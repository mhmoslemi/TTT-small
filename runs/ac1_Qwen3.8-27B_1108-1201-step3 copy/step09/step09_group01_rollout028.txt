def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def convolve_ab(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

2. LP fictitious play with Polyak-Nesterov averaging and PDLP solver

import numpy as np
from scipy.fft import rfft, irfft
import time

_lp_solver_cached = None

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def convolve_ab(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def get_lp_solver():
    global _lp_solver_cached
    if _lp_solver_cached is not None:
        return _lp_solver_cached
    import cvxpy as cp
    for solver_name in [cp.PDLP, cp.GUROBI, cp.MOSEK, cp.GLOP, cp.GLPK, cp.CBC, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return _lp_solver_cached

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def solve_lp_best_response(a_fixed, solver=None):
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.PDLP, cp.ECOS, cp.SCS, cp.GLPK, cp.GLOP, cp.CBC]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def score_of(a):
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a_norm = a / s
    return float(2.0 * len(a_norm) * fast_convolve(a_norm).max())

def exact_cd(a0, time_limit, verbose=False, label=""):
    """Coordinate descent with exact 1D line search via ternary + bisection."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    sum_a = 1.0
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            v = a[i]
            k_start = i
            k_end = i + n - 1

            if k_start > 0:
                unaff_left = float(conv[:k_start].max())
            else:
                unaff_left = 0.0
            if k_end < 2 * n - 2:
                unaff_right = float(conv[k_end + 1:].max())
            else:
                unaff_right = 0.0
            unaff_max = max(unaff_left, unaff_right)

            affected_conv = conv[k_start:k_end + 1]
            a_vals = a[0:n]

            has_2i = (2 * i >= k_start) and (2 * i <= k_end)
            idx_2i = 2 * i - k_start if has_2i else -1

            def obj_fn(delta):
                if delta <= -v + 1e-15:
                    return float('inf')
                linear = affected_conv + 2.0 * delta * a_vals
                lm = float(linear.max())
                if has_2i:
                    qv = affected_conv[idx_2i] + 2.0 * delta * v + delta * delta
                    return max(unaff_max, lm, qv)
                else:
                    return max(unaff_max, lm)

            # Check if delta=0 is optimal: try small perturbations
            obj_0 = max(unaff_max, float(affected_conv.max()))
            obj_neg = obj_fn(-0.01 * v) if v > 1e-15 else float('inf')
            obj_pos = obj_fn(0.01 * v)

            if obj_neg >= obj_0 - 1e-15 and obj_pos >= obj_0 - 1e-15:
                continue

            # Determine search direction
            if obj_neg < obj_pos:
                lo = -v
                hi = 0.0
            else:
                lo = 0.0
                hi = min(v, 1.0)

            # Also check the opposite direction briefly
            if obj_neg < obj_pos:
                obj_pos_check = obj_fn(0.5 * v)
                if obj_pos_check < obj_neg:
                    lo, hi = -v, v
            else:
                obj_neg_check = obj_fn(-0.5 * v)
                if obj_neg_check < obj_pos:
                    lo, hi = -v, v

            # Ternary search
            for _ in range(25):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                o1 = obj_fn(m1)
                o2 = obj_fn(m2)
                if o1 < o2:
                    hi = m2
                else:
                    lo = m1

            best_delta = (lo + hi) / 2.0
            best_obj = obj_fn(best_delta)

            # Compare with delta=0
            if best_obj >= obj_0 - 1e-14:
                continue

            # Apply update
            a[i] = max(v + best_delta, 0.0)
            sum_a += best_delta

            changes = (conv[k_start:k_end + 1] * 0 + 2.0 * best_delta * a_vals)
            if has_2i:
                changes[idx_2i] = 2.0 * best_delta * v + best_delta * best_delta
            conv[k_start:k_end + 1] += changes
            M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

            cur_score = 2.0 * n * M / (sum_a * sum_a)
            if cur_score < best_score - 1e-14:
                best_score = cur_score
                best_a = a.copy()
            improved = True

        if verbose and sweeps % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ECD {label} s={sweeps} ({elapsed:.0f}s) M={M:.8f} score={best_score:.6f}")

        if not improved:
            no_improve += 1
            if no_improve > 30:
                break
        else:
            no_improve = 0

    return best_a, best_score, sweeps

def fast_cd(a0, time_limit, verbose=False, label=""):
    """Fast coordinate descent with fixed step sizes (for quick polishing)."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    sum_a = 1.0
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue
            v = a[i]
            k_start = i
            k_end = i + n - 1

            if k_start > 0:
                unaff_left = float(conv[:k_start].max())
            else:
                unaff_left = 0.0
            if k_end < 2 * n - 2:
                unaff_right = float(conv[k_end + 1:].max())
            else:
                unaff_right = 0.0
            unaff_max = max(unaff_left, unaff_right)

            if unaff_max >= M - 1e-15:
                continue

            affected = conv[k_start:k_end + 1]
            a_vals = a[0:n]
            has_2i = (2 * i >= k_start) and (2 * i <= k_end)
            idx_2i = 2 * i - k_start if has_2i else -1

            best_delta = 0.0
            best_new_max = M

            for f in [0.05, 0.1, 0.2, 0.3, 0.5]:
                for sign in [-1, 1]:
                    delta = sign * f * v
                    new_val = v + delta
                    if new_val < -1e-15 or new_val > 1.0:
                        continue
                    new_aff = affected + 2.0 * delta * a_vals
                    if has_2i:
                        new_aff = new_aff.copy()
                        new_aff[idx_2i] = affected[idx_2i] + 2.0 * delta * v + delta * delta
                    nm = max(unaff_max, float(new_aff.max()))
                    if nm < best_new_max - 1e-15:
                        best_new_max = nm
                        best_delta = delta

            if best_delta != 0.0:
                a[i] = max(v + best_delta, 0.0)
                sum_a += best_delta
                changes = 2.0 * best_delta * a_vals
                if has_2i:
                    changes = changes.copy()
                    changes[idx_2i] = 2.0 * best_delta * v + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M / (sum_a * sum_a)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if verbose and sweeps % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FCD {label} s={sweeps} ({elapsed:.0f}s) M={M:.8f} score={best_score:.6f}")

        if not improved:
            no_improve += 1
            if no_improve > 50:
                break
        else:
            no_improve = 0

    return best_a, best_score, sweeps

def adam_logsumexp(a0, time_limit, lr=0.01, beta_init=20.0, verbose=False, label=""):
    """Adam optimizer with logsumexp-smoothed max convolution."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    beta = beta_init
    m = np.zeros(n)
    v_snap = np.zeros(n)
    t_count = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        t_count += 1
        conv = fast_convolve(a)
        max_conv = conv.max()

        # Logsumexp weights
        scaled = beta * (conv - max_conv)
        scaled = np.clip(scaled, -50, 0)
        weights = np.exp(scaled)
        Z = weights.sum()
        if Z < 1e-30:
            weights = np.zeros_like(conv)
            weights[conv.argmax()] = 1.0
            Z = 1.0
        weights = weights / Z

        # Gradient computation
        grad = np.zeros(n)
        peak_idx = np.where(weights > 1e-8)[0]
        for k in peak_idx:
            w = weights[k]
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            grad[j] += 2.0 * w * a[k - j]

        # Adam update
        t_f = float(t_count)
        m = 0.9 * m + 0.1 * grad
        v_snap = 0.999 * v_snap + 0.001 * grad * grad
        m_hat = m / (1.0 - 0.9 ** t_count)
        v_hat = v_snap / (1.0 - 0.999 ** t_count)
        a_raw = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # Project to simplex
        a = simplex_project(a_raw)

        # Update beta (increase temperature sharpness)
        beta *= 1.005
        if beta > 500:
            beta = 500

        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            lr *= 0.9
            no_improve = 0
            if lr < 1e-6:
                break

        if verbose and t_count % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam {label} t={t_count} ({elapsed:.0f}s) score={best_score:.6f} beta={beta:.1f}")

    return best_a, best_score

def lp_fictitious_play(a0, time_limit, solver=None, verbose=False, label=""):
    """LP fictitious play with Nesterov averaging and interleaved CD."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s

    b = a.copy()
    a_sequent = a.copy()
    b_sequent = b.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1

        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new
            gamma = 2.0 / (iters + 2.0)
            b_sequent = (1.0 - gamma) * b_sequent + gamma * b

        a_new, M_a = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new
            gamma = 2.0 / (iters + 2.0)
            a_sequent = (1.0 - gamma) * a_sequent + gamma * a

        # Every 5 iterations, do a quick CD polish on the sequential
        if iters % 5 == 0:
            for cand_raw in [a, b, a_sequent, b_sequent, 0.5 * (a_sequent + b_sequent)]:
                cand = np.maximum(cand_raw, 0)
                sc = cand.sum()
                if sc < 1e-10:
                    continue
                cand = cand / sc
                sc2 = score_of(cand)
                if sc2 < best_score:
                    best_score = sc2
                    best_a = cand.copy()

            # Quick CD on the best
            if iters % 20 == 0:
                t_cd = min(5.0, (time_limit - (time.monotonic() - t0)) * 0.1)
                if t_cd > 1:
                    a_cd, s_cd, _ = fast_cd(best_a, t_cd)
                    if s_cd < best_score:
                        best_score = s_cd
                        best_a = a_cd.copy()
                        # Update sequential with CD result
                        a_sequent = best_a.copy()

        if verbose and iters % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FP {label} it={iters} ({elapsed:.0f}s) best={best_score:.6f}")

    return best_a, best_score, iters

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def init_cosine(n, rng):
    """Initialize with a cosine-based profile."""
    t = np.arange(n) / n
    n_comp = int(rng.integers(2, 6))
    profile = np.ones(n)
    for j in range(n_comp):
        freq = rng.uniform(0.5, 3.0)
        amp = rng.uniform(0.1, 0.3)
        phase = rng.uniform(0, np.pi)
        profile += amp * np.cos(2 * np.pi * freq * t + phase)
    profile = np.maximum(profile, 0.01)
    s = profile.sum()
    return profile / s

def init_gaussian_mixture(n, rng):
    """Initialize with Gaussian mixture."""
    t = np.arange(n) / n
    n_gauss = int(rng.integers(2, 5))
    profile = np.zeros(n)
    for _ in range(n_gauss):
        center = rng.uniform(0.1, 0.9)
        width = rng.uniform(0.05, 0.2)
        amp = rng.uniform(0.5, 1.0)
        profile += amp * np.exp(-0.5 * ((t - center) / width) ** 2)
    profile = np.maximum(profile, 0.01)
    s = profile.sum()
    return profile / s

def peak_perturb(a0, rng, strength=0.05):
    """Perturb by targeting peak contributions."""
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a)
    M = conv.max()
    peak_thresh = M * 0.9
    peak_pos = np.where(conv >= peak_thresh)[0]

    pressure = np.zeros(n)
    for k in peak_pos:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            prod = a[i] * a[j]
            pressure[i] += prod
            if j != i:
                pressure[j] += prod * 0.5

    p_max = pressure.max()
    if p_max > 1e-20:
        pressure = pressure / p_max
    noise = rng.standard_normal(n) * 0.3
    a_new = a * (1.0 - strength * pressure + strength * noise)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a0.copy()
    return a_new / s

def check_and_update(a, best_seq, best_score, label=""):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)})")
    return best_seq, best_score

                a_cd, s_cd, _ = exact_cd(best_seq, t_cd, label=f"cd_after_adam{lr_try}")
                best_seq, best_score = check_and_update(a_cd, best_seq, best_score, f"cd_adam_{lr_try}")

    print(f"Phase 4 done ({time.monotonic()-t4:.0f}s): best={best_score:.6f}")

    # === Phase 5: Final Polishing ===
    print(f"\n=== Phase 5: Final Polish ===")
    t5 = time.monotonic()
    if best_seq is not None:
        # Intensive CD
        rem = deadline - time.monotonic() - 100
        if rem > 30:
            t_alloc = min(rem * 0.5, 200.0)
            print(f"  Final ECD budget={t_alloc:.0f}s")
            a_final, s_final, _ = exact_cd(best_seq, t_alloc, verbose=True, label="final")
            best_seq, best_score = check_and_update(a_final, best_seq, best_score, "final_ecd")

        # Quick LP polish
        rem = deadline - time.monotonic() - 50
        if rem > 20:
            t_lp = min(rem * 0.4, 60.0)
            a_lp, s_lp, _ = lp_fictitious_play(best_seq, t_lp, solver=solver,
                                                verbose=False, label="final_lp")
            best_seq, best_score = check_and_update(a_lp, best_seq, best_score, "final_lp")

        # Final CD
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            a_f, s_f, _ = exact_cd(best_seq, min(rem, 30.0), label="ultimate")
            best_seq, best_score = check_and_update(a_f, best_seq, best_score, "ultimate")

    print(f"Phase 5 done ({time.monotonic()-t5:.0f}s): best={best_score:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL: score={best_score:.6f} n={len(best_seq)}")
    return [float(x) for x in best_seq]
