def logsumexp_anneal(a0, time_limit, T_stages=None, maxiter_per_stage=5000):
    n = len(a0)
    z = np.log(np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)).copy()
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [50.0, 25.0, 12.5, 6.25, 3.125, 1.5, 0.75, 0.375, 0.19, 0.1, 0.05, 0.02, 0.01]
    
    n_stages = len(T_stages)
    iters_per = max(20, maxiter_per_stage // n_stages)
    best_a = np.asarray(a0, dtype=np.float64)
    best_score = evaluate_local(best_a)
    
    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        def func_lse(z_in):
            a_in = np.exp(z_in - z_in.max())
            a_in = a_in / a_in.sum()
            b_in = fast_convolve(a_in)
            max_b = b_in.max()
            exp_b = np.exp(b_in / stage_T - max_b / stage_T)
            log_sum_exp = max_b / stage_T + np.log(exp_b.sum())
            return (2.0 * n / stage_T) * log_sum_exp * 0  # placeholder
            # Actually let's compute properly:
            # f_T(a) = T * log(sum_k exp(conv(a,a)[k] / T))
            # = T * (max_b/T + log(sum exp((b-kmax)/T)))
            # = max_b + T * log(sum exp((b-kmax)/T))
        
        def func_lse2(z_in):
            a_in = np.exp(z_in - z_in.max())
            a_in = a_in / a_in.sum()
            b_in = fast_convolve(a_in)
            max_b = b_in.max()
            shifted = (b_in - max_b) / stage_T
            log_sum = np.log(np.exp(shifted).sum())
            return max_b + stage_T * log_sum
        
        def jac_lse(z_in):
            a_in = np.exp(z_in - z_in.max())
            a_in = a_in / a_in.sum()
            b_in = fast_convolve(a_in)
            max_b = b_in.max()
            shifted = (b_in - max_b) / stage_T
            exp_shifted = np.exp(shifted)
            w = exp_shifted / exp_shifted.sum()
            # grad of b_in w.r.t. a_in: b_in[k] = sum_j a[j]*a[k-j]
            # grad[k]_i = 2*a[k-i] (if valid)
            # total grad = sum_k w[k] * 2 * a[k-i]
            g = 2.0 * np.correlate(w, a_in, mode='valid')
            # Project gradient for softmax parameterization
            S = np.dot(g, a_in)
            return a_in * (g - S)
        
        try:
            res = minimize(
                fun=func_lse2, x0=z, jac=jac_lse, method='L-BFGS-B',
                options={'maxiter': iters_per, 'ftol': 1e-16,

            )

                best_a = a_new.copy()
        except Exception:
            pass
        
        if time.monotonic() - t0 > time_limit:
            break
    
    return best_a, best_score

def subgradient_refine(a0, time_limit, n_steps=100000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.05
    no_improve = 0
    momentum = np.zeros(n)
    beta = 0.9
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.9995
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        # Compute subgradient from active set
        w = b[active] - b[active].min() + 1e-10
        w = w / w.sum()
        
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += w[k - active[0]] * a[k - idx] if False else w[np.searchsorted(active, k)] * a[k - idx]
        
        # Simpler: g = 2 * correlate(y_active, a, 'valid')
        y_active = np.zeros(2 * n - 1)
        y_active[active] = w
        g = 2.0 * np.correlate(y_active, a, mode='valid')
        
        # Nesterov-style update
        momentum = beta * momentum - lr * g
        a_trial = a + momentum
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st
        
        score = 2.0 * n * fast_convolve(a_trial).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 20:
                lr *= 0.85
                momentum *= 0.5
                no_improve = 0
        lr = max(1e-7, lr)
        
        if step % 10000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad {elapsed:.0f}s: score={best_score:.10f} lr={lr:.8f}")
    
    return best_a, best_score

                break
        
        if not improved:
            # Random coordinate perturbation
            for _ in range(3):
                p1 = rng.integers(0, n)
                p2 = rng.integers(0, n)
                if p1 == p2:
                    continue
                a_trial = a.copy()
                transfer = a[p1] * rng.uniform(0.01, 0.05)
                a_trial[p1] -= transfer
                a_trial[p2] += transfer
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
        
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 500:
                break
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    
    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"LP solver: {solver}")
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 10 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Init: n={len(best_seq)}, score={best_score:.10f}")
    
    n_best = len(best_seq)
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")
    
    # === Phase 1: Nesterov LP at current length ===
    print(f"\n=== Phase 1: Nesterov LP n={n_best} ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 550.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_optimize(best_seq, t_alloc, solver=solver, n_max_iters=500000)
        try_update(a_lp, "nlp_main")
    
    # === Phase 2: LogSumExp annealing ===
    print(f"\n=== Phase 2: LogSumExp Annealing ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 350.0)
    if t_alloc > 30:
        T_stages = [30.0, 15.0, 8.0, 4.0, 2.0, 1.0, 0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.008]
        print(f"  Budget: {t_alloc:.0f}s, {len(T_stages)} stages")
        a_lse, _ = logsumexp_anneal(best_seq, t_alloc, T_stages=T_stages, maxiter_per_stage=8000)
        try_update(a_lse, "lse_anneal")
    
    # === Phase 3: Subgradient refinement ===
    print(f"\n=== Phase 3: Subgradient ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 15:
        a_sg, _ = subgradient_refine(best_seq, t_alloc, n_steps=100000)
        try_update(a_sg, "subgrad")
    
    # === Phase 4: Peak-directed descent ===
    print(f"\n=== Phase 4: Peak Descent ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 100.0)
    if t_alloc > 10:
        a_pd, _ = peak_directed_descent(best_seq, t_alloc, n_steps=50000)
        try_update(a_pd, "peak_desc")
    
    # === Phase 5: Final LogSumExp at very low T ===
    print(f"\n=== Phase 5: Final LSE (low T) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 150.0)
    if t_alloc > 15:
        T_low = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
        a_lse2, _ = logsumexp_anneal(best_seq, t_alloc, T_stages=T_low, maxiter_per_stage=6000)
        try_update(a_lse2, "lse_low_T")
    
    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-Length ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        # Try longer length
        for scale in [1.25, 1.5, 2.0]:
            rem = deadline - time.monotonic()
            if rem < 50:
                break
            alt_n = int(n_best * scale)
            if alt_n < 200 or alt_n > 5000:
                continue
            t_alt = min(rem * 0.3, 150.0)
            if t_alt < 30:
                continue
            print(f"  Trying n={alt_n} (scale={scale}), budget={t_alt:.0f}s")
            init_alt = interpolate_sequence(best_seq, alt_n)
            
            # Quick LP for smaller n
            if alt_n <= 2000:
                t_lp = t_alt * 0.5
                a_alt, _ = nesterov_lp_optimize(init_alt, t_lp, solver=solver, n_max_iters=100000)
                try_update(a_alt, f"lp_n{alt_n}")
            
            # LogSumExp polish
            rem = deadline - time.monotonic()
            if rem > 20 and best_seq is not None:
                T_alt = [10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02]
                a_alt2, _ = logsumexp_anneal(best_seq, rem * 0.3, T_stages=T_alt, maxiter_per_stage=3000)
                try_update(a_alt2, f"lse_n{alt_n}")
    
    # === Phase 7: Perturbation + re-optimize ===
    print(f"\n=== Phase 7: Perturbation Escape ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t_pert = min(rem * 0.3, 120.0)
        n_cur = len(best_seq)
        x = np.linspace(0, 2 * np.pi, n_cur)
        for freq in [1, 2, 3, 5]:
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            perturbation = 0.015 * np.cos(freq * x)
            perturbed = best_seq + perturbation
            perturbed = np.maximum(perturbed, 0)
            ps = perturbed.sum()
            if ps < 1e-15:
                continue
            perturbed = perturbed / ps
            t_sub = min(t_pert / 4.0, rem * 0.4)
            if t_sub < 10:
                continue
            a_pert, _ = logsumexp_anneal(perturbed, t_sub, T_stages=[5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05], maxiter_per_stage=2000)
            try_update(a_pert, f"pert_f{freq}")
    
    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        a_fin, _ = logsumexp_anneal(best_seq, rem * 0.5,
                                     T_stages=[2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01],
                                     maxiter_per_stage=4000)
        try_update(a_fin, "final_lse")
    
    rem = deadline - time.monotonic()
    if rem > 10:
        a_pd2, _ = peak_directed_descent(best_seq, rem * 0.5, n_steps=20000)
        try_update(a_pd2, "final_peak")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.10f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        best_seq = np.ones(100) / 100
        best_score = evaluate_local(best_seq)
    
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
