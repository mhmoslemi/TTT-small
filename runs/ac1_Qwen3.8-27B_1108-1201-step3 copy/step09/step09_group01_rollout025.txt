    """Build (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed, M, solver=None):
    """
    Given fixed a (non-negative) and threshold M, find b maximizing sum(b)
    subject to conv(a,b)_k <= M for all k, b >= 0.
    Returns (b, optimal_sum) or (None, None).
    """
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.SCS, cp.ECOS, cp.PDLP, cp.GLPK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = float(b_val.sum())
    if s < 1e-15:
        return None, None
    return b_val, s


def lp_fictitious_play(a_init, time_limit, solver=None, verbose=False):
    """
    LP Fictitious Play with Nesterov-Aginag-Tseng sequent averaging.
    The game: Row picks a, Column picks b, payoff is max_k conv(a,b)_k.
    Best response: given a, find b maximizing sum(b) s.t. conv(a,b)_k <= M.
    """
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    # Nesterov sequent
    a_seq = a.copy()
    b_seq = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    stall_count = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        M_a = float(fast_convolve(a).max())

        # Solve LP: max sum(b) s.t. conv(a,b)_k <= M_a, b >= 0
        b_br, S_b = solve_lp_best_response(a, M_a, solver)
        if b_br is None:
            stall_count += 1
            if stall_count > 50:
                break
            continue
        stall_count = 0

        # The best response b_br has sum = S_b.
        # If we normalize: b_norm = b_br / S_b, then max conv(a, b_norm) = M_a / S_b.
        # For auto-convolution at fixed point: S_b should be close to sum(a) = 1.
        
        # Nesterov sequent update
        gamma = 2.0 / (it + 2.0)
        b_seq = (1.0 - gamma) * b_seq + gamma * (b_br / S_b)
        
        # Now solve for a's best response given b_seq (or b_br/S_b)
        b_curr = b_br / S_b
        M_b = float(fast_convolve_dummy = convolve_ab(b_curr, b_curr)).max() if False else float(convolve_ab(b_curr, b_curr).max())
        
        a_br, S_a = solve_lp_best_response(b_curr, M_b, solver)
        if a_br is not None:
            a_seq = (1.0 - gamma) * a_seq + gamma * (a_br / S_a)

        # Evaluate candidates
        candidates = [(a, "a"), (b_curr, "b"), (a_seq, "a_seq"), (b_seq, "b_seq"),
                      (0.5 * (a_seq + b_seq), "avg")]
        for cand, name in candidates:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-10:
                continue
            cand_n = cand / sc
            cur_score = 2.0 * n * fast_convolve(cand_n).max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = cand_n.copy()
                if verbose and it % 10 == 0:
                    print(f"    FLP it={it}: new best={best_score:.8f} ({name})")

        # Update a for next iteration
        a = b_curr  # or use a_seq; using the latest iterate

    return best_a, best_score


def nesterov_fictitious_play_v2(a_init, time_limit, solver=None, verbose=False):
    """
    Improved Fictitious Play: 
    - Uses the sequent average as the "current strategy" for solving LPs
    - This gives better convergence properties (Aginag-Tseng)
    - Tracks both the iterate and the sequent for the best score
    """
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    # Sequent (Nesterov running average)
    a_s = a.copy()
    b_s = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        gamma = 2.0 / (it + 2.0)

        # Solve best response for b given current sequent a_s
        M_as = float(fast_convolve(a_s).max())
        b_br, S_b = solve_lp_best_response(a_s, M_as, solver)
        if b_br is None:
            # Fallback: use gradient step
            conv = fast_convolve(a_s)
            peak_idx = np.argmax(conv)
            grad = np.zeros(n)
            i_start = max(0, peak_idx - n + 1)
            i_end = min(n - 1, peak_idx)
            for i in range(i_start, i_end + 1):
                grad[i] = 2.0 * a_s[peak_idx - i]
            a_new = a_s * (1.0 - 0.01 * grad / max(grad.max(), 1e-10))
            a_new = simplex_project(a_new)
            a_s = (1.0 - gamma) * a_s + gamma * a_new
            continue

        b_norm = b_br / S_b
        b_s = (1.0 - gamma) * b_s + gamma * b_norm

        # Solve best response for a given current sequent b_s
        M_bs = float(convolve_ab(b_s, b_s).max())
        a_br, S_a = solve_lp_best_response(b_s, M_bs, solver)
        if a_br is not None:
            a_norm = a_br / S_a
            a_s = (1.0 - gamma) * a_s + gamma * a_norm
        else:
            a_s = b_s.copy()

        # Evaluate candidates
        cands = [a_s, b_s, 0.5 * (a_s + b_s), b_norm]
        if a_br is not None:
            cands.append(a_norm)
        for cand in cands:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-10:
                continue
            cand_n = cand / sc
            cur_score = 2.0 * n * fast_convolve(cand_n).max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = cand_n.copy()

    return best_a, best_score


def fast_coord_descent(a0, time_limit, verbose=False, label=""):
    """
    Fast coordinate descent with vectorized peak tracking.
    At each coordinate, try multiple step sizes to minimize the max convolution.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_max = float(conv[:k_start].max())
            else:
                unaff_max = -1e30
            if k_end < 2 * n - 2:
                unaff_right = float(conv[k_end + 1:].max())
                unaff_max = max(unaff_max, unaff_right)

            affected = conv[k_start:k_end + 1]

            best_delta = 0.0
            best_new_max = M

            # Try multiple step sizes (both directions)
            a_i = a[i]
            factors = [0.01, 0.03, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 0.9, 1.0]
            
            for f in factors:
                for sign in [-1, 1]:
                    delta = sign * f * a_i
                    new_val = a_i + delta
                    if new_val < -1e-15 or new_val > 1.0:
                        continue

                    # Compute new affected values
                    new_aff = affected + 2.0 * delta * a[k_start:k_end + 1 - i] if k_end - i < n else affected + 2.0 * delta * a[0:n]
                    # Actually: conv[k] += 2*delta*a[k-i] for k in [i, i+n-1]
                    # a[k-i] for k in [i, i+n-1] is a[0], a[1], ..., a[n-1]
                    # So the change is 2*delta*a[j] for j = 0,...,n-1
                    # Plus delta^2 at k=2i if 2i is in range
                    new_aff = affected.copy() + 2.0 * delta * a[:n - k_start + k_start]  # This is just a
                    # Wait, let me be more careful
                    # conv[k] for k in [i, i+n-1]: change = 2*delta*a[k-i]
                    # k-i ranges from 0 to n-1, so a[k-i] = a[0], a[1], ..., a[n-1] = a
                    new_aff = affected + 2.0 * delta * a
                    if 2 * i >= k_start and 2 * i <= k_end:
                        idx = 2 * i - k_start
                        new_aff = new_aff.copy()
                        new_aff[idx] += delta * delta  # quadratic term

                    new_max = max(unaff_max, float(new_aff.max()))
                    if new_max < best_new_max - 1e-16:
                        best_new_max = new_max
                        best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-15:
                a[i] = max(a[i] + best_delta, 0.0)
                sum_a = a.sum()

                # Update conv
                changes = 2.0 * best_delta * a
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] += best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M / (sum_a * sum_a)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 5:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def fine_coord_descent(a0, time_limit, verbose=False, label=""):
    """Very fine coordinate descent for final polishing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = float(conv[:k_start].max())
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = float(conv[k_end + 1:].max())
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            affected = conv[k_start:k_end + 1]
            a_i = a[i]

            best_delta = 0.0
            best_new_max = M

            # Very fine steps
            for factor in [0.999, 0.998, 0.997, 0.995, 0.993, 0.99, 0.985, 0.98,
                           1.001, 1.002, 1.003, 1.005, 1.007, 1.01, 1.015, 1.02]:
                delta = a_i * (factor - 1.0)
                new_val = a_i + delta
                if new_val < -1e-15:
                    continue

                new_aff = affected + 2.0 * delta * a
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_aff = new_aff.copy()
                    new_aff[idx] += delta * delta

                new_max = max(unaff_max, float(new_aff.max()))
                if new_max < best_new_max - 1e-17:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-16:
                a[i] = max(a[i] + best_delta, 0.0)
                sum_a = a.sum()

                changes = 2.0 * best_delta * a
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] += best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M / (sum_a * sum_a)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved and sweeps > 10:
            break

        if verbose and sweeps % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FineCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def smooth_gradient_descent(a0, time_limit, beta_init=50.0, lr_init=0.005, verbose=False):
    """
    Gradient descent on the smooth max approximation using log-sum-exp.
    With Nesterov momentum and adaptive parameters.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    v = np.zeros(n)  # momentum
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    beta = beta_init
    lr = lr_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        conv_a = fast_convolve(a)
        max_conv = float(conv_a.max())

        # Smooth max via log-sum-exp
        scaled = beta * (conv_a - max_conv)
        scaled = np.clip(scaled, -50.0, 0.0)
        weights = np.exp(scaled)
        Z = weights.sum()
        if Z < 1e-30:
            weights = np.zeros_like(conv_a)
            weights[conv_a.argmax()] = 1.0
            Z = 1.0
        p = weights / Z  # softmax distribution over convolution positions

        # Gradient: grad[i] = 2 * sum_k p[k] * a[k-i] (cross-correlation)
        # This is: 2 * (p correlated with a)[i]
        # Compute via the formula: grad[i] = 2 * sum_k p[k] * a[k-i] where 0<=k-i<n
        grad = np.zeros(n)
        for k in range(len(p)):
            if p[k] < 1e-12:
                continue
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            grad[j] += 2.0 * p[k] * a[k - j]

        # Nesterov momentum step in log-space
        v = 0.9 * v - lr * grad
        log_a = np.log(a + 1e-30) + v
        a_new = np.exp(log_a - log_a.max())
        a_new = simplex_project(a_new)

        # Evaluate
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Adaptive parameters
        if no_improve > 50:
            lr *= 0.9
            beta *= 1.3
            no_improve = 0
            v *= 0.5
            if lr < 1e-10:
                break
        elif it % 100 == 0:
            beta *= 1.01

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score
