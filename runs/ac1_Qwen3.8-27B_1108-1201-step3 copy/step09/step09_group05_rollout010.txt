import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    T = np.zeros((size, n))
    for j in range(n):
        Bj = rfft(np.zeros(size), fft_size)
        Bj[j] = 1.0
        T[:, j] = irfft(A * Bj, fft_size)[:size]
    return T


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def norm_sequence(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    size = 2 * n - 1
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build constraint using the matrix
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def lbgfs_refine(a0, time_limit, T_start=None, T_end=None, maxiter=10000, n_stages=10):
    """
    L-BFGS on log-sum-exp smoothed objective.
    CORRECTED: temperature goes from smooth (low T) to sharp (high T).
    """
    n = len(a0)
    if T_start is None:
        T_start = max(10.0, n / 20.0)
    if T_end is None:
        T_end = max(500.0, n * 2)
    
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    
    def objective(z_in, T_val):
        a = softmax_from_z(z_in)
        b = fast_convolve(a)
        Tb = T_val * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        log_sum_exp = max_Tb + np.log(exp_Tb.sum())
        return (2.0 * n / T_val) * log_sum_exp

    def grad(z_in, T_val):
        a = softmax_from_z(z_in)
        b = fast_convolve(a)
        Tb = T_val * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()
        grad_a = np.zeros(n)
        size_b = len(b)
        for k in range(size_b):
            if w[k] < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]
        S = np.dot(grad_a, a)
        return a * (grad_a - S) * (2.0 * n / T_val)

    # Temperature schedule: smooth (low T) to sharp (high T)
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(20, maxiter // n_stages)

    for stage_i, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            return objective(z_in, stage_T)
        
        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            return grad(z_in, stage_T)

        try:
            res = minimize(
                fun=func,
                x0=z,
                jac=jac,
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 500, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

        # Track progress
        a_cur = softmax_from_z(z)
        score = 2.0 * n * fast_convolve(a_cur).max()
        if stage_i % 3 == 0:
            print(f"    L-BFGS stage {stage_i+1}/{n_stages}, T={stage_T:.1f}, raw_score={score:.6f}")

    return softmax_from_z(z)


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=50000):
    """Nesterov dual averaging LP refinement."""
    n = len(a0)
    a = norm_sequence(a0)
    y = a.copy()
    best_a = a.copy()
    best_raw = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_raw
    stall_count = 0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        if it % 10 == 0 or it < 5:

            if score_a < best_raw:
                best_raw = score_a
                best_a = a.copy()

                    noise = rng_local.random(n) * 0.05
                    y = y + noise
                    y = np.maximum(y, 0)
                    y = y / y.sum()
                    stall_count = 0

    return best_a, best_raw


def fista_refine(a0, time_limit, n_max_iters=200000):
    """
    FISTA (proximal gradient with Nesterov acceleration) for the non-smooth objective.
    Minimizes g(a) = 2*n*max_k conv(a,a)[k] + indicator_simplex(a)
    using subgradient of the max.
    """
    n = len(a0)
    a = norm_sequence(a0)
    y = a.copy()
    best_a = a.copy()
    best_raw = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    t_fista = 1.0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Subgradient of max(conv(y,y)) at y
        b_y = fast_convolve(y)
        k_star = int(np.argmax(b_y))
        
        # Subgradient: d/dy max_k conv(y,y)[k] = 2 * conv(y, e_{k_star}) restricted to [0,n-1]
        grad = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        for i in range(js, je + 1):
            grad[i] = 2.0 * y[k_star - i]
        
        # Lipschitz estimate
        L = 2.0 * b_y.max() + 1e-6
        
        # Gradient step
        z = y - (1.0 / L) * grad
        # Project onto simplex (non-negative, sum=1)
        z = project_simplex(z)
        
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_fista ** 2)) / 2.0
        theta = (t_fista - 1.0) / t_new
        y = z + theta * (z - a)
        y = project_simplex(y)
        a = z
        t_fista = t_new
        
        # Track best
        if it % 50 == 0:
            raw = 2.0 * n * fast_convolve(a).max()
            if raw < best_raw:
                best_raw = raw
                best_a = a.copy()

    return best_a, best_raw


def project_simplex(v):
    """Project vector onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s

def structured_init(n, rng, mode=0):
    """Create structured initial sequence."""
    if mode == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif mode == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif mode == 3:
        # Asymmetric: concentrated toward one end
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x) * (1 + 0.3 * np.sin(3 * np.pi * x))
    elif mode == 4:
        # Two-peak asymmetric
        x = np.linspace(0, 1, n)
        a = np.exp(-30 * (x - 0.3) ** 2) + 0.7 * np.exp(-30 * (x - 0.8) ** 2)
    elif mode == 5:
        # Ramp with oscillation
        x = np.linspace(0, 1, n)
        a = x * (1 + 0.4 * np.cos(4 * np.pi * x))
    elif mode == 6:
        # Inverse ramp
        x = np.linspace(0, 1, n)
        a = (1 - x) * (1 + 0.3 * np.sin(5 * np.pi * x))
    else:
        a = np.ones(n) * (1 + 0.1 * np.sin(2 * np.pi * rng.random() * 3 * np.linspace(0, 1, n)))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        a_fista, _ = fista_refine(best_seq, t_alloc, n_max_iters=50000)
        try_update(a_fista, "fista")

    # ===== Phase 4: Peak shaving =====
    print(f"\n=== Phase 4: Peak Shaving (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.06, 90.0)
    if t_alloc > 10:
        a_ps, _ = peak_shave_refine(best_seq, t_alloc, n_steps=50000)
        try_update(a_ps, "peak_shave")

    # ===== Phase 5: Subgradient coordinate descent =====
    print(f"\n=== Phase 5: Subgrad Coord (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 120.0)
    if t_alloc > 10:
        a_sg, _ = subgradient_coord_descent(best_seq, t_alloc, n_steps=100000)
        try_update(a_sg, "subgrad_coord")

    # ===== Phase 6: Second LP round =====
    print(f"\n=== Phase 6: Nesterov LP r2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 200.0)
    if t_alloc > 20:
        a_lp2, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=50000)
        try_update(a_lp2, "nesterov_lp_2")

    # ===== Phase 7: L-BFGS very sharp polish =====
    print(f"\n=== Phase 7: L-BFGS Sharp Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 180.0)
    if t_alloc > 20:
        n_cur = len(best_seq)
        T_lo = max(100.0, n_cur * 0.3)
        T_hi = max(2000.0, n_cur * 5)
        print(f"  T: {T_lo:.1f} -> {T_hi:.1f}, budget={t_alloc:.0f}s")
        a_lb2 = lbgfs_refine(best_seq, t_alloc, T_start=T_lo, T_end=T_hi, maxiter=8000, n_stages=8)
        try_update(a_lb2, "lbgfs_sharp")

    # ===== Phase 8: Multi-length exploration =====
    print(f"\n=== Phase 8: Multi-Length (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    n_cur = len(best_seq)
    alt_lengths = [int(n_cur * m) for m in [0.85, 0.9, 1.1, 1.15, 1.3]]
    alt_lengths = [x for x in alt_lengths if 512 <= x <= 2048 and x != n_cur]
    # Add some specific values
    for fn in [800, 900, 1000, 1100, 1200, 1300]:
        if 512 <= fn <= 2048 and fn != n_cur and fn not in alt_lengths:
            alt_lengths.append(fn)
    alt_lengths = sorted(set(alt_lengths))

    n_tried = 0
    for alt_n in alt_lengths:
        rem = deadline - time.monotonic()
        if rem < 60 or n_tried >= 3:
            break
        n_tried += 1
        t_total = min(rem * 0.20, 180.0)
        if t_total < 30:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")

        # Initialize from best via interpolation + some structured init
        init_alt = interpolate_sequence(best_seq, alt_n)
        # Blend with a structured init for diversity
        init_struct = structured_init(alt_n, rng, mode=rng.integers(0, 7))
        init_alt = 0.7 * init_alt + 0.3 * init_struct
        init_alt = norm_sequence(init_alt)

        # LP
        t_lp = t_total * 0.5
        a_alt, _ = nesterov_lp_refine(init_alt, t_lp, solver=solver, n_max_iters=30000)
        try_update(a_alt, f"lp_n{alt_n}")

        # L-BFGS on the best at this length
        rem = deadline - time.monotonic()
        if rem > 20:
            t_lb = min(rem * 0.3, t_total * 0.4)
            if t_lb > 10:
                cur_best_for_n = best_seq if len(best_seq) == alt_n else a_alt
                a_alt2 = lbgfs_refine(cur_best_for_n, t_lb, 
                                        T_start=max(10.0, alt_n/50.0),
                                        T_end=max(500.0, alt_n*2),
                                        maxiter=5000, n_stages=8)
                try_update(a_alt2, f"lbgfs_n{alt_n}")

    # ===== Phase 9: Multi-start with different initializations =====
    print(f"\n=== Phase 9: Multi-Start (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        n_cur = len(best_seq)
        t_per_start = min(rem * 0.15, 120.0)
        
        for mode in [3, 4, 5, 6]:
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            init_ms = structured_init(n_cur, rng, mode=mode)
            # Blend with current best
            init_ms = 0.5 * norm_sequence(best_seq) + 0.5 * init_ms
            init_ms = norm_sequence(init_ms)
            
            t_sub = min(t_per_start / 2, rem * 0.8)
            # Quick L-BFGS
            a_ms = lbgfs_refine(init_ms, t_sub,
                                T_start=max(10.0, n_cur/50.0),
                                T_end=max(500.0, n_cur*2),
                                maxiter=5000, n_stages=8)
            try_update(a_ms, f"multistart_{mode}")

    # ===== Phase 10: Final combined polish =====
    print(f"\n=== Phase 10: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t_fin = min(rem * 0.25, 60.0)
        a_fin = lbgfs_refine(best_seq, t_fin, T_start=200.0, T_end=3000.0, maxiter=5000, n_stages=6)
        try_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 10:
        t_ps = min(rem * 0.25, 30.0)
        a_ps2, _ = peak_shave_refine(best_seq, t_ps, n_steps=30000)
        try_update(a_ps2, "final_ps")

    rem = deadline - time.monotonic()
    if rem > 10:
        t_sg = min(rem * 0.25, 30.0)
        a_sg2, _ = subgradient_coord_descent(best_seq, t_sg, n_steps=50000)
        try_update(a_sg2, "final_sg")

    rem = deadline - time.monotonic()
    if rem > 10:
        t_lp_fin = min(rem * 0.25, 20.0)
        a_lp_fin, _ = nesterov_lp_refine(best_seq, t_lp_fin, solver=solver, n_max_iters=5000)
        try_update(a_lp_fin, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
