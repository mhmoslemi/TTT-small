import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

    return float(2.0 * n * conv.max() / (s * s))


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def frank_wolfe_strong(a0, time_limit, verbose=True, top_k=30):
    """
    Frank-Wolfe with strong Wolfe line search.
    Guaranteed to monotonically decrease f for convex f on the simplex.
    Each iteration: compute subgradient, find best vertex, do exact line search.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.01:
            break

        # Compute smoothed subgradient using top-k peaks with softmax weighting
        k_act = min(top_k, 2 * n - 1)
        if k_act <= 5:
            peaks = np.arange(2 * n - 1)
            pv = conv
        else:
            peaks = np.argpartition(conv, -k_act)[-k_act:]
            pv = conv[peaks]

        # Adaptive temperature: tighter as we converge
        temp = 0.005 * max(M, 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / ws

        # Compute subgradient g_j = 2 * sum_k w_k * a[k-j] (for valid k-j)
        g = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            # g[i] += w_k * 2 * a[k-i] for i in [i_start, i_end]
            g[i_start:i_end + 1] += w[idx] * 2.0 * a[k - i_end:k - i_start + 1][::-1]

        # Find the vertex: j* = argmin g_j
        j_star = int(np.argmin(g))

        # Strong Wolfe line search along direction e_{j*} - a
        # f(t) = f((1-t)*a + t*e_{j*}) for t in [0,1]
        # Precompute cross term: conv(a, e_{j*})[k] = a[k-j*] if valid
        ae = np.zeros(2 * n - 1)
        ae[j_star:j_star + n] = a

        # f(t) = max_k [(1-t)^2*conv[k] + 2t(1-t)*ae[k] + t^2*delta_{k,2j*}]
        # Define: q_k(t) = (1-t)^2*c_k + 2t(1-t)*e_k + t^2*d_k
        # where c_k = conv[k], e_k = ae[k], d_k = 1 if k==2j* else 0

        c = conv
        e = ae
        d_k = 2 * j_star

        def f_of_t(t_val):
            f1 = (1.0 - t_val) ** 2
            f2 = 2.0 * t_val * (1.0 - t_val)
            f3 = t_val ** 2
            val = f1 * c + f2 * e
            if d_k < 2 * n - 1:
                val = val.copy()
                val[d_k] += f3
            return float(val.max())

        # Coarse search
        best_t = 0.0
        best_f = M
        candidates = [0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.2, 0.3, 0.5]
        for t_try in candidates:
            f_try = f_of_t(t_try)
            if f_try < best_f - 1e-15:
                best_f = f_try
                best_t = t_try

        # Golden section refinement
        if best_t > 0:
            lo = max(0.0, best_t * 0.5)
            hi = min(1.0, best_t * 1.5)
            if hi - lo < 1e-10:
                hi = lo + 0.1
            for _ in range(30):
                m1 = lo + (hi - lo) * 0.382
                m2 = lo + (hi - lo) * 0.618
                f1 = f_of_t(m1)
                f2 = f_of_t(m2)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            best_t = (lo + hi) / 2.0
            best_f = f_of_t(best_t)
        elif best_t == 0:
            # Try very small steps in case the optimal t is tiny
            for t_try in [0.001, 0.0005, 0.0001]:
                f_try = f_of_t(t_try)
                if f_try < best_f - 1e-15:
                    best_f = f_try
                    best_t = t_try

        if best_t < 1e-12:
            no_improve += 1
            if no_improve > 100:
                break
            # Perturb to escape
            a = a + np.random.standard_normal(n) * 1e-6
            a = np.maximum(a, 0)
            a = a / a.sum()
            conv = fast_convolve(a)
            M = float(conv.max())
            continue

        # Update
        a_new = (1.0 - best_t) * a
        a_new[j_star] += best_t

        # Verify and update
        M_new = best_f
        score_new = 2.0 * n * M_new

        if score_new < best_score - 1e-15:
            best_score = score_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        conv = fast_convolve(a)
        M = float(conv.max())

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW it={it} ({elapsed:.0f}s): best={best_score:.8f} M={M:.10f} t={best_t:.6f} j={j_star} no_imv={no_improve}")

    return best_a, best_score


def frank_wolfe_multi_vertex(a0, time_limit, verbose=True, top_k=30, num_vertices=5):
    """
    Frank-Wolfe with multiple vertex candidates.
    Instead of just argmin g, try the top few smallest g values.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.01:
            break

        # Compute smoothed subgradient
        k_act = min(top_k, 2 * n - 1)
        if k_act <= 5:
            peaks = np.arange(2 * n - 1)
            pv = conv
        else:
            peaks = np.argpartition(conv, -k_act)[-k_act:]
            pv = conv[peaks]

        temp = 0.005 * max(M, 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / ws

        g = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            g[i_start:i_end + 1] += w[idx] * 2.0 * a[k - i_end:k - i_start + 1][::-1]

        # Find top num_vertices smallest g values
        if num_vertices < n:
            vertex_candidates = np.argpartition(g, num_vertices)[:num_vertices]
        else:
            vertex_candidates = np.arange(n)

        best_t_overall = 0.0
        best_f_overall = M
        best_j_overall = 0

        for j_star in vertex_candidates:
            j_star = int(j_star)
            ae = np.zeros(2 * n - 1)
            ae[j_star:j_star + n] = a
            d_k = 2 * j_star
            c = conv
            e = ae

            def f_of_t_local(t_val, c=c, e=e, d_k=d_k):
                f1 = (1.0 - t_val) ** 2
                f2 = 2.0 * t_val * (1.0 - t_val)
                f3 = t_val ** 2
                val = f1 * c + f2 * e
                if d_k < 2 * n - 1:
                    val = val.copy()
                    val[d_k] += f3
                return float(val.max())

            # Quick evaluation at a few points
            for t_try in [0.02, 0.05, 0.1, 0.2]:
                f_try = f_of_t_local(t_try)
                if f_try < best_f_overall - 1e-15:
                    best_f_overall = f_try
                    best_t_overall = t_try
                    best_j_overall = j_star

        if best_t_overall < 1e-12:
            no_improve += 1
            if no_improve > 50:
                break
            a = a + np.random.standard_normal(n) * 1e-5
            a = np.maximum(a, 0)
            a = a / a.sum()
            conv = fast_convolve(a)
            M = float(conv.max())
            continue

        # Refine the best candidate with golden section
        j_star = best_j_overall
        ae = np.zeros(2 * n - 1)
        ae[j_star:j_star + n] = a
        d_k = 2 * j_star

        def f_of_t_refine(t_val):
            f1 = (1.0 - t_val) ** 2
            f2 = 2.0 * t_val * (1.0 - t_val)
            f3 = t_val ** 2
            val = f1 * conv + f2 * ae
            if d_k < 2 * n - 1:
                val = val.copy()
                val[d_k] += f3
            return float(val.max())

        lo = max(0.0, best_t_overall * 0.3)
        hi = min(1.0, best_t_overall * 2.0)
        if hi - lo < 1e-8:
            hi = lo + 0.05
        for _ in range(35):
            m1 = lo + (hi - lo) * 0.382
            m2 = lo + (hi - lo) * 0.618
            if f_of_t_refine(m1) < f_of_t_refine(m2):
                hi = m2
            else:
                lo = m1
        best_t_overall = (lo + hi) / 2.0
        best_f_overall = f_of_t_refine(best_t_overall)

        if best_f_overall >= M - 1e-14:
            no_improve += 1
            if no_improve > 50:
                break
            continue

        a_new = (1.0 - best_t_overall) * a
        a_new[j_star] += best_t_overall
        a = a_new
        conv = fast_convolve(a)
        M = float(conv.max())
        score_new = 2.0 * n * M

        if score_new < best_score - 1e-15:
            best_score = score_new
            best_a = a.copy()
            no_improve = 0

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FWm it={it} ({elapsed:.0f}s): best={best_score:.8f} M={M:.10f}")

    return best_a, best_score


def lp_guided_descent(a0, time_limit, solver, verbose=True):
    """
    Uses LP to find dual variables (peak weights) for a smoothed subgradient,
    then does a projected gradient step.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0
    lr = 0.05
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        # Solve LP for best response (gives us structure)
        b_lp, M_lp, _ = lp_best_response(a, solver)

        # Use b_lp as a direction: move toward it
        # But also use the subgradient for a proper descent direction
        k_act = min(30, 2 * n - 1)
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        pv = conv[peaks]
        temp = 0.003 * max(M, 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        w = w / w.sum()

        g = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            g[i_start:i_end + 1] += w[idx] * 2.0 * a[k - i_end:k - i_start + 1][::-1]

        # Gradient step + projection
        a_new = a - lr * g
        a_new = simplex_project(a_new)

        # Also try moving toward LP solution
        alpha = min(0.3, 5.0 / (it + 10.0))
        a_mix = (1 - alpha) * a_new + alpha * b_lp
        a_mix = np.maximum(a_mix, 0)
        a_mix = a_mix / a_mix.sum()

        # Pick the better of the two
        score_g = 2.0 * n * fast_convolve(a_new).max()
        score_m = 2.0 * n * fast_convolve(a_mix).max()

        if score_g <= score_m:
            a = a_new
            cur_score = score_g
        else:
            a = a_mix
            cur_score = score_m

        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 30:
            lr *= 0.8
            no_improve = 0
            if lr < 1e-8:
                break

        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LPgD it={it} ({elapsed:.0f}s): best={best_score:.8f} lr={lr:.6f}")

    return best_a, best_score


def adaptive_coord_descent(a0, time_limit, verbose=True, label=""):
    """
    Coordinate descent with adaptive step sizes and incremental convolution updates.
    Optimizes each coordinate by finding the optimal value analytically.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve_sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-18:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = conv[:k_start].max() if k_start < 2 * n - 1 else -1e30
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            a_shifted = a[k_start - i:k_end - i + 1]
            conv_affected = conv[k_start:k_end + 1]

            best_delta = 0.0
            best_new_M = M

            # Try multiplicative adjustments
            base = max(a[i], 1e-10)
            for factor in [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5]:
                delta = base * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = conv_affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-14:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-14:
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps > 20:
                break
        else:
            no_improve_sweeps = 0

        if verbose and sweeps % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD[{label}] sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

            print(f"    FineCD sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def diverse_init(n, rng, init_type=0):
    """Generate diverse initializations."""
    if init_type == 0:
        t = np.arange(n)
        center = (n - 1) / 2.0
        a = np.maximum(1.0 - np.abs(t - center) / center, 0)
    elif init_type == 1:
        t = np.arange(n)
        center = (n - 1) / 2.0
        x = (t - center) / center
        a = 0.3 + 0.7 * x * x
    elif init_type == 2:
        t = np.arange(n)
        center = (n - 1) / 2.0
        x = (t - center) / center
        power = rng.uniform(1.0, 3.0)
        a = np.maximum(0.5 * (1 + np.cos(np.pi * x)), 0) ** power
    elif init_type == 3:
        t = np.arange(n)
        c1 = n * rng.uniform(0.2, 0.35)
        c2 = n * rng.uniform(0.65, 0.8)
        w = n * rng.uniform(0.08, 0.2)
        a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    elif init_type == 4:
        # Zadoff-Chu like: a_j = |exp(2*pi*i*j*(j+1)/(2n))| pattern
        t = np.arange(n)
        phase = 2 * np.pi * t * (t + 1) / (2 * n)
        a = np.abs(np.exp(1j * phase))  # This is just 1, so add structure
        a = 0.5 + 0.5 * np.cos(2 * np.pi * t * rng.integers(1, 5) / n)
    elif init_type == 5:
        # Flat-top with random perturbation
        a = np.ones(n) + 0.1 * rng.standard_normal(n)
    else:
        t = np.arange(n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * rng.uniform(0.1, 0.5) * t / n)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def pair_swap_optimize(a0, time_limit, verbose=True):
    """
    Try swapping mass between pairs of coordinates to reduce peaks.
    More effective than single-coordinate moves for breaking symmetry.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    rounds = 0
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit:
        rounds += 1
        improved = False

        # Try random pairs
        num_tries = 200
        for _ in range(num_tries):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            if a[i] < 1e-15 or a[j] < 1e-15:
                continue

            # Try moving a small fraction of mass from i to j
            for frac in [0.01, 0.02, 0.05, 0.1]:
                delta = a[i] * frac
                new_a_i = a[i] - delta
                new_a_j = a[j] + delta
                if new_a_i < 0:
                    continue

                # Compute new M efficiently
                # conv changes at positions affected by i and j
                # conv_new[k] = conv[k] - 2*delta*a[k-i] + delta^2 (at k=2i)
                #                            - 2*delta*a[k-j] + delta^2 (at k=2j)
                # Wait, this isn't quite right for the cross term.
                # Let's just compute the change in the affected region.
                # Actually for efficiency, let's just try it and check max.

                # Affected range for coordinate i: [i, i+n-1]
                # Affected range for coordinate j: [j, j+n-1]
                # Union of affected ranges
                k_min = min(i, j)
                k_max = max(i + n - 1, j + n - 1)
                k_max = min(k_max, 2 * n - 2)

                # Unaffected max
                if k_min > 0:
                    unaff_l = conv[:k_min].max() if k_min < 2 * n - 1 else -1e30
                else:
                    unaff_l = -1e30
                if k_max < 2 * n - 2:
                    unaff_r = conv[k_max + 1:].max()
                else:
                    unaff_r = -1e30
                unaff_max = max(unaff_l, unaff_r)

                # Change in conv due to moving delta from i to j:
                # new_conv[k] = conv[k] - 2*delta*a[k-i] + delta^2*delta_{k,2i}
                #                              + 2*delta*a[k-j] - delta^2*delta_{k,2j}
                #                          + cross terms: -2*delta*a[k-i] was already there...
                # Actually: conv depends on a^2. If we change a[i] by -delta and a[j] by +delta:
                # New conv[k] = old conv[k] - 2*delta*a[k-i] + 2*delta*a[k-j]
                #              + delta^2*delta_{k,2i} - delta^2*delta_{k,2j}
                #              + 2*delta*(-delta)*delta_{k,i+j}  (cross term)
                # Wait, let me be more careful.
                # conv[k] = sum_p a[p]*a[k-p]
                # After: a'[i] = a[i]-delta, a'[j] = a[j]+delta
                # conv'[k] = conv[k] - 2*delta*a[k-i] + 2*delta*a[k-j]
                #           + delta^2*(delta_{k,2i} - delta_{k,2j})
                #           - 2*delta^2 * delta_{k,i+j}  (if i != j and both valid)
                # Wait, the cross term: when p=i and k-p=j, we get a[i]*a[j] which changes to
                # (a[i]-delta)(a[j]+delta) = a[i]*a[j] + delta*a[i] - delta*a[j] - delta^2
                # And when p=j and k-p=i: (a[j]+delta)(a[i]-delta) = a[i]*a[j] - delta*a[j] + delta*a[i] - delta^2
                # Total change from cross terms: 2*delta*(a[i]-a[j]) - 2*delta^2 at k=i+j
                # But this is already captured by the -2*delta*a[k-i] and +2*delta*a[k-j] terms
                # when k=i+j: -2*delta*a[j] + 2*delta*a[i] = 2*delta*(a[i]-a[j])
                # Plus the delta^2 terms.
                # So the formula is:
                # conv'[k] = conv[k] - 2*delta*a[k-i] + 2*delta*a[k-j]
                #           + delta^2*delta_{k,2i} - delta^2*delta_{k,2j}

                affected = conv[k_min:k_max + 1].copy()
                # Subtract 2*delta*a[k-i] for k in [k_min, k_max] where 0 <= k-i < n
                idx_i_start = max(k_min, i)
                idx_i_end = min(k_max, i + n - 1)
                if idx_i_start <= idx_i_end:
                    affected[idx_i_start - k_min:idx_i_end - k_min + 1] -= 2.0 * delta * a[k_min:idx_i_end + 1][idx_i_start - k_min:idx_i_end - k_min + 1]
                    # Hmm this indexing is getting complex. Let me use a simpler approach.

                # Actually, let me just do it more carefully with explicit indexing
                affected = conv[k_min:k_max + 1].copy()
                for k in range(k_min, k_max + 1):
                    change = 0.0
                    if 0 <= k - i < n:
                        change -= 2.0 * delta * a[k - i]
                    if 0 <= k - j < n:
                        change += 2.0 * delta * a[k - j]
                    if k == 2 * i:
                        change += delta * delta
                    if k == 2 * j:
                        change -= delta * delta
                    affected[k - k_min] += change

                new_M = max(unaff_max, float(affected.max()))
                if new_M < M - 1e-13:
                    a[i] = new_a_i
                    a[j] = new_a_j
                    conv[k_min:k_max + 1] = affected
                    M = new_M
                    improved = True
                    cur_score = 2.0 * n * M
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = a.copy()
                    break  # Move to next random pair

        if not improved and rounds > 100:
            break

        if verbose and rounds % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PairSwap round={rounds} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def accelerated_fw(a0, time_limit, verbose=True, top_k=30):
    """
    Accelerated Frank-Wolfe (using momentum-like averaging).
    Maintains a "ghost" point that moves faster, and the actual point is an average.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    # Ghost point (for computing gradient)
    z = a.copy()
    tk = 2.0 + 1e-8

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.01:
            break

        # Compute gradient at ghost point z
        conv_z = fast_convolve(z)
        k_act = min(top_k, 2 * n - 1)
        if k_act <= 5:
            peaks = np.arange(2 * n - 1)
            pv = conv_z
        else:
            peaks = np.argpartition(conv_z, -k_act)[-k_act:]
            pv = conv_z[peaks]

        temp = 0.01 * max(conv_z.max(), 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        w = w / w.sum()

        g = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            g[i_start:i_end + 1] += w[idx] * 2.0 * z[k - i_end:k - i_start + 1][::-1]

        # Linear minimization oracle: find vertex
        j_star = int(np.argmin(g))
        e_j = np.zeros(n)
        e_j[j_star] = 1.0

        # Update ghost: z = (tk-1)/tk * z + 1/tk * e_j
        z = ((tk - 1) / tk) * z + (1.0 / tk) * e_j

        # Update actual point: gamma = (2 + tk) / (2 + t0_val)  (standard accelerated FW)
        # Simpler: use gamma = 1 - 2/(tk+2)
        gamma = 1.0 - 2.0 / (tk + 2.0)
        a_new = gamma * a + (1.0 - gamma) * e_j

        # Strong step: line search between a and e_j
        # f(t) = f((1-t)*a + t*e_j)
        ae = np.zeros(2 * n - 1)
        ae[j_star:j_star + n] = a
        d_k = 2 * j_star
        c_cur = conv

        def f_of_t(t_val):
            f1 = (1.0 - t_val) ** 2
            f2 = 2.0 * t_val * (1.0 - t_val)
            f3 = t_val ** 2
            val = f1 * c_cur + f2 * ae
            if d_k < 2 * n - 1:
                val = val.copy()
                val[d_k] += f3
            return float(val.max())

        # Try the accelerated step and also do a line search
        f_accel = f_of_t(1.0 - gamma)  # t = 1-gamma moves from a toward e_j
        # Actually a_new = gamma*a + (1-gamma)*e_j, so t = 1-gamma in our parameterization
        # f(t) = f((1-t)*a + t*e_j), so t = 1-gamma

        # Compare with a few other t values
        best_t = 1.0 - gamma
        best_f = f_accel
        for t_try in [0.01, 0.05, 0.1, 0.2, 0.5]:
            f_try = f_of_t(t_try)
            if f_try < best_f:
                best_f = f_try
                best_t = t_try

        # Refine
        if 0 < best_t < 1:
            lo = max(0, best_t * 0.5)
            hi = min(1, best_t * 1.5)
            if hi - lo < 1e-8:
                hi = min(1, lo + 0.1)
            for _ in range(25):
                m1 = lo + (hi - lo) * 0.382
                m2 = lo + (hi - lo) * 0.618
                if f_of_t(m1) < f_of_t(m2):
                    hi = m2
                else:
                    lo = m1
            best_t = (lo + hi) / 2.0
            best_f = f_of_t(best_t)

        a = (1.0 - best_t) * a + best_t * e_j
        tk += 1.0
        conv = fast_convolve(a)
        M = float(conv.max())
        score_new = 2.0 * n * M

        if score_new < best_score - 1e-15:
            best_score = score_new
            best_a = a.copy()

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    AccFW it={it} ({elapsed:.0f}s): best={best_score:.8f} tk={tk:.0f}")

    return best_a, best_score
