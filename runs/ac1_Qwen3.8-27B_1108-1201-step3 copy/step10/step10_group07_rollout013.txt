def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

    if k_start > 0:

    # The quadratic adds delta^2 on top of the linear term at quad_idx

    lo = -a[i]

        best_delta = -a[i]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Exact 1D convex coordinate descent (bisection on subgradient)
2. Frank-Wolfe (conditional gradient) for fast global moves
3. LP-based best-response alternating minimization
4. Nesterov-accelerated subgradient for exploitation
5. Diverse structural restarts to escape local minima
6. Multi-scale time allocation with adaptive switching

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_of(a):
    """Compute 2*n*max(conv)/(sum)^2 quickly."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def find_lp_solver():
    """Find the fastest working LP solver."""
    if not hasattr(find_lp_solver, '_cached'):
        for solver_name in [cp.PDLP, cp.ECOS, cp.CLARABEL, cp.SCS]:
            try:
                x = cp.Variable(5, nonneg=True)
                prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
                prob.solve(solver=solver_name)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    find_lp_solver._cached = solver_name
                    return solver_name
            except Exception:
                continue
        find_lp_solver._cached = cp.ECOS
    return find_lp_solver._cached


def build_conv_matrix(a):
    """Build the convolution matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, solver):
    """Solve LP: min M s.t. conv(a,b) <= M, sum(b)=1, b>=0."""
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    ones = np.ones(2 * n - 1)
    constraints = [
        T @ b_var <= M_var * ones,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return a.copy(), float('inf')
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float('inf')
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return a.copy(), float('inf')
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else float('inf')
    return b_val, M_val


def exact_coord_descent(a0, time_limit, verbose=True, label=""):
    """
    Exact coordinate descent using bisection on the 1D convex subproblem.
    For each coordinate i, finds the optimal delta by binary searching
    where the subgradient of the 1D convex function contains zero.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve_sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-18:
                continue

            k_start = i
            k_end = i + n - 1

            # Unaffected max
            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:2 * n - 1].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            # Affected conv values: conv[k] for k in [i, i+n-1]
            v = conv[k_start:k_end + 1]
            # Slopes: a[k-i] for k in [i, i+n-1] which is just a[0:n] = a
            s_arr = a
            # Quadratic term index: k=2i corresponds to index (2i - k_start) = i in affected array
            quad_idx = i

            # Bisection to find optimal delta
            lo = -a[i] * 1.001
            hi = a[i] * 3.0 + 1.0 / n

            for _ in range(50):
                mid = (lo + hi) / 2.0
                # Compute values at mid
                vals = v + 2.0 * mid * s_arr
                vals = vals.copy()
                vals[quad_idx] += mid * mid
                j_star = int(np.argmax(vals))

                if j_star == quad_idx:
                    g = 2.0 * s_arr[quad_idx] + 2.0 * mid
                else:
                    g = 2.0 * s_arr[j_star]

                # Check if unaff_max dominates
                if unaff_max > vals.max() + 1e-15:
                    # Function is flat (constant = unaff_max), any delta works
                    # Prefer smaller |delta| for stability
                    if abs(mid) < abs(lo) and abs(mid) < abs(hi):
                        break
                    # Move toward 0
                    if lo < 0 < hi:
                        break
                    if lo < 0:
                        hi = mid
                    else:
                        lo = mid
                    continue

                if g > 1e-14:
                    hi = mid
                elif g < -1e-14:
                    lo = mid
                else:
                    break

            best_delta = (lo + hi) / 2.0
            if best_delta < -a[i]:
                best_delta = -a[i]

            # Evaluate at optimal delta
            new_M_candidate = unaff_max
            vals_new = v + 2.0 * best_delta * s_arr
            vals_new = vals_new.copy()
            vals_new[quad_idx] += best_delta * best_delta
            aff_max = float(vals_new.max())
            new_M = max(unaff_max, aff_max)

            if new_M < M - 1e-14:
                # Update a[i]
                old_ai = a[i]
                a[i] = old_ai + best_delta
                if a[i] < 0:
                    a[i] = 0.0
                    best_delta = -old_ai
                    # Recompute
                    vals_new = v + 2.0 * best_delta * s_arr
                    vals_new = vals_new.copy()
                    vals_new[quad_idx] += best_delta * best_delta
                    new_M = max(unaff_max, float(vals_new.max()))

                # Update convolution incrementally
                changes = 2.0 * best_delta * s_arr
                changes = changes.copy()
                changes[quad_idx] = 2.0 * best_delta * old_ai + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps > 15:
                break
        else:
            no_improve_sweeps = 0

        if verbose and sweeps % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ECD[{label}] sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f} M={M:.10f}")

    return best_a, best_score


def frank_wolfe_alternating(a0, time_limit, verbose=True, label=""):
    """
    Frank-Wolfe (conditional gradient) method for minimizing max_k conv(a,a)[k]
    over the simplex. Used as a fast global move between ECD sweeps.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        k_star = int(np.argmax(conv))

        # Subgradient: g[j] = 2*a[k_star - j] (for valid j)
        # This is 2*a reversed around position k_star
        g = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(n - 1, k_star)
        for j in range(j_start, j_end + 1):
            g[j] = 2.0 * a[k_star - j]

        # FW step: find j* = argmin_j g[j] (among all j, not just active)
        j_star_fw = int(np.argmin(g))
        if g[j_star_fw] >= M * 0.001:
            # No improvement possible with a single vertex
            # Try a combo step
            no_improve += 1
            if no_improve > 100:
                break
            # Perturb: pick random coordinate with small g
            candidates = np.where(g < g[j_star_fw] * 1.5 + 1e-10)[0]
            if len(candidates) > 0:
                j_star_fw = int(candidates[np.random.randint(len(candidates))])
            else:
                break

        # Combo step (lazy FW): find best alpha in [0,1]
        # a_new = (1-alpha)*a + alpha*e_{j_star}
        # This changes a[j_star] by alpha*(1/n - a[j_star]) approximately
        # (since e_{j_star} has 1 at position j_star and 0 elsewhere, but we're on simplex)
        # Actually e_{j_star} is a unit vector, so sum is 1.
        # a_new[j] = (1-alpha)*a[j] for j != j_star
        # a_new[j_star] = (1-alpha)*a[j_star] + alpha

        # Try several alpha values
        best_alpha = 0.0
        best_new_M = M
        for alpha in [0.01, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7]:
            delta = alpha * (1.0 - a[j_star_fw])  # change in a[j_star]
            # Other coordinates change by -alpha * a[j] each
            # conv change is complex; just evaluate
            a_trial = (1.0 - alpha) * a
            a_trial[j_star_fw] += alpha
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-15:
                continue
            a_trial = a_trial / ss
            c_trial = fast_convolve(a_trial)
            m_trial = float(c_trial.max())
            if m_trial < best_new_M - 1e-14:
                best_new_M = m_trial
                best_alpha = alpha

        if best_alpha > 0:
            a = (1.0 - best_alpha) * a
            a[int(j_star_fw)] += best_alpha
            a = np.maximum(a, 0)
            ss = a.sum()
            a = a / ss
            conv = fast_convolve(a)
            M = float(conv.max())
            cur_score = 2.0 * n * M
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 200:
            break

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW[{label}] it={it} ({elapsed:.0f}s): best={best_score:.8f} M={M:.10f}")

    return best_a, best_score


def subgradient_nesterov(a0, time_limit, verbose=True, label=""):
    """
    Nesterov-accelerated subgradient descent with multi-peak weighting.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    # Nesterov momentum
    y = a.copy()  # lookahead point
    v = np.zeros(n)  # velocity
    mu = 1.0  # momentum parameter (starts at 1, increases)

    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    lr = 0.1
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        mu = (1.0 + np.sqrt(1.0 + 4.0 * mu * mu)) / 2.0
        t_k = (1.0 + mu) / (2.0 + mu)  # Nesterov step size parameter

        # Evaluate at lookahead point y
        conv_y = fast_convolve(y)
        M_y = float(conv_y.max())

        # Multi-peak subgradient at y
        k_act = min(30, 2 * n - 1)
        peaks = np.argpartition(conv_y, -k_act)[-k_act:]
        pv = conv_y[peaks]
        temp = 0.005 * max(M_y, 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / ws

        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            sg[i_start:i_end + 1] += w[idx] * 2.0 * y[k - i_end:k - i_start + 1][::-1]

        # Nesterov update
        v = mu * v - lr * sg / (it + 1)
        y_new = simplex_project(y + v)

        # Update a
        a = (1.0 - t_k) * a + t_k * y_new
        a = np.maximum(a, 0)
        ss = a.sum()
        if ss < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / ss
        y = a.copy()

        conv = fast_convolve(a)
        M = float(conv.max())
        cur_score = 2.0 * n * M
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300:
            lr *= 0.8
            no_improve = 0
            if lr < 1e-10:
                break

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NSGD[{label}] it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def diverse_init(n, rng, init_type=0):
    """Generate diverse initializations."""
    if init_type == 0:
        t = np.arange(n)
        center = (n - 1) / 2.0
        a = np.maximum(1.0 - np.abs(t - center) / center, 0)
    elif init_type == 1:
        t = np.arange(n)
        center = (n - 1) / 2.0
        x = (t - center) / center
        a = 0.3 + 0.7 * x * x
    elif init_type == 2:
        t = np.arange(n)
        center = (n - 1) / 2.0
        x = (t - center) / center
        power = rng.uniform(1.0, 3.0)
        a = np.maximum(0.5 * (1 + np.cos(np.pi * x)), 0) ** power
    elif init_type == 3:
        t = np.arange(n)
        c1 = n * rng.uniform(0.2, 0.35)
        c2 = n * rng.uniform(0.65, 0.8)
        w = n * rng.uniform(0.08, 0.2)
        a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    elif init_type == 4:
        period = int(rng.integers(3, 10))
        t = np.arange(n)
        a = 0.5 + 0.4 * np.cos(2 * np.pi * t / period)
    elif init_type == 5:
        # Comb pattern: periodic spikes
        t = np.arange(n)
        period = int(rng.integers(4, 12))
        width = max(1, period // 8)
        a = np.zeros(n)
        for k in range(0, n, period):
            for w_idx in range(width):
                if k + w_idx < n:
                    a[k + w_idx] = 1.0
    elif init_type == 6:
        # Smooth noise
        freqs = rng.integers(1, 15, size=min(10, n))
        phases = rng.uniform(0, 2 * np.pi, size=len(freqs))
        coeffs = rng.uniform(0.1, 1.0, size=len(freqs))
        t = np.arange(n)
        a = np.ones(n)
        for fi in range(len(freqs)):
            a += coeffs[fi] * np.cos(2 * np.pi * freqs[fi] * t / n + phases[fi])
        a = np.maximum(a, 0)
    elif init_type == 7:
        # Bimodal with gap
        t = np.arange(n)
        gap = int(n * rng.uniform(0.1, 0.3))
        a = np.zeros(n)
        a[:n//2 - gap//2] = 1.0
        a[n//2 + gap//2:] = 1.0
        a = a + 0.1 * np.random.standard_normal(n)
        a = np.maximum(a, 0)
    else:
        a = np.maximum(1.0 + 0.5 * np.random.standard_normal(n), 0)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def lp_alternating(a0, time_limit, solver, verbose=True):
    """LP-based best-response alternating minimization with adaptive damping."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    conv_a = fast_convolve(a)
    best_score = 2.0 * n * float(conv_a.max())
    t0 = time.monotonic()
    it = 0
    stagnation = 0
    prev_score = best_score

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2.0:
            break

        b_new, M_lp = lp_best_response(a, solver)
        if not np.isfinite(M_lp):
            break

        score_b = 2.0 * n * float(fast_convolve(b_new).max())
        if score_b < best_score:
            best_score = score_b
            best_a = b_new.copy()

        # Adaptive damping
        gap = abs(score_b - 2.0 * n * float(fast_convolve(a).max()))
        if it <= 10:
            alpha = 1.0
        elif it <= 30:
            alpha = 0.8
        elif it <= 60:
            alpha = 0.6
        else:
            alpha = max(0.3, 8.0 / (it + 5.0))

        a = (1 - alpha) * a + alpha * b_new
        a = np.maximum(a, 0)
        ss = a.sum()
        if ss < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / ss

        score_a = 2.0 * n * float(fast_convolve(a).max())
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        # Stagnation detection
        if score_a > prev_score * (1 + 1e-6):
            stagnation += 1
        else:
            stagnation = max(0, stagnation - 1)
        prev_score = score_a

        # Random perturbation when stuck
        if stagnation > 20:
            noise_scale = 0.02 / np.sqrt(n)
            a_pert = a + np.random.standard_normal(n) * noise_scale
            a_pert = np.maximum(a_pert, 0)
            ps = a_pert.sum()
            if ps > 1e-10:
                a_pert = a_pert / ps
                score_pert = 2.0 * n * float(fast_convolve(a_pert).max())
                if score_pert < best_score * 1.005:
                    a = a_pert
            stagnation = 0

        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interleaved_search(a0, solver, time_limit, verbose=True):
    """
    Interleave ECD, FW, and LP for balanced exploration/exploitation.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()
    round_num = 0

    while time.monotonic() - t0 < time_limit:
        round_num += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 30:
            break

        # ECD phase: 40% of remaining (min 20s)
        t_ecd = min(remaining * 0.4, 80.0)
        if t_ecd > 15:
            a_ecd, s_ecd = exact_coord_descent(a, t_ecd, verbose=False, label=f"r{round_num}")
            if 2.0 * n * s_ecd < best_score:
                best_score = 2.0 * n * s_ecd
                best_a = a_ecd.copy()
                a = a_ecd.copy()
                if verbose:
                    print(f"    ECD improved: {best_score:.8f}")

        # FW phase: 20% of remaining (min 10s)
        remaining = time_limit - (time.monotonic() - t0)
        t_fw = min(remaining * 0.2, 40.0)
        if t_fw > 10 and time.monotonic() - t0 < time_limit - 30:
            a_fw, s_fw = frank_wolfe_alternating(a, t_fw, verbose=False, label=f"r{round_num}")
            sc_fw = 2.0 * n * s_fw
            if sc_fw < best_score:
                best_score = sc_fw
                best_a = a_fw.copy()
                a = a_fw.copy()
                if verbose:
                    print(f"    FW improved: {best_score:.8f}")

        # LP phase: solve LP once
        remaining = time_limit - (time.monotonic() - t0)
        if remaining > 60:
            try:
                b_lp, M_lp = lp_best_response(a, solver)
                if np.isfinite(M_lp):
                    sc_lp = 2.0 * n * float(fast_convolve(b_lp).max())
                    if sc_lp < best_score:
                        best_score = sc_lp
                        best_a = b_lp.copy()
                        if verbose:
                            print(f"    LP improved: {best_score:.8f}")
                    # Damped update
                    alpha = 0.7
                    a = (1 - alpha) * a + alpha * b_lp
                    a = np.maximum(a, 0)
                    ss = a.sum()
                    if ss > 1e-15:
                        a = a / ss
            except Exception:
                pass

        if verbose and round_num % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Interleaved round={round_num} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def propose_candidate():
    """
    Multi-phase optimization combining exact coordinate descent, Frank-Wolfe,
    LP best-response, and Nesterov-accelerated subgradient methods.
    """
    deadline = time.monotonic() + 1770.0
    rng = np.random.default_rng(42)

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is not None:
        n = len(parent)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    else:
        n = 1024
        best_seq = np.ones(n) / n
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Using default init: n={n}, score={best_score:.8f}")

    solver = find_lp_solver()
    print(f"LP solver: {solver}")

    def try_update(candidate, label):
        nonlocal best_seq, best_score
        candidate = np.maximum(candidate, 0.0)
        s = candidate.sum()
        if s < 0.01:
            return
        candidate = candidate / s
        try:
            sc = float(evaluate_sequence(candidate.tolist()))
        except Exception:
            sc = float('inf')
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = candidate.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(candidate)})")

    # === Phase 1: Interleaved ECD + FW + LP (main workhorse, 50%) ===
    print(f"\n=== Phase 1: Interleaved Search (best={best_score:.8f}) ===")
    t1_budget = (deadline - time.monotonic()) * 0.50
    print(f"  Budget: {t1_budget:.0f}s")
    a1, s1 = interleaved_search(best_seq, solver, min(t1_budget, 600.0), verbose=True)
    try_update(a1, "interleaved")

    # === Phase 2: Focused ECD on best (25%) ===
    print(f"\n=== Phase 2: Focused ECD (best={best_score:.8f}) ===")
    t2_budget = (deadline - time.monotonic()) * 0.25
    print(f"  Budget: {t2_budget:.0f}s")
    a2, s2 = exact_coord_descent(best_seq, min(t2_budget, 300.0), verbose=True, label="focused")
    try_update(a2, "ecd_focused")

    # === Phase 3: LP alternating from best (15%) ===
    print(f"\n=== Phase 3: LP Alternating (best={best_score:.8f}) ===")
    t3_budget = (deadline - time.monotonic()) * 0.15
    if t3_budget > 30:
        print(f"  Budget: {t3_budget:.0f}s")
        a3, s3 = lp_alternating(best_seq, min(t3_budget, 200.0), solver, verbose=True)
        try_update(a3, "lp_alt")

    # === Phase 4: Nesterov SGD for exploration (5%) ===
    print(f"\n=== Phase 4: Nesterov SGD (best={best_score:.8f}) ===")
    t4_budget = (deadline - time.monotonic()) * 0.05
    if t4_budget > 20:
        print(f"  Budget: {t4_budget:.0f}s")
        a4, s4 = subgradient_nesterov(best_seq, min(t4_budget, 100.0), verbose=True, label="nest")
        try_update(a4, "nesterov")

    # === Phase 5: Diverse restarts with quick ECD (5%) ===
    print(f"\n=== Phase 5: Diverse Restarts (best={best_score:.8f}) ===")
    t5_end = deadline - 30
    init_types = [5, 6, 7, 0, 2]  # comb, smooth noise, bimodal, triangle, cosine
    for itype in init_types:
        if time.monotonic() > t5_end - 20:
            break
        init = diverse_init(n, rng, itype)
        # Quick ECD for 15s
        t_quick = min(15.0, t5_end - time.monotonic() - 10)
        if t_quick < 5:
            break
        a_r, s_r = exact_coord_descent(init, t_quick, verbose=False, label=f"restart{itype}")
        try_update(a_r, f"restart_{itype}")

    # === Final polish ===
    print(f"\n=== Final Polish (best={best_score:.8f}) ===")
    t_final = min(deadline - time.monotonic() - 5, 60.0)
    if t_final > 10:
        a_fp, s_fp = exact_coord_descent(best_seq, t_final, verbose=False, label="final")
        try_update(a_fp, "final_polish")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
