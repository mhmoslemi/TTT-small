_LP_SOLVER = None

def fast_convolve(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):

    s = a.sum()
    if s < 0.01:

    conv = fast_convolve(a)
    return 2.0 * n * float(conv.max()) / (s * s)

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = int(np.nonzero(cond)[0][-1])
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def smooth_max_grad(conv, tau):

    shifted = conv - conv.max()
    exp_vals = np.exp(tau * shifted)
    p = exp_vals / exp_vals.sum()
    return p

def smoothed_fista(a0, time_limit, verbose=False, label=""):
    """
    FISTA on the smoothed objective:
    f_tau(a) = (1/tau) * log(sum_k exp(tau * conv[k]))
    subject to: sum(a) = 1, a >= 0
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()
    
    # tau: smoothing parameter (larger = closer to max)
    tau = 30.0
    # FISTA parameters
    step = 1.0  # will adapt
    k_momentum = 1.0
    a_prev = a.copy()
    a_curr = a.copy()
    
    no_improve = 0
    iters = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1:
            break
        
        # Compute gradient of smoothed objective at a_curr
        conv = fast_convolve(a_curr)
        p = smooth_max_grad(conv, tau)
        
        # grad_i = 2 * sum_j p[i+j] * a[j] for valid indices
        # This is: 2 * (p_reversed * a) but only the "middle" part
        # Actually: grad[i] = 2 * sum_{j=0}^{n-1} p[i+j] * a[j] where i+j in [0, 2n-2]
        # This is a cross-correlation-like operation
        # grad = 2 * convolve(p, a_reversed) evaluated at specific positions
        # Let's compute it directly using FFT
        grad = np.zeros(n)
        # p has length 2n-1, a has length n
        # grad[i] = 2 * sum_{j=0}^{n-1} p[i+j] * a[j]
        # = 2 * (p * a_flipped)[i + n - 1] where * is linear convolution
        # Let's verify: (p * a_flipped)[k] = sum_j p[k-j] * a_flipped[j] = sum_j p[k-j] * a[n-1-j]
        # For k = i + n - 1: sum_j p[i + n - 1 - j] * a[n-1-j]
        # Let m = n-1-j: = sum_m p[i + n - 1 - (n-1-m)] * a[m] = sum_m p[i+m] * a[m]
        # Yes! So grad[i] = 2 * (p * a_flipped)[i + n - 1]
        
        a_flipped = a_curr[::-1]
        conv_len = (2*n-1) + n - 1  # length of full conv of p and a_flipped
        fft_size = 1 << (conv_len - 1).bit_length()
        P = rfft(p, fft_size)
        AF = rfft(a_flipped, fft_size)
        conv_pa = irfft(P * AF, fft_size)[:conv_len]
        grad = 2.0 * conv_pa[n-1:n-1+n]
        
        # Step size: use 1/L where L is a Lipschitz constant
        # For simplicity, use backtracking or fixed step
        # The Hessian of f_tau is bounded by 4*tau (roughly)
        step_size = 1.0 / (4.0 * tau * a_curr.max() * 2 * n)
        # Actually, let's use a simpler adaptive step
        step_size = 0.01 / max(a_curr.max(), 1e-10)
        
        # FISTA update
        a_next = simplex_project(a_curr - step_size * grad)
        
        # Momentum
        k_new = (1.0 + np.sqrt(1.0 + 4.0 * k_momentum * k_momentum)) / 2.0
        a_prev_new = a_next + ((k_momentum - 1.0) / k_new) * (a_next - a_curr)
        
        a_curr = a_next
        a_prev = a_prev_new
        k_momentum = k_new
        
        # Evaluate and track best
        conv_curr = fast_convolve(a_curr)
        score_curr = 2.0 * n * float(conv_curr.max())
        if score_curr < best_score - 1e-12:
            best_score = score_curr
            best_a = a_curr.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Also check the extrapolated point
        conv_prev = fast_convolve(a_prev)
        score_prev = 2.0 * n * float(conv_prev.max())
        if score_prev < best_score:
            best_score = score_prev
            best_a = a_prev.copy()
        
        # Gradually increase tau (less smoothing = closer to true max)
        if iters % 50 == 0:
            tau *= 1.05
            k_momentum = 1.0
            a_prev = a_curr.copy()
            
        if verbose and iters % 100 == 0:
            print(f"    FISTA {label} it={iters}: score={best_score:.8f}, tau={tau:.1f}")
        
        if no_improve > 500:
            break
    
    return best_a, best_score, iters
