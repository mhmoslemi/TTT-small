import numpy as np
from scipy.fft import rfft, irfft
import time

_lp_solver = None

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    import cvxpy as cp
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.PDLP, cp.SCS, cp.GLOP, cp.GLPK, cp.ECOS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def solve_lp_best_response(a_fixed, active_indices=None, solver=None):
    """Solve LP to find b minimizing max_{k in active} (a*b)_k subject to sum(b)=1, b>=0.
    If active_indices is None, use all lags."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    T = build_conv_matrix(a_fixed)
    
    if active_indices is not None:
        T = T[active_indices, :]
        m = len(active_indices)
    else:
        m = 2 * n - 1
    
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [
        T @ b_var <= t_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.PDLP, cp.GLPK, cp.SCS, cp.GLOP, cp.GUROBI, cp.MOSEK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(t_var.value)

def smart_coord_descent(a0, time_limit, verbose=False, label=""):

    n = len(a0)

        a = a / s

    conv = fast_convolve(a)

    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):

            if ai < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

                unaff_left = float(conv[:k_start].max())
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:

            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            affected = conv[k_start:k_end + 1]
            coeffs = a[(k_start - i):(k_end - i + 1)] + ai
            has_quad = (2 * i >= k_start) and (2 * i <= k_end)
            quad_idx = 2 * i - k_start if has_quad else -1

            candidates = [0.0, -ai]

            for j in range(len(affected)):
                c = coeffs[j]
                if c > 1e-15:
                    d = (unaff_max - affected[j]) / c
                    if -ai * 1.5 < d < ai * 1.5:

            top_k = min(8, len(affected))

            for j1 in range(len(top_idx)):

                    if abs(denom) > 1e-15:

                        if -ai * 1.5 < d < ai * 1.5:

            for frac in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:

            best_delta = 0.0
            best_new_max = M

            for delta in candidates:

                delta = max(delta, -ai)
                if abs(delta) < 1e-18:
                    continue

                new_affected = affected + delta * coeffs
                if has_quad:
                    new_affected = new_affected.copy()
                    new_affected[quad_idx] += delta * delta

                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-15:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-14:
                a[i] = max(ai + best_delta, 0.0)
                conv[k_start:k_end + 1] += best_delta * coeffs
                if has_quad:
                    conv[2 * i] += best_delta * best_delta
                M = best_new_max

                cur_score = 2.0 * n * M

                    best_score = cur_score

                conv[k_start:k_end + 1] += best_delta * coeffs
                if has_quad:
                    conv[2 * i] += best_delta * best_delta
                M = best_new_max
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved and sweeps > 20:
            break

    return best_a, best_score

def lp_fictitious_play(a0, time_limit, solver=None, label=""):
    """LP fictitious play with adaptive damping and direct LP evaluation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1

        # Solve best response
        b_new, M_b = solve_lp_best_response(a, solver=solver)
        if b_new is None:
            continue

        # Adaptive damping: faster early, slower later
        alpha = min(1.0, 15.0 / (iters + 10))
        
        # Averaged update
        a_new = (1 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        # Also try the LP solution directly (it might be better than the blend)
        score_b = 2.0 * n * fast_convolve(b_new).max()
        if score_b < best_score - 1e-14:
            best_score = score_b
            best_a = b_new.copy()

        # Try a blend with different alpha
        if iters % 3 == 0:
            a_blend = 0.5 * a + 0.5 * b_new
            a_blend = np.maximum(a_blend, 0)
            s_blend = a_blend.sum()
            if s_blend > 1e-15:
                a_blend = a_blend / s_blend
                score_blend = 2.0 * n * fast_convolve(a_blend).max()

        if iters % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LPFP {label} it={iters} ({elapsed:.0f}s): best={best_score:.6f}, M_b={M_b:.8f}")

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.max(np.arange(1, n+1)[(u * np.arange(1, n+1) - (cssv - 1)) > 0])
    theta = (cssv[rho-1] - 1.0) / rho
    return np.maximum(v - theta, 0.0)

def balanced_mirror_descent(a0, time_limit, label=""):
    """Mirror descent with balanced active-set gradient."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    lr = 0.5  # learning rate for mirror descent

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        
        # Find active set
        threshold = M * 0.95
        active = np.where(conv >= threshold)[0]
        
        # Compute balanced gradient
        # g_i = sum_{k in active} w_k * 2 * a_{k-i}
        # Use equal weights for simplicity
        grad = np.zeros(n)
        for k in active:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            grad[j] += 2.0 * a[k - j]
        
        # Normalize gradient
        g_max = grad.max()
        if g_max < 1e-15:
            break
        grad = grad / g_max
        
        # Mirror descent step
        log_a = np.log(a + 1e-30)
        log_a -= lr * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = simplex_project(a_new)
        
        # Evaluate
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        a = a_new
        
        # Decay learning rate
        lr *= 0.995
        if lr < 0.01:
            lr = 0.01
        
        if it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    BMD {label} it={it} ({elapsed:.0f}s): best={best_score:.6f}, M={M:.8f}")

    return best_a, best_score

def peak_equalization_move(a0, rng, label=""):
    """Try to equalize the top peaks by adjusting coordinates that contribute to multiple peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n

    best_score = 2.0 * n * M
    best_a = a.copy()
    
    # Find top peaks
    sorted_conv = np.sort(conv)[::-1]
    n_peaks = min(10, n)
    
    # For each pair of top peaks, try to reduce both simultaneously
    top_peaks = np.argsort(conv)[::-1][:n_peaks]
    
    for attempt in range(20):
        if len(top_peaks) < 2:
            break
        # Pick two peaks that are "close" in some sense
        k1, k2 = top_peaks[0], top_peaks[1]
        
        # Find coordinates that contribute to both peaks
        # Peak k1 involves a[i]*a[k1-i] for valid i
        # Peak k2 involves a[i]*a[k2-i] for valid i
        # Common contributors: i such that both k1-i and k2-i are in [0,n-1]
        delta_k = abs(k2 - k1)
        
        # Try reducing a coordinate that contributes heavily to both
        contrib_1 = np.zeros(n)
        contrib_2 = np.zeros(n)
        i_start1 = max(0, k1 - n + 1)
        i_end1 = min(n - 1, k1)
        for i in range(i_start1, i_end1 + 1):
            j1 = k1 - i
            contrib_1[i] = a[i] * a[j1]
        
        i_start2 = max(0, k2 - n + 1)
        i_end2 = min(n - 1, k2)
        for i in range(i_start2, i_end2 + 1):
            j2 = k2 - i
            contrib_2[i] = a[i] * a[j2]
        
        # Find coordinate with high contribution to both
        joint = contrib_1 + contrib_2
        # Weight by how much reducing helps both peaks
        help_both = np.minimum(contrib_1 / max(conv[k1], 1e-15), 
                               contrib_2 / max(conv[k2], 1e-15))
        target_idx = np.argmax(help_both * a)
        
        # Try reducing this coordinate and redistributing
        old_val = a[target_idx]
        if old_val < 1e-15:
            continue
        
        for frac in [0.1, 0.2, 0.3, 0.5]:
            reduce_amount = frac * old_val
            a_try = a.copy()
            a_try[target_idx] = old_val - reduce_amount
            # Redistribute to random other coordinates
            others = np.delete(np.arange(n), target_idx)
            redistribute_to = rng.choice(others, size=min(10, len(others)), replace=False)
            per_redist = reduce_amount / len(redistribute_to)
            a_try[redistribute_to] += per_redist
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            
            score = 2.0 * n * fast_convolve(a_try).max()
            if score < best_score - 1e-14:
                best_score = score
                best_a = a_try.copy()
                a = a_try.copy()
                conv = fast_convolve(a)
                M = conv.max()
    
    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    n = len(a_fixed)
    T_full = build_conv_matrix(a_fixed)

    if active_indices is not None:
        T = T_full[active_indices, :]
    else:
        T = T_full

    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [
        T @ b_var <= t_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(t_var), constraints)

    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.PDLP, cp.GLPK, cp.SCS, cp.GLOP, cp.GUROBI, cp.MOSEK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(t_var.value)


def simplex_project(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    cond = u * rho_candidates > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = rho_candidates[cond][-1]
    theta = (cssv[rho - 1] - 1.0) / rho
    return np.maximum(v - theta, 0.0)


def smart_coord_descent(a0, time_limit, verbose=False, label=""):
    """Coordinate descent with smart 1D candidate evaluation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            ai = a[i]
            if ai < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = float(conv[:k_start].max())
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = float(conv[k_end + 1:].max())
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            affected = conv[k_start:k_end + 1].copy()
            coeffs = a[(k_start - i):(k_end - i + 1)] + ai
            has_quad = (2 * i >= k_start) and (2 * i <= k_end)
            quad_idx = 2 * i - k_start if has_quad else -1

            candidates = [0.0, -ai]

            for j in range(len(affected)):
                c = coeffs[j]
                if c > 1e-15:
                    d = (unaff_max - affected[j]) / c
                    if -ai * 1.5 < d < ai * 1.5:
                        candidates.append(d)

            top_k = min(8, len(affected))
            top_idx = np.argsort(affected)[-top_k:]
            for j1 in range(len(top_idx)):
                for j2 in range(j1 + 1, len(top_idx)):
                    i1, i2 = top_idx[j1], top_idx[j2]
                    denom = coeffs[i1] - coeffs[i2]
                    if abs(denom) > 1e-15:
                        d = (affected[i2] - affected[i1]) / denom
                        if -ai * 1.5 < d < ai * 1.5:
                            candidates.append(d)

            for frac in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 0.9]:
                candidates.append(-frac * ai)
                candidates.append(frac * 0.3 * ai)

            best_delta = 0.0
            best_new_max = M

            for delta in candidates:
                if delta < -ai - 1e-12:
                    continue
                delta = max(delta, -ai)
                if abs(delta) < 1e-18:
                    continue

                new_affected = affected + delta * coeffs
                if has_quad:
                    new_affected = new_affected.copy()
                    new_affected[quad_idx] += delta * delta

                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-15:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-14:
                a[i] = max(ai + best_delta, 0.0)
                conv[k_start:k_end + 1] += best_delta * coeffs
                if has_quad:
                    conv[2 * i] += best_delta * best_delta
                M = best_new_max

                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if sweeps % 50 == 0:
            conv = fast_convolve(a)
            M = float(conv.max())
            cur_score = 2.0 * n * M
            if cur_score < best_score - 1e-14:
                best_score = cur_score
                best_a = a.copy()

        if not improved:
            no_improve += 1
            if no_improve > 200:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score, sweeps


def fine_coord_descent(a0, time_limit, label=""):
    """Very fine coordinate descent for final polishing with binary search refinement."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            ai = a[i]
            if ai < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_max = float(conv[:k_start].max())
            else:
                unaff_max = -1e30
            if k_end < 2 * n - 2:
                ur = float(conv[k_end + 1:].max())
                unaff_max = max(unaff_max, ur)

            affected = conv[k_start:k_end + 1]
            coeffs = a[(k_start - i):(k_end - i + 1)] + ai
            has_quad = (2 * i >= k_start) and (2 * i <= k_end)
            quad_idx = 2 * i - k_start if has_quad else -1

            best_delta = 0.0
            best_new_max = M

            # Fine grid search
            num_trials = 31
            deltas = np.linspace(-ai, ai * 0.02, num_trials)
            for delta in deltas:
                if abs(delta) < 1e-18:
                    continue
                new_affected = affected + delta * coeffs
                if has_quad:
                    na = new_affected.copy()
                    na[quad_idx] += delta * delta
                    new_affected = na
                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-16:
                    best_new_max = new_max
                    best_delta = delta

            # If we found a good delta, refine with binary search
            if best_delta != 0.0:
                lo = best_delta - abs(ai) * 0.005
                hi = best_delta + abs(ai) * 0.005
                lo = max(lo, -ai)
                for _ in range(8):
                    mid = (lo + hi) / 2.0
                    na = affected + mid * coeffs
                    if has_quad:
                        na = na.copy()
                        na[quad_idx] += mid * mid
                    nm = max(unaff_max, float(na.max()))
                    if nm < M - 1e-15:
                        lo = mid
                        if nm < best_new_max:
                            best_new_max = nm
                            best_delta = mid
                    else:
                        hi = mid

            if best_delta != 0.0 and best_new_max < M - 1e-15:
                a[i] = max(ai + best_delta, 0.0)
                conv[k_start:k_end + 1] += best_delta * coeffs
                if has_quad:
                    conv[2 * i] += best_delta * best_delta
                M = best_new_max
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved and sweeps > 20:
            break

    return best_a, best_score


def lp_fictitious_play(a0, time_limit, solver=None, label=""):
    """LP fictitious play with adaptive damping and direct LP evaluation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1

        b_new, M_b = solve_lp_best_response(a, solver=solver)
        if b_new is None:
            continue

        # Adaptive damping: use power-law decay
        alpha = min(1.0, 12.0 / (iters + 8) ** 0.8)

        a_new = (1 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        # Try LP solution directly
        score_b = 2.0 * n * fast_convolve(b_new).max()
        if score_b < best_score - 1e-14:
            best_score = score_b
            best_a = b_new.copy()

        # Try 50/50 blend
        if iters % 2 == 0:
            a_blend = 0.5 * a + 0.5 * b_new
            a_blend = np.maximum(a_blend, 0)
            s_blend = a_blend.sum()
            if s_blend > 1e-15:
                a_blend = a_blend / s_blend
                score_blend = 2.0 * n * fast_convolve(a_blend).max()
                if score_blend < best_score - 1e-14:
                    best_score = score_blend
                    best_a = a_blend.copy()

        if iters % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LPFP {label} it={iters} ({elapsed:.0f}s): best={best_score:.6f}, M_b={M_b:.8f}")

    return best_a, best_score, iters


def active_set_lp(a0, time_limit, solver=None, label=""):
    """LP focusing on active (peak) set for targeted improvement near optimum."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        M = conv.max()

        # Active set: lags within 3% of max
        threshold = M * 0.97
        active = np.where(conv >= threshold)[0]
        if 0 not in active:
            active = np.concatenate([active, [0]])

        b_new, M_b = solve_lp_best_response(a, active_indices=active, solver=solver)
        if b_new is None:
            continue

        alpha = min(1.0, 8.0 / (iters + 5))
        a_new = (1 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        score_b = 2.0 * n * fast_convolve(b_new).max()
        if score_b < best_score - 1e-14:
            best_score = score_b
            best_a = b_new.copy()

        if iters % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ASLP {label} it={iters} ({elapsed:.0f}s): best={best_score:.6f}, |A|={len(active)}")

    return best_a, best_score, iters


def balanced_mirror_descent(a0, time_limit, label=""):
    """Mirror descent with balanced active-set gradient (KKT-motivated)."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    lr = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()

        # Active set: top peaks
        threshold = M * 0.92
        active = np.where(conv >= threshold)[0]

        # Compute balanced gradient: g_i = sum_{k in active} (1/|A|) * 2 * a_{k-i}
        grad = np.zeros(n)
        w = 1.0 / len(active)
        for k in active:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            grad[j] += w * 2.0 * a[k - j]

        g_max = grad.max()
        if g_max < 1e-15:
            break
        grad = grad / g_max

        # Mirror descent
        log_a = np.log(a + 1e-30)
        log_a -= lr * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = simplex_project(a_new)

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        lr *= 0.99
        if lr < 0.05:
            lr = 0.05

        if it % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    BMD {label} it={it} ({elapsed:.0f}s): best={best_score:.6f}, M={M:.8f}")

    return best_a, best_score


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, best_seq, best_score, label=""):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score
