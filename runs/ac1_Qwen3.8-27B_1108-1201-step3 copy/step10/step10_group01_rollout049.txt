    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):

    if s < 0.01:

    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

                return _lp_solver

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np

import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


_lp_solver_name = None

def get_lp_solver():
    global _lp_solver_name
    if _lp_solver_name is not None:
        return _lp_solver_name
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_name = solver_name
                return _lp_solver_name
        except Exception:
            continue
    _lp_solver_name = cp.ECOS
    return _lp_solver_name


def solve_lp_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers_to_try = [solver] + [s for s in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP] if s != solver]
    for s in solvers_to_try:
        try:
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0)
                sm = b_val.sum()
                if sm < 1e-15:
                    continue
                return b_val / sm, float(M_var.value)
        except Exception:
            continue
    return None, None


def nesterov_lp(a0, time_limit, solver=None, max_iters=500000, stall_patience=15, seed=42):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall = 0
    rng = np.random.default_rng(seed)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b_new, M_lp = solve_lp_response(y, solver)
        if b_new is None:
            stall += 1
            if stall > 30:
                break
            continue

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 1000 == 0:
            if best_score < prev_best - 1e-10:
                stall = 0
            else:
                stall += 1
            prev_best = best_score
            if stall >= stall_patience:
                noise = rng.standard_normal(n) * 0.015 / np.sqrt(stall)
                y = np.maximum(y + noise, 0)
                y = y / y.sum()
                stall = 0
    return best_a, best_score


def lse_obj_and_grad(z, n, T_temp):
    a = np.exp(z - z.max())
    a = a / a.sum()
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp(b - max_b)
    w = exp_b / exp_b.sum()
    lse = max_b + np.log(exp_b.sum())
    fval = (2.0 * n / T_temp) * lse

    # grad_a[i] = 2 * sum_k w[k] * a[k-i] (FFT-accelerated)
    n_w = len(w)
    fft_size = 1 << (n + n_w - 1 - 1).bit_length()
    B = rfft(a[::-1], fft_size)
    W = rfft(w, fft_size)
    conv = irfft(B * W, fft_size)
    c = conv[n - 1: n - 1 + n]
    grad_a = 2.0 * c * (2.0 * n / T_temp)

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return fval, grad_z


def lbgfs_lse(a0, time_limit, T_stages=None, maxiter_total=20000, seed=0):
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [20.0, 10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005]

    n_stages = len(T_stages)
    iters_per = max(10, maxiter_total // n_stages)

    for T_stage in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func_z(z_local):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            f, g = lse_obj_and_grad(z_local, n, T_stage)
            return f

        def jac_z(z_local):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_local)
            f, g = lse_obj_and_grad(z_local, n, T_stage)
            return g

        try:
            res = minimize(
                fun=func_z, x0=z, jac=jac_z, method='L-BFGS-B',
                options={'maxiter': iters_per, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': iters_per * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    a_out = np.exp(z - z.max())
    a_out = a_out / a_out.sum()
    return a_out


def pairwise_cd(a0, time_limit, n_steps=100000, top_m=8, seed=0):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0
    rng = np.random.default_rng(seed)

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        max_b = b[peak_k]

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_contrib = idx[np.argsort(contribs)[::-1][:5]]

        improved = False
        # Try reducing top contributors and redistributing
        for pos in top_contrib:
            if a[pos] < 1e-12:
                continue
            for frac in [0.995, 0.99, 0.98, 0.97, 0.95]:
                delta = a[pos] * (1.0 - frac)
                if delta < 1e-15:
                    continue
                # Redistribute to a random position far from peak
                target_pos = pos
                for _ in range(3):
                    target_pos = rng.integers(0, n)
                    if abs(target_pos - peak_k) > n // 3:
                        break

                a_trial = a.copy()
                a_trial[pos] -= delta
                a_trial[target_pos] += delta
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st < 1e-15:
                    break
                a_trial = a_trial / st
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try pairwise swap: move mass from high to low
            pos_i = top_contrib[0] if len(top_contrib) > 0 else int(np.argmax(a))
            pos_j = int(np.argmin(a))
            if pos_i == pos_j:
                stall += 1
            else:
                delta = a[pos_i] * 0.01
                a_trial = a.copy()
                a_trial[pos_i] -= delta
                a_trial[pos_j] += delta
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-15:
                    a_trial = a_trial / st
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
            if not improved:
                stall += 1
                if stall > 500:
                    # Random restart perturbation
                    noise = rng.standard_normal(n) * 0.005
                    a = np.maximum(a + noise, 0)
                    a = a / a.sum()
                    stall = 0

        if step % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD {elapsed:.0f}s: score={best_score:.8f} stall={stall}")

    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
