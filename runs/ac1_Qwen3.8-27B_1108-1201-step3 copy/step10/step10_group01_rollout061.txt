This is fast to compute (just matrix-vector multiply with softmax weights).

```python
def mwu_best_response(a, n_iter=100, beta=10.0):
    """Multiplicative weight update for best response."""
    n = len(a)
    b = np.ones(n) / n
    T = build_conv_matrix(a)  # (2n-1) × n
    
    for t in range(n_iter):
        v = T @ b  # (2n-1,)
        # Softmax weights
        v_max = v.max()
        w = np.exp(beta * (v - v_max))
        w = w / w.sum()
        # Gradient
        g = T.T @ w  # (n,)
        # MWU update
        eta = 1.0 / beta  # step size
        b = b * np.exp(-eta * g)
        b = np.maximum(b, 1e-30)
        b = b / b.sum()
    
    return b
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast MWU-based Nesterov
accelerated alternation with FFT convolutions, occasional LP verification,
LogSumExp L-BFGS refinement, and multi-start diversity.
Key improvements: 10-100x faster best response via MWU+FFT enabling
many more outer iterations, adaptive temperature, and structured restarts.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp


def fast_convolve(a, b=None):
    """Linear convolution via FFT. If b is None, computes auto-convolution."""
    a = np.asarray(a, dtype=np.float64)
    if b is None:
        b = a
    else:
        b = np.asarray(b, dtype=np.float64)
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def evaluate_local(a):
    """Evaluate 2n*max(conv(a,a))/(sum(a))^2 with local computation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def mwu_best_response(a, n_mwu_iters=50, beta=10.0, eta=None):
    """
    Multiplicative Weight Update for best response.
    Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k].
    Uses FFT for fast convolution.
    """
    n = len(a)
    b = np.ones(n) / n
    if eta is None:
        eta = 1.0 / np.sqrt(beta)
    
    # Precompute: we need conv(a, b) at each step
    # and correlate(w, a) for the gradient
    
    for t in range(n_mwu_iters):
        # Compute v = conv(a, b)
        v = fast_convolve(a, b)  # length 2n-1
        
        # Softmax weights (top-heavy)
        v_max = v.max()
        # Clip to avoid overflow
        scaled = beta * (v - v_max)

        b = b * np.exp(-eta * g)
        b = np.maximum(b, 1e-30)
        s = b.sum()
        b = b / s

            sy = 1.0
```

def fast_convolve(a, b=None):

            if best_score < prev_best - 1e-10:
                stall = 0
            else:
                stall += 1
            prev_best = best_score

            if stall >= 8:
                noise = rng.standard_normal(n) * 0.01 / np.sqrt(stall)
                y = y + noise
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                else:
                    y = best_a.copy()
                stall = 0

            if it % print_interval == 0:
                elapsed = time.monotonic() - t0
                print(f"    N-MWU it={it} ({elapsed:.0f}s): best={best_score:.8f} beta={beta:.1f}")

    return best_a, best_score


def solve_lp_best_response(a_fixed, solver=None, max_iters=None):
    """Solve the LP best response exactly."""
    if solver is None:
        solver = _find_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    kwargs = {}
    if max_iters is not None:
        kwargs['max_iter'] = max_iters
    try:
        prob.solve(solver=solver, verbose=False, **kwargs)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.asarray(b_var.value, dtype=np.float64)
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s < 1e-15:
                return None, None
            b_val = b_val / s
            return b_val, float(M_var.value)
    except Exception:
        pass
    for alt in [cp.SCS, cp.ECOS, cp.CLARABEL, cp.PDLP]:
        if alt == solver:
            continue
        try:
            prob.solve(solver=alt, verbose=False, **kwargs)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s < 1e-15:
                    return None, None
                b_val = b_val / s
                return b_val, float(M_var.value)
        except Exception:
            continue
    return None, None


_lp_solver_cache = None


def _find_lp_solver():
    global _lp_solver_cache
    if _lp_solver_cache is not None:
        return _lp_solver_cache
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cache = solver_name
                return _lp_solver_cache
        except Exception:
            continue
    _lp_solver_cache = cp.ECOS
    return _lp_solver_cache


def logsumexp_lbgfs(a0, time_limit, T_stages=None, maxiter_per_stage=3000,
                    top_k_weight=3.0):
    """
    Multi-stage LogSumExp smoothed L-BFGS with peak emphasis.
    """
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [20.0, 12.0, 8.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5,
                    0.35, 0.25, 0.15, 0.1, 0.06, 0.04, 0.02, 0.01]

    n_stages = len(T_stages)
    iters_per = max(20, maxiter_per_stage // n_stages)

    for T_stage in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func_lse(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a_in = np.exp(z_in - z_in.max())
            a_in = a_in / a_in.sum()
            b_in = fast_convolve(a_in)
            max_b = b_in.max()
            exp_b = np.exp(np.clip((b_in - max_b) / T_stage, -40, 0))
            lse = max_b + T_stage * np.log(exp_b.sum())
            return (2.0 * n / T_stage) * lse * T_stage

        def jac_lse(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a_in = np.exp(z_in - z_in.max())
            a_in = a_in / a_in.sum()
            b_in = fast_convolve(a_in)
            max_b = b_in.max()
            exp_b = np.exp(np.clip((b_in - max_b) / T_stage, -40, 0))
            # Peak emphasis: boost weights for top peaks
            sorted_idx = np.argsort(b_in)[::-1]
            top_k = min(30, len(b_in))
            emphasis = np.ones(len(b_in))
            emphasis[sorted_idx[:top_k]] = top_k_weight
            w = exp_b * emphasis / (exp_b * emphasis).sum()
            g = zeros_and_fill_grad(a_in, w)
            S = np.dot(g, a_in)
            return a_in * (g - S)

        try:
            res = minimize(
                fun=func_lse, x0=z, jac=jac_lse, method='L-BFGS-B',
                options={'maxiter': iters_per, 'ftol': 1e-15,
                         'gtol': 1e-10, 'maxfun': iters_per * 500, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    a_out = np.exp(z - z.max())
    a_out = a_out / a_out.sum()
    return a_out


def zeros_and_fill_grad(a, w):
    """Compute gradient of sum_k w[k]*conv(a,a)[k] with respect to a."""
    n = len(a)
    b = fast_convolve(a)
    grad = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-14:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += w[k] * a[k - idx]
    grad *= 2.0
    return grad


def peak_coord_descent(a0, time_limit, n_steps=50000, top_m=8):
    """Coordinate descent targeting top convolution peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        top_peaks = np.argsort(b)[::-1][:min(top_m, len(b))]

        improved = False
        for peak_k in top_peaks[:3]:
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            order = np.argsort(contribs)[::-1][:4]
            for oi in order:
                pos = idx[oi]
                if a[pos] < 1e-12:
                    continue
                for factor in [0.99, 0.98, 0.97]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] * (1.0 - factor)
                    if diff < 1e-15:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] = others + diff * others / others_sum
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-12:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            peak_k = top_peaks[0]
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top3 = idx[np.argsort(contribs)[::-1][:3]]
            a_trial = a.copy()
            for pos in top3:
                if a[pos] > 1e-12:
                    a_trial[pos] *= 0.997
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                break
            a_trial = a_trial / st
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-12:
                a = a_trial
                best_score = score
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
                if stall > 500:
                    break

        if (step + 1) % 15000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakCD {elapsed:.0f}s: score={best_score:.8f}")

    return best_a, best_score


def subgradient_refine(a0, time_limit, n_steps=40000):
    """Subgradient method targeting active peak set."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.01
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.999
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st

        score = 2.0 * n * fast_convolve(a_trial).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 30:
            lr *= 0.9
            no_improve = 0
        lr = max(1e-7, lr)

        if (step + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad {elapsed:.0f}s: score={best_score:.8f} lr={lr:.6f}")

    return best_a, best_score


def structured_init(n, rng):
    """Create diverse structured initializations with small endpoints."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-5 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.35 * np.cos(2 * np.pi * 4 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.8 + x ** 1.8 + 0.05
    elif choice == 5:
        a = np.exp(-4 * (x - 0.3) ** 2) + 0.5 * np.exp(-4 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 1.0 / (1.0 + 8.0 * np.sin(np.pi * x) ** 2)
    elif choice == 7:
        a = np.cos(np.pi * (x - 0.5)) ** 3
        a = np.maximum(a, 0)
    elif choice == 8:
        a = 0.3 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.2 * np.cos(2 * np.pi * 5 * x)
        a = np.maximum(a, 0.01)
    else:
        a = np.exp(-3 * (x - 0.2)) ** 2 + np.exp(-3 * (x - 0.8)) ** 2 + 0.1
    a = np.maximum(a, 0)
    # Taper endpoints to reduce edge peaks
    taper = min(10, n // 10)
    if taper > 1:
        for i in range(taper):
            scale = (i + 1) / (taper + 1)
            a[i] *= scale
            a[n - 1 - i] *= scale
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def nesterov_lp_mwu_hybrid(a0, time_limit, solver=None, lp_every=300,
                            n_mwu_iters=25, n_warm=10,
                            beta_start=3.0, beta_end=60.0):
    """
    Hybrid: MWU for speed, occasional LP for exactness.
    Nesterov accelerated.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    avg_b = a.copy()
    count = 0
    it = 0
    prev_best = best_score
    stall = 0
    rng = np.random.default_rng(777)
    b_warm = a.copy()

    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, it / 6000.0)
        beta = beta_start + (beta_end - beta_start) * progress

        if (solver is not None) and (it % lp_every == 0) and (it > 0):
            # Occasional exact LP
            elapsed = time.monotonic() - t0
            if time_limit - elapsed > 5:
                b_lp, M_lp = solve_lp_best_response(y, solver)
                if b_lp is not None:
                    b_warm = b_warm * 0.3 + b_lp * 0.7
                    b_warm = np.maximum(b_warm, 1e-15)
                    b_warm = b_warm / b_warm.sum()
                    b_new = b_warm
                else:
                    n_it = n_warm if it < 200 else n_mwu_iters
                    b_new = mwu_best_response(y, b_warm, n_it, beta)
            else:
                n_it = n_warm if it < 200 else n_mwu_iters
                b_new = mwu_best_response(y, b_warm, n_it, beta)
        else:
            n_it = n_warm if it < 200 else n_mwu_iters
            b_new = mwu_best_response(y, b_warm, n_it, beta)
            b_warm = b_warm * 0.5 + b_new * 0.5
            b_warm = np.maximum(b_warm, 1e-15)
            b_warm = b_warm / b_warm.sum()

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        count += 1
        avg_b = ((count - 1) * avg_b + b_new) / count

        score_avg = 2.0 * n * fast_convolve(avg_b).max()
        if score_avg < best_score:
            best_score = score_avg
            best_a = avg_b.copy()
            stall = 0

        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0

        it += 1

        if it % 500 == 0:
            if best_score < prev_best - 1e-10:
                stall = 0
            else:
                stall += 1
            prev_best = best_score

            if stall >= 10:
                noise_scale = 0.015 / np.sqrt(stall)
                noise = rng.standard_normal(n) * noise_scale
                y = y + noise
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                stall = 0

            if it % 3000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    Hybrid it={it} ({elapsed:.0f}s): best={best_score:.8f} beta={beta:.1f}")

    return best_a, best_score
