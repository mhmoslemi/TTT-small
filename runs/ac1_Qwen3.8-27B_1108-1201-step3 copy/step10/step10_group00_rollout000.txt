def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def get_subgradient(a, active_peaks):
    """
    Compute the average subgradient of max_k (conv(a,a))_k over active peaks.
    
    For peak at position k: grad_j = 2*a[k-j] if 0 <= k-j < n, else 0.
    This is equivalent to 2 * (reversed a) shifted by (n-1-k).
    """
    n = len(a)
    a_rev = a[::-1].copy()
    grad = np.zeros(n)
    
    for k in active_peaks:
        # grad_j = 2 * a[k-j] for j in [max(0, k-n+1), min(n-1, k)]
        # In terms of a_rev[i] = a[n-1-i]:
        # a[k-j] = a_rev[n-1-k+j]
        # So grad[j] = 2 * a_rev[n-1-k+j]
        # This means grad is 2*a_rev shifted right by (n-1-k) positions
        
        shift = n - 1 - k
        if shift >= 0:
            # grad[j] = 2*a_rev[shift+j] for j = 0 to n-1-shift
            grad[:n-shift] += 2.0 * a_rev[shift:]
        else:
            # shift is negative, so grad[j] = 2*a_rev[j+shift] for j=-shift to n-1
            grad[-shift:] += 2.0 * a_rev[:n+shift]
    
    grad /= len(active_peaks)
    return grad

def mirror_descent_refine(a0, time_limit, n_steps, eta0=1.0):
    """Mirror descent with Polyak averaging on the simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Polyak averaging
    avg_a = np.zeros(n)
    
    for t in range(1, n_steps + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Active set
        threshold = max_b * 0.995
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        
        # Subgradient
        grad = get_subgradient(a, active)
        
        # Mirror descent step
        eta = eta0 / np.sqrt(t)
        log_w = -eta * grad
        log_w -= log_w.max()
        unnormalized = a * np.exp(log_w)
        s_new = unnormalized.sum()
        if s_new < 1e-300:
            a = np.ones(n) / n
        else:
            a = unnormalized / s_new
        
        # Polyak averaging
        avg_a += a / t
        
        score = 2.0 * n * fast_convolve(a).max()
        avg_score = 2.0 * n * fast_convolve(avg_a / (avg_a.sum() + 1e-300)).max()
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
        if avg_score < best_score:
            best_score = avg_score
            best_a = (avg_a / (avg_a.sum() + 1e-300)).copy()
    
    return best_a, best_score

def solve_lp_best_response(a_fixed, solver=None):
    """LP: given a fixed, find b>=0, sum(b)=1 minimizing max_k (T@b)_k."""
    import cvxpy as cp
    from scipy.sparse import csr_matrix
    
    n = len(a_fixed)
    size = 2 * n - 1
    
    # Build sparse convolution matrix
    rows, cols, vals = [], [], []
    for j in range(n):
        for k in range(j, j + n):
            rows.append(k)
            cols.append(j)
            vals.append(a_fixed[j])
    
    T = csr_matrix((vals, (rows, cols)), shape=(size, n))
    
    b_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    
    constraints = [T @ b_var <= t_var, cp.sum(b_var) == 1]
    objective = cp.Minimize(t_var)
    prob = cp.Problem(objective, constraints)
    
    solvers_to_try = [solver, cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS, cp.SCS]
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_b = b_val.sum()
                if s_b < 1e-15:
                    return None, None
                b_val = b_val / s_b
                return b_val, float(max(0, t_var.value))
        except Exception:
            continue
    
    return None, None

def lbgfs_logsumexp_refine(a0, time_limit, T_stages, maxiter_per_stage=5000):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    n_stages = len(T_stages)
    iters_per_stage = max(10, maxiter_per_stage // n_stages)
    
    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        def func(z_val):
            if time.monotonic() > deadline:
                return 1e15
            a_val = softmax_from_z(z_val)
            b_val = fast_convolve(a_val)
            max_b = b_val.max()
            exp_terms = np.exp((b_val - max_b) / stage_T)
            lse = max_b + stage_T * np.log(exp_terms.sum())
            return 2.0 * n * lse
        
        def jac(z_val):
            if time.monotonic() > deadline:
                return np.zeros_like(z_val)
            a_val = softmax_from_z(z_val)
            b_val = fast_convolve(a_val)
            max_b = b_val.max()
            exp_terms = np.exp((b_val - max_b) / stage_T)
            w = exp_terms / exp_terms.sum()
            
            grad_a = np.zeros(n)
            for k in range(len(b_val)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_val[k - idx]
            
            S = np.dot(grad_a, a_val)
            grad_z = a_val * (grad_a - S)
            return 2.0 * n * grad_z
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-14, 'maxfun': iters_per_stage * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    return softmax_from_z(z)

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
