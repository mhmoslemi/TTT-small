def fast_convolve(a):

    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence using local computation (sum=1 normalization)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver = None

def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return _lp_solver
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp(a_fixed, solver):
    """Solve LP: given fixed a, find b>=0, sum(b)=1 minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.asarray(b_var.value, dtype=np.float64)
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s > 1e-15:
                return b_val / s, float(M_var.value)
    except Exception:
        pass
    for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        if alt == solver:
            continue
        try:
            prob.solve(solver=alt)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s > 1e-15:
                    return b_val / s, float(M_var.value)
        except Exception:
            continue
    return None, None

def subgradient_nesterov(a0, time_limit, eta0=0.04, active_frac=0.998, max_iters=500000):
    """Subgradient method with Nesterov acceleration for the max-convolution problem."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = eta0
    no_improve = 0
    prev_a = a.copy()
    prev_g = None
    rng = np.random.default_rng(999)
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(y)
        max_b = b.max()
        
        # Adaptive active set
        threshold = max_b * active_frac
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        # Compute subgradient
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= max(1, len(active))
        
        # BB step size
        if it >= 3 and prev_g is not None:
            s_vec = a - prev_a
            y_vec = g - prev_g
            s_ds = np.dot(s_vec, s_vec)
            s_dy = np.dot(s_vec, y_vec)
            y_dy = np.dot(y_vec, y_vec)
            if s_ds > 1e-15 and y_dy > 1e-15:
                if abs(s_dy) > 1e-15:
                    eta = 0.5 * eta + 0.5 * s_ds / abs(s_dy)
                else:
                    eta = s_ds / y_dy
                eta = max(1e-6, min(eta, 0.3))
            else:
                eta *= 0.998
        
        prev_a = a.copy()
        prev_g = g.copy()
        
        # Exponential update (mirror descent)
        log_a = np.log(np.maximum(a, 1e-300))
        g_shifted = g - g.min()
        log_a_new = log_a - eta * g_shifted
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()
        
        # Nesterov
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * a_new
        y = np.maximum(y, 0)
        ys = y.sum()
        if ys < 1e-15:
            y = a_new.copy()
        else:
            y = y / ys
        
        a = a_new
        
        score = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        
        if min(score, score_y) < best_score - 1e-14:
            best_score = min(score, score_y)
            best_a = a.copy() if score < score_y else y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 500:
            noise = rng.standard_normal(n) * 0.01
            a = a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            y = a.copy()
            no_improve = 0
            eta = max(eta, eta0 * 0.5)
        
        if (it + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Subgrad iter={it+1} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def logsumexp_smooth(a0, time_limit, T_stages=None):
    """
    LogSumExp smoothed optimization with L-BFGS.
    Minimizes a smooth approximation of the max-convolution.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    z = np.log(np.maximum(a, 1e-300)).copy()
    t0 = time.monotonic()
    best_z = z.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    if T_stages is None:
        T_stages = [20.0, 10.0, 5.0, 2.5, 1.5, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]
    
    n_stages = len(T_stages)
    base_iters = max(50, 8000 // n_stages)
    
    for stage_i, T in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit - 5:
            break
        
        def func(z_val):
            a_val = _softmax(z_val)
            b_val = fast_convolve(a_val)
            m = b_val.max()
            return (2.0 * n / T) * (m + np.log(np.sum(np.exp(b_val - m))))
        
        def jac(z_val):
            a_val = _softmax(z_val)
            b_val = fast_convolve(a_val)
            m = b_val.max()
            exp_b = np.exp(b_val - m)
            w = exp_b / exp_b.sum()
            
            # grad of logsumexp(conv(a,a), T) w.r.t. a
            grad_a = np.zeros(n)
            size = 2 * n - 1
            for k in range(size):
                if w[k] < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_val[k - idx]
            
            # Chain rule through softmax
            S = np.dot(grad_a, a_val)
            grad_z = a_val * (grad_a - S)
            return grad_z * (2.0 * n / T)
        
        max_iter = base_iters
        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': max_iter * 100,
                                   'disp': False})
            z = res.x
        except Exception:
            pass
        
        a_check = _softmax(z)
        sc = 2.0 * n * fast_convolve(a_check).max()
        if sc < best_score:
            best_score = sc
            best_z = z.copy()
        
        if (stage_i + 1) % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LogSumExp stage={stage_i+1}/{n_stages} T={T:.3f} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return _softmax(best_z)


def _softmax(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def coordinate_polish(a0, time_limit, max_steps=100000):
    """
    Coordinate descent that targets peak-contributing elements.
    Uses simultaneous updates for efficiency.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()

    if s < 1e-15:

        a = a / s
    
    best_a = a.copy()

    t0 = time.monotonic()
    stall = 0
    rng = np.random.default_rng(555)
    
    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Find top peaks
        threshold = max_b * 0.9999
        peaks = np.where(b >= threshold)[0]
        if len(peaks) == 0:
            peaks = np.array([int(np.argmax(b))])
        
        improved = False
        for peak_k in peaks[:5]:
            if improved:
                break
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            
            # Try reducing top contributors
            top_m = min(8, len(idx))
            top_pos = idx[np.argsort(contribs)[::-1][:top_m]]
            
            for pos in top_pos:
                trial_val = a[pos] * 0.995
                diff = a[pos] - trial_val
                if diff < 1e-15:
                    continue
                
                # Redistribute to others proportionally
                a_trial = a.copy()
                a_trial[pos] = trial_val
                mask = np.ones(n, dtype=bool)
                mask[pos] = False
                a_trial[mask] = a[mask] + diff * a[mask] / (a.sum() - a[pos])
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                
                sc = 2.0 * n * fast_convolve(a_trial).max()
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

                # Try small random perturbation

                peak_k = int(np.argmax(b))
                js = max(0, peak_k - n + 1)

                contribs = a[idx] * a[peak_k - idx]
                top_pos = idx[np.argsort(contribs)[::-1][:6]]

                sc = 2.0 * n * fast_convolve(a_trial).max()

        else:
            stall = 0
        
        if (step + 1) % 20000 == 0:

def interpolate_seq(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
