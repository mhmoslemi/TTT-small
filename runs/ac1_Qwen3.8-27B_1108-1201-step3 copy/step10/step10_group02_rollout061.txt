    n = len(a)
    size = 2 * n - 1

    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_convolve(a):

    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def proj_simplex(v, s=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = s}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - s)
    if not cond.any():
        return np.full(n, s / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-strategy optimization:
1. Diverse initialization (ramps, Gaussians, Hann, structured, random-smooth)
2. Proximal gradient on simplex (Euclidean projection, adaptive step)
3. Nesterov-accelerated mirror descent (exponential weighting)
4. LogSumExp smoothed gradient descent with cooling schedule
5. Solution interpolation between diverse candidates
6. Coordinate descent for final peak-directed polish
7. Multi-scale refinement (optimize at coarser n, then upsample and refine)
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence score."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Project onto probability simplex {x>=0, sum(x)=1}."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def proj_simplex(v, s=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = s}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - s)
    if not cond.any():
        return np.full(n, s / n)
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_subgrad_weighted(a, top_idx, weights):
    """Compute weighted subgradient of max_k conv(a,a)[k]."""
    n = len(a)
    g = np.zeros(n)
    for w_k, k in zip(weights, top_idx):
        if w_k < 1e-15:
            continue
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        j = np.arange(i_start, i_end + 1)
        g[j] += 2.0 * w_k * a[k - j]
    return g


def proximal_gradient_optimize(a0, time_limit, K=8, L_init=4.0, verbose=False):
    """Proximal gradient on simplex with adaptive L and restarts."""
    a = normalize(np.asarray(a0, dtype=np.float64).copy())
    n = len(a)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    L = L_init
    it = 0
    no_improve = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        max_b = b.max()
        score = 2.0 * n * max_b

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Top K lags with geometric weighting
        sorted_idx = np.argsort(b)[::-1]
        top_idx = sorted_idx[:K]
        decay = np.exp(-np.arange(len(top_idx)) * 0.4)
        weights = decay / decay.sum()

        g = compute_subgrad_weighted(a, top_idx, weights)
        a_new = proj_simplex(a - g / L)
        score_new = 2.0 * n * fast_convolve(a_new).max()

        if score_new > score + 1e-12:
            L = min(L * 2.0, 100.0)
        else:
            L = max(1.5, L * 0.95)

        a = a_new
        it += 1

        # Stuck: random perturbation from best
        if no_improve > 3000:
            scale = 0.01 * min(1.0, no_improve / 3000.0)
            noise = rng.standard_normal(n) * scale
            a = normalize(best_a + noise)
            no_improve = 0
            L = L_init

        if verbose and it % 100000 == 0:
            print(f"    PG it={it} ({time.monotonic()-t0:.0f}s): best={best_score:.8f} L={L:.1f}")

    return best_a, best_score


def mirror_descent_optimize(a0, time_limit, eta0=0.04, K=12, verbose=False):
    """Nesterov-accelerated mirror descent (exponential weighting)."""
    a = normalize(np.asarray(a0, dtype=np.float64).copy())
    n = len(a)
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = eta0
    it = 0
    no_improve = 0
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(y)
        max_b = b.max()

        sorted_idx = np.argsort(b)[::-1]
        top_idx = sorted_idx[:K]
        decay = np.exp(-np.arange(len(top_idx)) * 0.25)
        weights = decay / decay.sum()

        g = compute_subgrad_weighted(y, top_idx, weights)

        # Mirror descent: a_new_i prop to a_i * exp(-eta * g_i)
        log_a = np.log(np.maximum(a, 1e-300))
        log_a_new = log_a - eta * g
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-300:
            a_new = a.copy()
        else:
            a_new = a_new / a_new_sum

        # Nesterov acceleration
        alpha = 2.0 / (it + 2.0)
        y = (1.0 - alpha) * y + alpha * a_new
        y = normalize(y)

        a = a_new
        it += 1

        score = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        cur_best = min(score, score_y)

        if cur_best < best_score - 1e-13:
            best_score = cur_best
            best_a = a.copy() if score <= score_y else y.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive eta
        if no_improve > 800:
            eta = max(0.002, eta * 0.97)
        if no_improve > 5000:
            scale = 0.015
            noise = rng.standard_normal(n) * scale
            a = normalize(best_a + noise)
            y = a.copy()
            no_improve = 0
            eta = eta0

        if verbose and it % 100000 == 0:
            print(f"    MD it={it} ({time.monotonic()-t0:.0f}s): best={best_score:.8f} eta={eta:.5f}")

    return best_a, best_score


def lse_smoothed_gradient_descent(a0, time_limit, T_schedule=None, verbose=False):
    """Gradient descent on LogSumExp-smoothed objective with simplex projection."""
    if T_schedule is None:
        T_schedule = [20.0, 12.0, 8.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.015, 0.01]

    a = normalize(np.asarray(a0, dtype=np.float64).copy())
    n = len(a)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for T in T_schedule:
        if time.monotonic() - t0 > time_limit - 3:
            break

        lr = 0.15
        for it in range(20000):
            if time.monotonic() - t0 > time_limit - 1:
                break

            b = fast_convolve(a)
            max_b = b.max()
            exp_b = np.exp((b - max_b) / T)
            w = exp_b / exp_b.sum()

            # Gradient
            g = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-10:
                    continue
                i_start = max(0, k - n + 1)
                i_end = min(k, n - 1)
                j = np.arange(i_start, i_end + 1)
                g[j] += 2.0 * w[k] * a[k - j]

            a_new = proj_simplex(a - lr * g)
            score_new = 2.0 * n * fast_convolve(a_new).max()

            if score_new > 2.0 * n * max_b + 1e-12:
                lr *= 0.5
                if lr < 1e-8:
                    break
            else:
                a = a_new
                lr = min(0.5, lr * 1.05)

                if score_new < best_score - 1e-13:
                    best_score = score_new
                    best_a = a.copy()

        if verbose:
            print(f"    LSE T={T:.3f} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, max_steps=100000, verbose=False):
    """Peak-directed coordinate descent with redistribution."""
    a = normalize(np.asarray(a0, dtype=np.float64).copy())
    n = len(a)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(4242)
    stall = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))

        i_start = max(0, peak_k - n + 1)
        i_end = min(peak_k, n - 1)
        j = np.arange(i_start, i_end + 1)
        contribs = a[j] * a[peak_k - j]

        top_m = min(8, len(j))
        top_positions = j[np.argsort(contribs)[::-1][:top_m]]

        improved = False
        for pos in top_positions:
            if a[pos] < 1e-12:
                continue
            for factor in [0.999, 0.997, 0.995, 0.99, 0.985, 0.98]:
                a_trial = a.copy()
                diff = a[pos] * (1.0 - factor)
                a_trial[pos] = a[pos] * factor
                mask = np.ones(n, dtype=bool)
                mask[pos] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    break
                a_trial[mask] += diff * a_trial[mask] / others_sum
                a_trial = normalize(a_trial)
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Periodic exploration: try increasing a non-peak coordinate
            if stall > 50 and stall % 50 == 0:
                pos = rng.integers(n)
                a_trial = a.copy()
                a_trial[pos] *= 1.005
                # Take from peak contributor
                pk_pos = top_positions[0]
                excess = a_trial[pos] - a[pos]
                a_trial[pk_pos] = max(0.0, a[pk_pos] - excess)
                a_trial = normalize(a_trial)
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
                    continue
            if stall > 5000:
                break
        else:
            stall = 0

        if verbose and (step + 1) % 20000 == 0:
            print(f"    CD {time.monotonic()-t0:.0f}s: score={best_score:.8f} stall={stall}")

    return best_a, best_score


def gen_ramp(n, power=1.0, reverse=False):
    """Power-law ramp."""
    x = np.arange(1, n + 1, dtype=np.float64)
    if reverse:
        x = n + 1 - x
    a = x ** power
    return normalize(a)


def gen_gaussian(n, center=0.5, width=0.25):
    """Gaussian bump."""
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - center) / width) ** 2)
    return normalize(a)


def gen_hann(n):
    """Hann window."""
    x = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / max(1, n - 1)))
    return normalize(a)


def gen_structured(n, choice=0):
    """Various structured initializations."""
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = (1.0 - x) ** 1.5
    elif choice == 1:
        a = x ** 2.0
    elif choice == 2:
        a = np.exp(-4.0 * x * (1.0 - x))
    elif choice == 3:
        a = 0.2 + 0.8 * (1.0 - x) ** 0.7
    elif choice == 4:
        a = np.sin(np.pi * x) ** 1.3
    elif choice == 5:
        a = np.exp(-3.0 * x) * (1.0 - x)
    elif choice == 6:
        a = 1.0 / (1.0 + np.exp(-30.0 * (x - 0.4)))
    else:
        a = (1.0 - x) * np.exp(-2.0 * (1.0 - x))
    a = np.maximum(a, 0)
    return normalize(a)


def gen_random_smooth(n, rng, n_smooth=8):
    """Random smooth non-negative sequence."""
    a = rng.standard_normal(n)
    kernel = np.ones(n_smooth) / n_smooth
    for _ in range(3):
        a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0)
    return normalize(a)


def interpolate_to_length(a, new_n):
    """Interpolate a sequence to a new length, preserving normalization."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a_norm = normalize(a)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    return normalize(a_interp)


def search_interpolation(a1, a2, n_samples=20, verbose=False):
    """Search along the line segment between two solutions."""
    best_a = a1.copy()
    best_score = 2.0 * len(a1) * fast_convolve(a1).max()
    n = len(a1)

    for i in range(1, n_samples):
        t = i / n_samples
        a_trial = (1.0 - t) * a1 + t * a2
        a_trial = normalize(a_trial)
        score = 2.0 * n * fast_convolve(a_trial).max()
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_trial.copy()

    if verbose:
        print(f"    Interpolation: best={best_score:.8f}")

    return best_a, best_score


def multi_start_quick_optimize(init_a, time_per_method, verbose=False):
    """Quick optimization using both PG and MD, return best."""
    a_pg, s_pg = proximal_gradient_optimize(init_a, time_per_method * 0.5, verbose=verbose)
    a_md, s_md = mirror_descent_optimize(init_a, time_per_method * 0.5, verbose=verbose)
    if s_pg <= s_md:
        return a_pg, s_pg
    else:
        return a_md, s_md


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Combines proximal gradient, mirror descent, LogSumExp GD, coordinate descent,
    solution interpolation, and multi-scale refinement.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)

        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = normalize(p)
            p_score = evaluate_local(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    # Generate diverse initializations
    n_base = 1200
    if parent is not None:
        n_base = len(parent)
        # Interpolate parent to n_base for consistent comparison
        if n_base != 1200:
            pass  # Keep native size

    initializations = []
    initializations.append(("parent", best_seq.copy() if best_seq is not None else gen_ramp(n_base)))
    initializations.append(("ramp1", gen_ramp(n_base, power=1.0)))
    initializations.append(("ramp2", gen_ramp(n_base, power=1.5)))
    initializations.append(("ramp_rev", gen_ramp(n_base, power=1.0, reverse=True)))
    initializations.append(("gauss", gen_gaussian(n_base, center=0.5, width=0.3)))
    initializations.append(("hann", gen_hann(n_base)))
    initializations.append(("struct0", gen_structured(n_base, 0)))
    initializations.append(("struct3", gen_structured(n_base, 3)))
    initializations.append(("struct5", gen_structured(n_base, 5)))
    initializations.append(("rand1", gen_random_smooth(n_base, rng, 8)))
    initializations.append(("rand2", gen_random_smooth(n_base, np.random.default_rng(seed + 1), 12)))

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Quick optimization from diverse starts ===
    print(f"\n=== Phase 1: Multi-start quick optimization (budget ~300s) ===")
    rem = deadline - time.monotonic()
    n_inits = len(initializations)
    t_per = max(15.0, rem * 0.25 / n_inits)
    print(f"  {n_inits} initializations, ~{t_per:.0f}s each")

    candidates = []
    for name, init_a in initializations:
        if time.monotonic() > deadline - 60:
            break
        t_alloc = min(t_per, deadline - time.monotonic() - 10)
        if t_alloc < 10:
            break
        try:
            a_opt, sc = multi_start_quick_optimize(init_a, t_alloc)
            candidates.append((sc, a_opt))
            try_update(a_opt, name)
        except Exception as e:
            print(f"  Error with {name}: {e}")
            candidates.append((float('inf'), init_a))

    # === Phase 2: Interpolation search ===
    print(f"\n=== Phase 2: Interpolation search ===")
    candidates.sort(key=lambda x: x[0])
    top_k = min(5, len(candidates))

    for i in range(top_k):
        for j in range(i + 1, top_k):
            if time.monotonic() > deadline - 120:
                break
            # Only interpolate same-length sequences
            if len(candidates[i][1]) != len(candidates[j][1]):
                continue
            a_interp, sc = search_interpolation(candidates[i][1], candidates[j][1], n_samples=30)
            if sc < best_score - 1e-13:
                try_update(a_interp, f"interp_{i}_{j}")
            candidates.append((sc, a_interp))

    # === Phase 3: Intensive optimization on best ===
    print(f"\n=== Phase 3: Intensive optimization (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc3 = max(60.0, rem * 0.40)
    print(f"  Budget: {t_alloc3:.0f}s")

    # Sub-phase 3a: Proximal gradient (longer run)
    t3a = t_alloc3 * 0.45
    a_pg, sc_pg = proximal_gradient_optimize(best_seq, t3a, K=10, L_init=3.5)
    try_update(a_pg, "PG_intensive")

    # Sub-phase 3b: Mirror descent (longer run)
    rem = deadline - time.monotonic()
    t3b = max(30.0, t_alloc3 * 0.35)
    a_md, sc_md = mirror_descent_optimize(best_seq, t3b, eta0=0.03, K=15)
    try_update(a_md, "MD_intensive")

    # Sub-phase 3c: LogSumExp gradient descent
