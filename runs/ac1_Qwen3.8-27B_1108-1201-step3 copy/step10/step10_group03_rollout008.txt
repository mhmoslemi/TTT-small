"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast scipy-based LP alternating
minimization with Nesterov acceleration, multi-n exploration, and subgradient
fine-tuning. Key improvements over previous: (1) direct scipy linprog for
much lower overhead than cvxpy, (2) systematic multi-n exploration with
adaptive time allocation, (3) Nesterov-averaged alternating minimization,
(4) subgradient polishing with adaptive step size.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def local_eval(a):
    """Fast local evaluation (with sum=1 normalization)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a_fixed, T=None):
    """Given fixed a (sum=1), find b >= 0, sum(b)=1 minimizing max_k conv(a,b)[k].
    Returns (b, M) or (None, None) on failure.
    """
    n = len(a_fixed)
    size = 2 * n - 1
    if T is None:
        T = build_conv_matrix(a_fixed)
    
    # LP: minimize M
    # subject to: T @ b - M <= 0  (size constraints)
    #             sum(b) = 1
    #             b >= 0
    # Variable: x = [b_0, ..., b_{n-1}, M]
    
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    
    # A_ub @ x <= b_ub
    # T @ b - M <= 0  =>  [T | -1] @ [b; M] <= 0
    A_ub = np.hstack([T, -np.ones((size, 1))])
    b_ub = np.zeros(size)
    
    # A_eq @ x == b_eq
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                      bounds=bounds, method='highs', options={'presolve': True})
        if res.status == 0:
            b_val = np.maximum(res.x[:n], 0)
            s = b_val.sum()
            if s > 1e-15:
                b_val = b_val / s
            M_val = res.x[n]
            return b_val, M_val
    except Exception:
        pass
    return None, None


def nesterov_lp_alternation(a0, time_limit, max_iters=50000):
    """Nesterov-accelerated LP alternating minimization."""
    n = len(a0)
    a = normalize(a0)
    b = a.copy()
    y_a = a.copy()
    y_b = b.copy()
    
    T_a = build_conv_matrix(a)  # Will rebuild when y_a changes significantly
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(12345)
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Step 1: Best response b given y_a
        T_ya = build_conv_matrix(y_a)
        b_new, M1 = lp_best_response(y_a, T=T_ya)
        if b_new is None:
            b_new = b
        
        # Nesterov update for b
        alpha_b = 2.0 / (it + 2)
        y_b = (1.0 - alpha_b) * y_b + alpha_b * b_new
        y_b = normalize(y_b)
        momentum_b = (it + 1) / (it + 1 + alpha_b * (it + 2))
        b = momentum_b * b + (1 - momentum_b) * b_new
        b = normalize(b)
        
        # Step 2: Best response a given y_b
        T_yb = build_conv_matrix(y_b)
        a_new, M2 = lp_best_response(y_b, T=T_yb)
        if a_new is None:
            a_new = a
        
        # Nesterov update for a
        alpha_a = 2.0 / (it + 3)
        y_a = (1.0 - alpha_a) * y_a + alpha_a * a_new
        y_a = normalize(y_a)
        momentum_a = (it + 1) / (it + 1 + alpha_a * (it + 3))
        a = momentum_a * a + (1 - momentum_a) * a_new
        a = normalize(a)
        
        # Track best
        sc_a = 2.0 * n * fast_convolve(a).max()
        sc_ya = 2.0 * n * fast_convolve(y_a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        if sc_ya < best_score:
            best_score = sc_ya
            best_a = y_a.copy()
        
        if (it + 1) % 500 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            
            if stall_count >= 5:
                noise = 0.03 / np.sqrt(stall_count)
                y_a = normalize(y_a + rng.standard_normal(n) * noise)
                y_b = normalize(y_b + rng.standard_normal(n) * noise)
                stall_count = 0
    
    return best_a, best_score


def subgradient_refine(a0, time_limit, n_steps=50000):
    """Subgradient method with Polyak step size and peak averaging."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Optimal lower bound for step size (use current best / some factor)
    lb = best_score * 0.9  # rough lower bound
    lr = 0.01
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        M = b.max()
        current_obj = 2.0 * n * M
        
        # Find peaks within epsilon of max
        threshold = M * 0.999
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        # Subgradient of max_k (a*a)[k] w.r.t. a
        # grad[k] = 2 * (a * a)[k] mapped back to a-space
        # (a*a)[k] = sum_i a[i] * a[k-i], so d/d a[j] = sum_k w[k] * (a[k-j] + a[j-k])... 
        # Actually for auto-convolution: d(conv(a,a))[k]/d(a[j]) = a[k-j] + a[j-k] (with valid indices)
        # But since we're using the max, we average over active peaks
        g = np.zeros(n)
        for k in active:
            # d dk of conv / da[j] = a[k-j] if 0<=k-j<n, plus a[j-k] if 0<=j-k<n
            # But conv(a,a)[k] = sum_{i: 0<=i<n, 0<=k-i<n} a[i]*a[k-i]
            # d/d a[j] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
            # Note: if k=j, both terms give a[0], so it's 2*a[0]... 
            # Wait no: d/d a[j] of sum_i a[i]*a[k-i] = a[k-j] (from first factor) + a[j-k] (from second factor, when i = k-j gives a[k-j]*a[j-k], derivative w.r.t a[j] is a[k-j] from first term and a[j-k] from... 
            # Let me be more careful.
            # conv(a,a)[k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[i] * a[k-i]
            # d/d a[j]:
            #   From first factor: a[k-j] if j is in [max(0,k-n+1), min(n-1,k)]
            #   From second factor: a[k-j] if (k-j) is in [max(0,k-n+1), min(n-1,k)], i.e., j is in [k-(min(n-1,k)), k-max(0,k-n+1)]
            #   = j in [max(0, k-n+1), min(n-1, k)]
            # So both conditions give the same range! Thus d/d a[j] = 2*a[k-j] for j in the valid range.
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)
        
        # Polyak step size
        gap = current_obj - lb
        if gap > 1e-15:
            step_size = lr * gap / (g @ g + 1e-30)
            step_size = min(step_size, lr)
        else:
            step_size = 0
        
        a_trial = a - step_size * g
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st
        
        sc = 2.0 * n * fast_convolve(a_trial).max()
        if sc < best_score - 1e-15:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
            lb = max(lb, sc * 0.99)
        else:
            no_improve += 1
            if no_improve >= 50:
                lr *= 0.9
                no_improve = 0
                # Small random perturbation
                a = normalize(a + rng.standard_normal(n) * 0.01)
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast scipy-based LP alternating
minimization with Nesterov acceleration, systematic multi-n exploration, and
subgradient fine-tuning. Key improvements: (1) direct scipy linprog with HiGHS
solver for much lower overhead than cvxpy, enabling 10x more LP iterations,
(2) Nesterov-averaged alternating minimization for faster convergence,
(3) systematic multi-n exploration with adaptive time allocation,
(4) subgradient polishing with adaptive step size and peak averaging.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def local_eval(a):
    """Fast local evaluation of the objective."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a_fixed, T=None):
    """Given fixed a (sum=1), find b >= 0, sum(b)=1 minimizing max_k conv(a,b)[k].
    Returns (b, M) or (None, None) on failure.
    """
    n = len(a_fixed)
    size = 2 * n - 1
    if T is None:
        T = build_conv_matrix(a_fixed)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_ub = np.hstack([T, -np.ones((size, 1))])
    b_ub = np.zeros(size)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(None, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5.0})
        if res.status == 0:
            b_val = np.maximum(res.x[:n], 0)
            s = b_val.sum()
            if s > 1e-15:
                b_val = b_val / s
            M_val = float(res.x[n])
            return b_val, M_val
    except Exception:
        pass
    return None, None


def nesterov_lp_alternation(a0, time_limit, max_iters=100000):
    """Nesterov-accelerated LP alternating minimization with stall detection."""
    n = len(a0)
    a = normalize(a0)
    b = a.copy()
    y_a = a.copy()
    y_b = b.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(7777)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Step 1: Best response b given y_a
        T_ya = build_conv_matrix(y_a)
        b_new, M1 = lp_best_response(y_a, T=T_ya)
        if b_new is None:
            b_new = y_b.copy()

        # Nesterov update for y_b
        alpha_b = 2.0 / (it + 2)
        y_b = (1.0 - alpha_b) * y_b + alpha_b * b_new
        y_b = normalize(y_b)
        # Momentum combo for b
        w_b = (it + 1) / (it + 1 + alpha_b * (it + 2))
        b = w_b * b + (1.0 - w_b) * b_new
        b = normalize(b)

        # Step 2: Best response a given y_b
        T_yb = build_conv_matrix(y_b)
        a_new, M2 = lp_best_response(y_b, T=T_yb)
        if a_new is None:
            a_new = y_a.copy()

        # Nesterov update for y_a
        alpha_a = 2.0 / (it + 3)
        y_a = (1.0 - alpha_a) * y_a + alpha_a * a_new
        y_a = normalize(y_a)
        w_a = (it + 1) / (it + 1 + alpha_a * (it + 3))
        a = w_a * a + (1.0 - w_a) * a_new
        a = normalize(a)

        # Track best (check both a and y_a, and b and y_b)
        sc_a = 2.0 * n * fast_convolve(a).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        if (it + 1) % 300 == 0:
            sc_ya = 2.0 * n * fast_convolve(y_a).max()
            if sc_ya < best_score:
                best_score = sc_ya
                best_a = y_a.copy()

            sc_b = 2.0 * n * fast_convolve(b).max()
            if sc_b < best_score:
                best_score = sc_b
                best_a = b.copy()

            sc_yb = 2.0 * n * fast_convolve(y_b).max()
            if sc_yb < best_score:
                best_score = sc_yb
                best_a = y_b.copy()

        if (it + 1) % 500 == 0:

                stall_count += 1

                y_b = normalize(y_b + rng.standard_normal(n) * noise_scale)
                stall_count = 0

            if (it + 1) % 5000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LP iter={it+1} ({elapsed:.0f}s): score={best_score:.10f}")

    return best_a, best_score


def subgradient_refine(a0, time_limit, n_steps=100000):
    """Subgradient method with adaptive step size and peak averaging."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    lb = best_score * 0.95
    lr = 0.02
    no_improve = 0
    rng = np.random.default_rng(42)

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        M = b.max()
        current_obj = 2.0 * n * M

        # Find peaks within threshold of max
        threshold = M * 0.998
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Subgradient
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)

        # Adaptive step size
        gap = current_obj - lb
        if gap > 1e-12:
            step_size = min(lr, lr * gap / (g @ g + 1e-30))
        else:
            step_size = lr * 0.1

        a_trial = a - step_size * g
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st

        sc = 2.0 * n * fast_convolve(a_trial).max()
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
            lb = max(lb, sc * 0.99)
        else:
            no_improve += 1
            if no_improve >= 100:
                lr *= 0.85
                no_improve = 0
                a = normalize(a + rng.standard_normal(n) * 0.02)

        if (step + 1) % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG step={step+1} ({elapsed:.0f}s): score={best_score:.10f}")

    return best_a, best_score


def damped_lp_alternation(a0, time_limit, max_iters=50000, damping=0.3):
    """LP alternating minimization with damping (simpler, sometimes more stable)."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Best response for a given current b (=a for auto-conv)
        T = build_conv_matrix(a)
        a_new, _ = lp_best_response(a, T=T)
        if a_new is not None:
            a_trial = (1.0 - damping) * a + damping * a_new
            a_trial = normalize(a_trial)
            sc = 2.0 * n * fast_convolve(a_trial).max()
            if sc < best_score:
                best_score = sc
                best_a = a_trial.copy()
                a = a_trial

        # Also try the other direction
        T2 = build_conv_matrix(a)
        b_new, _ = lp_best_response(a, T=T2)
        if b_new is not None:
            sc_b = 2.0 * n * fast_convolve(b_new).max()
            if sc_b < best_score:
                best_score = sc_b
                best_a = b_new.copy()
                a = b_new

        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    DampedLP iter={it+1} ({elapsed:.0f}s): score={best_score:.10f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Create diverse structured initial sequences."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 1.0 + 0.1 * np.cos(2 * np.pi * x) + 0.05 * np.cos(4 * np.pi * x)
    elif choice == 7:
        a = np.ones(n) + 0.2 * np.sin(2 * np.pi * 5 * x)
        a = np.maximum(a, 0.01)
    elif choice == 8:
        r = 0.5
        a = np.ones(n)
        m = max(1, int(r * n / 2))
        a[:m] = 0.5 * (1 + np.cos(np.pi * (np.arange(m) - m + 1) / m))
        a[-m:] = a[:m][::-1]
    else:
        a = np.ones(n) * (1.0 + 0.1 * rng.standard_normal())
        a = np.maximum(a, 0.01)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses fast scipy HiGHS LP alternating minimization with Nesterov acceleration,
    systematic multi-n exploration, and subgradient polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = local_eval(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = local_eval(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.10f}")

    n_best = len(best_seq)

    def try_update(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = local_eval(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    # === Phase 1: Fast LP Alternation on parent n ===
    print(f"\n=== Phase 1: Fast LP Alternation (n={n_best}, score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.30, 450.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        a_lp, sc_lp = nesterov_lp_alternation(best_seq, t1)
        try_update(a_lp, "lp_main")

    # === Phase 2: Subgradient refinement ===
    print(f"\n=== Phase 2: Subgradient Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.08, 80.0)
    if t2 > 15:
        a_sg, sc_sg = subgradient_refine(best_seq, t2)
        try_update(a_sg, "subgrad")

    # === Phase 3: Multi-n Exploration ===
    print(f"\n=== Phase 3: Multi-n Exploration (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()

    # Generate candidate n values
    n_candidates = set()
    # Around current best
    for mult in [0.75, 0.85, 0.9, 1.1, 1.2, 1.33, 1.5, 2.0]:
        cand_n = int(n_best * mult)
        if 200 <= cand_n <= 6000:
            n_candidates.add(cand_n)
    # Absolute values
    for abs_n in [400, 500, 600, 800, 1000, 1500, 1800, 2000, 2500, 3000, 4000]:
        if 200 <= abs_n <= 6000:
            n_candidates.add(abs_n)
    n_candidates.discard(n_best)

    # Sort by proximity to n_best
    n_list = sorted(n_candidates, key=lambda x: abs(x - n_best))

    for cand_n in n_list:
        rem = deadline - time.monotonic()
        if rem < 50:
            break

        # Time budget proportional to proximity
        proximity = 1.0 / (1.0 + abs(cand_n - n_best) / 800.0)
        t_sub = min(rem * 0.12 * proximity + 15, 90.0)
        if t_sub < 30:
            continue

        print(f"  Trying n={cand_n}, budget={t_sub:.0f}s")
        init_cand = interpolate_sequence(best_seq, cand_n)

        # LP alternation
        t_lp = t_sub * 0.75
        a_cand, _ = nesterov_lp_alternation(init_cand, t_lp)
        try_update(a_cand, f"lp_n{cand_n}")

        # Quick subgradient
        rem = deadline - time.monotonic()
        if rem > 15:
            t_sg = min(rem * 0.2, 25.0)
            a_cand2, _ = subgradient_refine(a_cand, t_sg)
            try_update(a_cand2, f"sg_n{cand_n}")

        # If this n gave a new best, try a few nearby n values
        if len(best_seq) == cand_n:
            for nearby in [cand_n - 50, cand_n + 50, cand_n - 100, cand_n + 100]:
                rem = deadline - time.monotonic()
                if rem < 30:
                    break
                if not (200 <= nearby <= 6000):
                    continue
                t_nb = min(rem * 0.08, 40.0)
                if t_nb < 20:
                    continue
                init_nb = interpolate_sequence(best_seq, nearby)
                a_nb, _ = nesterov_lp_alternation(init_nb, t_nb)
                try_update(a_nb, f"lp_n{nearby}_near")

    # === Phase 4: Final intensive LP on best ===
    print(f"\n=== Phase 4: Final LP Polish (score={best_score:.10f}, n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t_final = min(rem * 0.35, 180.0)
        print(f"  Budget: {t_final:.0f}s")
        a_fin, _ = nesterov_lp_alternation(best_seq, t_final)
        try_update(a_fin, "final_lp")

    # === Phase 5: Final subgradient ===
    print(f"\n=== Phase 5: Final Subgradient (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fs = min(rem * 0.5, 60.0)
        a_fs, _ = subgradient_refine(best_seq, t_fs)
        try_update(a_fs, "final_sg")

    # Final verification
    if best_seq is not None:
        fs = local_eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
        try:
            eval_score = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
