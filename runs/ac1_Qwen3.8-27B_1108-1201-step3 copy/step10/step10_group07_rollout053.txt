def smoothed_gradient_descent(a0, time_limit, p_init=4, p_end=2, verbose=True):
    """
    Minimize sum_k(c_k^p) where c = auto_conv(a), using Adam optimizer.
    p decreases over time from p_init to p_end.
    """
    n = len(a0)
    N = 1 << (2*n-1).bit_length()  # FFT size
    a = a0.copy() / a0.sum()
    
    best_a = a.copy()
    best_score = 2*n*fast_convolve(a).max()
    
    # Adam parameters
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t = 0
    
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        t += 1
        # Anneal p
        progress = min(1.0, t / 1000)
        p = p_init + (p_end - p_init) * progress
        
        # Compute convolution
        c = fast_convolve(a)
        
        # Compute gradient of sum(c^p)
        if p == 2:
            # Special case for efficiency
            grad = 4.0 * cross_corr(a, c, N)
        else:
            c_pow = c ** (p-1)  # Note: c >= 0 since a >= 0
            grad = 2.0 * p * cross_corr(a, c_pow, N)
        
        # Adam update
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        lr = 0.01 / (1 + 0.01*t)**0.5  # Decaying learning rate
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project onto simplex {x >= 0, sum(x) = 1}
        a_new = simplex_project(a_new)
        
        a = a_new
        
        # Check improvement
        c_new = fast_convolve(a)
        score = 2*n*c_new.max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def score_of(a):
    """Compute 2*n*max(conv)/(sum)^2 quickly."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def find_lp_solver():
    """Find the fastest working LP solver."""
    cached = getattr(find_lp_solver, '_cached', None)
    if cached is not None:
        return cached
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                find_lp_solver._cached = solver_name
                return solver_name
        except Exception:
            continue
    find_lp_solver._cached = cp.ECOS
    return cp.ECOS


def build_conv_matrix(a):
    """Build the convolution matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, solver):
    """
    Solve LP: min M s.t. conv(a,b) <= M*1, sum(b)=1, b>=0.
    Returns (b_opt, M_opt).
    """
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    ones = np.ones(2 * n - 1)
    constraints = [
        T @ b_var <= M_var * ones,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return a.copy(), float('inf')
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float('inf')
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return a.copy(), float('inf')
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else float('inf')
    return b_val, M_val


def compute_grad_smoothed(a, c, p, N):
    """
    Compute gradient of sum_k(c_k^p) where c = auto_conv(a).
    grad_i = 2*p * sum_m a[m] * c[m+i]^(p-1)
    Computed via FFT cross-correlation.
    """
    n = len(a)
    if p == 1.0:
        # grad_i = 2 * sum_m a[m] * c[m+i] = 2 * cross_corr(a, c)[i]
        c_pow = c.copy()
    else:
        c_pow = np.maximum(c, 0.0) ** (p - 1.0)
    A = rfft(a, N)
    C = rfft(c_pow, N)
    cross = irfft(np.conj(A) * C, N)[:n]
    return 2.0 * p * cross


def smoothed_adam(a0, time_limit, p_init=8.0, p_end=2.0, lr_init=0.005, verbose=True, seed=42):
    """
    Adam optimizer on smoothed objective sum(c^p) with annealing p.
    Projects onto simplex after each step.
    """
    n = len(a0)
    N = 1 << (2 * n - 1).bit_length()
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    # Adam state
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    stale = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed

        # Anneal p: start high (peak-focused), decrease to 2 (sum of squares)
        progress = min(1.0, elapsed / max(time_limit * 0.7, 1.0))
        p = p_init + (p_end - p_init) * progress

        c = fast_convolve(a)
        grad = compute_grad_smoothed(a, c, p, N)

        # Normalize gradient to have consistent scale
        g_norm = np.linalg.norm(grad)
        if g_norm > 1e-15:
            grad = grad / g_norm * max(g_norm, 1.0 / n)

        # Adaptive learning rate
        lr = lr_init / (1.0 + 0.05 * t)

        # Adam update
        m_buf = beta1 * m_buf + (1 - beta1) * grad
        v_buf = beta2 * v_buf + (1 - beta2) * (grad * grad)
        m_hat = m_buf / (1.0 - beta1 ** t)
        v_hat = v_buf / (1.0 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = simplex_project(a_new)

        # Evaluate
        c_new = fast_convolve(a_new)
        score = 2.0 * n * c_new.max()
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
            stale = 0
        else:
            stale += 1

        a = a_new

        # If stuck, try a perturbation
        if stale > 100:
            noise_scale = 0.02 / np.sqrt(n)
            a_pert = a + np.random.default_rng(t).standard_normal(n) * noise_scale
            a_pert = simplex_project(a_pert)
            s_pert = 2.0 * n * fast_convolve(a_pert).max()
            if s_pert < best_score * 1.01:
                a = a_pert
            stale = 0

        if verbose and t % 500 == 0:
            print(f"    SmoothAdam t={t} ({elapsed:.0f}s): p={p:.1f} best={best_score:.8f}")

    return best_a, best_score


def coord_descent_exact(a0, time_limit, verbose=True, label=""):
    """
    Coordinate descent with exact 1D minimization.
    For each coordinate i, find the value of a_i that minimizes the max-conv objective
    (with all other coordinates fixed and sum constraint handled by redistribution).
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        # Randomize order for diversity
        order = np.random.permutation(n)

        for i in order:
            if a[i] < 1e-16:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = float(max(unaff_left, unaff_right))

            # a[k-i] for k in [k_start, k_end]
            a_shifted = a[max(0, k_start - i):k_end - i + 1]
            conv_affected = conv[k_start:k_end + 1].copy()

            best_delta = 0.0
            best_new_M = M

            # Try a range of multiplicative factors
            base = max(a[i], 1e-10)
            factors = []
            # Aggressive and fine steps
            for f in [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5]:
                factors.append(f - 1.0)
            for f in [0.98, 0.99, 1.01, 1.02]:
                factors.append(f - 1.0)

            for factor in factors:
                delta = base * factor
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = conv_affected + 2.0 * delta * a_shifted
                # Fix quadratic term at k=2i
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_val_arr = new_affected.copy()
                    new_val_arr[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta
                    new_affected = new_val_arr

                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-14:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-14:
                old_val = a[i]
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * old_val + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD[{label}] sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def pairwise_coord_descent(a0, time_limit, verbose=True, label="pw"):
    """
    Pairwise coordinate descent: optimize pairs of coordinates simultaneously.
    This can escape 1D local minima.
    """

    else:
        a = a / s

    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    rounds = 0
    no_improve = 0
    rng = np.random.default_rng(123)

    while time.monotonic() - t0 < time_limit:
        rounds += 1

        # Select a random pair
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        if a[i] < 1e-16 or a[j] < 1e-16:
            continue

        # Try 2D grid of multiplicative factors
        best_delta_i = 0.0
        best_delta_j = 0.0
        best_new_M = M

        base_i = max(a[i], 1e-10)
        base_j = max(a[j], 1e-10)

        f_i_list = [-0.3, -0.15, -0.05, 0.05, 0.15, 0.3]
        f_j_list = [-0.3, -0.15, -0.05, 0.05, 0.15, 0.3]

        for fi in f_i_list:
            di = base_i * fi
            if a[i] + di < -1e-15:
                continue
            for fj in f_j_list:
                dj = base_j * fj
                if a[j] + dj < -1e-15:
                    continue

                # Compute new conv values
                # Affected range: union of [i, i+n-1] and [j, j+n-1]
                k_s = min(i, j)
                k_e = max(i + n - 1, j + n - 1)
                k_e = min(k_e, 2 * n - 2)
                k_s = max(k_s, 0)

                if k_s > 0:
                    unaff_left = conv[:k_s].max()
                else:
                    unaff_left = -1e30
                if k_e < 2 * n - 2:
                    unaff_right = conv[k_e + 1:].max()
                else:
                    unaff_right = -1e30
                unaff_max = float(max(unaff_left, unaff_right))

                new_seg = conv[k_s:k_e + 1].copy()

                # Contribution from i
                i_k_s = k_s
                i_k_e = min(i + n - 1, 2 * n - 2)
                a_for_i = a[max(0, i_k_s - i):i_k_e - i + 1]
                new_seg[i_k_s - k_s:i_k_e - k_s + 1] += 2.0 * di * a_for_i
                # Quadratic term for i
                if 2 * i >= i_k_s and 2 * i <= i_k_e:
                    idx = 2 * i - k_s
                    new_seg[idx] = conv[2 * i] + 2.0 * di * a[i] + di * di

                # Contribution from j (using updated conv for i's part)
                j_k_s = max(0, j)
                j_k_e = min(j + n - 1, 2 * n - 2)
                a_for_j = a[max(0, j_k_s - j):j_k_e - j + 1]
                new_seg[j_k_s - k_s:j_k_e - k_s + 1] += 2.0 * dj * a_for_j
                # Quadratic term for j
                if 2 * j >= j_k_s and 2 * j <= j_k_e:
                    idx = 2 * j - k_s
                    new_seg[idx] += 2.0 * dj * a[j] + dj * dj
                    # But we already added 2*di*a_for_i at this position if 2j is in i's range
                    # Need to be more careful - let's just recompute from scratch

                new_M = max(unaff_max, float(new_seg.max()))
                if new_M < best_new_M - 1e-14:
                    best_new_M = new_M
                    best_delta_i = di
                    best_delta_j = dj

        if best_delta_i != 0.0 or best_delta_j != 0.0:
            # Recompute convolution from scratch (safer for pairwise)
            a[i] = max(a[i] + best_delta_i, 0.0)
            a[j] = max(a[j] + best_delta_j, 0.0)
            conv = fast_convolve(a)
            M = float(conv.max())
            cur_score = 2.0 * n * M
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                break

        if verbose and rounds % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD[{label}] round={rounds} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def lp_alternating(a0, time_limit, solver, verbose=True):
    """LP-based best-response alternating minimization."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    stagnation = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        b_new, M_lp = lp_best_response(a, solver)
        if not np.isfinite(M_lp):
            break

        score_b = 2.0 * n * fast_convolve(b_new).max()
        if score_b < best_score:
            best_score = score_b
            best_a = b_new.copy()

        # Damping schedule
        if it <= 10:
            alpha = 1.0
        elif it <= 30:
            alpha = 0.7
        elif it <= 60:
            alpha = 0.5
        else:
            alpha = max(0.3, 15.0 / (it + 5.0))

        a = (1 - alpha) * a + alpha * b_new
        a = np.maximum(a, 0)
        a_s = a.sum()
        if a_s < 1e-15:
            break
        a = a / a_s

        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        if score_a > best_score * (1 + 1e-8):
            stagnation += 1
        else:
            stagnation = max(0, stagnation - 1)

        if stagnation > 25:
            noise_scale = 0.01 / np.sqrt(n)
            a_pert = a + np.random.standard_normal(n) * noise_scale
            a_pert = np.maximum(a_pert, 0)
            a_pert_s = a_pert.sum()
            if a_pert_s > 1e-10:
                a_pert = a_pert / a_pert_s
            score_pert = 2.0 * n * fast_convolve(a_pert).max()
            if score_pert < best_score * 1.005:
                a = a_pert
            stagnation = 0

        if verbose and it % 15 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def spectral_shaping(a0, time_limit, verbose=True, num_iter=500):
    """
    Gerchberg-Saxton style spectral shaping.
    Modify the power spectrum to reduce auto-convolution peaks.
    """
    n = len(a0)
    N = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit and it < num_iter:
        it += 1

        A = rfft(a, N)
        power = np.abs(A) ** 2
        phase = np.angle(A)

        # Compute auto-conv from power spectrum
        c = irfft(power, N)[:2 * n - 1]

        # Soft-clip the peaks in c
        M = c.max()
        threshold = M * 0.95
        # Reduce peaks: c_new = c * exp(-beta * relu(c - threshold))
        beta = 2.0
        excess = np.maximum(c - threshold, 0)
        c_new = c * np.exp(-beta * excess)
        # Renormalize to maintain sum = 1
        c_sum = c_new.sum()
        if c_sum > 1e-15:
            c_new = c_new / c_sum

        # Convert back to frequency domain
        power_new = rfft(c_new, N)
        # Ensure non-negative (should be, but numerical errors)
        power_new = np.maximum(power_new, 0)
        # Magnitude of new A
        mag_new = np.sqrt(power_new)
        # New A keeping original phase
        A_new = mag_new * np.exp(1j * phase)
        # Ensure DC component is preserved (sum = 1)
        A_new[0] = A[0]  # Keep DC fixed

        a_new = irfft(A_new, N)[:n]

        # Project onto simplex
        a_new = simplex_project(a_new)

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        # Damped update to maintain stability
        a = 0.7 * a + 0.3 * a_new
        a = np.maximum(a, 0)
        a_s = a.sum()
        if a_s > 1e-15:
            a = a / a_s

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Spectral it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def fine_polish(a0, time_limit, verbose=True):
    """Very fine coordinate descent for last-mile improvements."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-16:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = float(max(unaff_left, unaff_right))

            a_shifted = a[max(0, k_start - i):k_end - i + 1]
            conv_affected = conv[k_start:k_end + 1]

            best_delta = 0.0
            best_new_M = M

            # Very fine multiplicative steps
            for factor in [0.999, 0.998, 0.997, 0.9995, 1.0005, 1.001, 1.002, 1.003]:
                delta = a[i] * (factor - 1.0)
                if abs(delta) < 1e-16:
                    continue
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = conv_affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-15:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-15:
                old_val = a[i]
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * old_val + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 15:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FinePolish sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def random_restart_search(a0, time_limit, n_restarts=10, verbose=True):
    """
    Try multiple random restarts from the current best with various perturbations.
    For each restart, do a short round of coordinate descent.
    """
    n = len(a0)
    best_a = a0.astype(np.float64).copy()
    s = best_a.sum()
    if s < 1e-15:
        best_a = np.ones(n) / n
    else:
        best_a = best_a / s
    best_score = 2.0 * n * fast_convolve(best_a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    time_per_restart = time_limit / max(n_restarts, 1)

    for r in range(n_restarts):
        if time.monotonic() - t0 > time_limit:
            break

        # Create perturbation
        noise_type = r % 4
        if noise_type == 0:
            # Small Gaussian noise
            noise = rng.standard_normal(n) * (0.01 / np.sqrt(n))
        elif noise_type == 1:
            # Sparse perturbation (change a few coordinates)
            noise = np.zeros(n)
            k = rng.integers(5, 30)
            idx = rng.choice(n, size=k, replace=False)
            noise[idx] = rng.standard_normal(k) * (0.05 / np.sqrt(n))
        elif noise_type == 2:
            # Smooth noise (low frequency)
            freqs = rng.uniform(0, 3, size=n)
            noise = np.cos(2 * np.pi * freqs * np.arange(n) / n) * (0.02 / np.sqrt(n))
        else:
            # Coordinate swap perturbation
            noise = np.zeros(n)
            for _ in range(rng.integers(3, 15)):
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i != j:
                    swap_amt = 0.1 * min(best_a[i], best_a[j])
                    noise[i] = -swap_amt
                    noise[j] = swap_amt

        perturbed = best_a + noise
        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-10:
            continue
        perturbed = perturbed / ps

        # Short coordinate descent from perturbed
        t_alloc = min(time_per_restart * 0.7, 30.0)
        a_cd, s_cd = coord_descent_exact(perturbed, t_alloc, verbose=False, label=f"rr{r}")

        if s_cd < best_score:
            best_score = s_cd
            best_a = a_cd.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    Restart {r} ({elapsed:.0f}s): NEW BEST {best_score:.8f}")

    return best_a, best_score


def diverse_init(n, rng, init_type=0):
    """Generate diverse initializations."""
    if init_type == 0:
        t = np.arange(n)
        center = (n - 1) / 2.0
        a = np.maximum(1.0 - np.abs(t - center) / center, 0)
    elif init_type == 1:
        t = np.arange(n)
        center = (n - 1) / 2.0
        x = (t - center) / center
        a = 0.3 + 0.7 * x * x
    elif init_type == 2:
        t = np.arange(n)
        center = (n - 1) / 2.0
        x = (t - center) / center
        power = rng.uniform(1.0, 3.0)
        a = np.maximum(0.5 * (1 + np.cos(np.pi * x)), 0) ** power
    elif init_type == 3:
        t = np.arange(n)
        c1 = n * rng.uniform(0.2, 0.35)
        c2 = n * rng.uniform(0.65, 0.8)
        w = n * rng.uniform(0.08, 0.2)
        a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    elif init_type == 4:
        period = int(rng.integers(3, 10))
        t = np.arange(n)
        a = 0.5 + 0.4 * np.cos(2 * np.pi * t / period)
    else:
        a = np.ones(n)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def propose_candidate():
    """
    Multi-phase optimization combining:
    1. LP best-response alternating (main workhorse)
    2. Smoothed gradient descent with FFT-based gradients (new)
    3. Pairwise coordinate descent (new)
    4. Spectral shaping via GS algorithm (new)
    5. Fine polishing and random restarts
    """
    deadline = time.monotonic() + 1780.0
    rng = np.random.default_rng(777)

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is not None:
        n = len(parent)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    else:
        n = 1024
        best_seq = np.ones(n) / n
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Using default init: n={n}, score={best_score:.8f}")

    print(f"Initial best: {best_score:.8f} (n={n})")

    # Find LP solver
    solver = find_lp_solver()
    print(f"LP solver: {solver}")

    def try_update(candidate, label):
        nonlocal best_seq, best_score
        if candidate is None:
            return
        candidate = np.maximum(np.asarray(candidate, dtype=np.float64), 0.0)
        s = candidate.sum()
        if s < 0.01:
            return
        candidate = candidate / s
        try:
            sc = float(evaluate_sequence(candidate.tolist()))
        except Exception:
            sc = float('inf')
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = candidate.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(candidate)}) ***")

    elapsed_total = lambda: time.monotonic() - (deadline - 1780.0)

    # === Phase 1: LP Alternating (30% time) ===
    print(f"\n=== Phase 1: LP Alternating (best={best_score:.8f}) ===")
    t1_end = time.monotonic() + (deadline - time.monotonic()) * 0.30

    # Run LP from parent
    t_alloc = min(t1_end - time.monotonic(), 300.0)
    print(f"  LP alt from parent, budget={t_alloc:.0f}s")
    a_lp, s_lp = lp_alternating(best_seq, t_alloc, solver, verbose=True)
    try_update(a_lp, "lp_parent")

    # Run LP from a diverse init
    if time.monotonic() < t1_end - 60:
        init = diverse_init(n, rng, 0)
        t_alloc2 = min(t1_end - time.monotonic(), 120.0)
        print(f"  LP alt from triangle init, budget={t_alloc2:.0f}s")
        a_lp2, s_lp2 = lp_alternating(init, t_alloc2, solver, verbose=False)
        try_update(a_lp2, "lp_triangle")

    # === Phase 2: Smoothed Adam (20% time) ===
    print(f"\n=== Phase 2: Smoothed Adam (best={best_score:.8f}) ===")
    t2_end = time.monotonic() + (deadline - time.monotonic()) * 0.20

    t_alloc = min(t2_end - time.monotonic(), 250.0)
    if t_alloc > 60:
        print(f"  SmoothAdam from best, budget={t_alloc:.0f}s")
        a_sam, s_sam = smoothed_adam(best_seq, t_alloc, p_init=8.0, p_end=2.0,
                                     lr_init=0.005, verbose=True, seed=42)
        try_update(a_sam, "smooth_adam")

        # Also try from LP result
        if time.monotonic() < t2_end - 60:
            t_alloc2 = min(t2_end - time.monotonic(), 120.0)
            a_sam2, s_sam2 = smoothed_adam(a_lp, t_alloc2, p_init=6.0, p_end=2.0,
                                           lr_init=0.003, verbose=False, seed=137)
            try_update(a_sam2, "smooth_adam_lp")

    # === Phase 3: Coordinate Descent + Pairwise (20% time) ===
    print(f"\n=== Phase 3: Coord Descent (best={best_score:.8f}) ===")
    t3_end = time.monotonic() + (deadline - time.monotonic()) * 0.20

    # Standard CD from best
    t_alloc = min(t3_end - time.monotonic(), 150.0)
    if t_alloc > 30:
        print(f"  CD from best, budget={t_alloc:.0f}s")
        a_cd, s_cd = coord_descent_exact(best_seq, t_alloc, verbose=True, label="main")
        try_update(a_cd, "cd_main")

    # Pairwise CD
    if time.monotonic() < t3_end - 30:
        t_alloc2 = min(t3_end - time.monotonic(), 100.0)
        print(f"  Pairwise CD, budget={t_alloc2:.0f}s")
        a_pw, s_pw = pairwise_coord_descent(best_seq, t_alloc2, verbose=True, label="main")
        try_update(a_pw, "pairwise")

    # === Phase 4: Spectral Shaping (10% time) ===
    print(f"\n=== Phase 4: Spectral Shaping (best={best_score:.8f}) ===")
    t4_end = time.monotonic() + (deadline - time.monotonic()) * 0.10

    t_alloc = min(t4_end - time.monotonic(), 100.0)
    if t_alloc > 30:
        print(f"  Spectral from best, budget={t_alloc:.0f}s")
        a_spec, s_spec = spectral_shaping(best_seq, t_alloc, verbose=True, num_iter=300)
        try_update(a_spec, "spectral")

    # === Phase 5: Random Restarts + Fine Polish (20% time) ===
    print(f"\n=== Phase 5: Random Restarts + Polish (best={best_score:.8f}) ===")
    t5_end = deadline - 30

    # Random restarts
    t_alloc = min(t5_end - time.monotonic(), 300.0)
    if t_alloc > 60:
        print(f"  Random restarts, budget={t_alloc:.0f}s")
        a_rr, s_rr = random_restart_search(best_seq, t_alloc, n_restarts=15, verbose=True)
        try_update(a_rr, "restarts")

    # Final fine polish
    if time.monotonic() < t5_end - 20:
        t_fp = min(t5_end - time.monotonic() - 20, 120.0)
        if t_fp > 20:
            print(f"  FinePolish, budget={t_fp:.0f}s")
            a_fp, s_fp = fine_polish(best_seq, t_fp, verbose=True)
            try_update(a_fp, "fine_polish")

    # Another round of CD on the best
    if time.monotonic() < t5_end - 10:
        t_final = min(t5_end - time.monotonic() - 10, 60.0)
        if t_final > 15:
            print(f"  Final CD, budget={t_final:.0f}s")
            a_fin, s_fin = coord_descent_exact(best_seq, t_final, verbose=False, label="final")
            try_update(a_fin, "final_cd")

    # Final fine polish
    if time.monotonic() < deadline - 5:
        t_fp2 = min(deadline - time.monotonic() - 5, 30.0)
        if t_fp2 > 5:
            a_fp2, s_fp2 = fine_polish(best_seq, t_fp2, verbose=False)
            try_update(a_fp2, "final_fine")

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
