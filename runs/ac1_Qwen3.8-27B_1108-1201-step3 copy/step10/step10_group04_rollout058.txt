import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fft_convolve(a, b=None):
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def compute_grad_logsumexp(a, T):
    """Compute gradient of (2n/T)*log(sum exp(T*c)) w.r.t. a."""
    n = len(a)
    c = fft_convolve(a, a)
    Tc = T * c
    Tc_shifted = Tc - Tc.max()
    exp_Tc = np.exp(Tc_shifted)
    p = exp_Tc / exp_Tc.sum()
    # grad_a[i] = (4n/T) * sum_k p[k] * a[k-i]
    conv = fft_convolve(p[::-1], a)  # length 3n-2
    grad_a = (4.0 * n / T) * conv[n-1:2*n-1][::-1]
    return grad_a, p

def softmax(z):
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()

def softmax_grad(g_a, a):
    """Given gradient w.r.t. a and current a=softmax(z), compute grad w.r.t. z."""
    # da[i]/dz[j] = a[i]*(delta_ij - a[j])
    # grad_z[j] = sum_i grad_a[i] * a[i] * (delta_ij - a[j])
    #           = a[j] * (grad_a[j] - sum_i a[i]*grad_a[i])
    S = np.dot(a, g_a)
    return a * (g_a - S)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FISTA (accelerated gradient) on log-sum-exp smoothed objective with FFT-based gradients
2. Multi-stage temperature annealing
3. Diverse initializations (spectral, structured, interpolated parent)
4. Peak-shaving local search
5. Multi-length exploration
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fft_convolve(a, b=None):
    """Compute linear convolution using FFT."""
    if b is None:
        a = np.asarray(a, dtype=np.float64)
        b = a
    b = np.asarray(b, dtype=np.float64)
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def softmax(z):
    """Stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def grad_logsumexp_wrt_a(a, T):
    """
    Compute gradient of (2n/T)*log(sum_k exp(T*c[k])) w.r.t. a.
    where c = convolve(a, a).
    Returns (grad_a, objective_value).
    """
    n = len(a)
    c = fft_convolve(a, a)
    Tc = T * c
    Tc_max = Tc.max()
    exp_Tc = np.exp(Tc - Tc_max)
    p = exp_Tc / exp_Tc.sum()
    logsum = Tc_max + np.log(exp_Tc.sum())
    obj = (2.0 * n / T) * logsum

    # grad_a[i] = (4n/T) * sum_k p[k] * a[k-i]
    # This equals (4n/T) * convolve(p_reversed, a)[n-1 : 2n-1][::-1]
    p_rev = p[::-1]
    conv_pa = fft_convolve(p_rev, a)
    # conv_pa has length (2n-1) + n - 1 = 3n-2
    # We need indices [n-1, 2n-2] reversed
    grad_a = (4.0 * n / T) * conv_pa[n - 1:2 * n - 1][::-1]
    return grad_a, obj, p, c


def softmax_jacobian_applied(g_a, a):
    """
    Given grad w.r.t. a and a = softmax(z), compute grad w.r.t. z.
    grad_z[j] = a[j] * (g_a[j] - sum_i a[i]*g_a[i])
    """
    S = np.dot(a, g_a)
    return a * (g_a - S)


def fista_anneal(z_init, T_stages, iters_per_stage, time_budget, label=""):
    """
    FISTA with log-sum-exp smoothing and multi-stage temperature annealing.
    """
    t0 = time.monotonic()
    n = len(z_init)
    z = np.asarray(z_init, dtype=np.float64).copy()
    best_a = softmax(z)
    best_score = 2.0 * n * fft_convolve(best_a, best_a).max()
    original_best = best_score

    for si, T in enumerate(T_stages):
        if time.monotonic() - t0 > time_budget:
            break
        z = z.copy()
        y = z.copy()
        t_mom = 1.0
        step = 2.0 / (T * 2.0 * n)  # Initial step size (1/L where L is Lipschitz const)
        step = min(step, 1.0)

        for it in range(iters_per_stage):
            if time.monotonic() - t0 > time_budget:
                break

            a_y = softmax(y)
            g_a, obj_val, p, c = grad_logsumexp_wrt_a(a_y, T)
            g_z = softmax_jacobian_applied(g_a, a_y)

            g_norm = np.linalg.norm(g_z)
            if g_norm < 1e-15:
                break

            # Backtracking line search
            alpha = step
            z_trial = z - alpha * g_z
            a_trial = softmax(z_trial)

            # Compute trial objective
            c_trial = fft_convolve(a_trial, a_trial)
            Tc_trial = T * c_trial
            Tc_trial_max = Tc_trial.max()
            obj_trial = (2.0 * n / T) * (Tc_trial_max + np.log(np.sum(np.exp(Tc_trial - Tc_trial_max))))

            # Quadratic model from FISTA
            obj_model = obj_val + np.dot(g_z, z_trial - z) + (1.0 / (2.0 * alpha)) * np.dot(z_trial - z, z_trial - z)

            if obj_trial <= obj_model + 1e-12:
                step = min(alpha * 1.25, 10.0)
            else:
                step = alpha * 0.5
                continue

            # FISTA momentum update
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
            beta = (t_mom - 1.0) / t_new
            z_old = z.copy()
            z = z_trial
            y = z + beta * (z - z_old)
            t_mom = t_new

            # Track best (using true max, not log-sum-exp)
            actual_score = 2.0 * n * fft_convolve(a_trial, a_trial).max()
            if actual_score < best_score:
                best_score = actual_score
                best_a = a_trial.copy()

        # End of stage: update z from y
        z = y.copy()
        a_check = softmax(z)
        actual_score = 2.0 * n * fft_convolve(a_check, a_check).max()
        if actual_score < best_score:
            best_score = actual_score
            best_a = a_check.copy()
        elapsed = time.monotonic() - t0
        if si % 3 == 0 or si == len(T_stages) - 1:
            print(f"    [{label}] Stage {si}/{len(T_stages)} T={T:.2f} score={best_score:.6f} ({elapsed:.0f}s)")

    return best_a, best_score


def lbgfs_anneal(z_init, T_stages, iters_per_stage, time_budget, label=""):
    """L-BFGS with log-sum-exp, multi-stage annealing."""
    t0 = time.monotonic()
    n = len(z_init)
    z = np.asarray(z_init, dtype=np.float64).copy()
    best_a = softmax(z)
    best_score = 2.0 * n * fft_convolve(best_a, best_a).max()

    for si, T in enumerate(T_stages):
        if time.monotonic() - t0 > time_budget:
            break

        def func(z_in):
            a = softmax(z_in)
            c = fft_convolve(a, a)
            Tc = T * c
            Tc_max = Tc.max()
            if Tc_max > 700:
                Tc_max = 700
            return (2.0 * n / T) * (Tc_max + np.log(np.sum(np.exp(Tc - Tc_max))))

        def jac(z_in):
            a = softmax(z_in)
            c = fft_convolve(a, a)
            Tc = T * c
            Tc_max = Tc.max()
            if Tc_max > 700:
                Tc_max = 700
            exp_Tc = np.exp(Tc - Tc_max)
            p = exp_Tc / exp_Tc.sum()
            p_rev = p[::-1]
            conv_pa = fft_convolve(p_rev, a)
            grad_a = (4.0 * n / T) * conv_pa[n - 1:2 * n - 1][::-1]
            S = np.dot(a, grad_a)
            return a * (grad_a - S)

        try:
            res = minimize(func, z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': iters_per_stage, 'ftol': 1e-16,
                                    'gtol': 1e-12, 'maxfun': iters_per_stage * 500, 'disp': False})
            z = res.x
        except Exception:
            pass

        a_check = softmax(z)
        actual_score = 2.0 * n * fft_convolve(a_check, a_check).max()
        if actual_score < best_score:
            best_score = actual_score
            best_a = a_check.copy()

    return best_a, best_score


def peak_shaving(a_init, time_budget, n_iter=20000, label=""):
    """Local search: reduce the maximum autocorrelation peak by redistributing mass."""
    t0 = time.monotonic()
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fft_convolve(a, a).max()
    rng = np.random.default_rng(12345)

    for it in range(n_iter):
        if time.monotonic() - t0 > time_budget:
            break
        c = fft_convolve(a, a)
        k_max = int(np.argmax(c))

        # Find top contributors to the peak
        js = max(0, k_max - n + 1)
        je = min(n - 1, k_max)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_max - idx]
        top_srcs = idx[np.argsort(contrib)[::-1][:3]]

        # Find target positions (low contribution to peak)
        low_contrib = idx[np.argsort(contrib)[:3]]
        # Also consider positions outside the peak's support
        outside = [j for j in range(n) if j < js or j > je]
        if outside:
            outside_arr = np.array(outside)
            low_out = outside_arr[np.argsort(a[outside_arr])[:2]]
        else:
            low_out = low_contrib[:2]

        targets = list(low_out) + list(low_contrib)
        if not targets:
            targets = [0]

        improved = False
        for src in top_srcs:
            for dst in targets:
                if dst == src:
                    continue
                for frac in [0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    c_trial = fft_convolve(a_trial, a_trial)
                    score = 2.0 * n * c_trial.max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.03)
                a_trial = a.copy()
                amt = a[i] * frac
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss > 1e-15:
                    a_trial = a_trial / ss
                    score = 2.0 * n * fft_convolve(a_trial, a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()

            # Periodic larger perturbation
            if it % 500 == 499:
                count = rng.integers(n // 20, n // 5)
                idx_pert = rng.choice(n, count, replace=False)
                a_trial = a.copy()
                a_trial[idx_pert] += rng.normal(0, 0.01, count)
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss > 1e-15:
                    a_trial = a_trial / ss
                    score = 2.0 * n * fft_convolve(a_trial, a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()

    return best_a, best_score


def subgradient_fasta(a_init, time_budget, n_active_frac=0.05, label=""):
    """
    Fast subgradient method with Nesterov acceleration on the exact max.
    Works in log-parameterization for automatic constraint handling.
    """
    t0 = time.monotonic()
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    z = np.log(np.maximum(a, 1e-300)).copy()
    y = z.copy()
    t_mom = 1.0
    step = 0.5 / n
    best_a = a.copy()
    best_score = 2.0 * n * fft_convolve(a, a).max()

    for it in range(500000):
        if time.monotonic() - t0 > time_budget:
            break

        a_y = softmax(y)
        c = fft_convolve(a_y, a_y)
        M = c.max()

        # Active set
        thresh = M - n_active_frac * M
        active = np.where(c >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        # Weight by proximity to max
        weights = (c[active] - thresh) / (M - thresh + 1e-300)
        weights = weights / weights.sum()

        # Subgradient: grad_a[i] = 2n * sum_{k in active} w[k] * 2 * a_y[k-i]
        p_active = np.zeros(len(c))
        p_active[active] = weights
        p_rev = p_active[::-1]
        conv_pa = fft_convolve(p_rev, a_y)
        g_a = 2.0 * n * conv_pa[n - 1:2 * n - 1][::-1]

        g_z = softmax_jacobian_applied(g_a, a_y)
        g_norm = np.linalg.norm(g_z)
        if g_norm < 1e-15:
            break

        # Backtracking
        alpha = step
        improved = False
        for bt in range(8):
            z_trial = z - alpha * g_z
            a_trial = softmax(z_trial)
            c_trial = fft_convolve(a_trial, a_trial)
            score_trial = 2.0 * n * c_trial.max()

            if score_trial < 2.0 * n * M - 1e-15:
                step = min(alpha * 1.3, 5.0)
                improved = True
                break
            alpha *= 0.5

        if not improved:
            step = max(alpha * 0.5, 1e-6 / n)
            # Random kick if stuck many times
            if it % 200 == 199:
                rng = np.random.default_rng(it)
                z += rng.normal(0, 0.01, n)
            continue

        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        beta = (t_mom - 1.0) / t_new
        z_old = z.copy()
        z = z_trial
        y = z + beta * (z - z_old)
        t_mom = t_mom if t_new < 1000 else 1.0
        t_mom = t_new

        if score_trial < best_score:
            best_score = score_trial
            best_a = a_trial.copy()

    return best_a, best_score


def structured_init(n, rng, mode=0):
    """Generate structured initializations."""
    if mode == 0:
        # Cosine window
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif mode == 1:
        # Gaussian
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif mode == 2:
        # Sine bump
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif mode == 3:
        # Linear ramp (asymmetric!)
        a = np.linspace(0.1, 1.0, n)
    elif mode == 4:
        # Quadratic (asymmetric)
        x = np.linspace(0, 1, n)
        a = x ** 2 * (1 - x) + 0.1
    elif mode == 5:
        # Cosine with phase (asymmetric)
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 3 * x + 0.5)
    elif mode == 6:
        # Dual bump
        x = np.linspace(0, 1, n)
        a = np.exp(-20 * (x - 0.3) ** 2) + 0.5 * np.exp(-20 * (x - 0.7) ** 2) + 0.05
    elif mode == 7:
        # Random smooth
        a = rng.random(n)
        # Smooth it
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
    elif mode == 8:
        # Inverse Gaussian (peaked in middle)
        x = np.linspace(-1, 1, n)
        a = 1.0 / (0.3 + x ** 2)
    elif mode == 9:
        # Sinc-like
        x = np.linspace(-2, 2, n)
        a = np.sinc(x) ** 2
    else:
        a = np.ones(n)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using cubic-like interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    if old_n >= 4:
        from scipy.interpolate import PchipInterpolator
        try:
            interp = PchipInterpolator(old_x, f_norm)
            f_interp = interp(new_x)
        except Exception:
            f_interp = np.interp(new_x, old_x, f_norm)
    else:
        f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if not a_list:
            return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def optimize_length(a_init, time_budget, label=""):

        best_a = np.ones(n) / n
        s = 1.0
    best_a = best_a / s
    best_score = evaluate_safe(best_a)
    z_init = np.log(np.maximum(best_a, 1e-300))

    def try_update(a, src):
        nonlocal best_a, best_score
        if a is None:
            return
        a = np.maximum(a, 0)
        ss = a.sum()
        if ss < 0.01:
            return
        a = a / ss
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t0
            print(f"    NEW BEST [{src}]: {sc:.6f} (n={n}, {elapsed:.0f}s)")

    # Phase 1: FISTA subgradient on exact max (fast, many iterations)

        a_out, _ = subgradient_fasta(best_a, t1, n_active_frac=0.03, label=f"{label}_sg")
        try_update(a_out, f"sg_{label}")

    # Phase 2: FISTA log-sum-exp annealing
    rem = time_budget - (time.monotonic() - t0)
    t2 = min(rem * 0.30, 200.0)
    if t2 > 15:
        z_cur = np.log(np.maximum(best_a, 1e-300))
        # Multi-stage: start smooth, go sharp
        T_stages = np.geomspace(5.0, 500.0, 15)
        iters_per_stage = max(50, int(t2 / 0.5))  # ~0.5s per stage
        a_out, _ = fista_anneal(z_cur, T_stages, iters_per_stage, t2, label=f"{label}_fista")
        try_update(a_out, f"fista_{label}")

    # Phase 3: L-BFGS log-sum-exp (polishing)
    rem = time_budget - (time.monotonic() - t0)
    t3 = min(rem * 0.25, 150.0)
    if t3 > 10:
        z_cur = np.log(np.maximum(best_a, 1e-300))
        T_stages = np.geomspace(10.0, 200.0, 10)
        iters_per_stage = max(20, int(t3 / 3.0))
        a_out, _ = lbgfs_anneal(z_cur, T_stages, iters_per_stage, t3, label=f"{label}_lbgfs")
        try_update(a_out, f"lbgfs_{label}")

    # Phase 4: Subgradient again (refined active set)
    rem = time_budget - (time.monotonic() - t0)
    t4 = min(rem * 0.15, 100.0)
    if t4 > 10:
        a_out, _ = subgradient_fasta(best_a, t4, n_active_frac=0.01, label=f"{label}_sg2")
        try_update(a_out, f"sg2_{label}")

    # Phase 5: Peak shaving
    rem = time_budget - (time.monotonic() - t0)
    t5 = min(rem * 0.10, 60.0)
    if t5 > 5:
        a_out, _ = peak_shaving(best_a, t5, n_iter=50000, label=f"{label}_ps")
        try_update(a_out, f"ps_{label}")

    return best_a, best_score

    Main search function. Uses multi-length, multi-initialization strategy
    with FISTA, L-BFGS, and peak shaving.
    """
    t_start = time.monotonic()
    time_limit = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float('inf')

    # Try to get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n0 = 1200
        best_seq = structured_init(n0, rng, mode=0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n0}, score={best_score:.6f}")

            return

        int(n_base * 2.0),
        800, 1000, 1200, 1500, 1800, 2400
