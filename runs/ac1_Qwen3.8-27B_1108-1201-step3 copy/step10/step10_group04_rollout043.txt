import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def project_simplex(v):
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:

            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:

            try:

                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score
    momentum = 0.85
    v = np.zeros(n)

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2.0)
        alpha = min(alpha, 0.3)

        v = momentum * v + (1.0 - momentum) * (b_new - y)
        y = y + v
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = b_new.copy()
            sy = 1.0
        y = y / sy

        a = (1.0 - alpha) * a + alpha * b_new
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

            if best_score < prev_best - 1e-7:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 8:
                rng = np.random.default_rng(it * 13 + 37)
                noise = rng.random(n) * 0.05
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                a = a + noise * 0.5
                a = np.maximum(a, 0)
                a = a / a.sum()
                v = np.zeros(n)
                stall_count = 0

        if verbose and (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def peak_shaving(a0, time_limit, n_peaks=5, verbose=True):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_max = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(54321)
    no_improve = 0

    max_steps = max(5000, int(time_limit * 200))

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        peak_idx = np.argsort(b)[::-1]
        top_peaks = peak_idx[:n_peaks]

        k_worst = top_peaks[0]
        js = max(0, k_worst - n + 1)
        je = min(n - 1, k_worst)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_worst - idx]

        i_max_local = np.argmax(contrib)
        i_max = idx[i_max_local]

        if a[i_max] < 1e-10:
            i_max = np.argmax(a)

        affected = set()
        for k in top_peaks:
            ks = max(0, k - n + 1)
            ke = min(n - 1, k)
            for i in range(ks, ke + 1):
                affected.add(i)

        safe = [j for j in range(n) if j not in affected and j != i_max]
        if not safe:
            thresh = np.percentile(a, 30)
            safe = [j for j in range(n) if j != i_max and a[j] < thresh]
        if not safe:
            safe = [j for j in range(n) if j != i_max]

        if not safe:
            break

        found = False
        n_tries = min(5, len(safe))
        dests = rng.choice(safe, n_tries, replace=False)

        for j_dest in dests:
            for frac in [0.002, 0.005, 0.01, 0.03, 0.08]:
                amt = a[i_max] * frac
                if amt < 1e-12:
                    continue
                a_trial = a.copy()
                a_trial[i_max] -= amt
                a_trial[j_dest] += amt
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t < 0.01:
                    continue
                a_trial = a_trial / s_t
                b_trial = fast_convolve(a_trial)
                if b_trial.max() < best_max - 1e-10:
                    a = a_trial
                    best_max = b_trial.max()
                    best_a = a.copy()
                    no_improve = 0
                    found = True
                    break
            if found:
                break

        if not found:
            no_improve += 1
            if no_improve > 500:
                noise = rng.random(n) * 0.02
                a = a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                b = fast_convolve(a)
                if b.max() < best_max:
                    best_max = b.max()
                    best_a = a.copy()
                no_improve = 0
            elif no_improve > 2000:
                break

        if verbose and step > 0 and step % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave step={step} ({elapsed:.0f}s): max={best_max:.8f}")

    return best_a, best_max * (2.0 * n)


def cutting_plane_polish(a0, time_limit, verbose=True):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_max = fast_convolve(a).max()
    best_score = 2.0 * n * best_max
    t0 = time.monotonic()

    cutting_planes = []

    for outer_iter in range(200):
        if time.monotonic() - t0 > time_limit:
            break
        if time.monotonic() - t0 > time_limit - 5:
            break

        b = fast_convolve(a)
        max_b = b.max()

        if max_b < best_max - 1e-10:
            best_max = max_b
            best_score = 2.0 * n * max_b
            best_a = a.copy()
            if verbose:
                print(f"    CP outer={outer_iter}: new best max={best_max:.8f}")

        sorted_idx = np.argsort(b)[::-1]
        top_k = min(10, 2 * n - 1)
        top_vals = b[sorted_idx[:top_k]]
        threshold = top_vals[-1]

        active_k = sorted_idx[:top_k]

        a_var = cp.Variable(n, nonneg=True)
        constraints = [cp.sum(a_var) == 1]

        target_max = max_b * (1.0 - 0.005)

        for k in active_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad = 2.0 * a[k - idx]
            f_at_a = b[k]
            c_vec = np.zeros(n)
            c_vec[idx] = grad
            rhs = (target_max + f_at_a) / 2.0
            constraints.append(cp.sum(c_vec * a_var) <= rhs)

        for cp_coeff, cp_rhs in cutting_planes:
            constraints.append(cp.sum(cp_coeff * a_var) <= cp_rhs)

        prob = cp.Problem(cp.Minimize(0), constraints)
        solver = get_lp_solver()
        try:
            prob.solve(solver=solver)
        except Exception:
            prob.solve(solver=cp.SCS)

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            target_max = max_b * (1.0 + 0.01)
            constraints2 = [cp.sum(a_var) == 1]
            for k in active_k:
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad = 2.0 * a[k - idx]
                f_at_a = b[k]
                c_vec = np.zeros(n)
                c_vec[idx] = grad
                rhs = (target_max + f_at_a) / 2.0
                constraints2.append(cp.sum(c_vec * a_var) <= rhs)
            prob = cp.Problem(cp.Minimize(0), constraints2)
            try:
                prob.solve(solver=solver)
            except Exception:
                prob.solve(solver=cp.SCS)
            if prob.status not in ('optimal', 'optimal_inaccurate'):
                if len(cutting_planes) > 100:
                    cutting_planes = cutting_planes[:50]
                continue

        a_new = np.asarray(a_var.value, dtype=np.float64)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new

        b_new = fast_convolve(a_new)
        max_new = b_new.max()

        new_violations = []
        for k in active_k:
            if b_new[k] > target_max * 1.001:
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad = 2.0 * a_new[k - idx]
                f_at = b_new[k]
                c_vec = np.zeros(n)
                c_vec[idx] = grad
                rhs = (target_max + f_at) / 2.0
                new_violations.append((c_vec, rhs))

        cutting_planes.extend(new_violations)
        if len(cutting_planes) > 500:
            cutting_planes = cutting_planes[-300:]

        a = a_new

    return best_a, best_score


def lbgfs_logsumexp(a0, time_limit, T_start=100.0, T_end=0.05, n_stages=10):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    def lse_objective(z_in, T_val):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = softmax_from_z(z_in)
        b = fast_convolve(a)
        Tb = T_val * b
        max_Tb = min(Tb.max(), 700.0)
        exp_Tb = np.exp(Tb - max_Tb)
        log_sum_exp = max_Tb + np.log(exp_Tb.sum())
        return (2.0 * n / T_val) * log_sum_exp

    def lse_jacobian(z_in, T_val):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z_in)
        a = softmax_from_z(z_in)
        b = fast_convolve(a)
        Tb = T_val * b
        max_Tb = min(Tb.max(), 700.0)
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]
        S = np.dot(grad_a, a)
        return a * (grad_a - S)

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        def func(z_in):
            return lse_objective(z_in, stage_T)

        def jac(z_in):
            return lse_jacobian(z_in, stage_T)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': 50, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': 25000,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def anti_palindromic_init(n, rng, p=2.0):
    x = np.linspace(0, 1, n)
    base = np.abs(x - 0.5)
    a = (0.5 - base) ** p
    a = np.maximum(a, 0)
    noise = rng.random(n) * 0.1
    a = a + noise
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = (1 - 2 * np.abs(x - 0.5)) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = a ** 1.5
    elif choice == 4:
        x = np.linspace(0, 1, n)
        a = np.exp(-2 * (x - 0.3) ** 2) + np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 5:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.6 + 0.4 * np.cos(3 * x)
    elif choice == 6:
        a = anti_palindromic_init(n, rng, p=1.5)
    else:
        a = anti_palindromic_init(n, rng, p=3.0)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def quick_optimize(a_init, time_budget, solver):
    n = len(a_init)
    a = np.maximum(a_init, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_safe(a)
    t0 = time.monotonic()

    def try_update(arr, label):
        nonlocal best_a, best_score
        if arr is None:
            return
        arr = np.maximum(arr, 0)
        ss = arr.sum()
        if ss < 0.01:
            return
        arr = arr / ss
        sc = evaluate_safe(arr)
        if np.isfinite(sc) and sc < best_score - 1e-10:
            best_score = sc
            best_a = arr.copy()
            elapsed = time.monotonic() - t0
            print(f"      [{label}] {sc:.6f} (n={n}, t={elapsed:.0f}s)")

    t_lp = time_budget * 0.6
    a_lp, _ = nesterov_lp(best_a, t_lp, solver=solver, n_max_iters=100000, verbose=False)
    try_update(a_lp, "lp")

    rem = time_budget - (time.monotonic() - t0)
    if rem > 5:
        a_ps, _ = peak_shaving(best_a, rem * 0.7, verbose=False)
        try_update(a_ps, "ps")

    rem = time_budget - (time.monotonic() - t0)
    if rem > 5:
        a_cp, _ = cutting_plane_polish(best_a, rem * 0.8, verbose=False)
        try_update(a_cp, "cp")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    time_limit_global = total_start + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update_global(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - total_start
            print(f"  GLOBAL BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is None:
        n_base = 1200
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_best = len(best_seq)

    print(f"\n=== Phase 1: Explore nearby lengths (current best={best_score:.6f}, n={n_best}) ===")
    explore_lengths = sorted(set([
        int(n_best * 0.75),
        int(n_best * 0.85),
        n_best,
        int(n_best * 1.15),
        int(n_best * 1.3),
    ]))
    explore_lengths = [ln for ln in explore_lengths if 200 <= ln <= 5000]

    for ln in explore_lengths:
        rem = time_limit_global - time.monotonic()
        if rem < 60:
            break
        t_alloc = min(rem * 0.15, 150.0)
        if t_alloc < 30:
            break

        init_long = interpolate_sequence(best_seq, ln)
        sc_init = evaluate_safe(init_long)
        if sc_init < best_score:
            best_score = sc_init
            best_seq = init_long.copy()
            n_best = ln

        a_out, sc_out = quick_optimize(init_long, t_alloc, solver)
        try_update_global(a_out, f"explore_n{ln}")

    print(f"\n=== Phase 2: Main optimization at n={n_best} (best={best_score:.6f}) ===")
    rem = time_limit_global - time.monotonic()
    t_main = rem * 0.45
    if t_main > 100:
        a_main, sc_main = nesterov_lp(best_seq, t_main, solver=solver, n_max_iters=300000, verbose=True)
        try_update_global(a_main, f"main_lp_n{n_best}")

    print(f"\n=== Phase 3: Peak shaving (best={best_score:.6f}) ===")
    rem = time_limit_global - time.monotonic()
    t_ps = min(rem * 0.25, 300.0)
    if t_ps > 20:
        a_ps, _ = peak_shaving(best_seq, t_ps, n_peaks=5, verbose=True)
        try_update_global(a_ps, f"ps_n{n_best}")

    print(f"\n=== Phase 4: Cutting plane polish (best={best_score:.6f}) ===")
    rem = time_limit_global - time.monotonic()
    t_cp = min(rem * 0.30, 300.0)
    if t_cp > 20:
        a_cp, _ = cutting_plane_polish(best_seq, t_cp, verbose=True)
        try_update_global(a_cp, f"cp_n{n_best}")

    print(f"\n=== Phase 5: LogSumExp refinement (best={best_score:.6f}) ===")
    rem = time_limit_global - time.monotonic()
    t_lse = min(rem * 0.30, 200.0)
    if t_lse > 15:
        a_lse = lbgfs_logsumexp(best_seq, t_lse, T_start=50.0, T_end=0.05, n_stages=10)
        try_update_global(a_lse, f"lse_n{n_best}")

    print(f"\n=== Phase 6: Perturbation restarts (best={best_score:.6f}) ===")
    rem = time_limit_global - time.monotonic()
    n_restarts = min(5, int(rem / 60))
    for ri in range(n_restarts):
        rem = time_limit_global - time.monotonic()
        if rem < 50:
            break
        t_restart = min(rem * 0.2, 200.0)
        rng_r = np.random.default_rng(seed + 1000 + ri)
        noise_scale = 0.02 + 0.02 * ri
        perturbed = best_seq + rng_r.random(len(best_seq)) * noise_scale
        perturbed = np.maximum(perturbed, 0)
        perturbed = perturbed / perturbed.sum()
        a_r, _ = nesterov_lp(perturbed, t_restart, solver=solver, n_max_iters=50000, verbose=False)
        try_update_global(a_r, f"restart{ri}_n{len(best_seq)}")

    print(f"\n=== Phase 7: Final local polish (best={best_score:.6f}) ===")
    rem = time_limit_global - time.monotonic()
    if rem > 20:
        a_final_ps, _ = peak_shaving(best_seq, min(rem * 0.5, 100.0), n_peaks=8, verbose=False)
        try_update_global(a_final_ps, "final_ps")

    rem = time_limit_global - time.monotonic()
    if rem > 15:
        a_final_cp, _ = cutting_plane_polish(best_seq, min(rem * 0.6, 80.0), verbose=False)
        try_update_global(a_final_cp, "final_cp")

    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs) and fs < best_score:
            best_score = fs

    total_time = time.monotonic() - total_start
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)}, total_time={total_time:.0f}s)")
    return [float(x) for x in best_seq]
