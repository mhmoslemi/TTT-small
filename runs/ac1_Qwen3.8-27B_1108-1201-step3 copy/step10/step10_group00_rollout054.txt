    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    shifted = (b - max_b) / T
    exp_shifted = np.exp(shifted)
    log_sum = np.log(exp_shifted.sum())
    lse = max_b + T * log_sum

    w = exp_shifted / exp_shifted.sum()
    
    # Gradient of b w.r.t. a (autocorrelation)

def adam_step(params, grads, state, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    m = beta1 * state['m'] + (1 - beta1) * grads
    v = beta2 * state['v'] + (1 - beta2) * grads**2
    m_hat = m / (1 - beta1**state['t'])
    v_hat = v / (1 - beta2**state['t'])
    params -= lr * m_hat / (np.sqrt(v_hat) + eps)
    state['m'] = m
    state['v'] = v
    state['t'] += 1

    grad_z = a * (2.0 * n * grad_a - 2.0 * n * dot_a_grad)

    return obj, grad_z


def adam_loop(z, T, time_limit, lr, beta1=0.9, beta2=0.999, eps=1e-8,
              verbose_interval=5000, verbose_prefix=""):
    """Run Adam optimization at fixed temperature T for time_limit seconds."""
    n = len(z)
    z_cur = z.copy()
    m = np.zeros_like(z_cur)
    v = np.zeros_like(z_cur)
    t = 0
    t0 = time.monotonic()

    best_obj = float('inf')
    best_z = z_cur.copy()
    no_improve_count = 0

    while time.monotonic() - t0 < time_limit:
        obj, grad = compute_obj_grad(z_cur, T)

        if not np.isfinite(obj):
            z_cur = best_z.copy()
            continue

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_z = z_cur.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        # Stall detection: reduce effective lr if stuck
        effective_lr = lr
        if no_improve_count > 5000:
            effective_lr = lr * 0.1
            no_improve_count = 0

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        z_cur = z_cur - effective_lr * m_hat / (np.sqrt(v_hat) + eps)

        # Clip to prevent numerical issues
        if np.abs(z_cur).max() > 80:
            z_cur = np.clip(z_cur, -80, 80)

        if verbose_prefix and t % verbose_interval == 0:
            elapsed = time.monotonic() - t0
            a_check = softmax_safe(z_cur)
            b_check = fast_convolve(a_check)
            score_check = 2.0 * n * b_check.max()
            print(f"    {verbose_prefix} t={t} ({elapsed:.1f}s): "
                  f"LSE_obj={obj:.6f}, true_score={score_check:.8f}, T={T:.4f}")

    return best_z, best_obj


def evaluate_local(a):
    """Fast local evaluation of the objective."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def interpolate_to_length(a, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0)
    ss = a_interp.sum()
    if ss > 1e-15:
        a_interp = a_interp / ss
    return a_interp


def log_from_a(a):
    """Convert normalized a to log-space z (up to a constant)."""
    return np.log(np.maximum(a, 1e-300))


def structured_init(n, rng):
    """Create diverse structured initializations."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * rng.integers(2, 8) * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** rng.uniform(1.0, 2.5) + 0.05
    elif choice == 5:
        c1 = rng.uniform(0.2, 0.4)
        c2 = rng.uniform(0.6, 0.8)
        a = np.exp(-30 * (x - c1) ** 2) + 0.7 * np.exp(-30 * (x - c2) ** 2)
    elif choice == 6:
        a = 1.0 + 0.3 * np.sin(2 * np.pi * rng.uniform(1, 5) * x + rng.uniform(0, 6.28))
    else:
        a = np.ones(n) * (1.0 + rng.uniform(0.05, 0.2) * rng.standard_normal(n))
        a = np.maximum(a, 0.01)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def multi_stage_adam(z0, time_limit, n_stages=None):
    """Multi-stage Adam with geometric temperature annealing."""
    n = len(z0)
    if n_stages is None:
        n_stages = max(6, int(np.log2(max(1, n)) ))
    
    # Temperature schedule: geometric from high to very low
    T_start = max(2.0, n / 40.0)
    T_end = 0.005
    T_stages = np.geomspace(T_start, T_end, n_stages)

    z_cur = z0.copy()
    t0 = time.monotonic()
    best_obj = float('inf')
    best_z = z0.copy()

    for stage in range(n_stages):
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 2:
            break

        T = T_stages[stage]
        # Learning rate: start higher, decrease
        lr = 0.1 / (1.0 + 0.3 * stage)

        # Time for this stage
        stage_time = min(remaining / (n_stages - stage), 120.0)
        if stage == n_stages - 1:
            stage_time = remaining - 2

        if stage_time < 1:
            continue

        z_cur, obj_val = adam_loop(z_cur, T, stage_time, lr,
                                    verbose_interval=max(500, int(stage_time / 2)),
                                    verbose_prefix=f"Stage{stage+1}")

        a_check = softmax_safe(z_cur)
        score = 2.0 * n * fast_convolve(a_check).max()
        if score < best_obj:
            best_obj = score
            best_z = z_cur.copy()

        elapsed = time.monotonic() - t0
        print(f"  Stage {stage+1}/{n_stages} done (T={T:.4f}, {elapsed:.0f}s): "
              f"score={score:.8f}, best={best_obj:.8f}")

    return best_z, best_obj


def final_polish(z0, time_limit):
    """Ultra-low temperature Adam for final polishing."""
    n = len(z0)
    T_stages = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    
    z_cur = z0.copy()
    t0 = time.monotonic()
    best_obj = float('inf')
    best_z = z0.copy()

    for stage, T in enumerate(T_stages):
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        stage_time = min(remaining / (len(T_stages) - stage), 60.0)
        if stage_time < 1:
            continue

        lr = 0.02 / (1.0 + 0.5 * stage)
        z_cur, _ = adam_loop(z_cur, T, stage_time, lr,
                              verbose_interval=10000,
                              verbose_prefix=f"Polish{stage+1}")

        a_check = softmax_safe(z_cur)
        score = 2.0 * n * fast_convolve(a_check).max()
        if score < best_obj:
            best_obj = score
            best_z = z_cur.copy()

        print(f"  Polish stage {stage+1} (T={T:.4f}): score={score:.8f}")

    return best_z, best_obj


def diverse_restart(best_a, n, time_per_restart, rng, seed_offset=0):
    """Create a perturbed version of best_a and briefly optimize it."""
    a_pert = best_a.copy()
    # Add structured perturbation
    x = np.linspace(0, 2 * np.pi, n)
    n_freqs = 3
    for _ in range(n_freqs):
        freq = rng.integers(1, 10)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.01, 0.05)
        a_pert = a_pert + amp * np.cos(freq * x + phase)
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        a_pert = best_a.copy()
        s = 1.0
    a_pert = a_pert / s

    z = log_from_a(a_pert)
    # Brief Adam optimization
    T = max(0.1, n / 200.0)
    z_opt, _ = adam_loop(z, T, time_per_restart, lr=0.05,
                          verbose_interval=10000, verbose_prefix="Restart")
    a_opt = softmax_safe(z_opt)
    return a_opt


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses multi-resolution Adam with LSE smoothing, diverse restarts,
    and ultra-low-temperature final polish.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 15.0)

    best_a = None
    best_score = float('inf')

    # Try to load parent sequence
    parent_loaded = False
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
            best_a = parent.copy()
            best_score = p_score
            parent_loaded = True
    except Exception:
        pass

    # Determine working length
    if best_a is not None:
        n_work = len(best_a)
    else:
        n_work = 1200
        best_a = structured_init(n_work, rng)
        best_score = evaluate_local(best_a)
        print(f"Default init: n={n_work}, score={best_score:.8f}")

    n = n_work
    print(f"\nWorking with n={n}, initial score={best_score:.8f}")

    def try_update(a_candidate, label):
        nonlocal best_a, best_score
        if a_candidate is None:
            return
        a_candidate = np.maximum(np.asarray(a_candidate, dtype=np.float64), 0)
        s = a_candidate.sum()
        if s < 0.01:
            return
        a_candidate = a_candidate / s
        sc = evaluate_local(a_candidate)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_candidate.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a_candidate)})")

    # ============================================================
    # PHASE 1: Multi-resolution optimization
    # ============================================================
    print(f"\n{'='*60}")
    print(f"PHASE 1: Multi-resolution Adam (score={best_score:.8f})")
    print(f"{'='*60}")

    # Stage 1a: Coarse resolution (n/4)
    n_coarse = max(100, n // 4)
    elapsed = time.monotonic() - t_start
    t_alloc = min((deadline - time.monotonic()) * 0.15, 180.0)
    if t_alloc > 20 and parent_loaded:
        print(f"\n  Coarse: n={n_coarse}, budget={t_alloc:.0f}s")
        a_coarse = interpolate_to_length(best_a, n_coarse)
        z_coarse = log_from_a(a_coarse)
        z_coarse, _ = multi_stage_adam(z_coarse, min(t_alloc, 60.0), n_stages=5)
        a_coarse_opt = softmax_safe(z_coarse)
        # Interpolate back to full resolution
        a_fine_from_coarse = interpolate_to_length(a_coarse_opt, n)
        try_update(a_fine_from_coarse, "coarse_interp")

    # Stage 1b: Half resolution (n/2)
    n_half = max(200, n // 2)
    t_alloc = min((deadline - time.monotonic()) * 0.20, 250.0)
    if t_alloc > 20:
        print(f"\n  Half: n={n_half}, budget={t_alloc:.0f}s")
        a_half = interpolate_to_length(best_a, n_half)
        z_half = log_from_a(a_half)
        z_half, _ = multi_stage_adam(z_half, min(t_alloc, 100.0), n_stages=6)
        a_half_opt = softmax_safe(z_half)
        a_fine_from_half = interpolate_to_length(a_half_opt, n)
        try_update(a_fine_from_half, "half_interp")

    # Stage 1c: Full resolution
    t_alloc = min((deadline - time.monotonic()) * 0.35, 500.0)
    if t_alloc > 30:
        print(f"\n  Full: n={n}, budget={t_alloc:.0f}s")
        z_full = log_from_a(best_a)
        z_full, _ = multi_stage_adam(z_full, t_alloc, n_stages=8)
        a_full_opt = softmax_safe(z_full)
        try_update(a_full_opt, "adam_full")

    # ============================================================
    # PHASE 2: Diverse restarts
    # ============================================================
    print(f"\n{'='*60}")
    print(f"PHASE 2: Diverse restarts (score={best_score:.8f})")
    print(f"{'='*60}")

    remaining = deadline - time.monotonic()
    n_restarts = max(2, int(remaining / 80))
    n_restarts = min(n_restarts, 5)
    t_per_restart = min(remaining / (n_restarts + 1), 80.0)

    for i in range(n_restarts):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        print(f"\n  Restart {i+1}/{n_restarts} (budget={t_per_restart:.0f}s)")

        # Create diverse starting point
        if i == 0:
            # Use parent with small perturbation
            a_start = best_a + 0.01 * rng.standard_normal(n)
            a_start = np.maximum(a_start, 0)
            a_start = a_start / a_start.sum()
        elif i == 1:
            # Structured init
            a_start = structured_init(n, rng)
        elif i == 2:
            # Uniform with low-frequency modulation
            x = np.linspace(0, 1, n)
            a_start = 1.0 + 0.3 * np.cos(2 * np.pi * rng.uniform(0.5, 3.0) * x)
            a_start = np.maximum(a_start, 0.01)
            a_start = a_start / a_start.sum()
        elif i == 3:
            # Two-bump structure
            c1 = rng.uniform(0.1, 0.4)
            c2 = rng.uniform(0.6, 0.9)
            w1 = rng.uniform(0.01, 0.05)
            w2 = rng.uniform(0.01, 0.05)
            x = np.linspace(0, 1, n)
            a_start = np.exp(-(x - c1) ** 2 / (2 * w1 ** 2)) + \
                      rng.uniform(0.3, 0.8) * np.exp(-(x - c2) ** 2 / (2 * w2 ** 2))
            a_start = a_start / a_start.sum()
        else:
            # Large perturbation of best
            a_start = best_a * (1.0 + 0.15 * rng.standard_normal(n))
            a_start = np.maximum(a_start, 0)
            s = a_start.sum()
            if s < 1e-15:
                a_start = structured_init(n, rng)
            else:
                a_start = a_start / s

        z_start = log_from_a(a_start)
        t_sub = min(t_per_restart, remaining - 20)
        if t_sub > 10:
            z_opt, _ = multi_stage_adam(z_start, t_sub, n_stages=5)
            a_opt = softmax_safe(z_opt)
            try_update(a_opt, f"restart_{i}")

    # ============================================================
    # PHASE 3: Final polish
    # ============================================================
    print(f"\n{'='*60}")
    print(f"PHASE 3: Final polish (score={best_score:.8f})")
    print(f"{'='*60}")

    remaining = deadline - time.monotonic()
    if remaining > 30:
        t_polish = min(remaining * 0.6, 200.0)
        print(f"  Polish budget: {t_polish:.0f}s")
        z_best = log_from_a(best_a)
        z_polished, _ = final_polish(z_best, t_polish)
        a_polished = softmax_safe(z_polished)
        try_update(a_polished, "polish")

        # One more round of very brief Adam at lowest temp
        remaining = deadline - time.monotonic()
        if remaining > 15:
            T_ultra = 0.001
            z_curr = log_from_a(best_a)
            z_final, _ = adam_loop(z_curr, T_ultra, min(remaining - 5, 30.0),
                                     lr=0.005, verbose_interval=5000,
                                     verbose_prefix="UltraPolish")
            a_final = softmax_safe(z_final)
            try_update(a_final, "ultra_polish")

    # ============================================================
    # PHASE 4: Quick sweep at different lengths
    # ============================================================
    remaining = deadline - time.monotonic()
    if remaining > 100:
        print(f"\n{'='*60}")
        print(f"PHASE 4: Length sweep (score={best_score:.8f}, remaining={remaining:.0f}s)")
        print(f"{'='*60}")

        for n_try in [n - 100, n + 100, n + 200]:
            if n_try <= 100 or remaining < 40:
                continue
            t_sub = min((deadline - time.monotonic()) * 0.25, 80.0)
            if t_sub < 20:
                break
            print(f"\n  Trying n={n_try}, budget={t_sub:.0f}s")
            a_interp = interpolate_to_length(best_a, n_try)
            z_interp = log_from_a(a_interp)
            z_opt, _ = multi_stage_adam(z_interp, t_sub, n_stages=5)
            a_opt = softmax_safe(z_opt)
            try_update(a_opt, f"n_{n_try}")

    # Final verification
    if best_a is not None:
        final_score = evaluate_local(best_a)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\n{'='*60}")
        print(f"FINAL: score={best_score:.8f}, n={len(best_a)}")
        print(f"Total time: {time.monotonic() - t_start:.1f}s")
        print(f"{'='*60}")

    # Ensure we return a valid list
    if best_a is None:
        best_a = np.ones(1200) / 1200

    result = [float(max(0.0, min(1000.0, x))) for x in best_a]
    return result
