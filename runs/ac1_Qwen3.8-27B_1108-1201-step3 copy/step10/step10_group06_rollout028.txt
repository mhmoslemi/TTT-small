def golden_section_search(f, lo, hi, max_iter=35):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-16:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5

            a_new = (1.0 - best_t) * a + best_t * b
            a_new = np.maximum(a_new, 0)
            cs = a_new.sum()
            if cs < 1e-10:
                a_new = a
            else:
                a_new = a_new / cs
            score_new = compute_score(a_new)
            if score_new < best_score - 1e-13:
                best_score = score_new
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1
            a = a_new
        else:
            no_improve += 1
            alpha = 3.0 / (iters + 3.0)
            a = (1.0 - alpha) * a + alpha * b
            a = np.maximum(a, 0)
            cs = a.sum()
            if cs < 1e-10:
                a = np.ones(n) / n
            else:
                a = a / cs

        a_avg = (iters * a_avg + a) / (iters + 1)

        sc_avg = compute_score(a_avg)
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = a_avg.copy()

        sc_b = compute_score(b)
        if sc_b < best_score:
            best_score = sc_b
            best_a = b.copy()

        if verbose and iters % 5 == 0:
            print(f"    FP n={n} it={iters}: best={best_score:.8f}")

        if no_improve > 40:
            break

    return best_a, best_score, iters

def subgradient_descent(a0, time_limit, verbose=False):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    lr = 0.01
    no_improve = 0
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        max_val = float(conv.max())

        k_top = min(30, 2 * n - 1)
        peaks = np.argpartition(conv, -k_top)[-k_top:]
        pv = conv[peaks]
        temp = 0.002 * max_val
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            break
        w = w / ws

        grad = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-13:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            grad[j] += 2.0 * w[idx] * a[k - j]

        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-10:
            a_new = a
        else:
            a_new = a_new / sn

        sc = compute_score(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        lr *= 0.998
        if no_improve > 50:
            lr *= 0.8
            no_improve = 0
        if no_improve > 300:
            break

        if verbose and iters % 20 == 0:
            print(f"    SG it={iters}: best={best_score:.8f}")

    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def init_power_law(n, alpha, rng):
    t = np.arange(n, dtype=np.float64) + 1.0
    a = t ** alpha
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    a = a / s
    shift = rng.uniform(-0.1, 0.1) * n
    a_shifted = np.interp(t - shift - 1, np.arange(max(0, int(-shift)), n + max(0, int(shift))), a, left=0, right=0)
    a_shifted = np.maximum(a_shifted, 0)
    ss = a_shifted.sum()
    if ss > 1e-10:
        a_shifted = a_shifted / ss
    return a_shifted

def init_gaussian_asym(n, rng, center_frac=0.3):
    t = np.arange(n, dtype=np.float64)
    center = n * center_frac
    width = n * rng.uniform(0.15, 0.35)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_ramp_gaussian(n, rng):
    t = np.arange(n, dtype=np.float64)
    ramp = (t + 1) / (n + 1)
    center = n * rng.uniform(0.2, 0.5)
    width = n * rng.uniform(0.2, 0.4)
    gauss = np.exp(-0.5 * ((t - center) / width) ** 2)
    a = ramp * gauss
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        idx = np.nonzero(cond)[0][-1]
        theta = (cssv[idx] - 1.0) / (idx + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def basin_hopping_peak(a0, time_limit, rng, verbose=False):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    count = 0

    while time.monotonic() - t0 < time_limit:
        count += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 30:
            break
        t_alloc = min(rem * 0.35, 90.0)

        pert_type = count % 5
        perturbed = a.copy()

        if pert_type == 0:
            conv_p = fast_convolve(a)
            k_peak = int(np.argmax(conv_p))
            i_start = max(0, k_peak - n + 1)
            i_end = min(n - 1, k_peak)
            idx = np.arange(i_start, i_end + 1)
            contrib = a[k_peak - idx] * a[idx]
            top_frac = max(1, len(idx) // 5)
            top_idx = idx[np.argsort(contrib)[-top_frac:]]
            for ii in top_idx:
                perturbed[ii] *= (1.0 - 0.03 - 0.05 * rng.random())
            non_peak = np.setdiff1d(np.arange(n), idx)
            if len(non_peak) > 0:
                picks = rng.choice(non_peak, size=min(5, len(non_peak)), replace=False)
                for ii in picks:
                    perturbed[ii] *= (1.0 + 0.02 + 0.03 * rng.random())
        elif pert_type == 1:
            sigma = 0.015 + 0.025 * rng.random()
            perturbed = a + rng.standard_normal(n) * sigma
        elif pert_type == 2:
            shift = rng.integers(1, max(2, n // 8))
            shifted = np.roll(a, shift)
            mix = rng.uniform(0.2, 0.5)
            perturbed = (1.0 - mix) * a + mix * shifted
        elif pert_type == 3:
            blen = rng.integers(max(1, n // 16), max(2, n // 4))
            i1 = rng.integers(0, max(1, n - blen))
            i2 = rng.integers(0, max(1, n - blen))
            if abs(i1 - i2) > blen:
                temp = perturbed[i1:i1+blen].copy()
                perturbed[i1:i1+blen] = perturbed[i2:i2+blen]
                perturbed[i2:i2+blen] = temp
        else:
            n_squeeze = rng.integers(n // 10, n // 5)
            squeeze_idx = rng.choice(n, size=n_squeeze, replace=False)
            scale = rng.uniform(0.5, 0.8)
            for ii in squeeze_idx:
                perturbed[ii] *= scale
            expand_idx = np.setdiff1d(np.arange(n), squeeze_idx)
            boost = 1.0 + (1.0 - scale) * n_squeeze / max(1, len(expand_idx))
            perturbed[expand_idx] *= boost

        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-10:
            continue
        perturbed = perturbed / ps

        a_opt, s_opt, _ = exact_coord_descent(perturbed, t_alloc, label=f"bh{count}")
        if s_opt < best_score - 1e-10:
            best_score = s_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"    BH{count}: improved to {best_score:.8f}")

    return best_a, best_score

def multi_start_screen(n_values, time_budget, rng):
    results = []
    t0 = time.monotonic()
    for n in n_values:
        rem = time_budget - (time.monotonic() - t0)
        if rem < 10:
            break
        inits = [
            init_power_law(n, 1.0, rng),
            init_power_law(n, 0.5, rng),
            init_power_law(n, 2.0, rng),
            init_gaussian_asym(n, rng, 0.25),
            init_gaussian_asym(n, rng, 0.4),
            init_ramp_gaussian(n, rng),
            np.maximum(np.ones(n) * 0.1 + rng.standard_normal(n) * 0.05, 0),
        ]
        for j, init in enumerate(inits):
            rem2 = time_budget - (time.monotonic() - t0)
            if rem2 < 5:
                break
            init = np.maximum(init, 0)
            s_init = init.sum()
            if s_init < 1e-10:
                init = np.ones(n) / n
            else:
                init = init / s_init
            t_alloc = max(2.0, min(8.0, rem2 / 4.0))
            a_opt, s_opt, _ = exact_coord_descent(init, t_alloc, label=f"ms_n{n}_{j}")
            results.append((s_opt, a_opt, n))

    results.sort(key=lambda x: x[0])
    return results

def propose_candidate():
    """
    Multi-phase optimization combining parametric initialization, LP fictitious play,
    exact coordinate descent, subgradient descent, and peak-targeted basin hopping.
    Searches across multiple sequence lengths for the best score.
    """
    deadline = time.monotonic() + 1780.0
    rng = np.random.default_rng(42)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    # Phase 1: Multi-start screening across lengths
    print("\n=== Phase 1: Multi-Start Screening ===")
    t1 = 90.0
    n_parent = len(parent) if parent is not None else 1024
    n_values = sorted(set([n_parent, 512, 768, 1024, 1536, 2048]))
    n_values = [n for n in n_values if 256 <= n <= 2500]

    results = multi_start_screen(n_values, t1, rng)
    print(f"  Screened {len(results)} candidates, best={results[0][0]:.8f}")

    # Also try parent if available
    if parent is not None:
        for n_try in [n_parent, min(n_parent + 256, 2048), max(n_parent - 256, 512)]:
            if n_try < 256 or n_try > 2500:
                continue
            a_init = interpolate_seq(parent, n_try)
            t_alloc = min(10.0, t1 * 0.2)
            a_opt, s_opt, _ = exact_coord_descent(a_init, t_alloc, label=f"parent_n{n_try}")
            results.append((s_opt, a_opt, n_try))
        results.sort(key=lambda x: x[0])

    if results and results[0][0] < best_score:
        best_score = results[0][0]
        best_seq = results[0][1].copy()
        sc = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
        print(f"  Phase 1 best: {best_score:.8f} (n={len(best_seq)})")

    # Phase 2: LP Fictitious Play on best candidate
    print(f"\n=== Phase 2: LP Fictitious Play (best={best_score:.8f}) ===")
    n_work = len(best_seq)
    rem = deadline - time.monotonic() - 600
    if rem > 30:
        t_fp = min(rem * 0.55, 350.0)
        print(f"  FP budget={t_fp:.0f}s, n={n_work}")
        a_fp, s_fp, iters = lp_fictitious_play(best_seq, t_fp, solver=solver, verbose=True)
        sc = float(evaluate_sequence(a_fp.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_fp.copy()
            print(f"  NEW BEST FP: {sc:.8f}")
        else:
            print(f"  FP score: {s_fp:.8f} (eval: {sc:.8f})")

    # Phase 2b: Try LP at a different length
    rem = deadline - time.monotonic() - 500
    if rem > 60:
        alt_n = min(n_work + 256, 2500)
        if alt_n != n_work:
            a_alt = interpolate_seq(best_seq, alt_n)
            t_fp2 = min(rem * 0.4, 150.0)
            print(f"  FP at n={alt_n}, budget={t_fp2:.0f}s")
            a_fp2, s_fp2, _ = lp_fictitious_play(a_alt, t_fp2, solver=solver, verbose=True)
            sc2 = float(evaluate_sequence(a_fp2.tolist()))
            if np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_seq = a_fp2.copy()
                print(f"  NEW BEST FP2: {sc2:.8f} (n={alt_n})")

    # Phase 3: Coordinate descent polishing
    print(f"\n=== Phase 3: CD Polishing (best={best_score:.8f}) ===")
    n_work = len(best_seq)
    for rnd in range(4):
        rem = deadline - time.monotonic() - 350
        if rem < 20:
            break
        t_cd = min(rem * 0.3, 180.0)
        print(f"  CD round {rnd+1}, budget={t_cd:.0f}s")
        a_cd, s_cd, sw = exact_coord_descent(best_seq, t_cd, verbose=True, label=f"ph3_r{rnd}")
        sc = float(evaluate_sequence(a_cd.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_cd.copy()
            print(f"  NEW BEST CD: {sc:.8f}")
        else:
            if rnd > 0:
                break

    # Phase 4: Subgradient descent
    print(f"\n=== Phase 4: Subgradient (best={best_score:.8f}) ===")
    n_work = len(best_seq)
    rem = deadline - time.monotonic() - 250
    if rem > 30:
        t_sg = min(rem * 0.5, 120.0)
        print(f"  SG budget={t_sg:.0f}s")
        a_sg, s_sg = subgradient_descent(best_seq, t_sg, verbose=True)
        sc = float(evaluate_sequence(a_sg.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_sg.copy()
            print(f"  NEW BEST SG: {sc:.8f}")

    # Phase 5: Peak-targeted basin hopping
    print(f"\n=== Phase 5: Basin Hopping (best={best_score:.8f}) ===")
    n_work = len(best_seq)
    rem = deadline - time.monotonic() - 200
    if rem > 30:
        t_bh = min(rem * 0.6, 250.0)
        print(f"  BH budget={t_bh:.0f}s")
        a_bh, s_bh = basin_hopping_peak(best_seq, t_bh, rng, verbose=True)
        sc = float(evaluate_sequence(a_bh.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_bh.copy()
            print(f"  NEW BEST BH: {sc:.8f}")

    # Phase 6: Try longer sequences
    print(f"\n=== Phase 6: Longer Sequences (best={best_score:.8f}) ===")
    n_work = len(best_seq)
    for longer_n in [n_work + 512, n_work + 1024]:
        if longer_n > 3000:
            break
        rem = deadline - time.monotonic() - 150
        if rem < 30:
            break
        a_long = interpolate_seq(best_seq, longer_n)
        t_long = min(rem * 0.4, 120.0)
        print(f"  Trying n={longer_n}, budget={t_long:.0f}s")
        a_opt, s_opt, _ = exact_coord_descent(a_long, t_long, label=f"long_{longer_n}")
        sc = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_opt.copy()
            print(f"  NEW BEST long: {sc:.8f} (n={longer_n})")

    # Phase 7: Final coordinated polishing
    print(f"\n=== Phase 7: Final Polish (best={best_score:.8f}) ===")
    n_work = len(best_seq)
    rem = deadline - time.monotonic() - 30
    if rem > 15:
        t_final = min(rem * 0.6, 60.0)
        a_fin, s_fin, _ = exact_coord_descent(best_seq, t_final, label="final")
        sc = float(evaluate_sequence(a_fin.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_fin.copy()
            print(f"  NEW BEST final: {sc:.8f}")

    # One more quick pass
    rem = deadline - time.monotonic() - 10
    if rem > 5:
        a_last, s_last, _ = exact_coord_descent(best_seq, min(rem * 0.7, 20.0), label="last")
        sc = float(evaluate_sequence(a_last.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_last.copy()

    if best_seq is None:
        best_seq = np.ones(512) / 512

    # Final verification
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
