def project_simplex(v):
    """Project vector v onto the probability simplex (a>=0, sum(a)=1)."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1
    rho = np.where(u > cv)[0][-1]
    theta = (cv[rho] + 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def build_T_sparse(a):
    n = len(a)
    diag_data = [np.full(n, a[d]) for d in range(n)]
    T = diags(diag_data, list(range(n)), shape=(2 * n - 1, n), format='csr')
    return T

def solve_lp_best_response(a):

        else:
            a = a / sa
        sc_a = 2.0 * n * fast_convolve(a).max()
        sc_y = 2.0 * n * fast_convolve(y).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        if (it + 1) % 500 == 0:
            if best_score < prev_best - 1e-9:
                stall = 0
            else:
                stall += 1
            prev_best = best_score
            if stall >= 8:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 6)
                noise = 0.04 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall = 0
            if (it + 1) % 2000 == 0:
                print(f"    NL it={it+1} t={time.monotonic()-t0:.0f}s score={best_score:.8f}")

    return best_a, best_score

def lse_objective(z, n, T):
    z_shift = z - z.max()
    exp_z = np.exp(z_shift)
    a = exp_z / exp_z.sum()
    b = fast_convolve(a)
    max_b = b.max()
    lse = max_b + T * np.log(np.sum(np.exp((b - max_b) / T)))
    obj = 2.0 * n * lse
    w = np.exp((b - max_b) / T)
    w /= w.sum()
    grad_a = np.zeros(n)
    thresh = 1e-12
    active = np.where(w > thresh)[0]
    for k in active:
        js = max(0, k - n + 1)
        je = min(n, k + 1)
        idx = np.arange(js, je)
        grad_a[idx] += w[k] * a[k - idx]
    grad_a *= 2.0
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z

def _lse_func(z_x):
    return lse_objective(z_x, _globals_n, _globals_T)[0]

def _lse_jac(z_x):
    return lse_objective(z_x, _globals_n, _globals_T)[1]

def lbgfs_refine(a0, time_limit, T_stages, maxiter_stage=300):
    n = len(a0)
    _globals_n = n
    z = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()
    for T_stage in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        _globals_T = T_stage
        try:
            res = minimize(_lse_func, z, jac=_lse_jac, method='L-BFGS-B',
                           options={'maxiter': maxiter_stage, 'gtol': 1e-15,
                                    'ftol': 1e-20, 'disp': False})
            z = res.x
        except Exception:
            pass
    z_shift = z - z.max()
    exp_z = np.exp(z_shift)
    a = exp_z / exp_z.sum()
    return a

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u) - 1.0
    rho = np.where(u > cv)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    theta = (cv[rho[-1]] + 1.0) / (rho[-1] + 1)
    return np.maximum(v - theta, 0)

def topk_subgradient(a0, time_limit, k=3, max_iters=100000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = 0.02
    patience = 0
    scale = 1.0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        top_k = np.argsort(b)[-k:]
        g = np.zeros(n)
        for kk in top_k:
            js = max(0, kk - n + 1)
            je = min(n, kk + 1)
            idx = np.arange(js, je)
            g[idx] += a[kk - idx]
        g /= k
        g *= 2.0
        a_new = a - eta * scale * g
        a_new = project_simplex(a_new)
        sc = 2.0 * n * fast_convolve(a_new).max()
        if sc < best_score - 1e-15:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            patience = 0
        else:
            patience += 1
            if patience % 30 == 0:
                scale *= 0.7
                if scale < 0.005:
                    break
    return best_a, best_score

def structured_init(n, rng):
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4.0 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0.01)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = np.exp(-50 * (x - 0.25) ** 2) + np.exp(-50 * (x - 0.75) ** 2)
    elif choice == 7:
        a = np.exp(-10 * x) + np.exp(-10 * (1 - x)) + 0.1
    elif choice == 8:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 5 * x) + 0.2 * np.cos(2 * np.pi * 11 * x)
        a = np.maximum(a, 0.01)
    else:
        a = 1.0 / (1.0 + np.exp(-20 * (x - 0.5)))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def perturbed_seq(a, rng, strength=0.03):
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 10)
    phase = rng.uniform(0, 2 * np.pi)
    pert = strength * np.cos(freq * x + phase)
    a_p = a + pert
    a_p = np.maximum(a_p, 0)
    s = a_p.sum()
    if s < 1e-15:
        return a.copy()
    return a_p / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def multi_freq_perturb(a, rng, strength=0.02):
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    pert = np.zeros(n)
    for _ in range(3):
        freq = rng.integers(1, 12)
        phase = rng.uniform(0, 2 * np.pi)
        pert += (strength / 3.0) * np.cos(freq * x + phase)
    a_p = a + pert
    a_p = np.maximum(a_p, 0)
    s = a_p.sum()
    if s < 1e-15:
        return a.copy()
    return a_p / s
