def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def logsumexp_obj(z, n, T_temp):
    """Compute LogSumExp objective value."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    return 2.0 * n * lse


def logsumexp_obj_grad(z, n, T_temp):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse

    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z

        a = a / a.sum()

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    rng = np.random.default_rng(777)
    lp_times = []
    n_success = 0

        if time.monotonic() - t0 > time_limit:

        lp_t0 = time.monotonic()
        b_new, M_b = solve_lp_best_response(y, solver, use_symmetry=use_symmetry)
        lp_times.append(time.monotonic() - lp_t0)

        if b_new is None:
            continue
        n_success += 1

        avg_lp_time = np.mean(lp_times[-50:]) if len(lp_times) >= 10 else 1.0

        # Nesterov dual averaging step
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        if use_symmetry:
            y = (y + y[::-1]) / 2.0
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = np.maximum(a, 0)
        if use_symmetry:
            a = (a + a[::-1]) / 2.0

        if sa < 1e-15:

            a = a / sa

def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0.copy())
    t0 = time.monotonic()
    deadline = t0 + time_limit
    n_stages = len(T_stages)
    iters_per_stage = max(10, maxiter_per_stage // n_stages)

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        def func_lse(z_cur):
            if time.monotonic() > deadline:
                return 1e15
            obj, _ = logsumexp_obj_grad(z_cur, n, stage_T)
            return obj

        def jac_lse(z_cur):
            if time.monotonic() > deadline:
                return np.zeros_like(z_cur)
            _, g = logsumexp_obj_grad(z_cur, n, stage_T)
            return g

        try:
            res = minimize(
                fun=func_lse, x0=z, jac=jac_lse, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-25,
                         'gtol': 1e-18, 'maxfun': iters_per_stage * 500, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    a_out = softmax_from_z(z)
    # Enforce symmetry
    a_out = (a_out + a_out[::-1]) / 2.0
    a_out = a_out / a_out.sum()
    return a_out


def multiplicative_subgradient(a0, time_limit, n_steps):
    """Mirror descent (multiplicative weights) targeting active peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    a = (a + a[::-1]) / 2.0
    a = a / a.sum()

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta_base = 0.08
    no_improve = 0
    scale = 1.0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * (1.0 - 2e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        eta = eta_base * scale
        log_scale = -eta * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        a_new = (a_new + a_new[::-1]) / 2.0
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 30:
            scale *= 0.8
            no_improve = 0
            if scale < 0.005:
                break

    return best_a, best_score


def structured_init(n, rng):
    """Create a structured symmetric initial sequence."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    else:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)

    a = np.maximum(a, 0)
    a = (a + a[::-1]) / 2.0
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization and symmetry."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    f_interp = (f_interp + f_interp[::-1]) / 2.0
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def perturbed_sequence(a, rng, strength=0.03):
    """Create a symmetric perturbed version of sequence a."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 7)
    phase = rng.uniform(0, 2 * np.pi)
    perturbation = strength * np.cos(freq * x + phase)
    perturbation = (perturbation + perturbation[::-1]) / 2.0
    a_pert = a + perturbation
    a_pert = np.maximum(a_pert, 0)
    a_pert = (a_pert + a_pert[::-1]) / 2.0
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses Nesterov LP with symmetry, LogSumExp L-BFGS, multiplicative subgradient,
    and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Nesterov LP on parent (long) ===
    print(f"\n=== Phase 1: Nesterov LP n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 540.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver, n_max_iters=1000000, use_symmetry=True)
        local_update(a_lp, "nesterov_lp_1")

    # === Phase 2: LogSumExp L-BFGS fine polishing ===
    print(f"\n=== Phase 2: L-BFGS LogSumExp (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.22, 380.0)
    if t_alloc > 30:
        T_stages = [10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.001]
        print(f"  T_stages ({len(T_stages)}), budget={t_alloc:.0f}s")
        a_lb = lbgfs_refine(best_seq, t_alloc, T_stages, maxiter_per_stage=15000)
        local_update(a_lb, "lbgfs_fine")

    # === Phase 3: Multiplicative subgradient ===
    print(f"\n=== Phase 3: Multi SubGrad (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 130.0)
    if t_alloc > 15:
        a_ms, _ = multiplicative_subgradient(best_seq, t_alloc, n_steps=300000)
        local_update(a_ms, "mult_subgrad")

    # === Phase 4: Second Nesterov LP round ===
    print(f"\n=== Phase 4: Nesterov LP R2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 180.0)
    if t_alloc > 30:
        a_lp2, _ = nesterov_lp_refine(best_seq, t_alloc, solver, n_max_iters=500000, use_symmetry=True)
        local_update(a_lp2, "nesterov_lp_2")

    # === Phase 5: Multi-length exploration ===
    target_lengths = [1400, 1600, 1800, 2000, 1000]
    print(f"\n=== Phase 5: Multi-length (score={best_score:.8f}) ===")
    for target_n in target_lengths:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_sub = min(rem * 0.15, 200.0)
        if t_sub < 30:
            break
        print(f"  Trying n={target_n}, budget={t_sub:.0f}s")
        init_t = interpolate_sequence(best_seq, target_n)

        # Nesterov LP on new length
        t_lp = t_sub * 0.6
        a_t, _ = nesterov_lp_refine(init_t, t_lp, solver, n_max_iters=300000, use_symmetry=True)
        local_update(a_t, f"lp_n{target_n}")

        # Quick L-BFGS polish
        rem = deadline - time.monotonic()
        if rem > 20:
            t_lb = min(rem * 0.3, 60.0)
            if t_lb > 10:
                T_short = [3.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
                a_tb = lbgfs_refine(best_seq, t_lb, T_short, maxiter_per_stage=3000)
                local_update(a_tb, f"lbgfs_n{target_n}")

    # === Phase 6: Perturbation escape + re-optimize ===
    print(f"\n=== Phase 6: Perturb Escape (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        n_cur = len(best_seq)
        for trial in range(4):
            rem_sub = deadline - time.monotonic()
            if rem_sub < 30:
                break
            strength = 0.015 + 0.01 * trial
            a_pert = perturbed_sequence(best_seq, rng, strength=strength)
            t_sub = min(rem_sub * 0.25, 80.0)
            a_opt, _ = nesterov_lp_refine(a_pert, t_sub, solver, n_max_iters=200000, use_symmetry=True)
            local_update(a_opt, f"pert{trial}")
            # Quick polish
            rem_sub = deadline - time.monotonic()
            if rem_sub > 15 and best_seq is not None:
                T_quick = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
                a_pol = lbgfs_refine(best_seq, min(rem_sub * 0.3, 30.0), T_quick, maxiter_per_stage=2000)
                local_update(a_pol, f"pert{trial}_polish")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.4, 60.0)
        if t_fin > 10:
            T_final = [2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.001, 0.0005]
            a_fin = lbgfs_refine(best_seq, t_fin, T_final, maxiter_per_stage=8000)
            local_update(a_fin, "final_lbgfs")

    # === Phase 8: Final Nesterov squeeze ===
    rem = deadline - time.monotonic()
    if rem > 20:
        t_squeeze = min(rem * 0.5, 40.0)
        if t_squeeze > 10:
            a_sq, _ = nesterov_lp_refine(best_seq, t_squeeze, solver, n_max_iters=100000, use_symmetry=True)
            local_update(a_sq, "final_lp")

    # === Phase 9: Final multiplicative ===
    rem = deadline - time.monotonic()
    if rem > 10:
        a_fm, _ = multiplicative_subgradient(best_seq, min(rem * 0.5, 20.0), n_steps=100000)
        local_update(a_fm, "final_mult")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
