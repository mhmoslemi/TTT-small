def subgradient_polish(a_init, time_limit, n_active_frac=0.05, lr=0.1, max_iters=50000):
    """
    Subgradient method for polishing the solution.
    Minimizes max_k (a*a)_k using subgradient of the max function.
    """
    n = len(a_init)
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_score_val = _score_normalized(a)
    t0 = time.monotonic()
    
    # Mirror map: use softmax parameterization
    z = np.log(np.maximum(a, 1e-300)).copy()
    
    for t in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute subgradient of max_k (a*a)_k
        b = fast_convolve(a)
        max_b = b.max()
        if max_b < 1e-300:
            break
        
        # Active set
        threshold = max_b * (1.0 - n_active_frac)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        # Subgradient
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                g[j] += a[k - j]
        g /= len(active)
        
        # Mirror descent step
        alpha = lr / (1.0 + 0.01 * t)
        z_new = z - alpha * g
        a_new = np.exp(z_new - z_new.max())
        s = a_new.sum()
        if s < 1e-300:
            break
        a_new = a_new / s
        
        sc = _score_normalized(a_new)
        if sc < best_score_val - 1e-12:
            best_score_val = sc
            best_a = a_new.copy()
            z = z_new
            a = a_new
        else:
            # Halve learning rate if no improvement
            lr = max(lr * 0.95, 0.001)
    
    return best_a, best_score_val

            best_a = y.copy()

        if it % 50 == 0:
            if best_sc < prev_best - 1e-9:
                stall = 0
            else:
                stall += 1
            prev_best = best_sc
            if stall > 40:
                break

        if verbose and it % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"        NFW iter={it} ({elapsed:.0f}s): score={best_sc:.6f}")

    return best_a, best_sc


def basic_fw(a_init, time_limit, solver, max_iters=200000):
    """Basic Frank-Wolfe (non-accelerated) - more robust."""
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_sc = score_normalized(a)
    t0 = time.monotonic()
    stall = 0
    prev_best = best_sc

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = solve_lp_best_response(a, solver)
        if b is None:
            break

        a = normalize(b)
        sc = score_normalized(a)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()

        if it % 30 == 0:
            if best_sc < prev_best - 1e-9:
                stall = 0
            else:
                stall += 1
            prev_best = best_sc
            if stall > 15:
                break

    return best_a, best_sc


def mirror_descend_polish(a_init, time_limit, n_active_frac=0.05, verbose=False):
    """Mirror descent with subgradient for polishing."""
    a = normalize(a_init.copy())
    z = np.log(np.maximum(a, 1e-300)).copy()
    best_a = a.copy()
    best_sc = score_normalized(a)
    t0 = time.monotonic()
    lr = 0.5
    no_improve = 0

    for it in range(200000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        if max_b < 1e-300:
            break

        threshold = max_b * (1.0 - n_active_frac)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        g = np.zeros(len(a))
        n = len(a)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                g[j] += a[k - j]
        g /= len(active)

        z_new = z - lr * g
        exp_z = np.exp(z_new - z_new.max())
        s = exp_z.sum()
        if s < 1e-300:
            break
        a_new = exp_z / s

        sc = score_normalized(a_new)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a_new.copy()
            z = z_new
            a = a_new
            no_improve = 0
            lr = min(lr * 1.02, 2.0)
        else:
            no_improve += 1
            lr = max(lr * 0.97, 0.01)
            if no_improve > 500:
                rng = np.random.default_rng(it * 7 + 3)
                noise = rng.random(len(a)) * 0.05
                a = normalize(a + noise)
                z = np.log(np.maximum(a, 1e-300))
                no_improve = 0
                lr = 0.5

        if verbose and it % 50000 == 0 and it > 0:
            elapsed = time.monotonic() - t0
            print(f"        MD iter={it} ({elapsed:.0f}s): score={best_sc:.6f}")

    return best_a, best_sc


def init_uniform(n):
    return np.ones(n) / n


def init_cosine(n):
    x = np.linspace(0, 1, n)
    a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    return normalize(a)


def init_edge_conc(n, p=2.0):
    x = np.linspace(0, 1, n)
    a = x ** p + (1 - x) ** p
    return normalize(a)


def init_bimodal(n, sigma=0.15):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.2) / sigma) ** 2) + np.exp(-((x - 0.8) / sigma) ** 2)
    return normalize(a)


def init_central(n, sigma=0.2):
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.5) / sigma) ** 2)
    return normalize(a)


def init_ruler(n):
    """2-adic valuation based (dyadic) initialization."""
    a = np.ones(n)
    for j in range(n):
        v = 0
        m = j + 1
        while m % 2 == 0 and m > 0:
            v += 1
            m //= 2
        a[j] = 1.0 / (v + 1)
    return normalize(a)


def init_random(n, seed):
    rng = np.random.default_rng(seed)
    a = rng.random(n)
    return normalize(a)


def init_gaussian_multi(n, seed, n_peaks=5):
    rng = np.random.default_rng(seed)
    a = np.zeros(n)
    centers = rng.uniform(0, 1, n_peaks)
    sigmas = rng.uniform(0.05, 0.2, n_peaks)
    weights = rng.uniform(0.5, 1.0, n_peaks)
    x = np.linspace(0, 1, n)
    for c, s, w in zip(centers, sigmas, weights):
        a += w * np.exp(-((x - c) / s) ** 2)
    return normalize(a)


def init_parent_interpolated(parent, n):
    if parent is None:
        return init_uniform(n)
    old_n = len(parent)
    if old_n == n:
        return normalize(parent)
    p = normalize(parent)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, n)
    p_interp = np.interp(new_x, old_x, p)
    return normalize(p_interp)


def perturb(a, rng, strength=0.05):
    """Random perturbation to escape local minima."""
    a_trial = a.copy()
    k = max(2, len(a) // 5)
    idx = rng.choice(len(a), k, replace=False)
    a_trial[idx] += rng.normal(0, strength, k)
    return normalize(a_trial)


def tournament_select(candidates, scores, n_select=5):
    """Select top-n_select candidates by score."""
    order = np.argsort(scores)
    indices = order[:min(n_select, len(candidates))]
    return [candidates[i] for i in indices], [scores[i] for i in indices]


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function."""
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    t_deadline = t_global_start + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    def time_left():
        return max(0.0, t_deadline - time.monotonic())

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  GLOBAL BEST [{label}]: {sc:.6f} (n={len(a)}, t={time.monotonic()-t_global_start:.0f}s)")

    # Try to get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = normalize(p)
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    n_base = len(best_seq) if best_seq is not None else 1200
    print(f"Base length: {n_base}, initial best: {best_score:.6f}")

    # =========================================================
    # PHASE 1: Tournament of diverse initializations
    # =========================================================
    print(f"\n=== Phase 1: Tournament (n={n_base}) ===")
    t_alloc = min(time_left() * 0.25, 400.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")

        # Generate diverse initializations
        inits = []
        # From parent (if available)
        if parent is not None:
            inits.append(("parent", init_parent_interpolated(parent, n_base)))
        # Structured
        inits.append(("uniform", init_uniform(n_base)))
        inits.append(("cosine", init_cosine(n_base)))
        inits.append(("edge2", init_edge_conc(n_base, 2.0)))
        inits.append(("edge4", init_edge_conc(n_base, 4.0)))
        inits.append(("bimodal", init_bimodal(n_base)))
        inits.append(("central", init_central(n_base)))
        inits.append(("ruler", init_ruler(n_base)))
        # Random
        for i in range(6):
            inits.append((f"rand{i}", init_random(n_base, seed + 100 + i * 17)))
        # Multi-Gaussian
        for i in range(4):
            inits.append((f"gauss{i}", init_gaussian_multi(n_base, seed + 500 + i * 31)))

        # Quick evaluation and tournament
        per_init_time = t_alloc / len(inits)
        candidates = []
        cand_scores = []

        for label, a_init in inits:
            if time_left() < 10:
                break
            t_sub = min(per_init_time, time_left() - 5)
            if t_sub < 2:
                break
            # Quick Nesterov FW
            a_out, sc_out = nesterov_fw(a_init, t_sub, solver, max_iters=5000)
            eval_sc = evaluate_safe(a_out)
            if np.isfinite(eval_sc):
                candidates.append(a_out)
                cand_scores.append(eval_sc)
                print(f"    {label}: {eval_sc:.6f}")
            else:
                candidates.append(a_init)
                cand_scores.append(score_normalized(a_init))

        # Select top candidates
        if candidates:
            sel_cands, sel_scores = tournament_select(candidates, cand_scores, n_select=5)
            # Update global best from any candidate
            for c in candidates:
                try_update(c, "tournament")

            # Deep optimize the top candidates
            deep_budget_each = min(time_left() * 0.4 / max(1, len(sel_cands)), 200.0)
            for i, (a_c, sc_c) in enumerate(zip(sel_cands, sel_scores)):
                if time_left() < 60:
                    break
                t_sub = min(deep_budget_each, time_left() - 30)
                if t_sub < 20:
                    break
                print(f"  Deep optim candidate {i} (n={n_base}, init_score={sc_c:.6f}), budget={t_sub:.0f}s")
                a_deep, _ = nesterov_fw(a_c, t_sub, solver, max_iters=100000)
                try_update(a_deep, f"deep_{i}")

    # =========================================================
    # PHASE 2: Extended Nesterov FW on current best
    # =========================================================
    print(f"\n=== Phase 2: Extended FW (best={best_score:.6f}) ===")
    if best_seq is not None and time_left() > 120:
        t_alloc2 = min(time_left() * 0.30, 500.0)
        print(f"  Budget: {t_alloc2:.0f}s")
        a_ext, _ = nesterov_fw(best_seq, t_alloc2, solver, max_iters=200000, verbose=True)
        try_update(a_ext, "ext_fw")

    # =========================================================
    # PHASE 3: Mirror descent polishing
    # =========================================================
    print(f"\n=== Phase 3: Mirror Descent Polish (best={best_score:.6f}) ===")
    if best_seq is not None and time_left() > 60:
        t_alloc3 = min(time_left() * 0.15, 200.0)
        print(f"  Budget: {t_alloc3:.0f}s")
        for n_act in [0.03, 0.08, 0.15]:
            if time_left() < 20:
                break
            t_sub = min(t_alloc3 / 3, time_left() - 10)
            if t_sub < 10:
                break
            a_md, _ = mirror_descend_polish(best_seq, t_sub, n_active_frac=n_act)
            try_update(a_md, f"md_{n_act}")

    # =========================================================
    # PHASE 4: Perturbation + re-optimize (escape local minima)
    # =========================================================
    print(f"\n=== Phase 4: Perturbation Escapes (best={best_score:.6f}) ===")
    if best_seq is not None and time_left() > 90:
        t_alloc4 = min(time_left() * 0.15, 250.0)
        print(f"  Budget: {t_alloc4:.0f}s")
        n_perturb = max(3, int(t_alloc4 / 40))
        per_perturb = t_alloc4 / n_perturb

        for i in range(n_perturb):
            if time_left() < 30:
                break
            t_sub = min(per_perturb, time_left() - 15)
            if t_sub < 10:
                break
            # Generate perturbation
            a_pert = perturb(best_seq, rng, strength=0.03 + 0.02 * i)
            # Optimize the perturbed point
            a_out, _ = nesterov_fw(a_pert, t_sub, solver, max_iters=50000)
            try_update(a_out, f"perturb_{i}")

    # =========================================================
    # PHASE 5: Try different lengths
    # =========================================================
    print(f"\n=== Phase 5: Length Exploration (best={best_score:.6f}) ===")
    if best_seq is not None and time_left() > 120:
        t_alloc5 = min(time_left() * 0.20, 300.0)
        print(f"  Budget: {t_alloc5:.0f}s")

        lengths_to_try = []
        for mult in [0.75, 1.25, 1.5, 2.0]:
            ln = int(n_base * mult)
            if 200 <= ln <= 4000 and ln != n_base:
                lengths_to_try.append(ln)
        lengths_to_try = sorted(set(lengths_to_try))[:3]

        for ln in lengths_to_try:
            if time_left() < 60:
                break
            t_sub = min(t_alloc5 / max(1, len(lengths_to_try)), time_left() - 30)
            if t_sub < 30:
                break
            print(f"  Trying n={ln}, budget={t_sub:.0f}s")
            a_init_ln = init_parent_interpolated(best_seq, ln)
            a_out, _ = nesterov_fw(a_init_ln, t_sub, solver, max_iters=50000)
            try_update(a_out, f"len_{ln}")

    # =========================================================
    # PHASE 6: Final polishing
    # =========================================================
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}, n={len(best_seq) if best_seq is not None else 0}) ===")
    if best_seq is not None and time_left() > 20:
        # One more FW pass
        t_sub = min(time_left() * 0.4, 120.0)
        if t_sub > 10:
            a_fin, _ = nesterov_fw(best_seq, t_sub, solver, max_iters=50000)
            try_update(a_fin, "final_fw")

        # Mirror descent
        if time_left() > 15:
            t_sub2 = min(time_left() * 0.3, 60.0)
            if t_sub2 > 5:
                a_fin2, _ = mirror_descend_polish(best_seq, t_sub2, n_active_frac=0.03)
                try_update(a_fin2, "final_md")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    total_time = time.monotonic() - t_global_start
    print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq) if best_seq is not None else 0}, "
          f"time={total_time:.0f}s")

    if best_seq is not None:
        return [float(x) for x in best_seq]
    else:
        return [1.0] * 1200
