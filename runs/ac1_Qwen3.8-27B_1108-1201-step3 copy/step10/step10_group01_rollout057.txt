    """Solve min_x max_k (b_base[k] + lin[k]*x + quad[k]*x^2) s.t. x >= 0."""

        return np.max(b_base + lin*x + quad*x*x)
    
    def subgrad(x):
        vals = b_base + lin*x + quad*x*x
        gval = vals.max()
        active = np.where(vals >= gval - 1e-15)[0]
        return np.sum(lin[active] + 2*quad[active]*x)
    
    # Check if x=0 is optimal
    if subgrad(0) >= 0:
        return 0.0
    
    # Bisection
    for _ in range(60):
        x_mid = (x_lo + x_hi) / 2
        sg = subgrad(x_mid)
        if sg >= 0:
            x_hi = x_mid
        else:
            x_lo = x_mid
    
    return (x_lo + x_hi) / 2
