    """Gerchberg-Saxton style refinement."""
    n = len(a0)
    fft_size = 1 << (2*n-1).bit_length()
    a = np.maximum(a0, 0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = evaluate_local(a)
    
    for it in range(n_iters):
        # Time domain: project to non-negative simplex
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            a = a0.copy()
            s = a0.sum()
        a = a / s
        
        # Frequency domain: flatten
        A = rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Compress magnitude
        mag_new = mag ** gamma  # gamma < 1 compresses
        # Or: mag_new = mag / (mag + mag.mean())
        
        A_new = mag_new * np.exp(1j * phase)
        a = irfft(A_new, fft_size)[:n]
        
        # Check improvement
        score = evaluate_local(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score

def fast_convolve(a):

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

class LPMatvec:
    """Fast matrix-vector product for T @ b where T is the convolution matrix."""
    def __init__(self, a_fixed):

        self.fft_size = next_pow2(2 * self.n - 1)

def gp_refine(a0, n_iters, gamma=0.7, time_limit=60):
    """
    Gerchberg-Packsatz (GS) refinement:
    Alternates between time-domain projection and frequency-domain flattening.
    """
    n = len(a0)
    fft_size = next_pow2(2 * n - 1)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Frequency domain: flatten magnitude
        A = rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Compress: new_mag = mag^gamma (for gamma < 1, peaks are reduced)
        mag_new = mag ** gamma
        # Renormalize DC component to preserve sum
        dc_old = mag[0]
        dc_new = mag_new[0]
        if dc_old > 1e-15:
            scale = 1.0 / (dc_new / dc_old)
            mag_new *= scale
        
        A_new = mag_new * np.exp(1j * phase)
        a = irfft(A_new, fft_size)[:n]
        
        # Time domain: project to simplex
        a = project_simplex(a)
        
        # Check
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score


def coarse_to_fine(a_parent, target_n, time_limit, T):
    """
    Coarse-to-fine: downsample parent, optimize at each level, upsample.
    """
    n = len(a_parent)
    
    # Create a sequence of sizes from small to target_n
    sizes = []
    s = 100
    while s < target_n:
        sizes.append(s)
        s *= 2
    sizes.append(target_n)
    
    # Start from parent, downsample to smallest
    current_n = n
    a = a_parent.copy()
    
    # Downsample to sizes[0]
    while current_n > sizes[0]:
        target = max(50, current_n // 2)
        a = interpolate_to_length(a, target)
        current_n = target
    
    # Refine at each scale
    t0 = time.monotonic()
    for target_n in sizes:
        if time.monotonic() - t0 > time_limit:
            break
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 5:
            break
        
        # Upsample to target_n if needed
        if len(a) < target_n:
            a = interpolate_to_length(a, target_n)
        elif len(a) > target_n:
            a = interpolate_to_length(a, target_n)
        
        # Optimize at this scale

        a, score = nesterov_pg(a, n_it_time, T, max_iters=10000)
        
        elapsed = time.monotonic() - t0
        print(f"    CoarseToFine n={len(a)} ({elapsed:.0f}s): score={score:.8f}")
    
    return a


def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0)
    ss = a_interp.sum()
    if ss > 1e-15:
        a_interp = a_interp / ss
    return a_interp
