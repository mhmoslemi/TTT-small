def fast_convolve(a):

def golden_section_search(f, lo, hi, max_iter=35):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-14:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def basin_hopping(a0, time_limit, rng, verbose=False):

    n = len(a0)

            break

        pert = a.copy()

        if ptype == 0:

            conv_p = fast_convolve(a)
            top_k = min(8, 2 * n - 1)
            peaks = np.argpartition(conv_p, -top_k)[-top_k:]
            for k in peaks:
                k = int(k)
                i_start = max(0, k - n + 1)
                i_end = min(n - 1, k)
                idx = np.arange(i_start, i_end + 1)

                picks = rng.choice(idx, size=min(n_red, len(idx)), replace=False)
                for ii in picks:
                    pert[ii] *= (1.0 - 0.03 * rng.random())

            pert = a + rng.standard_normal(n) * sigma
        elif ptype == 2:

            i1 = rng.integers(0, n - blen)

                temp = pert[i1:i1+blen].copy()
                pert[i1:i1+blen] = pert[i2:i2+blen]
                pert[i2:i2+blen] = temp

            # Frequency perturbation
            A = rfft(pert, 2 * n)
            freq_mask = rng.random(n) < 0.1
            A[1:n] *= (1.0 + 0.1 * rng.standard_normal(np.sum(freq_mask)) * freq_mask[1:n] if np.sum(freq_mask) > 1 else 1.0)
            pert = irfft(A, 2 * n)[:n]
        else:
            # Mirror mixing
            shift = rng.integers(1, max(2, n // 5))
            shifted = np.roll(a, shift)
            mix = 0.6 + 0.3 * rng.random()
            pert = mix * a + (1 - mix) * shifted

        pert = np.maximum(pert, 0)
        ps = pert.sum()
        if ps < 1e-10:
            continue
        pert = pert / ps

        a_opt, s_opt, _ = exact_coord_descent(pert, t_alloc, label=f"bh_{count}")
        if s_opt < best_score - 1e-10:
            best_score = s_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"    BH {count} (type={ptype}): {best_score:.8f}")

    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def init_asymmetric(n, rng, shift=0.3):
    t = np.arange(n, dtype=np.float64)
    center = n * shift
    width = n * rng.uniform(0.15, 0.35)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_multi_peak(n, rng, n_peaks=3):
    """Initialize with multiple Gaussian peaks."""
    a = np.zeros(n)
    for _ in range(n_peaks):
        center = rng.uniform(0, n)
        width = rng.uniform(n * 0.05, n * 0.2)
        amp = rng.uniform(0.5, 1.5)
        t = np.arange(n, dtype=np.float64)
        a += amp * np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_fft_based(n, rng):
    """Initialize using flat-ish spectrum approach."""
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    # Create spectrum that's somewhat flat but with random phases
    amps = rng.uniform(0.8, 1.2, n_freq)
    amps[0] = 3.0  # Strong DC component for non-negativity
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)[:n]
    a = np.abs(a)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def block_coord_descent(a0, time_limit, verbose=False, label=""):
    """Block coordinate descent: update pairs of coordinates jointly."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s
    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        # Process coordinates in pairs
        for i in range(0, n - 1, 2):
            if time.monotonic() - t0 > time_limit:
                break
            j = i + 1
            if a[i] < 1e-14 and a[j] < 1e-14:
                continue
            vi, vj = a[i], a[j]

            # Affected convolution positions for i
            k_lo_i, k_hi_i = i, min(i + n - 1, 2 * n - 2)
            # Affected for j
            k_lo_j, k_hi_j = j, min(j + n - 1, 2 * n - 2)
            k_lo = min(k_lo_i, k_lo_j)
            k_hi = max(k_hi_i, k_hi_j)

            # Unaffected max
            unaff_max = 0.0
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            # Build the affected convolution values as function of (xi, xj)
            # This is complex for block updates, so let's do a 2D search
            # Sample a grid and then refine
            best_xi, best_xj = vi, vj
            best_local = 2.0 * n * max(unaff_max, float(conv[k_lo:k_hi+1].max()))

            # Coarse grid
            for fi in [0.5, 0.75, 1.0, 1.25, 1.5, 2.0]:
                xi = vi * fi
                if xi < 0:
                    continue
                for fj in [0.5, 0.75, 1.0, 1.25, 1.5, 2.0]:
                    xj = vj * fj
                    if xj < 0:
                        continue
                    # Compute new affected conv values
                    new_conv = conv[k_lo:k_hi+1].copy()
                    # Update for i
                    k_lo_i_local = k_lo_i - k_lo
                    k_hi_i_local = k_hi_i - k_lo
                    partners_i = np.arange(k_lo_i, k_hi_i + 1) - i
                    delta_i = xi - vi
                    for idx in range(k_hi_i_local - k_lo_i_local + 1):
                        k = k_lo_i + idx
                        if k == 2 * i:
                            new_conv[idx] += delta_i * (2.0 * vi + delta_i)
                        else:
                            p = k - i
                            if 0 <= p < n:
                                new_conv[idx] += 2.0 * delta_i * a[p]
                    # Update for j
                    k_lo_j_local = k_lo_j - k_lo
                    k_hi_j_local = k_hi_j - k_lo
                    delta_j = xj - vj
                    for idx in range(k_hi_j_local - k_lo_j_local + 1):
                        k = k_lo_j + idx
                        if k == 2 * j:
                            new_conv[idx] += delta_j * (2.0 * vj + delta_j)
                        else:
                            p = k - j
                            if 0 <= p < n:
                                new_conv[idx] += 2.0 * delta_j * a[p]

                    ns = current_sum + delta_i + delta_j
                    if ns < 0.01:
                        continue
                    lm = max(unaff_max, float(new_conv.max()))
                    sc = 2.0 * n * lm / (ns * ns)
                    if sc < best_local - 1e-14:
                        best_local = sc
                        best_xi, best_xj = xi, xj

            if abs(best_xi - vi) > 1e-12 or abs(best_xj - vj) > 1e-12:
                # Apply update
                delta_i = best_xi - vi
                delta_j = best_xj - vj
                a[i] = best_xi
                a[j] = best_xj
                current_sum += delta_i + delta_j

                # Update conv
                for k in range(k_lo_i, k_hi_i + 1):
                    p = k - i
                    if 0 <= p < n:
                        if k == 2 * i:
                            conv[k] += delta_i * (2.0 * vi + delta_i)
                        else:
                            conv[k] += 2.0 * delta_i * a[p]
                for k in range(k_lo_j, k_hi_j + 1):
                    p = k - j
                    if 0 <= p < n:
                        if k == 2 * j:
                            conv[k] += delta_j * (2.0 * vj + delta_j)
                        else:
                            conv[k] += 2.0 * delta_j * a[p]

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
        if verbose and sweeps % 5 == 0:
            print(f"    BCD {label} sweep={sweeps}: best={best_score:.8f}")
    return best_a, best_score

def propose_candidate():
    """
    Multi-phase optimization combining LP Fictitious Play, smooth-max gradient
    descent, exact coordinate descent, block coordinate descent, and basin hopping.
    Explores multiple sequence lengths and initialization strategies.
    """
    deadline = time.monotonic() + 1780.0
    rng = np.random.default_rng(42)
    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception as e:
        print(f"Parent load failed: {e}")

    # Phase 1: Diverse initialization and quick screening
    print("\n=== Phase 1: Diverse Init + Quick CD ===")
    t1_start = time.monotonic()
    n_parent = len(parent) if parent is not None else 1152

    n_values = sorted(set([n_parent, 1024, 1280, 768, 1536, 896]))
    n_values = [n for n in n_values if 256 <= n <= 2560]

    candidates = []
    for n in n_values[:5]:
        if time.monotonic() - t1_start > 100:
            break
        inits = []
        if parent is not None:
            inits.append(interpolate_seq(parent, n))
        inits.append(init_asymmetric(n, rng, 0.25))
        inits.append(init_asymmetric(n, rng, 0.6))
        inits.append(init_multi_peak(n, rng, 3))
        inits.append(init_multi_peak(n, rng, 5))
        inits.append(init_fft_based(n, rng))
        inits.append(np.ones(n) / n)

        for j, cand in enumerate(inits[:7]):
            t_left = 100.0 - (time.monotonic() - t1_start)
            if t_left < 3:
                break
            t_alloc = max(1.0, min(8.0, t_left / 8.0))
            a_opt, s_opt, _ = exact_coord_descent(cand, t_alloc, label=f"p1_{n}_{j}")
            candidates.append((s_opt, a_opt, n))

    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 1: {len(candidates)} candidates, best={candidates[0][0]:.8f}")
    for s_c, a_c, n_c in candidates[:4]:
        sc = float(evaluate_sequence(a_c.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_c.copy()
            print(f"  NEW BEST p1: {sc:.8f} (n={n_c})")

    n_work = len(best_seq)
    print(f"  Working with n={n_work}, best={best_score:.8f}")

    # Phase 2: LP Fictitious Play (main optimization)
    print(f"\n=== Phase 2: LP Fictitious Play (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 600
    if rem > 30:
        t_fp = min(rem * 0.6, 500.0)
        print(f"  FP budget={t_fp:.0f}s")
        a_fp, s_fp, iters = lp_fictitious_play(best_seq, t_fp, solver=solver, verbose=True)
        sc = float(evaluate_sequence(a_fp.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_fp.copy()
            print(f"  NEW BEST FP: {sc:.8f}")

    # Phase 3: Try different lengths with FP
    print(f"\n=== Phase 3: Multi-length FP (best={best_score:.8f}) ===")
    for alt_n in [n_work + 160, n_work - 160, n_work + 320]:
        if alt_n < 300 or alt_n > 2500:
            continue
        rem = deadline - time.monotonic() - 400
        if rem < 30:
            break
        t_fp2 = min(rem * 0.25, 120.0)
        a_init = interpolate_seq(best_seq, alt_n)
        a_fp2, s_fp2, _ = lp_fictitious_play(a_init, t_fp2, solver=solver)
        sc2 = float(evaluate_sequence(a_fp2.tolist()))
        if np.isfinite(sc2) and sc2 < best_score:
            best_score = sc2
            best_seq = a_fp2.copy()
            n_work = alt_n
            print(f"  NEW BEST FP n={alt_n}: {sc2:.8f}")

    # Phase 4: Smooth-max gradient descent
    print(f"\n=== Phase 4: Smooth-Max Descent (best={best_score:.8f}) ===")
    n_work = len(best_seq)
    rem = deadline - time.monotonic() - 300
    if rem > 20:
        t_smd = min(rem * 0.4, 120.0)
        print(f"  SMD budget={t_smd:.0f}s")
        a_smd, s_smd = smooth_max_descent(best_seq, t_smd, verbose=True, label="ph4")
        sc = float(evaluate_sequence(a_smd.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_smd.copy()
            print(f"  NEW BEST SMD: {sc:.8f}")

    # Phase 5: Exact coordinate descent polishing
    print(f"\n=== Phase 5: ECD Polishing (best={best_score:.8f}) ===")
    n_work = len(best_seq)
    for rnd in range(4):
        rem = deadline - time.monotonic() - 250
        if rem < 15:
            break
        t_cd = min(rem * 0.2, 100.0)
        print(f"  ECD round {rnd+1}, budget={t_cd:.0f}s")
        a_cd, s_cd, sw = exact_coord_descent(best_seq, t_cd, verbose=True, label=f"ph5_r{rnd}")
        sc = float(evaluate_sequence(a_cd.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_cd.copy()
            print(f"  NEW BEST ECD: {sc:.8f}")
        else:
            # Perturb and retry
            perturbed = best_seq + rng.standard_normal(n_work) * 0.015
            perturbed = np.maximum(perturbed, 0)
            perturbed = perturbed / perturbed.sum()
            a_cd2, s_cd2, _ = exact_coord_descent(perturbed, min(t_cd * 0.5, 50.0))
            sc2 = float(evaluate_sequence(a_cd2.tolist()))
            if np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_seq = a_cd2.copy()
                print(f"  NEW BEST ECD-pert: {sc2:.8f}")

    # Phase 6: Block coordinate descent
    print(f"\n=== Phase 6: Block CD (best={best_score:.8f}) ===")
    n_work = len(best_seq)
    rem = deadline - time.monotonic() - 200
    if rem > 20:
        t_bcd = min(rem * 0.4, 80.0)
        print(f"  BCD budget={t_bcd:.0f}s")
        a_bcd, s_bcd, _ = block_coord_descent(best_seq, t_bcd, verbose=True, label="ph6")
        sc = float(evaluate_sequence(a_bcd.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_bcd.copy()
            print(f"  NEW BEST BCD: {sc:.8f}")

    # Phase 7: Basin hopping
    print(f"\n=== Phase 7: Basin Hopping (best={best_score:.8f}) ===")
    n_work = len(best_seq)
    rem = deadline - time.monotonic() - 100
    if rem > 20:
        t_bh = min(rem * 0.6, 200.0)
        print(f"  BH budget={t_bh:.0f}s")
        a_bh, s_bh = basin_hopping(best_seq, t_bh, rng, verbose=True)
        sc = float(evaluate_sequence(a_bh.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_bh.copy()
            print(f"  NEW BEST BH: {sc:.8f}")

    # Phase 8: Final polishing
    print(f"\n=== Phase 8: Final Polish (best={best_score:.8f}) ===")
    n_work = len(best_seq)
    rem = deadline - time.monotonic() - 30
    if rem > 10:
        t_last = min(rem * 0.5, 60.0)
        a_last, s_last, _ = exact_coord_descent(best_seq, t_last, label="final_cd")
        sc = float(evaluate_sequence(a_last.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_last.copy()
            print(f"  NEW BEST final: {sc:.8f}")

    rem = deadline - time.monotonic() - 10
    if rem > 5:
        a_fin, s_fin, _ = exact_coord_descent(best_seq, min(rem * 0.7, 20.0), label="fin")
        sc = float(evaluate_sequence(a_fin.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_fin.copy()
            print(f"  NEW BEST fin: {sc:.8f}")

    if best_seq is None:
        best_seq = np.ones(512) / 512

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
