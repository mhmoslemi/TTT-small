_lp_solver = None

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    import cvxpy as cp
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.GLOP, cp.PDLP, cp.SCS, cp.GLPK, cp.ECOS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS

def solve_lp_best_response(a_fixed, solver=None):
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.GLPK, cp.SCS, cp.PDLP, cp.GLOP]:

                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None, None
    b_val = b_val / s
    return b_val, float(M_var.value), prob

def nesterov_lp_fp(a0, time_limit, solver=None, label=""):
    """Nesterov-accelerated LP fictitious play with Polyak averaging."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    iters = 0
    y = a.copy()  # Nesterov virtual point

    # Polyak average
    polyak_a = a.copy()
    total_weight = 0.0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        t = time.monotonic() - t0

        # Nesterov extrapolation
        gamma_it = 1.0 / np.sqrt(iters + 1.0)
        y_new = (1.0 + gamma_it) * a - gamma_it * y
        y_new = np.maximum(y_new, 0)
        y_s = y_new.sum()
        if y_s < 1e-15:
            y_new = a.copy()
        else:
            y_new = y_new / y_s
        y = y_new

        # Solve LP best response against y
        b_new, M_b, _ = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        # Adaptive step size
        alpha = 2.0 / (iters + 2.0)

        # Update
        a_new = (1 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        # Polyak averaging
        total_weight += 1.0
        w = 1.0 / total_weight
        polyak_a = (1.0 - w) * polyak_a + w * a_new
        polyak_a = np.maximum(polyak_a, 0)
        ps = polyak_a.sum()
        if ps > 1e-15:
            polyak_a = polyak_a / ps

        # Evaluate all candidates
        for cand, cl in [(a_new, 'a'), (b_new, 'b'), (polyak_a, 'p')]:
            if cand is None:
                continue
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-15:
                continue
            cand = cand / cs
            score = 2.0 * n * fast_convolve(cand).max()
            if score < best_score - 1e-14:
                best_score = score
                best_a = cand.copy()

        a = a_new

        if iters % 10 == 0:
            print(f"    NLPF {label} it={iters} ({t:.0f}s): best={best_score:.8f}, M_b={M_b:.10f}, alpha={alpha:.4f}")

    return best_a, best_score, iters

def interleaved_lp_cd(a0, time_limit, solver=None, label=""):
    """Interleaved LP and coordinate descent."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    iters = 0
    y = a.copy()

    while time.monotonic() - t0 < time_limit:
        iters += 1
        t = time.monotonic() - t0

        # Every 3 iterations, do an LP step
        if iters % 3 == 1:
            gamma_it = 1.0 / np.sqrt(iters + 1.0)
            y_new = (1.0 + gamma_it) * a - gamma_it * y
            y_new = np.maximum(y_new, 0)
            y_s = y_new.sum()
            if y_s < 1e-15:
                y_new = a.copy()
            else:
                y_new = y_new / y_s
            y = y_new

            b_new, M_b, _ = solve_lp_best_response(y, solver)
            if b_new is not None:
                alpha = 2.0 / (iters + 2.0)
                a_new = (1 - alpha) * a + alpha * b_new
                a_new = np.maximum(a_new, 0)
                s_new = a_new.sum()
                if s_new > 1e-15:
                    a = a_new / s_new

        # CD step (short budget)
        cd_time = min(5.0, (time_limit - (time.monotonic() - t0)) * 0.3)
        if cd_time > 1.0:
            a_cd, s_cd, _ = enhanced_coord_descent(a, cd_time, top_k=8, label=f"{label}_cd{iters}")
            if s_cd < best_score - 1e-14:
                best_score = s_cd
                best_a = a_cd.copy()
            a = a_cd

        # Evaluate current
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()

        if iters % 5 == 0:
            print(f"    ILP {label} it={iters} ({t:.0f}s): best={best_score:.8f}")

    return best_a, best_score, iters

def fine_cd(a0, time_limit, label=""):
    """Very fine coordinate descent for final polishing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            ai = a[i]
            if ai < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_max = float(conv[:k_start].max())
            else:
                unaff_max = -1e30
            if k_end < 2 * n - 2:
                ur = float(conv[k_end + 1:].max())
                unaff_max = max(unaff_max, ur)

            affected = conv[k_start:k_end + 1]
            coeffs = a[(k_start - i):(k_end - i + 1)] + ai
            has_quad = (2 * i >= k_start) and (2 * i <= k_end)
            quad_idx = 2 * i - k_start if has_quad else -1

            best_delta = 0.0
            best_new_max = M

            # Fine grid around current value
            for factor in [0.9999, 0.9998, 0.9995, 0.999, 0.998, 0.997, 0.995,
                          0.99, 0.985, 0.98, 0.97, 0.95,
                          1.0001, 1.0002, 1.0005, 1.001, 1.002, 1.003, 1.005,
                          1.01, 1.015, 1.02, 1.03, 1.05]:
                delta = ai * (factor - 1.0)
                if delta < -ai:
                    continue
                new_affected = affected + delta * coeffs
                if has_quad:
                    nm = max(float(new_affected.max()), float(new_affected[quad_idx] + delta * delta))
                else:
                    nm = float(new_affected.max())
                new_max = max(unaff_max, nm)
                if new_max < best_new_max - 1e-16:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-15:
                a[i] = max(ai + best_delta, 0.0)
                conv[k_start:k_end + 1] += best_delta * coeffs
                if has_quad:
                    conv[2 * i] += best_delta * best_delta
                M = best_new_max
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved and sweeps > 20:
            break

        if sweeps % 10 == 0:
            conv = fast_convolve(a)
            M = float(conv.max())

    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def spectral_interp(f, new_n):
    """Interpolate via FFT for better spectral properties."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s

    # Zero-pad FFT of old sequence to new length
    fft_size = 1 << (2 * max(old_n, new_n) - 1).bit_length()
    F = rfft(f_norm, fft_size)

    # Truncate to appropriate number of frequency bins
    n_bins = new_n // 2 + 1
    F_new = F[:n_bins]
    if old_n > new_n:
        # Slightly smooth in frequency domain
        F_new[-min(10, n_bins // 10):] *= 0.9

    f_new = irfft(F_new, 2 * n_bins - 1)[:new_n]
    f_new = np.maximum(f_new, 0)
    fs = f_new.sum()
    if fs > 1e-15:
        f_new = f_new / fs
    else:
        f_new = np.interp(np.linspace(0, 1, new_n), np.linspace(0, 1, old_n), f_norm)
        f_new = np.maximum(f_new, 0)
        fs = f_new.sum()
        if fs > 1e-15:
            f_new = f_new / fs
    return f_new

def check_update(a, best_seq, best_score, label=""):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
    return best_seq, best_score

def perturb_escape(a0, rng, mode=0):
    """Perturb to escape local optima."""
    n = len(a0)
    a = a0.copy()

    if mode == 0:
        # Peak-targeted reduction
        conv = fast_convolve(a)
        M = conv.max()
        threshold = M * 0.85
        peak_pos = np.where(conv >= threshold)[0]
        pressure = np.zeros(n)
        for k in peak_pos:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            pressure[j] += a[j] * a[k - j]
        p_max = pressure.max()
        if p_max > 1e-20:
            pressure /= p_max
        noise = rng.standard_normal(n) * 0.02
        a_new = a * (1.0 - 0.08 * pressure + noise)
        a_new = np.maximum(a_new, 0)
    elif mode == 1:
        # Random mask
        frac = 0.15
        mask = rng.random(n) > frac
        n_keep = max(10, mask.sum())
        if n_keep < n // 4:
            mask = np.ones(n, dtype=bool)
        a_new = a * mask
        a_new[mask] *= (1.0 + rng.standard_normal(mask.sum()) * 0.2)
        a_new = np.maximum(a_new, 0)
    else:
        # Gaussian smoothing
        kernel_size = max(3, n // 50)
        if kernel_size % 2 == 0:
            kernel_size += 1
        kernel = np.exp(-0.5 * (np.arange(kernel_size) - kernel_size // 2) ** 2 / (kernel_size // 4 + 1))
        kernel /= kernel.sum()
        a_new = np.convolve(a, kernel, mode='same')
        a_new = np.maximum(a_new, 0)

    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def propose_candidate():
    """
    Multi-phase optimization combining enhanced coordinate descent,
    Nesterov-accelerated LP fictitious play, interleaved hybrid,
    multi-length exploration, and fine polishing.
    """
    deadline = time.monotonic() + 1785.0
    rng = np.random.default_rng(42)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 1024
        best_seq = np.ones(n_init) / n_init
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Fallback init: n={n_init}, score={best_score:.8f}")

    n_best = len(best_seq)

    # === Phase 1: Enhanced coordinate descent ===
    print(f"\n=== Phase 1: Enhanced CD (best={best_score:.8f}) ===")
    t1 = min(200.0, (deadline - time.monotonic()) * 0.12)
    print(f"  Budget: {t1:.0f}s, n={n_best}")
    a1, s1, sw1 = enhanced_coord_descent(best_seq, t1, top_k=12, verbose=True, label="p1")

    # === Phase 2: Nesterov LP Fictitious Play ===

    print(f"\n=== Phase 3: Interleaved LP+CD (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 600
    t3 = min(max(rem * 0.3, 60), 400.0)
    print(f"  Budget: {t3:.0f}s, n={len(best_seq)}")
    a3, s3, it3 = interleaved_lp_cd(best_seq, t3, solver=solver, label="p3")
    best_seq, best_score = check_update(a3, best_seq, best_score, "p3_ilpcd")

    # === Phase 4: Perturb & re-optimize ===
    print(f"\n=== Phase 4: Perturb Loop (best={best_score:.8f}) ===")
    n_best = len(best_seq)
    for rnd in range(3):
        rem = deadline - time.monotonic() - 300
        if rem < 60:
            break
        perturbed = perturb_escape(best_seq, rng, mode=rnd)
        t4a = min(rem * 0.2, 60.0)
        a4, s4, _ = enhanced_coord_descent(perturbed, t4a, top_k=8, label=f"p4r{rnd}")
        best_seq, best_score = check_update(a4, best_seq, best_score, f"p4_pert_r{rnd}")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-Length (best={best_score:.8f}) ===")
    n_best = len(best_seq)
    alt_lengths = []
    for alt in [n_best - 64, n_best - 32, n_best + 32, n_best + 64, 960, 1088, 1152, 1280]:
        if 512 <= alt <= 2048 and alt != n_best and alt not in alt_lengths:
            alt_lengths.append(alt)
    alt_lengths = alt_lengths[:4]

    for n_alt in alt_lengths:
        rem = deadline - time.monotonic() - 200
        if rem < 40:
            break
        t5 = min(rem * 0.3, 80.0)
        init_alt = spectral_interp(best_seq, n_alt)
        a5, s5, _ = enhanced_coord_descent(init_alt, t5, top_k=10, label=f"p5_n{n_alt}")
        best_seq, best_score = check_update(a5, best_seq, best_score, f"p5_n{n_alt}")

                a5f, s5f = fine_cd(best_seq, min(rem2 * 0.3, 40.0), label=f"p5f_n{len(best_seq)}")

    print(f"\n=== Phase 6: Final Polish (best={best_score:.8f}) ===")

        # Fine CD

    # Final fine polish
    rem = deadline - time.monotonic() - 15
    if rem > 10:
        a6c, s6c = fine_cd(best_seq, rem * 0.8, label="p6c")
        best_seq, best_score = check_update(a6c, best_seq, best_score, "p6c_fine")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
