def project_simplex(v):
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n)/n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) / np.sum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - cssv[-1]) + 1)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def compute_subgradient(a, n_active_frac=0.1):
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    
    if max_b < 1e-300: 
        return np.zeros(n), max_b
    
    threshold = max_b * (1 - n_active_frac)  # Avoid tiny values
    active_idx = np.where(b >= threshold)[0]
    
    if len(active_idx) == 0:
        active_idx = np.array([np.argmax(a)] if n > 0 else [])
