def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_cross_convolve(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def spectral_init(n, rng):
    """Generate a sequence with good spectral properties using random phase DFT."""
    # Random phase in frequency domain
    freq = np.zeros(n)
    freq[0] = 1.0  # DC component
    # Choose random phases for the positive frequencies
    for k in range(1, n // 2 + 1):
        freq[k] = np.exp(1j * rng.uniform(0, 2 * np.pi))
        if k < n - k:  # Conjugate symmetry for real signal
            freq[n - k] = freq[k].conjugate()
    if n % 2 == 0:
        freq[n // 2] = np.exp(1j * rng.uniform(0, 2 * np.pi)).real  # Nyquist is real
    
    # Inverse DFT
    a = irfft(freq, n).real
    a = np.abs(a)  # Project to non-negative (take absolute value)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(n) / n
    return a

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    if size <= 0:
        return np.array([0.0])
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

        # Actually: grad[i] = sum_{j: 0<=j<n, 0<=i+j<L} a[j] * weights[i+j]
        # where L = len(weights) = 2n-1
        # Let's compute this directly for efficiency (n up to ~2048, so O(n^2) might be too slow)
        # Use FFT: this is a cross-correlation
        # Define b[j] = a[j], c[k] = weights[k]
        # We want d[i] = sum_j b[j] * c[i+j] for i = 0, ..., n-1
        # This is the "anti-convolution" or we can reverse one:
        # d[i] = sum_j b[j] * c[i+j] = (b * c_rev)[i + n - 1] where c_rev[k] = c[2n-2-k]... 
        # Actually let's just use the direct formula with numpy:
        # For each i: grad[i] = dot(a[:min(n, L-i)], weights[i:i+min(n, L-i)])
        # This is O(n * min(n, L)) = O(n^2) which for n=1024 is ~1M ops, fine.

        L = len(weights)
        for i in range(n):
            j_end = min(n, L - i)
            if j_end > 0:
                grad[i] = 2.0 * np.dot(a[:j_end], weights[i:i + j_end])

        # Momentum update
        v = momentum * v + grad
        a_new = a - lr * v
        a_new = simplex_project(np.maximum(a_new, 0))

        # Evaluate
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Adaptive parameters
        if no_improve > 50:
            lr *= 0.85
            beta *= 1.2
            momentum = min(0.99, momentum + 0.01)
            no_improve = 0
            if lr < 1e-12:
                break
        elif it % 20 == 0:
            beta *= 1.005

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GradGD {label} it={it} ({elapsed:.0f}s): score={best_score:.6f}, beta={beta:.1f}, lr={lr:.6f}")

    return best_a, best_score


def kaiser_init(n, beta=4.0):
    """Kaiser window initialization."""
    from scipy.special import i0
    k = np.arange(n)
    x = (2.0 * k - (n - 1)) / (n - 1) if n > 1 else np.array([0.0])
    a = i0(beta * np.sqrt(np.maximum(1 - x**2, 0)))
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a


def spectral_init_random_phase(n, rng):
    """Random phase DFT initialization."""
    freq = np.ones(n)
    phases = rng.uniform(0, 2 * np.pi, n)
    phases[0] = 0.0
    if n % 2 == 0:
        phases[n // 2] = 0.0
    for i in range(1, n // 2):
        phases[n - i] = -phases[i]
    spectrum = np.exp(1j * phases)
    a = irfft(spectrum, n).real
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def spectral_init_tapered(n, rng, taper=2.0):
    """Tapered spectrum initialization."""
    freq_mag = np.ones(n)
    # Apply a taper to reduce high-frequency content
    k = np.arange(n)
    half = n / 2.0
    freq_mag = 1.0 / (1.0 + taper * (k / half - 1.0)**4)
    # Ensure conjugate symmetry
    freq_mag_mag = np.abs(freq_mag)
    phases = rng.uniform(0, 2 * np.pi, n)
    phases[0] = 0.0
    if n % 2 == 0:
        phases[n // 2] = 0.0
    for i in range(1, n // 2):
        phases[n - i] = -phases[i]
    spectrum = freq_mag_mag * np.exp(1j * phases)
    a = irfft(spectrum, n).real
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def perturb_structured(a0, rng, strength=0.05):
    """Perturb focusing on high-autocorrelation regions."""
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a)
    M = conv.max()
    if M < 1e-20:
        return a

    # Identify peak positions
    threshold = M * 0.85
    peaks = np.where(conv >= threshold)[0]

    # Compute pressure on each coordinate
    pressure = np.zeros(n)
    for k in peaks:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            pressure[i] += a0[i] * a0[j]

    p_max = pressure.max()
    if p_max > 1e-20:
        pressure = pressure / p_max

    # Apply perturbation: reduce high-pressure, add some noise
    noise = rng.standard_normal(n)
    a_new = a * np.maximum(1.0 - strength * pressure + 0.3 * strength * noise, 0.01)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def perturb_random(a0, rng, frac=0.15):
    """Random perturbation."""
    n = len(a0)
    a = a0.copy()
    mask = rng.random(n) < frac
    a[mask] = a[mask] * (1.0 + 0.5 * rng.standard_normal(mask.sum()))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    else:
        a = a0 / a0.sum()
    return a


def perturb_block(a0, rng, block_size=16, strength=0.3):
    """Perturb a random block of coordinates."""
    n = len(a0)
    a = a0.copy()
    n_blocks = n // block_size
    if n_blocks < 1:
        return a
    b = rng.integers(0, n_blocks)
    start = b * block_size
    end = min(start + block_size, n)
    a[start:end] = a[start:end] * np.maximum(1.0 + strength * rng.standard_normal(end - start), 0.01)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, best_seq, best_score, label=""):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score


def fine_coord_descent(a0, time_limit, label=""):
    """Very fine coordinate descent with tight factor search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0

    factors = [0.999, 0.998, 0.995, 0.99, 0.985, 0.98, 0.97,
               1.001, 1.002, 1.005, 1.01, 1.015, 1.02, 1.03]

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            ai = a[i]
            if ai < 1e-18:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            unaff_max = -1e30
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            affected = conv[k_start:k_end + 1]
            idx = np.arange(k_start, k_end + 1)
            coeffs = 2.0 * a[idx - i]
            quad_offset = 2 * i - k_start if (2 * i >= k_start and 2 * i <= k_end) else -1

            best_delta = 0.0
            best_new_max = M

            for factor in factors:
                delta = ai * (factor - 1.0)
                if delta < -ai - 1e-15:
                    continue
                new_aff = affected + delta * coeffs
                if quad_offset >= 0:
                    new_aff_ao = new_aff[quad_offset] + delta * delta
                else:
                    new_aff_ao = 0.0
                new_max = max(unaff_max, float(new_aff.max()))
                if new_max < best_new_max - 1e-16:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-15:
                a[i] = a[i] + best_delta
                conv_update_delta(conv, a, i, best_delta, n)
                M = best_new_max
                improved = True

        if sweeps % 50 == 0:
            conv = fast_convolve(a)
            M = float(conv.max())

        cur_score = 2.0 * n * M
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()

        if not improved and sweeps > 20:
            break

    return best_a, best_score


def propose_candidate():
    """
    Multi-phase optimization:
    1. Generate diverse initializations (parent, Kaiser, spectral)
    2. Exact coordinate descent
    3. Smoothed gradient descent
    4. Perturbation + re-optimization loop
    5. Multi-length exploration
    6. Final polishing
    """
    deadline = time.monotonic() + 1785.0
    rng = np.random.default_rng(42)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 1024
        best_seq = kaiser_init(n_init, beta=4.0)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Kaiser init: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)
    print(f"Starting: n={n_best}, best={best_score:.6f}")

    # === Phase 1: Exact coordinate descent ===
    print(f"\n=== Phase 1: Exact CD (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(250.0, rem * 0.18)
    print(f"  Budget: {t1:.0f}s, n={n_best}")
    a1, s1 = coord_descent(best_seq, t1, verbose=True, label="p1")
    best_seq, best_score = check_and_update(a1, best_seq, best_score, "p1_cd")

    # === Phase 2: Smoothed gradient descent ===
    print(f"\n=== Phase 2: Smoothed GD (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(150.0, rem * 0.12)
    if rem > 100:
        print(f"  Budget: {t2:.0f}s, n={n_best}")
        a2, s2 = smoothed_gradient_descent(best_seq, t2, beta_init=40.0, lr_init=0.03, label="p2")
        best_seq, best_score = check_and_update(a2, best_seq, best_score, "p2_gd")

    # === Phase 3: Another CD pass ===
    print(f"\n=== Phase 3: CD pass 2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(150.0, rem * 0.12)
    if rem > 80:
        print(f"  Budget: {t3:.0f}s")
        a3, s3 = coord_descent(best_seq, t3, verbose=False, label="p3")
        best_seq, best_score = check_and_update(a3, best_seq, best_score, "p3_cd")

    # === Phase 4: Perturbation + re-optimization loop ===
    print(f"\n=== Phase 4: Perturb Loop (best={best_score:.6f}) ===")
    for rnd in range(6):
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t4 = min(rem * 0.12, 120.0)

        # Choose perturbation type
        pert_type = rnd % 3
        if pert_type == 0:
            perturbed = perturb_structured(best_seq, rng, strength=0.03 + 0.015 * rnd)
            plabel = f"p4_struct_{rnd}"
        elif pert_type == 1:
            perturbed = perturb_random(best_seq, rng, frac=0.1 + 0.05 * rnd)
            plabel = f"p4_rand_{rnd}"
        else:
            perturbed = perturb_block(best_seq, rng, block_size=8 + 8 * rnd, strength=0.2 + 0.1 * rnd)
            plabel = f"p4_block_{rnd}"

        # Optimize perturbed version
        a4, s4 = coord_descent(perturbed, t4, verbose=False, label=plabel)
        best_seq, best_score = check_and_update(a4, best_seq, best_score, plabel)

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-Length (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    alt_lengths = set()
    for alt in [n_best // 2, int(n_best * 1.25), int(n_best * 1.5), n_best * 2]:
        if 256 <= alt <= 4096:
            alt_lengths.add(alt)
    alt_lengths = sorted(alt_lengths)[:4]

    for n_alt in alt_lengths:
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t5 = min(rem * 0.15, 100.0)
        if n_alt == len(best_seq):
            init_alt = best_seq.copy()
        else:
            init_alt = interpolate_seq(best_seq, n_alt)
        a5, s5 = coord_descent(init_alt, t5, verbose=False, label=f"p5_n{n_alt}")
        best_seq, best_score = check_and_update(a5, best_seq, best_score, f"p5_n{n_alt}")

    # === Phase 6: Spectral re-initialization + optimization ===
    print(f"\n=== Phase 6: Spectral Init + Opt (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        t6 = min(rem * 0.2, 120.0)
        n_try = len(best_seq)
        for init_type in ['kaiser', 'spectral', 'tapered']:
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            t6i = min(rem * 0.3, 60.0)
            if init_type == 'kaiser':
                a_init = kaiser_init(n_try, beta=5.0)
            elif init_type == 'spectral':
                a_init = spectral_init_random_phase(n_try, rng)
            else:
                a_init = spectral_init_tapered(n_try, rng, taper=3.0)

            a6, s6 = coord_descent(a_init, t6i, verbose=False, label=f"p6_{init_type}")
            best_seq, best_score = check_and_update(a6, best_seq, best_score, f"p6_{init_type}")

    # === Phase 7: Final intensive polishing ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t7 = min(rem * 0.4, 120.0)
        print(f"  Fine CD budget: {t7:.0f}s")
        a7, s7 = fine_coord_descent(best_seq, t7, label="p7")
        best_seq, best_score = check_and_update(a7, best_seq, best_score, "p7_fine")

    rem = deadline - time.monotonic()
    if rem > 20:
        t7b = min(rem * 0.5, 80.0)
        a7b, s7b = coord_descent(best_seq, t7b, verbose=False, label="p7b")
        best_seq, best_score = check_and_update(a7b, best_seq, best_score, "p7b_cd")

    rem = deadline - time.monotonic()
    if rem > 15:
        a7c, s7c = fine_coord_descent(best_seq, rem * 0.7, label="p7c")
        best_seq, best_score = check_and_update(a7c, best_seq, best_score, "p7c_fine")

    # Fallback
    if best_seq is None:
        best_seq = kaiser_init(1024, beta=4.0)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
