    """Full convolution a*a using FFT."""

    size = 2*n - 1

    if conv_len > fft_size:
        fft_size *= 2
    A_rev = rfft(a[::-1].copy(), fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)
    grad_a = 2.0 * conv[n - 1:n - 1 + n]

    # Transform to z-space: grad_z[i] = a[i]*(grad_a[i] - dot(grad_a, a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj_val, grad_z


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a_fixed):
    """Build the (2n-1) x n convolution matrix T where (T@b) = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T

def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0. Returns (b, M)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, max_iters=100000):
    """
    Nesterov-accelerated LP best-response iteration.
    Properly implements the accelerated scheme for the zero-sum game:
      min_a max_k (a*a)_k
    Uses running average to track best solution.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    y = a.copy()  # Extrapolation point
    a_prev = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    # Running average (often better than any single iterate)
    avg_a = np.zeros(n)
    n_avg = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        # Solve LP: find b* minimizing max_k (y * b*)_k
        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue

        # Nesterov step with t_k schedule
        alpha_t = 2.0 / (t + 2.0)

        # Update y (extrapolation point)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_new = y_new / y_new.sum()

        # Nesterov extrapolation: a_{t+1} from y
        if t == 1:
            a_new = b_star.copy()
        else:
            # momentum = (t-1)/(t+2) for Nesterov
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s

        y = y_new
        a_prev = a
        a = a_new

        # Check score of a
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        # Check score of y
        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        # Running average
        avg_a += a
        n_avg += 1
        if n_avg >= 10 and n_avg % 20 == 0:
            avg_check = avg_a / n_avg
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()

        if t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}, LP_M={M_val:.8f}")

    return best_a, best_score


def logsumexp_optimize(a0, time_limit, T_start=100.0, T_end=0.05, n_stages=20):
    """
    LogSumExp-smoothed optimization with L-BFGS and multi-stage annealing.
    Uses FFT-based gradients for O(n log n) per iteration.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a_init = a_init / a_init.sum()
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        # Scale max iterations based on remaining time and stage
        iters_per_stage = max(15, min(80, int(40 + 20 * (si / n_stages))))

        def func(z_in):
            val, _ = lse_obj_grad_z(z_in, stage_T)
            return val

        def jac(z_in):
            _, g = lse_obj_grad_z(z_in, stage_T)
            return g

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': iters_per_stage * 200,
                         'disp': False}
            )
            z = res.x
            a = softmax_fn(z)
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
        except Exception:
            pass

    return best_a, best_score


def peak_shaving(a0, time_limit, seed=123):
    """
    Coordinate-descent peak shaving: directly reduce top autocorrelation peaks
    by moving mass from high-contributing positions to low-contributing ones.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:8]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            # Find highest contributor
            src_local = np.argmax(contrib)
            src = idx[src_local]

            # Find target: position not in any top peak area, or low value
            peak_areas = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_areas.update(range(js2, je2 + 1))

            candidates = [j for j in range(n) if j not in peak_areas]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            # Pick a random candidate with low value
            candidates.sort(key=lambda j: a[j])
            dst_pool = candidates[:max(1, len(candidates) // 3)]
            dst = dst_pool[rng.integers(len(dst_pool))]

            for frac in [0.003, 0.007, 0.015, 0.03, 0.05]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random exploration
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            if stall > 200:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 1.5
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.6 + 0.3 * np.cos(3 * x) + 0.1 * np.cos(7 * x)
    else:
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Uses multi-phase optimization with:
    - LP Nesterov at multiple lengths
    - FFT-based LogSumExp gradient optimization
    - Peak shaving for final polish
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)  # Leave 60s margin

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Time budget: {time_budget:.0f}s")

    # Get parent sequence
    best_seq = None
    best_score = float('inf')
    parent = None

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Determine starting length
    n_base = len(best_seq) if best_seq is not None else 1200

    if parent is None:
        n_base = 1200
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: LP Nesterov at n={n_base} (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.30, 500.0)
    if t1 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t1, solver)
        try_update(a_out, f"lp_n{n_base}")

    # Phase 2: Gradient refinement at same length
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: LogSumExp at n={n_base} (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 15:
        a_out, _ = logsumexp_optimize(best_seq, t2, T_start=50.0, T_end=0.05, n_stages=15)
        try_update(a_out, f"lse_n{n_base}")

    # Phase 3: Try n = round(n_base * 1.33) ~ 1600
    n_long1 = int(round(n_base * 4 / 3))
    n_long1 = max(n_long1, n_base + 100)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: LP Nesterov at n={n_long1} (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.25, 400.0)
    if t3 > 30:
        init_long = interpolate_sequence(best_seq, n_long1)
        a_out, _ = nesterov_lp_optimize(init_long, t3, solver)
        try_update(a_out, f"lp_n{n_long1}")

    # Phase 4: Try n = round(n_base * 1.67) ~ 2000
    n_long2 = int(round(n_base * 5 / 3))
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: LogSumExp at n={n_long2} (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 20:
        init_long2 = interpolate_sequence(best_seq, n_long2)
        a_out, _ = logsumexp_optimize(init_long2, t4, T_start=200.0, T_end=0.05, n_stages=20)
        try_update(a_out, f"lse_n{n_long2}")

    # Phase 5: LP at the longer length (if time permits)
    remaining = time_budget - (time.monotonic() - t_global_start)
    n_lp2 = n_long1  # Use the length that worked well in Phase 3
    print(f"\n=== Phase 5: LP Nesterov refinement at n={n_lp2} (remaining={remaining:.0f}s) ===")
    t5 = min(remaining * 0.25, 350.0)
    if t5 > 30:
        init5 = interpolate_sequence(best_seq, n_lp2)
        # If we have a good solution at this length, use it; otherwise interpolate
        a_out, _ = nesterov_lp_optimize(init5, t5, solver)
        try_update(a_out, f"lp_refine_n{n_lp2}")

    # Phase 6: Final peak shaving on the best solution
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 6: Peak shaving (remaining={remaining:.0f}s) ===")
    t6 = min(remaining * 0.25, 150.0)
    if t6 > 5:
        a_out, _ = peak_shaving(best_seq, t6, seed=456)
        try_update(a_out, "peak_shave_1")

    # Phase 7: Final LogSumExp polish
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 7: Final LogSumExp polish (remaining={remaining:.0f}s) ===")
    t7 = min(remaining * 0.30, 120.0)
    if t7 > 10:
        a_out, _ = logsumexp_optimize(best_seq, t7, T_start=5.0, T_end=0.02, n_stages=12)
        try_update(a_out, "final_lse")

    # Phase 8: One more peak shave
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 10:
        t8 = min(remaining * 0.5, 60.0)
        if t8 > 5:
            a_out, _ = peak_shaving(best_seq, t8, seed=789)
            try_update(a_out, "peak_shave_2")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
