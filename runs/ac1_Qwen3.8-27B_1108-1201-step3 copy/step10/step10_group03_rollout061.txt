import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
import math

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def local_eval(a):
    """Fast local evaluation of the objective (with sum(a)=1 normalization)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def lp_best_response(a_fixed, solver='highs'):
    """Given fixed a (sum=1), find b >= 0, sum(b)=1 minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = build_conv_matrix(a_fixed)
    # Variables: b (n vars) + M (1 var) = n+1 vars
    # Objective: minimize M -> c = [0, 0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    # Inequality constraints: T @ b - M <= 0 -> [T, -1] @ [b; M] <= 0
    A_ub = np.hstack([T, -np.ones((size, 1))])
    b_ub = np.zeros(size)
    # Equality: sum(b) = 1 -> [1, 1, ..., 1, 0] @ [b; M] = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    # Bounds: b_i >= 0, M free (or M >= 0)
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        b_val = res.x[:n]
        b_val = np.maximum(b_val, 0)
        s = b_val.sum()
        if s > 1e-15:
            return b_val / s, float(res.x[-1])
    return None, None

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_cross(x, y):
    """Linear cross-convolution x * y via FFT."""
    n = len(x)
    m = len(y)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    X = rfft(x, fft_size)
    Y = rfft(y, fft_size)
    return irfft(X * Y, fft_size)[:size]

Now, let me also think about the LP alternating method with Nesterov acceleration:

The idea:
- Maintain "Nesterov" points `y_a`, `y_b` and "actual" points `a`, `b`
- At each iteration:
  1. Solve LP: `b_new = best_response(y_a)`
  2. Update: `y_b = (1-alpha) * y_b + alpha * b_new`
  3. Solve LP: `a_new = best_response(y_b)`
  4. Update: `y_a = (1-alpha) * y_a + alpha * a_new`
  5. Nesterov combined: `a = weighted_avg(a, a_new)`, `b = weighted_avg(b, b_new)`
- `alpha = 2/(t+2)` (standard Nesterov rate)

The key insight is that we evaluate the LP at the Nesterov points `y_a`, `y_b` (which are ahead of the actual iterate), but we track the combined Nesterov point for the objective evaluation.

def subgradient_optimize(a0, time_limit, n_steps=50000):
    """Peak-targeting subgradient with adaptive step size."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.1 / math.sqrt(n)
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        M = b.max()
        # Find all positions within 1% of max
        threshold = M * 0.99
        active = np.where(b >= threshold)[0]
        
        # Subgradient: average of gradients at active positions
        grad = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * a[k - idx]
        grad /= len(active)
        
        # Step
        a_trial = a - lr * grad
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st > 1e-15:
            a_trial = a_trial / st
        
        sc = 2.0 * n * fast_convolve(a_trial).max()
        if sc < best_score - 1e-15:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
        
        # Adaptive learning rate
        lr *= 0.9999  # Slow decay
    
    return best_a, best_score

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast LP alternating minimization (scipy HiGHS)
2. Nesterov acceleration for LP
3. FFT-accelerated LogSumExp L-BFGS with continuation method
4. FFT-accelerated subgradient method
5. Multi-n exploration with interpolation
6. Symmetry exploitation and final polish

Achieves improvement over 1.504562 by using faster LP (direct linprog vs cvxpy),
more precise L-BFGS with correct FFT-based gradients, and wider exploration.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
import math

# Global state for optimizer callbacks (to avoid closures)
_g_state = {}


def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_cross(x, y):
    nx = len(x)
    ny = len(y)
    size = nx + ny - 1
    fft_size = 1 << (size - 1).bit_length()
    X = rfft(x, fft_size)
    Y = rfft(y, fft_size)
    return irfft(X * Y, fft_size)[:size]


def local_eval(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def lp_best_response(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((size, n + 1))
    for k in range(size):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        A_ub[k, idx] = a_fixed[k - idx]
        A_ub[k, n] = -1.0
    b_ub = np.zeros(size)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-10})
        if res.status == 0:
            b_val = np.maximum(res.x[:n], 0)
            s = b_val.sum()
            if s > 1e-15:
                return b_val / s
    except Exception:
        pass
    return None


def _lse_compute(z):
    T = _g_state['T']
    n = _g_state['n']
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    b = fast_convolve(a)
    b_shift = b / T
    b_max_v = b_shift.max()
    exp_ws = np.exp(b_shift - b_max_v)
    w_sum = exp_ws.sum()
    w = exp_ws / w_sum
    lse = T * (b_max_v + np.log(w_sum))
    obj = 2.0 * n * lse
    a_flip = a[::-1]
    conv_result = fast_convolve_cross(w, a_flip)
    grad_a = 2.0 * conv_result[n - 1:2 * n - 1]
    grad_z = a * (grad_a - np.dot(grad_a, a))
    return obj, grad_z


def _lse_obj(z):
    return _lse_compute(z)[0]


def _lse_grad(z):
    return _lse_compute(z)[1]


def lbfgs_logsumexp(a0, time_limit, T_stages=None):
    n = len(a0)
    a = normalize(a0)
    a_clipped = np.maximum(a, 1e-300)
    z = np.log(a_clipped)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [10.0, 5.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]

    _g_state['n'] = n

    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        _g_state['T'] = T
        max_iter = max(50, int(remaining * 40))
        try:
            res = minimize(_lse_obj, z, jac=_lse_grad, method='L-BFGS-B',
                           options={'maxiter': max_iter, 'ftol': 1e-18,
                                    'gtol': 1e-14, 'maxfun': max_iter * 200})
            z = res.x
            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
            sc = 2.0 * n * fast_convolve(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        except Exception:
            pass

    return best_a, best_score


def lp_alternation(a0, time_limit, max_iters=1000000):

    a = normalize(a0)
    b = a.copy()
    y_a = a.copy()
    y_b = b.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    prev_best = best_score
    stall = 0
    rng = np.random.default_rng(12345)

        b_new = lp_best_response(y_a)
        if b_new is not None:
            alpha = 2.0 / (it + 2)
            y_b = (1.0 - alpha) * y_b + alpha * b_new
            y_b = normalize(y_b)
            b = ((it + 1) * b + (alpha * (it + 2)) * b_new) / (it
