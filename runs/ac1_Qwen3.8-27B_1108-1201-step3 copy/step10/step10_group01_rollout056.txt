def compute_gradient_softmax(a, b, T):
    """Compute gradient of LogSumExp-smoothed objective wrt log-parameters z.
    a: current point on simplex (sum=1), shape (n,)
    b: conv(a,a), shape (2n-1,)
    T: temperature
    Returns: gradient wrt z, shape (n,)
    """
    n = len(a)
    # Compute w = softmax(T * b)
    Tb = T * b
    Tb_shifted = Tb - Tb.max()
    w = np.exp(Tb_shifted)
    w = w / w.sum()
    
    # Compute g_i = sum_k w_k * a_{k-i}
    w_flipped = w[::-1]
    N = 1
    while N < 3*n - 2:

    h = np.dot(w, b)
    
    # Gradient: 2n * a_i * (g_i - h)
    grad = 2.0 * n * a * (g - h)
    return grad

Multi-phase optimization minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses: (1) LogSumExp-smoothed L-BFGS with FFT-based gradients and
ultra-low temperatures, (2) adaptive subgradient with Barzilai-Borwein
step size and Nesterov momentum, (3) LP best-response alternation,
(4) multi-length exploration, (5) final coordinate-descent polish.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1
    while N < size:
        N <<= 1
    A = rfft(a, N)
    return irfft(A * A, N)[:size]


def evaluate_local(a):
    """Evaluate 2n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def lse_value(a, T):
    """LogSumExp-smoothed max of conv(a,a), scaled."""
    b = fast_convolve(a)
    n = len(a)
    Tb = T * b
    return (2.0 * n / T) * (Tb.max() + np.log(np.exp(Tb - Tb.max()).sum()))


def lse_gradient_z(a, T):
    """Gradient of lse_value wrt log-parameters z (softmax parameterization)."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    Tb_shift = Tb - Tb.max()
    w = np.exp(Tb_shift)
    w /= w.sum()
    
    # Compute g_i = sum_k w_k * a_{k-i} via cross-correlation
    w_flipped = w[::-1]
    N = 1
    while N < 3 * n - 2:
        N <<= 1
    conv_res = irfft(rfft(w_flipped, N) * rfft(a, N), N)
    g = conv_res[n-1:2*n-1][::-1]
    
    h = np.dot(w, b)
    grad = 2.0 * n * a * (g - h)
    return grad


def lse_grad_direct(z, T):
    """Compute gradient from z directly (for L-BFGS)."""
    a = np.exp(z - z.max())
    a /= a.sum()
    return lse_gradient_z(a, T)


def lse_func_direct(z, T):
    """Compute objective from z directly (for L-BFGS)."""
    a = np.exp(z - z.max())
    a /= a.sum()
    return lse_value(a, T)


def lbgfs_refine(a0, time_limit, T_stages=None, maxiter=5000):
    """L-BFGS optimization with LogSumExp smoothing and multi-stage cooling."""
    if T_stages is None:
        T_stages = [20.0, 10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.15, 0.08, 0.04, 0.02, 0.01]
    
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()
    
    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        def func(z):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            return lse_func_direct(z, T)
        
        def jac(z):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z)
            return lse_grad_direct(z, T)
        
        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                          options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                                   'maxfun': maxiter * 100, 'disp': False})
            z = res.x
        except Exception:
            pass
    
    a_out = np.exp(z - z.max())
    a_out /= a_out.sum()
    return a_out

def eval_score(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def lse_objective(z, T):
    a = np.exp(z - z.max())
    a /= a.sum()
    b = fast_convolve(a)
    Tb = T * b
    return (2.0 * len(a) / T) * (Tb.max() + np.log(np.exp(Tb - Tb.max()).sum()))


def lse_gradient(z, T):
    a = np.exp(z - z.max())
    a /= a.sum()
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    w = np.exp(Tb - Tb.max())
    w /= w.sum()
    # Cross-correlation: g[i] = sum_k w[k] * a[k-i]
    w_flipped = w[::-1]
    N = 1
    while N < 3 * n - 2:
        N <<= 1
    conv_res = irfft(rfft(w_flipped, N) * rfft(a, N), N)
    g = conv_res[n - 1:2 * n - 1][::-1]
    h = np.dot(w, b)
    return 2.0 * n * a * (g - h)


def _cb_obj(z):
    return lse_objective(z, _gt[0])


def _cb_jac(z):
    return lse_gradient(z, _gt[0])


def lbgfs_optimize(a0, time_limit, T_stages=None, maxiter=5000):
    if T_stages is None:
        T_stages = [30.0, 15.0, 8.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.25,
                    0.12, 0.06, 0.03, 0.015, 0.008, 0.004]
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()
    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        if time_limit - (time.monotonic() - t0) < 5:
            break
        _gt[0] = T
        try:
            res = minimize(fun=_cb_obj, x0=z, jac=_cb_jac, method='L-BFGS-B',
                           options={'maxiter': maxiter, 'ftol': 1e-18,
                                    'gtol': 1e-14, 'maxfun': maxiter * 200})
            z = res.x
        except Exception:
            pass
    a_out = np.exp(z - z.max())
    a_out /= a_out.sum()
    return a_out


def subgradient_optimize(a0, time_limit, top_k=20, momentum=0.9,
                         lr0=0.05, n_steps=100000):
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()
    v = np.zeros(n)  # momentum
    lr = lr0
    no_improve = 0
    prev_a = a.copy()
    prev_g = None

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        # Adaptive top-K
        sorted_b = np.sort(b)[::-1]
        gap = sorted_b[min(top_k, len(b) - 1)] / max_b if max_b > 0 else 1.0
        k_eff = min(top_k, max(5, int(top_k * gap)))
        
        # Get top-K peaks
        top_idx = np.argsort(b)[::-1][:k_eff]
        
        # Average subgradient over top-K
        g = np.zeros(n)
        for k in top_idx:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= k_eff
        g *= 2.0 * n

        # Nesterov momentum
        v = momentum * v + (1 - momentum) * g
        
        # BB step size (if possible)
        if prev_g is not None:
            s_vec = a - prev_a
            y_vec = g - prev_g
            sy = np.dot(s_vec, y_vec)
            ss = np.dot(s_vec, s_vec)
            if sy > 1e-14:
                lr_bb = ss / sy
                lr = 0.5 * lr + 0.5 * np.clip(lr_bb, lr * 0.1, lr * 10.0)
            else:
                lr *= 0.95

        # Update
        a_trial = a - lr * v
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st

        score = eval_score(a_trial)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                lr *= 0.9
                no_improve = 0
            if lr < 1e-8:
                break

        prev_a = a.copy()
        prev_g = g.copy()

    return best_a, best_score


def solve_lp_br(a_fixed, solver=None):
    """Solve LP best response: min M s.t. T@b <= M, sum(b)=1, b>=0."""
    if solver is None:
        solver = _find_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    constr = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constr)
    for sol in [solver] + [cp.SCS, cp.ECOS, cp.CLARABEL]:
        try:
            prob.solve(solver=sol, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                bv = np.asarray(b_var.value, dtype=np.float64)
                bv = np.maximum(bv, 0)
                s = bv.sum()
                if s > 1e-15:
                    bv /= s
                    return bv, float(M_var.value)
        except Exception:
            continue
    return None, None


_solver_cache = [None]


def _find_solver():
    if _solver_cache[0] is not None:
        return _solver_cache[0]
    for name in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(2, nonneg=True)
            p = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            p.solve(solver=name, verbose=False)
            if p.status in ('optimal', 'optimal_inaccurate'):
                _solver_cache[0] = name
                return name
        except Exception:
            continue
    _solver_cache[0] = cp.ECOS
    return cp.ECOS


def lp_alternation(a0, time_limit, n_iters=5000, solver=None):
    """Nesterov LP alternation."""
    n = len(a0)
    a = normalize(a0)
    y = a.copy()
    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()
    stall = 0
    prev_best = best_score

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_val = solve_lp_br(y, solver)
        if b_new is None:
            stall += 1
            if stall > 30:
                break
            continue

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy > 1e-15:
            y /= sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / ((it + 1) + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa > 1e-15:
            a /= sa

        sc_a = eval_score(a)
        sc_y = eval_score(y)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-10:
                stall = 0
            else:
                stall += 1
            prev_best = best_score

            if stall >= 8:
                rng = np.random.default_rng(it)
                y += 0.01 * rng.standard_normal(n)
                y = np.maximum(y, 0)
                y /= y.sum()
                stall = 0

    return best_a, best_score


def coord_descent_polish(a0, time_limit, n_steps=50000):
    """Coordinate descent targeting top peaks."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_k = int(np.argmax(b))
        js = max(0, max_k - n + 1)
        je = min(n - 1, max_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[max_k - idx]
        order = np.argsort(contribs)[::-1][:5]

        improved = False
        for oi in order:
            pos = int(idx[oi])
            if a[pos] < 1e-12:
                continue
            for factor in [0.98, 0.97, 0.95]:
                a_t = a.copy()
                a_t[pos] *= factor
                diff = a[pos] * (1 - factor)
                mask = np.ones(n, dtype=bool)
                mask[pos] = False
                others = a_t[mask].sum()
                if others < 1e-15:
                    break
                a_t[mask] += diff * a[mask] / a[mask].sum()
                a_t = a_t / a_t.sum()
                sc = eval_score(a_t)
                if sc < best_score - 1e-13:
                    a = a_t
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try increasing a small element
            min_pos = int(np.argmin(a))
            for factor in [1.05, 1.1, 1.2]:
                a_t = a.copy()
                a_t[min_pos] *= factor
                a_t = a_t / a_t.sum()
                sc = eval_score(a_t)
                if sc < best_score - 1e-13:
                    a = a_t
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            break

    return best_a, best_score


def interpolate_length(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a_norm = normalize(a)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_new = np.interp(new_x, old_x, a_norm)
    f_new = np.maximum(f_new, 0)
    return normalize(f_new)


def structured_init(n, rng):
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.exp(-3.0 * x)
    elif choice == 1:
        a = np.exp(-3.0 * (1 - x))
    elif choice == 2:
        a = np.exp(-2.0 * x) + 0.3 * np.exp(-2.0 * (1 - x))
    elif choice == 3:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    else:
        a = np.exp(-4 * (x - 0.3) ** 2) + 0.2
    a = np.maximum(a, 0)
    return normalize(a)
