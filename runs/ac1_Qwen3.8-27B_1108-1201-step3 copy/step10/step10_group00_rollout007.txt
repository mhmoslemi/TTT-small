"""

import numpy as np

from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        lo = j
        hi = j + n
        T[lo:hi, j] = a
    return T


def solve_lp_best_response(a_fixed, n):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k] via LP."""
    size = 2 * n - 1
    # Variables: b_0, ..., b_{n-1}, M  (total n+1)
    # Objective: minimize M -> c = [0,...,0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0

    # Constraints: T_a @ b - M <= 0 for each k (2n-1 inequalities)
    # In standard form: A_ub @ x <= b_ub
    # T_a @ b - M*1 <= 0
    # So row k of A_ub: [T_a[k,0], ..., T_a[k,n-1], -1], b_ub[k] = 0
    T_a = build_conv_matrix(a_fixed)
    A_ub = np.hstack((T_a, -np.ones((size, 1))))
    b_ub = np.zeros(size)

    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    # Bounds: b_j >= 0, M unbounded below (but will be >= 0 from constraints)
    bounds = [(0, None)] * n + [(None, None)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True, 'dual_feasibility_tol': 1e-12, 'primal_feasibility_tol': 1e-12})

    if res.status != 0:
        return None, None

    b_val = res.x[:n]
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    M_val = max(0.0, res.x[-1])
    return b_val, M_val


def coordinate_descent_polish(a0, n_sweeps=5, tol=1e-14):
    """
    Exact coordinate descent: for each coordinate j, solve the 1D convex
    minimization problem to find the optimal a[j] given all other coordinates fixed.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a0.copy()
    a = a / s

    conv = fast_convolve(a)
    current_max = conv.max()
    best_a = a.copy()
    best_max = current_max

    for sweep in range(n_sweeps):
        improved = False
        for j in range(n):
            old_val = a[j]
            if old_val < 1e-15:
                # Try a small positive value
                # The change in conv[k] for k where j contributes:
                # For k in [j, j+n-1] intersected with [0, 2n-2]:
                pass

            valid_k_lo = max(0, j)
            valid_k_hi = min(2 * n - 2, j + n - 1)

            # For each affected lag k, compute the quadratic in x = a[j]:
            # conv_new[k] = conv_old[k] - old_contribution + new_contribution
            # old_contribution to conv[k] from a[j]:
            #   if k-j in [0,n-1] and j in [max(0,k-n+1), min(n-1,k)]:
            #     if k == 2*j: a[j]^2
            #     else: 2*a[j]*a[k-j]
            # (Note: both j and k-j must be in the valid range for the sum)

            # Precompute the "base" (conv without a[j] terms) and coefficients
            # for each affected k
            affected = np.arange(valid_k_lo, valid_k_hi + 1)
            k_valid = (affected >= 0) & (affected <= 2 * n - 2)
            affected = affected[k_valid]

            if len(affected) == 0:
                continue

            # For each k in affected, determine if j and k-j are both in valid range
            # j is always valid (0 <= j <= n-1)
            # k-j must be in [0, n-1]
            kj = affected - j
            both_valid = (kj >= 0) & (kj <= n - 1)
            affected = affected[both_valid]
            kj = kj[both_valid]

            if len(affected) == 0:
                continue

            # Compute base_k = conv[k] minus the terms involving a[j]
            # and the linear/quadratic coefficients
            is_diag = (affected == 2 * j)  # k == 2j means the term is a[j]^2
            off_diag = ~is_diag

            base = np.zeros(len(affected))
            coeff_lin = np.zeros(len(affected))
            coeff_quad = np.zeros(len(affected))

            for idx in range(len(affected)):
                k = affected[idx]
                current_conv_k = conv[k]
                if is_diag[idx]:
                    # Term is a[j]^2
                    base[idx] = current_conv_k - old_val * old_val
                    coeff_quad[idx] = 1.0
                else:
                    # Term is 2*a[j]*a[k-j]
                    aj = a[kj[idx]]
                    base[idx] = current_conv_k - 2 * old_val * aj
                    coeff_lin[idx] = 2 * aj

            # Unaffected lags keep their conv values
            # Overall max = max(max(affected new values), max(unaffected conv values))
            mask = np.ones(2 * n - 1, dtype=bool)
            mask[affected] = False
            unaffected_max = conv[mask].max() if mask.any() else 0.0

            # 1D convex minimization: min_x max( max_k(base_k + coeff_lin_k*x + coeff_quad_k*x^2), unaffected_max )
            # subject to x >= 0
            # Use bisection on the subgradient

            # The function f(x) = max(max_k(base_k + L_k*x + Q_k*x^2), unaffected_max)
            # is convex in x >= 0 (max of convex functions)
            # Find x where f'(x) = 0 or x = 0

            # Upper bound on x: the current value is old_val, and the optimal won't be much larger
            x_lo = 0.0
            x_hi = max(old_val * 2.0, 1e-10) * 4 + 0.01 / n  # generous upper bound

            # Bisection: find where the derivative changes sign
            # f'(x) = max over active k of (L_k + 2*Q_k*x)
            # At the minimum, f'(x) should be <= 0 from the left and >= 0 from the right

            # Actually, let's use a simpler approach: ternary search since f is convex
            # f(x) = max(max_k(base_k + L_k*x + Q_k*x^2), unaffected_max)

            def f_1d(x):
                vals = base + coeff_lin * x + coeff_quad * x * x
                return max(vals.max(), unaffected_max)

            # Golden section search
            gr = (np.sqrt(5) + 1) / 2
            a_g = x_lo
            b_g = x_hi
            c_g = b_g - (b_g - a_g) / gr
            d_g = a_g + (b_g - a_g) / gr
            fc = f_1d(c_g)
            fd = f_1d(d_g)

            for _ in range(60):  # 60 iterations gives precision ~ (1/phi)^60 ~ 1e-13
                if fc < fd:
                    b_g = d_g
                    d_g = c_g
                    fd = fc
                    c_g = b_g - (b_g - a_g) / gr
                    fc = f_1d(c_g)
                else:
                    a_g = c_g
                    c_g = d_g
                    fc = fd
                    d_g = a_g + (b_g - a_g) / gr
                    fd = f_1d(d_g)

            x_opt = (a_g + b_g) / 2
            x_opt = max(0.0, x_opt)

            if abs(x_opt - old_val) > 1e-12:
                a[j] = x_opt
                improved = True
                # Update conv
                conv = fast_convolve(a)
                new_max = conv.max()
                if new_max < best_max - 1e-15:
                    best_max = new_max
                    best_a = a.copy()
                else:
                    # Roll back
                    a[j] = old_val

        if not improved:
            break

    # Renormalize
    s = a.sum()
    if s < 1e-15:
        s = best_a.sum()
        a = best_a.copy()
        if s < 1e-15:
            return a0.copy()
    a = a / s
    return a


def nesterov_lp_refine(a0, time_limit, n_max_iters=500000):
    """Nesterov-accelerated LP alternating with adaptive step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)

        lp_time = time.monotonic() - lp_t0

        if b_new is None:
            continue

        avg_lp_time = np.mean(lp_times[-20:]) if len(lp_times) >= 10 else 0.0

        # Adaptive step size: use geometric mean of recent LP times
        # Faster LP -> can use larger steps
        alpha = 2.0 / (it + 2)

        # Nesterov dual averaging
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        # Stall detection
        if (it + 1) % 100 == 0:

                stall_count += 1

            if stall_count >= 8:

            print(f"    LP too slow ({avg_lp_time:.2f}s/iter), stopping at iter {it+1}")

    return best_a, best_score


def logsumexp_obj_grad(z, n, T_temp):
    """Compute LogSumExp objective and gradient w.r.t. log-domain z."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        a = np.ones(n) / n
    else:
        a = exp_z / s

    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse

    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage=3000):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    iters_per_stage = max(10, maxiter_per_stage // len(T_stages))

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        def func(zv):
            if time.monotonic() > deadline:
                return 1e15
            obj, _ = logsumexp_obj_grad(zv, n, stage_T)
            return obj

        def jac(zv):
            if time.monotonic() > deadline:
                return np.zeros_like(zv)
            _, g = logsumexp_obj_grad(zv, n, stage_T)
            return g

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': iters_per_stage * 500, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        a = np.ones(n) / n
    else:
        a = exp_z / s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def symmetrize(a):
    """Force symmetry: a[j] = a[n-1-j]."""
    return (a + a[::-1]) / 2.0


def structured_init(n, rng):
    """Create a structured initial sequence."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    else:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

        init_2000 = interpolate_sequence(best_seq, 2000)
        a_2000, sc_2000 = full_pipeline_for_n(init_2000, 2000, t_n2000)
        local_update(a_2000, "n2000")

    # === Phase 6: Perturbation escapes on best ===
    print(f"\n=== Phase 6: Perturbation escapes (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        for trial in range(4):
            rem_sub = deadline - time.monotonic()
            if rem_sub < 20:
                break
            x = np.linspace(0, 2 * np.pi, len(best_seq))
            freq = rng.integers(1, 8)
            phase = rng.uniform(0, 2 * np.pi)
            strength = 0.01 + 0.015 * trial
            perturbation = strength * np.cos(freq * x + phase)
            a_pert = np.maximum(best_seq + perturbation, 0)
            s = a_pert.sum()
            if s < 1e-15:
                continue
            a_pert = a_pert / s
            t_sub = min(rem_sub * 0.2, 80.0)
            a_opt, sc_opt = full_pipeline_for_n(a_pert, len(a_pert), t_sub)
            local_update(a_opt, f"pert{trial}")

    # === Phase 7: Final ultra-fine polish ===
    print(f"\n=== Phase 7: Final polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        # CD with many sweeps
        t_cd_final = min(rem * 0.4, 60.0)
        n_sweeps_final = max(5, int(t_cd_final / 2))
        a_cd = coordinate_descent_polish(best_seq, n_sweeps=n_sweeps_final)
        local_update(a_cd, "final_cd")

        rem = deadline - time.monotonic()
        if rem > 10:
            # Ultra-fine L-BFGS
            T_stages = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
            a_lb = lbgfs_refine(best_seq, min(rem * 0.5, 30.0), T_stages, maxiter_per_stage=1000)
            local_update(a_lb, "final_lbgfs")

            rem = deadline - time.monotonic()
            if rem > 10:
                a_cd2 = coordinate_descent_polish(best_seq, n_sweeps=3)
                local_update(a_cd2, "final_cd2")

    # Final verification with evaluate_sequence
    if best_seq is not None:
        final_eval = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
        if np.isfinite(final_eval) and final_eval < best_score:
            best_score = final_eval
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
        # Also try with evaluate_sequence to make sure
        print(f"  evaluate_sequence confirms: {final_eval:.10f}")

    if best_seq is None:
        best_seq = np.ones(1200) / 1200

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
