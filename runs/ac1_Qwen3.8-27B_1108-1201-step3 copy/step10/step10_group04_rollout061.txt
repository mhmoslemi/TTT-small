```python
def fast_convolve(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))

def project_simplex(v):
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

        try:

            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
```

```python
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- LP best-response iteration with momentum (main workhorse, using HiGHS)
- Subgradient mirror descent (complementary method)
- LogSumExp L-BFGS for final polishing

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    v_sum = v.sum()
    if v_sum < 1e-15:
        return np.ones(n) / n
    v = v / v_sum  # Normalize first
    # Actually, project onto sum=1, v>=0
    # Standard algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def build_conv_structure(n):
    """Precompute the row/col indices for the convolution matrix."""
    size = 2 * n - 1
    rows = np.empty(n * (n + 1), dtype=np.int32)  # Upper bound on nnz
    cols = np.empty(n * (n + 1), dtype=np.int32)
    idx = 0
    for k in range(size):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        cnt = j_end - j_start + 1
        rows[idx:idx+cnt] = k
        cols[idx:idx+cnt] = np.arange(j_start, j_end + 1)
        idx += cnt
    return rows[:idx], cols[:idx], (size, n)

    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    res = linprog(c, A_ub=A_ub, b_ub=np.zeros(size),
                  A_eq=A_eq, b_eq=b_eq,

    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(res.x[n])
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP best-response with HiGHS (main workhorse, ~5ms/iter for n=1200)
2. Subgradient mirror descent (for escaping local minima)
3. LogSumExp L-BFGS (for final polishing)
4. Multi-length exploration (n=800 to n=2000)
Key: exploits LP structure for fast best-response, momentum + adaptive steps.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    v_sum = v.sum()
    if v_sum < 1e-15:
        return np.ones(n) / n
    if v_sum != 1.0:
        v = v / v_sum
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    result = np.maximum(v - theta, 0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(n) / n
    return result

def build_conv_structure(n):
    size = 2 * n - 1
    rows_list = []
    cols_list = []
    for k in range(size):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows_list.append(k)
            cols_list.append(j)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)

class LPBestResponse:
    def __init__(self, n):
        self.n = n
        self.size = 2 * n - 1
        self.T_row, self.T_col = build_conv_structure(n)
        self.A_ub_template = np.zeros((self.size, n + 1))
        self.A_ub_template[:, n] = -1.0
        self.c = np.zeros(n + 1)
        self.c[n] = 1.0
        self.A_eq = np.zeros((1, n + 1))
        self.A_eq[0, :n] = 1.0
        self.b_eq = np.array([1.0])
        self.b_ub = np.zeros(self.size)
        self.bounds = [(0, None)] * n + [(-np.inf, None)]

    def solve(self, a_fixed):
        A_ub = self.A_ub_template.copy()
        A_ub[self.T_row, self.T_col] = a_fixed[self.T_col]
        res = linprog(self.c, A_ub=A_ub, b_ub=self.b_ub,
                      A_eq=self.A_eq, b_eq=self.b_eq,
                      bounds=self.bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                               'primal_feasibility_tolerance': 1e-9})
        if res.status != 0:
            return None, None
        b_val = res.x[:self.n].copy()
        b_val = np.maximum(b_val, 0)
        s = b_val.sum()
        if s < 1e-15:
            return None, None
        b_val = b_val / s
        return b_val, float(res.x[self.n])

def lp_best_response_iter(a0, time_limit, alpha=0.25, momentum=0.0, restart_every=2000, noise_scale=0.02):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    lp_solver = LPBestResponse(n)
    t0 = time.monotonic()
    it = 0
    rng = np.random.default_rng(12345)
    v = np.zeros(n)
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b, M_val = lp_solver.solve(a)
        if b is None:
            break

        if momentum > 0:
            v = momentum * v + (b - a)
            a_new = project_simplex(a + alpha * v)
        else:
            a_new = (1.0 - alpha) * a + alpha * b
            a_new = np.maximum(a_new, 0)
            ss = a_new.sum()
            if ss < 1e-15:
                a_new = a
            else:
                a_new = a_new / ss

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1

        a = a_new
        it += 1

        if stall >= restart_every:
            noise = rng.random(n) * noise_scale
            a = a + noise
            a = project_simplex(a)
            v = np.zeros(n)
            stall = 0
            alpha = max(0.05, alpha * 0.7)

        if it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-BR iter={it} ({elapsed:.0f}s): score={best_score:.6f} alpha={alpha:.3f}")

    return best_a, best_score

def subgradient_mirror_descent(a0, time_limit, T_start=1.0, T_end=0.1):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    z = np.log(np.maximum(a, 1e-30))
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.5

    for step in range(200000):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        active_thresh = max_b * 0.95
        active = np.where(b >= active_thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        g = np.zeros(n)
        for k in active:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                k_j = k - j
                if 0 <= k_j < n:
                    g[j] += 2.0 * a[k_j]
        g /= len(active)

        z_new = z - lr * g
        a_new = np.exp(z_new - z_new.max())
        a_new = a_new / a_new.sum()

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            z = z_new
            a = a_new
            lr = min(lr * 1.05, 2.0)
        else:
            lr = max(lr * 0.95, 0.01)
            if lr <= 0.011:
                noise = np.random.default_rng(step * 7).random(n) * 0.01
                a = a + noise
                a = project_simplex(a)
                z = np.log(np.maximum(a, 1e-30))
                lr = 0.3

        if step % 10000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGradMD iter={step} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score

def logsumexp_lbfgs(a0, time_limit, T_start=50.0, T_end=0.5, n_stages=8):
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30)).copy()
    t0 = time.monotonic()

    def lse_objective(z_in, T_val):
        a = np.exp(z_in - z_in.max())
        a = a / a.sum()
        b = fast_convolve(a)
        Tb = T_val * b
        max_Tb = Tb.max()
        if max_Tb > 500:
            max_Tb = 500
        exp_Tb = np.exp(Tb - max_Tb)
        log_sum_exp = max_Tb + np.log(exp_Tb.sum())
        return (2.0 * n / T_val) * log_sum_exp

    def lse_jacobian(z_in, T_val):
        a = np.exp(z_in - z_in.max())
        a = a / a.sum()
        b = fast_convolve(a)
        Tb = T_val * b
        max_Tb = Tb.max()
        if max_Tb > 500:
            max_Tb = 500
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-10:
                continue
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                k_j = k - j
                if 0 <= k_j < n:
                    grad_a[j] += 2.0 * w[k] * a[k_j]
        S = np.dot(grad_a, a)
        return a * (grad_a - S)

    temps = np.geomspace(T_start, T_end, n_stages)
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func(z_in):
            return lse_objective(z_in, stage_T)

        def jac(z_in):
            return lse_jacobian(z_in, stage_T)

        try:
            maxiter = max(10, int(min(50, remaining / 2.0)))
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-16,
                         'gtol': 1e-12, 'maxfun': maxiter * 200,
                         'disp': False}
            )
            z = res.x
        except Exception as e:
            print(f"    LSE-LBFGS stage {si} error: {e}")

    a_out = np.exp(z - z.max())
    a_out = a_out / a_out.sum()
    return a_out

def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp

def structured_init(n, rng):
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * np.linspace(0, 3, n))
    elif choice == 4:
        a = np.ones(n)
        a[rng.choice(n, n // 4, replace=False)] *= 0.5
    else:
        a = np.exp(-np.abs(np.linspace(-1, 1, n)) ** 1.5)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
```
