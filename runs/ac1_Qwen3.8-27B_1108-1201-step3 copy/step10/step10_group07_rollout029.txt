        M = conv.max()
        score = 2.0 * n * M

        if score < best_score - 1e-12:
            best_score = score
            best_p = p.copy()
            stall = 0
        else:
            stall += 1

        if stall > 300:
            lr *= 0.7
            stall = 0
            if lr < 1e-8:
                break

        # Find top peaks
        k_act = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        peaks = peaks[np.argsort(conv[peaks])[::-1]]

        # Softmax weights (sharper for more focus on the top peak)
        temp = 0.005 * max(M, 1e-10)
        pv = conv[peaks]
        w = np.exp((pv - pv[0]) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / ws

        # Subgradient via weighted sum of shifted p
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-13:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            sg[i_start:i_end + 1] += w[idx] * 2.0 * p[k - i_end:k - i_start + 1][::-1]

        # Mirror descent step: multiplicative weights
        log_p = np.log(np.maximum(p, 1e-30))
        log_p_new = log_p - lr * sg
        log_p_new -= log_p_new.max()
        p_new = np.exp(log_p_new)
        p_sum = p_new.sum()
        if p_sum < 1e-30:
            p_new = p.copy()
        else:
            p_new = p_new / p_sum

        # Check if this step improves (for adaptive lr)
        conv_new = fast_convolve(p_new)
        score_new = 2.0 * n * conv_new.max()

        if score_new > score * 1.05:
            # Step was too aggressive, backtrack
            lr *= 0.5
            # Try a smaller step
            log_p_new = log_p - lr * sg
            log_p_new -= log_p_new.max()
            p_new = np.exp(log_p_new) / np.exp(log_p_new).sum()
            conv_new = fast_convolve(p_new)
            score_new = 2.0 * n * conv_new.max()

        p = p_new
        conv = conv_new

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD[{label}] it={it} ({elapsed:.0f}s): best={best_score:.8f} lr={lr:.6f}")

    return best_p, best_score


def coord_descent(p, time_limit, verbose=True, label="", top_factors=None):
    """
    Coordinate descent with adaptive factor exploration.
    Solves the 1D convex subproblem for each coordinate.
    """
    n = len(p)
    p = np.maximum(p.astype(np.float64), 0.0).copy()
    s = p.sum()
    if s < 1e-15:
        p = np.ones(n) / n
    else:
        p = p / s

    conv = fast_convolve(p)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_p = p.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    if top_factors is None:
        top_factors = [0.0, 0.2, 0.4, 0.6, 0.75, 0.85, 0.92, 1.0,
                       1.08, 1.15, 1.25, 1.4, 1.6, 2.0, 3.0]

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if p[i] < 1e-16:
                # Try setting from zero
                if M > 0:
                    for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
                        new_val = p.sum() * frac / n
                        k_s = i
                        k_e = min(i + n - 1, 2 * n - 2)
                        p_shifted = p[k_s - i:k_e - i + 1]
                        new_aff = conv[k_s:k_e + 1] + 2.0 * new_val * p_shifted
                        idx = 2 * i - k_s
                        if 0 <= idx < len(new_aff):
                            new_aff = new_aff.copy()
                            new_aff[idx] += new_val * new_val
                        unaff = 0.0
                        if k_s > 0:
                            unaff = max(unaff, float(conv[:k_s].max()))
                        if k_e < 2 * n - 2:
                            unaff = max(unaff, float(conv[k_e + 1:].max()))
                        new_M = max(unaff, float(new_aff.max()))
                        if new_M < M - 1e-13:
                            p[i] = new_val
                            M = new_M
                            conv[k_s:k_e + 1] = new_aff
                            improved = True
                            break

                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            # Unaffected max
            unaff_max = 0.0
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            p_shifted = p[k_start - i:k_end - i + 1]
            conv_aff = conv[k_start:k_end + 1]

            best_val = p[i]
            best_M = M

            for factor in top_factors:
                new_val = p[i] * factor
                if new_val < 0:
                    continue

                delta = new_val - p[i]
                new_aff = conv_aff + 2.0 * delta * p_shifted
                idx = 2 * i - k_start
                if 0 <= idx < len(new_aff):
                    new_aff = new_aff.copy()
                    new_aff[idx] = conv[2 * i] + 2.0 * delta * p[i] + delta * delta

                new_M = max(unaff_max, float(new_aff.max()))
                if new_M < best_M - 1e-14:
                    best_M = new_M
                    best_val = new_val

            if best_val != p[i] and best_M < M - 1e-14:
                delta = best_val - p[i]
                p[i] = best_val
                changes = 2.0 * delta * p_shifted
                idx = 2 * i - k_start
                if 0 <= idx < len(conv[k_start:k_end + 1]):
                    changes = changes.copy()
                    changes[idx] = 2.0 * delta * (p[i] - delta) + delta * delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_p = p.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 15:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD[{label}] sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f} M={M:.10f}")

    return best_p, best_score


def fine_polish(p, time_limit, verbose=True, label=""):
    """Very fine coordinate descent for last-mile improvement."""
    n = len(p)
    p = np.maximum(p.astype(np.float64), 0.0).copy()
    s = p.sum()
    if s < 1e-15:
        p = np.ones(n) / n
    else:
        p = p / s

    conv = fast_convolve(p)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_p = p.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    factors = [0.995, 0.997, 0.998, 0.999,
               1.001, 1.002, 1.003, 1.005]

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if p[i] < 1e-14:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            unaff_max = 0.0
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            p_shifted = p[k_start - i:k_end - i + 1]
            conv_aff = conv[k_start:k_end + 1]

            best_val = p[i]
            best_M = M

            for factor in factors:
                new_val = p[i] * factor
                delta = new_val - p[i]
                new_aff = conv_aff + 2.0 * delta * p_shifted
                idx = 2 * i - k_start
                if 0 <= idx < len(new_aff):
                    new_aff = new_aff.copy()
                    new_aff[idx] = conv[2 * i] + 2.0 * delta * p[i] + delta * delta
                new_M = max(unaff_max, float(new_aff.max()))
                if new_M < best_M - 1e-15:
                    best_M = new_M
                    best_val = new_val

            if best_val != p[i] and best_M < M - 1e-15:
                delta = best_val - p[i]
                p[i] = best_val
                changes = 2.0 * delta * p_shifted
                idx = 2 * i - k_start
                if 0 <= idx < (k_end - k_start + 1):
                    changes = changes.copy()
                    changes[idx] = 2.0 * delta * (p[i] - delta) + delta * delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_p = p.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 10:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP[{label}] sweep={sweeps} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_p, best_score


def interpolate_sequence(p, new_n):
    """Interpolate a probability vector to a new length."""
    n = len(p)
    if new_n == n:
        return p.copy()
    x_old = np.linspace(0, 1, n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    q = np.interp(x_new, x_old, p)
    q = np.maximum(q, 0)
    s = q.sum()
    if s < 1e-15:
        q = np.ones(new_n) / new_n
    else:
        q = q / s
    return q


def random_init(n, rng, init_type=0):
    """Generate diverse initializations."""
    if init_type == 0:
        # U-shape (edges high)
        t = np.arange(n)
        center = (n - 1) / 2.0
        x = (t - center) / center
        a = 0.2 + 0.8 * x * x
    elif init_type == 1:
        # Cosine window
        t = np.arange(n)
        center = (n - 1) / 2.0
        x = (t - center) / center
        a = 0.5 * (1 + np.cos(np.pi * x)) ** 2
    elif init_type == 2:
        # W-shape
        t = np.arange(n)
        a = 0.3 + 0.7 * (1 - np.cos(4 * np.pi * t / n)) / 2
    elif init_type == 3:
        # Two humps
        c1 = n * rng.uniform(0.2, 0.35)
        c2 = n * rng.uniform(0.65, 0.8)
        w = n * rng.uniform(0.05, 0.15)
        t = np.arange(n)
        a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    elif init_type == 4:
        # Periodic
        period = int(max(3, n // rng.integers(50, 200)))
        t = np.arange(n)
        a = 0.5 + 0.4 * np.cos(2 * np.pi * t / period)
    elif init_type == 5:
        # Random smooth
        freqs = rng.uniform(0.5, 5.0, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        t = np.arange(n)
        a = np.ones(n)
        for f, ph in zip(freqs, phases):
            a += 0.2 * np.cos(2 * np.pi * f * t / n + ph)
        a = np.maximum(a, 0)
    else:
        a = np.ones(n)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def propose_candidate():
    """
    Multi-scale optimization strategy:
    1. Polish parent (n=1024) with coordinate descent
    2. Try mirror descent from multiple inits at n=1024
    3. Interpolate to n=2048, optimize
    4. Interpolate to n=4096, optimize
    5. Fine polish the best result
    Returns the best sequence found.
    """
    deadline = time.monotonic() + 1785.0
    rng = np.random.default_rng(42)

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is not None:
        n_parent = len(parent)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    else:
        n_parent = 1024
        best_seq = random_init(n_parent, rng, 0)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default init: n={n_parent}, score={best_score:.8f}")

    def try_update(candidate, label):
        nonlocal best_seq, best_score
        candidate = np.maximum(candidate, 0.0)
        s = candidate.sum()
        if s < 0.01 or not np.isfinite(s):
            return
        candidate = candidate / s
        try:
            sc = float(evaluate_sequence(candidate.tolist()))
        except Exception:
            sc = float('inf')
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = candidate.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(candidate)}) ***")

    # === Phase 1: Polish parent with coordinate descent ===
    print(f"\n=== Phase 1: Polish parent n={n_parent} (best={best_score:.8f}) ===")
    t1 = 200.0
    t1_end = time.monotonic() + min(t1, deadline - time.monotonic() - 10)

    a_cd, s_cd = coord_descent(best_seq, t1_end - time.monotonic(), verbose=True, label="parent_cd")
    try_update(a_cd, "cd_parent")

    # === Phase 2: Mirror descent from various inits ===
    print(f"\n=== Phase 2: Mirror descent (best={best_score:.8f}) ===")
    t2_end = time.monotonic() + min(300.0, deadline - time.monotonic() - 10)

    # MD from best
    remaining = t2_end - time.monotonic()
    if remaining > 30:
        t_alloc = min(remaining * 0.5, 120.0)
        a_md, s_md = mirror_descent(best_seq, t_alloc, top_k=30, verbose=True, lr_init=0.2, label="best")
        try_update(a_md, "md_best")
        # Quick CD after MD
        if time.monotonic() < t2_end - 20:
            a_cd2, s_cd2 = coord_descent(a_md, min(t2_end - time.monotonic(), 60.0),
                                         verbose=False, label="md_cd")
            try_update(a_cd2, "md_cd")

    # MD from diverse inits
    for init_type in [0, 2, 3, 5]:
        if time.monotonic() > t2_end - 30:
            break
        init = random_init(n_parent, rng, init_type)
        t_alloc = min(t2_end - time.monotonic(), 80.0)
        if t_alloc < 15:
            break
        a_md2, s_md2 = mirror_descent(init, t_alloc, top_k=20, verbose=False, lr_init=0.3,
                                      label=f"init{init_type}")
        try_update(a_md2, f"md_init{init_type}")

    # === Phase 3: Interpolate to n=2048 and optimize ===
    print(f"\n=== Phase 3: n=2048 (best={best_score:.8f}) ===")
    n2 = 2048
    t3_end = time.monotonic() + min(500.0, deadline - time.monotonic() - 50)

    # Interpolate best to n2
    q2 = interpolate_sequence(best_seq, n2)
    sc_q2 = float(evaluate_sequence(q2.tolist()))
    print(f"  Interpolated to n={n2}: score={sc_q2:.8f}")
    try_update(q2, "interp_2048")

    # Mirror descent on n=2048
    if time.monotonic() < t3_end - 60:
        t_alloc = min(t3_end - time.monotonic() - 60, 180.0)
        a_md3, s_md3 = mirror_descent(q2, t_alloc, top_k=30, verbose=True, lr_init=0.15, label="2048")
        try_update(a_md3, "md_2048")

    # CD on n=2048
    if time.monotonic() < t3_end - 30:
        t_alloc = min(t3_end - time.monotonic(), 200.0)
        a_cd3, s_cd3 = coord_descent(best_seq if len(best_seq) == n2 else q2,
                                     t_alloc, verbose=True, label="2048_cd")
        try_update(a_cd3, "cd_2048")

    # Second round MD on n=2048 from best
    if time.monotonic() < t3_end - 30:
        t_alloc = min(t3_end - time.monotonic(), 100.0)
        if len(best_seq) == n2:
            a_md4, s_md4 = mirror_descent(best_seq, t_alloc, top_k=40, verbose=True,
                                          lr_init=0.1, label="2048_r2")
            try_update(a_md4, "md_2048_r2")

    # === Phase 4: Interpolate to n=4096 and optimize ===
    print(f"\n=== Phase 4: n=4096 (best={best_score:.8f}) ===")
    n3 = 4096
    t4_end = time.monotonic() + min(500.0, deadline - time.monotonic() - 50)

    # Interpolate best (preferably n=2048) to n=4096
    base_for_interp = best_seq if len(best_seq) == n2 else interpolate_sequence(best_seq, n2)
    q3 = interpolate_sequence(base_for_interp, n3)
    sc_q3 = float(evaluate_sequence(q3.tolist()))
    print(f"  Interpolated to n={n3}: score={sc_q3:.8f}")
    try_update(q3, "interp_4096")

    # Mirror descent on n=4096
    if time.monotonic() < t4_end - 60:
        t_alloc = min(t4_end - time.monotonic() - 60, 200.0)
        a_md5, s_md5 = mirror_descent(q3, t_alloc, top_k=30, verbose=True, lr_init=0.1, label="4096")
        try_update(a_md5, "md_4096")

    # CD on n=4096 (fewer sweeps, longer per sweep)
    if time.monotonic() < t4_end - 30:
        t_alloc = min(t4_end - time.monotonic(), 200.0)
        cur = best_seq if len(best_seq) == n3 else q3
        a_cd4, s_cd4 = coord_descent(cur, t_alloc, verbose=True, label="4096_cd")
        try_update(a_cd4, "cd_4096")

    # === Phase 5: Fine polish the best ===
    print(f"\n=== Phase 5: Fine polish (best={best_score:.8f}, n={len(best_seq)}) ===")
    t5_end = deadline - 30

    if time.monotonic() < t5_end:
        t_alloc = min(t5_end - time.monotonic(), 200.0)
        a_fp, s_fp = fine_polish(best_seq, t_alloc, verbose=True, label="final")
        try_update(a_fp, "fine_polish")

    # One more CD pass
    if time.monotonic() < deadline - 20:
        t_alloc = min(deadline - time.monotonic() - 20, 60.0)
        a_cd5, s_cd5 = coord_descent(best_seq, t_alloc, verbose=False, label="final_cd")
        try_update(a_cd5, "final_cd")

    # Final fine polish
    if time.monotonic() < deadline - 10:
        t_alloc = min(deadline - time.monotonic() - 10, 30.0)
        a_fp2, s_fp2 = fine_polish(best_seq, t_alloc, verbose=False, label="final_fp")
        try_update(a_fp2, "final_fp")

    # Verify
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.10f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
