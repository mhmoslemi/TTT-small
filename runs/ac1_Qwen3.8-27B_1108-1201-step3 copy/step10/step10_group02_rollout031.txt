Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a dual QP approach.
The key innovation is reformulating the problem as a convex optimization
over the dual variable p (a probability distribution on convolution indices),
solving the inner QP via Hankel matrix inversion, and using adaptive
gradient descent on the dual simplex. Combined with LP refinement and
coordinate descent polishing.
"""

import time
from scipy.fft import rfft, irfft

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def project_to_simplex(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def build_hankel(p, n):
    """Build n×n Hankel matrix Q[i,j] = p[i+j]."""
    Q = np.empty((n, n), dtype=np.float64)
    for j in range(n):
        Q[:, j] = p[j:j+n]
    return Q

def dual_qp_optimize(a0, time_limit, n_inner_steps=2000):
    """
    Dual QP optimization: iterate on p to minimize 1^T Q(p)^{-1} 1.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = evaluate_local(a)

    sigma = n / 3.0
    p = np.exp(-0.5 * ((k_idx - (n-1)) / sigma)**2)
    p = p / p.sum()
    
    # Adaptive parameters
    eta = 0.05
    eta_decay = 0.9995
    eta_min = 0.001
    no_improve = 0
    
    for it in range(n_inner_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if time.monotonic() - t0 > time_limit - 2:
            break
        
        # Build Q
        Q = build_hankel(p, n)
        
        # Add regularization for stability
        # Check condition number (approximately)
        try:
            L = np.linalg.cholesky(Q + 1e-10 * np.eye(n))
            # Solve Qv = 1
            one = np.ones(n)
            Ly = np.linalg.solve(L, one)
            v = np.linalg.solve(L.T, Ly)
        except np.linalg.LinAlgError:

        # To minimize g, we do p += eta * (v*v) (since grad is negative)

        p_new = p + eta * denom
        p = project_to_simplex(p_new)
        
        eta = max(eta_decay * eta, eta_min)
        
        if (it + 1) % 100 == 0:

        improved = False
        # Try reducing the top contributor
        top_pos = idx[np.argmax(contribs)]
        
        for factor in [0.999, 0.998, 0.995, 0.99, 0.98]:
            a_trial = a.copy()
            a_trial[top_pos] *= factor
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            score = evaluate_local(a_trial)
            if score < best_score:
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True
                break
        
        if not improved:
            # Try increasing a non-peak-contributing position
            non_peak = np.setdiff1d(np.arange(n), idx)
            if len(non_peak) > 0:
                # Find position with smallest contribution to peak
                pos = non_peak[np.argmin(a[non_peak])]
                for factor in [1.005, 1.01, 1.02]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    a_trial = a_trial / a_trial.sum()
                    score = evaluate_local(a_trial)
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
        
        if not improved:
            stall += 1
            if stall > 500:
                break
        else:
            stall = 0
        
        if (step + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score
