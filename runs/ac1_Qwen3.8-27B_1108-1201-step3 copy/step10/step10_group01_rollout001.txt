    # d/d a[i] [(a*a)_k] = a[k-i] + a[k-i] = 2*a[k-i] (for most i, with correction at 2i=k)
    # Actually: d/d a[i] [sum_j a[j] * a[k-j]] 
    # = sum_j [δ(j-i) * a[k-j] + a[j] * δ(k-j-i)]
    # = a[k-i] + a[k-i]  (if both k-i in [0,n) and i in [0,n))
    # = 2*a[k-i] when 0 <= k-i < n
    # But when i = k-i (i.e., 2i = k), we count the same term twice, so it's just a[k-i] = a[i].
    # For the purpose of the gradient, the correction at 2i=k is negligible when n is large.
    # So approximately: grad[i] = 2 * sum_k w[k] * a[k-i] = 2 * (w ⋆ a)[i] (cross-correlation)
    
    # Using FFT: cross_correlation(w, a)[i] = IFFT(FFT(w) * conj(FFT(a)))[i]
    W = rfft(w, fft_size)
    A_fft = rfft(a, fft_size)
    corr = irfft(W * np.conj(A_fft), fft_size)[:n]
    return 2.0 * corr

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def conv_gradient_weighted(a, w):
    """Gradient of sum_k w[k]*(a*a)[k] w.r.t. a.
    ≈ 2 * cross_correlation(w, a) for the auto-conv case."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(W * np.conj(A), fft_size)[:n]
    return 2.0 * corr

def evaluate_local(a):

    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def get_lp_solver():
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x[0] >= 0.3])
            prob.solve(solver=solver_name, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

def solve_lp(a_fixed, solver=None, eps=1e-8):
    """Solve: min M s.t. (a*b)_k <= M, b>=0, sum(b)=1.
    Returns (b, M)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    size = 2 * n - 1
    # Build T matrix: (T @ b)[k] = (a * b)[k]
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a_fixed
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    if solver is None:
        solver = get_lp_solver()
    try:
        prob.solve(solver=solver, verbose=False)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0)
            s = b_val.sum()
            if s < 1e-15:
                return None, None
            return b_val / s, float(M_var.value)
    except Exception:
        pass
    # Fallback
    for alt in [cp.SCS, cp.ECOS, cp.CLARABEL, cp.PDLP]:
        if alt == solver:
            continue
        try:
            prob.solve(solver=alt, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0)
                s = b_val.sum()
                if s < 1e-15:
                    continue
                return b_val / s, float(M_var.value)
        except Exception:
            continue
    return None, None

def subgradient_momentum(a0, time_limit, n_steps=100000, top_k=10, 
                          initial_lr=0.05, decay=0.9995, momentum=0.9):
    """Subgradient descent with momentum on the top-k peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    v = np.zeros(n)  # momentum
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = initial_lr
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Top-k peaks with weights
        top_idx = np.argsort(b)[::-1][:top_k]
        top_vals = b[top_idx]
        # Softmax weighting
        w_top = np.exp(top_vals - top_vals[0])
        w_top /= w_top.sum()
        
        # Weighted gradient
        g = np.zeros(n)
        for ki, wi in zip(top_idx, w_top):
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            idx = np.arange(js, je + 1)
            g[idx] += wi * a[ki - idx]
        
        # Momentum update
        v = momentum * v - lr * g
        a_new = a + v
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / sn
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            # Try the update anyway (momentum might help)
            a = a_new
        
        # Adaptive learning rate
        if no_improve >= 30:
            lr *= 0.9
            v *= 0.5  # reduce momentum on stall
            no_improve = 0
        
        if step % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGradMo {elapsed:.0f}s: score={best_score:.8f} lr={lr:.6f}")
    
    return best_a, best_score

def logsumexp_lbgfs(a0, time_limit, T_stages=None):
    """L-BFGS with LogSumExp smoothing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [10.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]
    
    n_stages = len(T_stages)
    
    for T_stage in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        iters = max(20, min(5000, int(remaining * 3)))
        
        def func(z):
            a = np.exp(z - z.max())
            a = a / a.sum()
            b = fast_convolve(a)
            max_b = b.max()
            exp_b = np.exp(T_stage * (b - max_b))
            lse = max_b + np.log(exp_b.sum()) / T_stage
            return 2.0 * n * lse
        
        def jac(z):
            a = np.exp(z - z.max())
            a = a / a.sum()
            b = fast_convolve(a)
            max_b = b.max()
            exp_b = np.exp(T_stage * (b - max_b))
            w = exp_b / exp_b.sum()
            # grad of 2n * lse w.r.t. a
            g_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                g_a[idx] += 2.0 * w[k] * a[k - idx]
            # grad of lse w.r.t. z: d/dz [lse] = d/dz [f(a)] * da/dz
            # a = softmax(z), so da_i/dz_j = a_i(δ_ij - a_j)
            # d f / d z_j = sum_i (df/da_i) * a_i * (δ_ij - a_j)
            # = a_j * (df/da_j - sum_i df/da_i * a_i)
            S = np.dot(g_a, a)
            grad_z = a * (g_a - S)
            return 2.0 * n * grad_z  # Wait, I already included 2n in g_a
        
        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                          options={'maxiter': iters, 'ftol': 1e-18, 
                                   'gtol': 1e-14, 'maxfun': iters * 200, 'disp': False})
            z = res.x
        except Exception:
            pass
        
    a_out = np.exp(z - z.max())
    a_out = a_out / a_out.sum()
    return a_out

def spectral_flatten(a0, time_limit, n_steps=5000, alpha_init=0.3):
    """Spectral flattening: push |DFT(a)|^2 toward uniform."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    alpha = alpha_init
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute power spectrum
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        A = rfft(a, fft_size)
        P = A.real**2 + A.imag**2  # |A|^2
        
        # Target: flatten P toward its mean
        P_mean = P.mean()
        # Modified flatten: compress dynamic range
        P_new = P - alpha * (P - P_mean)
        P_new = np.maximum(P_new, P_mean * 0.1)  # don't go too negative
        
        # Now we need a_new such that |DFT(a_new)|^2 ≈ P_new, a_new >= 0, sum(a_new) = 1
        # Use "amaas" (all at once magnitude and phase): keep phase, update magnitude
        phase = np.angle(A)
        magnitude_new = np.sqrt(P_new)
        A_new = magnitude_new * np.exp(1j * phase)
        a_new = irfft(A_new, fft_size)[:n]
        
        # Project onto simplex (non-negative, sum=1)
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / sn
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            a_half = (a + a_new) / 2
            a_half = np.maximum(a_half, 0)
            a_half = a_half / a_half.sum()
            score_half = 2.0 * n * fast_convolve(a_half).max()
            if score_half < best_score - 1e-14:
                best_score = score_half
                best_a = a_half.copy()
                a = a_half
        
        # Decay alpha
        alpha *= 0.999
        
        if step % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Spectral {elapsed:.0f}s: score={best_score:.8f} alpha={alpha:.4f}")
    
    return best_a, best_score

def nesterov_lp_optimize(a0, time_limit, solver=None, max_iters=50000, stall_patience=15):
    """Nesterov-accelerated LP alternation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall = 0
    rng = np.random.default_rng(42)
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, M_b = solve_lp(y, solver)
        if b_new is None:
            stall += 1
            if stall > 30:
                break
            continue
        
        # Nesterov dual averaging
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy
        
        # Primal iterated average
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        if (it + 1) % 500 == 0:
            if best_score < prev_best - 1e-9:
                stall = 0
            else:
                stall += 1
            prev_best = best_score
            
            if stall >= stall_patience:
                noise = rng.standard_normal(n) * 0.02
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall = 0
    
    return best_a, best_score

def peak_shave(a0, time_limit, n_steps=50000, n_peaks=5, n_contrib=10):
    """Aggressive peak shaving: reduce top contributors to each peak."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Get top peaks
        top_peaks = np.argsort(b)[::-1][:n_peaks]
        
        improved = False
        for pk in top_peaks:
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[pk - idx]
            # Sort by contribution
            order = np.argsort(contribs)[::-1][:n_contrib]
            
            for oi in order:
                pos = idx[oi]
                if a[pos] < 1e-12:
                    continue
                # Try reducing this position
                for factor in [0.99, 0.985, 0.98, 0.97, 0.95]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    delta = a[pos] * (1.0 - factor)
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others_sum = a_trial[mask].sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] += delta * a_trial[mask] / others_sum
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Simultaneous reduction
            pk = top_peaks[0]
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[pk - idx]
            top = idx[np.argsort(contribs)[::-1][:3]]
            a_trial = a.copy()
            for pos in top:
                if a[pos] > 1e-12:
                    a_trial[pos] *= 0.999
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                break
            a_trial /= st
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-13:
                a = a_trial
                best_score = score
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
                if stall > 500:
                    break
    
    return best_a, best_score

def structured_init(n, rng):
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.05
    elif choice == 5:
        a = np.exp(-3 * (x - 0.25) ** 2) + 0.4 * np.exp(-3 * (x - 0.75) ** 2)
    elif choice == 6:
        a = np.exp(-4 * (x - 0.4) ** 2) + 0.3
    else:
        a = (1 - x) ** 0.7 + 0.1
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
