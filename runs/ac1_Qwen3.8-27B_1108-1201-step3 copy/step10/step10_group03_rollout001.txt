from scipy.optimize import linprog
import time

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def local_eval(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def solve_best_response_lp(a_fixed, use_highs=True):
    """LP: given a_fixed (sum=1), find b>=0, sum(b)=1 min max_k conv(a_fixed,b)_k."""
    n = len(a_fixed)
    size = 2 * n - 1
    
    # Build conv matrix T: T[k,j] = a_fixed[k-j] if 0<=k-j<n
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a_fixed
    
    # LP: min M s.t. T@b - M <= 0, sum(b)=1, b>=0, M free
    # x = [M, b_0, ..., b_{n-1}]
    c = np.zeros(n + 1)
    c[0] = 1.0
    
    A_ub = np.zeros((size, n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = T
    b_ub = np.zeros(size)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([1.0])
    
    lb = [None] + [0.0] * n
    ub = [None] + [None] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=list(zip(lb, ub)), method='highs')
    
    if res.success:
        b_val = np.maximum(res.x[1:], 0)
        s = b_val.sum()
        if s > 1e-15:
            b_val = b_val / s
            return b_val, float(res.x[0])
    return None, None

def multiplicative_update(a0, time_limit, n_steps=100000):
    """Entropy-regularized (multiplicative) updates for flattening."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = 1.0 / n  # initial step size
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        M = b.max()
        
        # Find active peaks
        threshold = M * 0.99
        active = np.where(b >= threshold)[0]
        
        # Subgradient of mean active peaks
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)
        g *= 2.0
        
        # Multiplicative update
        a_new = a * np.exp(-eta * g)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        sc = 2.0 * n * fast_convolve(a_new).max()
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            eta = min(eta * 1.005, 2.0)
        else:
            eta *= 0.98
            # Still update a slightly
            a = 0.99 * a + 0.01 * a_new
            a = normalize(a)
        
        eta = max(eta, 1e-8)
    
    return best_a, best_score

def subgradient_polish(a0, time_limit, n_steps=50000):

    n = len(a0)

    t0 = time.monotonic()

        M = b.max()
        
        # All peaks within 0.1% of max
        threshold = M * 0.999
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)
        
        a_trial = a - lr * g
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st
        
        sc = 2.0 * n * fast_convolve(a_trial).max()
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
            lr = min(lr * 1.01, 0.05)
        else:
            lr *= 0.95
        lr = max(lr, 1e-10)
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
