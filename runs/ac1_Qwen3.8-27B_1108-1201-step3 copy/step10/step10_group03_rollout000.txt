def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def local_eval(a):

    if s < 0.01:
        return float('inf')
    a = a / s
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())

    if s < 1e-15:

def get_lp_solver():
    """Find a working LP solver."""

            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def solve_lp_best_response(a_fixed, solver, warm_start=None):

            g2[idx] += 2.0 * a[k - idx]
        g2 /= len(active2)
        
        g = 0.7 * g + 0.3 * g2
        
        # Projected gradient step with backtracking
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            a_trial = a - trial_lr * g
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st
            sc = 2.0 * n * fast_convolve(a_trial).max()
            if sc < best_score:
                best_score = sc
                best_a = a_trial.copy()
                a = a_trial
                no_improve = 0
                lr = min(trial_lr * 1.05, 0.1)
                break
        else:
            no_improve += 1
            if no_improve >= 10:
                lr *= 0.9
                no_improve = 0
        lr = max(1e-8, lr)
    
    return best_a, best_score


def logsumexp_lbgfs(a0, time_limit, T_stages=None):
    """L-BFGS on the LogSumExp-smoothed objective with log-parametrization."""
    n = len(a0)
    a = normalize(a0)
    a_clipped = np.maximum(a, 1e-300)
    z = np.log(a_clipped) - np.log(a_clipped.sum())
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [50.0, 20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
    
    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        max_iter = max(30, int(remaining * 25))
        
        def lse_obj(z):
            z_max = z.max()
            exp_z = np.exp(z - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_shift = b / T
            b_max = b_shift.max()
            lse = b_max + np.log(np.exp(b_shift - b_max).sum())
            return (2.0 * n) * lse
        
        def lse_grad(z):
            z_max = z.max()
            exp_z = np.exp(z - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            b_shift = b / T
            b_max = b_shift.max()
            exp_ws = np.exp(b_shift - b_max)
            w = exp_ws / exp_ws.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_cur[k - idx]
            S = np.dot(grad_a, a_cur)
            return a_cur * (grad_a - S)
        
        try:
            res = minimize(
                lse_obj, z, jac=lse_grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-20, 'gtol': 1e-14,
                         'maxfun': max_iter * 300, 'disp': False}
            )
            z = res.x
            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
            sc = 2.0 * n * fast_convolve(a_new).max()
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        except Exception:
            pass
    
    return best_a, best_score


def fourier_refine(a0, n_freq, time_limit):
    """Refine in the Fourier domain."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    fft_size = 1 << (2 * n - 1).bit_length()
    A = rfft(a, fft_size)
    
    n_free = 2 * n_freq + 1
    x0 = np.zeros(n_free)
    x0[0] = A[0].real
    for i in range(1, n_freq + 1):
        x0[2 * i - 1] = A[i].real
        x0[2 * i] = A[i].imag
    
    A_rest = A[n_freq + 1:].copy()
    
    def fourier_obj(x):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        A_new = np.zeros(fft_size // 2 + 1, dtype=complex)
        A_new[0] = x[0]
        for i in range(1, n_freq + 1):
            A_new[i] = x[2 * i - 1] + 1j * x[2 * i]
        A_new[n_freq + 1:] = A_rest
        a_candidate = irfft(A_new, fft_size)[:n]
        a_candidate = np.maximum(a_candidate, 0)
        s = a_candidate.sum()
        if s < 1e-10:
            return 1e15
        a_candidate = a_candidate / s
        b = fast_convolve(a_candidate)
        return 2.0 * n * b.max()
    
    try:
        res = minimize(
            fourier_obj, x0, method='Nelder-Mead',
            options={'maxiter': 3000, 'xatol': 1e-12, 'fatol': 1e-12, 'adaptive': True}
        )
        x_opt = res.x
        A_new = np.zeros(fft_size // 2 + 1, dtype=complex)
        A_new[0] = x_opt[0]
        for i in range(1, n_freq + 1):
            A_new[i] = x_opt[2 * i - 1] + 1j * x_opt[2 * i]
        A_new[n_freq + 1:] = A_rest
        a_candidate = irfft(A_new, fft_size)[:n]
        a_candidate = np.maximum(a_candidate, 0)
        s = a_candidate.sum()
        if s > 1e-10:
            a_candidate = a_candidate / s
            sc = 2.0 * n * fast_convolve(a_candidate).max()
            if sc < best_score:
                best_score = sc
                best_a = a_candidate.copy()
    except Exception:
        pass
    
    return best_a, best_score


def skewed_window_init(n, skew, width, rng):
    """Create a skewed window initialization."""
    x = np.linspace(0, 1, n)
    # Base window (raised cosine or similar)
    base = np.cos(np.pi * (x - 0.5)) ** 2
    base = np.maximum(base, 0)
    # Apply skew: multiply by exponential tilt
    tilt = np.exp(skew * (x - 0.5))
    a = base * tilt
    # Optionally add width control
    if width != 1.0:
        x_shifted = (x - 0.5) / width + 0.5
        mask = (x_shifted >= 0) & (x_shifted <= 1)
        base2 = np.zeros(n)
        base2[mask] = np.cos(np.pi * (x_shifted[mask] - 0.5)) ** 2
        a = base2 * np.exp(skew * (x - 0.5))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def diverse_init(n, rng):
    """Create diverse initializations."""
    x = np.linspace(0, 1, n)
    choice = rng.integers(0, 12)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 1.0 / (1.0 + np.exp(-50 * (x - 0.5)))
        a = a / (a.max() + 1e-15)
    elif choice == 7:
        a = 1.0 + 0.1 * np.cos(2 * np.pi * x) + 0.05 * np.cos(4 * np.pi * x)
    elif choice == 8:
        # Skewed raised cosine
        a = np.cos(np.pi * x / 2) ** 2 * np.exp(2.0 * (x - 0.5))
    elif choice == 9:
        # Asymmetric Gaussian mixture
        a = np.exp(-3 * (x - 0.35) ** 2) + 0.7 * np.exp(-3 * (x - 0.65) ** 2)
    elif choice == 10:
        # Power-law decay from one end
        a = (1 - x) ** 0.8 + 0.05
    else:
        # Random smooth function
        coeffs = rng.standard_normal(5)
        a = np.ones(n)
        for k, c in enumerate(coeffs):
            a += c * np.cos(2 * np.pi * (k + 1) * x)
        a = np.maximum(a, 0)
    
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def water_filling_refine(a0, time_limit, n_steps=20000):
    """Water-filling: iteratively reduce peaks by redistributing mass from peak-contributing positions."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        M = b.max()
        peak_k = int(np.argmax(b))
        
        # Find positions that contribute most to the peak
        # Contribution of position j to b[peak_k] is a[j] * a[peak_k - j]
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contributions = a[idx] * a[peak_k - idx]
        
        # Identify "excess" positions: those with high a but low contribution elsewhere
        # Strategy: reduce a at positions with high a and high contribution to peak
        total_contrib = contributions.sum()
        if total_contrib < 1e-15:
            break
        
        # Fraction of each position's contribution to the peak
        frac = contributions / total_contrib
        
        # Also look at second peak
        b_sorted = np.sort(b)[::-1]
        if len(b_sorted) > 1 and b_sorted[1] > 0.99 * M:
            second_peak = int(np.argmax(np.where(b >= 0.999 * M, b, -1)))
            if second_peak != peak_k:
                js2 = max(0, second_peak - n + 1)
                je2 = min(n - 1, second_peak)
                idx2 = np.arange(js2, je2 + 1)
                contributions2 = a[idx2] * a[second_peak - idx2]
                # Combine
                combined = np.zeros(n)
                combined[idx] = frac
                combined[idx2] += contributions2 / max(contributions2.sum(), 1e-15) * 0.5
        else:
            combined = np.zeros(n)
            combined[idx] = frac
        
        # Gradient step: reduce a where it contributes to peaks
        g = combined
        # Normalize gradient
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        g = g / g_norm
        
        # Adaptive step size
        for lr in [0.01, 0.005, 0.002, 0.001]:
            a_trial = a - lr * g
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-10:
                continue
            a_trial = a_trial / st
            sc = 2.0 * n
