This is more stable because:
- It focuses on the most important peaks
- It's less sensitive to noise in the smaller peaks
- The gradient is spread across K positions (more stable)

But it's non-smooth (the set of top-K changes discontinuously). We can smooth it by using a soft top-K:
f_{T,K,β}(a) = T * log(sum_k (softmax(conv/β)_k)^α * exp(conv[k]/T))

    """Generate a sequence using the Gerchberg-Saxton algorithm for flat spectrum."""
    if rng is None:

    # A[0] = sum(a) = 1

def logsumexp_adam(a0, time_limit, solver=None):
    """LogSumExp optimization with Adam, using log-parametrization."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Log-parametrization
    a_clip = np.maximum(a, 1e-20)
    z = np.log(a_clip)
    z = z - z.mean()  # center
    
    # Adam state
    m_z = np.zeros(n)
    v_z = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.1
    t_adam = 0
    
    # Temperature schedule
    T = 5.0
    T_min = 0.01
    T_decay = 0.999
    
    # Adaptive: lower T when improvement stalls
    no_improve = 0
    
    for step in range(500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        t_adam += 1
        
        # Compute a from z
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_cur = exp_z / exp_z.sum()
        
        # Compute conv and LogSumExp
        b = fast_convolve(a_cur)
        b_shift = b / T
        b_max = b_shift.max()
        exp_ws = np.exp(b_shift - b_max)
        w = exp_ws / exp_ws.sum()
        lse = T * (b_max + np.log(exp_ws.sum()))
        
        # Gradient w.r.t. a
        grad_a = np.zeros(n)
        # For efficiency, only process significant weights
        significant = w > 1e-10
        for k in np.where(significant)[0]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a_cur[k - idx]
        
        # Chain rule for soft
        S = np.dot(grad_a, a_cur)
        grad_z = a_cur * (grad_a - S)
        
        # Adam update
        m_z = beta1 * m_z + (1 - beta1) * grad_z
        v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
        m_hat = m_z / (1 - beta1 ** t_adam)
        v_hat = v_z / (1 - beta2 ** t_adam)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Evaluate
        sc = 2.0 * n * fast_convolve(a_cur).max()
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive temperature
        if no_improve > 200:
            T = max(T_min, T * T_decay ** 200)
            no_improve = 0
        
        if (step + 1) % 5000 == 0:
            print(f"    LSE-Adam step={step+1}: T={T:.4f}, score={best_score:.10f}")
    
    return best_a, best_score

        y_a = (1.0 - alpha) * y_a + alpha * b_new
        y_a = normalize(y_a)
        
        a = ((it + 2) * a + (alpha * (it + 3)) * b_new) / (it + 2 + alpha * (it + 3))
        a = normalize(a)
        
        sc = 2.0 * n * fast_convolve(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        sc_y = 2.0 * n * fast_convolve(y_a).max()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y_a.copy()
        
        if (it + 1) % 2000 == 0:
            if best_score < prev_best - 1e-9:
                stall = 0
            else:
                stall += 1
            prev_best = best_score
            
            if stall >= 5:
                noise = 0.01 / math.sqrt(stall)
                y_a = normalize(y_a + rng.standard_normal(n) * noise)
                stall = 0
            
            print(f"    BiDirLP iter={it+1}: score={best_score:.10f}")
    
    return best_a, best_score


def logsumexp_fft_optimize(a0, time_limit, extra_iters=0):
    """
    LogSumExp optimization with Adam, using FFT for O(n log n) gradients.
    Uses adaptive temperature: lowers T when progress stalls.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    # FFT size for cross-correlation: need (2n-1) + n - 1 = 3n-2
    corr_fft_size = 1 << (3 * n - 2 - 1).bit_length()
    
    # Log-parametrization
    a_clip = np.maximum(a, 1e-20)
    z = np.log(a_clip) - np.log(a_clip).mean()
    
    # Adam state
    m_z = np.zeros(n)
    v_z = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    lr = 0.5
    t_adam = 0
    
    # Adaptive temperature
    T = 3.0
    T_min = 0.005
    no_improve = 0
    total_steps = 0
    
    while time.monotonic() - t0 < time_limit:
        t_adam += 1
        total_steps += 1
        
        # Compute a from z (softmax)
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_cur = exp_z / exp_z.sum()
        
        # Compute convolution
        b = fast_convolve(a_cur)
        
        # Compute softmax weights for LogSumExp
        b_shift = b / T
        b_max = b_shift.max()
        exp_ws = np.exp(b_shift - b_max)
        w = exp_ws / exp_ws.sum()
        
        # FFT-accelerated gradient: grad_a[j] = 2 * sum_k w[k] * a[k-j]
        # This is the cross-correlation of w (length 2n-1) and a (length n)
        W = rfft(w, corr_fft_size)
        A = rfft(a_cur, corr_fft_size)
        cross = irfft(np.conj(W) * A, corr_fft_size)
        grad_a = 2.0 * cross[:n]
        
        # Chain rule for softmax parameterization
        S = np.dot(grad_a, a_cur)
        grad_z = a_cur * (grad_a - S)
        
        # Adam update
        m_z = b1 * m_z + (1 - b1) * grad_z
        v_z = b2 * v_z + (1 - b2) * grad_z ** 2
        m_hat = m_z / (1 - b1 ** t_adam)
        v_hat = v_z / (1 - b2 ** t_adam)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Evaluate
        sc = 2.0 * n * b.max()
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()
            no_improve = 0
            # Slightly increase lr on improvement
            lr = min(lr * 1.005, 2.0)
        else:
            no_improve += 1
        
        # Adaptive temperature and learning rate
        if no_improve > 500:
            T = max(T_min, T * 0.7)
            lr = max(0.01, lr * 0.8)
            no_improve = 0
        
        if total_steps % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-FFT step={total_steps} ({elapsed:.0f}s): T={T:.4f}, lr={lr:.4f}, score={best_score:.10f}")
    
    return best_a, best_score


def peak_shaving_lp(a0, solver, n_steps=100):
    """LP-based peak shaving: iteratively reduce the highest convolution peaks."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > 30:
            break
        
        b = fast_convolve(a)
        M = b.max()
        
        # Find positions near the peak
        threshold = M * 0.95
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        # Build LP: find direction d that reduces active peaks
        # Linearize: b[k] + 2 * sum_j a[k-j] * d[j] <= M * (1 - reduction)
        m_act = len(active)
        G = np.zeros((m_act, n))
        for i, k in enumerate(active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            G[i, idx] = 2.0 * a[k - idx]
        
        d_var = cp.Variable(n)
        reduction_var = cp.Variable()
        
        constraints = [
            cp.sum(d_var) == 0,
            a + d_var >= 0,
            G @ d_var <= -reduction_var * np.ones(m_act),
        ]
        
        prob = cp.Problem(cp.Maximize(reduction_var), constraints)
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                d_val = np.asarray(d_var.value, dtype=np.float64)
                for step_size in [1.0, 0.5, 0.25, 0.1]:
                    a_trial = a + step_size * d_val
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    sc = 2.0 * n * fast_convolve(a_trial).max()
                    if sc < best_score:
                        best_score = sc
                        best_a = a_trial.copy()
                        a = a_trial
                        break
        except Exception:
            break
    
    return best_a, best_score


def coordinate_descent_peaks(a0, time_limit):
    """Coordinate descent targeting the peak positions."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    b = fast_convolve(a)
    M = b.max()
    
    # Find a wide band of "important" indices (near peaks)
    peak_threshold = M * 0.8
    peak_positions = np.where(b >= peak_threshold)[0]
    
    # For each peak position k, the indices j that affect it are [max(0,k-n+1), min(n-1,k)]
    # Collect all such indices
    important_indices = set()
    for k in peak_positions:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            important_indices.add(j)
    important_list = list(important_indices)
    
    if len(important_list) == 0:
        important_list = list(range(n))
    
    # Limit to manageable size
    if len(important_list) > 500:
        # Sample uniformly
        important_list = list(rng.choice(important_list, 500, replace=False))
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        idx = important_list[it % len(important_list)]
        
        # Optimize a[idx]: find optimal value keeping sum=1
        # conv[k] for affected k is quadratic in a[idx]
        # For simplicity, try a small perturbation
        for delta_frac in [0.01, -0.01, 0.005, -0.005]:
            a_trial = a.copy()
            a_trial[idx] = max(0, a[idx] * (1 + delta_frac))
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st
            b_trial = fast_convolve(a_trial)
            sc = 2.0 * n * b_trial.max()
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a_trial.copy()
                a = a_trial
                break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Phase 1: Multi-n initialization with GS + structured + LP
    Phase 2: Bidirectional LP alternation
    Phase 3: FFT-accelerated LogSumExp Adam with adaptive temperature
    Phase 4: Peak shaving LP
    Phase 5: Final polish (low-T LSE + coordinate descent)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    
    best_seq = None
    best_score = float('inf')
    best_n = 0
    
    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = local_eval(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
            best_n = len(parent)
    except Exception:
        pass
    
    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, 0)
        best_score = local_eval(best_seq)
        best_n = n_init
        print(f"Default init: n={best_n}, score={best_score:.10f}")
    
    def try_update(a, label=""):
        nonlocal best_seq, best_score, best_n
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = local_eval(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            best_n = len(a)
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")
    
    # === Phase 1: Multi-n initialization ===
    print(f"\n=== Phase 1: Multi-n Init (current best: {best_score:.10f}, n={best_n}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.15, 200.0)
    if t1 > 30:
        n_candidates = set()
        # Include parent n and neighbors
        for mult in [0.6, 0.75, 0.85, 1.0, 1.1, 1.25, 1.5, 1.75, 2.0]:
            cand_n = int(best_n * mult)
            if 400 <= cand_n <= 5000:
                n_candidates.add(cand_n)
        # Also add some absolute values
        for an in [600, 800, 1000, 1200, 1500, 2000, 2500, 3000]:
            if 400 <= an <= 5000:
                n_candidates.add(an)
        
        n_candidates = sorted(n_candidates)
        print(f"  Trying n values: {n_candidates}")
        
        sub_budget = t1 / max(len(n_candidates), 1)
        for cand_n in n_candidates:
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            t_sub = min(sub_budget, rem * 0.5)
            if t_sub < 10:
                break
            
            # Try multiple initializations
            inits = []
            # GS init
            inits.append(gs_init(cand_n, n_iters=100, seed=cand_n))
            # Structured inits
            for si in range(3):
                inits.append(structured_init(cand_n, seed=cand_n + si * 100))
            # Interp from best
            if best_seq is not None and len(best_seq) != cand_n:
                inits.append(interpolate_sequence(best_seq, cand_n))
            
            for ii, init in enumerate(inits):
                rem = deadline - time.monotonic()
                if rem < 10:
                    break
                sc = local_eval(init)
                if sc < best_score:
                    print(f"    n={cand_n}, init{ii}: {sc:.10f}")
                    best_seq = init.copy()
                    best_score = sc
                    best_n = cand_n
                # Quick LP polish
                if rem > 15:
                    a_lp, _ = bidirectional_lp(init, min(10, rem * 0.3), solver)
                    try_update(a_lp, f"lp_n{cand_n}_i{ii}")
            
            print(f"  After n={cand_n}: best={best_score:.10f}")
    
    print(f"  Phase 1 done: best_score={best_score:.10f}, best_n={best_n}")
    
    # === Phase 2: Bidirectional LP on best ===
    print(f"\n=== Phase 2: Bidirectional LP (n={best_n}, score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.20, 300.0)
    if t2 > 30:
        print(f"  Budget: {t2:.0f}s")
        a_lp, sc_lp = bidirectional_lp(best_seq, t2, solver)
        try_update(a_lp, "bidi_lp_main")
    
    # === Phase 3: FFT LogSumExp Adam ===
    print(f"\n=== Phase 3: LSE-FFT Adam (n={best_n}, score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.35, 500.0)
    if t3 > 30:
        print(f"  Budget: {t3:.0f}s")
        a_lse, sc_lse = logsumexp_fft_optimize(best_seq, t3)
        try_update(a_lse, "lse_fft")
        
        # Second round with lower starting T (modify initial T via a fresh call)
        rem = deadline - time.monotonic()
        if rem > 60:
            t3b = min(rem * 0.3, 200.0)
            print(f"  Second LSE-FFT round, budget: {t3b:.0f}s")
            a_lse2, sc_lse2 = logsumexp_fft_optimize(best_seq, t3b)
            try_update(a_lse2, "lse_fft_2")
    
    # === Phase 4: Peak shaving LP ===
    print(f"\n=== Phase 4: Peak Shaving LP (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.10, 100.0)
    if t4 > 15:
        a_ps, sc_ps = peak_shaving_lp(best_seq, solver, n_steps=200)
        try_update(a_ps, "peak_shave")
    
    # === Phase 5: Final polish ===
    print(f"\n=== Phase 5: Final Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    
    if rem > 30:
        # Low-T LSE
        t5 = min(rem * 0.4, 150.0)
        if t5 > 20:
            a_fin, sc_fin = logsumexp_fft_optimize(best_seq, t5)
            try_update(a_fin, "final_lse")
        
        # Peak shaving again
        rem = deadline - time.monotonic()
        if rem > 20:
            a_ps2, _ = peak_shaving_lp(best_seq, solver, n_steps=100)
            try_update(a_ps2, "final_peak")
        
        # Try different n one more time
        rem = deadline - time.monotonic()
        if rem > 60:
            for mult in [0.9, 1.0, 1.1]:
                cand_n = int(best_n * mult)
                if cand_n == best_n:
                    continue
                if 400 <= cand_n <= 5000:
                    rem = deadline - time.monotonic()
                    if rem < 30:
                        break
                    init_new = interpolate_sequence(best_seq, cand_n)
                    a_new, _ = logsumexp_fft_optimize(init_new, min(rem * 0.3, 60.0))
                    try_update(a_new, f"polish_n{cand_n}")
    
    # === Final verification ===
    if best_seq is not None:
        fs = local_eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.10f}, n={len(best_seq)}")
        try:
            eval_score = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
    
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
