```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based LogSumExp alternating optimization (bilinear split)
2. LP best-response with Nesterov momentum (ADMM-style)
3. Multi-length exploration (n=960, 1200, 1500, 1800, 2400)
4. Targeted peak-reduction perturbations using LP dual info
5. Aggressive temperature annealing for fine convergence
6. Random restarts with structured perturbations
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT. Returns array of length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    """Linear cross-convolution of a and b via FFT. a has length n, b has length n."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def lse_selfconv_grad(a, T):
    """
    Compute LSE(conv(a,a); T) and its gradient w.r.t. a.
    LSE approximates max(conv(a,a)) as T -> 0.
    Returns (obj, grad) where obj = T*log(sum(exp(b/T))) and grad = d obj / da.
    """
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp((b - max_b) / T)
    sum_exp = exp_b.sum()
    w = exp_b / sum_exp
    lse_val = max_b + T * np.log(sum_exp)

    n = len(a)
    # grad[j] = 2 * sum_k w[k] * a[k-j] (a[m]=0 for m not in [0,n-1])
    # Use FFT: convolve w_reversed with a_padded
    w_rev = w[::-1]  # length 2n-1
    a_pad = np.zeros(2 * n - 1)
    a_pad[:n] = a

    fft_size = 1 << (len(w_rev) + len(a_pad) - 2).bit_length()
    Wr = rfft(w_rev, fft_size)
    Ap = rfft(a_pad, fft_size)
    conv_res = irfft(Wr * Ap, fft_size)[:len(w_rev) + len(a_pad) - 1]

    # grad[j] = 2 * conv_res[2n-2-j]
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * conv_res[idx]

    return lse_val, grad


def lse_crossconv_grad_wrt_b(a_fixed, b, T):
    """
    Compute LSE(conv(a_fixed, b); T) and its gradient w.r.t. b.
    Here conv is linear in b, so this is a convex optimization in b.
    """
    n = len(a_fixed)
    c = cross_convolve(a_fixed, b)  # length 2n-1
    max_c = c.max()
    exp_c = np.exp((c - max_c) / T)
    sum_exp = exp_c.sum()
    w = exp_c / sum_exp
    lse_val = max_c + T * np.log(sum_exp)

    # grad_b[j] = sum_k w[k] * a_fixed[k-j] (a_fixed[m]=0 for m not in [0,n-1])
    # This is the cross-correlation of w and a_fixed
    w_rev = w[::-1]  # length 2n-1
    a_pad = np.zeros(2 * n - 1)
    a_pad[:n] = a_fixed

    fft_size = 1 << (len(w_rev) + len(a_pad) - 2).bit_length()
    Wr = rfft(w_rev, fft_size)
    Ap = rfft(a_pad, fft_size)
    conv_res = irfft(Wr * Ap, fft_size)[:len(w_rev) + len(a_pad) - 1]

    # grad_b[j] = conv_res[2n-2-j]
    idx = 2 * n - 2 - np.arange(n)
    grad_b = conv_res[idx]

    return lse_val, grad_b


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(np.clip(z_shifted, -700, 0))
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def logsumexp_obj_grad_logdomain(z, n, T):
    """
    Compute LSE objective and gradient w.r.t. log-domain parameter z.
    a = softmax(z), obj = LSE(conv(a,a); T)
    """
    a = softmax_from_z(z)
    lse_val, grad_a = lse_selfconv_grad(a, T)
    # Chain rule: grad_z = a * (grad_a - sum(grad_a * a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return lse_val, grad_z


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.CLP, cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS, cp.GLOP, cp.CBC]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix_sparse(a):
    """Build sparse convolution matrix T such that T @ b = convolve(a, b)."""
    from scipy.sparse import lil_matrix
    n = len(a)
    size = 2 * n - 1
    T = lil_matrix((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T.tocsr()


def solve_lp_best_response(a_fixed, solver, max_time=30.0):
    """
    Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k].
    Returns (b_val, M_val) or (None, None) on failure.
    """
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix_sparse(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = []
    if solver:
        solvers_to_try.append(solver)
    for s in [cp.GUROBI, cp.MOSEK, cp.CLP, cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS, cp.GLOP, cp.CBC]:
        if s not in solvers_to_try:
            solvers_to_try.append(s)

    start_t = time.monotonic()
    for s in solvers_to_try:
        if time.monotonic() - start_t > max_time:
            break
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_val = b_val.sum()
                if s_val < 1e-15:
                    return None, None
                b_val = b_val / s_val
                return b_val, float(max(0, M_var.value))
        except Exception:
            continue
    return None, None


def evaluate_local(a):
    """Evaluate sequence score locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize_a(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def alternating_lse_optimize(a0, time_limit, T_stages, max_iter_per_stage):
    """
    Alternating LogSumExp optimization:
    - Fix a, optimize LSE(conv(a,b)) w.r.t. b using L-BFGS in log domain
    - Fix b, optimize LSE(conv(b,a)) w.r.t. a using L-BFGS in log domain
    - Average a and b at the end
    """
    n = len(a0)
    t0 = time.monotonic()
    deadline = t0 + time_limit

    a = normalize_a(a0.copy())
    b = a.copy()
    best_a = a.copy()
    best_score = evaluate_local(a)

    for T in T_stages:
        if time.monotonic() >= deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 3:
            break
        iters = max(5, min(max_iter_per_stage, int(remaining * 50)))

        for semi_iter in range(min(iters, 2000)):
            if time.monotonic() >= deadline:
                break

            # Optimize b given a: LSE(conv(a, b)) is convex in b
            z_b = z_from_a(b)

            def func_b(z):
                b_tmp = softmax_from_z(z)
                L, _ = lse_crossconv_grad_wrt_b(a, b_tmp, T)
                return L

            def jac_b(z):
                b_tmp = softmax_from_z(z)
                _, g_b = lse_crossconv_grad_wrt_b(a, b_tmp, T)
                S = np.dot(g_b, b_tmp)
                return b_tmp * (g_b - S)

            try:
                res = minimize(fun=func_b, x0=z_b, jac=jac_b, method='L-BFGS-B',
                               options={'maxiter': 50, 'ftol': 1e-18, 'gtol': 1e-16, 'disp': False})
                if res.status in (0, 1, 2):
                    b = softmax_from_z(res.x)
            except Exception:
                pass

            # Optimize a given b: LSE(conv(b, a)) is convex in a
            z_a = z_from_a(a)

            def func_a(z):
                a_tmp = softmax_from_z(z)
                L, _ = lse_crossconv_grad_wrt_b(b, a_tmp, T)
                return L

            def jac_a(z):
                a_tmp = softmax_from_z(z)
                _, g_a = lse_crossconv_grad_wrt_b(b, a_tmp, T)
                S = np.dot(g_a, a_tmp)
                return a_tmp * (g_a - S)

            try:
                res = minimize(fun=func_a, x0=z_a, jac=jac_a, method='L-BFGS-B',
                               options={'maxiter': 50, 'ftol': 1e-18, 'gtol': 1e-16, 'disp': False})
                if res.status in (0, 1, 2):
                    a = softmax_from_z(res.x)
            except Exception:
                pass

            # Check convergence
            if semi_iter % 20 == 0:
                avg = normalize_a(0.5 * a + 0.5 * b)
                sc = evaluate_local(avg)
                if sc < best_score:
                    best_score = sc
                    best_a = avg.copy()

    # Final: average a and b
    final = normalize_a(0.5 * a + 0.5 * b)
    final_score = evaluate_local(final)
    if final_score < best_score:
        best_score = final_score
        best_a = final.copy()

    return best_a, best_score


def lbgfs_logsumexp_refine(a0, time_limit, T_stages):
    """
    Direct L-BFGS on LogSumExp(self-convolution) in log domain.
    This is non-convex but can find good local minima.
    """
    n = len(a0)
    z = z_from_a(a0.copy())
    t0 = time.monotonic()
    deadline = t0 + time_limit
    best_a = a0.copy()
    best_score = evaluate_local(a0)
    best_z = z.copy()

    for T in T_stages:
        if time.monotonic() >= deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 3:
            break
        max_iter = max(20, min(500, int(remaining * 30)))

        def func(z_in):
            if time.monotonic() > deadline:
                return 1e15
            obj, _ = logsumexp_obj_grad_logdomain(z_in, n, T)
            return obj

        def jac(z_in):
            if time.monotonic() > deadline:
                return np.zeros_like(z_in)
            _, g = logsumexp_obj_grad_logdomain(z_in, n, T)
            return g

        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': max_iter, 'ftol': 1e-25, 'gtol': 1e-20,
                                    'maxfun': max_iter * 200, 'disp': False})
            if res.x is not None:
                new_a = softmax_from_z(res.x)
                sc = evaluate_local(new_a)
                if sc < best_score:
                    best_score = sc
                    best_a = new_a.copy()
                    best_z = res.x.copy()
                z = res.x.copy()
        except Exception:
            pass

    return best_a, best_score


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=500000):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    a = normalize_a(a0.copy())
    y = a.copy()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()

    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(12345)

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver, max_time=10.0)
        if b_new is None:
            break

        # Nesterov dual averaging
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = normalize_a(y)

        a = normalize_a(((it + 1) * a + (it + 2) * b_new) / (2 * it + 3))

        score_a = evaluate_local(a)
        score_y = evaluate_local(y)
        score_avg = evaluate_local(normalize_a(0.5 * a + 0.5 * b_new))

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        if score_avg < best_score:
            best_score = score_avg
            best_a = normalize_a(0.5 * a + 0.5 * b_new).copy()

        # Stall detection
        if (it + 1) % 50 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 8:
                # Perturb to escape
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 8)
                noise = 0.03 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = normalize_a(y + noise)
                stall_count = 0

            if (it + 1) % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def targeted_peak_reduction(a, threshold_frac=0.01, n_steps=10000, eta_base=0.02):
    """
    Multiplicative weights targeting active peaks.
    Reduces the maximum convolution value by penalizing contributions to peaks.
    """
    n = len(a)
    a = normalize_a(a.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    eta = eta_base
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > 300:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Active set: peaks within threshold of max
        threshold = max_b * (1.0 - threshold_frac)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        # Subgradient: average gradient over active peaks
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            indices = np.arange(js, je + 1)
            g[indices] += a[k - indices]
        g /= len(active)

        # Multiplicative update
        log_update = -eta * g
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = normalize_a(a_new)

        score = evaluate_local(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 50:
            eta *= 0.7
            no_improve = 0
            if eta < 1e-6:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    s_new = f_interp.sum()
    if s_new > 1e-15:
        f_interp = f_interp / s_new
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def perturbed_sequence(a, rng, strength=0.03):
    """Create a perturbed version of sequence a."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    perturbation = np.zeros(n)
    for _ in range(rng.integers(1, 5)):
        freq = rng.integers(1, 12)
        phase = rng.uniform(0, 2 * np.pi)
        amp = strength / 2
        perturbation += amp * np.sin(freq * x + phase)
    a_pert = a + perturbation
    return normalize_a(a_pert)


def mirrored_sequence(a):
    """Create a symmetric version of the sequence."""
    n = len(a)
    half_n = (n + 1) // 2
    a_sym = np.zeros(n)
    a_sym[:half_n] = a[:half_n]
    if n % 2 == 0:
        a_sym[half_n:] = a_sym[half_n-1::-1]
    else:
        a_sym[half_n:] = a_sym[half_n-1::-1]
    return normalize_a(a_sym)

        a_alt, _ = alternating_lse_optimize(best_seq, t_alloc, T_stages, max_iter_per_stage=3000)
        try_update(a_alt, "alt_lse_phase2")

    # === Phase 3: L-BFGS LogSumExp direct ===
    print(f"\n=== Phase 3: L-BFGS LSE direct (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 200.0)
    if t_alloc > 20:
        T_stages3 = [5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]
        print(f"  T_stages: {len(T_stages3)}, budget={t_alloc:.0f}s")
        a_lbg, _ = lbgfs_logsumexp_refine(best_seq, t_alloc, T_stages3)
        try_update(a_lbg, "lbgfs_phase3")

    # === Phase 4: Targeted peak reduction ===
    print(f"\n=== Phase 4: Targeted Peak Reduction (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.05, 80.0)
    if t_alloc > 15:
        a_tp, _ = targeted_peak_reduction(best_seq, threshold_frac=0.005, n_steps=50000, eta_base=0.03)
        try_update(a_tp, "targeted_phase4")

    # === Phase 5: Multi-length exploration ===
    target_lengths = [960, 1500, 1800, 2400]
    target_lengths = [n for n in target_lengths if n != n_best]
    # Sort by expected benefit (prefer lengths close to current best, then larger)
    target_lengths.sort(key=lambda n: abs(n - n_best))

    print(f"\n=== Phase 5: Multi-length exploration (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    n_targets = len(target_lengths)
    for i, n_target in enumerate(target_lengths):
        rem = deadline - time.monotonic()
        # Allocate time: more for earlier targets
        time_share = rem / (n_targets - i) * 0.8
        if time_share < 40:
            continue

        print(f"\n  --- n={n_target}, budget={time_share:.0f}s ---")
        init = interpolate_sequence(best_seq, n_target)

        # LP refinement
        t_lp = time_share * 0.5
        a_lp_t, _ = nesterov_lp_refine(init, t_lp, solver, n_max_iters=100000)
        try_update(a_lp_t, f"lp_n{n_target}")

        # Alt-LSE
        rem_sub = deadline - time.monotonic()
        if rem_sub > 20:
            t_alt = min(rem_sub * 0.3, time_share * 0.3)
            T_stages_t = [5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03]
            a_alt_t, _ = alternating_lse_optimize(best_seq if len(best_seq) == n_target else init,
                                                  t_alt, T_stages_t, max_iter_per_stage=2000)
            try_update(a_alt_t, f"alt_n{n_target}")

    # === Phase 6: Aggressive LP on the best ===
    print(f"\n=== Phase 6: Aggressive LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s, n={len(best_seq)}")
        a_lp6, _ = nesterov_lp_refine(best_seq, t_alloc, solver, n_max_iters=500000)
        try_update(a_lp6, "lp_aggressive")

    # === Phase 7: Random restarts ===
    print(f"\n=== Phase 7: Random Restarts (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        n_restart = min(6, int(rem / 60))
        print(f"  {n_restart} restarts, ~{rem/n_restart:.0f}s each")
        for trial in range(n_restart):
            rem_sub = deadline - time.monotonic()
            if rem_sub < 30:
                break
            strength = 0.02 + 0.02 * trial
            a_pert = perturbed_sequence(best_seq, rng, strength=strength)
            # Also try mirrored version
            if trial % 2 == 1:
                a_pert = mirrored_sequence(a_pert)

            t_sub = min(rem_sub * 0.5, 120.0)
            # Quick LP
            a_r, _ = nesterov_lp_refine(a_pert, t_sub, solver, n_max_iters=50000)
            try_update(a_r, f"restart{trial}")

            # Quick LSE
            rem_sub2 = deadline - time.monotonic()
            if rem_sub2 > 20:
                T_short = [3.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02]
                a_r2, _ = lbgfs_logsumexp_refine(a_r, min(rem_sub2 * 0.3, 40.0), T_short)
                try_update(a_r2, f"restart{trial}_lse")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        # Very fine LSE
        t_fin = min(rem * 0.4, 60.0)
        if t_fin > 10:
            T_fine = [1.0, 0.5, 0.3, 0.2, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.007, 0.005]
            a_fin, _ = lbgfs_logsumexp_refine(best_seq, t_fin, T_fine)
            try_update(a_fin, "final_lse")

        # Final LP
        rem = deadline - time.monotonic()
        if rem > 20:
            a_fin2, _ = nesterov_lp_refine(best_seq, min(rem * 0.5, 30.0), solver, n_max_iters=20000)
            try_update(a_fin2, "final_lp")

        # Final targeted reduction
        rem = deadline - time.monotonic()
        if rem > 15:
            a_fin3, _ = targeted_peak_reduction(best_seq, threshold_frac=0.003, n_steps=20000, eta_base=0.01)
            try_update(a_fin3, "final_targeted")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}")
        # Call evaluate_sequence for verification
        try:
            eval_score = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
            print(f"evaluate_sequence confirms: {eval_score:.8f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
```
