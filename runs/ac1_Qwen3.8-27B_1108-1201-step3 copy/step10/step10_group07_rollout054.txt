import numpy as np

import time
import cvxpy as cp

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def convolve_ab(a, b):
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def score_of(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))

def simplex_project(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def find_lp_solver():
    cached = getattr(find_lp_solver, '_cached', None)
    if cached is not None:
        return cached
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                find_lp_solver._cached = solver_name
                return solver_name
        except Exception:
            continue
    find_lp_solver._cached = cp.ECOS
    return cp.ECOS

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def lp_best_response(a, solver, warm_start=None):
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    ones = np.ones(2 * n - 1)
    
    constraints = [
        T @ b_var <= M_var * ones,
        cp.sum(b_var) == 1
    ]
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    # Warm start
    if warm_start is not None:
        b_var.value = np.maximum(warm_start, 0)
        s = b_var.value.sum()
        if s > 0:
            b_var.value = b_var.value / s
        try:
            Ms = float((T @ b_var.value).max())
            M_var.value = Ms
        except:
            pass
    
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return a.copy(), float('inf')
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float('inf')
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return a.copy(), float('inf')
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else float('inf')
    return b_val, M_val

def nesterov_smoothed_gradient(a0, time_limit, verbose=True):
    """
    Nesterov-accelerated gradient descent on the smoothed objective:
    f_epsilon(a) = 2*n*epsilon*log(sum(exp(conv(a,a)/epsilon))) / (sum(a))^2
    
    We work with the simplex-normalized version to simplify.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    # Nesterov momentum
    y = a.copy()
    momentum = 0.0
    
    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    
    t0 = time.monotonic()
    it = 0
    
    # Adaptive smoothing parameter
    eps_init = 0.001
    eps = eps_init
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * momentum * momentum)) / 2.0
        new_momentum = (t_k - 1.0) / (t_k + 1.0)
        
        # Nesterov step
        y = a + new_momentum * (a - y) if it > 1 else a.copy()
        
        # Compute smoothed gradient at y
        conv_y = fast_convolve(y)
        M_y = conv_y.max()
        
        # Log-sum-exp gradient
        shifted = conv_y / eps - (conv_y.max() / eps)
        weights = np.exp(shifted)
        weights /= weights.sum()
        
        # Gradient of max_k conv(a,a)[k] w.r.t. a[j]
        # = sum_k weights[k] * 2 * a[k-j] (for valid k-j)
        grad = np.zeros(n)
        for k in range(len(conv_y)):
            if weights[k] < 1e-14:
                continue
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            grad[i_start:i_end + 1] += weights[k] * 2.0 * y[k - i_end:k - i_start + 1][::-1]
        
        # Learning rate (for the smoothed problem on the simplex)
        lr = 2.0 / ((it + 1) ** 2 * np.linalg.norm(grad, 2) ** 2 + 1e-10)
        
        a_new = y - lr * grad
        a_new = simplex_project(a_new)
        
        # Evaluate
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        momentum = new_momentum
        
        # Adaptively decrease epsilon
        if it % 50 == 0:
            eps *= 0.9
            if eps < 1e-6:
                eps = 1e-6
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NGD it={it} ({elapsed:.0f}s): best={best_score:.8f} eps={eps:.6f}")
    
    return best_a, best_score

def lp_alternating(a0, time_limit, solver, verbose=True):

def diverse_init(n, rng, init_type=0):
    if init_type == 0:
        t = np.arange(n)
        center = (n - 1) / 2.0
        a = np.maximum(1.0 - np.abs(t - center) / center, 0)
    elif init_type == 1:
        t = np.arange(n)
        center = (n - 1) / 2.0
        x = (t - center) / center
        a = 0.3 + 0.7 * x * x
    elif init_type == 2:
        t = np.arange(n)
        center = (n - 1) / 2.0
        x = (t - center) / center
        power = rng.uniform(1.0, 3.0)
        a = np.maximum(0.5 * (1 + np.cos(np.pi * x)), 0) ** power
    elif init_type == 3:
        t = np.arange(n)
        c1 = n * rng.uniform(0.2, 0.35)
        c2 = n * rng.uniform(0.65, 0.8)
        w = n * rng.uniform(0.08, 0.2)
        a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    elif init_type == 4:
        period = int(rng.integers(3, 10))
        t = np.arange(n)
        a = 0.5 + 0.4 * np.cos(2 * np.pi * t / period)
    elif init_type == 5:
        t = np.arange(n)
        a = 1.0 - 0.5 * np.cos(4 * np.pi * t / n)
        a = np.maximum(a, 0.01)
    elif init_type == 6:
        # Sparse comb
        a = np.zeros(n)
        spacing = rng.integers(3, 8)
        for i in range(0, n, spacing):
            a[i] = 1.0
    elif init_type == 7:
        # Random with smoothing
        a = rng.random(n)
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, 'same')
    else:
        a = rng.random(n)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def mild_polish(a0, time_limit, verbose=True):
    """Very fine coordinate descent for last-mile polishing."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        
        for i in range(n):
            if a[i] < 1e-16:
                continue
            
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            
            unaff_max = 0.0
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))
            
            a_shifted = a[k_start - i:k_end - i + 1]
            conv_affected = conv[k_start:k_end + 1]
            
            best_delta = 0.0
            best_new_M = M
            
            for factor in [0.997, 0.995, 0.993, 0.998, 1.002, 1.005, 1.007, 1.003]:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue
                
                new_affected = conv_affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] = conv_affected[idx] + 2.0 * delta * a[i] + delta ** 2
                
                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-15:
                    best_new_M = new_M
                    best_delta = delta
            
            if best_delta != 0.0 and best_new_M < M - 1e-15:
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta ** 2
                conv[k_start:k_end + 1] += changes

                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 15:
                break
        else:
            no_improve = 0

def subgradient_descent(a0, time_limit, lr_init=0.02, top_k=20, verbose=True):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        M = conv.max()
        score = 2.0 * n * M
        
        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 150:
            lr *= 0.8
            no_improve = 0
            if lr < 1e-11:
                break
        
        # Multi-peak subgradient
        k_act = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        pv = conv[peaks]
        
        temp = 0.002 * max(M, 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / ws
        
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            sg[i_start:i_end + 1] += w[idx] * 2.0 * a[k - i_end:k - i_start + 1][::-1]
        
        a_new = a - lr * sg
        a_new = simplex_project(a_new)
        a = a_new
        conv = fast_convolve(a)
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD it={it} ({elapsed:.0f}s): best={best_score:.8f} lr={lr:.6f}")
    
    return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based best-response alternating minimization with warm-starting (primary)
2. Nesterov-accelerated smoothed gradient descent (log-sum-exp smoothing)
3. Exact coordinate descent with 1D subproblem solving (ternary search + analytical candidates)
4. Subgradient descent with multi-peak softmax weighting
5. Fine-grained polishing for last-mile improvements
With diverse initializations and adaptive time allocation.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def score_of(a):
    """Compute 2*n*max(conv)/(sum)^2 quickly."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def find_lp_solver():
    """Find the fastest working LP solver."""
    cached = getattr(find_lp_solver, '_cached', None)
    if cached is not None:
        return cached
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                find_lp_solver._cached = solver_name
                return solver_name
        except Exception:
            continue
    find_lp_solver._cached = cp.ECOS
    return cp.ECOS


def build_conv_matrix(a):
    """Build the convolution matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, solver, warm_start=None):
    """
    Solve LP: min M s.t. conv(a,b) <= M, sum(b)=1, b>=0.
    Returns (b_opt, M_opt).
    """
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    ones = np.ones(2 * n - 1)

    constraints = [
        T @ b_var <= M_var * ones,
        cp.sum(b_var) == 1
    ]

    prob = cp.Problem(cp.Minimize(M_var), constraints)

    # Warm start
    if warm_start is not None:
        b_var.value = np.maximum(warm_start, 0)
        s = b_var.value.sum()
        if s > 0:
            b_var.value = b_var.value / s
        try:
            Ms = float((T @ b_var.value).max())
            M_var.value = Ms
        except Exception:
            pass

    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return a.copy(), float('inf')

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float('inf')

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return a.copy(), float('inf')
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else float('inf')
    return b_val, M_val


def eval_1d_delta(d, conv_affected, a_shifted, a_i, has_quad, quad_idx, unaff_max):
    """Evaluate the 1D objective at delta=d."""
    if d < -a_i - 1e-14:
        return float('inf')
    new_aff = conv_affected + 2.0 * d * a_shifted
    if has_quad:
        new_aff = new_aff.copy()
        new_aff[quad_idx] = conv_affected[quad_idx] + 2.0 * d * a_i + d * d
    return max(unaff_max, float(new_aff.max()))


def nesterov_smoothed_gradient(a0, time_limit, verbose=True):
    """
    Nesterov-accelerated gradient descent on the smoothed objective
    using log-sum-exp approximation of the max.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)

            y = a + new_momentum * (a - y)
        else:
            new_momentum = 0.0

        y = np.maximum(y, 0)

        if sy < 1e-15:
            y = np.ones(n) / n
            sy = 1.0
        y = y / sy

        shifted = conv_y / eps - (conv_y.max() / eps)
        weights = np.exp(shifted)
        weights /= weights.sum()

        grad = np.zeros(n)

            grad[i_start:i_end + 1] += weights[k] * 2.0 * y[k - i_end:k - i_start + 1][::-1]

        # Learning rate
        gnorm = np.linalg.norm(grad, 2)
        if gnorm < 1e-15:
            break
        lr = 1.0 / ((it + 1) * gnorm)

        a_new = y - lr * grad
        a_new = simplex_project(a_new)

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        a = a_new
        momentum = new_momentum

        # Adaptively decrease epsilon
        if it % 80 == 0:
            eps *= 0.85
            if eps < 5e-5:
                eps = 5e-5

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NGD it={it} ({elapsed:.0f}s): best={best_score:.8f} eps={eps:.6f}")

    return best_a, best_score


def exact_coord_descent(a0, time_limit, verbose=True, label=""):
    """
    Coordinate descent with exact 1D subproblem solving.
    Uses ternary search + analytical candidate evaluation.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve_sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        sweep_improvements = 0

        for i in range(n):
            if a[i] < 1e-16:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            # Unaffected max
            unaff_max = 0.0
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            a_shifted = a[k_start - i:k_end - i + 1]
            conv_affected = conv[k_start:k_end + 1]

            has_quad = (2 * i >= k_start) and (2 * i <= k_end)
            quad_idx = 2 * i - k_start if has_quad else -1

            # Quick analytical candidates
            best_delta = 0.0
            best_new_M = max(unaff_max, float(conv_affected.max()))

            # Candidate 1: flatten the peak to unaff_max
            peak_idx = int(np.argmax(conv_affected))
            slope_peak = 2.0 * a_shifted[peak_idx]
            if slope_peak > 1e-14 and conv_affected[peak_idx] > unaff_max:
                d_cand = (unaff_max - conv_affected[peak_idx]) / slope_peak
                if -a[i] - 1e-14 <= d_cand:
                    val = eval_1d_delta(d_cand, conv_affected, a_shifted, a[i], has_quad, quad_idx, unaff_max)
                    if val < best_new_M:
                        best_new_M = val
                        best_delta = d_cand

            # Candidate 2: quadratic root at quad_idx (if quad_idx is the peak)
            if has_quad and quad_idx == peak_idx:
                b_coef = 2.0 * a[i]
                c_coef = conv_affected[quad_idx] - unaff_max
                disc = b_coef * b_coef - 4.0 * c_coef
                if disc >= 0:
                    sqrt_disc = np.sqrt(disc)
                    d1 = (-b_coef - sqrt_disc) / 2.0
                    d2 = (-b_coef + sqrt_disc) / 2.0
                    for d_c in [d1, d2]:
                        if -a[i] - 1e-14 <= d_c:
                            val = eval_1d_delta(d_c, conv_affected, a_shifted, a[i], has_quad, quad_idx, unaff_max)
                            if val < best_new_M:
                                best_new_M = val
                                best_delta = d_c

            # Ternary search around best_delta
            if best_new_M < M - 1e-13:
                search_lo = max(-a[i], best_delta - a[i] * 0.8)
                search_hi = best_delta + a[i] * 0.8 + 1e-6
                for _ in range(25):
                    m1 = search_lo + (search_hi - search_lo) / 3.0
                    m2 = search_hi - (search_hi - search_lo) / 3.0
                    f1 = eval_1d_delta(m1, conv_affected, a_shifted, a[i], has_quad, quad_idx, unaff_max)
                    f2 = eval_1d_delta(m2, conv_affected, a_shifted, a[i], has_quad, quad_idx, unaff_max)
                    if f1 < f2:
                        search_hi = m2
                    else:
                        search_lo = m1
                best_delta = (search_lo + search_hi) / 2.0
                best_new_M = eval_1d_delta(best_delta, conv_affected, a_shifted, a[i], has_quad, quad_idx, unaff_max)

            if best_delta != 0.0 and best_new_M < M - 1e-13:
                new_val = a[i] + best_delta
                if new_val < 0:
                    best_delta = -a[i]
                    new_val = 0.0
                    best_new_M = eval_1d_delta(best_delta, conv_affected, a_shifted, a[i], has_quad, quad_idx, unaff_max)
                a[i] = new_val
                changes = 2.0 * best_delta * a_shifted
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta ** 2
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
                sweep_improvements += 1

        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps > 15:
                break
        else:
            no_improve_sweeps = 0

        if verbose and sweeps % 40 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD[{label}] sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f} M={M:.10f} imp={sweep_improvements}")

    return best_a, best_score


def lp_alternating(a0, time_limit, solver, verbose=True):
    """LP-based best-response alternating minimization with warm-starting and adaptive damping."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    stagnation = 0
    warm_start = a.copy()
    rng = np.random.default_rng(123)

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        b_new, M_lp = lp_best_response(a, solver, warm_start=warm_start)
        if not np.isfinite(M_lp):
            break

        score_b = 2.0 * n * fast_convolve(b_new).max()
        if score_b < best_score:
            best_score = score_b
            best_a = b_new.copy()
            warm_start = b_new.copy()

        # Adaptive damping
        if it <= 8:
            alpha = 1.0
        elif it <= 20:
            alpha = 0.85
        elif it <= 50:
            alpha = 0.6
        elif it <= 100:
            alpha = 0.45
        else:
            alpha = max(0.25, 6.0 / (it + 5.0))

        a = (1.0 - alpha) * a + alpha * b_new
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
            s = 1.0
        a = a / s

        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            warm_start = a.copy()

        # Stagnation
        if score_a > best_score * (1.0 + 1e-5):
            stagnation += 1
        else:
            stagnation = max(0, stagnation - 1)

        if stagnation > 15:
            noise_scale = 0.015 / np.sqrt(n)
            a_pert = a + rng.standard_normal(n) * noise_scale
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            score_pert = 2.0 * n * fast_convolve(a_pert).max()
            if score_pert < best_score * 1.005:
                a = a_pert
                warm_start = a_pert.copy()
            stagnation = 0

        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP it={it} ({elapsed:.0f}s): best={best_score:.8f} M_lp={M_lp:.10f}")

    return best_a, best_score


def diverse_init(n, rng, init_type=0):
    """Generate diverse initializations."""
    if init_type == 0:
        t = np.arange(n)
        center = (n - 1) / 2.0
        a = np.maximum(1.0 - np.abs(t - center) / center, 0)
    elif init_type == 1:
        t = np.arange(n)
        center = (n - 1) / 2.0
        x = (t - center) / center
        a = 0.3 + 0.7 * x * x
    elif init_type == 2:
        t = np.arange(n)
        center = (n - 1) / 2.0
        x = (t - center) / center
        power = rng.uniform(1.5, 3.0)
        a = np.maximum(0.5 * (1 + np.cos(np.pi * x)), 0) ** power
    elif init_type == 3:
        t = np.arange(n)
        c1 = n * rng.uniform(0.2, 0.35)
        c2 = n * rng.uniform(0.65, 0.8)
        w = n * rng.uniform(0.08, 0.2)
        a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    elif init_type == 4:
        period = int(rng.integers(4, 12))
        t = np.arange(n)
        a = 0.5 + 0.4 * np.cos(2.0 * np.pi * t / period)
    elif init_type == 5:
        t = np.arange(n)
        a = 1.0 - 0.5 * np.cos(4.0 * np.pi * t / n)
        a = np.maximum(a, 0.01)
    elif init_type == 6:
        # Sparse comb
        a = np.zeros(n)
        spacing = int(rng.integers(3, 7))
        for i in range(0, n, spacing):
            a[i] = 1.0
    elif init_type == 7:
        # Smoothed random
        a = rng.random(n)
        kernel = np.ones(7) / 7
        a = np.convolve(a, kernel, 'same')
    elif init_type == 8:
        # W-shape (multi-peak)
        t = np.arange(n)
        a = 1.0 - 0.6 * np.cos(6.0 * np.pi * t / n)
        a = np.maximum(a, 0.05)
    else:
        a = rng.random(n)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def mild_polish(a0, time_limit, verbose=True):
    """Very fine coordinate descent for last-mile polishing."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-16:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            unaff_max = 0.0
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            a_shifted = a[k_start - i:k_end - i + 1]
            conv_affected = conv[k_start:k_end + 1]

            best_delta = 0.0
            best_new_M = M

            for factor in [0.996, 0.994, 0.992, 0.998, 1.002, 1.005, 1.007, 1.003]:
                delta = a[i] * (factor - 1.0)
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = conv_affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] = conv_affected[idx] + 2.0 * delta * a[i] + delta * delta

                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-15:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-15:
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta ** 2
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 12:
                break
        else:
            no_improve = 0

    return best_a, best_score


def subgradient_descent(a0, time_limit, lr_init=0.015, top_k=30, verbose=True):
    """Subgradient descent with multi-peak softmax weighting and simplex projection."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        M = conv.max()
        score = 2.0 * n * M

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 120:
            lr *= 0.82
            no_improve = 0
            if lr < 1e-11:
                break

        k_act = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        pv = conv[peaks]

        temp = 0.002 * max(M, 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / ws

        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            sg[i_start:i_end + 1] += w[idx] * 2.0 * a[k - i_end:k - i_start + 1][::-1]

        a_new = a - lr * sg
        a_new = simplex_project(a_new)
        a = a_new
        conv = fast_convolve(a)

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD it={it} ({elapsed:.0f}s): best={best_score:.8f} lr={lr:.6f}")

    return best_a, best_score

            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_fin / a_fin.sum()
                print(f"  NEW BEST [final_cd]: {sc:.8f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
