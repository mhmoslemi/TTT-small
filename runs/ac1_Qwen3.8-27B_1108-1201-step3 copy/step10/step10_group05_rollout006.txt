def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def convolve_ab(a, b):

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

                improved = True

            conv = fast_convolve(a)
            M = float(conv.max())

        if not improved:
            no_improve += 1
            if no_improve > 200:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.7f}")

    return best_a, best_score, sweeps

def fine_coord_descent(a0, time_limit, label=""):
    n = len(a0)
    a = normalize_seq(np.asarray(a0, dtype=np.float64))
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0

    factors = [0.9999, 0.9998, 0.9995, 0.999, 0.998, 0.997, 0.995, 0.99,
               1.0001, 1.0002, 1.0005, 1.001, 1.002, 1.003, 1.005, 1.01]

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            ai = a[i]
            if ai < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            unaff_max = -1e30
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            affected = conv[k_start:k_end + 1]
            coeffs = a[k_start - i:k_end - i + 1] + ai
            has_quad = (2 * i >= k_start) and (2 * i <= k_end)
            quad_idx = 2 * i - k_start if has_quad else -1

            best_delta = 0.0
            best_new_max = M

            for factor in factors:
                delta = ai * (factor - 1.0)
                if delta < -ai:
                    continue
                new_affected = affected + delta * coeffs
                if has_quad:
                    new_affected = new_affected.copy()
                    new_affected[quad_idx] += delta * delta
                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-16:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-15:
                a[i] = max(ai + best_delta, 0.0)
                conv[k_start:k_end + 1] += best_delta * coeffs
                if has_quad:
                    conv[2 * i] += best_delta * best_delta
                M = best_new_max
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved and sweeps > 20:
            break

    return best_a, best_score

def lp_fictitious_play(a0, time_limit, solver=None, label=""):
    n = len(a0)
    a = normalize_seq(np.asarray(a0, dtype=np.float64))
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    iters = 0
    b = a.copy()

    while time.monotonic() - t0 < time_limit:
        iters += 1
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is None:
            break

        # Adaptive damping: start aggressive, decay
        alpha = min(1.0, 20.0 / (iters + 10) ** 0.8)
        a_new = (1 - alpha) * a + alpha * b_new
        a_new = normalize_seq(a_new)

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        score_b = 2.0 * n * fast_convolve(b_new).max()
        if score_b < best_score - 1e-14:
            best_score = score_b
            best_a = b_new.copy()

        if iters % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LPFP {label} it={iters} ({elapsed:.0f}s): best={best_score:.7f}")

    return best_a, best_score, iters

def spectral_perturb(a0, rng, strength=0.05):
    """Perturb in frequency domain to escape local optima."""
    n = len(a0)
    a = a0.copy()
    A = rfft(a)
    # Perturb high-frequency bins more
    freqs = np.arange(len(A))
    high_mask = freqs >= n * 0.3
    phase = rng.standard_normal(len(A)) * strength * high_mask
    magnitude = rng.standard_normal(len(A)) * strength * 0.5 * high_mask
    A_perturbed = A * (1.0 + magnitude) * np.exp(1j * phase)
    # Make it consistent for real signal
    A_perturbed[0] = A_perturbed[0].real
    if n % 2 == 0:
        A_perturbed[-1] = A_perturbed[-1].real
    a_new = irfft(A_perturbed, n)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a0.copy()
    return a_new / s

def peak_targeted_perturb(a0, rng, strength=0.05):
    """Perturb by targeting peak-contributing coordinates."""
    n = len(a0)
    conv = fast_convolve(a)
    M = conv.max()
    peak_threshold = M * 0.85
    peak_positions = np.where(conv >= peak_threshold)[0]

    pressure = np.zeros(n)
    for k in peak_positions:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            pressure[i] += a0[i] * a0[j]

    p_max = pressure.max()
    if p_max > 1e-20:
        pressure = pressure / p_max

    noise = rng.standard_normal(n)
    a_new = a0 * (1.0 - strength * pressure + 0.5 * strength * noise)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a0.copy()
    return a_new / s

def random_restructure(a0, rng, frac=0.15):
    """Randomly restructure to escape local optima."""
    n = len(a0)
    a = a0.copy()
    keep = rng.random(n) > frac
    n_keep = keep.sum()
    if n_keep < max(10, n // 4):
        keep = np.ones(n, dtype=bool)
    a[~keep] = 0
    a[keep] = a[keep] * (1.0 + rng.standard_normal(n_keep) * 0.4)
    a = np.maximum(a, 0)
    zero_mask = (a < 1e-15)
    if zero_mask.any():
        n_add = max(1, int(zero_mask.sum() * 0.3))
        add_idx = np.where(zero_mask)[0]
        add_idx = rng.choice(add_idx, size=min(n_add, len(add_idx)), replace=False)
        a[add_idx] = rng.random(n_add) * a.max() * 0.5
    s = a.sum()
    if s < 1e-15:
        return a0.copy()
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n, endpoint=False)
    new_x = np.linspace(0, 1, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def cyclic_shift_perturb(a0, rng, shift_amount=None):
    """Perturb by applying a small cyclic-shift-like deformation."""
    n = len(a0)
    if shift_amount is None:
        shift_amount = rng.integers(1, max(2, n // 50))
    # Create a small deformation: blend with a shifted version
    a_shifted = np.roll(a0, shift_amount)
    # Handle the discontinuity at the boundary
    blend = 0.1
    a_new = (1 - blend) * a0 + blend * a_shifted
    return normalize_seq(a_new)

def subgradient_nesterov(a0, time_limit, label=""):
    """Subgradient method with Nesterov acceleration."""
    n = len(a0)
    a = normalize_seq(np.asarray(a0, dtype=np.float64))
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    gamma = 1.0
    y = a.copy()
    it = 0
    avg_a = a.copy()

    while time.monotonic() - t0 < time_limit:
        it += 1
        step = 1.0 / np.sqrt(it + 1)

        y = (1.0 + gamma) * a - gamma * y
        y = simplex_project(np.maximum(y, 0))

        conv_y = fast_convolve(y)
        M = conv_y.max()
        threshold = M * 0.95
        near_peak = np.where(conv_y >= threshold)[0]
        rho = 1.0 / len(near_peak)

        grad = np.zeros(n)
        for k in near_peak:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j_arr = np.arange(i_start, i_end + 1)
            grad[j_arr] += rho * 2.0 * y[k - j_arr]

        a_new = a - step * grad
        a_new = simplex_project(a_new)

        gamma_it = 2.0 / (it + 2.0)
        avg_a = (1.0 - gamma_it) * avg_a + gamma_it * a_new

        for cand in [a_new, avg_a]:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-10:
                continue
            cand = cand / sc
            score = 2.0 * n * fast_convolve(cand).max()
            if score < best_score - 1e-14:
                best_score = score
                best_a = cand.copy()

        a = a_new
        gamma = (1.0 + np.sqrt(1.0 + 4.0 * gamma * gamma)) / 2.0

    return best_a, best_score

def block_coordinate_descent(a0, time_limit, block_size=4, label="", rng=None):
    """Block coordinate descent: update small groups of coordinates together."""
    n = len(a0)
    a = normalize_seq(np.asarray(a0, dtype=np.float64))
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False

        # Pick a random block of coordinates
        if rng is not None:
            start = rng.integers(0, n)
        else:
            start = (iters * block_size) % n

        indices = [((start + i) % n) for i in range(block_size)]
        indices = sorted(set(indices))

        # For each coordinate in the block, do a local search
        for i in indices:
            ai = a[i]
            if ai < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            unaff_max = -1e30
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            affected = conv[k_start:k_end + 1]
            coeffs = a[k_start - i:k_end - i + 1] + ai
            has_quad = (2 * i >= k_start) and (2 * i <= k_end)
            quad_idx = 2 * i - k_start if has_quad else -1

            best_delta = 0.0
            best_new_max = M
            # Try a range of multipliers
            for mult in np.linspace(-1.0, 0.5, 30):
                delta = mult * ai
                if delta < -ai:
                    continue
                new_affected = affected + delta * coeffs
                if has_quad:
                    new_affected = new_affected.copy()
                    new_affected[quad_idx] += delta * delta
                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-15:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-14:
                a[i] = max(ai + best_delta, 0.0)
                conv[k_start:k_end + 1] += best_delta * coeffs
                if has_quad:
                    conv[2 * i] += best_delta * best_delta
                M = best_new_max
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved and iters > 100:
            break

    return best_a, best_score

def check_and_update(a, best_seq, best_score, label=""):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.7f} (n={len(a)})")
    return best_seq, best_score
