import numpy as np
import time
import math
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def local_eval(a):
    """Fast local evaluation of the objective."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def lse_obj_and_grad(a, T):
    """Compute LSE_T(conv(a,a)) and its gradient."""
    n = len(a)
    b = fast_convolve(a)
    bT = b / T
    bT_max = bT.max()
    exp_ws = np.exp(bT - bT_max)
    w = exp_ws / exp_ws.sum()
    lse_val = T * (bT_max + np.log(exp_ws.sum()))
    
    # Gradient: grad[j] = 2 * sum_k w[k] * a[k-j]
    # Using FFT cross-correlation
    a_pad = np.zeros(2 * n - 1)
    a_pad[:n] = a
    N = 1 << (4 * n - 1).bit_length()
    W = rfft(w, N)
    A = rfft(a_pad, N)
    corr = irfft(W * np.conj(A), N)
    grad = 2.0 * corr[:n]
    
    return lse_val, grad, b, w

def nesterov_lse_optimize(a0, time_limit, T_schedule=None):
    """
    Nesterov-accelerated multiplicative weights on LSE-smoothed objective.
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    if T_schedule is None:
        T_schedule = [10.0, 5.0, 3.0, 2.0, 1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.007, 0.005]
    
    # Nesterov momentum
    y = a.copy()
    k_global = 0
    
    for T in T_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        # For each T, run a batch of Nesterov iterations
        # Adaptive number of iterations based on remaining time
        n_iters = max(50, int(remaining / len(T_schedule[T_schedule.index(T) if T in T_schedule else 0]) * 10))
        
        beta = 1.0  # momentum parameter (will update)
        a_prev = a.copy()
        
        for k in range(n_iters):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Nesterov lookahead
            beta_new = (1.0 + math.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
            y = a + (beta / beta_new) * (a - a_prev)
            y = normalize(y)
            beta = beta_new
            
            # Compute gradient at y
            val, grad, b, w = lse_obj_and_grad(y, T)
            
            # Center the gradient (remove component in direction of 1)
            # For the simplex constraint, the effective gradient is grad - mean(grad)
            grad_centered = grad - grad.mean()
            
            # Adaptive step size
            # The LSE is Lipschitz with constant ~2*n/T (roughly)
            # Safe step: 1 / (2 * max|grad|)
            gnorm = np.max(np.abs(grad_centered))
            if gnorm < 1e-15:
                break
            lr = 0.5 / gnorm  # Conservative step
            
            # Multiplicative update (mirror descent step)
            z = grad_centered * lr  # z = lr * grad
            # a_new = y * exp(-z) / sum(y * exp(-z))
            log_a_new = np.log(np.maximum(y, 1e-300)) - z
            log_a_new -= log_a_new.max()  # numerical stability
            a_new = np.exp(log_a_new)
            a_new = a_new / a_new.sum()
            
            a_prev = a.copy()
            a = a_new
            
            # Track best
            sc = 2.0 * n * fast_convolve(a).max()
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        print(f"    T={T:.4f}: score={best_score:.10f}")
    
    return best_a, best_score

**Hankel/Active-set approach:**

Given an active set S = {k_1, ..., k_m} and weights lambda_1, ..., lambda_m (sum = 1):
1. Form the n x n matrix P where P[j, k] = sum_i lambda_i * [j + k = k_i] (with the convention that a[x] = 0 for x out of [0, n-1]).

Wait, I need to be more careful. The KKT condition is:
`sum_i lambda_i * a[k_i - j] = c` for all j = 0, ..., n-1.

Here, a[k_i - j] is defined as a[x] if 0 <= x < n, and 0 otherwise.

So: `sum_i lambda_i * a[k_i - j] = c`

Let me define the matrix P where `(P @ a)[j] = sum_i lambda_i * a[k_i - j]`.

`(P @ a)[j] = sum_k P[j, k] * a[k]`

So `P[j, k] = sum_i lambda_i * [k = k_i - j] = sum_i lambda_i * [j + k = k_i]`.

P is a symmetric matrix (P[j,k] = P[k,j] since [j+k = k_i] is symmetric in j and k). P is a sum of "Hankel" matrices (anti-diagonals).

Actually, P is itself a Hankel matrix! P[j,k] depends only on j+k. Specifically:
`P[j,k] = sum_i lambda_i * delta(j+k, k_i)`

where delta(m, k_i) = 1 if m = k_i, 0 otherwise.

So P is a Hankel matrix with:
- P[0, k_i] = lambda_i (for k_i < n)
- P[j, k_i - j] = lambda_i (when 0 <= k_i - j < n)

The matrix P has non-zero entries only on the anti-diagonals j + k = k_i for i = 1, ..., m.

If m = n (active set has n elements), P is a full Hankel matrix and likely invertible.
If m < n, P has rank at most m and is singular.

So we need m >= n for P to be invertible! That means we need at least n active lags.

But the total number of lags is 2n-1, so we need at least n of them to be active. That's a lot!

In practice, at the optimum, maybe only a subset of the lags are "truly active" (exactly at the max), and the rest are slightly below. The KKT conditions with inequalities are:
- For active lags k_i: conv(a,a)[k_i] = M
- For inactive lags k: conv(a,a)[k] <= M
- Multiplier lambda_k >= 0 for active, = 0 for inactive.

The "soft" version (using LSE) automatically handles this: the weights w are concentrated on the active lags but spread over all lags.

So the generic (full-rank) case requires n active lags. This means the KKT-based approach gives a unique solution only when n lags are active.
