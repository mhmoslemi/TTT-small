def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def compute_subgradient(a, active_frac=0.05):
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    if max_b < 1e-300:
        return np.zeros(n), max_b
    threshold = max_b - active_frac * max_b
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    # Weighted subgradient
    weights = np.maximum(b[active] - threshold, 0.0)
    wsum = weights.sum()
    if wsum < 1e-300:
        weights = np.ones(len(active))
    else:
        weights = weights / wsum
    g = np.zeros(n)
    for i, k in enumerate(active):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += weights[i] * a[k - idx]
    g *= 2.0
    return g, max_b


def project_simplex(v):
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def nesterov_lp_fixed(a0, time_limit, solver=None, n_max_iters=200000):
    """Fixed Nesterov-accelerated LP best-response."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:

    a = a / s
    y = a.copy()
    best_a = a.copy()

    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score
    restart_count = 0

            break

        if b_new is None:
            continue

        # Nesterov step size
        alpha = 2.0 / ((it + 1.0) ** 0.5 + 1.0)
        # Clamp for stability
        alpha = min(alpha, 0.5)

        # Update y (implicit iterate)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        # Update a (explicit iterate) - simple averaging
        if it == 0:
            a = b_new.copy()
        else:
            a = (1.0 - alpha) * a + alpha * b_new
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        # Evaluate

        score_y = 2.0 * n * fast_convolve(y).max()

            best_a = a.copy()
            stall_count = 0
        if score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

def lse_objective(z_in, T_val, n, t0, time_limit):
    if time.monotonic() - t0 > time_limit:
        return 1e15
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    return (2.0 * n / T_val) * log_sum_exp


def lse_jacobian(z_in, T_val, n, t0, time_limit):
    if time.monotonic() - t0 > time_limit:
        return np.zeros_like(z_in)
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    grad_a *= 2.0
    S = np.dot(grad_a, a)
    return a * (grad_a - S)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        ss = a.sum()
        if ss < 0.01:
            return
        a = a / ss
        sc = 2.0 * len(a) * fast_convolve(a).max()
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t0
            print(f"  NEW BEST [{src_label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Phase A: Quick mirror descent for warmup
    rem = time_budget - (time.monotonic() - t0)
    t_a = min(rem * 0.12, 80.0)
    if t_a > 5:
        a_md, _ = mirror_descent_adaptive(best_a, t_a, active_frac=0.04)
        try_update(a_md, f"md_{label}")

    # Phase B: Two-phase LP + subgradient (main workhorse)
    rem = time_budget - (time.monotonic() - t0)
    t_b = min(rem * 0.50, 500.0)
    if t_b > 20:
        a_2p, _ = two_phase_lp_subgradient(best_a, t_b, solver=solver, lp_frac=0.65)
        try_update(a_2p, f"2phase_{label}")

    # Phase C: LogSumExp L-BFGS
    rem = time_budget - (time.monotonic() - t0)
    t_c = min(rem * 0.22, 200.0)
    if t_c > 10:
        a_lb = lbgfs_logsumexp(best_a, t_c, T_start=50.0, T_end=0.03, n_stages=15)
        try_update(a_lb, f"lse_{label}")

    # Phase D: Peak pressure local search
    rem = time_budget - (time.monotonic() - t0)
    t_d = min(rem * 0.12, 80.0)
    if t_d > 5:
        a_pp, _ = peak_pressure_search(best_a, t_d, n_steps=30000)
        try_update(a_pp, f"pp_{label}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function with adaptive time allocation."""
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    t_deadline = t_start + max(1.0, float(budget_s) - 50.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Time budget: {budget_s}s, deadline at t={t_deadline - t_start:.0f}s")

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = 2.0 * len(parent) * fast_convolve(parent).max()
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_base = 960
        best_seq = structured_init(n_base, rng)
        best_score = 2.0 * n_base * fast_convolve(best_seq).max()
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    def try_global_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = 2.0 * len(a) * fast_convolve(a).max()
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  GLOBAL NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: Optimize at base length ===
    print(f"\n=== Phase 1: Optimize at n={n_base} (best={best_score:.6f}) ===")
    rem = t_deadline - time.monotonic()
    t1 = min(rem * 0.42, 650.0)
    if t1 > 60:
        print(f"  Budget: {t1:.0f}s")
        a_out, _ = optimize_at_length(best_seq, n_base, t1, solver, label=f"n{n_base}")
        try_global_update(a_out, f"opt_n{n_base}")

    # === Phase 2: Try longer lengths ===
    print(f"\n=== Phase 2: Longer lengths (best={best_score:.6f}) ===")
    longer_lengths = []
    for mult in [1.2, 1.5, 2.0]:
        ln = int(n_base * mult)
        if ln > n_base and ln <= 5000:
            longer_lengths.append(ln)
    longer_lengths = sorted(set(longer_lengths))[:3]

    for ln in longer_lengths:
        rem = t_deadline - time.monotonic()
        if rem < 150:
            break
        t_alloc = min(rem * 0.28, 350.0)
        if t_alloc < 60:
            break
        print(f"  Trying n={ln}, budget={t_alloc:.0f}s")

        init_long = interpolate_sequence(best_seq, ln)
        sc_init = 2.0 * ln * fast_convolve(init_long).max()
        print(f"    Interpolated score: {sc_init:.6f}")
        if sc_init < best_score:
            best_score = sc_init
            best_seq = init_long.copy()

        a_out, _ = optimize_at_length(init_long, ln, t_alloc, solver, label=f"n{ln}")
        try_global_update(a_out, f"opt_long_n{ln}")

        # Extra LP refinement if this length is competitive
        if a_out is not None:
            sc_out = 2.0 * len(a_out) * fast_convolve(a_out).max()
            if sc_out <= best_score * 1.01:
                rem2 = t_deadline - time.monotonic()
                if rem2 > 60:
                    t2 = min(rem2 * 0.2, 150.0)
                    if t2 > 30:
                        a_ref, _ = nesterov_lp_fixed(a_out, t2, solver=solver)
                        try_global_update(a_ref, f"lp_refine_n{ln}")

    # === Phase 3: Try shorter lengths ===
    print(f"\n=== Phase 3: Shorter lengths (best={best_score:.6f}) ===")
    shorter_lengths = []
    for mult in [0.75, 0.5, 0.66]:
        sn = int(n_base * mult)
        if sn < n_base and sn >= 200:
            shorter_lengths.append(sn)
    shorter_lengths = sorted(set(shorter_lengths))[:2]

    for sn in shorter_lengths:
        rem = t_deadline - time.monotonic()
        if rem < 100:
            break
        t_alloc = min(rem * 0.18, 150.0)
        if t_alloc < 30:
            break
        print(f"  Trying n={sn}, budget={t_alloc:.0f}s")
        init_short = interpolate_sequence(best_seq, sn)
        a_out, _ = optimize_at_length(init_short, sn, t_alloc, solver, label=f"n{sn}")
        try_global_update(a_out, f"opt_short_n{sn}")

    # === Phase 4: Final polish on the best ===
    print(f"\n=== Phase 4: Final Polish (best={best_score:.6f}, n={len(best_seq)}) ===")
    rem = t_deadline - time.monotonic()
    if rem > 30:
        # LP refinement
        t_lp = min(rem * 0.30, 120.0)
        if t_lp > 10:
            a_fin, _ = nesterov_lp_fixed(best_seq, t_lp, solver=solver)
            try_global_update(a_fin, "final_lp")

        # LogSumExp fine-tune with low temperature
        rem = t_deadline - time.monotonic()
        if rem > 20:
            t_lse = min(rem * 0.30, 80.0)
            if t_lse > 10:
                a_fin2 = lbgfs_logsumexp(best_seq, t_lse, T_start=5.0, T_end=0.02, n_stages=12)
                try_global_update(a_fin2, "final_lse")

        # Peak pressure
        rem = t_deadline - time.monotonic()
        if rem > 15:
            t_pp = min(rem * 0.25, 60.0)
            if t_pp > 5:
                a_fin3, _ = peak_pressure_search(best_seq, t_pp, n_steps=20000)
                try_global_update(a_fin3, "final_pp")

        # One more LP pass
        rem = t_deadline - time.monotonic()
        if rem > 15:
            t_lp2 = min(rem * 0.40, 40.0)
            if t_lp2 > 5:
                a_fin4, _ = nesterov_lp_fixed(best_seq, t_lp2, solver=solver)
                try_global_update(a_fin4, "final_lp2")

        # Final subgradient
        rem = t_deadline - time.monotonic()
        if rem > 10:
            t_sg = min(rem * 0.50, 30.0)
            if t_sg > 5:
                a_fin5, _ = mirror_descent_adaptive(best_seq, t_sg, active_frac=0.02)
                try_global_update(a_fin5, "final_sg")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE (evaluate_sequence): {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nNo valid sequence found!")

    elapsed_total = time.monotonic() - t_start
    print(f"Total elapsed: {elapsed_total:.1f}s")

    if best_seq is None:
        return [1.0 / max(n_base, 1)] * n_base
    return [float(x) for x in best_seq]
