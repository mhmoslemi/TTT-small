import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def subgradient_max_conv(a):
    """Compute subgradient of max_k (a*a)[k] at point a.
    Returns (max_val, subgradient_vector)"""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    # Active set: indices where b[k] is close to max
    threshold = max_b * (1 - 1e-6)
    active = np.where(b >= threshold)[0]
    
    # Subgradient: average of Q_k @ a for k in active set
    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]
    g /= len(active)
    
    return max_b, g

def project_simplex(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase hybrid approach:
1. Fast Nesterov-accelerated subgradient with FFT-based convolution (no LP)
2. LogSumExp-smoothed L-BFGS in log-domain with aggressive temperature annealing
3. LP-guided Nesterov refinement (limited, with cvxpy)
4. Multi-length exploration with interpolation
5. Coordinate descent + final L-BFGS polish
Key innovations: fast subgradient with no LP bottleneck, log-domain optimization,
adaptive active set, and structured multi-phase time allocation.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Project vector onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_subgradient(a, tol_frac=1e-4):
    """Compute subgradient of max_k (a*a)[k] at point a (sum(a)=1).
    Returns (max_val, subgradient_vector)."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    threshold = max_b * (1.0 - tol_frac)
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(b)])

    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        # grad_k[j] = 2*a[k-j] for j in [js, je]
        idx = np.arange(js, je + 1)
        g[idx] += 2.0 * a[k - idx]
    g /= len(active)
    return max_b, g


def logsumexp_objective_grad_z(z, n, T):
    """Compute LogSumExp-smoothed objective and gradient w.r.t. log-domain z.
    a = softmax(z), objective = 2*n*T*logsumexp(b/T) where b = conv(a,a)."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()

    b = fast_convolve(a)
    b_scaled = b / T
    b_max = b_scaled.max()
    exp_b = np.exp(b_scaled - b_max)
    sum_exp = exp_b.sum()
    lse = b_max + np.log(sum_exp)
    obj = 2.0 * n * T * lse

    # Weights for each lag
    w = exp_b / sum_exp

    # Gradient of lse w.r.t. a: sum_k w_k * grad_k(a)
    grad_a = np.zeros(n)
    # Only process significant weights
    w_thresh = 1e-12
    significant = w > w_thresh
    for k in np.where(significant)[0]:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * 2.0 * a[k - idx]

    # Chain rule through softmax: grad_z = a * (grad_a - dot(grad_a, a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    grad_z *= 2.0 * n * T

    return obj, grad_z


def nesterov_subgradient_refine(a0, time_limit, tol_frac=1e-4, seed=123):
    """Nesterov-accelerated subgradient method with FFT-based convolution."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    # Nesterov parameters
    z = a.copy()  # extrapolation point
    t_mom = 1.0
    eta_base = 0.5  # base step size
    eta = eta_base
    momentum = 0.9
    v = np.zeros(n)  # momentum velocity
    stall_count = 0
    prev_best = best_score
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        batch_start = time.monotonic()

        # Process in mini-batches for efficiency
        n_batch = max(1, min(1000, int(0.02 / max(time.monotonic() - batch_start, 1e-9))))
        for _ in range(n_batch):
            if time.monotonic() - t0 > time_limit:
                break

            # Compute subgradient at z
            max_b, g = compute_subgradient(z, tol_frac)
            F = 2.0 * n * max_b

            # Momentum update
            v = momentum * v + (1.0 - momentum) * g

            # Step with diminishing step size
            eta = eta_base / (1.0 + 0.001 * iter_count)
            a_candidate = z - eta * v

            # Project to simplex
            a_candidate = project_simplex(a_candidate)

            # Nesterov extrapolation
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
            beta = (t_mom - 1.0) / t_new
            z = a_candidate + beta * (a_candidate - a)
            z = np.maximum(z, 0)
            sz = z.sum()
            if sz > 1e-15:
                z = z / sz
            t_mom = t_new
            a = a_candidate
            iter_count += 1

        # Periodic evaluation
        if iter_count % 500 == 0 or time.monotonic() - batch_start > 5.0:
            score_a = 2.0 * n * fast_convolve(a).max()
            score_z = 2.0 * n * fast_convolve(np.maximum(z, 0)).max()
            if score_a < best_score:
                best_score = score_a
                best_a = a.copy()
            if score_z < best_score:
                best_score = score_z
                best_a = np.maximum(z, 0).copy()
                best_a /= best_a.sum()

            # Stall detection
            if best_score < prev_best - 1e-8:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count > 20:
                # Perturbation to escape
                noise = 0.02 * rng.standard_normal(n)
                a = a + noise
                a = project_simplex(a)
                z = a.copy()
                v = np.zeros(n)
                t_mom = 1.0
                stall_count = 0

            if iter_count % 10000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NesterovSubGrad iter={iter_count} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage=2000, seed=456):
    """L-BFGS with LogSumExp smoothing and temperature annealing in log-domain."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()

    for stage_idx, stage_T in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func(z_vec):
            obj, _ = logsumexp_objective_grad_z(z_vec, n, stage_T)
            return obj

        def jac(z_vec):
            _, g = logsumexp_objective_grad_z(z_vec, n, stage_T)
            return g

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-12, 'maxfun': maxiter_per_stage * 200,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass

        # Evaluate true objective at current point
        a_current = softmax_from_z(z)
        score = 2.0 * n * fast_convolve(a_current).max()
        if score < best_score:
            best_score = score
            best_a = a_current.copy()

        elapsed = time.monotonic() - t0
        print(f"    L-BFGS stage {stage_idx+1}/{len(T_stages)} T={stage_T}: score={best_score:.8f} ({elapsed:.0f}s)")

    return best_a, best_score


def lp_best_response(a_fixed, solver=None):
    """Solve LP: min M s.t. T@b <= M, sum(b)=1, b>=0. Returns (b, M)."""
    import cvxpy as cp
    n = len(a_fixed)
    size = 2 * n - 1

    # Build constraint matrix T (size x n): T[k,j] = a[k-j] if valid
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    # Build constraints: (a*b)[k] <= M for all k
    # (a*b)[k] = sum_j a[k-j]*b[j] (with valid indices)
    # Use sparse representation
    rows = []
    cols = []
    vals = []
    for k in range(size):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a_fixed[k - j])

    T_mat = cp.SparseMatrix((vals, (rows, cols)), (size, n))
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = []
    if solver:
        solvers_to_try.append(solver)
    solvers_to_try.extend([cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS, cp.GUROBI, cp.MOSEK])

    for s in solvers_to_try:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                sb = b_val.sum()
                if sb < 1e-15:
                    return None, None
                b_val = b_val / sb
                return b_val, float(max(0, M_var.value))
        except Exception:
            continue
    return None, None


def nesterov_lp_refine(a0, time_limit, solver, max_iters=100000):
    """Nesterov LP alternating method."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(789)
    lp_times = []

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        lp_t0 = time.monotonic()
        b_new, M_b = lp_best_response(y, solver)
        lp_times.append(time.monotonic() - lp_t0)

        if b_new is None:
            continue

        # Nesterov update
        alpha = 2.0 / (it + 2.0)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-9:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 8:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 5)
                noise = 0.03 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0

            if (it + 1) % 500 == 0:
                avg_lp = np.mean(lp_times[-20:]) if len(lp_times) >= 5 else 1.0
                print(f"    NesterovLP iter={it+1}: score={best_score:.8f} avg_lp={avg_lp*1000:.0f}ms")

        # Break if LP too slow
        if len(lp_times) > 20 and np.mean(lp_times[-20:]) > 2.0:
            print(f"    LP too slow, stopping at iter {it+1}")
            break

    return best_a, best_score


def coordinate_descent_refine(a0, time_limit, n_sweeps=50):
    """Coordinate descent: optimize one coordinate at a time."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(321)

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break

        # Shuffle coordinate order
        order = rng.permutation(n)
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            # Find optimal a[j] given others fixed
            # (a*a)[k] as function of a[j]: quadratic in a[j]
            # For each k, (a*a)[k] = a[j]*a[k-j] + a[k-j]*a[j] + rest
            #   = 2*a[j]*a[k-j] + rest_k (if k-j is valid and != j)
            #   or a[j]^2 + rest_k (if k-j == j, i.e., k = 2j)
            # We want to minimize max_k (a*a)[k] by choosing a[j] in [0, 1-sum_others]

            a_old = a[j]
            # Try a few candidate values
            candidates = [a_old * 0.5, a_old * 0.8, a_old * 1.2, a_old * 1.5,
                          a_old * 0.2, 0.0, a_old]
            for c in candidates:
                a[j] = c
                # Renormalize
                s_new = a.sum()
                if s_new < 1e-15:
                    a[j] = a_old
                    continue
                a_trial = a / s_new
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-12:
                    best_score = score
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    break
            a[j] = a_old  # restore if no improvement found in candidates

    return best_a, best_score


def structured_init(n, rng):
    """Create a structured initial sequence."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-6 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 2 + 0.05
    elif choice == 5:
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    elif choice == 6:
        # Multi-frequency
        a = 1.0 + 0.2 * np.cos(2 * np.pi * 3 * x) + 0.1 * np.cos(2 * np.pi * 7 * x)
        a = np.maximum(a, 0)
    else:
        # Random smooth
        a = np.exp(-0.5 * ((x - rng.uniform(0.2, 0.8)) / 0.15) ** 2) + 0.1
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Evaluate sequence score locally (fast, assuming sum(a)=1)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a_norm = a / s
    n = len(a_norm)
    b = fast_convolve(a_norm)
    return float(2.0 * n * b.max())

    for solver_name in [cp.GUROBI, cp.MOSEK, cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def perturb_sequence(a, rng, strength=0.02):
    """Create perturbed version of sequence."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 8)
    phase = rng.uniform(0, 2 * np.pi)
    perturbation = strength * np.cos(freq * x + phase)
    a_pert = a + perturbation
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s
