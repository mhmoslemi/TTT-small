"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multiplicative weight (mirror descent) method on the simplex
2. QP-based fictitious play (L2-norm best response)
3. Exact coordinate descent with FFT-accelerated convolution
4. Structured basin hopping
5. Multi-scale continuation
"""
import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * float(conv.max()) / (s * s)

def compute_score_normalized(a):
    """Assume a is already normalized (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * float(conv.max())

def subgradient_at_peaks(a, conv, n_peaks=20, temperature=None):
    """Compute subgradient of max_k conv[k] at the top peaks."""
    n = len(a)
    max_val = float(conv.max())
    if max_val <= 0:
        return np.zeros(n)
    
    # Get top peaks
    k_act = min(n_peaks, len(conv))
    peaks = np.argpartition(conv, -k_act)[-k_act:]
    pv = conv[peaks]
    
    if temperature is None:
        temperature = 0.01 * max_val
    
    w = np.exp((pv - pv.max()) / temperature)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.ones_like(w) / len(w)
    else:
        w = w / w_sum
    
    grad = np.zeros(n)
    for idx, k in enumerate(peaks):
        if w[idx] < 1e-12:
            continue
        k = int(k)
        # ∂conv(k)/∂a[i] = 2*a[k-i] if 0 <= k-i < n
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        j = np.arange(i_start, i_end + 1)
        partners = k - j  # these are k-i values
        grad[j] += 2.0 * w[idx] * a[partners]
    
    return grad

Let me also think about the QP approach more carefully. The QP best response is:
minimize (Qb)^T(Qb) = b^T Q^T Q b
subject to: sum(b) = 1, b >= 0

where Q is the (2n-1) x n convolution matrix.

This is a standard QP that can be solved with:
- Projected gradient: b_new = projection_simplex(b - eta * 2*Q^T*Q*b)
- Or cvxpy

The projected gradient method is O(n^2) per iteration (for Q^T Q vector product), which is comparable to ECD.

Actually, Q^T Q can be precomputed! Q^T Q is an n x n matrix where:
(Q^T Q)[i,j] = sum_k Q[k,i] * Q[k,j]

Q[k,i] = a[k-i] if 0 <= k-i < n, else 0.

So (Q^T Q)[i,j] = sum_k a[k-i] * a[k-j] (over valid k)
= sum_m a[m] * a[m + (i-j)] (where m = k-i, and m+(i-j) = k-j must be valid)

This is exactly the cross-correlation of a with itself shifted by (i-j)!

So Q^T Q = C where C[i,j] = corr(a, a)[i-j] (the autocorrelation of a).

This means Q^T Q is a Toeplitz matrix (constant along diagonals)! We can multiply by it in O(n log n) using FFT.

This is a huge speedup for the QP iterations.

So the QP best response algorithm becomes:
1. Precompute the autocorrelation r = conv(a, a) (which we already have)
2. To compute (Q^T Q) b = C b where C is Toeplitz with first column r[0:n] and first row r[0]:n]:
   - Use FFT: C b = IFFT(FFT(r_padded) * FFT(b_padded)) (circular convolution, then take first n elements)
   - Actually, C b[i] = sum_j C[i,j] b[j] = sum_j r[|i-j|] b[j] (approximately, for the Toeplitz structure)
   - More precisely: C[i,j] = r[i-j] if i>=j, r[j-i] if j>i (symmetric Toeplitz)
   - This is a convolution: (C b)[i] = sum_j r[|i-j|] b[j]
   - We can compute this as: circular convolution of r_sym with b, where r_sym[k] = r[k] for k=0..n-1, r_sym[k] = r[n-k] for k=n..2n-1
   
   Actually, let me think about this more carefully.
   
   r[k] = conv(a)[k] for k = 0, ..., 2n-2 (the full linear autocorrelation)
   
   C[i,j] = sum_{k: max(0,i+j-n+1) <= k <= min(i,j+n-1)} ...

   (Q^T Q)[i,j] = sum_k Q[k,i] * Q[k,j]

   Q[k,j] = a[k-j] if 0 <= k-j < n, else 0
   
   So (Q^T Q)[i,j] = sum_{k: 0<=k-i<n AND 0<=k-j<n} a[k-i] * a[k-j]
   = sum_{m: max(0,i-n+1) <= m <= min(i, n-1) AND max(0,j-n+1) <= m+(i-j) <= min(j, n-1) + (i-j)} ...
   
   This is getting messy. Let me just note that (Q^T Q) b can be computed as:
   
   (Q^T Q b)[i] = Q^T (Q b)[i] = sum_k Q[k,i] (Qb)[k]
   = sum_k a[k-i] * (Qb)[k]  (for valid k-i)
   
   And (Qb)[k] = sum_j Q[k,j] b[j] = sum_j a[k-j] b[j] (for valid k-j)

   So (Q^T Q b)[i] = sum_k a[k-i] * conv(a,b)[k] (for 0 <= k-i < n)
   = sum_{m=0}^{n-1} a[m] * conv(a,b)[m+i]
   = (a * conv(a,b))[i] in some sense...
   
   Actually: (Q^T Q b)[i] = sum_{m=0}^{n-1} a[m] * (Qb)[m+i]
   
   And (Qb)[m+i] = conv(a,b)[m+i] = sum_j a[m+i-j] b[j]
   
   So (Q^T Q b)[i] = sum_m a[m] * sum_j a[m+i-j] b[j]
   = sum_j b[j] * sum_m a[m] * a[m+i-j]
   = sum_j b[j] * conv(a,a)[i-j]   (where conv(a,a)[k] = sum_m a[m]*a[m+k]... wait, need to be careful with indexing)
   
   Let me define R[k] = conv(a,a)[k] = sum_i a[i]*a[k-i] (the full linear autocorrelation, length 2n-1).
   
   Then (Q^T Q b)[i] = sum_j b[j] * R[i-j]
   
   where R[k] = 0 for k < 0 or k >= 2n-1.
   
   So (Q^T Q b)[i] = sum_j b[j] * R[i-j] = (b * R)[i] (circular or linear convolution of b with R).
   
   Since R has length 2n-1 and b has length n, the linear convolution has length 3n-2. We only need the first n elements.
   
   We can compute this with FFT:
   - Zero-pad both to length >= 3n (or next power of 2)
   - FFT both, multiply, IFFT, take first n elements

Actually, one more consideration: the simplex projection. The standard algorithm for projecting onto the probability simplex is:
1. Sort the vector in descending order

3. theta = (sum_{j=1}^{rho} u[j] - 1) / rho
4. Return max(v - theta, 0)

This is O(n log n) per projection.

For the QP, the iteration is:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated QP fictitious play (L2-norm best response)
2. Soft-max gradient descent (smooth max approximation)
3. Multiplicative weight (mirror descent) on the simplex
4. Exact coordinate descent with golden section
5. Structured basin hopping with diverse perturbations
6. Multi-restart across different sequence lengths

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(a, b):
    na = len(a)
    nb = len(b)
    size = na + nb - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * float(conv.max()) / (s * s)


def compute_score_norm(a):
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * float(conv.max())


def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def qp_matvec(a_fixed, b):
    """Compute Q^T Q b where Q is the convolution matrix of a_fixed.
    Uses FFT for O(n log n) complexity."""
    n = len(a_fixed)
    R = fast_convolve(a_fixed)  # length 2n-1
    # (Q^T Q b)[i] = sum_j b[j] * R[i-j] for i=0..n-1
    # This is the linear convolution of b with R, taking first n elements
    size = n + len(R) - 1
    fft_size = 1 << (size - 1).bit_length()
    B = rfft(b, fft_size)
    Rc = rfft(R, fft_size)
    result = irfft(B * Rc, fft_size)[:n]
    return result


def qp_best_response(a_fixed, n_iter=200, eta0=0.1):
    """Solve min b^T Q^T Q b s.t. sum(b)=1, b>=0 using projected gradient."""
    n = len(a_fixed)
    b = np.ones(n) / n
    eta = eta0
    best_b = b.copy()
    best_obj = float('inf')
    
    for it in range(n_iter):
        g = qp_matvec(a_fixed, b)
        b_new = simplex_project(b - eta * 2.0 * g)
        obj = float(b_new @ qp_matvec(a_fixed, b_new))
        if obj < best_obj:
            best_obj = obj
            best_b = b_new.copy()
        b = b_new
        eta *= 0.995
    
    return best_b


def softmax_gradient_descent(a0, time_limit, n_peaks=50, beta_range=(5, 50), verbose=False):
    """Optimize using soft-max approximation of the max function."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = compute_score_norm(a)
    t0 = time.monotonic()
    lr = 0.1
    iters = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        max_val = float(conv.max())
        
        # Soft-max weights over all lags
        beta = 20.0 + 20.0 * min(1.0, iters / 100.0)
        scaled = beta * (conv - max_val)
        exp_scaled = np.exp(scaled)
        w = exp_scaled / exp_scaled.sum()
        
        # Subgradient: grad[i] = sum_k w[k] * 2*a[k-i] (with bounds)
        grad = np.zeros(n)
        # Use FFT for efficiency: grad = 2 * (w * a_reversed) truncated
        # Actually, grad[i] = 2 * sum_k w[k] * a[k-i]
        # = 2 * (w reversed convolved with a)[offset]
        # Let's just do it directly for the top peaks
        k_act = min(n_peaks, 2 * n - 1)
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        pw = w[peaks]
        
        for idx in range(len(peaks)):
            if pw[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            partners = k - j
            grad[j] += 2.0 * pw[idx] * a[partners]
        
        a_new = a - lr * grad
        a_new = simplex_project(a_new)
        
        sc = compute_score_norm(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        lr *= 0.999
        
        if no_improve > 50:
            lr *= 0.8
            no_improve = 0
            # Add small perturbation to escape
            a = a + 0.01 * np.random.randn(n)
            a = simplex_project(a)
        
        if no_improve > 300:
            break
        
        if verbose and iters % 50 == 0:
            print(f"      SMD it={iters}: score={best_score:.8f}")
    
    return best_a, best_score


def multiplicative_weight_descent(a0, time_limit, verbose=False):
    """Mirror descent with entropy regularization on the simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = compute_score_norm(a)
    t0 = time.monotonic()
    eta = 0.5
    iters = 0
    no_improve = 0
    n_peaks = 30
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        max_val = float(conv.max())
        
        # Get top peaks with softmax weights
        k_act = min(n_peaks, 2 * n - 1)
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        pv = conv[peaks]
        temp = 0.05 * max_val
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            break
        w = w / ws
        
        grad = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            partners = k - j
            grad[j] += 2.0 * w[idx] * a[partners]
        
        # Multiplicative weight update
        log_a = np.log(a + 1e-30)
        log_a_new = log_a - eta * grad
        a_new = np.exp(log_a_new - log_a_new.max())
        a_new = a_new / a_new.sum()
        
        sc = compute_score_norm(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Averaged iterate
            a_avg = (iters * a + a_new) / (iters + 1)
            sc_avg = compute_score_norm(a_avg)
            if sc_avg < best_score:
                best_score = sc_avg
                best_a = a_avg.copy()
        
        a = a_new
        eta = 2.0 / np.sqrt(iters + 1.0)
        
        if no_improve > 100:
            break
    
    return best_a, best_score

            if has_quad:

                c_pt = hi - gr * (hi - lo)
                d_pt = lo + gr * (hi - lo)
                
                def score_at(x):
                    nv = c_vals + 2.0 * x * a_partners
                    if has_quad:
                        nv = nv.copy()
                        nv[quad_idx] = c_vals[quad_idx] + x * x
                    return 2.0 * n * max(unaff_max, float(nv.max()))
                
                fc = score_at(c_pt)
                fd = score_at(d_pt)
                for _ in range(20):
                    if fc <= fd:
                        hi = d_pt
                        d_pt = c_pt
                        fd = fc
                        c_pt = hi - gr * (hi - lo)
                        fc = score_at(c_pt)
                    else:
                        lo = c_pt
                        c_pt = d_pt
                        fc = fd
                        d_pt = lo + gr * (hi - lo)
                        fd = score_at(d_pt)
                x_gold = (lo + hi) * 0.5
                sg = score_at(x_gold)
                if sg < best_local:
                    best_local = sg
                    best_x = x_gold
            
            if abs(best_x - v) > 1e-13:
                delta = best_x - v
                a[i] = best_x
                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes
                
                cur_score = 2.0 * n * float(conv.max())
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
        
        if verbose and sweeps % 10 == 0:
            print(f"      ECD {label} sweep={sweeps}: best={best_score:.8f}")
    
    return best_a, best_score


def lp_best_response(a_fixed, solver_name=None):
    """Solve LP: min M s.t. conv(a_fixed * b) <= M, sum(b)=1, b>=0."""
    import cvxpy as cp
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build conv matrix sparsely
    # conv(a*b)[k] = sum_i a_fixed[i] * b[k-i]
    # We need: for each k, sum_i a_fixed[i] * b[k-i] <= M
    # This is: sum_i a_fixed[k-i] * b[i] <= M (reindexing)
    
    # Use scipy sparse for the constraint matrix
    from scipy.sparse import lil_matrix
    T = lil_matrix((2 * n - 1, n))
    for i in range(n):
        for k in range(i, i + n):
            if k < 2 * n - 1:
                T[k, i] = a_fixed[k - i]
    
    T_csr = T.tocsr()
    constraints = [T_csr @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    solvers_to_try = []
    if solver_name:
        solvers_to_try.append(solver_name)
    solvers_to_try.extend(['ECOS', 'CLARABEL', 'SCS', 'GLPK', 'PDLP'])
    
    for sol in solvers_to_try:
        try:
            prob.solve(solver=sol)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s < 1e-15:
                    continue
                b_val = b_val / s
                M_val = float(M_var.value)
                return b_val, M_val
        except Exception:
            continue
    return None, None


def fictitious_play_qp(a0, time_limit, n_iter_per_step=100, verbose=False):
    """Fictitious play using QP best response (L2 minimization)."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = compute_score_norm(a)
    t0 = time.monotonic()
    iters = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 10:
            break
        
        # QP best response
        b = qp_best_response(a, n_iter=n_iter_per_step)
        
        # Try different mixing fractions
        best_t = 0.0
        best_ls = best_score
        for trial_t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            cand = (1.0 - trial_t) * a + trial_t * b
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-10:
                continue
            cand = cand / cs
            sc = compute_score_norm(cand)
            if sc < best_ls:
                best_ls = sc
                best_t = trial_t
        
        if best_t > 0:
            a_new = (1.0 - best_t) * a + best_t * b
            a_new = np.maximum(a_new, 0)
            cs = a_new.sum()
            if cs < 1e-10:
                a_new = a
            else:
                a_new = a_new / cs
            score_new = compute_score_norm(a_new)
            if score_new < best_score - 1e-12:
                best_score = score_new
                best_a = a_new.copy()
            a = a_new
        else:
            # Use averaged iterate
            a = 0.7 * a + 0.3 * b
            a = np.maximum(a, 0)
            a = a / a.sum()
        
        if verbose and iters % 5 == 0:
            print(f"      FQ it={iters}: best={best_score:.8f}")
    
    return best_a, best_score


def basin_hopping(a0, time_limit, rng, verbose=False):
    """Basin hopping with diverse perturbations followed by ECD."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score_norm(a)
    t0 = time.monotonic()
    count = 0
    
    while time.monotonic() - t0 < time_limit:
        count += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 30:
            break
        t_alloc = min(rem * 0.3, 60.0)
        
        pert_type = count % 5
        perturbed = a.copy()
        
        if pert_type == 0:
            # Gaussian noise
            sigma = 0.03 + 0.04 * rng.random()
            perturbed = a + rng.standard_normal(n) * sigma
        elif pert_type == 1:
            # Peak-focused reduction
            conv_p = fast_convolve(a)
            top_k = min(20, 2 * n - 1)
            peaks = np.argpartition(conv_p, -top_k)[-top_k:]
            for k in peaks:
                k = int(k)
                i_start = max(0, k - n + 1)
                i_end = min(n - 1, k)
                idx = np.arange(i_start, i_end + 1)
                n_pick = max(1, len(idx) // 3)
                picks = rng.choice(idx, size=n_pick, replace=False)
                for ii in picks:
                    perturbed[ii] *= (1.0 - 0.05 * rng.random())
        elif pert_type == 2:
            # Block permutation
            n_blocks = rng.integers(3, 8)
            block_size = n // n_blocks
            perm = rng.permutation(n_blocks)
            for i in range(n_blocks):
                start = i * block_size
                end = (i + 1) * block_size if i < n_blocks - 1 else n
                pstart = perm[i] * block_size
                pend = (perm[i] + 1) * block_size if perm[i] < n_blocks - 1 else n
                perturbed[pstart:pend] = a[start:end]
        elif pert_type == 3:
            # Frequency domain perturbation
            fft_size = 1 << (2 * n - 1).bit_length()
            A = rfft(a, fft_size)
            A_magnitude = np.abs(A)
            A_magnitude *= (1.0 + 0.1 * rng.standard_normal(len(A_magnitude)))
            A_magnitude = np.maximum(A_magnitude, 0)
            A_new = A_magnitude * np.exp(1j * np.angle(A))
            a_fd = irfft(A_new, fft_size)[:n]
            perturbed = a_fd
        else:
            # Asymmetric shift
            shift = rng.integers(1, n // 5)
            shifted = np.zeros(n)
            shifted[shift:] = a[:n-shift]
            perturbed = 0.6 * a + 0.4 * shifted
        
        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-10:
            continue
        perturbed = perturbed / ps
        
        a_opt, s_opt = exact_coord_descent(perturbed, t_alloc, label=f"bh_{count}")
        if s_opt < best_score - 1e-10:
            best_score = s_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"      BH {count}: improved to {best_score:.8f}")
    
    return best_a, best_score


def init_gaussian(n, rng, center_frac=0.4, width_frac=0.25):
    t = np.arange(n, dtype=np.float64)
    center = n * center_frac
    width = n * width_frac
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def init_parametric(n, rng):
    """Initialize with a parametric family that should give low peak."""
    t = np.arange(n, dtype=np.float64)
    x = t / n
    # Sum of cosines with random frequencies and phases
    n_harm = rng.integers(2, 6)
    a = np.ones(n)
    for _ in range(n_harm):
        freq = rng.uniform(0.5, 3.0)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.1, 0.4)
        a += amp * np.cos(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def init_from_spectrum(n, rng):
    """Initialize by inverting a flattened spectrum."""
    fft_size = 1 << (2 * n - 1).bit_length()
    n_freq = fft_size // 2 + 1
    # Flat spectrum with small random variations
    amp = np.ones(n_freq) * (1.0 + 0.1 * rng.standard_normal(n_freq))
    amp = np.maximum(amp, 0.1)
    phase = rng.uniform(0, 2 * np.pi, n_freq)
    A = amp * np.exp(1j * phase)
    a = irfft(A, fft_size)[:n]
    a = np.abs(a)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    new_x = np.linspace(0, 1, new_n)
    old_x = np.linspace(0, 1, old_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
