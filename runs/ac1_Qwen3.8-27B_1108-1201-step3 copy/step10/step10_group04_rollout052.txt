def fast_convolve(a):
    """Fast autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_cross(a, b):
    """Fast cross-correlation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a_fixed):
    """Build the convolution matrix T such that T@b = conv(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    af = np.asarray(a_fixed)
    for j in range(n):
        T[j:j + n, j] = af
    return T

    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def compute_subgradient(a, n_active_frac=0.1):
    """Compute subgradient of max_k (a*a)_k using active set."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    if max_b < 1e-300:
        return np.zeros(n), max_b
    threshold = max_b - n_active_frac * max_b
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]
    g /= len(active)
    return g, max_b

def lbgfs_logsumexp_enhanced(a0, time_limit, T_start=500.0, T_end=0.01, n_stages=15):
    """
    L-BFGS with LogSumExp, enhanced with:
    - More stages for better annealing
    - Gradient checkpointing
    - Better temperature schedule (geometric with finer steps)
    """
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    
    def lse_objective(z_in, T_val):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = softmax_from_z(z_in)
        b = fast_convolve(a)
        Tb = T_val * b
        max_Tb = Tb.max()
        if max_Tb > 700:
            max_Tb = 700
        exp_Tb = np.exp(Tb - max_Tb)
        log_sum_exp = max_Tb + np.log(exp_Tb.sum())
        return (2.0 * n / T_val) * log_sum_exp
    
    def lse_jacobian(z_in, T_val):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z_in)
        a = softmax_from_z(z_in)
        b = fast_convolve(a)
        Tb = T_val * b
        max_Tb = Tb.max()
        if max_Tb > 700:
            max_Tb = 700
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()
        # Compute gradient w.r.t. a, then chain rule
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]
        S = np.dot(grad_a, a)
        return a * (grad_a - S)
    
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(30, int(200 / n_stages))
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        def func(z_in):
            return lse_objective(z_in, stage_T)
        
        def jac(z_in):
            return lse_jacobian(z_in, stage_T)
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-15, 'maxfun': iters_per_stage * 1000,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    return softmax_from_z(z)

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp

import numpy as np

def fast_convolve(a):
    """Fast autocorrelation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_cross(a, b):
    """Fast cross-correlation via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a_fixed):
    """Build convolution matrix T such that T@b = full_conv(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    af = np.asarray(a_fixed, dtype=np.float64)
    for j in range(n):
        T[j:j + n, j] = af
    return T


def softmax_from_z(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Log of positive array."""
    return np.log(np.maximum(a, 1e-300))


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:

            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_cg(a_fixed, b_init=None, solver=None, max_constraints=250):
    """
    Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0
    using constraint generation (cutting planes).
    Returns (b_opt, M_opt).
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    size = 2 * n - 1

    if b_init is None:
        b = np.ones(n) / n
    else:
        b = np.maximum(np.asarray(b_init, dtype=np.float64), 0)
        s = b.sum()
        if s < 1e-15:
            b = np.ones(n) / n
        else:
            b = b / s

    n_constraints = min(max_constraints, size)

    for cg_iter in range(12):
        # Select constraints based on current b
        conv = fast_convolve_cross(a_fixed, b)
        M_cur = conv.max()

        if size > n_constraints:
            top_idx = np.argsort(conv)[::-1][:n_constraints]
        else:
            top_idx = np.arange(size)

        T_sub = build_conv_matrix(a_fixed)[top_idx]
        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [T_sub @ b_var <= M_var, cp.sum(b_var) == 1]

        prob = cp.Problem(cp.Minimize(M_var), constraints)

        solved = False
        for s_name in [solver, cp.PDLP, cp.SCS, cp.ECOS]:
            try:
                prob.solve(solver=s_name)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
                    break
            except Exception:
                continue

        if not solved:
            return b, float(M_cur)

        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0)
        s_val = b_val.sum()
        if s_val < 1e-15:
            return b, float(M_cur)
        b_val = b_val / s_val

        # Check for violated constraints
        full_conv = fast_convolve_cross(a_fixed, b_val)
        M_new = full_conv.max()
        M_sub = full_conv[top_idx].max()

        if M_new - M_sub < 1e-7 * max(M_new, 1e-10):
            return b_val, float(M_new)

        # Grow constraint set
        b = b_val
        n_constraints = min(n_constraints * 2, size)

    return b_val, float(M_new)


def compute_subgradient(a, n_active_frac=0.1):
    """Compute subgradient of max_k (a*a)_k using active set."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    if max_b < 1e-300:
        return np.zeros(n), max_b
    threshold = max_b - n_active_frac * max_b
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]
    g /= len(active)
    return g, max_b

def lbgfs_logsumexp_opt(a0, time_limit, T_start=300.0, T_end=0.01, n_stages=12):
    """L-BFGS with LogSumExp, multi-stage annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    def lse_obj(z_in, T_val):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = softmax_from_z(z_in)
        b = fast_convolve(a)
        Tb = T_val * b
        max_Tb = Tb.max()
        if max_Tb > 700:
            max_Tb = 700
        exp_Tb = np.exp(Tb - max_Tb)
        log_sum_exp = max_Tb + np.log(exp_Tb.sum())
        return (2.0 * n / T_val) * log_sum_exp

    def lse_grad(z_in, T_val):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z_in)
        a = softmax_from_z(z_in)
        b = fast_convolve(a)
        Tb = T_val * b
        max_Tb = Tb.max()
        if max_Tb > 700:
            max_Tb = 700
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]
        S = np.dot(grad_a, a)
        return a * (grad_a - S)

    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(20, int(150 // n_stages))

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        def func(z_in):
            return lse_obj(z_in, stage_T)

        def jac(z_in):
            return lse_grad(z_in, stage_T)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-14, 'maxfun': iters_per_stage * 500,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def peak_shaving_opt(a0, time_limit, n_steps=50000):
    """Peak shaving local search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')

    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

            break
        if no_improve > 3000:
            noise = rng.random(n) * 0.02
            a = a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            no_improve = 0

        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_k - idx]
        src = idx[np.argmax(contrib)]

        improved = False
        dst_candidates = np.argsort(a)[:15]
        for dst in dst_candidates:
            if dst == src:
                continue
            for frac in [0.02, 0.05, 0.1, 0.15]:
                amt = a[src] * frac
                if amt < 1e-14:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    no_improve = 0
                    improved = True
                    break
            if improved:
                break

        if not improved:
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.01, 0.08)
                a_trial = a.copy()
                amt = a[i] * frac
                a_trial[i] -= amt

                        best_score = score

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    else:
        f_interp = np.ones(new_n)
