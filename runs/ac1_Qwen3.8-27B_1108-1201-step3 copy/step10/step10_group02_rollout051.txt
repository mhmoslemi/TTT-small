"""
Saddle-point accelerated optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses FFT-based cross-correlation for O(n log n) iterations, Nesterov acceleration,
adaptive step sizes, periodic LP refinement, and coordinate descent polishing.
Key innovation: treats the max as a game (minimax) with a learned peak distribution,
giving a smooth weighted subgradient that adapts to which peaks are most problematic.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft


def fast_auto_conv(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = rfft(a, N)
    return irfft(A * A, N)[:size]


def cross_corr_weighted(a, p):
    """
    Compute r[j] = sum_i a[i] * p[i+j] for j=0..n-1.
    This is Q(p) @ a, the 'gradient' of the bilinear form w.r.t. a.
    """
    n = len(a)
    n_p = len(p)
    size = n + n_p - 1
    N = 1 << (size - 1).bit_length()
    a_rev = a[::-1]
    result = irfft(rfft(a_rev, N) * rfft(p, N), N)
    return result[n - 1: n - 1 + n]


def evaluate_local(a):
    """Evaluate the score locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a = a / s
    n = len(a)
    b = fast_auto_conv(a)
    return float(2.0 * n * b.max())


def build_conv_matrix_rows(a, peak_indices):
    """Build rows of the convolution matrix for specific peak indices."""
    n = len(a)
    rows = []
    for k in peak_indices:
        row = np.zeros(n)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        row[js:je + 1] = a[k - js:k - js + n][::-1]
        rows.append(row)
    return np.array(rows)


def solve_lp_best_response(a_fixed, peak_indices=None, solver=None):
    """Solve LP: min M s.t. conv(a_fixed, b)[k] <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    if peak_indices is None:
        peak_indices = list(range(2 * n - 1))
    else:
        peak_indices = list(peak_indices)

    T_rows = build_conv_matrix_rows(a_fixed, peak_indices)
    constraints = [T_rows @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    for s in [solver, cp.ECOS, cp.SCS, cp.CLARABEL]:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                bs = b_val.sum()
                if bs < 1e-15:
                    return None, None
                return b_val / bs, float(M_var.value)
        except Exception:
            continue
    return None, None

    t0 = time.monotonic()
    iteration = 0
    no_improve = 0

def coord_descent_2var(a0, time_limit, n_rounds=200000):

        best_factor = 1.0
        for factor in [0.99, 0.98, 0.97, 0.95, 0.93, 0.90, 0.85, 0.80]:
            delta = a[top_pos] * (1.0 - factor)
            if delta < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[top_pos] = a[top_pos] * factor
            # Redistribute delta proportionally to other top contributors
            mask = np.ones(n, dtype=bool)
            mask[top_pos] = False
            others = a_trial[mask].copy()
            others_sum = others.sum()
            if others_sum < 1e-15:
                break
            a_trial[mask] = others + delta * others / others_sum
            a_trial = a_trial / a_trial.sum()

            # Quick score estimation
            At = rfft(a_trial, N)
            conv_t = irfft(At * At, N)[:2*n-1]
            score = 2.0 * n * conv_t.max()

            if score < best_local:
                best_local = score
                best_factor = factor

        if best_local < best_score - 1e-14:
            a[top_pos] *= best_factor
            delta = a0_sum if (a0_sum := a.sum()) else 1.0  # placeholder
            # Redo properly
            a_prev_sum = a[top_pos] / best_factor  # approx
            a = a / a.sum()
            best_score = best_local
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall % 1000 == 0 and stall > 5000:
                # Perturb and try
                a = a + rng.standard_normal(n) * 0.002
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0

        if (rnd + 1) % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Coord2V {elapsed:.0f}s: best={best_score:.8f}")

    return best_a, best_score


def frank_wolfe_refine(a0, time_limit, n_iters=5000000):
    """
    Frank-Wolfe (conditional gradient) with away steps for the convex
    reformulation: minimize max_k conv(a,a)[k] s.t. a in simplex.
    Uses the subgradient of the max to determine the FW direction.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    support_size = 0
    N = 1 << ((2 * n - 1) - 1).bit_length()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute convolution and find active peak
        A = rfft(a, N)
        conv_a = irfft(A * A, N)[:2*n-1]
        max_b = conv_a.max()
        # Use top peaks for smoother gradient
        threshold = max_b * 0.999
        active = np.where(conv_a >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(conv_a))])

        # Subgradient: g = (2/|S|) * sum_{k in S} (P_k a)
        g = np.zeros(n)
        for k in active[:min(len(active), 50)]:  # Limit to top 50
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= max(1, min(len(active), 50))

        # FW step: find min of <g, s> over simplex = argmin g
        min_idx = int(np.argmin(g))

        # Away step: check if removing a support point helps
        if support_size > 1:
            # Find worst support point (highest g value among small coords)
            small_idx = np.where(a > 1e-8)[0]
            if len(small_idx) > 1:
                away_idx = small_idx[np.argmax(g[small_idx])]
                # Test away step
                a_away = a.copy()
                a_away[away_idx] = 0
                a_away_sum = a_away.sum()
                if a_away_sum > 1e-10:
                    a_away /= a_away_sum
                    Aa = rfft(a_away, N)
                    conv_away = irfft(Aa * Aa, N)[:2*n-1]
                    if conv_away.max() < max_b * 0.9999:
                        a = a_away
                        support_size = max(0, support_size - 1)

        # Standard FW update
        t = 2.0 / (it + 2)
        a = (1.0 - t) * a + t * (np.eye(n)[min_idx])
        a = np.maximum(a, 0)
        a_sum = a.sum()
        if a_sum < 1e-15:
            a = a[0] if False else np.ones(n) / n
        else:
            a = a / a_sum

        if a[min_idx] > 1e-8 and not (a[min_idx] > 0.5 and support_size < 50):
            support_size = min(n, support_size + 1)

        # Evaluate periodically
        if (it + 1) % 2000 == 0:
            score = 2.0 * n * conv_a.max()  # approximate (stale)
            # Quick recompute for best tracking
            A_eval = rfft(a, N)
            conv_eval = irfft(A_eval * A_eval, N)[:2*n-1]
            score_eval = 2.0 * n * conv_eval.max()
            if score_eval < best_score - 1e-12:
                best_score = score_eval
                best_a = a.copy()

        if (it + 1) % 100000 == 0:
            print(f"    FW iter={it+1} ({time.monotonic()-t0:.0f}s): "
                  f"best={best_score:.8f} support={support_size}")

    return best_a, best_score


def structured_init(n, rng, style=0):
    """Generate structured initial sequences with known good properties."""
    x = np.linspace(0, 1, n)
    if style == 0:
        # Raised cosine
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif style == 1:
        # Gaussian center
        a = np.exp(-5.0 * (x - 0.5) ** 2 / 0.1)
    elif style == 2:
        # Power law
        a = (1 - x * (1 - x)) ** 2
    elif style == 3:
        # Two humps
        a = np.exp(-8 * (x - 0.3) ** 2 / 0.05) + np.exp(-8 * (x - 0.7) ** 2 / 0.05)
    elif style == 4:
        # Flat with cosine edge tapering
        a = np.ones(n)
        taper = max(1, n // 10)
        for i in range(taper):
            w = 0.5 * (1 + np.cos(np.pi * i / taper))
            a[i] = w
            a[n-1-i] = w
    elif style == 5:
        # Symmetric peak
        a = np.sin(np.pi * x) ** 3
    else:
        a = np.cos(np.pi * x / 2) ** 1.5
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining saddle point, LP-accelerated methods,
    and coordinate descent to minimize 2*n*max(conv(a,a))/sum(a)^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    # Find LP solver
    solver = None
    for s in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solver = s
                break
        except Exception:
            continue
    print(f"LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng, style=0)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Saddle Point Optimizer (main workhorse) ===
    print(f"\n=== Phase 1: Saddle Point (score={best_score:.8f}, n={n_best}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.40, 650.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_sp, _ = saddle_point_optimize(best_seq, t_alloc, solver=solver,
                                        eta_a0=0.25, eta_p0=0.25,
                                        nesterov=True, adaptive=True)
        try_update(a_sp, "saddle_point_p1")

    # === Phase 2: LP-Accelerated Saddle Point ===
    print(f"\n=== Phase 2: LP-Accelerated SP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 30 and solver is not None:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = lp_accelerated_saddle(best_seq, t_alloc, solver=solver,
                                        lp_every=150, n_peaks=15)
        try_update(a_lp, "lp_saddle")

    # === Phase 3: Second Saddle Point with different params ===
    print(f"\n=== Phase 3: Saddle Point v2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        # Start from current best with slight perturbation
        a_pert = best_seq + rng.standard_normal(len(best_seq)) * 0.01
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()
        a_sp2, _ = saddle_point_optimize(a_pert, t_alloc, solver=solver,
                                         eta_a0=0.15, eta_p0=0.15,
                                         nesterov=True, adaptive=True)
        try_update(a_sp2, "saddle_point_p2")

    # === Phase 4: Coordinate Descent with 2-variable sub-solves ===
    print(f"\n=== Phase 4: Coord Descent 2V (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 10:
        a_cd, _ = coord_descent_2var(best_seq, t_alloc, n_rounds=100000)
        try_update(a_cd, "coord_2var")

    # === Phase 5: Final Saddle Point polish with low step size ===
    print(f"\n=== Phase 5: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.05, 60.0)
    if t_alloc > 5:
        a_fin, _ = saddle_point_optimize(best_seq, t_alloc, solver=solver,
                                         eta_a0=0.05, eta_p0=0.05,
                                         nesterov=True, adaptive=True)
        try_update(a_fin, "final_polish")

    # === Phase 6: Try different n values ===
    print(f"\n=== Phase 6: Multi-n exploration (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        for n_try in [int(n_best * 0.8), int(n_best * 1.2)]:
            if rem < 15:
                break
            if n_try < 100 or n_try > 3000:
                continue
            rem = deadline - time.monotonic()
            t_alloc = min(rem * 0.4, 60.0)
            if t_alloc < 5:
                break
            # Interpolate parent to new size
            old_n = len(best_seq)
            old_x = np.linspace(0, 1, old_n)
            new_x = np.linspace(0, 1, n_try)
            a_interp = np.interp(new_x, old_x, best_seq)
            a_interp = np.maximum(a_interp, 0)
            a_interp = a_interp / a_interp.sum()

            a_new_n, _ = saddle_point_optimize(a_interp, t_alloc, solver=solver,
                                               eta_a0=0.3, eta_p0=0.3,
                                               nesterov=True, adaptive=True)
            try_update(a_new_n, f"n={n_try}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
