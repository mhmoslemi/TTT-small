def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def normalize(a):

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    """Pairwise mass-transfer moves."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    else:
        a = a / s

    conv = fast_convolve(a).copy()
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1

        # Pick pairs based on active peaks
        peak_thresh = M * 0.9
        peak_positions = np.where(conv >= peak_thresh)[0]
        if len(peak_positions) == 0:
            peak_positions = np.array([np.argmax(conv)])

        improved = False
        rng_pair = np.random.default_rng(hash((iters, n)) % 2**31)

        # For each peak, try transferring mass within its support
        for k in peak_positions[:min(20, len(peak_positions))]:
            if time.monotonic() - t0 > time_limit:
                break
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            indices = np.arange(i_start, i_end + 1)
            if len(indices) < 2:
                continue

            # Try a few random pairs
            n_tries = min(len(indices), 5)
            for _ in range(n_tries):
                pair = rng_pair.choice(indices, size=2, replace=False)
                i, j = int(pair[0]), int(pair[1])
                ai, aj = a[i], a[j]
                if ai < 1e-15 or aj < 1e-15:
                    continue

                # Transfer delta from j to i (or vice versa)
                # conv change for +delta at i, -delta at j:
                # For each k': delta * [coeff_i(k') - coeff_j(k')]
                # where coeff_i(k') = a[k'-i] + (a[i] if k'-i != i else 0) for valid k'-i
                # This is complex, so we just evaluate a few candidate deltas

                # Compute affected range
                ks = min(i, j)
                ke = max(i + n - 1, j + n - 1)
                ke = min(ke, 2 * n - 2)

                unaff = -1e30
                if ks > 0:
                    unaff = max(unaff, float(conv[:ks].max()))
                if ke < 2 * n - 2:
                    unaff = max(unaff, float(conv[ke + 1:].max()))

                # Compute linear coefficients for the transfer
                # delta at i: affects conv[k] for k in [i, i+n-1]
                # -delta at j: affects conv[k] for k in [j, j+n-1]
                lin = np.zeros(ke - ks + 1)
                for k in range(ks, ke + 1):
                    c = 0.0
                    ki = k - i
                    if 0 <= ki < n:
                        c += a[ki]
                        if ki != i:
                            c += ai
                    kj = k - j
                    if 0 <= kj < n:
                        c -= a[kj]
                        if kj != j:
                            c -= aj
                    lin[k - ks] = c

                # Quadratic terms
                quad_i = (2 * i >= ks) and (2 * i <= ke)
                quad_j = (2 * j >= ks) and (2 * j <= ke)

                # Try candidate deltas
                max_delta = min(ai, aj) * 0.5
                best_d = 0.0
                best_m = M

                for frac in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
                    for sign in [1, -1]:
                        d = sign * frac * max_delta
                        if abs(d) > min(ai, aj) + 1e-15:
                            continue
                        new_vals = conv[ks:ke + 1] + d * lin
                        if quad_i:
                            new_vals[2 * i - ks] += d * d
                        if quad_j:
                            new_vals[2 * j - ks] += d * d
                        nm = max(unaff, float(new_vals.max()))
                        if nm < best_m - 1e-16:
                            best_m = nm
                            best_d = d

                if best_d != 0.0 and best_m < M - 1e-15:
                    a[i] = ai + best_d
                    a[j] = aj - best_d
                    a = np.maximum(a, 0)
                    conv = fast_convolve(a).copy()
                    M = float(conv.max())
                    cur_score = 2.0 * n * M
                    if cur_score < best_score - 1e-14:
                        best_score = cur_score
                        best_a = a.copy()
                    improved = True

        if iters % 50 == 0:
            conv = fast_convolve(a).copy()
            M = float(conv.max())

        if not improved and iters > 100:
            break

        if verbose and iters % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Pairwise {label} it={iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def lp_fictitious_play(a0, time_limit, label=""):
    """LP fictitious play with adaptive damping."""
    import cvxpy as cp

    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    iters = 0

    # Prebuild T matrix
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a  # This changes as a changes, so we rebuild

    # Find a solver
    solver = None
    for sn in [cp.GUROBI, cp.MOSEK, cp.GLOP, cp.PDLP, cp.SCS, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=sn)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solver = sn
                break
        except Exception:
            continue
    if solver is None:
        solver = cp.ECOS

    b = a.copy()

    while time.monotonic() - t0 < time_limit:
        iters += 1

        # Rebuild T with current a
        T = np.zeros((2 * n - 1, n))
        for j in range(n):
            T[j:j + n, j] = a

        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver)
        except Exception:
            try:
                prob.solve(solver=cp.SCS)
            except Exception:
                continue

        if prob.status not in ('optimal', 'optimal_inaccurate'):
            continue

        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0)
        sv = b_val.sum()
        if sv < 1e-15:
            continue
        b_val = b_val / sv

        # Adaptive damping
        alpha = min(1.0, 5.0 / (iters + 3))
        a_new = (1 - alpha) * a + alpha * b_val
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        # Also check b directly
        score_b = 2.0 * n * fast_convolve(b_val).max()
        if score_b < best_score - 1e-14:
            best_score = score_b
            best_a = b_val.copy()

        if iters % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LPFP {label} it={iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def multiplicative_weights(a0, time_limit, label=""):
    """Multiplicative weights / no-regret approach for the max-convolution problem."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    eta = 2.0
    w = np.ones(2 * n - 1) / (2 * n - 1)

    while time.monotonic() - t0 < time_limit:
        it += 1

        conv_a = fast_convolve(a)
        # Update weights (no-regret on the lag dimension)
        w = w * np.exp(eta * conv_a / max(1e-15, conv_a.max()))
        w = w / w.sum()

        # Compute gradient-like direction
        # grad[j] = sum_k w[k] * a[k-j] (for valid k-j)
        grad = np.zeros(n)
        active_k = np.where(w > 1e-12)[0]
        for k in active_k:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                grad[i] += w[k] * a[k - i]

        # Mirror descent step
        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 0)
        ss = a.sum()
        if ss < 1e-15:
            a = np.ones(n) / n
            ss = 1.0
        a = a / ss

        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()

        # Adaptive eta
        if it % 50 == 0:
            eta *= 0.95
            if eta < 0.01:
                break

        if it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MW {label} it={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def active_set_perturb(a0, rng, strength=0.03):
    """Perturb by reducing mass at coordinates contributing to peaks."""
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a)
    M = conv.max()
    peak_threshold = M * 0.85
    peak_positions = np.where(conv >= peak_threshold)[0]

    pressure = np.zeros(n)
    for k in peak_positions:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            pressure[i] += a[i] * a[j]

    p_max = pressure.max()
    if p_max > 1e-20:
        pressure = pressure / p_max

    # Reduce at high-pressure coordinates, add to low-pressure
    mask_high = pressure > 0.7
    mask_low = pressure < 0.3
    n_high = mask_high.sum()
    n_low = mask_low.sum()

    if n_high > 0 and n_low > 0:
        a[mask_high] *= (1.0 - strength)
        a[mask_low] *= (1.0 + strength * n_high / max(1, n_low))
        a = np.maximum(a, 0)
        ss = a.sum()
        if ss > 1e-15:
            a = a / ss
    else:
        # Fallback: Gaussian perturbation
        a = a * (1.0 + rng.standard_normal(n) * strength * 0.5)
        a = np.maximum(a, 0)
        ss = a.sum()
        if ss > 1e-15:
            a = a / ss

    return a


def structured_restart(a0, rng, frac=0.15):
    """Partially zero out and refill to escape local optima."""
    n = len(a0)
    a = a0.copy()
    n_zero = max(10, int(n * frac))
    zero_idx = rng.choice(n, size=n_zero, replace=False)
    a[zero_idx] = 0
    a = np.maximum(a, 0)
    ss = a.sum()
    if ss < 0.01:
        return a0.copy()
    # Redistribute some mass to zeroed positions
    for idx in zero_idx:
        a[idx] = rng.random() * a.max() * 0.3
    a = np.maximum(a, 0)
    ss = a.sum()
    if ss > 1e-15:
        a = a / ss
    return a


def check_and_update(a, best_seq, best_score, label=""):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
    return best_seq, best_score

            parent = p / p.sum()
    except Exception:
        pass

    best_seq = None
    best_score = float('inf')

        best_score = fs

    return [float(x) for x in best_seq]
