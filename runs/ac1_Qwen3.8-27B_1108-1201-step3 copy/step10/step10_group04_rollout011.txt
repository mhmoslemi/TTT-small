**Exact line search implementation:**

Given `a` (current) and `b*` (LP solution), find `t* = argmin_t max_k ((1-t)*a + t*b*) * ((1-t)*a + t*b*)_k`.

Let `c(t) = (1-t)*a + t*b*`. Then `(c(t)*c(t))_k = sum_j c(t)_j * c(t)_{k-j}`.

This is quadratic in t:
`(c(t)*c(t))_k = (1-t)^2 (a*a)_k + 2t(1-t) (a*b*)_k + t^2 (b**b*)_k`
`= [(a*a)_k - 2(a*b*)_k + (b**b*)_k] t^2 + 2[(a*b*)_k - (a*a)_k] t + (a*a)_k`

Let `A_k = (b**b*)_k - 2(a*b*)_k + (a*a)_k`, `B_k = 2((a*b*)_k - (a*a)_k)`, `C_k = (a*a)_k`.

Then `f_k(t) = A_k t^2 + B_k t + C_k` and we want `min_t max_k f_k(t)`.

Since each `f_k` is a quadratic (possibly concave if `A_k < 0`), the max is not necessarily convex. But in practice, since `b*` is the best response to `a`, we expect `A_k >= 0` for most k (the "curvature" should be non-negative when moving in the direction of the best response).

Actually, `A_k = (b**b*)_k - 2(a*b*)_k + (a*a)_k = ((b*-a)*(b*-a))_k >= 0` ... wait, that's not right either. Let me expand:

`(b*-a) * (b*-a)` at position k = `sum_j (b*_j - a_j)(b*_{k-j} - a_{k-j})`

This is not necessarily non-negative (it's a convolution of a signed sequence with itself).

Hmm, so the line search is not trivially convex. But we can still use it:
1. Evaluate `f(t) = max_k f_k(t)` at several points in [0, 1]
2. Find the minimum using a 1D search (golden section, or just grid search + refinement)

Since we need to evaluate `f(t)` which requires computing the convolution of `c(t)` with itself, each evaluation costs O(n log n) with FFT. With ~20 evaluations, this is fast.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize_scalar, minimize


def fast_convolve(a):
    """Fast linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_cross(a, b):
    """Fast linear cross-correlation/convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a_fixed):
    """Build the convolution matrix T where (T @ b)_k = (a*b)_k."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def line_search_lp(a, b_star, n_evaluate=20):
    """
    Exact line search: find t in [0,1] minimizing max_k ((1-t)*a + t*b_star) * (...)_k
    Returns optimal t and the resulting sequence.
    """
    n = len(a)
    
    # Precompute the quadratic coefficients
    aa = fast_convolve(a)
    ab = fast_convolve_cross(a, b_star)
    bb = fast_convolve(b_star)
    
    # f_k(t) = A_k * t^2 + B_k * t + C_k
    A_coeffs = bb - 2 * ab + aa  # coefficient of t^2
    B_coeffs = 2 * (ab - aa)     # coefficient of t
    C_coeffs = aa                 # constant term
    
    def objective(t):
        vals = A_coeffs * t * t + B_coeffs * t + C_coeffs
        return np.max(vals)
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = objective(c)
    fd = objective(d)
    
    for _ in range(40):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = objective(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = objective(d)
    
    t_opt = (lo + hi) / 2
    a_new = (1 - t_opt) * a + t_opt * b_star
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return t_opt, a_new


def solve_lp_best_response(a_fixed, solver=None, top_k_weight=0):
    """
    Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0.
    If top_k_weight > 0, add a quadratic penalty on top peaks (approximated).
    Returns (b, M).
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def lp_iteration_with_line_search(a, solver, n_iters=10000, time_limit=100.0, line_search=True):
    """
    LP-based alternating minimization with exact line search.
    """
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    prev_b = None
    stall_count = 0
    prev_best = best_score
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is None:
            stall_count += 1
            if stall_count > 20:
                break
            continue
        
        # Try the LP solution directly
        score_b = 2.0 * n * fast_convolve(b_new).max()
        if score_b < best_score - 1e-12:
            best_score = score_b
            best_a = b_new.copy()
        
        if line_search:
            t_opt, a_new = line_search_lp(a, b_new)
            score_a = 2.0 * n * fast_convolve(a_new).max()
            if score_a < best_score - 1e-12:
                best_score = score_a
                best_a = a_new.copy()
            a = a_new
        else:
            # Fallback: fixed step
            alpha = 0.5

            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s > 1e-15:
                a_new = a_new / s
            a = a_new

                a = a / a.sum()

            elapsed = time.monotonic() - t0
            print(f"    LP-LS iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

def multi_peak_lp(a_fixed, solver=None, n_peaks=5):
    """
    LP that minimizes a weighted sum of top peaks.
    Minimize M s.t. T(a)@b <= M for the top n_peaks positions,
    and T(a)@b <= 2M for the rest.
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    size = 2 * n - 1
    T = build_conv_matrix(a_fixed)
    
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # First, find approximately which peaks are on top
    # Use uniform b to get a rough idea
    b_uniform = np.ones(n) / n
    b_conv = T @ b_uniform
    top_positions = np.argsort(b_conv)[::-1][:n_peaks]
    
    constraints = [
        T[top_positions] @ b_var <= M_var,
        cp.sum(b_var) == 1,
        b_var >= 0,
    ]
    # Add loose constraints for other positions
    other_positions = np.setdiff1d(np.arange(size), top_positions)
    constraints.append(T[other_positions] @ b_var <= 2 * M_var)
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def spectral_flatness_step(a, n_steps=100, alpha=0.5):
    """
    Spectral approach: modify the amplitude spectrum to flatten the autocorrelation.
    """
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    # FFT size
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    for step in range(n_steps):
        A = rfft(a, fft_size)
        # Power spectrum
        P = np.abs(A) ** 2
        # Target: flat power spectrum
        # Modify: new amplitude = P^alpha (compresses dynamic range)
        P_target = P ** alpha
        # New magnitude
        new_mag = np.sqrt(P_target)
        # Maintain phase
        phase = np.angle(A)
        A_new = new_mag * np.exp(1j * phase)
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Reduce alpha
            # Actually, let's try a different approach: blend
            a_blend = 0.5 * a + 0.5 * a_new
            a_blend = np.maximum(a_blend, 0)
            a_blend = a_blend / a_blend.sum()
            score_blend = 2.0 * n * fast_convolve(a_blend).max()
            if score_blend < best_score - 1e-12:
                best_score = score_blend
                best_a = a_blend.copy()
                a = a_blend
            else:
                break
    
    return best_a, best_score


def coordinate_descent_peaks(a, time_limit=30.0, n_peaks=10):
    """
    Coordinate descent focused on reducing top peaks.
    """
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        sorted_idx = np.argsort(b)[::-1]
        top_positions = sorted_idx[:n_peaks]
        
        improved = False
        for k in top_positions:
            # Find the pair (j, k-j) with maximum product a[j]*a[k-j]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            products = a[idx] * a[k - idx]
            src = idx[np.argmax(products)]
            partner = k - src
            
            # Try moving mass from src to a random position
            for _ in range(10):
                dst = rng.integers(n)
                if dst == src or dst == partner:
                    continue
                for frac in [0.05, 0.1, 0.15, 0.2]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        break
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-12:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            no_improve += 1
            if no_improve > 500:
                # Random perturbation
                k = rng.integers(max(1, n // 20), n // 5)
                idx = rng.choice(n, k, replace=False)
                a_trial = a.copy()
                a_trial[idx] += rng.normal(0, 0.01, k)
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score + 0.01:  # Allow slight worsening
                    a = a_trial
                no_improve = 0
        else:
            no_improve = 0
    
    return best_a, best_score


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def lse_optimize(a0, time_limit, T_start=50.0, T_end=0.1, n_stages=15):
