import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft, fft, ifft
from scipy.sparse import csr_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate 2n*max(conv(a,a))/(sum(a))^2 (scale invariant)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def project_simplex(v):
    """Project vector v onto the simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def matvec_Qq(q, a, fft_cache=None):
    """Compute Q_q @ a where (Q_q)_{jl} = q_{j+l}. Uses FFT cross-correlation."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (2 * size - 1).bit_length()
    q_ext = np.zeros(N)
    q_ext[:size] = q
    a_ext = np.zeros(N)
    a_ext[:n] = a
    Q_f = fft(q_ext)
    A_f = fft(a_ext)
    c = ifft(Q_f * np.conj(A_f))[:n].real
    return c


def qp_game_iteration(a0, n_steps=50000, time_limit=300.0, top_k=30,
                      print_every=5000, seed=42):
    """
    QP-Game iteration: alternates between solving min a^T Q_q a (QP)
    and updating q to be uniform over active peaks.
    Uses Nesterov-accelerated projected gradient with FFT for Qq @ a.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    # Initialize q as uniform
    q = np.ones(2 * n - 1) / (2 * n - 1)

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    # Nesterov parameters
    y = a.copy()
    t_nest = 1.0
    # Estimate Lipschitz constant
    Qa = matvec_Qq(q, y)
    L_est = 2.0 * np.linalg.norm(Qa, ord=np.inf)
    L_est = max(L_est, 1.0)

    rng = np.random.default_rng(seed)
    stall = 0
    prev_best = best_score

    for it in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute gradient at y: grad = 2 * Q_q @ y
        Qy = matvec_Qq(q, y)
        grad = 2.0 * Qy

        # Nesterov step
        step_size = 1.0 / max(L_est, 1e-10)
        a_new = y - step_size * grad
        a_new = project_simplex(a_new)

        # Update Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        beta = (t_nest - 1.0) / t_new
        y = a_new + beta * (a_new - a)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy
        t_nest = t_new
        a = a_new

        # Evaluate
        if (it + 1) % 10 == 0:
            b = fast_convolve(a)
            score = 2.0 * n * b.max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

            # Update q: uniform over top peaks
            if (it + 1) % 20 == 0:
                b = fast_convolve(a)
                max_b = b.max()
                threshold = max_b * 0.98
                active = np.where(b >= threshold)[0]
                if len(active) < top_k:
                    # Expand to top_k
                    top_idx = np.argsort(b)[::-1][:top_k]
                    active = top_idx
                else:
                    active = active[:top_k]
                q_new = np.zeros(2 * n - 1)
                q_new[active] = 1.0 / len(active)
                q = q_new
                # Update Lipschitz estimate
                Qy_check = matvec_Qq(q, y)
                L_new = 2.0 * np.linalg.norm(Qy_check, ord=np.inf)
                L_est = max(L_est * 0.95, L_new, 1.0)

        # Stall detection
        if (it + 1) % 500 == 0:
            if best_score < prev_best - 1e-10:
                stall = 0
            else:
                stall += 1
            prev_best = best_score

            if stall >= 8:
                # Perturb to escape
                noise = 0.05 * rng.standard_normal(n)
                a_pert = a + noise
                a_pert = project_simplex(a_pert)
                a = a_pert
                y = a.copy()
                t_nest = 1.0
                # Reset q
                b = fast_convolve(a)
                top_idx = np.argsort(b)[::-1][:top_k]
                q = np.zeros(2 * n - 1)
                q[top_idx] = 1.0 / len(top_idx)
                stall = 0

            if print_every and (it + 1) % print_every == 0:
                print(f"      QPGame iter={it+1} ({time.monotonic()-t0:.0f}s): "
                      f"score={best_score:.10f} L={L_est:.2f} top_k={top_k} stall={stall}")

    return best_a, best_score


def build_sparse_T(a):
    """Build sparse (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    # Column j has non-zeros at rows j, j+1, ..., j+n-1 with values a[0],...,a[n-1]
    rows = np.concatenate([np.arange(j, j + n) for j in range(n)])
    cols = np.repeat(np.arange(n), n)
    vals = np.tile(a, n)
    T = csr_matrix((vals, (rows, cols)), shape=(size, n))
    return T


def solve_lp_best_response(a_fixed, solver=None, T_matrix=None, max_iter=5000):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()

    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    if T_matrix is None:
        T_matrix = build_sparse_T(a_fixed)

    constraints = [T_matrix @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    try:
        prob.solve(solver=solver, max_iter=max_iter, verbose=False)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.asarray(b_var.value, dtype=np.float64)
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s < 1e-15:
                return None, None, T_matrix
            b_val = b_val / s
            return b_val, float(M_var.value), T_matrix
    except Exception:
        pass

    # Fallback solvers
    for alt in [cp.SCS, cp.ECOS, cp.CLARABEL, cp.PDLP]:
        if alt == solver:
            continue
        try:
            prob.solve(solver=alt, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s < 1e-15:
                    return None, None, T_matrix
                b_val = b_val / s
                return b_val, float(M_var.value), T_matrix
        except Exception:
            continue
    return None, None, T_matrix


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def lp_alternation(a0, time_limit=600.0, solver=None, max_iters=100000,
                   stall_patience=15, print_interval=2000):
    """LP alternation with warm-started sparse matrices."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall = 0
    rng = np.random.default_rng(12345)

    # Pre-build initial T
    T = build_sparse_T(a)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Solve LP: find best response b
        b_new, M_b, T = solve_lp_best_response(a, solver, T, max_iter=3000)
        if b_new is None:
            stall += 1
            if stall > 30:
                break
            continue

        # Update a (blend toward b)
        alpha = 2.0 / (it + 2)
        a = (1.0 - alpha) * a + alpha * b_new
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = best_a.copy()
        else:
            a = a / sa

        # Track best
        b_conv = fast_convolve(a)
        score = 2.0 * n * b_conv.max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # Periodic evaluation and stall detection
        if (it + 1) % 500 == 0:
            if best_score < prev_best - 1e-10:
                stall = 0
            else:
                stall += 1
            prev_best = best_score

            if stall >= stall_patience:
                # Perturbation
                noise_scale = 0.03 / max(1, np.sqrt(stall))
                noise = rng.standard_normal(n) * noise_scale
                a = a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                T = build_sparse_T(a)
                stall = 0

            if print_interval and (it + 1) % print_interval == 0:
                print(f"      LP iter={it+1} ({time.monotonic()-t0:.0f}s): "
                      f"score={best_score:.10f} M={M_b:.10f} stall={stall}")

    return best_a, best_score


def active_set_qp_refine(a0, time_limit=200.0, top_k_schedule=None,
                         print_every=5000):
    """
    Refine using active-set QP with progressively shrinking active sets.
    Starts with a large active set and narrows to the true peaks.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    if top_k_schedule is None:

    for top_k in top_k_schedule:
        if time.monotonic() - t0 > time_limit:

        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        # Get current active set
        b = fast_convolve(a)
        max_b = b.max()
        active = np.argsort(b)[::-1][:top_k]
        q = np.zeros(2 * n - 1)
        q[active] = 1.0 / len(active)

        # Nesterov-accelerated projected gradient on QP
        y = a.copy()
        t_nest = 1.0
        Qy = matvec_Qq(q, y)
        L_est = max(2.0 * np.linalg.norm(Qy, ord=np.inf), 1.0)

        phase_iters = min(20000, int(remaining * 200))

        for it in range(phase_iters):
            if time.monotonic() - t0 > time_limit:
                break

            Qy = matvec_Qq(q, y)
            grad = 2.0 * Qy
            step_size = 1.0 / L_est
            a_new = y - step_size * grad
            a_new = project_simplex(a_new)

            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
            beta = (t_nest - 1.0) / t_new
            y = a_new + beta * (a_new - a)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy > 1e-15:
                y = y / sy
            t_nest = t_new
            a = a_new

            if (it + 1) % 50 == 0:
                b = fast_convolve(a)
                score = 2.0 * n * b.max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()

        # Update active set after each phase
        b = fast_convolve(best_a)
        # Check if the max is actually at our active set
        max_at_active = b[active].max()
        global_max = b.max()
        if global_max > max_at_active * 1.001:
            # Expand active set to include the true max
            new_active = np.argsort(b)[::-1][:max(top_k, 10)]
            active = new_active
            q = np.zeros(2 * n - 1)
            q[active] = 1.0 / len(active)

        if print_every:
            print(f"      ActQP top_k={top_k} ({time.monotonic()-t0:.0f}s): "
                  f"score={best_score:.10f}")

    return best_a, best_score


def coord_descent_peaks(a0, time_limit=100.0, n_steps=100000, top_m=6):
    """Coordinate descent targeting top convolution peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        top_peaks = np.argsort(b)[::-1][:min(top_m, len(b))]

        improved = False
        for peak_k in top_peaks[:3]:
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            order = np.argsort(contribs)[::-1][:3]
            for oi in order:
                pos = idx[oi]
                if a[pos] < 1e-14:
                    continue
                for factor in [0.995, 0.99, 0.98]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] * (1.0 - factor)
                    if diff < 1e-16:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others_sum = a_trial[mask].sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] += diff * a_trial[mask] / others_sum
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Try simultaneous reduction of top contributors
            peak_k = top_peaks[0]
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top3 = idx[np.argsort(contribs)[::-1][:2]]
            a_trial = a.copy()
            for pos in top3:
                if a[pos] > 1e-14:
                    a_trial[pos] *= 0.999
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                break
            a_trial = a_trial / st
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-13:
                a = a_trial
                best_score = score
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
                if stall > 500:
                    break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Create diverse structured initializations."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.05
    elif choice == 5:
        a = np.exp(-3 * (x - 0.25) ** 2) + 0.4 * np.exp(-3 * (x - 0.75) ** 2)
    elif choice == 6:
        a = np.exp(-4 * (x - 0.4) ** 2) + 0.3
    elif choice == 7:
        a = (1 - x) ** 0.7 + 0.1
    elif choice == 8:
        # Asymmetric with bump
        a = np.exp(-5 * (x - 0.3) ** 2) + 0.5 * np.exp(-5 * (x - 0.7) ** 2) + 0.05
    else:
        # Gentle cosine
        a = 0.6 + 0.4 * np.cos(np.pi * x)
        a = np.maximum(a, 0)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

    print(f"\n=== Phase 1: QP-Game Iteration ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.25, 400.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s, n={len(best_seq)}")
        a_qpg, _ = qp_game_iteration(best_seq, n_steps=50000,
                                     time_limit=t1, top_k=40,
                                     print_every=10000, seed=seed)
        try_update(a_qpg, "qp_game_1")

    # ============ PHASE 2: Active Set QP Refinement ============
    print(f"\n=== Phase 2: Active Set QP, score={best_score:.10f} ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.20, 300.0)
    if t2 > 20:
        print(f"  Budget: {t2:.0f}s")
        a_asq, _ = active_set_qp_refine(best_seq, time_limit=t2,
                                        print_every=5000)
        try_update(a_asq, "active_qp_2")

    # ============ PHASE 3: LP Alternation ============
    print(f"\n=== Phase 3: LP Alternation, score={best_score:.10f} ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.30, 500.0)
    if t3 > 40:
        print(f"  Budget: {t3:.0f}s, n={len(best_seq)}")
        a_lp, _ = lp_alternation(best_seq, time_limit=t3, solver=solver,
                                  max_iters=50000, stall_patience=12,
                                  print_interval=3000)
        try_update(a_lp, "lp_3")

    # ============ PHASE 4: Second QP-Game Pass (on LP result) ============
    print(f"\n=== Phase 4: QP-Game Pass 2, score={best_score:.10f} ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.15, 200.0)
    if t4 > 15:
        # Use smaller top_k for more focused optimization
        a_qpg2, _ = qp_game_iteration(best_seq, n_steps=30000,
                                      time_limit=t4, top_k=15,
                                      print_every=5000, seed=seed + 1)
        try_update(a_qpg2, "qp_game_4")

    # ============ PHASE 5: Coordinate Descent Polish ============
    print(f"\n=== Phase 5: Coord Descent, score={best_score:.10f} ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.10, 150.0)
    if t5 > 10:
        a_cd, _ = coord_descent_peaks(best_seq, time_limit=t5,
                                       n_steps=200000, top_m=5)
        try_update(a_cd, "coord_5")

    # ============ PHASE 6: Multi-length exploration ============
    print(f"\n=== Phase 6: Multi-length, score={best_score:.10f} ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        n_cur = len(best_seq)
        alt_lengths = []
        for factor in [0.75, 1.25, 1.5]:
            alt_n = int(n_cur * factor)
            if 200 <= alt_n <= 5000 and alt_n not in alt_lengths:
                alt_lengths.append(alt_n)

        for alt_n in alt_lengths[:2]:
            if rem < 30:
                break
            t_alt = min(rem * 0.4, 120.0)
            if t_alt < 15:
                break
            print(f"  Trying n={alt_n}, t={t_alt:.0f}s")
            init_alt = interpolate_sequence(best_seq, alt_n)
            # Quick QP-game
            a_q, _ = qp_game_iteration(init_alt, n_steps=10000,
                                       time_limit=t_alt * 0.5, top_k=30,
                                       print_every=5000, seed=seed + alt_n)
            try_update(a_q, f"qp_n{alt_n}")
            rem = deadline - time.monotonic()
            if rem > 15:
                # Then LP
                a_l, _ = lp_alternation(a_q, time_limit=t_alt * 0.45,
                                         solver=solver, max_iters=20000,
                                         stall_patience=10, print_interval=2000)
                try_update(a_l, f"lp_n{alt_n}")
            rem = deadline - time.monotonic()

    # ============ PHASE 7: Multi-start diversity ============
    print(f"\n=== Phase 7: Multi-start, score={best_score:.10f} ===")
    rem = deadline - time.monotonic()
    if rem > 50:
        n_cur = len(best_seq)
        t_each = min(rem / 4.0, 60.0)
        for pi in range(3):
            if time.monotonic() > deadline - 20:
                break
            if t_each < 10:
                break
            # Generate perturbation
            if pi == 0:
                x = np.linspace(0, 2 * np.pi, n_cur)
                pert = 0.04 * np.cos(2 * x) + 0.02 * np.cos(4 * x)
            elif pi == 1:
                pert = 0.03 * rng.standard_normal(n_cur)
            else:
                x = np.linspace(0, 2 * np.pi, n_cur)
                pert = 0.03 * np.sin(3 * x) + 0.01 * np.cos(7 * x)

            a_pert = best_seq + pert
            a_pert = np.maximum(a_pert, 0)
            ps = a_pert.sum()
            if ps < 1e-15:
                continue
            a_pert = a_pert / ps

            # Quick QP-game on perturbation
            a_res, _ = qp_game_iteration(a_pert, n_steps=15000,
                                         time_limit=t_each, top_k=25,
                                         print_every=5000, seed=seed + 100 + pi)
            try_update(a_res, f"perturb_qp_{pi}")
            rem = deadline - time.monotonic()

    # ============ PHASE 8: Final Active Set QP + Coord Descent ============
    print(f"\n=== Phase 8: Final Polish, score={best_score:.10f} ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin1 = min(rem * 0.5, 60.0)
        if t_fin1 > 10:
            a_fin1, _ = active_set_qp_refine(best_seq, time_limit=t_fin1,
                                              top_k_schedule=[20, 10, 5, 3, 2, 1],
                                              print_every=3000)
            try_update(a_fin1, "final_asq")
        rem = deadline - time.monotonic()
        if rem > 10:
            t_fin2 = min(rem * 0.6, 30.0)
            if t_fin2 > 5:
                a_fin2, _ = coord_descent_peaks(best_seq, time_limit=t_fin2,
                                                 n_steps=100000, top_m=4)
                try_update(a_fin2, "final_cd")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
