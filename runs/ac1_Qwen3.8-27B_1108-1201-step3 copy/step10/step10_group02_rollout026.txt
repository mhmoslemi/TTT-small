    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

def simplex_project(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    v = v.copy()

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0][-1]

    s = result.sum()
    if s > 1e-15:

    else:

    return result

                eta = 0.7 * eta + 0.3 * eta_new
                eta = max(1e-6, min(eta, 0.5))
            else:
                eta *= 0.95

        prev_a = a.copy()
        prev_g = g.copy()

        # Subgradient step (in log-space for positivity)
        log_a = np.log(np.maximum(a, 1e-300))
        g_centered = g - g.mean()
        log_a_new = log_a - eta * g_centered
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-300:
            a_new = a.copy()
        else:
            a_new = a_new / a_new_sum

        # Nesterov update
        if nesterov:
            alpha = 2.0 / (iter_count + 2.0)
            y = (1.0 - alpha) * y + alpha * a_new
            y = np.maximum(y, 0)
            ys = y.sum()
            if ys < 1e-15:
                y = a_new.copy()
            else:
                y = y / ys

        a = a_new
        iter_count += 1

        # Evaluate periodically (not every iteration for speed)
        if iter_count % 20 == 0 or iter_count < 50:
            score = evaluate_local(a)
            if score < best_score - 1e-12:
                best_score = score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

        # Symmetrize periodically
        if sym_period > 0 and iter_count % sym_period == 0:
            a = symmetrize(a)
            a = a / a.sum()
            y = a.copy()

        # Restart if stuck
        if no_improve >= 300:
            rng = np.random.default_rng(iter_count * 31 + 7)
            noise_scale = 0.003 + 0.003 * min(1.0, no_improve / 600.0)
            noise = rng.standard_normal(n) * noise_scale
            a = a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            y = a.copy()
            no_improve = 0
            eta = max(eta, eta0 * 0.5)
            if no_improve > 1500:
                break

        if iter_count % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SubGrad iter={iter_count} ({elapsed:.0f}s): best={best_score:.8f} eta={eta:.6f}")

    return best_a, best_score


def gauss_newton_refine(a0, time_limit, top_k=5, max_inner_iters=50):
    """
    Gauss-Newton refinement: solve system of equations
    (a*a)_{k_j} = t for top-k peaks, plus sum(a)=1.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    prev_score = best_score
    stall = 0

    outer_iter = 0
    while time.monotonic() - t0 < time_limit:
        outer_iter += 1
        b = fast_convolve(a)
        max_b = b.max()

        # Find top-k peak positions
        peaks = np.argpartition(b, -top_k)[-top_k:]
        peaks = peaks[np.argsort(b[peaks])[::-1]]

        # Current residual
        F = np.zeros(top_k + 1)
        F[:top_k] = b[peaks] - max_b  # should be ~0 at optimum
        F[top_k] = a.sum() - 1.0

        # Build Jacobian J (top_k+1) x n
        # Row j: d(a*a)_{k_j}/da_i = 2*a[k_j - i] for valid i
        # Last row: 1 (from sum constraint)
        J = np.zeros((top_k + 1, n))
        for j, k in enumerate(peaks):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            J[j, idx] = 2.0 * a[k - idx]
        J[top_k, :] = 1.0

        # Gauss-Newton: solve J^T J delta = -J^T F
        JtJ = J.T @ J  # n x n
        JtF = J.T @ F  # n

        # Add small regularization for stability
        JtJ[np.diag_indices(n)] += 1e-10

        try:
            delta = np.linalg.solve(JtJ, -JtF)
        except np.linalg.LinAlgError:
            delta = -0.01 * JtF

        # Limit step size
        delta_norm = np.linalg.norm(delta)
        if delta_norm > 0.1:
            delta = delta * 0.1 / delta_norm

        a_new = a + delta
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()

        score = evaluate_local(a_new)
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            # Try half step
            a_half = a + 0.5 * delta
            a_half = np.maximum(a_half, 0)
            a_half = a_half / a_half.sum()
            score_half = evaluate_local(a_half)
            if score_half < best_score - 1e-13:
                best_score = score_half
                best_a = a_half.copy()
                a = a_half
                stall = 0
            else:
                stall += 1
                if stall > max_inner_iters:
                    # Try restart
                    rng = np.random.default_rng(outer_iter * 17)
                    noise = rng.standard_normal(n) * 0.005
                    a = a + noise
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    stall = 0

        if outer_iter % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Gauss-Newton iter={outer_iter} ({elapsed:.0f}s): score={best_score:.8f} stall={stall}")

        if stall > max_inner_iters * 3:
            break

    return best_a, best_score


def coordinate_descent_refine(a0, time_limit, top_k_peaks=3, top_m_coords=8):
    """
    Aggressive coordinate descent: for each peak, find the coordinates
    that contribute most, and try multiplicative updates.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    stall = 0

    factors = [0.999, 0.998, 0.997, 0.995, 0.99, 0.985, 0.98, 1.01, 1.02, 1.03, 1.05]

    step = 0
    while time.monotonic() - t0 < time_limit:
        step += 1
        b = fast_convolve(a)
        max_b = b.max()

        # Find top peaks
        peaks = np.argpartition(b, -top_k_peaks)[-top_k_peaks:]
        peaks = peaks[np.argsort(b[peaks])[::-1]]

        improved = False
        for peak_k in peaks:
            if improved:
                break
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_positions = idx[np.argsort(contribs)[::-1][:top_m_coords]]

            for pos in top_positions:
                if a[pos] < 1e-12:
                    continue
                for factor in factors:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    if a_trial[pos] < 0:
                        continue
                    diff = a[pos] - a_trial[pos]
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] = others + diff * others / others_sum
                    a_trial = a_trial / a_trial.sum()
                    score = evaluate_local(a_trial)
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            if stall > 500:
                # Try simultaneous decrease of top contributors
                a_trial = a.copy()
                peak_k = int(np.argmax(b))
                js = max(0, peak_k - n + 1)
                je = min(n - 1, peak_k)
                idx = np.arange(js, je + 1)
                contribs = a[idx] * a[peak_k - idx]
                top_positions = idx[np.argsort(contribs)[::-1][:6]]
                for pos in top_positions:
                    if a_trial[pos] > 1e-12:
                        a_trial[pos] *= 0.998
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = evaluate_local(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
                elif stall > 2000:
                    break
            elif stall % 100 == 0:
                # Small random perturbation
                rng = np.random.default_rng(step * 13)
                a = a + rng.standard_normal(n) * 0.002
                a = np.maximum(a, 0)
                a = a / a.sum()
        else:
            stall = 0

        if (step + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      CoordDesc step={step} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def logsumexp_smooth_optimize(a0, time_limit, T_final=0.1):
    """
    Optimize using LogSumExp smoothing with decreasing temperature.
    Uses scipy L-BFGS-B in the softmax parameterization.
    """
    from scipy.optimize import minimize

    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    z = np.log(np.maximum(a, 1e-300)).copy()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = evaluate_local(a)

    T_stages = [10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.15, T_final]
    n_stages = len(T_stages)
    time_per_stage = time_limit / n_stages

    for T in T_stages:
        if time.monotonic() - t0 > time_limit - 5:
            break

        stage_start = time.monotonic()
        stage_time = min(time_per_stage, time_limit - (stage_start - t0))

        def lse_obj(z_inner):
            a_inner = np.exp(z_inner - z_inner.max())
            a_inner = a_inner / a_inner.sum()
            b = fast_convolve(a_inner)
            max_b = b.max()
            exp_b = np.exp((b - max_b) / T)
            lse = max_b + T * np.log(exp_b.sum() / len(b))
            return (2.0 * n * lse)

        def lse_grad(z_inner):
            a_inner = np.exp(z_inner - z_inner.max())
            a_inner = a_inner / a_inner.sum()
            b = fast_convolve(a_inner)
            max_b = b.max()
            exp_b = np.exp((b - max_b) / T)
            w = exp_b / exp_b.sum()

            # Gradient of LSE w.r.t. a
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += w[k] * 2.0 * a_inner[k - idx]

            # Convert to z-gradient via softmax
            S = np.dot(grad_a, a_inner)
            grad_z = a_inner * (grad_a - S)
            return 2.0 * n * grad_z

        maxiter = max(30, int(stage_time * 50))
        try:
            res = minimize(fun=lse_obj, x0=z, jac=lse_grad, method='L-BFGS-B',
                           options={'maxiter': maxiter, 'ftol': 1e-18,
                                    'gtol': 1e-12, 'maxfun': maxiter * 200,
                                    'disp': False})
            z = res.x
        except Exception:
            pass

        a_cur = np.exp(z - z.max())
        a_cur = a_cur / a_cur.sum()
        sc = evaluate_local(a_cur)
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()
            print(f"      LSE T={T}: score={best_score:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses accelerated subgradient as main workhorse, Gauss-Newton for
    structural refinement, LogSumExp L-BFGS for smooth optimization,
    and coordinate descent for final polish.
    Exploits symmetry and multi-n search for diversity.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        x = np.linspace(0, 1, n_init)
        best_seq = np.cos(np.pi * x / 2) ** 2
        best_seq = np.maximum(best_seq, 0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Quick multi-n exploration ===
    print(f"\n=== Phase 1: Multi-n exploration (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.03, 40.0)

    if t_alloc > 10:
        n_candidates = [800, 1000, 1200, 1500, 2000]
        n_candidates = [n for n in n_candidates if n > 100]
        time_per_n = t_alloc / len(n_candidates)

        for n_try in n_candidates:
            if time.monotonic() > deadline - 10:
                break
            a_try = interpolate_sequence(best_seq, n_try)
            sc = evaluate_local(a_try)
            print(f"    n={n_try}: score={sc:.8f}")
            if sc < best_score:
                best_score = sc
                best_seq = a_try.copy()

    n_best = len(best_seq)
    print(f"  Best n after exploration: {n_best}, score={best_score:.8f}")

    # === Phase 2: Main accelerated subgradient (top-3 peaks) ===
    print(f"\n=== Phase 2: Accel SubGradient top-3 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 550.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s, n={n_best}")
        a_md, _ = accelerated_subgradient(best_seq, t_alloc, top_k=3,
                                          eta0=0.04, nesterov=True, sym_period=300)
        try_update(a_md, "subgrad_p2")

    # === Phase 3: Accelerated subgradient with top-5 (more robust) ===
    print(f"\n=== Phase 3: Accel SubGradient top-5 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_md2, _ = accelerated_subgradient(best_seq, t_alloc, top_k=5,
                                           eta0=0.02, nesterov=True, sym_period=500)
        try_update(a_md2, "subgrad_p3")

    # === Phase 4: Gauss-Newton refinement ===
    print(f"\n=== Phase 4: Gauss-Newton (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_gn, _ = gauss_newton_refine(best_seq, t_alloc, top_k=5, max_inner_iters=100)
        try_update(a_gn, "gauss_newton")

    # === Phase 5: LogSumExp L-BFGS ===
    print(f"\n=== Phase 5: LogSumExp L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lse, _ = logsumexp_smooth_optimize(best_seq, t_alloc, T_final=0.1)
        try_update(a_lse, "logsumexp")

    # === Phase 6: Coordinate descent polish ===
    print(f"\n=== Phase 6: Coord Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 15:
        a_cd, _ = coordinate_descent_refine(best_seq, t_alloc,
                                            top_k_peaks=3, top_m_coords=10)
        try_update(a_cd, "coord_desc")

    # === Phase 7: Final subgradient with very small eta ===
    print(f"\n=== Phase 7: Final refinement (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.07, 80.0)
    if t_alloc > 10:
        a_fin, _ = accelerated_subgradient(best_seq, t_alloc, top_k=2,
                                           eta0=0.005, nesterov=True, sym_period=0)
        try_update(a_fin, "final_subgrad")

    # === Phase 8: Final coordinate descent ===
    print(f"\n=== Phase 8: Final CD polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.05, 40.0)
    if t_alloc > 5:
        a_polish, _ = coordinate_descent_refine(best_seq, t_alloc,
                                                top_k_peaks=2, top_m_coords=5)
        try_update(a_polish, "final_polish")

    # Final: try symmetrization
    if best_seq is not None:
        sym_seq = symmetrize(best_seq)
        sym_seq = sym_seq / sym_seq.sum()
        sym_score = evaluate_local(sym_seq)
        if sym_score < best_score:
            best_score = sym_score
            best_seq = sym_seq
            print(f"  Symmetrization improved: {best_score:.8f}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
