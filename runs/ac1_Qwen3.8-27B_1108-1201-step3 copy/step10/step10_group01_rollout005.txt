        prob.solve(solver=solver, verbose=False)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0)
            s = b_val.sum()
            if s < 1e-15:
                return None, None, None
            b_val /= s
            M_val = float(M_var.value)
            y_dual = None
            if get_dual:
                try:
                    y_dual = np.asarray(conv_constraints.dual_value, dtype=np.float64)
                    y_dual = np.maximum(y_dual, 0)
                except:
                    pass
            return b_val, M_val, y_dual
    except:
        pass
    # Fallback
    for alt in [cp.SCS, cp.ECOS, cp.CLARABEL]:
        if alt == solver:
            continue
        try:
            prob.solve(solver=alt, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0)
                s = b_val.sum()
                if s < 1e-15:
                    return None, None, None
                b_val /= s
                M_val = float(M_var.value)
                return b_val, M_val, None
        except:
            continue
    return None, None, None

def nesterov_lp(a0, time_limit, solver, max_iters=50000, get_dual_period=10):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall = 0
    rng = np.random.default_rng(123)
    y_dual_last = None
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        want_dual = (it % get_dual_period == 0)
        b_new, M_val, y_dual = solve_lp(y, solver, get_dual=want_dual)
        if b_new is None:
            stall += 1
            if stall > 40:
                break
            continue
        if y_dual is not None:
            y_dual_last = y_dual
        
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy
        
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        if (it + 1) % 1000 == 0:
            if best_score < prev_best - 1e-10:
                stall = 0
            else:
                stall += 1
            prev_best = best_score
            
            if stall >= 20:
                scale = 0.005 + 0.005 * min(stall, 10)
                y = y + rng.standard_normal(n) * scale
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall = 0
    
    return best_a, y_dual_last
