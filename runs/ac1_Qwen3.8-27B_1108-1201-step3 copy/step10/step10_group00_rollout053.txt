    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    c = np.zeros(size)
    c[:n] = a
    r = np.zeros(n)
    r[0] = a[0]
    return toeplitz(c, r)

def get_lp_solver():
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

def solve_lp_best_response(a_fixed, primary_solver):
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solvers_to_try = list(dict.fromkeys([primary_solver, cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]))
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_sum = b_val.sum()
                if s_sum < 1e-15:
                    return None, None
                b_val = b_val / s_sum
                return b_val, float(max(0.0, M_var.value))
        except Exception:
            continue
    return None, None

def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def softmax_from_z(z):
    z_shift = z - z.max()
    exp_z = np.exp(z_shift)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))

def power_mean_obj_grad(z, n, p):
    a = softmax_from_z(z)
    c = fast_convolve(a)
    c = np.maximum(c, 0)
    cpowers = c ** p
    sum_cp = cpowers.sum()
    if sum_cp < 1e-300:
        return 1e10, np.zeros(n)
    obj = 2.0 * n * sum_cp ** (1.0 / p)
    if p <= 1.0:
        weights = np.ones_like(c)
    else:
        weights = c ** (p - 1)
    d = np.correlate(weights, a, mode='valid')
    factor = sum_cp ** (1.0 / p - 1.0)
    grad_a = 2.0 * factor * d
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z

def lse_obj_grad(z, n, T):
    a = softmax_from_z(z)
    c = fast_convolve(a)
    max_c = c.max()
    exp_terms = np.exp((c - max_c) / T)
    lse = max_c + T * np.log(exp_terms.sum())
    obj = 2.0 * n * lse
    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(c)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z

def nesterov_lp(a0, time_limit, solver, max_iters):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(12345)
    lp_times = []
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        lp_t0 = time.monotonic()
        b_new, M_b = solve_lp_best_response(y, solver)
        lp_times.append(time.monotonic() - lp_t0)
        
        if b_new is None:
            continue
        
        avg_lp = np.mean(lp_times[-30:]) if len(lp_times) >= 10 else 1.0
        
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy
        
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            
            if stall_count >= 8:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 5)
                noise = 0.04 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0
            
            if (it + 1) % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} avg_lp={avg_lp*1000:.0f}ms")
        
        if len(lp_times) > 50 and avg_lp > 2.0:
            print(f"    LP too slow ({avg_lp:.2f}s), stopping at iter {it+1}")
            break
    
    return best_a, best_score

def lbgfs_powermean(a0, time_limit, p_stages, maxiter_per_stage):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    
    for p_val in p_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        def func(z_cur):
            if time.monotonic() > deadline:
                return 1e15
            o, _ = power_mean_obj_grad(z_cur, n, p_val)
            return o
        
        def jac(z_cur):
            if time.monotonic() > deadline:
                return np.zeros_like(z_cur)
            _, g = power_mean_obj_grad(z_cur, n, p_val)
            return g
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': min(maxiter_per_stage, max(10, int(remaining * 50))),
                         'ftol': 1e-20, 'gtol': 1e-14, 'maxfun': 5000, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    return softmax_from_z(z)

def lbgfs_lse(a0, time_limit, T_stages, maxiter_per_stage):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    deadline = t0 + time_limit
    
    for T_val in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        def func(z_cur):
            if time.monotonic() > deadline:
                return 1e15
            o, _ = lse_obj_grad(z_cur, n, T_val)
            return o
        
        def jac(z_cur):
            if time.monotonic() > deadline:
                return np.zeros_like(z_cur)
            _, g = lse_obj_grad(z_cur, n, T_val)
            return g
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': min(maxiter_per_stage, max(10, int(remaining * 50))),
                         'ftol': 1e-20, 'gtol': 1e-14, 'maxfun': 5000, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    return softmax_from_z(z)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def structured_init(n, rng):
    choice = rng.integers(0, 5)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = (1 + np.cos(np.pi * x)) / 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-3 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x)
        a = np.maximum(a, 0)
    else:
        a = np.exp(-4 * (x - 0.25) ** 2) + 0.6 * np.exp(-4 * (x - 0.75) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def perturb_sequence(a, rng, strength=0.03):
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 7)
    phase = rng.uniform(0, 2 * np.pi)
    a_pert = a + strength * np.cos(freq * x + phase)
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s

def multiplicative_subgrad(a0, time_limit, n_steps):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta_base = 0.05
    no_improve = 0
    scale = 1.0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * (1.0 - 5e-4)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([0])
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)
        eta = eta_base * scale
        log_scale = -eta * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= 30:
            scale *= 0.7
            no_improve = 0
            if scale < 0.01:
                break
    return best_a, best_score
