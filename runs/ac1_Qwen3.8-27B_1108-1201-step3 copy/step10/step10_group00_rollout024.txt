```python
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

    s = a.sum()
    if s < 0.01:

    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

                  bounds=bounds, method='highs')

        return None, None
    
    b_val = np.asarray(res.x[:n], dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
```

26. **One more optimization idea: "Spectral" approach**

In the frequency domain, `conv(a,a)` has FFT equal to `|A(f)|^2` where A is the FFT of a. So `max(conv(a,a))` is the maximum of `IFFT(|A(f)|^2)`.

This is the "peak" of the autocorrelation. We want to minimize this peak.

There's a connection to "constant envelope" or "low PAPR" sequences. The problem of minimizing the peak of the autocorrelation is related to the "minimum peak-to-average power ratio" (PAPR) problem.

For complex sequences, there are good algorithms (like the Golay complementary sequence construction). But for real non-negative sequences, the problem is different.

I don't think this spectral approach will give a direct algorithm, but it might give insight into good initializations.

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast LP alternating with scipy HiGHS (no cvxpy overhead) for many iterations
2. FFT-based LogSumExp gradient descent in log-domain with adaptive temperature annealing
3. Multi-length exploration (n=1000, 1400, 1600, 1800)
4. Structured perturbation restarts and final fine-tuning
Key improvement over previous: uses scipy linprog directly (much faster per LP solve),
allowing 5-10x more LP iterations, plus more aggressive temperature annealing.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def solve_lp_response(a_fixed):
    n = len(a_fixed)
    T_mat = build_conv_matrix(a_fixed)
    size = 2 * n - 1
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_ub = np.zeros((size, n + 1))
    A_ub[:, :n] = T_mat
    A_ub[:, n] = -1.0
    b_ub = np.zeros(size)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if res.status != 0:
            return None, None
        b_val = np.asarray(res.x[:n], dtype=np.float64)
        b_val = np.maximum(b_val, 0)
        s = b_val.sum()
        if s < 1e-15:
            return None, None
        b_val = b_val / s
        M_val = max(0.0, float(res.x[n]))
        return b_val, M_val
    except Exception:
        return None, None


def lse_obj_grad_z(z, T_temp):
    n = len(z)
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp((b - max_b) / T_temp)
    w = exp_b / exp_b.sum()
    lse = max_b + T_temp * np.log(exp_b.sum())
    obj = 2.0 * n * lse
    N = 1 << (3 * n - 2).bit_length()
    A_fft = rfft(a, N)
    W_fft = rfft(w, N)
    c = irfft(A_fft * np.conj(W_fft), N)[:n]
    grad_a = (4.0 * n / T_temp) * c
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lse_gradient_descent(z0, time_limit, T_start, T_min, T_decay, steps_per_T, momentum):
    z = z0.copy().astype(np.float64)
    n = len(z)
    t0 = time.monotonic()
    best_obj = 2.0 * n * fast_convolve(softmax_from_z(z)).max()
    best_z = z.copy()
    T = T_start
    v = np.zeros_like(z)
    while T > T_min and time.monotonic() - t0 < time_limit:
        prev_obj = best_obj
        for step in range(steps_per_T):
            if time.monotonic() - t0 > time_limit:
                break
            obj, grad = lse_obj_grad_z(z, T)
            v = momentum * v - 0.01 * grad
            z_new = z + v
            z_obj, _ = lse_obj_grad_z(z_new, T)
            if z_obj < obj:
                z = z_new
            else:
                v = 0.5 * v
            a_cur = softmax_from_z(z)
            cur_obj = 2.0 * n * fast_convolve(a_cur).max()
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_z = z.copy()
        if best_obj > prev_obj - 1e-12:
            T *= T_decay
    return best_z, best_obj


def nesterov_lp_alternate(a0, time_limit, max_iters):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(12345)
    iter_count = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b_new, M_b = solve_lp_response(y)
        if b_new is None:
            continue
        iter_count += 1
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        if (it + 1) % 500 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            if stall_count >= 5:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 8)
                noise = 0.03 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                a = a + noise * 0.5
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall_count = 0
            elapsed = time.monotonic() - t0
            print(f"    LP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 0.3 + 0.4 * (1 - np.abs(2 * x - 1))
    else:
        a = np.exp(-3 * (x - 0.5) ** 2) + 0.2 * np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def fine_gradient_refine(a0, time_limit, T_stages, steps_per_stage):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_obj = 2.0 * n * fast_convolve(a0).max()
    best_a = a0.copy()
    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        z, obj = lse_gradient_descent(z, remaining, T, T * 0.5, 0.8,
                                      min(steps_per_stage, int(remaining * 50)), 0.9)
        a_new = softmax_from_z(z)
        cur_obj = 2.0 * n * fast_convolve(a_new).max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a_new.copy()
    return best_a, best_obj


def perturbed_sequence(a, rng, strength):
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 10)
    phase = rng.uniform(0, 2 * np.pi)
    perturbation = strength * np.cos(freq * x + phase)
    a_pert = a + perturbation
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s
```
