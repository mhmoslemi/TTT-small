import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_conv2(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def smoothed_max_optimize(a0, time_limit, beta_init=10.0, beta_final=1000.0, lr_init=0.1, label=""):

    s = a.sum()

    else:

    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    
    beta = beta_init
    lr = lr_init
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute softmax weights
        conv = fast_convolve(a)
        c_max = conv.max()
        scaled = beta * (conv - c_max)
        scaled = np.clip(scaled, -100, 0)
        weights = np.exp(scaled)
        Z = weights.sum()
        weights /= Z
        
        # Gradient: grad[j] = 2 * sum_k weights[k] * a[k-j] for valid indices
        # This is 2 * conv(weights, a) (linear convolution, truncated to length n)
        grad = 2.0 * fast_conv2(weights, a)[:n]

"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_conv2(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def score_of(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return 2.0 * n
    a_norm = a / s
    return 2.0 * n * fast_convolve(a_norm).max()


def smooth_max_optimize(a0, time_limit, beta_init=5.0, lr_init=0.05, label=""):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    beta = beta_init
    lr = lr_init
    it = 0
    no_improve = 0
    gamma = 1.0
    y = a.copy()

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov lookahead
        if it > 1:
            y = (1.0 + gamma) * a - gamma * y
        y = simplex_project(np.maximum(y, 0))

        # Compute softmax weights on autocorrelation
        conv_y = fast_convolve(y)
        c_max = conv_y.max()
        scaled = beta * (conv_y - c_max)
        scaled = np.clip(scaled, -80, 0)
        weights = np.exp(scaled)
        Z = weights.sum()
        if Z < 1e-30:
            weights = np.zeros_like(conv_y)
            weights[conv_y.argmax()] = 1.0
            Z = 1.0
        weights /= Z

        # Gradient via FFT: grad = 2 * conv(weights, y) truncated to n
        grad = 2.0 * fast_conv2(weights, y)[:n]

        # Mirror descent step
        log_a = np.log(a + 1e-30)
        log_a -= lr * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = simplex_project(a_new)

        # Evaluate
        score = score_of(a_new)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        gamma = (1.0 + np.sqrt(1.0 + 4.0 * gamma * gamma)) / 2.0

        # Adaptive parameters
        if no_improve > 40:
            lr *= 0.7
            beta *= 1.4
            no_improve = 0
            if lr < 1e-12:
                break
        elif it % 15 == 0:
            beta *= 1.03

        if it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SMO[{label}] it={it} ({elapsed:.0f}s): score={best_score:.6f} beta={beta:.1f} lr={lr:.6f}")

    return best_a, best_score


def peak_directed_perturb(a0, rng, strength=0.05):
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a)
    M = conv.max()
    peak_thr = M * 0.85
    peak_positions = np.where(conv >= peak_thr)[0]

    pressure = np.zeros(n)
    for k in peak_positions:
        w = conv[k] - peak_thr
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            pressure[i] += w * a[i] * a[j]

    p_max = pressure.max()
    if p_max > 1e-20:
        pressure = pressure / p_max

    # Reduce mass at high-pressure positions, add to low-pressure
    target = a * (1.0 - strength * pressure)
    noise = rng.standard_normal(n) * 0.1 * strength
    target += np.abs(noise) * a * strength
    target = np.maximum(target, 0)
    s = target.sum()
    if s < 1e-15:
        return a.copy()
    return target / s


def random_restructure(a0, rng, frac=0.2):
    n = len(a0)
    a = a0.copy()
    keep = rng.random(n) > frac
    n_keep = keep.sum()
    if n_keep < max(10, n // 5):
        keep = np.ones(n, dtype=bool)
        n_keep = n
    a[~keep] = 0
    a[keep] = a[keep] * (1.0 + rng.standard_normal(n_keep) * 0.3)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy()
    return a / s

        for i in range(n):

                a[i] = max(ai + best_delta, 0.0)
                conv[k_start:k_end + 1] += best_delta * coeffs
                if has_quad:
                    conv[2 * i] += best_delta * best_delta
                M = best_new_max

                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score

                improved = True

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        a4b, s4b = smart_coord_descent(a4a, t4b, label=f"p4b_n{n_alt}")
        best_seq, best_score = check_and_update(a4b, best_seq, best_score, f"p4_n{n_alt}")

    # === Phase 5: Continued smooth optimization on best ===
    print(f"\n=== Phase 5: Extended Smooth Opt (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.25, 300.0)
    if t5 > 30:
        print(f"  Budget: {t5:.0f}s, n={len(best_seq)}")
        a5, s5 = smooth_max_optimize(best_seq, t5, beta_init=20.0, lr_init=0.02, label="p5")
        best_seq, best_score = check_and_update(a5, best_seq, best_score, "p5_smooth")

    # === Phase 6: Final intensive polishing ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t6a = min(rem * 0.4, 200.0)
        a6a, s6a = smart_coord_descent(best_seq, t6a, label="p6a", verbose=True)
        best_seq, best_score = check_and_update(a6a, best_seq, best_score, "p6a_cd")

    rem = deadline - time.monotonic()
    if rem > 20:
        t6b = min(rem * 0.5, 120.0)
        a6b, s6b = fine_cd(best_seq, t6b, label="p6b")
        best_seq, best_score = check_and_update(a6b, best_seq, best_score, "p6b_fine")

    rem = deadline - time.monotonic()
    if rem > 10:
        a6c, s6c = fine_cd(best_seq, rem * 0.7, label="p6c")
        best_seq, best_score = check_and_update(a6c, best_seq, best_score, "p6c_fine")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
