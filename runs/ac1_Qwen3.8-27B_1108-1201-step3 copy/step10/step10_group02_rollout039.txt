Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-accelerated LogSumExp L-BFGS,

coordinate descent. Key improvements: O(n log n) gradient via FFT cross-correlation,
finer temperature annealing, adaptive active-set management, and multi-scale
interpolation to explore different sequence lengths.

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence using local computation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def cross_correlation(w, a, N=None):
    """Compute (w star a)[m] = sum_k w[k]*a[k-m] for m=0..len(a)-1 via FFT."""
    if N is None:
        size = len(w) + len(a) - 1
        N = 1 << (size - 1).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    r = ifft(W * np.conj(A), N).real
    return r[:len(a)]


def logsumexp_value_and_grad(a, T):
    """Compute LogSumExp smoothed objective and gradient using FFT."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp((b - max_b) / T)
    S = exp_b.sum()
    w = exp_b / S

    # Gradient: 2 * cross_correlation(w, a)
    grad = 2.0 * cross_correlation(w, a)

    # Objective: (2n/T) * (max_b + T*log(S)) = (2n/T) * logsumexp(b/T)
    obj = (2.0 * n / T) * (max_b + T * np.log(S))

    return obj, grad

    """L-BFGS on log-sum-exp with fine temperature schedule and FFT gradient."""
    n = len(a0)

    if s < 1e-15:
        return a

    t0 = time.monotonic()

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    for T in T_stages:

            return func_grad(a_inner)[0]

def build_conv_matrix_cols(a_fixed, n_peaks=None):
    """Build matrix T such that T @ b = convolve(a_fixed, b), using FFT acceleration."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))

    if n_peaks is not None:
        idx = np.argsort(-np.abs(T.sum(axis=1)))[:n_peaks]
        return T[idx, :], idx
    return T, np.arange(size)

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = eta0
    no_improve = 0
    prev_a = a.copy()
    prev_g = None

    for it in range(10000000):
        if time.monotonic() - t0 > time_limit - 1:
            break

        b = fast_convolve(y)
        max_b = b.max()

        # Use top peaks for subgradient
        threshold = max_b * (1.0 - (1.0 - top_k_frac) * 0.01)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Subgradient via cross-correlation with indicator of active set
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / len(active)
        g = 2.0 * cross_correlation(w, y)

        # BB step size
        if it >= 3 and prev_g is not None:
            s_vec = a - prev_a
            y_vec = g - prev_g
            s_dot_s = np.dot(s_vec, s_vec)
            s_dot_y = np.dot(s_vec, y_vec)
            y_dot_y = np.dot(y_vec, y_vec)
            if s_dot_s > 1e-18 and y_dot_y > 1e-18:
                if abs(s_dot_y) > 1e-18:
                    eta_bb = s_dot_s / abs(s_dot_y)
                else:
                    eta_bb = s_dot_s / y_dot_y
                eta = 0.5 * eta + 0.5 * eta_bb
                eta = max(1e-6, min(eta, 0.5))
            else:
                eta *= 0.999

        prev_a = a.copy()
        prev_g = g.copy()

        # Mirror descent update
        log_a = np.log(np.maximum(a, 1e-300))
        g_shifted = g - g.min()
        log_a_new = log_a - eta * g_shifted
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()

        # Nesterov
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * a_new
        y = np.maximum(y, 0)
        ys = y.sum()
        if ys < 1e-15:
            y = a_new.copy()
        else:
            y = y / ys

        a = a_new
        score = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if min(score, score_y) < best_score - 1e-14:
            best_score = min(score, score_y)
            best_a = (a if score < score_y else y).copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 300:
            rng = np.random.default_rng(it * 13 + 7)
            noise = rng.standard_normal(n) * 0.003
            a = np.maximum(a + noise, 0)
            a = a / a.sum()
            y = a.copy()
            no_improve = 0
            eta = max(eta, 0.02)

        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ActiveMD iter={it+1} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def coordinate_descent_targeted(a0, time_limit, n_trials=100000):
    """Targeted coordinate descent on peak-contributing positions."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    stall = 0

    for step in range(n_trials):
        if time.monotonic() - t0 > time_limit - 1:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Focus on the top few peaks
        peak_indices = np.argsort(-b)[:min(5, 2 * n - 1)]
        improved = False

        for peak_k in peak_indices:
            if improved:
                break
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_m = min(4, len(idx))
            top_pos = idx[np.argsort(contribs)[::-1][:top_m]]

            for pos in top_pos:
                if a[pos] < 1e-14:
                    continue
                for factor in [0.9999, 0.9998, 0.9995, 0.999, 0.998, 0.996, 0.993]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] * (1.0 - factor)
                    if diff < 1e-18:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others = a_trial[mask]
                    osum = others.sum()
                    if osum < 1e-15:
                        break
                    a_trial[mask] = others + diff * others / osum
                    a_trial = a_trial / a_trial.sum()
                    sc = evaluate_local(a_trial)
                    if sc < best_score - 1e-15:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            if stall > 500:
                # Simultaneous reduction of top contributors
                b = fast_convolve(a)
                peak_k = int(np.argmax(b))
                js = max(0, peak_k - n + 1)
                je = min(n - 1, peak_k)
                idx = np.arange(js, je + 1)
                contribs = a[idx] * a[peak_k - idx]
                top_positions = idx[np.argsort(contribs)[::-1][:3]]
                a_trial = a.copy()
                for pos in top_positions:
                    if a_trial[pos] > 1e-14:
                        a_trial[pos] *= 0.9995
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = evaluate_local(a_trial)
                if sc < best_score - 1e-15:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    stall = 0
                elif stall > 2000:
                    break
        else:
            stall = 0

        if (step + 1) % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc {elapsed:.0f}s: score={best_score:.10f}")

    return best_a, best_score


def dual_guided_refine(a0, time_limit, n_iters=5000):
    """
    Uses dual variable p (distribution over peaks) to guide optimization.
    At each step, solve the QP: min_a a^T Q(p) a s.t. a>=0, sum a=1
    where Q(p)[i,j] = p[i+j]. Then update p to concentrate on new peaks.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    # Initialize p as uniform over peaks
    p = np.ones(2 * n - 1) / (2 * n - 1)

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit - 1:
            break

        # Compute Q(p) implicitly: Q(p)[i,j] = p[i+j] if i+j < 2n-1
        # The quadratic form a^T Q(p) a = sum_{i,j} a[i]*a[j]*p[i+j] = sum_k p[k]*conv(a,a)[k]
        # Gradient: 2 * (p_star_a) where (p_star_a)[m] = sum_k p[k]*a[k-m]
        
        # Use mirror descent on the smooth objective a^T Q(p) a
        conv_val = fast_convolve(a)
        obj_val = np.dot(p, conv_val)
        
        # Gradient of obj w.r.t. a
        g = 2.0 * cross_correlation(p, a)
        
        # Mirror descent step
        eta = 0.01 / (1 + 0.001 * it)
        log_a = np.log(np.maximum(a, 1e-300))
        g_shifted = g - g.min()
        log_a_new = log_a - eta * g_shifted
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()
        a = a_new

        # Update p: concentrate on peaks of conv(a,a)
        conv_val = fast_convolve(a)
        max_c = conv_val.max()
        # Softmax with temperature
        T_p = 0.01 * max_c
        exp_c = np.exp((conv_val - max_c) / T_p)
        p = exp_c / exp_c.sum()

        score = 2.0 * n * conv_val.max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if (it + 1) % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    DualGuide iter={it+1} ({elapsed:.0f}s): score={best_score:.10f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


_lp_solver_cached = None


def get_lp_solver():
    """Find a working LP solver."""
    global _lp_solver_cached
    if _lp_solver_cached is not None:
        return _lp_solver_cached
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return _lp_solver_cached


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses FFT-accelerated LogSumExp L-BFGS, dual-guided mirror descent,
    LP alternating, and targeted coordinate descent with multi-n exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        n_init = 1200
        x = np.linspace(0, 1, n_init)
        best_seq = np.sin(np.pi * x) ** 1.5
        best_seq = np.maximum(best_seq, 0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.10f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01 or not np.isfinite(s):
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    # === Phase 1: FFT-accelerated LogSumExp L-BFGS (main workhorse) ===
    print(f"\n=== Phase 1: LogSumExp L-BFGS (score={best_score:.10f}, n={n_best}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 480.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        T_stages = [8.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.35, 0.25,
                    0.18, 0.12, 0.08, 0.05, 0.03, 0.02, 0.01, 0.005]
        a_lbs = logsumexp_lbgfs_fine(best_seq, t_alloc, T_stages=T_stages,
                                     maxiter_per_stage=8000)
        try_update(a_lbs, "lbgfs_p1")

    # === Phase 2: Nesterov LP Alternating ===
    print(f"\n=== Phase 2: Nesterov LP (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp = nesterov_lp_refine(best_seq, t_alloc, solver=solver, max_iters=30000)
        try_update(a_lp, "nesterov_lp")

    # === Phase 3: Second LogSumExp L-BFGS pass (finer) ===
    print(f"\n=== Phase 3: Fine LogSumExp L-BFGS (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 320.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        T_fine = [2.0, 1.0, 0.5, 0.3, 0.2, 0.12, 0.07, 0.04, 0.02, 0.01, 0.005, 0.003, 0.001]
        a_lbs2 = logsumexp_lbgfs_fine(best_seq, t_alloc, T_stages=T_fine,
                                      maxiter_per_stage=10000)
        try_update(a_lbs2, "lbgfs_p3")

    # === Phase 4: Dual-guided refinement ===
    print(f"\n=== Phase 4: Dual Guided (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 180.0)
    if t_alloc > 15:
        print(f"  Budget: {t_alloc:.0f}s")
        a_dual = dual_guided_refine(best_seq, t_alloc, n_iters=10000)
        try_update(a_dual, "dual_guide")

    # === Phase 5: Targeted Coordinate Descent ===
    print(f"\n=== Phase 5: Coord Descent (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 120.0)
    if t_alloc > 10:
        print(f"  Budget: {t_alloc:.0f}s")
        a_cd = coordinate_descent_targeted(best_seq, t_alloc, n_trials=150000)
        try_update(a_cd, "coord_desc")

    # === Phase 6: Multi-n exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        # Try a few nearby n values
        for n_try in [n_best - 100, n_best + 100, n_best + 200]:
            if time.monotonic() > deadline - 40:
                break
            if n_try < 500 or n_try > 3000:
                continue
            a_interp = interpolate_sequence(best_seq, n_try)
            # Quick refinement with L-BFGS
            t_sub = min(30.0, (deadline - time.monotonic()) * 0.3)
            if t_sub < 10:
                break
            T_quick = [3.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
            a_ref = logsumexp_lbgfs_fine(a_interp, t_sub, T_stages=T_quick,
                                         maxiter_per_stage=3000)
            try_update(a_ref, f"n={n_try}")

    # === Phase 7: Final coordinate descent polish ===
    print(f"\n=== Phase 7: Final Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.4, rem - 5)
    if t_alloc > 5:
        print(f"  Budget: {t_alloc:.0f}s")
        a_polish = coordinate_descent_targeted(best_seq, t_alloc, n_trials=200000)
        try_update(a_polish, "final_polish")

    # === Phase 8: One more LP pass if time remains ===
    print(f"\n=== Phase 8: Final LP (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_alloc = min(rem - 5, 60.0)
        if t_alloc > 10:
            a_lp2 = nesterov_lp_refine(best_seq, t_alloc, solver=solver, max_iters=10000)
            try_update(a_lp2, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\n{'='*60}")
        print(f"FINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
        print(f"{'='*60}")

    if best_seq is None:
        # Fallback: uniform sequence
        best_seq = np.ones(100) / 100.0
        best_score = evaluate_local(best_seq)
        print(f"Fallback: score={best_score:.10f}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
