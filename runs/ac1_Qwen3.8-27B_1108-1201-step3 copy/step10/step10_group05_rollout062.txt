    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

            elapsed = time.monotonic() - t0
            print(f"    ECD {label} sw={sweeps} ({elapsed:.0f}s): score={best_score:.6f}, M={M:.8f}")
    
    return best_a, best_score, sweeps

def smoothed_gradient_descent(a0, time_limit, beta=50.0, lr=0.1, label=""):
    """
    Frequency-domain smoothed gradient descent.
    Gradient of sum_k w[k]*conv[k] w.r.t. a[i] = 2*(w*a)[i] where w = softmax(beta*conv)
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        conv = fast_convolve(a)
        M = conv.max()
        score = 2.0 * n * M
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Compute weights (softmax)
        scaled = beta * (conv - M)
        scaled = np.clip(scaled, -50, 0)
        w = np.exp(scaled)
        w = w / w.sum()
        
        # Gradient: g[i] = 2 * sum_k w[k] * a[k-i] (for valid k-i)
        # This is 2 * (w * a) restricted to valid range
        # Using convolution: (w * a)[i] = sum_k w[k] * a[k-i]
        # But we need sum_k w[k] * a[k-i] where 0<=k-i<n
        # This is the "cross-correlation" style: g[i] = 2 * sum_j w[i+j] * a[j]
        # Which is the same as 2 * (full_convolution of w and a, reversed and indexed)
        
        # Actually: d/d a[i] conv[k] = 2*a[k-i] (for valid k-i)
        # So gradient[i] = sum_k w[k] * 2*a[k-i]
        # Let's compute this via FFT: it's the悲伤convolution of w with a, but indexed differently
        # gradient = 2 * ifft(fft(w) * conj(fft(a))) ... no
        # 
        # gradient[i] = 2 * sum_{k: 0<=k-i<n} w[k] * a[k-i]
        # Let j = k-i: gradient[i] = 2 * sum_{j: 0<=j<n, 0<=i+j<2n-1} w[i+j] * a[j]
        # This is a "sliding dot product": for each i, dot(w[i:i+n], a)
        # 
        # Efficient: use FFT. Define W as the full convolution spectrum.
        # Actually: sum_j w[i+j]*a[j] = (w_reversed * a)[some_index]
        # Let w_rev[j] = w[N-1-j] for appropriate N
        # Then sum_j w[i+j]*a[j] = (w_rev * a)[N-1-i] ... this is getting complicated.
        
        # Simpler: just use the direct formula with FFT
        # grad[i] = 2 * sum_j w[i+j] * a[j] for j where 0<=i+j<2n-1
        # This equals 2 * (w_padded * a_flipped)[N - 1 - i] where N is FFT size
        # 
        # Actually the simplest: use the fact that conv(a, b)[k] = sum_j a[j]*b[k-j]
        # We want sum_j w[i+j]*a[j]. Let b[j] = a[j], and define c[m] = w[m] for m in [0,2n-2].
        # sum_j w[i+j]*a[j] = sum_j a[j]*w[i+j] = (a * w_rev)[?]
        # 
        # Let me just do it directly for each i (O(n^2) total, fine for n~1024)
        grad = np.zeros(n)
        for i in range(n):
            j_max = min(n - 1, 2 * n - 2 - i)
            grad[i] = 2.0 * np.dot(w[i:i + j_max + 1], a[:j_max + 1])
        
        # Mirror descent step (exponential)
        log_a = np.log(a + 1e-30)
        log_a -= lr * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a = simplex_project(a_new)
        
        # Adaptive parameters
        if no_improve > 30:
            lr *= 0.85
            beta *= 1.2
            no_improve = 0
            if lr < 1e-8:
                break
        
        if it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SMD {label} it={it} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score

def perturb_sequence(a, rng, strength=0.05, method="gaussian"):
    """Perturb a sequence for exploration."""
    n = len(a)
    if method == "gaussian":
        noise = rng.standard_normal(n) * strength
        a_new = a * (1.0 + noise)
    elif method == "peak_targeted":
        conv = fast_convolve(a)
        M = conv.max()
        threshold = M * 0.85
        peak_pos = np.where(conv >= threshold)[0]
        pressure = np.zeros(n)
        for k in peak_pos:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                pressure[j] += a[j] * a[k - j]
        p_max = pressure.max()
        if p_max > 1e-20:
            pressure /= p_max
        a_new = a * (1.0 - strength * pressure + 0.3 * strength * rng.standard_normal(n))
    elif method == "zero_restart":
        frac = rng.uniform(0.1, 0.4)
        mask = rng.random(n) > frac
        if mask.sum() < n // 4:
            mask = np.ones(n, dtype=bool)
        a_new = a * mask
        a_new[mask] *= (1.0 + rng.standard_normal(mask.sum()) * 0.3)
    elif method == "shift":
        shift = rng.integers(1, max(2, n // 8))
        a_new = np.roll(a, shift)
        a_new = np.abs(a_new)
    else:
        a_new = a.copy()
    
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def symmetrize(a):
    """Make sequence palindromic."""
    return 0.5 * (a + a[::-1])

def check_score(a):
    """Evaluate sequence and return score."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return float(evaluate_sequence(a.tolist()))
