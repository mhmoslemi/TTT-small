import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate 2n*max(conv(a,a))/(sum(a))^2 with local computation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def solve_lp_fast(a_fixed):
    """Solve LP: min M s.t. (Tb)_k <= M, sum(b)=1, b>=0 using scipy HiGHS.
    Returns (b_opt, M_opt) or (None, None)."""
    n = len(a_fixed)
    size = 2 * n - 1
    
    # Build A_ub: (size x (n+1)) matrix
    # Row k: [a_fixed convolved structure for b, -1 for M]
    # (Tb)_k = sum_j T[k,j] b_j <= M
    # Row k of T: T[k,j] = a_fixed[k-j] if 0<=k-j<n else 0
    
    # Build the matrix efficiently using the toeplitz-like structure
    A_ub = np.zeros((size, n + 1))
    for j in range(n):
        # a_fixed[k-j] is nonzero when 0 <= k-j < n, i.e., j <= k < j+n
        k_start = j
        k_end = min(j + n, size)
        A_ub[k_start:k_end, j] = a_fixed[:k_end - k_start]
    A_ub[:, n] = -1.0  # subtract M
    
    b_ub = np.zeros(size)
    
    # Equality constraint: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Objective: minimize M
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Bounds: b_j in [0, inf], M in [-inf, inf]
    bounds = [(0, None)] * n + [(-np.inf, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if res.status == 0:
            b_opt = np.maximum(res.x[:n], 0)
            s = b_opt.sum()
            if s < 1e-15:
                return None, None
            b_opt = b_opt / s
            M_opt = res.x[n]
            return b_opt, float(M_opt)
    except Exception:
        pass
    return None, None


def nesterov_lp_optimize(a0, time_limit, max_iters=500000, stall_patience=30,
                         alpha0=0.5, print_interval=5000):
    """Nesterov-accelerated LP alternation with adaptive parameters."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score + 1.0
    stall = 0
    rng = np.random.default_rng(12345)
    
    # Running averages for Nesterov
    a_avg = a.copy()
    count = 0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP: find b that minimizes max_k (y*b)_k
        b_new, M_b = solve_lp_fast(y)
        if b_new is None:
            stall += 1
            if stall > 50:
                break
            continue
        
        # Adaptive alpha
        alpha = 2.0 / (it + 2.0)
        
        # Nesterov momentum update on y
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy
        
        # Primal iterated average
        count += 1
        a_avg = ((count - 1) * a_avg + alpha * b_new) / count
        a_avg = np.maximum(a_avg, 0)
        sa = a_avg.sum()
        if sa > 1e-15:
            a_avg = a_avg / sa
        
        # Evaluate and track best
        score_y = 2.0 * n * fast_convolve(y).max()
        score_avg = 2.0 * n * fast_convolve(a_avg).max()
        
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        if score_avg < best_score:
            best_score = score_avg
            best_a = a_avg.copy()
        
        # Stall detection with perturbation
        if (it + 1) % 1000 == 0:
            if best_score < prev_best - 1e-10:
                stall = 0
            else:
                stall += 1
            prev_best = best_score
            
            if stall >= stall_patience:
                # Perturb and restart
                noise_scale = 0.01 / max(1.0, np.sqrt(float(stall)))
                noise = rng.standard_normal(n) * noise_scale
                y = y + noise
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                a_avg = y.copy()
                stall = 0
                prev_best = evaluate_local(y)
            
            if print_interval and (it + 1) % print_interval == 0:
                elapsed = time.monotonic() - t0
                print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} M_LP={M_b:.8f}")
    
    return best_a, best_score


def subgradient_optimize(a0, time_limit, n_steps=200000, print_interval=20000):
    """Subgradient method with active-set averaging and adaptive step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Polyak-like step size
    g_norm_sq_sum = 0.0
    lr = 0.05
    no_improve = 0
    momentum = np.zeros(n)
    mu = 0.9  # momentum coefficient
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Active set: positions within 0.1% of max
        threshold = max_b * (1.0 - 0.001)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        # Compute subgradient: average of gradients at active positions
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            # grad[k] = a[k - idx] flipped
            g[idx] += a[k - idx]
        g /= len(active)
        
        # Momentum update
        momentum = mu * momentum + (1 - mu) * g
        g_use = momentum
        
        # Normalized step
        g_norm = np.linalg.norm(g_use)
        if g_norm < 1e-15:
            break
        
        # Step size: decrease if no improvement
        a_trial = a - lr * g_use / g_norm
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            a_trial = a.copy()
        else:
            a_trial = a_trial / st
        
        score = 2.0 * n * fast_convolve(a_trial).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 20:
                lr *= 0.9
                no_improve = 0
        
        lr = max(1e-6, min(0.1, lr))
        
        if step % print_interval == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad {elapsed:.0f}s: score={best_score:.8f} lr={lr:.6f} |A|={len(active)}")
    
    return best_a, best_score


def logsumexp_lbgfs(a0, time_limit, T_stages=None, print_interval=None):
    """L-BFGS with LogSumExp smoothing, multi-stage temperature cooling."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [20.0, 10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1,
                    0.05, 0.03, 0.02, 0.01]
    
    n_stages = len(T_stages)
    iters_per = max(50, 50000 // n_stages)
    
    for si, T_stage in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        def func_loc(z_vec):
            a_vec = np.exp(z_vec - z_vec.max())
            a_vec = a_vec / a_vec.sum()
            b_vec = fast_convolve(a_vec)
            max_b = b_vec.max()
            exp_b = np.exp(b_vec - max_b)
            lse = max_b + np.log(exp_b.sum())
            return (2.0 * n / T_stage) * lse
        
        def jac_loc(z_vec):
            a_vec = np.exp(z_vec - z_vec.max())
            a_vec = a_vec / a_vec.sum()
            b_vec = fast_convolve(a_vec)
            max_b = b_vec.max()
            exp_b = np.exp(b_vec - max_b)
            w = exp_b / exp_b.sum()
            # Compute gradient efficiently using convolution structure
            # grad_a[j] = sum_k w[k] * a[k-j] (for valid k-j)
            # This is the convolution of w_padded with a
            # w is length 2n-1, a is length n
            # grad_a[j] = sum_{k: 0<=k-j<2n-1} w[k] * a[k-j]
            # Let l = k-j, then k = j+l, l in [max(0,j-(2n-2)), min(n-1, 2n-2-j)]
            # Actually: grad_a[j] = sum_l a[l] * w[j+l] where 0<=j+l<2n-1
            # This is (a * w_reversed)[n-1-j] ... let me just use direct computation
            # via FFT for speed
            # w_conv[j] = sum_l a[l] * w[j+l] for l=0..n-1, j+l in 0..2n-2
            # = (a flipped) convolved with w, evaluated at specific points
            # Simpler: use the fact that grad_a = (T^T w) where T is the conv matrix
            # T^T w [j] = sum_k T[k,j] w[k] = sum_k a[k-j] w[k] (k-j in [0,n-1])
            # = sum_{l=0}^{n-1} a[l] w[l+j]
            # This is a cross-correlation!
            # Use FFT: corr = ifft(fft(a) * conj(fft(w_padded)))
            w_padded = np.zeros(2 * n)
            w_padded[:2*n-1] = w
            A = rfft(a, 2*n)
            W = rfft(w_padded, 2*n)
            corr = irfft(A * np.conj(W), 2*n)
            # corr[j] = sum_l a[l] * w_padded[j+l] (circular)
            # We need sum_l a[l] * w[l+j] which is corr with w reversed
            # Actually let me just do it directly for correctness
            grad_a = np.zeros(n)
            for j in range(n):
                k_start = j
                k_end = min(j + n, 2 * n - 1)
                l = np.arange(0, k_end - k_start)
                grad_a[j] = np.dot(a[l], w[k_start:k_end])
            grad_a *= 2.0 / T_stage
            S = np.dot(grad_a, a_vec)
            return a_vec * (grad_a - S)
        
        try:
            res = minimize(
                fun=func_loc, x0=z, jac=jac_loc, method='L-BFGS-B',
                options={'maxiter': iters_per, 'ftol': 1e-18,
                         'gtol': 1e-10, 'maxfun': iters_per * 500, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    a_out = np.exp(z - z.max())
    a_out = a_out / a_out.sum()
    return a_out


def peak_coord_descent(a0, time_limit, n_steps=100000, top_m=6, print_interval=20000):
    """Peak-directed coordinate descent targeting top convolution peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        top_peaks = np.argsort(b)[::-1][:min(top_m, len(b))]
        
        improved = False
        for peak_k in top_peaks[:3]:
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            order = np.argsort(contribs)[::-1][:4]
            
            for oi in order:
                pos = idx[oi]
                if a[pos] < 1e-12:
                    continue
                for factor in [0.995, 0.99, 0.985]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] * (1.0 - factor)
                    if diff < 1e-15:
                        break
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] = others + diff * others / others_sum
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall > 200:
                break
            # Random small perturbation when stuck
            if stall % 50 == 0:
                rng = np.random.default_rng(step)
                pos = rng.integers(0, n)
                a[pos] *= 0.99
                a = np.maximum(a, 0)
                a = a / a.sum()
        
        if step % print_interval == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakCD {elapsed:.0f}s: score={best_score:.8f}")
    
    return best_a, best_score


def frequency_smooth(a0, time_limit, n_iters=5000, print_interval=1000):
    """Frequency-domain smoothing: reduce high-frequency content to flatten peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute FFT
        A = rfft(a, n)
        
        # Apply a low-pass filter (reduce high frequencies)
        # Use a cosine-tapered window in frequency domain
        freqs = np.arange(n // 2 + 1)
        # Taper: keep low freqs, reduce high freqs
        taper = np.ones(len(freqs))
        cutoff = int(0.3 * len(freqs))
        for i in range(cutoff, len(freqs)):
            t = (i - cutoff) / max(1, len(freqs) - cutoff)
            taper[i] = 1.0 - 0.5 * t * t
        
        A_smooth = A * taper
        # Rebuild real signal
        a_smooth = irfft(A_smooth, n)
        a_smooth = np.maximum(a_smooth, 0)
        ss = a_smooth.sum()
        if ss < 1e-15:
            break
        a_smooth = a_smooth / ss
        
        # Blend with current
        blend = 0.1 + 0.3 * (it / n_iters)  # increase blending over time
        a_trial = (1 - blend) * a + blend * a_smooth
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            break
        a_trial = a_trial / st
        
        score = evaluate_local(a_trial)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
        
        if it % print_interval == 0:
            elapsed = time.monotonic() - t0
            print(f"    FreqSmooth {elapsed:.0f}s: score={best_score:.8f}")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
