import numpy as np
from scipy.fft import rfft, irfft
import time

# --- Helper Functions ---

def fast_convolve(a):
    """Compute linear convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    """Compute the objective function value."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * float(conv.max()) / (s * s)

def solve_lp_response(a, solver=None):
    """Solve the LP: min M s.t. T_a @ b <= M, sum(b)=1, b>=0"""
    import cvxpy as cp
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    # Build T_a efficiently
    # T_a[k, i] = a[k-i] if 0 <= k-i < n else 0
    # For each row k, the non-zero entries are at columns max(0, k-n+1) to min(n-1, k)
    # We'll build it as a sparse matrix
    from scipy.sparse import csr_matrix
    rows = []
    cols = []
    vals = []
    for i in range(n):
        for k in range(i, i + n):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    T = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except:
        prob.solve(solver='ECOS')
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(b_var.value, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val /= s
    return b_val, float(M_var.value)

def golden_section_search(f, lo, hi, max_iter=35):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-14:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5

def coordinate_descent(a0, time_limit, verbose=False, label=""):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()

    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue
            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            unaff_max = 0.0
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)
            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]
            if has_quad:
                quad_idx = 2 * i - k_lo

            upper = max(3.0 * v, 10.0 / n)

            def _score(x, _c=c_vals, _ap=a_partners, _qm=unaff_max,
                       _hq=has_quad, _qi=quad_idx, _cs=current_sum,
                       _v=v, _n=n):
                ns = _cs + (x - _v)
                if ns < 0.01:
                    return 1e10
                nv = _c + 2.0 * x * _ap

                    nv = nv.copy()
                    nv[_qi] = _c[_qi] + x * x
                nm = max(_qm, float(nv.max()))
                return 2.0 * _n * nm / (ns * ns)

            best_x = golden_section_search(_score, 0.0, upper, max_iter=28)

            if abs(best_x - v) > 1e-13:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta
                changes = 2.0 * delta * a_partners

                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

def subgradient_step(a, conv, lr, top_k=15, temp_scale=0.005):
    n = len(a)
    max_val = float(conv.max())
    if max_val < 1e-15:
        return a
    k_act = min(top_k, 2 * n - 1)
    peaks = np.argpartition(conv, -k_act)[-k_act:]
    pv = conv[peaks]
    temp = temp_scale * max_val
    w = np.exp((pv - pv.max()) / temp)
    ws = w.sum()
    if ws < 1e-15:
        return a
    w = w / ws

    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-12:
            continue
        k = int(peaks[idx])
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        j = np.arange(i_start, i_end + 1)
        grad[j] += 2.0 * w[idx] * a[k - j]

    a_new = a - lr * grad
    return np.maximum(a_new, 0)

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

        conv = fast_convolve(a)
        a_new = subgradient_step(a, conv, lr)
        a_new = simplex_project(a_new)
        sc = compute_score(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        a_avg = (iters * a_avg + a_new) / (iters + 1)
        if iters % 50 == 0:
            sc_avg = compute_score(a_avg)
            if sc_avg < best_score:
                best_score = sc_avg
                best_a = a_avg.copy()
        a = a_new
        lr *= 0.998
        if no_improve > 200:
            lr = max(lr * 0.5, 1e-5)
            no_improve = 0
        if no_improve > 500:
            break
        if verbose and iters % 100 == 0:
            print(f"    SG it={iters}: best={best_score:.8f}, lr={lr:.6f}")

    return best_a, best_score

def lp_fictitious_play(a0, time_limit, solver=None, verbose=False):
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    a_avg = a.copy()
    no_improve = 0
    decay = 5.0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 10:
            break

        b, M_val, dual = solve_lp_response(a, solver)
        if b is None:
            continue

        # Adaptive mixing: try multiple step sizes
        best_t = 0.0
        best_ls = best_score
        trial_ts = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
        for trial_t in trial_ts:
            cand = (1.0 - trial_t) * a + trial_t * b
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-10:
                continue
            cand = cand / cs
            sc = compute_score(cand)
            if sc < best_ls:
                best_ls = sc
                best_t = trial_t

        if best_t > 0:
            a_new = (1.0 - best_t) * a + best_t * b
            a_new = np.maximum(a_new, 0)
            cs = a_new.sum()
            if cs < 1e-10:
                a_new = a
            else:
                a_new = a_new / cs
            score_new = compute_score(a_new)
            if score_new < best_score - 1e-13:
                best_score = score_new
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1
            a = a_new
        else:
            no_improve += 1
            alpha = decay / (iters + decay)
            a = (1.0 - alpha) * a + alpha * b
            a = np.maximum(a, 0)
            cs = a.sum()
            if cs < 1e-10:
                a = np.ones(n) / n
            else:
                a = a / cs

        a_avg = (iters * a_avg + a) / (iters + 1)

        # Check averaging and best response
        for cand in [a, a_avg, b]:
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-10:
                continue
            cand = cand / cs
            sc = compute_score(cand)
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()

        if verbose and iters % 5 == 0:
            print(f"    FP n={n} it={iters}: best={best_score:.8f}, M={M_val:.6f}" if M_val is not None else f"    FP n={n} it={iters}: best={best_score:.8f}")

        if no_improve > 40:
            # Try a subgradient kick to escape
            conv_now = fast_convolve(a)
            a_kick = subgradient_step(a, conv_now, 0.01)
            a_kick = simplex_project(a_kick)
            sc_kick = compute_score(a_kick)
            if sc_kick < best_score:
                best_score = sc_kick
                best_a = a_kick.copy()
            a = 0.5 * a + 0.5 * a_kick
            a = np.maximum(a, 0)
            a = a / a.sum()
            no_improve = 0

    return best_a, best_score, iters

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def init_asymmetric(n, rng, center_frac=0.3, width_frac=0.25):
    t = np.arange(n, dtype=np.float64)
    center = n * center_frac
    width = n * width_frac
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    # Add a secondary bump for more asymmetry
    a2_center = n * (1.0 - center_frac) * 0.7
    a2_width = n * 0.1
    a += 0.3 * np.exp(-0.5 * ((t - a2_center) / a2_width) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_flat_spectrum(n, rng):
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = rng.uniform(0.8, 1.2, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def basin_hopping(a0, time_limit, rng, verbose=False):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    count = 0

    while time.monotonic() - t0 < time_limit:
        count += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 30:
            break
        t_alloc = min(rem * 0.3, 90.0)

        pert_type = count % 5
        perturbed = a.copy()

        if pert_type == 0:
            # Peak-directed reduction
            conv_p = fast_convolve(a)
            top_k = min(8, 2 * n - 1)
            peaks = np.argpartition(conv_p, -top_k)[-top_k:]
            for k in peaks:
                i_start = max(0, int(k) - n + 1)
                i_end = min(n - 1, int(k))
                idx = np.random.default_rng(count * 7 + k).choice(
                    np.arange(i_start, i_end + 1),
                    size=max(1, (i_end - i_start + 1) // 5), replace=False)
                for ii in idx:
                    perturbed[ii] *= (1.0 - 0.08 * rng.random())
        elif pert_type == 1:
            sigma = 0.03 + 0.02 * rng.random()
            perturbed = a + rng.standard_normal(n) * sigma
        elif pert_type == 2:
            # Block rotation for asymmetry
            blen = rng.integers(max(1, n // 10), max(2, n // 4))
            i1 = rng.integers(0, max(1, n - blen))
            block = perturbed[i1:i1+blen].copy()
            perturbed[i1:i1+blen] = np.roll(block, rng.integers(1, blen))
        elif pert_type == 3:
            # Mirror mix
            shift = rng.integers(1, max(2, n // 5))
            shifted = np.roll(a, shift)
            perturbed = 0.7 * a + 0.3 * shifted
        else:
            # Random coordinate perturbation
            n_pert = max(5, n // 20)
            idx = rng.choice(n, size=n_pert, replace=False)
            perturbed[idx] *= (1.0 + rng.standard_normal(n_pert) * 0.15)

        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-10:
            continue
        perturbed = perturbed / ps

        a_opt, s_opt, _ = coordinate_descent(perturbed, t_alloc, label=f"bh_{count}")
        if s_opt < best_score - 1e-10:
            best_score = s_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"    BH {count}: improved to {best_score:.8f}")

    return best_a, best_score

def propose_candidate():
