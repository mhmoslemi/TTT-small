"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive mirror descent with
thermodynamic peak weighting, coordinate descent polish, peak equalization,
and aggressive multi-n exploration. Key innovations: (1) multiplicative weight
updates that naturally respect the simplex constraint, (2) adaptive temperature
scheduling for peak selection, (3) coordinate-wise quadratic minimization,
(4) systematic exploration of sequence lengths with interpolation-based transfer.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def obj_normalized(a):
    """Evaluate 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def local_eval(a):
    """Full evaluation matching evaluate_sequence."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def compute_subgradient(a, b, n_hot_frac=0.1, T=None):
    """Compute weighted subgradient of max_k (a*a)[k].
    Uses thermodynamic weighting: w[k] proportional to exp(b[k]/T).
    """
    n = len(a)
    size = 2 * n - 1
    if T is None or T <= 0:
        # Hard max: average over top n_hot_frac fraction
        n_hot = max(1, int(n_hot_frac * size))
        top_idx = np.argpartition(b, -n_hot)[-n_hot:]
        w = np.zeros(size)
        w[top_idx] = 1.0 / n_hot
    else:
        # Soft: Boltzmann weights
        b_shift = b - b.max()
        w = np.exp(b_shift / T)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(size)
            w[int(np.argmax(b))] = 1.0
        else:
            w /= w_sum

    g = np.zeros(n)
    active = np.where(w > 1e-14)[0]
    for k in active:
        wk = w[k]
        if wk < 1e-14:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += 2.0 * wk * a[k - idx]
    return g


def mirror_descent_optimize(a0, time_limit, T_init=0.05, T_cool=0.9995, eta_init=0.3):
    """Mirror descent (multiplicative weights) with adaptive temperature and step size."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = obj_normalized(a)
    t0 = time.monotonic()
    T = T_init
    eta = eta_init
    no_improve = 0
    step = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        g = compute_subgradient(a, b, T=T)

        # Mirror descent step
        log_a = np.log(np.maximum(a, 1e-300))
        log_a -= eta * g
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = a_new / a_new.sum()

        sc = obj_normalized(a_new)
        if sc < best_score - 1e-15:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            # Slightly increase eta if improving
            eta = min(eta * 1.005, 2.0)
        else:
            no_improve += 1
            if no_improve >= 20:
                eta *= 0.9
                T *= T_cool
                no_improve = 0

        step += 1
        if step % 5000 == 0:
            el = time.monotonic() - t0
            print(f"    MD step={step} ({el:.0f}s): score={best_score:.10f}, eta={eta:.4f}, T={T:.6f}")

    return best_a, best_score


def coordinate_descent(a0, time_limit, max_rounds=50000):
    """Coordinate descent: optimize one element at a time using local quadratic model."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = obj_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    for rnd in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        # Random coordinate
        i = rng.integers(0, n)

        # Compute current convolution
        b = fast_convolve(a)
        M = b.max()

        # Find optimal a[i] by trying a few values
        # The peaks affected by a[i] are at positions k in [i, min(i+n-1, 2n-2)]
        # (a*a)[k] = C_k + a[i]*D_k where D_k = 2*a[k-i] (for k != 2i) or 2*a[i] (for k=2i)
        # Actually D_k depends on a[i] itself when k=2i, making it quadratic.
        # Simplification: treat it as linear for the step, solve for best step.

        a_i_old = a[i]
        if a_i_old < 1e-15:
            # Try small positive value
            a[i] = 0.5 / n
        else:
            # Try reducing a[i] and redistributing to a random other coordinate
            # Or try a local linear search
            best_local = a_i_old
            best_local_sc = M
            for frac in [0.5, 0.7, 0.8, 0.9, 1.1, 1.3, 1.5]:
                a_trial = a.copy()
                delta = a_i_old * (frac - 1.0)
                a_trial[i] = a_i_old * frac
                # Redistribute delta to a random other position
                j = rng.integers(0, n)
                if j != i:
                    a_trial[j] = max(0, a_trial[j] + delta)
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                b_trial = fast_convolve(a_trial)
                sc_trial = b_trial.max()
                if sc_trial < best_local_sc:
                    best_local_sc = sc_trial
                    best_local_frac = frac
                    best_local_j = j
            if best_local_frac != 1.0:
                delta = a_i_old * (best_local_frac - 1.0)
                a[i] = max(0, a_i_old * best_local_frac)
                a[best_local_j] = max(0, a[best_local_j] + delta)
                a = normalize(a)

        # Check improvement
        sc = obj_normalized(a)
        if sc < best_score - 1e-15:
            best_score = sc
            best_a = a.copy()

        if (rnd + 1) % 10000 == 0:
            print(f"    CD round={rnd+1} ({time.monotonic()-t0:.0f}s): score={best_score:.10f}")

    return best_a, best_score


def peak_equalization(a0, time_limit, n_top=5):
    """Iteratively equalize the top peaks by solving small linear systems."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = obj_normalized(a)
    t0 = time.monotonic()

    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        M = b.max()

        # Find top peaks
        top_idx = np.argsort(b)[::-1][:n_top]
        peak_vals = b[top_idx]

        # If peaks are already well-equalized, try a different strategy
        peak_range = peak_vals.max() - peak_vals.min()
        if peak_range < 1e-10:
            # All top peaks are equal; try reducing them all
            break

        # For each top peak k, the gradient w.r.t. a[i] is 2*a[k-i]
        # We want to find a direction d that reduces all top peaks equally
        # subject to sum(d) = 0 and a + d >= 0

        # Build the system: for each top peak k, sum_i (2*a[k-i]) * d[i] = -alpha
        # where alpha is the reduction amount, and sum(d) = 0
        # This is an underdetermined system; use minimum norm solution

        m = len(top_idx)
        A_mat = np.zeros((m + 1, n))
        for i, k in enumerate(top_idx):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            A_mat[i, idx] = 2.0 * a[k - idx]
        A_mat[m, :] = 1.0  # sum constraint

        # Target: reduce each peak by alpha, sum of d = 0
        target = np.zeros(m + 1)
        alpha = peak_range * 0.1  # 10% of the range
        target[:m] = -alpha
        target[m] = 0.0

        # Minimum norm solution: d = A^T (A A^T)^{-1} target
        ATA = A_mat @ A_mat.T
        try:
            ATA_inv_target = np.linalg.solve(ATA + 1e-10 * np.eye(m + 1), target)
            d = A_mat.T @ ATA_inv_target
        except np.linalg.LinAlgError:
            break

        # Ensure a + d >= 0
        max_neg = np.minimum(0, a + d)
        if max_neg.min() < -1e-15:
            scale = -1e-15 / max_neg.min() if max_neg.min() != 0 else 1.0
            d *= max(0, scale) * 0.9

        a_trial = a + d
        a_trial = np.maximum(a_trial, 0)
        a_trial = a_trial / a_trial.sum()

        sc = obj_normalized(a_trial)
        if sc < best_score - 1e-15:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
        else:
            # Try smaller step
            for step_scale in [0.5, 0.25, 0.1]:
                a_trial = a + step_scale * d
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                sc = obj_normalized(a_trial)
                if sc < best_score - 1e-15:
                    best_score = sc
                    best_a = a_trial.copy()
                    a = a_trial
                    break
            else:
                # No improvement; add small noise
                a = a + rng_noise(a, 0.001)
                a = normalize(a)

    return best_a, best_score


def rng_noise(a, scale):
    """Add small Gaussian noise."""
    rng = np.random.default_rng(int(time.monotonic() * 1e6) % 100000)
    return a + rng.standard_normal(len(a)) * scale


def adaptive_subgradient(a0, time_limit, n_hot=10):
    """Adaptive subgradient with Polyak step size and momentum."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = obj_normalized(a)
    t0 = time.monotonic()
    momentum = np.zeros(n)
    beta = 0.9
    lr = 0.1
    best_lr = lr
    no_improve = 0

    for step in range(500000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        M = b.max()

        # Select hot peaks
        n_hot_actual = min(n_hot, 2 * n - 1)
        top_idx = np.argpartition(b, -n_hot_actual)[-n_hot_actual:]
        w = np.zeros(2 * n - 1)
        w[top_idx] = 1.0 / n_hot_actual

        # Compute subgradient
        g = np.zeros(n)
        for k in top_idx:
            wk = w[k]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * wk * a[k - idx]

        # Momentum update
        momentum = beta * momentum + (1 - beta) * g
        g_eff = momentum

        # Normalized step
        g_norm = np.linalg.norm(g_eff)
        if g_norm < 1e-15:
            break
        g_eff = g_eff / g_norm

        a_trial = a - lr * g_eff
        a_trial = np.maximum(a_trial, 0)
        s = a_trial.sum()
        if s < 1e-15:
            break
        a_trial = a_trial / s

        sc = obj_normalized(a_trial)
        if sc < best_score - 1e-15:
            best_score = sc
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
            best_lr = lr
        else:
            no_improve += 1
            if no_improve >= 30:
                lr *= 0.95
                no_improve = 0

    return best_a, best_score


def symmetrize(a):
    """Project to symmetric subspace."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    return normalize(a_sym)


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Create structured initial sequence with various shapes."""

        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:

    elif choice == 7:

        a = x ** 2 + (1 - x) ** 2 + 0.1
    else:
        # Bimodal
        a = np.exp(-20 * (x - 0.2) ** 2) + np.exp(-20 * (x - 0.8) ** 2) + 0.05
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

    print(f"\n=== Phase 8: Final Round (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        a_final, _ = mirror_descent_optimize(best_seq, min(rem * 0.4, 60.0),
                                             T_init=0.005, T_cool=0.99995, eta_init=0.1)
        try_update(a_final, "phase8_md")

        rem = deadline - time.monotonic()
        if rem > 15:
            a_final2, _ = adaptive_subgradient(best_seq, min(rem * 0.5, 40.0), n_hot=2)
            try_update(a_final2, "phase8_sg")

    # Final verification
    if best_seq is not None:
        fs = local_eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
        try:
            eval_score = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
