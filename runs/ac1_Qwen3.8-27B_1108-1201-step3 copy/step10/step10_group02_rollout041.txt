    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:

    return exp_z / s

                            best_score = sc_lp
                            best_a = a_lp.copy()
                            a = a_lp.copy()
                            y = a.copy()
                            no_improve = 0

        # Anti-symmetric perturbation
        if use_antisym and (it + 1) % 2000 == 0 and no_improve > 100:
            flip_a = a[::-1].copy()
            s_sym = 0.5 * (a + flip_a)
            # Add anti-symmetric component to reduce center peak
            d = np.zeros(n)
            for i in range(n // 2):
                if s_sym[i] > s_sym[n - 1 - i] + 1e-10:
                    d[i] = min(s_sym[n - 1 - i], 0.3 * s_sym[i])
                    d[n - 1 - i] = -d[i]
                elif s_sym[n - 1 - i] > s_sym[i] + 1e-10:
                    d[n - 1 - i] = min(s_sym[i], 0.3 * s_sym[n - 1 - i])
                    d[i] = -d[n - 1 - i]
            a_asym = s_sym + d
            a_asym = np.maximum(a_asym, 0)
            s_asym = a_asym.sum()
            if s_asym > 1e-10:
                a_asym /= s_asym
                sc_asym = evaluate_local(a_asym)
                if sc_asym < best_score:
                    best_score = sc_asym
                    best_a = a_asym.copy()
                    a = a_asym.copy()
                    y = a.copy()
                    no_improve = 0

        # Restart with noise if stuck
        if no_improve >= 300:
            noise_scale = 0.01 + 0.01 * min(no_improve / 300.0, 2.0)
            noise = rng.standard_normal(n) * noise_scale
            a = a + noise
            a = np.maximum(a, 0)
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
            else:
                a = a / s
            y = a.copy()
            no_improve = 0
            eta = max(eta, eta0)

        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} eta={eta:.6f} |A|={len(active)}")

    return best_a, best_score


def lp_alternating(a0, time_limit, solver=None, n_max_iters=100000,
                   nesterov=True, diversify_every=50):
    """
    LP-based alternating optimization with Nesterov acceleration.
    The main workhorse for this problem.
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall = 0
    rng = np.random.default_rng(77777)
    T = build_conv_matrix(a)  # Will rebuild as a changes

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Solve LP for best response
        M_current = fast_convolve(y).max()
        b_new, _ = solve_lp_best_response(y, M_current, solver)
        if b_new is None:
            continue

        # Nesterov update
        alpha = 2.0 / (it + 2.5)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        # Average iterate
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        # Evaluate
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0

        if (it + 1) % 50 == 0:
            if best_score < prev_best - 1e-12:
                stall = 0
            else:
                stall += 1
            prev_best = best_score

            if stall >= 8:
                # Diversification: mix with random or anti-symmetric perturbation
                if rng.random() < 0.5:
                    noise = rng.standard_normal(n) * (0.02 / np.sqrt(stall))
                    y = y + noise
                else:
                    flip_y = y[::-1].copy()
                    s_sym = 0.5 * (y + flip_y)
                    d = np.zeros(n)
                    for i in range(n // 2):
                        diff = s_sym[i] - s_sym[n - 1 - i]
                        if abs(diff) > 1e-12:
                            d[i] = 0.2 * diff
                            d[n - 1 - i] = -0.2 * diff
                    y = s_sym + d
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y /= sy
                stall = 0

        if (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} stall={stall}")

    return best_a, best_score


def logsumexp_objective_and_grad(z, T_temp):
    """Compute LogSumExp smoothed objective and gradient in softmax parameterization."""
    a = softmax_safe(z)
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp((b - max_b) / T_temp)
    w = exp_b / exp_b.sum()

    # Gradient of logsumexp(b/T) w.r.t. a: (1/T) * w * da_b/da_i
    # da_b_k/da_i = a[k-i] (for valid indices)
    grad_a = np.zeros(n)
    size = 2 * n - 1
    for k in range(size):
        if w[k] < 1e-14:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    grad_a /= T_temp

    # Log-sum-exp value
    logsumexp_val = max_b / T_temp + np.log(exp_b.sum())

    # Overall objective: 2n * logsumexp (since sum(a)=1 under softmax)
    obj = 2.0 * n * logsumexp_val

    # Convert to z-gradient (Jacobian of softmax)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)

    return obj, grad_z


def logsumexp_lbgfs(a0, time_limit, T_stages=None, maxiter_per_stage=2000):
    """L-BFGS on log-sum-exp smoothed objective with decreasing temperature."""
    n = len(a0)
    a0 = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    z = np.log(a0 / a0.sum()).copy()
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]

    n_stages = len(T_stages)
    iters_per_stage = max(20, maxiter_per_stage // n_stages)

    for si, T in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit - 5:
            break

        def func(z_inner):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            obj, _ = logsumexp_objective_and_grad(z_inner, T)
            return obj

        def jac(z_inner):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_inner)
            _, grad = logsumexp_objective_and_grad(z_inner, T)
            return grad

        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                                    'gtol': 1e-15, 'maxfun': iters_per_stage * 200,
                                    'disp': False})
            z = res.x
        except Exception:
            pass

        if (si + 1) % 3 == 0:
            a_curr = softmax_safe(z)
            sc = evaluate_local(a_curr)
            elapsed = time.monotonic() - t0
            print(f"    L-BFGS stage {si+1}/{n_stages} (T={T}) ({elapsed:.0f}s): score={sc:.8f}")

    return softmax_safe(z)


def coordinate_descent_refine(a0, time_limit, n_steps=100000, top_k=8, factors=None):
    """Coordinate descent targeting peak-contributing elements with wider search."""
    if factors is None:
        factors = [0.9999, 0.9998, 0.9995, 0.999, 0.998, 0.997, 0.995, 0.99, 0.985, 0.97]
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    stall = 0
    rng = np.random.default_rng(99999)

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Find all peaks within 0.1% of max
        threshold = max_b * (1.0 - 1e-4)
        peaks = np.where(b >= threshold)[0]
        if len(peaks) == 0:
            peaks = np.array([int(np.argmax(b))])
        # Limit peaks to top 20
        if len(peaks) > 20:
            peak_vals = b[peaks]
            top_idx = np.argsort(peak_vals)[::-1][:20]
            peaks = peaks[top_idx]

        improved = False
        for peak_k in peaks:
            if improved or time.monotonic() - t0 > time_limit:
                break
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_m = min(top_k, len(idx))
            top_positions = idx[np.argsort(contribs)[::-1][:top_m]]

            for pos in top_positions:
                if improved:
                    break
                if a[pos] < 1e-12:
                    continue
                for factor in factors:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] - a_trial[pos]
                    if diff < 1e-16:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others_sum = a_trial[mask].sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] += diff * a_trial[mask] / others_sum
                    a_trial = a_trial / a_trial.sum()
                    score = evaluate_local(a_trial)
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break

        if not improved:
            stall += 1
            # Try adding mass from low-contribution positions to high positions
            if stall > 200 and stall % 50 == 0:
                peak_k = int(np.argmax(fast_convolve(a)))
                js = max(0, peak_k - n + 1)
                je = min(n - 1, peak_k)
                idx = np.arange(js, je + 1)
                contribs = a[idx] * a[peak_k - idx]
                # Reduce top contributors slightly
                top_positions = idx[np.argsort(contribs)[::-1][:3]]
                a_trial = a.copy()
                for pos in top_positions:
                    if a[pos] > 1e-12:
                        a_trial[pos] *= 0.999
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = evaluate_local(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
                elif stall > 2000:
                    break
            elif stall > 5000:
                break
        else:
            stall = 0

        if (step + 1) % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD {elapsed:.0f}s: score={best_score:.8f} stall={stall}")

    return best_a, best_score


def antisymmetric_improve(a0, time_limit, n_trials=200):
    """Try anti-symmetric perturbations to reduce the convolution peak."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a, evaluate_local(a)
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(31415)

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        max_b = b[peak_k]

        # Strategy: transfer mass from positions contributing to peak
        # to positions that don't contribute to any peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        peak_positions = set(range(js, je + 1))

        # Find positions NOT in any near-peak
        all_peaks = set()
        threshold = max_b * 0.999
        for k in np.where(b >= threshold)[0]:
            js2 = max(0, int(k) - n + 1)
            je2 = min(n - 1, int(k))
            all_peaks.update(range(js2, je2 + 1))

        safe_positions = [i for i in range(n) if i not in all_peaks and a[i] > 1e-10]
        risky_positions = [i for i in range(n) if i in all_peaks and a[i] > 1e-10]

        if not safe_positions or not risky_positions:
            continue

        # Pick a risky position to reduce and safe position to increase
        r_pos = risky_positions[rng.integers(0, len(risky_positions))]
        s_pos = safe_positions[rng.integers(0, len(safe_positions))]

        for frac in [0.01, 0.02, 0.05, 0.1, 0.2]:
            a_trial = a.copy()
            transfer = a[r_pos] * frac
            a_trial[r_pos] -= transfer
            a_trial[s_pos] += transfer
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                break
            a_trial /= st
            score = evaluate_local(a_trial)
            if score < best_score - 1e-13:
                best_score = score
                best_a = a_trial.copy()
                a = a_trial
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    f = np.asarray(f, dtype=np.float64)
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Combines mirror descent, LP alternating, anti-symmetric perturbation,
    log-sum-exp L-BFGS, and coordinate descent with dimension exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        x = np.linspace(0, 1, n_init)
        best_seq = np.cos(np.pi * x / 2) ** 2
        best_seq = np.maximum(best_seq, 0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Quick anti-symmetric improvement (60s) ===
    print(f"\n=== Phase 1: Anti-symmetric refinement (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(60.0, rem * 0.05)
    if t_alloc > 10:
        a_as, _ = antisymmetric_improve(best_seq, t_alloc, n_trials=500)
        try_update(a_as, "antisym_p1")

    # === Phase 2: Mirror Descent with LP (400s) ===
    print(f"\n=== Phase 2: Mirror Descent + LP (score={best_score:.8f}, n={n_best}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(400.0, rem * 0.25)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_md, _ = mirror_descent_optimize(best_seq, t_alloc,
                                          eta0=0.05, active_frac=0.999,
                                          nesterov=True, bb_adapt=True,
                                          lp_every=800, solver=solver,
                                          use_antisym=True)
        try_update(a_md, "mirror_desc_p2")

    # === Phase 3: Aggressive LP Alternating (500s) ===
    print(f"\n=== Phase 3: LP Alternating (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(500.0, rem * 0.30)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = lp_alternating(best_seq, t_alloc, solver=solver,
                                 n_max_iters=100000, nesterov=True,
                                 diversify_every=50)
        try_update(a_lp, "lp_alt")

    # === Phase 4: LogSumExp L-BFGS (350s) ===
    print(f"\n=== Phase 4: LogSumExp L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(350.0, rem * 0.22)
    if t_alloc > 30:
        T_stages = [10.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]
        print(f"  Budget: {t_alloc:.0f}s, {len(T_stages)} temperature stages")
        a_lb = logsumexp_lbgfs(best_seq, t_alloc, T_stages=T_stages,
                               maxiter_per_stage=8000)
        try_update(a_lb, "lbgfs_p4")

    # === Phase 5: Second LP pass from improved solution (250s) ===
    print(f"\n=== Phase 5: LP Pass 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(250.0, rem * 0.18)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, _ = lp_alternating(best_seq, t_alloc, solver=solver,
                                  n_max_iters=50000, nesterov=True,
                                  diversify_every=30)
        try_update(a_lp2, "lp_p2")

    # === Phase 6: Coordinate Descent Polish (150s) ===
    print(f"\n=== Phase 6: Coord Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(150.0, rem * 0.12)
    if t_alloc > 15:
        a_cd, _ = coordinate_descent_refine(best_seq, t_alloc, n_steps=200000,
                                            top_k=10)
        try_update(a_cd, "coord_desc")

    # === Phase 7: Dimension Exploration (remaining time) ===
    print(f"\n=== Phase 7: Dimension Exploration (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        n_trial_list = [800, 1000, 1500, 2000]
        n_trial_list = [n for n in n_trial_list if n != n_best]
        # Sort by expected benefit (closer to n_best first)
        n_trial_list.sort(key=lambda x: abs(x - n_best))

        for n_trial in n_trial_list[:3]:
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            print(f"  Trying n={n_trial} (remaining {rem:.0f}s)")
            a_interp = interpolate_sequence(best_seq, n_trial)
            # Quick mirror descent at new dimension
            t_sub = min(rem * 0.25, 120.0)
            if t_sub > 10:
                a_new, _ = mirror_descent_optimize(a_interp, t_sub,
                                                   eta0=0.03, active_frac=0.999,
                                                   nesterov=True, bb_adapt=True,
                                                   lp_every=500, solver=solver,
                                                   use_antisym=True)
                try_update(a_new, f"dim_{n_trial}")

    # === Phase 8: Final polish (remaining) ===
    print(f"\n=== Phase 8: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_alloc = min(rem * 0.5, 100.0)
        if t_alloc > 10:
            # Final coordinate descent
            a_polish, _ = coordinate_descent_refine(best_seq, t_alloc,
                                                    n_steps=100000, top_k=12)
            try_update(a_polish, "final_cd")

        rem = deadline - time.monotonic()
        if rem > 15:
            # Final L-BFGS at very low temp
            t_alloc2 = min(rem * 0.8, 60.0)
            if t_alloc2 > 10:
                a_final = logsumexp_lbgfs(best_seq, t_alloc2,
                                          T_stages=[2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01],
                                          maxiter_per_stage=3000)
                try_update(a_final, "final_lbgfs")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
