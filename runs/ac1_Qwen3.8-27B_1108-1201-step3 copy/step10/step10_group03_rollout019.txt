5. Make a small step in this direction.

This is basically the subgradient method, but with a specific focus on reducing the "excess."

    fft_size = 1 << (size - 1).bit_length()

def compute_subgradient(a, b, active_frac=0.005):
    n = len(a)
    M = b.max()
    thresh = M * (1.0 - active_frac)
    active = np.where(b >= thresh)[0]
    g = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += 2.0 * a[k - idx]

    g -= g.mean()  # project to tangent space of simplex

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Use the algorithm from Duchi et al. 2008
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    return best_a, best_score


def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T


def get_lp_solver():
    for sn in [cp.PDLP, cp.SCS, cp.ECOS, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=sn)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return sn
        except:
            continue
    return cp.ECOS


def lp_best_response(a_fixed, solver):
    """Given fixed a (sum=1), find b >= 0, sum(b)=1 minimizing max conv(a,b)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    tried = [solver]
    for alt in [cp.PDLP, cp.SCS, cp.ECOS, cp.CLARABEL]:
        if alt not in tried:
            tried.append(alt)
    for s in tried:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                bv = np.asarray(b_var.value, dtype=np.float64)
                bv = np.maximum(bv, 0)
                sv = bv.sum()
                if sv > 1e-15:
                    return bv / sv
        except:
            continue
    return None


def lp_alternation(a0, time_limit, solver, max_iters=5000):
    """Bidirectional LP alternation with Nesterov averaging."""
    n = len(a0)
    a = normalize(a0)
    b = a.copy()
    y_a = a.copy()
    y_b = b.copy()
    best_a = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()
    prev_best = best_score
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Best response for b given y_a
        b_new = lp_best_response(y_a, solver)
        if b_new is not None:
            alpha = 2.0 / (it + 2)
            y_b = (1 - alpha) * y_b + alpha * b_new
            y_b = normalize(y_b)
        
        # Best response for a given y_b
        a_new = lp_best_response(y_b, solver)
        if a_new is not None:
            alpha = 2.0 / (it + 3)
            y_a = (1 - alpha) * y_a + alpha * a_new
            y_a = normalize(y_a)
        
        # Evaluate and track
        sc_a = local_score(y_a)
        sc_b = local_score(y_b)
        if sc_a < best_score:
            best_score = sc_a
            best_a = y_a.copy()
        if sc_b < best_score:
            best_score = sc_b
            best_a = y_b.copy()
        
        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-10:
                stall = 0
            else:
                stall += 1
            prev_best = best_score
            if stall >= 5:
                # Perturb to escape
                rng_local = np.random.default_rng(it)
                y_a = normalize(y_a + rng_local.standard_normal(n) * 0.01)
                y_b = normalize(y_b + rng_local.standard_normal(n) * 0.01)
                stall = 0
        
        if (it + 1) % 500 == 0:
            print(f"    LP iter={it+1} ({time.monotonic()-t0:.0f}s): score={best_score:.10f}")
    
    return best_a, best_score


def logsumexp_lbgfs(a0, time_limit, T_stages=None):
    """L-BFGS on LogSumExp-smoothed objective with log parametrization."""
    n = len(a0)
    a = normalize(a0)
    a_clipped = np.maximum(a, 1e-300)
    z = np.log(a_clipped) - np.log(a_clipped.sum())
    best_a = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [50.0, 20.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
    
    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        max_iter = max(50, int(remaining * 30))
        
        def lse_obj(z):
            z_max = z.max()
            exp_z = np.exp(z - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            shifted = b / T
            b_max = shifted.max()
            lse = b_max + np.log(np.sum(np.exp(shifted - b_max)))
            return 2.0 * n * lse
        
        def lse_grad(z):
            z_max = z.max()
            exp_z = np.exp(z - z_max)
            a_cur = exp_z / exp_z.sum()
            b = fast_convolve(a_cur)
            shifted = b / T
            b_max = shifted.max()
            exp_ws = np.exp(shifted - b_max)
            w = exp_ws / exp_ws.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                wk = w[k]
                if wk < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * wk * a_cur[k - idx]
            S = np.dot(grad_a, a_cur)
            return a_cur * (grad_a - S)
        
        try:
            res = minimize(lse_obj, z, jac=lse_grad, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-20, 
                                   'gtol': 1e-14, 'maxfun': max_iter * 100})
            z_new = res.x
            a_new = np.exp(z_new - z_new.max())
            a_new = a_new / a_new.sum()
            sc = local_score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                z = z_new
        except:
            pass
        
        if T <= 0.1:
            print(f"    LSE L-BFGS T={T}: score={best_score:.10f} ({time.monotonic()-t0:.0f}s)")
    
    return best_a, best_score


def msequence_init(n, m=10):
    """Create a soft m-sequence inspired initialization of length n."""
    # Generate m-sequence of length 2^m - 1
    L = (1 << m) - 1
    # Use a simple LFSR to generate the m-sequence
    # For m=10, we need a primitive polynomial of degree 10
    # x^10 + x^7 + 1 (one of the primitive polynomials)
    state = 1
    seq = []
    for i in range(L):
        seq.append(1 if (state & 1) else 0)
        # Feedback: x^10 + x^7 + 1
        bit = ((state >> 9) ^ (state >> 6) ^ (state & 1)) & 1
        state = ((state >> 1) | (bit << 9)) & ((1 << m) - 1)
    
    seq = np.array(seq, dtype=np.float64)
    # Make it balanced (it should be for m-sequence)
    # Map to [0, 1] range and interpolate to length n
    from numpy import interp
    x_old = np.linspace(0, 1, L)
    x_new = np.linspace(0, 1, n)
    seq_interp = interp(x_new, x_old, seq)
    seq_interp = np.maximum(seq_interp, 0.01)
    return normalize(seq_interp)


def structured_init(n, rng):
    """Create structured initial sequence."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        # Soft m-sequence
        a = msequence_init(n, m=min(10, max(4, int(math.log2(n)))))
    elif choice == 4:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0.01)
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        a = 1.0 / (1.0 + np.exp(-50 * (x - 0.5)))
        a = a / a.max()
    elif choice == 7:
        a = 1.0 + 0.1 * np.cos(2 * np.pi * x) + 0.05 * np.cos(4 * np.pi * x)
    elif choice == 8:
        # Random with smooth structure
        a = 1.0 + 0.2 * np.cos(2 * np.pi * rng.uniform(1, 5) * x) 
        a = np.maximum(a, 0.01)
    else:
        # Multi-frequency
        a = 1.0
        for k in range(3):
            a += 0.1 * np.cos(2 * np.pi * (k + 1) * rng.uniform(0.5, 3.0) * x + rng.uniform(0, 2*np.pi))
        a = np.maximum(a, 0.01)
    return normalize(a)
