def fast_convolve(a):

def fast_cross_correlate(a, w):
    """
    Compute c[j] = sum_{m=0}^{n-1} a[m] * w[m+j] for j=0..n-1.
    Uses FFT for O(n log n) complexity.
    """
    n = len(a)
    L = 1 << (3 * n - 2 - 1).bit_length()
    a_rev = a[::-1]
    A_fft = rfft(a_rev, L)
    W_fft = rfft(w, L)
    D = np.irfft(A_fft * W_fft, L)
    return D[n - 1:2 * n - 1]


def lse_obj_grad(z, T, n):
    """Compute LogSumExp objective and gradient in z-space."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()

    b = fast_convolve(a)

    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    w = exp_Tb / exp_Tb.sum()

    log_sum_exp = Tb_max + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp

    c = fast_cross_correlate(a, w)
    w_dot_b = np.dot(w, b)
    grad = 4.0 * n * a * (c - w_dot_b)

    return obj, grad

def subgradient_mirror_descent(a0, time_limit, n_active_frac=0.05):
    """Mirror descent on the simplex using subgradient of max(b)."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n

    t0 = time.monotonic()
    lr = 0.5
    no_improve = 0

    for step in range(200000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b - n_active_frac * max_b
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Subgradient: average of gradients at active peaks
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        # Backtracking line search
        alpha = lr
        improved = False
        for _ in range(8):
            z_new = z - alpha * g
            a_new = np.exp(z_new - z_new.max())
            a_new = a_new / a_new.sum()
            score_new = 2.0 * n * fast_convolve(a_new).max()
            if score_new < best_score - 1e-14:
                best_score = score_new
                best_a = a_new.copy()
                a = a_new
                z = z_new
                no_improve = 0
                lr = min(alpha * 1.05, 5.0)
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            lr = max(lr * 0.7, 0.01)

def lp_best_response_step(a_fixed, M_target=None):
    """
    Solve LP: min M s.t. conv(a_fixed, b)_k <= M for all k, sum(b)=1, b>=0.
    Returns (b_opt, M_opt).
    """
    n = len(a_fixed)
    size = 2 * n - 1

    # Build the convolution matrix T where (T@b)[k] = conv(a_fixed, b)[k]
    # T[k, j] = a_fixed[k-j] if 0 <= k-j < n, else 0
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed

    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    if M_target is not None:
        constraints.append(M_var <= M_target)

    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solver = None
    for s in [cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solver = s
                break
        except Exception:
            continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s_sum = b_val.sum()
    if s_sum < 1e-15:
        return None, None
    b_val = b_val / s_sum
    return b_val, float(M_var.value)


def nesterov_lp_v2(a0, time_limit, n_max_iters=50000):

    best_a = a.copy()

    t0 = time.monotonic()
    stall = 0
    prev_best = best_score

        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = b_new.copy()
            sy = b_new.sum()
        y_new = y_new / sy

        # Update a (averaged iterate)
        t_plus = it + 1.0
        beta = (1.0 + np.sqrt(4.0 * t_plus + 1.0)) / 4.0
        a = (1.0 - beta) * a + beta * y_new
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y_new.copy()
        else:
            a = a / sa
        y = y_new

        # Check scores
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score - 1e-12:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        if score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()
            stall = 0

        if (it + 1) % 100 == 0:
            if best_score < prev_best - 1e-8:
                stall = 0
            else:
                stall += 1
            prev_best = best_score

            if stall >= 10:
                rng = np.random.default_rng(it * 17 + 41)
                noise = rng.random(n) * 0.02
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall = 0

            if (it + 1) % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NesterovLP it={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 10)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif choice == 1:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 2:
        a = np.sin(np.pi * x) ** (0.7 + rng.random() * 0.6)
    elif choice == 3:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * rng.integers(1, 5) * x)
        a = np.maximum(a, 0) ** 1.5
    elif choice == 4:
        a = np.exp(-3 * (x - rng.uniform(0.3, 0.7)) ** 2)
    elif choice == 5:
        a = 0.5 + 0.3 * np.cos(6 * np.pi * x) + 0.2 * np.cos(10 * np.pi * x)
        a = np.maximum(a, 0)
    elif choice == 6:
        a = (1 + np.cos(2 * np.pi * rng.integers(2, 8) * x)) ** 2
    elif choice == 7:
        a = np.exp(-2 * x) + np.exp(-2 * (1 - x))
    elif choice == 8:
        a = np.abs(np.sin(np.pi * x * rng.uniform(1, 4)))
    else:
        a = np.ones(n) * (1 + 0.3 * np.cos(2 * np.pi * rng.random(3) * x))
        a = np.maximum(a, 0)

    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def optimize_at_length(a_init, time_budget, label=""):
    """
    Run multi-phase optimization at a fixed length.
    """
    n = len(a_init)
    best_a = np.maximum(a_init, 0).astype(np.float64)
    s = best_a.sum()
    if s < 1e-15:
        best_a = np.ones(n) / n
        s = 1.0
    best_a = best_a / s
    best_score = 2.0 * n * fast_convolve(best_a).max()
    t0 = time.monotonic()

    def try_update(a, src):
        nonlocal best_a, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        ss = a.sum()
        if ss < 1e-10:
            return
        a = a / ss
        sc = 2.0 * n * fast_convolve(a).max()
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t0
            print(f"    NEW BEST [{src}]: {sc:.6f} (n={n}, t={elapsed:.0f}s)")

    # Phase 1: LogSumExp L-BFGS (primary - most effective for beating 2)
    rem = time_budget - (time.monotonic() - t0)
    t1 = min(rem * 0.50, 400.0)
    if t1 > 20:
        print(f"  [{label}] Phase 1: LSE L-BFGS ({t1:.0f}s)")
        a_out, _ = lbgfs_logsumexp(best_a, t1, T_start=500.0, T_end=0.005, n_stages=30)
        try_update(a_out, f"lse_{label}")

    # Phase 2: Peak shaving (targeted local search)
    rem = time_budget - (time.monotonic() - t0)
    t2 = min(rem * 0.25, 150.0)
    if t2 > 10:
        print(f"  [{label}] Phase 2: Peak shaving ({t2:.0f}s)")
        a_out, _ = peak_shaving(best_a, t2, max_steps=30000)
        try_update(a_out, f"peak_{label}")

    # Phase 3: Second LSE pass from improved point
    rem = time_budget - (time.monotonic() - t0)
    t3 = min(rem * 0.15, 100.0)
    if t3 > 10:
        print(f"  [{label}] Phase 3: LSE refine ({t3:.0f}s)")
        a_out, _ = lbgfs_logsumexp(best_a, t3, T_start=50.0, T_end=0.003, n_stages=20)
        try_update(a_out, f"lse2_{label}")

    # Phase 4: Mirror descent (diversification)
    rem = time_budget - (time.monotonic() - t0)
    t4 = min(rem * 0.10, 60.0)
    if t4 > 5:
        print(f"  [{label}] Phase 4: Mirror descent ({t4:.0f}s)")
        a_out, _ = subgradient_mirror_descent(best_a, t4, n_active_frac=0.03)
        try_update(a_out, f"md_{label}")

    # Phase 5: Final peak shaving
    rem = time_budget - (time.monotonic() - t0)
    t5 = min(rem * 0.10, 60.0)
    if t5 > 5:
        print(f"  [{label}] Phase 5: Final peak shaving ({t5:.0f}s)")
        a_out, _ = peak_shaving(best_a, t5, max_steps=20000)
        try_update(a_out, f"peak2_{label}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function: multi-length, multi-phase optimization."""
    global time_limit_global
    rng = np.random.default_rng(seed)
    time_limit_global = time.monotonic() + max(1.0, float(budget_s) - 60.0)

    best_seq = None
    best_score = float('inf')

    # Try to get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    n_base = len(best_seq) if best_seq is not None else 1200

    if parent is None:
        n_base = 1200
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  GLOBAL NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # === Phase 1: Primary optimization at base length ===
    print(f"\n=== Phase 1: Optimize at n={n_base} (best={best_score:.6f}) ===")
    rem = time_limit_global - time.monotonic()
    t1 = min(rem * 0.45, 700.0)
    if t1 > 60:
        print(f"  Budget: {t1:.0f}s")
        a_out, _ = optimize_at_length(best_seq, t1, label=f"n{n_base}")
        try_update(a_out, f"opt_n{n_base}")

    # === Phase 2: Try nearby lengths ===
    print(f"\n=== Phase 2: Nearby lengths (best={best_score:.6f}) ===")
    # Try lengths around the base
    nearby = []
    for mult in [0.8, 0.9, 1.1, 1.25, 1.5]:
        ln = int(n_base * mult)
        if 200 <= ln <= 5000 and ln != n_base:
            nearby.append(ln)
    nearby = sorted(set(nearby))[:4]

    for ln in nearby:
        rem = time_limit_global - time.monotonic()
        if rem < 120:
            break
        t_alloc = min(rem * 0.20, 250.0)
        if t_alloc < 40:
            break
        print(f"  Trying n={ln}, budget={t_alloc:.0f}s")

        init_long = interpolate_sequence(best_seq, ln)
        sc_init = evaluate_safe(init_long)
        if sc_init < best_score:
            best_score = sc_init
            best_seq = init_long.copy()

        a_out, _ = optimize_at_length(init_long, t_alloc, label=f"n{ln}")
        try_update(a_out, f"opt_n{ln}")

    # === Phase 3: Multi-restart at best length ===
    print(f"\n=== Phase 3: Multi-restart at n={len(best_seq)} (best={best_score:.6f}) ===")
    rem = time_limit_global - time.monotonic()
    n_restarts = 3
    t_per_restart = rem * 0.15 / n_restarts

    for ri in range(n_restarts):
        rem = time_limit_global - time.monotonic()
        if rem < 60:
            break
        t_r = min(rem * 0.5 / (n_restarts - ri), 200.0)
        if t_r < 30:
            break

        # Diverse initialization: mix of best and random
        mix = rng.uniform(0.3, 0.7)
        a_rand = structured_init(len(best_seq), rng)
        a_init = mix * best_seq + (1 - mix) * a_rand
        a_init = np.maximum(a_init, 0)
        a_init = a_init / a_init.sum()

        print(f"  Restart {ri+1}/{n_restarts} (budget={t_r:.0f}s)")
        a_out, _ = optimize_at_length(a_init, t_r, label=f"r{ri+1}")
        try_update(a_out, f"restart_{ri+1}")

    # === Phase 4: Final intensive polish ===
    print(f"\n=== Phase 4: Final Polish (best={best_score:.6f}, n={len(best_seq)}) ===")
    rem = time_limit_global - time.monotonic()
    if rem > 30:
        # Very low temperature LSE
        t_lse = min(rem * 0.40, 200.0)
        if t_lse > 20:
            print(f"  Final LSE ({t_lse:.0f}s)")
            a_fin, _ = lbgfs_logsumexp(best_seq, t_lse, T_start=10.0, T_end=0.002, n_stages=20)
            try_update(a_fin, "final_lse")

        # Peak shaving
        rem = time_limit_global - time.monotonic()
        t_ps = min(rem * 0.30, 100.0)
        if t_ps > 10:
            print(f"  Final peak shaving ({t_ps:.0f}s)")
            a_fin2, _ = peak_shaving(best_seq, t_ps, max_steps=50000)
            try_update(a_fin2, "final_peak")

        # One more LSE
        rem = time_limit_global - time.monotonic()
        t_lse2 = min(rem * 0.30, 60.0)
        if t_lse2 > 10:
            a_fin3, _ = lbgfs_logsumexp(best_seq, t_lse2, T_start=5.0, T_end=0.002, n_stages=15)
            try_update(a_fin3, "final_lse2")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
