"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smooth subgradient descent (main workhorse)
2. LP Fictitious Play (periodic best-response pulls)
3. Exact coordinate descent with golden section (final polishing)
4. Multi-length basin hopping
5. Diverse initialization (asymmetric Gaussians, flat spectrum, random)

import numpy as np
from scipy.fft import rfft, irfft
import time

_LP_SOLVER = None

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * float(conv.max()) / (s * s)

def get_lp_solver():
    global _LP_SOLVER
    if _LP_SOLVER is not None:
        return _LP_SOLVER
    import cvxpy as cp
    for name in [cp.ECOS, cp.GLPK, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _LP_SOLVER = name
                return name
        except Exception:
            continue
    _LP_SOLVER = 'ECOS'
    return _LP_SOLVER

def solve_lp_best_response(a_fixed, solver=None):
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    # Build convolution matrix T (2n-1) x n
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j+n, j] = a_fixed
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in ['ECOS', 'GLPK', 'PDLP', 'SCS']:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    M_val = float(M_var.value)
    return b_val, M_val

def golden_section_search(f, lo, hi, max_iter=40):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-15:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5

def exact_coord_descent(a0, time_limit, verbose=False, label=""):
    """Exact coordinate descent with golden section search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue
            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            unaff_max = 0.0
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i == k_lo + np.arange(n_aff))  # check if 2i is in range
            # Actually, 2i is in [k_lo, k_hi] iff k_lo <= 2i <= k_hi
            has_quad = (k_lo <= 2 * i <= k_hi)
            quad_idx = 2 * i - k_lo if has_quad else -1

            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]

            def _score(x, _c=c_vals, _ap=a_partners, _qm=unaff_max,
                       _hq=has_quad, _qi=quad_idx, _cs=current_sum,
                       _v=v, _n=n):
                ns = _cs + (x - _v)
                if ns < 0.01:
                    return 1e10
                nv = _c + 2.0 * x * _ap
                if _hq:
                    nv = nv.copy()
                    nv[_qi] = _c[_qi] + x * x
                nm = max(_qm, float(nv.max()))
                return 2.0 * _n * nm / (ns * ns)

            upper = max(3.0 * v, 5.0 / n)
            best_x = golden_section_search(_score, 0.0, upper, max_iter=30)

            if abs(best_x - v) > 1e-13:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta
                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
        if verbose and sweeps % 5 == 0:
            print(f"    ECD {label} sweep={sweeps}: best={best_score:.8f}")

    return best_a, best_score, sweeps

def softmax_subgradient_step(a, t, lr):
    """One step of subgradient descent using log-sum-exp smoothing."""
    n = len(a)
    conv = fast_convolve(a)
    S = a.sum()
    
    # Log-sum-exp for numerical stability
    m = conv.max()
    M_soft = m + t * np.log(np.exp((conv - m) / t).sum())
    
    # Softmax weights
    w = np.exp((conv - m) / t)
    w /= w.sum()
    
    # Gradient computation
    # grad_p = (2n/S^2) * [sum_j a_j * w_{p+j} + w_{2p}*a_p - 2*M_soft*a_p/S]
    # The first term is a cross-correlation
    corr = np.zeros(n)
    for p in range(n):
        corr[p] = np.dot(a, w[p:p+n])
    
    diag_corr = np.zeros(n)
    for p in range(n):
        kp = 2 * p
        if kp < 2 * n - 1:
            diag_corr[p] = w[kp] * a[p]
    
    grad = (2.0 * n / (S * S)) * (corr + diag_corr - 2.0 * M_soft * a / S)
    
    return grad, M_soft, conv

def subgradient_descent(a0, time_limit, verbose=False, label=""):
    """Subgradient descent with adaptive step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    
    # Temperature schedule
    t_init = 0.1
    t_min = 0.005
    t_decay = 0.999
    
    # Learning rate schedule
    lr = 0.001
    lr_decay = 0.9995
    
    a_avg = a.copy()
    no_improve = 0
    t_temp = t_init
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        grad, M_soft, conv = softmax_subgradient_step(a, t_temp, lr)
        
        a_new = a - lr * grad
        # Project to simplex (non-negative, sum = 1)
        a_new = simplex_project(a_new)
        
        sc = compute_score(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a_avg = (iters * a_avg + a_new) / (iters + 1)
        sc_avg = compute_score(a_avg)
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = a_avg.copy()
        
        a = a_new
        
        # Adapt temperature and learning rate
        t_temp = max(t_min, t_temp * t_decay)
        if no_improve > 50:
            lr *= 0.8
            no_improve = 0
        if no_improve > 500:
            break
        
        if verbose and iters % 10 == 0:
            print(f"    SG {label} iter={iters}, t={t_temp:.4f}, lr={lr:.6f}: best={best_score:.8f}")
    
    return best_a, best_score

def simplex_project(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def init_asymmetric(n, rng, shift=0.3):
    t = np.arange(n, dtype=np.float64)
    center = n * shift
    width = n * rng.uniform(0.15, 0.4)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_flat_spectrum(n, rng):
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = rng.uniform(0.7, 1.3, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_random_smooth(n, rng):
    """Random smooth initialization using low-frequency sine mix."""
    a = np.zeros(n)
    for _ in range(rng.integers(3, 8)):
        freq = rng.uniform(0.1, 2.0)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.5, 1.0)
        a += amp * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

                best_score = score_new

                no_improve += 1

            a = (1.0 - alpha) * a + alpha * b
            a = np.maximum(a, 0)

                a = a / cs

            print(f"    FP n={n} it={iters}: best={best_score:.8f}")

            break

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

2. LP Fictitious Play with adaptive mixing (periodic best-response)

import numpy as np
from scipy.fft import rfft, irfft
import time

_LP_SOLVER = None


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * float(conv.max()) / (s * s)


def get_lp_solver():
    global _LP_SOLVER
    if _LP_SOLVER is not None:
        return _LP_SOLVER
    import cvxpy as cp
    for name in [cp.GLPK, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _LP_SOLVER = name
                return name
        except Exception:
            continue
    _LP_SOLVER = 'ECOS'
    return _LP_SOLVER


def solve_lp_best_response(a_fixed, solver=None):
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in ['ECOS', 'GLPK', 'PDLP', 'SCS']:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    M_val = float(M_var.value)
    return b_val, M_val


def golden_section_search_inline(f_eval, lo, hi, max_iter=35):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f_eval(c)
    fd = f_eval(d)
    for _ in range(max_iter):
        if hi - lo < 1e-14:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f_eval(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f_eval(d)
    return (lo + hi) * 0.5


def ecd_coordinate_score(x, c_vals, a_partners, unaff_max, has_quad, quad_idx, current_sum, v, n):
    ns = current_sum + (x - v)
    if ns < 0.01:
        return 1e10
    nv = c_vals + 2.0 * x * a_partners
    if has_quad:
        nv = nv.copy()
        nv[quad_idx] = c_vals[quad_idx] + x * x
    nm = max(unaff_max, float(nv.max()))
    return 2.0 * n * nm / (ns * ns)


def exact_coord_descent(a0, time_limit, verbose=False, label=""):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue
            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            unaff_max = 0.0
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (k_lo <= 2 * i <= k_hi)
            quad_idx = 2 * i - k_lo if has_quad else -1

            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]

            upper = max(3.0 * v, 8.0 / n)
            best_x = golden_section_search_inline(
                lambda x, _c=c_vals, _ap=a_partners, _qm=unaff_max,
                _hq=has_quad, _qi=quad_idx, _cs=current_sum, _v=v, _n=n:
                ecd_coordinate_score(x, _c, _ap, _qm, _hq, _qi, _cs, _v, _n),
                0.0, upper, max_iter=30
            )

            if abs(best_x - v) > 1e-13:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta
                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:

            if no_improve >= 3:
                break
        else:
            no_improve = 0
        if verbose and sweeps % 5 == 0:

    return best_a, best_score, sweeps


def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def sg_gradient(a, t_temp):
    n = len(a)
    conv = fast_convolve(a)
    S = a.sum()
    m = conv.max()
    M_soft = m + t_temp * np.log(np.exp((conv - m) / t_temp).sum())
    w = np.exp((conv - m) / t_temp)
    w /= w.sum()

    corr = np.empty(n)
    for p in range(n):
        corr[p] = np.dot(a, w[p:p + n])

    diag_corr = np.zeros(n)
    for p in range(n):
        kp = 2 * p
        if kp < 2 * n - 1:
            diag_corr[p] = w[kp] * a[p]

    grad = (2.0 * n / (S * S)) * (corr + diag_corr - 2.0 * M_soft * a / S)
    return grad, M_soft, conv

def init_asymmetric(n, rng, shift=0.3):
    t = np.arange(n, dtype=np.float64)
    center = n * shift
    width = n * rng.uniform(0.15, 0.35)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_dual_gaussian(n, rng):
    t = np.arange(n, dtype=np.float64)
    c1 = n * rng.uniform(0.2, 0.4)
    c2 = n * rng.uniform(0.6, 0.8)
    w1 = n * rng.uniform(0.08, 0.2)
    w2 = n * rng.uniform(0.08, 0.2)
    a = np.exp(-0.5 * ((t - c1) / w1) ** 2) + np.exp(-0.5 * ((t - c2) / w2) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_flat_spectrum(n, rng):
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = rng.uniform(0.6, 1.4, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_sine_mix(n, rng):
    a = np.zeros(n)
    for _ in range(rng.integers(3, 7)):
        freq = rng.uniform(0.1, 2.5)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.3, 1.0)
        a += amp * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

            t_alloc = min(60.0, (deadline - 500 - time.monotonic()) * 0.2)
            if t_alloc < 10:
                break
            a_cd, s_cd, _ = exact_coord_descent(a_current, t_alloc, label=f"p2_cd{cycle}")
            if s_cd < best_score:
                best_score = s_cd
                best_seq = a_cd.copy()
                a_current = a_cd.copy()
                print(f"  CYCLE {cycle} CD: NEW BEST {best_score:.8f}")

        elif step == 4:
            # Perturb and re-optimize to escape local minima
            n_w = len(a_current)
            perturbed = a_current + rng.standard_normal(n_w) * 0.015
            perturbed = np.maximum(perturbed, 0)
            perturbed = perturbed / perturbed.sum()
            t_alloc = min(40.0, (deadline - 500 - time.monotonic()) * 0.15)
            if t_alloc < 10:
                break
            a_p, s_p, _ = exact_coord_descent(perturbed, t_alloc, label=f"p2_pert{cycle}")
            if s_p < best_score:
                best_score = s_p
                best_seq = a_p.copy()
                a_current = a_p.copy()
                print(f"  CYCLE {cycle} PERT: NEW BEST {best_score:.8f}")
            else:
                a_current = best_seq.copy()

    print(f"\nAfter Phase 2: best={best_score:.8f}, n={len(best_seq)}")

    # Phase 3: Multi-length exploration
    print(f"\n=== Phase 3: Multi-length Exploration ===")
    for target_n in [1024, 1536, 2048, 2560, 3072]:
        if time.monotonic() > deadline - 350:
            break
        rem = deadline - time.monotonic() - 250
        if rem < 60:
            break
        if target_n == len(best_seq):
            continue

        a_interp = interpolate_seq(best_seq, target_n)
        t_sg = min(rem * 0.3, 50.0)
        a_new, s_new = subgradient_descent(a_interp, t_sg)

        t_cd = min(rem * 0.25, 40.0)
        a_new2, s_new2, _ = exact_coord_descent(a_new, t_cd)

        sc = compute_score(a_new2)
        if sc < best_score:
            best_score = sc
            best_seq = a_new2.copy()
            print(f"  NEW BEST at n={target_n}: {sc:.8f}")
        else:
            print(f"  n={target_n}: {sc:.8f} (no improvement)")

    print(f"\nAfter Phase 3: best={best_score:.8f}, n={len(best_seq)}")

    # Phase 4: Basin hopping
    print(f"\n=== Phase 4: Basin Hopping ===")
    n_work = len(best_seq)
    a_current = best_seq.copy()
    bh_count = 0

    while time.monotonic() < deadline - 150 and bh_count < 30:
        bh_count += 1
        pert_type = bh_count % 4

        if pert_type == 0:
            perturbed = a_current + rng.standard_normal(n_work) * 0.02
        elif pert_type == 1:
            shift = rng.integers(2, max(3, n_work // 12))
            perturbed = 0.65 * a_current + 0.35 * np.roll(a_current, shift)
        elif pert_type == 2:
            blen = max(4, rng.integers(n_work // 12, n_work // 5))
            i1 = rng.integers(0, n_work - blen)
            i2 = rng.integers(0, n_work - blen)
            perturbed = a_current.copy()
            if abs(i1 - i2) > blen:
                temp = perturbed[i1:i1 + blen].copy()
                perturbed[i1:i1 + blen] = perturbed[i2:i2 + blen]
                perturbed[i2:i2 + blen] = temp
        else:
            # Scale peaks down, valleys up
            conv_p = fast_convolve(a_current)
            peaks = conv_p > np.percentile(conv_p, 90)
            perturbed = a_current.copy()
            for k in range(len(peaks)):
                if peaks[k]:
                    i_start = max(0, k - n_work + 1)
                    i_end = min(n_work - 1, k)
                    perturbed[i_start:i_end + 1] *= 0.92

        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-10:
            continue
        perturbed = perturbed / ps

        t_opt = min((deadline - 150 - time.monotonic()) * 0.2, 45.0)
        if t_opt < 10:
            break
        a_opt, s_opt, _ = exact_coord_descent(perturbed, t_opt, label=f"bh{bh_count}")

        if s_opt < best_score - 1e-13:
            best_score = s_opt
            best_seq = a_opt.copy()
            a_current = a_opt.copy()
            print(f"  BH #{bh_count}: NEW BEST {best_score:.8f}")

    print(f"\nAfter Phase 4: best={best_score:.8f}, n={len(best_seq)}")

    # Phase 5: Final intensive polishing
    print(f"\n=== Phase 5: Final Polish ===")
    n_work = len(best_seq)
    rem = deadline - time.monotonic() - 10
    if rem > 15:
        t_cd = min(rem * 0.6, 120.0)
        print(f"  ECD final: budget={t_cd:.0f}s, n={n_work}")
        a_final, s_final, _ = exact_coord_descent(best_seq, t_cd, verbose=True, label="final")
        sc = float(evaluate_sequence(a_final.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_final.copy()
            print(f"  FINAL ECD: {sc:.8f}")

    # Last attempt
    rem = deadline - time.monotonic() - 3
    if rem > 5:
        a_last, s_last, _ = exact_coord_descent(best_seq, min(rem * 0.7, 20.0), label="last")
        sc = float(evaluate_sequence(a_last.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_last.copy()
            print(f"  LAST: {sc:.8f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'='*50}")
    return [float(x) for x in best_seq]
