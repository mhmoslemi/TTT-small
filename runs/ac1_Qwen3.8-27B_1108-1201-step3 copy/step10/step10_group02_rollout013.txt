Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based alternating

structure of the problem.

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def eval_score(a):
    """Evaluate the objective function."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        T[i:i+n, i] = a[i]
    return T

def solve_lp_best_response(a_fixed, solver='SCS', verbose=False):
    """
    Given fixed a (non-negative, sum=1), find b minimizing max_k convolve(a,b)[k]

    Returns (b, M) or (None, None) on failure.
    """
    n = len(a_fixed)
    size = 2 * n - 1
    
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    T = build_conv_matrix(a_fixed)
    
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    solvers_to_try = [solver] if solver else []
    for s in ['SCS', 'ECOS', 'CLARABEL', 'PDLP']:
        if s not in solvers_to_try:
            solvers_to_try.append(s)
    
    for sol in solvers_to_try:
        try:
            prob.solve(solver=sol)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_val = b_val.sum()
                if s_val < 1e-15:

def lp_alternating(a0, time_limit, solver='SCS', verbose=True):
    """
    LP alternating minimization with Nesterov acceleration.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()
    
    if verbose:
        print(f"  LP Alt start: score={best_score:.8f}, n={n}")
    
    # Pre-build nothing (T changes each iteration)
    it = 0
    stalled = 0
    prev_best = best_score
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Solve LP for best response to current a

        if b_new is None:
            break

        alpha = 2.0 / (it + 2)

        a_new = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
            s_new = 1
        a_new = a_new / s_new

        if score < best_score:

            stalled += 1

    if verbose:
        print(f"  LP Alt done: {it} iters, best={best_score:.8f}, time={time.monotonic()-t0:.0f}s")

def subgradient_polish(a0, time_limit, eta0=0.05, verbose=True):
    """
    Subgradient optimization with averaging for final polish.
    Uses the fact that the subgradient of max_k (a*a)[k] at a is
    a "shifted a" (averaged over active peaks).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()
    
    if verbose:
        print(f"  Subgrad start: score={best_score:.8f}")
    
    # Maintaining running average for convergence
    a_avg = a.copy()
    it = 0
    eta = eta0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        eta = eta0 / (1 + 0.01 * it)
        
        # Compute convolution
        b = fast_convolve(a)
        max_b = b.max()
        
        # Find active peaks (within 1% of max)
        threshold = max_b * 0.99
        active = np.where(b >= threshold)[0]
        
        # Compute subgradient: average of shifted-a over active peaks
        grad = np.zeros(n)
        for k in active:
            # (a*a)[k] = sum_i a[i] * a[k-i]
            # grad_a[i] = a[k-i] for valid k-i
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            grad[i_min:i_max+1] += a[k - i_max:k - i_min][::-1]
        
        grad /= max(1, len(active))
        
        # Mirror descent update (exponential weights on simplex)
        log_a = np.log(np.maximum(a, 1e-300))
        log_a_new = log_a - eta * grad
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()
        
        # Running average
        a_avg = ((it + 1) * a_avg + (2.0 / (it + 2)) * a_new) / (it + 1 + 2.0 / (it + 2))
        a_avg = np.maximum(a_avg, 0)
        a_avg = a_avg / a_avg.sum()
        
        a = a_new
        
        # Evaluate both a and a_avg
        score_a = eval_score(a)
        score_avg = eval_score(a_avg)
        
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_avg < best_score:
            best_score = score_avg
            best_a = a_avg.copy()
        
        if verbose and (it <= 5 or it % 500 == 0):
            elapsed = time.monotonic() - t0
            print(f"    Subgrad iter {it} ({elapsed:.0f}s): score={score_a:.8f}, avg={score_avg:.8f}, best={best_score:.8f}")
    
    if verbose:
        print(f"  Subgrad done: {it} iters, best={best_score:.8f}")
    
    return best_a, best_score

    s = a.sum()

        a = a / s

    best_score = eval_score(a)

        print(f"  CoordDesc start: score={best_score:.8f}")
    
    it = 0
    stalled = 0

        it += 1
        b = fast_convolve(a)
        max_b = b.max()
        
        # Find peaks
        threshold = max_b * 0.9999
        peaks = np.where(b >= threshold)[0]
        
        improved = False

            # Try reducing the top contributors

                    # Redistribute to other elements proportional to their values

                    mask[pos] = False
                    others = a_trial[mask]
                    others_sum = others.sum()

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based alternating minimization
with Nesterov acceleration, subgradient polishing with running averages, and
targeted coordinate descent. Key innovation: uses scipy HiGHS for fast LP solves
and exploits the convolutive matrix structure for efficient best-response computation.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def eval_score(a):
    """Evaluate the objective function locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for i in range(n):
        T[i:i+n, i] = a[i]
    return T


def solve_lp_scipy(a_fixed):
    """
    Solve LP using scipy linprog (HiGHS):
    minimize M
    s.t. T @ b <= M  (2n-1 constraints)
         sum(b) = 1
         b >= 0
    Variables: [b_0, ..., b_{n-1}, M]
    """
    n = len(a_fixed)
    size = 2 * n - 1
    T = build_conv_matrix(a_fixed)
    
    # Decision variables: x = [b_0,...,b_{n-1}, M]
    # Objective: minimize M = x[n]
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Inequality constraints: T @ b - M <= 0
    # i.e., [T, -1] @ x <= 0
    A_ub = np.zeros((size, n + 1))
    A_ub[:, :n] = T
    A_ub[:, n] = -1.0
    b_ub = np.zeros(size)
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: b_i >= 0, M unbounded below (but will be >= 0 at optimum)
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 120})
    
    if res.status == 0:
        b_val = res.x[:n]
        b_val = np.maximum(b_val, 0)
        s = b_val.sum()
        if s < 1e-15:
            return None, None
        b_val = b_val / s
        return b_val, float(res.x[n])
    return None, None


def solve_lp_cvxpy(a_fixed, solver='SCS'):
    """Solve LP using cvxpy."""
    n = len(a_fixed)
    size = 2 * n - 1
    
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    T = build_conv_matrix(a_fixed)
    
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    solvers_to_try = [solver, 'SCS', 'ECOS', 'CLARABEL', 'PDLP']
    seen = set()
    for sol in solvers_to_try:
        if sol in seen:
            continue
        seen.add(sol)
        try:
            prob.solve(solver=sol)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s_val = b_val.sum()
                if s_val < 1e-15:
                    continue
                b_val = b_val / s_val
                return b_val, float(M_var.value)
        except Exception:
            continue
    
    return None, None


def solve_best_response(a_fixed, method='scipy'):
    """Solve the LP best-response problem with fallback."""
    if method == 'scipy':
        result = solve_lp_scipy(a_fixed)
        if result[0] is not None:
            return result
        # Fallback to cvxpy
        return solve_lp_cvxpy(a_fixed)
    else:
        result = solve_lp_cvxpy(a_fixed)
        if result[0] is not None:
            return result
        return solve_lp_scipy(a_fixed)

        print(f"  LP Alt start: score={best_score:.8f}, n={n}, method={method}")
    
    it = 0
    single_times = []
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        step_t0 = time.monotonic()
        
        # Solve LP for best response to y (Nesterov point)
        b_new, M_val = solve_best_response(y, method=method)
        if b_new is None:
            if verbose:
                print(f"    LP failed at iter {it}, switching method")
            method = 'cvxpy' if method == 'scipy' else 'scipy'
            continue
        
        step_time = time.monotonic() - step_t0
        single_times.append(step_time)
        if len(single_times) > 50:
            single_times.pop(0)
        
        # Nesterov update
        alpha = 2.0 / (it + 2)
        
        # Update y (extrapolated point)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = b_new.copy()
            sy = 1.0
        y = y / sy

        weight_a = it + 1

def subgradient_polish(a0, time_limit, eta0=0.05, verbose=True):
    """
    Subgradient optimization with Nesterov acceleration and running averages.
    The subgradient of max_k (a*a)[k] is a weighted average of shifted copies of a.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    y = a.copy()  # Nesterov extrapolated point
    a_avg = a.copy()  # Running average
    
    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()
    
    if verbose:
        print(f"  Subgrad start: score={best_score:.8f}, eta0={eta0}")
    
    it = 0
    stalled = 0
    prev_best = best_score
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute gradient at Nesterov point y
        b = fast_convolve(y)
        max_b = b.max()
        
        # Find active peaks (within 0.5% of max)
        threshold = max_b * 0.995
        active = np.where(b >= threshold)[0]
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in active:
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            idx = np.arange(i_min, i_max + 1)
            grad[idx] += y[k - idx]
        
        grad /= max(1, len(active))
        
        # Mirror descent step: a_new[i] proportional to a[i] * exp(-eta * grad[i])
        eta = eta0 / np.sqrt(1 + it / 100)
        log_a = np.log(np.maximum(a, 1e-300))
        log_a_new = log_a - eta * (grad - grad.min())
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()
        
        # Nesterov update for y
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * a_new
        y = np.maximum(y, 0)
        y = y / y.sum()
        
        # Running average (Nesterov-type)
        w = alpha * (it + 2)
        a_avg = ((it + 1) * a_avg + w * a_new) / (it + 1 + w)
        a_avg = np.maximum(a_avg, 0)
        a_avg = a_avg / a_avg.sum()
        
        a = a_new
        
        # Evaluate
        score_a = eval_score(a)
        score_avg = eval_score(a_avg)
        score_y = eval_score(y)
        
        current_best = min(score_a, score_avg, score_y)
        if current_best < best_score:
            if score_avg <= score_a and score_avg <= score_y:
                best_a = a_avg.copy()
            elif score_y <= score_a:
                best_a = y.copy()
            else:
                best_a = a.copy()
            best_score = current_best
            stalled = 0
        else:
            stalled += 1
        
        if verbose and (it <= 5 or it % 200 == 0):
            elapsed = time.monotonic() - t0
            print(f"    it={it} ({elapsed:.0f}s): a={score_a:.8f}, "
                  f"avg={score_avg:.8f}, y={score_y:.8f}, best={best_score:.8f}")
        
        if stalled > 2000:
            # Add small noise to escape
            rng = np.random.default_rng(it)
            a = a + rng.standard_normal(n) * 0.01
            a = np.maximum(a, 0)
            a = a / a.sum()
            y = a.copy()
            stalled = 0
        
    if verbose:
        print(f"  Subgrad done: {it} iters, best={best_score:.8f}")
    
    return best_a, best_score


def coordinate_polish(a0, time_limit, verbose=True):
    """
    Targeted coordinate descent: reduce elements that contribute most to
    the maximum convolution peak, redistributing mass to others.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()
    
    if verbose:
        print(f"  CoordDescent start: score={best_score:.8f}")
    
    it = 0
    stalled = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(a)
        max_b = b.max()
        
        # Find peaks within 0.01% of max
        threshold = max_b * 0.9999
        peaks = np.where(b >= threshold)[0]
        
        improved = False
        for k in peaks:
            if improved:
                break
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            idx = np.arange(i_min, i_max + 1)
            contribs = a[idx] * a[k - idx]
            top_m = min(6, len(idx))
            top_positions = idx[np.argsort(contribs)[::-1][:top_m]]
            
            for pos in top_positions:
                if a[pos] < 1e-10:
                    continue
                for factor in [0.9995, 0.999, 0.998, 0.995, 0.99, 0.98]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] - a_trial[pos]
                    if diff < 1e-15:
                        break
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] = others + diff * others / others_sum
                    a_trial = a_trial / a_trial.sum()
                    
                    sc = eval_score(a_trial)
                    if sc < best_score - 1e-13:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            stalled += 1
            if stalled > 300:
                break
        else:
            stalled = 0
        
        if verbose and (it <= 5 or it % 500 == 0):
            elapsed = time.monotonic() - t0
            print(f"    CD it={it} ({elapsed:.0f}s): best={best_score:.8f}, stalled={stalled}")
    
    if verbose:
        print(f"  CoordDescent done: {it} iters, best={best_score:.8f}")
    
    return best_a, best_score


def two_sided_lp_alternating(a0, time_limit, verbose=True):
    """
    Two-sided LP alternating: alternate between solving LP for a and LP for b.
    Both players get exact best responses.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    b = a.copy()
    
    best_a = a.copy()
    best_b = b.copy()
    best_score = eval_score(a)
    t0 = time.monotonic()
    
    if verbose:
        print(f"  2Sided LP start: score={best_score:.8f}")
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Solve LP: minimize max_k (a*b)[k] over b, given a
        b_new, M1 = solve_best_response(a, method='scipy')
        if b_new is None:
            break
        
        # Solve LP: minimize max_k (b_new*a)[k] over a, given b_new
        a_new, M2 = solve_best_response(b_new, method='scipy')
        if a_new is None:
            a_new = b_new.copy()
            M2 = M1
        
        # Nesterov update
        alpha = 2.0 / (it + 2)
        a = (1.0 - alpha) * a + alpha * a_new
        a = np.maximum(a, 0)
        a = a / a.sum()
        b = (1.0 - alpha) * b + alpha * b_new
        b = np.maximum(b, 0)
        b = b / b.sum()
        
        # Evaluate
        score_a = eval_score(a)
        score_b = eval_score(b)
        
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_b < best_score:
            best_score = score_b
            best_b = b.copy()
        
        if verbose and (it <= 5 or it % 20 == 0):
            elapsed = time.monotonic() - t0
            print(f"    2Sided it={it} ({elapsed:.0f}s): "
                  f"a={score_a:.8f}, b={score_b:.8f}, best={best_score:.8f}")
        
        # Check if a and b have converged (close to each other)
        if it % 10 == 0:
            diff = np.abs(a - b).max()
            if diff < 1e-6:
                if verbose:
                    print(f"    Converged: ||a-b||_inf = {diff:.2e}")
                break
    
    if verbose:
        print(f"  2Sided LP done: {it} iters, best={best_score:.8f}")
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Phase 1: One-sided LP alternating with Nesterov acceleration (scipy HiGHS)
    Phase 2: Subgradient polish with running averages
    Phase 3: Targeted coordinate descent
    Phase 4: Short two-sided LP for final refinement
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    
    # Load parent sequence
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            a = p / p.sum()
            n = len(a)
            parent_score = eval_score(a)
            print(f"Parent: n={n}, score={parent_score:.8f}")
        else:
            raise ValueError("Invalid parent")
    except Exception:
        n = 1200
        a = np.ones(n) / n
        parent_score = 2.0
        print(f"Default init: n={n}, score={parent_score:.8f}")
    
    best_a = a.copy()
    best_score = eval_score(a)
    
    def try_update(a_trial, label):
        nonlocal best_a, best_score
        a_trial = np.maximum(np.asarray(a_trial, dtype=np.float64), 0)
        s = a_trial.sum()
        if s < 0.01:
            return
        a_trial = a_trial / s
        sc = eval_score(a_trial)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_trial.copy()
            print(f"  *** NEW BEST [{label}]: {best_score:.8f} ***")
    
    # === Phase 1: One-sided LP Alternating (main workhorse) ===
    print(f"\n{'='*60}")
    print(f"Phase 1: One-sided LP Alternating (score={best_score:.8f})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.50, 850.0)
    if t_alloc > 60:
        print(f"  Time budget: {t_alloc:.0f}s")
        a_lp, sc = lp_alternating(best_a, t_alloc, verbose=True, method='scipy')
        try_update(a_lp, "LP_one_sided")
    
    # === Phase 2: Subgradient polish ===
    print(f"\n{'='*60}")
    print(f"Phase 2: Subgradient Polish (score={best_score:.8f})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 350.0)
    if t_alloc > 30:
        print(f"  Time budget: {t_alloc:.0f}s")
        a_sg, sc = subgradient_polish(best_a, t_alloc, eta0=0.04, verbose=True)
        try_update(a_sg, "subgradient")
    
    # === Phase 3: Coordinate descent ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Coordinate Descent (score={best_score:.8f})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 15:
        print(f"  Time budget: {t_alloc:.0f}s")
        a_cd, sc = coordinate_polish(best_a, t_alloc, verbose=True)
        try_update(a_cd, "coord_descent")
    
    # === Phase 4: Two-sided LP (symmetric refinement) ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Two-sided LP (score={best_score:.8f})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.07, 100.0)
    if t_alloc > 20:
        print(f"  Time budget: {t_alloc:.0f}s")
        a_2s, sc = two_sided_lp_alternating(best_a, t_alloc, verbose=True)
        try_update(a_2s, "two_sided_LP")
    
    # === Phase 5: Final subgradient (very small eta) ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Final Subgradient (score={best_score:.8f})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.03, 40.0)
    if t_alloc > 10:
        a_fin, sc = subgradient_polish(best_a, t_alloc, eta0=0.01, verbose=False)
        try_update(a_fin, "final_subgrad")
    
    # Final evaluation
    fs = eval_score(best_a)
    if fs < best_score:
        best_score = fs
    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_a)})")
    print(f"{'='*60}")
    
    return [float(max(0.0, min(1000.0, x))) for x in best_a]
