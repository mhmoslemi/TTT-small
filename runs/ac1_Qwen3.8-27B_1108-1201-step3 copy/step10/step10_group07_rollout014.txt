"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based best-response alternating minimization with adaptive damping
2. L-BFGS on smooth (exponential) approximation of the max convolution
3. Pairwise and single-coordinate descent for local polishing
4. Multi-scale exploration (n=1024, 2048) to find better basins
5. Temperature annealing for the smooth optimization phase
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def score_of(a):
    """Compute 2*n*max(conv)/(sum)^2 quickly."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def find_lp_solver():
    """Find the fastest working LP solver."""
    cached = getattr(find_lp_solver, '_cached', None)
    if cached is not None:
        return cached
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                find_lp_solver._cached = solver_name
                return solver_name
        except Exception:
            continue
    find_lp_solver._cached = cp.ECOS
    return cp.ECOS


def build_conv_matrix(a):
    """Build the convolution matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, solver):
    """
    Solve LP: min M s.t. conv(a,b) <= M, sum(b)=1, b>=0.
    Returns (b_opt, M_opt, dual_info).
    """
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    ones = np.ones(2 * n - 1)
    conv_constraints = T @ b_var <= M_var * ones
    constraints = [
        conv_constraints,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, max_time=30)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_time=30)
        except Exception:
            return a.copy(), float('inf'), None

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float('inf'), None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return a.copy(), float('inf'), None
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else float('inf')
    return b_val, M_val, None


def lp_alternating(a0, time_limit, solver, verbose=True):
    """
    LP-based best-response alternating minimization with adaptive damping.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    stagnation = 0
    prev_score = best_score

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2.0:
            break

        b_new, M_lp, _ = lp_best_response(a, solver)
        if not np.isfinite(M_lp):
            break

        score_b = 2.0 * n * fast_convolve(b_new).max()
        if score_b < best_score - 1e-12:
            best_score = score_b
            best_a = b_new.copy()

        if it <= 10:
            alpha = 1.0
        elif it <= 30:
            alpha = 0.8
        elif it <= 60:
            alpha = 0.6
        else:
            alpha = max(0.2, 8.0 / (it + 5.0))

        a_new = (1 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a
        else:
            a_new = a_new / s_new

        score_a = 2.0 * n * fast_convolve(a_new).max()
        if score_a < best_score - 1e-12:
            best_score = score_a
            best_a = a_new.copy()

        if abs(score_a - prev_score) < 1e-10:
            stagnation += 1
        else:
            stagnation = max(0, stagnation - 1)
        prev_score = score_a

        if stagnation > 40:
            noise = 0.005 / np.sqrt(n)
            a_pert = a_new + np.random.default_rng(it).standard_normal(n) * noise
            a_pert = np.maximum(a_pert, 0)
            sp = a_pert.sum()
            if sp > 1e-10:
                a_pert = a_pert / sp
                score_pert = 2.0 * n * fast_convolve(a_pert).max()
                if score_pert < best_score * 1.005:
                    a_new = a_pert
            stagnation = 0

        a = a_new

        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP it={it} ({elapsed:.0f}s): best={best_score:.8f} cur={score_a:.8f}")

    return best_a, best_score


def smooth_obj_and_grad(x, T, n):
    """
    Compute smoothed objective sum(exp(conv/T)) and gradient in softmax params.
    x is the parameter vector, a = softmax(x).
    """
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    s = a.sum()
    a = a / s

    conv = fast_convolve(a)
    conv_shifted = conv - conv.max()
    w = np.exp(conv_shifted / T)

    # grad_a[i] = (2/T) * sum_{k=i}^{i+n-1} w[k] * a[k-i]
    # = (2/T) * conv(w, reverse(a))[i + n - 1]
    v = a[::-1]
    total_size = len(w) + n - 1
    fft_size = 1 << (total_size - 1).bit_length()
    W = rfft(w, fft_size)
    V = rfft(v, fft_size)
    conv_wa = irfft(W * V, fft_size)[:total_size]
    grad_a = (2.0 / T) * conv_wa[n - 1:n - 1 + n]

    # grad_x[j] = a[j] * (grad_a[j] - dot(grad_a, a))
    dot_ga_a = np.dot(grad_a, a)
    grad_x = a * (grad_a - dot_ga_a)

    obj = np.sum(w)
    return obj, grad_x


def lbfgs_smooth(a0, time_limit, T_init=None, T_final=None, verbose=True):
    """
    L-BFGS optimization of the smooth approximation using softmax parameterization.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    # Initialize x from a (log-softmax)
    a_safe = np.maximum(a, 1e-30)
    x = np.log(a_safe)
    x = x - x.mean()  # Center for numerical stability

    if T_init is None:
        conv0 = fast_convolve(a)
        T_init = 0.1 * conv0.max()
    if T_final is None:
        T_final = 0.02 * T_init

    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    iteration = [0]
    T_current = [T_init]

    # Anneal temperature over the optimization
    def anneal_T(frac_done):
        # Exponential annealing
        return T_init * (T_final / T_init) ** frac_done

    def fun_and_grad(x_eval):
        iteration[0] += 1
        elapsed = time.monotonic() - t0
        frac = min(elapsed / time_limit, 1.0)
        T_current[0] = anneal_T(frac)
        return smooth_obj_and_grad(x_eval, T_current[0], n)

    # Run L-BFGS with time limit
    result = minimize(
        fun_and_grad, x,
        method='L-BFGS-B',
        options={
            'maxiter': 100000,
            'maxfun': 100000,
            'gtol': 1e-8,
            'ftol': 1e-12,
            'maxls': 20
        }
    )

    # Get final solution
    x_final = result.x
    x_shifted = x_final - x_final.max()
    a_final = np.exp(x_shifted)
    s_final = a_final.sum()
    a_final = a_final / s_final
    a_final = np.maximum(a_final, 0)

    score_final = score_of(a_final)
    if score_final < best_score:
        best_score = score_final
        best_a = a_final.copy()

    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    L-BFGS ({elapsed:.0f}s): {iteration[0]} iters, best={best_score:.8f}, "
              f"final={score_final:.8f}, status={result.status}, T_end={T_current[0]:.6f}")

    return best_a, best_score


def two_pass_lbfgs(a0, time_limit, verbose=True):
    """
    Two-pass L-BFGS: first with large T (explore), then small T (exploit).
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv0 = fast_convolve(a)
    M0 = conv0.max()

    t0 = time.monotonic()
    # Phase 1: large T (explore all peaks)
    t1 = min(time_limit * 0.4, time_limit - 5)
    a1, s1 = lbfgs_smooth(a, t1, T_init=0.3 * M0, T_final=0.1 * M0, verbose=verbose)

    elapsed1 = time.monotonic() - t0
    remaining = time_limit - elapsed1

    # Phase 2: small T (focus on max peak)
    if remaining > 5:
        conv1 = fast_convolve(a1)
        M1 = conv1.max()
        a2, s2 = lbfgs_smooth(a1, remaining, T_init=0.1 * M1, T_final=0.01 * M1, verbose=verbose)

        if s2 < s1:
            a1 = a2
            s1 = s2

    if verbose:
        total_elapsed = time.monotonic() - t0
        print(f"    TwoPass L-BFGS ({total_elapsed:.0f}s): best={s1:.8f}")

    return a1, s1


def coord_descent_fine(a0, time_limit, verbose=True, label=""):
    """
    Fine coordinate descent with adaptive steps.
    Uses incremental convolution updates.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-18:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            a_shifted = a[k_start - i:k_end - i + 1]
            conv_affected = conv[k_start:k_end + 1]

            best_delta = 0.0
            best_new_M = M

            # Adaptive factors
            base = max(a[i], 1e-10)
            factors = [-0.5, -0.3, -0.1, -0.05, -0.01, 0.01, 0.05, 0.1, 0.3, 0.5]

            for factor in factors:
                delta = base * factor
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = conv_affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-15:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-15:
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 50:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD[{label}] sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def pairwise_cd(a0, time_limit, verbose=True, label=""):
    """
    Pairwise coordinate descent: transfer mass between pairs of coordinates.
    Focuses on pairs that affect the top peaks.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    rounds = 0
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit:
        rounds += 1
        improved = False

        # Find top peaks
        K = min(10, 2 * n - 1)
        peak_idx = np.argpartition(conv, -K)[-K:]
        peak_idx = peak_idx[np.argsort(conv[peak_idx])[::-1]]

        # For each peak, find contributing coordinates
        candidate_pairs = set()
        for k_star in peak_idx:
            k_star = int(k_star)
            if conv[k_star] < M * 0.9:
                break
            i_start = max(0, k_star - n + 1)
            i_end = min(n - 1, k_star)
            # Top contributing coordinates
            contributions = []
            for i in range(i_start, i_end + 1):
                j = k_star - i
                contributions.append((a[i] * a[j], i, j))
            contributions.sort(reverse=True)
            for contrib, i, j in contributions[:5]:
                if contrib > 0:
                    candidate_pairs.add((min(i, j), max(i, j)))

        # Also add some random pairs
        for _ in range(5):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                candidate_pairs.add((min(i, j), max(i, j)))

        for (i, j) in list(candidate_pairs)[:30]:
            if a[i] < 1e-15 or a[j] < 1e-15:
                continue

            # Linear change: delta transfers from i to j
            # conv[k] changes by: 2*delta*(a[k-j] - a[k-i]) + quadratic
            k_i_start = i
            k_i_end = min(i + n - 1, 2 * n - 2)
            k_j_start = j
            k_j_end = min(j + n - 1, 2 * n - 2)

            # Affected range is union of [k_i_start, k_i_end] and [k_j_start, k_j_end]
            aff_start = min(k_i_start, k_j_start)
            aff_end = max(k_i_end, k_j_end)

            if aff_start > 0:
                unaff_left = conv[:aff_start].max()
            else:
                unaff_left = -1e30
            if aff_end < 2 * n - 2:
                unaff_right = conv[aff_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            best_delta = 0.0
            best_new_M = M

            # Try several delta values
            base = min(a[i], a[j])
            for factor in [-0.3, -0.15, -0.05, 0.05, 0.15, 0.3]:
                delta = base * factor
                if delta == 0:
                    continue

                # Compute new conv in affected range
                new_conv_aff = conv[aff_start:aff_end + 1].copy()

                # Effect of changing a[i] by -delta
                for k in range(k_i_start, k_i_end + 1):
                    idx = k - aff_start
                    ki = k - i
                    if 0 <= ki < n:
                        new_conv_aff[idx] += 2.0 * (-delta) * a[ki]
                # Fix quadratic at k=2i
                if 2 * i >= k_i_start and 2 * i <= k_i_end:
                    idx = 2 * i - aff_start
                    new_conv_aff[idx] = conv[2 * i] + 2.0 * (-delta) * a[i] + delta * delta

                # Effect of changing a[j] by +delta
                for k in range(k_j_start, k_j_end + 1):
                    idx = k - aff_start
                    kj = k - j
                    if 0 <= kj < n:
                        new_conv_aff[idx] += 2.0 * delta * a[kj]
                # Fix quadratic at k=2j
                if 2 * j >= k_j_start and 2 * j <= k_j_end:
                    idx = 2 * j - aff_start
                    new_conv_aff[idx] = conv[2 * j] + 2.0 * delta * a[j] + delta * delta

                new_M = max(unaff_max, float(new_conv_aff.max()))
                if new_M < best_new_M - 1e-15:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-15:
                # Apply the change
                a[i] = max(a[i] - best_delta, 0.0)
                a[j] = max(a[j] + best_delta, 0.0)

                # Update conv
                # Redo: 
                # Change a[i] by -best_delta, a[j] by +best_delta
                for k in range(k_i_start, k_i_end + 1):
                    ki = k - i
                    if 0 <= ki < n:
                        conv[k] += 2.0 * (-best_delta) * a[ki]
                if 2 * i >= k_i_start and 2 * i <= k_i_end:
                    conv[2 * i] = conv[2 * i] - 2.0 * (-best_delta) * (a[i] + best_delta) + best_delta * best_delta
                    # Actually let me just recompute affected regions properly
                    # This is getting complex, let me just recompute conv for this pair

                # Simpler: just recompute the affected conv entries
                # Actually for correctness, let me just recompute full conv if n is small enough
                # For n <= 4096, this is fast enough with FFT
                if n <= 4096:
                    conv = fast_convolve(a)
                else:
                    # Manual update
                    pass

                M = float(conv.max())
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved and rounds > 30:
            break

        if verbose and rounds % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD[{label}] round={rounds} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def diverse_init(n, seed, init_type=0):
    """Generate diverse initializations."""
    rng = np.random.default_rng(seed)
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    if init_type == 0:
        a = np.maximum(1.0 - np.abs(t - center) / center, 0)
    elif init_type == 1:
        x = (t - center) / center
        a = 0.3 + 0.7 * x * x
    elif init_type == 2:
        x = (t - center) / center
        power = rng.uniform(1.0, 3.0)
        a = np.maximum(0.5 * (1 + np.cos(np.pi * x)), 0) ** power
    elif init_type == 3:
        c1 = n * rng.uniform(0.2, 0.35)
        c2 = n * rng.uniform(0.65, 0.8)
        w = n * rng.uniform(0.08, 0.2)
        a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    elif init_type == 4:
        period = int(rng.integers(3, 10))
        a = 0.5 + 0.4 * np.cos(2 * np.pi * t / period)
    elif init_type == 5:
        a = 1.0 - 0.5 * np.cos(4 * np.pi * t / n)
        a = np.maximum(a, 0.01)
    elif init_type == 6:
        # Gaussian
        w = n * rng.uniform(0.1, 0.3)
        a = np.exp(-0.5 * ((t - center) / w) ** 2)
    else:
        a = 0.5 + 0.4 * np.random.default_rng(seed + 1).random(n)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def propose_candidate():
    """
    Multi-phase optimization combining LP alternating, L-BFGS smooth
    optimization, and coordinate descent for final polishing.
    """
    deadline = time.monotonic() + 1770.0
    rng = np.random.default_rng(12345)

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is not None:
        n_main = len(parent)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    else:
        n_main = 1024
        best_seq = np.ones(n_main) / n_main
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default init: n={n_main}, score={best_score:.8f}")

    solver = find_lp_solver()
    print(f"LP solver: {solver}")

    def try_update(candidate, label):
        nonlocal best_seq, best_score
        candidate = np.maximum(candidate, 0.0)
        s = candidate.sum()
        if s < 0.01:
            return
        candidate = candidate / s
        try:
            sc = float(evaluate_sequence(candidate.tolist()))
        except Exception:
            sc = float('inf')
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = candidate.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(candidate)})")

    # === Phase 1: LP Alternating on main size (40% time) ===
    print(f"\n=== Phase 1: LP Alternating n={n_main} (best={best_score:.8f}) ===")
    t1_budget = (deadline - time.monotonic()) * 0.40
    t1_end = time.monotonic() + t1_budget
    print(f"  Budget: {t1_budget:.0f}s")

    a_lp, s_lp = lp_alternating(best_seq, t1_budget, solver, verbose=True)
    try_update(a_lp, "lp_parent")

    # === Phase 2: L-BFGS smooth optimization (20% time) ===
    print(f"\n=== Phase 2: L-BFGS Smooth Opt (best={best_score:.8f}) ===")
    t2_budget = (deadline - time.monotonic()) * 0.20
    t2_end = time.monotonic() + t2_budget
    print(f"  Budget: {t2_budget:.0f}s")

    # First pass with the LP result
    a_s1, s_s1 = two_pass_lbfgs(a_lp, min(t2_budget * 0.6, 300), verbose=True)
    try_update(a_s1, "lbfgs_lp")

    # Second pass from best so far
    if time.monotonic() < t2_end - 10:
        remaining2 = t2_end - time.monotonic()
        if remaining2 > 20:
            a_s2, s_s2 = two_pass_lbfgs(best_seq, remaining2, verbose=True)
            try_update(a_s2, "lbfgs_best")

    # === Phase 3: Try n=2048 (20% time) ===
    print(f"\n=== Phase 3: Try n=2048 (best={best_score:.8f}) ===")
    t3_budget = (deadline - time.monotonic()) * 0.20
    t3_end = time.monotonic() + t3_budget
    print(f"  Budget: {t3_budget:.0f}s")

    n2 = 2048
    if t3_budget > 30:
        # Initialize n2 from upsampling the best sequence or a fresh init
        init_n2 = diverse_init(n2, 777, 0)  # Triangle
        a_n2, s_n2 = lp_alternating(init_n2, min(t3_budget * 0.6, 300), solver, verbose=True)
        try_update(a_n2, "lp_n2048")

        if time.monotonic() < t3_end - 10:
            remaining3 = t3_end - time.monotonic()
            if remaining3 > 15:
                a_n2s, s_n2s = two_pass_lbfgs(a_n2, remaining3, verbose=True)
                try_update(a_n2s, "lbfgs_n2048")

    # === Phase 4: Coordinate descent polishing (15% time) ===
    print(f"\n=== Phase 4: CD Polishing (best={best_score:.8f}) ===")
    t4_budget = (deadline - time.monotonic()) * 0.15
    t4_end = time.monotonic() + t4_budget
    print(f"  Budget: {t4_budget:.0f}s")

    if t4_budget > 20:
        a_cd, s_cd = coord_descent_fine(best_seq, min(t4_budget * 0.7, 200), verbose=True, label="main")
        try_update(a_cd, "cd_main")

        if time.monotonic() < t4_end - 10:
            remaining4 = t4_end - time.monotonic()
            if remaining4 > 10:
                a_cd2, s_cd2 = coord_descent_fine(best_seq, remaining4, verbose=False, label="finer")
                try_update(a_cd2, "cd_finer")

    # === Phase 5: Final L-BFGS + CD on absolute best (5% time) ===
    print(f"\n=== Phase 5: Final Polish (best={best_score:.8f}) ===")
    t5_end = deadline - 30

    if time.monotonic() < t5_end - 10:
        remaining5 = t5_end - time.monotonic()
        if remaining5 > 10:
            a_fin, s_fin = two_pass_lbfgs(best_seq, remaining5 * 0.5, verbose=True)
            try_update(a_fin, "final_lbfgs")

    if time.monotonic() < t5_end - 5:
        remaining5b = t5_end - time.monotonic()
        if remaining5b > 5:
            a_fin2, s_fin2 = coord_descent_fine(best_seq, remaining5b, verbose=False, label="final")
            try_update(a_fin2, "final_cd")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
