import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def score_of_normalized(a):
    """Compute 2*n*max(conv)/(sum)^2 for normalized a (sum=1)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def simplex_project(v):
    n = len(v)
    if n == 0:
        return v.copy()
    v = v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                find_lp_solver._cached = solver_name
                return solver_name
        except Exception:

    find_lp_solver._cached = cp.ECOS
    return cp.ECOS


def lp_best_response(a, solver):
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    ones = np.ones(2 * n - 1)
    constraints = [
        T @ b_var <= M_var * ones,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return a.copy(), float('inf')
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float('inf')
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return a.copy(), float('inf')
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else float('inf')
    return b_val, M_val


def exact_coord_descent(a0, time_limit, verbose=True, label=""):
    """
    Coordinate descent where each step finds the exact optimal delta
    for that coordinate by 1D minimax.
    """

    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve_sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        
        for i in range(n):
            if a[i] < 1e-18:
                continue
            
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            
            # Unaffected max
            if k_start > 0:
                unaff_max = float(conv[:k_start].max())
            else:
                unaff_max = -np.inf
            if k_end < 2 * n - 2:
                right_max = float(conv[k_end + 1:].max())
                unaff_max = max(unaff_max, right_max)
            
            # Affected convolution values
            affected_conv = conv[k_start:k_end + 1].copy()
            # a[k-i] for k in [k_start, k_end]
            a_shifted = a[k_start - i:k_end - i + 1]
            
            # Find optimal delta
            # For k in [k_start, k_end], k != 2i:
            #   new_conv[k] = affected_conv[k - k_start] + 2*delta*a_shifted[k - k_start]
            # For k = 2i (if in range):
            #   new_conv[2i] = conv[2i] + 2*a[i]*delta + delta^2
            
            two_i = 2 * i
            two_i_in_range = (two_i >= k_start and two_i <= k_end)
            two_i_local = two_i - k_start if two_i_in_range else -1
            
            # Search range for delta
            delta_min = -a[i]  # Can't go below 0
            # For delta_max: we don't need to go too high
            # A reasonable upper bound: enough to push any affected value to 2*M
            delta_max = 2.0 * M / max(a_shifted.max(), 1e-15) + a[i]
            # Cap it
            delta_max = min(delta_max, 2.0 * a[i] + M)
            
            # Use ternary search on [delta_min, delta_max]
            # The function f(delta) = max(unaff_max, max_affected(delta)) is convex
            
            def obj_at_delta(d):
                new_affected = affected_conv + 2.0 * d * a_shifted
                if two_i_in_range:
                    new_affected[two_i_local] = conv[two_i] + 2.0 * a[i] * d + d * d
                new_M = max(unaff_max, float(new_affected.max()))
                return new_M
            
            # Ternary search
            lo = delta_min
            hi = delta_max
            for _ in range(50):

                    lo = m1

import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def score_of_normalized(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def simplex_project(v):
    n = len(v)
    if n == 0:
        return v.copy()
    v = v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def find_lp_solver():
    cached = getattr(find_lp_solver, '_cached', None)
    if cached is not None:
        return cached
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                find_lp_solver._cached = solver_name
                return solver_name
        except Exception:
            continue
    find_lp_solver._cached = cp.ECOS
    return cp.ECOS


def lp_best_response(a, solver):
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    ones = np.ones(2 * n - 1)
    constraints = [
        T @ b_var <= M_var * ones,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return a.copy(), float('inf')
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float('inf')
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return a.copy(), float('inf')
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else float('inf')
    return b_val, M_val


def gauss_bump_init(n, n_bumps, rng):
    a = np.zeros(n)
    for _ in range(n_bumps):
        center = rng.uniform(0.1, 0.9) * n
        width = rng.uniform(0.02, 0.15) * n
        amp = rng.uniform(0.3, 1.0)
        t = np.arange(n)
        a += amp * np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def diverse_init(n, rng, init_type):
    if init_type == 0:
        t = np.arange(n)

    elif init_type == 2:
        t = np.arange(n)

        power = rng.uniform(1.0, 3.0)
        a = np.maximum(0.5 * (1 + np.cos(np.pi * x)), 0) ** power
    elif init_type == 3:
        return gauss_bump_init(n, rng.integers(2, 6), rng)
    elif init_type == 4:
        period = int(rng.integers(3, 12))

        a = 0.5 + 0.4 * np.cos(2 * np.pi * t / period)
    elif init_type == 5:

        a = 0.5

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:

def exact_coord_descent(a0, time_limit, verbose=True, label=""):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:

    else:
        a = a / s
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            v = a[i]
            if v < 1e-18:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_max = float(conv[:k_start].max())
            else:
                unaff_max = -1e30
            if k_end < 2 * n - 2:
                rm = float(conv[k_end + 1:].max())
                unaff_max = max(unaff_max, rm)

            affected = conv[k_start:k_end + 1]
            a_sh = a[k_start - i:k_end - i + 1]

            two_i = 2 * i
            quad_in = (two_i >= k_start and two_i <= k_end)
            quad_loc = two_i - k_start if quad_in else -1

            # Generate candidate deltas
            candidates = [0.0, -v, -0.5 * v, 0.5 * v]

            # Deltas that bring top peaks down
            sorted_idx = np.argsort(affected)[::-1]
            for li in sorted_idx[:5]:
                if a_sh[li] > 1e-15:
                    # Delta to bring this peak to unaff_max
                    d1 = (unaff_max - affected[li]) / (2.0 * a_sh[li])
                    if -v * 1.01 < d1 < 2.0 * v + M:
                        candidates.append(d1)
                    # Delta to bring this peak to the second highest
                    if len(sorted_idx) > 1:
                        second_val = affected[sorted_idx[1]]
                        d2 = (second_val - affected[li]) / (2.0 * a_sh[li])
                        if -v * 1.01 < d2 < 2.0 * v + M:
                            candidates.append(d2)

            # Small multiplicative adjustments
            for frac in [0.99, 1.01, 0.97, 1.03, 0.95, 1.05]:
                candidates.append(v * (frac - 1.0))

            # Evaluate candidates
            best_d = 0.0
            best_m = M + 1e-15  # slightly above current to require improvement

            for d in candidates:
                if d == 0.0:
                    continue
                new_v = v + d
                if new_v < -1e-15:
                    continue
                new_aff = affected + 2.0 * d * a_sh
                if quad_in:
                    new_aff = new_aff.copy()
                    new_aff[quad_loc] = conv[two_i] + 2.0 * v * d + d * d
                new_M = max(unaff_max, float(new_aff.max()))
                if new_M < best_m:
                    best_m = new_M
                    best_d = d

            if best_d != 0.0 and best_m < M - 1e-14:
                a[i] = max(a[i] + best_d, 0.0)
                changes = 2.0 * best_d * a_sh
                if quad_in:
                    changes = changes.copy()
                    changes[quad_loc] = 2.0 * best_d * (v - best_d) + best_d * best_d
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 30:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD[{label}] sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f} M={M:.10f}")

    return best_a, best_score

        t_alloc = min(t1_end - time.monotonic(), 150.0)
        print(f"  LP round 2, budget={t_alloc:.0f}s")
        a_lp2, s_lp2 = lp_alternating_improved(best_seq, t_alloc, solver, verbose=True)
        try_update(a_lp2, "lp_r2")

    # === Phase 2: Exact Coordinate Descent (30% time) ===
    print(f"\n=== Phase 2: Exact CD (best={best_score:.8f}) ===")
    t2_end = time.monotonic() + (deadline - time.monotonic()) * 0.30

    t_alloc = min(t2_end - time.monotonic(), 250.0)
    if t_alloc > 20:
        print(f"  ExactCD budget={t_alloc:.0f}s")
        a_cd, s_cd = exact_coord_descent(best_seq, t_alloc, verbose=True, label="main")
        try_update(a_cd, "cd_main")

        # CD from LP result if different
        if time.monotonic() < t2_end - 30:
            t_alloc2 = min(t2_end - time.monotonic(), 80.0)
            a_cd2, s_cd2 = exact_coord_descent(a_lp, t_alloc2, verbose=False, label="lp")
            try_update(a_cd2, "cd_lp")

    # === Phase 3: Peak Equalization + Fine Polish (20% time) ===
    print(f"\n=== Phase 3: PeakEq + FinePolish (best={best_score:.8f}) ===")
    t3_end = time.monotonic() + (deadline - time.monotonic()) * 0.20

    if time.monotonic() < t3_end - 30:
        t_pe = min(t3_end - time.monotonic() - 30, 60.0)
        if t_pe > 15:
            print(f"  PeakEq budget={t_pe:.0f}s")
            a_pe, s_pe = peak_equalization(best_seq, t_pe, verbose=True)
            try_update(a_pe, "peak_eq")

    if time.monotonic() < t3_end - 10:
        t_fp = min(t3_end - time.monotonic() - 10, 80.0)
        if t_fp > 10:
            print(f"  FinePolish budget={t_fp:.0f}s")
            a_fp, s_fp = fine_polish(best_seq, t_fp, verbose=True)
            try_update(a_fp, "fine")

    # === Phase 4: Final polishing (5% time) ===
    print(f"\n=== Phase 4: Final Polish (best={best_score:.8f}) ===")
    t4_end = deadline - 15

    if time.monotonic() < t4_end - 5:
        t_final = min(t4_end - time.monotonic() - 5, 60.0)
        if t_final > 10:
            a_fin, s_fin = exact_coord_descent(best_seq, t_final, verbose=False, label="final")
            try_update(a_fin, "final_cd")

    if time.monotonic() < t4_end - 3:
        t_fin2 = min(t4_end - time.monotonic() - 3, 30.0)
        if t_fin2 > 5:
            a_fin2, s_fin2 = fine_polish(best_seq, t_fin2, verbose=False)
            try_update(a_fin2, "final_fine")

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
