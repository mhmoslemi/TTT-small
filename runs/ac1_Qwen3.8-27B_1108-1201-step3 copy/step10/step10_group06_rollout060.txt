def fast_convolve(a):
    """Fast linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    """Compute the objective function value."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * float(conv.max()) / (s * s)

def get_lp_solver():
    """Find an available LP solver."""
    global _LP_SOLVER
    if _LP_SOLVER is not None:
        return _LP_SOLVER
    import cvxpy as cp
    for name in [cp.GLPK, cp.PDLP, cp.ECOS, cp.SCS, cp.CBC]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _LP_SOLVER = name
                return name
        except Exception:
            continue
    _LP_SOLVER = 'ECOS'
    return _LP_SOLVER

def solve_lp_best_response(a_fixed, solver=None):
    """Solve the LP to find the best response b to a_fixed."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build constraint matrix T where T @ b gives the convolution a_fixed * b
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j+n, j] = a_fixed
    
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in ['ECOS', 'GLPK', 'PDLP', 'SCS']:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    M_val = float(M_var.value)
    return b_val, M_val

def golden_section_search(f, lo, hi, max_iter=50, tol=1e-14):
    """Golden section search for minimization."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < tol:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5

def exact_coord_descent(a0, time_limit, verbose=False, label=""):
    """Exact coordinate descent with golden section search on each coordinate."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0
    last_improve_sweep = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        # Randomize the order slightly in later sweeps to escape local minima
        if sweeps > 5:
            order = np.random.permutation(n)
        else:
            order = range(n)

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue
            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            # Max of unaffected convolution entries
            unaff_max = 0.0
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)
            quad_idx = 2 * i - k_lo if has_quad else -1

            # Precompute the "base" convolution values (what they'd be with a[i]=0)
            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]

            # Define the score function for a given x = a[i]
            def score_fn(x, _c=c_vals, _ap=a_partners, _qm=unaff_max,
                        _hq=has_quad, _qi=quad_idx, _cs=current_sum,
                        _v=v, _n=n):
                ns = _cs + (x - _v)
                if ns < 1e-10:
                    return 1e10
                # Compute new affected convolution values
                nv = _c + 2.0 * x * _ap
                if _hq:
                    nv[_qi] = _c[_qi] + x * x
                nm = max(_qm, float(nv.max()))
                return 2.0 * _n * nm / (ns * ns)

            # Search range: [0, upper]
            upper = max(5.0 * v, 10.0 / n)
            best_x = golden_section_search(score_fn, 0.0, upper, max_iter=35)

            if abs(best_x - v) > 1e-13:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta

                # Update convolution
                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-15:
                    best_score = cur_score
                    best_a = a.copy()
                    last_improve_sweep = sweeps
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 10 == 0:
            print(f"    ECD {label} sweep={sweeps}: best={best_score:.8f}")

    return best_a, best_score, sweeps

def lp_fictitious_play(a0, time_limit, solver=None, verbose=False):

    t0 = time.monotonic()
    iters = 0
    a_ergodic = a.copy()  # Ergodic average
    b_ergodic = a.copy()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 20:
            break

        # Solve LP for best response
        b, M_val = solve_lp_best_response(a, solver)
        if b is None:
            continue

        # Update ergodic averages
        a_ergodic = (iters * a_ergodic + a) / (iters + 1)
        b_ergodic = (iters * b_ergodic + b) / (iters + 1)

        # Try different step sizes
        best_t = 0.0
        best_candidate = a.copy()
        best_ls = best_score

        # Adaptively choose step sizes based on iteration
        if iters < 5:
            trials = [0.3, 0.5, 0.7, 1.0]
        elif iters < 20:
            trials = [0.1, 0.2, 0.3, 0.5, 0.7]
        else:
            trials = [0.05, 0.1, 0.15, 0.2, 0.3, 0.5]

        for trial_t in trials:
            cand = (1.0 - trial_t) * a + trial_t * b
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-10:
                continue
            cand = cand / cs
            sc = compute_score(cand)
            if sc < best_ls:
                best_ls = sc
                best_t = trial_t
                best_candidate = cand.copy()

        # Also try the ergodic averages
        for avg in [a_ergodic, b_ergodic]:
            avg = np.maximum(avg, 0)
            cs = avg.sum()
            if cs < 1e-10:
                continue
            avg = avg / cs
            sc = compute_score(avg)
            if sc < best_ls:
                best_ls = sc
                best_t = -1  # Flag for ergodic
                best_candidate = avg.copy()

        if best_ls < best_score - 1e-12:
            best_score = best_ls
            best_a = best_candidate.copy()
            no_improve = 0
            a = best_candidate.copy()
        else:
            no_improve += 1
            # Still update a using a decaying step
            alpha = 2.0 / (iters + 2.0)
            a = (1.0 - alpha) * a + alpha * b
            a = np.maximum(a, 0)
            cs = a.sum()
            if cs < 1e-10:
                a = np.ones(n) / n
            else:
                a = a / cs

        if verbose and iters % 5 == 0:
            print(f"    FP n={n} it={iters}: best={best_score:.8f}, M={M_val:.6f}")

        if no_improve > 50:
            break

    return best_a, best_score, iters

def init_asymmetric(n, rng, center_frac=0.3):
    """Initialize with a Gaussian bump."""
    t = np.arange(n, dtype=np.float64)
    center = n * center_frac
    width = n * rng.uniform(0.15, 0.35)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_flat_spectrum(n, rng):
    """Initialize with flat magnitude spectrum."""
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = np.ones(n_freq)
    a = irfft(amps * np.exp(1j * phases), n_fft)
    a = np.abs(a[:n])
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

1. LP-based fictitious play with ergodic averaging
2. Exact 1D coordinate descent with golden section search

import numpy as np
from scipy.fft import rfft, irfft
import time

_LP_SOLVER = None

def fast_convolve(a):
    """Fast linear convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * float(conv.max()) / (s * s)

def get_lp_solver():
    """Find an available LP solver."""
    global _LP_SOLVER
    if _LP_SOLVER is not None:
        return _LP_SOLVER
    import cvxpy as cp
    for name in [cp.GLPK, cp.PDLP, cp.ECOS, cp.SCS, cp.CBC]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _LP_SOLVER = name
                return name
        except Exception:
            continue
    _LP_SOLVER = 'ECOS'
    return _LP_SOLVER

def solve_lp_best_response(a_fixed, solver=None):
    """Solve LP: min M s.t. conv(a_fixed,b) <= M, b>=0, sum(b)=1."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed

    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in ['ECOS', 'GLPK', 'PDLP', 'SCS']:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    M_val = float(M_var.value)
    return b_val, M_val

def golden_section_search(f, lo, hi, max_iter=45, tol=1e-14):
    """Golden section search for unimodal minimization."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < tol:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5

def exact_coord_descent(a0, time_limit, verbose=False, label=""):
    """Exact coordinate descent with golden section on each coordinate."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        if sweeps <= 3:
            order = range(n)
        else:
            order = np.random.permutation(n)

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-15:
                continue
            v = a[i]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1

            unaff_max = 0.0
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))

            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)
            quad_idx = 2 * i - k_lo if has_quad else -1

            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]

            c_vals_copy = c_vals.copy()
            ap_copy = a_partners.copy()
            qm = unaff_max
            cs = current_sum
            hq = has_quad
            qi = quad_idx

            def score_fn(x, _c=c_vals_copy, _ap=ap_copy, _qm=qm,
                         _hq=hq, _qi=qi, _cs=cs, _v=v, _n=n):
                ns = _cs + (x - _v)
                if ns < 1e-10:
                    return 1e10
                nv = _c + 2.0 * x * _ap
                if _hq:
                    nv[_qi] = _c[_qi] + x * x
                nm = max(_qm, float(nv.max()))
                return 2.0 * _n * nm / (ns * ns)

            upper = max(5.0 * v, 8.0 / n)
            best_x = golden_section_search(score_fn, 0.0, upper, max_iter=35)

            if abs(best_x - v) > 1e-13:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta

                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-15:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 4:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 5 == 0:
            print(f"    ECD {label} sw={sweeps}: best={best_score:.8f}")

    return best_a, best_score, sweeps

def lp_fictitious_play(a0, time_limit, solver=None, verbose=False):
    """LP fictitious play with ergodic averaging and adaptive steps."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    a_ergodic = a.copy()
    b_ergodic = a.copy()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 15:
            break

        b, M_val = solve_lp_best_response(a, solver)
        if b is None:
            continue

        a_ergodic = (iters * a_ergodic + a) / (iters + 1)
        b_ergodic = (iters * b_ergodic + b) / (iters + 1)

        best_t = 0.0
        best_candidate = a.copy()
        best_ls = best_score

        if iters < 5:
            trials = [0.3, 0.5, 0.7, 1.0]
        elif iters < 20:
            trials = [0.1, 0.2, 0.3, 0.5, 0.7]
        else:
            trials = [0.05, 0.1, 0.15, 0.2, 0.3, 0.5]

        for trial_t in trials:
            cand = (1.0 - trial_t) * a + trial_t * b
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-10:
                continue
            cand = cand / cs
            sc = compute_score(cand)
            if sc < best_ls:
                best_ls = sc
                best_t = trial_t
                best_candidate = cand.copy()

        for avg in [a_ergodic, b_ergodic]:
            avg = np.maximum(avg, 0)
            cs = avg.sum()
            if cs < 1e-10:
                continue
            avg = avg / cs
            sc = compute_score(avg)
            if sc < best_ls:
                best_ls = sc
                best_candidate = avg.copy()
                best_t = -1

        # Also try mixing ergodic with current
        for mix_w in [0.3, 0.5, 0.7]:
            cand = (1.0 - mix_w) * best_candidate + mix_w * b
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-10:
                continue
            cand = cand / cs
            sc = compute_score(cand)
            if sc < best_ls:
                best_ls = sc
                best_candidate = cand.copy()

        if best_ls < best_score - 1e-13:
            best_score = best_ls
            best_a = best_candidate.copy()
            no_improve = 0
            a = best_candidate.copy()
        else:
            no_improve += 1
            alpha = 2.0 / (iters + 2.0)
            a = (1.0 - alpha) * a + alpha * b
            a = np.maximum(a, 0)
            cs = a.sum()
            if cs < 1e-10:
                a = np.ones(n) / n
            else:
                a = a / cs

        if verbose and iters % 5 == 0:
            print(f"    FP n={n} it={iters}: best={best_score:.8f} M={M_val:.6f}")

        if no_improve > 60:
            break

    return best_a, best_score, iters

def mirror_descent_refine(a0, time_limit, verbose=False):
    """Mirror descent (multiplicative weights) to flatten convolution."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()
    iters = 0
    a_avg = np.zeros(n)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        if no_improve > 300:
            break

        conv = fast_convolve(a)
        max_val = float(conv.max())
        if max_val < 1e-15:
            break

        # Focus on top peaks
        k_act = min(15, 2 * n - 1)
        peaks = np.argpartition(conv, -k_act)[-k_act:]
        pv = conv[peaks]
        temp = 0.005 * max_val
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            break
        w = w / ws

        grad = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            idxs = np.arange(i_start, i_end + 1)
            grad[idxs] += w[idx] * 2.0 * a[k - idxs]

        lr = 2.0 / (np.sqrt(iters + 1.0) * 0.5 + 1.0) * 0.3
        a_new = a * np.exp(-lr * grad / max_val)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        a_avg = (iters * a_avg + a_new) / (iters + 1)

        sc = 2.0 * n * float(fast_convolve(a_new).max())
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        sc_avg = 2.0 * n * float(fast_convolve(a_avg).max())
        if sc_avg < best_score - 1e-12:
            best_score = sc_avg
            best_a = a_avg.copy()

        a = a_new

        if verbose and iters % 50 == 0:
            print(f"    MD it={iters}: best={best_score:.8f}")

    return best_a, best_score

def init_asymmetric(n, rng, center_frac=0.3):
    """Gaussian bump initialization."""
    t = np.arange(n, dtype=np.float64)
    center = n * center_frac
    width = n * rng.uniform(0.15, 0.35)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_flat_spectrum(n, rng):
    """Flat magnitude spectrum initialization."""
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    a = irfft(np.exp(1j * phases), n_fft)
    a = np.abs(a[:n])
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_two_peaks(n, rng):
    """Two-peaked initialization."""
    t = np.arange(n, dtype=np.float64)
    c1 = n * rng.uniform(0.2, 0.4)
    c2 = n * rng.uniform(0.6, 0.8)
    w1 = n * rng.uniform(0.1, 0.2)
    w2 = n * rng.uniform(0.1, 0.2)
    a = np.exp(-0.5 * ((t - c1) / w1) ** 2) + np.exp(-0.5 * ((t - c2) / w2) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    """Interpolate to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def block_pair_descent(a0, time_limit, rng, verbose=False):
    """Block-pair coordinate descent: update pairs of coordinates."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    conv = fast_convolve(a).copy()
    current_sum = 1.0
    best_score = 2.0 * n * float(conv.max())
    best_a = a.copy()
    t0 = time.monotonic()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False

        for _ in range(n // 2):
            if time.monotonic() - t0 > time_limit:
                break
            i, j = rng.choice(n, size=2, replace=False)
            if a[i] < 1e-15 or a[j] < 1e-15:
                continue

            vi = a[i]
            vj = a[j]

            # Try a small range of (xi, xj) values
            best_xi, best_xj = vi, vj
            best_local = compute_score_dir(a, conv, current_sum, n)

            for fi in [0.8, 0.9, 1.0, 1.1, 1.2]:
                for fj in [0.8, 0.9, 1.0, 1.1, 1.2]:
                    xi = vi * fi
                    xj = vj * fj
                    if xi < 0 or xj < 0:
                        continue
                    # Compute new score approximately
                    ns = current_sum + (xi - vi) + (xj - vj)
                    if ns < 1e-10:
                        continue
                    # Quick estimate: just check if it helps
                    di = xi - vi
                    dj = xj - vj
                    # Update conv temporarily
                    old_conv_i = conv[i:i+n].copy()
                    old_conv_j = conv[j:j+n].copy()

                    # For coordinate i
                    p_i = np.arange(i, i+n) - i
                    ch_i = 2.0 * di * a[p_i]
                    if 2*i >= i and 2*i < i+n:
                        qi = 2*i - i
                        ch_i = ch_i.copy()
                        ch_i[qi] = di * (2.0*vi + di)
                    conv[i:i+n] += ch_i

                    # For coordinate j
                    p_j = np.arange(j, j+n) - j
                    ch_j = 2.0 * dj * a[p_j]
                    if 2*j >= j and 2*j < j+n:
                        qj = 2*j - j
                        ch_j = ch_j.copy()
                        ch_j[qj] = dj * (2.0*vj + dj)
                    conv[j:j+n] += ch_j

                    ns_score = 2.0 * n * float(conv.max()) / (ns * ns)

                    # Revert
                    conv[i:i+n] -= ch_i
                    conv[j:j+n] -= ch_j

                    if ns_score < best_local - 1e-14:
                        best_local = ns_score
                        best_xi = xi
                        best_xj = xj

            if best_xi != vi or best_xj != vj:
                di = best_xi - vi
                dj = best_xj - vj
                a[i] = best_xi
                a[j] = best_xj
                current_sum += di + dj

                p_i = np.arange(i, i+n) - i
                ch_i = 2.0 * di * a[p_i]
                if 2*i >= i and 2*i < i+n:
                    qi = 2*i - i
                    ch_i = ch_i.copy()
                    ch_i[qi] = di * (2.0*vi + di)
                conv[i:i+n] += ch_i

                p_j = np.arange(j, j+n) - j
                ch_j = 2.0 * dj * a[p_j]
                if 2*j >= j and 2*j < j+n:
                    qj = 2*j - j
                    ch_j = ch_j.copy()
                    ch_j[qj] = dj * (2.0*vj + dj)
                conv[j:j+n] += ch_j

                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved and iters > 5:
            break

        if verbose and iters % 10 == 0:
            print(f"    BPD it={iters}: best={best_score:.8f}")

    return best_a, best_score

def compute_score_dir(a, conv, current_sum, n):
    """Quick score computation from cached convolution."""
    return 2.0 * n * float(conv.max()) / (current_sum * current_sum)

def propose_candidate():
    """
    Multi-phase optimization:
    1. Diverse initialization at multiple lengths
    2. LP Fictitious Play with ergodic averaging
    3. Exact coordinate descent polishing (multiple rounds)
    4. Mirror descent refinement
    5. Block-pair descent for escape
    6. Final intensive polish
    """
    deadline = time.monotonic() + 1780.0
    rng = np.random.default_rng(777)
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
    except Exception as e:
        print(f"Parent load issue: {e}")

    # Phase 1: Diverse initialization
    print("\n=== Phase 1: Diverse Init ===")
    t1_start = time.monotonic()
    t1_budget = 150.0

    n_parent = len(parent) if parent is not None else 1024
    n_values = sorted(set([n_parent, 1024, 1280, 1536, 2048]))
    n_values = [nv for nv in n_values if 256 <= nv <= 2200]

    candidates = []
    for n in n_values:
        if time.monotonic() - t1_start > t1_budget:
            break
        inits = []
        if parent is not None:
            if n == n_parent:
                inits.append(parent.copy())
            else:
                inits.append(interpolate_seq(parent, n))

        inits.append(init_asymmetric(n, rng, 0.30))
        inits.append(init_asymmetric(n, rng, 0.45))
        inits.append(init_flat_spectrum(n, rng))
        inits.append(init_two_peaks(n, rng))

        for j, cand in enumerate(inits[:5]):
            rem = t1_budget - (time.monotonic() - t1_start)
            if rem < 3:
                break
            t_alloc = max(1.0, min(7.0, rem / 5.0))
            a_opt, s_opt, _ = exact_coord_descent(cand, t_alloc, label=f"p1_{n}_{j}")
            candidates.append((s_opt, a_opt, n))

    candidates.sort(key=lambda x: x[0])
    print(f"  Phase 1: {len(candidates)} cands, best={candidates[0][0]:.8f}")

    for s_c, a_c, n_c in candidates[:3]:
        sc = float(evaluate_sequence(a_c.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_c.copy()
            print(f"  NEW BEST p1: {sc:.8f} (n={n_c})")

    # Phase 2: LP Fictitious Play
    print(f"\n=== Phase 2: LP Fictitious Play (best={best_score:.8f}) ===")
    n_work = len(best_seq)
    rem = deadline - time.monotonic() - 550
    if rem > 30:
        t_fp = min(rem * 0.55, 300.0)
        print(f"  FP budget={t_fp:.0f}s, n={n_work}")
        a_fp, s_fp, iters = lp_fictitious_play(best_seq, t_fp, solver=solver, verbose=True)
        sc = float(evaluate_sequence(a_fp.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_fp.copy()
            n_work = len(best_seq)
            print(f"  NEW BEST FP: {sc:.8f}")

        alt_n = n_work + 256

                sc2 = float(evaluate_sequence(a_fp2.tolist()))

            print(f"  NEW BEST MD: {sc:.8f}")

            sc3 = float(evaluate_sequence(a_cd3.tolist()))
            if np.isfinite(sc3) and sc3 < best_score:

    # Phase 5: Block-pair descent
    print(f"\n=== Phase 5: Block-Pair (best={best_score:.8f}) ===")
    n_work = len(best_seq)

        a_bp, s_bp = block_pair_descent(best_seq, t_bp, rng, verbose=True)
        sc = float(evaluate_sequence(a_bp.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_bp.copy()
            print(f"  NEW BEST BP: {sc:.8f}")

    n_work = len(best_seq)

    if rem > 10:
        t_fin = min(rem * 0.6, 50.0)
        print(f"  Final budget={t_fin:.0f}s")
        a_fin, s_fin, _ = exact_coord_descent(best_seq, t_fin, verbose=True, label="final")
        sc = float(evaluate_sequence(a_fin.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_fin.copy()
            print(f"  NEW BEST final: {sc:.8f}")

    # Last second
    rem = deadline - time.monotonic() - 5
    if rem > 5:
        a_last, s_last, _ = exact_coord_descent(best_seq, min(rem * 0.8, 15.0), label="last")
        sc = float(evaluate_sequence(a_last.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_last.copy()

    if best_seq is None:
        best_seq = np.ones(512) / 512

    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
