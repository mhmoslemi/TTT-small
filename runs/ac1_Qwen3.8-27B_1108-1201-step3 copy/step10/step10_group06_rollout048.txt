Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Mirror descent with exponential weights (fast global moves)
2. LP Fictitious Play with exact best response
3. Enhanced coordinate descent with dual-guided step selection

5. Basin hopping with structured perturbations

import numpy as np
from scipy.fft import rfft, irfft
import time

_LP_SOLVER = None

def get_lp_solver():
    global _LP_SOLVER
    if _LP_SOLVER is not None:
        return _LP_SOLVER
    import cvxpy as cp
    for name in [cp.GLPK, cp.ECOS, cp.PDLP, cp.CBC, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _LP_SOLVER = name
                return name
        except Exception:
            continue
    _LP_SOLVER = 'ECOS'
    return 'ECOS'

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * float(conv.max()) / (s * s)

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

def solve_lp_best_response(a_fixed, solver=None):
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in ['ECOS', 'GLPK', 'PDLP', 'SCS', 'CBC']:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    M_val = float(M_var.value)
    return b_val, M_val

**Faster approach: Analytical solution for the 1D subproblem**

The score as a function of x is:
score(x) = 2*n * max(unaff_max, max_{k in affected}(c_k + 2*x*a_{k-i} or c_k + x²)) / (S + x - v)²

where S = sum(a), c_k are the convolution values with the x-dependent terms removed.

This is a ratio of (max of linear/quadratic functions) to (affine function)².

The maximum in the numerator is achieved at some specific k*. If we know k*, then the score becomes:
score(x) = 2*n * (c_{k*} + 2*x*a_{k*-i}) / (S + x - v)²  [for k* != 2i]
or
score(x) = 2*n * (c_{k*} + x²) / (S + x - v)²  [for k* = 2i]

In the first case, this is (linear) / (affine)², which has a unique minimum that can be found analytically.
In the second case, this is (quadratic) / (affine)², which also has an analytical minimum.

But the issue is: we don't know which k* is the bottleneck for all x values. The argmax changes as x varies.

However, we can use the following approach:
1. Evaluate score at x = v (current value) to find the current bottleneck k*
2. Solve the 1D problem assuming k* is the bottleneck
3. Check if the solution is valid (i.e., k* is still the bottleneck at the optimal x)
4. If not, identify the new bottleneck and repeat

This "active set" approach can converge in just a few iterations for most coordinates.

Let me code this up:

For the linear case (k* != 2i):
score(x) = 2*n * (c + p*x) / (S0 + x)² where c = c_{k*}, p = 2*a_{k*-i}, S0 = S - v

d(score)/dx = 2*n * [p*(S0+x)² - (c+p*x)*2*(S0+x)] / (S0+x)⁴
= 2*n * [p*(S0+x) - 2*(c+p*x)] / (S0+x)³
= 2*n * [p*S0 + p*x - 2c - 2p*x] / (S0+x)³
= 2*n * [p*S0 - 2c - p*x] / (S0+x)³

Setting to zero: p*S0 - 2c - p*x = 0
x* = (p*S0 - 2c) / p = S0 - 2c/p = S0 - c/a_{k*-i}

But we need x >= 0 and the constraint that other lags don't exceed this value.

For the quadratic case (k* = 2i):
score(x) = 2*n * (c + x²) / (S0 + x)²

d(score)/dx = 2*n * [2x*(S0+x)² - (c+x²)*2*(S0+x)] / (S0+x)⁴
= 2*n * [2x*(S0+x) - 2*(c+x²)] / (S0+x)³
= 2*n * [2x*S0 + 2x² - 2c - 2x²] / (S0+x)³
= 2*n * [2x*S0 - 2c] / (S0+x)³
= 4*n * (x*S0 - c) / (S0+x)³

Setting to zero: x*S0 - c = 0 → x* = c/S0

So in both cases, we get a simple formula for the unconstrained optimum! Then we just need to:
1. Clamp to x >= 0
2. Check that the bottleneck lag is still valid
3. If not, find the new bottleneck and iterate

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

        j = k_star - i
        if j == i:
            g[i] = 2.0 * a[i]
        else:
            g[i] = a[j]
    return g

def golden_section_search(f, lo, hi, max_iter=30):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-14:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fd = fc
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5

            cs = s

            def score_fn(x, _c=c_vals, _ap=partners_arr, _qm=unaff_max,
                         _hq=has_quad, _qi=quad_idx, _cs=cs, _n=n):
                ns = _cs + (x - 0.0)
                if ns < 0.01:
                    return 1e10
                nv = _c + 2.0 * x * _ap
                if _hq:
                    nv = nv.copy()
                    nv[_qi] = _c[_qi] + x * x
                nm = max(_qm, float(nv.max()))
                return 2.0 * _n * nm / (ns * ns)

            upper = max(3.0 * v, 5.0 / n)
            best_x = golden_section_search(score_fn, 0.0, upper, max_iter=25)

            if abs(best_x - v) > 1e-13:
                delta = best_x - v
                a[i] = best_x
                s = s + delta
                changes = 2.0 * delta * partners_arr
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes

                cur_score = 2.0 * n * float(conv.max()) / (s * s)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
        if verbose and sweeps % 5 == 0:
            print(f"    ECD sweep={sweeps}: best={best_score:.8f}")

    return best_a, best_score


def lp_fictitious_play(a0, time_limit, solver=None, verbose=False):
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    a_avg = a.copy()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 10:
            break

        b, M_val = solve_lp_best_response(a, solver)
        if b is None:
            no_improve += 1
            if no_improve > 10:
                break
            continue

        best_t = 0.0
        best_ls = best_score
        trial_ts = [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
        for trial_t in trial_ts:
            cand = (1.0 - trial_t) * a + trial_t * b
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-10:
                continue
            cand = cand / cs
            sc = compute_score(cand)
            if sc < best_ls:
                best_ls = sc
                best_t = trial_t

        if best_t > 0:
            a_new = (1.0 - best_t) * a + best_t * b
            a_new = np.maximum(a_new, 0)
            cs = a_new.sum()
            if cs < 1e-10:
                a_new = a.copy()
            else:
                a_new = a_new / cs
            score_new = compute_score(a_new)
            if score_new < best_score - 1e-12:
                best_score = score_new
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1
            a = a_new
        else:
            no_improve += 1
            alpha = 3.0 / (iters + 3.0)
            a = (1.0 - alpha) * a + alpha * b
            a = np.maximum(a, 0)
            cs = a.sum()
            if cs < 1e-10:
                a = np.ones(n) / n
            else:
                a = a / cs

        a_avg = (iters * a_avg + a) / (iters + 1)

        for cand in [a, a_avg, b]:
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-10:
                continue
            cand = cand / cs
            sc = compute_score(cand)
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()

        if verbose and iters % 5 == 0:
            print(f"    FP it={iters}: best={best_score:.8f}")

        if no_improve > 40:
            break

    return best_a, best_score


def basin_hopping(a0, time_limit, rng, verbose=False):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    count = 0

    while time.monotonic() - t0 < time_limit:
        count += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 20:
            break
        t_alloc = min(rem * 0.35, 100.0)

        pert_type = count % 5
        perturbed = a.copy()

        if pert_type == 0:
            conv_p = fast_convolve(a)
            top_k = min(15, 2 * n - 1)
            peaks = np.argpartition(conv_p, -top_k)[-top_k:]
            for k in peaks:
                ki = int(k)
                i_start = max(0, ki - n + 1)
                i_end = min(n - 1, ki)
                idx = np.arange(i_start, i_end + 1)
                n_pick = max(1, len(idx) // 5)
                picks = rng.choice(idx, size=min(n_pick, len(idx)), replace=False)
                for ii in picks:
                    perturbed[ii] *= (1.0 - 0.08 * rng.random())
        elif pert_type == 1:
            sigma = 0.01 + 0.04 * rng.random()
            perturbed = a + rng.standard_normal(n) * sigma
        elif pert_type == 2:
            blen = max(1, rng.integers(n // 10, max(2, n // 4)))
            i1 = rng.integers(0, max(1, n - blen))
            i2 = rng.integers(0, max(1, n - blen))
            if abs(i1 - i2) > blen:
                temp = perturbed[i1:i1 + blen].copy()
                perturbed[i1:i1 + blen] = perturbed[i2:i2 + blen]
                perturbed[i2:i2 + blen] = temp
        elif pert_type == 3:
            shift = rng.integers(1, max(2, n // 5))
            shifted = np.roll(a, int(shift))
            perturbed = 0.7 * a + 0.3 * shifted
        else:
            mask = rng.random(n) < 0.1
            perturbed[mask] *= rng.uniform(0.3, 1.7, mask.sum())

        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-10:
            continue
        perturbed = perturbed / ps

        a_opt, s_opt = analytical_coord_descent(perturbed, t_alloc)
        if s_opt < best_score - 1e-10:
            best_score = s_opt
            best_a = a_opt.copy()
            a = a_opt.copy()
            if verbose:
                print(f"    BH {count}: improved to {best_score:.8f}")

    return best_a, best_score


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def init_asymmetric(n, rng, shift=0.3):
    t = np.arange(n, dtype=np.float64)
    center = n * shift
    width = n * rng.uniform(0.15, 0.4)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_flat_spectrum(n, rng):
    n_fft = 1 << (n - 1).bit_length()
    n_freq = n_fft // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = rng.uniform(0.5, 1.5, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, n_fft)
    a = np.abs(a[:n])
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_dual_peak(n, rng):
    t = np.arange(n, dtype=np.float64)
    c1 = n * rng.uniform(0.2, 0.4)
    c2 = n * rng.uniform(0.6, 0.8)
    w = n * rng.uniform(0.1, 0.25)
    a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s
