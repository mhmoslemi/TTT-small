from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def subgradient_of_max_conv(a, active_set):
    """Compute subgradient of max_k conv(a)[k] at given active set."""
    n = len(a)
    g = np.zeros(n)
    for k in active_set:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]
    g /= len(active_set)
    return g

def project_simplex(v):
    """Project vector v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

    a = np.maximum(np.asarray(a, dtype=np.float64), 0)

    if s < 0.01:

    n = len(a)

    return float(2.0 * n * b.max() / (s * s))

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated mirror descent (multiplicative weights) - fast per iteration
2. Aggressive LogSumExp L-BFGS with fine-grained temperature annealing
3. FISTA-style proximal gradient as alternative
4. Multi-length exploration with diverse initializations
5. Adaptive method switching based on progress
Key differences from previous:
- No LP solves in main loop (much faster per iteration)
- More aggressive LogSumExp cooling
- Better step size adaptation
- FISTA as additional method
- More diverse starting points

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

# Global state for L-BFGS functions (to avoid closures)
_lse_state = {'n': 0, 'T': 1.0}

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def subgradient_max_conv(a, active_set):
    """Compute subgradient of max_k conv(a)[k] at given active set of k indices."""
    n = len(a)
    g = np.zeros(n)
    for k in active_set:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]
    g /= len(active_set)
    return g

def project_simplex(v):
    """Project vector v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def softmax_fn(z):
    """Softmax function (numerically stable)."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def lse_func(z):
    """LogSumExp objective for L-BFGS."""
    n = _lse_state['n']
    T = _lse_state['T']
    a = softmax_fn(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T)
    lse = max_b + T * np.log(exp_terms.sum())
    return 2.0 * n * lse

def lse_jac(z):
    """Gradient of LogSumExp objective w.r.t. z."""
    n = _lse_state['n']
    T = _lse_state['T']
    a = softmax_fn(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T)
    w = exp_terms / exp_terms.sum()
    
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return 2.0 * n * grad_z

def nesterov_mirror_descent(a0, time_limit, eta_init=0.1):
    """
    Nesterov-accelerated mirror descent (multiplicative weights).
    Minimizes max(conv(a,a)) on the simplex.
    """
    n = len(a0)
    a = normalize(a0).copy()
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = eta_init
    t_param = 1.0
    no_improve = 0
    
    for it in range(5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute convolution at lookahead point
        b = fast_convolve(y)
        max_b = b.max()
        threshold = max_b * (1.0 - 1e-5)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        
        # Subgradient
        g = subgradient_max_conv(y, active)
        
        # Mirror descent step
        log_y = np.log(np.maximum(y, 1e-300))
        logits = log_y - eta * g
        logits -= logits.max()
        a_new = np.exp(logits)
        sa = a_new.sum()
        if sa < 1e-300:
            a_new = y.copy()
        else:
            a_new = a_new / sa
        
        # Nesterov extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param)) / 2.0
        y_new = a_new + (t_param - 1.0) / t_new * (a_new - a)
        y_new = np.maximum(y_new, 0.0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy
        
        t_param = t_new
        a = a_new
        y = y_new
        
        # Track best
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive step size
        if no_improve >= 100:
            eta *= 0.75
            no_improve = 0
            if eta < 1e-8:
                break
        
        if it % 5000 == 0 and it > 0:
            elapsed = time.monotonic() - t0
            print(f"    NMD iter={it} ({elapsed:.0f}s): score={best_score:.8f} eta={eta:.6f}")
    
    return best_a, best_score

def structured_init(n, rng, pattern_id):
    """Create a structured initial sequence with given pattern."""
    x = np.linspace(0, 1, n)
    if pattern_id == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif pattern_id == 1:
        a = np.sin(np.pi * x) ** 2
    elif pattern_id == 2:
        a = np.exp(-4 * x * (1 - x))
    elif pattern_id == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif pattern_id == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif pattern_id == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif pattern_id == 6:
        # Triangle
        a = np.minimum(x, 1 - x) * 2 + 0.05
    elif pattern_id == 7:
        # Super-Gaussian
        a = np.exp(-8 * x * (1 - x))
    elif pattern_id == 8:
        # Bell with skirt
        a = np.exp(-5 * (x - 0.5)**2) + 0.1 * np.ones(n)
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def perturbed_sequence(a, rng, strength=0.03):
    """Create a perturbed version of sequence a."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 8)
    phase = rng.uniform(0, 2 * np.pi)
    perturbation = strength * np.cos(freq * x + phase)
    a_pert = a + perturbation
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s
