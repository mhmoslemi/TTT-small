def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

    a = np.maximum(np.asarray(a, dtype=np.float64), 0)

    if s < 0.01:

    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

    n = len(a)

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-accelerated gradient-based
optimization with log-sum-exp smoothing, peak-directed coordinate descent,
and multi-start diversity. No LP solves - pure first-order methods for speed.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def eval_score(a):
    a = np.maximum(np.asarray(a, float), 0)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max() / (s * s)

def lse_obj(z, T):
    zs = z - z.max()
    a = np.exp(zs)
    a /= a.sum()
    b = fast_convolve(a)
    mb = b.max()
    e = np.exp(T * (b - mb))
    return 2.0 * len(a) * (mb + np.log(e.sum()) / T)

def lse_grad(z, T):
    zs = z - z.max()
    a = np.exp(zs)
    a /= a.sum()
    b = fast_convolve(a)
    mb = b.max()
    e = np.exp(T * (b - mb))
    w = e / e.sum()
    n = len(a)
    # grad_a[i] = (2/T) * sum_k w[k] * a[k-i] * 1_{0<=k-i<n}
    # = (2/T) * cross_correlate(w, a)[i]
    # cross_correlate(w, a)[i] = convolve(w, a[::-1])[i+n-1]
    size = 3 * n - 2
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    Ar = rfft(a[::-1], fft_size)
    cc = irfft(W * Ar, fft_size)[:size]
    grad_a = (2.0 / T) * cc[n-1:2*n-1]
    # grad_z[i] = a[i] * (2n * grad_a[i] - dot(2n*grad_a, a))
    obj_grad = 2.0 * n * grad_a
    S = np.dot(obj_grad, a)
    return a * (obj_grad - S)

def run_lbgfs(a0, deadline, T_stages):
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    best_a = a0.copy()
    best_score = eval_score(a0)
    
    for T in T_stages:
        if time.monotonic() > deadline:
            break
        # Run L-BFGS for this temperature
        res = minimize(
            fun=lambda z_: lse_obj(z_, T),
            x0=z,
            jac=lambda z_: lse_grad(z_, T),
            method='L-BFGS-B',
            options={'maxiter': 5000, 'ftol': 1e-18, 'gtol': 1e-14, 'disp': False}
        )
        z = res.x
        a = np.exp(z - z.max())
        a /= a.sum()
        sc = eval_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    return best_a, best_score

    size = 2 * n - 1

    A = rfft(a, fft_size)

def cross_correlate_peak(w, a):
    """Compute cross-correlation (w ⋆ a)[i] = sum_k w[k]*a[k-i] for i=0..n-1."""
    n = len(a)
    size = (2 * n - 1) + n - 1  # = 3n-2
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    Ar = rfft(a[::-1], fft_size)
    result = irfft(W * Ar, fft_size)[:size]
    return result[n - 1: 2 * n - 1]


def lse_objective(z, T):
    """Log-sum-exp smoothed objective in log-parameterization."""
    zs = z - z.max()
    a = np.exp(zs)
    s = a.sum()
    if s < 1e-300:
        return 1e15
    a = a / s
    b = fast_convolve(a)
    mb = b.max()
    e = np.exp(T * (b - mb))
    ess = e.sum()
    if ess < 1e-300:
        return 1e15
    lse = mb + np.log(ess) / T
    return 2.0 * len(a) * lse


def lse_gradient(z, T):
    """Gradient of LSE objective w.r.t. log-parameters z."""
    zs = z - z.max()
    a = np.exp(zs)
    s = a.sum()
    a = a / s
    b = fast_convolve(a)
    mb = b.max()
    e = np.exp(T * (b - mb))
    ess = e.sum()
    w = e / ess
    n = len(a)
    # grad of lse w.r.t. a_i = (2/T) * (w ⋆ a)[i]
    cc = cross_correlate_peak(w, a)
    grad_a = (2.0 / T) * cc
    # grad of objective w.r.t. a: 2n * grad_a
    obj_grad = 2.0 * n * grad_a
    # grad w.r.t. z: a_i * (obj_grad_i - sum(a * obj_grad))
    S = np.dot(obj_grad, a)
    return a * (obj_grad - S)


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(1.0, v[0])])
    u = np.sort(v)[::-1]
    css = np.cumsum(u) - 1.0
    pos = u - css / np.arange(1, n + 1) > 0
    rho = np.where(pos)[0]

        return np.ones(n) / n
    rho = rho[-1]
    theta = (css[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
