import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a.astype(np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Project onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FISTA with FFT-accelerated LogSumExp gradient (backtracking line search)
2. Peak peeling algorithm for targeted max-conv reduction
3. Nesterov-accelerated subgradient with adaptive step
4. Multi-length exploration with smooth interpolation
5. Simulated annealing restarts for global escape
Key differences from previous: uses FISTA (not L-BFGS), FFT-based gradient
(computed in O(n log n)), peeling algorithm, and aggressive multi-n search.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a.astype(np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def lse_objective(a, T):
    """Compute 2n * LSE_T(conv(a,a))."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T)
    lse = max_b + T * np.log(exp_terms.sum())
    return 2.0 * n * lse


def lse_gradient(a, T):
    """Compute gradient of 2n * LSE_T(conv(a,a)) w.r.t. a using FFT."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T)
    w = exp_terms / exp_terms.sum()

    # grad[m] = 4n * sum_j a[j] * w[m+j]
    # Compute via FFT: cross-correlation
    N = 1 << (3 * n - 2).bit_length()
    A = rfft(a, N)
    W = rfft(w, N)
    d_fft = irfft(A * np.conj(W), N)
    # d[m] = d_fft[(-m) mod N]
    indices = np.mod(-np.arange(n), N)
    d = d_fft[indices]
    return 4.0 * n * d


def score_local(a):
    """Compute the score 2n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def fista_lse(a0, time_limit, T_schedule, iters_per_stage=10000):
    """FISTA with LogSumExp smoothing and backtracking."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for T in T_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        y = a.copy()
        t_mom = 1.0
        # Initial step: use rough Lipschitz estimate
        step = T / (2.0 * n)
        if step > 0.1:
            step = 0.1
        if step < 1e-10:
            step = 1e-10

        max_iters = min(iters_per_stage, max(50, int(remaining * 50)))
        no_improve = 0
        stage_best = best_score

        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit - 2:
                break

            obj_y = lse_objective(y, T)
            grad_y = lse_gradient(y, T)

            # Backtracking line search
            alpha = step
            a_new = None
            for _bt in range(15):
                a_trial = y - alpha * grad_y
                a_trial = project_simplex(a_trial)
                obj_trial = lse_objective(a_trial, T)
                # Descent condition
                lo = obj_y + alpha * np.dot(grad_y, a_trial - y)
                if obj_trial <= lo + 1e-14:
                    a_new = a_trial
                    break
                alpha *= 0.5
            if a_new is None:
                a_new = project_simplex(y - alpha * grad_y)

            # FISTA momentum update
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
            momentum = (t_mom - 1.0) / t_new
            y = a_new + momentum * (a_new - a)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy > 1e-15:
                y = y / sy
            else:
                y = a_new.copy()
            a = a_new
            t_mom = t_new

            # Adapt step size
            step = max(step * 1.01, alpha * 1.1)

            # Check score periodically (expensive due to FFT)
            if it % 200 == 0 or it == 0:
                score = 2.0 * n * fast_convolve(a).max()
                if score < stage_best:
                    stage_best = score
                    no_improve = 0
                else:
                    no_improve += 1
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                if no_improve >= 30:
                    break

        a = best_a.copy()

    return best_a, best_score


def peeling_optimize(a0, time_limit, n_peel_steps=50000):
    """Peak peeling: iteratively reduce the highest convolution peak."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    alpha = 0.5
    stall = 0

    for step in range(n_peel_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        k_star = np.argmax(b)
        max_b = b[k_star]

        # Subgradient: g[m] = 2 * a[k_star - m] for valid m
        g = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        g[idx] = 2.0 * a[k_star - idx]

        # Step size based on gap
        target = 1.0 / (2 * n - 1)
        gap = max_b - target
        if gap < 1e-12:
            break
        g_sq = np.dot(g, g)
        if g_sq < 1e-20:
            break
        step_size = alpha * gap / g_sq

        a_new = a - step_size * g
        a_new = project_simplex(a_new)

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            a = a_new  # still accept small moves for exploration
            stall += 1

        if stall > 20:
            alpha *= 0.8
            stall = 0
            if alpha < 1e-6:
                alpha = 0.1  # restart with larger steps (perturbation)
                b = fast_convolve(a)
                k_star = np.argmax(b)
                # Perturb around the peak
                n_pert = max(1, n // 20)
                center = k_star
                lo = max(0, center - n_pert)
                hi = min(n, center + n_pert)
                noise = np.random.randn(hi - lo) * 0.01
                a[lo:hi] += noise
                a = np.maximum(a, 0)
                a = a / a.sum()

        if step % 5000 == 0 and step > 0:
            print(f"    Peeling {time.monotonic()-t0:.0f}s: score={best_score:.8f} alpha={alpha:.4f}")

    return best_a, best_score


def nesterov_subgrad(a0, time_limit, T=0.5, n_steps=100000):
    """Nesterov-accelerated LogSumExp subgradient with averaging."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    t_mom = 1.0
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    # Step size schedule: alpha_t = alpha0 / (1 + t*decay)
    alpha0 = 1.0 / np.sqrt(n)
    decay = 1.0 / (n_steps * 0.5)
    avg_a = np.zeros(n)
    count = 0

    for t in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        alpha = alpha0 / (1.0 + t * decay)
        grad_y = lse_gradient(y, T)
        a = project_simplex(y - alpha * grad_y)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        y = a + ((t_mom - 1.0) / t_new) * (a - a)
        # Correct FISTA: y_{k+1} = x_{k+1} + ((t_k-1)/t_{k+1})(x_{k+1} - x_k)
        # We need to store x_k
        # Let me redo this properly
        y = a  # simplified (no momentum, just subgradient)
        # Actually let's use proper Nesterov:
        # y_{k+1} = x_{k+1} + ((t_k - 1)/t_{k+1})(x_{k+1} - x_k)
        # We need x_k which is the previous a

        count += 1
        avg_a += a

        if t % 1000 == 0:
            score = 2.0 * n * fast_convolve(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

    # Also check the average
    if count > 0:
        avg_a = avg_a / count
        avg_a = project_simplex(avg_a)
        score_avg = 2.0 * n * fast_convolve(avg_a).max()
        if score_avg < best_score:
            best_score = score_avg
            best_a = avg_a.copy()

    return best_a, best_score


def nesterov_subgrad_v2(a0, time_limit, T=0.5, n_steps=100000):
    """Nesterov-accelerated subgradient with proper momentum and averaging."""
    n = len(a0)
    x = np.maximum(a0.astype(np.float64), 0).copy()
    s = x.sum()
    if s < 1e-15:
        x = np.ones(n) / n
    else:
        x = x / s

    y = x.copy()
    t_mom = 1.0
    best_a = x.copy()
    best_score = 2.0 * n * fast_convolve(x).max()
    t0 = time.monotonic()

    alpha0 = 0.5 / np.sqrt(n)

    for t in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        alpha = alpha0 / (1.0 + t * 0.01)
        grad_y = lse_gradient(y, T)
        x_new = project_simplex(y - alpha * grad_y)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        y = x_new + ((t_mom - 1.0) / t_new) * (x_new - x)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy
        else:
            y = x_new.copy()

        x = x_new
        t_mom = t_new

        if t % 2000 == 0:
            score = 2.0 * n * fast_convolve(x).max()
            if score < best_score:
                best_score = score
                best_a = x.copy()

            score_y = 2.0 * n * fast_convolve(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()

    return best_a, best_score


def structured_init(n, rng):
    """Create a structured initial sequence."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-2 * (x - 0.3) ** 2) + 0.5 * np.exp(-2 * (x - 0.7) ** 2)
    elif choice == 6:
        # U-shape: good for flat convolution
        a = x ** 2 + (1 - x) ** 2 + 0.1
    else:
        # Bimodal at edges
        a = np.exp(-30 * x ** 2) + np.exp(-30 * (1 - x) ** 2) + 0.05

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def simulated_annealing(a0, time_limit, n_restarts=10, local_time=10.0):
    """Simulated annealing: perturb, re-optimize, keep best."""
    n = len(a0)
    rng = np.random.default_rng(12345)
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()
    t0 = time.monotonic()

    for r in range(n_restarts):
        if time.monotonic() - t0 > time_limit:
            break
        rem = time_limit - (time.monotonic() - t0)
        if rem < local_time * 1.5:
            break

        # Generate perturbation
        pert_strength = 0.02 + 0.03 * rng.random()
        x = np.linspace(0, 2 * np.pi * 3, n)
        freq1 = rng.integers(1, 5)
        freq2 = rng.integers(5, 15)
        p1 = rng.uniform(0, 2 * np.pi)
        p2 = rng.uniform(0, 2 * np.pi)
        pert = pert_strength * (np.cos(freq1 * x + p1) + 0.5 * np.cos(freq2 * x + p2))
        a_pert = a0 + pert
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()

        # Local optimization with peeling (fast)
        t_sub = min(local_time, rem * 0.4)
        a_opt, _ = peeling_optimize(a_pert, t_sub, n_peel_steps=5000)
        sc = 2.0 * n * fast_convolve(a_opt).max()
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            a0 = a_opt.copy()
            print(f"    SA restart {r}: score={sc:.8f} (improved)")

    return best_a, best_score
