- From the term `k-i=j`, i.e., `i=k-j`: `a[k-j]` (if `0 ≤ k-j < n`)
- If both conditions give the same term (i.e., `j = k-j`, so `k=2j`), we only count it once but the derivative is `2*a[j]`.

So: `d(a*a)_k / da[j] = a[k-j] + a[k-j]` if `0 ≤ k-j < n` and `k ≠ 2j`
`= 2*a[j]` if `k = 2j`
`= a[k-j]` if only one of the conditions holds (but this can't happen since both conditions are equivalent: `0 ≤ k-j < n`)

Wait, both conditions are the same: `0 ≤ k-j < n`. So:
- If `0 ≤ k-j < n`: derivative is `2*a[k-j]` if `k ≠ 2j`, else `2*a[j]` (same thing since `k-j = j` when `k=2j`)
- If `k-j < 0` or `k-j ≥ n`: derivative is 0

So: `d(a*a)_k / da[j] = 2*a[k-j]` if `0 ≤ k-j < n`, else 0.
(Since when `k=2j`, `k-j=j`, so `2*a[k-j] = 2*a[j]`, which is correct.)

Great, so the gradient is simple: `grad_k[j] = 2*a[k-j]` if `0 ≤ k-j < n`, else 0.

This means the gradient of `(a*a)_k` is just a shifted (and possibly reversed) copy of `a`, scaled by 2.

For the subgradient of `max_k (a*a)_k`, if the active set is `S`, then a subgradient is:
`sum_{k in S} w_k * 2*shifted_a(k)` where `w_k ≥ 0, sum w_k = 1`.

This is very efficient to compute!

Now, for the optimization, I'll use:
1. **Track the top M peaks** (M = 20 or so)
2. **Compute the weighted average gradient** over these peaks
3. **Take a projected gradient step** with backtracking
4. **Use Nesterov momentum** for acceleration

Let me also think about the projection. The constraint is `a ≥ 0` and `sum a = 1`. The projection onto this set is:
1. Subtract a Lagrange multiplier to make the sum = 1
2. Clip to 0
3. Renormalize

Actually, the projection onto `{a ≥ 0, sum a = 1}` (the probability simplex) is:
1. Start with `a - α * 1` for some α
2. Find α such that `sum max(0, a_j - α) = 1`
3. The result is `max(0, a - α)`

But since we're already on the simplex (we can maintain this), the projection is just clipping to 0 and renormalizing.

For efficiency, I'll maintain `a` on the simplex at all times: after each step, clip to 0 and renormalize.

def eval_score(a):

        return float('inf')

    b = fast_convolve(a)

def normalize(a):

def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a, find b >= 0, sum b = 1 minimizing max_k conv(a,b)_k."""
    if solver is None:
        solver = _get_solver()

    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    size = 2*n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a_fixed
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0)
            s = b_val.sum()
            if s < 1e-15:
                return None
            return b_val / s

    return None

def subgradient_optimize(a0, time_limit, top_m=30):

def logsumexp_lbgfs(a0, time_limit, T_stages=None):

    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.01]
    
    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        maxiter = max(10, int(remaining * 30))  # rough estimate

            a = np.exp(z - z.max())
            a = a / a.sum()
            b = fast_convolve(a)
            max_b = b.max()
            lse = max_b + np.log(np.exp(b - max_b).sum())
            return 2.0 * n * lse / T
        
        def jac(z):
            a = np.exp(z - z.max())
            a = a / a.sum()
            b = fast_convolve(a)
            max_b = b.max()
            exp_b = np.exp(b - max_b)
            w = exp_b / exp_b.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-12:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += w[k] * a[k - idx]
            grad_a *= 2.0 / T
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        
        try:
            res = minimize(func, z, jac=jac, method='L-BFGS-B',
                          options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
            z = res.x
        except:
            pass
    
    a_out = np.exp(z - z.max())
    a_out = a_out / a_out.sum()
    return a_out

    active = np.where(b >= threshold)[0]
    if len(active) > top_m:
        active = np.argsort(b)[::-1][:top_m]

        je = min(n - 1, k)
        idx = np.arange(js, je + 1)

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def eval_objective(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def get_lp_solver():
    """Find a working LP solver."""
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for s in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = s
                return s
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_response(a_fixed, solver=None):
    """Given fixed a (sum=1), find b>=0, sum b=1 minimizing max_k conv(a,b)_k."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0)
            s = b_val.sum()
            if s < 1e-15:
                return None
            return b_val / s
    except Exception:
        pass
    for alt in [cp.SCS, cp.ECOS, cp.CLARABEL, cp.PDLP]:
        if alt == solver:
            continue
        try:
            prob.solve(solver=alt, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0)
                s = b_val.sum()
                if s < 1e-15:
                    return None
                return b_val / s
        except Exception:
            continue
    return None


def nesterov_lp_optimize(a0, time_limit, solver=None):
    """Nesterov dual-averaging LP alternation."""
    n = len(a0)
    a = normalize(a0)
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0
    rng = np.random.default_rng(42)
    prev_best = best_score

    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        b_new = solve_lp_response(y, solver)
        if b_new is None:
            stall += 1
            if stall > 30:
                break
            continue
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = normalize(y)
        denom = (it + 1) + alpha * (it + 2)
        a = ((it + 1) * a + alpha * (it + 2) * b_new) / denom
        a = normalize(a)
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        if (it + 1) % 500 == 0:
            if best_score < prev_best - 1e-10:
                stall = 0
                prev_best = best_score
            else:
                stall += 1
            if stall >= 10:
                noise = 0.03 / np.sqrt(max(1, stall)) * rng.standard_normal(n)
                y = normalize(y + noise)
                stall = 0
        if (it + 1) % 3000 == 0:
            print(f"    NLP it={it+1} ({time.monotonic()-t0:.0f}s): score={best_score:.8f}")
    return best_a, best_score


def compute_subgradient(a, top_m=30):
    """Compute weighted subgradient of max_k (a*a)_k."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    threshold = max_b * (1.0 - 1e-6)
    active = np.where(b >= threshold)[0]
    if len(active) > top_m:
        active = np.argsort(b)[::-1][:top_m]
    if len(active) == 0:
        active = np.array([np.argmax(b)])
    w = b[active] / b[active].sum()
    grad = np.zeros(n)
    for i in range(len(active)):
        k = active[i]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += w[i] * 2.0 * a[k - idx]
    return grad, b, max_b


def subgradient_optimize(a0, time_limit, top_m=30):
    """Nesterov-accelerated subgradient with adaptive step size."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    v = np.zeros(n)
    mu = 0.82
    step = 0.004
    no_improve = 0

    for it in range(2000000):
        if time.monotonic() - t0 > time_limit:
            break
        y = normalize(a - step * v)
        grad, b, max_b = compute_subgradient(y, top_m)
        v = mu * v + (1.0 - mu) * grad
        a_new = normalize(y - step * v)
        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        a = a_new
        if no_improve > 150:
            step *= 0.75
            no_improve = 0
        mu = 0.82 * np.exp(-it / 30000.0)
        if it % 10000 == 0:
            print(f"    SG it={it} ({time.monotonic()-t0:.0f}s): score={best_score:.8f} step={step:.6f}")
    return best_a, best_score


def lse_objective(z):
    """LogSumExp-smoothed objective (uses global _lse_T)."""
    n = len(z)
    a = np.exp(z - z.max())
    a = a / a.sum()
    b = fast_convolve(a)
    max_b = b.max()
    lse = max_b + np.log(np.sum(np.exp(b - max_b)))
    return 2.0 * n * lse / _lse_T


def lse_gradient(z):
    """Gradient of LogSumExp-smoothed objective."""
    n = len(z)
    a = np.exp(z - z.max())
    a = a / a.sum()
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp(b - max_b)
    w = exp_b / exp_b.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-13:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]
    grad_a *= 2.0 / _lse_T
    S = np.dot(grad_a, a)
    return a * (grad_a - S)


def lbgfs_lse_optimize(a0, time_limit, T_stages=None):
    """Multi-stage LogSumExp L-BFGS optimization."""
    global _lse_T
    z = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()
    if T_stages is None:
        T_stages = [10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.01]
    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        _lse_T = T
        maxiter = max(20, int(remaining * 20))
        try:
            res = minimize(lse_objective, z, jac=lse_gradient, method='L-BFGS-B',
                           options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                                    'maxfun': maxiter * 200})
            z = res.x
        except Exception:
            pass
    a_out = np.exp(z - z.max())
    a_out = a_out / a_out.sum()
    return a_out


def freq_smooth(a, sigma=0.5):
    """Gaussian smooth in frequency domain (reduces high-freq oscillations)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    freqs = np.linspace(0, fft_size / 2, fft_size // 2 + 1)
    center = fft_size / 4.0
    gauss = np.exp(-(freqs - center) ** 2 / (2 * sigma * sigma))
    A_smooth = A * gauss
    a_smooth = irfft(A_smooth, fft_size)[:size]
    # Also take the original length portion
    a_smooth = a_smooth[:n]
    a_smooth = np.maximum(a_smooth, 0)
    s = a_smooth.sum()
    if s < 1e-15:
        return a.copy()
    return a_smooth / s


def ramp_init(n, skew=1.0):
    """Asymmetric power-law ramp."""
    x = np.linspace(0, 1, n)
    a = x ** skew + 0.05 * (1 - x) ** (1.0 + skew)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def two_bump_init(n, c1=0.3, c2=0.7, w1=1.0, w2=0.5, s1=15.0, s2=12.0):
    """Two asymmetric Gaussian bumps."""
    x = np.linspace(0, 1, n)
    a = w1 * np.exp(-s1 * (x - c1) ** 2) + w2 * np.exp(-s2 * (x - c2) ** 2)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def cos_skew_init(n, freq=3, phase=0.3, base=0.7):
    """Cosine-modulated asymmetric base."""
    x = np.linspace(0, 1, n)
    a = (1 - x) ** base * (1.0 + 0.4 * np.cos(2.0 * np.pi * freq * x + phase))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def exp_decay_init(n, rate=2.0):
    """Exponential decay from one end."""
    x = np.linspace(0, 1, n)
    a = np.exp(-rate * x) + 0.1
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a_norm = a / a.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f = np.interp(new_x, old_x, a_norm)
    f = np.maximum(f, 0)
    s = f.sum()
    if s > 1e-15:
        f = f / s
    else:
        f = np.ones(new_n) / new_n
    return f


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    Phase 1: LP Nash equilibrium at multiple lengths
    Phase 2: Direct subgradient refinement
    Phase 3: LogSumExp L-BFGS polishing
    Phase 4: Multi-start diversity with asymmetric inits
    Phase 5: Frequency-domain perturbation + short LP
    Phase 6: Final tight polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = eval_objective(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        parent = None

    if best_seq is None:
        best_seq = ramp_init(2000, skew=1.2)
        best_score = eval_objective(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    def do_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_objective(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: LP at best length ===
    n_best = len(best_seq)
    print(f"\n=== Phase 1: Primary LP n={n_best}, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.35, 500.0)
    if t1 > 40:
        print(f"  Budget: {t1:.0f}s")
        a_lp, _ = nesterov_lp_optimize(best_seq, t1, solver=solver)
        do_update(a_lp, "lp_main")

    # Try a larger length
    if time.monotonic() < deadline - 120:
        n_large = max(n_best + 500, 2500)
        n_large = min(n_large, 3500)
        t_large = min((deadline - time.monotonic()) * 0.20, 250.0)
        if t_large > 40:
            print(f"\n=== Phase 1b: LP at n={n_large} ===")
            init_large = interpolate_seq(best_seq, n_large)
            a_large, _ = nesterov_lp_optimize(init_large, t_large, solver=solver)
            do_update(a_large, f"lp_n{n_large}")

    # === Phase 2: Subgradient refinement ===
    print(f"\n=== Phase 2: Subgradient, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.18, 200.0)
    if t2 > 20:
        a_sg, _ = subgradient_optimize(best_seq, t2, top_m=30)
        do_update(a_sg, "subgrad")

    # === Phase 3: LogSumExp L-BFGS ===
    print(f"\n=== Phase 3: L-BFGS, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.22, 280.0)
    if t3 > 20:
        T_stages = [8.0, 4.0, 2.0, 1.0, 0.6, 0.4, 0.25, 0.15, 0.1, 0.06, 0.04, 0.02, 0.01]
        a_lb = lbgfs_lse_optimize(best_seq, t3, T_stages=T_stages)
        do_update(a_lb, "lbgfs_main")

    # === Phase 4: Multi-start diversity ===
    print(f"\n=== Phase 4: Multi-start, score={best_score:.8f} ===")
    n_cur = len(best_seq)
    rem = deadline - time.monotonic()
    if rem > 80:
        t_each = min(rem / 5.0, 55.0)
        inits = [
            ramp_init(n_cur, skew=rng.uniform(0.3, 0.7)),
            two_bump_init(n_cur, c1=rng.uniform(0.2, 0.35),
                          c2=rng.uniform(0.65, 0.8),
                          w1=1.0, w2=rng.uniform(0.3, 0.6)),
            cos_skew_init(n_cur, freq=rng.integers(2, 7),
                          phase=rng.uniform(0, 6.0), base=rng.uniform(0.5, 1.0)),
            exp_decay_init(n_cur, rate=rng.uniform(1.0, 4.0)),
        ]
        for i, a_init in enumerate(inits):
            if time.monotonic() > deadline - 30:
                break
            if t_each < 15:
                break
            print(f"  Start {i+1}/4, budget={t_each:.0f}s")
            a_res, _ = nesterov_lp_optimize(a_init, t_each, solver=solver)
            do_update(a_res, f"start_{i}")

    # === Phase 5: Frequency-domain perturbation ===
    print(f"\n=== Phase 5: Freq perturb, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 40 and best_seq is not None:
        t_freq = min(rem * 0.15, 60.0)
        if t_freq > 15:
            a_freq = freq_smooth(best_seq, sigma=rng.uniform(0.3, 1.0))
            # Short LP on the smoothed version
            a_fres, _ = nesterov_lp_optimize(a_freq, t_freq, solver=solver)
            do_update(a_fres, "freq_smooth")

    # === Phase 6: Final tight polish ===
    print(f"\n=== Phase 6: Final Polish, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 20 and best_seq is not None:
        t_fin = min(rem * 0.5, 80.0)
        if t_fin > 10:
            a_fin = lbgfs_lse_optimize(best_seq, t_fin,
                                        T_stages=[3.0, 1.5, 0.8, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.01, 0.005])
            do_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 15:
        a_sg2, _ = subgradient_optimize(best_seq, min(rem * 0.5, 30.0), top_m=20)
        do_update(a_sg2, "final_sg")

    rem = deadline - time.monotonic()
    if rem > 10 and best_seq is not None:
        a_fin2 = lbgfs_lse_optimize(best_seq, min(rem * 0.6, 15.0),
                                     T_stages=[1.0, 0.5, 0.3, 0.1, 0.05, 0.02, 0.01])
        do_update(a_fin2, "final_lbgfs2")

    # Final verification
    if best_seq is not None:
        fs = eval_objective(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
