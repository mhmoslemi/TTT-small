def fast_convolve(a):

    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Shaped initializations (linear, exponential, cosine ramps)
2. LP-based best-response alternating with top-K peak constraints
3. Pair-based coordinate descent targeting dominant lags

5. Fine coordinate polishing

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_of(a):
    """Compute 2*n*max(conv)/(sum)^2 quickly."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def find_lp_solver():
    """Find the fastest working LP solver."""
    cached = getattr(find_lp_solver, '_cached_solver', None)
    if cached is not None:
        return cached
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                find_lp_solver._cached_solver = solver_name
                return solver_name
        except Exception:
            continue
    find_lp_solver._cached_solver = cp.ECOS
    return cp.ECOS


def lp_best_response(a, top_k=300, solver=None):
    """
    Solve reduced LP: min M s.t. (a*b)[k] <= M for top-K lags, sum(b)=1, b>=0.
    """
    if solver is None:
        solver = find_lp_solver()
    n = len(a)
    conv = fast_convolve(a)
    K = min(top_k, 2 * n - 1)
    peak_lags = np.argpartition(conv, -K)[-K:]
    
    T = np.zeros((K, n))
    for idx, k in enumerate(peak_lags):
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            T[idx, i] = a[k - i]
    
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    ones = np.ones(K)
    constraints = [
        T @ b_var <= M_var * ones,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            return a / a.sum(), float('inf')
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a / a.sum(), float('inf')
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return a / a.sum(), float('inf')
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else float('inf')
    return b_val, M_val


def lp_alternating(a0, time_limit, solver, top_k=300, verbose=True):
    """LP-based best-response alternating minimization with reduced constraints."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    it = 0
    stagnation = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        
        b_new, M_lp = lp_best_response(a, top_k=top_k, solver=solver)
        if not np.isfinite(M_lp):
            break
        
        score_b = score_of(b_new)
        if score_b < best_score:
            best_score = score_b
            best_a = b_new.copy()
        
        if it <= 15:
            alpha = 1.0
        elif it <= 40:
            alpha = 0.7
        else:
            alpha = max(0.2, 10.0 / (it + 10.0))
        
        a = (1 - alpha) * a + alpha * b_new
        a = np.maximum(a, 0)
        a_sum = a.sum()
        if a_sum < 1e-15:
            break
        a = a / a_sum
        
        score_a = score_of(a)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        if score_a > best_score * (1 + 1e-5):
            stagnation += 1
        else:
            stagnation = max(0, stagnation - 1)
        
        if stagnation > 15:
            rng = np.random.default_rng(it)
            noise_scale = 0.03 / np.sqrt(n)
            a_pert = a + rng.standard_normal(n) * noise_scale
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            score_pert = score_of(a_pert)
            if score_pert < best_score * 1.01:
                a = a_pert
            stagnation = 0
        
        if verbose and it % 15 == 0:
            print(f"    LP it={it} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def coord_descent(a0, time_limit, verbose=True, fine=False):
    """Adaptive coordinate descent with incremental convolution updates."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0
    
    if fine:
        factors = [0.995, 0.998, 0.997, 0.999, 1.001, 1.003, 1.002, 1.005]
        tol = 1e-15
        max_no_improve = 30
    else:
        factors = [-0.5, -0.25, -0.1, -0.03, 0.03, 0.1, 0.25, 0.5]
        tol = 1e-14
        max_no_improve = 20
    
    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        
        for i in range(n):
            if a[i] < 1e-16:
                continue
            
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            
            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)
            
            a_shifted = a[k_start - i:k_end - i + 1]
            conv_affected = conv[k_start:k_end + 1]
            
            best_delta = 0.0
            best_new_M = M
            
            for factor in factors:
                if fine:
                    delta = a[i] * (factor - 1.0)
                else:
                    base = max(a[i], 1e-8)
                    delta = base * factor
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue
                
                new_affected = conv_affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta
                
                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - tol:
                    best_new_M = new_M
                    best_delta = delta
            
            if best_delta != 0.0 and best_new_M < M - tol:
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > max_no_improve:
                break
        else:
            no_improve = 0
        
        if verbose and sweeps % 20 == 0:
            print(f"    CD sweep={sweeps} ({time.monotonic()-t0:.0f}s): best={best_score:.8f} M={M:.10f}")
    
    return best_a, best_score


def subgradient_descent(a0, time_limit, lr_init=0.015, top_k=30, verbose=True):
    """Subgradient descent with multi-peak weighting and simplex projection."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        M = conv.max()
        score = 2.0 * n * M
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 100:
            lr *= 0.8
            no_improve = 0
            if lr < 1e-10:
                break
        
        k_act = min(top_k, 2 * n - 1)
        if k_act >= 2 * n - 1:
            peaks = np.arange(2 * n - 1)
            pv = conv.copy()
        else:
            peaks = np.argpartition(conv, -k_act)[-k_act:]
            pv = conv[peaks]
        
        temp = 0.005 * max(M, 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / ws
        
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-13:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            sg[i_start:i_end + 1] += w[idx] * 2.0 * a[k - i_end:k - i_start + 1][::-1]
        
        a_new = a - lr * sg
        a_new = simplex_project(a_new)
        a = a_new
        conv = fast_convolve(a)
        
        if verbose and it % 200 == 0:
            print(f"    SGD it={it} ({time.monotonic()-t0:.0f}s): best={best_score:.8f} lr={lr:.6f}")
    
    return best_a, best_score


def shaped_init(n, shape_type, param, rng=None):
    """Generate shaped initializations."""
    t = np.linspace(0, 1, n)
    if shape_type == 'linear':
        a = 1.0 - param * t
    elif shape_type == 'exp':
        a = np.exp(-param * t)
    elif shape_type == 'cosine':
        a = np.cos(param * np.pi * t / 2.0)
    elif shape_type == 'power':
        a = (1.0 - t) ** param
    elif shape_type == 'quadratic':
        a = 1.0 - param * t * t
    elif shape_type == 'ramp_up':
        a = 0.1 + 0.9 * t
    elif shape_type == 'bimodal':
        c1 = 0.3
        c2 = 0.7
        w = 0.15
        a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    elif shape_type == 'peak_center':
        a = 1.0 - param * np.abs(t - 0.5)
    elif shape_type == 'hann':
        a = 0.5 * (1 + np.cos(2 * np.pi * t))
        a = np.maximum(a, 0.001)
    elif shape_type == 'random_smooth':
        if rng is None:
            rng = np.random.default_rng(0)
        freq = rng.integers(1, 8)
        phase = rng.uniform(0, 2 * np.pi)
        a = 0.5 + 0.4 * np.cos(2 * np.pi * freq * t + phase)
    else:
        a = np.ones(n)
    
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def search_shapes(n, time_budget):
    """Search over shaped initializations to find the best starting point."""
    t0 = time.monotonic()
    best_a = None
    best_score = float('inf')
    
    shapes = [
        ('linear', [0.5, 1.0, 1.5, 1.9]),
        ('exp', [1.0, 2.0, 3.0, 4.0]),
        ('cosine', [0.5, 1.0, 1.5]),
        ('power', [1.0, 2.0, 3.0, 5.0]),
        ('quadratic', [0.5, 1.0, 1.5]),
        ('ramp_up', [1.0]),
        ('bimodal', [1.0]),
        ('peak_center', [0.5, 1.0, 1.5]),
        ('hann', [1.0]),
    ]
    
    for shape_type, params in shapes:
        if time.monotonic() - t0 > time_budget:
            break
        for p in params:
            a = shaped_init(n, shape_type, p)
            sc = score_of(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    # Also try random smooth
    rng = np.random.default_rng(123)
    for _ in range(20):
        if time.monotonic() - t0 > time_budget:
            break
        a = shaped_init(n, 'random_smooth', 0, rng)
        sc = score_of(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    if best_a is None:
        best_a = np.ones(n) / n
        best_score = score_of(best_a)
    
    return best_a, best_score

        print(f"  LP from shape, budget={t_alloc:.0f}s")

        try_update(a_lp2, "lp_shape")
    
    # Second round from best
    if time.monotonic() < t1_end - 40:
        t_alloc = min(t1_end - time.monotonic(), 200.0)
        print(f"  LP round 2, budget={t_alloc:.0f}s")
        a_lp3, s_lp3 = lp_alternating(best_seq, t_alloc, solver, top_k=500, verbose=True)
        try_update(a_lp3, "lp_r2")
    
    # Phase 1b: Try different lengths
    print(f"\n=== Phase 1b: Different lengths (best={best_score:.8f}) ===")
    if time.monotonic() < deadline - 300:
        t_alloc = min(deadline - time.monotonic() - 300, 120.0)
        best_seq, best_score = try_different_lengths(n, t_alloc, best_seq, best_score)
    
    # Phase 2: Subgradient Descent
    print(f"\n=== Phase 2: SGD (best={best_score:.8f}) ===")
    t2_end = time.monotonic() + (deadline - time.monotonic()) * 0.15
    
    if time.monotonic() < t2_end:
        t_alloc = min(t2_end - time.monotonic(), 200.0)
        print(f"  SGD budget={t_alloc:.0f}s")
        a_sgd, s_sgd = subgradient_descent(best_seq, t_alloc, lr_init=0.01, top_k=40, verbose=True)
        try_update(a_sgd, "sgd")
        
        # Also from LP result
        if time.monotonic() < t2_end - 30:
            a_sgd2, s_sgd2 = subgradient_descent(a_lp, min(t2_end - time.monotonic(), 80.0),
                                                  lr_init=0.005, top_k=60, verbose=False)
            try_update(a_sgd2, "sgd_lp")
    
    # Phase 3: Coordinate Descent
    print(f"\n=== Phase 3: Coordinate Descent (best={best_score:.8f}) ===")
    t3_end = time.monotonic() + (deadline - time.monotonic()) * 0.20
    
    if time.monotonic() < t3_end:
        t_alloc = min(t3_end - time.monotonic(), 300.0)
        print(f"  CD main budget={t_alloc:.0f}s")
        a_cd, s_cd = coord_descent(best_seq, t_alloc, verbose=True, fine=False)
        try_update(a_cd, "cd_main")
        
        if time.monotonic() < t3_end - 40:
            t_alloc2 = min(t3_end - time.monotonic(), 120.0)
            a_cd2, s_cd2 = coord_descent(a_lp, t_alloc2, verbose=True, fine=False)
            try_update(a_cd2, "cd_lp")
    
    # Phase 4: Fine Polishing
    print(f"\n=== Phase 4: Fine Polish (best={best_score:.8f}) ===")
    t4_end = deadline - 20
    
    if time.monotonic() < t4_end - 30:
        t_fp = min(t4_end - time.monotonic() - 30, 120.0)
        print(f"  Fine CD budget={t_fp:.0f}s")
        a_fp, s_fp = coord_descent(best_seq, t_fp, verbose=True, fine=True)
        try_update(a_fp, "fine")
    
    # Final CD pass
    if time.monotonic() < t4_end - 5:
        t_fin = min(t4_end - time.monotonic() - 5, 30.0)
        if t_fin > 5:
            a_fin, s_fin = coord_descent(best_seq, t_fin, verbose=False, fine=True)
            try_update(a_fin, "final")
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
