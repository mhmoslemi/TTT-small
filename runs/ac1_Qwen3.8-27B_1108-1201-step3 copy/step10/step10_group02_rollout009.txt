    global _lp_solver_cached
    if _lp_solver_cached is not None:
        return _lp_solver_cached
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return _lp_solver_cached


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solved = False
    try:
        prob.solve(solver=solver)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            solved = True
    except Exception:
        pass
    if not solved:
        for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
                    break
            except Exception:
                continue
    if not solved:
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def pressure_subgradient(a, conv, target=None, max_active=150):
    """Compute pressure-weighted subgradient of max(conv)."""
    n = len(a)
    if target is None:
        target = conv.mean()
    pressure = np.maximum(conv - target, 0)
    total = pressure.sum()
    if total < 1e-30:
        k_max = int(np.argmax(conv))
        js = max(0, k_max - n + 1)
        je = min(n - 1, k_max)
        idx = np.arange(js, je + 1)
        g = np.zeros(n)
        g[idx] = 2.0 * a[k_max - idx]
        return g
    thresh = max(pressure.max() * 0.005, 1e-15)
    active = np.where(pressure > thresh)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(conv))])
    if len(active) > max_active:
        top_idx = active[np.argsort(pressure[active])[::-1][:max_active]]
    else:
        top_idx = active
    g = np.zeros(n)
    for k in top_idx:
        w = pressure[k] / total
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += 2.0 * w * a[k - idx]
    return g


def mirror_descent_pressure(a0, time_limit, eta0=0.05):
    """Mirror descent with pressure-weighted subgradient and BB step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = score(a)
    t0 = time.monotonic()
    eta = eta0
    no_improve = 0
    prev_a = a.copy()
    prev_g = None

    for it in range(3000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_convolve(y)
        max_conv = conv_y.max()

        g = pressure_subgradient(y, conv_y)

        if it >= 2 and prev_g is not None:
            s_vec = a - prev_a
            y_vec = g - prev_g
            s_dot_s = np.dot(s_vec, s_vec)
            s_dot_y = np.dot(s_vec, y_vec)
            y_dot_y = np.dot(y_vec, y_vec)
            if s_dot_s > 1e-18 and y_dot_y > 1e-18:
                if abs(s_dot_y) > 1e-18:
                    eta_bb = s_dot_s / abs(s_dot_y)
                else:
                    eta_bb = s_dot_s / y_dot_y
                eta = 0.6 * eta + 0.4 * eta_bb
                eta = max(1e-6, min(eta, 0.5))
            else:
                eta *= 0.995

        prev_a = a.copy()
        prev_g = g.copy()

        log_a = np.log(np.maximum(a, 1e-300))
        g_shifted = g - g.min()
        log_a_new = log_a - eta * g_shifted
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new = a_new / a_new.sum()

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * a_new
        y = np.maximum(y, 0)
        ys = y.sum()
        if ys < 1e-15:
            y = a_new.copy()
        else:
            y = y / ys

        a = a_new

        sc = score(a)
        sc_y = 2.0 * n * conv_y.max()
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a.copy()
            no_improve = 0
        elif sc_y < best_score - 1e-13:
            best_score = sc_y
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 300:
            rng = np.random.default_rng(it * 31 + 7)
            noise_scale = 0.003 * np.sqrt(min(no_improve / 300, 10.0))
            noise = rng.standard_normal(n) * noise_scale
            a = a + noise
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            y = a.copy()
            no_improve = 0
            eta = max(eta, eta0 * 0.5)

        if (it + 1) % 5000 == 0:
            print(f"  MD iter={it+1} ({time.monotonic()-t0:.0f}s): best={best_score:.8f} eta={eta:.6f}")

    return best_a, best_score


def nesterov_lp_refine(a0, time_limit, solver=None):
    """Nesterov-accelerated LP best-response alternating."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    if a.sum() < 1e-15:
        a = np.ones(n) / n
    a = a / a.sum()
    y = a.copy()
    best_a = a.copy()
    best_score = score(a)
    t0 = time.monotonic()
    stall = 0
    prev_best = best_score
    rng = np.random.default_rng(12345)

    for it in range(200000):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        y = y / y.sum()

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        a = a / a.sum()

        sc_a = score(a)
        sc_y = 2.0 * n * fast_convolve(y).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        if (it + 1) % 50 == 0:
            if best_score < prev_best - 1e-11:
                stall = 0
            else:
                stall += 1
            prev_best = best_score
            if stall >= 15:
                freq = rng.uniform(1, 15)
                pert = np.sin(2 * np.pi * freq * np.arange(n) / n) * (0.008 / np.sqrt(stall))
                y = y + pert
                y = np.maximum(y, 1e-10)
                y = y / y.sum()
                stall = 0

        if (it + 1) % 500 == 0:
            print(f"  NP-LP iter={it+1} ({time.monotonic()-t0:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def softmax_safe(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def lse_obj_grad(z, T):
    """Compute log-sum-exp smoothed objective and gradient in z-space."""
    a = softmax_safe(z)
    n = len(a)
    conv = fast_convolve(a)
    exp_conv = np.exp(conv / T - conv.max() / T)
    w = exp_conv / exp_conv.sum()
    logsumexp = conv.max() / T + np.log(exp_conv.sum())
    obj = (2.0 * n) * T * logsumexp

    grad_a = np.zeros(n)
    size = 2 * n - 1
    for k in range(size):
        if w[k] < 1e-14:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lse_lbgfs(a0, time_limit, T_stages=None):
    """L-BFGS on log-sum-exp smoothed objective with decreasing temperature."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [30.0, 15.0, 8.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02]

    iters_per_stage = max(30, 8000 // len(T_stages))

    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        if time.monotonic() - t0 > time_limit - 5:
            break

        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            obj, _ = lse_obj_grad(z_in, T)
            return obj

        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            _, grad = lse_obj_grad(z_in, T)
            return grad

        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                                    'gtol': 1e-14, 'maxfun': iters_per_stage * 200,
                                    'disp': False})
            z = res.x
        except Exception:
            pass

    return softmax_safe(z)


def coordinate_descent_intensive(a0, time_limit, n_passes=500000):
    """Intensive coordinate descent with 1D line search on peak-contributing coordinates."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score(a)
    t0 = time.monotonic()
    stall = 0
    rng = np.random.default_rng(777)
    multipliers_coarse = [0.999, 0.998, 0.995, 0.99, 0.985, 0.98, 1.001, 1.002, 1.005, 1.01]
    multipliers_fine = [0.9997, 0.9995, 0.999, 0.998, 0.997, 0.995,
                        1.0003, 1.0005, 1.001, 1.002, 1.003, 1.005]

    for p in range(n_passes):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(a)
        max_conv = conv.max()
        peak_k = int(np.argmax(conv))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

        top_m = min(40, len(idx))
        order = np.argsort(contribs)[::-1][:top_m]
        top_positions = idx[order]

        n_extra = min(15, n - len(top_positions))
        if n_extra > 0:
            candidates = np.setdiff1d(np.arange(n), top_positions)
            extra = rng.choice(candidates, size=n_extra, replace=False)
            top_positions = np.concatenate([top_positions, extra])

        improved = False
        muls = multipliers_fine if stall > 50 else multipliers_coarse

        for pos in top_positions:
            if time.monotonic() - t0 > time_limit:
                break
            if a[pos] < 1e-12:
                continue

            best_mult = 1.0
            best_sc = best_score
            for mult in muls:
                a_trial = a.copy()
                a_trial[pos] = a[pos] * mult
                if a_trial[pos] < 0:
                    continue
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = score(a_trial)
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_mult = mult

            if best_mult != 1.0:
                a[pos] *= best_mult
                a = a / a.sum()
                if best_sc < best_score:
                    best_score = best_sc
                    best_a = a.copy()
                    improved = True

        if not improved:
            stall += 1
            if stall > 100:
                freq = rng.uniform(0.5, 25)
                amp = 0.004 / np.sqrt(min(stall / 100, 20.0))
                pert = np.sin(2 * np.pi * freq * np.arange(n) / n) * amp
                a = a + pert
                a = np.maximum(a, 1e-10)
                a = a / a.sum()
                stall = 0
                sc = score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
        else:
            stall = max(0, stall - 2)

        if (p + 1) % 2000 == 0:
            print(f"  CD pass={p+1} ({time.monotonic()-t0:.0f}s): score={best_score:.8f} stall={stall}")

    return best_a, best_score


def pairwise_coordinate_descent(a0, time_limit, n_passes=50000):
    """Pairwise coordinate descent: adjust pairs of coordinates simultaneously."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(444)
    stall = 0

    for p in range(n_passes):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(a)
        peak_k = int(np.argmax(conv))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_positions = idx[np.argsort(contribs)[::-1][:8]]

        improved = False
        for i in range(len(top_positions)):
            for j in range(i + 1, len(top_positions)):
                pos1 = top_positions[i]
                pos2 = top_positions[j]
                if a[pos1] < 1e-12 or a[pos2] < 1e-12:
                    continue
                for f1 in [0.995, 0.98, 0.95]:
                    for f2 in [1.01, 1.02, 1.05]:
                        a_trial = a.copy()
                        a_trial[pos1] = a[pos1] * f1
                        a_trial[pos2] = a[pos2] * f2
                        ss = a_trial.sum()
                        if ss < 1e-15:
                            continue
                        a_trial = a_trial / ss
                        sc = score(a_trial)
                        if sc < best_score - 1e-14:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 200:
                break
        else:
            stall = 0

        if (p + 1) % 5000 == 0:
            print(f"  PCD pass={p+1} ({time.monotonic()-t0:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = score(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        x = np.linspace(0, 1, n_init)
        best_seq = np.cos(np.pi * x / 2) ** 2
        best_seq = np.maximum(best_seq, 0)
        best_seq = best_seq / best_seq.sum()
        best_score = score(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Mirror Descent with pressure weighting ===
    print(f"\n=== Phase 1: Pressure Mirror Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_md, _ = mirror_descent_pressure(best_seq, t_alloc, eta0=0.04)
        try_update(a_md, "MD_pressure")

    # === Phase 2: LP Best-Response with Nesterov ===
    print(f"\n=== Phase 2: Nesterov LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver)
        try_update(a_lp, "nesterov_lp")

    # === Phase 3: Log-Sum-Exp L-BFGS ===
    print(f"\n=== Phase 3: LSE L-BFGS (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 20:
        T_stages = [20.0, 10.0, 6.0, 4.0, 2.5, 1.5, 1.0, 0.6, 0.4, 0.25, 0.15, 0.1, 0.05, 0.02, 0.01]
        print(f"  Budget: {t_alloc:.0f}s")
        a_lse = lse_lbgfs(best_seq, t_alloc, T_stages=T_stages)
        try_update(a_lse, "lse_lbgfs")

    # === Phase 4: Intensive Coordinate Descent ===
    print(f"\n=== Phase 4: Intensive Coord Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 500.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_cd, _ = coordinate_descent_intensive(best_seq, t_alloc, n_passes=500000)
        try_update(a_cd, "coord_desc")

    # === Phase 5: Pairwise Coordinate Descent ===
    print(f"\n=== Phase 5: Pairwise Coord Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.06, 60.0)
    if t_alloc > 10:
        a_pcd, _ = pairwise_coordinate_descent(best_seq, t_alloc, n_passes=50000)
        try_update(a_pcd, "pairwise_cd")

    # === Phase 6: Final Mirror Descent polish (very low eta) ===
    print(f"\n=== Phase 6: Final MD Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.02, 20.0)
    if t_alloc > 5:
        a_md2, _ = mirror_descent_pressure(best_seq, t_alloc, eta0=0.005)
        try_update(a_md2, "MD_polish")

    # === Phase 7: Final Coord Descent polish ===
    print(f"\n=== Phase 7: Final CD Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = rem
    if t_alloc > 5:
        a_final, _ = coordinate_descent_intensive(best_seq, t_alloc, n_passes=200000)
        try_update(a_final, "final_cd")

    # Final verification
    if best_seq is not None:
        fs = score(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
