def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    if n == 0:

                y = y / ys
        else:
            y = a.copy()

        conv_y = fast_convolve(y)
        M_y = float(conv_y.max())

        # Find top-K active peaks
        K = min(30, 2 * n - 1)
        peaks = np.argpartition(conv_y, -K)[-K:]
        peak_vals = conv_y[peaks]
        threshold = peak_vals.min() * 0.95
        active_peaks = peaks[peak_vals >= threshold]

        # Softmax weighting
        temp = 0.005 * max(M_y, 1e-10)
        w = np.exp((conv_y[active_peaks] - conv_y[active_peaks].max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(active_peaks)) / len(active_peaks)
        else:
            w = w / ws

        # Compute subgradient
        sg = np.zeros(n)
        for idx in range(len(active_peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(active_peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            sg[i_start:i_end + 1] += w[idx] * 2.0 * y[k - i_end:k - i_start + 1][::-1]

        # Update
        a_new = a - eta * sg
        a_new = simplex_project(a_new)

        # Evaluate
        conv_new = fast_convolve(a_new)
        M_new = float(conv_new.max())
        cur_score = 2.0 * n * M_new

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_new.copy()

        # Update momentum
        momentum += 1.0
        prev_a = a.copy()
        a = a_new
        conv = conv_new

        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            if verbose:
                print(f"    NSG it={it} ({elapsed:.0f}s): best={best_score:.8f} M={M_new:.10f}")

        if M_y < best_M * (1 - 1e-8) and it > 100:
            # Check stagnation
            pass

    return best_a, best_score


def mirror_descent_lp(a0, time_limit, solver, verbose=True):
    """
    LP best-response with exponentiated gradient (mirror descent) updates.
    This naturally stays in the simplex and can be more effective than Euclidean updates.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a + 1e-10  # ensure strictly positive for log
    s = a.sum()
    a = a / s

    best_a = a.copy()
    conv = fast_convolve(a)
    best_M = float(conv.max())
    best_score = 2.0 * n * best_M

    t0 = time.monotonic()
    it = 0
    stagnation = 0
    last_best = best_score

    while time.monotonic() - t0 < time_limit:
        it += 1
        if it > 3000:
            break

        b_new, M_lp, active = lp_best_response(a, solver)
        if not np.isfinite(M_lp):
            break

        # Multi-alpha line search
        alphas = [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1, 0.05]
        best_alpha = 0.0
        best_f = float('inf')

        for alpha in alphas:
            if alpha == 1.0:
                candidate = np.maximum(b_new, 0.0)
            else:
                candidate = (1 - alpha) * a + alpha * b_new
                candidate = np.maximum(candidate, 0.0)
            cs = candidate.sum()
            if cs < 1e-15:
                continue
            candidate = candidate / cs
            f_candidate = float(fast_convolve(candidate).max())
            if f_candidate < best_f:
                best_f = f_candidate
                best_alpha = alpha

        if best_alpha > 0:
            if best_alpha == 1.0:
                a = np.maximum(b_new, 0.0)
            else:
                a = (1 - best_alpha) * a + best_alpha * b_new
                a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = np.ones(n) / n
            else:
                a = a / s

            conv = fast_convolve(a)
            M_cur = float(conv.max())
            cur_score = 2.0 * n * M_cur

            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                stagnation = 0
            else:
                stagnation += 1
        else:
            stagnation += 1

        # Stagnation handling
        if stagnation > 30:
            # Try perturbation
            noise_scale = 0.02 / np.sqrt(n)
            a_pert = a + np.random.standard_normal(n) * noise_scale
            a_pert = np.maximum(a_pert, 0.0)
            ps = a_pert.sum()
            if ps > 1e-15:
                a_pert = a_pert / ps
                sp = float(fast_convolve(a_pert).max()) * 2.0 * n
                if sp < best_score * 1.001:
                    a = a_pert
                    stagnation = 0

            if stagnation > 100:
                # More aggressive restart
                noise_scale = 0.05 / np.sqrt(n)
                a_pert = a + np.random.standard_normal(n) * noise_scale
                a_pert = np.maximum(a_pert, 0.0)
                ps = a_pert.sum()
                if ps > 1e-15:
                    a_pert = a_pert / ps
                    sp = float(fast_convolve(a_pert).max()) * 2.0 * n
                    if sp < best_score:
                        a = a_pert
                        best_a = a.copy()
                        best_score = sp
                        best_M = float(fast_convolve(a).max())
                    stagnation = 0

                if stagnation > 200:
                    break

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MDSL it={it} ({elapsed:.0f}s): best={best_score:.8f} alpha={best_alpha:.2f}")

    return best_a, best_score


def coordinate_descent_golden(a0, time_limit, verbose=True):
    """Coordinate descent with golden section line search per coordinate."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    # Precompute affected ranges
    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-16:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            a_shifted = a[max(0, k_start - i):k_end - i + 1]
            conv_affected = conv[k_start:k_end + 1].copy()

            # Golden section search for best value of a[i]
            # Try a range of values
            lo = 0.0
            hi = min(2.0 * a[i] + 0.001, 1.0)

            # First try a few fixed factors
            best_val = a[i]
            best_new_M = M

            factors = [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0]
            for fac in factors:
                new_val = a[i] * fac
                if new_val < 0 or new_val > 1.0:
                    continue
                delta = new_val - a[i]
                new_aff = conv_affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_aff = new_aff.copy()
                    new_aff[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta
                new_M = max(unaff_max, float(new_aff.max()))
                if new_M < best_new_M - 1e-14:
                    best_new_M = new_M
                    best_val = new_val

            if best_val != a[i] and best_new_M < M - 1e-14:
                delta = best_val - a[i]
                a[i] = max(best_val, 0.0)
                changes = 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * delta * (a[i] - delta) + delta * delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def fine_polish(a0, time_limit, verbose=True):
    """Very fine coordinate descent for last-mile improvements."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-16:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            a_shifted = a[k_start - i:k_end - i + 1]
            conv_affected = conv[k_start:k_end + 1]

            best_delta = 0.0
            best_new_M = M

            for factor in [0.999, 0.997, 0.995, 0.998, 1.002, 1.004, 1.006, 1.003]:
                delta = a[i] * (factor - 1.0)
                if abs(delta) < 1e-18:
                    continue
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = conv_affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-15:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-15:
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * delta * np.sign(best_delta)
                    # Fix: the quadratic term should be delta^2
                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 15:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FinePolish sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def diverse_init(n, init_type, rng):
    """Generate diverse initializations."""
    if init_type == 0:
        t = np.arange(n)
        center = (n - 1) / 2.0
        a = np.maximum(1.0 - np.abs(t - center) / center, 0.01)
    elif init_type == 1:
        t = np.arange(n, dtype=np.float64)
        center = (n - 1) / 2.0
        x = (t - center) / center
        a = 0.3 + 0.7 * x * x
    elif init_type == 2:
        t = np.arange(n, dtype=np.float64)
        center = (n - 1) / 2.0
        x = (t - center) / center
        power = rng.uniform(0.3, 1.0)
        a = np.maximum(0.5 * (1 + np.cos(np.pi * x)), 0.01) ** power
    elif init_type == 3:
        t = np.arange(n, dtype=np.float64)
        c1 = n * rng.uniform(0.15, 0.3)
        c2 = n * rng.uniform(0.7, 0.85)
        w = n * rng.uniform(0.05, 0.15)
        a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    elif init_type == 4:
        # Edge-heavy
        t = np.arange(n, dtype=np.float64)
        center = (n - 1) / 2.0
        x = (t - center) / center
        a = 1.0 / (0.1 + np.abs(x) ** 0.5)
    elif init_type == 5:
        # U-shape with tuning
        t = np.arange(n, dtype=np.float64)
        center = (n - 1) / 2.0
        x = (t - center) / center
        p = rng.uniform(1.5, 3.0)
        a = np.abs(x) ** p + 0.05
    else:
        t = np.arange(n, dtype=np.float64)
        center = (n - 1) / 2.0
        x = (t - center) / center
        a = 1.0 / (1.0 + (rng.uniform(0.5, 5.0) * x) ** 2)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def propose_candidate():
    """
    Multi-phase optimization to minimize 2n*max(auto_conv)/(sum)^2.
    Phase 1: LP best-response with line search (main driver)
    Phase 2: Nesterov subgradient descent
    Phase 3: Coordinate descent + fine polish
    """
    deadline = time.monotonic() + 1790.0
    rng = np.random.default_rng(42)

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is not None:
        n = len(parent)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    else:
        n = 1024
        best_seq = np.ones(n) / n
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default init: n={n}, score={best_score:.8f}")

    solver = find_lp_solver()
    print(f"LP solver: {solver}")

    def try_update(candidate, label):
        nonlocal best_seq, best_score
        candidate = np.maximum(candidate, 0.0)
        s = candidate.sum()
        if s < 0.01:
            return
        candidate = candidate / s
        try:
            sc = float(evaluate_sequence(candidate.tolist()))
        except Exception:
            sc = float('inf')
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = candidate.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(candidate)})")

    # === Phase 1: LP best-response with line search ===
    print(f"\n=== Phase 1: LP Best-Response (best={best_score:.8f}) ===")
    t1_budget = (deadline - time.monotonic()) * 0.50
    t1_end = time.monotonic() + t1_budget

    # Run from parent
    t_alloc = min(t1_end - time.monotonic(), 500.0)
    print(f"  LP from parent, budget={t_alloc:.0f}s")
    a_lp, s_lp = mirror_descent_lp(best_seq, t_alloc, solver, verbose=True)
    try_update(a_lp, "lp_parent")

    # Run from diverse initializations
    inits = [
        diverse_init(n, rng, 4),  # Edge-heavy
        diverse_init(n, rng, 5),  # U-shape tuned
        diverse_init(n, rng, 1),  # U-shape
        diverse_init(n, rng, 2),  # Cosine
    ]

    for idx, init in enumerate(inits):
        if time.monotonic() > t1_end - 30:
            break
        remaining = t1_end - time.monotonic()
        if remaining < 60:
            break
        t_alloc = min(remaining * 0.3, 150.0)
        print(f"  LP init {idx+1}/{len(inits)}, budget={t_alloc:.0f}s")
        a_tmp, s_tmp = mirror_descent_lp(init, t_alloc, solver, verbose=False)
        try_update(a_tmp, f"lp_init{idx+1}")

    # Second LP round on best
    if time.monotonic() < t1_end - 60:
        t_alloc = min(t1_end - time.monotonic(), 300.0)
        print(f"  LP round 2, budget={t_alloc:.0f}s")
        a_lp2, s_lp2 = mirror_descent_lp(best_seq, t_alloc, solver, verbose=True)
        try_update(a_lp2, "lp_round2")

    # === Phase 2: Nesterov Subgradient ===
    print(f"\n=== Phase 2: Nesterov Subgradient (best={best_score:.8f}) ===")
    t2_budget = (deadline - time.monotonic()) * 0.20
    t2_end = time.monotonic() + t2_budget

    t_alloc = min(t2_end - time.monotonic(), 200.0)
    if t_alloc > 30:
        print(f"  NSG budget={t_alloc:.0f}s")
        a_nsg, s_nsg = subgradient_with_nesterov(best_seq, t_alloc, verbose=True)
        try_update(a_nsg, "nesterov")

    # === Phase 3: Coordinate Descent ===
    print(f"\n=== Phase 3: Coordinate Descent (best={best_score:.8f}) ===")
    t3_budget = (deadline - time.monotonic()) * 0.20
    t3_end = time.monotonic() + t3_budget

    t_alloc = min(t3_end - time.monotonic(), 250.0)
    if t_alloc > 30:
        print(f"  CD budget={t_alloc:.0f}s")
        a_cd, s_cd = coordinate_descent_golden(best_seq, t_alloc, verbose=True)
        try_update(a_cd, "cd")

    # === Phase 4: Fine Polish ===
    print(f"\n=== Phase 4: Fine Polish (best={best_score:.8f}) ===")
    t4_end = deadline - 15

    if time.monotonic() < t4_end - 20:
        t_fp = min(t4_end - time.monotonic() - 10, 150.0)
        if t_fp > 15:
            print(f"  FinePolish budget={t_fp:.0f}s")
            a_fp, s_fp = fine_polish(best_seq, t_fp, verbose=True)
            try_update(a_fp, "fine_pollish")

    # Final CD
    if time.monotonic() < t4_end - 5:
        t_fin = min(t4_end - time.monotonic() - 5, 60.0)
        if t_fin > 10:
            a_fin, s_fin = coordinate_descent_golden(best_seq, t_fin, verbose=False)
            try_update(a_fin, "final_cd")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
