def fast_convolve(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):

    conv = fast_convolve(a)

def golden_section_search(f, lo, hi, max_iter=40):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-15:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5

            a_partners = a[partners]

                if k == 2 * i:

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    deadline = time.monotonic() + 1790.0
    rng = np.random.default_rng(777)

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and p.sum() > 0.01:
            parent = p / p.sum()
    except Exception:
        pass

    if parent is None:
        parent = init_gaussian(1024, rng, 0.3, 0.25)

    best_seq = parent.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"Parent: n={len(parent)}, score={best_score:.8f}")

    # Phase 1: Multi-length screening
    print("\n=== Phase 1: Multi-length Screening ===")
    t1_start = time.monotonic()
    n_parent = len(parent)
    n_candidates = list(set([n_parent, n_parent - 128, n_parent + 128,
                              768, 1024, 1280, 1536, 2048]))
    n_candidates = [n for n in n_candidates if 256 <= n <= 4096]
    n_candidates.sort(key=lambda x: abs(x - n_parent))
    n_candidates = n_candidates[:5]

    screened = []
    for n in n_candidates:
        if time.monotonic() - t1_start > 60:
            break
        t_left = 60.0 - (time.monotonic() - t1_start)
        if t_left < 3:
            break
        t_alloc = min(12.0, t_left / 3.0)
        init = interpolate_seq(parent, n)
        a_opt, s_opt, _ = exact_coord_descent(init, t_alloc, label=f"scr_n{n}")
        screened.append((s_opt, a_opt, n))
        print(f"  Screen n={n}: score={s_opt:.8f} ({t_alloc:.0f}s)")

    screened.sort(key=lambda x: x[0])
    for s_c, a_c, n_c in screened:
        sc = float(evaluate_sequence(a_c.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_c.copy()
            print(f"  NEW BEST screen: {sc:.8f} (n={n_c})")

    print(f"  Phase 1 best: {best_score:.8f} (n={len(best_seq)})")

    # Phase 2: Soft-max L-BFGS (new primary driver)
    print(f"\n=== Phase 2: Soft-max L-BFGS (best={best_score:.8f}) ===")
    n_work = len(best_seq)
    rem = deadline - time.monotonic() - 600
    if rem > 30:
        t_lbfgs = min(rem * 0.5, 300.0)
        print(f"  L-BFGS budget={t_lbfgs:.0f}s, n={n_work}")
        a_lbfgs, s_lbfgs = lbfgs_soft_max(best_seq, t_lbfgs, verbose=True)
        sc = float(evaluate_sequence(a_lbfgs.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_lbfgs.copy()
            print(f"  NEW BEST L-BFGS: {sc:.8f}")

        # Try L-BFGS on the second-best screened candidate too
        if len(screened) > 0 and screened[0][2] != n_work:
            rem2 = deadline - time.monotonic() - 500
            if rem2 > 30:
                t2 = min(rem2 * 0.3, 150.0)
                s_c2, a_c2, n_c2 = screened[0]
                if n_c2 != n_work:
                    a_lbfgs2, s_lbfgs2 = lbfgs_soft_max(a_c2, t2, verbose=True)
                    sc2 = float(evaluate_sequence(a_lbfgs2.tolist()))
                    if np.isfinite(sc2) and sc2 < best_score:
                        best_score = sc2
                        best_seq = a_lbfgs2.copy()
                        print(f"  NEW BEST L-BFGS n={n_c2}: {sc2:.8f}")

    # Phase 3: Coordinate descent polishing
    print(f"\n=== Phase 3: ECD Polishing (best={best_score:.8f}) ===")
    n_work = len(best_seq)
    for rnd in range(4):
        rem = deadline - time.monotonic() - 400
        if rem < 15:
            break
        t_cd = min(rem * 0.2, 120.0)
        print(f"  ECD round {rnd + 1}, budget={t_cd:.0f}s")
        a_cd, s_cd, sw = exact_coord_descent(best_seq, t_cd, verbose=True, label=f"ph3_r{rnd}")
        sc = float(evaluate_sequence(a_cd.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_cd.copy()
            print(f"  NEW BEST ECD: {sc:.8f}")
        else:
            perturbed = best_seq + rng.standard_normal(n_work) * 0.015
            perturbed = np.maximum(perturbed, 0)
            perturbed = perturbed / perturbed.sum()
            a_cd2, s_cd2, _ = exact_coord_descent(perturbed, min(t_cd * 0.5, 60.0))
            sc2 = float(evaluate_sequence(a_cd2.tolist()))
            if np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_seq = a_cd2.copy()
                print(f"  NEW BEST ECD-pert: {sc2:.8f}")

    # Phase 4: Peak-targeted search
    print(f"\n=== Phase 4: Peak-Targeted Search (best={best_score:.8f}) ===")
    n_work = len(best_seq)
    rem = deadline - time.monotonic() - 250
    if rem > 20:
        t_pts = min(rem * 0.6, 180.0)
        print(f"  PTS budget={t_pts:.0f}s")
        a_pts, s_pts = peak_targeted_search(best_seq, t_pts, rng, verbose=True)
        sc = float(evaluate_sequence(a_pts.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_pts.copy()
            print(f"  NEW BEST PTS: {sc:.8f}")

    # Phase 5: Basin hopping
    print(f"\n=== Phase 5: Basin Hopping (best={best_score:.8f}) ===")
    n_work = len(best_seq)
    rem = deadline - time.monotonic() - 150
    if rem > 20:
        t_bh = min(rem * 0.5, 120.0)
        print(f"  BH budget={t_bh:.0f}s")
        a_bh, s_bh = basin_hopping(best_seq, t_bh, rng, verbose=True)
        sc = float(evaluate_sequence(a_bh.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_bh.copy()
            print(f"  NEW BEST BH: {sc:.8f}")

    # Phase 6: One more L-BFGS pass on the current best
    rem = deadline - time.monotonic() - 100
    if rem > 20:
        t_l2 = min(rem * 0.4, 80.0)
        print(f"\n=== Phase 6: L-BFGS round 2 (best={best_score:.8f}) ===")
        a_l2, s_l2 = lbfgs_soft_max(best_seq, t_l2, verbose=True)
        sc = float(evaluate_sequence(a_l2.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_l2.copy()
            print(f"  NEW BEST L-BFGS2: {sc:.8f}")

    # Phase 7: Fine polishing
    print(f"\n=== Phase 7: Fine Polish (best={best_score:.8f}) ===")
    n_work = len(best_seq)
    rem = deadline - time.monotonic() - 30
    if rem > 10:
        t_fin = min(rem * 0.6, 60.0)
        a_fin, s_fin = fine_polish(best_seq, t_fin, rng, verbose=True)
        sc = float(evaluate_sequence(a_fin.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_fin.copy()
            print(f"  NEW BEST Fine: {sc:.8f}")

    # Final ECD pass
    rem = deadline - time.monotonic() - 10
    if rem > 5:
        a_last, s_last, _ = exact_coord_descent(best_seq, min(rem * 0.7, 30.0), label="last")
        sc = float(evaluate_sequence(a_last.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_last.copy()
            print(f"  NEW BEST last: {sc:.8f}")

    if best_seq is None:
        best_seq = np.ones(512) / 512

    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
