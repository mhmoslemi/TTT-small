"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive-temperature L-BFGS
with FFT-accelerated LogSumExp gradients, multi-length exploration, and
peak-directed coordinate descent refinement.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate 2n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


class LSEOptimizer:
    """
    LogSumExp-smoothed objective with FFT-based gradient.
    """
    def __init__(self, a0, T=1.0):
        self.n = len(a0)
        self.T = T
        # Precompute FFT sizes
        self.fft_size_conv = 1 << (2 * self.n - 2 - 1).bit_length()  # for convolution
        self.fft_size_grad = 1 << (3 * self.n - 3 - 1).bit_length()  # for gradient
        self.a_rev = a0[::-1].copy()
        self.z = np.log(np.maximum(a0, 1e-300))
    
    def get_a(self, z):
        a = np.exp(z - z.max())
        return a / a.sum()
    
    def obj_grad(self, z):
        a = self.get_a(z)
        b = fast_convolve(a)
        
        max_b = b.max()
        scaled = (b - max_b) / self.T
        exp_scaled = np.exp(scaled)
        sum_exp = exp_scaled.sum()
        lse = max_b + self.T * np.log(sum_exp)
        obj = 2.0 * self.n * lse
        
        w = exp_scaled / sum_exp  # softmax weights, shape (2n-1,)

        # g_a[i] = 4n * sum_m a[m] * w[i+m]

        G = irfft(A_rev * W, self.fft_size_grad)
        crosscorr = G[self.n - 1:2 * self.n - 1]

        # Softmax gradient
        S = np.dot(g_obj_da, a)
        grad_z = a * (g_obj_da - S)
        
        return obj, grad_z

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive-temperature L-BFGS
with FFT-accelerated LogSumExp gradients, multi-length exploration, and
peak-directed coordinate descent refinement.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def lse_func(z, n, T, fft_size):
    a = np.exp(z - z.max())
    a = a / a.sum()
    b = fast_convolve(a)
    max_b = b.max()
    scaled = (b - max_b) / T
    exp_s = np.exp(scaled)
    sum_exp = exp_s.sum()
    lse = max_b + T * np.log(sum_exp)
    return 2.0 * n * lse


def lse_grad(z, n, T, fft_size):
    a = np.exp(z - z.max())
    a = a / a.sum()
    b = fast_convolve(a)
    max_b = b.max()
    scaled = (b - max_b) / T
    exp_s = np.exp(scaled)
    sum_exp = exp_s.sum()
    w = exp_s / sum_exp
    a_rev = a[::-1]
    A_rev = rfft(a_rev, fft_size)
    W = rfft(w, fft_size)
    G = irfft(A_rev * W, fft_size)
    crosscorr = G[n - 1:2 * n - 1]
    g_obj_da = 4.0 * n * crosscorr
    S = np.dot(g_obj_da, a)
    return a * (g_obj_da - S)

    ss = f_interp.sum()
    return f_interp / ss
