```
def fast_convolve(a):

    c = irfft(A * A, N)[:size]
```

```
def weighted_gradient(a, w):
    n=len(a); L=len(w)
    if L != 1:
       w = 0.5 * (w + w[::-1])
    s = float(w.sum())
    if s <= 0:
       return np.zeros(n)
    w = w / s
    N = 1 << (n + L - 2).bit_length()

    conv = irfft(A * W, N)
    # indices L-1 down to L-n
    idx = np.arange(L-1, L-1-n-1, -1)
    d = 2.0 * conv[idx]
    # clean

    return d
```

```python
"""
Fast flat-autocorrelation search for nonnegative step functions.

The objective is
    score(a) = 2*n*max_k conv(a,a)[k] / (sum(a))^2.

This program starts from the provided parent sequence when available, tries
several lengths, and refines sequences using:
  - logarithmic mirror descent with FFT-accelerated weighted gradients,
  - temperature-annealed LogSumExp L-BFGS in log-probability space,
  - short final pair-exchange polishing.

All heavy convolutions/correlations are done with FFTs, enabling many
iterations per second for n up to a few thousand.
"""

import numpy as np
import time

try:
    from scipy.fft import rfft, irfft
except Exception:
    from numpy.fft import rfft, irfft

try:
    from scipy.optimize import minimize
except Exception:
    minimize = None


GLOBAL_BEST_SEQ = None
GLOBAL_BEST_SCORE = float("inf")
GLOBAL_BEST_LABEL = ""
GLOBAL_BETA = 1.0
GLOBAL_DEADLINE = 0.0


def normalize_sequence(a):
    """Clip to nonnegative finite values and normalize to sum 1."""
    try:
        a = np.asarray(a, dtype=np.float64)
    except Exception:
        return None
    if a.ndim != 1 or a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-15:
        return None
    return a / s


def fast_convolve(a):
    """Linear auto-convolution, symmetrized to remove tiny FFT asymmetries."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = rfft(a, N)
    c = irfft(A * A, N)[:size]
    if size > 1:
        c = 0.5 * (c + c[::-1])
    c = np.where(c > 0.0, c, 0.0)
    return c


def weighted_gradient(a, w):
    """
    Gradient of sum_k w[k]*conv(a,a)[k] with respect to a.

    For each k,
        b_k = sum_i a_i a_{k-i}
    and db_k/da_i = 2*a_{k-i} if valid.
    Therefore the gradient is 2 * correlation(a, w).
    """
    n = len(a)
    L = len(w)
    if L == 1:
        w = w.copy()
    else:
        w = 0.5 * (w + w[::-1])
    s = float(w.sum())
    if s <= 0 or not np.isfinite(s):
        return np.zeros(n, dtype=np.float64)
    w = w / s

    N = 1 << (n + L - 2).bit_length()
    A = rfft(a, N)
    W = rfft(w, N)
    conv = irfft(A * W, N)

    idx = np.arange(L - 1, L - 1 - n - 1, -1)
    d = 2.0 * conv[idx]
    d = np.where(d > 0.0, d, 0.0)
    return d


def make_weights(b, beta, top_k=0):
    """Nonnegative weights over convolution lags, summing to 1."""
    b = np.asarray(b, dtype=np.float64)
    L = len(b)

    if top_k is not None and top_k > 0:
        k = int(min(top_k, L))
        if k <= 0:
            return None
        if k >= L:
            idx = np.arange(L)
        else:
            idx = np.argsort(b)[-k:]
        w = np.zeros(L, dtype=np.float64)
        w[idx] = 1.0
    else:
        beta = float(beta)
        if not np.isfinite(beta) or beta <= 0.0:
            w = np.ones(L, dtype=np.float64)
        else:
            bm = float(b.max())
            if not np.isfinite(bm):
                w = np.ones(L, dtype=np.float64)
            else:
                e = np.exp(beta * (b - bm))
                s = float(e.sum())
                if s <= 0 or not np.isfinite(s):
                    w = np.zeros(L, dtype=np.float64)
                    w[int(np.argmax(b))] = 1.0
                else:
                    w = e / s

    if L > 1:
        w = 0.5 * (w + w[::-1])
    s = float(w.sum())
    if s <= 0 or not np.isfinite(s):
        w = np.ones(L, dtype=np.float64)
        s = float(L)
    return w / s


def evaluate_local(a):
    """Local score assuming normalization is safe."""
    a = normalize_sequence(a)
    if a is None:
        return float("inf")
    n = len(a)
    b = fast_convolve(a)
    bm = float(b.max())
    return float(2.0 * n * bm)


def consider_sequence(a, label):
    """Update global best if a improves it."""
    global GLOBAL_BEST_SEQ, GLOBAL_BEST_SCORE, GLOBAL_BEST_LABEL
    a = normalize_sequence(a)
    if a is None:
        return
    sc = evaluate_local(a)
    if np.isfinite(sc) and sc < GLOBAL_BEST_SCORE - 1e-15:
        GLOBAL_BEST_SCORE = sc
        GLOBAL_BEST_SEQ = a.copy()
        GLOBAL_BEST_LABEL = label
        print(f"  NEW BEST [{label}] score={sc:.12f} n={len(a)}")


def load_parent():
    """Load the externally supplied parent sequence if present."""
    try:
        p = globals().get("height_sequence_1", None)
        if p is None:
            return None
        p = np.asarray(p, dtype=np.float64)
        if p.ndim != 1 or p.size == 0:
            return None
        p = np.where(np.isfinite(p), p, 0.0)
        p = np.maximum(p, 0.0)
        s = float(p.sum())
        if s <= 1e-15:
            return None
        return p / s
    except Exception:
        return None


def structured_init(n, seed=123):
    """Reasonable asymmetric initial density used only if parent is missing."""
    x = np.linspace(0.0, 1.0, n)
    a = 0.25 + 0.75 * x
    a = a + 0.15 * x * (1.0 - x)
    a = a + 0.05 * np.cos(10.0 * x)
    a = np.maximum(a, 0.0)
    return normalize_sequence(a)


def interpolate_sequence(a, new_n):
    """Interpolate a normalized sequence to a new length."""
    a = normalize_sequence(a)
    new_n = int(new_n)
    if new_n <= 0:
        return None
    if a is None:
        return np.full(new_n, 1.0 / new_n)
    if len(a) == new_n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, len(a))
    x_new = np.linspace(0.0, 1.0, new_n)
    b = np.interp(x_new, x_old, a)
    return normalize_sequence(b)


def softmax_log(z):
    """Stable softmax for log-probability coordinates."""
    z = np.asarray(z, dtype=np.float64)
    z = np.nan_to_num(z, nan=0.0, posinf=0.0, neginf=0.0)
    z = z - float(z.max())
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0 or not np.isfinite(s):
        return np.full(z.shape, 1.0 / z.size)
    return e / s


def lse_objective(z):
    """
    LogSumExp-smooth objective for L-BFGS.

    GLOBAL_BETA sets the inverse temperature. Returns a score-like value,
    i.e. 2*n*smooth_max_b.
    """
    if GLOBAL_DEADLINE > 0.0 and time.monotonic() > GLOBAL_DEADLINE:
        return 1e100

    z = np.asarray(z, dtype=np.float64)
    if not np.all(np.isfinite(z)):
        return 1e100

    a = softmax_log(z)
    n = len(a)
    b = fast_convolve(a)
    bm = float(b.max())

    beta = float(GLOBAL_BETA)
    if beta <= 0 or not np.isfinite(beta):
        return float(2.0 * n * b.sum())

    e = np.exp(beta * (b - bm))
    Z = float(e.sum())
    if Z <= 0 or not np.isfinite(Z):
        return float(2.0 * n * bm)

    smooth = bm + float(np.log(Z)) / beta
    return float(2.0 * n * smooth)


def lse_gradient(z):
    """Gradient of lse_objective with respect to log-probabilities."""
    if GLOBAL_DEADLINE > 0.0 and time.monotonic() > GLOBAL_DEADLINE:
        return np.zeros_like(z, dtype=np.float64)

    z = np.asarray(z, dtype=np.float64)
    if not np.all(np.isfinite(z)):
        return np.zeros_like(z, dtype=np.float64)

    a = softmax_log(z)
    n = len(a)
    b = fast_convolve(a)
    bm = float(b.max())

    beta = float(GLOBAL_BETA)
    if beta <= 0 or not np.isfinite(beta):
        return np.zeros_like(a, dtype=np.float64)

    e = np.exp(beta * (b - bm))
    Z = float(e.sum())
    if Z <= 0 or not np.isfinite(Z):
        w = np.zeros_like(b)
        w[int(np.argmax(b))] = 1.0
    else:
        w = e / Z
        if len(w) > 1:
            w = 0.5 * (w + w[::-1])
        ws = float(w.sum())
        if ws <= 0 or not np.isfinite(ws):
            w = np.full_like(b, 1.0 / len(b))
        else:
            w = w / ws

    d = weighted_gradient(a, w)
    mean_ad = float(np.dot(a, d))
    grad = 2.0 * n * a * (d - mean_ad)
    return grad


def md_refine(a, time_limit, beta_mults):
    """
    Monotone logarithmic mirror descent.

    beta_mults are multipliers m such that beta = m*n. Each multiplier gets
    roughly equal wall-clock budget.
    """
    a = normalize_sequence(a)
    if a is None or len(beta_mults) == 0:
        return a, float("inf")

    start = time.monotonic()
    end = start + max(0.0, float(time_limit))
    n = len(a)

    b0 = fast_convolve(a)
    best_score = float(2.0 * n * b0.max())
    best_a = a.copy()

    main_factors = (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0)
    tiny_factors = (0.002, 0.0005)
    stage_time = max(1.0, float(time_limit) / float(len(beta_mults)))

    for mult in beta_mults:
        if time.monotonic() >= end:
            break
        stage_end = min(end, time.monotonic() + stage_time)
        beta = float(mult) * float(n)

        while time.monotonic() < stage_end:
            b = fast_convolve(a)
            bm = float(b.max())
            cur_score = float(2.0 * n * bm)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()

            w = make_weights(b, beta, 0)
            if w is None:
                break

            d = weighted_gradient(a, w)
            if d.size == 0:
                break
            maxd = float(np.max(d))
            if not np.isfinite(maxd) or maxd <= 1e-300:
                break
            meand = float(np.mean(d))
            if not np.isfinite(meand) or meand <= 1e-300:
                meand = maxd

            scale = 1.0 / meand
            cap = 10.0 / maxd
            if not np.isfinite(scale) or scale <= 0.0:
                scale = cap
            if scale > cap:
                scale = cap

            best_trial = None
            best_trial_score = cur_score

            for factor in main_factors + tiny_factors:
                eta = float(factor) * scale
                if eta <= 0.0 or not np.isfinite(eta):
                    continue
                exp_arg = -eta * d
                trial = a * np.exp(exp_arg)
                s = float(trial.sum())
                if s <= 0.0 or not np.isfinite(s):
                    continue
                trial = trial / s

                tb = fast_convolve(trial)
                ts = float(2.0 * n * tb.max())
                if np.isfinite(ts) and ts < best_trial_score - 1e-15:
                    best_trial_score = ts
                    best_trial = trial

            if best_trial is not None:
                a = best_trial
                if best_trial_score < best_score:
                    best_score = best_trial_score
                    best_a = a.copy()
            else:
                break

    return best_a, best_score


def lbgfs_refine(a, time_limit, beta_mults, maxiter_per_stage=1200):
    """
    Annealed LogSumExp L-BFGS in log-space.

    Uses actual score tracking, so a smoothed-improving but actual-worsening
    iterate is not adopted unless it is close.
    """
    if minimize is None:
        return a, float("inf")

    a = normalize_sequence(a)
    if a is None or len(beta_mults) == 0:
        return a, float("inf")

    start = time.monotonic()
    end = start + max(0.0, float(time_limit))
    n = len(a)
    best_a = a.copy()
    best_score = evaluate_local(a)
    stage_time = max(1.0, float(time_limit) / float(len(beta_mults)))

    for mult in beta_mults:
        if time.monotonic() >= end:
            break
        stage_end = min(end, time.monotonic() + stage_time)
        remaining = stage_end - time.monotonic()
        if remaining < 2.0:
            break

        beta = float(mult) * float(n)
        GLOBAL_BETA = beta
        GLOBAL_DEADLINE = stage_end

        z = np.log(np.maximum(a, 1e-300))
        scale_n = 1200.0 / float(n)
        maxiter = int(np.clip(float(maxiter_per_stage) * scale_n, 80, 5000))
        maxfun = int(max(500, 30 * maxiter))

        options = {
            "maxiter": maxiter,
            "maxfun": maxfun,
            "ftol": 1e-15,
            "gtol": 1e-10,
            "maxls": 100,
            "disp": False,
        }

        start_score = evaluate_local(a)
        try:
            res = minimize(
                fun=lse_objective,
                x0=z,
                jac=lse_gradient,
                method="L-BFGS-B",
                options=options,
            )
        except Exception:
            res = None

        if res is not None:
            x = np.asarray(res.x, dtype=np.float64)
            if np.all(np.isfinite(x)):
                a_new = softmax_log(x)
                sc = evaluate_local(a_new)
                if np.isfinite(sc):
                    if sc < best_score:
                        best_score = sc
                        best_a = a_new.copy()
                    if sc < start_score + 1e-9:
                        a = a_new
        else:
            break

    return best_a, best_score


def pair_refine(a, time_limit, beta_mult=500.0, top_k=0, seed=0):
    """
    Simple projected two-coordinate exchange polish.

    Moves mass from a coordinate with large weighted gradient to one with small
    weighted gradient, accepting only actual score improvements.
    """
    a = normalize_sequence(a)
    if a is None:
        return a, float("inf")

    start = time.monotonic()
    end = start + max(0.0, float(time_limit))
    n = len(a)
    b0 = fast_convolve(a)
    best_score = float(2.0 * n * b0.max())
    best_a = a.copy()
    rng = np.random.default_rng(seed)

    fracs = (0.001, 0.003, 0.01, 0.03, 0.1, 0.3, 0.5)
    beta = float(beta_mult) * float(n)

    for _ in range(1000000):
        if time.monotonic() >= end:
            break

        b = fast_convolve(a)
        bm = float(b.max())
        cur_score = float(2.0 * n * bm)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        w = make_weights(b, beta, top_k)
        if w is None:
            break
        d = weighted_gradient(a, w)

        mask = a > 1e-10
        if int(mask.sum()) < 2:
            break
        idx = np.where(mask)[0]
        di = d[idx]
        if di.max() - di.min() <= 1e-14:
            break

        accepted = False

        i_main = idx[int(np.argmax(di))]
        j_main = idx[int(np.argmin(di))]
        if i_main != j_main:
            for frac in fracs:
                delta = float(frac) * float(a[i_main])
                if delta <= 0.0:
                    continue
                trial = a.copy()
                trial[i_main] = a[i_main] - delta
                trial[j_main] = a[j_main] + delta
                trial = np.maximum(trial, 0.0)
                s = float(trial.sum())
                if s <= 0.0 or not np.isfinite(s):
                    continue
                trial = trial / s
                ts = float(2.0 * n * fast_convolve(trial).max())
                if np.isfinite(ts) and ts < best_score - 1e-15:
                    best_score = ts
                    best_a = trial.copy()
                    a = trial
                    accepted = True
                    break

        if not accepted:
            order = np.argsort(di)
            k = min(10, len(order))
            bottom = order[:k]
            top = order[-k:]
            for _ in range(12):
                ii = top[int(rng.integers(k))]
                jj = bottom[int(rng.integers(k))]
                ii = int(idx[ii])
                jj = int(idx[jj])
                if ii == jj:
                    continue
                for frac in (0.001, 0.01, 0.1):
                    delta = float(frac) * float(a[ii])
                    if delta <= 0.0:
                        continue
                    trial = a.copy()
                    trial[ii] = a[ii] - delta
                    trial[jj] = a[jj] + delta
                    trial = np.maximum(trial, 0.0)
                    s = float(trial.sum())
                    if s <= 0.0 or not np.isfinite(s):
                        continue
                    trial = trial / s
                    ts = float(2.0 * n * fast_convolve(trial).max())
                    if np.isfinite(ts) and ts < best_score - 1e-15:
                        best_score = ts
                        best_a = trial.copy()
                        a = trial
                        accepted = True
                        break
                if accepted:
                    break

        if not accepted:
            break

    return best_a, best_score


def run_pipeline(a, pipe_time, seed, label, hard_deadline=None):
    """Multi-stage refinement for one starting sequence."""
    a = normalize_sequence(a)
    if a is None:
        return
    pipe_time = max(0.0, float(pipe_time))
    if pipe_time <= 0:
        return

    t0 = time.monotonic()
    end = t0 + pipe_time
    if hard_deadline is not None:
        end = min(end, float(hard_deadline))
    budget = max(0.0, end - t0)
    if budget <= 0:
        return

    consider_sequence(a, label + "_start")
    n = len(a)

    # 1) Broad mirror descent.
    t1 = min(budget * 0.32, 700.0)
    if t1 > 10.0:
        res, sc = md_refine(a, t1, [0.5, 2.0, 8.0, 30.0])
        consider_sequence(res, label + "_md1")
        a = normalize_sequence(res)
        if a is None:
            return

    # 2) Medium LogSumExp L-BFGS.
    rem = end - time.monotonic()
    if rem > 40.0:
        t2 = min(rem * 0.5, 700.0)
        if t2 > 20.0:
            res, sc = lbgfs_refine(a, t2, [1.0, 5.0, 20.0, 80.0], maxiter_per_stage=1200)
            consider_sequence(res, label + "_lbfgs1")
            a = normalize_sequence(res)
            if a is None:
                return

    # 3) High mirror descent.
    rem = end - time.monotonic()
    if rem > 30.0:
        t3 = min(rem * 0.40, 350.0)
        if t3 > 10.0:
            res, sc = md_refine(a, t3, [30.0, 100.0, 300.0, 1000.0])
            consider_sequence(res, label + "_md2")
            a = normalize_sequence(res)
            if a is None:
                return

    # 4) High L-BFGS.
    rem = end - time.monotonic()
    if rem > 25.0:
        t4 = min(rem * 0.40, 300.0)
        if t4 > 15.0:
            res, sc = lbgfs_refine(a, t4, [40.0, 150.0, 500.0], maxiter_per_stage=900)
            consider_sequence(res, label + "_lbfgs2")
            a = normalize_sequence(res)
            if a is None:
                return

    # 5) Pair polishing.
    rem = end - time.monotonic()
    if rem > 15.0:
        t5 = min(rem * 0.5, 160.0)
        if t5 > 5.0:
            res, sc = pair_refine(a, t5, beta_mult=500.0, top_k=0, seed=int(seed))
            consider_sequence(res, label + "_pair")
            a = normalize_sequence(res)
            if a is None:
                return

    # 6) Final high mirror descent.
    rem = end - time.monotonic()
    if rem > 10.0:
        t6 = min(rem * 0.6, 120.0)
        if t6 > 5.0:
            res, sc = md_refine(a, t6, [300.0, 1000.0, 3000.0])
            consider_sequence(res, label + "_final_md")


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main entry point. Searches over several sequence lengths and refines each
    with a fast gradient pipeline.
    """
    global GLOBAL_BEST_SEQ, GLOBAL_BEST_SCORE, GLOBAL_BEST_LABEL, GLOBAL_DEADLINE

    seed = int(seed)
    rng = np.random.default_rng(seed)

    actual_budget = max(1.0, float(budget_s) - 30.0)
    start = time.monotonic()
    deadline = start + actual_budget
    GLOBAL_DEADLINE = deadline

    GLOBAL_BEST_SEQ = None
    GLOBAL_BEST_SCORE = float("inf")
    GLOBAL_BEST_LABEL = "none"

    parent = load_parent()
    if parent is not None:
        consider_sequence(parent, "parent")

    if GLOBAL_BEST_SEQ is None:
        fb = structured_init(1200, seed + 7)
        consider_sequence(fb, "fallback")

    if GLOBAL_BEST_SEQ is None:
        fb = np.ones(1200, dtype=np.float64)
        consider_sequence(fb, "uniform")

    if GLOBAL_BEST_SEQ is None:
        return [0.0]

    n_best = len(GLOBAL_BEST_SEQ)

    # Build a small, diverse list of lengths.
    candidates = [
        n_best,
        int(n_best * 1.25),
        int(n_best * 1.60),
        int(n_best * 2.10),
        int(n_best * 0.75),
        int(n_best * 0.55),
        1000,
        1500,
        1800,
        2200,
    ]

    lengths = []
    for c in candidates:
        c = int(c)
        if c < 250 or c > 3000:
            continue
        close = False
        for e in lengths:
            if abs(c - e) < 80:
                close = True
                break
        if not close:
            lengths.append(c)
        if len(lengths) >= 5:
            break

    print(f"Parent best: n={n_best}, score={GLOBAL_BEST_SCORE:.12f}")
    print(f"Lengths to explore: {lengths}")

    frac_list = [0.42, 0.24, 0.15, 0.10, 0.09]
    for idx, n_len in enumerate(lengths):
        if time.monotonic() >= deadline - 70.0:
            break
        remaining = deadline - time.monotonic()
        if remaining < 70.0:
            break

        frac = frac_list[idx] if idx < len(frac_list) else 0.08
        cap = 1000.0 if idx == 0 else 550.0
        t_alloc = min(remaining * frac, cap)
        if t_alloc < 45.0:
            break

        src = GLOBAL_BEST_SEQ if GLOBAL_BEST_SEQ is not None else parent
        a_init = interpolate_sequence(src, n_len)
        if a_init is None:
            continue

        print(f"\nPipeline n={n_len}, budget={t_alloc:.1f}s, current best={GLOBAL_BEST_SCORE:.12f}")
        run_pipeline(a_init, t_alloc, seed + idx, f"n{n_len}", hard_deadline=deadline)

    # Small perturbation/refinement basin to escape very shallow local minima.
    if GLOBAL_BEST_SEQ is not None and deadline - time.monotonic() > 90.0:
        n_best = len(GLOBAL_BEST_SEQ)
        x = np.linspace(0.0, 2.0 * np.pi, n_best)

        for pi in range(3):
            if deadline - time.monotonic() < 50.0:
                break

            a = GLOBAL_BEST_SEQ.copy()
            if pi == 0:
                skew = np.linspace(-1.0, 1.0, n_best)
                a = a * (1.0 + 0.018 * skew)
            elif pi == 1:
                a = a + 0.010 * rng.standard_normal(n_best)
            else:
                a = a * (1.0 + 0.018 * np.sin(3.0 * x))

            a = normalize_sequence(a)
            if a is None:
                continue
            consider_sequence(a, f"perturb{pi}")

            t_quick = min((deadline - time.monotonic()) * 0.35, 180.0)
            if t_quick < 30.0:
                break

            res, sc = md_refine(a, t_quick, [50.0, 200.0, 800.0])
            consider_sequence(res, f"perturb{pi}_md")

            if deadline - time.monotonic() > 30.0:
                res, sc = lbgfs_refine(res, min((deadline - time.monotonic()) * 0.4, 90.0),
                                       [80.0, 300.0], maxiter_per_stage=700)
                consider_sequence(res, f"perturb{pi}_lbfgs")

    # Final universal polish on the global best.
    if GLOBAL_BEST_SEQ is not None and deadline - time.monotonic() > 15.0:
        t_final = min(deadline - time.monotonic() - 2.0, 90.0)
        if t_final > 5.0:
            res, sc = pair_refine(GLOBAL_BEST_SEQ, t_final, beta_mult=800.0,
                                  top_k=0, seed=seed + 999)
            consider_sequence(res, "final_pair")

    final = GLOBAL_BEST_SEQ
    if final is None:
        final = np.ones(1200, dtype=np.float64)
    final = normalize_sequence(final)
    if final is None:
        final = np.ones(1200, dtype=np.float64)

    print(f"\nFINAL local score: {GLOBAL_BEST_SCORE:.12f}, n={len(final)}, label={GLOBAL_BEST_LABEL}")

    out = [float(max(0.0, min(1000.0, x))) for x in final]
    try:
        ev = evaluate_sequence(out)
        print(f"FINAL evaluate_sequence: {ev}")
    except Exception as e:
        print("Could not call evaluate_sequence:", e)

    return out
```
