def _fast_convolve(a):
    """Fast autocorrelation using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def _log_so(a):

def _build_conv_matrix(a_fixed):
    """Build matrix T such that (T @ b)_k = (a_fixed * b)_k."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a_fixed
    return T


def _lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. T(a)@b <= M*1, sum(b)=1, b>=0."""

    b_val = np.asarray(b_var.value, dtype=np.float64)

    s = b_val.sum()

def _smoothed_objective(z, T):
    """Log-sum-exp smoothed objective in log-space."""
    a = _softmax(z)
    b = _fast_convolve(a)
    Tb = T * b
    max_Tb = np.max(Tb)
    if max_Tb > 500:
        max_Tb = 500
    exp_Tb = np.exp(Tb - max_Tb)
    lse = max_Tb + np.log(exp_Tb.sum())
    return (2.0 * len(a) / T) * lse


def _smoothed_gradient(z, T):
    """Gradient of log-sum-exp smoothed objective."""
    n = len(z)
    a = _softmax(z)
    b = _fast_convolve(a)
    Tb = T * b
    max_Tb = np.max(Tb)
    if max_Tb > 500:
        max_Tb = 500
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    # Gradient w.r.t. a: 2 * sum_k w[k] * a shifted
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    # Gradient w.r.t. z via softmax Jacobian
    S = np.dot(grad_a, a)
    return a * (grad_a - S)

def _lse_optimize(z0, time_limit, T_start=50.0, T_end=0.1, n_stages=8):

    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        def func(z_in):
            return _smoothed_objective(z_in, T)
        
        def jac(z_in):
            return _smoothed_gradient(z_in, T)
        
        max_iter = max(30, int(remaining * 2))
        try:

                fun=func, x0=z, jac=jac, method='L-BFGS-B',

                         'disp': False}

                if not dst_candidates:

                dst = rng.choice(dst_candidates)
                frac = rng.uniform(0.01, 0.05)
                amt = a[src] * frac
                a_trial[src] -= amt
                a_trial[dst] += amt
        elif move == 1:
            # Gaussian perturbation
            k = rng.integers(max(2, n // 20), n // 5)
            idx = rng.choice(n, k, replace=False)
            a_trial[idx] += rng.normal(0, T * 0.05, k)

            i, j = rng.integers(0, n, 2)

                a_trial2[i] += amt

                s2 = _fast_convolve(np.maximum(a_trial2, 0)).max()

        elif move == 3:

            j = rng.integers(i + 1, n)

def _interpolate(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    result = np.interp(new_x, old_x, a_norm)
    result = np.maximum(result, 0)
    ss = result.sum()
    if ss > 1e-15:
        result = result / ss
    return result

def _fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def _log_space(a):
    return np.log(np.maximum(a, 1e-300))


def _build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def _lp_best_response(a_fixed, solver):
    import cvxpy as cp
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = _build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def _subgradient_best_response(a_fixed):
    """Fast subgradient-based best response (no LP needed)."""
    n = len(a_fixed)
    # We want to find b that minimizes max_k (a*b)_k
    # Use mirror descent: b_{t+1} = softmax(log(b_t) - eta * g_t)
    b = np.ones(n) / n
    z = np.zeros(n)
    eta = 0.5
    for _ in range(50):
        conv = _fast_convolve(a_fixed) * 1.0 + np.zeros(1)  # conv of a_fixed with delta at 0
        # Actually we need cross-correlation: (a_fixed * b)[k] = sum_j a_fixed[j] * b[k-j]
        # Let's compute it properly
        Ab = np.convolve(a_fixed, b)
        max_k = np.argmax(Ab)
        # Subgradient w.r.t. b: a_fixed shifted
        g = np.zeros(n)
        js = max(0, max_k - n + 1)
        je = min(n - 1, max_k)
        idx = np.arange(js, je + 1)
        g[idx] = a_fixed[max_k - idx]
        z = z - eta * g
        b = _softmax(z)
        eta *= 0.9
    return b


def _smoothed_obj_grad(z, T):
    """Combined objective and gradient for LSE smoothing."""
    n = len(z)
    a = _softmax(z)
    b = _fast_convolve(a)
    Tb = T * b
    max_Tb = np.max(Tb)
    if max_Tb > 500:
        max_Tb = 500.0
    exp_Tb = np.exp(Tb - max_Tb)
    lse = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * lse
    # Gradient
    w = exp_Tb / exp_Tb.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def _lse_optimize(a0, time_limit, T_start=50.0, T_end=0.05, n_stages=10):
    """Multi-stage LSE-smoothed L-BFGS optimization."""
    n = len(a0)
    z = _log_space(a0).copy()
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, T in enumerate(temps):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        remaining = time_limit - elapsed
        if remaining < 10:
            break
        
        max_iter = max(50, int(remaining * 3))
        
        res = _lse_stage_minimize(z, T, max_iter)
        if res is not None:
            z = res
    
    return _softmax(z)


def _lse_stage_minimize(z, T, max_iter):
    """Run one stage of L-BFGS on the LSE-smoothed objective."""
    n = len(z)
    
    def func(z_in):
        obj, _ = _smoothed_obj_grad(z_in, T)
        return obj
    
    def jac(z_in):
        _, g = _smoothed_obj_grad(z_in, T)
        return g
    
    try:
        res = minimize(
            fun=func, x0=z, jac=jac, method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': 1e-15,
                     'gtol': 1e-12, 'maxfun': max_iter * 200,
                     'disp': False, 'maxls': 50}
        )
        return res.x
    except Exception:
        return z


def _nesterov_fp(a0, time_limit, solver, max_iters=500000):
    """Clean Nesterov-accelerated fictitious play."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    y = a.copy()
    best_a = a.copy()
    b_conv = _fast_convolve(a)
    best_score = 2.0 * n * b_conv.max()
    t0 = time.monotonic()
    prev_best = best_score
    no_improve_count = 0
    
    for it in range(max_iters):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        
        # Best response to y
        b_new = None
        if solver is not None:
            b_new, _ = _lp_best_response(y, solver)
        if b_new is None:
            b_new = _subgradient_best_response(y)
        
        # Nesterov step: alpha = 2/(t+2)
        alpha = 2.0 / (it + 2.0)
        
        # Update y (lookahead point)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy
        
        # Update a (running average / actual iterate)
        a = (1.0 - alpha) * a + alpha * b_new
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        # Evaluate
        if it % 10 == 0 or it < 10:
            sc_a = 2.0 * n * _fast_convolve(a).max()
            if sc_a < best_score - 1e-12:
                best_score = sc_a
                best_a = a.copy()
            sc_y = 2.0 * n * _fast_convolve(y).max()
            if sc_y < best_score - 1e-12:
                best_score = sc_y
                best_a = y.copy()
        
        # Monitor convergence every 2000 iters
        if (it + 1) % 2000 == 0:
            sc_check = 2.0 * n * _fast_convolve(a).max()
            if sc_check < best_score:
                best_score = sc_check
                best_a = a.copy()
            
            if best_score < prev_best - 1e-8:
                no_improve_count = 0
            else:
                no_improve_count += 1
            prev_best = best_score
            
            if elapsed % 60 < 4:
                print(f"    NFP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")
            
            # Perturb if stalled
            if no_improve_count >= 5:  # 10000 iters without improvement
                rng = np.random.default_rng(it * 13 + 1)
                noise = rng.random(n) * 0.03
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                no_improve_count = 0
    
    return best_a, best_score


def _peak_annealing(a0, time_limit, rng=None):
    """Simulated annealing with peak-targeting moves."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()

    best_score = 2.0 * n * b.max()

    n_steps = max(200, int(time_limit * 60))
    
    for it in range(n_steps):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        T = 0.05 * (0.8 ** (it / max(1, n_steps // 10)))
        
        move = rng.integers(0, 5)
        a_trial = a.copy()
        
        if move == 0:
            # Peak targeting: reduce the maximum autocorrelation peak
            b_cur = _fast_convolve(a)
            peak_k = int(np.argmax(b_cur))
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[peak_k - idx]
            total = contrib.sum()
            if total > 1e-15:
                src = idx[np.argmax(contrib)]
                # Move to a position that doesn't contribute to the peak
                non_peak = [j for j in range(n) if j < js or j > je]
                if non_peak:
                    dst = non_peak[rng.integers(len(non_peak))]
                else:
                    dst = rng.integers(n)
                    if dst == src:
                        dst = (dst + 1) % n
                frac = rng.uniform(0.02, 0.08)
                amt = a[src] * frac
                a_trial[src] -= amt
                a_trial[dst] += amt
        elif move == 1:
            # Gaussian perturbation on random subset
            k = rng.integers(max(2, n // 15), n // 4)
            idx = rng.choice(n, k, replace=False)
            a_trial[idx] += rng.normal(0, T * 0.03, k)
        elif move == 2:
            # Asymmetric shift: move mass from one side to the other
            side = rng.integers(0, 2)
            amt_total = a.sum() * rng.uniform(0.01, 0.03)
            if side == 0:
                # Move from left to right
                src_region = slice(0, n // 3)
                dst_region = slice(2 * n // 3, n)
            else:
                src_region = slice(2 * n // 3, n)
                dst_region = slice(0, n // 3)
            src = src_region.start + rng.integers(src_region.stop - src_region.start)
            dst = dst_region.start + rng.integers(dst_region.stop - dst_region.start)
            amt = min(a[src], amt_total)
            a_trial[src] -= amt
            a_trial[dst] += amt
        elif move == 3:
            # Segment scaling (asymmetric)
            i = rng.integers(0, n // 2)
            j = rng.integers(i + 1, n)
            factor = rng.uniform(0.6, 1.4)
            a_trial[i:j + 1] *= factor
        else:
            # Random pair transfer
            i, j = rng.integers(0, n, 2)
            if i != j:
                amt = min(a[i], a[j]) * rng.uniform(0.1, 0.4)
                a_trial[i] -= amt
                a_trial[j] += amt
        
        a_trial = np.maximum(a_trial, 0)
        ss = a_trial.sum()
        if ss < 1e-15:
            continue
        a_trial = a_trial / ss
        
        score = 2.0 * n * _fast_convolve(a_trial).max()
        delta = score - current_score
        
        if delta < 0 or rng.random() < np.exp(-max(delta, 0) / max(T, 1e-10)):
            a = a_trial
            current_score = score
            if score < best_score:
                best_score = score
                best_a = a.copy()
    
    return best_a, best_score

        order = rng.permutation(n)

def _struct_init(n, rng):

def _interpolate_seq(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    result = np.interp(new_x, old_x, a_norm)
    result = np.maximum(result, 0)
    ss = result.sum()
    if ss > 1e-15:
        result = result / ss
    return result

    t_d = min(rem * 0.1, 100.0)
    if t_d > 10:
        rng = np.random.default_rng(int(time.monotonic() * 1000) % 1000000)
        a_out, _ = _peak_annealing(best_a, t_d, rng=rng)
        try_update(a_out, "anneal")
    
    # Phase E: Coordinate descent
    rem = time_budget - (time.monotonic() - t0)
    t_e = min(rem * 0.1, 60.0)
    if t_e > 5:
        a_out, _ = _coord_descent(best_a, t_e, max_rounds=30)
        try_update(a_out, "coord")
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function: finds the best sequence minimizing the evaluation function."""
    global _GLOBAL_DEADLINE
    rng = np.random.default_rng(seed)
    _GLOBAL_DEADLINE = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    solver = _find_lp_solver()
    print(f"LP solver: {solver}")
    
    best_seq = None
    best_score = float('inf')
    
    # Try to load parent
    n_base = 1200
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-6:
            parent = p / p.sum()
            p_score = _eval_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
            n_base = len(parent)
    except Exception:
        pass
    
    if best_seq is None:
        best_seq = _struct_init(n_base, rng)
        best_score = _eval_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    
    def global_update(candidate, label):
        nonlocal best_seq, best_score
        if candidate is None:
            return
        c = np.maximum(candidate, 0).astype(np.float64)
        ss = c.sum()
        if ss < 0.01:
            return
        sc = _eval_safe(c)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = c.copy()
            print(f"  ** GLOBAL BEST [{label}]: {sc:.6f} (n={len(c)}) **")
    
    # === Phase 1: Diverse initialization ===
    print(f"\n=== Phase 1: Diverse Init (best={best_score:.6f}, n={n_base}) ===")
    t_init = min(60.0, (budget_s - 30) * 0.03)
    t0 = time.monotonic()
    
    # Quick refine the parent
    a_quick = _quick_mirror_descend(best_seq, n_iters=100)
    global_update(a_quick, "quick_init")
    
    # Try diverse inits
    for trial in range(8):
        if time.monotonic() - t0 > t_init:
            break
        a_try = _struct_init(n_base, rng)
        a_try = _quick_mirror_descend(a_try, n_iters=50)
        global_update(a_try, f"diversify_{trial}")
    
    # === Phase 2: Main optimization at base length ===
    print(f"\n=== Phase 2: Main Opt at n={n_base} (best={best_score:.6f}) ===")
    rem = _GLOBAL_DEADLINE - time.monotonic()
    t_main = min(rem * 0.5, 700.0)
    if t_main > 60:
        print(f"  Budget: {t_main:.0f}s")
        a_out, sc_out = _optimize_length(best_seq, t_main, solver)
        global_update(a_out, "main_opt")
    
    # === Phase 3: LSE refinement on best ===
    print(f"\n=== Phase 3: LSE Refinement (best={best_score:.6f}) ===")
    rem = _GLOBAL_DEADLINE - time.monotonic()
    t_lse = min(rem * 0.2, 200.0)
    if t_lse > 20:
        print(f"  Budget: {t_lse:.0f}s")
        a_lse = _lse_optimize(best_seq, t_lse, T_start=5.0, T_end=0.03, n_stages=12)
        global_update(a_lse, "lse_deep")
        
        # Another Nesterov pass after LSE
        rem = _GLOBAL_DEADLINE - time.monotonic()
        t_np = min(rem * 0.15, 150.0)
        if t_np > 20:
            a_np, _ = _nesterov_fp(best_seq, t_np, solver=solver)
            global_update(a_np, "nesterov2")
    
    # === Phase 4: Peak targeting and annealing ===
    print(f"\n=== Phase 4: Peak Targeting (best={best_score:.6f}) ===")
    rem = _GLOBAL_DEADLINE - time.monotonic()
    t_peak = min(rem * 0.15, 150.0)
    if t_peak > 15:
        print(f"  Budget: {t_peak:.0f}s")
        rng2 = np.random.default_rng(int(time.monotonic() * 7) % 1000000)
        a_pk, _ = _peak_annealing(best_seq, t_peak, rng=rng2)
        global_update(a_pk, "peak_anneal")
    
    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-Length (best={best_score:.6f}) ===")
    rem = _GLOBAL_DEADLINE - time.monotonic()
    
    candidate_lengths = []
    for mult in [1.2, 1.5, 0.75, 0.6]:
        ln = int(n_base * mult)
        if 100 <= ln <= 5000 and ln != n_base:
            candidate_lengths.append(ln)
    candidate_lengths = sorted(set(candidate_lengths))[:3]
    
    for ln in candidate_lengths:
        rem = _GLOBAL_DEADLINE - time.monotonic()
        if rem < 80:
            break
        t_alloc = min(rem * 0.25, 200.0)
        if t_alloc < 40:
            break
        print(f"  Trying n={ln}, budget={t_alloc:.0f}s")
        
        init_long = _interpolate_seq(best_seq, ln)
        sc_init = _eval_safe(init_long)
        if np.isfinite(sc_init) and sc_init < best_score:
            best_score = sc_init
            best_seq = init_long.copy()
            print(f"    Interpolation improved: {sc_init:.6f}")
        
        a_out, _ = _optimize_length(init_long, t_alloc, solver)
        global_update(a_out, f"len_{ln}")
    
    # === Phase 6: Final polish ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}, n={len(best_seq)}) ===")
    rem = _GLOBAL_DEADLINE - time.monotonic()
    if rem > 10:
        # Coordinate descent
        t_cd = min(rem * 0.25, 60.0)
        if t_cd > 5:
            a_cd, _ = _coord_descent(best_seq, t_cd, max_rounds=20)
            global_update(a_cd, "coord_final")
        
        # Final LSE with low temperature
        rem = _GLOBAL_DEADLINE - time.monotonic()
        t_fl = min(rem * 0.3, 80.0)
        if t_fl > 10:
            a_fl = _lse_optimize(best_seq, t_fl, T_start=2.0, T_end=0.02, n_stages=8)
            global_update(a_fl, "lse_final")
        
        # One more Nesterov
        rem = _GLOBAL_DEADLINE - time.monotonic()
        t_fn = min(rem * 0.3, 60.0)
        if t_fn > 10:
            a_fn, _ = _nesterov_fp(best_seq, t_fn, solver=solver)
            global_update(a_fn, "nesterov_final")
        
        # Final peak annealing
        rem = _GLOBAL_DEADLINE - time.monotonic()
        t_fp = min(rem * 0.2, 40.0)
        if t_fp > 5:
            rng3 = np.random.default_rng(99999)
            a_fp, _ = _peak_annealing(best_seq, t_fp, rng=rng3)
            global_update(a_fp, "anneal_final")
    
    # Final verification
    if best_seq is not None:
        fs = _eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
    
    print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
