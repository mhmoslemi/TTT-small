def project_simplex(v):
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho_val = rho[-1]
    theta = (cssv[rho_val] - 1) / (rho_val + 1.0)
    return np.maximum(v - theta, 0)

    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = a ** 1.5
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(3 * x) + 0.2 * np.cos(5 * x)
    else:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
        a = a * (1 + 0.1 * np.cos(20 * np.pi * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def nesterov_lp_improved(a0, time_limit, solver=None, n_max_iters=200000, verbose=True):
    """Improved Nesterov-accelerated LP best-response with adaptive step."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    a_prev = a.copy()
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    t_counter = 1.0
    prev_best = best_score
    stall_count = 0
    fp_norm_history = []

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_counter * t_counter))
        beta = (t_counter - 1.0) / t_new if t_new > 1e-10 else 0.0
        t_counter = t_new

        y = a + beta * (a - a_prev)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        # LP best response
        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        # Update: blend current a with LP solution
        alpha = 2.0 / (t_counter + 2.0)
        a_prev = a.copy()
        a = (1.0 - alpha) * a + alpha * b_new
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = b_new.copy()
        else:
            a = a / sa

        # Track best
        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score - 1e-12:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        # Fixed-point check: ||a - b_new||
        fp_norm = np.linalg.norm(a - b_new) / max(1.0, np.linalg.norm(a))
        fp_norm_history.append(fp_norm)

        # Convergence / stall detection
        if it % 200 == 0 and it > 0:
            if best_score >= prev_best - 1e-9:
                stall_count += 1
            else:
                stall_count = 0
            prev_best = best_score

            if stall_count >= 15:
                # Perturb to escape stagnation
                rng = np.random.default_rng(it * 17 + 41)
                noise = rng.random(n) * 0.02
                a = a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                a_prev = a.copy()
                t_counter = 1.0
                stall_count = 0

        if verbose and (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.7f}, fp={fp_norm:.4f}")

    return best_a, best_score


def peak_annihilation(a0, time_limit, n_steps=100000, verbose=False):
    """Novel local search that specifically targets and reduces top autocorrelation peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        # Get top peaks
        sorted_idx = np.argsort(b)[::-1]
        n_peaks = min(5, len(b))
        top_peaks = sorted_idx[:n_peaks]
        top_vals = b[top_peaks]

        # For each top peak, find contributing pairs and try to reduce
        improved = False
        for pi in range(n_peaks):
            k = top_peaks[pi]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            total_contrib = contrib.sum()
            if total_contrib < 1e-15:
                continue

            # Find top contributors
            top_c = np.argsort(contrib)[::-1][:3]
            for ci in top_c:
                src = idx[ci]
                partner = k - src
                # Find destination: position that minimizes impact on top peaks
                # A position j impacts peak k if j is in [max(0,k-n+1), min(n-1,k)]
                peak_areas = set()
                for k2 in top_peaks:
                    js2 = max(0, k2 - n + 1)
                    je2 = min(n - 1, k2)
                    peak_areas.update(range(js2, je2 + 1))

                # Find positions outside peak areas with low values
                outside = [j for j in range(n) if j not in peak_areas]
                if not outside:
                    outside = list(range(n))
                # Sort by current value (prefer moving to low-value positions)
                outside_sorted = sorted(outside, key=lambda j: a[j])
                candidates = outside_sorted[:10]

                for dst in candidates:
                    if dst == src or dst == partner:
                        continue
                    # Try different fractions
                    for frac in [0.005, 0.01, 0.02, 0.05]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        a_trial = np.maximum(a_trial, 0)
                        ss = a_trial.sum()
                        if ss < 1e-15:
                            continue
                        a_trial = a_trial / ss
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-12:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            no_improve = 0
                            break
                    if improved:
                        break
            if improved:
                break

        if not improved:
            no_improve += 1
            # Random perturbation when stuck
            if no_improve % 5 == 0:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.02)
                    a_trial = a.copy()
                    amt = a[i] * frac
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss > 1e-15:
                        a_trial = a_trial / ss
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score + 0.001:  # Allow small increases
                            a = a_trial
                            score_cur = 2.0 * n * fast_convolve(a).max()
            
            if no_improve > 5000:
                break

        if verbose and step % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakAnn {elapsed:.0f}s: score={best_score:.7f}")

    return best_a, best_score


def fourier_papr_refine(a0, time_limit, n_iters=5000):
    """Fourier-domain refinement inspired by PAPR reduction algorithms."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        A = rfft(a)
        # Compute autocorrelation in frequency domain
        power = np.abs(A) ** 2
        acorr = irfft(power, 2 * n - 1)[:2 * n - 1]
        peak_idx = int(np.argmax(acorr))
        peak_val = acorr[peak_idx]
        mean_val = acorr.mean()
        papr = peak_val / max(mean_val, 1e-15)

        if papr < 1.01:
            break

        # Phase-only modification to reduce peak
        # Mx algorithm style: shift peak energy
        phase_mask = np.ones(len(A), dtype=np.complex128)
        # Random phase perturbation weighted by peak location
        delta = rng.uniform(0.01, 0.1)
        phase_mask = np.exp(1j * delta * np.sin(2 * np.pi * peak_idx * np.arange(len(A)) / len(A)))
        A_new = A * phase_mask
        a_new = irfft(A_new, 2 * n - 1)[:n]
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss > 1e-15:
            a_new = a_new / ss
            score = 2.0 * n * fast_convolve(a_new).max()
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
                a = a_new

        # Also try direct time-domain adjustment
        if it % 10 == 0:
            b = fast_convolve(a)
            k_peak = int(np.argmax(b))
            js = max(0, k_peak - n + 1)
            je = min(n - 1, k_peak)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_peak - idx]
            # Reduce the largest contributor
            src = idx[np.argmax(contrib)]
            # Spread to average
            a_trial = a.copy()
            amt = a[src] * 0.01
            a_trial[src] -= amt
            a_trial += amt / n
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss > 1e-15:
                a_trial = a_trial / ss
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score:
                    best_score = score
                    best_a = a_trial.copy()
                    a = a_trial

    return best_a, best_score


def multiplicative_weights(a0, time_limit, n_iters=50000):
    """Multiplicative weights / exponential weights for the game."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    # FPSZ (Freund, Schapire, Servedio, Warmuth) style
    eta = 1.0 / np.sqrt(n)
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Find the maximizer (best response of opponent)
        b = fast_convolve(a)
        k_star = int(np.argmax(b))
        
        # Compute the "loss" vector: how much each coordinate contributes to the peak
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        loss = np.zeros(n)
        loss[idx] = a[k_star - idx]
        
        # Multiplicative update: decrease coordinates with high loss
        a = a * np.exp(-eta * loss)
        a = np.maximum(a, 1e-30)
        a = a / a.sum()
        
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

    return best_a, best_score


def optimize_at_length(a_init, time_budget, solver, label=""):
    """Run multi-phase optimization at a fixed length."""
    n = len(a_init)
    best_a = np.maximum(a_init.copy(), 0)
    s = best_a.sum()
    if s < 1e-15:
        best_a = np.ones(n) / n
    best_a = best_a / best_a.sum()
    best_score = 2.0 * n * fast_convolve(best_a).max()
    t0 = time.monotonic()

    def try_update(a, src_label):
        nonlocal best_a, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        ss = a.sum()
        if ss < 0.01:
            return
        a = a / ss
        sc = 2.0 * n * fast_convolve(a).max()
        if sc < best_score - 1e-10:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t0
            print(f"  IMPROVED [{src_label} n={n}]: {sc:.7f} ({elapsed:.0f}s)")

    # Phase A: Nesterov LP (main workhorse) - 60% of budget
    rem = time_budget - (time.monotonic() - t0)
    t_a = rem * 0.55
    if t_a > 20:
        a_lp, _ = nesterov_lp_improved(best_a, t_a, solver=solver, n_max_iters=100000, verbose=True)
        try_update(a_lp, f"nlp_{label}")

    # Phase B: Peak annihilation - 20% of budget
    rem = time_budget - (time.monotonic() - t0)
    t_b = rem * 0.35
    if t_b > 10:
        a_pa, _ = peak_annihilation(best_a, t_b, n_steps=200000)
        try_update(a_pa, f"peak_{label}")

    # Phase C: Fourier PAPR refinement - 10% of budget
    rem = time_budget - (time.monotonic() - t0)
    t_c = rem * 0.4
    if t_c > 5:
        a_fp, _ = fourier_papr_refine(best_a, t_c, n_iters=20000)
        try_update(a_fp, f"fourier_{label}")

    # Phase D: Second Nesterov LP pass - remaining budget
    rem = time_budget - (time.monotonic() - t0)
    t_d = rem
    if t_d > 10:
        a_lp2, _ = nesterov_lp_improved(best_a, t_d, solver=solver, n_max_iters=50000, verbose=False)
        try_update(a_lp2, f"nlp2_{label}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function."""
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    t_end = t_start + max(1.0, float(budget_s) - 50.0)  # 50s buffer

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Try to get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = 2.0 * len(parent) * fast_convolve(parent).max()
            print(f"Parent: n={len(parent)}, score={p_score:.7f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def time_remaining():
        return t_end - time.monotonic()

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = 2.0 * len(a) * fast_convolve(a).max()
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  GLOBAL NEW BEST [{label}]: {sc:.7f} (n={len(a)}, t={elapsed:.0f}s)")

    # Phase 1: Quick multi-length exploration (15% of time)
    print(f"\n=== Phase 1: Multi-length Exploration (best={best_score:.7f}) ===")
    rem = time_remaining()
    t_explore = min(rem * 0.12, 150.0)

    if parent is not None:
        n_base = len(parent)
    else:
        n_base = 1200
        best_seq = structured_init(n_base, rng)
        best_score = 2.0 * n_base * fast_convolve(best_seq).max()

    # Try a few nearby lengths
    lengths_to_try = []

        ln = n_base + offset
        if 200 <= ln <= 3000 and ln != n_base:
            lengths_to_try.append(ln)
    lengths_to_try = sorted(set(lengths_to_try))[:6]

    per_length_time = t_explore / max(1, len(lengths_to_try)) if lengths_to_try else 0

    for ln in lengths_to_try:
        if time_remaining() < 30:
            break
        t_alloc = min(per_length_time, time_remaining() * 0.5)
        if t_alloc < 10:
            break
        print(f"  Exploring n={ln} (budget={t_alloc:.0f}s)")
        init = interpolate_sequence(best_seq, ln)
        sc = 2.0 * ln * fast_convolve(init).max()
        cur = init
        if sc < best_score:
            best_score = sc
            best_seq = cur.copy()
            print(f"    Interpolated n={ln}: {sc:.7f}")
        
        # Quick Nesterov LP
        a_out, _ = nesterov_lp_improved(cur, t_alloc * 0.7, solver=solver,
                                         n_max_iters=50000, verbose=False)
        try_update(a_out, f"explore_n{ln}")

        # Quick peak search
        rem2 = time_remaining()
        if rem2 > 10:
            a_pa, _ = peak_annihilation(a_out, min(t_alloc * 0.3, rem2 * 0.5), n_steps=20000)
            try_update(a_pa, f"peak_n{ln}")

    # Phase 2: Main optimization at best length (50% of remaining time)
    print(f"\n=== Phase 2: Main Optimization n={len(best_seq)} (best={best_score:.7f}) ===")
    rem = time_remaining()

        a_pa2, _ = peak_annihilation(best_seq, t_peak, n_steps=500000)

    # Final verification

        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.7f} (n={len(best_seq)})")
    
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
