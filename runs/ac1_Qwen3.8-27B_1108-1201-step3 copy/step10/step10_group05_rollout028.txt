So minimizing `sum_k c[k]^2` is equivalent to minimizing `sum_j |A[j]|^4` (for the circular version, which equals the linear version when N >= 2n-1).

This is the "constancy of the spectrum" problem - we want `|A[j]|` to be as constant as possible.

The constraint `A[0] = N` (DC component) is fixed. For the other components, we want them all to be equal in magnitude.

If we could make `|A[j]|^2 = N/(2n-1)` for j = 1,...,N-1 (approximately), then the sequence would have a "flat spectrum" (except at DC).

But the constraint `a[m] >= 0` limits what's achievable.

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1

    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def init_uniform(n):
    return np.ones(n) / n

def init_sine(n, freq=1, amp=0.3, phase=0.0):
    m = np.arange(n)
    a = 1.0 + amp * np.cos(2 * np.pi * freq * m / n + phase)
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    return a

def init_gaussian(n, sigma_frac=0.3):
    m = np.arange(n) - n / 2
    a = np.exp(-0.5 * (m / (sigma_frac * n)) ** 2)

def init_flat_spectrum(n, rng):
    """Try to create a sequence with a flat spectrum."""
    N = 1 << (2 * n - 2).bit_length()  # FFT size >= 2n-1
    # Set target: |A[k]|^2 = constant for all k
    target_power = 1.0 / N
    # Random phases
    phases = rng.random(N // 2 + 1) * 2 * np.pi
    # DC component
    A = np.zeros(N // 2 + 1, dtype=complex)
    A[0] = 1.0  # DC = sum(a) = 1
    for k in range(1, N // 2 + 1):
        magnitude = np.sqrt(2.0 * target_power * (N - 1))  # Adjust for DC
        A[k] = magnitude * np.exp(1j * phases[k])
    # Inverse FFT
    a_full = irfft(A, N)
    a = a_full[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return init_uniform(n)
    a = a / s
    return a

        sweeps += 1

        for i in range(n):

                continue

            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            affected = conv[k_start:k_end + 1]
            coeffs = a[(k_start - i):(k_end - i + 1)] + ai
            has_quad = (2 * i >= k_start) and (2 * i <= k_end)
            quad_idx = 2 * i - k_start if has_quad else -1

            # Generate candidate deltas
            candidates = [0.0, -ai]

            # Where individual lines hit unaff_max

                c = coeffs[j]

                    if -ai * 1.5 < d < ai * 1.5:
                        candidates.append(d)

            # Where top affected lines cross each other
            top_k = min(5, len(affected))
            top_idx = np.argsort(affected)[-top_k:]

                    i1, i2 = top_idx[j1], top_idx[j2]
                    denom = coeffs[i1] - coeffs[i2]

            best_delta = 0.0

            for delta in candidates:
                if delta < -ai - 1e-12:
                    continue
                delta = max(delta, -ai)

                    continue

                new_affected = affected + delta * coeffs

                    new_affected[quad_idx] += delta * delta

                a[i] = max(ai + best_delta, 0.0)

def spectral_perturb(a0, rng, strength=0.05):
    """Perturb based on spectral analysis of the convolution peak."""
    n = len(a0)
    a = a0.copy()
    conv = fast_convolve(a)
    M = conv.max()
    
    # Find peak positions
    threshold = M * 0.85
    peak_positions = np.where(conv >= threshold)[0]
    
    if len(peak_positions) == 0:
        return a.copy()
    
    # Compute pressure on each coordinate
    pressure = np.zeros(n)
    for k in peak_positions:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            pressure[i] += a[i] * a[j]
    
    p_max = pressure.max()
    if p_max > 1e-20:
        pressure = pressure / p_max
    
    # Perturb: reduce high-pressure, increase low-pressure
    noise = rng.standard_normal(n)
    a_new = a * (1.0 - strength * pressure + 0.3 * strength * noise)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

**Approach: Alternating Minimization with Frequency-Domain Constraints**

The idea is to alternate between:
1. In the time domain: minimize the peak of the convolution (coordinate descent)
2. In the frequency domain: adjust the spectrum to be flatter

For the frequency domain step:
- Compute the DFT of the current sequence (zero-padded to length N >= 2n-1)
- The peak of the convolution corresponds to constructive interference in the time domain
- To reduce the peak, we can adjust the phases of the DFT coefficients to reduce constructive interference at the peak location

But this is complex. Let me think of a simpler approach.

**Simpler approach: Iterative Peak Reduction**

1. Compute `c = conv(a, a)`
2. Find the peak: `c[k*] = M`
3. The peak at position k* is `sum_i a[i]*a[k*-i]`
4. To reduce this, we can decrease the product `a[i]*a[k*-i]` for the contributing pairs
5. This suggests decreasing either `a[i]` or `a[k*-i]` (whichever is larger)
6. But we need to maintain `sum(a) = 1`, so we increase other coordinates

This is essentially what coordinate descent does, but we can be more targeted.

**Better idea: Use the dual problem.**

The primal problem is:
minimize M
subject to: (a*a)[k] <= M for all k
sum(a) = 1
a >= 0

The Lagrangian is:
L = M + sum_k lambda_k ((a*a)[k] - M) - mu (sum(a) - 1) - sum_i nu_i * a_i

At the optimum, the KKT conditions give us:

def sumsq_gradient_descent(a0, time_limit, alpha=1.0, threshold_frac=0.8, lr_init=0.1, label=""):
    """Gradient descent on sum of squared convolution values with peak penalty."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        threshold = threshold_frac * M
        
        # Compute gradient of sum_k c[k]^2
        # grad[i] = 2 * sum_k c[k] * (a[k-i] + a[i]) (with bounds)
        grad = np.zeros(n)
        for i in range(n):
            # sum_k c[k] * a[k-i]
            k_min = max(0, i)
            k_max = min(2*n-2, i + n - 1)
            # a[k-i] for k in [k_min, k_max] means a[0] to a[min(n-1, k_max-i)]
            j_max = min(n-1, k_max - i)
            if j_max >= 0:
                grad[i] += 2.0 * np.dot(conv[k_min:k_max+1], a[:j_max+1]) + 2.0 * a[i] * conv.sum()
        
        # Add peak penalty gradient
        peak_mask = conv > threshold
        if peak_mask.any():
            peak_excess = np.where(peak_mask, conv - threshold, 0.0)
            for i in range(n):
                k_min = max(0, i)
                k_max = min(2*n-2, i + n - 1)
                j_max = min(n-1, k_max - i)
                if j_max >= 0:
                    grad[i] += 2.0 * alpha * (np.dot(peak_excess[k_min:k_max+1], a[:j_max+1]) + a[i] * peak_excess.sum())
        
        # Gradient step
        a_new = a - lr * grad
        a_new = simplex_project(a_new)
        
        # Evaluate
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        # Adaptive learning rate
        if no_improve > 30:
            lr *= 0.9
            no_improve = 0
            if lr < 1e-8:
                break
        elif it % 50 == 0:
            lr *= 0.99
        
        if it % 20 == 0:
            print(f"    SSQ {label} it={it}: score={best_score:.6f}, lr={lr:.6f}")
    
    return best_a, best_score
