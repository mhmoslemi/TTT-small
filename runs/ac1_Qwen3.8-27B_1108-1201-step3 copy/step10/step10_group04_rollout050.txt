import time
from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.maximum(0, np.where(u * np.arange(1, n + 1) > (cssv - 1))[0][:1][0])
    # Actually the standard algorithm:
    # rho = max {j : u_j > (sum_{i=1}^j u_i - 1) / j}
    tmp = u * np.arange(1, n + 1) - (cssv - 1)
    valid = tmp > 0
    if not valid.any():

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    tmp = u * np.arange(1, n + 1) - (cssv - 1)
    valid = tmp > 0
    if not valid.any():
        return np.ones(n) / n
    rho = np.where(valid)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    return g


def compute_multi_gradient(a, n_active=5, temperature=1.0):
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    if max_b < 1e-300:
        return np.zeros(n), max_b
    sorted_idx = np.argsort(b)[::-1]
    top_k = sorted_idx[:n_active]
    top_vals = b[top_k]
    exp_vals = np.exp(top_vals / max(max_b, 1e-300) / temperature - 1.0)

        js = max(0, k - n + 1)
        je = min(n - 1, k)
        g[js:je + 1] += 2.0 * wi * a[je:js - 1:-1]
    return g, max_b

def lbgfs_logsumexp(a_init, time_budget, T_start=50.0, T_end=0.05, n_stages=10):
    n = len(a_init)
    a = np.asarray(a_init, dtype=np.float64)
    z = np.log(np.maximum(a, 1e-300)).copy()
    t0 = time.monotonic()

    def lse_obj(z_in, T_val):
        a_local = np.exp(z_in - z_in.max())
        a_local = a_local / a_local.sum()
        b = fast_convolve(a_local)
        Tb = T_val * b
        max_Tb = min(Tb.max(), 700)
        exp_Tb = np.exp(Tb - max_Tb)
        log_sum_exp = max_Tb + np.log(exp_Tb.sum())
        return (2.0 * n / T_val) * log_sum_exp

    def lse_jac(z_in, T_val):
        a_local = np.exp(z_in - z_in.max())
        a_local = a_local / a_local.sum()
        b = fast_convolve(a_local)
        Tb = T_val * b
        max_Tb = min(Tb.max(), 700)
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a_local[k - idx]
        S = np.dot(grad_a, a_local)
        return a_local * (grad_a - S)

    temps = np.geomspace(T_start, T_end, n_stages)
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_budget:
            break
        remaining = time_budget - (time.monotonic() - t0)
        if remaining < 1:
            break

        def func(z_in):
            return lse_obj(z_in, stage_T)

        def jac(z_in):
            return lse_jac(z_in, stage_T)

        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': 50, 'ftol': 1e-16,
                                    'gtol': 1e-12, 'maxfun': 250, 'disp': False})
            z = res.x
        except Exception:
            pass

    a_out = np.exp(z - z.max())
    a_out = a_out / a_out.sum()
    return a_out

        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            src = idx[np.argmax(contrib)]
            dst = rng.integers(n)
            if dst == src:
                dst = (dst + 1) % n

            for frac in [0.02, 0.05]:
                amt = a[src] * frac
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-12:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved and step % 500 == 0:
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.01, 0.05)
                a_trial = a.copy()
                amt = a[i] * frac
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss > 1e-15:
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-12:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()

    return best_a, best_score


def structured_init(n, rng):
    choice = rng.integers(0, 12)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-6 * x * (1 - x))
    elif choice == 4:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n))
        a = a ** 2
    elif choice == 5:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(3 * x) + 0.2 * np.cos(5 * x)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
        a = a * (1 + 0.15 * np.cos(15 * np.pi * x))
    elif choice == 7:
        x = np.linspace(-1, 1, n)
        a = np.exp(-4 * x ** 2) * (1 + 0.2 * np.cos(10 * x))
    elif choice == 8:
        a = np.ones(n) * 0.5
        n_half = n // 2
        taper = np.linspace(0.3, 1.0, n_half)
        a[:n_half] = taper
        a[n_half:] = taper[::-1]
    elif choice == 9:
        freqs = np.fft.rfftfreq(n)
        target = np.exp(-5.0 * np.abs(freqs) * n)
        a = np.fft.irfft(target, n)
        a = np.abs(a)
    elif choice == 10:
        x = np.linspace(0, 1, n)
        a = (1 - 2 * x) ** 2
        a = np.maximum(a, 0.01)
    else:
        a = rng.random(n)
        a = a + 0.5 * np.ones(n)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def optimize_at_length(a_init, time_budget, solver, label=""):
    best_a = np.maximum(np.asarray(a_init, dtype=np.float64), 0)
    s = best_a.sum()
    if s < 1e-15:
        best_a = np.ones(len(a_init)) / len(a_init)
    best_a = best_a / best_a.sum()
    best_score = evaluate_safe(best_a)
    t0 = time.monotonic()
    n = len(best_a)

    def try_update(a, src_label):
        nonlocal best_a, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        ss = a.sum()
        if ss < 0.01:
            return
        a = a / ss
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-10:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t0
            print(f"    NEW BEST [{src_label}]: {sc:.6f} (n={n}, t={elapsed:.0f}s)")

    # Phase A: FISTA multi-point subgradient
    rem = time_budget - (time.monotonic() - t0)
    t_a = min(rem * 0.35, 120.0)
    if t_a > 5:
        a_fista, _ = fista_multipoint(best_a, n_iters=500000, n_active=5,
                                      temperature=0.8, lr_init=0.8,
                                      restart_every=3000, time_limit=t_a,
                                      t0=time.monotonic())
        try_update(a_fista, f"fista_{label}")

        # Second FISTA pass with different parameters
        rem = time_budget - (time.monotonic() - t0)
        t_a2 = min(rem * 0.20, 60.0)
        if t_a2 > 5:
            a_fista2, _ = fista_multipoint(best_a, n_iters=500000, n_active=8,
                                           temperature=0.5, lr_init=0.5,
                                           restart_every=5000, time_limit=t_a2,
                                           t0=time.monotonic())
            try_update(a_fista2, f"fista2_{label}")

    # Phase B: LP-based Nesterov refinement
    rem = time_budget - (time.monotonic() - t0)
    t_b = min(rem * 0.35, 400.0)
    if t_b > 15:
        a_lp, _ = nesterov_lp_refine(best_a, t_b, solver=solver, n_max_iters=100000)
        try_update(a_lp, f"lp_{label}")

    # Phase C: LogSumExp L-BFGS
    rem = time_budget - (time.monotonic() - t0)
    t_c = min(rem * 0.15, 80.0)
    if t_c > 5:
        a_lse = lbgfs_logsumexp(best_a, t_c, T_start=20.0, T_end=0.05, n_stages=8)
        try_update(a_lse, f"lse_{label}")

    # Phase D: Peak equalization
    rem = time_budget - (time.monotonic() - t0)
    t_d = min(rem * 0.15, 50.0)
    if t_d > 3:
        a_pe, _ = peak_equalization(best_a, t_d, n_steps=15000)
        try_update(a_pe, f"pe_{label}")

    # Phase E: Final FISTA polish
    rem = time_budget - (time.monotonic() - t0)
    t_e = min(rem * 0.15, 40.0)
    if t_e > 3:
        a_fin, _ = fista_multipoint(best_a, n_iters=200000, n_active=3,
                                    temperature=0.3, lr_init=0.3,
                                    restart_every=10000, time_limit=t_e,
                                    t0=time.monotonic())
        try_update(a_fin, f"polish_{label}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    global time_limit_global
    rng = np.random.default_rng(seed)
    time_limit_global = time.monotonic() + max(1.0, float(budget_s) - 50.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Get parent sequence if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    n_base = len(best_seq) if best_seq is not None else 1200

    if parent is None:
        n_base = 1200
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    def try_global_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** GLOBAL NEW BEST [{label}]: {sc:.6f} (n={len(a)}) ***")

    # Phase 1: Multi-start FISTA at base length
    print(f"\n=== Phase 1: Multi-start at n={n_base} (best={best_score:.6f}) ===")
    rem = time_limit_global - time.monotonic()
    t1_total = min(rem * 0.35, 500.0)

    n_starts = 6
    per_start_time = t1_total / n_starts

    start_inits = [best_seq.copy()]
    for si in range(n_starts - 1):
        start_inits.append(structured_init(n_base, rng))

    for si, init in enumerate(start_inits):
        rem = time_limit_global - time.monotonic()
        if rem < 10:
            break
        t_alloc = min(per_start_time, rem * 0.3)
        if t_alloc < 5:
            break
        print(f"  Start {si+1}/{n_starts}, budget={t_alloc:.0f}s")

        a_out, _ = fista_multipoint(init, n_iters=500000, n_active=5,
                                    temperature=0.7, lr_init=0.7,
                                    restart_every=2000, time_limit=t_alloc,
                                    t0=time.monotonic())
        try_global_update(a_out, f"multi_start_{si}")

    # Phase 2: Full optimization at base length
    print(f"\n=== Phase 2: Full optimization at n={n_base} (best={best_score:.6f}) ===")
    rem = time_limit_global - time.monotonic()
    t2 = min(rem * 0.30, 400.0)
    if t2 > 30:
        print(f"  Budget: {t2:.0f}s")
        a_out, _ = optimize_at_length(best_seq, t2, solver, label=f"n{n_base}")
        try_global_update(a_out, f"full_opt_n{n_base}")

    # Phase 3: Try different lengths
    print(f"\n=== Phase 3: Length exploration (best={best_score:.6f}) ===")
    candidate_lengths = []
    for mult in [1.25, 1.5, 2.0, 0.75]:
        ln = int(n_base * mult)
        if 400 <= ln <= 4000 and ln != n_base:
            candidate_lengths.append(ln)
    candidate_lengths = sorted(set(candidate_lengths))[:4]

    for ln in candidate_lengths:
        rem = time_limit_global - time.monotonic()
        if rem < 80:
            break
        t_alloc = min(rem * 0.20, 250.0)
        if t_alloc < 30:
            break
        print(f"  Trying n={ln}, budget={t_alloc:.0f}s")

        init_long = interpolate_sequence(best_seq, ln)
        sc_init = evaluate_safe(init_long)
        if sc_init < best_score:
            best_score = sc_init
            best_seq = init_long.copy()
            print(f"    Interpolation improved! score={sc_init:.6f}")

        a_out, _ = optimize_at_length(init_long, t_alloc, solver, label=f"n{ln}")
        try_global_update(a_out, f"len_n{ln}")

    # Phase 4: Final polish
    print(f"\n=== Phase 4: Final Polish (best={best_score:.6f}, n={len(best_seq)}) ===")
    rem = time_limit_global - time.monotonic()
    if rem > 30:
        # LP refinement
        t_lp = min(rem * 0.35, 150.0)
        if t_lp > 10:
            a_fin, _ = nesterov_lp_refine(best_seq, t_lp, solver=solver, n_max_iters=50000)
            try_global_update(a_fin, "final_lp")

        # LSE polish
        rem = time_limit_global - time.monotonic()
        if rem > 20:
            t_lse = min(rem * 0.25, 80.0)
            if t_lse > 5:
                a_fin2 = lbgfs_logsumexp(best_seq, t_lse, T_start=5.0, T_end=0.02, n_stages=8)
                try_global_update(a_fin2, "final_lse")

        # Peak equalization
        rem = time_limit_global - time.monotonic()
        if rem > 15:
            t_pe = min(rem * 0.20, 40.0)
            if t_pe > 5:
                a_fin3, _ = peak_equalization(best_seq, t_pe, n_steps=10000)
                try_global_update(a_fin3, "final_pe")

        # One more FISTA pass
        rem = time_limit_global - time.monotonic()
        if rem > 10:
            t_f = min(rem * 0.30, 30.0)
            if t_f > 5:
                a_fin4, _ = fista_multipoint(best_seq, n_iters=100000, n_active=3,
                                             temperature=0.3, lr_init=0.4,
                                             restart_every=5000, time_limit=t_f,
                                             t0=time.monotonic())
                try_global_update(a_fin4, "final_fista")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
