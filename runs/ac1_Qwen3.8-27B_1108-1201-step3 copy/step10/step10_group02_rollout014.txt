"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Parametric initialization exploiting monotone structure insight
2. LP alternating with Nesterov acceleration (main workhorse)
3. FFT-based mirror descent for fast per-iteration updates
4. LogSumExp-smoothed L-BFGS for precise refinement
5. Peak-directed coordinate descent for final polish
Key innovation: spectral flat-autocorrelation target for initialization,
adaptive active set, and multi-length exploration.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence using local computation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver_cached = None


def get_lp_solver():
    """Find a working LP solver."""
    global _lp_solver_cached
    if _lp_solver_cached is not None:
        return _lp_solver_cached
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return _lp_solver_cached
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return _lp_solver_cached


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    solved = False
    try:
        prob.solve(solver=solver)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            solved = True
    except Exception:
        pass
    if not solved:
        for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
                    break
            except Exception:
                continue
    if not solved:
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def compute_subgradient_fft(a, b, n_peaks=50):
    """Compute subgradient of max(conv(a,b)) using FFT for the top peaks."""
    n = len(a)
    size = 2 * n - 1
    # Find top peaks
    top_idx = np.argsort(b)[::-1][:n_peaks]
    # Weight by proximity to max
    max_b = b.max()
    weights = np.exp(-(b[top_idx] - max_b) / (max_b * 0.01 + 1e-300))
    weights = weights / weights.sum()

    # Compute gradient: grad[i] = sum_k w_k * b_peak_contribution
    # For peak at position k: contribution to a[i] is a[k-i] (if valid)
    # Use FFT: grad = sum_k w_k * reverse(a)[k] convolved with delta
    # Actually: grad[i] = sum_k w_k * a[k-i] where k-i in [0,n-1]
    # This is: grad = sum_k w_k * (a reversed, shifted by k)
    # More efficiently: grad = ifft(fft(sum_k w_k * exp(-ik*omega)) * fft(a))
    # Let's use direct computation for clarity (n_peaks is small)
    g = np.zeros(n)
    for k, w in zip(top_idx, weights):
        js = max(0, int(k) - n + 1)
        je = min(n - 1, int(k))
        if js > je:
            continue
        idx = np.arange(js, je + 1)
        g[idx] += w * a[int(k) - idx]
    return g


def softmax_safe(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def parametric_search(n, time_limit, rng):
    """Search over parametric families for good initialization."""
    t0 = time.monotonic()
    best_a = None
    best_score = float('inf')

    x = np.linspace(0, 1, n)

    # Family 1: Power law decreasing: a[i] = (1 - x[i])^p + eps
    for p in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0, 5.0, 7.0, 10.0]:
        a = (1 - x) ** p + 0.01
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            continue
        a = a / s
        sc = evaluate_local(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    # Family 2: Exponential decay: a[i] = exp(-alpha * x[i])
    for alpha in [0.5, 1.0, 1.5, 2.0, 3.0, 5.0, 7.0, 10.0, 15.0]:
        a = np.exp(-alpha * x) + 0.001
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            continue
        a = a / s
        sc = evaluate_local(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    # Family 3: Two-part: high then low
    for frac in [0.3, 0.4, 0.5, 0.6, 0.7]:
        for ratio in [1.5, 2.0, 3.0, 5.0, 10.0]:
            a = np.ones(n)
            split = int(n * frac)
            a[:split] = ratio
            a = np.maximum(a, 0)
            s = a.sum()
            if s < 1e-15:
                continue
            a = a / s
            sc = evaluate_local(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    # Family 4: Smoothed step: a[i] = 1/(1+exp(-k*(x - c)))
    for k in [5, 10, 20, 50]:
        for c in [0.3, 0.4, 0.5, 0.6, 0.7]:
            a = 1.0 / (1.0 + np.exp(-k * (x - c)))
            a = np.maximum(a, 0.001)
            s = a.sum()
            if s < 1e-15:
                continue
            a = a / s
            sc = evaluate_local(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    # Family 5: Cosine taper (decreasing): a[i] = cos(pi * x[i] / 2)^p
    for p in [1, 2, 3, 4, 6]:
        a = np.cos(np.pi * x / 2) ** p
        a = np.maximum(a, 0.001)
        s = a.sum()
        if s < 1e-15:
            continue
        a = a / s
        sc = evaluate_local(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    # Family 6: Linear decreasing with floor
    for slope in [0.5, 1.0, 2.0, 3.0, 5.0]:
        a = 1.0 - slope * x
        a = np.maximum(a, 0.01)
        s = a.sum()
        if s < 1e-15:
            continue
        a = a / s
        sc = evaluate_local(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    # Family 7: Super-position of decreasing and increasing (to break symmetry)
    for alpha in [0.3, 0.5, 0.7]:
        a = alpha * (1 - x) ** 2 + (1 - alpha) * x
        a = np.maximum(a, 0.001)
        s = a.sum()
        if s < 1e-15:
            continue
        a = a / s
        sc = evaluate_local(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    if time.monotonic() - t0 > time_limit:
        return best_a, best_score

    # Spectral initialization: target flat autocorrelation
    # |A(omega)| = 1/sqrt(2n-1) * |Dirichlet(omega)|
    target_mag = np.ones(2 * n - 1)
    fft_size = 1 << ((2 * n - 1) - 1).bit_length()
    # Compute target |A(omega)|
    omega = np.fft.rfftfreq(fft_size, d=1.0)
    # Dirichlet kernel of length 2n-1
    D = np.zeros_like(omega)
    for k in range(2 * n - 1):
        D += np.exp(-2j * np.pi * omega * k)
    target_mag_full = np.abs(D) / np.sqrt(2 * n - 1)
    # Take rfft size
    half_size = fft_size // 2 + 1
    target_mag_rfft = target_mag_full[:half_size]

    # Start with random phase
    a_init = np.ones(n) / n
    for iteration in range(200):
        A = rfft(a_init, fft_size)
        # Replace magnitude with target, keep phase
        angle = np.angle(A)
        A_new = target_mag_rfft * np.exp(1j * angle)
        a_new = irfft(A_new, fft_size)[:n]
        # Project to non-negative and normalize
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a_init = a_new
        sc = evaluate_local(a_init)
        if sc < best_score:
            best_score = sc
            best_a = a_init.copy()

    return best_a, best_score


def mirror_descent_optimize(a0, time_limit, eta0=0.05, active_frac=0.999,
                            nesterov=True, bb_adapt=True, n_peaks=30):
    """Mirror descent with BB step size and Nesterov acceleration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    eta = eta0
    no_improve = 0
    prev_a = a.copy()
    prev_g = None

    for it in range(10000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute convolution at Nesterov point
        b = fast_convolve(y)
        max_b = b.max()

        # Use top peaks weighted by proximity
        n_top = min(n_peaks, n)
        top_idx = np.argsort(b)[::-1][:n_top]
        weights = np.exp(-(b[top_idx] - max_b) / (max_b * 0.005 + 1e-300))
        weights /= weights.sum()

        # Compute subgradient
        g = np.zeros(n)
        for k, w in zip(top_idx, weights):
            js = max(0, int(k) - n + 1)
            je = min(n - 1, int(k))
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            g[idx] += w * y[int(k) - idx]

        # BB step size
        if bb_adapt and it >= 2 and prev_g is not None:
            s_vec = a - prev_a
            y_vec = g - prev_g
            s_dot_s = np.dot(s_vec, s_vec)
            s_dot_y = np.dot(s_vec, y_vec)
            y_dot_y = np.dot(y_vec, y_vec)
            if s_dot_s > 1e-15 and y_dot_y > 1e-15:
                if abs(s_dot_y) > 1e-15:
                    eta_bb = s_dot_s / abs(s_dot_y)
                    eta = 0.5 * eta + 0.5 * eta_bb
                else:
                    eta = s_dot_s / y_dot_y
                eta = max(1e-6, min(eta, 0.5))
            else:
                eta *= 0.999

        prev_a = a.copy()
        prev_g = g.copy()

        # Mirror descent update
        log_a = np.log(np.maximum(a, 1e-300))
        g_shifted = g - g.min()
        log_a_new = log_a - eta * g_shifted
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-300:
            a_new = a.copy()
            a_new_sum = 1.0
        a_new = a_new / a_new_sum

        # Nesterov
        if nesterov:
            alpha = 2.0 / (it + 2)
            y = (1.0 - alpha) * y + alpha * a_new
            y = np.maximum(y, 0)
            ys = y.sum()
            if ys < 1e-15:
                y = a_new.copy()
            else:
                y = y / ys

        a = a_new

        # Evaluate
        score = evaluate_local(a)
        score_y = 2.0 * n * fast_convolve(y).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        elif score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 500:
            rng = np.random.default_rng(it * 7 + 13)
            noise = rng.standard_normal(n) * (0.003 + 0.003 * np.sqrt(no_improve / 500.0))
            a = a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            y = a.copy()
            no_improve = 0
            eta = max(eta, eta0 * 0.5)

        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            if elapsed > 2:
                print(f"  MD iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} eta={eta:.6f}")

    return best_a, best_score


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=100000):
    """Nesterov LP alternating for refinement."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()

    y = a.copy()
    best_a = a.copy()

    t0 = time.monotonic()
    prev_best = best_score
    stall_rounds = 0
    rng = np.random.default_rng(42424)
    lp_time_total = 0.0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Check if we have enough time for LP + rest
        if lp_time_total > time_limit * 0.8:
            break

        lp_start = time.monotonic()
        b_new, M_b = solve_lp_best_response(y, solver)
        lp_time = time.monotonic() - lp_start
        lp_time_total += lp_time
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)

        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:

        if score_y < best_score:

                stall_rounds = 0
            else:
                stall_rounds += 1

                noise = rng.standard_normal(n) * noise_scale
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_rounds = 0

            if (it + 1) % 200 == 0:

def logsumexp_obj_and_grad(z, T):
    """Compute LogSumExp objective and gradient (sum-normalized)."""
    a = softmax_safe(z)
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp(b - max_b)
    w = exp_b / exp_b.sum()

    # Gradient of (2n/T) * logsumexp(b) w.r.t. a
    # d/d a_i [logsumexp(b)] = sum_k w_k * db_k/d a_i = sum_k w_k * a[k-i]
    grad_a = np.zeros(n)
    size = 2 * n - 1
    # Use FFT for efficiency
    # grad_a[i] = sum_k w[k] * a[k-i] (valid indices)
    # This is a correlation: grad_a = convolve(w_padded, a) appropriately
    # Simpler: direct loop (size is 2n-1, but w is sparse in practice for low T)
    for k in range(size):
        if w[k] < 1e-14:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a[k - idx]

    grad_a *= 2.0 * n / T

    # Convert to z-gradient (softmax parameterization)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)

    logsumexp = max_b + np.log(exp_b.sum())
    obj = (2.0 * n / T) * logsumexp
    return obj, grad_z


def logsumexp_lbgfs(a0, time_limit, T_stages=None, maxiter_per_stage=3000):
    """L-BFGS on log-sum-exp smoothed objective with decreasing temperature."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [15.0, 8.0, 4.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]

    n_stages = len(T_stages)
    iters_per_stage = max(30, maxiter_per_stage // n_stages)

    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        if time.monotonic() - t0 > time_limit - 3:
            break

        def func(z_inner):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            obj, _ = logsumexp_obj_and_grad(z_inner, T)
            return obj

        def jac(z_inner):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_inner)
            _, grad = logsumexp_obj_and_grad(z_inner, T)
            return grad

        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                                    'gtol': 1e-14, 'maxfun': iters_per_stage * 200,
                                    'disp': False})
            z = res.x
            sc = evaluate_local(softmax_safe(z))
            if (T_stages.index(T) if T in T_stages else 0) % 3 == 0:

def coordinate_descent_refine(a0, time_limit, n_steps=100000):
    """Coordinate descent targeting peak-contributing elements."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Find peaks within 0.01% of max
        threshold = max_b * 0.9999
        peaks = np.where(b >= threshold)[0]
        if len(peaks) == 0:
            peaks = np.array([int(np.argmax(b))])

        improved = False
        for peak_k in peaks[:3]:  # Limit to top 3 peaks
            if improved:
                break
            js = max(0, int(peak_k) - n + 1)
            je = min(n - 1, int(peak_k))
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[int(peak_k) - idx]
            top_m = min(8, len(idx))
            top_positions = idx[np.argsort(contribs)[::-1][:top_m]]

            for pos in top_positions:
                if a[pos] < 1e-12:
                    continue
                for factor in [0.9995, 0.999, 0.998, 0.997, 0.995, 0.99, 0.985, 0.98]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] - a_trial[pos]
                    if diff < 1e-16:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] = others + diff * others / others_sum
                    score = evaluate_local(a_trial)
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            if stall > 500:
                # Try simultaneous small adjustments
                a_trial = a.copy()
                peak_k = int(np.argmax(b))
                js = max(0, peak_k - n + 1)
                je = min(n - 1, peak_k)
                idx = np.arange(js, je + 1)
                contribs = a[idx] * a[peak_k - idx]
                top_positions = idx[np.argsort(contribs)[::-1][:5]]
                for pos in top_positions:
                    if a[pos] > 1e-12:
                        a_trial[pos] *= 0.999
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = evaluate_local(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
                else:
                    if stall > 2000:
                        break
        else:
            stall = 0

        if (step + 1) % 20000 == 0:
            elapsed = time.monotonic() - t0
            if elapsed > 1:
                print(f"  CoordDesc ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score
