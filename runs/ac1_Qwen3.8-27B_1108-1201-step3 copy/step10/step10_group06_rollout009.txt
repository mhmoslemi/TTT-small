def next_pow2(m):

def fast_convolve(a):

def golden_section_search_func(f, lo, hi, max_iter=35):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if hi - lo < 1e-16:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = f(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = f(d)
    return (lo + hi) * 0.5

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            v = a[i]
            if v < 1e-15:
                continue
            
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)
            n_aff = k_hi - k_lo + 1
            
            unaff_max = 0.0
            if k_lo > 0:
                unaff_max = max(unaff_max, float(conv[:k_lo].max()))
            if k_hi < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_hi + 1:].max()))
            
            partners = np.arange(k_lo, k_hi + 1) - i
            a_partners = a[partners]
            has_quad = (2 * i >= k_lo) and (2 * i <= k_hi)
            
            c_vals = np.empty(n_aff)
            for idx in range(n_aff):
                k = k_lo + idx
                if k == 2 * i:
                    c_vals[idx] = conv[k] - v * v
                else:
                    c_vals[idx] = conv[k] - 2.0 * v * a_partners[idx]
            
            quad_idx = 2 * i - k_lo if has_quad else -1
            
            upper = max(3.0 * v, 5.0 / n)
            
            def _score(x):
                return coord_descent_one_score(x, c_vals, a_partners, unaff_max, 
                                               has_quad, quad_idx, current_sum, v, n)
            
            best_x = golden_section_search_func(_score, 0.0, upper, max_iter=25)
            
            if abs(best_x - v) > 1e-14:
                delta = best_x - v
                a[i] = best_x
                current_sum += delta
                changes = 2.0 * delta * a_partners
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = delta * (2.0 * v + delta)
                conv[k_lo:k_hi + 1] += changes
                
                cur_score = 2.0 * n * float(conv.max()) / (current_sum * current_sum)
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
        
        if verbose and sweeps % 5 == 0:
            print(f"    CD sweep={sweeps}: best={best_score:.8f}")
    
    return best_a, best_score

def init_gaussian(n, rng, center_frac, width_frac):
    t = np.arange(n, dtype=np.float64)
    center = n * center_frac
    width = n * width_frac
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_flat_spectrum(n, rng):
    L = next_pow2(n)
    n_freq = L // 2 + 1
    phases = rng.uniform(0, 2.0 * np.pi, n_freq)
    amps = rng.uniform(0.5, 1.5, n_freq)
    spectrum = amps * np.exp(1j * phases)
    a = irfft(spectrum, L)[:n]
    a = np.abs(a)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def basin_hopping(a0, time_limit, rng, verbose=False):
    """Basin hopping with mirror descent as inner solver."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, float), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * float(fast_convolve(a).max())
    t0 = time.monotonic()
    count = 0
    
    while time.monotonic() - t0 < time_limit:
        count += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 30:
            break
        t_alloc = min(rem * 0.4, 100.0)
        
        # Choose perturbation type
        pert_type = count % 3
        perturbed = a.copy()
        
        if pert_type == 0:
            # Gaussian noise
            sigma = 0.02 + 0.02 * rng.random()
            perturbed = a + rng.standard_normal(n) * sigma
        elif pert_type == 1:
            # Block swap
            blen = int(rng.integers(n // 10, max(n // 5, 2)))
            i1 = int(rng.integers(0, max(n - blen, 1)))
            i2 = int(rng.integers(0, max(n - blen, 1)))
            if abs(i1 - i2) > blen:
                temp = perturbed[i1:i1+blen].copy()
                perturbed[i1:i1+blen] = perturbed[i2:i2+blen]
                perturbed[i2:i2+blen] = temp
        else:
            # Shifted mix
            shift = int(rng.integers(1, max(n // 8, 2)))
            shifted = np.roll(a, shift)
            perturbed = 0.7 * a + 0.3 * shifted
            perturbed = np.roll(perturbed, -shift // 2)  # try to center
        
        perturbed = np.maximum(perturbed, 0)
        ps = perturbed.sum()
        if ps < 1e-10:
            continue
        perturbed = perturbed / ps
        
        # Optimize the perturbed version
        a_opt, s_opt = mirror_descent(perturbed, t_alloc, initial_eta=0.5, tau_init=0.005)
        
        if s_opt < best_score - 1e-10:
            best_score = s_opt
            best_a = a_opt.copy()
            if verbose:
                print(f"    BH {count}: improved to {best_score:.8f}")
    
    return best_a, best_score
