def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

        try:
            res = minimize(obj, z, jac=grad, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-20,
                                   'gtol': 1e-14, 'maxfun': max_iter * 50})
            z = res.x
            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
            sc = local_eval(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
        except Exception:
            pass
    
    return best_a, best_score


def nesterov_subgradient(a0, time_limit, lr_init=0.01, momentum=0.9):
    """Nesterov accelerated subgradient method."""
    n = len(a0)
    a = normalize(a0)
    v = np.zeros(n)  # velocity
    best_a = a.copy()
    best_score = local_eval(a)
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    
    for step in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov look-ahead
        y = a + momentum * v
        
        b = fast_convolve(y)
        M = b.max()
        # Find active set
        threshold = M * 0.999
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        # Subgradient (average over active peaks)
        g = np.zeros(n)
        for k in active:
            js_lo = max(0, k - n + 1)
            js_hi = min(n - 1, k)
            idx = np.arange(js_lo, js_hi + 1)
            g[idx] += 2.0 * y[k - idx]
        g /= len(active)
        
        # Gradient step
        v = momentum * v - lr * g
        a_new = a + v
        a_new = normalize(a_new)
        a = a_new
        
        sc = local_eval(a)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
            no_improve = 0
            lr = min(lr * 1.005, 0.05)
        else:
            no_improve += 1
            if no_improve % 100 == 0:
                lr *= 0.9
                v *= 0.5
                no_improve = 0
        
        lr = max(lr, 1e-8)
    
    return best_a, best_score


def peak_shaving(a0, time_limit):
    """
    Direct peak shaving: iteratively reduce the maximum by solving
    a small optimization problem focused on the active peaks.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = local_eval(a)
    t0 = time.monotonic()
    lr = 0.05
    
    for step in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        M = b.max()
        # Active peaks (within 0.1% of max)
        threshold = M * 0.9999
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        
        # We want to minimize the mean of active peaks
        # Gradient of mean active peak
        g = np.zeros(n)
        for k in active:
            js_lo = max(0, k - n + 1)
            js_hi = min(n - 1, k)
            idx = np.arange(js_lo, js_hi + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)
        
        # Projected gradient step with backtracking
        for trial in range(20):
            step_size = lr / (2 ** trial)
            a_trial = a - step_size * g
            a_trial = normalize(a_trial)
            sc = local_eval(a_trial)
            if sc < best_score:
                best_score = sc
                best_a = a_trial.copy()
                a = a_trial
                lr = min(lr * 1.02, 0.1)
                break
        else:
            lr *= 0.5
            if lr < 1e-10:
                break
    
    return best_a, best_score


def interpolate_to_n(a, new_n):
    """Interpolate sequence to new length, maintaining sum=1."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    a = normalize(a)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a)
    return normalize(a_interp)


def structured_init(n, rng):
    """Create structured initial sequence."""
    x = np.linspace(0, 1, n)
    choice = rng.integers(0, 6)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.01)
    elif choice == 2:
        a = np.exp(-2 * (x - 0.5)**2 / (0.15**2))
    elif choice == 3:
        a = 0.7 + 0.3 * np.cos(2 * np.pi * 2 * x)
        a = np.maximum(a, 0.01)
    elif choice == 4:
        a = np.sin(np.pi * x) ** 0.5
    else:
        a = 1.0 + 0.2 * np.cos(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x)
    return normalize(a)


def symmetric_project(a):
    """Project onto symmetric sequences and then onto simplex."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    return normalize(a_sym)


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Primary method: adaptive-temperature LogSumExp L-BFGS.
    Secondary: Nesterov subgradient, peak shaving, multi-n exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = local_eval(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = local_eval(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.10f}")
    
    n_best = len(best_seq)
    
    def try_update(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.asarray(a, dtype=np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = local_eval(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")
    
    # === Phase 1: Adaptive LogSumExp L-BFGS on parent n ===
    print(f"\n=== Phase 1: Adaptive LogSumExp L-BFGS (n={n_best}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.35, 600.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        a1, sc1 = lbgfs_adaptive(best_seq, t1, T_start=None, T_decay=0.75)
        try_update(a1, "lbgfs_phase1")
    
    # === Phase 2: Peak shaving ===
    print(f"\n=== Phase 2: Peak Shaving (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.15, 200.0)
    if t2 > 15:
        a2, sc2 = peak_shaving(best_seq, t2)
        try_update(a2, "peak_shave")
    
    # === Phase 3: Nesterov subgradient ===
    print(f"\n=== Phase 3: Nesterov Subgradient (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.15, 200.0)
    if t3 > 15:
        a3, sc3 = nesterov_subgradient(best_seq, t3)
        try_update(a3, "nesterov")
    
    # === Phase 4: Second LogSumExp round ===
    print(f"\n=== Phase 4: LogSumExp Round 2 (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.20, 300.0)
    if t4 > 20:
        a4, sc4 = lbgfs_adaptive(best_seq, t4, T_start=None, T_decay=0.8, 
                                  min_T=1e-6)
        try_update(a4, "lbgfs_r2")
        # Peak shave after
        rem = deadline - time.monotonic()
        if rem > 20:
            a4b, _ = peak_shaving(best_seq, min(rem * 0.3, 60.0))
            try_update(a4b, "peak_shave_r2")

    print(f"\n=== Phase 5: Multi-n Exploration (score={best_score:.10f}) ===")
