If j = k-j (i.e., k = 2j): the two terms are the same (`a_j * a_j`), so `∂C_k / ∂a_j = 2*a_j`

In either case: `∂C_k / ∂a_j = 2*a_{k-j}` (where we interpret `a_{k-j} = 0` if k-j is out of range).

`grad_j = 0` for invalid j

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Mirror descent (multiplicative weights) with FFT-accelerated gradients
   and adaptive softmax smoothing
2. Binary-search coordinate descent exploiting 1D convexity
3. LP-based best-response alternating minimization for final polish
4. Multi-start with diverse initializations and adaptive restarts
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_of(a):
    """Compute 2*n*max(conv)/(sum)^2 quickly."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def compute_smooth_grad(w, a):
    """
    Compute grad_j = 2 * sum_k w[k] * a[k-j] for valid k-j.
    Uses FFT for efficiency.
    """
    n = len(a)
    # cross_correlation(w, a)[j] = sum_m w[m+j] * a[m]
    # = (rev(w) * a)[(2n-2) - j]
    w_rev = w[::-1]
    total_len = (2 * n - 1) + n - 1  # = 3n - 2
    fft_size = 1 << (total_len - 1).bit_length()
    W_rev = rfft(w_rev, fft_size)
    A = rfft(a, fft_size)
    conv_wa = irfft(W_rev * A, fft_size)
    # Extract: cross[j] = conv_wa[(2n-2) - j] for j in [0, n-1]
    indices = (2 * n - 2) - np.arange(n)
    cross = conv_wa[indices]
    return 2.0 * cross


def mirror_descent(a0, time_limit, tau_init, eta_init, verbose=True):
    """
    Mirror descent (multiplicative weights) for minimizing max conv on simplex.
    Uses adaptive smoothing and FTRL-style accumulation.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    
    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    
    t0 = time.monotonic()
    it = 0
    eta = eta_init
    tau = tau_init
    z = np.zeros(n)  # FTRL accumulator
    
    patience = 0
    best_patience_score = best_score
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        
        # Compute convolution
        conv = fast_convolve(a)
        M = conv.max()
        score = 2.0 * n * M
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        # Adaptive parameters
        if patience > 100:
            # Decrement tau (sharper) or reset
            if tau > 10.0:
                tau *= 0.9
            else:
                # Restart with perturbation
                noise = np.random.standard_normal(n) * 0.01 / np.sqrt(n)
                a = a + noise
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                z = np.zeros(n)
                patience = 0
            eta *= 0.9
        
        # Compute softmax weights
        w = np.exp(tau * (conv - M))
        w_sum = w.sum()
        w = w / w_sum
        
        # Compute gradient via FFT
        grad = compute_smooth_grad(w, a)
        
        # FTRL update
        z += eta * grad
        # a_new = exp(-z / eta_current) ... actually FTRL is:

        # = exp(-sum_s g_s * eta_s / eta_t) / Z

        # Scale to avoid overflow
        exponent -= exponent.max()
        a_new = a * np.exp(exponent)
        a_sum = a_new.sum()
        if a_sum < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / a_sum
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        a = a_new
        
        # Decay eta
        if it % 50 == 0:
            eta *= 0.99
            tau *= 0.995
        
        if verbose and it % 500 == 0:

            print(f"    MD it={it} ({elapsed:.0f}s): best={best_score:.8f} M={M:.6f} eta={eta:.6f} tau={tau:.1f}")
    
    return best_a, best_score
