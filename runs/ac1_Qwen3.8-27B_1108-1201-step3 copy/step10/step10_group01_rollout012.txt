    """Solve LP with only active constraints."""
    n = len(a)

    # Only use active rows
    T_active = T[active_constraints, :]

    prob = cp.Problem(cp.Minimize(M), constraints)

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(a, b):
    """Cross-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def evaluate_local(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2 locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def interpolate_sequence(f, new_n):
    """Interpolate normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def build_conv_matrix_rows(a, rows):
    """Build sparse rows of convolution matrix T where T[k,:] @ b = conv(a,b)[k]."""
    n = len(a)

        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        T[i, idx] = a[k - idx]
    return T

def mirror_descent_step(a, b, temperature, lr):
    """
    Fast mirror descent step: given fixed a, compute a descent direction
    for b using weighted convolution (weights concentrated on peaks).
    Returns updated b.
    """
    n = len(a)
    cross = fast_cross_convolve(a, b)
    max_c = cross.max()
    if max_c < 1e-15:
        return b.copy()
    # Weighted: emphasize peaks
    w = np.exp(temperature * (cross - max_c) / max_c)
    w = w / w.sum()
    # Gradient direction for b: g_j = sum_k w_k * a[k-j]
    # This is the cross-convolution of w with a, but in the right direction
    # g = conv(w, a) shifted... actually:
    # cross_conv[a,b][k] = sum_j a[j]*b[k-j]
    # d/d b_j [sum_k w_k * cross[k]] = sum_k w_k * a[k-j]
    # = (w * a reversed) convolved... let me compute directly
    # sum_k w_k * a[k-j] for valid k, j
    # This is the cross-correlation of w and a
    # = conv(flip(w), a) shifted? No...
    # Let's use: g_j = sum_k w[k] * a[k-j] where 0<=k-j<n and 0<=k<2n-1
    # j ranges 0..n-1, k ranges 0..2n-2
    # k-j ranges j..2n-2-j, but also 0<=k-j<n so k-j in [max(0,k-n+1), min(n-1,k)]
    # Actually: for fixed j, k ranges from j to j+n-1 (since k-j in [0, n-1])
    # So g_j = sum_{k=j}^{j+n-1} w[k] * a[k-j] = (w * flip(a)) ... 
    # This is the cross-correlation of w with reversed a, or equivalently:
    # g = irfft(rfft(w, 2n) * rfft(a, 2n)) ... no that's convolution
    # Let me just compute it as: g[j] = dot(w[j:j+n], a)
    g = np.zeros(n)
    for j in range(n):
        idx = np.arange(j, min(j + n, len(w)))
        g[j] = np.dot(w[idx], a[:len(idx)])
    # Mirror descent: b_new_j ∝ b_j * exp(-lr * g_j)
    log_b = np.log(np.maximum(b, 1e-300))
    log_b -= lr * g
    log_b -= log_b.max()
    b_new = np.exp(log_b)
    b_new = b_new / b_new.sum()
    return b_new


def mirror_descent_step_fast(a, b, temperature, lr):
    """
    Fast mirror descent using FFT.
    g_j = sum_k w[k] * a[k-j] = (w ⋆ a)[j] (cross-correlation)
    We can compute this via FFT: g = ifft(conj(FFT(w)) * FFT(a))
    but we need to be careful about the exact indexing.
    
    Actually: cross_conv[a,b][k] = sum_j a[j]*b[k-j]
    So d/d b_j [sum_k w_k * cross_conv[k]] = sum_k w_k * a[k-j]
    
    Let's define: for each j, g[j] = sum_{k=0}^{2n-2} w[k] * a[k-j] (where 0<=k-j<n)
    = sum_{m=0}^{n-1} w[j+m] * a[m] where m = k-j
    = (w ⋆ a)[j] (cross-correlation of w with a, shifted)
    
    In FFT terms: if W = FFT(w, 2n), A = FFT(a, 2n)
    then cross-correlation = ifft(conj(W) * A) ... but we need the right shift.
    
    Let me just use a direct approach with FFT:
    g[j] = sum_{m=0}^{n-1} w[j+m] * a[m]
    This is the convolution of w with reverse(a), shifted by j.
    Let ar = reverse(a), then (w * ar)[j+n-1] = sum_m w[m] * ar[j+n-1-m] = sum_m w[m]*a[m-1-(j+n-1-m)]... 
    
    This is getting confusing. Let me use a simpler FFT approach:
    Define w_padded of length 2n-1, a_padded of length n.
    g[j] = sum_{m=0}^{n-1} w[j+m] * a[m]
    
    This is equivalent to: g = correlate(w, a) where correlate(w,a)[j] = sum_m w[j+m]*a[m]
    In numpy: np.correlate(w, a, mode='valid') gives exactly this!
    """
    n = len(a)
    cross = fast_cross_convolve(a, b)
    max_c = cross.max()
    if max_c < 1e-15:
        return b.copy()
    w = np.exp(temperature * (cross - max_c) / max_c)
    w = w / w.sum()
    # g[j] = sum_{m=0}^{n-1} w[j+m] * a[m]
    g = np.correlate(w, a, mode='valid')
    # Mirror descent
    log_b = np.log(np.maximum(b, 1e-300))
    log_b -= lr * g
    log_b -= log_b.max()
    b_new = np.exp(log_b)
    s = b_new.sum()
    if s > 1e-15:
        b_new = b_new / s
    return b_new


def nesterov_md_optimize(a0, b0, time_limit, solver, n_md_per_lp=20,

    sb = b.sum()

        b = a.copy()
        sb = 1.0
    b = b / sb
    
    # Nesterov momentum
    y = b.copy()  # extrapolated point
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    stall = 0
    prev_best = best_score
    rng = np.random.default_rng(123)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Determine if we do LP or MD step
        if it % n_md_per_lp == n_md_per_lp - 1:
            # Do an LP step
            cross = fast_cross_convolve(a, y)
            max_c = cross.max()
            # Active set: top constraints

            b_new, M_val = solve_lp_active_set(a, active, max_c, solver, max_iter=300)
            if b_new is None:

        else:

            y = mirror_descent_step_fast(a, y, temperature, lr)

def lbgfs_logsumexp(a0, time_limit, T_stages=None, maxiter_total=20000):
    """L-BFGS with LogSumExp smoothing, multi-stage temperature cooling."""
    n = len(a0)
    a0 = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a0.sum()
    if s < 1e-15:
        a0 = np.ones(n) / n
        s = 1.0
    a0 = a0 / s
    z = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [20.0, 10.0, 5.0, 3.0, 2.0, 1.0, 0.6, 0.4, 0.25, 0.15, 0.1, 0.06, 0.04, 0.02]

    n_stages = len(T_stages)
    iters_per = max(20, maxiter_total // n_stages)

    for T_stage in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func_loc(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a_in = np.exp(z_in - z_in.max())
            a_in = a_in / a_in.sum()
            b_in = fast_convolve(a_in)
            max_b = b_in.max()
            exp_b = np.exp(b_in - max_b)
            lse = max_b + np.log(exp_b.sum())
            return (2.0 * n / T_stage) * lse

        def jac_loc(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a_in = np.exp(z_in - z_in.max())
            a_in = a_in / a_in.sum()
            b_in = fast_convolve(a_in)
            max_b = b_in.max()
            exp_b = np.exp(b_in - max_b)
            w = exp_b / exp_b.sum()
            # Gradient of sum_k w_k * conv(a,a)[k] w.r.t. a
            # d/d a_j [conv(a,a)[k]] = a[k-j] (for valid k-j)
            # So grad_j = sum_k w_k * a[k-j] = correlate(w, a)[j]
            grad_a = np.correlate(w, a_in, mode='valid')
            grad_a *= 2.0 / T_stage
            S = np.dot(grad_a, a_in)
            return a_in * (grad_a - S)

        try:
            res = minimize(
                fun=func_loc, x0=z, jac=jac_loc, method='L-BFGS-B',
                options={'maxiter': iters_per, 'ftol': 1e-16,
                         'gtol': 1e-12, 'maxfun': iters_per * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    a_out = np.exp(z - z.max())
    a_out = a_out / a_out.sum()
    return a_out


def peak_coord_descent(a0, time_limit, n_steps=100000, top_m=8):
    """Coordinate descent targeting top convolution peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        top_peaks = np.argsort(b)[::-1][:min(top_m, len(b))]

        improved = False
        for peak_k in top_peaks[:3]:
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            order = np.argsort(contribs)[::-1][:3]
            for oi in order:
                pos = idx[oi]
                if a[pos] < 1e-12:
                    continue
                for factor in [0.99, 0.985, 0.98, 0.97, 0.96]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] * (1.0 - factor)
                    if diff < 1e-15:
                        break
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] = others + diff * others / others_sum
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 200:
                break

    return best_a, best_score


def multi_scale_init(target_n, solver, time_budget):
    """
    Hierarchical initialization: start small, optimize, upscale.
    Returns a good initial sequence of length target_n.
    """
    scales = [40, 80, 160, 320, 640]
    scales = [s for s in scales if s < target_n]
    if not scales:
        return np.ones(target_n) / target_n
    
    # Add target
    if target_n not in scales:
        scales.append(target_n)
    
    t_total = time.monotonic() + time_budget
    n_scales = len(scales)
    current = np.ones(scales[0]) / scales[0]
    
    for si, n in enumerate(scales):
        rem = t_total - time.monotonic()
        if rem < 2:
            if n == target_n:
                break
            current = interpolate_sequence(current, target_n)
            break
        
        if si == 0:
            # Initialize at smallest scale
            x = np.linspace(0, 1, n)
            # Use a structured init that's neither too flat nor too peaked
            current = np.sin(np.pi * x) ** 0.8
            s = current.sum()
            current = current / s
        else:
            # Upscale from previous
            current = interpolate_sequence(current, n)
        
        # Allocate time proportional to scale
        time_for_scale = rem / (n_scales - si) * 0.8
        if time_for_scale < 3:
            time_for_scale = 3
        
        # Run MD optimization at this scale
        b_init = current.copy()
        a_opt, score = nesterov_md_optimize(
            current, b_init, min(time_for_scale, rem * 0.6),
            solver, n_md_per_lp=15, temperature=8.0, lr=0.03,
            print_every=500
        )
        current = a_opt
        print(f"  Multi-scale: n={n}, score={score:.8f}")
    
    return current


def lp_nesterov_optimize(a0, time_limit, solver, max_iters=50000,
                         stall_patience=10):
    """Standard Nesterov LP alternation (for reference/comparison)."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall = 0
    rng = np.random.default_rng(999)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Use active set for speed
        cross = fast_cross_convolve(a, y)
        max_c = cross.max()
        K = min(len(cross), max(200, n // 2))
        active = np.argsort(cross)[::-1][:K]
        
        b_new, M_b = solve_lp_active_set(a, active, max_c, solver, max_iter=400)
        if b_new is None:
            b_new, M_b = solve_lp_full(a, solver, max_iter=600)
        if b_new is None:
            stall += 1
            if stall > 20:
                break
            continue

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        if (it + 1) % 500 == 0:
            if best_score < prev_best - 1e-9:
                stall = 0
            else:
                stall += 1
            prev_best = best_score

            if stall >= stall_patience:
                noise = rng.standard_normal(n) * 0.015
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall = 0

            elapsed = time.monotonic() - t0
            print(f"    LP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. Multi-scale hierarchical initialization
    2. Fast mirror-descent Nesterov optimization (main workhorse)
    3. Periodic LP correction
    4. LogSumExp L-BFGS refinement
    5. Peak coordinate descent polish
    6. Multi-length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1000
        x = np.linspace(0, 1, n_init)
        best_seq = np.sin(np.pi * x) ** 0.8
        s = best_seq.sum()
        best_seq = best_seq / s
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)
    print(f"Starting: n={n_best}, score={best_score:.8f}")

    # === Phase 1: Multi-scale initialization (if parent isn't already good) ===
    # Only do this if we have time and the parent isn't very good
    rem = deadline - time.monotonic()
    if rem > 300 and best_score > 1.50:
        print(f"\n=== Phase 1: Multi-scale init, remaining={rem:.0f}s ===")
        t_ms = min(rem * 0.15, 120.0)
        ms_result = multi_scale_init(n_best, solver, t_ms)
        try_update(ms_result, "multi_scale")
    elif rem > 200:
        print(f"\n=== Phase 1: Skipping multi-scale (parent good) ===")

    # === Phase 2: Main MD+Nesterov optimization ===
    print(f"\n=== Phase 2: MD Nesterov, n={n_best}, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.40, 550.0)
    if t2 > 30:
        print(f"  Budget: {t2:.0f}s")
        a_md, _ = nesterov_md_optimize(
            best_seq, best_seq.copy(), t2, solver,
            n_md_per_lp=25, temperature=10.0, lr=0.025,
            print_every=300
        )
        try_update(a_md, "md_nesterov")

    # === Phase 3: LP Nesterov (full, for correction) ===
    print(f"\n=== Phase 3: LP Nesterov, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.20, 250.0)
    if t3 > 30:
        print(f"  Budget: {t3:.0f}s")
        a_lp, _ = lp_nesterov_optimize(best_seq, t3, solver,
                                         max_iters=30000, stall_patience=12)
        try_update(a_lp, "lp_nesterov")

    # === Phase 4: LogSumExp L-BFGS refinement ===
    print(f"\n=== Phase 4: LogSumExp L-BFGS, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.18, 200.0)
    if t4 > 20:
        T_stages = [15.0, 8.0, 5.0, 3.0, 2.0, 1.0, 0.6, 0.4, 0.25, 0.15, 0.1, 0.06, 0.04, 0.02]
        a_lb = lbgfs_logsumexp(best_seq, t4, T_stages=T_stages, maxiter_total=30000)
        try_update(a_lb, "lbgfs")

    # === Phase 5: Peak coordinate descent ===
    print(f"\n=== Phase 5: Peak CD, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.08, 80.0)
    if t5 > 10:
        a_cd, _ = peak_coord_descent(best_seq, t5, n_steps=50000, top_m=8)
        try_update(a_cd, "peak_cd")

    # === Phase 6: Second MD pass with different params ===
    print(f"\n=== Phase 6: MD pass 2, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.15, 150.0)
    if t6 > 20:
        # Perturb best sequence for diversity
        x = np.linspace(0, 2 * np.pi, len(best_seq))
        pert = 0.02 * np.cos(3 * x) + 0.015 * np.sin(5 * x)
        a_pert = best_seq + pert
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()
        
        a_md2, _ = nesterov_md_optimize(
            a_pert, a_pert.copy(), t6, solver,
            n_md_per_lp=20, temperature=15.0, lr=0.02,
            print_every=200
        )
        try_update(a_md2, "md_pass2")

    # === Phase 7: Try longer sequence ===
    print(f"\n=== Phase 7: Longer sequence, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        n_cur = len(best_seq)
        alt_n = int(n_cur * 1.3)
        if alt_n <= 5000:
            t7 = min(rem * 0.25, 200.0)
            print(f"  Trying n={alt_n}, budget={t7:.0f}s")
            init_alt = interpolate_sequence(best_seq, alt_n)
            a_alt, _ = nesterov_md_optimize(
                init_alt, init_alt.copy(), t7, solver,
                n_md_per_lp=25, temperature=10.0, lr=0.025,
                print_every=300
            )
            try_update(a_alt, f"md_n{alt_n}")
            
            rem = deadline - time.monotonic()
            if rem > 30:
                a_alt2 = lbgfs_logsumexp(a_alt, min(rem * 0.5, 60.0),
                                          T_stages=[5.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05],
                                          maxiter_total=10000)
                try_update(a_alt2, f"lbgfs_n{alt_n}")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final polish, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.5, 50.0)
        a_fin = lbgfs_logsumexp(best_seq, t_fin,
                                 T_stages=[1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01],
                                 maxiter_total=15000)
        try_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 15:
        a_cd2, _ = peak_coord_descent(best_seq, min(rem * 0.5, 25.0),
                                       n_steps=30000, top_m=6)
        try_update(a_cd2, "final_cd")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
