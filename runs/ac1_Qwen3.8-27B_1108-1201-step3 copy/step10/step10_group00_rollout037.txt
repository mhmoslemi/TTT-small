def fast_conv2(a, b):
    """Linear convolution of a and b via FFT."""
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

    best_M = np.inf

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_conv(a, b=None):
    """Linear convolution of a and b (or a with itself) via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def eval_score(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2 locally."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    c = fast_conv(a)
    return float(2.0 * n * c.max() / (s * s))


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-300:
        return np.ones_like(a) / len(a)
    return a / s


def mirror_lp_solve(a_fixed, n_iter=400, eta_scale=1.0):
    """Solve LP: min M s.t. conv(a_fixed, b)[k] <= M, sum(b)=1, b>=0
    using mirror descent (multiplicative weights).
    Returns (b_best, M_best)."""
    n = len(a_fixed)
    b = np.ones(n) / n
    best_M = np.inf
    best_b = b.copy()

    for t in range(1, n_iter + 1):
        conv = fast_conv(a_fixed, b)
        M = conv.max()
        if M < best_M:
            best_M = M
            best_b = b.copy()

        # Active set: peaks within tolerance of max
        tol = max(M * 5e-4, 1e-12)
        active = np.where(conv >= M - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(conv)])

        # Subgradient: g[i] = (1/m) * sum_{k in active} a_fixed[k-i]
        g = np.zeros(n)
        for k in active:
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            if i_lo <= i_hi:
                g[i_lo:i_hi + 1] += a_fixed[k - i_hi:k - i_lo + 1][::-1]
        g /= len(active)

        # Step size: diminishing
        eta = eta_scale / np.sqrt(t + 1)

        # Mirror descent (multiplicative weights) update
        log_b = np.log(np.maximum(b, 1e-300))
        log_b -= eta * g
        log_b -= log_b.max()
        b = np.exp(log_b)
        s = b.sum()
        if s < 1e-300:
            break
        b /= s

    return best_b, best_M

def logsumexp_obj_grad(z, n, T):
    """Compute LogSumExp-smoothed objective and gradient w.r.t. z (log-domain)."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()
    b = fast_conv(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T)
    lse = max_b + T * np.log(exp_terms.sum())
    obj = 2.0 * n * lse

    # Gradient w.r.t. a
    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        grad_a[i_lo:i_hi + 1] += 2.0 * w[k] * a[k - i_hi:k - i_lo + 1][::-1]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage=2000, deadline=None):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    a = normalize(a0)
    z = np.log(np.maximum(a, 1e-300))
    t0 = time.monotonic()
    if deadline is None:
        deadline = t0 + time_limit
    n_stages = len(T_stages)
    iters_per_stage = max(10, maxiter_per_stage // n_stages)

    for stage_i, stage_T in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 2:
            break

        def func(z):
            if time.monotonic() > deadline:
                return 1e15
            obj, _ = logsumexp_obj_grad(z, n, stage_T)
            return obj

        def jac(z):
            if time.monotonic() > deadline:
                return np.zeros_like(z)
            _, g = logsumexp_obj_grad(z, n, stage_T)
            return g

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-12, 'maxfun': iters_per_stage * 200,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass

        # Update a and check score
        a = normalize(np.exp(z - z.max()))
        sc = eval_score(a)
        if stage_i % 3 == 0 or stage_i == n_stages - 1:
            print(f"    L-BFGS stage {stage_i} (T={stage_T}): score={sc:.8f}")

    return normalize(np.exp(z - z.max()))

def structured_init(n, rng):
    """Create a structured initial sequence."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    elif choice == 6:
        # Flat-top with smooth edges
        a = np.minimum(np.minimum(x * 10, (1 - x) * 10), 1.0)
        a = np.maximum(a, 0)
    else:
        # Raised cosine
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    a = normalize(a)
    return a


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    return normalize(f_interp)


def perturbed_sequence(a, rng, strength=0.03):
    """Create a perturbed version of sequence a."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 7)
    phase = rng.uniform(0, 2 * np.pi)
    perturbation = strength * np.cos(freq * x + phase)
    a_pert = a + perturbation
    return normalize(a_pert)

        h = np.ones(m) / m
    h = h / sh
    # Full sequence for initial evaluation
    a_full = np.concatenate([h, h[::-1]])
    a_full = normalize(a_full)
    best_a = a_full.copy()
    best_score = eval_score(a_full)

    y_h = h.copy()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Build full y for LP
        y_full = np.concatenate([y_h, y_h[::-1]])
        y_full = normalize(y_full)

        # Mirror LP on full space but project back to symmetric
        b_new, _ = mirror_lp_solve(y_full, n_iter=n_inner)
        # Symmetrize: take first half
        b_h = (b_new[:m] + b_new[m:2*m][::-1]) / 2.0
        b_h = np.maximum(b_h, 0)
        sb = b_h.sum()
        if sb < 1e-15:
            b_h = y_h.copy()
            sb = 1.0
        b_h = b_h / sb

        # Nesterov update on half
        alpha = 2.0 / (it + 2.0)
        y_h = (1.0 - alpha) * y_h + alpha * b_h
        y_h = np.maximum(y_h, 0)
        sy = y_h.sum()
        if sy < 1e-15:
            y_h = h.copy()
            sy = 1.0
        y_h = y_h / sy

        h = ((it + 1) * h + alpha * (it + 2) * b_h) / (it + 1 + alpha * (it + 2))
        h = np.maximum(h, 0)
        sh2 = h.sum()
        if sh2 < 1e-15:
            h = y_h.copy()
        else:
            h = h / sh2

        # Evaluate full symmetric sequence
        a_full = np.concatenate([h, h[::-1]])
        a_full = normalize(a_full)
        sc = eval_score(a_full)
        if sc < best_score:
            best_score = sc
            best_a = a_full.copy()

        y_full = np.concatenate([y_h, y_h[::-1]])
        y_full = normalize(y_full)
        sc_y = eval_score(y_full)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y_full.copy()

        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SymNL iter={it} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses fast mirror-descent LP (no external solver), Nesterov acceleration,
    LogSumExp L-BFGS polish, subgradient refinement, and multi-length search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = eval_score(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = eval_score(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = eval_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Quick Nesterov LP on parent (60s) ===
    print(f"\n=== Phase 1: Quick Nesterov LP n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 60.0)
    if t_alloc > 10:
        print(f"  Budget: {t_alloc:.0f}s, n_inner=400")
        a_out, sc = nesterov_lp_refine(best_seq, t_alloc, n_inner=400, eta_scale=1.0)
        local_update(a_out, "quick_nl")

    # === Phase 2: Extended Nesterov LP (200s) ===
    print(f"\n=== Phase 2: Extended Nesterov LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 200.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s, n_inner=500")
        a_out, sc = nesterov_lp_refine(best_seq, t_alloc, n_inner=500, eta_scale=1.5)
        local_update(a_out, "ext_nl1")

    # === Phase 3: LogSumExp L-BFGS polish ===
    print(f"\n=== Phase 3: L-BFGS LogSumExp (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 180.0)
    if t_alloc > 20:
        T_stages = [10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
        print(f"  T_stages ({len(T_stages)}), budget={t_alloc:.0f}s")
        a_out = lbgfs_refine(best_seq, t_alloc, T_stages, maxiter_per_stage=5000,
                             deadline=deadline - (deadline - time.monotonic() - t_alloc))
        local_update(a_out, "lbgfs1")

    # === Phase 4: Subgradient polish ===
    print(f"\n=== Phase 4: Subgradient Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 100.0)
    if t_alloc > 10:
        a_out, sc = subgradient_polish(best_seq, t_alloc, initial_eta=0.05)
        local_update(a_out, "subgrad1")

    # === Phase 5: Symmetric Nesterov LP ===
    print(f"\n=== Phase 5: Symmetric Nesterov LP (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 120.0)
    if t_alloc > 15:
        # Make parent symmetric
        n_cur = len(best_seq)
        a_sym = (best_seq + best_seq[::-1]) / 2.0
        a_sym = normalize(a_sym)
        print(f"  Budget: {t_alloc:.0f}s, symmetric n={n_cur}")
        a_out, sc = symmetric_nesterov_lp(a_sym, t_alloc, n_inner=400)
        local_update(a_out, "sym_nl")

    # === Phase 7: Nesterov LP Round 2 (from updated best) ===
    print(f"\n=== Phase 7: Nesterov LP Round 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 120.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_out, sc = nesterov_lp_refine(best_seq, t_alloc, n_inner=400, eta_scale=2.0)
        local_update(a_out, "nl_r2")

    # === Phase 6: Try n=1500 ===
    print(f"\n=== Phase 6: Try n=1500 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        t_n1500 = min(rem * 0.15, 180.0)
        print(f"  Budget: {t_n1500:.0f}s")
        init_1500 = interpolate_sequence(best_seq, 1500)
        a_1500, _ = nesterov_lp_refine(init_1500, t_n1500 * 0.6, n_inner=300)
        local_update(a_1500, "nl_n1500")

        rem = deadline - time.monotonic()
        if best_seq is not None and len(best_seq) == 1500 and rem > 30:
            T3 = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
            a_1500b = lbgfs_refine(best_seq, min(rem, 80.0), T3, maxiter_per_stage=3000,
                                   deadline=deadline)
            local_update(a_1500b, "lbgfs_n1500")

    # === Phase 8: Try n=2000 ===
    print(f"\n=== Phase 8: Try n=2000 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        t_n2000 = min(rem * 0.12, 150.0)
        print(f"  Budget: {t_n2000:.0f}s")
        init_2000 = interpolate_sequence(best_seq, 2000)
        a_2000, _ = nesterov_lp_refine(init_2000, t_n2000 * 0.7, n_inner=250)
        local_update(a_2000, "nl_n2000")

    # === Phase 9: Perturbation escape ===
    print(f"\n=== Phase 9: Perturbation Escape (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        t_pert = min(rem * 0.15, 150.0)
        n_cur = len(best_seq)
        for trial in range(4):
            rem_sub = deadline - time.monotonic()
            if rem_sub < 20:
                break
            strength = 0.015 + 0.01 * trial
            a_pert = perturbed_sequence(best_seq, rng, strength=strength)
            t_sub = min(rem_sub * 0.25, 40.0)
            a_opt, _ = nesterov_lp_refine(a_pert, t_sub, n_inner=300, eta_scale=1.0)
            local_update(a_opt, f"pert{trial}")

    # === Phase 10: Final L-BFGS polish ===
    print(f"\n=== Phase 10: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.30, 60.0)
        if t_fin > 10:
            T_fin = [2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
            a_fin = lbgfs_refine(best_seq, t_fin, T_fin, maxiter_per_stage=3000,
                                 deadline=deadline)
            local_update(a_fin, "final_lbgfs")

    # === Phase 11: Final subgradient ===
    rem = deadline - time.monotonic()
    if rem > 10:
        a_ms, _ = subgradient_polish(best_seq, min(rem * 0.4, 20.0), initial_eta=0.02)
        local_update(a_ms, "final_subgrad")

    # === Phase 12: Final LP ===
    rem = deadline - time.monotonic()
    if rem > 8:
        a_lp, _ = nesterov_lp_refine(best_seq, min(rem * 0.7, 15.0), n_inner=200, eta_scale=1.0)
        local_update(a_lp, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = eval_score(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
