    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)

    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1) / (k + 1)
        return np.maximum(v - theta, 0)
    return np.ones(n) / n

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based best-response alternating minimization (initial driver)
2. Exact 1D coordinate descent with convex subproblem solving per coordinate
3. Dual-guided multiplicative updates for escaping local minima
4. Multi-phase refinement with adaptive time allocation
5. Very fine coordinate descent polishing for last-mile improvements
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score assuming sum(a)=1: 2*n*max(conv)."""
    n = len(a)
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max())


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n


def find_solver():
    """Find the best available LP solver."""
    cached = getattr(find_solver, '_cached', None)
    if cached is not None:
        return cached
    for solver_name in [cp.CLARABEL, cp.MOSEK, cp.GUROBI, cp.ECOS, cp.SCS, cp.ECOS_BB]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                find_solver._cached = solver_name
                return solver_name
        except Exception:
            continue
    find_solver._cached = cp.ECOS
    return cp.ECOS


def lp_best_response(a, solver):
    """
    Solve LP: min M s.t. conv(a,b)_k <= M for all k, sum(b)=1, b>=0.
    Returns (b_opt, M_opt).
    """
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a

    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    ones = np.ones(2 * n - 1)
    constraints = [
        T @ b_var <= M_var * ones,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            return a.copy(), float('inf')

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float('inf')

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return a.copy(), float('inf')
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else float('inf')
    return b_val, M_val


def lp_alternating(a0, time_limit, solver, verbose=True):
    """LP-based best-response alternating minimization with adaptive damping."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_score = score_normalized(a)

    it = 0
    stagnation_count = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)

            break

        score_b = score_normalized(b_new)

        elif it <= 30:

        else:
            alpha = max(0.25, 8.0 / (it + 20.0))

                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps > 80:
                break
        else:
            no_improve_sweeps = 0

        if verbose and sweeps % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD[{label}] sweep={sweeps} ({elapsed:.0f}s): best={best_score:.10f} M={M:.12f}")

    return best_a, best_score


def fine_cd(a0, time_limit, verbose=True):
    """Very fine coordinate descent with multiplicative updates for polishing."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-18:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            L = k_end - k_start + 1

            unaff_max = 0.0
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            a_shifted = a[:L]
            conv_affected = conv[k_start:k_end + 1]
            ai = a[i]

            C = conv_affected - 2.0 * ai * a_shifted
            has_quad = (2 * i >= k_start) and (2 * i <= k_end)
            if has_quad:
                quad_idx = 2 * i - k_start
                C[quad_idx] = conv_affected[quad_idx] - ai * ai

            best_x = ai
            best_h = M

            # Very fine multiplicative steps
            for factor in [0.997, 0.998, 0.999, 1.001, 1.002, 1.003, 0.995, 1.005]:
                x_try = ai * factor
                if x_try < 0:
                    continue
                new_aff = C + 2.0 * x_try * a_shifted
                if has_quad:
                    new_aff = new_aff.copy()
                    new_aff[quad_idx] = C[quad_idx] + x_try * x_try
                h_try = max(unaff_max, float(new_aff.max()))
                if h_try < best_h - 1e-17:
                    best_h = h_try
                    best_x = x_try

            # Also try x=0 if ai is small
            if ai < 1e-8:
                new_aff = C.copy()
                if has_quad:
                    new_aff[quad_idx] = C[quad_idx]
                h0 = max(unaff_max, float(new_aff.max()))
                if h0 < best_h - 1e-17:
                    best_h = h0
                    best_x = 0.0

            if best_x != ai and best_h < M - 1e-14:
                a[i] = best_x
                conv[k_start:k_end + 1] = C + 2.0 * best_x * a_shifted
                if has_quad:
                    conv[k_start + quad_idx] = C[quad_idx] + best_x * best_x
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 30:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FineCD sweep={sweeps} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score


def dual_guided_update(a, conv, alpha=0.5):
    """
    Compute a dual-guided update using peak distribution.
    Uses the KKT condition: at optimum, the 'pressure' (T^T y)_j should be
    constant on the support of a.
    """
    n = len(a)
    M = float(conv.max())
    K = 40
    if 2 * n - 1 <= K:
        peak_idx = np.arange(2 * n - 1)
        w = np.ones(len(peak_idx)) / len(peak_idx)
    else:
        peak_idx = np.argpartition(conv, -K)[-K:]
        pv = conv[peak_idx]
        # Softmax weighting
        temp = 0.01 * M
        w = np.exp((pv - pv.max()) / temp)
        w = w / w.sum()

    # Compute pressure p_j = sum_k w_k * a[k-j] (for valid k-j)
    p = np.zeros(n)
    for idx, k in enumerate(peak_idx):
        k = int(k)
        if w[idx] < 1e-15:
            continue
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        # a[k-j] for j in [j_start, j_end] = a[k-j_end] to a[k-j_start]
        p[j_start:j_end + 1] += w[idx] * a[k - j_end:k - j_start + 1][::-1]

    # At optimum: p_j ≈ M for j in support(a)
    # Update: a_j *= (M / p_j)^alpha for j in support
    a_new = a.copy()
    for j in range(n):
        if a[j] > 1e-15:
            ratio = M / (p[j] + 1e-18)
            ratio = np.clip(ratio, 0.5, 2.0)
            a_new[j] = a[j] * (ratio ** alpha)

    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def diverse_init(n, rng, init_type):
    """Generate diverse initializations."""
    t = np.arange(n)
    center = (n - 1) / 2.0
    if init_type == 0:
        a = np.maximum(1.0 - np.abs(t - center) / (center + 1), 0)
    elif init_type == 1:
        x = (t - center) / (center + 1)
        a = 0.3 + 0.7 * x * x
    elif init_type == 2:
        x = (t - center) / (center + 1)
        a = np.maximum(0.5 * (1 + np.cos(np.pi * x)), 0) ** 1.5
    elif init_type == 3:
        c1 = n * 0.3
        c2 = n * 0.7
        w = n * 0.15
        a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    elif init_type == 4:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 3 * t / n)
    elif init_type == 5:
        a = 1.0 - 0.5 * np.cos(4 * np.pi * t / n)
        a = np.maximum(a, 0.01)
    else:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 5 * t / n) + 0.2 * np.cos(2 * np.pi * 7 * t / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def propose_candidate():
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses LP alternating, exact coordinate descent, dual-guided updates,
    and fine polishing with multiple restarts.
    """
    deadline = time.monotonic() + 1770.0
    rng = np.random.default_rng(12345)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
    except Exception:
        pass

    if parent is not None:
        n = len(parent)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    else:
        n = 1024
        best_seq = np.ones(n) / n
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"Initial: n={n}, score={best_score:.10f}")

    solver = find_solver()
    print(f"LP solver: {solver}")

    def try_update(candidate, lbl):
        nonlocal best_seq, best_score
        candidate = np.maximum(candidate, 0.0)
        s = candidate.sum()
        if s < 0.01:
            return
        candidate = candidate / s
        try:
            sc = float(evaluate_sequence(candidate.tolist()))
        except Exception:
            sc = float('inf')
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = candidate.copy()
            print(f"  NEW BEST [{lbl}]: {sc:.10f}")

    # === Phase 1: LP Alternating ===
    t1_budget = min(250.0, (deadline - time.monotonic()) * 0.25)
    if time.monotonic() < deadline - 60:
        print(f"\n=== Phase 1: LP Alternating ({t1_budget:.0f}s) ===")
        a_lp, s_lp = lp_alternating(best_seq, t1_budget, solver, verbose=True)
        try_update(a_lp, "lp")

    # === Phase 2: Exact Coordinate Descent (main workhorse) ===
    t2_budget = min(500.0, (deadline - time.monotonic()) * 0.4)
    if time.monotonic() < deadline - 100:
        print(f"\n=== Phase 2: Exact CD ({t2_budget:.0f}s) ===")
        a_cd, s_cd = exact_cd(best_seq, t2_budget, verbose=True, label="main")
        try_update(a_cd, "cd_main")

    # CD from LP result if different
    if time.monotonic() < deadline - 100:
        t2b = min(150.0, (deadline - time.monotonic()) * 0.2)
        if t2b > 30:
            print(f"\n  CD from LP ({t2b:.0f}s)")
            a_cd2, s_cd2 = exact_cd(a_lp, t2b, verbose=False, label="lp")
            try_update(a_cd2, "cd_lp")

    # === Phase 3: Dual-guided + CD ===
    if time.monotonic() < deadline - 80:
        t3 = min(200.0, (deadline - time.monotonic()) * 0.25)
        print(f"\n=== Phase 3: Dual-guided + CD ({t3:.0f}s) ===")
        conv_cur = fast_convolve(best_seq)
        # Multiple dual-guided updates
        a_dual = best_seq.copy()
        for dg_iter in range(5):
            if time.monotonic() > deadline - 80:
                break
            conv_d = fast_convolve(a_dual)
            a_dual = dual_guided_update(a_dual, conv_d, alpha=0.3)
        try_update(a_dual, "dual")

        # CD from dual-guided
        t3cd = min(t3 / 2, deadline - time.monotonic() - 80)
        if t3cd > 20:
            a_dcd, s_dcd = exact_cd(a_dual, t3cd, verbose=False, label="dual")
            try_update(a_dcd, "dual_cd")

    # === Phase 4: Fine Polish ===
    if time.monotonic() < deadline - 20:
        t4 = min(200.0, deadline - time.monotonic() - 20)
        print(f"\n=== Phase 4: Fine Polish ({t4:.0f}s) ===")
        a_fp, s_fp = fine_cd(best_seq, t4, verbose=True)
        try_update(a_fp, "fine")

    # === Phase 5: Random restart + CD ===
    if time.monotonic() < deadline - 50:
        t5 = min(100.0, deadline - time.monotonic() - 50)
        print(f"\n=== Phase 5: Restart exploration ({t5:.0f}s) ===")
        for init_type in range(4):
            if time.monotonic() > deadline - 50:
                break
            init = diverse_init(n, rng, init_type)
            # Mix with current best
            for mix in [0.3, 0.5, 0.7]:
                a_mix = (1 - mix) * best_seq + mix * init
                a_mix = np.maximum(a_mix, 0)
                a_mix = a_mix / a_mix.sum()
                try_update(a_mix, f"mix{init_type}_{mix}")
            # Quick CD from mix
            remaining = deadline - time.monotonic() - 50
            if remaining > 30:
                a_rcd, _ = exact_cd(a_mix, min(remaining * 0.3, 40), verbose=False, label=f"r{init_type}")
                try_update(a_rcd, f"rcd{init_type}")

    # === Phase 6: Final fine polish ===
    if time.monotonic() < deadline - 5:
        t6 = deadline - time.monotonic() - 5
        if t6 > 10:
            print(f"\n=== Phase 6: Final Polish ({t6:.0f}s) ===")
            a_final, s_final = fine_cd(best_seq, t6, verbose=True)
            try_update(a_final, "final")
            # One more exact CD pass
            remaining = deadline - time.monotonic() - 3
            if remaining > 10:
                a_fin2, _ = exact_cd(best_seq, remaining, verbose=False, label="fin2")
                try_update(a_fin2, "fin2")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
