def fast_convolve(a):
    n = len(a)

    return irfft(A * A, fft_size)[:size]

    """Compute the score without calling evaluate_sequence (for speed)."""
    a = np.asarray(a, dtype=np.float64)

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)

        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)

        return np.ones(n) / n

def de_search(a_init, time_limit, pop_size=20, F=0.8, CR=0.7, verbose=False):

    Differential Evolution in the sequence space.
    Uses a mutation strategy suited for this problem.
    """

    population = [a_init.copy()]
    best_score = evaluate_score(a_init)
    
    for i in range(1, pop_size):
        # Initialize others with perturbations
        a_new = a_init * (1.0 + 0.1 * np.random.standard_normal(n))
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s
        population.append(a_new)
    
    # Track scores
    scores = [evaluate_score(pop) for pop in population]
    
    gen = 0
    while time.monotonic() - t0 < time_limit:
        gen += 1
        improved = False
        
        # Sort by score
        order = np.argsort(scores)
        population = [population[i] for i in order]
        scores = [scores[i] for i in order]
        
        for i in range(pop_size):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Mutation: DE/rand/1
            r1, r2 = np.random.choice(pop_size, 2, replace=False)
            while r1 == i:
                r1 = np.random.choice(pop_size)
            while r2 == i or r2 == r1:
                r2 = np.random.choice(pop_size)
            
            mutant = population[r1] + F * (population[r2] - population[i])
            
            # Crossover
            trial = population[i].copy()
            for j in range(n):
                if np.random.random() < CR:
                    trial[j] = mutant[j]
            
            # Non-negativity and normalization
            trial = np.maximum(trial, 0)
            s = trial.sum()
            if s < 1e-15:
                continue
            trial = trial / s
            
            trial_score = evaluate_score(trial)
            
            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                improved = True
            
            if trial_score < best_score:
                best_score = trial_score
        
        if verbose and gen % 5 == 0:
            print(f"    DE gen={gen} ({time.monotonic()-t0:.0f}s): best={best_score:.6f}")
        
        # Adapt F
        if gen % 10 == 0:
            F = max(0.3, F * 0.95)
    
    return population[0], best_score

We want to reduce the peaks. In the frequency domain, the autocorrelation has DFT |A|^2 (where A is the FFT of zero-padded a). 

If we want to reduce the peak at lag k*, we need to reduce the contribution of frequencies where the phase alignment creates constructive interference at that lag.

A practical approach:
1. Compute c = a*a
2. Identify the peak lags
3. For each peak lag k*, compute the "pressure" vector p_k* where p_k*[i] = a[k*-i] (the contribution of coordinate i to lag k*)
4. Reduce coordinates that contribute most to the peaks
5. Redistribute the mass to coordinates that contribute less to the peaks

This is essentially a "gradient descent on the max" approach, but done in a more structured way.

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def blahut_iteration(a0, n_iter=100, alpha=0.3, verbose=False):
    """
    Blahut-like spectral design iteration.
    Flattens the power spectrum of the zero-padded sequence.
    """
    n = len(a0)
    a = normalize(a0.copy())
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    best_a = a.copy()
    best_score = eval_score_norm(a)
    
    for it in range(n_iter):
        # Zero-pad and compute FFT
        a_padded = np.zeros(fft_size)
        a_padded[:n] = a
        A = rfft(a_padded, fft_size)
        
        # Power spectrum
        power = np.abs(A) ** 2
        max_p = power.max()
        if max_p < 1e-30:
            break
        
        # Target power: blend current with flat
        # Flat target: total_power / n_freqs
        total_power = power.sum()
        n_freqs = len(power)
        flat_power = total_power / n_freqs
        
        # Blend
        new_power = (1 - alpha) * power + alpha * flat_power
        new_mag = np.sqrt(np.maximum(new_power, 0))
        
        # Preserve phase
        mag = np.abs(A)
        phase = np.zeros_like(A)
        mask = mag > 1e-30
        phase[mask] = A[mask] / mag[mask]
        
        A_new = new_mag * phase
        
        # Inverse FFT
        a_new_padded = irfft(A_new, fft_size)
        a_new = a_new_padded[:n]
        
        # Project to non-negative, normalize
        a_new = normalize(a_new)
        
        score = eval_score_norm(a_new)
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and it % 20 == 0:
            print(f"      Blahut it={it}: best={best_score:.8f}")
    
    return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Blahut-like spectral design (iterative time/frequency projection)
2. Exact 1D coordinate descent with golden-section search
3. Multiplicative weights guided by peak-lag pressure
4. Differential Evolution in reduced space for global exploration
5. Multi-length exploration and final fine-tuning
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def eval_score_norm(a):
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max()

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def blahut_iteration(a0, n_iter=200, alpha=0.4, verbose=False):
    n = len(a0)
    a = normalize(a0.copy())
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    best_a = a.copy()
    best_score = eval_score_norm(a)
    for it in range(n_iter):
        a_padded = np.zeros(fft_size)
        a_padded[:n] = a
        A = rfft(a_padded, fft_size)
        power = np.abs(A) ** 2
        max_p = power.max()
        if max_p < 1e-30:
            break
        total_power = power.sum()
        n_freqs = len(power)
        flat_power = total_power / n_freqs
        new_power = (1 - alpha) * power + alpha * flat_power
        new_mag = np.sqrt(np.maximum(new_power, 0))
        mag = np.abs(A)
        phase = np.ones_like(A)
        mask = mag > 1e-30
        phase[mask] = A[mask] / mag[mask]
        A_new = new_mag * phase
        a_new_padded = irfft(A_new, fft_size)
        a_new = a_new_padded[:n]
        a_new = normalize(a_new)
        score = eval_score_norm(a_new)
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
        a = a_new
        if verbose and it % 50 == 0:
            print(f"      Blahut it={it}: best={best_score:.8f}")
    return best_a, best_score

def coord_descent(a0, time_limit, verbose=False, label=""):
    n = len(a0)
    a = normalize(a0.copy())
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0
    sweep = 0
    while time.monotonic() - t0 < time_limit:
        sweep += 1
        improved = False
        for i in range(n):
            ai = a[i]
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            unaff_max = -1e30
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))
            affected = conv[k_start:k_end + 1].copy()
            k_indices = np.arange(k_start, k_end + 1)
            ki_minus_i = k_indices - i
            coeffs = a[ki_minus_i].copy() + ai
            has_quad = (2 * i >= k_start) and (2 * i <= k_end)
            quad_idx = 2 * i - k_start if has_quad else -1
            lo = -ai
            hi = max(0.5 * ai, 1e-10)
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c = hi - gr * (hi - lo)
            d = lo + gr * (hi - lo)
            fc = max(unaff_max, float((affected + c * coeffs).max()) + (c * c if has_quad and (2*i==c+k_start else 0)))
            # Recompute properly
            def obj_at(delta):
                if delta < -ai - 1e-12:
                    return 1e30
                new_aff = affected + delta * coeffs
                if has_quad:
                    new_aff = new_aff.copy()
                    new_aff[quad_idx] += delta * delta
                return max(unaff_max, float(new_aff.max()))
            fc = obj_at(c)
            fd = obj_at(d)
            for _ in range(45):
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - gr * (hi - lo)
                    fc = obj_at(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + gr * (hi - lo)
                    fd = obj_at(d)
            best_delta = (lo + hi) / 2.0
            best_new_max = obj_at(best_delta)
            # Check boundaries
            for delta in [0.0, -ai]:
                val = obj_at(delta)
                if val < best_new_max:
                    best_new_max = val
                    best_delta = delta
            if best_new_max < M - 1e-14 and abs(best_delta) > 1e-15:
                a[i] = max(ai + best_delta, 0.0)
                conv[k_start:k_end + 1] = affected + best_delta * coeffs
                if has_quad:
                    conv[k_end + 1 if False else 2 * i] += best_delta * best_delta
                # Fix: only add delta^2 at position 2*i in the full conv array
                conv[2 * i] = affected[quad_idx] + best_delta * coeffs[quad_idx] + best_delta * best_delta if has_quad else affected[quad_idx] + best_delta * coeffs[quad_idx]
                # Actually let's just recompute the affected part properly
                conv[k_start:k_end + 1] = affected + best_delta * coeffs
                if has_quad:
                    conv[2 * i] += best_delta * best_delta
                M = best_new_max
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        if not improved:
            no_improve += 1
            if no_improve > 50:
                break
        else:
            no_improve = 0
        if sweep % 100 == 0:
            conv = fast_convolve(a)
            M = float(conv.max())
        if verbose and sweep % 20 == 0:
            print(f"      CD {label} sweep={sweep} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")
    return best_a, best_score

def multiplicative_weights(a0, time_limit, verbose=False, label=""):
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = eval_score_norm(a)
    t0 = time.monotonic()
    it = 0
    eta = 0.3
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        if M < 1e-15:
            break
        threshold = M * 0.92
        peak_lags = np.where(conv >= threshold)[0]
        pressure = np.zeros(n)
        for k in peak_lags:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            for i in range(i_start, i_end + 1):
                ki = k - i
                if 0 <= ki < n:
                    pressure[i] += a[ki] + a[i]
        p_max = pressure.max()
        if p_max < 1e-15:
            break
        pressure = pressure / p_max
        a_new = a * np.exp(-eta * pressure)
        a_new = normalize(a_new)
        score = eval_score_norm(a_new)
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
        a = a_new
        eta = max(0.01, eta * 0.995)
        if verbose and it % 200 == 0:
            print(f"      MW {label} it={it} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")
    return best_a, best_score

def smart_perturb(a0, rng, conv=None):
    n = len(a0)
    if conv is None:
        conv = fast_convolve(a0)
    M = conv.max()
    if M < 1e-15:
        return a0.copy()
    threshold = M * 0.9
    peak_lags = np.where(conv >= threshold)[0]
    pressure = np.zeros(n)
    for k in peak_lags:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            ki = k - i
            if 0 <= ki < n:
                pressure[i] += a0[i] * a0[ki]
    p_max = pressure.max()
    if p_max < 1e-20:
        noise = rng.standard_normal(n)
        a_new = a0 + 0.05 * noise * a0
        return normalize(a_new)
    pressure = pressure / p_max
    strength = 0.05
    noise = rng.standard_normal(n)
    a_new = a0 * (1.0 - strength * pressure + 0.3 * strength * noise)
    return normalize(a_new)

def de_local(a0, time_limit, pop_size=12, verbose=False, label=""):
    n = len(a0)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    population = [a0.copy()]
    best_score = eval_score_norm(a0)
    for i in range(1, pop_size):
        a_new = a0 * (1.0 + 0.05 * rng.standard_normal(n))
        a_new = normalize(a_new)
        population.append(a_new)
    scores = [eval_score_norm(p) for p in population]
    gen = 0
    F = 0.5
    CR = 0.8
    while time.monotonic() - t0 < time_limit:
        gen += 1
        order = np.argsort(scores)
        population = [population[i] for i in order]
        scores = [scores[i] for i in order]
        for i in range(pop_size):
            if time.monotonic() - t0 > time_limit:
                break
            candidates = [j for j in range(pop_size) if j != i]
            if len(candidates) < 2:
                continue
            r1, r2 = rng.choice(candidates, 2, replace=False)
            mutant = population[r1] + F * (population[r2] - population[i])
            trial = population[i].copy()
            cr_mask = rng.random(n) < CR
            trial[cr_mask] = mutant[cr_mask]
            trial = normalize(trial)
            trial_score = eval_score_norm(trial)
            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
            if trial_score < best_score:
                best_score = trial_score
        F = max(0.2, F * 0.98)
        if verbose and gen % 10 == 0:
            print(f"      DE {label} gen={gen} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")
    best_idx = np.argmin(scores)
    return population[best_idx], best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
