                for fi in [0.6, 0.75, 0.85, 0.95, 1.05, 1.15, 1.25]:
                    for fj in [0.6, 0.75, 0.85, 0.95, 1.05, 1.15, 1.25]:
                        new_ai = ai * fi
                        new_aj = aj * fj
                        if new_ai < 0 or new_aj < 0:
                            continue

                        di = new_ai - ai
                        dj = new_aj - aj
                        if abs(di) < 1e-18 and abs(dj) < 1e-18:
                            continue

                        # Quick rejection: estimate contribution at peak k
                        new_k_approx = conv[k] + ai * dj + aj * di + di * dj
                        if new_k_approx >= M - 1e-13:
                            continue

                        # Full verification
                        a[i] = new_ai
                        a[j] = new_aj
                        conv_check = fast_convolve(a)
                        new_M = float(conv_check.max())

                        if new_M < best_new_M - 1e-14:
                            best_new_M = new_M
                            best_pair = (new_ai, new_aj)

                        # Restore
                        a[i] = ai
                        a[j] = aj

                if best_pair is not None:
                    a[i] = best_pair[0]
                    a[j] = best_pair[1]
                    conv = fast_convolve(a)
                    M = best_new_M
                    cur_score = 2.0 * n * M
                    if cur_score < best_score - 1e-14:
                        best_score = cur_score
                        best_a = a.copy()
                    improved = True

        if not improved and rounds > 5:
            break

        if verbose and rounds % 20 == 0:
            print(f"    PCD {label} round={rounds} ({time.monotonic()-t0:.0f}s): best={best_score:.6f}")

    return best_a, best_score, rounds


def sa_polish(a0, time_limit, verbose=False, label=""):
    """Simulated annealing with peak-guided moves."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()

    T0 = M * 0.01
    it = 0
    converg = 0.01
    no_improve = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        it += 1
        T = T0 * np.exp(-converg * it)
        if T < M * 1e-6:
            break

        # Choose move type
        move_type = rng.random()

        if move_type < 0.6:
            # Peak-guided: perturb a peak-contributing coordinate
            peak_k = np.argmax(conv)
            i_start = max(0, peak_k - n + 1)
            i_end = min(n - 1, peak_k)
            # Weight by contribution
            contribs = np.array([a[i] * a[peak_k - i] for i in range(i_start, i_end + 1)])
            total_c = contribs.sum()
            if total_c > 1e-30:
                probs = contribs / total_c
                idx = rng.choice(len(contribs), p=probs)
                i = i_start + idx
            else:
                i = rng.integers(n)
            delta = rng.standard_normal() * T * a[i] / max(M, 1e-30)
            a[i] = max(a[i] + delta, 0.0)
        elif move_type < 0.8:
            # Pair swap: move mass between two coordinates
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            amount = rng.standard_normal() * T * min(a[i], a[j]) / max(M, 1e-30)
            a[i] = max(a[i] + amount, 0.0)
            a[j] = max(a[j] - amount, 0.0)
        else:
            # Random coordinate perturbation
            i = rng.integers(n)
            delta = rng.standard_normal() * T * a[i] / max(M, 1e-30)
            a[i] = max(a[i] + delta, 0.0)

        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
            continue
        # Don't renormalize (score is scale-invariant)

        conv = fast_convolve(a)
        new_M = float(conv.max())
        new_score = 2.0 * n * new_M

        delta_score = new_score - 2.0 * n * M
        if delta_score < 0:
            M = new_M
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # Metropolis acceptance
            if rng.random() < np.exp(-delta_score / max(T * M, 1e-30)):
                M = new_M
                no_improve += 1
            else:
                # Reject
                a = best_a.copy()
                conv = fast_convolve(a)
                M = float(conv.max())
                no_improve += 1

        if no_improve > 1000:
            break

        if verbose and it % 5000 == 0:
            print(f"    SA {label} it={it} ({time.monotonic()-t0:.0f}s): best={best_score:.6f}, T={T/M:.6f}")

    return best_a, best_score


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def fine_cd(a0, time_limit, label=""):
    """Very fine coordinate descent for final polishing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for i in range(n):
            ai = a[i]
            if ai < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_max = float(conv[:k_start].max())
            else:
                unaff_max = -1e30
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            affected = conv[k_start:k_end + 1]
            coeffs = a[(k_start - i):(k_end - i + 1)] + ai
            has_quad = (2 * i >= k_start) and (2 * i <= k_end)
            quad_idx = 2 * i - k_start if has_quad else -1

            best_delta = 0.0
            best_new_max = M

            for factor in [0.9999, 0.9995, 0.999, 0.998, 0.997, 0.995,
                           0.99, 0.985, 0.98, 1.005, 1.01, 1.015, 1.02, 1.025]:
                delta = ai * (factor - 1.0)
                if delta < -ai - 1e-12:
                    continue
                delta = max(delta, -ai)
                if abs(delta) < 1e-18:
                    continue
                new_affected = affected + delta * coeffs
                if has_quad:
                    new_affected = new_affected.copy()
                    new_affected[quad_idx] += delta * delta
                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-16:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-15:
                a[i] = max(ai + best_delta, 0.0)
                conv[k_start:k_end + 1] += best_delta * coeffs
                if has_quad:
                    conv[2 * i] += best_delta * best_delta
                M = best_new_max
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            break

    return best_a, best_score


def check_and_update(a, best_seq, best_score, label=""):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score


def propose_candidate():
    """Main search combining enhanced CD, pairwise CD, SA polish, and multi-length exploration."""
    deadline = time.monotonic() + 1785.0
    rng = np.random.default_rng(12345)

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 1024
        best_seq = np.ones(n_init, dtype=np.float64) / n_init
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Fallback init: n={n_init}, score={best_score:.6f}")

    # === Phase 1: Smart CD (improved) ===
    remaining = deadline - time.monotonic()
    t1 = min(remaining * 0.30, 400.0)
    print(f"\n=== Phase 1: Smart CD (best={best_score:.6f}, budget={t1:.0f}s) ===")
    a1, s1, sw1 = smart_coord_descent(best_seq, t1, verbose=True, label="p1")
    best_seq, best_score = check_and_update(a1, best_seq, best_score, "p1_scd")

    # === Phase 2: Pairwise CD + Smart CD loop ===
    print(f"\n=== Phase 2: Pairwise + CD (best={best_score:.6f}) ===")
    remaining = deadline - time.monotonic()
    n_rounds = 4
    for rnd in range(n_rounds):
        if time.monotonic() > deadline - 300:
            break
        t2a = min(remaining * 0.35, 120.0)
        t2b = min(remaining * 0.35, 120.0)
        remaining = deadline - time.monotonic()

        # Pairwise CD
        a2p, s2p, _ = pairwise_cd(best_seq, t2a, verbose=True, label=f"p2r{rnd}_pcd")
        best_seq, best_score = check_and_update(a2p, best_seq, best_score, f"p2_pcd_r{rnd}")

        # Follow with CD
        if time.monotonic() < deadline - 250:
            t2c = min((deadline - time.monotonic()) * 0.3, 80.0)
            a2c, s2c, _ = smart_coord_descent(best_seq, t2c, verbose=False, label=f"p2r{rnd}_cd")
            best_seq, best_score = check_and_update(a2c, best_seq, best_score, f"p2_cd_r{rnd}")

    # === Phase 3: SA Polish ===
    print(f"\n=== Phase 3: SA Polish (best={best_score:.6f}) ===")
    remaining = deadline - time.monotonic()
    t3 = min(remaining * 0.15, 200.0)
    a3, s3 = sa_polish(best_seq, t3, verbose=True, label="p3")
    best_seq, best_score = check_and_update(a3, best_seq, best_score, "p3_sa")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-Length (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    alt_lengths = [x for x in [768, 1280, 1536, 2048] if x != n_best]

    for n_alt in alt_lengths[:3]:
        if time.monotonic() > deadline - 200:
            break
        remaining = deadline - time.monotonic()
        init_alt = interpolate_seq(best_seq, n_alt)
        t4a = min(remaining * 0.4, 80.0)
        a4, s4, _ = smart_coord_descent(init_alt, t4a, verbose=False, label=f"p4_n{n_alt}")
        best_seq, best_score = check_and_update(a4, best_seq, best_score, f"p4_n{n_alt}")

    # === Phase 5: Restart with perturbation ===
    print(f"\n=== Phase 5: Restart (best={best_score:.6f}) ===")
    for rnd in range(3):
        if time.monotonic() > deadline - 150:
            break
        # Perturb
        a_pert = best_seq.copy()
        n_p = len(a_pert)
        conv_p = fast_convolve(a_pert)
        peak_k = np.argmax(conv_p)
        # Reduce peak contributors
        i_start = max(0, peak_k - n_p + 1)
        i_end = min(n_p - 1, peak_k)
        strength = 0.05 + 0.02 * rnd
        for i in range(i_start, i_end + 1):
            j = peak_k - i
            if i == j:
                a_pert[i] *= (1 - strength * 0.5)
            else:
                a_pert[i] *= (1 - strength * rng.random() * 0.3)

        a_pert = np.maximum(a_pert, 0)
        s_p = a_pert.sum()
        if s_p > 1e-15:
            a_pert = a_pert / s_p

        t5 = min((deadline - time.monotonic()) * 0.4, 60.0)
        a5, s5, _ = smart_coord_descent(a_pert, t5, verbose=False, label=f"p5r{rnd}")
        best_seq, best_score = check_and_update(a5, best_seq, best_score, f"p5_r{rnd}")

    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    if time.monotonic() < deadline - 30:
        t6 = min((deadline - time.monotonic()) * 0.5, 120.0)
        a6, s6 = fine_cd(best_seq, t6, label="p6")
        best_seq, best_score = check_and_update(a6, best_seq, best_score, "p6_fine")

    # One more smart CD pass
    if time.monotonic() < deadline - 15:
        t6b = min((deadline - time.monotonic()) * 0.6, 60.0)
        a6b, s6b, _ = smart_coord_descent(best_seq, t6b, verbose=False, label="p6b")
        best_seq, best_score = check_and_update(a6b, best_seq, best_score, "p6b_scd")

    # Final fine pass
    if time.monotonic() < deadline - 5:
        a6c, s6c = fine_cd(best_seq, (deadline - time.monotonic()) * 0.7, label="p6c")
        best_seq, best_score = check_and_update(a6c, best_seq, best_score, "p6c_fine")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
