import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def lse_obj_grad(z, n, T):
    """
    Compute LogSumExp-smoothed objective and gradient w.r.t. z.
    a = softmax(z), objective = 2n * (1/T) * log(sum exp(T*b)) where b=conv(a,a).
    Uses FFT for O(n log n) gradient.
    """
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()

    # Compute b = conv(a, a)
    b = fast_convolve(a)

    # Compute LogSumExp
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    sum_exp = exp_Tb.sum()
    lse = (max_Tb + np.log(sum_exp)) / T
    obj = 2.0 * n * lse

    # Softmax weights w_k = exp(T*b_k) / sum exp(T*b_k)
    w = exp_Tb / sum_exp

    # Compute g[i] = sum_k w[k] * a[k-i] via FFT cross-correlation
    fft_size = 1 << (2 * n - 1).bit_length()
    FA = rfft(a, fft_size)
    FW = rfft(w, fft_size)
    cross = irfft(FA * np.conj(FW), fft_size)

    # g[i] = cross[(fft_size - i) % fft_size]
    indices = (fft_size - np.arange(n)) % fft_size
    g = cross[indices]

    # dF/da_i = 4n * g[i]
    dF_da = 4.0 * n * g

    # dF/dz_i = a_i * (dF_da_i - sum_j a_j * dF_da_j)
    S = np.dot(dF_da, a)
    grad_z = a * (dF_da - S)

    return obj, grad_z


def lbgfs_fft_optimize(a0, time_limit, T_stages=None, maxiter_total=20000):
    """L-BFGS on LogSumExp-smoothed objective with FFT gradients."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [30.0, 15.0, 8.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1, 0.06, 0.04, 0.025, 0.015]

    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()
    iters_per_stage = max(50, maxiter_total // len(T_stages))

    for T_stage in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func(z_loc):
            obj, _ = lse_obj_grad(z_loc, n, T_stage)
            return obj

        def jac(z_loc):
            _, grad = lse_obj_grad(z_loc, n, T_stage)
            return grad

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': iters_per_stage * 500,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass

        # Check actual (non-smoothed) score
        a_check = np.exp(z - z.max())
        a_check = a_check / a_check.sum()
        score = 2.0 * n * fast_convolve(a_check).max()
        if score < best_score:
            best_score = score
            best_a = a_check.copy()

    return best_a, best_score


def solve_lp_best_response(a_fixed, solver=None, verbose=False):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.asarray(b_var.value, dtype=np.float64)
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s < 1e-15:
                return None, None
            b_val = b_val / s
            return b_val, float(max(0, M_var.value))
    except Exception:
        pass
    for alt in [cp.SCS, cp.ECOS, cp.CLARABEL, cp.SCIP, cp.GLPK]:
        if alt == solver:
            continue
        try:
            prob.solve(solver=alt, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s < 1e-15:
                    return None, None
                b_val = b_val / s
                return b_val, float(max(0, M_var.value))
        except Exception:
            continue
    return None, None


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.SCS, cp.ECOS, cp.PDLP, cp.SCIP, cp.GLPK]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.SCS


def nesterov_lp_optimize(a0, time_limit, solver=None, max_iters=50000,

    n = len(a0)

        s = 1.0

    best_a = a.copy()

    t0 = time.monotonic()
    prev_best = best_score
    stall = 0
    rng = np.random.default_rng(12345)

        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            stall += 1
            if stall > 25:
                break
            continue

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()

        else:
            a = a / sa

        if score_y < best_score:

def subgradient_refine(a0, time_limit, n_steps=50000):
    """Subgradient method with adaptive step size on active peak set."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)

            elapsed = time.monotonic() - t0
            print(f"    SubGrad {elapsed:.0f}s: score={best_score:.10f} lr={lr:.6f}")

    return best_a, best_score


def polynomial_ramp_init(n, order, rng):
    """Create a polynomial ramp initialization: a_j ∝ (j+0.5)^order."""
    x = np.arange(n, dtype=np.float64) + 0.5
    a = x ** order
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

        a = x ** 2
    elif choice == 2:
        a = np.exp(3 * x)
    elif choice == 3:
        a = x ** 0.8 * (1 - 0.3 * x)
    elif choice == 4:
        a = (1 + 2 * x) * x
    else:
        a = x ** 1.2 * (1 - 0.2 * x ** 2)
    a = np.maximum(a, 1e-10)
    s = a.sum()
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 1e-10)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization with FFT-accelerated gradients.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Try to load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = 2.0 * len(parent) * fast_convolve(parent).max()
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = polynomial_ramp_init(n_init, 2, rng)
        best_score = 2.0 * len(best_seq) * fast_convolve(best_seq).max()
        print(f"Default init (poly ramp): n={len(best_seq)}, score={best_score:.10f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = 2.0 * len(a) * fast_convolve(a).max()
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    # === Phase 1: FFT-based LogSumExp L-BFGS (fast, do first to warm up) ===
    print(f"\n=== Phase 1: LogSumExp L-BFGS (FFT gradients), score={best_score:.10f} ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.12, 120.0)
    if t1 > 10:
        T_stages = [20.0, 10.0, 6.0, 4.0, 2.5, 1.5, 1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1, 0.06, 0.04, 0.025]
        print(f"  Budget: {t1:.0f}s, {len(T_stages)} T-stages")
        a_lb, _ = lbgfs_fft_optimize(best_seq, t1, T_stages=T_stages, maxiter_total=30000)
        try_update(a_lb, "lbgfs_fft_p1")

    # === Phase 2: Primary Nesterov LP ===
    print(f"\n=== Phase 2: Nesterov LP, score={best_score:.10f} ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.40, 550.0)
    if t2 > 40:
        print(f"  Budget: {t2:.0f}s")
        a_lp, _ = nesterov_lp_optimize(best_seq, t2, solver=solver,
                                        max_iters=100000, stall_patience=12)
        try_update(a_lp, "nlp_phase2")

    # === Phase 3: LogSumExp L-BFGS refinement (after LP) ===
    print(f"\n=== Phase 3: LogSumExp L-BFGS refine, score={best_score:.10f} ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.15, 150.0)
    if t3 > 10:
        T_stages = [8.0, 5.0, 3.0, 2.0, 1.2, 0.8, 0.5, 0.35, 0.2, 0.12, 0.07, 0.04, 0.02]
        a_lb2, _ = lbgfs_fft_optimize(best_seq, t3, T_stages=T_stages, maxiter_total=25000)
        try_update(a_lb2, "lbgfs_fft_p3")

    # === Phase 4: Peak-directed coordinate descent ===
    print(f"\n=== Phase 4: Peak CD, score={best_score:.10f} ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.10, 100.0)
    if t4 > 10:
        a_cd, _ = peak_coord_descent(best_seq, t4, n_steps=200000, top_m=8)
        try_update(a_cd, "peak_cd")

    # === Phase 5: Subgradient refinement ===
    print(f"\n=== Phase 5: Subgradient, score={best_score:.10f} ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.08, 80.0)
    if t5 > 8:
        a_sg, _ = subgradient_refine(best_seq, t5, n_steps=80000)
        try_update(a_sg, "subgrad")

    # === Phase 6: Second LP pass ===
    print(f"\n=== Phase 6: LP Pass 2, score={best_score:.10f} ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.15, 180.0)
    if t6 > 30:
        print(f"  Budget: {t6:.0f}s")
        a_lp2, _ = nesterov_lp_optimize(best_seq, t6, solver=solver,
                                         max_iters=80000, stall_patience=15)
        try_update(a_lp2, "nlp_phase6")

    # === Phase 7: Multi-start with diverse initializations ===
    print(f"\n=== Phase 7: Multi-start diversity, score={best_score:.10f} ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        n_cur = len(best_seq)
        t_each = min(rem / 4.0, 40.0)
        inits = []
        # Different polynomial orders
        inits.append(polynomial_ramp_init(n_cur, 3, rng))
        inits.append(polynomial_ramp_init(n_cur, 1, rng))
        inits.append(tuned_ramp_init(n_cur, rng))
        # Perturbed current best
        x = np.linspace(0, 2 * np.pi, n_cur)
        pert = 0.04 * np.cos(2 * x) + 0.02 * np.sin(3 * x)
        a_pert = best_seq + pert
        a_pert = np.maximum(a_pert, 0)
        inits.append(a_pert / a_pert.sum())

        for pi, init in enumerate(inits):
            if time.monotonic() > deadline - 30:
                break
            if t_each < 15:
                break
            print(f"  Multi-start {pi+1}/{len(inits)}, t={t_each:.0f}s")
            # Quick LP + L-BFGS
            a_ms, _ = nesterov_lp_optimize(init, t_each * 0.6, solver=solver,
                                            max_iters=15000, stall_patience=8)
            try_update(a_ms, f"ms_lp_{pi}")
            rem = deadline - time.monotonic()
            if rem > 15 and best_seq is not None:
                a_ms2, _ = lbgfs_fft_optimize(a_ms, min(rem * 0.3, 20.0),
                                               T_stages=[5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05],
                                               maxiter_total=5000)
                try_update(a_ms2, f"ms_lb_{pi}")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish, score={best_score:.10f} ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.4, 40.0)
        a_fin, _ = lbgfs_fft_optimize(best_seq, t_fin,
                                       T_stages=[2.0, 1.0, 0.6, 0.4, 0.25, 0.15, 0.1, 0.06, 0.04, 0.02, 0.01],
                                       maxiter_total=15000)
        try_update(a_fin, "final_lb")

    rem = deadline - time.monotonic()
    if rem > 15:
        a_cd2, _ = peak_coord_descent(best_seq, min(rem * 0.5, 20.0),
                                       n_steps=50000, top_m=5)
        try_update(a_cd2, "final_peak")

    rem = deadline - time.monotonic()
    if rem > 10:
        a_sg2, _ = subgradient_refine(best_seq, min(rem * 0.6, 15.0), n_steps=30000)
        try_update(a_sg2, "final_sg")

    # Final verification
    if best_seq is not None:
        fs = 2.0 * len(best_seq) * fast_convolve(best_seq).max()
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
