import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_cross(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def project_simplex(v):
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(np.maximum(a_fixed, 0))
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.SCS, cp.ECOS, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_increasing_avg(a0, time_limit, solver=None):
    """Nesterov LP with increasing average (IA) for optimal no-regret bound."""
    if solver is None:

    cumulative_b = np.zeros(n)

    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score

    for it in range(200000):
        if time.monotonic() - t0 > time_limit:
            break
        a_t = a.copy()
        b_new, M_b = solve_lp_best_response(a_t, solver)
        if b_new is None:
            stall_count += 1
            if stall_count > 20:
                break
            continue

        cumulative_b += b_new
        alpha = 1.0 / (it + 1)
        a = cumulative_b * alpha
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = a_t.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score - 1e-12:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0

        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-8:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            if stall_count > 30:

                a = a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall_count = 0
            elapsed = time.monotonic() - t0
            if (it + 1) % 1000 == 0:
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def gauss_newton_peak_eq(a0, time_limit, K=5):
    """Gauss-Newton active-set peak equalization for autocorrelation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    for iteration in range(50000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        score = 2.0 * n * max_b

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()

        sorted_idx = np.argsort(b)[::-1]
        active = sorted_idx[:K]
        active_vals = b[active]
        peak_spread = active_vals[0] - active_vals[-1]

        if peak_spread < 1e-10 * max_b:
            pass

        # Build Gauss-Newton system
        # G[k, j] = 2*a[k-j] if 0 <= k-j < n, else 0
        # System: G * delta - m * M = rhs, sum(delta) = 0
        rows = []
        rhs_vec = []
        for k in active:
            row = np.zeros(n)
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    row[j] = 2.0 * a[idx]
            rows.append(row)
            rhs_vec.append(0.0)  # We want G*delta = M - b[k], rearranged
        rows.append(np.ones(n))
        rhs_vec.append(0.0)

        G = np.array(rows)  # (K+1) x n
        # m vector: [1, 1, ..., 1, 0]
        m_vec = np.zeros(K + 1)
        m_vec[:K] = 1.0

        # rhs: for active peaks, we want G*delta - m*M = [b[k_j] - max_b] (to reduce to max_b)
        # Actually: we want b[k_j] + G_row_j @ delta = M for all j
        # => G_row_j @ delta - M = M - b[k_j]... 
        # Let's use: G_row_j @ delta = M - b[k_j], with M = max_b (target)
        # Then: G_row_j @ delta = max_b - b[k_j]
        # And sum(delta) = 0
        # System: [G; e] * delta = [max_b - b[k_j]; 0]
        # This is (K+1) equations in n unknowns. Min-norm solution.
        rhs = np.zeros(K + 1)
        for i, k in enumerate(active):
            rhs[i] = max_b - b[k]  # >= 0 since max_b is the max
        rhs[K] = 0.0

        # Min-norm: delta = G^T (G G^T)^{-1} rhs
        GGt = G @ G.T  # (K+1) x (K+1)
        try:
            GGt_inv = np.linalg.inv(GGt + 1e-12 * np.eye(K + 1))
        except np.linalg.LinAlgError:
            GGt_inv = np.linalg.pinv(GGt)
        delta = G.T @ (GGt_inv @ rhs)

        # Also try the "subgradient" direction as fallback
        grad_sub = np.zeros(n)
        for k in active:
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    grad_sub[j] += 2.0 * a[idx]
        grad_sub /= K
        grad_sub -= grad_sub.mean()

        # Line search over both directions
        improved = False
        for direction, scale in [(delta, 1.0), (grad_sub, 1.0)]:
            for t in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01]:
                step = t * scale
                a_trial = a + step * direction
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st < 1e-15:
                    break
                a_trial = a_trial / st
                score_trial = 2.0 * n * fast_convolve(a_trial).max()
                if score_trial < best_score - 1e-14:
                    a = a_trial
                    best_score = score_trial
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try a random perturbation to escape
            if iteration > 50:
                noise = rng.normal(0, 0.01, n)
                a_trial = a + noise
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-15:
                    a_trial = a_trial / st
                    score_trial = 2.0 * n * fast_convolve(a_trial).max()
                    if score_trial < best_score:
                        a = a_trial
                        best_a = a.copy()
                        best_score = score_trial
                    else:
                        # Stuck, break after many failed attempts
                        if iteration > 500:
                            break

    return best_a, best_score


def support_refinement(a0, time_limit, solver=None):
    """Zero out small elements and re-optimize to create structural gaps."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    # Try zeroing elements from smallest to largest
    order = np.argsort(a)
    n_zeroed = 0
    for idx in order:
        if time.monotonic() - t0 > time_limit:
            break
        if a[idx] > 0.01 / n:
            break  # Only zero very small elements
        a_trial = a.copy()
        a_trial[idx] = 0.0
        st = a_trial.sum()
        if st < 0.01:
            break
        a_trial = a_trial / st
        score_trial = 2.0 * n * fast_convolve(a_trial).max()
        if score_trial < best_score - 1e-10:
            a = a_trial
            best_a = a.copy()
            best_score = score_trial
            n_zeroed += 1

    if n_zeroed > 0 and time.monotonic() - t0 < time_limit - 30:
        # Re-optimize after zeroing
        remaining_time = time_limit - (time.monotonic() - t0)
        a_out, sc = nesterov_lp_increasing_avg(a, min(remaining_time, 60.0), solver)
        if sc < best_score:
            best_score = sc
            best_a = a_out.copy()

    return best_a, best_score


def lse_lbgfs_v2(a0, time_limit, T_start=50.0, T_end=0.02, n_stages=15):
    """L-BFGS with LogSumExp approximation, multi-stage annealing (improved)."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    global _GDEADLINE
    _GDEADLINE = t0 + time_limit

    def lse_obj_jac(z_in, T_val):
        a = softmax_from_z(z_in)
        b = fast_convolve(a)
        Tb = T_val * b
        max_Tb = Tb.max()
        if max_Tb > 700:
            max_Tb = 700
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()
        log_sum_exp = max_Tb + np.log(exp_Tb.sum())
        obj = (2.0 * n / T_val) * log_sum_exp

        # Jacobian
        grad_a = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]
        S = np.dot(grad_a, a)
        jac = a * (grad_a - S)
        return obj, jac

    temps = np.geomspace(T_start, T_end, n_stages)
    for si, stage_T in enumerate(temps):
        if time.monotonic() > _GDEADLINE:
            break
        iters = max(30, 80)

        def func(z_in):
            if time.monotonic() > _GDEADLINE:
                return 1e15
            o, _ = lse_obj_jac(z_in, stage_T)
            return o

        def jac(z_in):
            if time.monotonic() > _GDEADLINE:
                return np.zeros_like(z_in)
            _, j = lse_obj_jac(z_in, stage_T)
            return j

        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': iters, 'ftol': 1e-16,
                                    'gtol': 1e-12, 'maxfun': iters * 200,
                                    'disp': False})
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def mirror_descent_refined(a0, time_limit, n_active_frac=0.1):
    """Mirror descent with adaptive active set and backtracking."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a).copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.2
    no_improve = 0
    rng = np.random.default_rng(123)

    for step in range(100000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b - n_active_frac * max_b
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)
        g -= g.mean()

        improved = False
        alpha = lr
        for bt in range(8):
            z_new = z - alpha * g
            a_new = softmax_from_z(z_new)
            score_new = 2.0 * n * fast_convolve(a_new).max()
            if score_new < best_score - 1e-14:
                best_score = score_new
                best_a = a_new.copy()
                a = a_new
                z = z_new
                no_improve = 0
                lr = min(alpha * 1.05, 2.0)
                improved = True
                break
            alpha *= 0.5

        if not improved:
            no_improve += 1
            lr = max(lr * 0.9, 0.01)
            if no_improve >= 50:
                noise = rng.normal(0, 0.02, n)
                a = a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                z = z_from_a(a)
                no_improve = 0
                lr = 0.2

    return best_a, best_score
