        if Z < 1e-30:
            weights = np.zeros_like(conv)
            weights[conv.argmax()] = 1.0
            Z = 1.0
        weights = weights / Z

        # Compute gradient via active weights
        active = np.where(weights > 1e-10)[0]
        grad = np.zeros(n)
        for k in active:
            w = weights[k]
            if w < 1e-12:
                continue
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            grad[j] += 2.0 * w * a[k - j]

        # Momentum update
        v = momentum * v + (1.0 - momentum) * grad

        # Multiplicative update
        log_a = np.log(a + 1e-30)
        log_a -= lr * v
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        conv_new = fast_convolve(a_new)
        M_new = float(conv_new.max())
        score = 2.0 * n * M_new

        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        if no_improve > 40:
            lr *= 0.8
            beta *= 1.4
            momentum = max(0.3, momentum * 0.95)
            no_improve = 0
            if lr < 1e-12:
                break
        elif it % 200 == 0:
            beta *= 1.03

        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    IRLS {label} it={it} ({elapsed:.0f}s): score={best_score:.6f}, M={M:.8f}, lr={lr:.4f}, beta={beta:.1f}")

    return best_a, best_score

def pair_coordinate_descent(a0, time_limit, label=""):
    """Exact 2-coordinate minimax block descent."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(0, n - 1, 2):  # step by 2 to use disjoint pairs
            i2 = i + 1
            x0 = a[i]
            y0 = a[i2]
            if x0 < 1e-15 and y0 < 1e-15:
                continue

            # Determine affected lag range
            k_min = i
            k_max = min(i2 + n - 1, 2 * n - 2)

            # Affected convex values
            affected = conv[k_min:k_max + 1].copy()
            unaff_max = 0.0
            if k_min > 0:
                unaff_max = max(unaff_max, float(conv[:k_min].max()))
            if k_max < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_max + 1:].max()))

            # Compute coefficients for g_k(x,y) = base + Ax + By + Cxy + Dx^2 + Ey^2
            # base = g_k with a[i]=a[i+1]=0
            # For each k in [k_min, k_max]:
            #   contribution from a[i]: appears as a[i]*a[k-i] and a[k-i]*a[i]
            #   contribution from a[i+1]: appears as a[i+1]*a[k-i-1] and a[k-i-1]*a[i+1]
            #   cross term: a[i]*a[i+1] when k-i = i+1 (k=2i+1) or k-(i+1) = i (same)

            A_coeffs = np.zeros(k_max - k_min + 1)
            B_coeffs = np.zeros(k_max - k_min + 1)
            C_coeffs = np.zeros(k_max - k_min + 1)  # xy cross term
            D_coeffs = np.zeros(k_max - k_min + 1)  # x^2 term
            E_coeffs = np.zeros(k_max - k_min + 1)  # y^2 term
            base = np.zeros(k_max - k_min + 1)

            for idx, k in enumerate(range(k_min, k_max + 1)):
                # base: g_k with a[i]=a[i+1]=0
                # current g_k minus terms involving a[i] and a[i+1]
                gk_current = conv[k]
                # Remove a[i] contributions
                contrib_x = 0.0
                if 0 <= k - i < n and k - i != i2:
                    Contrib_x = x0 * a[k - i]
                elif k - i == i:
                    contrib_x = x0 * x0
                elif k - i == i2:
                    contrib_x = x0 * y0
                # Actually let me just compute from scratch
                # g_k = sum_j a_j * a[k-j]
                # Terms with j=i: a[i]*a[k-i] = x0 * a[k-i] (if 0<=k-i<n)
                # Terms with j=i+1: a[i+1]*a[k-i-1] = y0 * a[k-i-1] (if 0<=k-i-1<n)
                # Terms with k-j=i: a[k-i]*a[i] = a[k-i]*x0 (if 0<=k-i<n, k-i!=i)
                # Terms with k-j=i+1: a[k-i-1]*a[i+1] = a[k-i-1]*y0 (if 0<=k-i-1<n, k-i-1!=i+1)

                # x-linear: from j=i (a[i]*a[k-i]) when k-i != i and k-i != i+1
                #            from k-j=i (a[k-i]*a[i]) when k-i != i and k-i != i+1
                # x-quadratic: from j=i and k-j=i simultaneously, i.e., k=2i: a[i]^2
                # x*y cross: from j=i and k-j=i+1 (k=2i+1): a[i]*a[i+1]
                #             from j=i+1 and k-j=i (k=2i+1): a[i+1]*a[i]

                # Let me compute A_k (coeff of x), B_k (coeff of y), C_k (coeff of x*y),
                # D_k (coeff of x^2), E_k (coeff of y^2)

                A_k = 0.0  # linear in x
                B_k = 0.0  # linear in y
                C_k = 0.0  # x*y
                D_k = 0.0  # x^2
                E_k = 0.0  # y^2

                # From j=i: term is a[i] * a[k-i] = x * a[k-i]
                if 0 <= k - i < n:
                    if k - i == i:
                        D_k += 1.0  # x*x
                    elif k - i == i2:
                        C_k += 1.0  # x*y
                    else:
                        A_k += a[k - i]

                # From j=i+1: term is a[i+1] * a[k-i-1] = y * a[k-i-1]
                if 0 <= k - i - 1 < n:
                    if k - i - 1 == i:
                        C_k += 1.0  # y*x
                    elif k - i - 1 == i2:
                        E_k += 1.0  # y*y
                    else:
                        B_k += a[k - i - 1]

                A_coeffs[idx] = A_k
                B_coeffs[idx] = B_k
                C_coeffs[idx] = C_k
                D_coeffs[idx] = D_k
                E_coeffs[idx] = E_k
                base[idx] = gk_current - (x0 * A_k + y0 * B_k + x0 * y0 * C_k + x0*x0 * D_k + y0*y0 * E_k)

            # Now solve: minimize max_k [base_k + A_k*x + B_k*y + C_k*x*y + D_k*x^2 + E_k*y^2]
            # subject to x,y >= 0
            # Use grid search + refinement

            x_range = max(x0, y0, 1.0 / n) * 3.0
            x_range = min(x_range, 5.0 / n)

            # Coarse grid
            n_grid = 15
            x_vals = np.linspace(0, x_range, n_grid)
            y_vals = np.linspace(0, x_range, n_grid)

            best_f = M
            best_x, best_y = x0, y0

            for x_val in x_vals:
                for y_val in y_vals:
                    g_new = base + A_coeffs * x_val + B_coeffs * y_val + C_coeffs * x_val * y_val + D_coeffs * x_val * x_val + E_coeffs * y_val * y_val
                    f_val = max(unaff_max, float(g_new.max()))
                    if f_val < best_f:
                        best_f = f_val
                        best_x, best_y = x_val, y_val

            # Fine grid around best
            if best_f < M - 1e-12:
                dx = x_range / n_grid
                x_fine = np.linspace(max(0, best_x - 2*dx), best_x + 2*dx, 15)
                y_fine = np.linspace(max(0, best_y - 2*dx), best_y + 2*dx, 15)
                for x_val in x_fine:
                    if x_val < 0:
                        continue
                    for y_val in y_fine:
                        if y_val < 0:
                            continue
                        g_new = base + A_coeffs * x_val + B_coeffs * y_val + C_coeffs * x_val * y_val + D_coeffs * x_val * x_val + E_coeffs * y_val * y_val
                        f_val = max(unaff_max, float(g_new.max()))
                        if f_val < best_f:
                            best_f = f_val
                            best_x, best_y = x_val, y_val

            if best_f < M - 1e-13:
                # Apply update
                a[i] = best_x
                a[i2] = best_y
                for idx, k in enumerate(range(k_min, k_max + 1)):
                    conv[k] = base[idx] + A_coeffs[idx]*best_x + B_coeffs[idx]*best_y + C_coeffs[idx]*best_x*best_y + D_coeffs[idx]*best_x**2 + E_coeffs[idx]*best_y**2
                M = best_f
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if sweeps % 200 == 0:
            conv = fast_convolve(a)
            M = float(conv.max())

        if not improved:
            no_improve += 1
            if no_improve > 100:
                break
        else:
            no_improve = 0

        if sweeps % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PairCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score

def simulated_annealing_peak(a0, rng, time_limit, label=""):
    """Simulated annealing with moves that target the peak lag."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s

    best_a = a.copy()
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    t0 = time.monotonic()

    T_init = M * 0.01
    T_current = T_init
    cooling = 0.995
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit and T_current > T_init * 1e-6:
        it += 1
        T_current *= cooling

        conv = fast_convolve(a)
        M = float(conv.max())
        peak_k = int(conv.argmax())

        # Propose a move: shift mass from peak-contributing positions to random positions
        i_start = max(0, peak_k - n + 1)
        i_end = min(n - 1, peak_k)

        # Find the coordinate that contributes most to the peak
        contribs = np.zeros(n)
        for i in range(i_start, i_end + 1):
            j = peak_k - i
            contribs[i] = a[i] * a[j]

        total_contrib = contribs.sum()
        if total_contrib < 1e-20:
            # Random move
            i = rng.integers(n)
            j = rng.integers(n)
            delta = a[i] * 0.1
            a_new = a.copy()
            a_new[i] = max(0, a[i] - delta)
            a_new[j] = a[j] + delta
        else:
            # Move mass away from the peak-contributing coordinate
            weights = contribs / total_contrib
            i = rng.choice(n, p=weights)
            # Move to a random position that doesn't contribute much to the peak
            j = rng.integers(n)
            frac = rng.uniform(0.02, 0.15)
            delta = a[i] * frac
            a_new = a.copy()
            a_new[i] = max(0, a[i] - delta)
            a_new[j] = a[j] + delta

        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()

        conv_new = fast_convolve(a_new)
        M_new = float(conv_new.max())
        score_new = 2.0 * n * M_new
        score_cur = 2.0 * n * M

        # Accept if better or with probability exp(-delta/T)
        if score_new < score_cur:
            a = a_new
            M = M_new
            if score_new < best_score - 1e-14:
                best_score = score_new
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            delta = score_new - score_cur
            if delta < 1e-10:
                accept = True
            else:
                ratio = -delta / (T_current * 2.0 * n)
                if ratio > 50:
                    accept = False
                else:
                    accept = rng.random() < np.exp(min(ratio, 50))
            if accept:
                a = a_new
                M = M_new

        if no_improve > 500:
            # Restart temperature
            T_current = T_init * 0.1
            no_improve = 0
            # Slightly randomize
            noise = rng.standard_normal(n) * 0.01 * a.max()
            a = np.maximum(a + noise, 0)
            a = a / a.sum()

        if it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SA {label} it={it} ({elapsed:.0f}s): score={best_score:.6f}, T={T_current:.6f}")

    return best_a, best_score

def smart_coord_descent(a0, time_limit, label=""):
    """Fast single-coordinate descent with intelligent candidate generation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            ai = a[i]
            if ai < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            if k_start > 0:
                unaff_max = float(conv[:k_start].max())
            else:
                unaff_max = -1e30
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            affected = conv[k_start:k_end + 1]
            coeffs = a[(k_start - i):(k_end - i + 1)] + ai
            has_quad = (2 * i >= k_start) and (2 * i <= k_end)
            quad_idx = 2 * i - k_start if has_quad else -1

            # Candidate deltas
            candidates = [0.0]
            # Where affected lines hit unaff_max
            for j in range(len(affected)):
                c = coeffs[j]
                if c > 1e-15:
                    d = (unaff_max - affected[j]) / c
                    if -ai * 1.5 < d < ai * 1.5:
                        candidates.append(d)

            # Top affected line crossings
            top_k = min(4, len(affected))
            top_idx = np.argsort(affected)[-top_k:]
            for j1 in range(len(top_idx)):
                for j2 in range(j1 + 1, len(top_idx)):
                    i1, i2 = top_idx[j1], top_idx[j2]
                    denom = coeffs[i1] - coeffs[i2]
                    if abs(denom) > 1e-15:
                        d = (affected[i2] - affected[i1]) / denom
                        if -ai * 1.5 < d < ai * 1.5:
                            candidates.append(d)

            # Small perturbations
            for f in [0.01, 0.02, 0.05, 0.1, 0.2, 0.5, -0.5, -0.9, -0.99]:
                candidates.append(f * ai)

            best_delta = 0.0
            best_new_max = M

            for delta in candidates:
                if delta < -ai - 1e-12:
                    continue
                delta = max(delta, -ai)
                if abs(delta) < 1e-16:
                    continue
                new_affected = affected + delta * coeffs
                if has_quad:
                    new_affected = new_affected.copy()
                    new_affected[quad_idx] += delta * delta
                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-15:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-14:
                a[i] = max(ai + best_delta, 0.0)
                conv[k_start:k_end + 1] += best_delta * coeffs
                if has_quad:
                    conv[2 * i] += best_delta * best_delta
                M = best_new_max
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if sweeps % 100 == 0:
            conv = fast_convolve(a)
            M = float(conv.max())

        if not improved:
            no_improve += 1
            if no_improve > 200:
                break
        else:
            no_improve = 0

        if sweeps % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score

def spectral_shaping(a0, rng, time_limit, label=""):
    """
    Use the frequency domain to shape the spectrum toward flatness.
    Minimize the L-inf of the circular autocorrelation via gradient on log(a).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30).copy()
    a = a / a.sum()

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    # Use circular autocorrelation for spectral method (good approx for large n)
    A = rfft(a, n)
    P = np.abs(A)**2
    circ_autocorr = irfft(P, n)  # circular autocorrelation (real)
    M_circ = float(circ_autocorr.max())

    lr = 0.1
    beta = 10.0
    it = 0
    no_improve = 0
    momentum_v = np.zeros(n)
    momentum = 0.8

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Compute smoothed gradient in frequency domain
        A = rfft(a, n)
        P = np.abs(A)**2
        max_P = P.max()
        scaled = beta * (P - max_P)
        scaled = np.clip(scaled, -50, 0)
        w_P = np.exp(scaled)
        w_P = w_P / w_P.sum()

        # The circular autocorrelation is irfft(P)
        # Gradient of sum_k w_k * circ_autocorr[k] w.r.t. a:
        # circ_autocorr = ifft(|A|^2), A = fft(a)
        # d/d(a) of |A|^2 at frequency m: 2*Re[A(m)*conj(da/dA(m))]...
        # This is complex. Let's use time-domain gradient instead.

        # Time-domain gradient of smoothed circular autocorrelation:
        circ_ac = irfft(P, n)
        max_ac = circ_ac.max()
        scaled_ac = beta * (circ_ac - max_ac)
        scaled_ac = np.clip(scaled_ac, -50, 0)
        w_ac = np.exp(scaled_ac)
        w_ac = w_ac / w_ac.sum()

        # Gradient: grad_i = 2 * sum_k w_k * a[(k-i) mod n]
        # This is 2 * circular cross-correlation of a and w_ac
        # = 2 * ifft(fft(a) * conj(fft(w_ac)))
        fa = rfft(a, n)
        fw = rfft(w_ac, n)
        grad = 2.0 * irfft(fa * np.conj(fw), n)

        # Momentum
        momentum_v = momentum * momentum_v + (1 - momentum) * grad

        # Multiplicative update
        log_a = np.log(a)
        log_a -= lr * momentum_v
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = a_new / a_new.sum()

        # Evaluate with true (linear) autocorrelation
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        if no_improve > 60:
            lr *= 0.8
            beta *= 1.5
            no_improve = 0
            if lr < 1e-12:
                break
        elif it % 100 == 0:
            beta *= 1.02

        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Spectral {label} it={it} ({elapsed:.0f}s): score={best_score:.6f}, lr={lr:.4f}, beta={beta:.1f}")

    return best_a, best_score

    rem = deadline - time.monotonic()
    t3 = min(200.0, rem * 0.15)
    print(f"  Budget: {t3:.0f}s, n={len(best_seq)}")
    a3, s3 = spectral_shaping(best_seq, rng, t3, label="p3")
    sc3 = float(evaluate_sequence(a3.tolist()))
    if np.isfinite(sc3) and sc3 < best_score:
        best_score = sc3
        best_seq = a3.copy()
        print(f"  Spectral improved: {best_score:.6f}")

    # === Phase 4: Simulated Annealing ===
    print(f"\n=== Phase 4: Simulated Annealing (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(250.0, rem * 0.18)
    print(f"  Budget: {t4:.0f}s, n={len(best_seq)}")
    a4, s4 = simulated_annealing_peak(best_seq, rng, t4, label="p4")
    sc4 = float(evaluate_sequence(a4.tolist()))
    if np.isfinite(sc4) and sc4 < best_score:
        best_score = sc4
        best_seq = a4.copy()
        print(f"  SA improved: {best_score:.6f}")

    # === Phase 5: Smart CD Polish ===
    print(f"\n=== Phase 5: Smart CD Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    for rnd in range(3):
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t5 = min(rem * 0.25, 150.0)
        a5, s5 = smart_coord_descent(best_seq, t5, label=f"p5_{rnd}")
        sc5 = float(evaluate_sequence(a5.tolist()))
        if np.isfinite(sc5) and sc5 < best_score:
            best_score = sc5
            best_seq = a5.copy()
            print(f"  SCD r{rnd} improved: {best_score:.6f}")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-Length (best={best_score:.6f}) ===")
    n_cur = len(best_seq)
    alt_lengths = []
    for alt in [n_cur - 128, n_cur + 128, n_cur + 256, int(n_cur * 1.5), int(n_cur * 0.75)]:
        if 256 <= alt <= 4096 and alt != n_cur and alt not in alt_lengths:
            alt_lengths.append(alt)
    alt_lengths = alt_lengths[:3]

    for n_alt in alt_lengths:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t6 = min(rem * 0.25, 120.0)
        init_alt = interpolate_seq(best_seq, n_alt)
        # Quick optimize
        a6, s6 = smart_coord_descent(init_alt, t6, label=f"p6_n{n_alt}")
        sc6 = float(evaluate_sequence(a6.tolist()))
        if np.isfinite(sc6) and sc6 < best_score:
            best_score = sc6
            best_seq = a6.copy()
            print(f"  MultiLen n={n_alt} improved: {best_score:.6f}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t7 = min(rem * 0.5, 180.0)
        a7, s7 = smart_coord_descent(best_seq, t7, label="p7a")
        sc7 = float(evaluate_sequence(a7.tolist()))
        if np.isfinite(sc7) and sc7 < best_score:
            best_score = sc7
            best_seq = a7.copy()

    rem = deadline - time.monotonic()
    if rem > 20:
        a7b, s7b = pair_coordinate_descent(best_seq, min(rem * 0.4, 80.0), label="p7b")
        sc7b = float(evaluate_sequence(a7b.tolist()))
        if np.isfinite(sc7b) and sc7b < best_score:
            best_score = sc7b
            best_seq = a7b.copy()

    rem = deadline - time.monotonic()
    if rem > 10:
        a7c, s7c = smart_coord_descent(best_seq, rem * 0.6, label="p7c")
        sc7c = float(evaluate_sequence(a7c.tolist()))
        if np.isfinite(sc7c) and sc7c < best_score:
            best_score = sc7c
            best_seq = a7c.copy()

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
