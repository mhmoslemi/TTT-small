LogSumExp-smoothed subgradient with FFT-accelerated gradients,
adaptive coordinate descent, peak equalization heuristics,
and multi-n exploration from diverse initializations.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize
import time
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_correlate(w, a):
    """Compute c[i] = sum_k w[k] * a[k-i] using FFT (linear cross-correlation)."""
    n_w = len(w)
    n_a = len(a)
    N = 1 << (n_w + n_a - 1 - 1).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    c = ifft(W * np.conj(A))[:n_a].real
    return c


def local_eval(a):
    """Fast local evaluation of the objective (normalized sum=1)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalized_eval(a):
    """Evaluate assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_smoothed_gradient(a, T):
    """Compute gradient of LogSumExp-smoothed max-convolution w.r.t. a.
    Returns (gradient_array, lse_value)."""
    n = len(a)
    b = fast_convolve(a)
    # Softmax weights with temperature T
    b_shift = b / T
    b_max = b_shift.max()
    exp_ws = np.exp(b_shift - b_max)
    w = exp_ws / exp_ws.sum()
    lse = b_max + np.log(np.exp(b_shift - b_max).sum())
    # Gradient: grad_a[i] = 2 * sum_k w[k] * a[k-i] = 2 * cross_correlate(w, a)[i]
    grad_a = 2.0 * cross_correlate(w, a)
    return grad_a, float((2.0 * n) * lse)


def softmax_obj_and_grad(z, T, n):
    """Objective and gradient in log-parametrization.
    a = softmax(z), obj = 2n * LSE(conv(a,a)/T)."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    b = fast_convolve(a)
    b_shift = b / T
    b_max_val = b_shift.max()
    exp_ws = np.exp(b_shift - b_max_val)
    w = exp_ws / exp_ws.sum()
    lse = b_max_val + np.log(np.exp(b_shift - b_max_val).sum())
    obj_val = 2.0 * n * lse
    # Gradient w.r.t. a
    grad_a = 2.0 * (2.0 * n) * cross_correlate(w, a) * T  # * T because d LSE/d a = T * w * grad_conv
    # Actually: d/d a [LSE(b/T)] = sum_k w[k] * db[k]/da = sum_k w[k] * 2*a[k-i]
    # So d obj / d a = 2n * sum_k w[k] * 2*a[k-i] = 4n * cross_correlate(w, a)
    grad_a = 2.0 * n * 2.0 * cross_correlate(w, a)
    # Chain rule for softmax: d obj/dz = a * (grad_a - dot(grad_a, a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj_val, grad_z


def nesterov_logsumexp(a0, time_limit, T_stages=None, max_restarts=5):
    """Nesterov-accelerated LogSumExp-smoothed subgradient in log-parametrization."""
    n = len(a0)
    a = normalize(a0)
    a_clipped = np.maximum(a, 1e-300)
    z = np.log(a_clipped)
    z -= z.max()  # center for numerical stability
    best_a = a.copy()
    best_score = normalized_eval(a)
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [30.0, 15.0, 8.0, 4.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]

    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        max_iter = max(50, int(remaining * 40))

        for restart in range(max_restarts):
            if time.monotonic() - t0 > time_limit:
                break
            remaining = time_limit - (time.monotonic() - t0)
            if remaining < 5:
                break
            iters_this = max(30, int(remaining * 30 / (max_restarts - restart)))
            if restart > 0:
                rng = np.random.default_rng(int(time.monotonic() * 1000) + restart)
                z = z + rng.standard_normal(n) * 0.01
                z -= z.max()

            try:
                res = minimize(
                    lambda zz: softmax_obj_and_grad(zz, T, n)[0],
                    z,
                    jac=lambda zz: softmax_obj_and_grad(zz, T, n)[1],
                    method='L-BFGS-B',
                    options={'maxiter': iters_this, 'ftol': 1e-22, 'gtol': 1e-14,
                             'maxfun': iters_this * 300, 'disp': False}
                )
                z = res.x
                a_new = np.exp(z - z.max())
                a_new = a_new / a_new.sum()
                sc = normalized_eval(a_new)
                if sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    z = np.log(np.maximum(a_new, 1e-300))
                    z -= z.max()
            except Exception:
                pass

        # Track progress
        if time.monotonic() - t0 < time_limit:
            a_cur = np.exp(z - z.max())
            a_cur = a_cur / a_cur.sum()
            sc_cur = normalized_eval(a_cur)
            if sc_cur < best_score:
                best_score = sc_cur
                best_a = a_cur.copy()

    return best_a, best_score


def nesterov_projected_subgradient(a0, time_limit, n_steps=None):
    """Nesterov-accelerated projected subgradient on the simplex.
    Uses LogSumExp smoothing with adaptive temperature."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = normalized_eval(a)
    t0 = time.monotonic()

    if n_steps is None:
        n_steps = max(10000, int(time_limit * 2000))

    # Adaptive temperature schedule
    T_start = n * 0.1
    T_end = n * 0.001
    n_total = n_steps

    for step in range(n_total):
        if time.monotonic() - t0 > time_limit:
            break

        # Adaptive temperature: geometric decay
        frac = step / max(1, n_total - 1)
        T = T_start * (T_end / T_start) ** frac

        # Nesterov momentum
        if step == 0:
            y = a.copy()
            a_prev = a.copy()
            t_k = 1.0
        else:
            t_new = (1.0 + math.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            beta = (t_k - 1.0) / t_new
            t_k = t_new
            y = a + beta * (a - a_prev)
            y = np.maximum(y, 0)
            s_y = y.sum()
            if s_y > 1e-15:
                y = y / s_y
            else:
                y = a.copy()

        # Compute smoothed subgradient at y
        b = fast_convolve(y)
        b_shift = b / T
        b_max = b_shift.max()
        exp_ws = np.exp(b_shift - b_max)
        w = exp_ws / exp_ws.sum()
        grad = 2.0 * cross_correlate(w, y)

        # Step size: eta = c / sqrt(step + 1)
        eta = 1.0 / math.sqrt(step + 1.0)
        # Scale by problem size
        eta *= 2.0 / math.sqrt(n)

        # Update
        a_trial = y - eta * grad
        a_trial = np.maximum(a_trial, 0)
        s_trial = a_trial.sum()
        if s_trial > 1e-15:
            a_trial = a_trial / s_trial
        else:
            a_trial = a.copy()

        a_prev = a
        a = a_trial

        # Track best
        if (step + 1) % 50 == 0 or step < 10:
            sc = normalized_eval(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    # Final check
    sc = normalized_eval(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

    return best_a, best_score


def coordinate_descent_polish(a0, time_limit, n_passes=20):
    """Cyclic coordinate descent: for each coordinate, try several multipliers
    and keep the best."""
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_score = normalized_eval(a)
    t0 = time.monotonic()

    multipliers = [0.0, 0.25, 0.5, 0.75, 1.0, 1.25, 1.5, 2.0, 4.0]

    for p in range(n_passes):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            orig = a[i]
            if orig < 1e-15:
                # Try setting to small positive
                a[i] = 1e-6
                s = a.sum()
                a = a / s
                sc = normalized_eval(a)
                if sc < best_score - 1e-15:
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                else:
                    a[i] = 0.0
                    continue
                continue

            for mult in multipliers:
                a_trial = a.copy()
                a_trial[i] = orig * mult
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                sc = normalized_eval(a_trial)
                if sc < best_score - 1e-15:
                    best_score = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    orig = a[i]
                    improved = True
                    break

        if not improved and p >= 2:
            # Try with fewer multipliers for speed
            pass
        if p % 5 == 4:
            print(f"    CD pass {p+1}/{n_passes}: score={best_score:.10f}")

    return best_a, best_score


def peak_equalization(a0, time_limit, n_rounds=500):
    """Aggressively reduce the top peaks by adjusting contributing coordinates."""
    n = len(a0)
    a = normalize(a0).copy()
    best_a = a.copy()
    best_score = normalized_eval(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        M = b.max()
        # Find top peaks
        top_k = min(5, len(b))
        peak_indices = np.argsort(b)[-top_k:]
        peak_indices = peak_indices[::-1]  # descending order

        # For each peak, find the coordinate that contributes most and reduce it
        reduced = False
        for k in peak_indices[:3]:
            # Contribution of a[i] to b[k] is 2*a[k-i] (gradient)
            # Find i that maximizes a[i] * a[k-i] (the actual contribution)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contributions = a[idx] * a[k - idx]
            i_max = idx[np.argmax(contributions)]
            contrib_max = contributions.max()

            if contrib_max < 1e-15:
                continue

            # Try reducing a[i_max] and increasing a random other coordinate
            # to maintain the sum
            candidates = [0.1, 0.3, 0.5, 0.7, 0.9]
            for frac in candidates:
                a_trial = a.copy()
                reduce_amt = a[i_max] * frac
                a_trial[i_max] -= reduce_amt
                if a_trial[i_max] < 0:
                    a_trial[i_max] = 0
                    reduce_amt = a[i_max]
                # Distribute the reduction to coordinates that don't contribute to peaks
                # Simple: add to a random non-peak-contributing coordinate
                # Or: add uniformly
                a_trial += reduce_amt / n
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                sc = normalized_eval(a_trial)
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    reduced = True
                    break

            if reduced:
                break

        if not reduced and rnd % 50 == 49:
            # Perturb to escape local minimum
            noise = rng.standard_normal(n) * 0.005
            a = a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()

    return best_a, best_score


def structured_init(n, choice, rng):
    """Create various structured initial sequences."""
    x = np.linspace(0, 1, n)
    if choice == 0:
        # Uniform
        a = np.ones(n)
    elif choice == 1:
        # U-shaped (high at edges)
        a = (x - 0.5) ** 2 + 0.1
    elif choice == 2:
        # Cosine bell
        a = np.cos(np.pi * x) ** 2 + 0.01
    elif choice == 3:
        # Two peaks at edges
        a = np.exp(-20 * x ** 2) + np.exp(-20 * (x - 1) ** 2) + 0.05
    elif choice == 4:
        # Inverse U (dip in middle)
        a = 1.0 - 0.5 * np.sin(np.pi * x) ** 2
    elif choice == 5:
        # Power law at edges
        a = x ** 0.3 + (1 - x) ** 0.3 + 0.05
    elif choice == 6:
        # Slightly perturbed uniform
        a = 1.0 + 0.2 * np.cos(2 * np.pi * rng.integers(1, 5) * x)
        a = np.maximum(a, 0.01)
    elif choice == 7:
        # Linear ramp
        a = x + 0.1
    elif choice == 8:
        # Concentrated at center
        a = np.exp(-10 * (x - 0.5) ** 2)
    else:
        # Random
        a = rng.random(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def alternating_refinement(a0, b0, time_limit, n_iters=5000):
    """Alternating minimization: given a, find best b; given b, find best a.
    Uses gradient descent for the inner optimization (faster than LP)."""
    n = len(a0)
    a = normalize(a0)
    b = normalize(b0)
    best_a = a.copy()
    best_score = normalized_eval(a)
    best_b = b.copy()
    t0 = time.monotonic()

    def best_response_gd(fixed, n_inner=50):
        """Find best response to 'fixed' using gradient descent."""
        x = np.ones(n) / n
        lr = 1.0 / n
        for it in range(n_inner):
            # Compute conv(fixed, x)
            fft_size = 1 << (2 * n - 1 - 1).bit_length()
            F = rfft(fixed, fft_size)
            X = rfft(x, fft_size)
            conv = irfft(F * X, fft_size)[:2 * n - 1]
            # Subgradient: position of max
            k_max = np.argmax(conv)
            grad = np.zeros(n)
            js = max(0, k_max - n + 1)
            je = min(n - 1, k_max)
            idx = np.arange(js, je + 1)
            grad[idx] = fixed[k_max - idx]
            # Update
            x = x - lr * grad
            x = np.maximum(x, 0)
            s = x.sum()
            if s > 1e-15:
                x = x / s
        return x

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Update b given a
        b = best_response_gd(a)
        sc_b = normalized_eval(b)
        if sc_b < best_score:
            best_score = sc_b
            best_b = b.copy()
            best_a = b.copy()  # Use b as candidate for a too (symmetric problem)

        # Update a given b
        a = best_response_gd(b)
        sc_a = normalized_eval(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()

        if (it + 1) % 500 == 0:
            print(f"    AltRef iter {it+1}: score={best_score:.10f}")

    return best_a, best_score


def multi_start_optimize(n, time_limit, parent, rng):
    """Try multiple initializations for a given n and return the best."""
    best_a = None
    best_score = float('inf')

    # Initializations to try
    inits = []
    if parent is not None and len(parent) > 0:
        inits.append(interpolate_sequence(parent, n))
    inits.append(structured_init(n, 1, rng))  # U-shaped
    inits.append(structured_init(n, 3, rng))  # Two peaks at edges
    inits.append(structured_init(n, 0, rng))  # Uniform
    inits.append(structured_init(n, 4, rng))  # Inverse U

    per_init_time = time_limit / max(1, len(inits))

    for idx, init in enumerate(inits):
        if time.monotonic() > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0_multi) if 'time_limit' in dir() else per_init_time
        t_sub = min(per_init_time * 1.5, remaining * 0.9)
        if t_sub < 10:
            break

        # Phase 1: Nesterov subgradient (fast)
        t1 = t_sub * 0.4
        a1, s1 = nesterov_projected_subgradient(init, t1)
        if s1 < best_score:
            best_score = s1
            best_a = a1.copy()

        # Phase 2: LogSumExp L-BFGS (precise)
        t2 = t_sub * 0.4
        if best_a is not None and t2 > 5:
            T_stages = [10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02]
            a2, s2 = nesterov_logsumexp(best_a, t2, T_stages=T_stages)
            if s2 < best_score:
                best_score = s2
                best_a = a2.copy()

        # Phase 3: Coordinate descent polish
        t3 = t_sub * 0.2
        if best_a is not None and t3 > 3:
            a3, s3 = coordinate_descent_polish(best_a, t3, n_passes=10)
            if s3 < best_score:
                best_score = s3
                best_a = a3.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses Nesterov-accelerated smoothed subgradient, LogSumExp L-BFGS,
    coordinate descent, peak equalization, alternating refinement,
    and multi-n exploration with diverse initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = normalized_eval(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(a)})")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = normalized_eval(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.10f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, 1, rng)
        best_score = normalized_eval(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.10f}")

    n_best = len(best_seq)
    print(f"Starting score: {best_score:.10f}, n={n_best}")

    # === Phase 1: Intensive LogSumExp L-BFGS on parent ===
    print(f"\n=== Phase 1: Intensive LogSumExp L-BFGS (n={n_best}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.25, 400.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        T_stages = [40.0, 20.0, 10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
        a1, s1 = nesterov_logsumexp(best_seq, t1, T_stages=T_stages)
        try_update(a1, "lbgfs_main")

    # === Phase 2: Nesterov projected subgradient ===
    print(f"\n=== Phase 2: Nesterov Subgradient (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.15, 250.0)
    if t2 > 20:
        print(f"  Budget: {t2:.0f}s")
        a2, s2 = nesterov_projected_subgradient(best_seq, t2)
        try_update(a2, "nesterov_sg")

    # === Phase 3: Coordinate descent polish ===
    print(f"\n=== Phase 3: Coordinate Descent (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.12, 180.0)
    if t3 > 10:
        print(f"  Budget: {t3:.0f}s")
        a3, s3 = coordinate_descent_polish(best_seq, t3, n_passes=30)
        try_update(a3, "coord_descent")

    # === Phase 4: Peak equalization ===
    print(f"\n=== Phase 4: Peak Equalization (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.10, 150.0)
    if t4 > 10:
        print(f"  Budget: {t4:.0f}s")
        a4, s4 = peak_equalization(best_seq, t4, n_rounds=2000)
        try_update(a4, "peak_eq")

    # === Phase 5: Second round of L-BFGS from improved point ===
    print(f"\n=== Phase 5: L-BFGS Round 2 (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.15, 250.0)
    if t5 > 20:
        print(f"  Budget: {t5:.0f}s")
        T_stages2 = [5.0, 2.0, 1.0, 0.5, 0.3, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
        a5, s5 = nesterov_logsumexp(best_seq, t5, T_stages=T_stages2)
        try_update(a5, "lbgfs_r2")

    # === Phase 6: Multi-n exploration ===
    print(f"\n=== Phase 6: Multi-n Exploration (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        n_candidates = []
        base_n = n_best
        for mult in [0.7, 0.8, 0.9, 1.1, 1.25, 1.5, 2.0]:
            cand_n = int(base_n * mult)
            if 200 <= cand_n <= 6000 and cand_n != n_best and cand_n not in n_candidates:
                n_candidates.append(cand_n)
        # Also try some absolute values
        for abs_n in [500, 800, 1500, 2000, 3000]:
            if abs_n != n_best and abs_n not in n_candidates:
                n_candidates.append(abs_n)
        n_candidates = sorted(n_candidates)[:6]  # Limit to 6 candidates

        for cand_n in n_candidates:
            rem = deadline - time.monotonic()
            if rem < 50:
                break
            t_sub = min(rem * 0.25, 120.0)
            if t_sub < 30:
                break
            print(f"  Trying n={cand_n}, budget={t_sub:.0f}s")

            # Interpolate parent to new n
            init_cand = interpolate_sequence(best_seq, cand_n)

            # Quick optimization
            T_sub = [10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05]
            a_c, s_c = nesterov_logsumexp(init_cand, t_sub * 0.6, T_stages=T_sub)
            try_update(a_c, f"lbgfs_n{cand_n}")

            # Coordinate descent
            rem = deadline - time.monotonic()
            if rem > 20 and a_c is not None:
                a_cd, s_cd = coordinate_descent_polish(a_c, min(rem * 0.3, 30.0), n_passes=10)
                try_update(a_cd, f"cd_n{cand_n}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final Polish (score={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        # Very fine L-BFGS
        t_fin = min(rem * 0.4, 80.0)
        if t_fin > 10:
            T_fin = [1.0, 0.5, 0.3, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
            a_fin, s_fin = nesterov_logsumexp(best_seq, t_fin, T_stages=T_fin)
            try_update(a_fin, "final_lbgfs")

        # Final coordinate descent
        rem = deadline - time.monotonic()
        if rem > 15:
            a_fin_cd, s_fin_cd = coordinate_descent_polish(best_seq, min(rem * 0.4, 40.0), n_passes=20)
            try_update(a_fin_cd, "final_cd")

        # Final peak equalization
        rem = deadline - time.monotonic()
        if rem > 10:
            a_fin_pe, s_fin_pe = peak_equalization(best_seq, min(rem * 0.4, 30.0), n_rounds=1000)
            try_update(a_fin_pe, "final_pe")

    # Final verification
    if best_seq is not None:
        fs = local_eval(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
        try:
            eval_score = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
            print(f"evaluate_sequence confirms: {eval_score:.10f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")

    if best_seq is None:
        best_seq = np.ones(100) / 100

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
