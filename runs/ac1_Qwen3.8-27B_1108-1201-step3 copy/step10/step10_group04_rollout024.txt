Now, the log-sum-exp objective (up to constants) is:
`f(a) = 2n/T * log(sum_k exp(T * (a*a)_k))`

And we want to minimize this subject to `a >= 0, sum(a) = 1`.

For the constraint `sum(a) = 1`, we can use the reparameterization `a = softmax(z)` which automatically satisfies both constraints.

where `grad_a` is the gradient with respect to `a`.

This is the standard softmax parameterization gradient.

Wait, actually: `(a*a)_k = sum_j a_j * a_{k-j}`. The partial derivative with respect to `a_i` is:

For simplicity (and since the boundary cases are rare), let's use `2*a_{k-i}`.

So the subgradient component for index `i` is:
`g_i = (1/|S|) * sum_{k in S} 2*a_{k-i}`

This can also be computed via cross-correlation!

`g_i = (2/|S|) * sum_{k in S} a_{k-i}`

If S is the set of top K indices, and we define the indicator vector `e_S` (1 at positions in S, 0 elsewhere), then:
`g = (2/|S|) * cross_corr(e_S, a)`

This is again O(n log n) with FFT.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Efficient LogSumExp gradient descent with FFT-based gradient (O(n log n) per step)
2. Reduced-constraint LP best response with Nesterov acceleration (main workhorse)
3. Multiplicative weights update variant
4. Multi-length exploration with interpolation
5. Final polishing with combined methods
Key improvements: O(n log n) gradient via cross-correlation FFT, adaptive LP active set,
fixed Nesterov coefficients, and better time allocation.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_cross_convolve(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]

def cross_correlation(w, a):
    """Compute (w ⊛ a)[i] = sum_j w[i+j] * a[j] for i = 0..n-1"""
    n = len(a)
    L = 1 << (3 * n - 2 - 1).bit_length()
    w_padded = np.zeros(L)
    w_padded[:len(w)] = w
    a_padded = np.zeros(L)
    a_padded[:n] = a
    W = rfft(w_padded, L)
    A = rfft(a_padded, L)
    return irfft(np.conj(W) * A, L)[:n]

def lse_objective_and_grad(a, T):
    """Compute log-sum-exp smoothed objective and its gradient."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    sum_exp = exp_Tb.sum()
    obj = (2.0 * n / T) * (max_Tb + np.log(sum_exp))
    
    w = exp_Tb / sum_exp  # softmax weights
    corr = cross_correlation(w, a)
    grad_a = 2.0 * T * corr
    
    return obj, grad_a, w, b

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))

def subgradient_active(a, top_k=20):
    """Compute subgradient using top-k active set."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    # Select top k indices
    k = min(top_k, len(b))
    top_idx = np.argsort(b)[::-1][:k]
    
    # Create indicator vector
    e = np.zeros(2*n-1)
    e[top_idx] = 1.0
    
    # Subgradient: g_i = (2/k) * sum_{j in top_idx} a[j-i] (approx)
    corr = cross_correlation(e, a)
    g = (2.0 / k) * corr
    
    return g, max_b, b

def grad_wrapping_z(grad_a, a):
    """Wrap gradient from a-space to z-space for softmax parameterization."""
    S = np.dot(grad_a, a)
    return a * (grad_a - S)

def lse_gradient_descent(a0, time_limit, T_start=200.0, T_end=0.1, n_stages=10, 
                         max_iter_per_stage=500):
    """Log-sum-exp gradient descent with annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        iters = min(max_iter_per_stage, int(remaining * 20))
        
        for it in range(max(iters, 1)):
            if time.monotonic() - t0 > time_limit:
                break
            
            a = softmax_from_z(z)
            obj, grad_a, w, b = lse_objective_and_grad(a, T_val)
            grad_z = grad_wrapping_z(grad_a, a)
            
            # Backtracking line search
            alpha = 1.0 / (T_val * 0.5)  # initial step
            improved = False
            for bt in range(8):
                z_new = z - alpha * grad_z
                a_new = softmax_from_z(z_new)
                score_new = 2.0 * n * fast_convolve(a_new).max()
                if score_new < best_score - 1e-15:
                    best_score = score_new
                    best_a = a_new.copy()
                    z = z_new
                    improved = True
                    break
                alpha *= 0.5
            
            if not improved:
                break
        
        # Evaluate at end of stage
        a = softmax_from_z(z)
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score

_lp_solver = None

def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver

def build_conv_matrix_reduced(a_fixed, active_idx):
    """Build convolution matrix for only active constraints."""
    n = len(a_fixed)
    T = np.zeros((len(active_idx), n))
    for row, k in enumerate(active_idx):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        T[row, idx] = a_fixed[k - idx]
    return T

def solve_reduced_lp(a_fixed, top_k=30, solver=None, max_constraints=100):
    """Solve reduced LP: find b to minimize max conv(a,b) over active constraints."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    
    # Compute conv(a, a) to find active constraints
    b_conv = fast_convolve(a_fixed)
    
    # Start with top_k constraints
    k = min(top_k, len(b_conv))
    active = np.argsort(b_conv)[::-1][:k]
    
    for attempt in range(5):
        T = build_conv_matrix_reduced(a_fixed, active)
        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver)
        except Exception:
            break
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        
        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0)
        s = b_val.sum()
        if s < 1e-15:
            return None, None
        b_val = b_val / s
        
        M_val = float(M_var.value)
        
        # Check if any non-active constraint is violated
        cross_conv = fast_cross_convolve(a_fixed, b_val)
        violation = cross_conv - M_val
        max_violation = violation.max()
        
        if max_violation < 1e-8 * max(1.0, M_val):
            return b_val, M_val
        
        # Add violated constraints
        worst = np.argsort(violation)[::-1]
        new_active = np.unique(np.concatenate([active, worst[:10]]))
        if len(new_active) > max_constraints:
            new_active = worst[:max_constraints]
        active = new_active
    
    return b_val, M_val

def nesterov_lp_reduced(a0, time_limit, solver=None, top_k=30, n_max_iters=100000):
    """Nesterov-accelerated reduced LP best response."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, M_b = solve_reduced_lp(y, top_k=top_k, solver=solver)
        if b_new is None:
            try:
                b_new, M_b = solve_reduced_lp(y, top_k=50, solver=solver)
            except Exception:
                continue
        
        if b_new is None:
            continue
        
        # Nesterov step
        t = it + 1
        alpha = 2.0 / (t + 1.0 + np.sqrt((t + 1.0)**2 + 4.0 * t))
        # Actually standard Nesterov: alpha_t = 2/(t+2)
        alpha = 2.0 / (t + 2.0)
        
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy
        
        a = (1.0 - alpha) * a + alpha * b_new
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        # Evaluate
        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score - 1e-12:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Check y too
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        
        # Stall detection
        if it % 100 == 0 and it > 0:
            if best_score < prev_best - 1e-8:
                stall_count = 0
            prev_best = best_score
            
            if stall_count >= 200:
                # Perturb to escape
                rng = np.random.default_rng(it * 7 + 3)
                noise = rng.random(n) * 0.05
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0
        
        if (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score

def multiplicative_weights(a0, time_limit, solver=None, top_k=30, n_max_iters=50000):
    """Multiplicative weights update with LP best response."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    eta = 0.1  # learning rate
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, M_b = solve_reduced_lp(y, top_k=top_k, solver=solver)
        if b_new is None:
            continue
        
        # Multiplicative update: a_i <- a_i * b_i^eta / normalization
        # To avoid underflow, use log-space
        log_a = np.log(np.maximum(a, 1e-300))
        log_b = np.log(np.maximum(b_new, 1e-300))
        log_a_new = log_a + eta * log_b
        a = softmax_from_z(log_a_new)
        
        # Also update y with Nesterov
        alpha = 2.0 / (it + 3.0)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        y = y / y.sum()
        
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
        
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()
    
    return best_a, best_score

def peak_relocalization(a0, time_limit, n_steps=20000):
    """Targeted peak reduction: identify peaks and move mass away."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]
        
        # Find top contributors to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_k - idx]
        
        # Try moving mass from top contributor to a "safe" position
        src_local = np.argmax(contrib)
        src = idx[src_local]
        
        # Find safe destination (doesn't contribute to any top-3 peak)
        top3 = np.argsort(b)[::-1][:3]
        unsafe = set()
        for k2 in top3:
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            unsafe.update(range(js2, je2 + 1))
        
        safe_idx = [j for j in range(n) if j not in unsafe]
        if not safe_idx:
            safe_idx = [j for j in range(n) if j != src]
        
        dst = rng.choice(safe_idx)
        
        for frac in [0.005, 0.01, 0.02, 0.03, 0.05]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-15:
                continue
            a_trial = a_trial / ss
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-15:
                a = a_trial
                best_score = score
                best_a = a.copy()
                no_improve = 0
                break
        else:
            no_improve += 1
            if no_improve > 5000:
                break
    
    return best_a, best_score

def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * np.linspace(0, 1, n))
        a = np.maximum(a ** 2, 0.01)
    elif choice == 4:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 3
    else:
        x = np.linspace(-1, 1, n)
        a = np.exp(-2 * x ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Efficient LogSumExp gradient descent with O(n log n) FFT-based cross-correlation gradient
2. Reduced-constraint LP best response with Nesterov acceleration (main workhorse)
3. Multiplicative weights variant for escaping local optima
4. Multi-length exploration with interpolation and re-optimization
5. Peak relocalization local search for final polishing
Key improvements: fast cross-correlation gradient, adaptive LP active set,
proper Nesterov coefficients, better time allocation, and multi-scale search.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(a, b):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def cross_correlation(w, a):
    """Compute (w star a)[i] = sum_j w[i+j] * a[j] for i=0..n-1 using FFT."""
    n = len(a)
    L = 1 << (3 * n - 2 - 1).bit_length()
    w_padded = np.zeros(L)
    w_padded[:len(w)] = w
    a_padded = np.zeros(L)
    a_padded[:n] = a
    W = rfft(w_padded, L)
    A = rfft(a_padded, L)
    return irfft(np.conj(W) * A, L)[:n]


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def grad_a_to_z(grad_a, a):
    """Wrap gradient from a-space to z-space for softmax parameterization."""
    S = np.dot(grad_a, a)
    return a * (grad_a - S)


def lse_grad(a, T):
    """Compute gradient of log-sum-exp smoothed max of conv(a,a)."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    sum_exp = exp_Tb.sum()
    w = exp_Tb / sum_exp
    corr = cross_correlation(w, a)
    grad = 2.0 * T * corr
    return grad, w, b


def lse_gradient_descent(a0, time_limit, T_start=200.0, T_end=0.1, n_stages=10):
    """Log-sum-exp gradient descent with temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()
    temps = np.geomspace(T_start, T_end, n_stages)

    for si, T_val in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break
        max_iters = max(50, int(remaining * 30))

        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit:
                break
            a = softmax_from_z(z)
            grad_a, w, b = lse_grad(a, T_val)
            grad_z = grad_a_to_z(grad_a, a)
            g_norm = np.linalg.norm(grad_z)
            if g_norm < 1e-12:
                break

            alpha = 1.0 / (T_val * 0.3 + 1e-6)
            improved = False
            for bt in range(10):
                z_new = z - alpha * grad_z
                a_new = softmax_from_z(z_new)
                score_new = 2.0 * n * fast_convolve(a_new).max()
                if score_new < best_score - 1e-14:
                    best_score = score_new
                    best_a = a_new.copy()
                    z = z_new
                    improved = True
                    break
                alpha *= 0.5
            if not improved:
                break

        a = softmax_from_z(z)
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        elapsed = time.monotonic() - t0
        print(f"    LSE stage {si+1}/{n_stages} (T={T_val:.1f}, {elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def build_conv_matrix_reduced(a_fixed, active_idx):
    """Build convolution constraint matrix for subset of indices."""
    n = len(a_fixed)
    T = np.zeros((len(active_idx), n))
    for row, k in enumerate(active_idx):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        T[row, idx] = a_fixed[k - idx]
    return T


def solve_reduced_lp(a_fixed, top_k=30, solver=None, max_constraints=150):
    """Solve reduced LP: minimize max_k (a*b)_k over active constraints."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_conv = fast_convolve(a_fixed)
    k = min(top_k, len(b_conv))
    active = np.argsort(b_conv)[::-1][:k]
    b_val = None
    M_val = None

    for attempt in range(6):
        T = build_conv_matrix_reduced(a_fixed, active)
        b_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver)
        except Exception:
            break
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            for alt in [cp.ECOS, cp.SCS, cp.CLARABEL]:
                if alt == solver:
                    continue
                try:
                    prob.solve(solver=alt)
                    if prob.status in ('optimal', 'optimal_inaccurate'):
                        break
                except Exception:
                    continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0)
        s = b_val.sum()
        if s < 1e-15:
            return None, None
        b_val = b_val / s
        M_val = float(M_var.value)

        cross_conv = fast_cross_convolve(a_fixed, b_val)
        violation = cross_conv - M_val
        max_viol = violation.max()
        if max_viol < 1e-7 * max(1.0, abs(M_val)):
            return b_val, M_val
        worst = np.argsort(violation)[::-1]
        new_active = np.unique(np.concatenate([active, worst[:15]]))
        if len(new_active) > max_constraints:
            new_active = worst[:max_constraints]
        active = new_active

    return b_val, M_val

def peak_relocalization(a0, time_limit):
    """Targeted peak reduction via mass transfer from high contributors."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(54321)
    no_improve = 0

    for step in range(50000):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_k - idx]
        src_local = np.argmax(contrib)
        src = idx[src_local]

        top5 = np.argsort(b)[::-1][:5]
        unsafe = set()
        for k2 in top5:
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            unsafe.update(range(js2, je2 + 1))
        safe_idx = [j for j in range(n) if j not in unsafe]
        if not safe_idx:
            safe_idx = [j for j in range(n) if j != src]
        dst = rng.choice(safe_idx)

        found = False
        for frac in [0.005, 0.01, 0.02, 0.03, 0.05]:
            amt = a[src] * frac
            if amt < 1e-15:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-15:
                continue
            a_trial = a_trial / ss
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-14:
                a = a_trial
                best_score = score
                best_a = a.copy()
                no_improve = 0
                found = True
                break
        if not found:
            no_improve += 1
            if no_improve > 8000:
                break
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
