def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate 2n*max(conv(a,a))/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_T(a):
    """Build (2n-1) x n convolution matrix T such that T@b = convolve(a,b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def nesterov_lp_optimize(a0, time_limit, solver=None, max_iters=200000,
                         stall_patience=15, lp_max_iter=2000):
    """Nesterov LP alternation with Cesaro averaging and adaptive restarts."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall = 0
    rng = np.random.default_rng(12345)

    # Cesaro accumulator
    cesaro_a = np.zeros(n)
    total_iters = 0
    last_b = a.copy()

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver, max_iter=lp_max_iter, warm_b=last_b)
        if b_new is None:
            stall += 1
            if stall > 25:
                break
            continue
        last_b = b_new

        # Nesterov update
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        # Primal iterated average
        total_iters += 1
        cesaro_a += b_new
        a = cesaro_a / total_iters
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        # Track best (check both y and Cesaro average)
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        # Stall detection
        if (it + 1) % 200 == 0:
            if best_score < prev_best - 1e-10:
                stall = 0
            else:
                stall += 1
            prev_best = best_score

            if stall >= stall_patience:
                noise_scale = 0.015 / max(1, np.sqrt(stall))
                noise = rng.standard_normal(n) * noise_scale
                y = y + noise
                y = np.maximum(y, 0)
                sy2 = y.sum()
                if sy2 > 1e-15:
                    y = y / sy2
                # Reset Cesaro
                cesaro_a = y.copy()
                total_iters = 1
                a = y.copy()
                stall = 0

    return best_a, best_score

def mirror_descent_lp(a0, time_limit, n_steps=50000):

        # This is the multiplicative weights / exponential weights update

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    Phase 1: Multi-length quick LP exploration
    Phase 2: Long LP alternation on best length
    Phase 3: LogSumExp L-BFGS aggressive cooling
    Phase 4: Mirror descent LP hybrid
    Phase 5: Peak coordinate descent
    Phase 6: Final LP polish
    Phase 7: Final CD + subgradient
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Multi-length quick LP exploration ===
    print(f"\n=== Phase 1: Multi-length exploration, current best={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    lengths_to_try = [800, 1000, 1200, 1500, 2000]
    # Include current best length
    if n_best not in lengths_to_try:
        lengths_to_try.append(n_best)
    lengths_to_try.sort()

    t_each = min(rem / max(1, len(lengths_to_try)) * 0.8, 35.0)
    for ln in lengths_to_try:
        if time.monotonic() > deadline - 60:
            break
        if t_each < 10:
            break
        if ln == n_best:
            # Skip if we already have a good solution at this length
            # (will be handled in Phase 2)
            continue
        print(f"  Trying n={ln}, t={t_each:.0f}s")
        if parent is not None:
            init = interpolate_sequence(parent, ln)
        else:
            init = structured_init(ln, rng)
        a_res, _ = nesterov_lp_optimize(init, t_each, solver=solver,
                                         max_iters=50000, stall_patience=8,
                                         lp_max_iter=1000)
        try_update(a_res, f"lp_n{ln}")

    # === Phase 2: Long LP on best sequence ===
    print(f"\n=== Phase 2: Long LP, n={len(best_seq)}, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.35, 550.0)
    if t2 > 60:
        print(f"  Budget: {t2:.0f}s")
        a_lp, _ = nesterov_lp_optimize(best_seq, t2, solver=solver,
                                        max_iters=300000, stall_patience=12,
                                        lp_max_iter=3000)
        try_update(a_lp, "lp_long")

    # === Phase 3: LogSumExp L-BFGS aggressive cooling ===
    print(f"\n=== Phase 3: LogSumExp L-BFGS, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.20, 300.0)
    if t3 > 20:
        T_stages = [12.0, 6.0, 4.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.35,
                    0.25, 0.18, 0.12, 0.08, 0.05, 0.03, 0.02, 0.01]
        print(f"  {len(T_stages)} stages, budget={t3:.0f}s")
        a_lb = logsumexp_lbgfs(best_seq, t3, T_stages=T_stages, maxiter_per_stage=20000)
        try_update(a_lb, "lbgfs")

    # === Phase 4: Mirror descent LP hybrid ===
    print(f"\n=== Phase 4: Mirror Descent LP, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.12, 180.0)
    if t4 > 30:
        a_md, _ = mirror_descent_lp(best_seq, t4, n_steps=30000)
        try_update(a_md, "mirror_lp")

    # === Phase 5: Peak coordinate descent ===
    print(f"\n=== Phase 5: Peak CD, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.12, 150.0)
    if t5 > 15:
        a_cd, _ = peak_coord_descent(best_seq, t5, n_steps=200000, top_m=10)
        try_update(a_cd, "peak_cd")

    # === Phase 6: Final LP polish ===
    print(f"\n=== Phase 6: LP Polish, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.12, 150.0)
    if t6 > 30:
        a_lp2, _ = nesterov_lp_optimize(best_seq, t6, solver=solver,
                                         max_iters=100000, stall_patience=10,
                                         lp_max_iter=2000)
        try_update(a_lp2, "lp_polish")

    # === Phase 7: Final subgradient + CD ===
    print(f"\n=== Phase 7: Final refinement, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t_sg = min(rem * 0.4, 60.0)
        if t_sg > 10:
            a_sg, _ = subgradient_refine(best_seq, t_sg, n_steps=50000)
            try_update(a_sg, "subgrad_final")

        rem = deadline - time.monotonic()
        if rem > 25:
            t_cd = min(rem * 0.5, 40.0)
            if t_cd > 8:
                a_cd2, _ = peak_coord_descent(best_seq, t_cd, n_steps=80000, top_m=8)
                try_update(a_cd2, "cd_final")

        # Final LogSumExp with very low temperature
        rem = deadline - time.monotonic()
        if rem > 20:
            a_fin = logsumexp_lbgfs(best_seq, min(rem * 0.6, 30.0),
                                     T_stages=[1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01],
                                     maxiter_per_stage=8000)
            try_update(a_fin, "lbgfs_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
