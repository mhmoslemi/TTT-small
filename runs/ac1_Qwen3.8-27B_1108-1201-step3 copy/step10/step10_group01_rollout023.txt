"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using accelerated LP best-response
with Nesterov averaging, FFT-based LogSumExp L-BFGS refinement, and
targeted coordinate descent. Multi-phase with adaptive time allocation.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize, linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))

def solve_lp_highs(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a_fixed
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([T, -np.ones((size, 1))])
    b_ub = np.zeros(size)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.ones(1)
    bounds = [(0, None)] * n + [(-np.inf, np.inf)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        b_val = np.maximum(res.x[:n], 0)
        sv = b_val.sum()
        if sv < 1e-15:
            return None, None
        return b_val / sv, float(res.x[-1])
    return None, None

def solve_lp_cvxpy(a_fixed, solver_name):
    import cvxpy as cp
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a_fixed
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver_name, verbose=False)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0)
            sv = b_val.sum()
            if sv < 1e-15:
                return None, None
            return b_val / sv, float(M_var.value)
    except Exception:
        pass
    return None, None

def nesterov_lp(a0, time_limit, solver='highs'):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    prev_best = best_score
    stall = 0
    rng = np.random.default_rng(777)
    
    for it in range(500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        if solver == 'highs':
            b_new, M_b = solve_lp_highs(y)
        else:
            b_new, M_b = solve_lp_cvxpy(y, solver)
        
        if b_new is None:
            stall += 1
            if stall > 30:
                break
            continue
        
        alpha = 2.0 / (it + 2.0)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy
        
        w = (it + 1.0) + alpha * (it + 2.0)
        a = ((it + 1.0) * a + alpha * (it + 2.0) * b_new) / w
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        sc_a = evaluate_local(a)
        sc_y = evaluate_local(y)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        
        if (it + 1) % 2000 == 0:
            if best_score < prev_best - 1e-10:
                stall = 0
            else:
                stall += 1
            prev_best = best_score
            elapsed = time.monotonic() - t0
            if (it + 1) % 10000 == 0:
                print(f"  NLP it={it+1} ({elapsed:.0f}s) score={best_score:.8f} M={M_b:.8f}")
            if stall >= 10:
                ns = 0.015 / np.sqrt(stall)
                y = y + ns * rng.standard_normal(n)
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall = 0
    
    return best_a, best_score

def lse_lbgfs(a0, time_limit, T_stages=None):
    """LogSumExp L-BFGS with FFT-accelerated gradient."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    z = np.log(np.maximum(a, 1e-300))
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [30.0, 15.0, 8.0, 5.0, 3.0, 2.0, 1.5, 1.0, 0.7, 0.5, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.015, 0.01]
    
    # Precompute FFT size
    N_ff = 1 << (3 * n - 2).bit_length()
    
    best_a = a.copy()
    best_score = evaluate_local(a)
    
    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        max_iter = max(10, int(remaining * 50))  # ~50 L-BFGS iters per second target
        
        def func(z):
            a = np.exp(z - z.max())
            a = a / a.sum()
            b = fast_convolve(a)
            mb = b.max()
            return (2.0 * n / T) * (mb + np.log(np.exp(b - mb).sum()))
        
        def jac(z):
            a = np.exp(z - z.max())
            a = a / a.sum()
            b = fast_convolve(a)
            mb = b.max()
            w = np.exp(b - mb)
            w = w / w.sum()
            # grad_raw[j] = sum_k w[k] * a[k-j] (with k-j in [0, n-1])
            # This is cross-correlation of w and a
            W = fft(w, N_ff)
            A = fft(a, N_ff)
            grad_raw = ifft(W * np.conj(A)).real[:n]
            grad_raw *= 2.0 / T
            # Softmax gradient: a_j * (grad_raw_j - sum(a * grad_raw))
            S = np.dot(a, grad_raw)
            return a * (grad_raw - S)
        
        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': max_iter * 200})
            z = res.x
        except Exception:
            pass
        
        a_cur = np.exp(z - z.max())
        a_cur = a_cur / a_cur.sum()
        sc = evaluate_local(a_cur)
        if sc < best_score:
            best_score = sc
            best_a = a_cur.copy()
    
    return best_a, best_score
