def fast_convolve(a):
    """Compute a*a (full convolution) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a_fixed):
    """Build matrix T such that T @ b = a_fixed * b (convolution)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def softmax_from_z(z):
    """Convert log-vertices to simplex."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert simplex point to log-vertices."""
    return np.log(np.maximum(a, 1e-300))

_lp_solver = None

def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:

    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    """Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0. Returns (b, M)."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        prob.solve(solver=cp.ECOS)
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            prob = cp.Problem(cp.Minimize(M_var), constraints)
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def flat_spectrum_init(n, rng, n_iter=50):
    """Initialize using iterative flat-spectrum (Gerchberg-Saxton) approach."""
    # Target: flat |DFT| for all frequencies
    # A(0) should be sqrt(n) (since sum(a)=1 means A(0)=1 in normalized DFT)
    # Actually with a normalized to sum=1, A(0) = 1, and we want |A(k)| ~ 1/sqrt(n) for k != 0
    
    # Start with random phase
    A = np.zeros(n, dtype=complex)
    A[0] = 1.0  # DC component (sum = 1)
    phases = rng.uniform(0, 2 * np.pi, n)
    A[1:] = np.exp(1j * phases[1:]) / np.sqrt(n)
    
    for _ in range(n_iter):
        # Inverse DFT
        a = np.fft.ifft(A).real
        # Project to non-negative
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        # Forward DFT
        A_new = np.fft.fft(a)
        # Keep new phase, set amplitude
        A[0] = abs(A_new[0])  # Keep DC magnitude
        for k in range(1, n):
            phase_k = np.angle(A_new[k])
            A[k] = np.exp(1j * phase_k) / np.sqrt(n)
        A = A  # Already updated
    
    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def nesterov_lp_accelerated(a0, time_budget, solver=None, verbose=True):
    """
    Nesterov-accelerated LP best-response method.
    Uses proper Nesterov scheduling with periodic restarts.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Nesterov state
    y = a.copy()  # extrapolated point
    x_prev = a.copy()
    t_nesterov = 1.0
    
    restart_interval = 200  # restart every 200 iterations
    stall_threshold = 50
    stall_count = 0
    prev_best = best_score
    momentum_mix = 0.7  # blend Nesterov with simple averaging
    
    if verbose:
        print(f"    NesterovLP start: score={best_score:.6f}, n={n}")
    
    for it in range(1000000):
        if time.monotonic() - t0 > time_budget:
            break
        
        # Solve LP for best response to y
        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue
        
        # Nesterov update
        t_next = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nesterov * t_nesterov))
        alpha = (t_nesterov - 1.0) / t_next
        
        # New y (extrapolated point)
        y_new = y + alpha * (b_star - x_prev)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / sy
        
        # Update x (the actual iterate) = best response
        x_new = b_star.copy()
        
        # Track best
        score_x = 2.0 * n * fast_convolve(x_new).max()
        if score_x < best_score - 1e-12:
            best_score = score_x
            best_a = x_new.copy()
            stall_count = 0
        
        score_y = 2.0 * n * fast_convolve(y_new).max()
        if score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y_new.copy()
            stall_count = 0
        
        x_prev = x_new
        y = y_new
        t_nesterov = t_next
        
        # Periodic restart to avoid divergence
        if (it + 1) % restart_interval == 0:
            if best_score < prev_best - 1e-8:
                prev_best = best_score
                stall_count = 0
            else:
                stall_count += 1
            
            if stall_count >= stall_threshold:
                # Restart from best point
                y = best_a.copy()
                x_prev = best_a.copy()
                t_nesterov = 1.0
                stall_count = 0
                prev_best = best_score
                if verbose:
                    elapsed = time.monotonic() - t0
                    print(f"    NesterovLP restart at iter {it+1} ({elapsed:.0f}s): score={best_score:.6f}")
        
        if verbose and (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def subgradient_mirror_descent(a0, time_budget, verbose=False):

        max_b = b.max()
        # Active set: top few% of correlation peaks

        if len(active) == 0:

            idx = np.arange(js, je + 1)

        # Mirror descent step

            elapsed = time.monotonic() - t0
            print(f"    SubGradMD iter={it} ({elapsed:.0f}s): score={best_score:.6f}")

2. Nesterov-accelerated LP best-response with Polyak averaging and adaptive restarts

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a_fixed):
    """Build matrix T such that T @ b = full_convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    k_idx = np.arange(size)
    j_idx = np.arange(n)
    diff = k_idx[:, None] - j_idx[None, :]
    mask = (diff >= 0) & (diff < n)
    T[mask] = a_fixed[diff[mask]]
    return T


def softmax_from_z(z):
    """Convert log-space to simplex."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert simplex to log-space."""
    return np.log(np.maximum(a, 1e-300))


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


_lp_solver = None

def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    """Solve LP: min M s.t. conv(a_fixed,b)_k <= M for all k, sum(b)=1, b>=0."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            prob = cp.Problem(cp.Minimize(M_var), constraints)
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def flat_spectrum_init(n, rng, n_gs_iter=100):
    """Flat-spectrum initialization via Gerchberg-Saxton / IAA algorithm."""
    A = np.zeros(n, dtype=complex)
    A[0] = 1.0
    phases = rng.uniform(0, 2 * np.pi, n)
    A[1:] = np.exp(1j * phases[1:]) / np.sqrt(n)

    for _ in range(n_gs_iter):
        a = np.fft.ifft(A).real
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
        A_new = np.fft.fft(a)
        for k in range(1, n):
            A[k] = np.exp(1j * np.angle(A_new[k])) / np.sqrt(n)
        A[0] = abs(A_new[0])

    a = np.fft.ifft(A).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def structured_init(n, rng, method=0):
    """Generate structured initializations."""
    if method == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif method == 1:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    elif method == 2:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2 * x) + 0.2 * np.cos(5 * x) + 0.1 * np.cos(7 * x)
        a = np.maximum(a, 0)
    elif method == 3:
        x = np.linspace(-1, 1, n)
        a = np.exp(-2 * x ** 2)
    elif method == 4:
        a = np.minimum(np.linspace(0, 1, n), np.linspace(1, 0, n))
    else:
        a = rng.random(n) + 0.1
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def nesterov_lp_main(a0, time_budget, solver=None, verbose=True):
    """
    Nesterov-accelerated LP best-response with Polyak averaging.
    The main optimization workhorse.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    # Nesterov state
    x = a.copy()       # current iterate
    y = a.copy()       # extrapolated point (where we solve LP)
    x_prev = a.copy()
    t_nest = 1.0

    # Polyak average
    polyak_avg = a.copy()
    total_alpha = 0.0

    # Stall tracking
    stall_count = 0
    prev_best_at_check = best_score
    check_interval = 100

    if verbose:
        print(f"    [NestLP] Start: score={best_score:.8f}, n={n}")

    for it in range(500000):
        if time.monotonic() - t0 > time_budget:
            break

        # Solve LP for best response to y
        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue

        # Nesterov update
        t_next = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        alpha_nest = (t_nest - 1.0) / t_next

        # Extrapolate
        y_new = y + alpha_nest * (b_star - x_prev)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / sy

        # x update: move toward best response
        x_new = b_star.copy()

        # Update Polyak average
        alpha_polyak = 1.0 / (it + 1.0)
        polyak_avg = (1.0 - alpha_polyak) * polyak_avg + alpha_polyak * x_new
        total_alpha += alpha_polyak

        # Track best (check actual score, not just LP value)
        score_x = 2.0 * n * fast_convolve(x_new).max()
        if score_x < best_score - 1e-12:
            best_score = score_x
            best_a = x_new.copy()
            stall_count = 0

        # Also check y
        score_y = 2.0 * n * fast_convolve(y_new).max()
        if score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y_new.copy()
            stall_count = 0

        # Check Polyak average periodically
        if (it + 1) % 500 == 0:
            score_p = 2.0 * n * fast_convolve(polyak_avg).max()
            if score_p < best_score - 1e-12:
                best_score = score_p
                best_a = polyak_avg.copy()
                stall_count = 0

        x_prev = x_new
        x = x_new
        y = y_new
        t_nest = t_next

        # Stall detection and restart
        if (it + 1) % check_interval == 0:
            if best_score < prev_best_at_check - 1e-10:
                prev_best_at_check = best_score
                stall_count = 0
            else:
                stall_count += 1

            if stall_count >= 100:
                # Restart from best point with perturbation
                rng_r = np.random.default_rng(it * 7 + 42)
                noise_scale = 0.02 + 0.03 * min(stall_count / 200.0, 1.0)
                y = best_a + rng_r.normal(0, noise_scale, n)
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy < 1e-15:
                    y = best_a.copy()
                else:
                    y = y / sy
                x = y.copy()
                x_prev = y.copy()
                t_nest = 1.0
                polyak_avg = best_a.copy()
                stall_count = 0
                prev_best_at_check = best_score
                if verbose:
                    elapsed = time.monotonic() - t0
                    print(f"    [NestLP] Restart at iter {it+1} ({elapsed:.0f}s): score={best_score:.8f}")

        if verbose and (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    [NestLP] iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def lbgfs_logsumexp(a0, time_budget, T_start=30.0, T_end=0.05, n_stages=10):
    """L-BFGS with LogSumExp smooth approximation, multi-stage annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_budget:
            break
        remaining = time_budget - (time.monotonic() - t0)
        if remaining < 2:
            break
        max_iter = max(50, int(remaining * 15))

        def stage_func(z_in):
            a_local = softmax_from_z(z_in)
            b_local = fast_convolve(a_local)
            Tb_local = stage_T * b_local
            max_Tb = min(Tb_local.max(), 600.0)
            exp_Tb = np.exp(Tb_local - max_Tb)
            lse_val = max_Tb + np.log(exp_Tb.sum())
            return 2.0 * n / stage_T * lse_val

        def stage_grad(z_in):
            a_local = softmax_from_z(z_in)
            b_local = fast_convolve(a_local)
            Tb_local = stage_T * b_local
            max_Tb = min(Tb_local.max(), 600.0)
            exp_Tb = np.exp(Tb_local - max_Tb)
            w_local = exp_Tb / exp_Tb.sum()

            grad_a_local = np.zeros(n)
            for k in range(len(b_local)):
                wk = w_local[k]
                if wk < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a_local[idx] += 2.0 * wk * a_local[k - idx]

            S_local = np.dot(grad_a_local, a_local)
            grad_z_local = a_local * (grad_a_local - S_local)
            return 2.0 * n * grad_z_local

        try:
            res = minimize(
                fun=stage_func, x0=z, jac=stage_grad, method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16,
                         'gtol': 1e-12, 'maxfun': max_iter * 200,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def subgradient_mirror_descent(a0, time_budget, verbose=False):
    """Mirror descent with subgradient of max correlation in log-space."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    z = z_from_a(a).copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta = 0.15
    decay = 0.9995

    for it in range(1000000):
        if time.monotonic() - t0 > time_budget:
            break

        b = fast_convolve(a)
        max_b = b.max()
        if max_b < 1e-15:
            break
        threshold = max_b * 0.95
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)

        z = z - eta * g
        a_new = softmax_from_z(z)
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
        a = a_new
        eta *= decay

    return best_a, best_score


def peak_directional_search(a0, time_budget, verbose=False):
    """
    Directional search: identify peak-contributing coordinates and
    move mass from high contributors to low ones.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(555)

    for outer in range(100000):
        if time.monotonic() - t0 > time_budget:
            break

        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        max_val = b[peak_k]

        # Find contributing coordinates
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_k - idx]
        total_contrib = contrib.sum()
        if total_contrib < 1e-15:
            break

        # Try moving mass from top contributor to a random non-contributing position
        src_local = np.argmax(contrib)
        src = idx[src_local]

        # Find target: position with low value that doesn't contribute much to peak
        peak_set = set(idx.tolist())
        non_peak = np.array([j for j in range(n) if j not in peak_set])
        if len(non_peak) == 0:
            non_peak = np.array([j for j in range(n) if j != src])

        dst_candidates = non_peak[np.argsort(a[non_peak])[:max(1, n // 10)]]
        if len(dst_candidates) == 0:
            break
        dst = rng.choice(dst_candidates)

        improved = False
        for frac in [0.005, 0.01, 0.02, 0.03, 0.05]:
            amt = a[src] * frac
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-15:
                continue
            a_trial = a_trial / ss
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-14:
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True
                break

        if not improved:
            # Try a broader random move
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.005, 0.03)
                a_trial = a.copy()
                amt = a[i] * frac
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss > 1e-15:
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True

            if not improved and outer % 5000 == 0 and outer > 0:
                break

    return best_a, best_score

            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  ** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s) **")

    # === Phase 1: Main LP optimization from parent ===
    print(f"\n=== Phase 1: Nesterov LP from parent (score={best_score:.8f}) ===")
    rem = time_limit - time.monotonic()
    t1 = min(rem * 0.45, 700.0)
    if t1 > 60:
        print(f"  Budget: {t1:.0f}s")
        a_out, _ = nesterov_lp_main(best_seq, t1, solver=solver, verbose=True)
        try_update(a_out, "nestlp_parent")

    # === Phase 2: LSE L-BFGS polish ===
    print(f"\n=== Phase 2: LSE L-BFGS polish (score={best_score:.8f}) ===")
    rem = time_limit - time.monotonic()
    t2 = min(rem * 0.15, 150.0)
    if t2 > 15:
        print(f"  Budget: {t2:.0f}s")
        a_out = lbgfs_logsumexp(best_seq, t2, T_start=20.0, T_end=0.05, n_stages=8)
        try_update(a_out, "lse_polish")

    # === Phase 3: Another LP round from current best ===
    print(f"\n=== Phase 3: Second LP round (score={best_score:.8f}) ===")
    rem = time_limit - time.monotonic()
    t3 = min(rem * 0.20, 250.0)
    if t3 > 30:
        print(f"  Budget: {t3:.0f}s")
        a_out, _ = nesterov_lp_main(best_seq, t3, solver=solver, verbose=True)
        try_update(a_out, "nestlp_r2")

    # === Phase 4: Peak directional search ===
    print(f"\n=== Phase 4: Peak directional search (score={best_score:.8f}) ===")
    rem = time_limit - time.monotonic()
    t4 = min(rem * 0.10, 80.0)
    if t4 > 10:
        print(f"  Budget: {t4:.0f}s")
        a_out, _ = peak_directional_search(best_seq, t4, verbose=False)
        try_update(a_out, "peak_dir")

    # === Phase 5: Try flat-spectrum init at same or different length ===
    print(f"\n=== Phase 5: Flat-spectrum alternative (score={best_score:.8f}) ===")
    rem = time_limit - time.monotonic()
    if rem > 120:
        t5 = min(rem * 0.15, 150.0)
        # Try a few different flat-spectrum inits
        for trial in range(3):
            if time.monotonic() > time_limit:
                break
            t_trial = t5 / 3.0
            if t_trial < 20:
                break
            rng_trial = np.random.default_rng(seed + 1000 * trial)
            init = flat_spectrum_init(n_base, rng_trial, n_gs_iter=60)
            sc_init = evaluate_safe(init)
            print(f"  Flat-spectrum trial {trial}: score={sc_init:.8f}")
            if sc_init < best_score:
                try_update(init, f"flat_init_{trial}")

            # Quick LP optimization from this init
            a_out, _ = nesterov_lp_main(init, t_trial, solver=solver, verbose=False)
            try_update(a_out, f"flat_lp_{trial}")

    # === Phase 6: Try longer length ===
    print(f"\n=== Phase 6: Longer length exploration (score={best_score:.8f}) ===")
    rem = time_limit - time.monotonic()
    if rem > 150:
        n_long = int(n_base * 1.5)
        if n_long > n_base:
            t6 = min(rem * 0.20, 200.0)
            if t6 > 60:
                print(f"  Trying n={n_long}, budget={t6:.0f}s")
                init_long = interpolate_sequence(best_seq, n_long)
                sc_long = evaluate_safe(init_long)
                print(f"  Interpolated init: score={sc_long:.8f}")
                if sc_long < best_score:
                    try_update(init_long, "interp_long")

                a_out, _ = nesterov_lp_main(init_long, t6, solver=solver, verbose=False)
                try_update(a_out, f"lp_long_n{n_long}")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final polish (score={best_score:.8f}) ===")
    rem = time_limit - time.monotonic()
    if rem > 20:
        # Short LSE
        t7a = min(rem * 0.3, 60.0)
        if t7a > 5:
            a_out = lbgfs_logsumexp(best_seq, t7a, T_start=5.0, T_end=0.02, n_stages=6)
            try_update(a_out, "final_lse")

        # Short LP
        rem = time_limit - time.monotonic()
        t7b = min(rem * 0.4, 60.0)
        if t7b > 5:
            a_out, _ = nesterov_lp_main(best_seq, t7b, solver=solver, verbose=False)
            try_update(a_out, "final_lp")

        # Peak search
        rem = time_limit - time.monotonic()
        t7c = min(rem * 0.3, 30.0)
        if t7c > 3:
            a_out, _ = peak_directional_search(best_seq, t7c, verbose=False)
            try_update(a_out, "final_peak")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL: score={best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
