import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate sequence using local computation (normalized to sum=1)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver = None

                _lp_solver = solver_name
                return _lp_solver
        except Exception:

    _lp_solver = cp.ECOS

def solve_lp_best_response(a_fixed, solver=None, tol=1e-8):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:

    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    for s in [solver, cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:

                sv = b_val.sum()
                if sv < 1e-15:

    return None, None

            a_new = a.copy()
            a_sum = 1.0
        a_new = a_new / a_sum

        # Nesterov
        alpha_ns = 2.0 / (it + 2)
        y = (1.0 - alpha_ns) * y + alpha_ns * a_new
        y = np.maximum(y, 0)
        ys = y.sum()
        if ys < 1e-15:
            y = a_new.copy()
        else:
            y = y / ys

        a = a_new
        score = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        elif score_y < best_score - 1e-13:
            best_score = score_y
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 300:
            noise = rng.standard_normal(n) * 0.008
            a = best_a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            y = a.copy()
            no_improve = 0

        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD iter={it+1} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def coordinate_descent_refine(a0, time_limit, n_steps=50000):
    """Targeted coordinate descent on peak-contributing elements."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    stall = 0
    rng = np.random.default_rng(999)

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.9999
        peaks = np.where(b >= threshold)[0]

        improved = False
        for peak_k in peaks[:3]:  # Limit to top 3 peaks
            if improved:
                break
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_m = min(8, len(idx))
            top_positions = idx[np.argsort(contribs)[::-1][:top_m]]

            for pos in top_positions:
                if a[pos] < 1e-14:
                    continue
                for factor in [0.999, 0.998, 0.997, 0.995, 0.993, 0.99, 0.985, 0.98]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] - a_trial[pos]
                    if diff < 1e-16:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] = others + diff * others / others_sum
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * len(a_trial) * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            if stall > 500:
                # Try reducing top-2 peak contributors simultaneously
                a_trial = a.copy()
                peak_k = int(np.argmax(b))
                js = max(0, peak_k - n + 1)
                je = min(n - 1, peak_k)
                idx = np.arange(js, je + 1)
                contribs = a[idx] * a[peak_k - idx]
                top_positions = idx[np.argsort(contribs)[::-1][:3]]
                for pos in top_positions:
                    if a[pos] > 1e-12:
                        a_trial[pos] *= 0.998
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
                else:
                    stall += 1
                    if stall > 2000:
                        break
        else:
            stall = 0

        if (step + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD step={step+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def logsumexp_smooth_optimize(a0, time_limit):
    """
    Smoothed optimization using log-sum-exp approximation of max.
    Uses scipy minimize with L-BFGS-B.
    """
    from scipy.optimize import minimize
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    z0 = np.log(np.maximum(a_init, 1e-300))

    t0 = time.monotonic()
    T_stages = [20.0, 10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]
    z = z0.copy()

    for T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        if time.monotonic() - t0 > time_limit - 5:
            break

        def func(z_inner):
            a_inner = np.exp(z_inner - z_inner.max())
            a_inner = a_inner / a_inner.sum()
            b = fast_convolve(a_inner)
            shifted = b - b.max()
            lse = b.max() + np.log(np.exp(shifted).sum())
            return (2.0 * n / T) * lse

        def jac(z_inner):
            a_inner = np.exp(z_inner - z_inner.max())
            a_inner = a_inner / a_inner.sum()
            b = fast_convolve(a_inner)
            shifted = b - b.max()
            exp_b = np.exp(shifted)
            w = exp_b / exp_b.sum()
            grad_a = np.zeros(n)
            size = 2 * n - 1
            for k in range(size):
                if w[k] < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_inner[k - idx]
            S = np.dot(grad_a, a_inner)
            return a_inner * (grad_a - S)

        try:
            maxiter = max(50, int(300 * (time_limit - (time.monotonic() - t0)) / len(T_stages)))
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12,
                                    'maxfun': maxiter * 100, 'disp': False})
            z = res.x
        except Exception:
            pass

    a_out = np.exp(z - z.max())
    a_out = a_out / a_out.sum()
    return a_out


def structured_init(n, rng):
    """Create structured initial sequence with various patterns."""
    choice = rng.integers(0, 7)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        # Power law (heavily weighted toward end)
        p = rng.uniform(0.3, 2.0)
        a = ((x + 0.01) ** p)
    elif choice == 4:
        # Triangle
        a = np.minimum(x, 1 - x) ** 0.5
    elif choice == 5:
        # Asymmetric bell
        mu = rng.uniform(0.3, 0.7)
        sig = rng.uniform(0.1, 0.3)
        a = np.exp(-0.5 * ((x - mu) / sig) ** 2)
    else:
        # Fejer-like
        a = np.sinc(5 * (x - 0.5))
        a = np.maximum(a, 0)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def peak_reduction_refine(a0, time_limit):
    """
    Iteratively reduce the dominant peak by identifying contributing pairs
    and redistributing mass away from them.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(55555)

    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))

        # Find top contributors to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        total_contrib = contribs.sum()

        if total_contrib < 1e-15:
            break

        # Reduce top contributors and spread to least-contributing positions
        n_reduce = min(5, len(idx))
        top_pos = idx[np.argsort(contribs)[::-1][:n_reduce]]
        # Spread to positions that don't contribute to the peak
        all_pos = np.arange(n)
        peak_pos_set = set(idx.tolist())
        non_peak = all_pos[~np.isin(all_pos, peak_pos_set)]
        if len(non_peak) == 0:
            non_peak = all_pos

        reduction = 0.001  # Fraction to reduce
        mass_moved = 0.0
        for pos in top_pos:
            if a[pos] > 1e-12:
                delta = a[pos] * reduction
                a[pos] -= delta
                mass_moved += delta

        if mass_moved < 1e-15:
            break

        # Distribute to non-peak positions weighted by current value
        weights = a[non_peak] + 1e-10
        weights = weights / weights.sum()
        a[non_peak] += mass_moved * weights

        a = np.maximum(a, 0)
        a = a / a.sum()
        score = 2.0 * n * fast_convolve(a).max()

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
        else:
            # Revert and try smaller reduction
            a = best_a.copy()
            if reduction < 1e-6:
                break
            reduction *= 0.5

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining LP best-response, mirror descent,
    log-sum-exp smoothing, and coordinate descent. Uses multi-scale
    exploration and targeted peak reduction.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Multi-n exploration with quick LP ===
    print(f"\n=== Phase 1: Multi-n exploration (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    n_base = len(best_seq)
    n_candidates = [n_base]
    for delta in [-200, -100, 100, 200, 400, 600]:
        nc = n_base + delta
        if nc >= 400 and nc <= 5000:
            n_candidates.append(nc)
    n_candidates = sorted(set(n_candidates))

    t_per_n = min(rem * 0.25 / len(n_candidates), 60.0)
    for nc in n_candidates:
        if time.monotonic() > deadline - 100:
            break
        if nc == n_base:
            a_try = best_seq.copy()
        else:
            a_try = interpolate_sequence(best_seq, nc)
        t_alloc = min(t_per_n, deadline - time.monotonic() - 100)
        if t_alloc < 10:
            continue
        a_res, sc = nesterov_lp_optimize(a_try, t_alloc, solver=solver)
        try_update(a_res, f"lp_n{nc}")

    # === Phase 2: Deep LP on best n ===
    print(f"\n=== Phase 2: Deep LP (score={best_score:.8f}, n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.40, 600.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_optimize(best_seq, t_alloc, solver=solver,
                                       eta0=0.1, patience=150)
        try_update(a_lp, "deep_lp")

    # === Phase 3: Mirror Descent ===
    print(f"\n=== Phase 3: Mirror Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 20:
        a_md, _ = mirror_descent_optimize(best_seq, t_alloc, eta0=0.04,
                                          active_frac=0.999)
        try_update(a_md, "mirror_descent")

    # === Phase 4: LogSumExp L-BFGS ===
    print(f"\n=== Phase 4: LogSumExp (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 15:
        a_lse = logsumexp_smooth_optimize(best_seq, t_alloc)
        try_update(a_lse, "logsumexp")

    # === Phase 5: Peak reduction ===
    print(f"\n=== Phase 5: Peak Reduction (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 10:
        a_pr, _ = peak_reduction_refine(best_seq, t_alloc)
        try_update(a_pr, "peak_reduction")

    # === Phase 6: Coordinate Descent ===
    print(f"\n=== Phase 6: Coord Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 120.0)
    if t_alloc > 10:
        a_cd, _ = coordinate_descent_refine(best_seq, t_alloc, n_steps=100000)
        try_update(a_cd, "coord_descent")

    # === Phase 7: Final LP polish ===
    print(f"\n=== Phase 7: Final LP Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 100.0)
    if t_alloc > 10:
        a_fin, _ = nesterov_lp_optimize(best_seq, t_alloc, solver=solver,
                                        eta0=0.05, patience=100)
        try_update(a_fin, "final_lp")

    # === Phase 8: Try n perturbations ===
    print(f"\n=== Phase 8: N Perturbations (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        n_cur = len(best_seq)
        for dn in [-50, 50, -100, 100]:
            if time.monotonic() > deadline - 15:
                break
            n_try = n_cur + dn
            if n_try < 400:
                continue
            a_try = interpolate_sequence(best_seq, n_try)
            t_alloc = min(rem * 0.3, 40.0)
            a_res, _ = nesterov_lp_optimize(a_try, t_alloc, solver=solver)
            try_update(a_res, f"n_perturb_{dn:+d}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
