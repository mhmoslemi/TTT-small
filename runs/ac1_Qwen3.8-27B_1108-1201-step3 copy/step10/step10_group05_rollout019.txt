import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def convolve_ab(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

Now let me think about the multiplicative pressure update more carefully:

Given a (normalized, sum=1):
1. Compute b = a*a
2. M = max(b)
3. Active set S = {k : b[k] >= M - ε} where ε is small (e.g., 0.01*M)
4. Compute pressure: p[i] = sum_{k in S, k-i valid} a[k-i]
   Using FFT: p = correlate(S_indicator, a)[0:n]
5. At KKT: p[i] should be constant (= λ|S|) for all i with a[i] > 0
6. Update: a_new[i] = a[i] * (mean(p) / p[i])^η
   (This reduces high-pressure coordinates and increases low-pressure ones)
7. Renormalize: a_new = a_new / sum(a_new)

The step size η controls aggressiveness. Small η = slow but stable; large η = fast but may oscillate.

Then pressure: p[i] = sum_k w[k] * a[k-i] (weighted)

This gives a smooth approximation to the hard active set.

The "duality gap" is: M - sum_k y[k] * (a*a)[k] = sum_k y[k] * (M - (a*a)[k]) >= 0

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def convolve_ab(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)

def multiplicative_pressure_update(a, eta=0.3, soft_beta=50.0, hard_frac=0.95):
    """
    Multiplicative weight update based on KKT pressure structure.
    At optimum: sum_{k active} a[k-i] = const for all i in support.
    We use soft (log-sum-exp) or hard active sets.
    """
    n = len(a)
    b = fast_convolve(a)
    M = b.max()
    
    if soft_beta > 0:
        # Soft active set via log-sum-exp
        scaled = soft_beta * (b - M)
        scaled = np.clip(scaled, -50, 0)
        w = np.exp(scaled)
        w = w / w.sum()
    else:
        # Hard active set
        threshold = M * hard_frac
        w = (b >= threshold).astype(float)
        if w.sum() < 1e-10:
            w = np.zeros_like(b)
            w[b.argmax()] = 1.0
        else:
            w = w / w.sum()
    
    # Compute pressure: p[i] = sum_k w[k] * a[k-i] for valid k-i
    # p[i] = sum_{j=0}^{n-1} w[i+j] * a[j]
    # This is cross-correlation of w and a
    # Using FFT:
    fft_size = 1 << (3 * n).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    # Cross-correlation: correlate(w, a)[k] = sum_j w[j+k] * a[j]
    # This equals ifft(conj(A) * W_shifted) ... let me just use np approach
    # Actually: sum_j w[i+j] * a[j] for j in [0,n-1]
    # = (w * a_reversed)[i + n - 1] in full convolution mode
    # Let's use: conv_full = convolve(w, a[::-1], 'full')
    # Then p[i] = conv_full[i + n - 1] ... 
    # Actually let me just compute it directly with FFT
    # p[i] = sum_{j=0}^{n-1} w[i+j] * a[j]
    # Define w_ext of length 2n-1 (already is)
    # p[i] = <w[i:i+n], a>
    # This is the "valid" part of correlation
    # Use: p = np.correlate(w, a, 'full')[n-1:2*n-1]
    
    # Faster with FFT:
    # p[i] = sum_j w[i+j] * a[j]
    # Let's compute via: reverse a, convolve with w
    a_rev = a[::-1]
    fft_sz = 1 << (2 * n).bit_length()
    W_ext = rfft(w, fft_sz)
    Ar = rfft(a_rev, fft_sz)
    conv_full = irfft(W_ext * Ar, fft_sz)[:3 * n - 1]
    # conv_full[k] = sum_j w[j] * a_rev[k-j] = sum_j w[j] * a[n-1-(k-j)]
    # = sum_j w[j] * a[n-1-k+j]
    # We want p[i] = sum_j w[i+j] * a[j]
    # Let m = i+j, so j = m-i: p[i] = sum_m w[m] * a[m-i] for m in [i, i+n-1]
    # = sum_m w[m] * a[m-i]
    # conv_full[k] = sum_j w[j] * a[n-1-k+j]
    # Set n-1-k+j = m-i => j = m-i-n+1+k
    # This is getting confusing. Let me just use np.correlate for clarity (it's fast enough for n<=4096)
    
    p = np.correlate(w, a, mode='full')
    # np.correlate(w, a, 'full')[k] = sum_n w[n+k] * conj(a[n]) for valid n
    # We want p[i] = sum_j w[i+j] * a[j] for j in [0, n-1]
    # This is correlate(w, a, 'full')[i] for i in [0, n-1] (with appropriate offset)
    # np.correlate(w, a, 'full') has length len(w)+len(a)-1 = (2n-1)+n-1 = 3n-2
    # The "zero-lag" is at index len(a)-1 = n-1
    # correlate(w, a, 'full')[n-1+i] = sum_j w[j+i] * a[j] for valid j
    # So p[i] = correlate(w, a, 'full')[n-1+i]
    
    n_full = len(p)
    if n_full >= 2 * n - 1:
        p = p[n - 1 : 2 * n - 1]
    else:
        # Fallback: compute directly
        p = np.zeros(n)
        for i in range(n):
            idx = np.arange(i, min(i + n, len(w)))
            j_idx = idx - i
            p[i] = (w[idx] * a[j_idx]).sum()
    
    # Ensure p is positive
    p = np.maximum(p, 1e-15)
    
    # Target: mean of p over support of a
    support = a > 1e-10
    if support.sum() > 0:
        target = p[support].mean()
    else:
        target = p.mean()
    
    # Multiplicative update
    ratio = target / p
    a_new = a * np.power(ratio, eta)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def lp_fictitious_play_exp(a0, time_limit, solver=None, label="", alpha_init=0.5, alpha_decay=0.9995, alpha_min=0.01):

            has_quad = (2 * i >= k_start) and (2 * i <= k_end)
            quad_idx = 2 * i - k_start if has_quad else -1
            
            candidates = [0.0, -ai]
            
            # Where individual lines hit unaff_max
            for j in range(len(affected)):
                c = coeffs[j]
                if c > 1e-15:
                    d = (unaff_max - affected[j]) / c
                    if -ai * 1.5 < d < ai * 1.5:
                        candidates.append(d)
            
            # Where top affected lines cross
            top_k = min(5, len(affected))
            top_idx = np.argsort(affected)[-top_k:]
            for j1 in range(len(top_idx)):
                for j2 in range(j1 + 1, len(top_idx)):
                    i1, i2 = top_idx[j1], top_idx[j2]
                    denom = coeffs[i1] - coeffs[i2]
                    if abs(denom) > 1e-15:
                        d = (affected[i2] - affected[i1]) / denom
                        if -ai * 1.5 < d < ai * 1.5:
                            candidates.append(d)
            
            # Fractional moves
            for frac in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
                candidates.append(-frac * ai)
                candidates.append(frac * 0.5 * ai)
            
            best_delta = 0.0
            best_new_max = M
            
            for delta in candidates:
                if delta < -ai - 1e-12:
                    continue
                delta = max(delta, -ai)
                if abs(delta) < 1e-18:
                    continue
                
                new_affected = affected + delta * coeffs
                if has_quad:
                    new_affected = new_affected.copy()
                    new_affected[quad_idx] += delta * delta
                
                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-15:
                    best_new_max = new_max
                    best_delta = delta
            
            if best_delta != 0.0 and best_new_max < M - 1e-14:
                a[i] = max(ai + best_delta, 0.0)
                conv[k_start:k_end + 1] += best_delta * coeffs
                if has_quad:
                    conv[2 * i] += best_delta * best_delta
                M = best_new_max
                
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        
        if sweeps % 50 == 0:
            conv = fast_convolve(a)
            M = float(conv.max())
        
        if not improved:
            no_improve += 1
            if no_improve > 200:
                break
        else:
            no_improve = 0
        
        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score

def fine_coord_descent(a0, time_limit, label=""):
    """Very fine coordinate descent for final polishing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    
    factors = [0.9999, 0.9998, 0.9995, 0.999, 0.998, 0.997, 0.995, 0.993, 0.99,
               0.985, 0.98, 1.001, 1.002, 1.003, 1.005, 1.007, 1.01, 1.015, 1.02, 1.03]
    
    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        
        for i in range(n):
            ai = a[i]
            if ai < 1e-15:
                continue
            
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            
            if k_start > 0:
                unaff_max = float(conv[:k_start].max())
            else:
                unaff_max = -1e30
            if k_end < 2 * n - 2:
                ur = float(conv[k_end + 1:].max())
                unaff_max = max(unaff_max, ur)
            
            affected = conv[k_start:k_end + 1]
            coeffs = a[(k_start - i):(k_end - i + 1)] + ai
            has_quad = (2 * i >= k_start) and (2 * i <= k_end)
            quad_idx = 2 * i - k_start if has_quad else -1
            
            best_delta = 0.0
            best_new_max = M
            
            for factor in factors:
                delta = ai * (factor - 1.0)
                if delta < -ai:
                    continue
                new_affected = affected + delta * coeffs
                if has_quad:
                    new_affected = new_affected.copy()
                    new_affected[quad_idx] += delta * delta
                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max - 1e-16:
                    best_new_max = new_max
                    best_delta = delta
            
            if best_delta != 0.0 and best_new_max < M - 1e-15:
                a[i] = max(ai + best_delta, 0.0)
                conv[k_start:k_end + 1] += best_delta * coeffs
                if has_quad:
                    conv[2 * i] += best_delta * best_delta
                M = best_new_max
                cur_score = 2.0 * n * M
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        
        if not improved and sweeps > 20:
            break
    
    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def check_and_update(a, best_seq, best_score, label=""):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score

def lp_fp_with_pipe(a0, time_limit, solver=None, label=""):
    """
    Combined LP-FP that alternates between exponential-weighted and true-average modes,
    with occasional multiplicative pressure updates to escape plateaus.
    """
    n = len(a0)
    a = np.maximum(a0, 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    a_bar_exp = a.copy()
    a_sum = np.zeros(n)
    it = 0
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    alpha = 0.4
    alpha_decay = 0.999
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Alternate: every 10th iteration use true average, otherwise exponential
        if it % 10 == 0 and it > 10:
            a_bar = a_sum / (it // 10)
        else:
            a_bar = a_bar_exp
        
        b, M_b = solve_lp_best_response(a_bar, solver)
        if b is None:
            alpha *= 0.9
            continue
        
        # Update exponential average
        a_bar_exp = (1 - alpha) * a_bar_exp + alpha * b
        a_bar_exp = np.maximum(a_bar_exp, 0)
        ss = a_bar_exp.sum()
        if ss > 1e-15:
            a_bar_exp = a_bar_exp / ss
        
        # Update true sum (every 10th)
        if it % 10 == 0:
            a_sum += b
            a = b
        else:
            a = a_bar_exp
        
        # Occasional pressure update
        if it % 50 == 45:
            a_pressure = multiplicative_pressure_update(a_bar_exp, eta=0.2, soft_beta=50.0)
            for cand in [a_pressure]:
                cand = np.maximum(cand, 0)
                sc = cand.sum()
                if sc < 1e-15:
                    continue
                cand = cand / sc
                score = 2.0 * n * fast_convolve(cand).max()
                if score < best_score - 1e-14:
                    best_score = score
                    best_a = cand.copy()
        
        # Track best
        for cand in [a_bar_exp, b]:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-15:
                continue
            cand = cand / sc
            score = 2.0 * n * fast_convolve(cand).max()
            if score < best_score - 1e-14:
                best_score = score
                best_a = cand.copy()
        
        alpha = max(0.01, alpha * alpha_decay)
        
        if it % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LPFP-pipe {label} it={it} ({elapsed:.0f}s): best={best_score:.6f}, alpha={alpha:.4f}")
    
    return best_a, best_score
