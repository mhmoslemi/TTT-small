**Key insight for improvement:**

The LP best-response gives $b^*(a)$ = argmin $M$ s.t. $\text{conv}(a,b) \leq M, \sum b = 1, b \geq 0$.

At a fixed point $a = b^*(a)$, we have that $a$ is optimal against itself in the LP sense. But this doesn't necessarily mean $a*a$ is minimized.

A better approach might be to use a ** two-sided LP** or a **game-theoretic** formulation. Actually, what we want is: $\min_p \max_k (p*p)_k$ subject to $p \geq 0, \sum p = 1$.

This is a single-player problem. The LP approach works by alternating: given $p$, find $q$ that minimizes $\max_k (p*q)_k$ (which is the LP), then update $p$ towards $q$. This is like finding a fixed point of the best-response map.

                for idx_k, k in enumerate(S):

                    if 0 <= kj < n:

                F2[idx_j] = val - mu

            F3 = np.zeros(r)
            for idx_k, k in enumerate(S):
                val = 0.0

                        val += a[i] * a[k_int - i]

            F4 = a_T.sum() - 1.0

            # Block (1,2): F1 wrt lam = ones (1 x r)
            # Block (1,3): F1 wrt mu = 0
            # Block (1,4): F1 wrt M = 0
            
            # Block (2,1): F2 wrt a_T (m x m): [j,i] = 2*lam[j+i-T] if j+T[i] in S
            J21 = np.zeros((m, m))

                    if k_val in S:

            J[1:1+m, m+r] = -1.0

            # Row m+r+1: F4
            J[m+r+1, 0:m] = 1.0

            mu_new = mu + dx[m+r]

            step = 1.0
            for _ in range(20):

                lam_trial = lam + step * dx[m:m+r]
                if np.all(a_trial_T > -1e-15) and np.all(lam_trial > -1e-15):
                    break
                step *= 0.5
            
            a[T] = np.maximum(a[T] + step * dx[0:m], 0.0)
            # Renormalize
            s = a.sum()
            if s > 0:
                a = a / s
            lam = np.maximum(lam + step * dx[m:m+r], 0.0)
            lam = lam / lam.sum()
            mu = mu + step * dx[m+r]
            M_val = M_val + step * dx[m+r+1]
        
        # Update a and check improvement
        conv = fast_convolve(a)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP best-response alternating minimization for initial refinement
2. Active-set Newton method solving the KKT system of the constrained problem (key innovation)
3. Smooth log-sum-exp gradient descent as secondary optimizer
4. Fine coordinate descent polish for last-mile improvements

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def score_of(a):
    """Compute 2*n*max(conv)/(sum)^2 quickly."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

    """Find the fastest working LP solver."""
    cached = getattr(find_lp_solver, '_cached', None)
    if cached is not None:
        return cached
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                find_lp_solver._cached = solver_name
                return solver_name
        except Exception:
            continue
    find_lp_solver._cached = cp.ECOS
    return cp.ECOS


def build_conv_matrix(a):
    """Build convolution matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, solver):
    """Solve LP: min M s.t. conv(a,b) <= M, sum(b)=1, b>=0."""
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    ones = np.ones(2 * n - 1)
    constraints = [
        T @ b_var <= M_var * ones,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return a.copy(), float('inf')

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float('inf')

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return a.copy(), float('inf')
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else float('inf')
    return b_val, M_val


def lp_alternating(a0, time_limit, solver, verbose=False):
    """LP-based best-response alternating minimization."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()

    else:

    best_a = a.copy()
    conv = fast_convolve(a)

    t0 = time.monotonic()
    it = 0
    stagnation = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        b_new, M_lp = lp_best_response(a, solver)
        if not np.isfinite(M_lp):
            break

        score_b = 2.0 * n * fast_convolve(b_new).max()
        if score_b < best_score - 1e-13:
            best_score = score_b
            best_a = b_new.copy()

        if it <= 30:
            alpha = 1.0
        elif it <= 80:
            alpha = 0.5
        else:
            alpha = max(0.2, 8.0 / (it + 5.0))

        a_new = (1 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss > 1e-15:
            a_new = a_new / ss
        a = a_new

        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score - 1e-13:
            best_score = score_a
            best_a = a.copy()
