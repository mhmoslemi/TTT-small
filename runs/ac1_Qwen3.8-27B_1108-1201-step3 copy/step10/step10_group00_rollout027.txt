def simplex_proj(v, target_sum=1.0):
    n = len(v)
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def compute_smoothed_gradient(a, b, T_frac=0.01):

    log_w = (b - max_b) / T
    log_w -= log_w.max()
    w = np.exp(log_w)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(b)
        w[np.argmax(b)] = 1.0
    else:
        w /= w_sum
    # grad_i = 2 * sum_k w[k] * a[k-i] (where 0<=k-i<n)

def simplex_proj(v, target_sum=1.0):
    n = len(v)
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    if not np.any(cond):
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def compute_smoothed_gradient(a, b, T_frac=0.01):
    n = len(a)
    max_b = b.max()
    T = max_b * T_frac
    if T < 1e-12:
        T = 1e-12
    log_w = (b - max_b) / T
    log_w -= log_w.max()
    w = np.exp(log_w)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(b)
        w[np.argmax(b)] = 1.0
    else:
        w = w / w_sum
    grad = 2.0 * np.correlate(w, a, mode='valid')
    return grad

            best_score = sc
            best_a = a_ms2.copy()
            a = a_ms2.copy()

    # Phase D: Coordinate descent
    elapsed = time.monotonic() - start_time
    if time_budget - elapsed > 3:
        a_cd = coordinate_descent_pass(a, n_passes=2)
        sc = evaluate_local(a_cd)
        if sc < best_score:
            best_score = sc
            best_a = a_cd.copy()

    # Phase E: Final polish
    elapsed = time.monotonic() - start_time
    if time_budget - elapsed > 3:
        t5 = min(time_budget - elapsed, 15.0)
        a_fin, sc = multiplicative_subgradient(best_a, t5, n_steps=100000, T_frac_init=0.001, T_frac_min=0.0005)
        if sc < best_score:
            best_score = sc
            best_a = a_fin.copy()

    return best_a, best_score

def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Init: n={len(best_seq)}, score={best_score:.8f}")

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Optimize at parent length ===
    print(f"\n=== Phase 1: Optimize n={len(best_seq)} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 480.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_opt, sc = optimize_at_length(best_seq, t_alloc, t_start)
        local_update(a_opt, "phase1")

    # === Phase 2: Multi-length exploration ===
    n_cur = len(best_seq)
    target_lengths = sorted([800, 1000, 1400, 1500, 1800, 2000], key=lambda x: abs(x - n_cur))
    for target_n in target_lengths:
        rem = deadline - time.monotonic()
        if rem < 35:
            break
        if target_n == n_cur:
            continue
        t_alloc = min(rem * 0.18, 180.0)
        if t_alloc < 15:
            continue
        print(f"\n=== Trying n={target_n} (budget={t_alloc:.0f}s) ===")
        init_n = interpolate_sequence(best_seq, target_n)
        a_opt, sc = optimize_at_length(init_n, t_alloc, t_start)
        local_update(a_opt, f"n{target_n}")

    # === Phase 3: Random restarts ===
    print(f"\n=== Phase 3: Random Restarts (score={best_score:.8f}) ===")
    for trial in range(4):
        rem = deadline - time.monotonic()
        if rem < 25:
            break
        t_alloc = min(rem * 0.15, 100.0)
        if t_alloc < 10:
            break
        strength = 0.03 + 0.02 * trial
        a_pert = perturbed_sequence(best_seq, rng, strength=strength)
        a_opt, sc = multiplicative_subgradient(a_pert, t_alloc, n_steps=200000, T_frac_init=0.008, T_frac_min=0.002)
        local_update(a_opt, f"restart{trial}")
        rem = deadline - time.monotonic()
        if rem > 8:
            a_cd = coordinate_descent_pass(a_opt, n_passes=1)
            local_update(a_cd, f"restart{trial}_cd")

    # === Phase 4: Final refinement ===
    print(f"\n=== Phase 4: Final Refinement (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t_fin = min(rem * 0.35, 50.0)
        a_fin, sc = multiplicative_subgradient(best_seq, t_fin, n_steps=200000, T_frac_init=0.001, T_frac_min=0.0005)
        local_update(a_fin, "final_mult")

    rem = deadline - time.monotonic()
    if rem > 10:
        a_cd = coordinate_descent_pass(best_seq, n_passes=3)
        local_update(a_cd, "final_cd")

    rem = deadline - time.monotonic()
    if rem > 8:
        T_stages = [0.5, 0.2, 0.1, 0.05, 0.02]
        a_f, sc = fista_lse(best_seq, min(rem * 0.5, 20.0), T_stages, max_iter_per_stage=40)
        local_update(a_f, "final_fista")

    rem = deadline - time.monotonic()
    if rem > 5:
        a_p, sc = multiplicative_subgradient(best_seq, min(rem * 0.7, 15.0), n_steps=50000, T_frac_init=0.0005, T_frac_min=0.0005)
        local_update(a_p, "ultra_final")

    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
