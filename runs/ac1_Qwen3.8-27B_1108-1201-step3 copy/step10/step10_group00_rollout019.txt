"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated LogSumExp gradient with aggressive temperature annealing
2. Spectral-domain refinement (flatten power spectrum to reduce autocorrelation peaks)
3. Population-based adaptive search with diverse mutation strategies
4. Multi-length exploration with smooth interpolation
5. Block coordinate descent with analytic sub-solves
Key innovations: spectral shaping, block-wise optimization, and
adaptive population dynamics to escape local minima.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def lse_obj_grad(a, T):
    """Compute LogSumExp-smoothed objective and gradient w.r.t. a.
    Returns (obj, grad) where obj = 2n * LSE(c, T) and grad = d(obj)/da.
    Assumes a is a probability vector (sum=1, a>=0).
    """
    n = len(a)
    c = fast_convolve(a)  # length 2n-1
    c_max = c.max()
    exp_c = np.exp((c - c_max) / T)
    lse = c_max + T * np.log(exp_c.sum())
    obj = 2.0 * n * lse

    # Softmax weights
    w = exp_c / exp_c.sum()  # length 2n-1

    # Gradient: g_m = 2 * sum_k w[k]*a[k-m] - a_m * w[2m] (if 2m valid)
    # First term: cross-correlation of w and a
    N = 1 << (len(w) + n - 1).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    R = irfft(W * np.conj(A), N)
    b = R[:n]  # b[m] = sum_k w[k]*a[k-m] for m=0..n-1

    g = 2.0 * b
    # Subtract the diagonal correction term
    idx = 2 * np.arange(n)
    valid = idx < (2 * n - 1)
    g[valid] -= a[valid] * w[idx[valid]]

    return obj, g


def lse_obj_grad_z(z, n, T):
    """Compute LSE objective and gradient in log-parameterization.
    z are log-probabilities, a = softmax(z).
    """
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    a = exp_z / exp_z.sum()

    obj, grad_a = lse_obj_grad(a, T)

    # Transform gradient: grad_z = a * (grad_a - sum(grad_a * a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)

    return obj, grad_z


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def evaluate_local(a):
    """Evaluate score locally."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def normalize(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def spectral_refine(a, n_iters=500, learning_rate=0.01):
    """Refine sequence by flattening its power spectrum.
    The idea: make |FFT(a)|^2 as flat as possible to reduce autocorrelation peaks.
    """
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    fft_size = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        A = rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_mean = P.mean()

        # Target: flatten P. Gradient of (P - P_mean)^2 w.r.t. a
        # d(P)/d(a) is related to the phase of A
        # We want to minimize sum_k (P[k] - P_mean)^2
        # dP[k]/da = 2 * Re(A[k] * conj(dA[k]/da))
        # dA[k]/da = conj(FFT(e_k)) = conj(exp(-2pi i k/n))
        # This is complex, so let's use a simpler approach:
        # Perturb a in the direction that reduces P's variance

        # Compute gradient numerically via the relationship:
        # If we add eta * h to a, then P changes by ~ 2*Re(FFT(eta*h) * conj(A))
        # We want P_new to be closer to P_mean
        # So we want Re(FFT(h) * conj(A)) to have opposite sign to (P - P_mean)

        # Let r[k] = (P[k] - P_mean) / P_mean (relative deviation)
        r = (P - P_mean) / (P_mean + 1e-15)

        # We want h such that Re(FFT(h) * conj(A)) ~= -r * P_mean
        # In the frequency domain: FFT(h) ~= -r * P_mean * A / |A|^2 (where A != 0)
        # h = IFFT(-r * P_mean * A / |A|^2)
        # But we need h to be real, so:
        # FFT(h)[k] = -r[k] * P_mean * A[k] / (|A[k]|^2 + eps) for k > 0
        #           = -r[k] * P_mean * A[k] / (|A[k]|^2 + eps) * 2 (for conjugate symmetry)

        H = np.zeros(fft_size // 2 + 1, dtype=np.complex128)
        mask = P > 1e-10
        H[mask] = -r[mask] * P_mean * A[mask] / (P[mask] + 1e-10)
        # Enforce conjugate symmetry for real IFFT
        H[0] = float(H[0].real) if fft_size // 2 + 1 > 0 else 0
        h = irfft(H * 2, fft_size)[:n]  # factor 2 for the conjugate part

        # Ensure h is a valid direction (zero mean to preserve normalization)
        h -= h.mean()
        h_norm = np.linalg.norm(h)
        if h_norm < 1e-15:
            break
        h = h / h_norm

        # Try the perturbation
        for lr in [learning_rate, learning_rate * 0.1, learning_rate * 0.01]:
            a_new = a + lr * h
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                continue
            a_new = a_new / s
            sc = evaluate_local(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
            break

        if it % 100 == 0 and it > 0:
            learning_rate *= 0.7

    return best_a, best_score


def lbgfs_optimize(a0, time_limit, T_stages, maxiter_per_stage=2000):
    """L-BFGS optimization with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    for stage_T in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func(z_in):
            obj, _ = lse_obj_grad_z(z_in, n, stage_T)
            return obj

        def jac(z_in):
            _, g = lse_obj_grad_z(z_in, n, stage_T)
            return g

        iters = min(maxiter_per_stage, max(10, int(remaining * 30)))
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-18,
                         'gtol': 1e-16, 'maxfun': iters * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def multiplicative_weights(a0, time_limit, n_steps=50000, eta0=0.1):
    """Multiplicative weights (mirror descent) with adaptive step size."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    eta = eta0
    no_improve = 0

        if time.monotonic() - t0 > time_limit:

        c = fast_convolve(a)
        c_max = c.max()

        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([n - 1])

        # Compute subgradient (equal weight over active peaks)
        # g_m = (1/|S|) sum_{k in S} (2*a_{k-m} - a_m * delta_{2m,k})
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for m in range(js, je + 1):
                g[m] += a[k - m]
            if 2 * m <= k and 0 <= 2 * m <= 2 * n - 2:
                pass  # diagonal correction is higher order, skip for speed
        g /= len(active)

        # Multiplicative update with line search
        for scale in [1.0, 0.5, 0.25, 0.1]:
            log_alpha = -eta * scale * g
            log_alpha -= log_alpha.max()
            a_new = a * np.exp(log_alpha)
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                continue
            a_new = a_new / s
            sc = evaluate_local(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
                no_improve = 0
            break

        no_improve += 1
        if no_improve >= 20:
            eta *= 0.7
            no_improve = 0
            if eta < 1e-6:
                break

    return best_a, best_score


def block_coordinate_descent(a0, time_limit, block_size=20, n_rounds=1000):
    """Block coordinate descent: optimize one block at a time."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        # Pick a random block
        start = rng.integers(0, n - block_size)
        end = start + block_size

        # Optimize the block via line search
        a_block = a[start:end].copy()
        for it in range(50):
            # Compute gradient for the block
            c = fast_convolve(a)
            c_max = c.max()
            T = 0.01
            exp_c = np.exp((c - c_max) / T)
            w = exp_c / exp_c.sum()

            # Gradient for block coordinates
            g_block = np.zeros(block_size)
            for m in range(start, end):
                s = 0.0
                for k in range(m, min(m + n, 2 * n - 1)):
                    s += w[k] * a[k - m]
                g_block[m - start] = 2.0 * s
                if 2 * m <= 2 * n - 2:
                    g_block[m - start] -= a[m] * w[2 * m]

            # Gradient descent step on the block
            lr = 0.01 / (np.max(np.abs(g_block)) + 1e-10)
            a_new_block = a_block - lr * g_block
            a_new_block = np.maximum(a_new_block, 0)

            a_new = a.copy()
            a_new[start:end] = a_new_block
            a_new = normalize(a_new)
            sc = evaluate_local(a_new)

            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
                a_block = a[start:end].copy()
            else:
                lr *= 0.5
                if lr < 1e-8:
                    break

    return best_a, best_score


def diverse_init(n, rng):
    """Generate diverse initial sequences."""
    candidates = []
    x = np.linspace(0, 1, n)

    # Gaussian bump
    a = np.exp(-8 * (x - 0.5) ** 2)
    candidates.append(normalize(a))

    # Raised cosine
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    candidates.append(normalize(a))

    # Sine squared
    a = np.sin(np.pi * x) ** 2
    candidates.append(normalize(a))

    # Power law
    a = (x * (1 - x)) ** 0.5
    candidates.append(normalize(a))

    # Two bumps
    a = np.exp(-10 * (x - 0.3) ** 2) + np.exp(-10 * (x - 0.7) ** 2)
    candidates.append(normalize(a))

    # Cosine with modulation
    a = 0.5 + 0.4 * np.cos(2 * np.pi * 2 * x) + 0.1 * np.cos(2 * np.pi * 5 * x)
    a = np.maximum(a, 0)
    candidates.append(normalize(a))

    # Random smooth
    coeffs = rng.normal(0, 1, 10)
    a = np.zeros(n)
    for i, c in enumerate(coeffs):
        a += c * np.cos(2 * np.pi * (i + 1) * x + rng.uniform(0, 2 * np.pi))
    a = np.maximum(a + 0.5, 0)
    candidates.append(normalize(a))

    # Uniform
    candidates.append(np.ones(n) / n)

    # Edge-peaked
    a = np.exp(-5 * x) + np.exp(-5 * (1 - x))
    candidates.append(normalize(a))

    return candidates


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    return normalize(f_interp)


def perturb_sequence(a, rng, method='gaussian', strength=0.02):
    """Perturb a sequence using various methods."""
    n = len(a)
    if method == 'gaussian':
        noise = rng.normal(0, strength, n)
        a_new = a + noise
    elif method == 'low_freq':
        freq = rng.integers(1, 8)
        phase = rng.uniform(0, 2 * np.pi)
        x = np.linspace(0, 2 * np.pi, n)
        noise = strength * np.cos(freq * x + phase)
        a_new = a + noise
    elif method == 'block_swap':
        a_new = a.copy()
        block_size = rng.integers(10, 50)
        i = rng.integers(0, n - block_size)
        j = rng.integers(0, n - block_size)
        if abs(i - j) > block_size:
            a_new[i:i+block_size], a_new[j:j+block_size] = a[j:j+block_size].copy(), a[i:i+block_size].copy()
    elif method == 'edge_push':
        a_new = a.copy()
        edge = rng.integers(0, 2)
        w = rng.uniform(0.01, 0.05)
        if edge == 0:
            a_new[:20] += w
            a_new[20:] -= w * 20 / (n - 20)
        else:
            a_new[-20:] += w
            a_new[:-20] -= w * 20 / (n - 20)
    else:
        a_new = a + rng.normal(0, strength, n)

    return normalize(a_new)


def population_search(a_init, time_limit, pop_size=8, rng=None):
    """Population-based search with adaptive mutations."""
    if rng is None:
        rng = np.random.default_rng(456)

    n = len(a_init)
    population = [normalize(a_init.copy())]
    # Add diverse perturbations
    methods = ['gaussian', 'low_freq', 'block_swap', 'edge_push']
    for i in range(pop_size - 1):
        m = methods[i % len(methods)]
        strength = 0.02 + 0.01 * (i % 3)
        population.append(perturb_sequence(a_init, rng, m, strength))

    scores = [evaluate_local(p) for p in population]
    best_idx = np.argmin(scores)
    best_a = population[best_idx].copy()
    best_score = scores[best_idx]

    t0 = time.monotonic()
    gen = 0
    adaptive_strength = 0.02

    while time.monotonic() - t0 < time_limit:
        gen += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        # Sort population by score
        order = np.argsort(scores)
        population = [population[i] for i in order]
        scores = [scores[i] for i in order]

        # Update best
        if scores[0] < best_score:
            best_score = scores[0]
            best_a = population[0].copy()

        # Selection: keep top half
        keep = max(2, pop_size // 2)
        elites = population[:keep]

        # Generate new population
        new_pop = list(elites)

        # Mutations
        for i in range(pop_size - len(new_pop)):
            parent = elites[rng.integers(0, keep)]
            method = methods[rng.integers(0, len(methods))]
            strength = adaptive_strength * (1 + 0.5 * np.sin(gen * 0.1))
            child = perturb_sequence(parent, rng, method, strength)
            new_pop.append(child)

        # Crossover
        if pop_size > len(new_pop) and len(elites) >= 2:
            p1, p2 = elites[0], elites[1]
            t = rng.uniform(0.2, 0.8)
            child = normalize(t * p1 + (1 - t) * p2)
            new_pop.append(child)

        population = new_pop[:pop_size]
        scores = [evaluate_local(p) for p in population]

        # Adaptive: reduce strength if stuck
        if gen % 50 == 0:
            if scores[0] >= best_score - 1e-10:
                adaptive_strength *= 0.8
            else:
                adaptive_strength = min(0.05, adaptive_strength * 1.2)

        if gen % 100 == 0:
            print(f"    PopSearch gen={gen}: best={best_score:.8f} strength={adaptive_strength:.4f}")

    return best_a, best_score


def spectral_shape_optimize(a0, time_limit, n_iters=2000):
    """Optimize by directly shaping the power spectrum to be flat.
    Uses the fact that flat PSD => flat autocorrelation (in the limit).
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_local(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    t0 = time.monotonic()

    lr = 0.005
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        A = rfft(a, fft_size)
        P = np.abs(A) ** 2
        P_target = P.mean() * np.ones_like(P)

        # We want to move P toward P_target
        # Perturbation direction in frequency domain: -alpha * (P - P_target)
        # In time domain: h = IFFT( dP/da * conj(grad_P) )
        # Simplified: h = IFFT( -(P - P_target) * conj(A) / (P + eps) )
        H = np.zeros(fft_size // 2 + 1, dtype=np.complex128)
        valid = P > 1e-12
        H[valid] = -(P[valid] - P_target[valid]) * np.conj(A[valid]) / (P[valid] + 1e-10)
        H[0] = 0.0  # Don't change the DC component
        h = irfft(H, fft_size)[:n]
        h -= h.mean()
        h_norm = np.linalg.norm(h)
        if h_norm < 1e-15:
            break
        h = h / h_norm

        # Line search
        for scale in [lr, lr * 0.3, lr * 0.1]:
            a_new = a + scale * h
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                continue
            a_new = a_new / s
            sc = evaluate_local(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
            break

        if it % 200 == 0:
            lr *= 0.85

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses spectral shaping, population search, L-BFGS with LSE,
    and block coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)

    best_seq = None
    best_score = float('inf')

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        candidates = diverse_init(n_init, rng)
        scores = [evaluate_local(c) for c in candidates]
        best_idx = np.argmin(scores)
        best_seq = candidates[best_idx].copy()
        best_score = scores[best_idx]
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)
    print(f"Starting optimization: n={n_best}, score={best_score:.8f}")

    # === Phase 1: Spectral Shaping (novel) ===
    print(f"\n=== Phase 1: Spectral Shaping (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_spec, _ = spectral_shape_optimize(best_seq, t_alloc, n_iters=3000)
        local_update(a_spec, "spectral_1")

    # === Phase 2: L-BFGS with aggressive temperature annealing ===
    print(f"\n=== Phase 2: L-BFGS LSE (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 30:
        T_stages = [50, 20, 10, 5, 3, 2, 1, 0.7, 0.5, 0.3, 0.2, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.005, 0.002]
        print(f"  T_stages ({len(T_stages)} stages), budget={t_alloc:.0f}s")
        a_lb = lbgfs_optimize(best_seq, t_alloc, T_stages, maxiter_per_stage=3000)
        local_update(a_lb, "lbgfs_1")

    # === Phase 3: Population search ===
    print(f"\n=== Phase 3: Population Search (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_pop, _ = population_search(best_seq, t_alloc, pop_size=10, rng=rng)
        local_update(a_pop, "population")

    # === Phase 4: Block coordinate descent ===
    print(f"\n=== Phase 4: Block Coordinate Descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_bcd, _ = block_coordinate_descent(best_seq, t_alloc, block_size=15, n_rounds=5000)
        local_update(a_bcd, "block_cd")

    # === Phase 5: Spectral refinement (second round) ===
    print(f"\n=== Phase 5: Spectral Refinement 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 15:
        a_spec2, _ = spectral_shape_optimize(best_seq, t_alloc, n_iters=5000)
        local_update(a_spec2, "spectral_2")

    # === Phase 6: Fine L-BFGS polish ===
    print(f"\n=== Phase 6: Fine L-BFGS Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 20:
        T_stages2 = [2, 1, 0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001]
        a_lb2 = lbgfs_optimize(best_seq, t_alloc, T_stages2, maxiter_per_stage=2000)
        local_update(a_lb2, "lbgfs_fine")

    # === Phase 7: Try different lengths ===
    print(f"\n=== Phase 7: Length Exploration (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        for n_try in [n_best + 100, n_best - 100, n_best + 200]:
            if n_try < 100 or n_try > 5000:
                continue
            rem_sub = deadline - time.monotonic()
            if rem_sub < 30:
                break
            init = interpolate_sequence(best_seq, n_try)
            t_sub = min(rem_sub * 0.3, 60.0)
            T_short = [5, 2, 1, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
            a_try = lbgfs_optimize(init, t_sub, T_short, maxiter_per_stage=1000)
            local_update(a_try, f"length_{n_try}")

    # === Phase 8: Final multiplicative weights polish ===
    print(f"\n=== Phase 8: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        a_mw, _ = multiplicative_weights(best_seq, min(rem * 0.3, 30.0), n_steps=20000, eta0=0.05)
        local_update(a_mw, "final_mw")

    rem = deadline - time.monotonic()
    if rem > 10:
        T_final = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
        a_fin = lbgfs_optimize(best_seq, min(rem * 0.5, 30.0), T_final, maxiter_per_stage=1000)
        local_update(a_fin, "final_lbgfs")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
