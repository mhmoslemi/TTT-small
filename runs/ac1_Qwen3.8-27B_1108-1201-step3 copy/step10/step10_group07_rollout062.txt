import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time
import cvxpy as cp


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_of(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def simplex_project(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def softmax(z):

import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def score_of(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))


def simplex_project(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, solver):
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    ones = np.ones(2 * n - 1)
    constraints = [
        T @ b_var <= M_var * ones,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return a.copy(), float('inf')

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float('inf')

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return a.copy(), float('inf')
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else float('inf')
    return b_val, M_val

def adaptive_coord_descent(a0, time_limit, verbose=True, label=""):

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-16:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

                unaff_left = conv[:k_start].max()
            else:

            if k_end < 2 * n - 2:

            unaff_max = max(unaff_left, unaff_right)

            a_shifted = a[k_start - i:k_end - i + 1]

                new_val = a[i] + delta
                if new_val < -1e-15:

                new_affected = conv_affected + 2.0 * delta * a_shifted

                    new_affected_copy = new_affected.copy()

def multi_peak_shaving(a0, time_limit, verbose=True):
    """
    Focus on reducing the top peaks simultaneously.
    For each top peak, identify the most contributing pairs and
    try to reduce them while minimizing damage to other peaks.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    rounds = 0

    while time.monotonic() - t0 < time_limit:
        rounds += 1
        improved = False

        K = 8
        peak_indices = np.argpartition(conv, -K)[-K:]
        peak_indices = peak_indices[np.argsort(conv[peak_indices])[::-1]]

        for k_star in peak_indices:
            k_star = int(k_star)
            if conv[k_star] < M * 0.92:
                continue

            i_start = max(0, k_star - n + 1)
            i_end = min(n - 1, k_star)

            contributions = np.zeros(i_end - i_start + 1)
            for idx, i in enumerate(range(i_start, i_end + 1)):
                j = k_star - i
                contributions[idx] = a[i] * a[j]

            top_contrib = np.argsort(contributions)[::-1][:5]

            for tc in top_contrib:
                i = i_start + int(tc)
                j = k_star - i

                # Try reducing the larger of a[i], a[j]
                target = i if a[i] >= a[j] else j

                for frac in [0.05, 0.1, 0.15, 0.2, 0.3]:
                    delta = -a[target] * frac
                    if abs(delta) < 1e-15:
                        continue

                    k_s = target
                    k_e = min(target + n - 1, 2 * n - 2)
                    if k_s > 0:
                        unaff_l = conv[:k_s].max()
                    else:
                        unaff_l = -1e30
                    if k_e < 2 * n - 2:
                        unaff_r = conv[k_e + 1:].max()
                    else:
                        unaff_r = -1e30
                    unaff_max = max(unaff_l, unaff_r)

                    a_shifted = a[k_s - target:k_e - target + 1]
                    conv_aff = conv[k_s:k_e + 1]
                    new_aff = conv_aff + 2.0 * delta * a_shifted
                    if 2 * target >= k_s and 2 * target <= k_e:
                        idx2 = 2 * target - k_s
                        new_aff = new_aff.copy()
                        new_aff[idx2] = conv[2 * target] + 2.0 * delta * a[target] + delta * delta

                    new_M = max(unaff_max, float(new_aff.max()))
                    if new_M < M - 1e-12:
                        a[target] = max(a[target] + delta, 0.0)
                        conv[k_s:k_e + 1] = new_aff
                        M = new_M
                        improved = True
                        cur_score = 2.0 * n * M
                        if cur_score < best_score:
                            best_score = cur_score
                            best_a = a.copy()
                        break

        if not improved and rounds > 80:
            break

        if verbose and rounds % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MultiPeak round={rounds} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def diverse_init(n, rng, init_type=0):
    """Generate diverse initializations."""
    if init_type == 0:
        t = np.arange(n)
        center = (n - 1) / 2.0
        a = np.maximum(1.0 - np.abs(t - center) / center, 0)
    elif init_type == 1:
        t = np.arange(n)
        center = (n - 1) / 2.0
        x = (t - center) / center
        a = 0.3 + 0.7 * x * x
    elif init_type == 2:
        t = np.arange(n)
        center = (n - 1) / 2.0
        x = (t - center) / center
        power = rng.uniform(1.0, 3.0)
        a = np.maximum(0.5 * (1 + np.cos(np.pi * x)), 0) ** power
    elif init_type == 3:
        t = np.arange(n)
        c1 = n * rng.uniform(0.2, 0.35)
        c2 = n * rng.uniform(0.65, 0.8)
        w = n * rng.uniform(0.08, 0.2)
        a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    elif init_type == 4:
        period = int(rng.integers(3, 10))
        t = np.arange(n)
        a = 0.5 + 0.4 * np.cos(2 * np.pi * t / period)
    elif init_type == 5:
        t = np.arange(n)
        a = 1.0 - 0.5 * np.cos(4 * np.pi * t / n)
        a = np.maximum(a, 0.01)
    elif init_type == 6:
        # Sine with random phase
        t = np.arange(n)
        freq = rng.uniform(1.0, 5.0)
        phase = rng.uniform(0, 2 * np.pi)
        a = 0.5 + 0.4 * np.sin(2 * np.pi * freq * t / n + phase)
        a = np.maximum(a, 0)
    elif init_type == 7:
        # Two-frequency combination
        t = np.arange(n)
        f1 = rng.uniform(2.0, 6.0)
        f2 = rng.uniform(8.0, 20.0)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * f1 * t / n) + 0.2 * np.cos(2 * np.pi * f2 * t / n)
        a = np.maximum(a, 0)
    else:
        a = np.ones(n) / n

    a = np.maximum(a, 0)
    ss = a.sum()
    if ss < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / ss
    return a


def basin_hopping(a0, time_limit, solver, verbose=True):
    """
    Basin hopping: randomly perturb the sequence, then re-optimize
    using a quick LP + CD cycle. Accept if improved.
    """
    n = len(a0)
    best_a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = best_a.sum()
    if s < 1e-15:
        best_a = np.ones(n) / n
    else:
        best_a = best_a / best_a.sum()
    best_score = 2.0 * n * fast_convolve(best_a).max()
    t0 = time.monotonic()
    trial = 0

    while time.monotonic() - t0 < time_limit:
        trial += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 30:
            break

        # Random perturbation
        scale = 0.02 / np.sqrt(n) * max(1.0, 1.0 - 0.01 * trial)
        rng_local = np.random.default_rng(42 + trial * 137)
        pert = best_a + rng_local.standard_normal(n) * scale
        pert = np.maximum(pert, 0)
        ps = pert.sum()
        if ps < 1e-15:
            continue
        pert = pert / ps

        # Quick LP (just 2-3 iterations)
        a_temp = pert.copy()
        for _ in range(3):
            b_new, _ = lp_best_response(a_temp, solver)
            a_temp = 0.5 * a_temp + 0.5 * b_new
            a_temp = np.maximum(a_temp, 0)
            ss = a_temp.sum()
            if ss > 1e-15:
                a_temp = a_temp / ss

        # Quick coordinate descent (1-2 sweeps)
        a_temp, s_temp = adaptive_coord_descent(a_temp, min(remaining * 0.05, 15), verbose=False, label="bh")

        if s_temp < best_score:
            best_score = s_temp
            best_a = a_temp.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"    BH trial={trial} ({elapsed:.0f}s): NEW BEST={best_score:.8f}")

        if verbose and trial % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    BH trial={trial} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def logsumexp_optimization(a0, time_limit, verbose=True):
    """
    Optimize using log-sum-exp smooth approximation of the max.
    Uses projected gradient with adaptive temperature.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    T = 0.01  # temperature for log-sum-exp
    lr = 0.005
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = float(conv.max())
        score = 2.0 * n * M

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 80:
            T = max(0.001, T * 0.9)
            lr = max(1e-9, lr * 0.9)
            no_improve = 0
            if lr < 1e-13 and T < 0.002:
                break

        # Compute log-sum-exp gradient
        max_c = M
        exp_vals = np.exp((conv - max_c) / T)
        total = exp_vals.sum()
        if total < 1e-300:
            weights = np.ones(len(conv)) / len(conv)
        else:
            weights = exp_vals / total

        # Gradient: grad_a[k] = 2 * sum_j weights[j] * a[j-k] (for valid j-k)
        # Use FFT: cross-correlation of weights and a
        fft_size = 1 << (2 * n - 1).bit_length()
        W = rfft(weights, fft_size)
        A = rfft(a, fft_size)
        # Cross-correlation: sum_m weights[m+k]*a[m] = irfft(W * conj(A))
        grad_a = 2.0 * irfft(W * np.conj(A), fft_size)[:n]

        # Projected gradient step
        a_new = a - lr * grad_a
        a_new = simplex_project(a_new)
        a = a_new

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE it={it} ({elapsed:.0f}s): best={best_score:.8f} T={T:.5f}")

    return best_a, best_score


def propose_candidate():
    """
    Multi-phase optimization combining:
    - L2 norm optimization (smooth, fast)
    - Log-sum-exp optimization (intermediate smoothness)
    - LP alternating minimization (precise)
    - Basin hopping (escape local minima)
    - Coordinate descent (local refinement)
    - Fine polishing (last mile)
    """
    deadline = time.monotonic() + 1770.0
    rng = np.random.default_rng(12345)

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is not None:
        n = len(parent)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    else:
        n = 1024
        best_seq = np.ones(n) / n
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Using default init: n={n}, score={best_score:.8f}")

    solver = find_lp_solver()
    print(f"LP solver: {solver}")

    def try_update(candidate, label):
        nonlocal best_seq, best_score
        candidate = np.maximum(candidate, 0.0)
        s = candidate.sum()
        if s < 0.01:
            return
        candidate = candidate / s
        try:
            sc = float(evaluate_sequence(candidate.tolist()))
        except Exception:
            sc = float('inf')
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = candidate.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(candidate)})")

    # === Phase 1: L2 Optimization (fast, smooth) ===
    print(f"\n=== Phase 1: L2 Optimization (best={best_score:.8f}) ===")
    t1_alloc = 120.0
    t1_end = time.monotonic() + t1_alloc

    a_l2, s_l2 = l2_optimization(best_seq, t1_alloc, verbose=True)
    try_update(a_l2, "l2")

    # === Phase 2: Log-Sum-Exp Optimization ===
    print(f"\n=== Phase 2: LogSumExp Optimization (best={best_score:.8f}) ===")
    t2_alloc = 150.0
    t2_end = time.monotonic() + t2_alloc

    a_lse, s_lse = logsumexp_optimization(best_seq, t2_alloc, verbose=True)
    try_update(a_lse, "lse")

    # Also from L2 result
    if time.monotonic() < t2_end - 30:
        a_lse2, s_lse2 = logsumexp_optimization(a_l2, min(t2_end - time.monotonic(), 80), verbose=False)
        try_update(a_lse2, "lse_l2")

    # === Phase 3: LP Alternating (main workhorse) ===
    print(f"\n=== Phase 3: LP Alternating (best={best_score:.8f}) ===")
    t3_alloc = 400.0
    t3_end = time.monotonic() + t3_alloc

    # From current best
    t_alloc1 = min(t3_alloc * 0.4, 160.0)
    a_lp1, s_lp1 = lp_alternating(best_seq, t_alloc1, solver, verbose=True)
    try_update(a_lp1, "lp_main")

    # From L2 result
    if time.monotonic() < t3_end - 60:
        t_alloc2 = min(t3_end - time.monotonic() - 60, 120.0)
        a_lp2, s_lp2 = lp_alternating(a_l2, t_alloc2, solver, verbose=False)
        try_update(a_lp2, "lp_l2")

    # From diverse inits
    inits = [
        diverse_init(n, rng, 0),
        diverse_init(n, rng, 4),
        diverse_init(n, rng, 7),
    ]
    for idx, init in enumerate(inits):
        if time.monotonic() > t3_end:
            break
        remaining = t3_end - time.monotonic()
        if remaining < 30:
            break
        t_alloc3 = min(remaining * 0.3, 80.0)
        a_lp3, s_lp3 = lp_alternating(init, t_alloc3, solver, verbose=False)
        try_update(a_lp3, f"lp_init{idx}")

    # === Phase 4: Basin Hopping ===
    print(f"\n=== Phase 4: Basin Hopping (best={best_score:.8f}) ===")
    t4_alloc = 300.0
    t4_end = time.monotonic() + t4_alloc

    a_bh, s_bh = basin_hopping(best_seq, t4_alloc, solver, verbose=True)
    try_update(a_bh, "bh")

    # === Phase 5: Multi-Peak Shaving ===
    print(f"\n=== Phase 5: Multi-Peak Shaving (best={best_score:.8f}) ===")
    t5_alloc = 150.0
    t5_end = time.monotonic() + t5_alloc

    a_mp, s_mp = multi_peak_shaving(best_seq, t5_alloc, verbose=True)
    try_update(a_mp, "multi_peak")

    # === Phase 6: Coordinate Descent ===
    print(f"\n=== Phase 6: Coordinate Descent (best={best_score:.8f}) ===")
    t6_alloc = 200.0
    t6_end = time.monotonic() + t6_alloc

    t_cd1 = min(t6_alloc * 0.5, 100.0)
    a_cd1, s_cd1 = adaptive_coord_descent(best_seq, t_cd1, verbose=True, label="main")
    try_update(a_cd1, "cd_main")

    if time.monotonic() < t6_end - 30:
        t_cd2 = min(t6_end - time.monotonic() - 30, 80.0)
        a_cd2, s_cd2 = adaptive_coord_descent(a_lp1, t_cd2, verbose=True, label="lp")
        try_update(a_cd2, "cd_lp")

    # === Phase 7: Fine Polish ===
    print(f"\n=== Phase 7: Fine Polish (best={best_score:.8f}) ===")
    t7_alloc = 100.0
    t7_end = time.monotonic() + t7_alloc

    a_fp, s_fp = fine_polish(best_seq, t7_alloc, verbose=True)
    try_update(a_fp, "fine")

    # === Phase 8: Final LP + Polish cycle ===
    print(f"\n=== Phase 8: Final Cycle (best={best_score:.8f}) ===")
    t8_end = deadline - 20

    if time.monotonic() < t8_end - 60:
        t_final_lp = min(t8_end - time.monotonic() - 60, 80.0)
        a_fl, s_fl = lp_alternating(best_seq, t_final_lp, solver, verbose=False)
        try_update(a_fl, "final_lp")

    if time.monotonic() < t8_end - 20:
        t_final_cd = min(t8_end - time.monotonic() - 20, 40.0)
        a_fc, s_fc = adaptive_coord_descent(best_seq, t_final_cd, verbose=False, label="fin")
        try_update(a_fc, "final_cd")

    if time.monotonic() < t8_end - 5:
        a_ff, s_ff = fine_polish(best_seq, min(t8_end - time.monotonic() - 5, 20), verbose=False)
        try_update(a_ff, "final_fine")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
