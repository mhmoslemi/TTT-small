Since a[k-i] >= 0, we have delta <= (M_new - conv[k]) / (2*a[k-i]) for all k with a[k-i] > 0.

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_of(a):
    """Compute 2*n*max(conv)/(sum)^2 quickly."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (s * s))

def simplex_project(v):
    """Project vector v onto the simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        k = int(np.nonzero(cond)[0][-1])
        theta = (cssv[k] - 1.0) / (k + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def find_lp_solver():
    """Find the fastest working LP solver."""
    cached = getattr(find_lp_solver, '_cached', None)
    if cached is not None:
        return cached
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                find_lp_solver._cached = solver_name
                return solver_name
        except Exception:
            continue
    find_lp_solver._cached = cp.ECOS
    return cp.ECOS

def build_conv_matrix(a):
    """Build the convolution matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def lp_best_response(a, solver):
    """
    Solve LP: min M s.t. conv(a,b) <= M, sum(b)=1, b>=0.
    Returns (b_opt, M_opt).
    """
    n = len(a)
    T = build_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    ones = np.ones(2 * n - 1)
    constraints = [
        T @ b_var <= M_var * ones,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return a.copy(), float('inf')

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return a.copy(), float('inf')

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return a.copy(), float('inf')
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else float('inf')
    return b_val, M_val

def optimal_1d_coord_descent(a0, time_limit, verbose=True, label=""):
    """
    Coordinate descent with optimal 1D step using envelope method.
    For each coordinate, find the delta that minimizes the new max convolution.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            ai = a[i]
            if ai < 1e-18:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            
            # Unaffected max
            unaff_max = -1e30
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))
            unaff_max = max(unaff_max, 0.0)

            # Affected coefficients: conv_new[k] = conv[k] + 2*delta*a[k-i] + delta^2 * 1_{k=2i}
            affected_indices = np.arange(k_start, k_end + 1)
            a_shifted = a[affected_indices - i]
            
            # For the linear approximation (ignoring delta^2):
            # conv_new[k] ≈ conv[k] + 2*delta*a[k-i]
            # We want to minimize max(unaff_max, max_k (conv[k] + 2*delta*a[k-i]))
            
            # The optimal delta is where the envelope is minimized.
            # For delta < 0: convex[k] + 2*delta*a[k-i] decreases for larger a[k-i]
            # For delta > 0: convex[k] + 2*delta*a[k-i] increases for larger a[k-i]
            
            # Binary search for optimal delta
            # Lower bound: delta >= -ai (to keep a[i] >= 0)
            # Upper bound: we need conv_new[k] >= 0 for all k, but since we're minimizing max, the upper bound is determined by the constraint that we don't want to increase M too much.
            
            # Let's use a practical bisection approach
            lo = -ai
            hi = ai  # Try delta in [-ai, ai]
            
            # Evaluate at a few points to find a good bracket
            # The optimal delta should be near where the max of affected values equals unaff_max
            # or where two affected values are equal
            
            # Simple approach: evaluate at a grid and refine
            best_delta = 0.0
            best_new_M = M
            
            # First, try to reduce M
            # The maximum reduction in M is limited by the steepest affected value
            # delta that brings the max affected value down to unaff_max:
            max_affected_idx = int(np.argmax(conv[k_start:k_end+1]))
            max_affected_val = conv[k_start + max_affected_idx]
            a_at_max = a[k_start + max_affected_idx - i]
            
            if a_at_max > 1e-15 and max_affected_val > unaff_max:
                # delta to bring max_affected_val down to unaff_max
                delta_target = (unaff_max - max_affected_val) / (2.0 * a_at_max)
                delta_target = max(delta_target, -ai)
                delta_target = min(delta_target, hi)
                
                # Evaluate at delta_target and nearby
                for d in [delta_target * 0.5, delta_target * 0.75, delta_target, delta_target * 1.25, delta_target * 0.25]:
                    if d < lo or d > hi:
                        continue
                    new_affected = conv[k_start:k_end+1].copy() + 2.0 * d * a_shifted
                    if 2 * i >= k_start and 2 * i <= k_end:
                        idx = 2 * i - k_start
                        new_affected[idx] += d * d
                    new_M = max(unaff_max, float(new_affected.max()))
                    if new_M < best_new_M - 1e-15:
                        best_new_M = new_M
                        best_delta = d
            else:
                # Try small positive and negative deltas
                for factor in [-0.5, -0.2, -0.05, 0.05, 0.2, 0.5]:
                    d = ai * factor
                    if d < lo or d > hi:
                        continue
                    new_affected = conv[k_start:k_end+1].copy() + 2.0 * d * a_shifted
                    if 2 * i >= k_start and 2 * i <= k_end:
                        idx = 2 * i - k_start
                        new_affected[idx] += d * d
                    new_M = max(unaff_max, float(new_affected.max()))
                    if new_M < best_new_M - 1e-15:
                        best_new_M = new_M
                        best_delta = d

            if best_delta != 0.0 and best_new_M < M - 1e-14:
                a[i] = max(a[i] + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] += best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 10:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD[{label}] sweep={sweeps} ({elapsed:.0f}s): best={best_score:.10f} M={M:.12f}")

    return best_a, best_score

def peak_shaving(a0, time_limit, verbose=True):
    """
    Peak-targeted mass transfer: find the top peaks and
    transfer mass to reduce them.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    rounds = 0

    while time.monotonic() - t0 < time_limit:
        rounds += 1
        improved = False

        # Find top peaks
        K = 10
        peak_indices = np.argpartition(conv, -K)[-K:]
        peak_indices = sorted(peak_indices, key=lambda k: conv[k], reverse=True)

        for k_star in peak_indices:
            k_star = int(k_star)
            if conv[k_star] < M * 0.99:
                continue

            # Find contributing pairs
            i_start = max(0, k_star - n + 1)
            i_end = min(n - 1, k_star)

            # For each pair (i, k_star-i), try transferring mass
            # The product a[i]*a[j] (j = k_star-i) contributes to conv[k_star]
            # To reduce it, make a[i] and a[j] more unequal
            
            pairs = []
            for i in range(i_start, i_end + 1):
                j = k_star - i
                if i == j:
                    pairs.append((i, j, a[i] * a[j], 'self'))
                else:
                    pairs.append((i, j, 2.0 * a[i] * a[j], 'cross'))
            
            pairs.sort(key=lambda x: x[2], reverse=True)

            for (i, j, contrib, ptype) in pairs[:5]:
                if contrib < 1e-14:
                    continue
                
                if ptype == 'self':
                    # a[i]^2 contributes; reduce a[i]
                    for frac in [0.05, 0.1, 0.15, 0.2, 0.3]:
                        delta = -a[i] * frac
                        new_affected = conv[i:i+n].copy() + 2.0 * delta * a[0:n]
                        idx = 2 * i - i  # = i
                        # Actually, k=2i is at position 2i in conv
                        # The affected range is [i, i+n-1]
                        # k=2i is at offset 2i - i = i in the affected range
                        new_affected[i] += delta * delta
                        new_M = float(np.max(np.concatenate([conv[:i] if i > 0 else [], new_affected, conv[i+n:] if i+n < 2*n-1 else []])))
                        if new_M < M - 1e-12:
                            a[i] = max(a[i] + delta, 0.0)
                            conv = fast_convolve(a)
                            M = float(conv.max())
                            improved = True
                            cur_score = 2.0 * n * M
                            if cur_score < best_score:
                                best_score = cur_score
                                best_a = a.copy()
                            break
                else:
                    # a[i]*a[j] contributes (counted twice in conv[k_star])
                    # Transfer mass from smaller to larger to reduce product
                    if a[i] >= a[j]:
                        target, other = j, i
                    else:
                        target, other = i, j
                    
                    for frac in [0.05, 0.1, 0.15, 0.2]:
                        delta = a[target] * frac
                        # Move delta from 'target' to 'other'
                        if a[other] < 0:
                            continue
                        
                        # Try the transfer
                        new_a = a.copy()
                        new_a[target] -= delta
                        new_a[other] += delta
                        if new_a[target] < 0:
                            continue
                        
                        new_conv = fast_convolve(new_a)
                        new_M = float(new_conv.max())
                        if new_M < M - 1e-12:
                            a = new_a
                            conv = new_conv
                            M = new_M
                            improved = True
                            cur_score = 2.0 * n * M
                            if cur_score < best_score:
                                best_score = cur_score
                                best_a = a.copy()
                            break

        if not improved and rounds > 20:
            break

        if verbose and rounds % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave round={rounds} ({elapsed:.0f}s): best={best_score:.10f}")

    return best_a, best_score

def diverse_init(n, rng, init_type=0):
    """Generate diverse initializations."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    x = (t - center) / center  # in [-1, 1]
    
    if init_type == 0:
        # U-shaped (boosted edges)
        a = 0.3 + 0.7 * x * x
    elif init_type == 1:
        # Stronger U-shape
        a = 0.1 + 0.9 * x ** 4
    elif init_type == 2:
        # Bimodal
        c1 = n * 0.25
        c2 = n * 0.75
        w = n * 0.1
        a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
        a += 0.01 * np.ones(n)  # Ensure all positive
    elif init_type == 3:
        # Power-law from edges
        # a[i] ~ (1 + |x|)^p
        p = rng.uniform(2.0, 5.0)
        a = (1.0 + np.abs(x)) ** p
    elif init_type == 4:
        # Cosine with higher power
        power = rng.uniform(2.0, 5.0)
        a = np.maximum(0.5 * (1 + np.cos(np.pi * x)), 0) ** power + 0.001
    elif init_type == 5:
        # Inverse triangular (opposite of uniform convolution shape)
        tri = 1.0 - np.abs(x)  # triangular, 1 at center, 0 at edges
        a = 1.0 / (tri + 0.1)
    elif init_type == 6:
        # Sort of "optimal" shape: peaks at 1/4 and 3/4
        c1 = n * 0.25
        c2 = n * 0.75
        w = n * 0.15
        a = np.exp(-0.5 * ((t - c1) / w) ** 2) + np.exp(-0.5 * ((t - c2) / w) ** 2)
    elif init_type == 7:
        # Random smooth
        freq1 = rng.uniform(1, 5)
        freq2 = rng.uniform(3, 10)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * freq1 * t / n) + 0.2 * np.cos(2 * np.pi * freq2 * t / n)
    else:
        # Uniform
        a = np.ones(n)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def lp_alternating_improved(a0, time_limit, solver, verbose=True):
    """
    Improved LP-based best-response alternating minimization.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    stagnation = 0
    base_a = a.copy()  # Reference for perturbation

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        # Solve LP: best response b* to current a
        b_new, M_lp = lp_best_response(a, solver)
        if not np.isfinite(M_lp):
            break

        # Evaluate b_new
        conv_b = fast_convolve(b_new)
        score_b = 2.0 * n * conv_b.max()
        if score_b < best_score:
            best_score = score_b
            best_a = b_new.copy()

        # Damping schedule: starts at 1.0 (full iterate), decays to ~0.2
        if it <= 10:
            alpha = 1.0
        elif it <= 30:
            alpha = max(0.5, 1.0 - 0.03 * (it - 10))
        elif it <= 80:
            alpha = max(0.3, 0.5 - 0.005 * (it - 30))
        else:
            alpha = max(0.15, 0.3 - 0.002 * (it - 80))

        a = (1 - alpha) * a + alpha * b_new
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-15:
            a = a / s

        # Evaluate updated a
        conv_a = fast_convolve(a)
        score_a = 2.0 * n * conv_a.max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            base_a = a.copy()

        # Stagnation detection
        if score_a > best_score * (1 + 1e-9) and score_b > best_score * (1 + 1e-9):
            stagnation += 1
        else:
            stagnation = max(0, stagnation - 1)

        # Random perturbation when stuck
        if stagnation > 20:
            noise_scale = 0.02 / np.sqrt(n)
            a_pert = base_a + np.random.standard_normal(n) * noise_scale
            a_pert = np.maximum(a_pert, 0)
            s_p = a_pert.sum()
            if s_p > 1e-15:
                a_pert = a_pert / s_p
            score_pert = 2.0 * n * fast_convolve(a_pert).max()
            if score_pert < best_score * 1.01:
                a = a_pert
            elif stagnation > 50:
                # More aggressive reset
                a = base_a + np.random.standard_normal(n) * (0.05 / np.sqrt(n))
                a = np.maximum(a, 0)
                s_p = a.sum()
                if s_p > 1e-15:
                    a = a / s_p
            stagnation = 0

        if verbose and it % 25 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP it={it} ({elapsed:.0f}s): best={best_score:.10f} alpha={alpha:.3f}")

    return best_a, best_score

def subgradient_descent(a0, time_limit, lr_init=0.01, top_k=30, verbose=True):
    """Subgradient descent with multi-peak weighting and simplex projection."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        M = conv.max()
        score = 2.0 * n * M

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 100:
            lr *= 0.8
            no_improve = 0
            if lr < 1e-12:
                break

        # Multi-peak subgradient
        k_act = min(top_k, 2 * n - 1)
        if k_act >= 2 * n - 1:
            peaks = np.arange(2 * n - 1)
            pv = conv.copy()
        else:
            peaks = np.argpartition(conv, -k_act)[-k_act:]
            pv = conv[peaks]

        temp = 0.005 * max(M, 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.ones(len(peaks)) / len(peaks)
        else:
            w = w / ws

        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            sg[i_start:i_end + 1] += w[idx] * 2.0 * a[k - i_end:k - i_start + 1][::-1]

        a_new = a - lr * sg
        a_new = simplex_project(a_new)
        a = a_new
        conv = fast_convolve(a)

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD it={it} ({elapsed:.0f}s): best={best_score:.10f} lr={lr:.8f}")

    return best_a, best_score

def fine_polish(a0, time_limit, verbose=True):
    """Very fine coordinate descent for last-mile polishing."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            ai = a[i]
            if ai < 1e-18:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            unaff_max = -1e30
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))
            unaff_max = max(unaff_max, 0.0)

            a_shifted = a[k_start - i:k_end - i + 1]
            conv_affected = conv[k_start:k_end + 1]

            best_delta = 0.0
            best_new_M = M

            # Fine multiplicative steps
            for factor in [0.999, 0.998, 0.997, 0.995, 1.003, 1.005, 1.007, 1.008]:
                delta = ai * (factor - 1.0)
                new_val = ai + delta
                if new_val < -1e-16:
                    continue

                new_affected = conv_affected + 2.0 * delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx] += delta * delta

                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M - 1e-15:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-15:
                a[i] = max(ai + best_delta, 0.0)
                changes = 2.0 * best_delta * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx] += best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))
                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
