    """Solve the KKT system given weights w on active set S.
    
    Unknowns: a_i for i in support, and lambda.
    Equations:
      For i in support: sum_{k in S} 2*w[k]*a[k-i] = lambda
      sum_{i in support} a_i = 1
    """

        support = np.arange(n)
    I = support
    nI = len(I)
    
    # Build the system
    # (nI + 1) x (nI + 1) system
    # First nI equations: KKT conditions
    # Last equation: normalization
    
    A_mat = np.zeros((nI + 1, nI + 1))
    b_vec = np.zeros(nI + 1)
    
    # KKT equations
    for idx_i, i in enumerate(I):
        for k in S:
            j = k - i
            if 0 <= j < n and j in I:  # j is in the support
                j_idx = np.where(I == j)[0][0]  # slow!
                A_mat[idx_i, j_idx] += 2 * w[k]
        A_mat[idx_i, nI] = -1  # -lambda
    
    # Normalization
    A_mat[nI, :nI] = 1.0
    b_vec[nI] = 1.0
    
    # Solve
    sol = np.linalg.solve(A_mat, b_vec)
    a = np.zeros(n)
    a[I] = np.maximum(sol[:nI], 0)
    a /= a.sum()
    return a

        w /= w.sum()

        p_min = pressure.min()

        a_new = np.exp(scaled_a)

        # Mix (to avoid oscillation)

        a = (1 - alpha) * a + alpha * a_new
        
        # Evaluate

        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Increase beta slowly
        if it % 100 == 0:

```python
def solve_dual_lp(a, iterations=100, lr=0.1):
    """Solve the dual LP using projected gradient.
    max t s.t. (T^T w)_i >= t, sum(w)=1, w>=0
    Equivalent: max min_i (T^T w)_i s.t. w in simplex
    """
    n = len(a)
    m = 2*n - 1
    w = np.ones(m) / m  # initialize
    t = -np.inf
    
    for it in range(iterations):
        # Compute (T^T w)_i for all i
        pressure = hankel_mv(w, a)  # (T^T w)_i
        t = pressure.min()
        
        # Subgradient of min_i (T^T w)_i w.r.t. w:
        # = (T[:, argmin]) * (indicator of min) + average over ties
        min_indices = np.where(pressure >= t - 1e-10)[0]
        rho = 1.0 / len(min_indices)
        
        # Subgradient: grad_w[k] = rho * sum_i 1[i in min_indices] * T[k,i]
        # = rho * sum_{i in min_indices} a[k-i] (for valid k-i)
        grad_w = np.zeros(m)
        for i in min_indices:
            for k in range(max(0, i), min(m, i + n)):
                j = k - i
                if 0 <= j < n:
                    grad_w[k] += a[j]
        grad_w *= rho
        
        # Gradient ascent step
        w = w + lr * grad_w
        w = np.maximum(w, 0)
        w /= w.sum()
        
        # Decrease lr
        lr *= 0.99
    
    return w, t
```
