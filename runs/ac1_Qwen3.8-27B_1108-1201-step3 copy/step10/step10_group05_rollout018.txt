import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

    T = np.zeros((2 * n - 1, n))

    return T

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_ab(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def build_conv_matrix(a):

def solve_lp_best_response(a_fixed, solver=None):
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()

        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]

    try:

def lp_fp_with_average(a0, time_limit, solver=None, label=""):
    """LP Fictitious Play with running average and adaptive damping."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

        iters += 1

        # Solve best response LP
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is None:
            continue

        # Adaptive damping: slower decay for better convergence
        alpha = 1.0 / (1.0 + iters * 0.02)
        a_new = (1 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        a = a_new

        # Update running average (Nash converge at optimal rate)
        w = 1.0 / iters
        avg_a = (1 - w) * avg_a + w * a

        # Store history for potential use
        if len(b_history) > 100:
            b_history.pop(0)
        b_history.append(b_new.copy())

        # Evaluate current and average
        score_cur = 2.0 * n * fast_convolve(a).max()
        if score_cur < best_score - 1e-14:
            best_score = score_cur
            best_a = a.copy()

        score_avg = 2.0 * n * fast_convolve(avg_a).max()
        if score_avg < best_score - 1e-14:
            best_score = score_avg
            best_a = avg_a.copy()

        # Also try LP solution directly
        score_b = 2.0 * n * fast_convolve(b_new).max()
        if score_b < best_score - 1e-14:
            best_score = score_b
            best_a = b_new.copy()

        # Every 20 iterations, try average of last few LP solutions
        if iters % 20 == 0 and len(b_history) >= 5:
            recent_avg = np.mean(b_history[-10:], axis=0)
            recent_avg = np.maximum(recent_avg, 0)

                  f"cur={score_cur:.6f}, avg={score_avg:.6f}, best={best_score:.6f}")

def nesterov_smoothed_md(a0, time_limit, beta_init=30.0, lr_init=0.008, label=""):

        it += 1

        if it > 1:
            y = (1.0 + gamma) * a - gamma * y

        weights = np.exp(scaled)
        Z = weights.sum()
        if Z < 1e-30:
            weights = np.zeros_like(conv_y)

        log_a -= lr * grad

def peak_targeted_perturb(a0, rng, strength=0.05):

def random_restructure(a0, rng, frac=0.2):
    """Randomly restructure to escape local optima."""
    n = len(a0)
    a = a0.copy()
    keep = rng.random(n) > frac
    n_keep = keep.sum()
    if n_keep < max(10, n // 5):
        keep = np.ones(n, dtype=bool)
        n_keep = n
    a[~keep] = 0
    n_keep_actual = keep.sum()
    if n_keep_actual > 0:
        a[keep] = a[keep] * (1.0 + rng.standard_normal(n_keep_actual) * 0.3)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy()
    return a / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        noise = rng.standard_normal(n)
        # Smooth via FFT
        N = n
        A = rfft(noise)
        # Low-pass filter
        freqs = np.arange(N // 2 + 1)
        cutoff = N // 8
        A[freqs > cutoff] *= 0.1
        smooth = irfft(A, N)[:n]
        a = 1.0 + 0.3 * smooth / (np.max(np.abs(smooth)) + 1e-10)
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-15:
            a = a / s
            score = 2.0 * n * fast_convolve(a).max()
            candidates.append((a, score))

    # Two-bump structure
    for center_frac in [0.3, 0.4, 0.5, 0.6]:
        t = np.arange(n)
        centers = np.array([center_frac * n, (1 - center_frac) * n])
        width = n / 8.0
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-0.5 * ((t - c) / width) ** 2)
        a += 0.1  # baseline
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-15:
            a = a / s
            score = 2.0 * n * fast_convolve(a).max()
            candidates.append((a, score))

    if candidates:
        candidates.sort(key=lambda x: x[1])
        best = candidates[0][0].copy()
        best_score = candidates[0][1]

    return best, best_score


def subgradient_nesterov(a0, time_limit, label=""):
    """Subgradient method with Nesterov acceleration on the exact objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    gamma = 1.0
    y = a.copy()
    it = 0
    avg_a = a.copy()

    while time.monotonic() - t0 < time_limit:
        it += 1
        step = 1.0 / np.sqrt(it + 1)

        y = (1.0 + gamma) * a - gamma * y
        y = simplex_project(np.maximum(y, 0))

        conv_y = fast_convolve(y)
        M = conv_y.max()
        threshold = M - 0.02 * M
        near_peak = np.where(conv_y >= threshold)[0]
        rho = 1.0 / len(near_peak)

        grad = np.zeros(n)
        for k in near_peak:
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            grad[j] += rho * 2.0 * y[k - j]

        a_new = a - step * grad
        a_new = simplex_project(a_new)

        gamma_it = 2.0 / (it + 2.0)
        avg_a = (1.0 - gamma_it) * avg_a + gamma_it * a_new

        for cand in [a_new, avg_a]:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-10:
                continue
            cand = cand / sc
            score = 2.0 * n * fast_convolve(cand).max()
            if score < best_score - 1e-14:
                best_score = score
                best_a = cand.copy()

        a = a_new
        gamma = (1.0 + np.sqrt(1.0 + 4.0 * gamma * gamma)) / 2.0

    return best_a, best_score


def check_and_update(a, best_seq, best_score, label=""):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score
