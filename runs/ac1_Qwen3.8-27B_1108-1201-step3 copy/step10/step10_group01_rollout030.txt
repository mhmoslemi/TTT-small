"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using accelerated exponentiated gradient
(mirror descent with KL divergence), LP alternation, LogSumExp L-BFGS, and
peak-directed coordinate descent. The new key innovation is the accelerated
exponentiated gradient which provides O(1/T^2) convergence with O(n log n) 
cost per iteration (via FFT-based convolution), making it much faster than
LP-based approaches for the initial optimization phase.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate 2n*max(conv(a,a))/(sum(a))^2 with local computation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def accelerated_exponentiated_gradient(a0, time_limit, top_k_schedule=None,
                                         eta0=0.1, eta_decay=0.9999,
                                         momentum=0.9, max_iters=500000,
                                         print_interval=50000):
    """
    Accelerated mirror descent (exponentiated gradient) with KL divergence.
    
    Objective: f(a) = 2n * max(conv(a,a)) subject to a >= 0, sum(a) = 1.
    
    Uses:
    - Subgradient at top-k peaks for smoother gradient
    - Nesterov momentum for O(1/T^2) convergence
    - Adaptive step size (polyak-like)
    - Gradual increase of top-k for smooth convergence
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    if top_k_schedule is None:
        # Start aggressive (k=1), gradually smooth out
        top_k_schedule = [1, 1, 2, 3, 5, 8, 13, 21, 34]
    
    eta = eta0
    y = a.copy()  # Nesterov momentum variable
    v = np.zeros(n)  # Momentum velocity
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Determine top-k for this iteration
        prog = it / max(1, max_iters)
        k_idx = min(int(prog * len(top_k_schedule)), len(top_k_schedule) - 1)
        top_k = top_k_schedule[k_idx]
        
        # Nesterov extrapolation
        y = a + momentum * v if it > 0 else a.copy()
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy > 0:
            y = y / sy
        else:
            y = a.copy()
        
        # Compute convolution and find top-k peaks
        b = fast_convolve(y)
        max_b = b.max()
        
        # Get top-k peak positions
        if top_k >= len(b):
            top_peaks = np.arange(len(b))
        else:
            top_peaks = np.argpartition(b, -top_k)[-top_k:]
        
        # Compute subgradient (average over top-k peaks)
        g = np.zeros(n)
        for k in top_peaks:
            # d conv(a,a)[k] / da_j = a[k-j] + a[j] if k-j==j (i.e., k=2j)
            # More simply: for each j, contribution is a[k-j] if 0<=k-j<n
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            vals = y[k - idx]
            g[idx] += vals
        g /= len(top_peaks)
        
        # Multiplicative weight update (mirror descent step)
        # a_new_j = a_j * exp(-eta * g_j) / sum
        log_a = np.log(np.maximum(a, 1e-300))
        log_a_new = log_a - eta * g
        # Normalize (soft normalization for numerical stability)
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        # Nesterov momentum update
        v = a_new - a
        a = a_new
        
        # Track best
        score = 2.0 * n * b.max()  # approximate (using y's conv)
        # Actually compute exact score for a
        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        # Adaptive eta
        eta *= eta_decay
        eta = max(eta, 1e-6)
        
        if (it + 1) % print_interval == 0:
            elapsed = time.monotonic() - t0
            print(f"    AEG it={it+1} ({elapsed:.0f}s): score={best_score:.8f} eta={eta:.6f} k={top_k}")
    
    return best_a, best_score


def gradient_logsumexp(a, n, T):
    """Compute gradient of (2n/T) * log(sum(exp(T*b))) where b=conv(a,a)."""
    b = fast_convolve(a)
    max_b = b.max()
    exp_b = np.exp(T * (b - max_b))
    w = exp_b / exp_b.sum()
    
    grad = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-14:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += w[k] * a[k - idx]
    grad *= 2.0 * n / T
    S = np.dot(grad, a)
    return a * (grad - S)  # Projection to tangent space of simplex

def peak_coord_descent_improved(a0, time_limit, n_steps=200000, top_m=15):

                        break

                        mask[pos] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        break

                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()

                    break

def lp_best_response(a_fixed, solver=None):
    """Solve LP: min M s.t. conv(a_fixed, b)_k <= M, sum(b)=1, b>=0."""
    if not HAS_CVXPY:
        return None, None
    if solver is None:
        solver = _find_lp_solver()
    
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.asarray(b_var.value, dtype=np.float64)
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s < 1e-15:
                return None, None
            b_val = b_val / s
            return b_val, float(M_var.value)
    except Exception:
        pass
    for alt in [cp.SCS, cp.ECOS, cp.CLARABEL, cp.PDLP]:
        if alt == solver:
            continue
        try:
            prob.solve(solver=alt, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s < 1e-15:
                    return None, None
                b_val = b_val / s
                return b_val, float(M_var.value)
        except Exception:
            continue
    return None, None

def _find_lp_solver():
    global _lp_solver_cache
    if _lp_solver_cache is not None:

def nesterov_lp_optimize(a0, time_limit, solver=None, max_iters=100000,
                          stall_patience=15, print_interval=5000):
    """Nesterov dual-averaging LP alternation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall = 0
    rng = np.random.default_rng(777)
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, M_b = lp_best_response(y, solver)
        if b_new is None:
            stall += 1
            if stall > 25:
                break
            continue
        
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy
        
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        if (it + 1) % 500 == 0:
            if best_score < prev_best - 1e-10:
                stall = 0
            else:
                stall += 1
            prev_best = best_score
            
            if stall >= stall_patience:
                noise_scale = 0.02 / max(1, np.sqrt(stall))
                noise = rng.standard_normal(n) * noise_scale
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall = 0
            
            if print_interval and (it + 1) % print_interval == 0:
                elapsed = time.monotonic() - t0
                print(f"    NLP it={it+1} ({elapsed:.0f}s): score={best_score:.8f} M_LP={M_b:.8f}")
    
    return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Accelerated Exponentiated Gradient (mirror descent with KL) - fast O(n log n) iterations
2. LP Alternation with Nesterov dual averaging - the primary workhorse
3. LogSumExp-smoothed L-BFGS with aggressive multi-stage cooling
4. Peak-directed coordinate descent with smart redistribution
5. Multi-start diversity and alternative length exploration
Key innovation: the accelerated exponentiated gradient provides a fast
complementary optimization trajectory to the LP, and the improved
time management ensures we extract maximum value from each phase.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_local(a):
    """Evaluate 2n*max(conv(a,a))/(sum(a))^2 with local computation."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def accelerated_exponentiated_gradient(a0, time_limit, top_k_schedule=None,
                                       eta0=0.12, eta_decay=0.99995,
                                       momentum=0.85, max_iters=500000,
                                       print_interval=100000):
    """
    Accelerated mirror descent (exponentiated gradient) with KL divergence.
    Uses subgradient at top-k peaks with Nesterov momentum.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    if top_k_schedule is None:
        top_k_schedule = [1, 1, 2, 3, 5, 8, 13, 21, 34, 50, 80]

    eta = eta0
    v = np.zeros(n)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        prog = it / max(1, max_iters)
        k_idx = min(int(prog * len(top_k_schedule)), len(top_k_schedule) - 1)
        top_k = top_k_schedule[k_idx]

        # Nesterov extrapolation
        if it > 0:
            y = a + momentum * v
        else:
            y = a.copy()
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy
        else:
            y = a.copy()

        b = fast_convolve(y)
        max_b = b.max()

        if top_k >= len(b):
            top_peaks = np.arange(len(b))
        else:
            top_peaks = np.argpartition(b, -top_k)[-top_k:]

        g = np.zeros(n)
        for k in top_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += y[k - idx]
        g /= len(top_peaks)

        # Multiplicative weight update
        log_a = np.log(np.maximum(a, 1e-300))
        log_a_new = log_a - eta * g
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        v = a_new - a
        a = a_new

        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        eta *= eta_decay
        eta = max(eta, 1e-7)

        if print_interval > 0 and (it + 1) % print_interval == 0:
            elapsed = time.monotonic() - t0
            print(f"    AEG it={it+1} ({elapsed:.0f}s): score={best_score:.8f} eta={eta:.6f} k={top_k}")

    return best_a, best_score


def lbgfs_logsumexp_improved(a0, time_limit, T_stages=None, maxiter_per_stage=8000):
    """
    L-BFGS with LogSumExp smoothing in log-space for multi-stage cooling.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    a_init = a_init / a_init.sum()
    z = np.log(a_init)
    t0 = time.monotonic()

    if T_stages is None:
        T_stages = [15.0, 10.0, 7.0, 5.0, 3.5, 2.5, 1.8, 1.2, 0.8, 0.5,
                    0.35, 0.25, 0.18, 0.12, 0.08, 0.05, 0.03, 0.02, 0.015, 0.01]

    iters_per = max(30, maxiter_per_stage // len(T_stages))

    for T_stage in T_stages:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func_loc(z, n=n, T=T_stage, t0=t0, time_limit=time_limit):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = np.exp(z - z.max())
            a = a / a.sum()
            b = fast_convolve(a)
            max_b = b.max()
            exp_b = np.exp(T * (b - max_b))
            lse = max_b + np.log(exp_b.sum()) / T
            return (2.0 * n) * lse

        def jac_loc(z, n=n, T=T_stage, t0=t0, time_limit=time_limit):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z)
            a = np.exp(z - z.max())
            a = a / a.sum()
            b = fast_convolve(a)
            max_b = b.max()
            exp_b = np.exp(T * (b - max_b))
            w = exp_b / exp_b.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += w[k] * a[k - idx]
            grad_a *= 2.0 * n
            S = np.dot(grad_a, a)
            return a * (grad_a - S)

        try:
            res = minimize(
                fun=func_loc, x0=z, jac=jac_loc, method='L-BFGS-B',
                options={'maxiter': iters_per, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': iters_per * 400, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    a_out = np.exp(z - z.max())
    a_out = a_out / a_out.sum()
    return a_out


def peak_coord_descent_improved(a0, time_limit, n_steps=300000, top_m=12):
    """
    Coordinate descent with smart redistribution: reduces peak contributors
    and redistributes mass to non-contributing positions.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0
    rng = np.random.default_rng(12345)

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        top_peaks = np.argsort(b)[::-1][:min(top_m, len(b))]

        improved = False

        for peak_k in top_peaks[:5]:
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            order = np.argsort(contribs)[::-1]

            for oi in order[:6]:
                pos = idx[oi]
                if a[pos] < 1e-12:
                    continue
                for factor in [0.995, 0.99, 0.98, 0.96]:
                    a_trial = a.copy()
                    a_trial[pos] *= factor
                    diff = a[pos] * (1.0 - factor)
                    if diff < 1e-15:
                        break
                    mask = np.ones(n, dtype=bool)
                    mask[pos] = False
                    for pk in top_peaks[:3]:
                        js2 = max(0, pk - n + 1)
                        je2 = min(n - 1, pk)
                        idx2 = np.arange(js2, je2 + 1)
                        mask[idx2] = False
                    if mask.sum() < 10:
                        mask = np.ones(n, dtype=bool)
                        mask[pos] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        break
                    a_trial[mask] = others + diff * others / others_sum
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved and step % 200 == 0:
            n_perturb = int(rng.integers(2, 6))
            positions = rng.choice(n, size=n_perturb, replace=False)
            a_trial = a.copy()
            for pos in positions:
                a_trial[pos] *= (1.0 + rng.uniform(-0.04, 0.04))
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 1e-15:
                a_trial = a_trial / st
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True

        if not improved:
            stall += 1
            if stall > 800:
                noise = rng.standard_normal(n) * 0.008
                a_trial = a + noise
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-15:
                    a = a_trial / st
                stall = 0

        if (step + 1) % 100000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD it={step+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


_lp_solver_cache = None

def find_lp_solver():
    """Find a working LP solver."""
    global _lp_solver_cache
    if _lp_solver_cache is not None:
        return _lp_solver_cache
    if not HAS_CVXPY:
        return None
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cache = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cache = cp.SCS
    return cp.SCS


def lp_best_response(a_fixed, solver=None):
    """Solve LP: min M s.t. conv(a_fixed, b)_k <= M, sum(b)=1, b>=0."""
    if not HAS_CVXPY:
        return None, None
    if solver is None:
        solver = find_lp_solver()

    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed

    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    for sol in [solver] + [s for s in [cp.SCS, cp.ECOS, cp.CLARABEL, cp.PDLP] if s != solver]:
        try:
            prob.solve(solver=sol, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                b_val = np.asarray(b_var.value, dtype=np.float64)
                b_val = np.maximum(b_val, 0)
                s = b_val.sum()
                if s < 1e-15:
                    return None, None
                b_val = b_val / s
                return b_val, float(M_var.value)
        except Exception:
            continue
    return None, None


def nesterov_lp_optimize(a0, time_limit, solver=None, max_iters=100000,
                          stall_patience=12, print_interval=5000):
    """Nesterov dual-averaging LP alternation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall = 0
    rng = np.random.default_rng(777)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = lp_best_response(y, solver)
        if b_new is None:
            stall += 1
            if stall > 25:
                break
            continue

        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = 1.0
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 500 == 0:
            if best_score < prev_best - 1e-10:
                stall = 0
            else:
                stall += 1
            prev_best = best_score

            if stall >= stall_patience:
                noise_scale = 0.015 / max(1, np.sqrt(stall))
                noise = rng.standard_normal(n) * noise_scale
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall = 0

            if print_interval > 0 and (it + 1) % print_interval == 0:
                elapsed = time.monotonic() - t0
                print(f"    NLP it={it+1} ({elapsed:.0f}s): score={best_score:.8f} M_LP={M_b:.8f}")

    return best_a, best_score

            )
            sc = evaluate_local(a_alt)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_alt.copy()
                print(f"  *** NEW BEST [alt_n{alt_n}]: {sc:.8f}")
            rem = deadline - time.monotonic()
            if rem > 20 and best_seq is not None:
                a_alt2 = lbgfs_logsumexp_improved(
                    best_seq if len(best_seq) == alt_n else init_alt, rem * 0.3,
                    T_stages=[5.0, 2.0, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01],
                    maxiter_per_stage=5000
                )
                sc2 = evaluate_local(a_alt2)
                if np.isfinite(sc2) and sc2 < best_score:
                    best_score = sc2
                    best_seq = a_alt2.copy()
                    print(f"  *** NEW BEST [alt_lbgfs_n{alt_n}]: {sc2:.8f}")
            rem = deadline - time.monotonic()
            if rem < 20:
                break

    # Phase 8: Final polish
    print(f"\n=== Phase 8: Final Polish, score={best_score:.8f} ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t_fin = min(rem * 0.5, 25.0)
        if t_fin > 5:
            a_fin = lbgfs_logsumexp_improved(
                best_seq, t_fin,
                T_stages=[0.8, 0.4, 0.2, 0.1, 0.05, 0.03, 0.02, 0.01, 0.005],
                maxiter_per_stage=10000
            )
            sc = evaluate_local(a_fin)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_fin.copy()
                print(f"  *** NEW BEST [final_lbgfs]: {sc:.8f}")

    rem = deadline - time.monotonic()
    if rem > 8:
        a_cd2, _ = peak_coord_descent_improved(best_seq, min(rem * 0.7, 15.0),
                                                 n_steps=200000, top_m=8)
        sc = evaluate_local(a_cd2)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_cd2.copy()
            print(f"  *** NEW BEST [final_cd]: {sc:.8f}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
        try:
            seq_list = [float(max(0.0, x)) for x in best_seq]
            eval_score = evaluate_sequence(seq_list)
            if np.isfinite(eval_score):
                print(f"  evaluate_sequence confirms: {eval_score:.8f}")
        except Exception:
            pass
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
