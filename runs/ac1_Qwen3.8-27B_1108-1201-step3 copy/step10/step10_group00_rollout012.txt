"""

1. Spectral flattening (Gerchberg-Saxton) for fast global improvement
2. Nesterov-accelerated LP alternating with solver fallbacks
3. Ultra-fine LogSumExp L-BFGS (temperatures down to 0.001)

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver, preferring fast ones."""
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_mat = build_conv_matrix(a_fixed)
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    solvers_to_try = [solver, cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL, cp.GUROBI, cp.MOSEK]
    solved = False
    for s in solvers_to_try:
        if s is None:
            continue
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solved = True
                break
        except Exception:
            continue

    if not solved:
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(max(0, M_var.value))


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def logsumexp_obj_grad(z, n, T_temp):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    max_b = b.max()
    exp_terms = np.exp((b - max_b) / T_temp)
    lse = max_b + T_temp * np.log(exp_terms.sum())
    obj = 2.0 * n * lse

    w = exp_terms / exp_terms.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, 2.0 * n * grad_z


def lbgfs_obj(z, n, T_stage, deadline):
    """Objective for L-BFGS."""
    if time.monotonic() > deadline:
        return 1e15
    obj, _ = logsumexp_obj_grad(z, n, T_stage)
    return obj


def lbgfs_jac(z, n, T_stage, deadline):
    """Gradient for L-BFGS."""
    if time.monotonic() > deadline:
        return np.zeros_like(z)
    _, g = logsumexp_obj_grad(z, n, T_stage)
    return g


def spectral_flatten(a, n_iters=50, blend=0.4, deadline=None):
    """
    Iterative spectral flattening (Gerchberg-Saxton style).
    Flattens the power spectrum of a while maintaining non-negativity.
    """
    n = len(a)
    L = 1 << (2 * n - 1).bit_length()
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    for i in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        a_padded = np.zeros(L)
        a_padded[:n] = a
        F = np.fft.fft(a_padded)

        # Target: flatten magnitudes (keep DC component)
        mags = np.abs(F)
        # Use geometric mean of non-DC magnitudes as target
        non_dc = mags[1:]
        if non_dc.min() < 1e-15:
            target = np.mean(non_dc)
        else:
            target = np.exp(np.mean(np.log(non_dc + 1e-30)))

        # Blend magnitudes toward target
        phases = np.angle(F)
        F_new = F.copy()
        for k in range(1, L // 2 + 1):
            new_mag = (1 - blend) * mags[k] + blend * target
            F_new[k] = new_mag * np.exp(1j * phases[k])
            if k < L - k:
                F_new[L - k] = new_mag * np.exp(1j * phases[L - k])

        a_new = np.fft.ifft(F_new).real[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new

    return a


def peak_shave(a, time_limit, n_steps=100000):
    """
    Directly target the maximum of the autocorrelation using multiplicative updates.
    Focus on the top few peaks to reduce them fastest.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return None, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    eta = 0.02
    scale = 1.0
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Find top peaks (within 0.1% of max)
        threshold = max_b * (1.0 - 1e-3)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        # Compute weighted subgradient from active peaks
        weights = (b[active] - max_b * (1 - 1e-3)) / max(1e-30, max_b - max_b * (1 - 1e-3))
        weights = np.maximum(weights, 0)
        if weights.sum() < 1e-15:
            weights = np.ones(len(active))

        g = np.zeros(n)
        for i, k in enumerate(active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += weights[i] * a[k - idx]
        g /= max(1.0, weights.sum())

        # Multiplicative update
        eta_eff = eta * scale
        log_scale = -eta_eff * g
        log_scale -= log_scale.max()
        a_new = a * np.exp(log_scale)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 30:
            scale *= 0.7
            no_improve = 0
            if scale < 0.005:
                break

        if step % 20000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave {elapsed:.0f}s: score={best_score:.8f} scale={scale:.4f}")

    return best_a, best_score


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=1000000):
    """Nesterov dual averaging LP alternating with adaptive restarts."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    rng = np.random.default_rng(12345)
    lp_times = []

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        lp_t0 = time.monotonic()
        b_new, M_b = solve_lp_best_response(y, solver)
        lp_times.append(time.monotonic() - lp_t0)

        if b_new is None:
            continue

        if len(lp_times) >= 20:
            avg_lp_time = np.mean(lp_times[-50:])
            if avg_lp_time > 5.0 and len(lp_times) > 100:
                print(f"    LP too slow ({avg_lp_time:.2f}s/iter), stopping at iter {it+1}")
                break

        # Nesterov dual averaging
        alpha = 2.0 / (it + 2)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        # Stall detection
        if (it + 1) % 300 == 0:
            if best_score < prev_best - 1e-10:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 8:
                x = np.linspace(0, 2 * np.pi, n)
                freq = rng.integers(1, 7)
                noise = 0.03 * np.cos(freq * x + rng.uniform(0, 2 * np.pi))
                y = y + noise
                y = np.maximum(y, 0)
                sy = y.sum()
                if sy > 1e-15:
                    y = y / sy
                stall_count = 0

            if (it + 1) % 3000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def lbgfs_refine(a0, time_limit, T_stages, maxiter_per_stage, deadline=None):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    if deadline is None:
        deadline = t0 + time_limit
    n_stages = len(T_stages)
    iters_per_stage = max(30, maxiter_per_stage // n_stages)

    for stage_i, stage_T in enumerate(T_stages):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 3:
            break

        # Adjust iterations based on remaining time
        stages_left = n_stages - stage_i
        avail_time = remaining
        iters = max(20, min(iters_per_stage * 3, int(avail_time * 50 / stages_left)))

        def func(z_in):
            return lbgfs_obj(z_in, n, stage_T, deadline)

        def jac(z_in):
            return lbgfs_jac(z_in, n, stage_T, deadline)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-25,
                         'gtol': 1e-18, 'maxfun': iters * 500, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)

def structured_init(n, rng):
    """Create a structured initial sequence."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x)
        a = np.maximum(a, 0)
    elif choice == 4:
        a = (1 - x) ** 1.5 + 0.1
    elif choice == 5:
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    elif choice == 6:
        # Sinc-like
        center = n // 2
        t = (np.arange(n) - center) / center
        a = np.sinc(t)
        a = np.maximum(a, 0)
    else:
        # Triangle
        a = 1.0 - np.abs(2 * x - 1)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_local(a):
    """Evaluate sequence score locally (fast)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s * s))


def perturbed_sequence(a, rng, strength=0.03):
    """Create a perturbed version of sequence a."""
    n = len(a)
    x = np.linspace(0, 2 * np.pi, n)
    freq = rng.integers(1, 8)
    phase = rng.uniform(0, 2 * np.pi)
    perturbation = strength * np.cos(freq * x + phase)
    a_pert = a + perturbation
    a_pert = np.maximum(a_pert, 0)
    s = a_pert.sum()
    if s < 1e-15:
        return a.copy()
    return a_pert / s


def diversify_and_select(parent, n, rng, budget_s=15):
    """Try several perturbations and spectral variants, return the best."""
    candidates = [parent.copy()]
    # Add perturbed versions
    for strength in [0.02, 0.05, 0.1, 0.15]:
        candidates.append(perturbed_sequence(parent, rng, strength=strength))
    # Add spectral flattened version
    sf = spectral_flatten(parent, n_iters=10, blend=0.3)
    candidates.append(sf)

    best_c = parent.copy()
    best_sc = evaluate_local(parent)

    for c in candidates:
        sc = evaluate_local(c)
        if sc < best_sc:
            best_sc = sc
            best_c = c.copy()

    return best_c, best_sc


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses spectral flattening, Nesterov LP alternating, ultra-fine LogSumExp
    L-BFGS, peak shaving, and multi-round diversification.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_local(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 1200
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_local(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    n_best = len(best_seq)

    def local_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_local(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    rem = deadline - time.monotonic()
    print(f"Total budget: {rem:.0f}s, starting score: {best_score:.8f}")

    # === Phase 0: Diversification (15s) ===
    print(f"\n=== Phase 0: Diversification ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_div = min(rem * 0.01, 15.0)
        best_div, sc_div = diversify_and_select(best_seq, n_best, rng, budget_s=t_div)
        local_update(best_div, "diversified")

    # === Phase 1: Spectral Flattening (60s) ===
    print(f"=== Phase 1: Spectral Flattening (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_spec = min(rem * 0.04, 60.0)
    if t_spec > 10:
        print(f"  Budget: {t_spec:.0f}s")
        a_spec = spectral_flatten(best_seq, n_iters=200, blend=0.3, deadline=time.monotonic() + t_spec)
        local_update(a_spec, "spectral")
        # Also try with different blend
        a_spec2 = spectral_flatten(best_seq, n_iters=100, blend=0.6, deadline=time.monotonic() + min(t_spec, 20))
        local_update(a_spec2, "spectral_strong")

    # === Phase 2: Nesterov LP (main workhorse) ===
    print(f"=== Phase 2: Nesterov LP n={n_best} (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_lp = min(rem * 0.25, 400.0)
    if t_lp > 30:
        print(f"  Budget: {t_lp:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_lp, solver, n_max_iters=1000000)
        local_update(a_lp, "nesterov_lp_1")

    # === Phase 3: Ultra-fine LogSumExp L-BFGS ===
    print(f"=== Phase 3: L-BFGS LogSumExp fine (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_lbfgs = min(rem * 0.30, 500.0)
    if t_lbfgs > 30:
        # Very fine temperature schedule with many low-temp stages
        T_stages = [10.0, 5.0, 3.0, 2.0, 1.0, 0.5, 0.3, 0.2, 0.1,
                    0.05, 0.03, 0.02, 0.015, 0.01, 0.007, 0.005,
                    0.003, 0.002, 0.0015, 0.001]
        print(f"  T_stages: {len(T_stages)} stages, budget={t_lbfgs:.0f}s")
        a_lb = lbgfs_refine(best_seq, t_lbfgs, T_stages, maxiter_per_stage=20000,
                            deadline=time.monotonic() + t_lbfgs)
        local_update(a_lb, "lbgfs_fine")

    # === Phase 4: Peak Shaving ===
    print(f"=== Phase 4: Peak Shaving (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_peak = min(rem * 0.08, 100.0)
    if t_peak > 10:
        a_pk, _ = peak_shave(best_seq, t_peak, n_steps=200000)
        local_update(a_pk, "peak_shave")

    # === Phase 5: Second Nesterov LP round ===
    print(f"=== Phase 5: Nesterov LP Round 2 (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_lp2 = min(rem * 0.15, 200.0)
    if t_lp2 > 20:
        print(f"  Budget: {t_lp2:.0f}s")
        a_lp2, _ = nesterov_lp_refine(best_seq, t_lp2, solver, n_max_iters=500000)
        local_update(a_lp2, "nesterov_lp_2")

    # === Phase 6: Perturbation escape + quick re-optimize ===
    print(f"=== Phase 6: Perturbation Escape (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        for trial in range(3):
            rem_sub = deadline - time.monotonic()
            if rem_sub < 30:
                break
            a_pert = perturbed_sequence(best_seq, rng, strength=0.015 + 0.01 * trial)
            # Quick spectral + LP
            a_sp = spectral_flatten(a_pert, n_iters=20, blend=0.3)
            t_sub = min(rem_sub * 0.25, 60.0)
            if t_sub > 10:
                a_opt, _ = nesterov_lp_refine(a_sp, t_sub, solver, n_max_iters=200000)
                local_update(a_opt, f"pert_trial{trial}")

    # === Phase 7: Try different n ===
    print(f"=== Phase 7: Try different n (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        # Try n=1000
        t_n1000 = min(rem * 0.12, 120.0)
        if t_n1000 > 30:
            print(f"  Trying n=1000, budget={t_n1000:.0f}s")
            init_1000 = interpolate_sequence(best_seq, 1000)
            a_sp1000 = spectral_flatten(init_1000, n_iters=30, blend=0.3)
            a_1000, _ = nesterov_lp_refine(a_sp1000, t_n1000 * 0.7, solver, n_max_iters=100000)
            local_update(a_1000, "lp_n1000")

    rem = deadline - time.monotonic()
    if rem > 60:
        # Try n=1500
        t_n1500 = min(rem * 0.15, 120.0)
        if t_n1500 > 30:
            print(f"  Trying n=1500, budget={t_n1500:.0f}s")
            init_1500 = interpolate_sequence(best_seq, 1500)
            a_sp1500 = spectral_flatten(init_1500, n_iters=30, blend=0.3)
            a_1500, _ = nesterov_lp_refine(a_sp1500, t_n1500 * 0.7, solver, n_max_iters=100000)
            local_update(a_1500, "lp_n1500")
            # Quick L-BFGS on n=1500 if it's the best
            rem = deadline - time.monotonic()
            if best_seq is not None and len(best_seq) == 1500 and rem > 30:
                T_st2 = [3.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
                a_1500b = lbgfs_refine(best_seq, min(rem, 60.0), T_st2,
                                        maxiter_per_stage=5000, deadline=time.monotonic() + min(rem, 60.0))
                local_update(a_1500b, "lbgfs_n1500")

    # === Phase 8: Final polish ===
    print(f"=== Phase 8: Final Polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        # L-BFGS final
        t_fin = min(rem * 0.4, 60.0)
        if t_fin > 10:
            T_stages_fin = [2.0, 1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.003, 0.002]
            a_fin = lbgfs_refine(best_seq, t_fin, T_stages_fin, maxiter_per_stage=5000,
                                 deadline=time.monotonic() + t_fin)
            local_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 15:
        # Peak shave final
        t_ps = min(rem * 0.4, 30.0)
        if t_ps > 5:
            a_ps, _ = peak_shave(best_seq, t_ps, n_steps=50000)
            local_update(a_ps, "final_peak")

    rem = deadline - time.monotonic()
    if rem > 10:
        # Final LP
        t_flp = min(rem * 0.5, 20.0)
        if t_flp > 5:
            a_flp, _ = nesterov_lp_refine(best_seq, t_flp, solver, n_max_iters=50000)
            local_update(a_flp, "final_lp")

    # Final verification and output
    if best_seq is not None:
        fs = evaluate_local(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_start
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}) at {elapsed:.1f}s")
        # Verify with evaluate_sequence
        test_seq = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        try:
            eval_score = evaluate_sequence(test_seq)
            print(f"Verified with evaluate_sequence: {eval_score:.8f}")
        except Exception as e:
            print(f"Verification error: {e}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
