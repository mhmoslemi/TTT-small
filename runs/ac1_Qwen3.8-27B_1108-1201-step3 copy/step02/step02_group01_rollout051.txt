"""
Combined spectral-gradient and LP-based iterative optimization for minimizing
the normalized peak autocorrelation of non-negative step functions.
Uses FFT-based spectral shaping, LP refinement with adaptive line search,
and multi-scale peak-reduction gradient methods.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the score 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * np.max(conv) / (sum_a ** 2))


def normalize_seq(a, S=None):
    """Normalize sequence to have a given sum."""
    if S is None:
        S = np.sqrt(2.0 * len(a))
    s = np.sum(a)
    if s < 1e-12:
        return a * S / len(a)
    return a * S / s


def build_lp_matrix(a):
    """Build the convolution constraint matrix A where (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            A[k, j] = a[k - j]
    return A


def lp_solve(a, M):
    """
    Solve LP: maximize sum(g) subject to conv(a,g)[k] <= M for all k, g >= 0.
    Returns the optimal g.
    """
    n = len(a)
    A = build_lp_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.status == 0:
        return res.x
    return np.zeros(n)


def spectral_gradient(a):
    """
    Compute gradient of sum_k (R[k] - mean(R))^2 w.r.t. a.
    This pushes all autocorrelation lags towards the mean.
    Uses FFT for efficiency.
    """
    n = len(a)
    R = fft_convolve(a, a)
    mu = np.sum(R) / (2 * n - 1)
    E = R - mu

    # Zero-pad a to length 2n-1
    b = np.zeros(2 * n - 1)
    b[:n] = a

    # term2[j] = conv(E, b)[j]
    conv2 = fft_convolve(E, b)

    # term1[j] = conv(E_flipped, b)[2n-2-j]
    E_flipped = E[::-1]
    conv1_full = fft_convolve(E_flipped, b)

    # Extract the relevant portions
    # term2[j] for j in [0, n-1]: conv2[j]
    # term1[j] for j in [0, n-1]: conv1_full[2n-2-j]
    term2 = conv2[:n]
    term1 = conv1_full[2 * n - 2 : 2 * n - 2 - n : -1]

    grad = 2.0 * (term1 + term2)
    return grad


def peak_gradient(a, top_k=5):
    """
    Compute a weighted gradient focusing on the top-k peak lags.
    Returns gradient vector of length n.
    """
    n = len(a)
    R = fft_convolve(a, a)

    # Get top-k lags
    if top_k >= 2 * n - 1:
        peak_indices = np.arange(2 * n - 1)
        weights = np.ones(2 * n - 1)
    else:
        # Weight by how far above mean each lag is
        mu = np.mean(R)
        excess = np.maximum(R - mu, 0)
        # Get top_k indices
        peak_indices = np.argsort(excess)[-top_k:]
        weights = np.zeros(2 * n - 1)
        weights[peak_indices] = excess[peak_indices]
        # Normalize weights
        w_sum = np.sum(weights)
        if w_sum > 0:
            weights /= w_sum
        else:
            weights[np.argmax(R)] = 1.0

    # Compute gradient: sum over active lags of weight * dR[k]/da[j]
    # dR[k]/da[j] = a[k-j] + a[j-k] (with bounds)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] == 0:
            continue
        w = weights[k]
        for j in range(n):
            v = 0.0
            i1 = k - j
            if 0 <= i1 < n:
                v += a[i1]
            i2 = j - k
            if 0 <= i2 < n:
                v += a[i2]
            grad[j] += w * v
    return grad


def peak_gradient_fast(a, top_k=5):
    """
    Faster peak gradient using FFT-based cross-correlation.
    """
    n = len(a)
    R = fft_convolve(a, a)
    mu = np.mean(R)
    excess = np.maximum(R - mu, 0)

    if top_k >= 2 * n - 1:
        E = R - mu
        n_active = 2 * n - 1
    else:
        # Only use top k
        indices = np.argsort(excess)[-top_k:]
        E = np.zeros(2 * n - 1)
        E[indices] = R[indices] - mu
        n_active = top_k

    # grad[j] = sum_k E[k] * (a[k-j] + a[j-k])
    # = sum_k E[k]*a[k-j] + sum_k E[k]*a[j-k]
    # First term: cross-correlation of E and a
    # Second term: cross-correlation of E and reversed a (shifted)

    # Use FFT-based approach
    # term1[j] = sum_k E[k] * a[k-j] for valid indices
    # This equals: for the full convolution framework:
    # Let E_ext = E (length 2n-1), a_ext = a zero-padded to length 2n-1
    # term1[j] = sum_k E[k] * a_ext[k-j] = (E_rev * a_ext)[(2n-2) - j] where * is conv

    E_rev = E[::-1]
    a_ext = np.zeros(2 * n - 1)
    a_ext[:n] = a

    conv1 = fft_convolve(E_rev, a_ext)  # length (2n-1)+n-1 = 3n-2
    term1 = conv1[2 * n - 2 : 2 * n - 2 - n : -1]  # indices [2n-2, 2n-3, ..., n-2] -> reversed

    # term2[j] = sum_k E[k] * a[j-k] = (E * a_ext)[j]
    conv2 = fft_convolve(E, a_ext)
    term2 = conv2[:n]

    grad = term1 + term2
    return grad


def combine_gradients(a, alpha_peak=0.3, top_k=5):
    """
    Combine spectral and peak gradients.
    """
    grad_spec = spectral_gradient(a)
    grad_peak = peak_gradient_fast(a, top_k=top_k)

    # Normalize both gradients to have same magnitude
    gs_norm = np.linalg.norm(grad_spec)
    gp_norm = np.linalg.norm(grad_peak)

    if gs_norm < 1e-10 and gp_norm < 1e-10:
        return np.zeros(len(a))

    if gs_norm > 0:
        grad_spec = grad_spec / gs_norm
    if gp_norm > 0:
        grad_peak = grad_peak / gp_norm

    return alpha_peak * gp_norm * grad_peak + (1 - alpha_peak) * gs_norm * grad_spec


def line_search_blend(a, g, S, max_iter=20):
    """
    Find best blend parameter t to minimize max(conv) for (1-t)*a + t*g.
    Uses golden section search.
    """
    n = len(a)
    g_norm = normalize_seq(g, S)

    def obj(t):
        b = (1 - t) * a + t * g_norm
        b = np.maximum(b, 0)
        s = np.sum(b)
        if s < 1e-10:
            return np.inf
        b = b * S / s
        conv = fft_convolve(b, b)
        return np.max(conv)

    # Golden section search on [0, 1]
    gr = (np.sqrt(5) - 1) / 2
    lo, hi = 0.0, 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = obj(c)
    fd = obj(d)

    for _ in range(max_iter):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = obj(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj(d)

    t_best = (lo + hi) / 2
    return t_best, obj(t_best)

            stall_count += 1
            lr = 0.1 / n
            grad = combine_gradients(a, alpha_peak=0.5, top_k=10)
            a_new = a - lr * grad
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize_seq(a_new, S)
            new_cm = np.max(fft_convolve(a_new, a_new))
            if new_cm < M:
                a = a_new
                if new_cm < best_conv_max:
                    best_conv_max = new_cm
                    best_a = a_new.copy()
                    stall_count = 0

        if stall_count > 30:
            # Do a spectral gradient cleanup
            for _ in range(5):
                grad = spectral_gradient(a)
                gn = np.linalg.norm(grad)
                if gn < 1e-10:
                    break
                lr = 0.5 / (gn + 1e-10)
                a_new = a - lr * grad
                a_new = np.maximum(a_new, 0.0)
                a_new = normalize_seq(a_new, S)
                new_cm = np.max(fft_convolve(a_new, a_new))
                if new_cm < np.max(fft_convolve(a, a)):
                    a = a_new
                    if new_cm < best_conv_max:
                        best_conv_max = new_cm
                        best_a = a_new.copy()
            stall_count = 0

        iteration += 1

        if verbose and iteration % 50 == 0:
            score = 2.0 * n * best_conv_max / (S * S)
            elapsed = time.monotonic() - t0
            print(f"    LP iter {iteration} (t={elapsed:.0f}s): conv_max={best_conv_max:.6f}, score={score:.6f}")

    return best_a


def gradient_refine(a, time_limit, verbose=True):
    """
    Gradient-based optimization combining spectral and peak gradients.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_conv_max = np.max(fft_convolve(a, a))

    lr = 1.0 / n
    step = 0
    stall_count = 0
    best_alpha = 0.3
    top_k = 5

    while time.monotonic() - t0 < time_limit:
        # Adaptive parameters
        if step % 100 == 0:
            # Adjust top_k based on how spread out the peaks are
            R = fft_convolve(a, a)
            mu = np.mean(R)
            excess = R - mu
            significant_peaks = np.sum(excess > 0.1 * mu)
            if significant_peaks > 10:
                top_k = min(15, significant_peaks)
            elif significant_peaks > 5:
                top_k = 8
            else:
                top_k = 3

            # Adjust alpha based on progress
            if stall_count > 50:
                best_alpha = min(0.7, best_alpha + 0.05)
            elif stall_count == 0:
                best_alpha = max(0.1, best_alpha - 0.02)

        # Compute combined gradient
        grad = combine_gradients(a, alpha_peak=best_alpha, top_k=top_k)

        # Backtracking line search
        a_trial = a.copy()
        for backtrack in range(15):
            a_trial = a - lr * grad
            a_trial = np.maximum(a_trial, 0.0)
            a_trial = normalize_seq(a_trial, S)
            new_cm = np.max(fft_convolve(a_trial, a_trial))

            if new_cm < np.max(fft_convolve(a, a)) * (1 - 0.001) or backtrack == 14:
                break
            lr *= 0.5

        if new_cm < best_conv_max - 1e-12:
            best_conv_max = new_cm
            best_a = a_trial.copy()
            a = a_trial
            stall_count = 0
            lr *= 1.05
        elif new_cm < np.max(fft_convolve(a, a)):
            a = a_trial
            stall_count += 1
            lr *= 0.9
        else:
            stall_count += 1
            lr *= 0.7

        lr = min(lr, 5.0 / n)

        step += 1
        if verbose and step % 200 == 0:
            score = 2.0 * n * best_conv_max / (S * S)
            elapsed = time.monotonic() - t0
            print(f"    Grad step {step} (t={elapsed:.0f}s): score={score:.6f}, lr={lr:.5f}, top_k={top_k}")

    return best_a


def spectral_shaping(a, n_iters=50, beta=0.3, verbose=False):
    """
    Gerchberg-Saxton style spectral shaping to flatten the autocorrelation.
    In frequency domain, compress the power spectrum towards uniform.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a, S)
    best_a = a.copy()
    best_cm = np.max(fft_convolve(a, a))

    for it in range(n_iters):
        A = np.fft.fft(a)
        P = np.abs(A) ** 2
        P_mean = np.mean(P)

        # Compress: P_new = P * (P_mean / P)^beta = P^(1-beta) * P_mean^beta
        P_safe = np.maximum(P, 1e-20)
        P_new = P_safe ** (1 - beta) * P_mean ** beta

        # Reconstruct with original phase
        phase = np.angle(A)
        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(A_new))

        # Project onto non-negativity and normalize
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_seq(a_new, S)

        new_cm = np.max(fft_convolve(a_new, a_new))
        if new_cm < best_cm:
            best_cm = new_cm
            best_a = a_new.copy()
            a = a_new
        else:
            # Reduce beta and try again
            beta *= 0.95
            if beta < 0.01:
                break

    return best_a


def targeted_polish(a, time_limit, verbose=True):
    """
    Targeted local search focusing on reducing the top peaks.
    Uses coordinate and block updates with FFT evaluation.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_cm = np.max(fft_convolve(a, a))
    rng = np.random.default_rng(99999)

    while time.monotonic() - t0 < time_limit:
        R = fft_convolve(a, a)
        peak_idx = int(np.argmax(R))
        current_peak = R[peak_idx]

        # Strategy 1: Reduce top contributors to the peak
        contribs = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                contribs[i] = a[i] * a[j]

        top_k = max(5, n // 10)
        top_idx = np.argsort(contribs)[-top_k:]

        for _ in range(8):
            candidate = a.copy()
            # Reduce top contributors
            reduce_mask = rng.random(len(top_idx)) < 0.5
            for idx in top_idx[reduce_mask]:
                candidate[idx] *= np.exp(rng.normal(-0.03, 0.015))

            # Increase random other elements
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 0:
                n_inc = max(5, n // 15)
                inc_idx = rng.choice(other, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.03, 0.015, size=n_inc))

            candidate = np.maximum(candidate, 0.0)
            candidate = normalize_seq(candidate, S)

            new_cm = np.max(fft_convolve(candidate, candidate))
            if new_cm < best_cm:
                best_cm = new_cm
                best_a = candidate.copy()
                a = candidate
            elif new_cm < current_peak * 1.02 and rng.random() < 0.2:
                a = candidate

        # Strategy 2: Block perturbation
        if rng.random() < 0.4:
            width = max(2, n // 25)
            start = int(rng.integers(0, max(1, n - width)))
            candidate = a.copy()
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.02, size=width))
            candidate = np.maximum(candidate, 0.0)
            candidate = normalize_seq(candidate, S)
            new_cm = np.max(fft_convolve(candidate, candidate))
            if new_cm < best_cm:
                best_cm = new_cm
                best_a = candidate.copy()
                a = candidate

        if verbose:
            elapsed = time.monotonic() - t0
            if int(elapsed) % 5 == 0:
                score = 2.0 * n * best_cm / (S * S)
                print(f"    Polish t={elapsed:.0f}s: score={score:.6f}")

    return best_a


def interpolate_sequence(a, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, a)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining spectral shaping, LP refinement,
    gradient methods, and targeted local search.
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 15.0
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    def update_best(a):
        nonlocal best_score, best_seq
        a_clip = np.clip(a, 0.0, 1000.0)
        s = local_score(a_clip)
        if s < best_score:
            best_score = s
            best_seq = a_clip.copy()
            print(f"    *** New best: score={s:.6f}, n={len(a)} ***")
            return True
        return False

    # Get provided sequence
    provided_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.001, 1000.0)
        print(f"Provided sequence: n={len(provided_seq)}, score={local_score(provided_seq):.6f}")
        update_best(provided_seq)

    # === Phase 1: Spectral shaping (quick, explores new territory) ===
    print("\n=== Phase 1: Spectral shaping ===")
    t0 = time.monotonic()

    candidates = []
    if provided_seq is not None:
        candidates.append(provided_seq.copy())

    # Also try structured initializations
    for n_val in [640, 800]:
        # Chirp-like
        t = np.arange(n_val, dtype=np.float64)
        a_chirp = 1.0 + 0.4 * np.cos(np.pi * t * t / (n_val * 0.4))
        a_chirp = np.clip(a_chirp, 0.01, None)
        candidates.append(a_chirp)

        # Multi-tone
        a_mt = np.ones(n_val)
        for f in [3, 7, 13]:
            a_mt += 0.15 * np.cos(2 * np.pi * f * t / n_val + rng.uniform(0, 2 * np.pi))
        a_mt = np.clip(a_mt, 0.01, None)
        candidates.append(a_mt)

    for a_init in candidates:
        if time.monotonic() > deadline - 600:
            break
        a_shaped = spectral_shaping(a_init, n_iters=30, beta=0.3)
        update_best(a_shaped)

    print(f"  Phase 1 done (t={time.monotonic()-t0:.1f}s), best={best_score:.6f}")

    # === Phase 2: LP-based refinement (main workhorse) ===
    print("\n=== Phase 2: LP refinement ===")
    if best_seq is not None:
        n = len(best_seq)
        t_limit = min(600.0, (deadline - time.monotonic()) * 0.4)
        print(f"  LP refine (n={n}), t_limit={t_limit:.0f}s")
        result = lp_refine_v2(best_seq, t_limit, verbose=True)
        update_best(result)

        # Also try at a different length
        if provided_seq is not None:
            for n_try in [800, 960]:
                if time.monotonic() > deadline - 300:
                    break
                a_interp = interpolate_sequence(best_seq, n_try)
                a_interp = np.clip(a_interp, 0.01, None)
                t_per = min(60.0, (deadline - time.monotonic()) * 0.15)
                result2 = lp_refine_v2(a_interp, t_per, verbose=False)
                update_best(result2)

    print(f"  Phase 2 done, best={best_score:.6f}")

    # === Phase 3: Gradient refinement ===
    print("\n=== Phase 3: Gradient refinement ===")
    if best_seq is not None:
        n = len(best_seq)
        t_limit = min(300.0, (deadline - time.monotonic()) * 0.25)
        print(f"  Gradient refine (n={n}), t_limit={t_limit:.0f}s")
        result = gradient_refine(best_seq, t_limit, verbose=True)
        update_best(result)

    print(f"  Phase 3 done, best={best_score:.6f}")

    # === Phase 4: Another LP round on the improved sequence ===
    print("\n=== Phase 4: Second LP round ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 60
        if remaining > 60:
            n = len(best_seq)
            t_limit = min(remaining * 0.6, 200.0)
            print(f"  LP round 2 (n={n}), t_limit={t_limit:.0f}s")
            result = lp_refine_v2(best_seq, t_limit, verbose=True)
            update_best(result)

    print(f"  Phase 4 done, best={best_score:.6f}")

    # === Phase 5: Final targeted polish ===
    print("\n=== Phase 5: Targeted polish ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 10
        if remaining > 10:
            result = targeted_polish(best_seq, remaining, verbose=True)
            update_best(result)

    print(f"\n=== FINAL: best_score={best_score:.6f}, n={len(best_seq) if best_seq is not None else 0} ===")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"=== VERIFIED: evaluate_score={final_score:.6f} ===")
        return final_list

    # Fallback
    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]

    return [1.0] * 256
