import numpy as np
import time
from scipy.optimize import linprog

def fft_convolve(a, b=None):
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * np.max(conv) / (sum_a ** 2))

def normalize_seq(a, S=None):
    if S is None:
        S = np.sqrt(2.0 * len(a))
    s = np.sum(a)
    if s < 1e-12:
        return np.ones(len(a)) * S / len(a)
    return a * S / s

def log_sum_exp(c, T):
    """Smoothed max: (1/T) * log(sum(exp(T*c)))"""

"""
Multi-strategy optimization for minimizing normalized peak autocorrelation
of non-negative step functions. Combines LP-based refinement, variance-minimizing
gradient descent, smoothed peak reduction (log-sum-exp), and multiplicative
weight descent for large sequences. Uses adaptive temperature and multiple
sequence lengths to escape local minima.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b=None):
    """Compute linear convolution of two 1-D arrays using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute the score 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * np.max(conv) / (sum_a ** 2))


def normalize_seq(a, S=None):
    """Normalize sequence to have a given sum."""
    if S is None:
        S = np.sqrt(2.0 * len(a))
    s = np.sum(a)
    if s < 1e-12:
        return np.ones(len(a)) * S / len(a)
    return a * S / s


def build_lp_matrix(a):
    """Build the convolution constraint matrix A where (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            A[k, j] = a[k - j]
    return A


def lp_solve(a, M):
    """Solve LP: maximize sum(g) subject to conv(a,g)[k] <= M, g >= 0."""
    n = len(a)
    A = build_lp_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.status == 0:
        return res.x
    return np.zeros(n)


def lp_refine(a, time_limit, verbose=False):
    """LP-based iterative refinement."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_peak = np.max(fft_convolve(a, a))

    for iteration in range(10000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_a = fft_convolve(a, a)
        M = np.max(conv_a)
        g = lp_solve(a, M)
        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break

        c = normalize_seq(g)

        best_t = 0.0
        for t_try in [0.5, 0.4, 0.3, 0.25, 0.2, 0.15, 0.1, 0.08, 0.05, 0.03, 0.01]:
            b = (1 - t_try) * a + t_try * c
            b = np.clip(b, 0.0, None)
            b = normalize_seq(b, S)
            b_peak = np.max(fft_convolve(b, b))
            if b_peak <= M + 1e-10:
                best_t = t_try
                break

        if best_t == 0.0:
            # Try smaller steps
            for t_try in [0.005, 0.003, 0.001]:
                b = (1 - t_try) * a + t_try * c
                b = np.clip(b, 0.0, None)
                b = normalize_seq(b, S)
                b_peak = np.max(fft_convolve(b, b))
                if b_peak <= M + 1e-8:
                    best_t = t_try
                    break

        if best_t == 0.0:
            break

        a = (1 - best_t) * a + best_t * c
        a = np.clip(a, 0.0, None)
        a = normalize_seq(a, S)

        new_peak = np.max(fft_convolve(a, a))
        if new_peak < best_peak - 1e-12:
            best_peak = new_peak
            best_a = a.copy()

        if verbose and iteration % 200 == 0:
            print(f"      LP iter {iteration}: peak={best_peak:.6f}, score={best_peak:.6f}, t={best_t:.4f}")

        if best_t < 0.002:
            break

    return best_a


def variance_gradient(a):
    """
    Compute gradient of variance of conv(a,a) w.r.t. a.
    var(c) = mean(c^2) - mean(c)^2, where c = conv(a,a).
    d(var)/d(a[j]) = (4/(2n-1)) * sum_k (c[k] - mean_c) * a[k-j]
    """
    n = len(a)
    conv = fft_convolve(a, a)
    len_c = len(conv)
    mean_c = np.mean(conv)
    c_centered = conv - mean_c

    # grad[j] = (4/len_c) * sum_k c_centered[k] * a[k-j]
    # = (4/len_c) * correlate(c_centered, a, 'full')[j] for j=0,...,n-1
    corr = np.correlate(c_centered, a, mode='full')
    grad = (4.0 / len_c) * corr[:n]

    return grad, conv


def smoothed_peak_gradient(a, T):
    """
    Compute gradient of (1/T)*log(sum(exp(T*conv(a,a)[k]))) w.r.t. a.
    Uses softmax weights for the peaks.
    """
    n = len(a)
    conv = fft_convolve(a, a)
    c_max = np.max(conv)
    exp_Tc = np.exp(T * (conv - c_max))
    weights = exp_Tc / np.sum(exp_Tc)

    # grad[j] = 2 * sum_k weights[k] * a[k-j]
    corr = np.correlate(weights, a, mode='full')
    grad = 2.0 * corr[:n]

    return grad, conv


def gradient_optimize_combined(a, time_limit, verbose=False):
    """
    Combined gradient optimization using both variance minimization
    and smoothed peak reduction with adaptive temperature.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_peak = np.max(fft_convolve(a, a))

    # Adaptive parameters
    T = 5.0  # temperature for log-sum-exp
    T_max = 50.0
    lr = 1.0 / n
    lr_decay = 0.999
    step = 0
    stall_count = 0
    peak_history = []

    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a, a)
        current_peak = np.max(conv)
        current_var = np.var(conv)

        # Compute combined gradient
        # Weight between variance and peak gradient
        # Early: focus on peak; Later: focus on variance (flattening)
        progress = min(1.0, step / 2000.0)
        w_var = 0.3 + 0.5 * progress  # weight for variance gradient
        w_peak = 1.0 - w_var  # weight for peak gradient

        grad_var, _ = variance_gradient(a)
        grad_peak, _ = smoothed_peak_gradient(a, T)

        # Normalize gradients
        gv_norm = np.linalg.norm(grad_var)
        gp_norm = np.linalg.norm(grad_peak)
        if gv_norm > 1e-12:
            grad_var = grad_var / gv_norm
        if gp_norm > 1e-12:
            grad_peak = grad_peak / gp_norm

        combined_grad = w_var * grad_var + w_peak * grad_peak
        cg_norm = np.linalg.norm(combined_grad)
        if cg_norm > 1e-12:
            combined_grad = combined_grad / cg_norm

        # Gradient step
        a_new = a - lr * combined_grad
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_seq(a_new, S)

        new_peak = np.max(fft_convolve(a_new, a_new))

        if new_peak < best_peak:
            best_peak = new_peak
            best_a = a_new.copy()
            a = a_new
            stall_count = 0
            peak_history.append(new_peak)
            # Slightly increase lr on success
            lr = min(lr * 1.01, 3.0 / n)
        elif new_peak < current_peak:
            a = a_new
            stall_count += 1
            peak_history.append(new_peak)
        else:
            stall_count += 1
            lr *= 0.95
            peak_history.append(current_peak)

        # Adaptive temperature: increase T to flatten more
        T = min(T * 1.002, T_max)

        # Decay learning rate
        lr *= lr_decay
        lr = max(lr, 0.01 / n)

        # Reset if stuck
        if stall_count > 100:
            lr = max(lr * 0.5, 0.01 / n)
            T = max(T * 0.5, 2.0)
            stall_count = 0
            # Add small random perturbation
            a = a + 0.01 * np.random.default_rng(step).standard_normal(n)
            a = np.maximum(a, 0.0)
            a = normalize_seq(a, S)

        step += 1

        if verbose and step % 500 == 0:
            print(f"      Grad step {step}: peak={best_peak:.6f}, T={T:.1f}, lr={lr:.4f}")

    return best_a


def mwd_refine(a, time_limit, verbose=False):
    """
    Multiplicative Weight Descent alternative to LP for large sequences.
    Iteratively finds g that approximately maximizes sum(g) subject to conv(a,g) <= M.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_peak = np.max(fft_convolve(a, a))

    eta = 2.0 / np.sqrt(n)

    for iteration in range(50000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_a = fft_convolve(a, a)
        M = np.max(conv_a)

        # MWD to find g
        g = np.ones(n)
        for mwd_iter in range(50):
            v = fft_convolve(a, g)
            # Penalty for exceeding M
            violation = np.maximum(0.0, v - M)
            # Update g multiplicatively
            # g[j] *= exp(-eta * sum_k violation[k] * a[k-j] / (M + eps))
            # The gradient of sum_k violation[k]*conv(a,g)[k] w.r.t. g[j]
            # = sum_k violation[k] * a[k-j] (for valid indices)
            corr = np.correlate(violation, a, mode='full')
            grad = corr[:n]
            grad = np.maximum(grad, 0.0)
            g = g * np.exp(-eta * grad / (M + 1e-10))
            g = np.maximum(g, 1e-10)

        g_sum = np.sum(g)
        if g_sum < 1e-10:
            break

        c = normalize_seq(g)

        # Blend
        best_t = 0.0
        for t_try in [0.3, 0.2, 0.15, 0.1, 0.08, 0.05, 0.03, 0.01]:
            b = (1 - t_try) * a + t_try * c
            b = np.clip(b, 0.0, None)
            b = normalize_seq(b, S)
            b_peak = np.max(fft_convolve(b, b))
            if b_peak <= M + 1e-10:
                best_t = t_try
                break

        if best_t > 0:
            a = (1 - best_t) * a + best_t * c
            a = np.clip(a, 0.0, None)
            a = normalize_seq(a, S)
            new_peak = np.max(fft_convolve(a, a))
            if new_peak < best_peak:
                best_peak = new_peak
                best_a = a.copy()
        else:
            # MWD not converging well, break
            if iteration > 20:
                break

        if verbose and iteration % 100 == 0:
            print(f"      MWD iter {iteration}: peak={best_peak:.6f}")

    return best_a


def gerchberg_saxton_refine(a, time_limit, verbose=False):
    """
    Gerchberg-Saxton style iteration: push spectrum toward flatness
    while maintaining non-negativity.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_peak = np.max(fft_convolve(a, a))

    # Use DFT of length >= 2n-1
    N = 1
    while N < 2 * n - 1:
        N <<= 1

    for iteration in range(100000):
        if time.monotonic() - t0 > time_limit:
            break

        # Forward transform
        A = np.fft.rfft(a, N)

        # Target: flat magnitude (but keep DC component)
        target_mag = np.abs(A).copy()
        # Make it flatter: geometric mean of current and target
        # We want |A| to be more uniform
        # Use: new_mag = sqrt(|A| * target_flat) where target_flat is the geometric mean
        geo_mean = np.exp(np.mean(np.log(np.abs(A) + 1e-10)))
        # Blend toward flat
        alpha = 0.1
        new_mag = (1 - alpha) * np.abs(A) + alpha * geo_mean
        new_mag[0] = np.abs(A[0])  # Keep DC (preserves sum)

        # Reconstruct with new magnitude and old phase
        phase = np.angle(A)
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]

        # Enforce non-negativity
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_seq(a_new, S)

        new_peak = np.max(fft_convolve(a_new, a_new))
        if new_peak < best_peak:
            best_peak = new_peak
            best_a = a_new.copy()
            a = a_new

        if verbose and iteration % 1000 == 0:
            print(f"      GS iter {iteration}: peak={best_peak:.6f}")

    return best_a


def block_coordinate_descent(a, time_limit, verbose=False):
    """
    Block coordinate descent: update blocks of the sequence to reduce peak.
    For each block, solve a small optimization problem.
    """
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a, S)
    t0 = time.monotonic()
    best_a = a.copy()
    best_peak = np.max(fft_convolve(a, a))
    rng = np.random.default_rng(42)

    block_size = max(4, n // 64)
    n_blocks = n // block_size

    for iteration in range(200000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fft_convolve(a, a)
        current_peak = np.max(conv)
        peak_idx = int(np.argmax(conv))

        # Find the block that most contributes to the peak
        # conv[peak_idx] = sum_i a[i]*a[peak_idx-i]
        # Blocks that overlap with the "support" of the peak
        contrib = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                contrib[i] += a[i] * a[j]

        # Pick a random block weighted by contribution
        block_contribs = np.array([np.sum(contrib[b*block_size:(b+1)*block_size]) for b in range(n_blocks)])
        total = np.sum(block_contribs)
        if total > 1e-10:
            probs = block_contribs / total
        else:
            probs = np.ones(n_blocks) / n_blocks
        block_idx = rng.choice(n_blocks, p=probs)

        start = block_idx * block_size
        end = start + block_size

        # Try to optimize this block
        best_block = a[start:end].copy()
        best_block_peak = current_peak

        for _ in range(20):
            # Random perturbation of the block
            perturbation = a[start:end].copy() * np.exp(rng.normal(0, 0.1, block_size))
            candidate = a.copy()
            candidate[start:end] = perturbation
            candidate = np.clip(candidate, 0.0, None)
            candidate = normalize_seq(candidate, S)

            new_peak = np.max(fft_convolve(candidate, candidate))
            if new_peak < best_block_peak:
                best_block_peak = new_peak
                best_block = candidate[start:end].copy()

        if best_block_peak < current_peak:
            a[start:end] = best_block
            a = normalize_seq(a, S)
            current_peak = best_block_peak
            if current_peak < best_peak:
                best_peak = current_peak
                best_a = a.copy()

        if verbose and iteration % 1000 == 0:
            print(f"      BCD iter {iteration}: peak={best_peak:.6f}")

    return best_a


def make_initialization(n, init_type, rng):
    """Create various types of initial sequences."""
    t = np.arange(n, dtype=np.float64)
    if init_type == 'constant':
        a = np.ones(n)
    elif init_type == 'chirp':
        alpha = 0.3 + 0.3 * rng.random()
        a = 1.0 + alpha * np.cos(np.pi * t * t / (n * 0.5))
    elif init_type == 'sine':
        f = 1 + int(rng.integers(0, 4))
        amp = 0.2 + 0.3 * rng.random()
        phase = rng.uniform(0, 2 * np.pi)
        a = 1.0 + amp * np.cos(2 * np.pi * f * t / n + phase)
    elif init_type == 'multi_tone':
        a = np.ones(n)
        for _ in range(3):
            f = int(rng.integers(1, max(2, n // 4)))
            amp = rng.uniform(0.05, 0.2)
            phase = rng.uniform(0, 2 * np.pi)
            a += amp * np.cos(2 * np.pi * f * t / n + phase)
    elif init_type == 'random':
        a = 1.0 + 0.3 * rng.standard_normal(n)
    elif init_type == 'bimodal':
        center = n // 2
        sigma = n * (0.15 + 0.1 * rng.random())
        a = 1.0 + 0.5 * np.exp(-0.5 * ((t - center) / sigma) ** 2)
    elif init_type == 'triangular':
        center = n // 2
        a = 1.0 + 0.5 * (1.0 - np.abs(t - center) / (center + 1))
    else:
        a = np.ones(n)

    a = np.clip(a, 0.01, None)
    return a


def interpolate_sequence(a, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, a)

        update_best(result)

    # Phase 5: GS refinement
    print(f"\n=== Phase 5: GS refinement (best: {best_score:.6f}) ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            t_limit = min(120.0, remaining * 0.2)
            print(f"  GS refine, limit={t_limit:.0f}s")
            result = gerchberg_saxton_refine(best_seq, t_limit, verbose=True)
            update_best(result)

    # Phase 6: MWD refinement (faster than LP for large n)
    print(f"\n=== Phase 6: MWD refinement (best: {best_score:.6f}) ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            t_limit = min(120.0, remaining * 0.2)
            print(f"  MWD refine, limit={t_limit:.0f}s")
            result = mwd_refine(best_seq, t_limit, verbose=True)
            update_best(result)

    # Phase 7: Final LP polish
    print(f"\n=== Phase 7: Final LP polish (best: {best_score:.6f}) ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 10
        if remaining > 30:
            t_limit = min(remaining, 100.0)
            n = len(best_seq)
            if n <= 768:  # LP is too slow for very large n
                print(f"  Final LP (n={n}), limit={t_limit:.0f}s")
                result = lp_refine(best_seq, t_limit, verbose=True)
                update_best(result)
            else:
                # Use gradient for large n
                print(f"  Final gradient (n={n}), limit={t_limit:.0f}s")
                result = gradient_optimize_combined(best_seq, t_limit, verbose=False)
                update_best(result)

    # Phase 8: Block coordinate descent for final polish
    print(f"\n=== Phase 8: BCD polish (best: {best_score:.6f}) ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 5
        if remaining > 30:
            t_limit = min(remaining, 60.0)
            print(f"  BCD, limit={t_limit:.0f}s")
            result = block_coordinate_descent(best_seq, t_limit, verbose=False)
            update_best(result)

    # Phase 9: Final gradient pass
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 3
        if remaining > 10:
            result = gradient_optimize_combined(best_seq, remaining, verbose=False)
            update_best(result)

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)}, local={best_score:.6f} ===")
        return final_list

    # Fallback
    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]

    return [1.0] * 256
