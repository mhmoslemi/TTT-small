"""
Minimize the peak autocorrelation score for non-negative sequences.
Uses LP-based iterative optimization, gradient polish, and multi-start.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]

def score_fast(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))

def propose_candidate(seed=42, budget_s=1800):
    t_start = time.monotonic()
    deadline = t_start + budget_s - 10
    
    best_score = np.inf
    best_seq = None
    
    # Try constant sequences of various lengths
    for n in [64, 128, 256, 512, 1024, 2048, 4096, 8192]:
        a = np.ones(n)
        s = score_fast(a)
        if s < best_score:
            best_score = s
            best_seq = a.copy()
    
    print(f"Constant sequence best: n={len(best_seq)}, score={best_score:.6f}")
    
    # Verify with evaluate_sequence
    final_list = [float(x) for x in best_seq]
    eval_score = evaluate_sequence(final_list)
    print(f"evaluate_sequence: {eval_score:.6f}")
    
    # Try to improve with optimization
    # (If the constant sequence is truly optimal, this won't help,
    #  but let's try just in case)
    
    if time.monotonic() < deadline - 60:
        # Try LP-based optimization from a non-constant start
        from scipy.optimize import linprog
        
        for n in [128, 256, 512]:
            if time.monotonic() > deadline - 60:
                break
            # Start from a slightly perturbed constant
            rng = np.random.default_rng(seed + n)
            a0 = np.ones(n) + 0.1 * rng.standard_normal(n)
            a0 = np.clip(a0, 0.1, None)
            
            # Run a few LP iterations
            for _ in range(5):
                if time.monotonic() > deadline - 60:
                    break
                c = fft_convolve(a0, a0)
                M = np.max(c)
                
                # Build A matrix for LP: A[k,j] = a0[k-j] if valid
                A = np.zeros((2*n-1, n))
                for j in range(n):
                    A[j:j+n, j] = a0
                c_obj = -np.ones(n)
                b_ub = np.full(2*n-1, M)
                
                try:
                    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                                  bounds=[(0, None)]*n, method='highs',
                                  options={'time_limit': 5.0})
                    if res.success:

                        Sb = np.sum(b)
                        if Sb > 1e-10:

                                    eval_s = evaluate_sequence([float(x) for x in best_seq])
                                    print(f"  n={n} improved: {eval_s:.6f}")
                                a0 = a_new
                                break
                except:
                    pass
    
    # Final
    final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    final_score = evaluate_sequence(final_list)
    print(f"FINAL: n={len(final_list)}, score={final_score:.6f}")

  In this case, min(A^2, 2A^2+2B^2) = A^2 (if A^2 < 2A^2+2B^2, i.e., A^2 > 2B^2, i.e., A > sqrt(2)*B). Contradiction with A < sqrt(2)*B.
  So in the range B < A < sqrt(2)*B, max = 2A^2+2B^2 (same as before).
  And for A > sqrt(2)*B, max = A^2.
  Minimize A^2 subject to A > sqrt(2)*(1/2-A): A > sqrt(2)/2 - sqrt(2)*A, A(1+sqrt(2)) > sqrt(2)/2, A > sqrt(2)/(2(1+sqrt(2))) = (sqrt(2)*(sqrt(2)-1))/(2) = (2-sqrt(2))/2 ≈ 0.293.
  Min A^2 in this range: A = 0.293, A^2 ≈ 0.086. But 2A^2+2B^2 at this point: B = 0.207, 2*(0.086+0.043) = 0.258. So max = 0.258 > 0.086.
  
  So the constraint is always the 2A^2+2B^2 term, and the minimum is at A=B=1/4, giving max = 1/4.
