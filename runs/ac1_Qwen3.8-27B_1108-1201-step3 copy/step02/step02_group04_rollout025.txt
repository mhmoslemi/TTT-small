def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def objective_normalized(a):
    """Compute 2*n*max(conv) / (sum)^2 for non-negative a."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max() / (s * s)


def lp_direction(f, M, n, timeout=10):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)[k]<=M for all k.
    Returns normalized g (sum=1) or None."""
    # Cross-convolution: (f*g)[k] = sum_{j} f[max(0,k-n+1)..min(k,n-1)] * g[k-j]
    # Actually (f*g)[k] = sum_j f[j]*g[k-j] where both are defined on [0,n-1]
    # A[k,i] = f[k-i] if 0<=k-i<=n-1, else 0
    # So A is (2n-1) x n, A[k,i] = f[k-i] if 0 <= k-i <= n-1
    
    # Build sparse matrix
    rows = []
    cols = []
    vals = []
    for i in range(n):
        # Column i: A[k,i] = f[k-i] for k such that 0 <= k-i <= n-1
        # i.e., k ranges from i to i+n-1
        k_range = range(i, i + n)
        f_idx = range(0, n)  # k-i ranges from 0 to n-1
        rows.extend(k_range)
        cols.extend([i] * n)
        vals.extend(f.tolist())
    
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                           options={'disp': False, 'time_limit': timeout})
    if res.success:
        g = np.maximum(res.x, 0.0)
        gs = g.sum()
        if gs > 1e-12:
            return g / gs
    return None

def mirror_descent_step(a, conv, temp_param, eta):
    """One mirror descent step on the simplex."""
    n = len(a)
    max_val = conv.max()
    if max_val < 1e-15:
        return a
    
    # Soft weight based on convolution peaks
    scaled = temp_param * conv / max_val
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-12:
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= w_sum
    
    # Compute subgradient
    sg = np.zeros(n)
    thresh = w.max() * 0.01
    active = np.where(w > thresh)[0]
    for k in active:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        # sg[j0] += w[k] * 2 * a[k-j0] for j0 in [j_start, j_end]
        # a[k-j0] for j0 in [j_start, j_end] is a[k-j_end : k-j_start+1] reversed
        sg[j_start:j_end + 1] += w[k] * 2.0 * a[k - j_end:k - j_start + 1][::-1]
    
    # Mirror step: multiplicative update
    # f_new = f * exp(-eta * sg)
    # Then project to simplex
    lf = -eta * sg
    lf -= lf.max()
    a_new = a * np.exp(lf)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-12:
        a_new = np.ones(n) / n
    else:
        a_new /= s

def gen_inits(n, rng):
    """Generate diverse initializations of length n, each summing to 1."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uni'] = np.ones(n) / n
    
    for sf in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'g{int(sf * 100)}'] = g
    
    r = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = r / r.sum()
    
    # Bimodal
    bi = np.zeros(n)
    w = max(1, n // 5)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    inits['bi'] = bi / bi.sum()
    
    # Cosine
    cb = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    inits['cos'] = cb / cb.sum()
    
    # Random
    rr = rng.dirichlet(np.ones(n) * 0.5)
    inits['rnd'] = rr
    
    # Triangle
    tri = np.maximum(1.0 - np.abs(x - 0.5 * n) / (0.5 * n), 0.01)
    inits['tri'] = tri / tri.sum()
    
    # Power law
    pl = (n - x + 1) ** 0.5
    inits['power'] = pl / pl.sum()
    
    # Sparse-ish
    sp = np.zeros(n)
    indices = rng.choice(n, size=n // 4, replace=False)
    sp[indices] = 1.0
    inits['sparse'] = sp / sp.sum()
    
    # Alternating
    alt = np.ones(n)
    alt[::2] = 0.3
    inits['alt'] = alt / alt.sum()
    
    return inits


def lp_polish(f0, time_limit, lp_timeout=8):
    """LP-based alternating optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    f_sum = f.sum()
    if f_sum < 1e-12:
        f = np.ones(n) / n
    else:
        f /= f_sum
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()
    
    t = 0.5
    patience = 0
    iter_count = 0
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        g = lp_direction(f, M, n, timeout=lp_timeout)
        if g is None:
            break
        
        if np.max(np.abs(g - f)) < 1e-12:
            break
        
        # Backtracking line search with more points
        improved = False
        for t_try in [t, 0.8*t, 0.6*t, 0.4*t, 0.2*t, 0.1*t]:
            f_new = (1.0 - t_try) * f + t_try * g
            f_new = np.maximum(f_new, 0.0)
            s = f_new.sum()
            if s < 1e-12:
                continue
            f_new /= s
            conv_new = fast_convolve(f_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < best_obj - 1e-12:
                f = f_new
                M = conv_new.max()
                best_obj = obj_new
                best_f = f.copy()
                patience = 0
                t = min(1.0, t * 1.05)
                improved = True
                break
        
        if not improved:
            patience += 1
            t *= 0.85
            if patience > 30:
                break
        
        if iter_count % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {iter_count}, obj={best_obj:.8f}, t={t:.3f}, elapsed={elapsed:.1f}s")
    
    return best_f, best_obj

def peak_shave(f0, time_limit, alpha=0.05):
    """Directly reduce peaks by adjusting values at positions that contribute to the max."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    f /= f.sum()
    
    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        max_k = np.argmax(conv)
        max_val = conv[max_k]
        
        if max_val < 1e-10:
            break
        
        # Find which positions contribute most to the peak
        # (f*f)[max_k] = sum_j f[j]*f[max_k-j]
        j_start = max(0, max_k - n + 1)
        j_end = min(n - 1, max_k)
        contributions = f[j_start:j_end+1] * f[max_k - j_end : max_k - j_start + 1][::-1]
        total_contrib = contributions.sum()
        
        if total_contrib < 1e-15:
            break
        
        # Reduce the largest contributors
        # Move mass from high-contributing positions to low ones
        contrib_rank = np.argsort(contributions)[::-1]
        
        # Amount to reduce
        reduce_amount = alpha * max_val / total_contrib
        f_new = f.copy()
        
        for idx in contrib_rank[:n//4]:
            j = j_start + int(idx)
            f_new[j] -= reduce_amount * contributions[int(idx)]
        
        f_new = np.maximum(f_new, 0)

Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using multi-phase optimization:
1. LP-based direction finding with adaptive line search
2. Mirror descent polishing with parameter sweeping
3. Subgradient refinement
4. Targeted coordinate adjustments
5. Random restarts to escape local minima

import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def obj_from_conv(a, conv):
    """Compute objective from precomputed convolution."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    return 2.0 * n * conv.max() / (s * s)


def solve_lp_cross(f, M, n, timeout=8):
    """Solve max sum(g) s.t. g>=0, (f*g)[k]<=M for all k=0..2n-2.
    Cross-conv: (f*g)[k] = sum_i f[i]*g[k-i] (valid when 0<=k-i<=n-1).
    Returns g (sum=1) or None."""
    # A[k,i] = f[k-i] if 0<=k-i<=n-1
    # Column i: rows k=i to k=i+n-1, values f[0], f[1], ..., f[n-1]
    cols = np.repeat(np.arange(n), n)
    rows = np.concatenate([np.arange(i, i+n) for i in range(n)])
    vals = np.tile(f, n)
    
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                           options={'disp': False, 'time_limit': timeout})
    if res.success:
        g = np.maximum(res.x, 0.0)
        gs = g.sum()
        if gs > 1e-12:
            return g / gs
    return None

def mirror_step(f, conv, temp, eta):
    """One mirror descent step."""
    n = len(f)
    max_c = conv.max()
    if max_c < 1e-15:
        return f
    
    scaled = temp * conv / max_c
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-12:
        w = np.zeros(2*n-1)
        w[np.argmax(conv)] = 1.0
    else:
        w /= ws
    
    sg = np.zeros(n)
    thresh = w.max() * 0.005
    active = np.where(w > thresh)[0]
    for k in active:
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        sg[js:je+1] += w[k] * 2.0 * f[k-je:k-js+1][::-1]
    
    lf = -eta * sg
    lf -= lf.max()
    f_new = f * np.exp(lf)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-12:
        f_new = np.ones(n) / n
    else:
        f_new /= s
    return f_new

def subgradient_polish(f, time_limit, eta0=0.1, verbose=False):
    """Subgradient method with adaptive step size on the probability simplex."""
    f = np.maximum(f.astype(np.float64), 0.0)
    n = len(f)
    f /= f.sum()
    
    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    eta = eta0
    patience = 0
    
    while time.monotonic() - t0 < time_limit:
        max_c = conv.max()
        if max_c < 1e-15:
            break
        
        # Subgradient at the peak
        # Use soft max for stability
        temp = max_c * 0.05
        weights = np.exp((conv - max_c) / temp)
        weights /= weights.sum()
        
        sg = np.zeros(n)
        thresh = weights.max() * 0.01
        active = np.where(weights > thresh)[0]
        for k in active:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            sg[js:je+1] += weights[k] * 2.0 * f[k-je:k-js+1][::-1]
        
        # Projected gradient step
        sg_centered = sg - sg.mean()
        f_new = f - eta * sg_centered
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-12:
            f_new = f.copy()
        else:
            f_new /= s
        
        conv = fast_convolve(f_new)
        obj = 2.0 * n * conv.max()
        
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f_new.copy()
            f = f_new
            patience = 0
        else:
            patience += 1
            if patience % 50 == 0:
                eta *= 0.8
                if eta < 0.001:
                    break
    
    if verbose:
        print(f"    SubGrad: obj={best_obj:.8f}, el={time.monotonic()-t0:.1f}s")
    
    return best_f, best_obj


def gen_inits(n, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uni'] = np.ones(n) / n
    
    for sf in [0.1, 0.15, 0.2, 0.3]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        inits[f'g{int(sf*100)}'] = g / g.sum()
    
    r = np.arange(1, n+1, dtype=np.float64)
    inits['ramp'] = r / r.sum()
    
    cb = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    inits['cos'] = cb / cb.sum()
    
    inits['rnd'] = rng.dirichlet(np.ones(n) * 0.5)
    
    tri = np.maximum(1.0 - np.abs(x - 0.5*n) / (0.5*n), 0.01)
    inits['tri'] = tri / tri.sum()
    
    # Bimodal
    bi = np.zeros(n)
    w = max(1, n // 5)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    inits['bi'] = bi / bi.sum()
    
    return inits

def smart_coordinate_descent(f, time_limit, verbose=False):
    """Coordinate descent that targets the convolution peaks."""
    f = np.maximum(f.astype(np.float64), 0.0)
    n = len(f)
    f /= f.sum()
    
    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and no_improve < 500:
        # Find peaks
        peak_thresh = conv.max() * 0.95
        peak_positions = np.where(conv > peak_thresh)[0]
        
        # Find "safe" positions (those not contributing to any peak)
        # A position j contributes to peak k if |k-j| < n
        # Safe positions are those that don't contribute to any peak
        contributing = np.zeros(n, dtype=bool)
        for k in peak_positions:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            contributing[js:je+1] = True
        
        safe_positions = np.where(~contributing)[0]
        if len(safe_positions) == 0:
            # All positions contribute to peaks; use the least-contributing ones
            contrib_count = np.zeros(n)
            for k in peak_positions:
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                contrib_count[js:je+1] += 1
            safe_positions = np.where(contrib_count == contrib_count.min())[0]
        
        # For the main peak, try to reduce it by moving mass from contributing to safe positions
        main_peak = int(np.argmax(conv))
        js = max(0, main_peak - n + 1)
        je = min(n - 1, main_peak)
        
        # The contribution from position j to the main peak is f[j] * f[main_peak - j]
        # We want to reduce this
        # Strategy: move a bit of mass from the highest-contributing position to a safe position
        
        contribs = f[js:je+1] * f[main_peak - je : main_peak - js + 1][::-1]
        
        if contribs.max() < 1e-15:
            no_improve += 1
            continue
        
        # Find the position with the highest contribution
        max_idx = int(np.argmax(contribs))
        j_from = js + max_idx
        
        # Find a safe position to move mass to
        # Prefer positions that are "far" from all peaks
        if len(safe_positions) > 0:
            # Pick the safe position with the smallest value (to not create new peaks)
            j_to = safe_positions[np.argmin(f[safe_positions])]
        else:
            j_to = (j_from + n // 2) % n  # Move to opposite side
        
        if j_from == j_to:
            no_improve += 1
            continue
        
        # Try a small transfer
        delta = 0.001 * f[j_from]
        for scale in [1.0, 0.5, 0.2, 0.1]:
            d = delta * scale
            if f[j_from] - d < 0:
                continue
            
            f[j_from] -= d
            f[j_to] += d
            f = np.maximum(f, 0)
            f /= f.sum()
            
            conv = fast_convolve(f)
            obj = 2.0 * n * conv.max()
            
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
                break
            else:
                # Undo
                f[j_from] += d
                f[j_to] -= d
                f = np.maximum(f, 0)
                f /= f.sum()
        
        else:
            no_improve += 1
    
    if verbose:
        print(f"    CoordDesc: obj={best_obj:.8f}, el={time.monotonic()-t0:.1f}s")
    
    return best_f, best_obj

At the optimum (KKT conditions), the following should hold:
- For all `k` where `(a*a)[k] = max`: the "gradient" of `(a*a)[k]` with respect to `a` should be in the normal cone of the simplex at `a`.
- For all `j` where `a[j] > 0`: the derivative of the Lagrangian should be zero.
- For all `j` where `a[j] = 0`: the derivative should be non-negative (we shouldn't want to increase `a[j]`).

This gives us a system of equations that characterizes the optimum. We could try to solve this system numerically.

The KKT conditions (with `sum(a) = 1`, `a >= 0`):
Let `K* = {k : (a*a)[k] = M}` be the set of active peaks, where `M = max_k (a*a)[k]`.
Let `J+ = {j : a[j] > 0}` be the support of `a`.

For `j in J+`: sum_{k in K*} lambda_k * d(a*a)[k]/da[j] = nu (Lagrange multiplier for sum constraint)
For `j not in J+`: sum_{k in K*} lambda_k * d(a*a)[k]/da[j] >= nu

Where `lambda_k >= 0` and `sum_{k in K*} lambda_k = 1` (normalization of the subgradient).

1. Given `a`, compute `K*` and the subgradient.

import scipy.sparse as sparse
import time


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def solve_lp_cross(f, M, n, timeout=8):
    """Solve max sum(g) s.t. g>=0, (f*g)[k]<=M for all k. Returns g (sum=1) or None."""
    cols = np.repeat(np.arange(n), n)
    rows = np.concatenate([np.arange(i, i + n) for i in range(n)])
    vals = np.tile(f, n)
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                           options={'disp': False, 'time_limit': timeout})
    if res.success:
        g = np.maximum(res.x, 0.0)
        gs = g.sum()
        if gs > 1e-12:
            return g / gs
    return None


def lp_iterate(f, time_limit, lp_timeout=8, verbose=False):
    """LP-based alternating optimization. Returns (best_f, best_normalized_obj)."""
    f = np.maximum(f.astype(np.float64), 0.0)
    n = len(f)
    s = f.sum()
    if s < 1e-12:
        f = np.ones(n) / n
    else:
        f = f / s
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()
    t0 = time.monotonic()
    t = 0.5
    patience = 0
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        g = solve_lp_cross(f, M, n, timeout=lp_timeout)
        if g is None:
            break
        if np.max(np.abs(g - f)) < 1e-12:
            break
        improved = False
        for t_try in [t, 0.7 * t, 0.5 * t, 0.3 * t, 0.15 * t, 0.05 * t]:
            f_new = (1.0 - t_try) * f + t_try * g
            f_new = np.maximum(f_new, 0.0)
            sn = f_new.sum()
            if sn < 1e-12:
                continue
            f_new /= sn
            conv_new = fast_convolve(f_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < best_obj - 1e-13:
                f = f_new
                M = conv_new.max()
                best_obj = obj_new
                best_f = f.copy()
                patience = 0
                t = min(1.0, t * 1.03)
                improved = True
                break
        if not improved:
            patience += 1
            t *= 0.85
            if patience > 40:
                break
        if verbose and it % 20 == 0:
            print(f"    LP it={it}, obj={best_obj:.8f}, t={t:.3f}, el={time.monotonic()-t0:.1f}s")
    return best_f, best_obj


def mirror_step(f, conv, temp, eta):
    """One mirror descent step on probability simplex."""
    n = len(f)
    max_c = conv.max()
    if max_c < 1e-15:
        return f
    scaled = temp * conv / max_c
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-12:
        w = np.zeros(2 * n - 1)
        w[np.argmax(conv)] = 1.0
    else:
        w /= ws
    sg = np.zeros(n)
    thresh = w.max() * 0.005
    active = np.where(w > thresh)[0]
    for k in active:
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        sg[js:je + 1] += w[k] * 2.0 * f[k - je:k - js + 1][::-1]
    lf = -eta * sg
    lf -= lf.max()
    f_new = f * np.exp(lf)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-12:
        f_new = np.ones(n) / n
    else:
        f_new /= s
    return f_new


def mirror_polish(f, time_limit, temp=50.0, eta=0.3, verbose=False):
    """Mirror descent polishing loop."""
    f = np.maximum(f.astype(np.float64), 0.0)
    n = len(f)
    f /= f.sum()
    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    patience = 0
    cur_eta = eta
    cur_temp = temp
    while time.monotonic() - t0 < time_limit:
        f = mirror_step(f, conv, cur_temp, cur_eta)
        conv = fast_convolve(f)
        obj = 2.0 * n * conv.max()
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f.copy()
            patience = 0
        else:
            patience += 1
            if patience % 30 == 0:
                cur_eta *= 0.7
                cur_temp *= 1.3
                if cur_eta < 0.005:
                    cur_eta = eta * 0.3
                    cur_temp = temp
            if patience > 300:
                break
    if verbose:
        print(f"    Mirror(t={temp:.0f},e={eta:.2f}): obj={best_obj:.8f}, el={time.monotonic()-t0:.1f}s")
    return best_f, best_obj


def subgradient_polish(f, time_limit, eta0=0.05, verbose=False):
    """Subgradient method with adaptive step on simplex."""
    f = np.maximum(f.astype(np.float64), 0.0)
    n = len(f)
    f /= f.sum()
    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    eta = eta0
    patience = 0
    while time.monotonic() - t0 < time_limit:
        max_c = conv.max()
        if max_c < 1e-15:
            break
        temp_s = max_c * 0.05
        weights = np.exp((conv - max_c) / temp_s)
        weights /= weights.sum()
        sg = np.zeros(n)
        thresh = weights.max() * 0.01
        active = np.where(weights > thresh)[0]
        for k in active:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            sg[js:je + 1] += weights[k] * 2.0 * f[k - je:k - js + 1][::-1]
        sg_c = sg - sg.mean()
        f_new = f - eta * sg_c
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-12:
            f_new = f.copy()
        else:
            f_new /= s
        conv = fast_convolve(f_new)
        obj = 2.0 * n * conv.max()
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f_new.copy()
            f = f_new
            patience = 0
        else:
            patience += 1
            if patience % 50 == 0:
                eta *= 0.8
                if eta < 0.0005:
                    break
    if verbose:
        print(f"    SubGrad(eta0={eta0}): obj={best_obj:.8f}, el={time.monotonic()-t0:.1f}s")
    return best_f, best_obj


def peak_shave(f, time_limit, verbose=False):
    """Targeted peak shaving: move mass from peak-contributing to safe positions."""
    f = np.maximum(f.astype(np.float64), 0.0)
    n = len(f)
    f /= f.sum()
    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    alpha = 0.02
    while time.monotonic() - t0 < time_limit and no_improve < 200:
        peak_thresh = conv.max() * 0.95
        peak_positions = np.where(conv > peak_thresh)[0]
        contributing = np.zeros(n, dtype=np.int32)
        for k in peak_positions:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            contributing[js:je + 1] += 1
        safe_mask = contributing == 0
        if not safe_mask.any():
            min_c = contributing.min()
            safe_mask = contributing == min_c
        safe_positions = np.where(safe_mask)[0]
        if len(safe_positions) == 0:
            safe_positions = np.array([0])
        main_peak = int(np.argmax(conv))
        js = max(0, main_peak - n + 1)
        je = min(n - 1, main_peak)
        contribs = f[js:je + 1] * f[main_peak - je:main_peak - js + 1][::-1]
        if contribs.max() < 1e-15:
            break
        j_from = js + int(np.argmax(contribs))
        safe_vals = f[safe_positions]
        j_to = safe_positions[np.argmin(safe_vals)]
        if j_from == j_to:
            no_improve += 1
            continue
        improved = False
        for scale in [alpha, alpha * 0.5, alpha * 0.2, alpha * 0.1]:
            d = scale * f[j_from]
            if d < 0 or f[j_from] - d < 0:
                continue
            f[j_from] -= d
            f[j_to] += d
            f = np.maximum(f, 0)
            f /= f.sum()
            conv = fast_convolve(f)
            obj = 2.0 * n * conv.max()
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
                improved = True
                break
            else:
                f[j_from] += d
                f[j_to] -= d
                f = np.maximum(f, 0)
                f /= f.sum()
        if not improved:
            no_improve += 1
            alpha *= 0.85
            if alpha < 0.001:
                break
    if verbose:
        print(f"    PeakShave: obj={best_obj:.8f}, el={time.monotonic()-t0:.1f}s")
    return best_f, best_obj


def gen_inits(n, rng):
    """Generate diverse initializations summing to 1."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    inits['uni'] = np.ones(n) / n
    for sf in [0.1, 0.15, 0.2, 0.3]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        inits[f'g{int(sf * 100)}'] = g / g.sum()
    r = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = r / r.sum()
    cb = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    inits['cos'] = cb / cb.sum()
    inits['rnd'] = rng.dirichlet(np.ones(n) * 0.5)
    tri = np.maximum(1.0 - np.abs(x - 0.5 * n) / (0.5 * n), 0.01)
    inits['tri'] = tri / tri.sum()
    bi = np.zeros(n)
    w = max(1, n // 5)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    inits['bi'] = bi / bi.sum()
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Main entry: multi-phase optimization combining LP, mirror descent, subgradient,
    and peak shaving. Focuses on parent sequence with parameter sweeps and restarts."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 20.0)
    best_seq = None
    best_score = float('inf')

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_conv = fast_convolve(parent)
            print(f"Parent: n={len(p)}, norm_obj={2.0*len(p)*p_conv.max():.8f}")
    except Exception:
        pass

    # Phase 1: Quick scan (150s)
    print("=== Phase 1: Quick scan ===")
    t_p1 = time.monotonic() + 150
    results = []

    scan_lengths = [512]
    if parent is not None and len(parent) != 512:
        scan_lengths.insert(0, len(parent))

    for n in scan_lengths:
        if time.monotonic() > t_p1:
            break
        inits = gen_inits(n, rng)
        if parent is not None:
            if parent.size == n:
                inits['par'] = parent.copy()
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
                pp = np.maximum(pp, 0.0)
                if pp.sum() > 1e-10:
                    inits['par'] = pp / pp.sum()
        time_per = max(3.0, (t_p1 - time.monotonic()) / max(1, len(inits) * len(scan_lengths)))
        for name, a0 in inits.items():
            if time.monotonic() > t_p1 - 3:
                break
            ta = min(time_per, t_p1 - time.monotonic() - 3)
            if ta < 2:
                break
            f_opt, _ = lp_iterate(a0, time_limit=ta, lp_timeout=4)
            score = float(evaluate_sequence(f_opt.tolist()))
            if np.isfinite(score):
                results.append((score, n, name, f_opt))
                if score < best_score:
                    best_score = score
                    best_seq = f_opt.copy()
                    print(f"  n={n:4d} {name:8s} score={score:.6f}")
    print(f"Phase 1 best: {best_score:.6f}")

    # Phase 2: Deep LP on top candidates (400s)
    print("=== Phase 2: Deep LP ===")
    t_p2 = time.monotonic() + 400
    results.sort(key=lambda x: x[0])
    top = results[:4]
    for idx, (sc, n, name, f) in enumerate(top):
        if time.monotonic() > t_p2 - 10:
            break
        ta = min((t_p2 - time.monotonic() - 10) / max(1, len(top) - idx), 120.0)
        if ta < 5:
            break
        f2, _ = lp_iterate(f, time_limit=ta, lp_timeout=10, verbose=True)
        sc2 = float(evaluate_sequence(f2.tolist()))
        if np.isfinite(sc2) and sc2 < best_score:
            best_score = sc2
            best_seq = f2.copy()
            print(f"  DeepLP[{idx}] n={n} {name} score={sc2:.6f}")

    # Phase 3: Multi-parameter polishing (600s)
    print("=== Phase 3: Polishing ===")
    t_p3 = time.monotonic() + 600
    if best_seq is not None:
        current = best_seq.copy()
        # Mirror descent with sweep
        mirror_params = [
            (20.0, 0.5), (40.0, 0.35), (60.0, 0.25), (100.0, 0.18),
            (150.0, 0.12), (250.0, 0.08), (400.0, 0.05), (600.0, 0.03)
        ]
        for mi, (tp, eta_m) in enumerate(mirror_params):
            if time.monotonic() > t_p3 - 20:
                break
            ta = min((t_p3 - time.monotonic() - 20) / max(1, len(mirror_params) - mi), 70.0)
            if ta < 3:
                break
            f3, _ = mirror_polish(current, time_limit=ta, temp=tp, eta=eta_m, verbose=True)
            sc3 = float(evaluate_sequence(f3.tolist()))
            if np.isfinite(sc3) and sc3 < best_score:
                best_score = sc3
                best_seq = f3.copy()
                current = f3.copy()
                print(f"  -> score={sc3:.6f}")
            # Subgradient between mirrors
            if time.monotonic() < t_p3 - 5:
                ta2 = min(15.0, t_p3 - time.monotonic() - 5)
                f4, _ = subgradient_polish(current, time_limit=ta2, eta0=0.03, verbose=False)
                sc4 = float(evaluate_sequence(f4.tolist()))
                if np.isfinite(sc4) and sc4 < best_score:
                    best_score = sc4
                    best_seq = f4.copy()
                    current = f4.copy()
                    print(f"  -> subgrad score={sc4:.6f}")

    # Phase 4: Peak shaving + more LP (350s)
    print("=== Phase 4: Peak shave + LP ===")
    t_p4 = time.monotonic() + 350
    if best_seq is not None:
        current = best_seq.copy()
        # Alternating peak shave and LP
        for round_i in range(6):
            if time.monotonic() > t_p4 - 10:
                break
            ta_ps = min(40.0, (t_p4 - time.monotonic() - 10) / max(1, 6 - round_i))
            if ta_ps < 3:
                break
            f5, _ = peak_shave(current, time_limit=ta_ps, verbose=True)
            sc5 = float(evaluate_sequence(f5.tolist()))
            if np.isfinite(sc5) and sc5 < best_score:
                best_score = sc5
                best_seq = f5.copy()
                current = f5.copy()
                print(f"  Shave r{round_i} score={sc5:.6f}")
            if time.monotonic() < t_p4 - 5:
                ta_lp = min(30.0, t_p4 - time.monotonic() - 5)
                f6, _ = lp_iterate(current, time_limit=ta_lp, lp_timeout=8)
                sc6 = float(evaluate_sequence(f6.tolist()))
                if np.isfinite(sc6) and sc6 < best_score:
                    best_score = sc6
                    best_seq = f6.copy()
                    current = f6.copy()
                    print(f"  LP r{round_i} score={sc6:.6f}")

    # Phase 5: Random restarts + final polish (300s)
    print("=== Phase 5: Restarts ===")
    t_p5 = deadline
    if best_seq is not None:
        current = best_seq.copy()
        n_best = len(current)
        for ri in range(10):
            if time.monotonic() > t_p5 - 10:
                break
            ta_tot = min((t_p5 - time.monotonic() - 10) / max(1, 10 - ri), 50.0)
            if ta_tot < 5:
                break
            noise_scale = 0.02 + 0.01 * ri
            noise = rng.normal(0, noise_scale, size=n_best)
            f_pert = current + noise
            f_pert = np.maximum(f_pert, 0.0)
            f_pert /= f_pert.sum()
            ta_lp = ta_tot * 0.5
            f_r, _ = lp_iterate(f_pert, time_limit=ta_lp, lp_timeout=5)
            sc_r = float(evaluate_sequence(f_r.tolist()))
            if time.monotonic() < t_p5 - 5:
                ta_m = min(15.0, t_p5 - time.monotonic() - 5)
                f_r2, _ = mirror_polish(f_r, time_limit=ta_m, temp=50, eta=0.2)
                sc_r2 = float(evaluate_sequence(f_r2.tolist()))
            else:
                f_r2, sc_r2 = f_r, sc_r
            best_local = min(sc_r, sc_r2) if np.isfinite(sc_r) and np.isfinite(sc_r2) else (sc_r if np.isfinite(sc_r) else sc_r2)
            if np.isfinite(best_local) and best_local < best_score:
                best_score = best_local
                best_seq = (f_r if sc_r <= sc_r2 else f_r2).copy()
                current = best_seq.copy()
                print(f"  Restart {ri} score={best_local:.6f}")

        # Final polish
        if time.monotonic() < t_p5 - 5:
            ta_f = t_p5 - time.monotonic() - 5
            f_fin, _ = mirror_polish(current, time_limit=ta_f * 0.5, temp=80, eta=0.15)
            sc_fin = float(evaluate_sequence(f_fin.tolist()))
            if np.isfinite(sc_fin) and sc_fin < best_score:
                best_score = sc_fin
                best_seq = f_fin.copy()
                print(f"  Final mirror score={sc_fin:.6f}")
            if time.monotonic() < t_p5 - 3:
                ta_f2 = t_p5 - time.monotonic() - 3
                f_fin2, _ = lp_iterate(best_seq, time_limit=ta_f2, lp_timeout=10)
                sc_fin2 = float(evaluate_sequence(f_fin2.tolist()))
                if np.isfinite(sc_fin2) and sc_fin2 < best_score:
                    best_score = sc_fin2
                    best_seq = f_fin2.copy()
                    print(f"  Final LP score={sc_fin2:.6f}")

    if best_seq is None:
        best_seq = np.ones(512)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL: n={len(best_seq)}, score={best_score:.6f}")
    return [float(x) for x in best_seq]
