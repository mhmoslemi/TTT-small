import time

    n = len(a)

    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')

def normalize(a, target_sum):
    """Normalize sequence to have target_sum."""
    S = a.sum()
    if S < 1e-15:
        return a.copy()
    return a * (target_sum / S)


def soft_max_weights(c, tau):
    """Compute soft-max weights over autocorrelation values."""
    c_max = c.max()
    shifted = tau * (c - c_max)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -500, 0)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    return w


def soft_max_gradient(a, tau):
    """Compute gradient of soft-max autocorrelation w.r.t. a using FFT."""
    n = len(a)
    c = np.convolve(a, a)
    w = soft_max_weights(c, tau)
    # grad[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
    # Use cross-correlation via FFT for speed
    # This equals 2 * (w cross-correlated with a) at the valid positions
    a_rev = a[::-1]

    grad = 2.0 * result[n-1:n-1+n]

            s = compute_obj(a_new)
            if s < best_score - 1e-10:
                best_score = s
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 20:
                    # Reset momentum, increase tau
                    momentum *= 0.5
                    no_improve = 0
        
        a = a_new
    
    return best_a, best_score


def lp_phase(a, deadline, max_iters=30):
    """
    LP-based refinement: iteratively solve LP and blend.
    Returns (best_a, best_score).
    """
    n = len(a)
    if n > 512:
        return a, compute_obj(a)
    
    S_t = np.sqrt(2.0 * n)
    a = normalize(a.copy(), S_t)
    best_a = a.copy()
    best_score = compute_obj(a)
    
    for it in range(max_iters):
        if time.monotonic() > deadline - 20:
            break
        
        c = np.convolve(a, a)
        M = float(c.max())
        
        g = lp_refine(a, M)
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-6:
            break
        g = normalize(g, S_t)
        
        # Try multiple blend factors
        improved = False
        for t_blend in [0.6, 0.4, 0.25, 0.15, 0.08, 0.04]:
            a_new = (1.0 - t_blend) * a + t_blend * g
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize(a_new, S_t)
            s = compute_obj(a_new)
            if s < best_score - 1e-10:
                best_score = s
                best_a = a_new.copy()
                a = a_new
                improved = True
                break
        
        if not improved:
            # Small perturbation to escape
            pert = np.random.default_rng(it).normal(0, 0.02, n)
            a = np.maximum(a + pert, 0)
            a = normalize(a, S_t)
            s = compute_obj(a)
            if s < best_score:
                best_score = s
                best_a = a.copy()
            else:
                break
    
    return best_a, best_score


def coordinate_descent(a, deadline, max_rounds=5):
    """
    Coordinate descent: optimize one coordinate at a time via golden section.
    """
    n = len(a)
    if n > 256:
        return a, compute_obj(a)
    
    S_t = np.sqrt(2.0 * n)
    a = normalize(a.copy(), S_t)
    best_score = compute_obj(a)
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline - 10:
            break
        for i in range(n):
            if time.monotonic() > deadline - 10:
                break
            orig = a[i]
            # Try a range of values
            candidates = [0.0, orig * 0.3, orig * 0.5, orig * 0.7, orig * 1.3, orig * 1.7, orig * 2.0]
            best_val = orig
            best_s = best_score
            for val in candidates:
                a[i] = val
                a_tmp = normalize(a.copy(), S_t)
                s = compute_obj(a_tmp)
                if s < best_s:
                    best_s = s
                    best_val = val
            a[i] = best_val
            a = normalize(a, S_t)
            if best_s < best_score:
                best_score = best_s
    
    return a, best_score


def hard_max_phase(a, deadline, phase_duration):
    """Phase using hard-max gradient (single peak)."""
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = normalize(a.copy(), S_t)
    best_a = a.copy()
    best_score = compute_obj(a)
    momentum = np.zeros(n)
    
    end_time = min(deadline, time.monotonic() + phase_duration)
    it = 0
    no_improve = 0
    
    while time.monotonic() < end_time:
        it += 1
        lr = max(0.01, 0.1 * (0.999 ** it))
        
        grad = hard_max_gradient(a)
        grad -= grad.mean()
        
        momentum = 0.9 * momentum - lr * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new, S_t)
        
        if it % 100 == 0:
            s = compute_obj(a_new)
            if s < best_score - 1e-10:
                best_score = s
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 15:
                    momentum *= 0.3
                    no_improve = 0
        
        a = a_new
    
    return best_a, best_score


def optimize_full(a0, deadline):
    """
    Full multi-phase optimization from a starting point.
    Phase 1: Soft-max gradient descent (broad exploration)
    Phase 2: Hard-max gradient descent (focused refinement)
    Phase 3: LP-based refinement (structured improvement)
    Phase 4: Coordinate descent (fine polishing)
    """
    n = len(a0)
    total_time = deadline - time.monotonic()
    if total_time < 30:
        return a0, compute_obj(a0)
    
    # Allocate time to phases
    t1 = min(total_time * 0.4, 300)  # Soft-max
    t2 = min(total_time * 0.2, 150)  # Hard-max
    t3 = min(total_time * 0.25, 200)  # LP
    t4 = max(20, total_time * 0.1)  # Coordinate descent
    
    # Phase 1: Soft-max gradient descent
    a, score = gradient_descent_phase(a0, deadline, t1)
    best_a = a.copy()
    best_score = score
    print(f"      Phase 1 (soft-max): score={best_score:.8f}")
    
    # Phase 2: Hard-max gradient descent
    if time.monotonic() < deadline - t3 - t4 - 20:
        a2, score2 = hard_max_phase(best_a, deadline, t2)
        if score2 < best_score:
            best_a = a2.copy()
            best_score = score2
            print(f"      Phase 2 (hard-max): score={best_score:.8f}")
    
    # Phase 3: LP refinement
    if n <= 512 and time.monotonic() < deadline - t4 - 10:
        a3, score3 = lp_phase(best_a, deadline - t4 - 10)
        if score3 < best_score:
            best_a = a3.copy()
            best_score = score3
            print(f"      Phase 3 (LP): score={best_score:.8f}")
    
    # Phase 4: Coordinate descent
    if n <= 256 and time.monotonic() < deadline - 10:
        a4, score4 = coordinate_descent(best_a, deadline)
        if score4 < best_score:
            best_a = a4.copy()
            best_score = score4
            print(f"      Phase 4 (coord): score={best_score:.8f}")
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Multi-start optimization with diverse initializations.
    Returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)
    
    best_seq = None
    best_score = float('inf')
    
    def track(a, label):
        nonlocal best_seq, best_score
        s = compute_obj(a)
        if s < best_score:
            best_score = s
            best_seq = a.copy()
            print(f"  [{label}] NEW BEST: score={s:.8f}, n={len(a)}")
        return s
    
    print(f"=== Starting optimization, budget={budget_s}s ===")
    
    # --- Try uniform sequences (theoretical optimum = 2.0) ---
    for n in [128, 256, 384, 512, 640]:
        a_unif = np.ones(n)
        s = track(a_unif, f"uniform n={n}")
    
    # --- Try provided sequence ---
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        if gs.size >= 8 and np.all(np.isfinite(gs)):
            s = track(gs, "provided")
            print(f"  Provided sequence: n={len(gs)}, score={s:.8f}")
    except NameError:
        pass
    
    # --- Priority 1: Optimize from provided sequence (if good) ---
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        if gs.size >= 8 and np.all(np.isfinite(gs)):
            if time.monotonic() < deadline - 120:
                print(f"\n  Optimizing from provided sequence (n={len(gs)})...")
                t_alloc = min(400.0, deadline - time.monotonic() - 120)
                a_opt, s_opt = optimize_full(gs, time.monotonic() + t_alloc)
                track(a_opt, "provided_opt")
    except NameError:
        pass
    
    # --- Priority 2: Optimize from diverse random starts ---
    lengths = [256, 384, 512]
    
    for n in lengths:
        if time.monotonic() > deadline - 90:
            break
        
        num_starts = 2
        for trial in range(num_starts):
            if time.monotonic() > deadline - 60:
                break
            
            # Diverse initialization
            if trial == 0:
                a0 = rng.random(n) * 0.5 + 0.25
            else:
                # Cosine-modulated start
                i = np.arange(n, dtype=np.float64)
                a0 = 0.5 + 0.3 * np.cos(2 * np.pi * i * (trial + 2) / n)
                a0 = np.maximum(a0, 0.01)
            
            t_alloc = min(250.0, deadline - time.monotonic() - 60)
            a_opt, s_opt = optimize_full(a0, time.monotonic() + t_alloc)
            track(a_opt, f"rand n={n} t={trial}")
    
    # --- Priority 3: Focused optimization on best sequence found ---
    if best_seq is not None and time.monotonic() < deadline - 60:
        print(f"\n  Final refinement on best sequence (score={best_score:.8f})...")
        remaining = deadline - time.monotonic() - 20
        a_final, s_final = optimize_full(best_seq, time.monotonic() + remaining)
        track(a_final, "final_refine")
    
    # --- Final validation ---
    if best_seq is not None:
        final_arr = np.maximum(best_seq, 0.0)
        final_arr = np.minimum(final_arr, 1000.0)
        final_score = compute_obj(final_arr)
        print(f"\n=== Final: n={len(final_arr)}, score={final_score:.8f} ===")
        
        # Verify with evaluate_sequence
        try:
            eval_arr = [float(x) for x in final_arr]
            eval_score = evaluate_sequence(eval_arr)
            print(f"  evaluate_sequence: {eval_score:.8f}")
        except Exception as e:
            print(f"  evaluate_sequence error: {e}")
        
        return [float(x) for x in final_arr]
    
    # Fallback
    print("  Falling back to uniform n=256")
    return [1.0] * 256
