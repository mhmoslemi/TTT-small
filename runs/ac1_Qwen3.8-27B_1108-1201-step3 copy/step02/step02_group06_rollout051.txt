3. Find the largest k such that t_k < y_(k) (or t_k <= y_(k) and t_k > y_(k+1))
4. t = t_k

This is O(n log n) due to sorting.

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

def project_simplex(y):
    """Project y onto {a >= 0, sum(a) = 1}."""
    n = len(y)
    # Sort in decreasing order
    u = np.sort(y)[::-1]
    # Find rho: largest k such that 1 + k*u[k] > sum(u[:k+1])
    # (using 0-indexed: u[0] >= u[1] >= ...)
    cssv = np.cumsum(u)
    cond = 1.0 + np.arange(1, n+1) * u > cssv
    rho = np.sum(cond)
    if rho == 0:
        # This shouldn't happen for valid input, but handle gracefully
        return np.ones(n) / n
    theta = (cssv[rho-1] - 1.0) / rho
    return np.maximum(y - theta, 0.0)

def smooth_max_grad(a, tau):
    """Compute smoothed max of conv(a,a) and its gradient."""
    n = len(a)
    conv = fft_convolve(a, a)
    conv_max = np.max(conv)
    shifted = tau * (conv - conv_max)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp
    F_smooth = conv_max + np.log(sum_exp) / tau
    
    # Compute gradient
    # grad[i] = 2 * c[i] where c[i] = sum_l a[l] * w[i+l]
    # Use matrix approach
    I, L = np.meshgrid(np.arange(n), np.arange(n), indexing='ij')
    idx = I + L
    valid = (idx <= 2*n - 2)
    W = np.where(valid, w[idx], 0.0)
    c = W @ a
    grad = 2.0 * c
    
    return F_smooth, grad, w

def smooth_subgradient_optimize(a_init, time_limit, tau_start=10.0, tau_end=200.0, verbose=False):
    """Optimize using smoothed subgradient method with projection."""
    a = a_init.astype(np.float64).copy()
    a = a / np.sum(a)  # normalize to sum = 1
    n = len(a)
    
    best_a = a.copy()
    best_score = local_score(a)
    
    t0 = time.monotonic()
    step_count = 0
    # Use a fixed step size scheme
    # eta = c / sqrt(t)
    # Choose c based on initial gradient norm
    
    # First, compute initial gradient to estimate scale
    _, grad0, _ = smooth_max_grad(a, tau_start)
    g0_norm = np.linalg.norm(grad0)
    eta_c = 1.0 / max(g0_norm, 1e-6)  # step size constant
    
    # Track best for line search
    best_F = best_score / (2 * n)  # = max(conv) for sum=1
    F_history = []
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        # Anneal tau from tau_start to tau_end
        tau = tau_start * (tau_end / tau_start) ** elapsed_frac
        
        F_smooth, grad, w = smooth_max_grad(a, tau)
        
        # Step size: adaptive
        eta = eta_c / np.sqrt(step_count + 1)
        
        # Take step
        a_new = a - eta * grad
        a_new = project_simplex(a_new)
        
        # Compute actual score
        score = local_score(a_new)
        
        # Line search: if score is much worse, try smaller step
        if score > best_score * 1.5:
            # Try smaller step
            for factor in [0.5, 0.25, 0.1]:
                a_try = a - eta * factor * grad
                a_try = project_simplex(a_try)
                score_try = local_score(a_try)
                if score_try < score:
                    a_new = a_try
                    score = score_try
                    break
        
        a = a_new
        step_count += 1
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
            best_F = score / (2 * n)
        
        # Adapt step size if stuck
        if step_count % 100 == 0 and verbose:
            print(f"      Step {step_count}, tau={tau:.1f}, score={score:.6f}, best={best_score:.6f}")
        
        # Periodic restart with perturbation
        if step_count % 500 == 0 and score > best_score * 1.1:
            a = best_a + 0.01 * np.random.standard_normal(n)
            a = np.clip(a, 0, None)
            a = a / np.sum(a)
            eta_c *= 0.5
    
    return best_a, best_score

Actually, a better approach might be to use the **mirror descent** or **exponential weights** method, which works well with the simplex constraint.

Mirror descent with negative entropy (softmax) parameterization:
- a_i = exp(theta_i) / sum_j exp(theta_j)
- Update: theta <- theta - eta * (grad / a) (element-wise)
  (This is the mirror map: the Bregman projection)

Or equivalently, a <- a * exp(-eta * grad) / sum(a * exp(-eta * grad))

This is the **multiplicative weights** update, which is known to converge linearly for mirror descent on the simplex.

Let me think about this more carefully. The mirror descent update for minimizing f(a) over the simplex with negative entropy is:

theta_{t+1} = theta_t - eta * ∇f(a_t) / a_t  (component-wise division)

Then a_{t+1,i} = exp(theta_{t+1,i}) / sum_j exp(theta_{t+1,j})
= a_t,i * exp(-eta * grad_i) / sum_j a_t,j * exp(-eta * grad_j)

This is the "exponential tilt" update. It's equivalent to:
a_{t+1} ∝ a_t * exp(-eta * grad)

This should work well! The step size eta controls the aggressiveness.

For the smoothed objective, this becomes:
a_{t+1} ∝ a_t * exp(-eta * 2 * c_t)
where c_t[i] = sum_l a_t[l] * w_t[i+l]

This is very efficient to compute!

def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
