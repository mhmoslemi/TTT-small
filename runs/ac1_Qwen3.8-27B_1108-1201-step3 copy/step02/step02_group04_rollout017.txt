```python
def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)

    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)

def smooth_max_val(x, t):
    """Compute log-sum-exp approximation to max."""
    m = x.max()
    if t * (x[-1] - x[0]) > 700:

    n = len(a)
    conv = fast_convolve(a)
    
    # Compute softmax weights
    m = conv.max()
    shifted = t * (conv - m)
    # Clamp to avoid overflow
    shifted = np.clip(shifted, -700, 700)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    
    # Compute gradient: g[j] = 2 * sum_k w[k] * a[k-j] for 0 <= k-j < n
    # = 2 * cross_corr(w, a)[j]
    # cross_corr(w, a)[j] = sum_m w[j+m] * a[m]
    # = (w * flip(a))[j + n - 1]
    
    a_flip = a[::-1]
    fft_size = 1 << (len(w) + len(a_flip) - 1 - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    AF = np.fft.rfft(a_flip, fft_size)
    conv_wa = np.fft.irfft(W * AF, fft_size)
    # cross_corr[j] = conv_wa[j + n - 1]
    g = 2.0 * conv_wa[n-1 : 2*n-1]
    
    # Projected gradient step
    a_new = a - eta * g
    # Project onto simplex
    a_new = project_simplex(a_new)
    
    conv_new = fast_convolve(a_new)
    obj = 2.0 * n * conv_new.max()
    return a_new, obj

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    a = np.maximum(a0, 0.0).astype(np.float64)

    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t = t_start
    eta = eta_start
    patience = 0

        batch_size = 10
        for _ in range(batch_size):

                break
            a_new, obj = projected_gradient_smooth(a, t, eta)
            if obj < best_obj:
                best_obj = obj

                patience = 0
            else:
                patience += 1
            a = a_new

            if eta < 0.01:

        if iter_count % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PGD iter {iter_count}, obj={best_obj:.8f}, t={t:.1f}, eta={eta:.4f}, elapsed={elapsed:.1f}s")

def coordinate_descent(a0, time_limit, max_rounds=50):
    """Coordinate descent: optimize one element at a time."""
    a = np.maximum(a0, 0.0).astype(np.float64)
    n = len(a)
    a_sum = a.sum()
    a = a / a_sum
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    for rnd in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        
        # Random order
        order = rng.permutation(n)
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            # Try to optimize a[j] while keeping sum fixed
            # Fix all other elements, vary a[j]
            # The constraint is sum(a) = 1, so if we change a[j] by delta,
            # we need to redistribute -delta among other elements.
            # Simpler: just try a few values of a[j] and renormalize.
            
            old_val = a[j]
            for frac in [0.5, 0.7, 0.85, 1.0, 1.2, 1.3, 1.5, 2.0, 0.0, 0.1]:
                a_try = a.copy()
                a_try[j] = old_val * frac
                if a_try.sum() < 1e-10:
                    continue
                a_try = a_try / a_try.sum()
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj - 1e-12:
                    a = a_try
                    best_obj = obj_try
                    best_a = a.copy()
                    improved = True
                    break
            
            # Also try small perturbations
            for delta in [0.01, -0.01, 0.05, -0.05]:
                a_try = a.copy()
                a_try[j] = max(0, a[j] + delta / n)
                if a_try.sum() < 1e-10:
                    continue
                a_try = a_try / a_try.sum()
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj - 1e-12:
                    a = a_try
                    best_obj = obj_try
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            # Try random perturbations
            for _ in range(10):
                a_try = a + rng.normal(0, 0.01, n)
                a_try = np.maximum(a_try, 0)
                if a_try.sum() < 1e-10:
                    continue
                a_try = a_try / a_try.sum()
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj - 1e-12:
                    a = a_try
                    best_obj = obj_try
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            break
    
    return best_a, best_obj
```

Hmm, coordinate descent with this many coordinates and the expensive convolution evaluations might be too slow. Each evaluation of the convolution takes O(n log n) time.

Let me reconsider. For n=512:
- FFT convolution: ~512 * log(512) ≈ 512 * 9 ≈ 4600 operations, plus overhead. Probably ~50 microseconds.
- Coordinate descent with 512 coordinates * ~10 trials * 50 μs = ~256 ms per round.
- 50 rounds = ~13 seconds. That's feasible!

But the issue is that simple coordinate descent might not converge well. Let me think of a better approach.

**Better approach: Use the analytical structure**

For a given j, if I fix all elements except a[j], the convolution values become:
`conv[k] = conv_without_j[k] + 2*a[j]*a[k-j]` (for 0<=k-j<n, k!=2j)
`conv[k] = conv_without_j[k] + a[j]^2 + 2*a[j]*a[k-j]` (for k=2j, if in range)

Wait, this isn't quite right. Let me be more careful.

`conv[k] = sum_i a[i]*a[k-i]`

If I change a[j] to a[j]+δ:
`conv_new[k] = conv[k] + δ*(a[k-j] + a[j]) + δ^2*(1 if k=2j else 0)`
Wait, that's not right either.

`conv_new[k] = sum_i a_new[i]*a_new[k-i]`
`= conv[k] + δ*a[k-j] + δ*a[j] + δ^2*(1 if k=2j else 0)`
Hmm, let me be more careful.

The terms involving a[j] in conv[k] are:
- i=j: a[j]*a[k-j] (contributes a[j]*a[k-j])
- i=k-j: a[k-j]*a[j] (contributes a[k-j]*a[j])
- If j=k-j (k=2j): only one term a[j]^2

So conv[k] = (terms not involving a[j]) + a[j]*a[k-j] + a[k-j]*a[j] (if j≠k-j)
= (terms not involving a[j]) + 2*a[j]*a[k-j] (if j≠k-j)
= (terms not involving a[j]) + a[j]^2 (if j=k-j)

If a[j] -> a[j]+δ:
conv_new[k] = (terms not involving a[j]) + 2*(a[j]+δ)*a[k-j] (if j≠k-j)
= conv[k] + 2δ*a[k-j]

conv_new[k] = (terms not involving a[j]) + (a[j]+δ)^2 (if j=k-j)
= conv[k] + 2δ*a[j] + δ^2

So in both cases: conv_new[k] = conv[k] + 2δ*a[k-j] + δ^2*(1 if k=2j else 0)

Let me define: `linear_part[k] = 2δ*a[k-j]` (with a[k-j]=0 if k-j not in [0,n-1])
and `quad_part[k] = δ^2` if k=2j else 0.

So conv_new[k] = conv[k] + linear_part[k] + quad_part[k]

The objective is to minimize max_k conv_new[k] subject to sum(a_new) = 1.
If I only change a[j], the sum constraint means I need to adjust the other elements. This makes it complicated.

Alternative: normalize after each step. So I change a[j] and then normalize. The normalization doesn't change the "shape" much for small changes.

Actually, let me just use the PGD approach with the smooth max, which is cleaner.

```python
def lbfgs_optimize(a0, time_limit, t_start=10.0, t_end=5000.0):

            a_local = a_local / a_local.sum()
            conv = fast_convolve(a_local)
            return 2.0 * n * smooth_max_val(conv, t)
        
        res = optimize.minimize(obj, b, method='L-BFGS-B', 
                                options={'maxiter': 50, 'gtol': 1e-8, 'disp': False})

        a_local = np.exp(b_opt - b_opt.max())
        a_local = a_local / a_local.sum()

        obj_val = 2.0 * n * conv.max()

            best_obj = obj_val
            best_a = a_local.copy()

        b = b_opt  # Continue from current point
```

This is a continuation method: start with smooth (small t) and gradually sharpen (increase t).

The issue is that L-BFGS-B with 512 dimensions and expensive function evaluations (each eval is an FFT) might be slow. Each function eval is ~50μs, and L-BFGS might need several hundred evals per phase. With 50 maxiter * ~5 evals per iter = 250 evals per phase, that's ~12ms per phase. With 10 phases, that's ~120ms total. Should be fine!

Let me combine multiple approaches:

1. **PGD with smooth max** (fast, good for initial optimization)
2. **L-BFGS with continuation** (more precise local optimization)
3. **LP-guided** (global direction, from previous work)
4. **Random restarts** (diversity)

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase strategy:
- Phase 1: L-BFGS continuation with smooth max (log-sum-exp) on softmax parameterization
- Phase 2: Projected gradient descent with annealed temperature
- Phase 3: LP-guided alternating optimization
- Phase 4: Coordinate descent polish
- Multiple diverse initializations including previous best
"""

import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def smooth_max_val(x, t):
    m = x.max()
    shifted = t * (x - m)
    shifted = np.clip(shifted, -700, 700)
    return m + np.log(np.sum(np.exp(shifted))) / t

def smooth_max_weights(x, t):
    """Return softmax weights for the smooth max."""
    m = x.max()
    shifted = t * (x - m)
    shifted = np.clip(shifted, -700, 700)
    e = np.exp(shifted)
    return e / e.sum()

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def compute_grad_conv(a, w):
    """Compute gradient of sum_k w[k]*conv[k] w.r.t. a.
    grad[j] = 2 * sum_k w[k] * a[k-j] for 0<=k-j<n"""
    n = len(a)
    a_flip = a[::-1]
    total_len = len(w) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    AF = np.fft.rfft(a_flip, fft_size)
    result = np.fft.irfft(W * AF, fft_size)
    # cross_corr[j] = result[j + n - 1] for j=0..n-1
    return 2.0 * result[n-1 : 2*n-1]

def lbfgs_continuation(a0, time_limit, t_list=None, max_iter_per_phase=100):
    """L-BFGS with continuation on temperature."""
    a = np.maximum(a0, 0.0).astype(np.float64)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    if t_list is None:
        t_list = [5, 10, 20, 50, 100, 200, 500, 1000, 2000, 4000, 8000]
    
    b = np.log(a + 1e-30)
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    prev_obj = np.inf
    
    for t in t_list:
        if time.monotonic() - t0 > time_limit:
            break
        
        phase_start = time.monotonic()
        
        def obj_and_box(b):
            a_local = np.exp(b - b.max())
            a_local = a_local / a_local.sum()
            conv = fast_convolve(a_local)
            return 2.0 * n * smooth_max_val(conv, t)
        
        res = optimize.minimize(obj_and_box, b, method='L-BFGS-B',
                                options={'maxiter': max_iter_per_phase, 'gtol': 1e-6})
        
        b = res.x
        a_local = np.exp(b - b.max())
        a_local = a_local / a_local.sum()
        conv = fast_convolve(a_local)
        obj_val = 2.0 * n * conv.max()
        
        if obj_val < best_obj:
            best_obj = obj_val
            best_a = a_local.copy()
        
        phase_time = time.monotonic() - phase_start
        if phase_time > 0.1:
            # If a phase takes too long, skip remaining
            if time.monotonic() - t0 > time_limit * 0.7:
                break
    
    return best_a, best_obj

def pgd_optimize(a0, time_limit, t_start=5.0, t_end=5000.0):
    """Projected gradient descent with annealed temperature."""
    a = np.maximum(a0, 0.0).astype(np.float64)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    t = t_start
    eta = 1.0 / (2.0 * t * fast_convolve(a).max() + 1e-10)
    # Actually, let me use a fixed eta and adapt it
    eta = 0.5
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    patience = 0
    iters = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        w = smooth_max_weights(conv, t)
        g = compute_grad_conv(a, w)
        
        a_new = project_simplex(a - eta * g)
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1
            # Backtrack
            eta *= 0.8
            if patience % 20 == 0:
                t = min(t * 1.3, t_end)
                eta = 1.0
        
        a = a_new
        iters += 1
        
        if iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PGD iter {iters}, obj={best_obj:.8f}, t={t:.1f}, elapsed={elapsed:.1f}s")
    
    return best_a, best_obj

def make_lp_structure(n):
    rows_list = []
    cols_list = []
    for i in range(n):
        for j in range(n):
            rows_list.append(i + j)
            cols_list.append(i)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)

def solve_lp(f, M, rows, cols, n, timeout=15):
    vals = np.tile(f, n)
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                           options={'disp': False, 'time_limit': timeout})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

def lp_optimize(f0, time_limit, rows=None, cols=None):
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    f_sum = f.sum()
    if f_sum < 1e-12:
        f = np.ones(n) / n
    else:
        f = f / f_sum
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()
    
    if rows is None or cols is None:
        rows, cols = make_lp_structure(n)
    
    t0 = time.monotonic()
    t = 1.0
    patience = 0
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        g = solve_lp(f, M, rows, cols, n, timeout=min(10, max(1, (time_limit - (time.monotonic()-t0))//3)))
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-14:
            break
        
        improved = False
        for t_try in [t, 0.7*t, 0.5*t, 0.3*t, 0.1*t]:
            f_new = (1.0 - t_try) * f + t_try * g
            f_new = np.maximum(f_new, 0.0)
            s = f_new.sum()
            if s < 1e-12:
                continue
            f_new = f_new / s
            conv_new = fast_convolve(f_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < best_obj - 1e-14:
                f = f_new
                M = conv_new.max()
                best_obj = obj_new
                best_f = f.copy()
                patience = 0
                t = min(1.0, t * 1.05)
                improved = True
                break
        
        if not improved:
            patience += 1
            t *= 0.9
            if patience > 30:
                break
    
    return best_f, best_obj

def gen_inits(n, rng):
    inits = {}
    x = np.arange(n, dtype=np.float64)
    mid = 0.5 * (n - 1)
    
    inits['uniform'] = np.ones(n) / n
    
    for sf in [0.05, 0.1, 0.15, 0.2, 0.3]:
        g = np.exp(-0.5 * ((x - mid) / (sf * n))**2)
        g = g / g.sum()
        inits[f'gauss_{int(sf*100)}'] = g
    
    # Cosine taper
    cb = 0.5 * (1 + np.cos(np.pi * x / (n - 1)))
    cb[cb < 0.01] = 0.01
    inits['cosine'] = cb / cb.sum()
    
    # Triangle
    tri = 1.0 - np.abs(x - mid) / mid
    tri[tri < 0.01] = 0.01
    inits['triangle'] = tri / tri.sum()
    
    # Bump (two peaks)
    bump = np.exp(-0.5*((x - 0.25*n)/(0.1*n))**2) + np.exp(-0.5*((x - 0.75*n)/(0.1*n))**2)
    inits['bump'] = bump / bump.sum()
    
    # Random
    rr = rng.dirichlet(np.ones(n) * 0.5)
    inits['random'] = rr
    
    # Sparse random
    sp = np.zeros(n)
    mask = rng.random(n) < 0.3
    sp[mask] = rng.random(mask.sum()) + 0.1
    inits['sparse'] = sp / sp.sum()
    
    return inits
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase strategy:
- L-BFGS continuation with smooth max (log-sum-exp) on softmax parameterization
- Projected gradient descent with annealed temperature
- LP-guided alternating optimization for global direction
- Multi-start with diverse initializations including previous best
- Final perturbation polish
"""

import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def smooth_max_val(x, t):
    """Numerically stable log-sum-exp approximation to max."""
    m = x.max()
    shifted = t * (x - m)
    shifted = np.clip(shifted, -700, 700)
    return m + np.log(np.sum(np.exp(shifted))) / t


def smooth_max_weights(x, t):
    """Return softmax weights for smooth max."""
    m = x.max()
    shifted = t * (x - m)
    shifted = np.clip(shifted, -700, 700)
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return w
    return e / s


def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 0:
        return v
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_grad_conv(a, w):
    """Gradient of sum_k w[k]*conv[k] w.r.t. a. grad[j] = 2*sum_k w[k]*a[k-j]."""
    n = len(a)
    a_flip = a[::-1]
    total_len = len(w) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    AF = np.fft.rfft(a_flip, fft_size)
    result = np.fft.irfft(W * AF, fft_size)
    return 2.0 * result[n - 1:2 * n - 1]


def lbfgs_continuation(a0, time_limit, t_list=None, max_iter_per_phase=80):
    """L-BFGS with continuation on temperature using softmax parameterization."""
    a = np.maximum(a0, 0.0).astype(np.float64)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    if t_list is None:
        t_list = [5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000, 10000]

    b = np.log(a + 1e-30)
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()

    t0 = time.monotonic()

    for t in t_list:
        if time.monotonic() - t0 > time_limit:
            break

        def obj_func(b_param):
            a_local = np.exp(b_param - b_param.max())
            a_local = a_local / a_local.sum()
            conv = fast_convolve(a_local)
            return 2.0 * n * smooth_max_val(conv, t)

        try:
            res = optimize.minimize(obj_func, b, method='L-BFGS-B',
                                    options={'maxiter': max_iter_per_phase,
                                             'gtol': 1e-7, 'ftol': 1e-10})
            b = res.x
        except Exception:
            break

        a_local = np.exp(b - b.max())
        a_local = a_local / a_local.sum()
        conv = fast_convolve(a_local)
        obj_val = 2.0 * n * conv.max()

        if obj_val < best_obj:
            best_obj = obj_val
            best_a = a_local.copy()

    return best_a, best_obj


def pgd_optimize(a0, time_limit, t_start=10.0, t_end=10000.0):
    """Projected gradient descent with annealed temperature."""
    a = np.maximum(a0, 0.0).astype(np.float64)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    t = t_start
    eta = 0.3

    best_a = a.copy()

    t0 = time.monotonic()
    patience = 0
    iters = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        w = smooth_max_weights(conv, t)
        g = compute_grad_conv(a, w)

        # Backtracking line search
        improved = False
        for eta_try in [eta, 0.7 * eta, 0.5 * eta, 0.3 * eta, 0.1 * eta]:
            a_new = project_simplex(a - eta_try * g)
            conv_new = fast_convolve(a_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < best_obj - 1e-14:
                a = a_new
                best_obj = obj_new
                best_a = a.copy()
                patience = 0
                eta = min(1.0, eta * 1.02)
                improved = True
                break

        if not improved:
            patience += 1
            eta *= 0.8
            if patience >= 30:
                t = min(t * 1.5, t_end)
                eta = 0.3
                patience = 0
                if t >= t_end and eta < 0.01:
                    break

        iters += 1
        if iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PGD iter {iters}, obj={best_obj:.8f}, t={t:.1f}, eta={eta:.4f}")

    return best_a, best_obj


def make_lp_structure(n):
    """Precompute LP constraint matrix structure."""
    rows_list = []
    cols_list = []
    for i in range(n):
        for j in range(n):
            rows_list.append(i + j)
            cols_list.append(i)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)


def solve_lp(f, M, rows, cols, n, timeout=10):
    """Solve max sum(g) s.t. g>=0, (f*g)[k]<=M."""
    vals = np.tile(f, n)
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                           options={'disp': False, 'time_limit': timeout})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def lp_optimize(f0, time_limit, rows=None, cols=None):
    """LP-guided alternating optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    f_sum = f.sum()
    if f_sum < 1e-12:
        f = np.ones(n) / n
    else:
        f = f / f_sum

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()

    if rows is None or cols is None:
        rows, cols = make_lp_structure(n)

    t0 = time.monotonic()
    t = 1.0
    patience = 0

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        lp_timeout = min(12, max(2, remaining // 4))
        g = solve_lp(f, M, rows, cols, n, timeout=lp_timeout)
        if g is None:
            break

        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-14:
            break

        improved = False
        for t_try in [t, 0.7 * t, 0.5 * t, 0.3 * t, 0.1 * t]:
            f_new = (1.0 - t_try) * f + t_try * g
            f_new = np.maximum(f_new, 0.0)
            s = f_new.sum()
            if s < 1e-12:
                continue
            f_new = f_new / s
            conv_new = fast_convolve(f_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < best_obj - 1e-14:
                f = f_new
                M = conv_new.max()
                best_obj = obj_new
                best_f = f.copy()
                patience = 0
                t = min(1.0, t * 1.03)
                improved = True
                break

        if not improved:
            patience += 1
            t *= 0.9
            if patience > 25:
                break

    return best_f, best_obj


def gen_inits(n, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    mid = 0.5 * (n - 1)

    inits['uniform'] = np.ones(n) / n

    for sf in [0.05, 0.08, 0.12, 0.18, 0.25, 0.35]:
        g = np.exp(-0.5 * ((x - mid) / (sf * n)) ** 2)
        g = g / g.sum()
        inits[f'g{int(sf * 100)}'] = g

    cb = 0.5 * (1 + np.cos(np.pi * x / (n - 1)))
    cb[cb < 0.005] = 0.005
    inits['cosine'] = cb / cb.sum()

    tri = 1.0 - np.abs(x - mid) / mid
    tri[tri < 0.005] = 0.005
    inits['triangle'] = tri / tri.sum()

    bump = (np.exp(-0.5 * ((x - 0.25 * n) / (0.08 * n)) ** 2) +
            np.exp(-0.5 * ((x - 0.75 * n) / (0.08 * n)) ** 2))
    inits['bump'] = bump / bump.sum()

    # Flat top
    ft = np.ones(n)
    w = int(0.1 * n)
    ft[:w] = np.linspace(0.01, 1, w)
    ft[-w:] = np.linspace(1, 0.01, w)
    inits['flattop'] = ft / ft.sum()

    # Power law
    pw = np.power(np.abs(x - mid) / mid + 0.1, 0.5)
    inits['power'] = pw / pw.sum()

    rr = rng.dirichlet(np.ones(n) * 0.3)
    inits['rand1'] = rr

    rr2 = rng.dirichlet(np.ones(n) * 0.8)
    inits['rand2'] = rr2

    # Alternating pattern
    alt = np.ones(n)
    alt[::2] = 0.3
    inits['alt'] = alt / alt.sum()

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """Main entry: multi-phase optimization with diverse strategies."""
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
    except Exception:
        pass

    target_n = 512
    if parent is not None:
        target_n = parent.size

    print(f"Target n={target_n}, parent available: {parent is not None}")

    # Precompute LP structure
    rows, cols = make_lp_structure(target_n)

    # Generate initializations
    inits = gen_inits(target_n, rng)
    if parent is not None:
        if parent.size >= target_n:
            pp = np.maximum(parent[:target_n], 0.0)
        else:
            pp = np.interp(np.linspace(0, 1, target_n),
                           np.linspace(0, 1, parent.size), parent)
            pp = np.maximum(pp, 0.0)
        if pp.sum() > 1e-10:
            inits['parent'] = pp / pp.sum()

    # Also try parent at different lengths
    if parent is not None:
        for alt_n in [256, 1024]:
            if alt_n != target_n:
                pp2 = np.interp(np.linspace(0, 1, alt_n),
                                np.linspace(0, 1, parent.size), parent)
                pp2 = np.maximum(pp2, 0.0)
                if pp2.sum() > 1e-10:
                    inits[f'parent_n{alt_n}'] = (pp2 / pp2.sum(), alt_n)

    # Separate same-length and different-length inits
    inits_same = {}
    inits_diff = {}
    for k, v in inits.items():
        if isinstance(v, tuple):
            inits_diff[k] = v
        else:
            inits_same[k] = v

    # Phase 1: L-BFGS continuation
    print("\n=== Phase 1: L-BFGS Continuation ===")
    t_phase1_start = time.monotonic()
    remaining_total = deadline - t_phase1_start
    # Allocate: 55% phase1, 20% phase2, 15% phase3, 5% phase4, 5% buffer
    t_p1 = remaining_total * 0.55
    t_p2 = remaining_total * 0.20
    t_p3 = remaining_total * 0.15
    t_p4 = remaining_total * 0.05
    t_buffer = remaining_total * 0.05

    n_inits = len(inits_same) + len(inits_diff)
    t_per_init = t_p1 / max(1, n_inits)

    results = []  # (score, name, sequence, length)

    for name, a0 in inits_same.items():
        if time.monotonic() > deadline - (t_p2 + t_p3 + t_p4 + t_buffer):
            break
        ta = min(t_per_init, deadline - time.monotonic() - t_p2 - t_p3 - t_p4 - t_buffer)
        if ta < 2:
            continue

        a_opt, obj = lbfgs_continuation(a0, time_limit=ta)
        score = float(evaluate_sequence(a_opt.tolist()))
        results.append((score, name, a_opt, target_n))

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  [{name}] score={score:.6f}")

    # Different lengths
    for name, (a0, alt_n) in inits_diff.items():
        if time.monotonic() > deadline - (t_p2 + t_p3 + t_p4 + t_buffer):
            break
        ta = min(t_per_init, deadline - time.monotonic() - t_p2 - t_p3 - t_p4 - t_buffer)
        if ta < 2:
            continue
        rows_alt, cols_alt = make_lp_structure(alt_n)
        a_opt, obj = lbfgs_continuation(a0, time_limit=ta)
        score = float(evaluate_sequence(a_opt.tolist()))
        results.append((score, name, a_opt, alt_n))

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  [{name}] score={score:.6f}")

    print(f"Phase 1 done. Best: {best_score:.6f}")

    # Phase 2: PGD refinement on top results
    print("\n=== Phase 2: PGD Refinement ===")
    results.sort(key=lambda x: x[0] if np.isfinite(x[0]) else 1e9)
    top_k = min(5, len([r for r in results if np.isfinite(r[0])]))
    t_per_pgd = t_p2 / max(1, top_k)

    for idx in range(top_k):
        sc, name, a, n_seq = results[idx]
        if not np.isfinite(sc):
            continue
        if time.monotonic() > deadline - t_p3 - t_p4 - t_buffer:
            break
        ta = min(t_per_pgd, deadline - time.monotonic() - t_p3 - t_p4 - t_buffer)
        if ta < 3:
            continue

        print(f"  PGD: {name} (n={n_seq}, score={sc:.6f}), t={ta:.1f}s")
        a_opt, obj = pgd_optimize(a, time_limit=ta)
        score = float(evaluate_sequence(a_opt.tolist()))

        if np.isfinite(score):
            results.append((score, f'pgd_{name}', a_opt, n_seq))
            if score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                print(f"    NEW BEST: pgd_{name} score={score:.6f}")

    results.sort(key=lambda x: x[0] if np.isfinite(x[0]) else 1e9)

    # Phase 3: LP polish on the overall best
    print("\n=== Phase 3: LP-Guided Polish ===")
    if best_seq is not None and np.isfinite(best_score):
        n_best = len(best_seq)
        rows_b, cols_b = make_lp_structure(n_best)
        remaining3 = deadline - time.monotonic() - t_p4 - t_buffer
        if remaining3 > 10:
            # Split: LP + L-BFGS from LP result
            t_lp = remaining3 * 0.6
            t_lbfgs = remaining3 * 0.4

            print(f"  LP on best (n={n_best}, score={best_score:.6f}), t={t_lp:.1f}s")
            a_lp, obj_lp = lp_optimize(best_seq, time_limit=t_lp, rows=rows_b, cols=cols_b)
            score_lp = float(evaluate_sequence(a_lp.tolist()))

            if np.isfinite(score_lp) and score_lp < best_score:
                best_score = score_lp
                best_seq = a_lp.copy()
                print(f"    NEW BEST: LP score={score_lp:.6f}")

            if time.monotonic() < deadline - t_p4 - t_buffer:
                tr = min(t_lbfgs, deadline - time.monotonic() - t_p4 - t_buffer)
                if tr > 5:
                    print(f"  L-BFGS from LP result, t={tr:.1f}s")
                    a_f, obj_f = lbfgs_continuation(
                        a_lp, time_limit=tr,
                        t_list=[50, 100, 200, 500, 1000, 2000, 5000, 10000])
                    score_f = float(evaluate_sequence(a_f.tolist()))
                    if np.isfinite(score_f) and score_f < best_score:
                        best_score = score_f
                        best_seq = a_f.copy()
                        print(f"    NEW BEST: post-LP L-BFGS score={score_f:.6f}")

    # Phase 4: Final perturbation polish
    print("\n=== Phase 4: Final Polish ===")
    if best_seq is not None and time.monotonic() < deadline - 5:
        a = best_seq.copy()
        n_f = len(a)
        a = a / a.sum()
        t0 = time.monotonic()
        max_time = deadline - t0 - 5
        scale = 0.005
        best_local = best_score
        stall = 0

        while time.monotonic() - t0 < max_time:
            # Multiple perturbation trials
            for _ in range(200):
                if time.monotonic() - t0 > max_time:
                    break
                a_try = a + rng.normal(0, scale / n_f, n_f)
                a_try = np.maximum(a_try, 0)
                s = a_try.sum()
                if s < 1e-10:
                    continue
                a_try = a_try / s
                conv = fast_convolve(a_try)
                obj = 2.0 * n_f * conv.max()
                if obj < best_local - 1e-12:
                    a = a_try
                    best_local = obj
                    stall = 0
                else:
                    stall += 1

            if stall > 500:
                scale *= 0.7
                stall = 0
                if scale < 1e-6:
                    break

        final_score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(final_score) and final_score < best_score:
            best_score = final_score
            best_seq = a.copy()
            print(f"  Polish improved: {best_score:.6f}")

    # Phase 5: One more L-BFGS from absolute best with fine temperature
    if best_seq is not None and time.monotonic() < deadline - 3:
        tr = min(30.0, deadline - time.monotonic() - 3)
        if tr > 5:
            n_best = len(best_seq)
            a_f, _ = lbfgs_continuation(best_seq, time_limit=tr,
                                         t_list=[100, 200, 500, 1000, 2000, 5000])
            score_f = float(evaluate_sequence(a_f.tolist()))
            if np.isfinite(score_f) and score_f < best_score:
                best_score = score_f
                best_seq = a_f.copy()
                print(f"  Final L-BFGS: {best_score:.6f}")

    if best_seq is None:
        best_seq = np.ones(128)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL: n={len(best_seq)}, score={best_score:.8f}")
    return [float(x) for x in best_seq]
```
