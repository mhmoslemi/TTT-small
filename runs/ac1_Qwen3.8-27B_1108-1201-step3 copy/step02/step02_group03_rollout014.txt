"""
Optimizes non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses improved gradient descent (top-k peaks, backtracking line search, Nesterov momentum),
LP-based refinement, and pairwise coordinate optimization.

import numpy as np
import time
from scipy.optimize import linprog


def compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize_sum(a, target=1.0):
    """Normalize a to have the given sum."""
    S = a.sum()
    if S < 1e-15:
        return np.ones_like(a) * target / len(a)
    return a * (target / S)


def top_k_gradient(a, k=10, tau=None):
    """
    Compute gradient using top-k peaks of autocorrelation.
    Returns gradient vector of length n.
    """
    n = len(a)
    c = np.convolve(a, a)
    
    if tau is not None:
        # Soft-max over all lags
        c_max = c.max()
        w = np.exp(tau * (c - c_max))
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w /= w_sum
    else:
        # Hard top-k
        c_sorted_indices = np.argsort(c)[::-1]
        top_k = c_sorted_indices[:k]
        c_top = c[top_k]
        # Geometric weights
        c_max = c_top[0]
        if c_max > 0:
            ratios = c_top / c_max
            w_top = ratios ** 2  # Quadratic weighting
            w_top /= w_top.sum()
        else:
            w_top = np.ones(k) / k
        w = np.zeros_like(c)
        for idx, wi in zip(top_k, w_top):
            w[idx] = wi
    
    # Compute gradient: grad[i] = 2 * sum_k w[k] * a[k-i] (for valid k-i)
    # Use FFT for efficiency
    # grad[i] = 2 * (w * a_reversed)[i + n - 1] in the "valid" correlation
    # Actually: grad[i] = 2 * sum_{j} w[i+j] * a[j] for j=0..n-1
    # This is the cross-correlation of w (length 2n-1) with a (length n)
    
    # For efficiency, use direct computation for small n or FFT for large n
    if n <= 1024:
        grad = np.zeros(n)
        for i in range(n):
            # sum_j w[i+j] * a[j] for j=0..n-1
            # w has indices i, i+1, ..., i+n-1
            w_slice = w[i:i+n]
            if len(w_slice) < n:
                w_slice = np.pad(w_slice, (0, n - len(w_slice)))
            grad[i] = 2.0 * np.dot(w_slice, a)
    else:
        # FFT-based
        a_rev = a[::-1]
        conv_result = np.convolve(w, a_rev, mode='full')
        grad = 2.0 * conv_result[n-1:n-1+n]
    
    return grad


def gradient_with_line_search(a, lr_init=0.5, k_peaks=10, tau=None, max_backtrack=10):
    """
    One gradient descent step with backtracking line search.
    Returns (new_a, success).
    """
    n = len(a)
    S_target = a.sum()
    
    grad = top_k_gradient(a, k=k_peaks, tau=tau)
    
    # Project gradient to maintain sum
    grad_proj = grad - grad.mean()
    
    # Ensure we're moving in a descent direction
    # The directional derivative of the objective in direction d is:
    # d(obj)/d(alpha) at alpha=0 ≈ (2n/S^2) * sum_k w_k * c_k' * ... 
    # Simplified: if grad points in ascent direction, flip it
    # Actually, we want to DECREASE the peak, so we go AGAINST the gradient of the peak
    # The gradient of c[k*] w.r.t. a points in the direction that INCREASES c[k*]
    # So we should go in the direction -grad to decrease the peak
    
    direction = -grad_proj
    
    # Check non-negativity constraint
    # a + alpha * direction >= 0
    # For direction[i] < 0: alpha <= a[i] / (-direction[i])
    neg_mask = direction < 0
    if np.any(neg_mask):
        max_alpha = np.min(a[neg_mask] / (-direction[neg_mask])) * 0.95
    else:
        max_alpha = lr_init
    
    alpha = min(lr_init, max_alpha)
    
    # Backtracking line search
    best_a = a.copy()
    best_obj = compute_obj(a)
    
    for _ in range(max_backtrack):
        a_new = a + alpha * direction
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize_sum(a_new, S_target)
        
        obj_new = compute_obj(a_new)
        if obj_new < best_obj - 1e-12:
            best_a = a_new
            best_obj = obj_new
            break
        alpha *= 0.5
    
    return best_a, best_obj


def nesterov_gradient_descent(a0, deadline, k_peaks=10, tau=None, phase_label=""):
    """
    Nesterov accelerated gradient descent with adaptive parameters.
    """
    n = len(a0)
    a = a0.copy()
    S_target = a.sum()
    v = np.zeros(n)  # momentum
    best = a.copy()
    best_obj = compute_obj(a)
    
    lr = 0.1
    beta = 0.9  # momentum coefficient
    no_improve = 0
    
    it = 0
    while time.monotonic() < deadline:
        it += 1
        
        # Nesterov look-ahead
        a_look = a - beta * v
        
        # Compute gradient at look-ahead point
        grad = top_k_gradient(a_look, k=k_peaks, tau=tau)
        grad_proj = grad - grad.mean()
        direction = -grad_proj
        
        # Update momentum
        v = beta * v + lr * direction
        
        # Determine step size (backtracking)
        alpha = 1.0
        a_new = a + alpha * v
        max_backtracks = 0
        
        while max_backtracks < 15:
            a_new = a + alpha * v
            a_new = np.maximum(a_new, 0.0)
            
            # Check sum
            S_new = a_new.sum()
            if S_new < 1e-10:
                alpha *= 0.5
                max_backtracks += 1
                continue
            a_new = normalize_sum(a_new, S_target)
            
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-12:
                break
            alpha *= 0.5
            max_backtracks += 1
        
        if obj_new < best_obj - 1e-10:
            a = a_new
            best = a.copy()
            best_obj = obj_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                # Reduce learning rate and reset momentum
                lr *= 0.5
                v *= 0.5
                if lr < 1e-6:
                    break
                no_improve = 0
        
        # Adaptive k_peaks: increase as we converge
        if it % 1000 == 0:
            k_peaks = min(50, 10 + it // 500)
        
        # Adaptive tau for soft-max
        if tau is not None and it % 500 == 0:
            tau = min(100.0, tau * 1.5)
    
    return best, best_obj


def lp_refine(a, M=None):
    """
    LP refinement: max sum(g) s.t. conv(a,g) <= M, g >= 0.
    Returns optimal g or None.
    """
    n = len(a)
    if n > 1024:
        return None
    
    if M is None:
        c = np.convolve(a, a)
        M = float(c.max())
    
    c_obj = -np.ones(n)
    
    # Build constraint matrix
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        for m in range(j, j + n):
            A[m, j] = a[m - j]
    
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def pairwise_coord_opt(a, deadline, max_rounds=5):
    """
    Pairwise coordinate optimization: for each pair (i,j), find optimal
    redistribution of mass between them.
    """
    n = len(a)
    S_target = a.sum()
    best = a.copy()
    best_obj = compute_obj(a)
    
    # Precompute which pairs are "important" (contribute to peak)
    c = np.convolve(best, best)
    k_star = int(np.argmax(c))
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        improved = False
        # Try pairs that contribute to the peak
        for i in range(n):
            if time.monotonic() > deadline:
                break
            j = k_star - i
            if 0 <= j < n and j > i:  # Only consider valid pairs, avoid duplicates
                # Try redistributing mass between a[i] and a[j]
                total = best[i] + best[j]
                if total < 1e-10:
                    continue
                
                # Golden section search on t in [0, 1]: a[i] = t*total, a[j] = (1-t)*total
                best_t = 0.5  # current split: a[i]/total
                best_t = best[i] / total if total > 0 else 0.5
                
                # Try several values of t
                candidates = [0.0, 0.1, 0.3, best_t * 0.5, best_t * 1.5, 0.7, 0.9, 1.0]
                candidates = [max(0.0, min(1.0, t)) for t in candidates]
                candidates = list(set([round(t, 4) for t in candidates]))
                
                best_t_val = best_t
                best_s = best_obj
                
                for t in candidates:
                    best[i] = t * total
                    best[j] = (1 - t) * total
                    s = compute_obj(best)
                    if s < best_s - 1e-12:
                        best_s = s
                        best_t_val = t
                
                if best_t_val != best_t:
                    best[i] = best_t_val * total
                    best[j] = (1 - best_t_val) * total
                    best_obj = best_s
                    improved = True
        
        if not improved:
            break
    
    return best, best_obj


def coordinate_descent(a, deadline, max_rounds=3):

```python
def hinge_loss_gradient(a, t, n_lags=None):
    """
    Compute gradient of sum_k max(0, c[k] - t)^2 w.r.t. a.
    This encourages all convolution values to be below t.
    """
    n = len(a)
    c = np.convolve(a, a)
    
    # Only active lags
    excess = np.maximum(c - t, 0)
    active = excess > 0
    
    if not np.any(active):
        return np.zeros(n)
    
    # Gradient: grad[i] = 4 * sum_{k: c[k]>t} (c[k]-t) * a[k-i]
    # = 4 * (excess * a_reversed) correlation
    w = excess  # weight for each lag
    a_rev = a[::-1]
    conv_result = np.convolve(w, a_rev, mode='full')
    grad = 4.0 * conv_result[n-1:n-1+n]
    
    return grad
```
