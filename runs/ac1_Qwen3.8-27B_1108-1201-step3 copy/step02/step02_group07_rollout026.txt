"""
Soft-max gradient descent for minimizing peak autocorrelation ratio of
non-negative sequences. Uses adaptive beta (soft-max sharpness), Adam
optimizer, multi-start with parametric and random initializations, and
progressive refinement.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute the score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def compute_score_and_grad(a, beta=50.0):
    """Compute score and gradient using soft-max approximation of max."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf, np.zeros(n)
    
    c = fft_convolve(a, a)
    
    # Soft-max of c
    c_max = np.max(c)
    exp_c = np.exp(beta * (c - c_max))
    p = exp_c / np.sum(exp_c)
    m_smooth = c_max + np.log(np.sum(exp_c)) / beta
    
    # Compute g[i] = sum_j a[j] * p[i+j] for valid indices

    p_rev = p[::-1]
    conv_ap = fft_convolve(a, p_rev)
    g = conv_ap[::-1]
    
    # Gradient: df/da[i] = 4*n/S^3 * (S*g[i] - m_smooth)
    grad = (4.0 * n / (S ** 3)) * (S * g - m_smooth)
    
    score = 2.0 * n * m_smooth / (S * S)
    return score, grad


def adam_step(params, grads, state, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer step. params is dict, grads is dict."""
    m = state['m']
    v = state['v']
    t = state['t'] + 1
    state['t'] = t
    
    for key in params:
        m[key] = beta1 * m[key] + (1 - beta1) * grads[key]
        v[key] = beta2 * v[key] + (1 - beta2) * grads[key] ** 2
        m_hat = m[key] / (1 - beta1 ** t)
        v_hat = v[key] / (1 - beta2 ** t)
        params[key] = params[key] - lr * m_hat / (np.sqrt(v_hat) + eps)
    
    return params

        S = 1.0
    a = a / S

    best_score = compute_score(a)

    t0 = time.monotonic()
    iter_count = 0
    patience = 0
    max_patience = 200
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 1.0:

        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)

        a = a - lr * (state['m'] * 0.9 + 0.1 * grad) / (np.sqrt(state['v'] * 0.999 + 0.001 * grad**2) + 1e-8)
        state['m'] = 0.9 * state['m'] + 0.1 * grad

                patience = 0

"""
Multi-strategy optimization for minimizing peak autocorrelation ratio of
non-negative sequences. Combines:
1. Parametric global search with few parameters
2. Soft-max gradient descent (Adam, adaptive beta, FFT-based gradient)
3. Analytic cyclic coordinate descent (exact 1D optimization per coordinate)
4. Simulated annealing for escaping local minima
Uses the provided sequence as warm start and explores multiple lengths.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]


def compute_score(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def soft_max_score_and_grad(a, beta=50.0):
    """Compute smoothed score and gradient using soft-max of max(conv)."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return np.inf, np.zeros(n)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    exp_c = np.exp(beta * (c - c_max))
    p = exp_c / np.sum(exp_c)
    m_smooth = c_max + np.log(np.sum(exp_c)) / beta
    # g[i] = sum_j a[j] * p[i+j] via FFT
    p_rev = p[::-1].copy()
    conv_ap = fft_convolve(a, p_rev)
    g = conv_ap[::-1]
    grad = (4.0 * n / (S ** 3)) * (S * g - m_smooth)
    score = 2.0 * n * m_smooth / (S * S)
    return score, grad


def make_cosine_init(n, rng, order=1):
    """Cosine-based window initialization."""
    t = np.arange(n, dtype=np.float64) / max(1, n - 1)
    a = 1.0
    for k in range(1, order + 2):
        coeff = 0.15 / k * rng.standard_normal()
        a = a + coeff * np.cos(np.pi * k * (2 * t - 1))
    a = np.abs(a)
    a = np.clip(a, 0.01, None)
    return a


def make_gaussian_init(n, rng, center=0.5, width=0.3):
    t = np.arange(n, dtype=np.float64)
    c = center + 0.05 * (2 * rng.random() - 1)
    w = width * (0.5 + rng.random())
    a = np.exp(-0.5 * ((t - c) / w) ** 2)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.001, None)


def make_polynomial_init(n, rng, degree=4):
    t = 2.0 * np.arange(n, dtype=np.float64) / max(1, n - 1) - 1.0
    coeffs = rng.normal(0, 0.3, size=degree + 1)
    coeffs[0] = 1.0 + abs(coeffs[0])
    a = np.zeros(n)
    for d, c in enumerate(coeffs):
        a += c * t ** d
    a = np.abs(a) + 0.05
    return a


def optimize_gradient(a0, time_limit, beta_start=10.0, beta_end=200.0,
                      lr_init=0.05, verbose=False):
    """Soft-max gradient descent with Adam and adaptive beta."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    S = float(np.sum(a))
    if S < 1e-10:
        a = np.ones(n) / n
        S = 1.0
    a = a / S

    best_a = a.copy()
    best_score = compute_score(a)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    t0 = time.monotonic()
    stall = 0
    max_stall = 150

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 0.5:
            break

        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / max(1.0, time_limit))
        beta = beta_start * (beta_end / beta_start) ** (progress ** 0.7)
        lr = lr_init * (1.0 / (1.0 + 0.5 * progress))

        score, grad = soft_max_score_and_grad(a, beta)
        if not np.isfinite(score):
            break

        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a_new, 1e-8, None)

        S_new = float(np.sum(a))
        if S_new < 0.01:
            a = a * (0.1 / max(S_new, 1e-15))

        stall += 1
        if t % 30 == 0:
            exact_s = compute_score(a)
            if exact_s < best_score - 1e-10:
                best_score = exact_s
                best_a = a.copy()
                stall = 0
            elif verbose and t % 300 == 0:
                print(f"      [{elapsed:5.1f}s] it={t} beta={beta:.1f} best={best_score:.6f}")

        if stall >= max_stall:
            rng = np.random.default_rng(t * 31 + 7)
            a = best_a * np.exp(0.08 * rng.standard_normal(n))
            a = np.clip(a, 1e-8, None)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            stall = 0

    final_s = compute_score(a)
    if final_s < best_score:
        best_score = final_s
        best_a = a.copy()
    return best_a, best_score


def coordinate_descent(a0, time_limit, verbose=False):
    """Cyclic coordinate descent with analytic optimal step computation."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    S = float(np.sum(a))
    a = a / S
    best_a = a.copy()
    best_score = compute_score(a)

    t0 = time.monotonic()
    pass_count = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 1.0:
            break

        c = fft_convolve(a, a)
        k_star = int(np.argmax(c))
        m_val = c[k_star]
        improved = False

        for i in range(n):
            if time.monotonic() - t0 >= time_limit - 0.5:
                break

            u_val = 0.0
            idx = k_star - i
            if 0 <= idx < n:
                u_val = a[idx]

            if u_val < 1e-15:
                opt_delta = -a[i] * 0.5
            elif k_star == 2 * i:
                disc = S * S - 2.0 * a[i] * S + m_val
                if disc > 0:
                    opt_delta = -S + np.sqrt(disc)
                else:
                    opt_delta = -0.3 * a[i]
            else:
                opt_delta = m_val / (2.0 * u_val * S) - 1.0

            # Clamp delta
            max_delta = 0.5 * a[i]
            opt_delta = np.clip(opt_delta, -max_delta, max_delta)

            a_new = a.copy()
            a_new[i] = a[i] + opt_delta
            if a_new[i] < 1e-10:
                a_new[i] = 1e-10

            new_score = compute_score(a_new)
            if new_score < compute_score(a) - 1e-12:
                a = a_new
                S = float(np.sum(a))
                improved = True

        c = fft_convolve(a, a)
        k_star = int(np.argmax(c))
        m_val = c[k_star]

        if improved:
            exact_s = compute_score(a)
            if exact_s < best_score:
                best_score = exact_s
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 2

        pass_count += 1
        if verbose and pass_count % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"      [CD {elapsed:5.1f}s] pass={pass_count} best={best_score:.6f} stall={stall}")

        if stall >= 30:
            break

    return best_a, best_score


def simulated_annealing(a0, time_limit, verbose=False):
    """Simulated annealing with structured perturbations."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    S = float(np.sum(a))
    a = a / S
    best_a = a.copy()
    best_score = compute_score(a)

    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    temp = 0.02
    cool = 0.9997
    iter_count = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 0.5:
            break

        a_cand = a.copy()
        r = rng.random()

        if r < 0.3:
            idx = rng.integers(0, n)
            a_cand[idx] *= np.exp(rng.normal(-0.02, 0.05))
        elif r < 0.5:
            width = max(1, int(rng.integers(2, max(3, n // 10))))
            start = int(rng.integers(0, max(1, n - width)))
            a_cand[start:start + width] *= np.exp(rng.normal(0.0, 0.03, size=width))
        elif r < 0.7:
            c = fft_convolve(a, a)
            peak = int(np.argmax(c))
            contribs = np.zeros(n)
            for j in range(n):
                idx2 = peak - j
                if 0 <= idx2 < n:
                    contribs[j] = a[j] * a[idx2]
            top_k = max(5, n // 10)
            top_idx = np.argsort(contribs)[-top_k:]
            a_cand[top_idx] *= np.exp(rng.normal(-0.04, 0.02, size=top_k))
            others = np.setdiff1d(np.arange(n), top_idx)
            if len(others) > 0:
                chosen = rng.choice(others, size=min(10, len(others)), replace=False)
                a_cand[chosen] *= np.exp(rng.normal(0.04, 0.02, size=len(chosen)))
        elif r < 0.85:
            a_cand *= np.exp(rng.normal(0, 0.01, size=n))
        else:
            # Mirror perturbation
            a_cand = a * (1.0 + 0.1 * rng.standard_normal(n))

        a_cand = np.clip(a_cand, 1e-8, None)
        S_cand = float(np.sum(a_cand))
        if S_cand < 0.01:
            continue

        cand_score = compute_score(a_cand)
        cur_score = compute_score(a)

        if cand_score < cur_score:
            a = a_cand
            if cand_score < best_score:
                best_score = cand_score
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        else:
            if rng.random() < np.exp(-(cand_score - cur_score) / max(temp, 1e-15)):
                a = a_cand
            stall += 1

        temp *= cool
        iter_count += 1

        if verbose and iter_count % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      [SA {elapsed:5.1f}s] it={iter_count} temp={temp:.6f} best={best_score:.6f}")

        if stall >= 500:
            break

    return best_a, best_score


def parametric_search(n, time_budget):
    """Search over parametric families using scipy minimize."""
    t0 = time.monotonic()
    best_a = None
    best_s = np.inf

    # Family 1: polynomial
    def poly_score(params):
        coeffs = params
        t = 2.0 * np.arange(n, dtype=np.float64) / max(1, n - 1) - 1.0
        a = np.zeros(n)
        d = len(coeffs) - 1
        for j in range(n):
            val = 0.0
            tval = t[j]
            for k in range(d, -1, -1):
                val = val * tval + coeffs[k]
            a[j] = max(0.01, val)
        return compute_score(a)

    # Family 2: Gaussian mixture
    def gauss_score(params):
        centers = params[0:n // 2]
        widths = np.abs(params[n // 2:]) + 0.05
        a = np.zeros(n)
        t = np.arange(n, dtype=np.float64) / max(1, n - 1)
        for c, w in zip(centers, widths):
            a += np.exp(-0.5 * ((t - c) / w) ** 2)
        a = np.clip(a, 0.001, None)
        return compute_score(a)

    rng = np.random.default_rng(42)

    # Polynomial search
    for degree in [2, 4, 6]:
        if time.monotonic() - t0 > time_budget * 0.5:
            break
        x0 = np.zeros(degree + 1)
        x0[0] = 1.0
        x0[1::2] = 0.3 * rng.standard_normal(degree // 2 + 1)
        x0[2::2] = 0.1 * rng.standard_normal((degree - 1) // 2 + 1)

        def obj_p(x):
            return poly_score(x)

        try:
            res = minimize(obj_p, x0, method='Nelder-Mead',
                          options={'maxiter': 1000, 'xatol': 0.01, 'fatol': 1e-8})
            if res.fun < best_s:
                t = 2.0 * np.arange(n, dtype=np.float64) / max(1, n - 1) - 1.0
                a = np.zeros(n)
                d = len(res.x) - 1
                for j in range(n):
                    val = 0.0
                    for k in range(d, -1, -1):
                        val = val * t[j] + res.x[k]
                    a[j] = max(0.01, val)
                best_s = res.fun
                best_a = a
        except Exception:
            pass

    # Gaussian search
    for ncomp in [2, 3]:
        if time.monotonic() - t0 > time_budget * 0.8:
            break
        x0 = np.zeros(2 * ncomp)
        for i in range(ncomp):
            x0[i] = 0.2 + 0.6 * i / max(1, ncomp - 1)
            x0[ncomp + i] = 0.2
        try:
            res = minimize(gauss_score, x0, method='Nelder-Mead',
                          options={'maxiter': 2000, 'xatol': 0.01, 'fatol': 1e-8})
            if res.fun < best_s:
                centers = res.x[0:ncomp]
                widths = np.abs(res.x[ncomp:]) + 0.05
                a = np.zeros(n)
                t = np.arange(n, dtype=np.float64) / max(1, n - 1)
                for c, w in zip(centers, widths):
                    a += np.exp(-0.5 * ((t - c) / w) ** 2)
                a = np.clip(a, 0.001, None)
                best_s = res.fun
                best_a = a
        except Exception:
            pass

    return best_a, best_s

                best_score = s_c
                best_seq = f_c.copy()
                print(f"  After final CD: {best_score:.6f}")

    # === Finalize ===
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        # Ensure sum is large enough
        if sum(final_list) < 1.0:
            scale = 1.0 / max(sum(final_list), 1e-10)
            final_list = [x * scale for x in final_list]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'='*60}")
        print(f"FINAL: score={final_score:.6f}, n={len(final_list)}")
        print(f"{'='*60}")
        return final_list

    # Fallback
    return [1.0] * 256
