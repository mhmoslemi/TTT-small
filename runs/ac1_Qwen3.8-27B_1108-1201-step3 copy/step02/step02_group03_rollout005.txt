"""
Optimizes a non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses magnitude equalization (Gerchberg-Saxton style), soft-max gradient descent,
LP-based Frank-Wolfe steps, and multi-restart strategy.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import fft, ifft

def compute_obj(a):
    """Compute the objective."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize(a, target=None):
    """Normalize sequence to have a specific sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)


def magnitude_equalization_step(a, alpha=0.7, iters=100):
    """
    Gerchberg-Saxton style magnitude equalization.
    Compresses the dynamic range of the DFT magnitudes.
    """
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = normalize(a.copy(), S_t)
    
    for _ in range(iters):
        A = fft(a)
        # Compress magnitudes
        mags = np.abs(A)
        phases = np.angle(A)
        # Keep DC component fixed
        dc = mags[0]
        mags_compressed = np.ones_like(mags)
        mags_compressed[0] = dc
        for m in range(1, n):
            mags_compressed[m] = (mags[m] ** alpha) * (dc ** (1 - alpha))
        # Actually, let's be more careful. We want to compress toward the mean.
        # mags_compressed[m] = mags[m] ** alpha * (mean_mags) ** (1-alpha)
        # But we need to keep DC fixed.
        # Let's use: for m >= 1, new_mag = mags[m] * (target/mags[m])^beta
        # where target = geometric mean of non-DC mags
        
        A_new = mags_compressed * np.exp(1j * phases)
        a_new = np.real(ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = normalize(a_new, S_t)
        a = a_new
    
    return a


def gradient_soft_max(a, tau):
    """
    Compute gradient of soft-max autocorrelation.
    grad[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
    where w[k] = exp(tau * c[k]) / sum(exp(tau * c)))
    """
    n = len(a)
    c = np.convolve(a, a)
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    
    # grad[i] = 2 * sum_j w[i+j] * a[j]
    # Use the convolution trick
    a_rev = a[::-1]
    conv_result = np.convolve(w, a_rev, mode='full')
    grad = 2.0 * conv_result[n-1:n-1+n]
    
    return grad


def gradient_hard(a):
    """Compute gradient at the hard max."""
    n = len(a)
    c = np.convolve(a, a)
    k_star = int(np.argmax(c))
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    return grad


def gradient_descent_step(a, lr, momentum_buf, momentum_decay, grad):
    """One step of projected gradient descent with momentum."""
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a_curr = normalize(a, S_t)
    
    # Project out mean to maintain sum constraint
    grad = grad - grad.mean()
    
    momentum_buf[:] = momentum_decay * momentum_buf - lr * grad
    a_new = a_curr + momentum_buf
    a_new = np.maximum(a_new, 0.0)
    a_new = normalize(a_new, S_t)
    
    return a_new


def lp_frank_wolfe(f, M):
    """
    Frank-Wolfe step: maximize sum(g) s.t. conv(f, g) <= M, g >= 0.
    This gives the direction of steepest descent for the sum objective
    while maintaining the convolution constraint.
    """
    n = len(f)
    if n > 1024:
        return None
    
    c_obj = -np.ones(n)
    
    # Build constraint matrix efficiently
    # A[m, j] = f[m-j] if 0 <= m-j < n, else 0
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        m_start = j
        m_end = min(j + n, 2 * n - 1)
        A[m_start:m_end, j] = f[:m_end - m_start]
    
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

"""
Optimizes a non-negative sequence to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses quadratic-gradient descent, soft-max gradient, LP-based Frank-Wolfe steps,
magnitude equalization initialization, and multi-restart with line search.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import fft, ifft


def compute_obj(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize(a, target=None):
    """Normalize sequence to have a specific sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)


def gradient_quadratic(a):
    """
    Gradient of sum(c[k]^2) w.r.t. a, where c = conv(a, a).
    grad[i] = 4 * sum_j c[i+j] * a[j]
    Computed via convolution.
    """
    n = len(a)
    c = np.convolve(a, a)
    a_rev = a[::-1]
    # grad[i] = 4 * conv(c, a_rev)[n-1+i]
    conv_ca = np.convolve(c, a_rev, mode='full')
    grad = 4.0 * conv_ca[n - 1: n - 1 + n]
    return grad


def gradient_soft_max(a, tau):
    """
    Gradient of soft-max(c) w.r.t. a.
    w[k] = exp(tau*(c[k]-max(c))) / sum(exp(tau*(c-max(c))))
    grad[i] = 2 * sum_j w[i+j] * a[j]
    """
    n = len(a)
    c = np.convolve(a, a)
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    a_rev = a[::-1]
    conv_result = np.convolve(w, a_rev, mode='full')
    grad = 2.0 * conv_result[n - 1: n - 1 + n]
    return grad


def gradient_top_k(a, k=5):
    """
    Gradient weighted by top-k autocorrelation peaks.
    """
    n = len(a)
    c = np.convolve(a, a)
    # Get top-k indices
    top_idx = np.argsort(c)[-k:]
    w = np.zeros_like(c)
    for idx in top_idx:
        w[idx] = 1.0
    w /= w.sum()
    a_rev = a[::-1]
    conv_result = np.convolve(w, a_rev, mode='full')
    grad = 2.0 * conv_result[n - 1: n - 1 + n]
    return grad


def log_grad_step(a, lr, momentum_buf, momentum_decay, grad, S_t):
    """One step of gradient descent with momentum, projecting onto {a>=0, sum(a)=S_t}."""
    n = len(a)
    a_curr = normalize(a, S_t)
    # Project out mean direction to maintain sum
    grad_c = grad - grad.mean()
    momentum_buf[:] = momentum_decay * momentum_buf - lr * grad_c
    a_new = a_curr + momentum_buf
    a_new = np.maximum(a_new, 0.0)
    a_new = normalize(a_new, S_t)
    return a_new


def lp_frank_wolfe(f, M, time_limit=20):
    """
    Frank-Wolfe: maximize sum(g) s.t. conv(f, g) <= M (elementwise), g >= 0.
    Returns optimal g or None.
    """
    n = len(f)
    if n > 1024:
        return None

    c_obj = -np.ones(n)

    # Build constraint matrix: A[m, j] = f[m-j] if 0 <= m-j < n
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        m_start = j
        m_end = min(j + n, 2 * n - 1)
        length = m_end - m_start
        A[m_start:m_end, j] = f[:length]

    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def lp_step_linesearch(a, M, S_t, time_limit=20):
    """LP step with line search for optimal blending."""
    g = lp_frank_wolfe(a, M, time_limit=time_limit)
    if g is None:
        return a, False

    g_sum = g.sum()
    if g_sum < 1e-6:
        return a, False

    g_norm = g * (S_t / g_sum)

    # Line search
    best_score = compute_obj(a)
    best_t = 0.0
    best_a = a.copy()

    for t in [0.9, 0.7, 0.5, 0.35, 0.2, 0.1, 0.05]:
        a_new = (1.0 - t) * a + t * g_norm
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new, S_t)
        s_new = compute_obj(a_new)
        if s_new < best_score:
            best_score = s_new
            best_t = t
            best_a = a_new

    return best_a, best_t > 0


def magnitude_equalization(a, alpha=0.5, iters=100):
    """
    Gerchberg-Saxton style: compress DFT magnitude dynamic range.
    """
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = normalize(a.copy(), S_t)

    for _ in range(iters):
        A = fft(a)
        mags = np.abs(A)
        phases = np.angle(A)
        dc = mags[0]
        if dc < 1e-12:
            break
        # Compress non-DC magnitudes toward dc
        mags_new = mags.copy()
        for m in range(1, n):
            mags_new[m] = mags[m] ** alpha * dc ** (1.0 - alpha)
        A_new = mags_new * np.exp(1j * phases)
        a_new = np.real(ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a = normalize(a_new, S_t)

    return a

                best_score = s_mag

                no_improve = 0

                    no_improve = 0

    # Phase 3: Soft-max gradient (sharper, targets peaks more directly)

    tau = 5.0
    lr = 0.2

    no_improve = 0

        it += 1
        tau = 5.0 + 3.0 * it
        tau = min(tau, 100.0)
        lr = 0.2 * (0.9999 ** it)
        lr = max(lr, 0.01)

        # Alternate between soft-max and top-k gradients
        if it % 10 == 0:
            grad = gradient_top_k(a, k=3)
        else:
            grad = gradient_soft_max(a, tau)

        a = log_grad_step(a, lr, momentum_buf, 0.9, grad, S_t)

        if it % 500 == 0:
            s = compute_obj(a)
            if s < best_score - 1e-9:
                best_score = s
                best = a.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 5:
                    momentum_buf[:] = 0
                    no_improve = 0
                    if tau > 50:
                        break

    if verbose:
        print(f"      After soft-grad: {best_score:.6f} (it={it})")

    # Phase 4: LP-based Frank-Wolfe (most powerful single step)
    t4_end = time.monotonic() + min(200, max(30, (deadline - time.monotonic()) * 0.35))
    if n <= 768 and time.monotonic() < t4_end:
        a = best.copy()
        no_improve_lp = 0
        it_lp = 0

        while time.monotonic() < t4_end and it_lp < 40:
            it_lp += 1
            c = np.convolve(a, a)
            M = float(c.max())

            # Slightly aggressive M to push harder
            M_aggr = M * 0.999

            a_new, improved = lp_step_linesearch(a, M, S_t, time_limit=15)
            if not improved:
                # Try aggressive
                a_new, improved = lp_step_linesearch(a, M_aggr, S_t, time_limit=15)

            if improved:
                a = a_new
                s_new = compute_obj(a)
                if s_new < best_score:
                    best = a.copy()
                    best_score = s_new
                no_improve_lp = 0
            else:
                no_improve_lp += 1
                # Small gradient nudge to escape
                grad = gradient_soft_max(a, 20.0)
                a = log_grad_step(a, 0.05, np.zeros(n), 0.0, grad, S_t)
                s = compute_obj(a)
                if s < best_score:
                    best = a.copy()
                    best_score = s
                    no_improve_lp = 0
                if no_improve_lp > 6:
                    break

        if verbose:
            print(f"      After LP: {best_score:.6f} (it={it_lp})")

    # Phase 5: Final fine gradient polish
    t5_end = deadline
    if time.monotonic() < t5_end - 2:
        a = best.copy()
        momentum_buf = np.zeros(n)
        tau = 20.0
        lr = 0.05
        it = 0

        while time.monotonic() < t5_end - 2 and it < 50000:
            it += 1
            lr = 0.05 * (0.99995 ** it)
            lr = max(lr, 0.005)
            grad = gradient_soft_max(a, tau)
            a = log_grad_step(a, lr, momentum_buf, 0.85, grad, S_t)

            if it % 500 == 0:
                s = compute_obj(a)
                if s < best_score - 1e-10:
                    best_score = s
                    best = a.copy()

    if verbose:
        print(f"      Final: {best_score:.6f}")

    return best, best_score


def generate_starts(rng, lengths):
    """Generate diverse starting sequences for each length."""
    starts = []

    # From height_sequence_1
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        if gs.size >= 8 and np.all(np.isfinite(gs)):
            for L in lengths:
                if len(gs) >= L:
                    starts.append(gs[:L].copy())
                else:
                    reps = L // len(gs) + 1
                    starts.append(np.tile(gs, reps)[:L].copy())
            # Perturbed versions
            for L in lengths:
                base = gs[:L].copy() if len(gs) >= L else np.tile(gs, L // len(gs) + 1)[:L]
                for std in [0.1, 0.2, 0.3]:
                    perturbed = np.maximum(base + rng.normal(0, std, L), 0)
                    starts.append(perturbed)
                # Shifted
                for shift in [1, 3, 7]:
                    starts.append(np.roll(base, shift))
    except (NameError, Exception):
        pass

    # Random and structured starts for each length
    for n in lengths:
        i = np.arange(n, dtype=np.float64)
        starts.append(rng.random(n) * 0.5 + 0.25)
        starts.append(0.5 + 0.5 * np.cos(2 * np.pi * i * 3.0 / n))
        starts.append(0.5 + 0.5 * np.cos(2 * np.pi * i * 7.0 / n + 0.3))
        starts.append(np.abs(np.sin(2 * np.pi * i * 2.0 / n)) + 0.15)
        starts.append(np.exp(-0.5 * ((i - n / 2.0) / (n * 0.25)) ** 2))
        starts.append(np.exp(-0.5 * ((i - n * 0.3) / (n * 0.12)) ** 2) +
                      np.exp(-0.5 * ((i - n * 0.7) / (n * 0.12)) ** 2))
        # Multi-frequency
        starts.append(0.3 + 0.3 * np.cos(2 * np.pi * i * 3.0 / n) +
                      0.2 * np.cos(2 * np.pi * i * 5.0 / n + 1.0))

    return starts


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Multi-phase optimization with:
    - Magnitude equalization initialization
    - Quadratic gradient descent (smooth surrogate)
    - Soft-max / top-k gradient descent
    - LP-based Frank-Wolfe with line search
    - Multi-restart with perturbation
    Returns the best non-negative sequence found.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    print(f"=== Starting optimization, budget={budget_s}s ===")

    # Load known best
    gs = None
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        if gs.size >= 8 and np.all(np.isfinite(gs)):
            s = compute_obj(gs)
            print(f"  height_sequence_1: n={len(gs)}, score={s:.6f}")
            best_score = s
            best_seq = gs.copy()
        else:
            gs = None
    except (NameError, Exception) as e:
        print(f"  height_sequence_1 unavailable: {e}")

    # Primary length: use native length of known best, or 384
    if best_seq is not None:
        primary_n = len(best_seq)
    else:
        primary_n = 384

    print(f"  Primary length: {primary_n}")

    # Phase A: Optimize the known best thoroughly
    if best_seq is not None:
        print(f"\n--- Phase A: Optimize known best (n={primary_n}) ---")
        t_alloc = min(400.0, (deadline - time.monotonic()) * 0.35)
        seq, score = optimize_sequence(best_seq, time.monotonic() + t_alloc, verbose=True)
        best_seq = seq
        best_score = score
        print(f"  Phase A result: {best_score:.6f}")

    # Phase B: Try diverse starts at primary length
    print(f"\n--- Phase B: Diverse starts at n={primary_n} ---")
    lengths_to_try = [primary_n]
    # Also try nearby lengths if time permits
    for alt_n in [256, 512]:
        if alt_n != primary_n:
            lengths_to_try.append(alt_n)

    all_starts = generate_starts(rng, lengths_to_try)
    # Sort: primary length first
    all_starts.sort(key=lambda x: (0 if len(x) == primary_n else 1, len(x)))

    # Deduplicate
    seen = set()
    unique_starts = []
    for s in all_starts:
        key = (len(s), tuple(np.round(s[:10], 2)))
        if key not in seen:
            seen.add(key)
            unique_starts.append(s)
    all_starts = unique_starts

    print(f"  {len(all_starts)} diverse starts")

    t_phase_b_end = deadline - 300  # Leave 300s for Phase C
    for i, a0 in enumerate(all_starts):
        if time.monotonic() > t_phase_b_end:
            break
        time_left = t_phase_b_end - time.monotonic()
        n_starts_left = len(all_starts) - i
        t_alloc = min(180.0, time_left / max(1, n_starts_left) * 0.85)
        t_alloc = max(30.0, t_alloc)

        if t_alloc < 20:
            break

        n_a0 = len(a0)
        seq, score = optimize_sequence(a0, time.monotonic() + t_alloc, verbose=False)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"  Start {i}: NEW BEST {best_score:.6f} (n={n_a0})")
        else:
            print(f"  Start {i}: {score:.6f} (n={n_a0})")

    # Phase C: Focus remaining time on the best solution
    print(f"\n--- Phase C: Final focus on best (score={best_score:.6f}, n={len(best_seq)}) ---")
    remaining = deadline - time.monotonic() - 10

    # Multiple rounds of optimization from the best
    for round_num in range(3):
        if remaining < 30:
            break
        # Create perturbed version
        if round_num == 0:
            start = best_seq.copy()
        elif round_num == 1:
            start = np.maximum(best_seq + rng.normal(0, 0.05, len(best_seq)), 0)
        else:
            start = np.maximum(best_seq + rng.normal(0, 0.15, len(best_seq)), 0)

        t_alloc = min(remaining - 20, 200.0)
        seq, score = optimize_sequence(start, time.monotonic() + t_alloc, verbose=True)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"  Round {round_num}: NEW BEST {best_score:.6f}")
        else:
            print(f"  Round {round_num}: {score:.6f}")
        remaining = deadline - time.monotonic() - 10

    # Final validation
    if best_seq is not None:
        final_score = compute_obj(best_seq)
        print(f"\n=== FINAL: n={len(best_seq)}, score={final_score:.6f} ===")
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"  evaluate_sequence confirms: {eval_score:.6f}")
        except Exception as e:
            print(f"  evaluate_sequence error: {e}")
        return [float(x) for x in best_seq]

    # Fallback
    print("Falling back to uniform sequence.")
    return [1.0] * 384
