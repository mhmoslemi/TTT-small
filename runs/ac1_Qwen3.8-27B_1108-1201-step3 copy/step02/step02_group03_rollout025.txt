Optimizes non-negative sequence to minimize 2*n*max(autocorr)/(sum)^2.
Primary method: iterative LP (convolution constraint) with adaptive step sizes.
Secondary: smooth gradient descent, simulated annealing for escape, coordinate polish.
"""

import numpy as np

from scipy.optimize import linprog

def compute_obj(a):
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(np.max(c))
    S = float(np.sum(a))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (S * S)

def normalize_sum(a, target=None):
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = float(np.sum(a))
    if S < 1e-15:
        return a.copy()
    return a * (target / S)

def solve_lp(f, M):
    n = len(f)
    c_obj = -np.ones(n)
    # Build A: A[k, j] = f[k-j] if 0 <= k-j < n
    A = np.zeros((2*n - 1, n))
    for j in range(n):
        for k in range(j, j + n):
            A[k, j] = f[k - j]
    b = np.full(2*n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

def soft_gradient(a, tau):
    n = len(a)
    c = np.convolve(a, a)
    c_max = c.max()
    w = np.exp(tau * (c - c_max))
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= w_sum
    a_rev = a[::-1]
    conv_result = np.convolve(w, a_rev, mode='full')
    grad = 2.0 * conv_result[n-1:n-1+n]
    return grad

def lp_iteration(f, max_iters=200, verbose=True):
    """Main LP-based optimization loop."""
    n = len(f)
    S_t = np.sqrt(2.0 * n)
    f = normalize_sum(f.copy(), S_t)
    best_f = f.copy()
    best_score = compute_obj(f)
    stagnation = 0
    
    for it in range(max_iters):
        c = np.convolve(f, f)
        M = float(c.max())
        
        g = solve_lp(f, M)
        if g is None:
            if verbose:
                print(f"    LP failed at iter {it}")
            break
        
        g_sum = float(np.sum(g))
        f_sum = float(np.sum(f))
        
        if g_sum <= f_sum * (1 + 1e-8):
            stagnation += 1
            if stagnation > 5:
                if verbose:
                    print(f"    LP stalled at iter {it}, score={best_score:.6f}")
                break
            # Try slightly larger M to escape
            g = solve_lp(f, M * 1.01)
            if g is None:
                break
            g_sum = float(np.sum(g))
            if g_sum <= f_sum * (1 + 1e-8):
                break
        
        # Normalize g
        g_norm = g * (S_t / g_sum)
        
        # Line search: find best blend factor
        best_t = 0
        best_s = best_score
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            f_new = (1.0 - t) * f + t * g_norm
            f_new = np.maximum(f_new, 0.0)
            f_new = normalize_sum(f_new, S_t)
            s = compute_obj(f_new)
            if s < best_s:
                best_s = s
                best_t = t
        
        if best_t == 0:
            stagnation += 1
            if stagnation > 3:
                break
            # Nudge with small random perturbation
            f = f + 0.01 * np.random.randn(n)
            f = np.maximum(f, 0.0)
            f = normalize_sum(f, S_t)
            continue
        
        f = (1.0 - best_t) * f + best_t * g_norm
        f = np.maximum(f, 0.0)
        f = normalize_sum(f, S_t)
        
        if best_s < best_score - 1e-10:
            best_score = best_s
            best_f = f.copy()
            stagnation = 0
            if verbose and it % 10 == 0:
                print(f"    LP iter {it}: score={best_score:.6f}, t={best_t:.2f}")
        else:
            stagnation += 1
    
    return best_f, best_score

def gradient_polish(f, max_iters=5000, verbose=False):
    """Gradient descent with momentum on the max autocorrelation."""
    n = len(f)
    S_t = np.sqrt(2.0 * n)
    f = normalize_sum(f.copy(), S_t)
    best_f = f.copy()
    best_score = compute_obj(f)
    momentum = np.zeros(n)
    decay = 0.9
    lr = 0.3
    tau = 5.0
    stagnation = 0
    
    for it in range(max_iters):
        tau = 5.0 + 15.0 * (it / 500.0)
        tau = min(tau, 40.0)
        lr = 0.3 * (0.9995 ** it)
        lr = max(lr, 0.01)
        
        grad = soft_gradient(f, tau)
        grad -= grad.mean()  # project out sum constraint
        
        momentum = decay * momentum - lr * grad
        f_new = f + momentum
        f_new = np.maximum(f_new, 0.0)
        f_new = normalize_sum(f_new, S_t)
        
        s = compute_obj(f_new)
        if s < best_score - 1e-10:
            best_score = s
            best_f = f_new.copy()
            stagnation = 0
        else:
            stagnation += 1
        
        f = f_new
        
        if stagnation > 100:
            break
    
    return best_f, best_score

    n = len(f)

"""
Optimizes non-negative sequence to minimize 2*n*max(autocorr)/(sum)^2.
Primary method: Iterative LP (convolution constraint linearization) with adaptive
step sizes and line search. Secondary: smooth gradient descent, multiplicative
updates, and coordinate descent polishing. Multi-start with diverse initializations
and multiple sequence lengths.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(np.max(c))
    S = float(np.sum(a))
    if S < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (S * S)


def normalize_sum(a, target=None):
    """Normalize sequence to have a specific sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = float(np.sum(a))
    if S < 1e-15:
        return a.copy()
    return a * (target / S)


def build_lp_matrix(f):
    """Build the constraint matrix A for the LP: A[k,j] = f[k-j] if valid."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = j + n
        A[k_start:k_end, j] = f[:n]
    return A


def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0. Returns g or None."""
    n = len(f)
    c_obj = -np.ones(n)
    A = build_lp_matrix(f)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def soft_max_gradient(a, tau):
    """Compute gradient of soft-max autocorrelation w.r.t. a using FFT."""
    n = len(a)
    c = np.convolve(a, a)
    c_max = c.max()
    w = np.exp(tau * (c - c_max))
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= w_sum
    a_rev = a[::-1]
    conv_result = np.convolve(w, a_rev, mode='full')
    grad = 2.0 * conv_result[n - 1:n - 1 + n]
    return grad


def hard_max_gradient(a):
    """Compute gradient of hard-max autocorrelation w.r.t. a."""
    n = len(a)
    c = np.convolve(a, a)
    k_star = int(np.argmax(c))
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    return grad


def lp_based_optimize(f, time_limit, verbose=True):
    """
    Main LP-based optimization loop.
    Iterates: solve LP, line search for best blend, update.
    """
    n = len(f)
    S_t = np.sqrt(2.0 * n)
    f = normalize_sum(f.copy(), S_t)
    best_f = f.copy()
    best_score = compute_obj(f)
    stagnation = 0
    it = 0
    t0 = time.monotonic()

    if verbose:
        print(f"  LP start: n={n}, score={best_score:.6f}")

    while time.monotonic() - t0 < time_limit:
        it += 1
        c = np.convolve(f, f)
        M = float(c.max())

        g = solve_lp(f, M)
        if g is None:
            if verbose:
                print(f"    LP failed at iter {it}")
            break

        g_sum = float(np.sum(g))
        f_sum = float(np.sum(f))
        ratio = g_sum / f_sum if f_sum > 1e-15 else 1.0

        if ratio <= 1.0 + 1e-7:
            stagnation += 1
            if stagnation <= 3:
                # Try slightly relaxed M
                g2 = solve_lp(f, M * 1.005)
                if g2 is not None:
                    g2_sum = float(np.sum(g2))
                    if g2_sum > f_sum * (1 + 1e-7):
                        g = g2
                        g_sum = g2_sum
                        ratio = g_sum / f_sum
                    else:
                        if stagnation > 2:
                            break
                else:
                    if stagnation > 2:
                        break
            else:
                break

        # Normalize g
        g_norm = g * (S_t / g_sum)

        # Line search: try multiple blend strategies
        best_t = 0.0
        best_s = best_score
        best_f_new = f.copy()

        # Additive blends
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            f_new = (1.0 - t) * f + t * g_norm
            f_new = np.maximum(f_new, 0.0)
            f_new = normalize_sum(f_new, S_t)
            s = compute_obj(f_new)
            if s < best_s:
                best_s = s
                best_t = t
                best_f_new = f_new.copy()

        # Geometric (multiplicative) blends
        for t in [0.3, 0.5, 0.7]:
            safe_f = np.maximum(f, 1e-10)
            safe_g = np.maximum(g_norm, 1e-10)
            f_new = np.power(safe_f, 1.0 - t) * np.power(safe_g, t)
            f_new = np.maximum(f_new, 0.0)
            f_new = normalize_sum(f_new, S_t)
            s = compute_obj(f_new)
            if s < best_s:
                best_s = s
                best_t = t
                best_f_new = f_new.copy()

        if best_s < best_score - 1e-12:
            f = best_f_new
            best_f = f.copy()
            best_score = best_s
            stagnation = 0
            if verbose and (it % 5 == 0 or it <= 5):
                print(f"    iter {it}: score={best_score:.6f}, ratio={ratio:.4f}, t={best_t:.2f}")
        else:
            stagnation += 1
            if stagnation > 3:
                # Small random nudge to escape
                f = f + 0.02 * np.random.randn(n)
                f = np.maximum(f, 0.0)
                f = normalize_sum(f, S_t)
                stagnation = 0
            elif stagnation > 8:
                break

    if verbose:
        print(f"  LP done: {it} iters, final score={best_score:.6f}")
    return best_f, best_score


def gradient_optimize(f, time_limit, verbose=False):
    """Gradient descent with momentum on soft-max autocorrelation."""
    n = len(f)
    S_t = np.sqrt(2.0 * n)
    f = normalize_sum(f.copy(), S_t)
    best_f = f.copy()
    best_score = compute_obj(f)
    momentum = np.zeros(n)
    decay = 0.9
    stagnation = 0
    it = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        it += 1
        tau = 5.0 + 15.0 * (it / 300.0)
        tau = min(tau, 50.0)
        lr = 0.3 * (0.9997 ** it)
        lr = max(lr, 0.01)

        grad = soft_max_gradient(f, tau)
        grad -= grad.mean()

        momentum = decay * momentum - lr * grad
        f_new = f + momentum
        f_new = np.maximum(f_new, 0.0)
        f_new = normalize_sum(f_new, S_t)

        s = compute_obj(f_new)
        if s < best_score - 1e-10:
            best_score = s
            best_f = f_new.copy()
            stagnation = 0
        else:
            stagnation += 1

        f = f_new

        if stagnation > 200:
            break

    if verbose:
        print(f"  Grad done: {it} iters, final score={best_score:.6f}")
    return best_f, best_score


def coordinate_polish(f, max_rounds=10, time_limit=30.0):
    """Coordinate descent: try changing individual elements."""
    n = len(f)
    S_t = np.sqrt(2.0 * n)
    f = normalize_sum(f.copy(), S_t)
    best_score = compute_obj(f)
    t0 = time.monotonic()

    for rnd in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            orig = f[i]
            # Generate candidates based on current value and peak structure
            c = np.convolve(f, f)
            k_star = int(np.argmax(c))
            j = k_star - i
            candidates = [0.0, orig * 0.5, orig * 1.5]
            if 0 <= j < n and j != i:
                candidates.extend([orig * 0.3, orig * 0.7, orig * 2.0, orig * 0.1])

            best_val = orig
            best_s = best_score
            for val in candidates:
                f[i] = val
                f = normalize_sum(f, S_t)
                s = compute_obj(f)
                if s < best_s - 1e-12:
                    best_s = s
                    best_val = val
            if best_val != orig:
                f[i] = best_val
                f = normalize_sum(f, S_t)
                best_score = best_s
                improved = True
            else:
                f[i] = orig

        if not improved:
            break

    return f, best_score


def multi_scale_optimize(f, time_limit, verbose=True):
    """
    Multi-scale optimization: coarsen, optimize, refine.
    This helps escape local minima by working at different resolutions.
    """
    n = len(f)
    if n < 64 or time_limit < 30:
        return lp_based_optimize(f, time_limit, verbose)

    # Coarsen by a factor of 2
    n_coarse = n // 2
    f_coarse = f[:n_coarse].copy()
    f_coarse = normalize_sum(f_coarse)

    # Optimize coarse
    t_coarse = time_limit * 0.4
    f_coarse, _ = lp_based_optimize(f_coarse, t_coarse, verbose=False)

    # Upsample and use as starting point for fine optimization
    f_fine = np.zeros(n)
    f_fine[:n_coarse] = f_coarse
    # Fill the rest by interpolation or just use the original
    f_fine = normalize_sum(f_fine)

    # Blend with original
    f_blend = 0.5 * f + 0.5 * f_fine
    f_blend = np.maximum(f_blend, 0.0)
    f_blend = normalize_sum(f_blend)

    # Fine optimization
    t_fine = time_limit * 0.5
    return lp_based_optimize(f_blend, t_fine, verbose)
