"""
Optimizes non-negative sequences to minimize 2*n*max(convolve(a,a))/(sum(a))^2.
Uses a combination of:
- Annealed soft-max gradient descent with momentum and FFT-based operations
- Spectral shaping (Gerchberg-Saxton style) for flat-spectrum design
- Active-set subgradient descent on hard peaks
- Coordinate descent polishing
- Multi-start with diverse initializations and multiple sequence lengths
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize(a, target=None):
    """Normalize sequence to have a specific sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)


def fft_convolve(a):
    """Compute full convolution using FFT."""
    n = len(a)
    N = 2 * n - 1
    N_fft = 1
    while N_fft < N:
        N_fft *= 2
    A = np.fft.rfft(a, N_fft)
    return np.fft.irfft(A * A, N_fft)[:N]


def fft_gradient(a, weights):
    """
    Compute gradient given weights on convolution lags.
    grad[i] = 2 * sum_{j=0}^{n-1} weights[i+j] * a[j]
    Using FFT-based cross-correlation.
    """
    n = len(a)
    # conv(weights, a_reversed)[i+n-1] = sum_j weights[i+j] * a[j]
    a_rev = a[::-1]
    needed = len(weights) + n - 1
    N_fft = 1
    while N_fft < needed:
        N_fft *= 2
    W = np.fft.rfft(weights, N_fft)
    Ar = np.fft.rfft(a_rev, N_fft)
    conv = np.fft.irfft(W * Ar, N_fft)
    return 2.0 * conv[n-1:n-1+n]


def hard_max_gradient(a):
    """Compute subgradient at the peak lag."""
    n = len(a)
    c = fft_convolve(a)
    k_star = int(np.argmax(c))
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    return grad, k_star


def soft_max_gradient(a, tau):
    """Compute soft-max gradient using FFT."""
    n = len(a)
    c = fft_convolve(a)
    c_shifted = c - c.max()
    w = np.exp(np.clip(tau * c_shifted, -500, 500))
    w_sum = w.sum()
    if w_sum < 1e-300:
        return hard_max_gradient(a)[0]
    w /= w_sum
    return fft_gradient(a, w)


def spectral_shaping_step(a, n_iters=50, alpha=0.7):
    """
    Gerchberg-Saxton style spectral shaping.
    Alternates between time-domain projection (a>=0, sum=S) and
    frequency-domain flattening.
    """
    n = len(a)
    S = a.sum()
    N = 2 * n - 1
    N_fft = 1
    while N_fft < N:
        N_fft *= 2
    
    a_work = a.copy()
    S_target = S
    
    for _ in range(n_iters):
        # Frequency domain: flatten magnitude
        A = np.fft.rfft(a_work, N_fft)
        mag = np.abs(A)
        phase = A / np.where(mag > 1e-30, mag, 1.0)
        
        # Compress dynamic range: new_mag = mag^alpha (alpha < 1 compresses)
        mag_min = mag.min()
        mag_max = mag.max()
        if mag_max < 1e-30:
            break
        # Normalize then compress
        mag_norm = mag / mag_max
        mag_compressed = mag_norm ** alpha
        mag_new = mag_compressed * mag_max
        
        # Alternative: truncate top and fill bottom
        # target_mag = S / np.sqrt(N_fft) * np.ones_like(mag)
        # mag_new = 0.5 * mag + 0.5 * target_mag
        
        A_new = mag_new * phase
        a_new = np.fft.irfft(A_new, N_fft)[:n]
        
        # Time domain: project onto a >= 0, sum = S_target
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-12:
            break
        a_new = a_new * (S_target / sn)
        
        a_work = a_new
    
    return a_work


def spectral_shaping_truncate(a, n_iters=30, frac=0.3):
    """
    Spectral shaping with truncation: cut the top frac% of power spectrum
    and redistribute to the bottom.
    """
    n = len(a)
    S = a.sum()
    N = 2 * n - 1
    N_fft = 1
    while N_fft < N:
        N_fft *= 2
    
    a_work = a.copy()
    
    for _ in range(n_iters):
        A = np.fft.rfft(a_work, N_fft)
        P = np.abs(A) ** 2
        phase = A / np.where(np.abs(A) > 1e-30, np.abs(A), 1.0)
        
        # Truncate top
        threshold = np.percentile(P, 100 * (1 - frac))
        excess = np.maximum(P - threshold, 0).sum()
        deficit_mask = P < threshold
        deficit = (threshold - P[deficit_mask]).sum()
        if deficit < 1e-30:
            break
        scale = excess / deficit
        
        P_new = P.copy()
        P_new[P > threshold] = threshold
        P_new[deficit_mask] = threshold - (threshold - P[deficit_mask]) * (1 + scale)
        # Ensure non-negative
        P_new = np.maximum(P_new, 0)
        
        mag_new = np.sqrt(P_new)
        A_new = mag_new * phase
        a_new = np.fft.irfft(A_new, N_fft)[:n]
        
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-12:
            break
        a_new = a_new * (S / sn)
        a_work = a_new
    
    return a_work


def gradient_descent_phase(a, deadline, phase_time, mode='soft'):
    """
    Gradient descent phase with momentum and annealing.
    mode: 'soft' for annealed soft-max, 'hard' for active-set
    """
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = normalize(a.copy(), S_t)
    best = a.copy()
    best_score = compute_obj(a)
    
    momentum = np.zeros(n)
    mu = 0.9
    lr = 0.5
    
    if mode == 'soft':
        tau_start = 3.0
        tau_end = 50.0
        n_expected = int(phase_time / 0.005)  # rough estimate
        n_expected = max(n_expected, 1000)
    else:
        tau_start = 0
        tau_end = 0
        n_expected = int(phase_time / 0.01)
        n_expected = max(n_expected, 500)
    
    no_improve = 0
    patience = 15
    
    for it in range(n_expected + 10000):
        if time.monotonic() > deadline:
            break
        
        # Anneal
        frac = min(1.0, it / max(1, n_expected))
        lr = 0.5 * (1.0 - 0.8 * frac) + 0.02
        
        if mode == 'soft':
            tau = tau_start + (tau_end - tau_start) * frac
            grad = soft_max_gradient(a, tau)
        else:
            grad, _ = hard_max_gradient(a)
        
        # Project gradient to maintain sum constraint
        grad = grad - grad.mean()
        
        # Momentum update
        momentum = mu * momentum - lr * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new, S_t)
        
        a = a_new
        
        # Check improvement periodically
        if it % 100 == 0:
            s = compute_obj(a)
            if s < best_score - 1e-9:
                best_score = s
                best = a.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve >= patience:
                    # Reset momentum and reduce lr
                    momentum *= 0.5
                    lr *= 0.5
                    no_improve = 0
                    if lr < 0.005:
                        break
    
    return best, best_score


def active_set_descent(a, deadline, max_iters=2000):
    """
    Active-set subgradient descent: focus on all lags near the peak.
    """
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = normalize(a.copy(), S_t)
    best = a.copy()
    best_score = compute_obj(a)
    
    momentum = np.zeros(n)
    mu = 0.85
    lr = 0.3
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = fft_convolve(a)
        c_max = c.max()
        
        # Find active lags (within 5% of max)
        threshold = c_max * 0.95
        active = np.where(c >= threshold)[0]
        
        # Compute weighted subgradient (average over active lags)
        weights = c[active] - threshold + c_max * 0.05  # positive weights
        weights = weights / weights.sum()
        
        grad = np.zeros(n)
        for k in active:
            w_k = (c[k] - threshold + c_max * 0.05)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += w_k * 2.0 * a[j]
        
        grad = grad / (c.max() * 0.05 * len(active) + 1e-30)
        grad = grad - grad.mean()
        
        momentum = mu * momentum - lr * grad
        a_new = a + momentum
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new, S_t)
        a = a_new
        
        if it % 50 == 0:
            s = compute_obj(a)
            if s < best_score - 1e-9:
                best_score = s
                best = a.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 10:
                    momentum *= 0.3
                    lr = max(0.01, lr * 0.7)
                    no_improve = 0
                    if lr < 0.005:
                        break
    
    return best, best_score


def coordinate_descent_polish(a, deadline, max_rounds=5):
    """
    Coordinate descent: try perturbing individual elements.
    """
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = normalize(a.copy(), S_t)
    best_score = compute_obj(a)
    c = fft_convolve(a)
    k_star = int(np.argmax(c))
    
    patience = 0
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        improved = False
        
        # Process elements in random order
        for i in np.random.default_rng(rnd).permutation(n):
            if time.monotonic() > deadline:
                break
            
            orig = a[i]
            if orig < 1e-6:
                continue
            
            # Try several candidates
            candidates = [0.0, orig * 0.3, orig * 0.7, orig * 1.3, orig * 1.7]
            
            # Add peak-aware candidates
            j = k_star - i
            if 0 <= j < n and j != i:
                if a[j] > orig:
                    candidates.append(orig * 0.2)
                    candidates.append(orig * 0.5)
                else:
                    candidates.append(orig * 1.5)
                    candidates.append(orig * 2.0)
            
            best_val = orig
            best_s = best_score
            
            for val in candidates:
                a[i] = val
                a_norm = normalize(a, S_t)
                a[:] = a_norm
                s = compute_obj(a)
                if s < best_s - 1e-10:
                    best_s = s
                    best_val = val
            
            if best_val != orig:
                a[i] = best_val
                a = normalize(a, S_t)
                best_score = best_s
                improved = True
                # Update peak
                c = fft_convolve(a)
                k_star = int(np.argmax(c))
            
            a[i] = orig  # restore for next iteration
        
        if not improved:
            patience += 1
            if patience >= 2:
                break
            # Try adding noise to escape
            a = a + np.random.default_rng(42 + rnd).normal(0, 0.01, n)
            a = np.maximum(a, 0)
            a = normalize(a, S_t)
            best_score = compute_obj(a)
        else:
            patience = 0
    
    return a, best_score


def lp_refine(a, deadline, max_iters=50):
    """
    LP-based refinement: maximize sum(g) s.t. convolve(a,g) <= M, g >= 0.
    """
    n = len(a)
    if n > 600:
        return a, compute_obj(a)
    
    S_t = np.sqrt(2.0 * n)
    a = normalize(a.copy(), S_t)
    best = a.copy()
    best_score = compute_obj(a)
    
    no_improve = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = fft_convolve(a)
        M = float(c.max())
        
        # Build constraint matrix (banded structure)
        c_obj = -np.ones(n)
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            for m in range(j, j + n):
                A[m, j] = a[m - j]
        
        b = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 30})
            if res.status != 0:
                break
            g = res.x
        except Exception:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-6:
            break
        g_norm = g * (S_t / g_sum)
        
        # Try blending
        improved = False
        for t_blend in [0.5, 0.3, 0.15, 0.05]:
            a_new = (1.0 - t_blend) * a + t_blend * g_norm
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize(a_new, S_t)
            s_new = compute_obj(a_new)
            if s_new < best_score - 1e-10:
                a = a_new
                best = a.copy()
                best_score = s_new
                improved = True
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 5:
                break
            # Nudge with gradient
            grad, _ = hard_max_gradient(a)
            grad -= grad.mean()
            a = a - 0.05 * grad
            a = np.maximum(a, 0)
            a = normalize(a, S_t)
            s = compute_obj(a)
            if s < best_score:
                best = a.copy()
                best_score = s
                no_improve = 0
        else:
            no_improve = 0
    
    return best, best_score


def optimize_single(a0, deadline):
    """
    Full optimization pipeline for a single starting point.
    """
    n = len(a0)
    a0 = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a0.sum() < 1e-6:
        a0 = np.ones(n) * 0.5
    a0 = normalize(a0)
    
    best = a0.copy()
    best_score = compute_obj(a0)
    
    print(f"    n={n}, initial={best_score:.6f}", flush=True)
    
    # Phase 1: Soft-max gradient descent
    t_phase = min(60.0, (deadline - time.monotonic()) * 0.25)
    a, s = gradient_descent_phase(a0, time.monotonic() + t_phase, t_phase, mode='soft')
    if s < best_score:
        best, best_score = a, s
    print(f"    After soft-GD: {best_score:.6f}", flush=True)
    
    if time.monotonic() > deadline - 10:
        return best, best_score
    
    # Phase 2: Spectral shaping
    t_phase = min(30.0, (deadline - time.monotonic()) * 0.15)
    n_gs_iters = max(10, int(t_phase / 0.002))
    a_spec = spectral_shaping_step(best, n_iters=n_gs_iters, alpha=0.6)
    a_spec = normalize(a_spec)
    s_spec = compute_obj(a_spec)
    if s_spec < best_score:
        best, best_score = a_spec, s_spec
        print(f"    After spectral: {best_score:.6f}", flush=True)
    else:
        # Try truncation variant
        a_spec2 = spectral_shaping_truncate(best, n_iters=n_gs_iters//2, frac=0.4)
        a_spec2 = normalize(a_spec2)
        s_spec2 = compute_obj(a_spec2)
        if s_spec2 < best_score:
            best, best_score = a_spec2, s_spec2
            print(f"    After spectral-trunc: {best_score:.6f}", flush=True)
    
    if time.monotonic() > deadline - 10:
        return best, best_score
    
    # Phase 3: Active-set descent
    t_phase = min(60.0, (deadline - time.monotonic()) * 0.25)
    a, s = active_set_descent(best, time.monotonic() + t_phase)
    if s < best_score:
        best, best_score = a, s
    print(f"    After active-set: {best_score:.6f}", flush=True)
    
    if time.monotonic() > deadline - 10:
        return best, best_score
    
    # Phase 4: Second soft-GD (refinement)
    t_phase = min(40.0, (deadline - time.monotonic()) * 0.2)
    a, s = gradient_descent_phase(best, time.monotonic() + t_phase, t_phase, mode='soft')
    if s < best_score:
        best, best_score = a, s
    print(f"    After 2nd soft-GD: {best_score:.6f}", flush=True)
    
    if n <= 512 and time.monotonic() > deadline - 30:
        return best, best_score
    
    # Phase 5: LP refinement (for smaller n)
    if n <= 400 and time.monotonic() < deadline - 20:
        t_phase = min(60.0, (deadline - time.monotonic()) * 0.3)
        a, s = lp_refine(best, time.monotonic() + t_phase)
        if s < best_score:
            best, best_score = a, s
        print(f"    After LP: {best_score:.6f}", flush=True)
    
    # Phase 6: Coordinate descent polish
    if n <= 512 and time.monotonic() < deadline - 5:
        a, s = coordinate_descent_polish(best, deadline - 2, max_rounds=3)
        if s < best_score:
            best, best_score = a, s
        print(f"    After coord-desc: {best_score:.6f}", flush=True)
    
    return best, best_score


def generate_starts(rng, lengths):
    """Generate diverse starting sequences."""
    starts = []
    
    # From height_sequence_1 if available
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        if gs.size >= 8 and np.all(np.isfinite(gs)):
            for L in lengths:
                if L <= len(gs):
                    starts.append(gs[:L].copy())
            # Perturbed versions
            for L in [min(len(gs), 384), min(len(gs), 512)]:
                for attempt in range(3):
                    noise = rng.normal(0, 0.1, L)
                    a = np.maximum(gs[:L] + noise, 0)
                    if a.sum() > 1e-6:
                        starts.append(a)
                # Shuffled
                idx = rng.permutation(L)
                starts.append(gs[idx].copy())
    except (NameError, UnboundLocalError):
        pass
    
    # Random starts
    for n in lengths:
        starts.append(rng.random(n) * 0.5 + 0.25)
        starts.append(rng.exponential(0.5, n) + 0.1)
    
    # Structured starts
    for n in lengths:
        i = np.arange(n, dtype=np.float64)
        starts.append(0.5 + 0.5 * np.cos(2 * np.pi * i * 3.0 / n))
        starts.append(0.5 + 0.5 * np.cos(2 * np.pi * i * 7.0 / n))
        starts.append(np.abs(np.sin(2 * np.pi * i * 2.0 / n)) + 0.1)
        starts.append(np.abs(np.sin(2 * np.pi * i * 5.0 / n) + 0.3 * np.cos(2*np.pi*i*3.0/n)) + 0.05)
        # Gaussian
        center = n / 2.0
        starts.append(np.exp(-0.5 * ((i - center) / (n * 0.2)) ** 2) + 0.05)
        # Two bumps
        starts.append(np.exp(-0.5 * ((i - n*0.25) / (n*0.1))**2) + np.exp(-0.5 * ((i - n*0.75) / (n*0.1))**2) + 0.05)
        # Triangular
        starts.append(np.abs(1.0 - 2.0 * i / n) + 0.1)
        # Inverse (peaks at edges)
        starts.append(np.abs(2.0 * i / n - 1.0) + 0.1)
    
    # Sparse-like starts
    for n in [256, 384, 512]:
        a = np.zeros(n)
        a[::2] = rng.random(n // 2 + 1) * 0.5 + 0.1
        starts.append(a[:n])
    
    # "Anti-palindrome" starts (small center, large edges)
    for n in [384, 512]:
        i = np.arange(n, dtype=np.float64)
        starts.append(np.exp(-0.5 * ((i - n/2) / (n*0.15))**2) * 0.2 + 0.1)
    
    return starts


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Uses multi-phase optimization from diverse
    starting points with spectral shaping, gradient methods, and LP.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)
    
    best_seq = None
    best_score = float('inf')
    
    print(f"Starting optimization, budget={budget_s}s", flush=True)
    
    # Lengths to try (prioritize 384 which was previous best, plus others)
    lengths = [384, 256, 512, 768]
    
    # Generate starts
    starts = generate_starts(rng, lengths)
    
    # Sort: prioritize length 384 first, then by length
    def start_priority(s):
        n = len(s)
        length_order = {384: 0, 256: 1, 512: 2, 768: 3}.get(n, 4)
        return length_order
    
    starts.sort(key=start_priority)
    
    # Deduplicate
    unique_starts = []
    seen = set()
    for s in starts:
        key = (len(s), round(float(s.sum()), 0))
        if key not in seen:
            seen.add(key)
            unique_starts.append(s)
    starts = unique_starts
    
    print(f"Total starts: {len(starts)}", flush=True)
    
    # Time allocation: shorter sequences get less time per attempt
    # but we try more of them
    results = []
    
    for idx, a0 in enumerate(starts):
        if time.monotonic() > deadline - 45.0:
            break
        
        n = len(a0)
        remaining = deadline - time.monotonic() - 30
        
        # Allocate time based on length and position
        if n <= 256:
            t_alloc = min(80.0, remaining * 0.15)
        elif n == 384:
            t_alloc = min(120.0, remaining * 0.2)
        elif n == 512:
            t_alloc = min(100.0, remaining * 0.15)
        else:
            t_alloc = min(80.0, remaining * 0.1)
        
        t_alloc = max(20.0, t_alloc)
        
        if time.monotonic() + t_alloc > deadline - 30:
            t_alloc = deadline - time.monotonic() - 30
            if t_alloc < 15:
                break
        
        print(f"\n[Start {idx}/{''.join(str(x) for x in [len(a) for a in starts]])}] n={n}, time_alloc={t_alloc:.0f}s", flush=True)
        
        seq, score = optimize_single(a0, time.monotonic() + t_alloc)
        
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"  *** NEW BEST: score={best_score:.6f}, n={n} ***", flush=True)
        results.append((n, score))
    
    # Second pass: focus on the best length with perturbed best solution
    if best_seq is not None:
        best_n = len(best_seq)
        print(f"\n=== Second pass at n={best_n} ===", flush=True)
        
        extra_starts = []
        for trial in range(5):
            noise = rng.normal(0, 0.05, best_n)
            a0 = np.maximum(best_seq + noise, 0)
            if a0.sum() > 1e-6:
                extra_starts.append(a0)
        
        # Also try spectral-shaped version as start
        a_spec = spectral_shaping_step(best_seq, n_iters=100, alpha=0.5)
        if a_spec.sum() > 1e-6:
            extra_starts.append(a_spec)
        
        for idx, a0 in enumerate(extra_starts):
            if time.monotonic() > deadline - 20.0:
                break
            remaining = deadline - time.monotonic() - 15
            t_alloc = min(100.0, remaining / max(1, len(extra_starts) - idx))
            if t_alloc < 15:
                break
            
            seq, score = optimize_single(a0, time.monotonic() + t_alloc)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"  *** NEW BEST (pass 2): score={best_score:.6f} ***", flush=True)
    
    # Final polish
    if best_seq is not None and time.monotonic() < deadline - 10:
        print(f"\n=== Final polish (score={best_score:.6f}) ===", flush=True)
        remaining = deadline - time.monotonic() - 5
        seq, score = optimize_single(best_seq, time.monotonic() + remaining)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
    
    # Validate
    if best_seq is not None:
        final_score = compute_obj(best_seq)
        print(f"\nFinal: n={len(best_seq)}, score={final_score:.6f}", flush=True)
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence: {eval_score:.6f}", flush=True)
        except Exception as e:
            print(f"evaluate_sequence error: {e}", flush=True)
        return [float(x) for x in best_seq]
    
    # Fallback
    print("Falling back to uniform sequence.")
    return [1.0] * 384
