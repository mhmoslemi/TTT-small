def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]

def compute_gradient(x, beta):
    """Compute gradient of 2n*LSE(beta*conv(x,x))/(sum(x))^2"""
    n = len(x)
    S = x.sum()
    B = fast_convolve(x)
    
    # Softmax of beta * B
    beta_B = beta * B
    beta_B -= beta_B.max()
    exp_B = np.exp(beta_B)
    p = exp_B / exp_B.sum()
    
    LSE = (1.0/beta) * np.log(exp_B.sum() + 1e-300)
    
    # grad_i = (4n/S^2) * sum_k p_k * x_{k-i} - 2n*LSE/S^3
    # sum_k p_k * x_{k-i} for valid k-i: this is cross-correlation
    # c_i = sum_{k} p_k * x_{k-i} where 0 <= k-i < n
    # = sum_{j=0}^{n-1} x_j * p_{j+i} where j+i < 2n-1
    
    c = np.zeros(n)
    for i in range(n):
        # k ranges from i to min(i+n-1, 2n-2)
        k_end = min(i + n, 2*n - 1)
        c[i] = np.dot(p[i:k_end], x[:k_end-i])
    
    grad = (4.0*n/S**2) * c - (2.0*n*LSE/S**3)
    return grad

def smooth_optimize(x0, time_limit, beta_start=100, beta_end=10000):
    """Optimize using Adam on the smooth LSE objective."""
    n = len(x0)
    x = x0.copy().astype(np.float64)
    x = np.maximum(x, 0)
    x /= x.sum()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = 0.1
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    beta = beta_start
    best_x = x.copy()
    best_obj = 2.0 * n * fast_convolve(x).max()
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        # Update beta (anneal)
        progress = min(1.0, t / 500.0)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        grad = compute_gradient(x, beta)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        x_new = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        x_new = np.maximum(x_new, 0)
        s = x_new.sum()
        if s < 1e-12:
            x_new = x
        else:
            x_new /= s
        
        # Check improvement
        conv = fast_convolve(x_new)
        obj = 2.0 * n * conv.max()
        
        if obj < best_obj - 1e-10:
            best_obj = obj
            best_x = x_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                lr *= 0.8
                no_improve = 0
                if lr < 1e-6:
                    break
        
        x = x_new
    
    return best_x, best_obj

"""
Hybrid smooth-gradient + LP optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses Adam with log-sum-exp smoothing, periodic LP jumps, and structured perturbations.
"""

import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_smooth_gradient(x, beta):
    """Compute gradient of 2n*LSE(beta*conv(x,x))/(sum(x))^2 using FFT."""
    n = len(x)
    S = x.sum()
    B = fast_convolve(x)

    beta_B = beta * B
    beta_B -= beta_B.max()
    exp_B = np.exp(beta_B)
    p = exp_B / exp_B.sum()

    LSE = (1.0 / beta) * np.log(exp_B.sum() + 1e-300)

    # c_i = sum_k x[k] * p[k+i] (cross-correlation at lag i)
    L = 1 << (3 * n).bit_length()
    X = np.fft.rfft(x, L)
    P = np.fft.rfft(p, L)
    c = np.fft.irfft(X * np.conj(P), L)[:n]

    grad = (4.0 * n / (S * S)) * c - (2.0 * n * LSE / (S ** 3))
    return grad


def adam_smooth_optimize(x0, time_limit, beta_start=50.0, beta_end=50000.0, lr=0.05):
    """Adam optimization on the smooth LSE objective with annealing."""
    n = len(x0)
    x = np.maximum(x0.astype(np.float64), 0.0)
    s = x.sum()
    if s < 1e-12:
        x = np.ones(n) / n
    else:
        x /= s

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8

    best_x = x.copy()
    best_obj = 2.0 * n * fast_convolve(x).max()

    t0 = time.monotonic()
    no_improve = 0
    patience = 300

    while time.monotonic() - t0 < time_limit:
        t += 1
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        beta = beta_start * (beta_end / beta_start) ** (progress ** 0.7)

        grad = compute_smooth_gradient(x, beta)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        x_new = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        x_new = np.maximum(x_new, 0.0)
        s = x_new.sum()
        if s < 1e-12:
            x_new = x
        else:
            x_new /= s

        conv = fast_convolve(x_new)
        obj = 2.0 * n * conv.max()

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_x = x_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > patience:
                lr *= 0.7
                no_improve = 0
                patience = int(patience * 1.3)
                if lr < 1e-7:
                    break

        x = x_new

        if t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Adam t={t}, obj={best_obj:.8f}, beta={beta:.0f}, lr={lr:.2e}, elapsed={elapsed:.1f}s")

    return best_x, best_obj


def make_lp_structure(n):
    """Precompute row/col indices for cross-convolution constraint matrix."""
    rows_list = []
    cols_list = []
    for i in range(n):
        for j in range(n):
            rows_list.append(i + j)
            cols_list.append(i)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)


def solve_lp(f, M, rows, cols, n, timeout=20):
    """Solve max sum(g) s.t. g>=0, (f*g)[k]<=M for all k."""
    vals = np.tile(f, n)
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                           options={'disp': False, 'time_limit': timeout})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def lp_alternate(f0, time_limit, n_lp_calls=8):
    """LP-guided alternating with backtracking line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    f /= f.sum()

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()

    rows, cols = make_lp_structure(n)

    t0 = time.monotonic()
    t = 1.0
    patience = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        iter_count += 1
        g = solve_lp(f, M, rows, cols, n, timeout=15)
        if g is None:
            break

        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g /= g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            break

        improved = False
        for t_try in [t, 0.7 * t, 0.5 * t, 0.3 * t, 0.1 * t]:
            f_new = (1.0 - t_try) * f + t_try * g
            f_new = np.maximum(f_new, 0.0)
            s = f_new.sum()
            if s < 1e-12:
                continue
            f_new /= s
            conv_new = fast_convolve(f_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < best_obj - 1e-12:
                f = f_new
                M = conv_new.max()
                best_obj = obj_new
                best_f = f.copy()
                patience = 0
                t = min(1.0, t * 1.05)
                improved = True
                break

        if not improved:
            patience += 1
            t *= 0.85
            if patience > 60:
                break

        if iter_count % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP iter {iter_count}, obj={best_obj:.8f}, elapsed={elapsed:.1f}s")

    return best_f, best_obj


def mirror_polish(f0, time_limit, t_param=80.0, eta0=0.2):
    """Multiplicative mirror descent polishing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    f /= f.sum()

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    eta = eta0
    patience = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        max_conv = conv.max()
        if max_conv < 1e-15:
            break

        scaled = t_param * conv / max_conv
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-12:
            w = np.zeros(len(conv))
            w[int(np.argmax(conv))] = 1.0
        else:
            w /= w_sum

        sg = np.zeros(n)
        thresh = w.max() * 0.001
        active = np.where(w > thresh)[0]
        for k in active:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            sg[j_start:j_end + 1] += w[k] * f[k - j_end:k - j_start + 1][::-1]

        sg_c = sg - sg.mean()
        lf = -eta * sg_c
        lf -= lf.max()
        f = f * np.exp(lf)
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s < 1e-12:
            f = np.ones(n) / n
        else:
            f /= s

        conv = fast_convolve(f)
        obj = 2.0 * n * conv.max()

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
            patience = 0
        else:
            patience += 1
            if patience % 20 == 0:
                eta *= 0.6
                if eta < 0.005:
                    t_param = min(t_param * 1.5, 1000.0)
                    eta = 0.15
            if patience > 200:
                break

    return best_f, best_obj


def structured_perturb(x, rng, strength=0.1):
    """Add structured (low-frequency) perturbations to escape local minima."""
    n = len(x)
    x_new = x.copy()
    # Add smooth random perturbation
    freqs = np.random.choice(np.arange(1, n // 4 + 1), size=3, replace=False)
    phases = rng.uniform(0, 2 * np.pi, size=3)
    amp = rng.uniform(0.5, 1.5, size=3)
    for i in range(3):
        x_new += strength * amp[i] * np.sin(2 * np.pi * freqs[i] * np.arange(n) / n + phases[i])
    x_new = np.maximum(x_new, 0.0)
    s = x_new.sum()
    if s > 1e-12:
        x_new /= s
    return x_new


def gen_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uni'] = np.ones(n) / n

    for sf in [0.1, 0.15, 0.2, 0.3]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'g{int(sf * 100)}'] = g

    bi = np.zeros(n)
    w = max(1, n // 5)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    inits['bi'] = bi / bi.sum()

    cb = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    inits['cos1'] = cb / cb.sum()

    cb2 = 0.5 * (1 + np.cos(4 * np.pi * x / n))
    inits['cos2'] = cb2 / cb2.sum()

    # Multi-frequency
    mf = 0.5 * (1 + 0.3 * np.cos(2 * np.pi * x / n) + 0.2 * np.cos(6 * np.pi * x / n))
    mf = np.maximum(mf, 0.01)
    inits['mf'] = mf / mf.sum()

    # Raised cosine (Bartlett-like)
    rc = np.sin(np.pi * x / n) ** 2
    rc = np.maximum(rc, 0.01)
    inits['rc'] = rc / rc.sum()

    # Power law
    r = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = r / r.sum()
    inits['inv'] = (1.0 / r) / (1.0 / r).sum()

    if parent is not None:
        if parent.size >= n:
            pp = np.maximum(parent[:n], 0.0)
            if pp.sum() > 1e-10:
                inits['par'] = pp / pp.sum()
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, parent.size), parent)
            pp = np.maximum(pp, 0.0)
            if pp.sum() > 1e-10:
                inits['par'] = pp / pp.sum()

    return inits

    if remaining > 50 and top:
        print(f"Phase 2: LP refinement on top {len(top)}, {remaining:.0f}s remaining")
        t_each = remaining / len(top)
        for idx, (sc, n, name, f) in enumerate(top):
            if time.monotonic() > deadline - 25:
                break
            ta = min(t_each, deadline - time.monotonic() - 25)
            if ta < 10:
                break

            # LP optimization
            try:
                f2, obj2 = lp_alternate(f, time_limit=min(ta * 0.5, deadline - time.monotonic() - 20))
                sc2 = float(evaluate_sequence(f2.tolist()))
                if np.isfinite(sc2) and sc2 < best_score:
                    best_score = sc2
                    best_seq = f2.copy()
                    print(f"  LP[{idx}] n={n} {name} score={sc2:.6f}")
            except Exception as e:
                print(f"  LP error [{idx}]: {e}")

            # Smooth polish on LP result
            if time.monotonic() < deadline - 15:
                tr = min(ta * 0.3, deadline - time.monotonic() - 15)
                try:
                    f3, _ = adam_smooth_optimize(f2, time_limit=tr, beta_start=200, beta_end=50000, lr=0.03)
                    sc3 = float(evaluate_sequence(f3.tolist()))
                    if np.isfinite(sc3) and sc3 < best_score:
                        best_score = sc3
                        best_seq = f3.copy()
                        print(f"  Smooth[{idx}] n={n} {name} score={sc3:.6f}")
                except Exception:
                    pass

    # Phase 3: Deep iteration on the best with perturbations
    if best_seq is not None and time.monotonic() < deadline - 20:
        print("Phase 3: Deep optimization with perturbations")
        n_best = len(best_seq)
        current = best_seq.copy() / best_seq.sum()
        remaining = deadline - time.monotonic() - 15

        # Multiple rounds of optimize -> perturb -> optimize
        n_rounds = min(5, int(remaining // 30))
        for round_i in range(n_rounds):
            if time.monotonic() > deadline - 15:
                break
            ta = (deadline - time.monotonic() - 15) / (n_rounds - round_i)
            if ta < 10:
                break

            # Optimize
            try:
                f_opt, _ = adam_smooth_optimize(current, time_limit=ta * 0.5,
                                                beta_start=100, beta_end=80000, lr=0.03)
                sc = float(evaluate_sequence(f_opt.tolist()))
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = f_opt.copy()
                    print(f"  Deep[{round_i}] score={sc:.6f}")
                current = f_opt
            except Exception:
                pass

            # Perturb
            if time.monotonic() < deadline - 20 and round_i < n_rounds - 1:
                current = structured_perturb(best_seq / best_seq.sum(), rng, strength=0.05 / (round_i + 1))

    # Phase 4: Final polish with mirror descent
    if best_seq is not None and time.monotonic() < deadline - 8:
        print("Phase 4: Final mirror polish")
        tr = deadline - time.monotonic() - 8
        for tp, eta in [(100.0, 0.15), (300.0, 0.08), (500.0, 0.04)]:
            if tr < 3:
                break
            try:
                f5, _ = mirror_polish(best_seq, time_limit=tr / 3.0, t_param=tp, eta0=eta)
                sc5 = float(evaluate_sequence(f5.tolist()))
                if np.isfinite(sc5) and sc5 < best_score:
                    best_score = sc5
                    best_seq = f5.copy()
                    print(f"  Mirror tp={tp:.0f} score={sc5:.6f}")
                tr = deadline - time.monotonic() - 8
            except Exception:
                pass

    # Phase 5: One final LP push
    if best_seq is not None and time.monotonic() < deadline - 5:
        tr = deadline - time.monotonic() - 5
        if tr > 10:
            try:
                f6, _ = lp_alternate(best_seq, time_limit=tr * 0.7)
                sc6 = float(evaluate_sequence(f6.tolist()))
                if np.isfinite(sc6) and sc6 < best_score:
                    best_score = sc6
                    best_seq = f6.copy()
                    print(f"  Final LP score={sc6:.6f}")
            except Exception:
                pass

    if best_seq is None:
        best_seq = np.ones(128)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final validation
    final_score = float(evaluate_sequence([float(x) for x in best_seq]))
    if not np.isfinite(final_score):
        best_seq = np.ones(512)
        final_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL: n={len(best_seq)}, score={final_score:.6f}")
    return [float(x) for x in best_seq]
