"""
Multi-phase optimization for minimizing normalized peak autocorrelation.
Combines fast subgradient descent, LP-based refinement, and simulated annealing.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)

    """Subgradient descent on the peak autocorrelation."""
    f = f_init.astype(np.float64).copy()

A better approach might be to use a smooth approximation of the max (like the log-sum-exp) and compute its gradient.

The log-sum-exp of the autocorrelation: $\phi_\tau(a) = \tau \log \sum_k \exp((a*a)[k]/\tau)$

As $\tau \to 0$, this approaches $\max_k (a*a)[k]$. The gradient is:
$\frac{\partial \phi_\tau}{\partial a_j} = \sum_k \frac{\exp((a*a)[k]/\tau)}{\sum_l \exp((a*a)[l]/\tau)} \cdot \frac{\partial (a*a)[k]}{\partial a_j}$

"""
Multi-phase optimization for minimizing normalized peak autocorrelation of
non-negative step functions. Combines LP-based directional optimization,
smooth gradient descent (annealed log-sum-exp), and simulated annealing.
Key improvements: better multi-start strategy, annealed gradient with
adaptive temperature, peak-focused SA moves, and dynamic time allocation.

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def build_constraint_A(f):
    """Build constraint matrix A for the LP. A[k,j] = f[k-j] if 0<=k-j<n."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. f*g <= M, g >= 0."""
    n = len(f)
    A = build_constraint_A(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                     method='highs', options={'presolve': True, 'time_limit': 20})
    if result.success and result.x is not None:
        return np.maximum(result.x, 0.0)
    return None


def lp_optimize(f_init, time_limit, t_init=0.3, verbose=False):
    """LP-based iterative optimization with adaptive step size and restarts."""
    f = f_init.astype(np.float64).copy()
    n = len(f)
    sum_f = np.sum(f)
    best_f = f.copy()
    best_score = local_score(f)
    t0 = time.monotonic()
    t = t_init
    stagnant = 0
    iteration = 0
    rng_local = np.random.default_rng(12345 + n)
    consecutive_no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(f, f)
        M = np.max(conv)

        g = solve_lp(f, M)
        if g is None:
            break

        sum_g = np.sum(g)
        if sum_g < 1e-10:
            break

        g_rescaled = g * (sum_f / sum_g)

        f_new = (1.0 - t) * f + t * g_rescaled
        f_new = np.clip(f_new, 0.0, None)
        sum_new = np.sum(f_new)
        if sum_new < 1e-10:
            t *= 0.5
            continue
        f_new = f_new * (sum_f / sum_new)

        score_new = local_score(f_new)
        iteration += 1

        if score_new < best_score - 1e-14:
            f = f_new
            best_score = score_new
            best_f = f.copy()
            stagnant = 0
            consecutive_no_improve = 0
            if t < 0.7:
                t = min(0.7, t * 1.03)
        else:
            f = f_new
            stagnant += 1
            consecutive_no_improve += 1
            if stagnant > 15:
                t *= 0.75
                stagnant = 0
            if stagnant > 50:
                # More aggressive perturbation
                f = f + 0.05 * rng_local.standard_normal(n)
                f = np.clip(f, 0.0, None)
                f = f * (sum_f / np.sum(f))
                t = 0.25
                stagnant = 0

                f_random = f_random / np.sum(f_random) * sum_f
                f = 0.7 * f + 0.3 * f_random
                f = np.clip(f, 0.0, None)
                f = f * (sum_f / np.sum(f))
                t = 0.4
                stagnant = 0
                consecutive_no_improve = 0

            print(f"    LP iter={iteration}, t={t:.3f}, score={local_score(f):.6f}, "
                  f"best={best_score:.6f}, elapsed={time.monotonic()-t0:.1f}s")

    return best_f, best_score


def annealed_gradient(f_init, time_limit, verbose=False):
    """
    Gradient descent using annealed log-sum-exp approximation of max.
    Uses parameterization a = softmax(theta) * S to ensure positivity.
    Temperature anneals from high (smooth) to low (sharp).
    """
    n = len(f_init)
    S = np.sqrt(2.0 * n)
    f_safe = np.clip(f_init, 1e-8, None)
    theta = np.log(f_safe / np.sum(f_safe))

    momentum1 = np.zeros(n)
    momentum2 = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    base_lr = 0.5
    step_count = 0
    best_theta = theta.copy()
    best_score = local_score(f_init)
    t0 = time.monotonic()

    # Temperature schedule: start high, end low
    tau_start = 5.0 / n
    tau_end = 0.1 / n

    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        # Exponential annealing
        tau = tau_start * (tau_end / tau_start) ** elapsed_frac

        # Compute softmax parameterization
        t_shifted = theta - np.max(theta)
        e = np.exp(t_shifted)
        a = e / np.sum(e) * S

        # Compute smooth gradient via log-sum-exp
        conv = fft_convolve(a, a)
        conv_max = np.max(conv)
        shifted_conv = (conv - conv_max) / tau
        exp_conv = np.exp(shifted_conv)
        sum_exp = np.sum(exp_conv)
        w = exp_conv / sum_exp

        # Gradient of sum_k w_k * (a*a)[k] with respect to a
        # d(a*a)[k]/da_j = 2*a[k-j] if valid
        # We need: grad_a[j] = sum_k w_k * 2 * a[k-j] (over valid k,j)
        # This is 2 * (rev_a * w) viewed appropriately
        rev_a = a[::-1]
        corr = fft_convolve(rev_a, w)
        # corr[m] = sum_j rev_a[j] * w[m-j] = sum_j a[n-1-j] * w[m-j]
        # We want: for each j, sum_k w[k] * a[k-j]
        # Let l = k-j, so k = j+l: sum_l w[j+l] * a[l]
        # This is (w * a)[j] in cross-correlation sense
        # Actually: sum_k w[k] * a[k-j] = cross-correlation of w and a at lag j
        # = sum_k w[k] * a[k-j]
        # Using FFT: this is ifft(fft(w) * conj(fft(a))) but let's use convolution
        # sum_k w[k] * a[k-j] = (w * a_reversed)[n-1-j] ... let me think differently
        # 
        # Alternative: use the fact that sum_k w[k] * (a*a)[k]'s gradient w.r.t. a[j]
        # = sum_k w[k] * 2*a[k-j] (for valid k-j)
        # = 2 * sum_k w[k] * a[k-j] * 1[0<=k-j<n]
        # Let m = k-j: = 2 * sum_m w[m+j] * a[m] * 1[0<=m<n, 0<=m+j<2n-1]
        # This is 2 * (w_shifted * a) where w_shifted depends on j...
        # 
        # Simpler approach: use the convolution trick
        # grad_a[j] = 2 * sum_k w[k] * a[k-j] for valid indices
        # = 2 * (w full-convoluted with a_flipped)[something]
        
        # Let's compute it directly using the correlation
        # cross_corr(j) = sum_k w[k] * a[k-j] for valid k, k-j
        # This equals: for each j, sum over valid k of w[k]*a[k-j]
        # = sum over m=k-j of w[m+j]*a[m]
        # Let's compute via FFT: ifft(fft(w, 2n) * conj(fft(a, 2n))) gives circular cross-corr
        # For linear: pad to size >= 2n
        
        size_cc = 1
        while size_cc < 2 * n:
            size_cc <<= 1
        fw = np.fft.rfft(w, size_cc)
        fa = np.fft.rfft(a, size_cc)
        cross_corr = np.fft.irfft(fw * np.conj(fa), size_cc)
        # cross_corr[j] = sum_k w[k] * a[k-j] (circular, but we padded so it's linear for j in [0,n-1])
        # Actually for linear cross-correlation:
        # cross_corr[j] = sum_k w[k] * a[k-j] where we need 0<=k<len(w) and 0<=k-j<len(a)
        # With padding, this should work for j in [0, n-1]
        
        grad_a = 2.0 * cross_corr[:n]
        
        # Subtract mean to maintain normalization (constraint sum(a) = S)
        inner = np.dot(a, grad_a) / S
        g_theta = a * (grad_a - inner)

        # Adam update
        step_count += 1
        momentum1 = beta1 * momentum1 + (1.0 - beta1) * g_theta
        momentum2 = beta2 * momentum2 + (1.0 - beta2) * (g_theta ** 2)
        m1_hat = momentum1 / (1.0 - beta1 ** step_count)
        m2_hat = momentum2 / (1.0 - beta2 ** step_count)
        lr = base_lr / np.sqrt(1.0 + 0.005 * step_count)
        theta = theta - lr * m1_hat / (np.sqrt(m2_hat) + eps)

        # Compute score
        t_shifted = theta - np.max(theta)
        a_check = np.exp(t_shifted) / np.sum(np.exp(t_shifted)) * S
        score = local_score(a_check)

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_theta = theta.copy()

        if verbose and step_count % 200 == 0:
            print(f"    GradStep={step_count}, score={score:.6f}, best={best_score:.6f}, tau={tau:.5f}")

    t_shifted = best_theta - np.max(best_theta)
    a_best = np.exp(t_shifted) / np.sum(np.exp(t_shifted)) * S
    return a_best, local_score(a_best)


def sa_optimize(f_init, time_limit, rng, aggressive=False):
    """Simulated annealing with peak-focused moves."""
    f = f_init.astype(np.float64).copy()
    n = len(f)
    sum_f = np.sum(f)
    best_f = f.copy()
    best_score = local_score(f)
    current_score = best_score
    t0 = time.monotonic()
    
    if aggressive:
        temperature = 0.008
        cooling = 0.9998
    else:
        temperature = 0.004
        cooling = 0.99985
    
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        candidate = f.copy()
        r = rng.random()

        if r < 0.30:
            # Peak-focused: reduce elements contributing to peak, increase others
            conv = fft_convolve(f, f)
            peak_idx = int(np.argmax(conv))
            contribs = np.zeros(n)
            j_min = max(0, peak_idx - n + 1)
            j_max = min(peak_idx, n - 1)
            for i in range(j_min, j_max + 1):
                contribs[i] = f[i] * f[peak_idx - i]
            top_k = max(4, n // 8)
            sorted_idx = np.argsort(contribs)[::-1]
            top_idx = sorted_idx[:top_k]
            candidate[top_idx] *= np.exp(rng.normal(-0.04, 0.015, size=top_k))
            other_idx = np.setdiff1d(np.arange(n), top_idx)
            if len(other_idx) > 0:
                n_inc = max(3, min(8, len(other_idx) // 4))
                inc_idx = rng.choice(other_idx, size=n_inc, replace=False)
                candidate[inc_idx] *= np.exp(rng.normal(0.04, 0.015, size=n_inc))
        elif r < 0.50:
            # Block modification
            n_blocks = rng.integers(1, 4)
            for _ in range(n_blocks):
                width = max(1, int(rng.integers(1, max(2, n // 5))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.03, size=width))
        elif r < 0.70:
            # Smooth perturbation
            candidate *= np.exp(rng.normal(0.0, 0.01, size=n))
            # Add small structured component
            freq = rng.integers(2, max(3, n // 4))
            candidate += 0.01 * f * np.sin(2.0 * np.pi * freq * np.arange(n) / n)
        elif r < 0.85:
            # Shift + modify
            shift = int(rng.integers(1, max(2, n // 15)))
            candidate = np.roll(f, shift)
            candidate *= np.exp(rng.normal(0.0, 0.008, size=n))
        else:
            # Large random jump (to escape local optima)
            if rng.random() < 0.3:
                # Blend with random
                f_rand = np.abs(rng.standard_normal(n)) + 0.05
                f_rand = f_rand / np.sum(f_rand) * sum_f
                candidate = 0.6 * f + 0.4 * f_rand
            else:
                # Moderate perturbation
                candidate *= np.exp(rng.normal(0.0, 0.03, size=n))

        candidate = np.clip(candidate, 0.0, None)
        cand_sum = np.sum(candidate)
        if cand_sum < 1e-10:
            continue
        candidate = candidate * (sum_f / cand_sum)

        new_score = local_score(candidate)
        iter_count += 1

        if new_score < current_score:
            f = candidate
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_f = candidate.copy()
        elif temperature > 1e-6 and rng.random() < np.exp(-(new_score - current_score) / temperature):
            f = candidate
            current_score = new_score

        temperature *= cooling

        if aggressive and iter_count % 500 == 0:
            # Periodic restart
            if rng.random() < 0.05:
                f = best_f + 0.1 * rng.standard_normal(n)
                f = np.clip(f, 0.0, None)
                f = f * (sum_f / np.sum(f))
                current_score = local_score(f)

    return best_f, best_score


def make_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    t = np.arange(n, dtype=np.float64)

    # 1. Provided-like: near uniform with noise
    a = 1.0 + 0.15 * rng.standard_normal(n)
    inits.append(np.clip(a, 0.01, None))

    # 2. Chirp
    alpha = 0.2 + 0.4 * rng.random()
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a += 0.03 * rng.standard_normal(n)
    inits.append(np.clip(a, 0.01, None))

    # 3. Multitone
    a = np.ones(n)
    for _ in range(5):
        k = int(rng.integers(1, max(2, n // 4)))
        amp = rng.uniform(0.08, 0.25)
        phase = rng.uniform(0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a += 0.02 * rng.standard_normal(n)
    inits.append(np.clip(a, 0.01, None))

    # 4. Ramp + noise
    a = 1.0 + 0.3 * t / n
    a += 0.05 * rng.standard_normal(n)
    inits.append(np.clip(a, 0.01, None))

    # 5. Bimodal
    a = 1.0 + 0.3 * np.cos(2.0 * np.pi * t / n)
    a += 0.04 * rng.standard_normal(n)
    inits.append(np.clip(a, 0.01, None))

    # 6. Exponential decay shape
    a = np.exp(-0.5 * np.abs(t - n/2) / (n/6)) + 0.2
    a += 0.03 * rng.standard_normal(n)
    inits.append(np.clip(a, 0.01, None))

    # 7. Pure random
    a = 0.5 + 0.8 * rng.random(n)
    inits.append(a)

    return inits

    """
    Multi-phase optimization:
    Phase 1: Multi-start LP optimization across different initializations and lengths.
    Phase 2: Annealed gradient refinement on best candidates.
    Phase 3: Aggressive SA polish.
    Phase 4: Final LP refinement.
    """
    t_global = time.monotonic()
    deadline = t_global + float(budget_s) - 30.0
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Check provided sequence
    provided_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s_prov = local_score(provided_seq)
        print(f"Provided sequence: n={len(provided_seq)}, score={s_prov:.6f}")
        if s_prov < best_score:
            best_score = s_prov
            best_seq = provided_seq.copy()

    n_prov = len(provided_seq) if provided_seq is not None else 256

    # === Phase 1: Multi-start LP optimization ===
    print("\n" + "=" * 60)
    print("Phase 1: Multi-start LP optimization")
    print("=" * 60)

    # Search configurations: (length, init_idx, time_fraction)
    # Try multiple lengths and initializations
    search_configs = []

    # Primary length (provided)
    for idx in range(7):
        search_configs.append((n_prov, idx, 1.0))

    # Secondary lengths
    for n_alt in [128, 192, 320, 384, 512]:
        if n_alt != n_prov:
            for idx in [0, 1, 3]:
                search_configs.append((n_alt, idx, 0.6))

    total_weight = sum(c[2] for c in search_configs)
    phase1_budget = (deadline - t_global) * 0.50  # 50% for phase 1

    # Sort by expected value: provided length first, then others
    search_configs.sort(key=lambda x: (0 if x[0] == n_prov else 1, -x[2]))

    for n, init_idx, weight in search_configs:
        if time.monotonic() > deadline - 60:
            break

        alloc = max(20.0, weight / total_weight * phase1_budget)
        alloc = min(alloc, deadline - time.monotonic() - 20)
        if alloc < 15:
            break

        # Generate initialization
        if init_idx == 0 and provided_seq is not None and n == n_prov:
            a0 = provided_seq.astype(np.float64).copy()
        else:
            inits = make_inits(n, rng)
            a0 = inits[min(init_idx, len(inits) - 1)]

        s0 = local_score(a0)
        # Normalize for fair comparison
        a0 = a0 * (np.sqrt(2.0 * n) / np.sum(a0))

        print(f"  n={n}, init_idx={init_idx}, initial_score={local_score(a0):.6f}, alloc={alloc:.0f}s")
        t_start = time.monotonic()

        f_best, score = lp_optimize(a0, alloc, t_init=0.25, verbose=False)
        elapsed = time.monotonic() - t_start
        print(f"  -> score={score:.6f}, time={elapsed:.1f}s")

        if score < best_score:
            best_score = score
            best_seq = f_best.copy()
            print(f"  *** New best: {best_score:.6f} (n={n}, init={init_idx}) ***")

    # === Phase 2: Annealed gradient refinement ===
    print("\n" + "=" * 60)
    print("Phase 2: Annealed gradient refinement")
    print("=" * 60)

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        grad_time = min(300.0, remaining * 0.40)
        if grad_time > 30:
            print(f"  Polishing n={len(best_seq)}, time={grad_time:.0f}s")
            a_grad, score_grad = annealed_gradient(best_seq, grad_time, verbose=True)
            print(f"  Gradient result: {score_grad:.6f}")
            if score_grad < best_score:
                best_score = score_grad
                best_seq = a_grad.copy()
                print(f"  *** New best: {best_score:.6f} ***")

            # Also try gradient from a different start
            if time.monotonic() < deadline - 60:
                n_g = len(a_grad)
                inits = make_inits(n_g, rng)
                a0 = inits[1]  # chirp
                a0 = a0 * (np.sqrt(2.0 * n_g) / np.sum(a0))
                t_rem = deadline - time.monotonic() - 10
                if t_rem > 30:
                    a_g2, score_g2 = annealed_gradient(a0, min(100, t_rem), verbose=False)
                    if score_g2 < best_score:
                        best_score = score_g2
                        best_seq = a_g2.copy()
                        print(f"  *** New best (2nd grad): {best_score:.6f} ***")

    # === Phase 3: Aggressive SA ===
    print("\n" + "=" * 60)
    print("Phase 3: Simulated annealing polish")
    print("=" * 60)

    if best_seq is not None:
        remaining = deadline - time.monotonic()
        sa_time = min(300.0, remaining * 0.50)
        if sa_time > 30:
            print(f"  SA polish, n={len(best_seq)}, time={sa_time:.0f}s")
            a_sa, score_sa = sa_optimize(best_seq, sa_time, rng, aggressive=True)
            print(f"  SA result: {score_sa:.6f}")
            if score_sa < best_score:
                best_score = score_sa
                best_seq = a_sa.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # === Phase 4: Final LP refinement ===
    print("\n" + "=" * 60)
    print("Phase 4: Final LP refinement")
    print("=" * 60)

    if best_seq is not None and time.monotonic() < deadline - 40:
        remaining = deadline - time.monotonic()
        final_lp_time = min(120.0, remaining * 0.80)
        if final_lp_time > 15:
            print(f"  Final LP, n={len(best_seq)}, time={final_lp_time:.0f}s")
            a_final, score_final = lp_optimize(best_seq, final_lp_time, t_init=0.15)
            print(f"  Final LP result: {score_final:.6f}")
            if score_final < best_score:
                best_score = score_final
                best_seq = a_final.copy()
                print(f"  *** New best: {best_score:.6f} ***")

    # === Phase 5: Quick final SA if time permits ===
    if best_seq is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            a_s5, score_s5 = sa_optimize(best_seq, remaining - 20, rng, aggressive=False)
            if score_s5 < best_score:
                best_score = score_s5
                best_seq = a_s5.copy()
                print(f"  *** Final SA new best: {best_score:.6f} ***")

    # Final verification and return
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'=' * 60}")
        print(f"FINAL RESULT: score={final_score:.6f}, n={len(final_list)}")
        print(f"Deadline buffer: {deadline - time.monotonic():.1f}s remaining")
        print(f"{'=' * 60}")
        return final_list

    # Fallback
    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]
    return [1.0] * 256
