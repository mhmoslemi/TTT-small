    Compute gradient of the soft-max of the autocorrelation.
    Uses top-K peaks for more robust optimization.
    """
    n = len(a)
    b = np.convolve(a, a)

        idx = np.argmax(b)
        w = np.zeros_like(b)
        w[idx] = 1.0
    else:

        b_shifted = b - b.max()
        w = np.exp(tau * b_shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:

    conv = np.convolve(w, a_flip, mode='full')
    # The valid part: conv[n-1 : n-1+n]
    grad = 2.0 * conv[n-1:n-1+n]
    
    return grad, b.max()

If `a[i] -> a[i] + delta`:
`b_new[k] = sum_j a'[j] * a'[k-j]`
`= sum_j a[j] * a[k-j] + delta * sum_j [delta_{j,i} * a[k-j] + a[j] * delta_{k-j,i}] + delta^2 * delta_{j,i} * delta_{k-j,i}`

The first correction: `delta * (a[k-i] + a[k-i]) = 2*delta*a[k-i]` (if `k-i != i`, i.e., `k != 2i`)
The second correction: if `k = 2i`, we get an extra `delta^2` term.

More precisely:
`b_new[k] = b[k] + 2*delta*a[k-i] + delta^2 * [k == 2i]`

(where `a[k-i]` is 0 if `k-i` is out of range [0, n-1])

So the change is quadratic in delta. For a given `k`, the new value is:
`b_new[k] = b[k] + 2*delta*a[k-i] + delta^2 * [k == 2i]`

We want to minimize `max_k b_new[k]`.

For the peak at position `k*`, we want `b_new[k*]` to be as small as possible:
`b_new[k*] = b[k*] + 2*delta*a[k*-i]`

If `a[k*-i] > 0`, we should decrease `delta` (make it negative) to reduce `b[k*]`.
If `a[k*-i] < 0`... but `a >= 0`, so `a[k*-i] >= 0`.

So decreasing `a[i]` (negative delta) always helps reduce `b[k*]` (if `a[k*-i] > 0`).

But we also need to check that other lags don't increase too much. And we need to maintain the sum constraint.

This is getting complex. Let me just implement a practical version:

For each coordinate `i`, try several candidate values and pick the best one. This is the coordinate descent approach, which is simpler and still effective.

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 for non-negative sequences.
Uses log-sum-exp smoothed gradients, adaptive temperature, coordinate descent,
and multi-start with spectral and structured initializations at multiple lengths.
"""

import numpy as np
import time


def compute_obj(a):
    """Compute the objective value."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = np.convolve(a, a)
    return float(2.0 * n * b.max() / (S * S))


def normalize_sum(a, target):
    """Normalize a to have the given sum."""
    S = a.sum()
    if S < 1e-15:
        return np.full_like(a, target / len(a))
    return a * (target / S)


def lse_gradient(a, tau):
    """
    Compute gradient of the log-sum-exp smoothed max of autocorrelation.
    Returns (gradient, current_max, lse_value).
    """
    n = len(a)
    b = np.convolve(a, a)
    
    # Compute soft weights
    b_shifted = b - b.max()
    weights = np.exp(tau * b_shifted)
    w_sum = weights.sum()
    if w_sum < 1e-300:
        weights = np.zeros_like(b)
        weights[np.argmax(b)] = 1.0
    else:
        weights /= w_sum
    
    # Gradient via cross-correlation
    # grad[i] = 2/tau * sum_k weights[k] * a[k-i] (for valid k-i)
    # = 2/tau * (weights cross-correlated with a)
    # Using convolution: conv(weights, a_flip)[n-1+i] = sum_j weights[n-1+i-j]*a_flip[j]
    # = sum_j weights[n-1+i-j]*a[n-1-j]
    # Let k = n-1-i, then we want sum_k weights[k]*a[k-i]... 
    
    # Actually, let's use: grad[i] = 2 * sum_k weights[k] * a[k-i] (for 0<=k-i<n)
    # = 2 * sum_k weights[k] * a[k-i]
    # This is the cross-correlation of weights with a, evaluated at lag i
    
    # conv(weights, a_flip) where a_flip = a[::-1]
    # conv[m] = sum_j weights[j] * a_flip[m-j] = sum_j weights[j] * a[n-1-(m-j)]
    # = sum_j weights[j] * a[n-1-m+j]
    # Let k = j, then: sum_k weights[k] * a[n-1-m+k]
    # We want: sum_k weights[k] * a[k-i]
    # So we need n-1-m+k = k-i, which gives n-1-m = -i, m = n-1+i
    
    a_flip = a[::-1]
    conv_result = np.convolve(weights, a_flip, mode='full')
    # grad[i] = 2 * conv_result[n-1+i]
    grad = 2.0 * conv_result[n-1:n-1+n]
    
    return grad, float(b.max())


def lse_value(a, tau):
    """Compute the log-sum-exp value of the autocorrelation."""
    b = np.convolve(a, a)
    b_max = b.max()
    if tau < 1e-10:
        return float(b_max)
    return float(b_max + np.log(np.exp(tau * (b - b_max)).sum()) / tau)

def spectral_init(n, rng):
    """Initialize using random phase spectrum."""
    N = 2 * n
    # Random phases with conjugate symmetry
    phases = rng.random(N) * 2 * np.pi
    # Create conjugate symmetric spectrum
    A = np.exp(1j * phases)
    # Enforce conjugate symmetry: A[k] = conj(A[N-k])
    for k in range(1, N // 2):
        A[k] = np.exp(1j * phases[k])
        A[N - k] = np.conj(A[k])
    A[0] = 1.0
    if N % 2 == 0:
        A[N // 2] = 1.0
    
    x = np.real(np.fft.ifft(A))
    a = np.abs(x[:n]) + 0.1
    return normalize_sum(a, np.sqrt(2.0 * n))


def structured_init(n, rng):
    """Initialize with a structured pattern."""
    # Sum of random cosines with incommensurate frequencies
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    n_harms = rng.integers(3, 8)
    for _ in range(n_harms):
        freq = rng.uniform(1, n // 2)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.3, 1.0)
        a += amp * np.cos(2 * np.pi * i * freq / n + phase)
    a = np.maximum(a, 0.01)
    return normalize_sum(a, np.sqrt(2.0 * n))


def sparse_init(n, rng):
    """Initialize with a sparse-like pattern (many small values, few large)."""
    a = np.full(n, 0.1)
    n_large = max(1, n // 8)
    positions = rng.choice(n, size=n_large, replace=False)
    a[positions] = rng.uniform(0.5, 2.0, size=n_large)
