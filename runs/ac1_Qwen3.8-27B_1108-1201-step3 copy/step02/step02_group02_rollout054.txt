"""
Hybrid LP + Coordinate Descent optimization for minimizing peak autocorrelation
of non-negative step functions. Combines LP-based direction finding with
coordinate descent for local polish.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def build_A(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def golden_section_search(f_func, a, b, tol=1e-8, max_iter=50):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f_func(c)
    fd = f_func(d)
    for _ in range(max_iter):
        if abs(b - a) < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f_func(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f_func(d)
    return (a + b) / 2.0, min(fc, fd)


def lp_min_peak(f, S):
    """LP: minimize max(f*x) s.t. sum(x)=S, x>=0."""
    n = len(f)
    A = build_A(f)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    bounds = [(0, None)] * n + [(0, None)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x[:n], 0.0)
    return None

"""
Hybrid LP + Coordinate Descent optimization for minimizing peak linear
autocorrelation of non-negative step functions. Uses LP-based direction
finding (min-peak formulation) combined with coordinate descent polish
and multiple starting points for global exploration.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Fast computation of the evaluation score."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def build_A(f):
    """Build (2n-1) x n constraint matrix where A[k,i] = f[k-i] if valid."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def golden_section_search(f_func, a, b, tol=1e-8, max_iter=50):
    """Golden section search for minimum of f_func on [a, b]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f_func(c)
    fd = f_func(d)
    for _ in range(max_iter):
        if abs(b - a) < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f_func(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f_func(d)
    x_opt = (a + b) / 2.0
    f_opt = f_func(x_opt)
    return x_opt, min(f_opt, fc, fd)


def lp_min_peak(f, S):
    """LP: minimize max_k (f*x)[k] subject to sum(x)=S, x>=0, t>=0.
    Returns the optimal x direction."""
    n = len(f)
    A = build_A(f)
    # Variables: [x_0, ..., x_{n-1}, t]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize t
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])
    bounds = [(0.0, None)] * n + [(0.0, None)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x[:n], 0.0)
    return None


def lp_max_sum(f, M):
    """LP: maximize sum(b) subject to (f*b)[k]<=M, b>=0."""
    n = len(f)
    A = build_A(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def lp_iterate(f0, time_limit, rng, verbose=False):
    """LP-based iterative optimization with line search."""
    n = len(f0)
    S = np.sqrt(2.0 * n)
    f = f0.astype(np.float64).copy()
    f = np.clip(f, 0, None)
    sum_f = np.sum(f)
    if sum_f < 0.01:
        f = np.ones(n) * (S / n)
    else:
        f = f * (S / sum_f)

    current_score = local_score(f)
    best_f = f.copy()
    best_score = current_score
    t0 = time.monotonic()
    stall_count = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        M = np.max(fft_convolve(f, f))

        # Try min-peak LP first
        g = lp_min_peak(f, S)
        if g is not None and np.sum(g) > 1e-10:
            pass
        else:
            # Fallback to max-sum LP
            g = lp_max_sum(f, M)
            if g is None or np.sum(g) < 1e-10:
                break

        iter_count += 1
        g = g * (S / np.sum(g))

        # Line search via golden section
        def line_score(t):
            f_new = (1.0 - t) * f + t * g
            f_new = np.clip(f_new, 0, None)
            sf = np.sum(f_new)
            if sf < 0.01:
                return np.inf
            f_new = f_new * (S / sf)
            return local_score(f_new)

        t_opt, score_opt = golden_section_search(line_score, 0.0, 1.0, tol=1e-6, max_iter=40)

        if score_opt < current_score - 1e-12:
            f = (1.0 - t_opt) * f + t_opt * g
            f = np.clip(f, 0, None)
            sf = np.sum(f)
            if sf > 0.01:
                f = f * (S / sf)
            current_score = score_opt
            stall_count = 0
            if current_score < best_score:
                best_score = current_score
                best_f = f.copy()
                if verbose:
                    elapsed = time.monotonic() - t0
                    if iter_count % 200 == 0 or best_score < 1.51:
                        print(f"    *** New best: {best_score:.6f} (iter={iter_count}, t={elapsed:.1f}s)")
        else:
            stall_count += 1
            if stall_count > 40:
                # Perturb to escape local minimum
                scale = 0.01 + 0.01 * min(stall_count / 200.0, 1.0)
                f = f * np.exp(rng.normal(0, scale, n))
                f = np.clip(f, 0, None)
                sf = np.sum(f)
                if sf > 0.01:
                    f = f * (S / sf)
                    current_score = local_score(f)
                stall_count = 0

        if verbose and iter_count % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    iter={iter_count}, score={current_score:.6f}, best={best_score:.6f}, t={elapsed:.1f}s")

    return best_f, best_score


def coordinate_descent(a_init, time_limit=300, rng=None, verbose=False):
    """Coordinate descent with incremental autocorrelation updates."""
    if rng is None:
        rng = np.random.default_rng(123)
    a = a_init.astype(np.float64).copy()
    n = len(a)
    best_a = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()
    cycle = 0

    while time.monotonic() - t0 < time_limit:
        cycle += 1
        b = fft_convolve(a, a).copy()
        S_total = np.sum(a)
        any_improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            S = S_total - old_val
            if S < 0.005 and old_val < 0.005:
                continue

            # Precompute shifted array: shifted[k] = a[k-j] for valid k
            shifted = np.zeros(2 * n - 1)
            shifted[j:j + n] = a.copy()

            two_j = 2 * j
            has_two_j = two_j < 2 * n - 1
            b_two_j = b[two_j] if has_two_j else 0.0

            # Target function for 1D optimization of a[j]
            def target(x):
                delta = x - old_val
                # Compute new max_b
                new_b = b + 2.0 * delta * shifted
                if has_two_j:
                    new_b[two_j] = b_two_j + x * x - old_val * old_val
                new_max = np.max(new_b)
                new_sum = S + x
                if new_sum < 0.01:
                    return np.inf
                return 2.0 * n * new_max / (new_sum * new_sum)

            # Determine search range
            mean_val = S / max(n - 1, 1)
            if old_val > 1e-8:
                lower = 0.0
                upper = max(3.0 * old_val, 2.0 * mean_val + 0.1)
            else:
                lower = 0.0
                upper = max(2.0 * mean_val, 0.5)

            # Evaluate at current point
            cur_val = target(old_val)

            # Golden section search
            x_opt, score_opt = golden_section_search(target, lower, upper, tol=1e-7, max_iter=30)

            # Also check boundaries
            score_at_0 = target(0.0)
            score_at_upper = target(upper)
            best_x = x_opt
            best_s = score_opt
            if score_at_0 < best_s:
                best_s = score_at_0
                best_x = 0.0
            if score_at_upper < best_s:
                best_s = score_at_upper
                best_x = upper

            if best_s < cur_val - 1e-12:
                x_new = best_x
                a[j] = x_new
                S_total = S + x_new
                any_improved = True

                # Update b incrementally
                if has_two_j:
                    b[two_j] = b_two_j + x_new * x_new - old_val * old_val
                k_range = np.arange(j, j + n)
                if has_two_j:
                    mask = k_range != two_j
                    b[k_range[mask]] += 2.0 * (x_new - old_val) * a[k_range[mask] - j]
                else:
                    b[k_range] += 2.0 * (x_new - old_val) * a[k_range - j]

        # Periodic recomputation to avoid numerical drift
        if cycle % 10 == 0:
            b = fft_convolve(a, a).copy()

        current_score = local_score(a)
        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()
            if verbose:
                elapsed = time.monotonic() - t0
                if best_score < 1.51 or cycle % 50 == 0:
                    print(f"    CD cycle={cycle}, new best={best_score:.6f}, elapsed={elapsed:.1f}s")

        if verbose and cycle % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD cycle={cycle}, score={current_score:.6f}, best={best_score:.6f}, elapsed={elapsed:.1f}s")

        if not any_improved and cycle > 20:
            # Small perturbation to escape
            a = a * np.exp(rng.normal(0, 0.005, n))
            a = np.clip(a, 0, None)

    return best_a, best_score


def make_chirp_init(n, alpha, rng):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multi_tone_init(n, rng):
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k in range(1, 8):
        amp = 0.3 / k
        freq = rng.uniform(0.3, 2.5) * k
        phase = rng.uniform(0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    a += 0.04 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_random_init(n, rng, spread=0.3):
    a = 1.0 + spread * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def resample_sequence(seq, new_n):
    x_old = np.linspace(0, 1, len(seq))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, seq)

                best_score = s_best
                best_seq = f_best.copy()
            cd_best, cd_score = coordinate_descent(best_seq, t_cd, rng, verbose=True)
            if cd_score < best_score:
                best_score = cd_score
                best_seq = cd_best.copy()
            print(f"  Phase 6 result: {best_score:.6f}")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 512
