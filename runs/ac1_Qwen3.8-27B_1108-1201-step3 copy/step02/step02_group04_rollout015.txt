"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using soft-subgradient mirror descent
with adaptive temperature, periodic LP-based restarts, spectral initializations,
and multi-scale optimization. Combines stochastic subgradient methods with
deterministic LP directions to escape local minima and approach the theoretical
lower bound of 2n/(2n-1).
"""

import numpy as np
import time
import scipy.optimize as optimize
import scipy.sparse as sparse


def fast_convolve(a):
    """Full linear auto-convolution of a with itself using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(A * A, fft_size)
    return result[:size]


def compute_soft_subgradient(a, conv, T):
    """Compute soft subgradient of max(conv) using softmax weighting.
    Returns (grad, w) where grad has length n and w has length 2n-1."""
    n = len(a)
    conv_shifted = conv - conv.max()
    w = np.exp(T * conv_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(conv))] = 1.0
    else:
        w /= w_sum
    grad = 2.0 * np.correlate(w, a, mode='valid')
    return grad, w


def mirror_descent_step(a, grad_c, eta):
    """Multiplicative (mirror) descent step with centered gradient."""
    log_update = -eta * grad_c
    log_update = np.clip(log_update, -50.0, 50.0)
    a_new = a * np.exp(log_update)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    a_new /= s
    return a_new

    else:

def make_lp_structure(n):
    """Precompute row/col indices for cross-convolution constraint matrix."""
    rows_list = []
    cols_list = []
    for i in range(n):
        for j in range(n):
            rows_list.append(i + j)
            cols_list.append(i)
    return np.array(rows_list, dtype=np.int32), np.array(cols_list, dtype=np.int32)


def solve_lp_direction(f, M, rows, cols, n, timeout=10):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)[k]<=M for all k."""
    vals = np.tile(f, n)
    A = sparse.csc_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                           options={'disp': False, 'time_limit': timeout})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def lp_blend_optimize(f0, time_limit):
    """LP-based alternating optimization with line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    f_sum = f.sum()

        if g_sum < 1e-12:
            break
        g /= g_sum

        if np.max(np.abs(g - f)) < 1e-14:
            break

        improved = False
        for t_try in [t, 0.7 * t, 0.5 * t, 0.3 * t, 0.1 * t]:
            f_new = (1.0 - t_try) * f + t_try * g
            f_new = np.maximum(f_new, 0.0)

def spectral_init(n, rng, profile='flat'):
    """Generate initialization based on spectral properties.
    Creates a sequence whose DFT magnitude has a desired profile."""
    if profile == 'flat':
        # Uniform magnitude in frequency domain -> white noise in time
        phase = rng.uniform(0, 2 * np.pi, n)
        A = np.exp(1j * phase)
        a = np.real(np.fft.ifft(A))
    elif profile == 'lowpass':
        # Low-pass magnitude -> smooth in time
        freqs = np.fft.fftfreq(n)
        A = np.exp(-2.0 * np.pi * np.abs(freqs) * n * 0.1)
        phase = rng.uniform(0, 2 * np.pi, n)
        A = A * np.exp(1j * phase)
        a = np.real(np.fft.ifft(A))
    elif profile == 'bandpass':
        freqs = np.fft.fftfreq(n)
        center = 0.1 / (2 * np.pi)
        bandwidth = 0.05 / (2 * np.pi)
        A = np.exp(-((np.abs(freqs) - center) / bandwidth) ** 2)
        phase = rng.uniform(0, 2 * np.pi, n)
        A = A * np.exp(1j * phase)
        a = np.real(np.fft.ifft(A))
    else:
        a = rng.dirichlet(np.ones(n) * 0.5)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    return a


def gen_initializations(n, rng, parent=None):
    """Generate diverse initializations of length n, each summing to 1."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uni'] = np.ones(n) / n

    # Gaussian with various widths
    for sf in [0.08, 0.12, 0.18, 0.25, 0.35]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'g{int(sf * 100)}'] = g

    # Bi-modal
    bi = np.zeros(n)
    w = max(1, n // 5)
    bi[:w] = 1.0
    bi[-w:] = 1.0
    inits['bi'] = bi / bi.sum()

    # Cosine
    cb = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    inits['cos'] = cb / cb.sum()

    # Triangle
    tri = np.maximum(1.0 - np.abs(x - 0.5 * n) / (0.5 * n), 0.01)
    inits['tri'] = tri / tri.sum()

    # Spectral initializations
    inits['spec_flat'] = spectral_init(n, rng, 'flat')
    inits['spec_low'] = spectral_init(n, rng, 'lowpass')
    inits['spec_band'] = spectral_init(n, rng, 'bandpass')

    # Random Dirichlet
    inits['rnd'] = rng.dirichlet(np.ones(n) * 0.3)

    # Smoothed random
    r = rng.random(n)
    # Smooth with a simple kernel
    kernel = np.ones(max(1, n // 32)) / max(1, n // 32)
    r_smooth = np.convolve(r, kernel, mode='same')
    r_smooth = np.maximum(r_smooth, 0.0)
    inits['rnd_smooth'] = r_smooth / r_smooth.sum()

    # Parent-based
    if parent is not None:
        p = np.maximum(parent, 0.0) if len(parent) == n else None
        if p is None:
            # Resample parent to length n
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            p = np.maximum(pp, 0.0)
        s = p.sum()
        if s > 1e-10:
            inits['parent'] = p / s
            # Perturbed parent
            noise = rng.normal(0, 0.02, n)
            pp = p / s + noise
            pp = np.maximum(pp, 0.0)
            s2 = pp.sum()
            if s2 > 1e-10:
                inits['parent_pert'] = pp / s2

    return inits

                best_score = sc_final
                best_seq = f_final.copy()
                print(f"  Final: score={sc_final:.6f}")
        except Exception:
            pass

        # One more round with different params
        if time.monotonic() < deadline - 5:
            tr2 = deadline - time.monotonic() - 5
            try:
                f_final2, _ = combined_optimize(best_seq, time_limit=tr2,
                                                T_init=50.0, eta_init=0.01)
                sc_final2 = float(evaluate_sequence(f_final2.tolist()))
                if np.isfinite(sc_final2) and sc_final2 < best_score:
                    best_score = sc_final2
                    best_seq = f_final2.copy()
                    print(f"  Final2: score={sc_final2:.6f}")
            except Exception:
                pass

    if best_seq is None:
        best_seq = np.ones(512)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL: n={len(best_seq)}, score={best_score:.6f}")
    return [float(x) for x in best_seq]
