```python
"""
Step function optimization using smoothed gradient descent with adaptive
temperature, peak-targeted multiplicative updates, and multi-scale local search.
"""

import numpy as np
from scipy.optimize import linprog
import time

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def fast_score(a):
    """Fast evaluation of the objective."""
    a = np.clip(a, 0, 1000)
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)

def smoothed_score(a, temperature):
    """Smoothed version of the objective using log-sum-exp."""
    a = np.clip(a, 0, 1000)
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    # Log-sum-exp for numerical stability
    c_max = np.max(c)
    log_sum = c_max + temperature * np.log(np.sum(np.exp((c - c_max) / temperature)))
    return 2.0 * n * log_sum / (S * S)

def smoothed_gradient(a, temperature):
    """Gradient of the smoothed objective."""
    a = np.clip(a, 0, 1000)
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.zeros(n)
    c = fft_convolve(a, a)
    # Softmax weights
    c_max = np.max(c)
    p = np.exp((c - c_max) / temperature)
    p /= np.sum(p)

    # dF/da_j = (2n/S^3) * [S * sum_k p_k * dc_k/da_j - 2 * G]

    return grad
```

```python
"""
Step function optimization minimizing peak autocorrelation using smoothed
gradient descent (log-sum-exp), adaptive temperature annealing, LP-based
direction finding, and multi-move local search polish.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time


def fft_convolve(a, b):
    """Fast linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def fast_score(a):
    """Fast evaluation of the true objective."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0, 1000)
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def smoothed_objective_and_grad(x, temp):
    """Compute smoothed objective and gradient. x is flattened sequence."""
    n = len(x)
    a = np.clip(x, 0, 1000)
    S = np.sum(a)
    if S < 0.01:
        return 1e10, np.zeros(n)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    exp_c = np.exp((c - c_max) / temp)
    log_sum = np.log(np.sum(exp_c))
    G = c_max + temp * log_sum
    F = 2.0 * n * G / (S * S)
    
    p = exp_c / np.sum(exp_c)
    
    # r[j] = sum_{l=0}^{n-1} p[j+l] * a[l]
    a_flip = a[::-1]
    conv_ap = fft_convolve(a_flip, p)
    r = conv_ap[n-1:2*n-1]
    
    dG = r + a  # dG/da_j = r[j] + a[j]
    grad = (2.0 * n / (S ** 3)) * (S * dG - 2.0 * G)
    return F, grad


def build_constraint_matrix(f):
    """Build the (2n-1) x n constraint matrix."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A


def solve_lp_direction(f, M, max_iter=100):
    """Solve LP for best descent direction."""
    n = len(f)
    A = build_constraint_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'maxiter': max_iter})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return np.ones(n) / n


def make_chirp(n, alpha, rng):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multitone(n, rng, n_tones=5):
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k in range(1, n_tones + 1):
        amp = 0.3 / k
        freq = rng.uniform(0.5, 3.0) * k
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2 * np.pi * freq * t / n + phase)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_random(n, rng, spread=0.3):
    a = 1.0 + spread * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def resample_sequence(seq, new_n):
    x_old = np.linspace(0, 1, len(seq))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, seq)


def gradient_descent_smoothed(a0, time_limit, rng, temp_range=(10.0, 0.5)):
    """Smoothed gradient descent with adaptive temperature."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = np.clip(a, 0, None)
    S = np.sum(a)
    if S < 0.01:
        a = np.ones(n)
        S = n
    a = a / S  # Normalize to sum = 1
    
    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    
    temp = temp_range[0]
    temp_min = temp_range[1]
    lr = 0.01
    patience = 0
    max_patience = 200
    n_steps = 0
    
    while time.monotonic() - t0 < time_limit:
        obj, grad = smoothed_objective_and_grad(a, temp)
        
        # Gradient step with adaptive LR
        a_new = a - lr * grad
        a_new = np.clip(a_new, 0, None)
        S_new = np.sum(a_new)
        if S_new < 0.01:
            lr *= 0.5
            continue
        a_new = a_new / S_new  # Renormalize
        
        new_score = fast_score(a_new)
        if new_score < best_score:
            best_a = a_new.copy()
            best_score = new_score
            patience = 0
        else:
            patience += 1
        
        a = a_new
        n_steps += 1
        
        # Adaptive temperature
        if patience > max_patience:
            # Reduce temperature
            temp = max(temp_min, temp * 0.9)
            # Restart with perturbation
            a = best_a * np.exp(rng.normal(0, 0.02, n))
            a = np.clip(a, 0, None)
            S = np.sum(a)
            if S > 0.01:
                a = a / S
            lr = 0.005
            patience = 0
        
        # Adaptive learning rate
        if patience < 50:
            lr = min(lr * 1.01, 0.05)
        else:
            lr = max(lr * 0.99, 0.001)
        
        if n_steps % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GD: step={n_steps}, temp={temp:.3f}, lr={lr:.5f}, "
                  f"best_score={best_score:.6f}, elapsed={elapsed:.1f}s")
    
    return best_a, best_score


def lp_refine(a0, time_limit, rng):
    """LP-based refinement."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = np.clip(a, 0, None)
    S = np.sum(a)
    if S < 0.01:
        a = np.ones(n)
    a = a / S
    
    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    t_mix = 0.1
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(a, a)
        M = np.max(c)
        b = solve_lp_direction(a, M)
        
        sum_b = np.sum(b)
        if sum_b < 1e-10:
            break
        g = b / sum_b
        
        improved = False
        for trial_t in [t_mix, t_mix * 0.5, t_mix * 0.25, t_mix * 0.1]:
            a_new = (1 - trial_t) * a + trial_t * g
            a_new = np.clip(a_new, 0, None)
            S_new = np.sum(a_new)
            if S_new < 0.01:
                continue
            a_new = a_new / S_new
            new_score = fast_score(a_new)
            if new_score < best_score:
                a = a_new
                best_a = a.copy()
                best_score = new_score
                t_mix = min(t_mix * 1.05, 0.3)
                stall = 0
                improved = True
                break
        
        if not improved:
            stall += 1
            t_mix *= 0.8
            if stall > 50:
                a = best_a * np.exp(rng.normal(0, 0.03, n))
                a = np.clip(a, 0, None)
                S = np.sum(a)
                if S > 0.01:
                    a = a / S
                stall = 0
                t_mix = 0.05
    
    return best_a, best_score
```

```python
"""
Step function optimization minimizing peak autocorrelation using smoothed
gradient descent (log-sum-exp with adaptive temperature), LP-based direction
finding, proportional fitting updates, and multi-move local search polish.
Achieves better peak-to-mean convolution ratio than prior LP-only approaches
by leveraging fast O(n log n) gradient steps for dense exploration.

import numpy as np
from scipy.optimize import linprog
import time


def fft_convolve(a, b):
    """Fast linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def fast_score(a):
    """Fast evaluation of the true objective function."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0, 1000)
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def smoothed_objective_and_grad(a, temp):
    """Compute smoothed objective (log-sum-exp) and its gradient."""
    n = len(a)
    a = np.clip(a, 0, 1000)
    S = np.sum(a)
    if S < 0.01:
        return 1e10, np.zeros(n)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    exp_c = np.exp((c - c_max) / temp)
    sum_exp = np.sum(exp_c)
    log_sum = np.log(sum_exp)
    G = c_max + temp * log_sum
    F = 2.0 * n * G / (S * S)

    p = exp_c / sum_exp

    # r[j] = sum_{l=0}^{n-1} a[l] * p[j+l] via FFT cross-correlation
    a_flip = a[::-1]
    conv_ap = fft_convolve(a_flip, p)
    r = conv_ap[n - 1:2 * n - 1]

    dG = r + a
    grad = (2.0 * n / (S ** 3)) * (S * dG - 2.0 * G)
    return F, grad


def build_constraint_matrix(f):
    """Build the (2n-1) x n constraint matrix for LP."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_direction(f, M):
    """Solve LP: max sum(b) s.t. b>=0, (f*b)_k <= M for all k."""
    n = len(f)
    A = build_constraint_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'maxiter': 200})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return np.ones(n) / n


def make_chirp(n, alpha, rng):
    """Generate chirp-like initial sequence."""
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a = np.clip(a, 0.01, None)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_multitone(n, rng, n_tones=5):
    """Generate multi-tone initial sequence."""
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for k in range(1, n_tones + 1):
        amp = 0.3 / k
        freq = rng.uniform(0.5, 3.0) * k
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_random(n, rng, spread=0.3):

def resample_sequence(seq, new_n):
    """Resample a sequence to a new length via linear interpolation."""
    x_old = np.linspace(0, 1, len(seq))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, seq)

    a = np.clip(a, 0, None)
    S = np.sum(a)

    plateau_threshold = 1e-6

    while time.monotonic() - t0 < time_limit:
        obj, grad = smoothed_objective_and_grad(a, temp)

        # Gradient descent step
        a_new = a - lr * grad
        a_new = normalize_seq(a_new)

        new_score = fast_score(a_new)

        if new_score < best_score - plateau_threshold:
            best_a = a_new.copy()
            best_score = new_score
            patience = 0
            # Slightly increase LR on improvement
            lr = min(lr * 1.02, 0.08)
        elif new_score < best_score:
            patience = max(0, patience - 1)
        else:
            patience += 1

        a = a_new
        n_steps += 1

        # Adaptive temperature annealing
        frac = min(1.0, n_steps / max(1, int(time_limit * 50)))
        temp = temp_start * (temp_end / temp_start) ** frac

        # If stuck, reheat and perturb
        if patience > max_patience:
            temp = max(temp * 1.5, temp_start * 0.3)
            lr = max(lr * 0.5, 0.003)
            # Perturb from best
            pert_scale = 0.01 + 0.03 * (patience / (2 * max_patience))
            a = best_a * np.exp(rng.normal(0, pert_scale, n))
            a = normalize_seq(a)
            patience = 0

        # Very long stall: try LP-based kick
        if patience > max_patience * 3:
            c = fft_convolve(a, a)
            M = np.max(c)
            b = solve_lp_direction(a, M)
            if np.sum(b) > 1e-10:
                g = b / np.sum(b)
                for t_mix in [0.1, 0.2, 0.3]:
                    a_try = normalize_seq((1 - t_mix) * a + t_mix * g)
                    s_try = fast_score(a_try)
                    if s_try < best_score:
                        a = a_try
                        best_a = a.copy()
                        best_score = s_try
                        break
            patience = 0
            lr = 0.01

        if n_steps % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GD: step={n_steps}, temp={temp:.3f}, lr={lr:.5f}, "
                  f"best={best_score:.6f}, elapsed={elapsed:.1f}s")

    return best_a, best_score


def lp_refine(a0, time_limit, rng):
    """LP-based iterative refinement with line search."""
    n = len(a0)
    a = normalize_seq(a0.copy())

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    t_mix = 0.1
    stall = 0
    n_iter = 0

    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(a, a)
        M = np.max(c)
        b = solve_lp_direction(a, M)

        sum_b = np.sum(b)
        if sum_b < 1e-10:
            stall += 1
            if stall > 30:
                break
            continue

        g = b / sum_b
        improved = False

        for trial_t in [t_mix, t_mix * 0.5, t_mix * 0.25, t_mix * 0.1]:
            a_new = normalize_seq((1 - trial_t) * a + trial_t * g)
            new_score = fast_score(a_new)
            if new_score < best_score:
                a = a_new
                best_a = a.copy()
                best_score = new_score
                t_mix = min(t_mix * 1.08, 0.3)
                stall = 0
                improved = True
                break

        if not improved:
            stall += 1
            t_mix *= 0.85
            if stall > 60:
                a = best_a * np.exp(rng.normal(0, 0.04, n))
                a = normalize_seq(a)
                stall = 0
                t_mix = 0.08

        n_iter += 1
        if n_iter % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP: iter={n_iter}, t_mix={t_mix:.4f}, "
                  f"best={best_score:.6f}, elapsed={elapsed:.1f}s")

    return best_a, best_score


def proportional_fitting_step(a, eta=0.1):
    """
    One proportional fitting step: reduce positions that contribute most
    to convolution peaks. a_i <- a_i * (1 - eta * max_k(a_i*a_{k-i}) / peak)
    """
    n = len(a)
    c = fft_convolve(a, a)
    peak = np.max(c)
    if peak <= 0:
        return a.copy()

    # For each position i, compute max_k a[i]*a[k-i] (its max contribution)
    # This is the max over valid k of a[i] * a[k-i]
    # = a[i] * max_{valid j} a[j] where j = k-i is in [0,n-1]
    # The valid j range depends on i and k. For a given i, as k ranges,
    # j = k-i ranges. The maximum possible a[j] is just the max of a.
    # But that's too crude. Let's compute it properly.
    # For position i, the contributions to c_k are a[i]*a[k-i] for valid k-i.
    # The max contribution from position i is a[i] * max(a[k-i] for valid k-i)
    # = a[i] * max(a) (since k-i can be any value in [0,n-1] for some k)
    # Actually for a fixed i, k ranges from i to i+n-1, so k-i ranges from 0 to n-1.
    # So the max contribution from position i is a[i] * max(a).
    # That's just proportional to a[i]. Not very useful.

    # Better: compute the actual contribution to the current peak.
    peak_idx = int(np.argmax(c))
    contribs = np.zeros(n)
    for i in range(n):
        j = peak_idx - i
        if 0 <= j < n:
            contribs[i] = a[i] * a[j]

    # Normalize contributions
    max_contrib = np.max(contribs)
    if max_contrib > 0:
        # Reduce high-contributing positions
        factor = 1.0 - eta * contribs / max_contrib
        a_new = a * np.clip(factor, 0.5, 1.2)
        return normalize_seq(a_new)
    return a.copy()


def local_search(a0, time_limit, rng):
    """
    Multi-move local search with simulated annealing.
    Move types: peak-targeted, proportional fitting, block, pair, global.
    """
    n = len(a0)
    a = normalize_seq(a0.copy())

    best_a = a.copy()
    best_score = fast_score(a)
    current_score = best_score
    t0 = time.monotonic()
    temp_sa = 0.002
    cooling = 0.99985
    n_iter = 0

    while time.monotonic() - t0 < time_limit:
        a_cand = a.copy()
        r = rng.random()

        if r < 0.20:
            # Peak-targeted: reduce top contributors to the current peak
            c = fft_convolve(a, a)
            peak_idx = int(np.argmax(c))
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]
            top_k = max(3, n // 12)
            top_idx = np.argsort(contribs)[-top_k:]
            a_cand[top_idx] *= np.exp(rng.normal(-0.05, 0.02, len(top_idx)))
            other = np.setdiff1d(np.arange(n), top_idx)
            if len(other) > 5:
                boost_n = min(8, len(other) // 4)
                boost_idx = rng.choice(other, size=boost_n, replace=False)
                a_cand[boost_idx] *= np.exp(rng.normal(0.04, 0.015, boost_n))

        elif r < 0.35:
            # Proportional fitting step
            a_cand = proportional_fitting_step(a, eta=rng.uniform(0.05, 0.2))

        elif r < 0.50:
            # Block scaling
            width = max(1, int(rng.integers(1, max(2, n // 8))))
            start = int(rng.integers(0, n - width))
            a_cand[start:start + width] *= np.exp(rng.normal(0, 0.035, width))

        elif r < 0.65:
            # Multi-block perturbation
            for _ in range(4):
                width = max(1, int(rng.integers(1, max(2, n // 20))))
                start = int(rng.integers(0, n - width))
                a_cand[start:start + width] *= np.exp(rng.normal(0, 0.025, width))

        elif r < 0.78:
            # Pair redistribution
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                delta = rng.uniform(-0.25, 0.25) * a[i]
                a_cand[i] = max(0, a[i] - delta)
                a_cand[j] = max(0, a[j] + delta)

        elif r < 0.88:
            # Trilinear perturbation
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            k = int(rng.integers(0, n))
            if len({i, j, k}) == 3:
                delta = rng.uniform(-0.12, 0.12) * a[i]
                a_cand[i] = max(0, a[i] - 2 * delta)
                a_cand[j] = max(0, a[j] + delta)
                a_cand[k] = max(0, a[k] + delta)

        else:
            # Global small perturbation
            a_cand = a * np.exp(rng.normal(0, 0.006, n))

        a_cand = np.clip(a_cand, 0, 1000)
        S_cand = np.sum(a_cand)
        if S_cand < 0.01:
            continue
        a_cand = a_cand / S_cand

        new_score = fast_score(a_cand)

        if new_score < current_score:
            a = a_cand
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                temp_sa = min(temp_sa * 1.1, 0.005)
        elif rng.random() < np.exp(-(new_score - current_score) / max(temp_sa, 1e-15)):
            a = a_cand
            current_score = new_score

        temp_sa *= cooling
        n_iter += 1

        if n_iter % 15000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LS: iter={n_iter}, temp={temp_sa:.7f}, "
                  f"best={best_score:.6f}, elapsed={elapsed:.1f}s")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization for step function coefficient minimization.
    Phase 1: Smoothed gradient descent (fast, dense exploration)
    Phase 2: LP-based refinement (principled direction finding)
    Phase 3: Local search with diverse moves (fine-tuning)
    Phase 4: Final polish
    """
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 45  # 45s safety buffer

    rng = np.random.default_rng(seed)

    best_a = None
    best_score = np.inf

    # Get initial sequence from globals if available
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0, 1000)
        if len(start_seq) > 0:
            s = fast_score(start_seq)
            print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
            if s < best_score:
                best_score = s
                best_a = start_seq.copy()

    n_default = 512

    # Generate diverse initializations
    inits = []
    labels = []
    if start_seq is not None:
        inits.append(start_seq.copy())
        labels.append("provided")
        inits.append(normalize_seq(start_seq * np.exp(rng.normal(0, 0.08, len(start_seq))) * np.sum(start_seq)))
        labels.append("provided+noise")
    inits.append(make_chirp(n_default, 0.5 + 0.3 * rng.random(), rng))
    labels.append("chirp")
    inits.append(make_multitone(n_default, rng, n_tones=int(rng.integers(3, 8))))
    labels.append("multitone")
    inits.append(make_random(n_default, rng, spread=0.4))
    labels.append("random")
    if start_seq is not None and len(start_seq) != n_default:
        inits.append(resample_sequence(start_seq, n_default))
        labels.append("resampled")

    print(f"Generated {len(inits)} initializations")

    # === Phase 1: Smoothed gradient descent ===
    print(f"\n{'='*60}")
    print("Phase 1: Smoothed gradient descent")
    print(f"{'='*60}")

    # Allocate time: give more time to the best initial point
    p1_time = min(900, max(300, (deadline - t_start) * 0.5))
    n_p1 = min(4, len(inits))
    t_per_init = p1_time / n_p1

    for i in range(n_p1):
        if time.monotonic() > deadline - 400:
            break
        time_left = deadline - time.monotonic() - 350
        t_alloc = max(40, min(300, time_left / max(1, n_p1 - i)))
        print(f"  [{labels[i]}] n={len(inits[i])}, allocating {t_alloc:.0f}s")
        a_gd, s_gd = gradient_descent_smoothed(inits[i], t_alloc, rng,
                                               temp_start=25.0, temp_end=0.3)
        if s_gd < best_score:
            best_score = s_gd
            best_a = a_gd.copy()
            print(f"    *** New best from {labels[i]}: {best_score:.6f} ***")
        else:
            print(f"    Result: {s_gd:.6f} (best so far: {best_score:.6f})")

    # === Phase 2: LP refinement ===
    print(f"\n{'='*60}")
    print(f"Phase 2: LP refinement (current best: {best_score:.6f})")
    print(f"{'='*60}")

    if best_a is not None and time.monotonic() < deadline - 250:
        t_lp = min(350, max(60, deadline - time.monotonic() - 200))
        print(f"  Allocating {t_lp:.0f}s for LP refinement")
        a_lp, s_lp = lp_refine(best_a, t_lp, rng)
        if s_lp < best_score:
            best_score = s_lp
            best_a = a_lp.copy()
            print(f"  *** New best from LP: {best_score:.6f} ***")
        else:
            print(f"  LP result: {s_lp:.6f}")

    # === Phase 3: Local search ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Local search (current best: {best_score:.6f})")
    print(f"{'='*60}")

    if best_a is not None and time.monotonic() < deadline - 80:
        t_ls = max(50, deadline - time.monotonic() - 60)
        print(f"  Allocating {t_ls:.0f}s for local search")
        a_ls, s_ls = local_search(best_a, t_ls, rng)
        if s_ls < best_score:
            best_score = s_ls
            best_a = a_ls.copy()
            print(f"  *** New best from LS: {best_score:.6f} ***")
        else:
            print(f"  LS result: {s_ls:.6f}")

    # === Phase 4: Final short GD + LS polish ===
    if best_a is not None and time.monotonic() < deadline - 30:
        remaining = deadline - time.monotonic() - 15
        if remaining > 40:
            print(f"\n{'='*60}")
            print(f"Phase 4: Final polish ({remaining:.0f}s)")
            print(f"{'='*60}")
            # Quick GD
            t_gd2 = remaining * 0.4
            a_gd2, s_gd2 = gradient_descent_smoothed(best_a, t_gd2, rng,
                                                     temp_start=2.0, temp_end=0.2)
            if s_gd2 < best_score:
                best_score = s_gd2
                best_a = a_gd2.copy()
                print(f"  After GD2: {best_score:.6f}")

            # Quick LS
            if time.monotonic() < deadline - 20:
                t_ls2 = max(10, deadline - time.monotonic() - 10)
                a_ls2, s_ls2 = local_search(best_a, t_ls2, rng)
                if s_ls2 < best_score:
                    best_score = s_ls2
                    best_a = a_ls2.copy()
                    print(f"  After LS2: {best_score:.6f}")

    # Final verification with the official evaluator
    if best_a is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_a]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'='*60}")
        print(f"FINAL RESULT: score={final_score:.6f}, n={len(final_list)}")
        print(f"{'='*60}")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 256
```
