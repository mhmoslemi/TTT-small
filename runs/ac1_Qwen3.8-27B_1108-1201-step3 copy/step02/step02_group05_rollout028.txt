    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()

    weights = exp_shifted / sum_exp
    a_flip = a[::-1]
    size = 1 << (3*n).bit_length()
    A = np.fft.rfft(a_flip, size)
    W = np.fft.rfft(weights, size)
    corr = np.fft.irfft(A * W, size)
    grad_a = 2.0 * n * 2.0 * corr[n-1:n-1+n]
    
    # grad of a wrt z: da[i]/dz[j] = a[i]*(delta_{ij} - a[j])
    # grad z = J^T * grad_a where J[i,j] = da[i]/dz[j] = a[i]*(delta_{ij} - a[j])

    mean_grad = np.dot(a, grad_a)

    return obj, grad_z

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smoothed gradient optimization
with softmax parameterization, adaptive beta, and multi-start strategy.
Uses FFT-based gradient computation for efficiency.
"""

import numpy as np
import scipy.optimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2*n-1]


def obj_norm(a):
    """Objective assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()


def smooth_obj_and_grad(z, beta):
    """Smoothed objective and gradient in softmax parameterization."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)
    
    conv = fast_convolve(a)
    max_c = conv.max()
    shifted = beta * (conv - max_c)
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    obj = 2.0 * n * (max_c + np.log(sum_exp) / beta)
    
    weights = exp_shifted / sum_exp
    
    # Gradient of max_approx wrt a
    a_flip = a[::-1]
    size = 1 << (3 * n - 2).bit_length()
    # Need convolution of a_flip (length n) and weights (length 2n-1)
    # Result has length n + (2n-1) - 1 = 3n - 2
    # We need indices n-1 to 2n-3 (n values)
    A = np.fft.rfft(a_flip, size)
    W = np.fft.rfft(weights, size)
    corr = np.fft.irfft(A * W, size)
    grad_a = 4.0 * n * corr[n-1:n-1+n]
    
    # Transform to z-space
    mean_grad = np.dot(a, grad_a)
    grad_z = a * (grad_a - mean_grad)
    
    return obj, grad_z


def optimize_from_start(a0, time_limit, seed=0):
    """Optimize from initial sequence using smoothed gradient + L-BFGS-B."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a0 = np.maximum(a0, 1e-10).astype(np.float64)
    a0 = a0 / a0.sum()
    
    z0 = np.log(a0 + 1e-15)  # initial z
    z0 = z0 - z0.mean()  # center for numerical stability
    
    best_a = a0.copy()
    best_obj = obj_norm(a0)
    t0 = time.monotonic()
    
    # Progressive beta schedule
    betas = [1.0, 3.0, 10.0, 30.0, 100.0, 300.0, 1000.0]
    
    for beta in betas:
        if time.monotonic() - t0 > time_limit:
            break
        
        # Time allocation for this beta
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 5:
            break
        beta_time = min(remaining * 0.4, 120.0)  # spend at most 40% or 120s per beta
        beta_time = max(beta_time, 3.0)
        
        # Multiple restarts for lower beta
        n_starts = 3 if beta < 10 else 1
        
        for start in range(n_starts):
            if time.monotonic() - t0 > time_limit:
                break
            
            if start == 0:
                z_start = z0.copy()
            else:
                z_start = z0 + rng.normal(0, 0.1, n)
            
            try:
                res = scipy.optimize.minimize(
                    smooth_obj_and_grad,
                    z_start,
                    args=(beta,),
                    method='L-BFGS-B',
                    jac=True,
                    options={'maxiter': 200, 'maxfun': 500, 'ftol': 1e-10,
                             'gtol': 1e-8, 'maxls': 20}
                )
                if res.success or res.fun < np.inf:
                    z_opt = res.x
                    z_max = z_opt.max()
                    exp_z = np.exp(z_opt - z_max)
                    a_opt = exp_z / exp_z.sum()
                    a_opt = np.maximum(a_opt, 0)
                    o = obj_norm(a_opt)
                    if o < best_obj:
                        best_obj = o
                        best_a = a_opt.copy()
                        z0 = z_opt - z_opt.mean()
            except Exception as e:
                pass
    
    return best_a, best_obj


def lp_optimize(f0, time_limit):
    """LP-based optimization (kept as backup)."""
    f = np.maximum(f0.astype(np.float64), 0.0).copy()
    s = f.sum()
    if s < 1e-12:
        f = np.ones_like(f) / len(f)
    else:
        f /= s

    best_f = f.copy()
    best_obj = obj_norm(f)
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 50:
        conv = fast_convolve(f)
        M = conv.max()
        if M < 1e-15:
            break

        n = len(f)
        # Build LP
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            end = min(2 * n - 1, j + n)
            A[j:end, j] = f[:end - j]
        c = -np.ones(n)
        b = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        try:
            res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
            if res.success:
                g = np.maximum(res.x, 0.0)
                gs = g.sum()
                if gs > 1e-12:
                    g /= gs
                    # Line search
                    best_t = 0
                    best_t_obj = best_obj
                    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                        a_new = (1-t)*f + t*g
                        a_new = np.maximum(a_new, 0)
                        ns = a_new.sum()
                        if ns < 1e-12:
                            continue
                        a_new /= ns
                        o = obj_norm(a_new)
                        if o < best_t_obj:
                            best_t_obj = o
                            best_t = t
                    if best_t > 0 and best_t_obj < best_obj - 1e-10:
                        a_new = (1-best_t)*f + best_t*g
                        a_new = np.maximum(a_new, 0)
                        a_new /= a_new.sum()
                        f = a_new
                        best_obj = best_t_obj
                        best_f = f.copy()
                        no_improve = 0
                    else:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

    return best_f, best_obj


def local_refine(a, time_limit):
    """Local refinement by redistributing mass away from peaks."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
        return a, obj_norm(a)
    a /= s
    best_obj = obj_norm(a)
    t0 = time.monotonic()
    stuck = 0

    while time.monotonic() - t0 < time_limit and stuck < 100:
        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))
        
        # Find elements contributing to the peak
        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]
        
        max_c = contrib.max()
        if max_c < 1e-15:
            break
        
        # Sort by contribution
        order = np.argsort(contrib)[::-1]
        top = order[:min(20, n//4)]
        bottom = order[-min(20, n//4):]
        
        found = False
        for di in top:
            if found:
                break
            for si in bottom:
                if found:
                    break
                if di == si:
                    continue
                for frac in [0.05, 0.1, 0.2, 0.4, 0.6]:
                    amount = a[di] * frac
                    if amount < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[di] -= amount
                    a_trial[si] += amount
                    a_trial = np.maximum(a_trial, 0)
                    ts = a_trial.sum()
                    if ts < 1e-12:
                        continue
                    a_trial /= ts
                    o = obj_norm(a_trial)
                    if o < best_obj - 1e-12:
                        best_obj = o
                        a = a_trial.copy()
                        found = True
                        break

        if not found:
            stuck += 1
        else:
            stuck = 0

    return a, best_obj


def get_parent(n):
    """Get parent sequence resized to length n."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim != 1 or p.size == 0 or not np.isfinite(p).all():
            return None
        if p.size >= n:
            pc = p[:n].copy()
        else:
            pc = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, p.size), p)
        pc = np.maximum(pc, 0.0)
        s = pc.sum()
        if s < 1e-10:
            return None
        return pc / s
    except Exception:
        return None

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase strategy combining:
1. Smoothed gradient optimization (softmax parameterization + L-BFGS-B with FFT gradients)
2. LP-based descent direction finding (iterative refinement)
3. Local mass-redistribution search
4. Multi-start with diverse initializations and sequence lengths
"""

import numpy as np
import scipy.optimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2 * n - 1]


def obj_norm(a):
    """Objective: 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()


def smooth_obj_and_grad(z, beta):
    """Smoothed objective and gradient in softmax parameterization.
    z is the log-parameter; a = softmax(z). Objective is 2n * soft_max_beta(conv(a,a)).
    """
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    a = exp_z / exp_z.sum()
    n = len(a)

    conv = fast_convolve(a)
    max_c = conv.max()
    shifted = beta * (conv - max_c)
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    obj = 2.0 * n * (max_c + np.log(sum_exp) / beta)

    weights = exp_shifted / sum_exp

    # grad_a[i] = 4n * sum_k weights[k] * a[k-i] for valid k-i in [0,n)
    # = 4n * (a_flip * weights)[n-1+i] via linear convolution
    a_flip = a[::-1]
    size = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a_flip, size)
    W = np.fft.rfft(weights, size)
    corr = np.fft.irfft(A * W, size)
    grad_a = 4.0 * n * corr[n - 1:n - 1 + n]

    # Transform to z-space: grad_z[j] = a[j] * (grad_a[j] - dot(a, grad_a))
    mean_grad = np.dot(a, grad_a)
    grad_z = a * (grad_a - mean_grad)

    return obj, grad_z


def optimize_smoothed(a0, time_limit, seed=0):
    """Optimize using smoothed gradient with progressive beta via L-BFGS-B."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a0 = np.maximum(a0.astype(np.float64), 1e-10)
    a0 = a0 / a0.sum()

    z0 = np.log(a0)
    z0 = z0 - z0.mean()

    best_a = a0.copy()
    best_obj = obj_norm(a0)
    t0 = time.monotonic()

    betas = [1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0, 1000.0]
    n_betas = len(betas)

    for bi, beta in enumerate(betas):
        if time.monotonic() - t0 > time_limit:
            break
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed
        if remaining < 3:
            break

        # Allocate time: earlier betas get less time
        frac = (bi + 1) / n_betas
        beta_time = remaining * (1.0 / (n_betas - bi)) * 0.8
        beta_time = max(2.0, min(beta_time, 120.0))

        n_starts = 2 if beta < 5 else 1
        for start in range(n_starts):
            if time.monotonic() - t0 > time_limit:
                break
            if start == 0:
                z_start = z0.copy()
            else:
                z_start = z0 + rng.normal(0, 0.05, n)

            try:
                res = scipy.optimize.minimize(
                    smooth_obj_and_grad,
                    z_start,
                    args=(beta,),
                    method='L-BFGS-B',
                    jac=True,
                    options={'maxiter': 300, 'maxfun': 1000,
                             'ftol': 1e-12, 'gtol': 1e-8, 'maxls': 30}
                )
                z_opt = res.x
                z_opt = z_opt - z_opt.mean()
                z_max = z_opt.max()
                exp_z = np.exp(z_opt - z_max)
                a_opt = exp_z / exp_z.sum()
                a_opt = np.maximum(a_opt, 0)
                o = obj_norm(a_opt)
                if o < best_obj:
                    best_obj = o
                    best_a = a_opt.copy()
                    z0 = z_opt.copy()
            except Exception:
                pass

    return best_a, best_obj


def lp_one_step(f, M):
    """Single LP step: find g>=0, sum(g)=1, s.t. (f*g)[k] <= M for all k."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(2 * n - 1, j + n)
        A[j:end, j] = f[:end - j]
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return None


def lp_optimize(f0, time_limit):
    """Iterative LP-based optimization with line search."""
    f = np.maximum(f0.astype(np.float64), 0.0).copy()
    s = f.sum()
    if s < 1e-12:
        f = np.ones_like(f) / len(f)
    else:
        f /= s

    best_f = f.copy()
    best_obj = obj_norm(f)
    t0 = time.monotonic()
    no_improve = 0
    max_no_improve = 40

    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        conv = fast_convolve(f)
        M = conv.max()
        if M < 1e-15:
            break

        g = lp_one_step(f, M)
        if g is None:
            no_improve += 1
            continue

        if np.allclose(g, f, atol=1e-10):
            no_improve += 3
            continue

        # Line search
        best_t = 0.0
        best_t_obj = best_obj
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_new = (1.0 - t) * f + t * g
            a_new = np.maximum(a_new, 0.0)
            ns = a_new.sum()
            if ns < 1e-12:
                continue
            a_new /= ns
            o = obj_norm(a_new)
            if o < best_t_obj:
                best_t_obj = o
                best_t = t

        if best_t > 0 and best_t_obj < best_obj - 1e-12:
            a_new = (1.0 - best_t) * f + best_t * g
            a_new = np.maximum(a_new, 0.0)
            a_new /= a_new.sum()
            f = a_new
            best_obj = best_t_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

    return best_f, best_obj


def local_refine(a, time_limit):
    """Local refinement: redistribute mass from peak contributors to safe positions."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-12:
        a = np.ones(n) / n
        return a, obj_norm(a)
    a /= s
    best_obj = obj_norm(a)
    t0 = time.monotonic()
    stuck = 0
    max_stuck = 80

    while time.monotonic() - t0 < time_limit and stuck < max_stuck:
        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))

        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = a[i] * a[j]

        max_c = contrib.max()
        if max_c < 1e-15:
            break

        order = np.argsort(contrib)
        n_swap = min(15, n // 8)
        top = order[-n_swap:]
        bottom = order[:n_swap]

        found = False
        for di in top:
            if found:
                break
            for si in bottom:
                if found:
                    break
                if di == si:
                    continue
                for frac in [0.05, 0.1, 0.2, 0.35, 0.5, 0.7]:
                    amount = a[di] * frac
                    if amount < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[di] -= amount
                    a_trial[si] += amount
                    a_try = np.maximum(a_trial, 0)
                    ts = a_try.sum()
                    if ts < 1e-12:
                        continue
                    a_try /= ts
                    o = obj_norm(a_try)
                    if o < best_obj - 1e-12:
                        best_obj = o
                        a = a_try.copy()
                        found = True
                        break

        if not found:
            stuck += 1
        else:
            stuck = 0

    return a, best_obj


def get_parent(n):
    """Get parent sequence resized to length n."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)

        else:
            pc = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, p.size), p)
        pc = np.maximum(pc, 0.0)
        s = pc.sum()
        if s < 1e-10:
            return None
        return pc / s
    except Exception:
        return None


def make_inits(n, rng, parent):
    """Generate diverse initializations for length n."""
    x = np.arange(n, dtype=np.float64)
    inits = {}

    if parent is not None:
        inits['parent'] = parent.copy()

    inits['uniform'] = np.ones(n) / n

    for sf in [0.25, 0.4, 0.6]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g = np.maximum(g, 1e-10)
        inits['gauss_%s' % str(sf)] = g / g.sum()

    if n > 1:
        hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n - 1)))
        hann = np.maximum(hann, 1e-10)
        inits['hann'] = hann / hann.sum()

    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    edges = np.zeros(n)
    m = max(1, n // 4)
    edges[:m] = 1.0
    edges[-m:] = 1.0
    inits['edges'] = edges / edges.sum()

    # Alternating
    alt = np.ones(n)
    alt[1::2] = 0.3
    inits['alt'] = alt / alt.sum()

    return inits
