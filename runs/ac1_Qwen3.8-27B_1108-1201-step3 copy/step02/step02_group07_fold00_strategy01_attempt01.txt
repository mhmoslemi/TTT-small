    """Evaluates the autocorrelation score of the sequence."""
    n = len(seq)
    a = np.clip(np.array(seq), 0.0, 1000.0)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    conv = rfft(rfft(a, n)[0: n] * rfft(a, n)[0: n])
    max_val = np.max(np.abs(conv))
    return 2.0 * n * max_val / (S ** 2)

def score_fast(seq):
    """Fast scoring function using FFT."""
    if not isinstance(seq, list):
        return np.inf
    a = np.asarray(seq, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    conv = rfft(rfft(a, n) * rfft(a, n))
    max_val = np.max(np.abs(conv))
    return 2.0 * n * max_val / (S ** 2)

def crossover(parent1, parent2, rate=0.7):
    """One-point crossover between two parent sequences."""
    if random.random() < rate:
        crossover_index = random.randint(1, len(parent1)-2)
        child = parent1[0:crossover_index] + parent2[crossover_index:]
    else:
        child = parent1[:]  # Or parent2[:], but we use parent1
    return child

def mutate(individual, std=0.05):
    """Add Gaussian noise to an individual and clip to [0, 1000]."""
    new = np.clip(individual + std * np.random.randn, 0.0, 1000.0)
    return new

def tournament_select(population, sizes, k=3, score_func=score_fast):
    """Select individuals based on tournament selection."""
    selected = []
    for _ in range(len(sizes)):
        candidates = random.sample(population, k)
        winner = min(candidates, key=lambda x: score_func(x))
        selected.append(winner)
    return selected
