```
def fft_square(a):
    a=np.asarray(a,float)
    n=a.size
    m=2*n-1
    nfft=1
    while nfft < m: nfft <<=1
    fa=np.fft.rfft(a,nfft)
    c=np.fft.irfft(fa*fa,nfft)[:m]
    c=np.maximum(c,0.0)
    return c
```

```
def coordinate_descent(a, deadline, rng, max_sweeps=1000, grid_size=40):
    a = np.asarray(a, dtype=float).copy()
    a = np.maximum(a, 0.0)
    n = a.size
    if n == 0: return a, inf
    S = float(a.sum())
    if S <= 1e-300:
        a = np.ones(n) * (np.sqrt(2*n)/n)
        S = float(a.sum())
    m = 2*n -1
    c = fft_square(a)
    M = float(c.max())
    if M <= 0: M = 1e-300
    bestR = (S*S)/M
    best = a.copy()
    best_score = 2.0*n*M/(S*S)
    no_improve = 0
    t_start = time.monotonic()
    for sweep in range(max_sweeps):
        if time.monotonic() >= deadline: break
        improved = False
        order = rng.permutation(n)
        max_a = float(np.max(a))
        # For each i
        for i in order:
            if time.monotonic() >= deadline: break
            x = float(a[i])
            S0 = S - x
            L = np.zeros(m, dtype=float)
            # assignment
            L[i:i+n] = 2.0 * a
            qi = 2*i
            if qi < m:
                L[qi] = 0.0
            # base
            base = c - L * x
            if qi < m:
                base[qi] = base[qi] - x*x
            # clean
            np.maximum(base, 0.0, out=base)
            # maybe clean c? c should be nonnegative
            # candidates
            upper = max(2.0, 10.0*x if x > 0 else 0.0, 3.0*max_a, np.sqrt(10.0*float(c.max()) + 1.0))
            upper = min(upper, 100.0)
            if upper < 1e-6: upper = 1e-6
            low = 1e-10
            if upper <= low:
                xs = np.array([low, upper])
            else:
                xs = np.logspace(np.log10(low), np.log10(upper), grid_size)
            extra = [0.0, x, 0.5*x, 2.0*x, upper*0.05, upper*0.5]
            xs = np.concatenate((xs, np.array(extra, dtype=float)))
            xs = xs[np.isfinite(xs) & (xs >= 0.0) & (xs <= upper)]
            if xs.size == 0: continue
            # to reduce duplicates, maybe not
            C = base[:, None] + L[:, None] * xs[None, :]
            if qi < m:
                C[qi, :] += xs * xs
            Mx = C.max(axis=0)
            Sx = S0 + xs
            R = np.full(xs.size, -1.0)
            valid = (Sx > 1e-300) & (Mx > 1e-300)
            if np.any(valid):
                R[valid] = Sx[valid] * Sx[valid] / Mx[valid]
            idx = int(np.argmax(R))
            if R[idx] > bestR + 1e-12:
                a[i] = float(xs[idx])
                c = C[:, idx].copy()
                c = np.maximum(c, 0.0)
                S = float(Sx[idx])
                bestR = float(R[idx])
                # Update best_score? best_score=2n/bestR; but use actual M later
                best = a.copy()
                best_score = 2.0*n / bestR  # since R=S^2/M
                improved = True
        if sweep % 5 == 0:
            c = fft_square(a)
            S = float(a.sum())
            M = float(c.max())
            if M > 0 and S > 0:
                R = S*S/M
                if R > bestR:
                    bestR = R; best = a.copy(); best_score=2*n/R
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 2:

    # final recompute

    S = float(best.sum())
    M = float(c.max())
    if S > 1e-300 and M > 1e-300:
        score = 2.0*n*M/(S*S)
    else:
        score = float('inf')
    return best, score
```

```python
"""
Search for a non-negative sequence minimizing
    2*n*max(convolve(a,a)) / sum(a)^2.

The main tools are:
- FFT-based convolution and gradients,
- coordinate descent on the scale-invariant ratio (sum^2 / max convolution),
- Adam-style ascent/descent on log-variables with a smoothed maximum,
- LP-based construction directions,
- several diverse starts and local escape attempts.
"""

import time
import numpy as np

try:
    from scipy.optimize import linprog
except Exception:
    linprog = None

INF = float('inf')


def next_power_of_two(n):
    n = int(n)
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def fft_square(a):
    """Full convolution of a with itself, using FFT."""
    a = np.asarray(a, dtype=float)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=float)
    m_len = 2 * n - 1
    nfft = next_power_of_two(m_len)
    fa = np.fft.rfft(a, nfft)
    c = np.fft.irfft(fa * fa, nfft)[:m_len]
    return np.maximum(c, 0.0)


def corr_weights(w, a):
    """
    For w of length 2n-1 and a of length n, return h[i]
    = sum_{j=0}^{n-1} w[i+j] * a[j], i=0..n-1.
    """
    n = a.size
    L = w.size
    total = L + n - 1
    nfft = next_power_of_two(total)
    fw = np.fft.rfft(w, nfft)
    fa_rev = np.fft.rfft(a[::-1], nfft)
    conv = np.fft.irfft(fw * fa_rev, nfft)
    return np.ascontiguousarray(conv[n - 1:n - 1 + n].real)


def soft_weights(c, t, flat_fraction=0.0):
    """
    Softmax weights over convolution coefficients.
    Also returns a smooth approximation of max(c).
    """
    m_len = c.size
    if m_len == 0:
        return np.empty(0), 0.0
    cmax = float(np.max(c))
    if cmax <= 0.0:
        return np.ones(m_len, dtype=float) / m_len, 0.0

    if t <= 1e-8:
        w = np.ones(m_len, dtype=float) / m_len
        return w, float(np.mean(c))

    z = t * (c - cmax)
    expz = np.exp(z)
    s = float(np.sum(expz))
    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros(m_len, dtype=float)
        w[int(np.argmax(c))] = 1.0
        softM = cmax
    else:
        w = expz / s
        softM = cmax + float(np.log(s)) / t

    if flat_fraction > 0.0:
        w = (1.0 - flat_fraction) * w + (flat_fraction / m_len)

    return w, float(softM)


def normalize_sum(a, target):
    a = np.asarray(a, dtype=float).copy()
    s = float(np.sum(a))
    if s <= 1e-300 or target <= 0.0 or not np.isfinite(target):
        return a
    return a * (target / s)


def compute_score(a):
    """Evaluate 2*n*max(conv(a,a))/sum(a)^2, matching the public evaluator."""
    a = np.asarray(a, dtype=float)
    if a.size == 0:
        return INF
    S = float(np.sum(a))
    if S < 0.01:
        return INF
    c = fft_square(a)
    M = float(np.max(c))
    if not np.isfinite(M) or M <= 0.0:
        return INF
    return float(2.0 * a.size * M / (S * S))


def sqrt_coeffs(n):
    """Coefficients of (1-z)^(-1/2), truncated to length n."""
    n = int(n)
    a = np.zeros(n, dtype=float)
    if n <= 0:
        return a
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * (2.0 * k - 1.0) / (2.0 * k)
    return a


def update_best(best_score, best_seq, candidate):
    if candidate is None:
        return best_score, best_seq
    c = np.asarray(candidate, dtype=float)
    if c.size == 0 or not np.all(np.isfinite(c)):
        return best_score, best_seq
    c = np.maximum(c, 0.0)
    s = compute_score(c)
    if np.isfinite(s) and s < best_score:
        return float(s), c.copy()
    return best_score, best_seq


def coord_eval(base, L, qi, S0, x):
    """Evaluate coordinate line objective at one x."""
    c = base + L * x
    if qi >= 0:
        c[qi] += x * x
    M = float(np.max(c))
    S = S0 + x
    if S <= 1e-300 or M <= 1e-300 or not np.isfinite(S) or not np.isfinite(M):
        return -1.0
    return float((S * S) / M)


def coordinate_descent(a, deadline, rng, max_sweeps=5000, grid_size=40):
    """
    Coordinate descent maximizing R = sum(a)^2 / max(conv(a,a)).
    This is equivalent to minimizing 2*n/R.
    """
    a = np.asarray(a, dtype=float).copy()
    a = np.maximum(a, 0.0)
    n = a.size
    if n == 0:
        return a, INF

    S0 = float(np.sqrt(2.0 * n))
    a = normalize_sum(a, S0)
    S = float(np.sum(a))
    if S <= 1e-300:
        a = np.ones(n, dtype=float) * (S0 / n)
        S = float(np.sum(a))

    m_len = 2 * n - 1
    c = fft_square(a)
    M = float(np.max(c))
    if M <= 1e-300:
        M = 1e-300
    bestR = float(S * S / M)
    best = a.copy()
    best_score = float(2.0 * n / bestR)

    no_improve_sweeps = 0

    for sweep in range(max_sweeps):
        if time.monotonic() >= deadline:
            break

        improved = False
        order = rng.permutation(n)
        max_a = float(np.max(a))

        for i in order:
            if time.monotonic() >= deadline:
                break

            x = float(a[i])
            S0i = S - x

            L = np.zeros(m_len, dtype=float)
            L[i:i + n] = 2.0 * a
            qi = 2 * i
            if qi < m_len:
                L[qi] = 0.0

            base = c - L * x
            if qi < m_len:
                base[qi] = base[qi] - x * x
            np.maximum(base, 0.0, out=base)

            cmax_now = float(np.max(c))
            upper = max(
                2.0,
                10.0 * x if x > 0.0 else 0.0,
                3.0 * max_a,
                float(np.sqrt(10.0 * cmax_now + 1.0))
            )
            upper = min(upper, 100.0)
            low = 1e-10
            if upper <= low:
                upper = 1e-6

            xs = np.logspace(np.log10(low), np.log10(upper), int(grid_size))
            extra = np.array([0.0, x, 0.5 * x, 2.0 * x, 0.05 * upper, 0.5 * upper], dtype=float)
            xs = np.concatenate((xs, extra))
            xs = xs[np.isfinite(xs) & (xs >= 0.0) & (xs <= upper)]

            if xs.size == 0:
                continue

            C = base[:, None] + L[:, None] * xs[None, :]
            if qi < m_len:
                C[qi, :] += xs * xs

            Mx = np.max(C, axis=0)
            Sx = S0i + xs
            R = np.full(xs.size, -1.0, dtype=float)
            valid = (Sx > 1e-300) & (Mx > 1e-300) & np.isfinite(Sx) & np.isfinite(Mx)
            if np.any(valid):
                R[valid] = (Sx[valid] * Sx[valid]) / Mx[valid]

            idx = int(np.argmax(R))

            # Local golden-style refinement around the grid winner.
            if (xs.size > 2 and 0 < idx < xs.size - 1 and
                    R[idx] > bestR * (1.0 - 1e-8)):
                lo = float(xs[idx - 1])
                hi = float(xs[idx + 1])
                best_x = float(xs[idx])
                best_val = float(R[idx])

                for _ in range(10):
                    m1 = lo + (hi - lo) / 3.0
                    m2 = hi - (hi - lo) / 3.0
                    r1 = coord_eval(base, L, qi, S0i, m1)
                    r2 = coord_eval(base, L, qi, S0i, m2)
                    if r1 >= r2:
                        hi = m2
                    else:
                        lo = m1
                    cm = 0.5 * (lo + hi)
                    rm = coord_eval(base, L, qi, S0i, cm)
                    if rm > best_val:
                        best_val = rm
                        best_x = cm

                if best_val > bestR + 1e-12:
                    a[i] = best_x
                    Cnew = base + L * best_x
                    if qi < m_len:
                        Cnew[qi] += best_x * best_x
                    c = np.maximum(Cnew, 0.0)
                    S = S0i + best_x
                    bestR = best_val
                    best_score = float(2.0 * n / bestR)
                    best = a.copy()
                    improved = True
                continue

            if R[idx] > bestR + 1e-12:
                a[i] = float(xs[idx])
                c = np.maximum(C[:, idx], 0.0)
                S = float(Sx[idx])
                bestR = float(R[idx])
                best_score = float(2.0 * n / bestR)
                best = a.copy()
                improved = True

        # Periodically recompute and re-normalize to remove drift.
        if sweep % 5 == 0:
            c = fft_square(a)
            S = float(np.sum(a))
            if S > 1e-300:
                a = normalize_sum(a, S0)
                c = fft_square(a)
                S = float(np.sum(a))
            M = float(np.max(c))
            if M > 1e-300 and S > 1e-300:
                R = float(S * S / M)
                if R > bestR:
                    bestR = R
                    best = a.copy()
                    best_score = float(2.0 * n / R)

        if improved:
            no_improve_sweeps = 0
        else:
            no_improve_sweeps += 1
            if no_improve_sweeps >= 2:
                break

    c = fft_square(best)
    S = float(np.sum(best))
    M = float(np.max(c))
    if S < 0.01 or M <= 1e-300:
        return best, INF
    return best, compute_score(best)


def adam_min_soft(a, deadline, t_start=1.0, t_end=80.0, lr=0.08, max_iter=500000):
    """
    Optimize in log-space using a smoothed-max gradient.
    The sum is fixed to sqrt(2n) at every step.
    """
    a = np.asarray(a, dtype=float).copy()
    a = np.maximum(a, 0.0)
    n = a.size
    if n == 0:
        return a, INF

    S0 = float(np.sqrt(2.0 * n))
    a = normalize_sum(a, S0)
    x = np.log(np.maximum(a, 1e-300))

    mbuf = np.zeros(n, dtype=float)
    vbuf = np.zeros(n, dtype=float)
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8

    rng = np.random.default_rng(1234567 + n)

    best = a.copy()
    best_score = compute_score(a)
    stable = 0
    last_print = -1.0
    it = 0

    while time.monotonic() < deadline and it < max_iter:
        it += 1

        a = np.exp(np.clip(x, -50.0, 50.0))
        s = float(np.sum(a))
        if s <= 1e-300:
            break
        a = normalize_sum(a, S0)

        c = fft_square(a)
        M = float(np.max(c))
        if not np.isfinite(M) or M <= 1e-300:
            break

        score = float(2.0 * n * M / (S0 * S0))
        if score < best_score - 1e-10:
            best_score = score
            best = a.copy()
            stable = 0
        else:
            stable += 1

        if stable > 8000:
            x = x + 0.25 * rng.standard_normal(n)
            mbuf[:] = 0.0
            vbuf[:] = 0.0
            stable = 0

        now = time.monotonic()
        if now - last_print > 30.0:
            print(f"    adam it={it} t={t:.2f} score={best_score:.6f}")
            last_print = now

        frac = min(1.0, float(it) / 30000.0)
        t = t_start * (t_end / t_start) ** frac

        flat = 0.0
        if t > 10.0:
            flat = min(0.25, 0.05 + 0.008 * (t - 10.0))

        w, _ = soft_weights(c, t, flat)
        h = corr_weights(w, a)
        g = 2.0 * a * h

        if not np.all(np.isfinite(g)):
            break

        normg = float(np.linalg.norm(g))
        if normg < 1e-14:
            break

        mbuf = b1 * mbuf + (1.0 - b1) * g
        vbuf = b2 * vbuf + (1.0 - b2) * (g * g)
        mhat = mbuf / (1.0 - b1 ** it)
        vhat = vbuf / (1.0 - b2 ** it)

        step = lr * mhat / (np.sqrt(vhat) + eps)
        step = np.nan_to_num(step, nan=0.0, posinf=0.0, neginf=0.0)
        step -= float(np.mean(step))
        step = np.clip(step, -0.35, 0.35)
        x = x - step

        if it % 1000 == 0:
            x = x + 0.02 * rng.standard_normal(n)

        x = np.nan_to_num(x, nan=0.0, posinf=50.0, neginf=-50.0)
        a = np.exp(np.clip(x, -50.0, 50.0))
        s = float(np.sum(a))
        if s <= 1e-300:
            break
        a = normalize_sum(a, S0)
        x = np.log(np.maximum(a, 1e-300))

    final_score = compute_score(best)
    if not np.isfinite(final_score):
        final_score = best_score
    return best, final_score


def lp_direction(f):
    """
    Solve LP: maximize sum(g) subject to convolve(f,g) <= 1,
    0 <= g <= 100.
    f should have max convolution about 1.
    """
    if linprog is None:
        return None
    n = f.size
    if n < 2 or n > 768:
        return None

    m_len = 2 * n - 1
    A = np.zeros((m_len, n), dtype=float)
    for j in range(n):
        A[j:j + n, j] = f

    b = np.full(m_len, 1.0 + 1e-9, dtype=float)
    c_obj = -np.ones(n, dtype=float)

    try:
        res = linprog(
            c_obj,
            A_ub=A,
            b_ub=b,
            bounds=(0.0, 100.0),
            method='highs',
            options={'presolve': True}
        )
        ok = False
        if hasattr(res, 'success'):
            ok = bool(res.success)
        if hasattr(res, 'status'):
            ok = ok or int(res.status) == 0
        if ok and hasattr(res, 'x'):
            g = np.asarray(res.x, dtype=float)
            g = np.maximum(g, 0.0)
            if float(np.sum(g)) > 1e-12 and np.all(np.isfinite(g)):
                return g
    except Exception:
        pass

    return None


def lp_blend_optimize(a, deadline, rng, max_calls=3):
    """Use LP directions and blends, with short coordinate polish."""
    a = np.asarray(a, dtype=float).copy()
    a = np.maximum(a, 0.0)
    n = a.size
    if n == 0:
        return a, INF

    S0 = float(np.sqrt(2.0 * n))
    best_local = a.copy()
    best_local_score = compute_score(a)

    if n > 640:
        return best_local, best_local_score

    f = normalize_sum(a, S0)

    for call in range(max_calls):
        if time.monotonic() >= deadline:
            break

        f = normalize_sum(f, S0)
        c = fft_square(f)
        M = float(np.max(c))
        if M <= 1e-300:
            break

        f_norm = f / np.sqrt(M)
        g = lp_direction(f_norm)
        if g is None:
            break

        cgg = fft_square(g)
        Mgg = float(np.max(cgg))
        if Mgg <= 1e-300:
            break
        g = g / np.sqrt(Mgg)

        call_best = f_norm.copy()
        call_best_score = best_local_score

        candidates = []
        candidates.append(g.copy())
        for th in (0.15, 0.35, 0.60, 1.0):
            candidates.append((1.0 - th) * f_norm + th * g)

        if np.all(f_norm > 1e-300) and np.all(g > 1e-300):
            for th in (0.3, 0.5, 0.7):
                h = np.exp((1.0 - th) * np.log(f_norm) + th * np.log(g))
                candidates.append(h)

        for h in candidates:
            h = np.maximum(h, 0.0)
            if float(np.sum(h)) <= 1e-300:
                continue
            sc = compute_score(h)
            if np.isfinite(sc):
                if sc < best_local_score:
                    best_local_score = sc
                    best_local = h.copy()
                if sc < call_best_score:
                    call_best_score = sc
                    call_best = h.copy()

        f = normalize_sum(call_best, S0)

        if time.monotonic() < deadline - 12.0:
            sub_deadline = min(deadline, time.monotonic() + min(30.0, deadline - time.monotonic()))
            seq, sc = coordinate_descent(f, sub_deadline, rng, grid_size=24, max_sweeps=4)
            if np.isfinite(sc) and sc < best_local_score:
                best_local = seq
                best_local_score = sc
                f = seq

    final_score = compute_score(best_local)
    if not np.isfinite(final_score):
        final_score = best_local_score
    return best_local, final_score


def generate_starts(rng, lengths, base):
    """Create diverse non-negative starts."""
    starts = []

    for n in lengths:
        n = int(n)
        if n < 2:
            continue

        # Random lognormal.
        a = np.exp(rng.normal(0.0, 0.6, n))
        if np.all(np.isfinite(a)) and float(np.sum(a)) > 0.0:
            starts.append(np.maximum(a, 0.0))

        # One-sided sqrt(1/(1-z)) sequence.
        a = sqrt_coeffs(n)
        starts.append(a.copy())

        # Reversed one-sided sequence.
        a = sqrt_coeffs(n)[::-1].copy()
        starts.append(a)

        # Two bursts: early and late.
        a = np.zeros(n, dtype=float)
        L = max(2, n // 3)
        R = max(2, n // 4)
        a[:L] = sqrt_coeffs(L)
        a[n - R:] = sqrt_coeffs(R)
        starts.append(a)

        # Base interpolation and perturbed base.
        if base is not None and base.size > 1:
            bp = np.maximum(np.asarray(base, dtype=float), 0.0)
            if np.all(np.isfinite(bp)) and float(np.sum(bp)) > 0.0:
                if n == bp.size:
                    starts.append(bp.copy())
                else:
                    old_x = np.linspace(0.0, 1.0, bp.size)
                    new_x = np.linspace(0.0, 1.0, n)
                    arr = np.interp(new_x, old_x, bp)
                    arr = np.maximum(arr, 0.0)
                    if float(np.sum(arr)) > 0.0:
                        starts.append(arr.copy())

                    arr2 = arr * np.exp(rng.normal(0.0, 0.4, n))
                    arr2 = np.maximum(arr2, 0.0)
                    if float(np.sum(arr2)) > 0.0:
                        starts.append(arr2.copy())

                a0 = bp if bp.size == n else sqrt_coeffs(n)
                for p in (0.75, 1.25):
                    arr = np.power(np.maximum(a0, 1e-300), p)
                    if np.all(np.isfinite(arr)) and float(np.sum(arr)) > 0.0:
                        starts.append(arr)

    out = []
    for s in starts:
        s = np.asarray(s, dtype=float)
        if s.size > 0 and np.all(np.isfinite(s)) and float(np.sum(s)) > 0.0:
            out.append(np.maximum(s, 0.0))
    return out


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search routine. Returns the best non-negative sequence found.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + max(60.0, float(budget_s) - 45.0)

    best_score = INF
    best_seq = None

    # Use the provided previous construction if possible.
    base = None
    try:
        hs = np.asarray(height_sequence_1, dtype=float)
        if hs.size >= 8 and np.all(np.isfinite(hs)):
            base = np.maximum(hs, 0.0)
    except Exception:
        base = None

    if base is None or base.size < 8 or float(np.sum(base)) <= 0.0:
        base = sqrt_coeffs(384)

    a0 = normalize_sum(base, np.sqrt(2.0 * base.size))
    best_score, best_seq = update_best(best_score, best_seq, a0)
    print(f"initial score={best_score:.6f}, n={best_seq.size if best_seq is not None else 0}")

    # Stage 1: strong coordinate descent on the best known sequence.
    now = time.monotonic()
    reserve = 430.0
    alloc = min(820.0, deadline - now - reserve)
    if alloc > 45.0:
        seq, sc = coordinate_descent(best_seq.copy(), now + alloc, rng, grid_size=56)
        best_score, best_seq = update_best(best_score, best_seq, seq)
        print(f"after base coordinate: score={best_score:.6f}, n={best_seq.size if best_seq is not None else 0}")

    # Stage 2: Adam on log-variables, then coordinate polish.
    now = time.monotonic()
    reserve = 320.0
    alloc = min(200.0, deadline - now - reserve)
    if alloc > 45.0 and best_seq is not None and best_seq.size <= 1024:
        seq, sc = adam_min_soft(best_seq.copy(), now + alloc)
        best_score, best_seq = update_best(best_score, best_seq, seq)
        print(f"after adam: score={best_score:.6f}, n={best_seq.size if best_seq is not None else 0}")

        cand = seq if seq is not None and np.all(np.isfinite(seq)) else best_seq
        now = time.monotonic()
        alloc2 = min(110.0, deadline - now - 180.0)
        if alloc2 > 30.0:
            seq2, sc2 = coordinate_descent(cand.copy(), now + alloc2, rng, grid_size=48)
            best_score, best_seq = update_best(best_score, best_seq, seq2)
            print(f"after adam coordinate: score={best_score:.6f}, n={best_seq.size if best_seq is not None else 0}")

    # Stage 3: LP-based directions, if length is moderate.
    now = time.monotonic()
    reserve = 260.0
    alloc = min(210.0, deadline - now - reserve)
    if alloc > 45.0 and best_seq is not None and best_seq.size <= 576:
        seq, sc = lp_blend_optimize(best_seq.copy(), now + alloc, rng, max_calls=3)
        best_score, best_seq = update_best(best_score, best_seq, seq)
        print(f"after lp: score={best_score:.6f}, n={best_seq.size if best_seq is not None else 0}")

        now = time.monotonic()
        alloc2 = min(90.0, deadline - now - 140.0)
        if alloc2 > 25.0:
            seq2, sc2 = coordinate_descent(best_seq.copy(), now + alloc2, rng, grid_size=48)
            best_score, best_seq = update_best(best_score, best_seq, seq2)

    # Stage 4: Explore other lengths and starting shapes.
    now = time.monotonic()
    if now < deadline - 180.0 and best_seq is not None:
        best_n = int(best_seq.size)
        candidate_lengths = [384, 512, 640, 768]
        lengths = [l for l in candidate_lengths if l != best_n and l <= 768]
        lengths = lengths[:3]

        if lengths:
            starts = generate_starts(rng, lengths, best_seq)
            scored_pairs = []

            for idx, arr in enumerate(starts):
                now = time.monotonic()
                if now > deadline - 150.0:
                    break
                arr = np.maximum(np.asarray(arr, dtype=float), 0.0)
                if arr.size == 0 or not float(np.sum(arr)):
                    continue
                arr = normalize_sum(arr, np.sqrt(2.0 * arr.size))
                s = compute_score(arr)
                starts[idx] = arr
                if np.isfinite(s):
                    scored_pairs.append((s, idx))

            scored_pairs.sort()
            max_starts = min(4, len(scored_pairs))

            for rank, (s, idx) in enumerate(scored_pairs[:max_starts]):
                now = time.monotonic()
                if now > deadline - 130.0:
                    break

                remaining = deadline - now - 130.0
                if remaining < 70.0:
                    break

                arr = np.asarray(starts[idx], dtype=float).copy()
                print(f"explore start n={arr.size} initial_score={s:.6f}")

                alloc_cd = min(110.0, remaining * 0.45)
                seq_cd, sc_cd = coordinate_descent(arr.copy(), now + alloc_cd, rng, grid_size=36, max_sweeps=250)
                best_score, best_seq = update_best(best_score, best_seq, seq_cd)

                cand = arr
                if np.isfinite(sc_cd) and sc_cd < s:
                    cand = seq_cd

                now = time.monotonic()
                alloc_adam = min(90.0, deadline - now - 80.0)
                if alloc_adam > 30.0:
                    seq_ad, sc_ad = adam_min_soft(cand.copy(), now + alloc_adam, t_start=5.0, t_end=80.0, lr=0.06)
                    best_score, best_seq = update_best(best_score, best_seq, seq_ad)

                    cand2 = seq_ad if np.isfinite(sc_ad) and sc_ad < compute_score(cand) else cand
                    now = time.monotonic()
                    alloc_cf = min(60.0, deadline - now - 55.0)
                    if alloc_cf > 20.0:
                        seq_cf, sc_cf = coordinate_descent(cand2.copy(), now + alloc_cf, rng, grid_size=40, max_sweeps=150)
                        best_score, best_seq = update_best(best_score, best_seq, seq_cf)

    # Stage 5: Escape attempts around the current best.
    if best_seq is not None:
        last_escape_score = best_score
        while time.monotonic() < deadline - 70.0:
            now = time.monotonic()
            alloc = min(85.0, deadline - now - 55.0)
            if alloc < 25.0:
                break

            n = int(best_seq.size)
            a_esc = best_seq * np.exp(rng.normal(0.0, 0.15, n))
            a_esc = np.maximum(a_esc, 0.0)
            a_esc = normalize_sum(a_esc, np.sqrt(2.0 * n))

            seq, sc = coordinate_descent(a_esc, now + alloc, rng, grid_size=40, max_sweeps=250)
            old_score = best_score
            best_score, best_seq = update_best(best_score, best_seq, seq)

            if not np.isfinite(best_score):
                break
            if best_score >= last_escape_score and time.monotonic() > deadline - 120.0:
                break
            last_escape_score = best_score
            print(f"escape score={best_score:.6f}")

    # Final polish on the global best.
    if best_seq is not None and time.monotonic() < deadline - 15.0:
        now = time.monotonic()
        alloc = min(320.0, deadline - now - 12.0)
        if alloc > 20.0:
            seq, sc = coordinate_descent(best_seq.copy(), now + alloc, rng, grid_size=72)
            best_score, best_seq = update_best(best_score, best_seq, seq)
            print(f"after final coordinate: score={best_score:.6f}, n={best_seq.size if best_seq is not None else 0}")

    # Fallback.
    if best_seq is None or not np.isfinite(best_score):
        fallback = sqrt_coeffs(384)
        fallback = normalize_sum(fallback, np.sqrt(2.0 * fallback.size))
        return [float(x) for x in fallback]

    # Final safe normalization.
    arr = np.asarray(best_seq, dtype=float).copy()
    arr = np.maximum(arr, 0.0)
    arr = normalize_sum(arr, np.sqrt(2.0 * arr.size))

    amax = float(np.max(arr))
    if amax > 1000.0:
        arr = arr * (1000.0 / amax)

    s = float(np.sum(arr))
    if s < 0.02:
        if s > 1e-300:
            arr = arr * (1.0 / s)
        else:
            arr = np.ones_like(arr) * 0.01

    # Optional official validation.
    try:
        official = float(evaluate_sequence([float(x) for x in arr]))
        if np.isfinite(official):
            print(f"official score={official:.6f}, local score={compute_score(arr):.6f}")
    except Exception:
        pass

    return [float(x) for x in arr]
```
