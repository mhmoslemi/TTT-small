    """Solve LP: max sum(g) s.t. conv(a,g)[k] <= M, g >= 0."""
    n = len(a)
    # Build the constraint matrix efficiently
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            A[k, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]

    return np.zeros(n)

            a = a_new
            stall = 0
            lr = min(lr * 1.05, lr_max)
        elif new_peak < peak_val:
            a = a_new
            stall += 1
            if stall > 30:
                lr *= 0.7
                stall = 0
        else:
            stall += 1
            lr *= 0.8
            if lr < lr_min:
                lr = lr_min
                # Try a random restart with small perturbation
                if stall > 100:
                    rng = np.random.default_rng(42 + step_count)
                    a = a * np.exp(rng.normal(0, 0.01, n))
                    a = np.maximum(a, 0)
                    a = normalize(a, S)
                    stall = 0
                    lr = 0.01 * S / n

        step_count += 1
        if verbose and step_count % 200 == 0:
            score = 2.0 * n * best_conv_max / (S * S)
            print(f"      SGD step {step_count}: score={score:.6f}, lr={lr:.2e}")

    return best_a


def coordinate_descent(a, time_limit, verbose=False):
    """Coordinate descent: optimize individual or paired coordinates."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize(a, S).copy()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = local_score(a)

    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(a, a)
        peak_idx = int(np.argmax(conv))
        peak_val = conv[peak_idx]

        # Find coordinates that most contribute to the peak
        # conv[peak_idx] = sum_j a[j] * a[peak_idx - j]
        contribs = np.zeros(n)
        lo = max(0, peak_idx - n + 1)
        hi = min(peak_idx, n - 1)
        for j in range(lo, hi + 1):
            other = peak_idx - j
            if 0 <= other < n:
                contribs[j] = a[j] * a[other]
                contribs[other] = a[j] * a[other]

        top_k = max(5, n // 16)
        top_idx = np.argsort(contribs)[-top_k:]

        # Try reducing top contributors and increasing non-contributors
        improved = False
        for _ in range(8):
            candidate = a.copy()

            # Randomly reduce some top contributors
            n_reduce = max(2, top_k // 3)
            reduce_idx = rng.choice(top_idx, size=n_reduce, replace=False)
            for idx in reduce_idx:
                factor = np.exp(rng.normal(-0.03, 0.01))
                candidate[idx] *= factor

            # Increase some non-top coordinates
            non_top = np.setdiff1d(np.arange(n), top_idx)
            if len(non_top) > 0:
                n_inc = max(2, len(non_top) // 20)
                inc_idx = rng.choice(non_top, size=n_inc, replace=False)
                for idx in inc_idx:
                    candidate[idx] *= np.exp(rng.normal(0.03, 0.01))

            candidate = np.maximum(candidate, 0.0)
            candidate = normalize(candidate, S)
            new_score = local_score(candidate)

            if new_score < best_score:
                best_score = new_score
                best_a = candidate.copy()
                a = candidate
                improved = True

        if not improved:
            # Try paired swaps: reduce one, increase another
            i1 = int(rng.integers(0, n))
            i2 = int(rng.integers(0, n))
            if i1 != i2:
                candidate = a.copy()
                delta = a[i1] * rng.uniform(0.01, 0.05)
                candidate[i1] -= delta
                candidate[i2] += delta
                if candidate[i1] < 0:
                    candidate[i2] += candidate[i1]
                    candidate[i1] = 0
                candidate = np.maximum(candidate, 0.0)
                candidate = normalize(candidate, S)
                new_score = local_score(candidate)
                if new_score < best_score:
                    best_score = new_score
                    best_a = candidate.copy()
                    a = candidate

        if verbose and int(time.monotonic() - t0) % 30 == 0:
            print(f"      CD t={time.monotonic()-t0:.0f}s: score={best_score:.6f}")

    return best_a


def simulated_annealing(a, time_limit, verbose=False):
    """Simulated annealing with structured moves."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize(a, S).copy()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = local_score(a)
    current_score = best_score

    rng = np.random.default_rng(999)

    # Temperature schedule
    T_start = 0.01 * best_score
    T_end = 0.0001 * best_score
    total_iters = int(time_limit * 50)  # estimate iterations
    it = 0

    while time.monotonic() - t0 < time_limit:
        T = T_start * (T_end / T_start) ** (float(it) / max(total_iters, 1))
        if T < T_end:
            T = T_end

        candidate = a.copy()
        move_type = rng.random()

        if move_type < 0.4:
            # Gaussian perturbation on random block
            width = int(rng.integers(max(1, n // 100), max(2, n // 10)))
            start = int(rng.integers(0, n - width))
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.02 * T / T_end ** 0.5, width))
        elif move_type < 0.7:
            # Targeted: reduce at peak-contributing positions
            conv = fft_convolve(candidate, candidate)
            pk = int(np.argmax(conv))
            lo = max(0, pk - n + 1)
            hi = min(pk, n - 1)
            j = int(rng.integers(lo, hi + 1))
            candidate[j] *= np.exp(rng.normal(-0.02, 0.01))
            # Compensate elsewhere
            k = int(rng.integers(0, n))
            if k != j:
                candidate[k] *= np.exp(rng.normal(0.02, 0.01))
        else:
            # Mirror flip: swap a[j] and a[n-1-j]
            j = int(rng.integers(0, n // 2))
            candidate[j], candidate[n - 1 - j] = candidate[n - 1 - j], candidate[j]

        candidate = np.maximum(candidate, 0.0)
        candidate = normalize(candidate, S)
        new_score = local_score(candidate)

        delta = new_score - current_score
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            a = candidate
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = candidate.copy()

        it += 1
        if verbose and it % 500 == 0:
            print(f"      SA iter {it}: T={T:.6f}, best={best_score:.6f}, curr={current_score:.6f}")

    return best_a


def try_different_length(a, n_target, time_per_method=20):
    """Interpolate to target length and optimize."""
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_target)
    a_interp = np.interp(x_new, x_old, a)
    a_interp = np.maximum(a_interp, 0.01)
    return a_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    1. LP-based refinement
    2. Subgradient descent with Adam
    3. Coordinate descent
    4. Simulated annealing polish
    5. Final LP refinement
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 30.0
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    def update_best(a):
        nonlocal best_score, best_seq
        a_clip = np.clip(a, 0.0, 1000.0)
        s = local_score(a_clip)
        if s < best_score:
            best_score = s
            best_seq = a_clip.copy()
            return True
        return False

    # Get provided sequence
    provided_seq = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            provided_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
            print(f"Provided sequence: n={len(provided_seq)}, score={local_score(provided_seq):.6f}")
            update_best(provided_seq)
    except Exception as e:
        print(f"Could not load provided sequence: {e}")

    if provided_seq is None:
        # Create a default starting sequence
        n = 640
        provided_seq = np.ones(n) * 0.5 + 0.3 * np.sin(np.pi * np.arange(n) / n)
        provided_seq = np.maximum(provided_seq, 0.01)
        update_best(provided_seq)

    n_prov = len(provided_seq)

    # === Phase 1: LP refinement ===
    print(f"\n=== Phase 1: LP refinement (n={n_prov}) ===")
    remaining = deadline - time.monotonic()
    t_lp = min(180.0, remaining * 0.25)
    print(f"  Time budget: {t_lp:.0f}s")
    result = lp_refine_sequence(provided_seq, t_lp, verbose=True)
    if update_best(result):
        print(f"  LP improved to: {best_score:.6f}")
    else:
        print(f"  LP no improvement: still {best_score:.6f}")

    # === Phase 2: Subgradient descent ===
    print(f"\n=== Phase 2: Subgradient descent (n={n_prov}) ===")
    remaining = deadline - time.monotonic()
    t_sgd = min(300.0, remaining * 0.3)
    print(f"  Time budget: {t_sgd:.0f}s")
    result = subgradient_descent(best_seq, t_sgd, verbose=True)
    if update_best(result):
        print(f"  SGD improved to: {best_score:.6f}")
    else:
        print(f"  SGD no improvement: still {best_score:.6f}")

    # === Phase 3: Coordinate descent ===
    print(f"\n=== Phase 3: Coordinate descent (n={n_prov}) ===")
    remaining = deadline - time.monotonic()
    t_cd = min(300.0, remaining * 0.3)
    print(f"  Time budget: {t_cd:.0f}s")
    result = coordinate_descent(best_seq, t_cd, verbose=True)
    if update_best(result):
        print(f"  CD improved to: {best_score:.6f}")
    else:
        print(f"  CD no improvement: still {best_score:.6f}")

    # === Phase 4: Simulated annealing ===
    print(f"\n=== Phase 4: Simulated annealing (n={n_prov}) ===")
    remaining = deadline - time.monotonic()
    t_sa = min(200.0, remaining * 0.3)
    print(f"  Time budget: {t_sa:.0f}s")
    result = simulated_annealing(best_seq, t_sa, verbose=True)
    if update_best(result):
        print(f"  SA improved to: {best_score:.6f}")
    else:
        print(f"  SA no improvement: still {best_score:.6f}")

    # === Phase 5: Final LP polish ===
    print(f"\n=== Phase 5: Final LP polish ===")
    remaining = deadline - time.monotonic()
    t_final = min(remaining - 20, 120.0)
    if t_final > 30:
        print(f"  Time budget: {t_final:.0f}s")
        result = lp_refine_sequence(best_seq, t_final, verbose=True)
        if update_best(result):
            print(f"  Final LP improved to: {best_score:.6f}")

    # === Phase 6: Try nearby lengths ===
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Phase 6: Try nearby lengths ===")
        for n_try in [512, 768]:
            if time.monotonic() > deadline - 40:
                break
            a_try = try_different_length(best_seq, n_try)
            t_sub = min(40.0, (deadline - time.monotonic()) * 0.4)
            if t_sub > 15:
                result = lp_refine_sequence(a_try, t_sub, verbose=False)
                if update_best(result):
                    print(f"  n={n_try}: improved to {best_score:.6f}")
                    result = subgradient_descent(result, min(30.0, (deadline - time.monotonic()) * 0.3), verbose=False)
                    update_best(result)
                    print(f"  n={n_try} after SGD: {best_score:.6f}")

    # Final verification
    print(f"\n=== FINAL ===")
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = local_score(best_seq)
        print(f"  n={len(final_list)}, local_score={final_score:.6f}")
        # Verify with evaluate_sequence
        try:
            eval_score = evaluate_sequence(final_list)
            print(f"  evaluate_sequence score={eval_score:.6f}")
        except Exception as e:
            print(f"  Could not verify: {e}")
        return final_list

    # Fallback
    return [1.0] * 256
