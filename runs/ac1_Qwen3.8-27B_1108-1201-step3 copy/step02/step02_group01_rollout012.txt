def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):

def normalize_seq(a, S=None):
    """Normalize to given sum."""
    if S is None:
        S = np.sqrt(2.0 * len(a))
    s = np.sum(a)
    if s < 1e-12:

def build_lp_matrix(a):
    """Build convolution matrix A where (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    # Use structured construction
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            A[k, j] = a[k - j]
    return A

def lp_solve_fast(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_lp_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.status == 0:
        return res.x
    return np.zeros(n)

def subgradient_optimize(a, time_limit, verbose=False):

    n = len(a)

        for idx in top_idx:

            a = candidate

"""
Multi-strategy optimization for minimizing normalized peak autocorrelation
of non-negative step functions. Combines LP-based directional search with
proper line search, multi-peak shaving, spectral initialization, diverse
starts, and simulated annealing restarts across multiple sequence lengths.

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return float(2.0 * n * np.max(conv) / (sum_a ** 2))


def normalize_seq(a, S=None):
    """Normalize to given sum."""
    if S is None:
        S = np.sqrt(2.0 * len(a))
    s = np.sum(a)
    if s < 1e-12:
        return a * S / len(a)
    return a * S / s


def build_lp_matrix(a):
    """Build convolution matrix A where (A@g)[k] = conv(a,g)[k]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            A[k, j] = a[k - j]
    return A


def lp_solve(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_lp_matrix(a)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.status == 0:
        return res.x
    return np.zeros(n)


def line_search_blend(a, g, S, M_current):

    n = len(a)
    best_t = 0.0
    best_peak = M_current

    # Coarse search

        b = (1 - t_try) * a + t_try * g
        b = np.clip(b, 0.0, None)
        b = normalize_seq(b, S)
        peak = np.max(fft_convolve(b, b))
        if peak < best_peak:
            best_peak = peak
            best_t = t_try

    # Fine search

        lo = max(0.005, best_t - 0.04)
        hi = min(1.0, best_t + 0.04)

            b = (1 - t_try) * a + t_try * g
            b = np.clip(b, 0.0, None)
            b = normalize_seq(b, S)
            peak = np.max(fft_convolve(b, b))
            if peak < best_peak:
                best_peak = peak
                best_t = t_try

        # We want to reduce a[i] where it contributes to multiple hot lags

def spectral_init(n, rng):
    """Create initialization from flat-ish spectrum."""
    # Random phases
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    phases = rng.uniform(0, 2 * np.pi, size // 2 + 1)
    # Make magnitude slightly non-uniform to help
    mags = 1.0 + 0.1 * rng.standard_normal(size // 2 + 1)
    mags = np.abs(mags) + 0.5
    spectrum = mags * np.exp(1j * phases)
    spectrum[0] = np.abs(spectrum[0])
    # Conjugate symmetry
    if size % 2 == 0:
        spectrum[-1] = np.abs(spectrum[-1])
    full = np.zeros(size, dtype=complex)
    full[:size // 2 + 1] = spectrum
    for i in range(size // 2 + 1, size):
        full[i] = np.conj(full[size - i])
    a = np.real(np.fft.ifft(full))[:n]
    a = np.abs(a) + 0.1
    return a


def subgradient_step(a, S, lr, rng):
    """One subgradient step with random tie-breaking among peaks."""
    n = len(a)
    conv = fft_convolve(a, a)
    peak_val = np.max(conv)
    # Pick a random peak among those within 1% of max
    hot = np.where(conv > 0.99 * peak_val)[0]
    peak_idx = hot[rng.integers(len(hot))]

    grad = np.zeros(n)
    for j in range(n):
        val = 0.0
        i1 = peak_idx - j
        if 0 <= i1 < n:
            val += a[i1]
        i2 = j - peak_idx
        if 0 <= i2 < n:
            val += a[i2]
        grad[j] = val

    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    a_new = normalize_seq(a_new, S)
    return a_new

    n = len(a)
    S = np.sqrt(2.0 * n)

    no_improve = 0

        if time.monotonic() - t0 > time_limit:
            break

        a_new = subgradient_step(a, S, lr, rng)

            best_peak = new_peak
            best_a = a_new.copy()
            a = a_new
            no_improve = 0

        else:
            no_improve += 1

            a = np.clip(a, 0.001, None)

            no_improve = 0

    return best_a

        a = 1.0 + 0.4 * np.exp(-0.5 * ((t - c1) / sigma) ** 2) + 0.4 * np.exp(-0.5 * ((t - c2) / sigma) ** 2)
    elif init_type == 'barker_like':
        barker = np.array([1, 1, 1, -1, 1, -1, 1], dtype=float)
        a = np.ones(n)
        for i in range(n):
            a[i] = 1.0 + 0.25 * barker[i % 7]
    elif init_type == 'costlike':
        cost = np.array([0, 1, 1, 0, 1, 0, 1], dtype=float)
        a = 1.0 + 0.3 * np.array([cost[i % 7] for i in range(n)])
    elif init_type == 'gaussian_mixture':
        a = np.ones(n)
        for _ in range(3):
            c = rng.uniform(0, n)
            s = n * rng.uniform(0.05, 0.2)
            a += 0.3 * np.exp(-0.5 * ((t - c) / s) ** 2)
    elif init_type == 'log_cosine':
        a = 1.0 + 0.2 * np.cos(np.pi * np.log1p(t) / np.log1p(n))
    elif init_type == 'sawtooth_smooth':
        period = max(7, n // 8)
        a = 1.0 + 0.15 * np.sin(2 * np.pi * t / period) * np.exp(-t / n)
    else:
        a = np.ones(n)
    a = np.clip(a, 0.01, None)
    return a


def interpolate_to_length(a, n_new):
    """Interpolate sequence to new length."""
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, a)


def optimize_single(a, time_budget, verbose=False):
    """Run full optimization pipeline on a single sequence."""
    n = len(a)
    t_start = time.monotonic()

    # Phase 1: LP refinement (most effective)
    t_lp = time_budget * 0.45
    a = lp_refine(a, t_lp, verbose=verbose)

    elapsed = time.monotonic() - t_start
    remaining = time_budget - elapsed - 5
    if remaining < 5:
        return a

    # Phase 2: Multi-peak shaving
    t_mp = min(remaining * 0.2, 60.0)
    a = multi_peak_shave(a, t_mp, verbose=verbose)

    elapsed = time.monotonic() - t_start
    remaining = time_budget - elapsed - 5
    if remaining < 5:
        return a

    # Phase 3: Subgradient
    t_sub = min(remaining * 0.3, 60.0)
    a = subgradient_optimize(a, t_sub, verbose=verbose)

    elapsed = time.monotonic() - t_start
    remaining = time_budget - elapsed - 5
    if remaining < 5:
        return a

    # Phase 4: Final LP polish
    a = lp_refine(a, remaining, verbose=verbose)

    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy search: LP directional search with line search,
    multi-peak shaving, subgradient, spectral inits, diverse starts,
    and multi-length exploration.
    """
    t_global = time.monotonic()
    deadline = t_global + budget_s - 25.0
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    def update_best(a):
        nonlocal best_score, best_seq
        a_clip = np.clip(a, 0.0, 1000.0)
        s = local_score(a_clip)
        if s < best_score:
            best_score = s
            best_seq = a_clip.copy()
            return True
        return False

    # Get provided sequence
    provided_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        try:
            provided_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
            provided_score = local_score(provided_seq)
            print(f"Provided: n={len(provided_seq)}, score={provided_score:.6f}")
            update_best(provided_seq)
        except Exception:
            pass

    # === Phase A: Deep optimization of provided sequence ===
    if provided_seq is not None:
        n = len(provided_seq)
        t_budget = min(400.0, (deadline - time.monotonic()) * 0.3)
        print(f"\n=== Phase A: Deep optimize provided (n={n}), budget={t_budget:.0f}s ===")
        result = optimize_single(provided_seq.copy(), t_budget, verbose=True)
        if update_best(result):
            print(f"  Improved to: {best_score:.6f}")

    # === Phase B: Try provided at other lengths ===
    if provided_seq is not None:
        n_orig = len(provided_seq)
        other_lengths = [448, 512, 576, 640, 704, 768, 832, 896, 960, 1024]
        other_lengths = [x for x in other_lengths if x != n_orig]
        for n_try in other_lengths:
            if time.monotonic() > deadline - 250:
                break
            t_budget = min(50.0, (deadline - time.monotonic()) * 0.08)
            a0 = interpolate_to_length(provided_seq, n_try)
            a0 = np.clip(a0, 0.01, None)
            # Quick LP only
            result = lp_refine(a0, t_budget, verbose=False)
            if update_best(result):
                print(f"  *** n={n_try}: score={best_score:.6f} ***")

    # === Phase C: Diverse initializations ===
    init_types = ['chirp', 'sine', 'multi_tone', 'random', 'bimodal',
                  'barker_like', 'costlike', 'gaussian_mixture', 'log_cosine',
                  'sawtooth_smooth', 'triangle']
    n_values = [256, 320, 384, 448, 512, 576, 640, 704, 768]

    for n in n_values:
        if time.monotonic() > deadline - 150:
            break
        for init_type in init_types[:6]:
            if time.monotonic() > deadline - 130:
                break
            t_budget = min(35.0, (deadline - time.monotonic()) * 0.06)
            a0 = make_initialization(n, init_type, rng)
            result = optimize_single(a0, t_budget, verbose=False)
            if update_best(result):
                print(f"  *** n={n}, init={init_type}: score={best_score:.6f} ***")

    # === Phase D: Spectral initializations ===
    for n in [384, 512, 640, 768]:
        if time.monotonic() > deadline - 100:
            break
        for trial in range(3):
            if time.monotonic() > deadline - 90:
                break
            t_budget = min(30.0, (deadline - time.monotonic()) * 0.05)
            a0 = spectral_init(n, rng)
            result = optimize_single(a0, t_budget, verbose=False)
            if update_best(result):
                print(f"  *** spectral n={n}, trial={trial}: score={best_score:.6f} ***")

    # === Phase E: Deep optimization on current best ===
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 20
        if remaining > 100:
            t_budget = min(remaining * 0.5, 400.0)
            print(f"\n=== Phase E: Deep optimize best (score={best_score:.6f}), budget={t_budget:.0f}s ===")
            n = len(best_seq)
            result = optimize_single(best_seq.copy(), t_budget, verbose=True)
            if update_best(result):
                print(f"  Improved to: {best_score:.6f}")

    # === Phase F: Perturb best and re-optimize ===
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 15
        if remaining > 30:
            n_best = len(best_seq)
            for perturb_scale in [0.03, 0.05, 0.08]:
                if time.monotonic() > deadline - 15:
                    break
                t_budget = min(40.0, (deadline - time.monotonic()) * 0.2)
                a_perturbed = best_seq * np.exp(rng.normal(0, perturb_scale, n_best))
                a_perturbed = np.clip(a_perturbed, 0.01, None)
                result = optimize_single(a_perturbed, t_budget, verbose=False)
                if update_best(result):
                    print(f"  *** perturb {perturb_scale}: score={best_score:.6f} ***")

    # === Phase G: Final LP polish ===
    if best_seq is not None:
        remaining = deadline - time.monotonic() - 5
        if remaining > 10:
            result = lp_refine(best_seq.copy(), remaining, verbose=False)
            if update_best(result):
                print(f"  Final polish: score={best_score:.6f}")

    # Final
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: eval={final_score:.6f}, n={len(final_list)}, local={best_score:.6f} ===")
        return final_list

    if provided_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in provided_seq]

    return [1.0] * 256
