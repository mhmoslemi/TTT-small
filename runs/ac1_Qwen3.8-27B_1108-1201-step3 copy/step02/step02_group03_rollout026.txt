def compute_obj(a):
    """Compute objective using FFT-based autocorrelation."""
    n = len(a)
    fsize = 1
    while fsize < 2 * n - 1:
        fsize *= 2
    F = np.fft.rfft(a, fsize)
    c = np.fft.irfft(F * F, fsize)[:2 * n - 1]
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def compute_autocorr(a):
    """Compute full autocorrelation via FFT."""
    n = len(a)
    fsize = 1
    while fsize < 2 * n - 1:
        fsize *= 2
    F = np.fft.rfft(a, fsize)
    return np.fft.irfft(F * F, fsize)[:2 * n - 1]


def compute_grad_soft(a, w):
    """
    Gradient of sum_k w[k]*c[k] w.r.t. a.
    grad[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
    Computed via FFT-based cross-correlation.
    """
    n = len(a)
    a_flip = a[::-1]
    fsize = 1
    while fsize < 3 * n - 1:
        fsize *= 2
    Fw = np.fft.rfft(w, fsize)
    Fa = np.fft.rfft(a_flip, fsize)
    conv = np.fft.irfft(Fw * Fa, fsize)
    grad = 2.0 * conv[n - 1:n + n - 1]
    return grad


def compute_grad_hard(a, k_star):
    """Gradient at the single peak lag k_star. grad[i] = 2*a[k_star - i] if valid."""
    n = len(a)
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    return grad


def normalize(a, target):
    """Normalize to target sum, ensuring non-negativity."""
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-12:
        return np.ones_like(a) * (target / len(a))

def soft_max_weights(c, tau):
    """Compute softmax weights over autocorrelation values."""
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= ws
    return w

def lp_refine_small(a, M, deadline):
    """LP-based refinement for small n. Solves max sum(g) s.t. conv(a,g) <= M, g>=0."""
    n = len(a)
    if n > 512:
        return None
    
    S_t = np.sqrt(2.0 * n)
    
    c_obj = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        for m in range(j, j + n):
            A[m, j] = a[m - j]
    
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': max(1, (deadline - time.monotonic()))})
        if res.status == 0:
            g = res.x
            g_sum = g.sum()
            if g_sum > 1e-6:
                g_norm = normalize(g, S_t)
                return g_norm
    except Exception:
        pass
    return None

    # Periodic patterns
    for freq in [3, 5, 7, 11]:
        inits.append(0.5 + 0.4 * np.cos(2 * np.pi * freq * i / n))
    
    # Two-lobe structure
    inits.append(np.exp(-((i - n*0.25)/(n*0.1))**2) + 0.3 * np.exp(-((i - n*0.75)/(n*0.1))**2) + 0.05)
    
    # Sparse-like
    a = np.zeros(n)
    a[::2] = rng.random(n // 2 + 1) * 0.5 + 0.2
    inits.append(a[:n])
    
    # Golay-like (alternating blocks)
    block_size = max(1, n // 16)
    a = np.zeros(n)
    for b in range(n // block_size + 1):
        a[b*block_size:(b+1)*block_size] = rng.choice([0.1, 0.8, 0.4])
    inits.append(a[:n])
    
    return inits


def optimize_single(a0, deadline, target_sum=None):
    """Full optimization pipeline for a single starting sequence."""
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    n = len(a)
    if a.sum() < 1e-6:
        a = np.ones(n) * 0.5
    
    if target_sum is None:
        target_sum = np.sqrt(2.0 * n)
    
    a = normalize(a, target_sum)
    best_a = a.copy()
    best_score = compute_obj(a)
    
    elapsed = time.monotonic() - (deadline - (deadline - time.monotonic()))
    remaining = deadline - time.monotonic()
    
    # Phase 1: Nesterov gradient descent with soft-max (50% of time)
    phase1_deadline = time.monotonic() + remaining * 0.45
    a, score = nesterov_grad_descent(a, phase1_deadline, n, target_sum)
    if score < best_score:
        best_score = score
        best_a = a.copy()
    
    print(f"      Phase 1 (Nesterov): {best_score:.6f}")
    
    # Phase 2: Peak chasing (25% of time)
    if time.monotonic() < deadline - 5:
        phase2_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.4
        a, score = peak_chasing(best_a, phase2_deadline, target_sum, n_peaks=3)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        print(f"      Phase 2 (Peak chase): {best_score:.6f}")
    
    # Phase 3: Nesterov again with different params (15% of time)
    if time.monotonic() < deadline - 5:
        a2 = best_a + np.random.default_rng(99).normal(0, 0.02, n)
        a2 = normalize(a2, target_sum)
        phase3_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.3
        a, score = nesterov_grad_descent(a2, phase3_deadline, n, target_sum)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        print(f"      Phase 3 (Nesterov 2): {best_score:.6f}")
    
    # Phase 4: Local perturbation search (remaining time)
    if time.monotonic() < deadline - 2:
        a, score = local_perturbation_search(best_a, deadline - 1.0, target_sum)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        print(f"      Phase 4 (Perturb): {best_score:.6f}")
    
    # Phase 5: LP refinement if n is small
    if n <= 384 and time.monotonic() < deadline - 10:
        c = compute_autocorr(best_a)
        M = float(c.max()) * 1.01  # slightly relaxed
        lp_deadline = deadline - 5
        g = lp_refine_small(best_a, M, lp_deadline)
        if g is not None:
            # Try blending
            for t_blend in [0.5, 0.3, 0.7, 0.2]:
                a_blend = (1 - t_blend) * best_a + t_blend * g
                a_blend = normalize(a_blend, target_sum)
                s_blend = compute_obj(a_blend)
                if s_blend < best_score:
                    best_score = s_blend
                    best_a = a_blend.copy()
            if best_score < compute_obj(best_a):
                print(f"      Phase 5 (LP): {best_score:.6f}")
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Multi-start optimization with FFT-based gradient descent,
    peak chasing, and local perturbation search across multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Try to use previous best
    prev_seq = None
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        if gs.size >= 64 and np.all(np.isfinite(gs)):
            prev_seq = gs
            print(f"Using previous best: n={len(gs)}, score={compute_obj(gs):.6f}")
    except NameError:
        pass
    
    # Lengths to try - focus on ranges likely to give good results
    lengths = [256, 384, 512, 768]
    
    # Allocate time: more time for lengths near the previous best
    # Previous best was n=384, so give it extra attention
    time_weights = {256: 0.15, 384: 0.35, 512: 0.30, 768: 0.20}
    
    n_inits_per_length = 3
    
    for n_len in lengths:
        if time.monotonic() > deadline - 60:
            break
        
        time_alloc = (deadline - time.monotonic()) * time_weights.get(n_len, 0.2)
        time_per_init = time_alloc / n_inits_per_length
        
        if time_per_init < 15:
            n_inits_per_length = max(1, int(time_alloc / 20))
            time_per_init = time_alloc / max(1, n_inits_per_length)
        
        print(f"\n  === n={n_len}, {n_inits_per_length} inits, {time_per_init:.0f}s each ===")
        
        inits = generate_initializations(n_len, rng, prev_seq)
        
        # Shrink if too many
        if len(inits) > n_inits_per_length * 2:
            # Pick diverse ones
            inits = inits[:n_inits_per_length * 2]
        
        for init_idx in range(min(n_inits_per_length, len(inits))):
            if time.monotonic() > deadline - 30:
                break
            
            a0 = inits[init_idx % len(inits)]
            init_deadline = min(time.monotonic() + time_per_init, deadline - 20)
            
            if init_deadline - time.monotonic() < 10:
                break
            
            print(f"    Init {init_idx} (n={n_len}):")
            a_opt, score = optimize_single(a0, init_deadline)
            
            if score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                print(f"    *** NEW BEST: {best_score:.6f} (n={n_len}) ***")
    
    # Final polish on the best sequence
    if best_seq is not None and time.monotonic() < deadline - 10:
        print(f"\n  Final polish (score={best_score:.6f}, n={len(best_seq)})...")
        remaining = deadline - time.monotonic() - 5
        n = len(best_seq)
        target_sum = np.sqrt(2.0 * n)
        
        # Quick Nesterov pass
        a_opt, score = nesterov_grad_descent(best_seq, time.monotonic() + remaining * 0.5, n, target_sum)
        if score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"    Polish 1: {best_score:.6f}")
        
        # Peak chasing
        if time.monotonic() < deadline - 5:
            a_opt, score = peak_chasing(best_seq, time.monotonic() + (deadline - time.monotonic() - 3) * 0.5, target_sum)
            if score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                print(f"    Polish 2 (peak): {best_score:.6f}")
        
        # Perturbation
        if time.monotonic() < deadline - 3:
            a_opt, score = local_perturbation_search(best_seq, deadline - 2, target_sum, n_trials=100)
            if score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                print(f"    Polish 3 (perturb): {best_score:.6f}")
    
    # Validate and return
    if best_seq is not None:
        # Ensure all values are valid
        best_seq = np.clip(best_seq, 0.0, 1000.0)
        final_score = compute_obj(best_seq)
        
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"\n  Final: n={len(best_seq)}, internal={final_score:.6f}, eval={eval_score:.6f}")
        except Exception as e:
            print(f"\n  Final: n={len(best_seq)}, internal={final_score:.6f}, eval error: {e}")
            eval_score = final_score
        
        return [float(x) for x in best_seq]
    
    # Fallback
    print("Falling back to uniform sequence.")
    return [1.0] * 256
