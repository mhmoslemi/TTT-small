```python
import numpy as np
import time

def fft_convolve(a, b=None):
    """Compute linear convolution using FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def score(seq):
    """Fast score computation."""
    a = np.asarray(seq, dtype=np.float64)
    a = np.clip(a, 0, 1000)
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)

def subgradient_step(a, lr):
    """One subgradient descent step."""
    n = len(a)
    S = np.sum(a)
    c = fft_convolve(a, a)
    max_c = np.max(c)
    
    # Find active set (positions where c is close to max)
    thresh = max_c * (1 - 1e-8)
    active = np.where(c >= thresh)[0]
    
    # Compute subgradient
    grad = np.zeros(n)
    S_active = len(active)
    for k in active:
        # g_j(k) = 2*a_j if k==2j, 2*a_{k-j} if 0<=k-j<n and k!=2j, 0 otherwise
        # Vectorized: for this k, the subgradient is 2*a[(k-j) mod n] for valid j
        # More precisely: grad[j] += 2*a[k-j] if 0<=k-j<n (and if k==2j, it's 2*a[j] which is the same as 2*a[k-j])
        # So actually: grad[j] += 2*a[k-j] for all j where 0 <= k-j < n
        # i.e., j in range[k-n+1, k+1] intersect [0, n-1]
        
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            # grad[j] += 2 * a[k-j] for j in [j_start, j_end]
            # k-j ranges from k-j_end to k-j_start
            # = from k - min(n-1, k) to k - max(0, k-n+1)
            # If k < n-1: j in [0, k], k-j in [k, 0], so a[k-j] = a[k], a[k-1], ..., a[0]
            # If k >= n-1: j in [k-n+1, n-1], k-j in [n-1, k-n+1], so a[k-j] = a[n-1], ..., a[k-n+1]
            
            indices = np.arange(j_start, j_end + 1)
            a_indices = k - indices
            grad[j_start:j_end+1] += 2.0 * a[a_indices]
    
    grad /= S_active
    
    # Update with projection
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0)
    s_new = np.sum(a_new)
    if s_new < 1e-10:
        return a.copy()
    a_new = a_new * (S / s_new)
    return a_new
```

```python
"""
Optimized step function search using subgradient descent, coordinate descent,
and local search. Targets the minimization of 2n*max autocorrelation / (sum)^2.
"""

import numpy as np
import time

def fft_convolve(a, b=None):
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def fast_score(a):
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)

def subgradient_descent(a0, time_limit, rng, verbose=False):
    """Subgradient descent with adaptive step size."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    S = np.sum(a)
    a = np.maximum(a, 0)
    if np.sum(a) < 0.01:
        a = np.ones(n) / n
        S = 1.0
    a = a * (S / np.sum(a))  # normalize to keep sum = S
    
    current_score = fast_score(a)
    best_a = a.copy()
    best_score = current_score
    
    t0 = time.monotonic()
    lr = 0.1  # initial learning rate
    patience = 0
    max_patience = 500
    
    while time.monotonic() - t0 < time_limit:
        # Compute autocorrelation
        c = fft_convolve(a, a)
        c_max = np.max(c)
        mean_c = np.sum(c) / (2 * n - 1)
        
        # Find active set
        thresh = c_max - 0.01 * (c_max - mean_c) if c_max > mean_c else c_max
        active = np.where(c >= thresh)[0]
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in active:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                indices = k - np.arange(j_start, j_end + 1)
                grad[j_start:j_end + 1] += 2.0 * a[indices]
        grad /= len(active)
        
        # Update
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            break
        a_new = a_new * (S / s_new)
        
        new_score = fast_score(a_new)
        
        if new_score < current_score:
            a = a_new
            current_score = new_score
            patience = 0
            lr = min(lr * 1.02, 0.5)
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()
        else:
            patience += 1
            lr *= 0.95
            if patience > max_patience:
                # Restart with perturbation
                a = best_a * np.exp(rng.normal(0, 0.05, n))
                a = np.maximum(a, 0)
                a = a * (S / np.sum(a))
                current_score = fast_score(a)
                lr = 0.1
                patience = 0
                max_patience = min(max_patience * 1.2, 2000)
        
        if verbose and patience % 1000 == 0:
            print(f"    sg iter, score={current_score:.6f}, best={best_score:.6f}, lr={lr:.4f}")
    
    return best_a, best_score

def coordinate_descent(a0, time_limit, rng, verbose=False):
    """Coordinate descent with exact 1D minimization."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 0.01:
        a = np.ones(n) / n
        S = 1.0
    a = a * (S / np.sum(a))
    
    current_score = fast_score(a)
    best_a = a.copy()
    best_score = current_score
    
    t0 = time.monotonic()
    sweep = 0
    
    while time.monotonic() - t0 < time_limit:
        aa = fft_convolve(a, a)
        improved = False
        
        for j in range(n):
            a_j = a[j]
            S_j = S - a_j  # sum of all other elements
            
            # For this j, (a*a)_k as function of x (new value of a[j]):
            # k in [j, j+n-1] (capped at 2n-2):
            #   if k == 2j: x^2 + (aa[k] - a_j^2)
            #   else: 2*a[k-j]*x + (aa[k] - 2*a_j*a[k-j])
            # k outside: aa[k] (constant)
            
            # Find max of constants (outside range)
            k_lo = max(0, j)
            k_hi = min(2 * n - 2, j + n - 1)
            
            if k_lo > 0:
                max_outside_left = np.max(aa[0:k_lo])
            else:
                max_outside_left = 0
            
            if k_hi < 2 * n - 2:
                max_outside_right = np.max(aa[k_hi + 1:])
            else:
                max_outside_right = 0
            
            max_outside = max(max_outside_left, max_outside_right)
            if k_lo == 0 and k_hi == 2 * n - 2:
                max_outside = 0  # no outside terms
            
            # Candidates for optimal x
            candidates = [0.0, a_j]
            
            # Quadratic term balance: x = C_2j / S_j if 2j is in range
            if 2 * j >= k_lo and 2 * j <= k_hi:
                C_2j = aa[2 * j] - a_j * a_j
                if S_j > 0 and C_2j > 0:
                    candidates.append(C_2j / S_j)
            
            # Linear term candidates (top few)
            # x_k = S + a_j - aa[k]/a[k-j] for k in range, k != 2j
            lin_candidates = []
            for k in range(k_lo, k_hi + 1):
                if k == 2 * j:
                    continue
                idx = k - j
                if 0 <= idx < n and a[idx] > 1e-10:
                    C_k = aa[k] - 2 * a_j * a[idx]
                    x_k = S + a_j - aa[k] / a[idx]
                    if x_k > 0:
                        lin_candidates.append(x_k)
            
            # Only keep top few linear candidates (by proximity to current a_j)
            if lin_candidates:
                lin_candidates = sorted(lin_candidates, key=lambda x: abs(x - a_j))
                candidates.extend(lin_candidates[:10])
            
            # Add some interpolation points
            candidates.append(max(0, a_j * 0.5))
            candidates.append(a_j * 1.5)
            
            # Evaluate at each candidate
            best_x = a_j
            best_s = current_score
            
            for x in candidates:
                x = max(0, x)
                # Compute max autocorrelation
                g_max = max_outside
                for k in range(k_lo, k_hi + 1):
                    if k == 2 * j:
                        g = x * x + (aa[k] - a_j * a_j)
                    else:
                        idx = k - j
                        if 0 <= idx < n:
                            g = 2 * a[idx] * x + (aa[k] - 2 * a_j * a[idx])
                        else:
                            g = aa[k]
                    if g > g_max:
                        g_max = g
                
                denom = (S_j + x) ** 2
                if denom < 1e-10:
                    continue
                s_val = 2 * n * g_max / denom
                if s_val < best_s:
                    best_s = s_val
                    best_x = x
            
            if best_x != a_j:
                a[j] = best_x
                improved = True
        
        # Update score
        new_score = fast_score(a)
        if new_score < best_score:
            best_score = new_score
            best_a = a.copy()
        
        current_score = new_score
        sweep += 1
        
        if verbose and sweep % 10 == 0:
            print(f"    CD sweep {sweep}, score={current_score:.6f}, best={best_score:.6f}")
        
        # Check for convergence
        if sweep > 50 and best_score == current_score:
            break
    
    return best_a, best_score
```

"""
Step function optimization: minimize 2n*max(autocorrelation)/(sum)^2
using subgradient descent, coordinate descent, and targeted local search.
"""

import numpy as np
import time


def fft_convolve(a, b=None):
    """Fast linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def fast_score(a):
    """Compute the objective score."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0, 1000)
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (s * s))


def normalize_nonneg(a, target_sum=None):
    """Clip to non-negative and normalize."""
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-15:
        return a
    if target_sum is None:
        target_sum = s
    return a * (target_sum / s)


def subgradient_descent(a0, time_limit, rng, verbose=False):
    """Subgradient descent with adaptive step size."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 0.01:
        a = np.ones(n)
        S = n
    a = a / S
    
    current_score = fast_score(a)
    best_a = a.copy()
    best_score = current_score
    
    t0 = time.monotonic()
    lr = 0.03
    stall = 0
    max_patience = 300
    iter_count = 0
    
    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(a, a)
        c_max = np.max(c)
        
        # Active set
        thresh = c_max * 0.9995
        active = np.where(c >= thresh)[0]
        
        # Subgradient
        grad = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                grad[js:je + 1] += 2.0 * a[k - np.arange(js, je + 1)]
        grad /= len(active)
        
        # Update
        a_new = a - lr * grad
        a_new = normalize_nonneg(a_new, target_sum=1.0)
        
        ns = fast_score(a_new)
        iter_count += 1
        
        if ns < current_score - 1e-12:
            a = a_new
            current_score = ns
            stall = 0
            lr = min(lr * 1.005, 0.2)
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()
        else:
            stall += 1
            lr *= 0.98
            if stall >= max_patience:
                # Perturb to escape
                noise_scale = min(0.15, 0.05 * (stall // max_patience + 1))
                a = best_a * np.exp(rng.normal(0, noise_scale, n))
                a = normalize_nonneg(a, target_sum=1.0)
                current_score = fast_score(a)
                lr = 0.03
                stall = 0
                max_patience = min(max_patience * 1.5, 1000)
        
        if verbose and iter_count % 5000 == 0:
            print(f"    SG: iter={iter_count}, score={current_score:.8f}, best={best_score:.8f}, lr={lr:.5f}")
    
    return best_a, best_score


def coordinate_descent(a0, time_limit, rng, verbose=False):
    """Coordinate descent with vectorized candidate evaluation."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = np.maximum(a, 0)
    S = np.sum(a)
    if S < 0.01:
        a = np.ones(n)
        S = n
    a = a / S  # normalize to sum=1
    
    current_score = fast_score(a)
    best_a = a.copy()
    best_score = current_score
    
    t0 = time.monotonic()
    sweep = 0
    no_improve_sweeps = 0
    
    while time.monotonic() - t0 < time_limit:
        aa = fft_convolve(a, a)
        aa_max = np.max(aa)
        
        improved = False
        total_improvement = 0
        
        for j in range(n):
            a_j = a[j]
            S_j = 1.0 - a_j  # sum of others (since total sum = 1)
            
            if S_j < 1e-10:
                continue
            
            # k range where a_j contributes
            k_lo = j
            k_hi = min(2 * n - 2, j + n - 1)
            
            # Max outside range
            outside_vals = []
            if k_lo > 0:
                outside_vals.append(np.max(aa[:k_lo]))
            if k_hi < 2 * n - 2:
                outside_vals.append(np.max(aa[k_hi + 1:]))
            max_out = max(outside_vals) if outside_vals else 0.0
            
            # Build candidates
            candidates = [a_j, 0.0]
            
            # Quadratic balance point
            if 2 * j >= k_lo and 2 * j <= k_hi:
                C_2j = aa[2 * j] - a_j * a_j
                if C_2j > 0:
                    candidates.append(C_2j / S_j)
            
            # Linear candidates from top peaks
            k_indices = np.arange(k_lo, k_hi + 1)
            # Exclude 2j from linear terms
            mask = k_indices != 2 * j
            k_lin = k_indices[mask]
            
            if len(k_lin) > 0:
                a_idx = k_lin - j  # indices into a
                # Valid: 0 <= a_idx < n (should be by construction)
                valid = (a_idx >= 0) & (a_idx < n)
                k_lin = k_lin[valid]
                a_idx = a_idx[valid]
                
                if len(k_lin) > 0:
                    a_vals = a[a_idx]
                    c_vals = aa[k_lin] - 2 * a_j * a_vals
                    # x_k = (a_val * S_j - c_val) / a_val = S_j - c_val/a_val
                    # But the condition for this to be the optimum is different
                    # x_k = 1 + a_j - aa[k]/a[k-j] (since S = 1)
                    valid_a = a_vals > 1e-10
                    if np.any(valid_a):
                        x_candidates = (1.0 + a_j - aa[k_lin[valid_a]] / a_vals[valid_a])
                        x_candidates = x_candidates[x_candidates > 0]
                        # Keep top 5 closest to a_j
                        if len(x_candidates) > 5:
                            x_candidates = x_candidates[np.argsort(np.abs(x_candidates - a_j))[:5]]
                        candidates.extend(x_candidates.tolist())
            
            # Add a few heuristic points
            candidates.append(a_j * 0.5)
            candidates.append(a_j * 2.0)
            
            # Clip candidates to reasonable range
            candidates = [max(0, min(c, 1.0)) for c in candidates]
            
            # Evaluate candidates vectorized
            cand_arr = np.array(candidates)
            
            # Compute g_max for each candidate
            # Linear terms
            g_max_arr = np.full(len(cand_arr), max_out)
            
            if len(k_lin) > 0:
                slopes = 2.0 * a[a_idx]
                intercepts = aa[k_lin] - 2 * a_j * a[a_idx]
                # values[i, k] = slopes[k] * cand[i] + intercepts[k]
                vals = np.outer(cand_arr, slopes) + intercepts[None, :]
                g_max_arr = np.maximum(g_max_arr, vals.max(axis=1))
            
            # Quadratic term
            if 2 * j >= k_lo and 2 * j <= k_hi:
                C_2j = aa[2 * j] - a_j * a_j
                quad_vals = cand_arr ** 2 + C_2j
                g_max_arr = np.maximum(g_max_arr, quad_vals)
            
            # Compute scores

            valid_denom = denom > 1e-10
            scores_arr = np.full(len(cand_arr), np.inf)
            scores_arr[valid_denom] = 2.0 * n * g_max_arr[valid_denom] / denom[valid_denom]
            
            best_idx = np.argmin(scores_arr)
            
            if scores_arr[best_idx] < current_score - 1e-12:
                a[j] = cand_arr[best_idx]
                current_score = scores_arr[best_idx]
                improved = True
                total_improvement += 1
        
        if improved:
            no_improve_sweeps = 0
        else:
            no_improve_sweeps += 1
        
        if fast_score(a) < best_score:
            best_score = fast_score(a)
            best_a = a.copy()
        
        sweep += 1
        
        if verbose and sweep % 50 == 0:
            print(f"    CD: sweep={sweep}, score={fast_score(a):.8f}, best={best_score:.8f}")
        
        if no_improve_sweeps > 10:
            # Try a perturbation
            a = best_a * np.exp(rng.normal(0, 0.03, n))
            a = normalize_nonneg(a, target_sum=1.0)
            current_score = fast_score(a)
            no_improve_sweeps = 0
    
    return best_a, best_score


def local_search(a0, time_limit, rng, verbose=False):

    n = len(a0)
    a = a0.astype(np.float64).copy()

    current_score = fast_score(a)

    best_score = current_score
    t0 = time.monotonic()
    temp = 0.002

        peak_idx = int(np.argmax(c))

            # Contributors to c[peak_idx]

            local_indices = np.arange(js, je + 1)

                top_global = local_indices[top_local]
                candidate[top_global] *= np.exp(rng.normal(-0.03, 0.01, len(top_global)))
                
                # Increase non-peak contributors

        if csum < 0.01:
