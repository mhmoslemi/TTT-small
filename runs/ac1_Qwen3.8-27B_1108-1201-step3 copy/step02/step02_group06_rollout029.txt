"""
Multi-phase optimization for minimizing normalized peak autocorrelation of
non-negative sequences. Combines:
1. Gradient descent with log-sum-exp smoothing (primary method)
2. LP-based refinement (for escaping local minima)
3. Targeted simulated annealing (final polishing)
4. Coordinate descent (fine-tuning)
Explores multiple sequence lengths and initializations.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def local_score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (sum_a ** 2)


def grad_lse(a, tau):
    """Compute LSE-smoothed peak and gradient w.r.t. a."""
    n = len(a)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    c_shift = (c - c_max) / tau
    exp_c = np.exp(c_shift)
    sum_exp = np.sum(exp_c)
    lse_val = c_max + tau * np.log(sum_exp)
    w = exp_c / sum_exp
    # Gradient
    full_conv = fft_convolve(w[::-1], a)
    grad_a = 2.0 * full_conv[n-1:2*n-1][::-1]
    return lse_val, grad_a, c


def gradient_optimize(a_init, time_limit, tau_start=0.5, tau_end=0.05, lr=1.0, 
                       patience_limit=300, verbose=False):
    """Adam gradient descent with LSE smoothing and adaptive restarts."""
    n = len(a_init)
    a = np.clip(a_init, 1e-10, None)
    a = a / np.sum(a)
    
    theta = np.log(a)
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    step = 0
    best_theta = theta.copy()
    best_score = local_score(a)
    t0 = time.monotonic()
    patience = 0
    rng = np.random.default_rng(999)
    
    while time.monotonic() - t0 < time_limit:
        elapsed_frac = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_start * (tau_end / tau_start) ** (elapsed_frac ** 0.5)
        
        theta_shift = theta - np.max(theta)
        a = np.exp(theta_shift) / np.sum(np.exp(theta_shift))
        
        lse_val, grad_a, _ = grad_lse(a, tau)
        
        inner = np.dot(grad_a, a)
        grad_theta = a * (grad_a - inner)
        
        step += 1
        m = beta1 * m + (1-beta1) * grad_theta
        v = beta2 * v + (1-beta2) * grad_theta**2
        m_hat = m / (1 - beta1**step)
        v_hat = v / (1 - beta2**step)
        lr_t = lr / np.sqrt(1 + 0.05 * step)
        theta = theta - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        
        theta_shift = theta - np.max(theta)
        a_new = np.exp(theta_shift) / np.sum(np.exp(theta_shift))
        score = local_score(a_new)
        
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_theta = theta.copy()
            patience = 0
        else:
            patience += 1
        
        if patience >= patience_limit:
            theta = best_theta + 0.05 * rng.standard_normal(n)
            patience = 0
            # Also reset Adam state
            m = np.zeros(n)
            v = np.zeros(n)
        
        if verbose and step % 500 == 0:
            print(f"    Grad step={step}, tau={tau:.4f}, score={score:.6f}, best={best_score:.6f}")
    
    theta_shift = best_theta - np.max(best_theta)
    a_best = np.exp(theta_shift) / np.sum(np.exp(theta_shift))
    return a_best, best_score


def build_constraint_A(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_optimize(f_init, time_limit, t_init=0.3, verbose=False):
    """LP-based iterative optimization."""
    f = f_init.astype(np.float64).copy()
    n = len(f)
    sum_f = np.sum(f)
    best_f = f.copy()
    best_score = local_score(f)
    t0 = time.monotonic()
    t = t_init
    stagnant = 0
    iteration = 0
    rng_local = np.random.default_rng(777)
    
    A = build_constraint_A(f)
    
    while time.monotonic() - t0 < time_limit:
        conv = fft_convolve(f, f)
        M = np.max(conv)
        
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n
        result = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': 30})
        if not result.success or result.x is None:
            break
        
        g = result.x
        sum_g = np.sum(g)
        if sum_g < 1e-10:
            break
        
        g_rescaled = g * (sum_f / sum_g)
        f_new = (1.0 - t) * f + t * g_rescaled
        f_new = np.clip(f_new, 0.0, None)
        sum_new = np.sum(f_new)
        if sum_new < 1e-10:
            t *= 0.5
            continue
        f_new = f_new * (sum_f / sum_new)
        
        score_new = local_score(f_new)
        iteration += 1
        
        if score_new < best_score - 1e-12:
            f = f_new
            best_score = score_new
            best_f = f.copy()
            stagnant = 0
        else:
            f = f_new
            stagnant += 1
            if stagnant > 30:
                t *= 0.7
                stagnant = 0
                # Perturbation
                f = f + 0.01 * rng_local.standard_normal(n)
                f = np.clip(f, 0.0, None)
                f = f * (sum_f / np.sum(f))
                # Rebuild A
                A = build_constraint_A(f)
                stagnant = 0
        
        if verbose and iteration % 20 == 0:
            print(f"    LP iter={iteration}, t={t:.3f}, score={local_score(f):.6f}, best={best_score:.6f}")
    
    return best_f, best_score
