"""
Multi-method optimization for minimizing peak convolution of non-negative sequences.
Combines subgradient descent, LP-based direction finding, and local peak-shaving
search with multiple initializations and sequence lengths.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = rfft(a, size)
    fb = rfft(b, size)
    return irfft(fa * fb, size)[:n + m - 1]


def score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def subgradient_of_max_conv(a, top_k=3):
    """Compute subgradient of max(conv(a,a)) with respect to a."""
    n = len(a)
    c = fft_convolve(a, a)
    # Get top_k positions
    if top_k >= len(c):
        top_idx = np.arange(len(c))
        weights = np.ones(len(c)) / len(c)
    else:
        # Get indices of top_k values
        top_idx = np.argsort(c)[-top_k:]
        # Use inverse as weights (or just uniform)
        weights = np.ones(top_k) / top_k
    
    g = np.zeros(n)
    for k_i, w in zip(top_idx, weights):
        ki = int(k_i)
        # d(conv_ki)/da[j] = 2*a[ki-j] if 0 <= ki-j < n
        lo = max(0, ki - (n - 1))
        hi = min(n - 1, ki)
        for j in range(lo, hi + 1):
            g[j] += w * 2.0 * a[ki - j]
    return g, c


def subgradient_descent(f_init, time_limit, lr=0.01, momentum=0.9, top_k=5, verbose=True):
    """Subgradient descent with momentum to minimize the score."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    vel = np.zeros(n)
    it = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 1.0:
            break
        
        g, c = subgradient_of_max_conv(f, top_k=top_k)
        S = float(np.sum(f))
        max_c = float(np.max(c))
        
        # Full subgradient of score = 2n*max_c/S^2
        # d(score)/da[j] = 2n * (g[j]*S - 2*max_c) / S^3
        full_g = 2.0 * n * (g * S - 2.0 * max_c) / (S ** 3)
        
        vel = momentum * vel + (1.0 - momentum) * full_g
        f = f - lr * vel
        f = np.clip(f, 0.0, 1000.0)
        
        it += 1
        if it % 100 == 0:
            s = score_fast(f)
            if s < best_score:
                best_score = s
                best_f = f.copy()
                stall = 0
            else:
                stall += 1
            if verbose:
                print(f"    [SGD {time.monotonic()-t0:5.1f}s] it={it} score={s:.6f} best={best_score:.6f}")
        
        if stall >= 10:
            # Adaptive step size: reduce if stalling
            lr *= 0.9
            stall = 0
        
        if stall >= 50:
            # Perturb to escape local minimum
            rng = np.random.default_rng(int(time.monotonic() * 1000))
            f += 0.01 * rng.standard_normal(n)
            f = np.clip(f, 0.01, 1000.0)
            lr = 0.01
            stall = 0
    
    s = score_fast(f)
    if s < best_score:
        best_score = s
        best_f = f.copy()
    return best_f, best_score


def build_A_matrix(f):
    """Build to
