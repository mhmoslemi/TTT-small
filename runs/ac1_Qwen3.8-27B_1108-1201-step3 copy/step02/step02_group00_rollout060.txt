```python
                    # For affected k != 2i: b_k(x) = b_k - 2*x_old*a_{k-i} + 2*x*a_{k-i}
                    # For k = 2i (if in range): b_k(x) = b_k - x_old^2 + x^2
                    # For unaffected k: b_k(x) = b_k
                    
                    # Unaffected max
                    if k_start > 0:
                        m_unaff_left = b[:k_start].max() if k_start > 0 else 0
                    else:
                        m_unaff_left = 0
                    
                    if k_end < 2*n-2:
                        m_unaff_right = b[k_end+1:].max() if k_end + 1 < 2*n else 0
                    else:
                        m_unaff_right = 0
                    
                    m_unaff = max(m_unaff_left, m_unaff_right, 0)
                    
                    # Affected indices
                    k_aff = np.arange(k_start, k_end + 1)
                    
                    # b_k at x_old (should equal b[k_aff])
                    # b_k(x) for k in k_aff:
                    #   if k == 2i: b_k = b[2i] - x_old^2 + x^2
                    #   else: b_k = b[k] - 2*x_old*a[k-i] + 2*x*a[k-i]
                    
                    a_ki = a[k_aff - i]  # a_{k-i} for each affected k
                    b_aff = b[k_aff].copy()
                    
                    # Remove x_old contribution
                    b_base = b_aff - 2 * x_old * a_ki  # this is correct for k != 2i
                    # For k = 2i, the contribution is x_old^2, not 2*x_old*a_{2i-i} = 2*x_old*a_i = 2*x_old^2
                    # So we need to fix: b_base[2i - k_start] should be b[2i] - x_old^2, not b[2i] - 2*x_old*a_i
                    if 2 * i >= k_start and 2 * i <= k_end:
                        idx_2i = 2 * i - k_start
                        b_base[idx_2i] = b[2*i] - x_old * x_old
                    
                    # Now b_k(x) = b_base_k + 2*x*a_{k-i} for k != 2i
                    #                        b_k(x) = b_base_k + x^2 for k = 2i
                    b_at_x = b_base.copy()
                    b_at_x += 2 * x * a_ki
                    if 2 * i >= k_start and 2 * i <= k_end:
                        idx_2i = 2 * i - k_start
                        b_at_x[idx_2i] = b_base[idx_2i] + x * x
                    
                    m_aff = b_at_x.max()
                    mb = max(m_unaff, m_aff)
                
                if S_other + x < 1e-15:
                    return float('inf')
                return 2.0 * n * mb / ((S_other + x) ** 2)
            
            # Find min of Q over candidates
            Q_vals = np.array([eval_Q(x) for x in x_candidates])
            min_idx = np.argmin(Q_vals)
            x_rough = x_candidates[min_idx]
            
            # Refine with golden section search near x_rough
            if x_rough > 0 and x_rough < x_max:
                # Narrow the range
                left = max(0.0, x_candidates[max(0, min_idx-1)])
                right = x_candidates[min(min_idx+1, len(x_candidates)-1)]
                if right - left < 1e-10:
                    x_opt = x_rough
                else:
                    # Golden section
                    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
                    for _ in range(30):
                        m1 = right - gr * (right - left)
                        m2 = left + gr * (right - left)
                        if eval_Q(m1) < eval_Q(m2):
                            right = m2
                        else:
                            left = m1
                        if right - left < 1e-10 * max(1, x_rough):
                            break
                    x_opt = (left + right) / 2
            else:
                x_opt = x_rough
            
            x_new = max(0.0, x_opt)
            
            if abs(x_new - x_old) > 1e-12:
                a[i] = x_new
                _update_b(b, a, i, x_old, x_new)
                S = S_other + x_new
        
        # End of sweep
        max_b = b.max()
        cur_obj = 2.0 * n * max_b / (S * S)
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve_sweeps = 0
        else:
            no_improve_sweeps += 1
        
        if verbose and sweep % 10 == 0:
            print(f"  Sweep {sweep}: obj={cur_obj:.8f}, best={best_obj:.8f}, S={S:.4f}")
    
    return best_a, best_obj


def _update_b(b, a, i, x_old, x_new):
    """Update b incrementally when a[i] changes from x_old to x_new."""
    n = len(a)
    dx = x_new - x_old
    if abs(dx) < 1e-15:
        return
    
    # Affected k: i <= k <= i + n - 1 (and k < 2n-1)
    k_start = i
    k_end = min(i + n - 1, 2 * n - 2)
    
    for k in range(k_start, k_end + 1):
        if k == 2 * i:
            b[k] += x_new * x_new - x_old * x_old
        else:
            b[k] += 2 * dx * a[k - i]
```

Wait, there's an issue with the `_update_b` function. When `k == 2i`, the old contribution was `x_old^2` and the new is `x_new^2`. When `k != 2i` but `k` is affected, the old contribution was `2*x_old*a[k-i]` and the new is `2*x_new*a[k-i]`. So the change is `2*dx*a[k-i]`.

But wait, I need to be careful. The convolution `b_k = sum_j a_j * a_{k-j}`. When we change `a_i` from `x_old` to `x_new`:
- Terms where `j = i`: change from `x_old * a_{k-i}` to `x_new * a_{k-i}`, i.e., `dx * a_{k-i}`
- Terms where `k-j = i` (i.e., `j = k-i`): change from `a_{k-i} * x_old` to `a_{k-i} * x_new`, i.e., `dx * a_{k-i}`

If `j = i` and `j = k-i` are the same (i.e., `k = 2i`), then there's only ONE term that changes, from `x_old^2` to `x_new^2`, i.e., change is `x_new^2 - x_old^2 = (x_new - x_old)(x_new + x_old)`.

If `k ≠ 2i`, there are two (possibly different) terms, each changing by `dx * a_{k-i}`, so total change is `2*dx*a_{k-i}`.

This matches what I have. Good.

But there's a subtle issue: when `k = 2i`, the term `a_j * a_{k-j}` with `j = i` is `a_i * a_i = x^2`. There's no separate "j = k-i" term because `k-i = i = j`. So the change is `x_new^2 - x_old^2`.

For `k ≠ 2i` but `0 <= k-i < n`: the term `j=i` gives `a_i * a_{k-i}` and the term `j=k-i` gives `a_{k-i} * a_i`. These are two separate terms (since `j=i ≠ j=k-i`), each contributing `a_i * a_{k-i}`, so total contribution from `a_i` is `2 * a_i * a_{k-i}`. After change: `2 * x_new * a_{k-i}`. Change: `2 * dx * a_{k-i}`. ✓

For `k` where `k-i < 0` or `k-i >= n`: no terms involve `a_i`, so no change. ✓

Great, the update formula is correct.

Now, there's another issue with my `eval_Q` function. When I compute `b_base`, I'm doing:
```
b_base = b_aff - 2 * x_old * a_ki
```
But for `k = 2i`, the contribution is `x_old^2`, not `2*x_old*a_i = 2*x_old^2`. So I need to handle that case separately, which I do with the `if 2*i >= k_start` check.

Wait, but `a_ki` for `k = 2i` is `a[2i - i] = a[i] = x_old`. So `2 * x_old * a_ki = 2 * x_old^2`. But the actual contribution is `x_old^2`. So `b_base[2i] = b[2i] - 2*x_old^2`, but it should be `b[2i] - x_old^2`. That's why I have the correction:
```
b_base[idx_2i] = b[2*i] - x_old * x_old
```
This is correct.

Now let me think about performance. For each coordinate update, I'm calling `eval_Q` about 50-80 times (20 uniform + a few special + 30 golden section). Each `eval_Q` call takes `O(n)` time. So one coordinate update is `O(n * 80)`. A full sweep is `O(n^2 * 80)`. For `n=256`: `256 * 256 * 80 ≈ 5.2M` operations per sweep. In Python, this might take a few seconds per sweep. We could probably do 10-50 sweeps in the time budget.

Actually, the `eval_Q` function in Python with numpy should be fairly fast for `n=256`. Let me estimate: each call involves a few numpy operations on arrays of size ~256. That's probably 10-50 microseconds. So 80 calls per coordinate = ~4ms. 256 coordinates = ~1 second per sweep. That's good - we can do many sweeps.

Let me also think about whether the coordinate descent will converge to a good solution. The issue is that the objective is non-convex (it's a ratio), so coordinate descent might get stuck. But in practice, it often works well for this type of problem.

To improve robustness, I'll:
1. Try multiple initializations
2. Use random coordinate ordering
3. Periodically perturb the solution
4. Try different lengths

Let me also add a **simulated annealing** component: occasionally accept a worse solution to escape local minima.

Actually, let me think about a potentially better approach: **alternating between coordinate descent and a "rebalancing" step**.

The rebalancing step would be: given the current `a`, compute `b = conv(a,a)`, identify the peaks of `b`, and redistribute mass from coordinates that contribute most to the peaks to coordinates that don't.

This is similar to the gradient approach but more targeted.

Let me now write the complete solution:

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using coordinate descent with:
- Exact 1D subproblem solving via golden section search
- Incremental FFT-free convolution updates
- Multiple initializations and lengths
- Simulated annealing for escaping local minima
- Symmetric sequence exploitation
"""

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def _update_b(b, a, i, x_old, x_new):
    """Update b incrementally when a[i] changes from x_old to x_new."""
    n = len(a)
    dx = x_new - x_old
    if abs(dx) < 1e-15:
        return
    k_start = i
    k_end = min(i + n - 1, 2 * n - 2)
    for k in range(k_start, k_end + 1):
        ki = k - i
        if ki < 0 or ki >= n:
            continue
        if k == 2 * i:
            b[k] += x_new * x_new - x_old * x_old
        else:
            b[k] += 2.0 * dx * a[ki]

def _eval_q(x, S_other, b, a, i, x_old, n):
    """Evaluate 2*n*max_b(x)/(S_other+x)^2 without modifying b."""
    if S_other + x < 1e-15:
        return float('inf')
    
    k_start = i
    k_end = min(i + n - 1, 2 * n - 2)
    
    if k_start > k_end:
        mb = b.max()
    else:
        # Unaffected max
        m_unaff = 0.0
        if k_start > 0:
            m_unaff = max(m_unaff, float(b[:k_start].max()))
        if k_end + 1 < 2 * n - 1:
            m_unaff = max(m_unaff, float(b[k_end+1:].max()))
        
        # Affected
        k_aff = np.arange(k_start, k_end + 1)
        a_ki = a[k_aff - i]
        b_base = b[k_aff].astype(np.float64) - 2.0 * x_old * a_ki
        
        # Fix k=2i case
        if 2 * i >= k_start and 2 * i <= k_end:
            idx_2i = 2 * i - k_start
            b_base[idx_2i] = b[2 * i] - x_old * x_old
        
        b_at_x = b_base + 2.0 * x * a_ki
        if 2 * i >= k_start and 2 * i <= k_end:
            idx_2i = 2 * i - k_start
            b_at_x[idx_2i] = b_base[idx_2i] + x * x
        
        m_aff = float(b_at_x.max())
        mb = max(m_unaff, m_aff)
    
    return 2.0 * n * mb / ((S_other + x) ** 2)

def coordinate_descent(a_init, time_limit, seed=42, anneal=False):
    """
    Coordinate descent to minimize 2*n*max_b/(sum_a)^2.
    Returns (best_a, best_objective).
    """
    np.random.seed(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-15:
        a = np.ones(len(a), dtype=np.float64) / len(a)
        S = 1.0
    
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    best_obj = 2.0 * n * max_b / (S * S)
    best_a = a.copy()
    
    t0 = time.monotonic()
    sweep = 0
    no_improve = 0
    max_no_improve = 50
    temp = 0.01 if anneal else 0.0
    
    while time.monotonic() - t0 < time_limit:
        sweep += 1
        obj_start = 2.0 * n * b.max() / (S * S)
        improved = False
        
        order = np.random.permutation(n)
        
        for i in order:
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            
            x_old = a[i]
            S_other = S - x_old
            
            # Find optimal x
            if x_old < 1e-10:
                # Try small positive values
                x_candidates = np.array([0.0, S_other/n*0.01, S_other/n*0.05, S_other/n*0.1, S_other/n*0.5])
            else:
                x_max = x_old * 20
                x_min = 0.0
                # Logarithmic + linear sampling
                if x_min < x_max * 0.01:
                    log_pts = np.exp(np.linspace(np.log(max(x_min, 1e-10)), np.log(x_max), 15))
                else:
                    log_pts = np.array([x_min])
                lin_pts = np.linspace(max(x_min, 0), x_max, 15)
                special = np.array([x_old * 0.1, x_old * 0.5, x_old * 0.8, x_old, x_old * 1.2, x_old * 2])
                x_candidates = np.concatenate([log_pts, lin_pts, special])
                x_candidates = np.unique(x_candidates[x_candidates >= 0])
            
            # Evaluate
            Q_vals = np.array([_eval_q(x, S_other, b, a, i, x_old, n) for x in x_candidates])
            min_idx = int(np.argmin(Q_vals))
            x_rough = x_candidates[min_idx]
            
            # Golden section refinement
            if 0 < min_idx < len(x_candidates) - 1:
                left = x_candidates[min_idx - 1]
                right = x_candidates[min_idx + 1]
            elif min_idx == 0:
                left = max(0.0, x_rough - x_candidates[min_idx] * 0.5 if len(x_candidates) > 1 else 0.0)
                right = x_candidates[min_idx + 1] if len(x_candidates) > 1 else x_rough * 2
                if left == right:
                    left = 0.0
                    right = max(right, 1e-6)
            else:
                left = x_candidates[min_idx - 1]
                right = x_rough + (x_rough - left) * 0.5
                if right <= left:
                    right = left + 1e-6
            
            # Ensure left < right
            if right - left < 1e-15:
                x_opt = x_rough
            else:
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                for _ in range(40):
                    m1 = right - gr * (right - left)
                    m2 = left + gr * (right - left)
                    if _eval_q(m1, S_other, b, a, i, x_old, n) < _eval_q(m2, S_other, b, a, i, x_old, n):
                        right = m2
                    else:
                        left = m1
                    if right - left < 1e-12 * max(1.0, abs(x_rough)):
                        break
                x_opt = (left + right) / 2.0
            
            x_new = max(0.0, x_opt)
            
            # Simulated annealing: occasionally accept worse moves
            if anneal and temp > 0.001:
                Q_new = _eval_q(x_new, S_other, b, a, i, x_old, n)
                Q_old = _eval_q(x_old, S_other, b, a, i, x_old, n)
                if Q_new > Q_old and np.random.random() < np.exp(-(Q_new - Q_old) / temp):
                    pass  # accept
                elif Q_new > Q_old:
                    x_new = x_old  # reject
            
            if abs(x_new - x_old) > 1e-14:
                a[i] = x_new
                _update_b(b, a, i, x_old, x_new)
                S = S_other + x_new
                improved = True
        
        # End of sweep
        cur_obj = 2.0 * n * b.max() / (S * S)
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if anneal and no_improve >= max_no_improve:
            # Perturb and reset
            noise = np.random.standard_normal(n) * 0.003 * S / n
            a = np.maximum(a + noise, 0.0)
            S = a.sum()
            b = fast_convolve(a)
            no_improve = 0
            max_no_improve = int(max_no_improve * 1.5)
            temp *= 0.7
        
        if not anneal and no_improve >= 30:
            break  # converged
    
    return best_a, best_obj

def symmetric_coordinate_descent(a_init, time_limit, seed=42):
    """
    Coordinate descent exploiting symmetry: a[i] = a[n-1-i].
    Only optimizes the first half of coordinates.
    """
    np.random.seed(seed)
    a = np.maximum(a_init.astype(np.float64), 0.0).copy()
    # Enforce symmetry
    n = len(a)
    a = (a + a[::-1]) / 2.0
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n, dtype=np.float64) / n
        S = 1.0
    
    b = fast_convolve(a)
    max_b = b.max()
    best_obj = 2.0 * n * max_b / (S * S)
    best_a = a.copy()
    
    t0 = time.monotonic()
    half_n = (n + 1) // 2
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        no_improve += 1
        obj_start = 2.0 * n * b.max() / (S * S)
        
        order = np.random.permutation(half_n)
        
        for i in order:
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            
            j = n - 1 - i  # symmetric partner
            x_old = a[i]
            
            if i == j:
                # Center element (odd n): S_other doesn't double
                S_other = S - x_old
                factor = 1
            else:
                S_other = S - 2 * x_old
                factor = 2
            
            # Finding optimal x is more complex with symmetry
            # When a[i] and a[j] both change to x:
            #   sum changes by factor * (x - x_old)
            #   b_k changes accordingly
            
            # For simplicity, use the same 1D approach but account for both positions
            # This is complex; let's just do a simple line search
            
            x_max = max(x_old * 10, S / n) if x_old > 0 else S / n
            x_candidates = np.linspace(0, x_max, 30)
            
            best_x = x_old
            best_q = _eval_q_sym(x_old, S, b, a, i, j, x_old, n)
            
            for x in x_candidates:
                if x < 0:
                    continue
                q = _eval_q_sym(x, S, b, a, i, j, x_old, n)
                if q < best_q:
                    best_q = q
                    best_x = x
            
            # Refine
            if best_x > 0 and best_x < x_max:
                idx = int(np.argmin([_eval_q_sym(x, S, b, a, i, j, x_old, n) for x in x_candidates]))
                lo = max(0, x_candidates[max(0, idx-1)])
                hi = x_candidates[min(len(x_candidates)-1, idx+1)]
                for _ in range(30):
                    m1 = hi - (np.sqrt(5.0)-1.0)/2.0 * (hi - lo)
                    m2 = lo + (np.sqrt(5.0)-1.0)/2.0 * (hi - lo)
                    if _eval_q_sym(m1, S, b, a, i, j, x_old, n) < _eval_q_sym(m2, S, b, a, i, j, x_old, n):
                        hi = m2
                    else:
                        lo = m1
                    if hi - lo < 1e-12 * max(1.0, best_x):
                        break
                best_x = (lo + hi) / 2.0
            
            x_new = max(0.0, best_x)
            
            if abs(x_new - x_old) > 1e-14:
                a[i] = x_new
                a[j] = x_new
                _update_b_sym(b, a, i, j, x_old, x_new)
                S += factor * (x_new - x_old)
        
        cur_obj = 2.0 * n * b.max() / (S * S)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= 50:
            break
    
    return best_a, best_obj

def _eval_q_sym(x, S, b, a, i, j, x_old, n):
    """Evaluate objective when a[i]=a[j]=x (symmetric update)."""
    S_new = S + (1 if i == j else 2) * (x - x_old)
    if S_new < 1e-15:
        return float('inf')
    
    # Compute max_b with a[i]=a[j]=x
    # Instead of doing it exactly, approximate by computing the affected b values
    # This is approximate; for exact, we'd need to track all changes
    
    # Actually, let's just compute the full convolution - it's O(n log n) via FFT
    # But that's expensive inside the inner loop. Let's use the incremental approach.
    
    # For now, use a simpler approach: compute b directly
    a_new = a.copy()
    a_new[i] = x
    if i != j:
        a_new[j] = x
    b_new = fast_convolve(a_new)
    return 2.0 * n * b_new.max() / (S_new * S_new)

def _update_b_sym(b, a, i, j, x_old, x_new):
    """Update b when a[i] and a[j] both change from x_old to x_new."""
    n = len(a)
    _update_b(b, a, i, x_old, x_new)
    if i != j:
        _update_b(b, a, j, x_old, x_new)

def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Ramps
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = ramp / ramp.sum()
    inits['ramp_rev'] = ramp[::-1] / ramp.sum()
    
    # Triangular
    tri = np.minimum(x, n - 1 - x) + 1.0
    inits['triangle'] = tri / tri.sum()
    
    # U-shaped (bimodal at edges)
    u = 1.0 + 4.0 * ((x - 0.5*(n-1)) / (0.5*(n-1)))**2
    inits['u_shape'] = u / u.sum()
    
    # V-shaped
    v = np.abs(x - 0.5*(n-1)) + 1.0
    inits['v_shape'] = v / v.sum()
    
    # Gaussians
    for sf in [0.1, 0.2, 0.3, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5*(n-1)) / (sf*n))**2)
        inits[f'gauss_{sf}'] = g / g.sum()
    
    # Power law
    for p in [0.5, 1.0, 2.0, 3.0]:
        pl = (x + 1.0)**p
        inits[f'power_{p}'] = pl / pl.sum()
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['random_sparse'] = rng.dirichlet(np.ones(n) * 0.1)
    
    # From parent
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
```

```python
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using exact coordinate descent with:

- Incremental O(n) convolution updates (no FFT in inner loop)

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def _update_b(b, a, i, x_old, x_new):
    """Update b incrementally when a[i] changes from x_old to x_new."""
    n = len(a)
    dx = x_new - x_old
    if abs(dx) < 1e-15:
        return
    k_start = i
    k_end = min(i + n - 1, 2 * n - 2)
    for k in range(k_start, k_end + 1):
        ki = k - i
        if ki < 0 or ki >= n:
            continue
        if k == 2 * i:
            b[k] += x_new * x_new - x_old * x_old
        else:
            b[k] += 2.0 * dx * a[ki]


def _eval_q(x, S_other, b, a, i, x_old, n):
    """Evaluate 2*n*max_b(x)/(S_other+x)^2 for coordinate i set to x."""
    S_new = S_other + x
    if S_new < 1e-15:
        return 1e30

    k_start = i
    k_end = min(i + n - 1, 2 * n - 2)

    if k_start > k_end:
        mb = float(b.max())
    else:
        # Unaffected portion
        m_unaff = 0.0
        if k_start > 0:
            m_unaff = max(m_unaff, float(b[:k_start].max()))
        if k_end + 1 < 2 * n - 1:
            m_unaff = max(m_unaff, float(b[k_end + 1:].max()))

            b_at_x[idx_2i] = b_base[idx_2i] + x * x

        m_aff = float(b_at_x.max())
        mb = max(m_unaff, m_aff)

            if hi - lo > 1e-14:
                for _ in range(45):
                    m1 = hi - golden_ratio * (hi - lo)
                    m2 = lo + golden_ratio * (hi - lo)
                    q1 = _eval_q(m1, S_other, b, a, i, x_old, n)
                    q2 = _eval_q(m2, S_other, b, a, i, x_old, n)
                    if q1 < q2:
                        hi = m2
                    else:
                        lo = m1
                    if hi - lo < 1e-13 * max(1.0, abs(x_rough)):
                        break
                x_opt = (lo + hi) * 0.5
            else:
                x_opt = x_rough

            x_new = max(0.0, x_opt)

            # Annealing: sometimes accept worse moves
            if anneal and temp > 1e-5 and abs(x_new - x_old) > 1e-12:
                q_new = _eval_q(x_new, S_other, b, a, i, x_old, n)
                q_old = _eval_q(x_old, S_other, b, a, i, x_old, n)
                if q_new > q_old:
                    delta = q_new - q_old
                    if np.random.random() < np.exp(-delta / temp):
                        pass  # accept
                    else:
                        x_new = x_old  # reject

            if abs(x_new - x_old) > 1e-14:
                a[i] = x_new
                _update_b(b, a, i, x_old, x_new)
                S = S_other + x_new
                improved_this_sweep = True

        # End of sweep
        if sweep % 200 == 0:
            b = fast_convolve(a).astype(np.float64)  # recompute for stability

        cur_obj = 2.0 * n * float(b.max()) / (S * S)

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD sweep {sweep}: obj={cur_obj:.8f}, best={best_obj:.8f}, S={S:.4f}, t={elapsed:.1f}s")

        # Perturbation / restart
        if anneal and no_improve >= max_no_improve:
            noise = np.random.standard_normal(n) * 0.005
            a = np.maximum(a + noise, 0.0)
            S = a.sum()
            if S > 1e-10:
                a = a / S
                S = 1.0
            b = fast_convolve(a).astype(np.float64)
            no_improve = 0
            max_no_improve = int(max_no_improve * 1.3) + 10
            temp *= 0.6

        if not anneal and no_improve >= max_no_improve:
            break

    # Final: normalize and compute true objective
    final_S = float(best_a.sum())
    final_b = fast_convolve(best_a)
    final_max = float(final_b.max())
    final_obj = 2.0 * n * final_max / (final_S * final_S) if final_S > 1e-15 else 1e30

    return best_a, final_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n (all normalized to sum=1)."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    mid = 0.5 * (n - 1)

    inits['uniform'] = np.ones(n) / n

    # U-shaped (high at edges)
    u = 1.0 + 4.0 * ((x - mid) / (0.5 * (n - 1) + 1e-9)) ** 2
    inits['u_shape'] = u / u.sum()

    # Strong U
    u2 = 1.0 + 20.0 * ((x - mid) / (0.5 * (n - 1) + 1e-9)) ** 2
    inits['u_strong'] = u2 / u2.sum()

    # V-shaped
    v = np.abs(x - mid) + 0.5
    inits['v_shape'] = v / v.sum()

    # Triangle (peak at center)
    tri = np.minimum(x, n - 1 - x) + 1.0
    inits['triangle'] = tri / tri.sum()

    # Gaussians
    for sf in [0.1, 0.15, 0.2, 0.3, 0.5]:
        g = np.exp(-0.5 * ((x - mid) / (sf * n)) ** 2)
        inits[f'gauss_{sf}'] = g / g.sum()

    # Bimodal (two bumps at 1/4 and 3/4)
    bi = np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2) + np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2)
    inits['bimodal'] = bi / bi.sum()

    # Edge concentration
    edge = np.zeros(n)
    w = max(1, n // 8)
    edge[:w] = 1.0
    edge[-w:] = 1.0
    inits['edges'] = edge / edge.sum()

    # Power laws
    for p in [0.5, 1.0, 2.0]:
        pl = (x + 1.0) ** p
        inits[f'power_{p}'] = pl / pl.sum()

    # Random
    inits['random1'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['random2'] = rng.dirichlet(np.ones(n) * 0.1)
    inits['random3'] = rng.dirichlet(np.ones(n) * 2.0)

    # From parent
    if parent is not None and len(parent) > 0:
        if n == len(parent):
            pp = np.maximum(parent, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        elif len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps

    return inits


def upsample(a, n_new):
    """Upsample sequence a to length n_new via interpolation."""
    n_old = len(a)
    if n_old == n_new:
        return a.astype(np.float64).copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    result = np.interp(x_new, x_old, a)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    return result


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase coordinate descent search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(10.0, float(budget_s) - 25.0)

    best_sequence = None
    best_score = float('inf')

    def try_sequence(a, label=""):
        nonlocal best_sequence, best_score
        if a is None or len(a) == 0:
            return
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print(f"  *** NEW BEST: {label}, n={len(a)}, score={score:.8f}")

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: len={len(p)}, sum={p.sum():.4f}")
    except Exception:
        pass

    # ========== Phase 1: Quick exploration ==========
    print("=== Phase 1: Quick Exploration ===")
    t_phase1_end = time.monotonic() + 200.0

    # Try parent and simple transforms
    if parent is not None:
        pn = parent / parent.sum()
        try_sequence(pn, "parent_norm")
        psym = (pn + pn[::-1]) / 2.0
        try_sequence(psym, "parent_sym")

    lengths = [96, 128, 160, 192, 256, 320, 384, 512]
    best_per_length = {}

    for n in lengths:
        if time.monotonic() > t_phase1_end - 20:
            break
        inits = generate_initializations(n, parent, rng)
        init_names = list(inits.keys())

        # Prioritize diverse shapes
        priority = ['parent', 'parent_ds', 'parent_up', 'u_shape', 'u_strong',
                    'edges', 'bimodal', 'gauss_0.2', 'gauss_0.3', 'random1', 'uniform']
        init_names = [nm for nm in priority if nm in inits] + \
                     [nm for nm in init_names if nm not in priority]
        init_names = init_names[:7]

        t_per = max(8.0, 150.0 / (len(lengths) * 5))

        for name in init_names:
            if time.monotonic() > t_phase1_end - 15:
                break
            t_alloc = min(t_per, t_phase1_end - time.monotonic() - 10)
            if t_alloc < 5:
                break

            a0 = inits[name]
            sv = int(rng.integers(0, 99999))
            a_opt, obj_opt = coordinate_descent(a0, time_limit=t_alloc, seed=sv)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  P1: n={n}, {name}, score={score:.8f}")
                if n not in best_per_length or score < best_per_length[n][0]:
                    best_per_length[n] = (score, a_opt.copy())

    print(f"Phase 1 done. Best: {best_score:.8f}")

    # ========== Phase 2: Deep optimization ==========
    print("=== Phase 2: Deep Optimization ===")
    t_phase2_end = deadline - 180.0

    # Build list of starting points for deep runs
    deep_starts = []

    # Best found so far
    if best_sequence is not None:
        deep_starts.append(('best', best_sequence.copy()))

    # Best per length
    for n, (sc, a) in sorted(best_per_length.items(), key=lambda x: x[1][0]):
        if len(deep_starts) >= 6:
            break
        key = f'len_{n}'
        if not any(key in s[0] for s in deep_starts):
            deep_starts.append((key, a.copy()))

    # Add some fresh starts at promising lengths
    best_n = len(best_sequence) if best_sequence is not None else 256
    for extra in ['u_shape', 'bimodal', 'edges']:
        inits_best_n = generate_initializations(best_n, parent, rng)
        if extra in inits_best_n:
            deep_starts.append((f'fresh_{extra}', inits_best_n[extra]))

        # Also try a slightly different length
        n2 = best_n + 64
        inits_n2 = generate_initializations(n2, parent, rng)
        if extra in inits_n2:
            deep_starts.append((f'fresh_{extra}_{n2}', inits_n2[extra]))

    deep_starts = deep_starts[:10]
    remaining2 = t_phase2_end - time.monotonic()
    t_per_deep = max(15.0, remaining2 / len(deep_starts))

    for idx, (label, a0) in enumerate(deep_starts):
        if time.monotonic() > t_phase2_end - 20:
            break
        t_alloc = min(t_per_deep, t_phase2_end - time.monotonic() - 15)
        if t_alloc < 8:
            break

        # Try two runs: one with annealing, one without
        for run_idx, use_anneal in enumerate([False, True]):
            t_single = t_alloc / 2.0
            if t_single < 5:
                break
            sv = int(rng.integers(0, 99999))
            a_opt, obj_opt = coordinate_descent(a0, time_limit=t_single, seed=sv, anneal=use_anneal)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  P2: {label} anneal={use_anneal}, n={len(a_opt)}, score={score:.8f}")

    print(f"Phase 2 done. Best: {best_score:.8f}")

    # ========== Phase 3: Multi-resolution and perturbation ==========
    print("=== Phase 3: Multi-Resolution & Perturbation ===")
    t_phase3_end = deadline - 80.0

    if best_sequence is not None:
        remaining3 = t_phase3_end - time.monotonic()
        n_best = len(best_sequence)

        # Try different lengths by up/down sampling the best
        for factor in [0.75, 1.0, 1.25, 1.5]:
            n_try = max(32, int(n_best * factor))
            if n_try == n_best:
                continue
            if time.monotonic() > t_phase3_end - 30:
                break
            a_up = upsample(best_sequence, n_try)
            sv = int(rng.integers(0, 99999))
            t_alloc = min(60.0, t_phase3_end - time.monotonic() - 20)
            if t_alloc < 5:
                break
            a_opt, obj_opt = coordinate_descent(a_up, time_limit=t_alloc, seed=sv, anneal=False)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  P3 upsample n={n_try}: score={score:.8f}")

        # Perturbation + re-optimize (multiple times)
        for pert_round in range(5):
            if time.monotonic() > t_phase3_end - 20:
                break
            n_cur = len(best_sequence)
            scales = [0.002, 0.005, 0.01, 0.02, 0.05][pert_round % 5]
            noise = rng.standard_normal(n_cur) * scales
            a_pert = np.maximum(best_sequence + noise, 0.0)
            s_pert = a_pert.sum()
            if s_pert < 1e-10:
                continue
            a_pert = a_pert / s_pert
            sv = int(rng.integers(0, 99999))
            t_alloc = min(40.0, t_phase3_end - time.monotonic() - 15)
            if t_alloc < 3:
                break
            use_anneal = (pert_round % 2 == 0)
            a_opt, obj_opt = coordinate_descent(a_pert, time_limit=t_alloc, seed=sv, anneal=use_anneal)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  P3 perturb r{pert_round}: score={score:.8f}")

    print(f"Phase 3 done. Best: {best_score:.8f}")

    # ========== Phase 4: Final polish ==========
    print("=== Phase 4: Final Polish ===")
    if best_sequence is not None:
        remaining4 = deadline - time.monotonic() - 5.0
        if remaining4 > 5:
            # Long polished run
            sv = int(rng.integers(0, 99999))
            t_alloc = remaining4 * 0.6
            a_opt, obj_opt = coordinate_descent(best_sequence, time_limit=t_alloc, seed=sv, anneal=False)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  P4 polish: score={score:.8f}")

            # One more with annealing
            remaining4 = deadline - time.monotonic() - 5.0
            if remaining4 > 5:
                sv = int(rng.integers(0, 99999))
                a_opt, obj_opt = coordinate_descent(best_sequence, time_limit=remaining4 * 0.4,
                                                    seed=sv, anneal=True)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  P4 anneal: score={score:.8f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.8f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
```
