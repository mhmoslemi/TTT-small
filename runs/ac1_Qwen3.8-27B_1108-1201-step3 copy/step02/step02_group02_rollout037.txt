```python
import numpy as np
import time

def fft_convolve(a, b):
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def compute_score(a):
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)

def compute_gradient(a, beta):
    """Compute gradient of soft-minimax objective."""
    n = len(a)
    c = fft_convolve(a, a)
    # Soft max weights
    max_c = np.max(c)
    shifted = beta * (c - max_c)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / np.sum(exp_shifted)
    
    # grad[i] = 2 * sum_j w[i+j] * a[j]
    # = 2 * (a_rev * w)[n-1+i]
    a_rev = a[::-1]
    conv = fft_convolve(a_rev, w)
    grad = 2.0 * conv[n-1:2*n-1]
    return grad, c

def optimize(a_init, time_limit, beta_start=10.0, beta_end=1000.0, rng=None):
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = np.clip(a, 0, None)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s  # Normalize to sum = 1
    
    if rng is None:
        rng = np.random.default_rng()
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    lr = 0.01
    beta = beta_start
    total_iters = 5000  # Will be adjusted by time
    n_iters = 0
    
    while time.monotonic() - t0 < time_limit:
        prog = min(1.0, n_iters / total_iters)
        beta = beta_start * (beta_end / beta_start) ** prog
        
        grad, c = compute_gradient(a, beta)
        
        # Adaptive step size
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-10:
            # Scale gradient to have unit norm, then use lr as step size
            step = lr * grad / grad_norm
        else:
            step = 0
        
        a_new = a - step
        a_new = np.clip(a_new, 0, None)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            a_new = a
        else:
            a_new = a_new / s_new
        
        new_score = compute_score(a_new)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()

        if new_score <= compute_score(a) * (1 + 0.01):
            a = a_new
        else:
            # Try smaller step
            a_new2 = a - step * 0.5
            a_new2 = np.clip(a_new2, 0, None)
            s_new2 = np.sum(a_new2)
            if s_new2 > 1e-10:
                a_new2 = a_new2 / s_new2
            if compute_score(a_new2) <= compute_score(a):
                a = a_new2
        
        # Occasionally perturb
        if n_iters % 100 == 0:
            a = a * np.exp(rng.normal(0, 0.02, n))
            s = np.sum(a)
            if s > 1e-10:

        lr = max(0.001, 0.01 * (1 - prog) ** 0.5)
```

```python
"""
Soft-minimax optimization with exponentiated gradient for minimizing peak autocorrelation.
Uses FFT-based gradient computation for O(n log n) per iteration.
"""

import numpy as np
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def compute_score(a):
    """Compute the evaluation score."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-10:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (s * s)


def compute_soft_gradient(a, beta):
    """
    Compute the gradient of the soft-minimax objective.
    Returns (gradient, convolution, max_c).
    """
    n = len(a)
    c = fft_convolve(a, a)
    max_c = np.max(c)
    shifted = beta * (c - max_c)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w = w / w_sum
    
    # grad[i] = 2 * (a_rev * w)[n-1+i]
    a_rev = a[::-1]
    conv = fft_convolve(a_rev, w)
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad, c, max_c


def optimize_expgrad(a_init, time_limit, rng, beta_start=30.0, beta_end=500.0, 
                      eta_start=0.05, eta_min=0.005, verbose=False):
    """
    Optimize using exponentiated gradient with soft-minimax.
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = np.clip(a, 0, None)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iter_count = 0
    no_improve_count = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        prog = min(1.0, elapsed / time_limit)
        
        # Schedule
        beta = beta_start + (beta_end - beta_start) * (prog ** 0.5)
        eta = eta_start * (eta_min / eta_start) ** (prog ** 0.7)
        
        # Compute gradient
        grad, c, max_c = compute_soft_gradient(a, beta)
        
        # Normalize gradient for stable updates
        grad_mean = np.mean(grad)
        grad_centered = grad - grad_mean
        
        # Exponentiated gradient step
        a_new = a * np.exp(-eta * grad_centered)
        a_new = np.clip(a_new, 1e-15, None)
        a_new = a_new / a_new.sum()
        
        new_score = 2.0 * n * max_c / 1.0  # Since sum(a) = 1, score = 2*n*max_c
        # Actually recompute properly
        new_score = compute_score(a_new)
        
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
        
        a = a_new
        iter_count += 1
        
        # Periodic exploration
        if no_improve_count > 200:
            # Try a perturbation
            a_pert = a * np.exp(rng.normal(0, 0.03, n))
            a_pert = np.clip(a_pert, 1e-15, None)
            a_pert = a_pert / a_pert.sum()
            pert_score = compute_score(a_pert)
            if pert_score < best_score:
                a = a_pert
                best_score = pert_score
                best_a = a_pert.copy()
            no_improve_count = 0
        
        if verbose and iter_count % 10000 == 0:
            print(f"    iter={iter_count}, score={compute_score(a):.6f}, best={best_score:.6f}, beta={beta:.1f}, eta={eta:.4f}")
    
    return best_a, best_score


def polish_subgradient(a_init, time_limit, rng, verbose=False):
    """
    Polish using subgradient descent on the actual peak (hard max).
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = np.clip(a, 0, None)
    s = np.sum(a)
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iter_count = 0
    
    # Find active set size to use
    active_frac = 0.1  # Use top 10% of peaks
    
    while time.monotonic() - t0 < time_limit:
        c = fft_convolve(a, a)
        max_c = np.max(c)
        
        # Find active set: positions where c[k] is within a small fraction of max
        threshold = max_c * (1.0 - 0.01)
        active = np.where(c >= threshold)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        
        # Compute gradient from active set
        grad = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for i in range(lo, hi + 1):
                grad[i] += 2.0 * a[k - i]
        grad /= len(active)
        
        # Center the gradient
        grad -= np.mean(grad)
        
        # Step
        eta = 0.01 / (1.0 + iter_count * 0.001)
        a_new = a - eta * grad
        a_new = np.clip(a_new, 0, None)
        s_new = np.sum(a_new)
        if s_new < 1e-10:
            a_new = a
        else:
            a_new = a_new / s_new
        
        new_score = compute_score(a_new)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
        
        a = a_new
        iter_count += 1
    
    return best_a, best_score
```

Actually, let me think about what makes a good initial sequence. The key property we want is "flat autocorrelation". 

For a random sequence with a[i] ~ Bernoulli(p) (i.e., 0 or 1 with probability p), the autocorrelation is approximately:
- At lag 0: p (if we're looking at the normalized version)
- At other lags: approximately p (by independence)

So random 0/1 sequences already have fairly flat autocorrelation! The issue is that for 0/1 sequences of length n, the sum is random and the peak might be slightly above average.

For a continuous sequence, we want a[i] > 0 everywhere (to avoid the peak being too high at the self-correlation). A random positive sequence should work well as a starting point.

Let me also think about the "ideal" sequence. If we could make c[k] = S^2/(2n-1) for all k, we'd achieve the bound. This means:
- sum_k c[k] = S^2 (which is always true)
- c[k] is constant for all k

This is the "flat top" autocorrelation. For non-negative sequences, this is related to "Golay complementary sequences" or "low-PAPR sequences".

One known construction: if a is a "Barker sequence" (all-off-peak autocorrelation <= 1), then for a{0,1} sequence of length n, the off-peak values are bounded. But Barker sequences are very short (max length 13 for binary).

For non-negative real sequences, we can do better. The key is to have the sequence vary smoothly so that the autocorrelation falls off slowly.

Actually, for our problem, we want the autocorrelation to be as FLAT as possible (not falling off). This is the opposite of what we usually want in signal processing!

A sequence with flat autocorrelation is one whose Fourier transform has constant magnitude. For a real non-negative sequence, this means the DFT should have roughly constant magnitude.

A random sequence has approximately flat autocorrelation (by the central limit theorem, the autocorrelation at non-zero lags is approximately Gaussian with mean ~0 and variance ~S^4/n for large n). But we need non-negative values, so the mean of the off-peak autocorrelation is positive.

For a sequence with a[i] = 1 + ε*x[i] where x[i] are i.i.d. with mean 0, the autocorrelation at lag k ≠ 0 is:
