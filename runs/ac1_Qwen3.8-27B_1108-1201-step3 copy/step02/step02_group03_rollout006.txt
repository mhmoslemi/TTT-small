def normalize(a, target):

def build_conv_matrix(f):
    """Build the (2n-1) x n matrix A where (A @ g)[k] = convolve(f,g)[k]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = j
        hi = j + n
        A[lo:hi, j] = f[:hi - lo]
    return A


def lp_direct(f, target_sum, time_limit=30):
    """

    A_ub[:, -1] = -1.0

        hi = j + n

def gradient_of_max_conv(a, tau=None):
    """
    Compute gradient of max_k conv(a,a)[k] or soft-max version w.r.t. a.
    Returns gradient vector of length n.
    """
    n = len(a)
    c = np.convolve(a, a)

    if tau is None:
        k_star = int(np.argmax(c))
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
    else:
        c_shifted = c - c.max()
        w = np.exp(tau * c_shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
        else:
            w /= w_sum
        # grad[i] = 2 * sum_j w[i+j] * a[j]
        a_rev = a[::-1]
        conv_result = np.convolve(w, a_rev, mode='full')
        grad = 2.0 * conv_result[n - 1:n - 1 + n]

    return grad


def nesterov_step(a, grad_fn, lr, momentum, beta, target_sum):
    """One Nesterov momentum step with projection."""
    n = len(a)
    # Lookahead
    y = a + beta * momentum
    g = grad_fn(y)
    # Project out mean to maintain sum constraint (approximately)
    g = g - g.mean()
    # Update
    momentum = beta * momentum - lr * g
    a_new = y + momentum
    a_new = np.maximum(a_new, 0.0)
    a_new = normalize(a_new, target_sum)
    return a_new, momentum

    # Phase 4: Local polish (coordinate descent for smaller n)
    remaining = deadline - time.monotonic()
    if remaining > 10 and n <= 512:
        t_phase4 = time.monotonic() + remaining * 0.8
        a = best.copy()
        polish_deadline = t_phase4
        improved_any = True
        rounds = 0
        while improved_any and rounds < 5 and time.monotonic() < polish_deadline:
            rounds += 1
            improved_any = False
            c = np.convolve(a, a)
            k_star = int(np.argmax(c))
            for i in range(n):
                if time.monotonic() > polish_deadline:
                    break
                orig = a[i]
                c_local = np.convolve(a, a)
                # The peak contribution from a[i] is 2*a[i]*a[k_star-i] (if valid)
                j = k_star - i
                if 0 <= j < n:
                    # Try reducing a[i] proportionally to its contribution
                    contrib = a[i] * a[j]
                    peak = c_local.max()
                    if peak > 0 and contrib > 0:
                        # Try reducing a[i]
                        for factor in [0.5, 0.7, 0.85]:
                            a[i] = orig * factor
                            a = normalize(a, target_sum)
                            s = compute_obj(a)
                            if s < best_score - 1e-10:
                                best_score = s
                                best = a.copy()
                                improved_any = True
                                orig = a[i]
                                break
                            a[i] = orig
                        else:
                            a[i] = orig
                    a = normalize(a, target_sum)

        if improved_any:
            print(f"    Phase 4 (polish): score={best_score:.6f}, rounds={rounds}")

    return best, best_score


def generate_starts(rng, lengths):
    """Generate diverse starting sequences."""
    starts = []

    # From height_sequence_1 if available
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        if gs.size >= 16 and np.all(np.isfinite(gs)):
            for L in lengths:
                if L <= len(gs):
                    starts.append(gs[:L].copy())
                else:
                    # Pad by tiling
                    reps = int(np.ceil(L / len(gs)))
                    padded = np.tile(gs, reps)[:L]
                    starts.append(padded.copy())
            # Perturbed versions
            for L in lengths[:3]:
                for trial in range(2):
                    noise = rng.normal(0, 0.1, min(L, len(gs)))
                    perturbed = np.maximum(gs[:L] + noise, 0)
                    starts.append(perturbed.copy())
    except NameError:
        pass
    except Exception:
        pass

    # Random starts
    for n in lengths:
        for _ in range(2):
            starts.append(rng.random(n) * 0.5 + 0.25)

    # Structured starts: anti-palindromic (to reduce center peak)
    for n in lengths:
        half = n // 2
        left = rng.random(half) * 0.5 + 0.25
        # Anti-palindromic: right side is inverse of left
        right = 0.75 - left
        if n % 2 == 1:
            mid = [0.5]
            starts.append(np.concatenate([left, mid, right]))
        else:
            starts.append(np.concatenate([left, right]))

    # Periodic starts (comb-like)
    for n in lengths:
        for period in [2, 3, 4]:
            base = rng.random(period) * 0.5 + 0.2
            a = np.tile(base, n // period + 1)[:n]
            starts.append(a.copy())

    # Sine-modulated starts
    for n in lengths:
        i = np.arange(n, dtype=np.float64)
        for freq in [2, 3, 5]:
            a = 0.5 + 0.4 * np.cos(2 * np.pi * i * freq / n)
            starts.append(np.maximum(a, 0.01).copy())

    # Gaussian window
    for n in lengths:
        i = np.arange(n, dtype=np.float64)
        center = n / 2.0
        a = np.exp(-0.5 * ((i - center) / (n * 0.15)) ** 2)
        starts.append((a + 0.05).copy())

    return starts


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Uses multi-phase optimization (Nesterov gradient,
    direct LP refinement, local polish) from multiple diverse starting points.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(0.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    print(f"Starting optimization, budget={budget_s}s")

    # Promising lengths
    lengths = [256, 384, 512, 768]

    # Generate starts
    starts = generate_starts(rng, lengths)

    # Sort by length
    starts.sort(key=lambda x: len(x))

    # Deduplicate
    seen = set()
    unique_starts = []
    for s in starts:
        key = (len(s), tuple(np.round(s[:8], 2)))
        if key not in seen:
            seen.add(key)
            unique_starts.append(s)
    starts = unique_starts

    print(f"Total starting sequences: {len(starts)}")

    # Phase A: Quick evaluation of all starts to find promising ones
    scored_starts = []
    for a0 in starts:
        a_norm = normalize(np.clip(a0, 0, 1000), np.sqrt(2.0 * len(a0)))
        s = compute_obj(a_norm)
        scored_starts.append((s, a_norm))
    scored_starts.sort(key=lambda x: x[0])

    # Take top starts for full optimization
    top_k = min(12, len(scored_starts))
    top_starts = [a for _, a in scored_starts[:top_k]]

    print(f"Top {top_k} starts by initial score:")
    for i, (s, a) in enumerate(scored_starts[:top_k]):
        print(f"  [{i}] n={len(a)}, score={s:.6f}")

    # Phase B: Full optimization on top starts
    n_starts = len(top_starts)
    for idx, a0 in enumerate(top_starts):
        if time.monotonic() > deadline - 60.0:
            break

        remaining = deadline - time.monotonic() - 30
        # Allocate time: earlier starts get less, later (more promising) get more
        if idx < 3:
            t_alloc = min(90.0, remaining / (n_starts - idx) * 1.2)
        elif idx < 6:
            t_alloc = min(150.0, remaining / (n_starts - idx) * 1.5)
        else:
            t_alloc = min(200.0, remaining / (n_starts - idx) * 2.0)

        t_alloc = max(30.0, t_alloc)
        if time.monotonic() + t_alloc > deadline - 30:
            t_alloc = deadline - time.monotonic() - 30
            if t_alloc < 30:
                break

        print(f"\n  [start {idx}] n={len(a0)}, initial_score={compute_obj(normalize(a0, np.sqrt(2*len(a0)))):.6f}, alloc={t_alloc:.0f}s")

        seq, score = optimize_single(a0, deadline - 30)

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"  [start {idx}] NEW BEST: score={best_score:.6f}, n={len(seq)}")
        else:
            print(f"  [start {idx}] score={score:.6f}, n={len(seq)} (not better)")

    # Phase C: If we have a good result, try to extend/improve with more LP iterations
    if best_seq is not None and time.monotonic() < deadline - 60:
        print(f"\nPhase C: Extended LP optimization on best (score={best_score:.6f})")
        remaining = deadline - time.monotonic() - 20

        # Try a few perturbed versions of the best
        extra_starts = []
        n_best = len(best_seq)
        for trial in range(4):
            noise = rng.normal(0, 0.05, n_best)
            perturbed = np.maximum(best_seq + noise, 0)
            extra_starts.append(perturbed)

        # Also try different lengths near the best
        for L in [max(64, n_best // 2), n_best, min(1024, n_best * 2)]:
            if L != n_best:
                if best_seq is not None and L <= len(best_seq):
                    extra_starts.append(best_seq[:L].copy())
                else:
                    extra_starts.append(rng.random(L) * 0.5 + 0.25)

        for idx, a0 in enumerate(extra_starts):
            if time.monotonic() > deadline - 30:
                break
            t_alloc = min(180.0, (deadline - time.monotonic() - 30) / max(1, len(extra_starts) - idx))
            if t_alloc < 30:
                break

            seq, score = optimize_single(a0, deadline - 30)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"  [extra {idx}] NEW BEST: score={best_score:.6f}")
            else:
                print(f"  [extra {idx}] score={score:.6f}")

    # Phase D: Final intensive LP polish on the absolute best
    if best_seq is not None and time.monotonic() < deadline - 20:
        print(f"\nPhase D: Final LP polish (score={best_score:.6f})")
        remaining = deadline - time.monotonic() - 10
        # Give all remaining time to one more optimization
        seq, score = optimize_single(best_seq, deadline - 5,
                                     phase_budgets={'grad': remaining * 0.1,
                                                    'lp': remaining * 0.8,
                                                    'polish': remaining * 0.1})
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"  Final polish: score={best_score:.6f}")

    # Final validation
    if best_seq is not None:
        final_score = compute_obj(best_seq)
        print(f"\nFinal: n={len(best_seq)}, score={final_score:.6f}")
        try:
            eval_score = float(evaluate_sequence([float(x) for x in best_seq]))
            print(f"evaluate_sequence score: {eval_score:.6f}")
        except Exception as e:
            print(f"evaluate_sequence error: {e}")
        # Ensure all values are valid
        best_seq = np.clip(best_seq, 0.0, 1000.0)
        return [float(x) for x in best_seq]

    # Fallback
    print("Falling back to uniform sequence.")
    return [1.0] * 256
