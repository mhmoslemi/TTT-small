Optimizes non-negative step function to minimize 2*n*max(autocorr)/(sum)^2.
Uses LP-guided Nesterov acceleration with simulated annealing perturbations
and golden-section coordinate descent polishing.

import numpy as np
import time
from scipy.optimize import linprog

def compute_obj(a):
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))

def normalize(a, target=None):
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)

def soft_max_gradient(a, tau):
    """Compute gradient of soft-max of autocorrelation."""
    n = len(a)
    c = np.convolve(a, a)
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    # grad[i] = 2 * sum_j w[i+j] * a[j]
    a_rev = a[::-1]
    conv_result = np.convolve(w, a_rev, mode='full')
    grad = 2.0 * conv_result[n-1:n-1+n]
    return grad

def lp_solve(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = len(f)
    if n > 1024:
        return None
    c_obj = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        for m in range(j, j + n):
            A[m, j] = f[m - j]
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None

def nesterov_step(a, grad, v, lr, mu):
    """One Nesterov step with momentum."""
    # Nesterov: y = a + mu * v
    # a_new = y - lr * grad(y)  (simplified: use grad(a) instead of grad(y))
    # v_new = a_new - y + v  (actually v_new = (a_new - y) / (1-mu) or similar)
    # Simplified Nesterov with momentum mu:
    v_new = mu * v - lr * grad
    a_new = a + v_new
    a_new = np.maximum(a_new, 0.0)
    return a_new, v_new

def coordinate_descent_golden(a, deadline, max_passes=5):
    """Coordinate descent with golden section search."""
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = normalize(a.copy(), S_t)
    best_score = compute_obj(a)
    
    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            orig = a[i]
            if orig < 1e-6:
                # Try setting to small positive value
                a[i] = 0.01
                a[n-1] = max(0, a[n-1] - 0.01)  # keep sum roughly constant
                # Actually, let's just try a few values
                candidates = [0.0, orig + 0.01, orig + 0.05]
            else:
                # Golden section search in [0, 2*orig]
                lo, hi = 0.0, orig * 2.5
                # Try golden section
                gr = (np.sqrt(5) - 1) / 2
                a1 = hi - gr * (hi - lo)
                a2 = lo + gr * (hi - lo)
                
                a[i] = a1
                a = normalize(a, S_t)
                f1 = compute_obj(a)
                a[i] = a2
                a = normalize(a, S_t)
                f2 = compute_obj(a)
                
                for _ in range(8):
                    if time.monotonic() > deadline:
                        break
                    if f1 < f2:
                        hi = a2
                        a2 = a1
                        f2 = f1
                        a1 = hi - gr * (hi - lo)
                        a[i] = a1
                        a = normalize(a, S_t)
                        f1 = compute_obj(a)
                    else:
                        lo = a1
                        a1 = a2
                        f1 = f2
                        a2 = lo + gr * (hi - lo)
                        a[i] = a2
                        a = normalize(a, S_t)
                        f2 = compute_obj(a)
                
                best_val = (lo + hi) / 2

                a = normalize(a, S_t)
                s = compute_obj(a)
                if s < best_score - 1e-10:
                    best_score = s
                    improved = True
                else:
                    a[i] = orig

"""
Optimizes non-negative step function to minimize 2*n*max(autocorr)/(sum)^2.
Uses LP-guided line search, Nesterov accelerated soft-max gradients,
simulated annealing exploration, and golden-section coordinate descent polishing.
Key differences from previous: proper line search along LP directions,
more aggressive LP iteration, Nesterov (not plain momentum), SA with restarts,
and finer coordinate descent.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_obj(a):
    """Compute objective: 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize(a, target=None):
    """Normalize sequence to have a specific sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)


def soft_max_grad(a, tau):
    """Gradient of soft-max of autocorrelation w.r.t. a."""
    n = len(a)
    c = np.convolve(a, a)
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    a_rev = a[::-1]
    conv_result = np.convolve(w, a_rev, mode='full')
    grad = 2.0 * conv_result[n - 1:n - 1 + n]
    return grad


def hard_max_grad(a):
    """Gradient at the single peak lag."""
    n = len(a)
    c = np.convolve(a, a)
    k_star = int(np.argmax(c))
    grad = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            grad[i] = 2.0 * a[j]
    return grad


def lp_solve(f, M):
    """LP: max sum(g) s.t. conv(f,g) <= M, g >= 0. Returns optimal g or None."""
    n = len(f)
    if n > 1024:
        return None
    c_obj = -np.ones(n)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_range = A[j:j + n, j]
        A_range[:] = f[:n]
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def line_search_lp(a, g_dir, S_t):
    """Line search along LP direction. Returns best (a_new, score)."""
    g_norm = g_dir * (S_t / max(g_dir.sum(), 1e-12))
    best_a = a.copy()
    best_s = compute_obj(a)
    for t in [0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.8, 0.5]:
        a_new = (1.0 - t) * a + t * g_norm
        a_new = np.maximum(a_new, 0.0)
        a_new = normalize(a_new, S_t)
        s_new = compute_obj(a_new)
        if s_new < best_s:
            best_s = s_new
            best_a = a_new.copy()
    return best_a, best_s


def nesterov_step(a, grad, v, lr, mu):
    """Nesterov momentum step."""
    v_new = mu * v - lr * grad
    a_new = a + v_new
    a_new = np.maximum(a_new, 0.0)
    return a_new, v_new


def golden_coord_descent(a, deadline, max_passes=5):
    """Coordinate descent with golden section search per coordinate."""
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = normalize(a.copy(), S_t)
    best_score = compute_obj(a)
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        improved = False
        for i in range(n):
            if time.monotonic() > deadline:
                break
            orig = a[i]
            lo = 0.0
            hi = max(orig * 3.0, 0.1)
            # Golden section
            a1 = hi - gr * (hi - lo)
            a2 = lo + gr * (hi - lo)
            a[i] = a1
            a = normalize(a, S_t)
            f1 = compute_obj(a)
            a[i] = a2
            a = normalize(a, S_t)
            f2 = compute_obj(a)
            for _ in range(10):
                if time.monotonic() > deadline:
                    break
                if f1 < f2:
                    hi = a2
                    a2 = a1
                    f2 = f1
                    a1 = hi - gr * (hi - lo)
                    a[i] = a1
                    a = normalize(a, S_t)
                    f1 = compute_obj(a)
                else:
                    lo = a1
                    a1 = a2
                    f1 = f2
                    a2 = lo + gr * (hi - lo)
                    a[i] = a2
                    a = normalize(a, S_t)
                    f2 = compute_obj(a)
            best_val = (lo + hi) * 0.5
            a[i] = best_val
            a = normalize(a, S_t)
            s = compute_obj(a)
            if s < best_score - 1e-10:
                best_score = s
                improved = True
            else:
                a[i] = orig
                a = normalize(a, S_t)
        if not improved:
            break
    return a, best_score


def pairwise_exchange(a, deadline, rng, n_trials=200):
    """Try pairwise value exchanges/mixes."""
    n = len(a)
    S_t = np.sqrt(2.0 * n)
    a = normalize(a.copy(), S_t)
    best_score = compute_obj(a)

    for _ in range(n_trials):
        if time.monotonic() > deadline:
            break
        i, j = rng.choice(n, 2, replace=False)
        ai, aj = a[i], a[j]
        if ai < 1e-8 and aj < 1e-8:
            continue
        # Try mixing
        for t in [0.3, 0.5, 0.7]:
            a[i] = (1 - t) * ai + t * aj
            a[j] = (1 - t) * aj + t * ai
            a = np.maximum(a, 0.0)
            a = normalize(a, S_t)
            s = compute_obj(a)
            if s < best_score - 1e-10:
                best_score = s
            else:
                a[i], a[j] = ai, aj
                a = normalize(a, S_t)

    return a, best_score


def optimize_sequence(a0, deadline, verbose=True):

    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)

    best = a.copy()
    best_score = compute_obj(a)
    S_t = np.sqrt(2.0 * n)

        print(f"    n={n}, initial_score={best_score:.6f}")

    v = np.zeros(n)
    mu = 0.9
    base_lr = 0.4
    no_improve = 0

        lr = base_lr * (0.99985 ** it)

        a, v = nesterov_step(a, grad, v, lr, mu)

            s = compute_obj(a)
            if s < best_score - 1e-9:

                best = a.copy()
                no_improve = 0
            else:
                no_improve += 1

                    no_improve = 0

            if g_sum < 1e-6:

                best = a.copy()
                best_score = s_new

                lr = 0.05 * (0.99 ** it_lp)
                a, v = nesterov_step(a, grad, v, max(lr, 0.005), 0.9)
                a = normalize(a, S_t)
                s = compute_obj(a)
                if s < best_score:
                    best = a.copy()
                    best_score = s

            if no_improve_lp > 10:
                # Escape: random perturbation

                a = a + rng_esc.normal(0, 0.05, n)
                a = np.maximum(a, 0.0)
                a = normalize(a, S_t)
                no_improve_lp = 0
                if compute_obj(a) < best_score:
                    best = a.copy()
                    best_score = compute_obj(a)

        if verbose:
            print(f"    P2 LP: score={best_score:.6f}, iters={it_lp}")

    # Phase 3: Simulated annealing with diverse moves
    if time.monotonic() < deadline - 40:
        rng = np.random.default_rng(777)
        a = best.copy()
        cur_s = best_score
        T = 0.01
        T_min = 1e-6

            move_type = rng.random()
            if move_type < 0.5:
                # Gaussian perturbation on random subset

                idx = rng.choice(n, size=n_p, replace=False)

            elif move_type < 0.75:
                # Multiplicative perturbation
                n_p = max(1, n // 6)
                idx = rng.choice(n, size=n_p, replace=False)
                a_new[idx] *= rng.uniform(0.6, 1.4, n_p)
            else:
                # Swap/mix two coordinates
                i, j = rng.choice(n, 2, replace=False)
                t_mix = rng.uniform(0.2, 0.8)
                old_i, old_j = a_new[i], a_new[j]
                a_new[i] = (1 - t_mix) * old_i + t_mix * old_j
                a_new[j] = (1 - t_mix) * old_j + t_mix * old_i

            a_new = np.maximum(a_new, 0.0)
            a_new = normalize(a_new, S_t)
            s_new = compute_obj(a_new)
            delta = s_new - cur_s
            if delta < 0 or rng.random() < np.exp(-delta / T):

                cur_s = s_new

        if s_pw < best_score:

            best_score = s_pw
        if verbose:

            starts.append(shuffled.copy())
    except NameError:
        pass

    # Structured starts at key lengths
    for n in [384, 320, 448]:
        starts.append(rng.random(n) * 0.5 + 0.25)
        i = np.arange(n, dtype=np.float64)
        starts.append(0.5 + 0.3 * np.cos(2 * np.pi * i * 3.0 / n))
        starts.append(np.abs(np.sin(2 * np.pi * i * 2.0 / n)) + 0.1)
        # Two-peak structure
        starts.append(np.exp(-((i - n * 0.3) / (n * 0.12)) ** 2) +
                      np.exp(-((i - n * 0.7) / (n * 0.12)) ** 2) + 0.05)

    # Deduplicate
    seen = set()
    unique_starts = []
    for s in starts:
        key = (len(s), round(float(s.sum()), 0), round(float(s.std()), 3))
        if key not in seen:
            seen.add(key)
            unique_starts.append(s)
    starts = unique_starts

    # Sort: prefer height_sequence_1 (longer, known good) first
    starts.sort(key=lambda x: (len(x) != 384, -len(x)))

    print(f"Total unique starts: {len(starts)}, budget={budget_s}s")

    # Process starts
    for idx, a0 in enumerate(starts):
        if time.monotonic() > deadline - 80:
            break

        remaining = deadline - time.monotonic()
        # Time allocation: first start gets most, later ones less
        if idx == 0:
            t_alloc = min(400.0, remaining * 0.40)
        elif idx == 1:
            t_alloc = min(250.0, remaining * 0.25)
        elif idx < 4:
            t_alloc = min(150.0, remaining * 0.15)
        else:
            t_alloc = min(100.0, remaining * 0.10)

        t_alloc = max(25.0, t_alloc)
        if time.monotonic() + t_alloc > deadline - 40:
            t_alloc = deadline - time.monotonic() - 40
            if t_alloc < 20:
                break

        print(f"\n--- Start {idx}: n={len(a0)}, alloc={t_alloc:.0f}s, remaining={remaining:.0f}s ---")
        seq, score = optimize_sequence(a0, time.monotonic() + t_alloc)

        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"  ** New best: score={best_score:.6f}, n={len(seq)}")
        else:
            print(f"  score={score:.6f} (not better than {best_score:.6f})")

    # Final intensive polish on the best sequence
    if best_seq is not None and time.monotonic() < deadline - 30:
        print(f"\n=== Final polish on best (score={best_score:.6f}, n={len(best_seq)}) ===")
        remaining = deadline - time.monotonic() - 12
        seq, score = optimize_sequence(best_seq, time.monotonic() + min(remaining, 180))
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"  Polished: score={best_score:.6f}")

        # Extra coordinate descent
        if len(best_seq) <= 512 and time.monotonic() < deadline - 15:
            a_c, s_c = golden_coord_descent(best_seq, deadline - 8, max_passes=4)
            if s_c < best_score:
                best_score = s_c
                best_seq = a_c
                print(f"  Extra coord desc: score={best_score:.6f}")

        # Extra pairwise
        if len(best_seq) <= 512 and time.monotonic() < deadline - 8:
            rng_final = np.random.default_rng(999)
            a_p, s_p = pairwise_exchange(best_seq, deadline - 5, rng_final, n_trials=200)
            if s_p < best_score:
                best_score = s_p
                best_seq = a_p
                print(f"  Extra pairwise: score={best_score:.6f}")

    # Final validation and return
    if best_seq is not None:
        final = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        fs = compute_obj(np.array(final))
        print(f"\n=== FINAL: n={len(final)}, score={fs:.6f} ===")
        return final

    print("Fallback: uniform sequence")
    return [1.0] * 384
