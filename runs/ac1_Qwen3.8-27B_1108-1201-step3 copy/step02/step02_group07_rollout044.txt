"""
Optimization for minimizing peak autocorrelation score of non-negative sequences.
Uses coordinate descent with closed-form optimal updates, simulated annealing,
and LP-based refinement. Key insight: for non-negative sequences, the max of
self-convolution is always at zero lag, so score = 2n*sum(a^2)/(sum(a))^2,
minimized at the constant sequence (score = 2). We verify this and search
for any improvements via multiple methods.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def score_fast(a):
    """Fast score computation using the zero-lag property for non-negative sequences."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    max_b = float(np.sum(a * a))
    return float(2.0 * n * max_b / (S * S))


def score_fft(a):
    """Score computation using FFT convolution for exact verification."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = rfft(a, size)
    conv = irfft(fa * fa, size)[:2 * n - 1]
    max_b = float(np.max(conv))
    return float(2.0 * n * max_b / (S * S))


def coordinate_descent(a, max_iters=2000, tol=1e-12):
    """
    Coordinate descent with closed-form optimal updates.
    For non-negative sequences, optimal a[i] = (sum_{j!=i} a[j]^2) / (sum_{j!=i} a[j]).
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    Q = float(np.sum(a * a))

    for it in range(max_iters):
        max_change = 0.0
        for i in range(n):
            A = S - a[i]
            B = Q - a[i] * a[i]
            if A > 1e-15:
                new_val = B / A
            else:
                new_val = a[i]
            new_val = max(0.0, min(1000.0, new_val))
            delta = new_val - a[i]
            if abs(delta) > 1e-14:
                max_change = max(max_change, abs(delta))
                Q += new_val * new_val - a[i] * a[i]
                S += delta
                a[i] = new_val
        if max_change < tol:
            break

    return a


def coordinate_descent_block(a, block_size, max_iters=500, tol=1e-12):
    """Block coordinate descent: optimize blocks of elements simultaneously."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    Q = float(np.sum(a * a))

    for it in range(max_iters):
        max_change = 0.0
        for start in range(0, n, block_size):
            end = min(start + block_size, n)
            idx = np.arange(start, end)
            A = S - np.sum(a[idx])
            B = Q - np.sum(a[idx] * a[idx])
            if A > 1e-15:
                # For a block, optimal values are all = B/A (same as individual)
                new_block = np.full(end - start, B / A)
            else:
                new_block = a[idx].copy()
            new_block = np.clip(new_block, 0.0, 1000.0)
            delta = np.sum(np.abs(new_block - a[idx]))
            max_change = max(max_change, delta / len(idx))
            Q += np.sum(new_block * new_block) - np.sum(a[idx] * a[idx])
            S += np.sum(new_block) - np.sum(a[idx])
            a[idx] = new_block
        if max_change < tol:
            break

    return a


def simulated_annealing(a, time_limit, seed=42):

def build_A_matrix(f):
    """Build matrix A where A @ b = conv(f, b) (full length 2n-1)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_direction(f, M, timeout=3.0):
    """Solve LP: max sum(b) s.t. conv(f,b) <= M, b >= 0."""
    n = len(f)
    if n > 2000:  # Too large for LP
        return None
    A = build_A_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lp_optimize(f_init, time_limit, t_step=0.08, verbose=False):
    """LP-based iterative optimization."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    f = np.clip(f, 0.01, 1000.0)
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 3.0:
            break

        S = float(np.sum(f))
        Q = float(np.sum(f * f))
        M = Q  # max is at zero lag for non-negative

        b = solve_lp_direction(f, M, timeout=2.0)
        if b is None:
            break

        Sb = float(np.sum(b))
        if Sb < 1e-12:
            break
        b = b * (S / Sb)

        # Line search
        best_t = t_step
        best_s = np.inf
        for t_try in [t_step, 0.5 * t_step, 0.25 * t_step, 2.0 * t_step]:
            f_cand = (1.0 - t_try) * f + t_try * b
            f_cand = np.clip(f_cand, 0.01, 1000.0)
            s_cand = score_fast(f_cand)
            if s_cand < best_s:
                best_s = s_cand
                best_t = t_try

        f = (1.0 - best_t) * f + best_t * b
        f = np.clip(f, 0.01, 1000.0)
        it += 1

        if best_s < best_score - 1e-10:
            best_score = best_s
            best_f = f.copy()
            stall = 0
        else:
            stall += 1

        if verbose and it % 20 == 0:
            print(f"    LP: it={it}, score={best_s:.6f}, best={best_score:.6f}, stall={stall}")

        if stall >= 50:
            rng_esc = np.random.default_rng(42 + it * 31)
            f += 0.05 * rng_esc.standard_normal(n)
            f = np.clip(f, 0.01, 1000.0)
            stall = 0

    return best_f, best_score


def gradient_descent(f_init, time_limit, lr=0.01, verbose=False):
    """Gradient descent on the score function (using zero-lag max)."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    f = np.clip(f, 0.01, 1000.0)
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    momentum = np.zeros(n)
    beta = 0.9
    it = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 1.0:
            break

        S = float(np.sum(f))
        Q = float(np.sum(f * f))

        # Gradient of 2n*Q/S^2 w.r.t. a[i]:
        # d/da[i] (2n * Q / S^2) = 2n * (2*a[i]*S^2 - Q*2*S) / S^4
        #                          = 4n * (a[i]*S - Q) / S^3
        grad = 4.0 * n * (f * S - Q) / (S * S * S)
        # This gradient points in the direction that increases the score
        # We want to decrease it, so we go opposite
        # But we need to be careful: we want to minimize, so gradient descent
        # Actually let's use the subgradient of max(conv) which for non-negative
        # is just 2*f at lag 0, giving gradient of Q = 2*f
        # Score = 2n*Q/S^2, d(score)/da[i] = 2n*(2*f[i]*S^2 - Q*2*S)/S^4 = 4n*f[i]*(S - Q/S)/S^3 * ... 
        # Simpler: let's just use coordinate descent which is exact

        momentum = beta * momentum + (1 - beta) * grad
        f = f - lr * momentum
        f = np.clip(f, 0.01, 1000.0)
        it += 1

        if it % 200 == 0:
            s = score_fast(f)
            if s < best_score:
                best_score = s
                best_f = f.copy()
            if verbose:
                print(f"    GD: it={it}, score={s:.6f}, best={best_score:.6f}")

    s = score_fast(f)
    if s < best_score:
        best_score = s
        best_f = f.copy()
    return best_f, best_score
