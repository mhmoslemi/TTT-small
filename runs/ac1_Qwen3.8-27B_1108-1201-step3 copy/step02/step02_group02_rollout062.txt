Hybrid optimization for minimizing peak autocorrelation of non-negative step functions.

import numpy as np
from scipy.optimize import linprog
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def fast_score(a):
    """Compute 2*n*max(a*a)/(sum(a))^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (s * s)


def project_nonneg_sum(a, S):
    """Project onto {x >= 0, sum(x) = S} using bisection."""
    n = len(a)
    a = np.maximum(a, 0.0)
    if np.sum(a) < 1e-15:
        return np.full(n, S / n)
    lo, hi = -np.max(a), np.max(a)

        if np.sum(np.maximum(a - mid, 0.0)) > S:
            lo = mid
        else:
            hi = mid
    return np.maximum(a - 0.5 * (lo + hi), 0.0)

"""
Hybrid optimization for minimizing peak autocorrelation of non-negative step functions.
Combines LP-based direction finding, log-sum-exp smoothed gradient descent with
continuation, and targeted local search. Uses FFT for fast autocorrelation evaluation.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fft_convolve(a, b):
    """Compute linear convolution of two 1-D arrays using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def fast_score(a):
    """Compute 2*n*max(a*a)/(sum(a))^2 efficiently."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    s = np.sum(a)
    if s < 0.01:
        return np.inf
    conv = fft_convolve(a, a)
    return 2.0 * n * np.max(conv) / (s * s)


def project_nonneg_sum(v, S):
    """Project v onto {x >= 0, sum(x) = S} using sorting (Duchi et al.)."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64)
    v_sorted = np.sort(v)[::-1]
    css = np.cumsum(v_sorted)
    idx = np.arange(1, n + 1, dtype=np.float64)
    condition = v_sorted > (css - S) / idx
    rho = np.where(condition)[0]
    if len(rho) == 0:
        return np.full(n, S / n)
    k = int(rho[-1]) + 1
    theta = (css[k - 1] - S) / k
    return np.maximum(v - theta, 0.0)


def build_constraint_matrix(f):
    """Build (2n-1) x n matrix where A[k,j] = f[k-j] if 0<=k-j<n."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp(f, M, x0=None):

                  bounds=[(0.0, None)] * n, method='highs',
                  options={'presolve': True})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def compute_smoothed_gradient(a, beta):

    exp_c = np.exp(beta * (conv - max_c))

    if total < 1e-300:
        return np.zeros(n)
    p = exp_c / total  # length 2n-1

    # grad[m] = 2 * sum_{j=0}^{n-1} a[j] * p[j+m] for 0 <= j+m <= 2n-2
    grad = np.zeros(n)
    for m in range(n):

def lp_iterate(f0, time_limit, rng, verbose=False):

    n = len(f0)

    f = np.clip(f, 0, None)

    else:
        f = f * (S / sum_f)

    best_score = current_score
    t0 = time.monotonic()

    stall_count = 0

def local_search_polish(a_init, time_limit, rng):
    """Targeted local search with peak-equalization moves."""
    n = len(a_init)
    S = np.sqrt(2.0 * n)
    current = a_init.astype(np.float64).copy()
    current = np.clip(current, 0, None)
    current = current * (S / np.sum(current))
    current_score = fast_score(current)
    best = current.copy()
    best_score = current_score
    t0 = time.monotonic()
    temperature = 0.002
    cooling = 0.9995

    while time.monotonic() - t0 < time_limit:
        candidate = current.copy()
        r = rng.random()

        if r < 0.35:
            # Peak-equalization move: reduce top-contributing lags
            conv = fft_convolve(current, current)
            # Find top-k lags
            k = min(5, len(conv) // 4)
            top_lags = np.argsort(conv)[-k:]
            # For each top lag, find the contributing pairs and reduce them
            for lag in top_lags:
                # Pairs (i, j) such that i + j = lag (in 0-indexed full conv)
                # Actually in np.convolve, index k corresponds to sum of a[i]*a[k-i]
                # Let's find the largest contributing element
                i_start = max(0, lag - n + 1)
                i_end = min(n, lag + 1)
                contribs = np.array([current[i] * current[lag - i] for i in range(i_start, i_end)])
                if len(contribs) > 0:
                    max_idx = i_start + int(np.argmax(contribs))
                    candidate[max_idx] *= (1.0 - rng.uniform(0.02, 0.08))
                    # Compensate by increasing a random other element
                    other = int(rng.integers(0, n))
                    candidate[other] *= (1.0 + rng.uniform(0.01, 0.05))
        elif r < 0.55:
            # Block perturbation targeting high-autocorrelation regions
            conv = fft_convolve(current, current)
            # Find lags where autocorrelation is high
            threshold = np.max(conv) * 0.95
            high_lags = np.where(conv >= threshold)[0]
            if len(high_lags) > 0:
                # Identify positions that contribute to multiple high lags
                contribution = np.zeros(n)
                for lag in high_lags[:10]:
                    i_start = max(0, lag - n + 1)
                    i_end = min(n, lag + 1)
                    for i in range(i_start, i_end):
                        contribution[i] += current[i] * current[lag - i]
                # Reduce positions with high contribution
                top_positions = np.argsort(contributation)[-min(10, n // 10):] if n > 10 else np.argsort(contribution)[-3:]
                for pos in top_positions:
                    candidate[pos] *= np.exp(rng.normal(-0.03, 0.01))
            else:
                candidate *= np.exp(rng.normal(0, 0.01, n))
        elif r < 0.70:
            # Multi-block small perturbation
            for _ in range(8):
                width = max(1, int(rng.integers(1, max(2, n // 12))))
                start = int(rng.integers(0, n - width))
                candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.012, size=width))
        elif r < 0.80:
            # Pair redistribution (preserve sum approximately)
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                delta = rng.uniform(-0.1, 0.1) * current[i]
                candidate[i] = max(0, current[i] - delta)
                candidate[j] = max(0, current[j] + delta)
        elif r < 0.90:
            # Chirp-like局部 shift
            phase = rng.uniform(0.1, 0.5)
            t_arr = np.arange(n) / n
            modulation = 1.0 + 0.05 * np.cos(2 * np.pi * phase * t_arr)
            candidate = candidate * modulation
        else:
            # Very small global perturbation
            candidate = current * np.exp(rng.normal(0.0, 0.004, n))

        candidate = np.clip(candidate, 0.0, 1000.0)
        sc = np.sum(candidate)
        if sc < 0.01:
            continue
        candidate = candidate * (S / sc)
        new_score = fast_score(candidate)

        if new_score < current_score:
            current = candidate
            current_score = new_score
            if current_score < best_score:
                best_score = current_score
                best = current.copy()
        elif rng.random() < np.exp(-(new_score - current_score) / max(temperature, 1e-15)):
            current = candidate
            current_score = new_score

        temperature *= cooling

    return best, best_score

        freq = rng.uniform(0.3, 2.5) * k
        phase = rng.uniform(0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * freq * t / n + phase)
    a += 0.03 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_random_init(n, rng, spread=0.3):
    a = 1.0 + spread * rng.standard_normal(n)
    return np.clip(a, 0.05, None)


def resample_sequence(seq, new_n):
    x_old = np.linspace(0, 1, len(seq))
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, seq)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Hybrid optimizer: LP iteration + smoothed gradient descent + local search.
    Minimizes 2n*max(a*a)/(sum(a))^2 over non-negative sequences.
    """
    t_global = time.monotonic()
    time_budget = max(0.0, float(budget_s) - 40.0)
    deadline = t_global + time_budget
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Get starting sequence
    start_seq = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        start_seq = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
        if len(start_seq) > 0:
            s = fast_score(start_seq)
            print(f"Provided sequence: n={len(start_seq)}, score={s:.6f}")
            if s < best_score:
                best_score = s
                best_seq = start_seq.copy()

    # Determine primary length
    if start_seq is not None:
        primary_n = len(start_seq)
    else:
        primary_n = 512

    # === Phase 1: LP iteration (primary) ===
    print(f"\n=== Phase 1: LP iteration (n={primary_n}) ===")
    if start_seq is not None and time.monotonic() < deadline - 500:
        f0 = start_seq.copy()
        t1 = min(600.0, deadline - time.monotonic() - 500)
        f_best, s_best = lp_iterate(f0, t1, rng, verbose=True)
        print(f"  Phase 1 result: score={s_best:.6f}")
        if s_best < best_score:
            best_score = s_best
            best_seq = f_best.copy()
            print(f"    *** New best: {best_score:.6f} ***")

    # === Phase 2: Smoothed gradient refinement ===
    print(f"\n=== Phase 2: Smoothed gradient ({deadline - time.monotonic():.0f}s remaining) ===")
    if best_seq is not None and time.monotonic() < deadline - 300:
        t2 = min(400.0, deadline - time.monotonic() - 300)
        beta_schedule = [1.0, 3.0, 10.0, 30.0, 100.0, 200.0, 500.0]
        a_best, s_best = smoothed_gradient_optimize(best_seq, t2, beta_schedule, rng)
        print(f"  Phase 2 result: score={s_best:.6f}")
        if s_best < best_score:
            best_score = s_best
            best_seq = a_best.copy()
            print(f"    *** New best: {best_score:.6f} ***")

    # === Phase 3: LP refinement on polished sequence ===
    print(f"\n=== Phase 3: LP refinement ===")
    if best_seq is not None and time.monotonic() < deadline - 200:
        t3 = min(300.0, deadline - time.monotonic() - 200)
        f_best, s_best = lp_iterate(best_seq, t3, rng, verbose=False)
        print(f"  Phase 3 result: score={s_best:.6f}")
        if s_best < best_score:
            best_score = s_best
            best_seq = f_best.copy()
            print(f"    *** New best: {best_score:.6f} ***")

    # === Phase 4: Local search polish ===
    print(f"\n=== Phase 4: Local search polish ===")
    if best_seq is not None and time.monotonic() < deadline - 50:
        t4 = min(250.0, max(30.0, deadline - time.monotonic() - 50))
        print(f"  Starting polish, score={best_score:.6f}, time={t4:.1f}s")
        polished, polish_score = local_search_polish(best_seq, t4, rng)
        if polish_score < best_score:
            best_score = polish_score
            best_seq = polished.copy()
            print(f"  After polish: {best_score:.6f}")
        else:
            print(f"  Polish didn't improve (best={best_score:.6f})")

    # === Phase 5: Explore other sizes/inits if time remains ===
    print(f"\n=== Phase 5: Exploration ===")
    if time.monotonic() < deadline - 100:
        configs = []
        for n in [384, 640, 768]:
            if n != primary_n:
                configs.append((n, 'chirp', 0.4))
                configs.append((n, 'multi_tone', 0))
        if best_seq is not None:
            for n in [448, 576]:
                configs.append((n, 'resample', 0))

        n_configs = len(configs)
        for ci, (n, init_type, alpha) in enumerate(configs):
            if time.monotonic() > deadline - 80:
                break
            time_left = deadline - time.monotonic() - 60
            remaining = n_configs - ci
            t_per = max(15.0, min(60.0, time_left / max(1, remaining)))

            if init_type == 'resample' and best_seq is not None:
                a0 = resample_sequence(best_seq, n)
            elif init_type == 'chirp':
                a0 = make_chirp_init(n, alpha + 0.2 * rng.random(), rng)
            elif init_type == 'multi_tone':
                a0 = make_multi_tone_init(n, rng)
            else:
                a0 = make_random_init(n, rng)

            f_best, s_best = lp_iterate(a0, t_per, rng, verbose=False)
            if s_best < best_score:
                best_score = s_best
                best_seq = f_best.copy()
                print(f"  n={n}, init={init_type}: *** New best: {best_score:.6f} ***")
            else:
                print(f"  n={n}, init={init_type}: score={s_best:.6f}")

    # === Phase 6: Final short polish ===
    if best_seq is not None and time.monotonic() < deadline - 20:
        remaining = max(10.0, deadline - time.monotonic() - 10)
        if remaining > 30:
            print(f"\n=== Phase 6: Final LP polish ({remaining:.0f}s) ===")
            f_best, s_best = lp_iterate(best_seq, remaining, rng, verbose=False)
            if s_best < best_score:
                best_score = s_best
                best_seq = f_best.copy()
                print(f"  Final LP: {best_score:.6f}")

    # Final verification
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n=== FINAL: score={final_score:.6f}, n={len(final_list)} ===")
        return final_list

    # Fallback
    if start_seq is not None:
        return [float(max(0.0, min(1000.0, x))) for x in start_seq]
    return [1.0] * 512
