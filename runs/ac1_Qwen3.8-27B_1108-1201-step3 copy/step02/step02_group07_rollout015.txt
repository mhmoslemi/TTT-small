Optimized search for minimum peak autocorrelation of non-negative sequences.
Combines LP-based direction finding with soft-max gradient polishing
and multi-scale time allocation.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]

    exp_s = np.exp(shifted)
    p = exp_s / np.sum(exp_s)
    
    # Gradient: g[j] = sum_k p[k] * a[k-j] (for valid k-j) + diagonal terms
    g = np.zeros(n)
    # Focus on top-k lags for efficiency
    top_idx = np.argsort(p)[-top_k:]
    for ki in top_idx:
        ki = int(ki)
        wk = p[ki]
        lo = max(0, ki - (n - 1))
        hi = min(n - 1, ki)
        for j in range(lo, hi + 1):
            g[j] += wk * a[ki - j]
        if ki % 2 == 0:
            jd = ki // 2
            if 0 <= jd < n:
                g[jd] += wk * a[jd]
    return g, c_max + np.log(np.sum(exp_s)) / beta


def gradient_polish(f_init, time_limit, beta_start=5.0, beta_end=50.0, verbose=True):
    """Soft-max gradient descent with momentum."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    S = float(np.sum(f))
    f = f / S  # normalize to sum=1
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    momentum = np.zeros(n)
    beta_m = 0.85
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 1.0:
            break
        
        progress = (time.monotonic() - t0) / time_limit
        beta = beta_start + (beta_end - beta_start) * progress ** 0.5
        
        g, _ = gradient_step(f, beta, top_k=30)
        
        # Adaptive learning rate
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-10:
            break
        lr = 0.01 / (1.0 + g_norm * 10)
        
        momentum = beta_m * momentum + (1.0 - beta_m) * g
        f = f - lr * momentum
        f = np.clip(f, 1e-6, None)
        f = f / float(np.sum(f))
        
        it += 1
        if it % 100 == 0:
            s = score_fast(f)
            if s < best_score:
                best_score = s
                best_f = f.copy()
                if verbose:
                    print(f"  [grad {time.monotonic()-t0:6.1f}s] it={it} score={s:.6f} best={best_score:.6f}")
    
    s = score_fast(f)
    if s < best_score:
        best_score = s
        best_f = f.copy()
    return best_f, best_score


def lp_optimize(f_init, time_limit, verbose=True):
    """LP-based iterative optimization."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    it = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 3.0:
            break
        
        c = fft_convolve(f, f)
        M = float(np.max(c))
        
        b = lp_direction(f, M, timeout=4.0)
        if b is None:
            # Try with slightly larger M
            b = lp_direction(f, M * 1.01, timeout=4.0)
            if b is None:
                break
        
        S = float(np.sum(f))
        Sb = float(np.sum(b))
        if Sb < 1e-12:
            break
        b = b * (S / Sb)
        
        # Line search
        best_t = 0.1
        best_s = np.inf
        for t_try in [0.1, 0.05, 0.2, 0.3]:
            f_cand = (1.0 - t_try) * f + t_try * b
            f_cand = np.clip(f_cand, 0.0, 1000.0)
            s_cand = score_fast(f_cand)
            if s_cand < best_s:
                best_s = s_cand
                best_t = t_try
        
        f = (1.0 - best_t) * f + best_t * b
        f = np.clip(f, 0.0, 1000.0)
        it += 1
        
        if best_s < best_score - 1e-10:
            best_score = best_s
            best_f = f.copy()
            stall = 0
        else:
            stall += 1
        
        if verbose and it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    [{elapsed:6.1f}s] it={it:4d} score={best_s:.6f} best={best_score:.6f}")
        
        if stall >= 30:
            # Escape: add noise
            rng = np.random.default_rng(42 + it)
            f += 0.02 * rng.standard_normal(n)
            f = np.clip(f, 0.01, 1000.0)
            stall = 0
    
    return best_f, best_score


def make_init(n, rng, init_type='hann'):
    """Generate initial sequence."""
    t = np.arange(n, dtype=np.float64)
    if init_type == 'hann':
        w = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / max(1, n - 1)))
        a = w + 0.1 * rng.standard_normal(n)
    elif init_type == 'bartlett':
        w = 1.0 - np.abs(2.0 * t / max(1, n - 1) - 1.0)
        a = 0.2 + 0.8 * w + 0.1 * rng.standard_normal(n)
    elif init_type == 'chirp':
        w = 0.5 * (1.0 + np.cos(np.pi * t / max(1, n - 1)))
        a = 0.5 + 0.3 * w + 0.3 * np.cos(2.0 * np.pi * t**2 / (2.0 * n)) + 0.05 * rng.standard_normal(n)
    elif init_type == 'random':
        a = 0.5 + rng.random(n)
    elif init_type == 'exponential':
        a = np.exp(-0.5 * t / n) * (1.0 + 0.2 * rng.standard_normal(n))
    else:
        a = np.ones(n) + 0.1 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)
