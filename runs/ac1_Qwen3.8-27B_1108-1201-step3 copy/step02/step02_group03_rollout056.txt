Optimizes non-negative sequences to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses LP-based alternating optimization with soft-max gradient polish.
"""

import numpy as np
import time
from scipy.optimize import linprog

def compute_obj(a):
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))

def normalize(a, target=None):
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)

def lp_solve(a, M, n=None):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M (elementwise), g >= 0."""
    if n is None:
        n = len(a)
    
    c_obj = -np.ones(n)
    # Build constraint matrix
    # A[m, j] = a[m-j] if 0 <= m-j < n
    A = np.zeros((2*n-1, n))
    for j in range(n):
        for m in range(j, j+n):
            A[m, j] = a[m-j]
    
    b = np.full(2*n-1, M)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.status == 0:
            return res.x
    except Exception as e:
        print(f"    LP error: {e}")
    return None

def gradient_softmax(a, tau):
    """Compute gradient of soft-max of conv(a,a)."""
    n = len(a)
    c = np.convolve(a, a)
    c_shifted = c - c.max()
    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= w_sum
    
    a_rev = a[::-1]
    conv_result = np.convolve(w, a_rev, mode='full')
    grad = 2.0 * conv_result[n-1:n-1+n]
    return grad, w

    n = len(a0)

        c = np.convolve(a, a)
        M = float(c.max())

        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-6:
            break
        
        g_norm = g * (S_t / g_sum)
        
        # Try multiple blending factors
        improved = False
        for t_blend in [1.0, 0.7, 0.5, 0.3, 0.15, 0.05]:
            a_new = (1.0 - t_blend) * a + t_blend * g_norm
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize(a_new, S_t)
            s_new = compute_obj(a_new)
            if s_new < best_score - 1e-10:
                a = a_new
                best = a.copy()
                best_score = s_new

    # Phase 2: Soft-max gradient descent
    t_phase2_end = time.monotonic() + (deadline - time.monotonic()) * 0.35

    momentum = np.zeros(n)

        a_new = a + momentum
        a_new = np.maximum(a_new, 0.0)

                a = a_new

        else:

    if verbose:

    # Phase 3: Final LP polish

                if s_new < best_score - 1e-10:
                    best_score = s_new

Score = 2n * (1/n) / 1 = 2.

For a non-uniform sequence, the peak could be higher or lower. The question is: can we make the peak LOWER than 1/n?

At k = n-1 (the middle): `conv(a,a)[n-1] = sum_{i=0}^{n-1} a[i]*a[n-1-i]`
By Cauchy-Schwarz: `sum a[i]*a[n-1-i] <= sqrt(sum(a[i]^2) * sum(a[n-1-i]^2)) = sum(a[i]^2)`
And `sum(a[i]^2) >= 1/n` (by Cauchy-Schwarz on sum a[i] = 1).

So the middle element >= 1/n... wait, that's the wrong direction. We have:
- `sum(a[i]^2) >= 1/n` (equality for uniform)
- `conv(a,a)[n-1] = sum a[i]*a[n-1-i] <= sum(a[i]^2)` (by Cauchy-Schwarz, with equality when a is palindromic)

So `conv(a,a)[n-1]` could be LESS than `sum(a[i]^2)` if a is not palindromic. And it could be less than 1/n!

For example, if a is supported on the first half: a[i] = 2/n for i = 0,...,n/2-1, and 0 for i >= n/2.
Then `conv(a,a)[n-1] = sum_{i=0}^{n/2-1} a[i]*a[n-1-i]`. But a[n-1-i] = 0 for i <= n/2-1 (since n-1-i >= n/2). So `conv(a,a)[n-1] = 0`!

But other elements might be large. `conv(a,a)[n/2-1] = sum_{i=0}^{n/2-1} a[i]*a[n/2-1-i] = sum_{i=0}^{n/2-1} (2/n)*(2/n) = (n/2)*(4/n^2) = 2/n`.

So the peak is 2/n, giving score = 2n * 2/n = 4. Much worse.

The issue is that concentrating mass creates large peaks at other lags.

So the optimal sequence needs to:
1. Be non-palindromic (to reduce the middle element below sum(a^2))
2. Be spread out (to avoid large peaks at other lags)

This is a delicate balance. The optimization approach (LP + gradient) should find the right balance.

"""
Optimizes non-negative sequences to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses LP-based alternating optimization (minimize cross-peak formulation),
soft-max gradient descent with adaptive temperature, and multi-start strategy.
Key improvements: direct peak-minimizing LP, tighter M scheduling, 
top-k soft-max gradient, and broader length exploration.
"""

import numpy as np
import time
from scipy.optimize import linprog


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    c = np.convolve(a, a)
    max_c = float(c.max())
    S = float(a.sum())
    if S < 0.01:
        return float('inf')
    return float(2.0 * n * max_c / (S * S))


def normalize(a, target=None):
    """Normalize sequence to have a specific sum."""
    if target is None:
        target = np.sqrt(2.0 * len(a))
    S = a.sum()
    if S < 1e-12:
        return a.copy()
    return a * (target / S)


def lp_min_peak(a, S, n=None):
    """
    Solve LP: minimize M' subject to conv(a,g)[k] <= M' for all k,
    sum(g) = S, g >= 0.
    Returns (g, M_prime) or (None, None).
    """
    if n is None:
        n = len(a)
    if n > 2048:
        return None, None

    # Variables: [g_0, ..., g_{n-1}, M']
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # Minimize M'

    # Inequality constraints: A[m,j] * g[j] - M' <= 0 for each lag m
    # A matrix: (2n-1) x (n+1)
    A_ub = np.zeros((2 * n - 1, n + 1))
    for j in range(n):
        for m in range(j, j + n):
            A_ub[m, j] = a[m - j]
    A_ub[:, -1] = -1.0  # -M'

    b_ub = np.zeros(2 * n - 1)

    # Equality: sum(g) = S
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([S])

    bounds = [(0.0, None)] * n + [(0.0, None)]

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if res.status == 0:
            return res.x[:n], float(res.x[n])
    except Exception:
        pass
    return None, None


def lp_max_sum(a, M, n=None):
    """
    Solve LP: max sum(g) subject to conv(a,g)[k] <= M for all k, g >= 0.
    Returns g or None.
    """
    if n is None:
        n = len(a)
    if n > 2048:
        return None

    c_obj = -np.ones(n)

    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        for m in range(j, j + n):
            A[m, j] = a[m - j]

    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return None


def gradient_topk(a, k=5, tau=None):
    """
    Compute gradient using top-k peak weighting.
    If tau is provided, use soft-max instead.
    """
    n = len(a)
    c = np.convolve(a, a)

    if tau is not None:
        c_shifted = c - c.max()
        w = np.exp(tau * c_shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0
        else:
            w /= w_sum
        a_rev = a[::-1]
        conv_result = np.convolve(w, a_rev, mode='full')
        grad = 2.0 * conv_result[n - 1:n - 1 + n]
        return grad

    # Top-k approach
    c_sorted = np.sort(c)[::-1]
    threshold = c_sorted[min(k, len(c_sorted)) - 1]
    w = np.where(c >= threshold, 1.0 / k, 0.0)
    # Smooth weights: use exponential based on rank
    if k > 1:
        ranks = np.argsort(np.argsort(c))[::-1].astype(float)
        w = np.exp(-0.5 * ranks / k)
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w = np.zeros_like(c)
            w[int(np.argmax(c))] = 1.0

    a_rev = a[::-1]
    conv_result = np.convolve(w, a_rev, mode='full')
    grad = 2.0 * conv_result[n - 1:n - 1 + n]
    return grad


def optimize_sequence(a0, deadline, verbose=True):
    """Full optimization pipeline for a single sequence."""
    n = len(a0)
    a = np.clip(np.asarray(a0, dtype=np.float64), 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = np.ones(n) * 0.5
    a = normalize(a)
    S_t = np.sqrt(2.0 * n)

    score = compute_obj(a)
    best = a.copy()
    best_score = score

    if verbose:
        print(f"    Start: n={n}, score={score:.6f}")

    remaining = deadline - time.monotonic()
    if remaining < 10:
        return best, best_score

    # === Phase 1: LP alternating optimization (min-peak formulation) ===
    t_end1 = time.monotonic() + remaining * 0.45
    it_lp = 0
    no_improve = 0
    M_current = float(np.convolve(a, a).max())

    while time.monotonic() < t_end1 and it_lp < 800:
        it_lp += 1
        c = np.convolve(a, a)
        M_current = float(c.max())

        # Solve min-peak LP
        g, M_prime = lp_min_peak(a, S_t)
        if g is None:
            # Fallback to max-sum LP
            g = lp_max_sum(a, M_current)
            if g is None:
                break
            M_prime = M_current

        g_sum = g.sum()
        if g_sum < 1e-6 or M_prime < 1e-12:
            break

        # If M' is already close to current peak and g is close to a, we're at a fixed point
        g_norm = g * (S_t / g_sum)

        # Try blending
        improved = False
        blend_factors = [0.5, 0.3, 0.7, 0.2, 0.8, 0.1, 0.4]
        for t_blend in blend_factors:
            a_new = (1.0 - t_blend) * a + t_blend * g_norm
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize(a_new, S_t)
            s_new = compute_obj(a_new)
            if s_new < best_score - 1e-10:
                a = a_new
                best = a.copy()
                best_score = s_new
                improved = True
                break

        if not improved:
            no_improve += 1
            # Try a small gradient nudge to escape
            if no_improve in [5, 10, 20]:
                grad = gradient_topk(a, k=3)
                grad -= grad.mean()
                a_new = a - 0.05 * grad
                a_new = np.maximum(a_new, 0.0)
                a_new = normalize(a_new, S_t)
                s_new = compute_obj(a_new)
                if s_new < best_score - 1e-10:
                    best_score = s_new
                    best = a_new.copy()
                    a = a_new
                    no_improve = 0
                elif no_improve >= 20:
                    break
        else:
            no_improve = 0

        if verbose and it_lp % 40 == 0:
            print(f"    LP min-peak iter {it_lp}: score={best_score:.6f}, M'={M_prime:.6f}")

    if verbose:
        print(f"    After LP: score={best_score:.6f}, iters={it_lp}")

    # === Phase 2: Soft-max gradient descent ===
    remaining = deadline - time.monotonic()
    if remaining > 20:
        t_end2 = time.monotonic() + remaining * 0.4
        a = best.copy()
        momentum = np.zeros(n)
        mu = 0.9
        no_improve = 0

        it = 0
        while time.monotonic() < t_end2 and it < 50000:
            it += 1
            # Adaptive temperature: start broad, sharpen
            tau = 5.0 + 50.0 * (it / 3000.0)
            tau = min(tau, 200.0)
            lr = 0.08 * (0.9999 ** it)
            lr = max(lr, 0.002)

            grad = gradient_topk(a, k=5, tau=tau)
            grad -= grad.mean()

            # Normalized gradient step
            g_norm_val = np.linalg.norm(grad)
            if g_norm_val > 1e-12:
                grad = grad / g_norm_val * 0.1

            momentum = mu * momentum - lr * grad
            a_new = a + momentum
            a_new = np.maximum(a_new, 0.0)
            a_new = normalize(a_new, S_t)

            if it % 50 == 0:
                s = compute_obj(a_new)
                if s < best_score - 1e-9:
                    best_score = s
                    best = a_new.copy()
                    a = a_new
                    no_improve = 0
                else:
                    no_improve += 1
                    if no_improve > 8:
                        # Reset and try harder
                        momentum[:] = 0
                        if it > 10000:
                            break
                        no_improve = 0
            else:
                a = a_new

        a = best.copy()
        if verbose:
            print(f"    After grad: score={best_score:.6f}, iters={it}")

    # === Phase 3: Final LP polish ===
    remaining = deadline - time.monotonic()
    if remaining > 15 and n <= 1024:
        t_end3 = time.monotonic() + remaining * 0.7
        a = best.copy()
        for it_polish in range(50):
            if time.monotonic() > t_end3:
                break
            c = np.convolve(a, a)
            M_cur = float(c.max())

            g, M_p = lp_min_peak(a, S_t)
            if g is None:
                g = lp_max_sum(a, M_cur)
                if g is None:
                    break

            g_sum = g.sum()
            if g_sum < 1e-6:
                break
            g_norm = g * (S_t / g_sum)

            improved = False
            for t_blend in [0.3, 0.5, 0.2, 0.4, 0.6]:
                a_new = (1.0 - t_blend) * a + t_blend * g_norm
                a_new = np.maximum(a_new, 0.0)
                a_new = normalize(a_new, S_t)
                s_new = compute_obj(a_new)
                if s_new < best_score - 1e-10:
                    best_score = s_new
                    best = a_new.copy()
                    a = a_new
                    improved = True
                    break

            if not improved:
                # Small perturbation
                if it_polish % 5 == 4:
                    noise = np.random.default_rng(it_polish).normal(0, 0.01, n)
                    a_new = np.maximum(a + noise, 0)
                    a_new = normalize(a_new, S_t)
                    s_new = compute_obj(a_new)
                    if s_new < best_score:
                        best_score = s_new
                        best = a_new.copy()
                        a = a_new
                if it_polish > 30 and not improved:
                    break

    return best, best_score


def generate_starts(rng, lengths):
    """Generate diverse starting sequences."""
    starts = []

    # From height_sequence_1 if available
    try:
        gs = np.asarray(height_sequence_1, dtype=np.float64)
        gs = gs[np.isfinite(gs)]
        if gs.size >= 16:
            for L in lengths:
                if L <= len(gs):
                    starts.append(gs[:L].copy())
                else:
                    # Interpolate/pad
                    frac = np.linspace(0, len(gs) - 1, L)
                    idx = np.floor(frac).astype(int)
                    starts.append(gs[idx])
            # Reversed
            for L in lengths[:2]:
                if L <= len(gs):
                    starts.append(gs[:L][::-1].copy())
            # Perturbed
            for L in lengths[:2]:
                if L <= len(gs):
                    p = gs[:L] + rng.normal(0, 0.05, L)
                    starts.append(np.maximum(p, 0))
    except NameError:
        pass

    # Random starts
    for n in lengths:
        starts.append(rng.random(n) * 0.4 + 0.3)
        starts.append(rng.random(n) * 0.6 + 0.2)

    # Structured starts
    for n in lengths[:2]:
        i = np.arange(n, dtype=np.float64)
        starts.append(0.4 + 0.3 * np.cos(2 * np.pi * i * 3.0 / n))
        starts.append(0.5 + 0.25 * np.sin(2 * np.pi * i * 7.0 / n))
        starts.append(np.abs(np.sin(2 * np.pi * i * 1.0 / n)) * 0.4 + 0.2)
        starts.append(0.5 + 0.3 * np.cos(2 * np.pi * i * 5.0 / n + 1.0))
        # Two-scale structure
        starts.append(0.3 + 0.2 * np.cos(2 * np.pi * i * 11.0 / n) +
                      0.15 * np.cos(2 * np.pi * i * 23.0 / n))

    return starts
