import numpy as np
import time
from scipy.optimize import linprog, minimize
from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lp_direction(a, M, n_aug=0):
    """
    Solve LP: max sum(g) s.t. g >= 0, conv(a,g)[k] <= M for all k.
    Uses scipy linprog.
    """
    n = len(a)
    size = 2 * n - 1
    
    # conv(a,g)[k] = sum_i a[i]*g[k-i] (valid indices)
    # We need: sum_i a[i]*g[k-i] <= M for all k
    # This is A @ g <= M * ones(size)
    # A is size x n, where A[k, j] = a[j] if k-j is in [0,n-1] and a[k-j]... 
    # Actually conv(a,g)[k] = sum_j a[j] * g[k-j] where both indices valid
    # = sum_j a[j] * g[k-j] for j in [max(0,k-n+1), min(n-1,k)]
    # So A[k,j] = a[j] if k-j in [0,n-1], else 0
    # Wait, that's not right either. Let me think again.
    # conv(a,g)[k] = sum_i a[i] * g[k-i] where i in [0,n-1] and k-i in [0,n-1]
    # So for a fixed k, the terms are: a[i] * g[k-i] for i in [max(0, k-(n-1)), min(n-1, k)]
    # In matrix form: (A @ g)[k] where A[k, j] = a[k-j] if k-j in [0,n-1], else 0
    # Wait no. (A @ g)[k] = sum_j A[k,j] * g[j]
    # We want (A @ g)[k] = sum_i a[i] * g[k-i]
    # Let j = k-i, so i = k-j: sum_j a[k-j] * g[j] where k-j in [0,n-1] and j in [0,n-1]
    # So A[k,j] = a[k-j] if 0 <= k-j <= n-1, else 0
    
    # Build A sparsely
    from scipy.sparse import diags, csr_matrix
    # For each k, A[k,:] has a reversed shifted
    # A is a "flip Toeplitz" matrix
    # Row k: a[k], a[k-1], ..., a[0], 0, ..., 0, 0, ..., 0 (shifted)
    # Actually A[k, j] = a[k-j] for max(0, k-n+1) <= j <= min(n-1, k)
    
    # Let's build it using the convolution structure
    # A @ g = conv(a, g) where both are length n
    # This is a Toeplitz matrix (with the flip)
    
    # Build A as dense for simplicity (n up to ~2048, size up to ~4095)
    # For n=256, A is 511 x 256 = 130K entries - fine
    # For n=2048, A is 4095 x 2048 = 8.4M entries - still ok
    
    A = np.zeros((size, n))
    for k in range(size):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        A[k, idx] = a[k - idx]
    
    # LP: minimize -sum(g) subject to A@g <= M*ones, g >= 0
    c = -np.ones(n)
    A_ub = A
    b_ub = np.full(size, M)
    
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x
    else:
        # Fallback: return uniform
        return np.ones(n) / n

def lp_iterate(a, time_limit, t_init=0.1, t_decay=0.999):
    """
    Iterative LP-based optimization.
    """
    a = a.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    t0 = time.monotonic()
    t = t_init
    
    while time.monotonic() - t0 < time_limit:
        # Solve LP
        g = lp_direction(a, M, n)
        
        if g is None or not np.all(np.isfinite(g)):
            break
        
        g = np.maximum(g, 0)
        gs = g.sum()
        if gs < 1e-15:
            break
        g = g / gs
        
        # Update
        a_new = (1 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        conv = fast_convolve(a_new)
        M_new = conv.max()
        obj_new = 2.0 * n * M_new
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        M = M_new
        t *= t_decay
    
    return best_a, best_obj

def subgradient_refine(a0, time_limit, top_k=20, temp_init=0.001):

        scaled -= scaled.max()

        # Center for simplex

        v = momentum * v - lr * sg

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP direction finding,
smooth log-sum-exp optimization, subgradient descent, and coordinate
refinement. Multi-length, multi-initialization search strategy.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

# All helper functions at top level

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(a, conv, temperature, top_k):
    """Compute weighted subgradient using top-k peaks."""
    n = len(a)
    size = len(conv)
    
    if top_k >= size:
        peaks = np.arange(size)
        peak_vals = conv.copy()
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]
    
    if temperature > 1e-15:
        scaled = peak_vals / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum
    else:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        wi = 2.0 * w[i]
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        sg[idx] += wi * a[k - idx]
    
    return sg

def subgradient_refine(a0, time_limit, top_k=20, temp_init=0.001, lr_init=0.02, momentum=0.85):
    """Subgradient descent on the exact max-convolution objective."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    v = np.zeros(n)
    temp = temp_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        sg = compute_subgradient(a, conv, temp, top_k)
        sg = sg - sg.mean()  # center for simplex
        
        v = momentum * v - lr * sg
        a = project_simplex(a + v)
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

            rng = np.random.default_rng()
            noise = rng.standard_normal(n) * 0.005
            a = project_simplex(a + noise)
            v *= 0.3
            no_improve = 0
            lr = max(0.003, lr * 0.5)
        
        lr *= 0.9998
        temp *= 1.0005
    
    return best_a, best_obj

def lp_solve_direction(a, M):
    """
    Solve LP: max sum(g) s.t. g >= 0, conv(a,g)[k] <= M for all k.
    Returns the optimal g or None.
    """
    n = len(a)
    size = 2 * n - 1
    
    # Build constraint matrix: A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # (A @ g)[k] = sum_j a[k-j] * g[j] = conv(a, g)[k]
    # Wait, let me verify: conv(a,g)[k] = sum_i a[i]*g[k-i] (valid i)
    # = sum_j a[k-j]*g[j] (valid j, with i=k-j)
    # So A[k,j] = a[k-j] if 0 <= k-j < n, else 0
    
    A = np.zeros((size, n))
    for k in range(size):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        A[k, idx] = a[k - idx]
    
    # LP: minimize -sum(g) subject to A@g <= M*ones, g >= 0
    c = -np.ones(n)
    b_ub = np.full(size, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    
    return None

def lp_iterate(a0, time_limit, t_init=0.1):
    """LP-based iterative optimization."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    t = t_init
    
    iterations = 0
    while time.monotonic() - t0 < time_limit:
        M = conv.max()
        g = lp_solve_direction(a, M)
        
        if g is None:
            break
        
        g = np.maximum(g, 0)
        gs = g.sum()
        if gs < 1e-15:
            break
        g = g / gs
        
        # Check if g is similar to a (convergence)
        cos_sim = np.dot(a, g) / (np.linalg.norm(a) * np.linalg.norm(g) + 1e-15)
        
        a_new = (1 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        conv = fast_convolve(a_new)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a_new.copy()
        
        a = a_new
        iterations += 1
        
        # Adaptive step size
        if cur_obj > best_obj * 1.01:
            t *= 0.8
        elif cos_sim > 0.999:
            break  # converged
        
        if time.monotonic() - t0 > time_limit:
            break
    
    return best_a, best_obj

def smooth_lse_optimize(a0, time_limit, temperature=10.0):
    """
    Smooth optimization using log-sum-exp approximation of max.
    Uses projected gradient with momentum on the simplex.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = 0.01
    v = np.zeros(n)
    momentum = 0.9
    temp = temperature
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        
        # Log-sum-exp gradient
        t_conv = temp * conv
        m = t_conv.max()
        exp_t = np.exp(t_conv - m)
        w = exp_t / exp_t.sum()
        
        # Gradient of LSE
        sg = np.zeros(n)
        thresh = w.max() * 0.01
        active = np.where(w > thresh)[0]
        for k_int in active:
            k = int(k_int)
            wi = w[k_int] * temp
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += wi * a[k - idx]
        
        sg = sg - sg.mean()
        
        v = momentum * v - lr * sg
        a = project_simplex(a + v)
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 500:
            no_improve = 0
            lr = max(0.002, lr * 0.6)
            temp = max(2.0, temp * 0.7)
        
        lr *= 0.9999
        momentum = min(0.97, momentum + 0.0001)
    
    return best_a, best_obj

def coordinate_refine(a0, time_limit, step_sizes=None):
    """
    Coordinate-wise local search: try perturbing each element slightly.
    Uses incremental convolution update for efficiency.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)

    n = len(a)

    t0 = time.monotonic()
    improved = True
    
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            for delta in step_sizes:
                a_new = a.copy()
                a_new[j] = max(0.0, a_new[j] + delta)
                s_new = a_new.sum()
                if s_new < 1e-10:
                    continue
                a_new = a_new / s_new
                
                # Incremental convolution update
                # conv_new[k] = conv[k] + 2 * delta_norm * a[k-j] (for valid k-j)
                # But we also need to account for the re-normalization...
                # For simplicity, just recompute

                    best_a = a_new.copy()
                    a = a_new.copy()
                    conv = conv_new
                    improved = True

    return best_a, best_obj

def generate_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp
    inits['ramp_rev'] = ramp[::-1].copy()
    
    tri = np.minimum(x + 1, n - x)
    tri /= tri.sum()
    inits['triangle'] = tri
    
    for sf in [0.2, 0.4, 0.6]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf}'] = g
    
    # Edge-concentrated
    edge = max(1, n // 5)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi
    
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                          np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                          np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps
    
    return inits

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_weighted_subgradient(a, conv, temperature, top_k):
    """Compute weighted subgradient using top-k convolution peaks."""
    n = len(a)
    size = len(conv)

    if top_k >= size:
        peaks = np.arange(size)
        peak_vals = conv.copy()
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]

    if temperature > 1e-18:
        scaled = peak_vals / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-18:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum
    else:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        wi = 2.0 * w[i]
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        sg[idx] += wi * a[k - idx]

    return sg


def subgradient_refine(a0, time_limit, top_k=20, temp_init=0.001, lr_init=0.02, momentum=0.85):
    """Subgradient descent on the exact max-convolution objective with momentum."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    v = np.zeros(n)
    temp = temp_init
    no_improve = 0
    perturb_count = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        sg = compute_weighted_subgradient(a, conv, temp, top_k)
        sg = sg - sg.mean()

        v = momentum * v - lr * sg
        a = project_simplex(a + v)

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 800:
            rng_local = np.random.default_rng(perturb_count * 1000)
            scale = 0.003 + 0.002 * perturb_count
            noise = rng_local.standard_normal(n) * scale
            a = project_simplex(a + noise)
            v *= 0.3
            no_improve = 0
            lr = max(0.003, lr * 0.6)
            perturb_count += 1

        lr *= 0.99975
        temp *= 1.0004

    return best_a, best_obj


def lp_solve_direction(a, M):
    """
    Solve LP: max sum(g) s.t. g >= 0, conv(a,g)[k] <= M for all k.
    Returns optimal g or None if infeasible/failed.
    """
    n = len(a)
    size = 2 * n - 1

    # Build constraint matrix A where (A@g)[k] = conv(a,g)[k]
    # A[k,j] = a[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((size, n))
    for k in range(size):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_end >= j_start:
            idx = np.arange(j_start, j_end + 1)
            A[k, idx] = a[k - idx]

    c = -np.ones(n)
    b_ub = np.full(size, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass

    return None


def lp_iterate(a0, time_limit, t_init=0.1):
    """LP-based iterative optimization using direction finding."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    t = t_init
    stall_count = 0

    while time.monotonic() - t0 < time_limit:
        M = conv.max()
        g = lp_solve_direction(a, M)

        if g is None:
            break

        gs = g.sum()
        if gs < 1e-12:
            break
        g = g / gs

        # Check convergence: if g is nearly parallel to a, we've stalled
        norm_a = np.linalg.norm(a)
        norm_g = np.linalg.norm(g)
        if norm_a * norm_g > 1e-15:
            cos_sim = np.dot(a, g) / (norm_a * norm_g)
        else:
            cos_sim = 1.0

        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        conv = fast_convolve(a_new)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1

        a = a_new

        # Adapt step size
        if cur_obj > best_obj * 1.005:
            t = max(0.01, t * 0.7)
        elif stall_count > 20:
            t = max(0.02, t * 0.8)
        elif cos_sim > 0.9995:
            break

        if time.monotonic() - t0 > time_limit:
            break

    return best_a, best_obj


def smooth_lse_optimize(a0, time_limit, temperature=10.0):
    """
    Smooth optimization using log-sum-exp approximation of max.
    Projected gradient with Nesterov-like momentum on the simplex.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = 0.008
    v = np.zeros(n)
    momentum = 0.9
    temp = temperature
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)

        # Log-sum-exp weights
        t_conv = temp * conv
        m = t_conv.max()
        if m < 1e-15:
            break
        exp_t = np.exp(t_conv - m)
        w = exp_t / exp_t.sum()

        # Gradient of LSE(t * conv(a,a))
        sg = np.zeros(n)
        thresh = w.max() * 0.005
        active = np.where(w > thresh)[0]
        for ki in active:
            k = int(ki)
            wi = w[ki] * temp
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_end >= j_start:
                idx = np.arange(j_start, j_end + 1)
                sg[idx] += wi * a[k - idx]

        sg = sg - sg.mean()

        v = momentum * v - lr * sg
        a = project_simplex(a + v)

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 400:
            no_improve = 0
            lr = max(0.001, lr * 0.6)
            temp = max(3.0, temp * 0.7)
            v *= 0.5

        lr *= 0.9999
        momentum = min(0.97, momentum + 0.00008)

    return best_a, best_obj


def coordinate_refine(a0, time_limit, step_sizes=None):
    """
    Coordinate-wise local search with incremental convolution updates.
    Cycles through elements and tries small +/- perturbations.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    size = 2 * n - 1

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    if step_sizes is None:
        step_sizes = [0.004, 0.002, 0.001, 0.0005, -0.004, -0.002, -0.001, -0.0005]

    t0 = time.monotonic()
    passed = True

    while passed and time.monotonic() - t0 < time_limit:
        passed = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            for delta in step_sizes:
                a_new = a.copy()
                new_val = a_new[j] + delta
                if new_val < 0.0:
                    continue
                a_new[j] = new_val
                s_new = a_new.sum()
                if s_new < 1e-10:
                    continue
                a_new = a_new / s_new

                # Incremental convolution: only positions k where |k-j| < n change
                # conv_new[k] = conv_a_new[k]
                # For efficiency, just recompute (FFT is fast for n~256)
                conv_new = fast_convolve(a_new)
                obj_new = 2.0 * n * conv_new.max()

                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new.copy()
                    conv = conv_new
                    passed = True
                    break

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations of length n, normalized to simplex."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    ramp = np.arange(1, n + 1, dtype=np.float64)
    ramp /= ramp.sum()
    inits['ramp'] = ramp
    inits['ramp_rev'] = ramp[::-1].copy()

    tri = np.minimum(x + 1.0, n - x)
    tri /= tri.sum()
    inits['triangle'] = tri

    for sf in [0.15, 0.3, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf:.2f}'] = g

    edge = max(1, n // 5)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    bump = np.exp(-x / (0.25 * n))
    bump /= bump.sum()
    inits['exp_decay'] = bump

    inits['random'] = rng.dirichlet(np.ones(n) * 0.4)

    # Cosine window
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    cos_w += 0.01
    cos_w /= cos_w.sum()
    inits['cosine'] = cos_w

    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                          np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                          np.linspace(0, 1, len(parent)), np.maximum(parent, 0))
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining LP direction finding, smooth LSE,
    subgradient descent, and coordinate refinement to minimize
    2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: length={len(p)}, sum={p.sum():.2f}")
    except Exception:
        pass

    # Phase 1: Quick scan over lengths and initializations
    print("=== Phase 1: Quick Length Scan ===")
    lengths = [128, 192, 256, 320, 384, 512]
    phase1_deadline = time.monotonic() + 280.0

    for n in lengths:
        if time.monotonic() > phase1_deadline - 15:
            break

        inits = generate_inits(n, parent, rng)
        priority = ['parent', 'parent_ds', 'parent_up', 'uniform', 'gauss_0.30',
                    'ramp', 'random', 'bimodal', 'exp_decay']
        init_names = [nm for nm in priority if nm in inits][:4]

        for name in init_names:
            if time.monotonic() > phase1_deadline - 10:
                break
            t_alloc = min(25.0, phase1_deadline - time.monotonic() - 8)
            if t_alloc < 8:
                break

            a0 = inits[name]
            a_opt, obj_opt = subgradient_refine(
                a0, time_limit=t_alloc, top_k=15, temp_init=0.002,
                lr_init=0.02, momentum=0.85)

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  P1: n={n}, {name}, score={score:.6f}")

    print(f"After Phase 1: best={best_score:.6f}")

    # Phase 2: LP-based optimization on best sequence
    print("=== Phase 2: LP-based Optimization ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 250.0
        if remaining > 30:
            t_lp = min(remaining * 0.35, 350.0)
            print(f"  LP allocation: {t_lp:.0f}s")
            a_opt, obj_opt = lp_iterate(best_sequence, time_limit=t_lp, t_init=0.12)

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  LP: score={score:.6f}")

    # Phase 3: Smooth LSE optimization with multiple temperatures
    print("=== Phase 3: Smooth LSE ===")
    if best_sequence is not None:
        for temp_val in [6.0, 12.0, 25.0]:
            if time.monotonic() > deadline - 150:
                break
            t_smooth = min(70.0, max(15.0, (deadline - time.monotonic() - 150) * 0.3))
            if t_smooth < 10:
                break

            a_opt, obj_opt = smooth_lse_optimize(
                best_sequence, time_limit=t_smooth, temperature=temp_val)

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  LSE t={temp_val}: score={score:.6f}")

    # Phase 4: Subgradient refinement with varied hyperparameters
    print("=== Phase 4: Subgradient Refinement ===")
    if best_sequence is not None:
        configs = [
            (10, 0.0004, 0.01, 0.92),
            (25, 0.001, 0.015, 0.88),
            (40, 0.002, 0.02, 0.85),
            (15, 0.0006, 0.008, 0.94),
            (30, 0.0015, 0.012, 0.90),
        ]
        for top_k_val, temp_i, lr_i, mom in configs:
            if time.monotonic() > deadline - 60:
                break
            t_sg = min(50.0, max(8.0, (deadline - time.monotonic() - 60) * 0.2))
            if t_sg < 5:
                break

            a_opt, obj_opt = subgradient_refine(
                best_sequence, time_limit=t_sg, top_k=top_k_val,
                temp_init=temp_i, lr_init=lr_i, momentum=mom)

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  SG k={top_k_val},t={temp_i}: score={score:.6f}")

    # Phase 5: Another LP pass if time remains
    print("=== Phase 5: LP Pass 2 ===")
    if best_sequence is not None:
        if time.monotonic() < deadline - 60:
            t_lp2 = min(80.0, deadline - time.monotonic() - 50)
            if t_lp2 > 15:
                a_opt, obj_opt = lp_iterate(best_sequence, time_limit=t_lp2, t_init=0.08)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  LP2: score={score:.6f}")

    # Phase 6: Coordinate refinement (local search)
    print("=== Phase 6: Coordinate Refinement ===")
    if best_sequence is not None and time.monotonic() < deadline - 25:
        t_coord = min(100.0, deadline - time.monotonic() - 20)
        a_opt, obj_opt = coordinate_refine(
            best_sequence, time_limit=t_coord,
            step_sizes=[0.004, 0.002, 0.001, 0.0005, -0.004, -0.002, -0.001, -0.0005])

        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Coord: score={score:.6f}")

    # Phase 7: Final micro-perturbation + quick subgradient
    print("=== Phase 7: Final Polish ===")
    if best_sequence is not None and time.monotonic() < deadline - 10:
        for i in range(5):
            if time.monotonic() > deadline - 5:
                break
            n_best = len(best_sequence)
            noise = rng.standard_normal(n_best) * 0.0015
            a_pert = project_simplex(best_sequence + noise)
            t_p = min(15.0, max(4.0, (deadline - time.monotonic() - 5) / 3.0))
            a_opt, _ = subgradient_refine(
                a_pert, time_limit=t_p, top_k=20, temp_init=0.0004,
                lr_init=0.006, momentum=0.93)

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Polish {i}: score={score:.6f}")

    # Final check with coordinate refinement if time allows
    if best_sequence is not None and time.monotonic() < deadline - 8:
        t_final = deadline - time.monotonic() - 5
        if t_final > 10:
            a_opt, _ = coordinate_refine(
                best_sequence, time_limit=t_final,
                step_sizes=[0.001, 0.0005, 0.0002, -0.001, -0.0005, -0.0002])
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final coord: score={score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
