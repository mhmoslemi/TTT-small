    if sum_a < 0.01:

    n = len(f)
    m = 2 * n - 1

        A[j:j+n, j] = f
    return A

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def score(a):
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)

def make_lp_A(f):
    n = len(f)
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):

Multi-phase optimization for minimizing normalized peak autocorrelation

import numpy as np
import time

def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]


def score(a):
    """Compute score = 2*n*max(conv(a,a))/sum(a)^2."""
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    if n == 0:
        return np.inf
    S = np.sum(a)
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return 2.0 * n * np.max(c) / (S * S)

    """Build constraint matrix: A[k,j] = f[k-j] if 0<=k-j<len(f)."""
    n = len(f)

            a[j] = best_x
            S_cur = S_prime + best_x

            if best_sj < best_s:
                best_s, best_a = best_sj, a.copy()
                improved = True

        sweep += 1
        if improved:
            no_improve_sweeps = 0
        else:
            no_improve_sweeps += 1
            if no_improve_sweeps >= 3:
                # Perturb to escape local minimum
                a *= np.exp(rng.normal(0.0, 0.04, n))
                a = np.clip(a, 0.0, None)
                a = a * (np.sum(f0) / np.sum(a))
                no_improve_sweeps = 0

    return best_a, best_s


def grad_phase(f0, t_limit, seed=31):
    """Adam gradient descent on log-softmax parameterization with annealed soft-max."""
    n = len(f0)
    S_target = np.sqrt(2.0 * n)
    f = np.clip(f0, 1e-12, None)
    theta = np.log(f / np.sum(f))

    m1 = np.zeros(n)
    m2 = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    base_lr = 0.6
    step = 0
    best_f = f.copy()
    best_s = score(f0)
    start = time.monotonic()
    tau = 6.0 / n
    tau_end = 0.4 / n

    while time.monotonic() - start < t_limit:
        frac = min(1.0, (time.monotonic() - start) / t_limit)
        tau = tau_start_val(tau, tau_end, frac)

        ts = theta - np.max(theta)
        e = np.exp(ts)
        a = e / np.sum(e) * S_target

        c = fft_convolve(a, a)
        cmax = np.max(c)
        w = np.exp((c - cmax) / tau)
        w /= np.sum(w)

        rev_a = a[::-1]
        corr = fft_convolve(rev_a, w)
        g_a = 2.0 * corr[n - 1:2 * n - 1]

        inner = np.dot(a, g_a) / S_target
        g_theta = a * (g_a - inner)

        step += 1
        m1 = b1 * m1 + (1 - b1) * g_theta
        m2 = b2 * m2 + (1 - b2) * (g_theta ** 2)
        m1h = m1 / (1.0 - b1 ** step)
        m2h = m2 / (1.0 - b2 ** step)
        lr = base_lr / np.sqrt(1.0 + 0.02 * step)
        theta -= lr * m1h / (np.sqrt(m2h) + eps)

        ts = theta - np.max(theta)
        a_new = np.exp(ts) / np.sum(np.exp(ts)) * S_target
        s_new = score(a_new)
        if np.isfinite(s_new) and s_new < best_s:
            best_s, best_f = s_new, a_new.copy()

    return best_f, best_s


def tau_start_val(tau, tau_end, frac):
    """Compute annealed temperature."""
    return tau * (tau_end / tau) ** frac


def sa_phase(f0, t_limit, seed=55):
    """Simulated annealing with diverse move operators."""
    n = len(f0)
    a = f0.astype(np.float64).copy()
    S = np.sum(a)
    best_a, best_s = a.copy(), score(a)
    cur_s = best_s
    temp = 0.0025
    cool = 0.99975
    rng = np.random.default_rng(seed)
    start = time.monotonic()

    while time.monotonic() - start < t_limit:
        cand = a.copy()
        r = rng.random()

        if r < 0.35:
            # Peak-targeted move: reduce peak contributors, boost others
            c = fft_convolve(a, a)
            pk = np.argmax(c)
            contribs = np.zeros(n)
            for i in range(n):
                j = pk - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]
            top_k = max(3, n // 10)
            top = np.argsort(contribs)[-top_k:]
            cand[top] *= np.exp(rng.normal(-0.03, 0.012, top_k))
            other = np.setdiff1d(np.arange(n), top)
            if len(other) > 2:
                inc = rng.choice(other, min(6, len(other)), replace=False)
                cand[inc] *= np.exp(rng.normal(0.03, 0.012, len(inc)))
        elif r < 0.55:
            # Block move
            w = max(1, int(rng.integers(1, max(2, n // 5))))
            st = int(rng.integers(0, n - w))
            cand[st:st + w] *= np.exp(rng.normal(0.0, 0.025, w))
        elif r < 0.75:
            # Double block
            for _ in range(2):
                w = max(1, int(rng.integers(1, max(2, n // 10))))
                st = int(rng.integers(0, n - w))
                cand[st:st + w] *= np.exp(rng.normal(0.0, 0.02, w))
        elif r < 0.85:
            # Shift + noise
            sh = int(rng.integers(1, max(2, n // 20)))
            cand = np.roll(a, sh)
            cand *= np.exp(rng.normal(0.0, 0.005, n))
        else:
            # Global noise
            cand *= np.exp(rng.normal(0.0, 0.007, n))

        cand = np.clip(cand, 0.0, None)
        cs = np.sum(cand)
        if cs < 1e-10:
            continue
        cand *= S / cs
        ns = score(cand)

        if ns < cur_s:
            a, cur_s = cand, ns
            if ns < best_s:
                best_s, best_a = ns, cand.copy()
        elif rng.random() < np.exp(-(ns - cur_s) / max(temp, 1e-12)):
            a, cur_s = cand, ns
        temp *= cool

    return best_a, best_s


def make_init_cos(n, alpha, rng):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(2.0 * np.pi * t / n)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_chirp(n, alpha, rng):
    t = np.arange(n, dtype=np.float64)
    a = 1.0 + alpha * np.cos(np.pi * t * t / n)
    a += 0.04 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_ramp(n, rng):
    a = np.linspace(1.2, 0.3, n)
    a += 0.05 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def make_init_random(n, rng):
    a = 0.5 + 0.5 * rng.random(n)
    return np.clip(a, 0.01, None)


def make_init_multitone(n, rng):
    t = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for _ in range(5):
        k = int(rng.integers(1, max(2, n // 4)))
        amp = rng.uniform(0.05, 0.2)
        ph = rng.uniform(0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * k * t / n + ph)
    a += 0.02 * rng.standard_normal(n)
    return np.clip(a, 0.01, None)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: LP -> CD -> Gradient -> SA -> Final CD.
    Explores multiple initializations and lengths.
    """
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 35.0
    rng = np.random.default_rng(seed)

    best_s = np.inf
    best_f = None

    # Check provided sequence
    prov = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        prov = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s_prov = score(prov)
        print(f"Provided: n={len(prov)}, score={s_prov:.8f}")
        best_s, best_f = s_prov, prov.copy()

    n0 = len(prov) if prov is not None else 256

    # === Phase 1: LP optimization from diverse starts ===
    print("\n" + "=" * 60)
    print("Phase 1: LP-based optimization (multiple starts)")
    print("=" * 60)

    # Define initializations to try
    inits = []
    if prov is not None:
        inits.append(('provided', prov.copy(), 0.28))
    inits.append(('cos', make_init_cos(n0, 0.3 + 0.3 * rng.random(), rng), 0.14))
    inits.append(('chirp', make_init_chirp(n0, 0.3 + 0.3 * rng.random(), rng), 0.14))
    inits.append(('ramp', make_init_ramp(n0, rng), 0.12))
    inits.append(('random', make_init_random(n0, rng), 0.12))
    inits.append(('multitone', make_init_multitone(n0, rng), 0.10))
    inits.append(('flat', np.clip(1.0 + 0.05 * rng.standard_normal(n0), 0.01, None), 0.10))

    total_frac = sum(x[2] for x in inits)
    phase1_budget = (deadline - t_start) * 0.45  # 45% of total for Phase 1
    p1_t0 = time.monotonic()

    for name, f_init, frac in inits:
        remaining = phase1_budget - (time.monotonic() - p1_t0)
        if remaining < 20:
            break
        alloc = max(15.0, frac / total_frac * phase1_budget)
        alloc = min(alloc, remaining - 5)
        if alloc < 10:
            break

        print(f"  LP [{name}] n={len(f_init)}, alloc={alloc:.0f}s")
        t_lp = time.monotonic()
        f_best, s_best = lp_phase(f_init, alloc, t_init=0.35, seed=hash(name) % 50000)
        elapsed = time.monotonic() - t_lp
        print(f"    -> score={s_best:.8f} ({elapsed:.1f}s)")
        if s_best < best_s:
            best_s, best_f = s_best, f_best.copy()
            print(f"    *** New best: {best_s:.8f} ***")

    # Also try a larger length (512) if time permits
    remaining = deadline - time.monotonic()
    if remaining > 100:
        n_big = 512
        f_big = make_init_cos(n_big, 0.4, rng)
        alloc_big = min(80.0, remaining * 0.15)
        print(f"\n  LP [big_cos] n={n_big}, alloc={alloc_big:.0f}s")
        f_best, s_best = lp_phase(f_big, alloc_big, t_init=0.3, seed=9999)
        print(f"    -> score={s_best:.8f}")
        if s_best < best_s:
            best_s, best_f = s_best, f_best.copy()
            print(f"    *** New best: {best_s:.8f} ***")

    # === Phase 2: Coordinate descent on best ===
    print("\n" + "=" * 60)
    print("Phase 2: Coordinate descent")
    print("=" * 60)
    if best_f is not None:
        remaining = deadline - time.monotonic()
        cd_time = min(350.0, remaining * 0.35)
        if cd_time > 20:
            print(f"  CD: n={len(best_f)}, time={cd_time:.0f}s")
            t_cd = time.monotonic()
            f_cd, s_cd = cd_phase(best_f, cd_time, seed=111)
            print(f"  -> score={s_cd:.8f} ({time.monotonic()-t_cd:.1f}s)")
            if s_cd < best_s:
                best_s, best_f = s_cd, f_cd.copy()
                print(f"  *** New best: {best_s:.8f} ***")

    # === Phase 3: Gradient polish ===
    print("\n" + "=" * 60)
    print("Phase 3: Gradient polishing")
    print("=" * 60)
    if best_f is not None:
        remaining = deadline - time.monotonic()
        grad_time = min(250.0, remaining * 0.35)
        if grad_time > 20:
            print(f"  Grad: n={len(best_f)}, time={grad_time:.0f}s")
            t_g = time.monotonic()
            f_g, s_g = grad_phase(best_f, grad_time, seed=222)
            print(f"  -> score={s_g:.8f} ({time.monotonic()-t_g:.1f}s)")
            if s_g < best_s:
                best_s, best_f = s_g, f_g.copy()
                print(f"  *** New best: {best_s:.8f} ***")

    # === Phase 4: Simulated annealing ===
    print("\n" + "=" * 60)
    print("Phase 4: Simulated annealing")
    print("=" * 60)
    if best_f is not None:
        remaining = deadline - time.monotonic()
        sa_time = min(180.0, remaining * 0.4)
        if sa_time > 15:
            print(f"  SA: n={len(best_f)}, time={sa_time:.0f}s")
            t_sa = time.monotonic()
            f_sa, s_sa = sa_phase(best_f, sa_time, seed=333)
            print(f"  -> score={s_sa:.8f} ({time.monotonic()-t_sa:.1f}s)")
            if s_sa < best_s:
                best_s, best_f = s_sa, f_sa.copy()
                print(f"  *** New best: {best_s:.8f} ***")

    # === Phase 5: Final CD polish ===
    print("\n" + "=" * 60)
    print("Phase 5: Final coordinate descent")
    print("=" * 60)
    if best_f is not None:
        remaining = deadline - time.monotonic()
        if remaining > 25:
            final_cd_time = remaining - 30
            print(f"  Final CD: n={len(best_f)}, time={final_cd_time:.0f}s")
            f_fin, s_fin = cd_phase(best_f, final_cd_time, seed=444)
            if s_fin < best_s:
                best_s, best_f = s_fin, f_fin.copy()
                print(f"  *** New best: {best_s:.8f} ***")

    # Final verification
    if best_f is not None:
        result = [float(max(0.0, min(1000.0, x))) for x in best_f]
        final_s = evaluate_sequence(result)
        print(f"\n{'=' * 60}")
        print(f"FINAL RESULT: score={final_s:.8f}, n={len(result)}")
        print(f"{'=' * 60}")
        return result

    # Fallback
    if prov is not None:
        return [float(max(0.0, min(1000.0, x))) for x in prov]
    return [1.0] * 256
