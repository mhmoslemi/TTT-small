"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase hybrid approach:
(1) LP-based direction finding with exact quadratic line search,
(2) Subgradient descent exploiting the active-set structure,
(3) Coordinate-wise single-variable optimization,
(4) Periodic perturbations to escape local minima.
Starts from the parent sequence and explores multiple lengths.
"""

import numpy as np
import scipy.optimize
import time


def convolve_fft(a, b=None):
    """Linear convolution of a with b (or a with itself if b is None) using FFT."""
    if b is None:
        b = a
    na = len(a)
    nb = len(b)
    size = na + nb - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def norm_seq(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    return a / s


def obj_norm(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = convolve_fft(a)
    return 2.0 * n * c.max()


def solve_lp_direction(f, M, n_constraints=None):
    """Find g >= 0 maximizing sum(g) subject to conv(f,g)_k <= M."""
    n = len(f)
    if n_constraints is None:
        n_constraints = 2 * n - 1
    A = np.zeros((n_constraints, n))
    for j in range(n):
        k_start = j
        k_end = min(n_constraints, j + n)
        for k in range(k_start, k_end):
            A[k, j] = f[k - j]
    c_vec = -np.ones(n)
    b_vec = np.full(n_constraints, M)
    bounds = [(0.0, None)] * n
    try:
        res = scipy.optimize.linprog(c_vec, A_ub=A, b_ub=b_vec, bounds=bounds, method='highs')
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return f.copy()


def quadratic_line_search(f, g, max_iter=80):
    """Find optimal t in [0,1] for (1-t)*f + t*g using quadratic structure."""
    n = len(f)
    M_arr = convolve_fft(f)
    L_arr = convolve_fft(f, g)
    Q_arr = convolve_fft(g)

    def score(t):
        vals = (1.0 - t) ** 2 * M_arr + 2.0 * t * (1.0 - t) * L_arr + t ** 2 * Q_arr
        return vals.max()

    # Use bisection on the derivative for the upper envelope
    # The function is the max of quadratics, which is convex in t
    lo = 0.0
    hi = 1.0
    best_t = 0.0
    best_val = score(0.0)

    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        # Numerical derivative
        eps = 1e-7
        val_mid = score(mid)
        if val_mid < best_val:
            best_val = val_mid
            best_t = mid
        if hi - lo < 1e-10:
            break
        d = (score(mid + eps) - score(mid - eps)) / (2.0 * eps)
        if d < 0:
            lo = mid
        else:
            hi = mid
    return best_t, best_val


def subgradient_step(a, step_size):
    """One subgradient descent step on max_k conv(a,a)_k."""
    n = len(a)
    conv = convolve_fft(a)
    M = conv.max()
    # Active set: lags where conv is close to max
    threshold = M * 0.95
    active = np.where(conv >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(conv))])

    grad = np.zeros(n)
    for k in active:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                grad[i] += 2.0 * a[j]
    grad /= len(active)

    a_new = a - step_size * grad
    a_new = norm_seq(a_new)
    return a_new, len(active)


def coordinate_descent(a, max_rounds=50, time_limit=30.0):
    """Optimize one coordinate at a time."""
    n = len(a)
    t0 = time.monotonic()
    best_obj = obj_norm(a)
    a = norm_seq(a.copy())

    for rnd in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            # Compute the quadratic coefficients for conv_k as function of a[i]=x
            saved = a[i]
            a[i] = 0.0
            conv0 = convolve_fft(a)
            a[i] = 1.0
            conv1 = convolve_fft(a)
            a[i] = saved

            # A_k = 1 if k = 2*i else 0
            # B_k = conv1_k - conv0_k - A_k
            # C_k = conv0_k
            A_k = np.zeros(2 * n - 1)
            if 2 * i < 2 * n - 1:
                A_k[2 * i] = 1.0
            B_k = conv1 - conv0 - A_k
            C_k = conv0

            # Minimize max_k(A_k * x^2 + B_k * x + C_k) / (1 - saved + x)^2 * 2*n
            # subject to x >= 0

            lo = 0.0
            hi = max(3.0 * saved, 0.05)
            gr = (np.sqrt(5.0) - 1.0) / 2.0

                if x < 0:
                    return np.inf
                s_new = 1.0 - saved + x
                if s_new < 1e-12:
                    return np.inf
                vals = A_k * x * x + B_k * x + C_k
                return 2.0 * n * vals.max() / (s_new * s_new)

            c = hi - gr * (hi - lo)
            d = lo + gr * (hi - lo)
            fc = f_of_x(c)
            fd = f_of_x(d)
            for _ in range(40):
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - gr * (hi - lo)
                    fc = f_of_x(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + gr * (hi - lo)
                    fd = f_of_x(d)

            x_opt = (lo + hi) / 2.0
            f_opt = f_of_x(x_opt)
            if f_opt < best_obj * (1.0 - 1e-10):
                a[i] = x_opt
                a = norm_seq(a)
                best_obj = obj_norm(a)
                improved = True

        if not improved:

    return a, best_obj


def lp_optimize(f0, time_limit=60.0):

                    no_improve = 0

def subgradient_optimize(f0, time_limit=60.0):

def make_initializations(n, rng):
    """Generate diverse initializations."""
    x = np.arange(n, dtype=np.float64)
    inits = {}
    inits['uniform'] = norm_seq(np.ones(n))

    for sf in [0.15, 0.25, 0.4, 0.6]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        inits['gauss_{}'.format(sf)] = norm_seq(g)

    if n > 1:
        hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n - 1)))
        inits['hann'] = norm_seq(hann)

    tri = np.minimum(x, n - 1 - x).astype(np.float64) + 0.5
    inits['triangle'] = norm_seq(tri)

    inits['dirichlet'] = norm_seq(rng.dirichlet(np.ones(n) * 0.5))
    inits['dirichlet2'] = norm_seq(rng.dirichlet(np.ones(n) * 2.0))

    m = max(1, n // 3)
    edges = np.zeros(n)
    edges[:m] = 1.0
    edges[-m:] = 1.0
    inits['edges'] = norm_seq(edges)

    return inits

def perturb_sequence(a, rng, strength=0.1):
    """Add random positive perturbation and renormalize."""
