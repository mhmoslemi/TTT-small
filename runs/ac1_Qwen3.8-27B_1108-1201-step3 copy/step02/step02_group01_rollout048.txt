    t0 = time.monotonic()

    best_a = a.copy()
    best_conv_max = float(np.max(fft_convolve(a, a)))

    y = a.copy()
    momenta = 0.9
    lr = 2.0 / (n * np.max(fft_convolve(a, a)) + 1e-10)
    step = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        conv_y = fft_convolve(y, y)
        peak_idx = int(np.argmax(conv_y))

        # Subgradient of conv[peak_idx] w.r.t. a[i]
        grad = np.zeros(n)
        for i in range(n):
            j = peak_idx - i
            if 0 <= j < n:
                grad[i] += y[j]
            j2 = i - peak_idx
            if 0 <= j2 < n:
                grad[i] += y[j2]
        # The actual gradient of sum_i a[i]*a[peak-i] w.r.t. a[i]
        # = a[peak-i] + a[i-peak] (when valid)

        # Nesterov update
        t_k = 2.0 / (step + 2.0)
        y_new = a - lr * grad
        y_new = np.maximum(y_new, 0.0)
        y = (1 - t_k) * y_new + t_k * (y - a)
        a = y_new
        a = normalize_seq(a, S)

        new_conv_max = float(np.max(fft_convolve(a, a)))
        if new_conv_max < best_conv_max:
            best_conv_max = new_conv_max
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Adaptive learning rate
        if stall > 50:
            lr *= 0.9
            stall = 0
        elif step % 200 == 0 and step > 0:
            lr *= 0.95

        step += 1
        if step % 2000 == 0 and verbose:
            score = 2.0 * n * best_conv_max / (S * S)
            print(f"    NSG step {step}: score={score:.6f}, lr={lr:.2e}")

    return best_a


def simulated_annealing(a, time_limit, verbose=False):
    """Simulated annealing with diverse move types."""
    n = len(a)
    S = np.sqrt(2.0 * n)
    a = normalize_seq(a, S)
    t0 = time.monotonic()

    best_a = a.copy()
    best_score = local_score(a)
    current_score = best_score

    T = 0.005
    T_min = 1e-7
    cooling = 0.998
    rng = np.random.default_rng(777)
    step = 0

    while time.monotonic() - t0 < time_limit and T > T_min:
        if step % 2000 == 0 and verbose:
            elapsed = time.monotonic() - t0
            print(f"    SA step {step}: T={T:.6f}, best={best_score:.6f}, elapsed={elapsed:.1f}s")

        move_type = rng.random()

        if move_type < 0.35:
            # Subgradient targeting: reduce peak contributors
            conv = fft_convolve(a, a)
            peak_idx = int(np.argmax(conv))
            contribs = np.zeros(n)
            for i in range(n):
                j = peak_idx - i
                if 0 <= j < n:
                    contribs[i] = a[i] * a[j]
            top_k = max(3, n // 10)
            top_idx = np.argsort(contribs)[-top_k:]
            candidate = a.copy()
            for idx in top_idx:
                candidate[idx] *= np.exp(rng.normal(-0.03, 0.015))
            # Boost some non-peak elements
            non_peak = np.setdiff1d(np.arange(n), top_idx)
            if len(non_peak) > 0:
                boost_idx = rng.choice(non_peak, size=max(2, n // 20), replace=False)
                candidate[boost_idx] *= np.exp(rng.normal(0.03, 0.015, size=len(boost_idx)))

        elif move_type < 0.6:
            # Gaussian perturbation
            candidate = a.copy()
            n_pert = max(2, n // 15)
            idx = rng.choice(n, size=n_pert, replace=False)
            candidate[idx] *= np.exp(rng.normal(0, 0.025, size=n_pert))

        elif move_type < 0.8:
            # Block modification
            width = max(1, n // 40)
            start = int(rng.integers(0, max(1, n - width)))
            candidate = a.copy()
            candidate[start:start + width] *= np.exp(rng.normal(0, 0.04, size=width))

        else:
            # Swap-based
            candidate = a.copy()
            i, j = tuple(rng.choice(n, size=2, replace=False))
            delta = rng.uniform(-0.15, 0.15) * max(a[i], a[j], 0.01)
            candidate[i] = max(0.0, a[i] - delta)
            candidate[j] = max(0.0, a[j] + delta)

        candidate = np.clip(candidate, 0.0, None)
        cs = np.sum(candidate)
        if cs < 1e-10:
            T *= cooling
            step += 1
            continue
        candidate = candidate * S / cs
        new_score = local_score(candidate)

        if new_score < current_score or rng.random() < np.exp((current_score - new_score) / T):
            a = candidate
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

        T *= cooling
        step += 1

    return best_a
