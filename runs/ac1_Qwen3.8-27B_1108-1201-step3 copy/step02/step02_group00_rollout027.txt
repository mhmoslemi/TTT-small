```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth approximation of the max
(softmax) combined with Nesterov-accelerated projected gradient descent on the
simplex, mirror descent, and multi-restart with diverse initializations.
Key innovation: using the true gradient of the smooth max (via FFT-based
convolution) enables use of accelerated methods with BB step sizes, giving
much faster convergence than subgradient methods.
"""

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u * np.arange(1, n + 1) > (cssv - 1))
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_max_and_weights(conv, tau):
    """Compute smooth max of conv and the softmax weights."""
    max_c = conv.max()
    scaled = tau * (conv - max_c)
    exp_vals = np.exp(scaled)
    w_sum = exp_vals.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
        return max_c, w
    weights = exp_vals / w_sum
    smooth_val = max_c + np.log(w_sum) / tau
    return smooth_val, weights

def compute_gradient_fft(a, weights):
    """Compute gradient of smooth_max(conv(a,a)) wrt a using FFT cross-correlation.
    grad[j] = 2 * sum_k weights[k] * a[k-j]
    """
    n = len(a)
    corr_len = len(weights) + n - 1  # = 3n - 2
    N = 1 << (corr_len - 1).bit_length()
    W_fft = np.fft.rfft(weights, N)
    A_fft = np.fft.rfft(a, N)
    # Cross-correlation: ifft(conj(W) * A)
    corr = np.fft.irfft(np.conj(W_fft) * A_fft, N)
    return 2.0 * corr[:n]

def objective_from_conv(n, conv):
    """Compute the objective value from the convolution."""
    return 2.0 * n * conv.max()
```

```python
def optimize_nesterov(a0, time_limit, tau_init=10, tau_max=500, 
                      tau_growth=1.001, momentum_init=0.9, momentum_growth=0.0002,
                      bb_step=True):
    """
    Nesterov-accelerated projected gradient descent on the simplex.
    Minimizes smooth_max(conv(a,a)) where tau is annealed.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    y = a.copy()  # Nesterov lookahead
    v = np.zeros(n)  # momentum
    momentum = momentum_init
    tau = tau_init
    lr = 1.0  # will be set by BB or initial
    
    conv = fast_convolve(a)
    best_obj = objective_from_conv(n, conv)
    best_a = a.copy()
    
    t0 = time.monotonic()
    
    # Previous state for BB step
    a_prev = a.copy()
    g_prev = None
    g_prev_diff = None
    
    step = 0
    while time.monotonic() - t0 < time_limit:
        # Compute gradient at Nesterov point y
        conv_y = fast_convolve(y)
        _, weights = smooth_max_and_weights(conv_y, tau)
        g = compute_gradient_fft(y, weights)
        
        # BB step size
        if bb_step and g_prev is not None:
            s_diff = y - a_prev
            g_diff = g - g_prev
            s_dot = np.dot(s_diff, s_diff)
            g_dot = np.dot(s_diff, g_diff)
            if s_dot > 1e-15 and abs(g_dot) > 1e-15:
                lr_new = s_dot / abs(g_dot)
                lr = 0.5 * (lr + lr_new)  # average of last two BB steps
                lr = min(lr, 2.0)  # cap
                lr = max(lr, 1e-6)
            # else keep lr
        
        # Nesterov update
        v = momentum * v - lr * g
        a_new = project_simplex(y + v * 0.5 + lr * g * 0.5)  
        # Actually, standard Nesterov:
        # y_new = (1-momentum) * a_new + momentum * (a_new + v)
        # Simplified: a_new = project(a - lr*grad_at_y), y_new = a_new + momentum*(a_new - a)
        
        # Let me use the more standard FISTA form:
        # y_k = x_k + (t_{k-1} - 1)/(t_k) * (x_k - x_{k-1})
        # x_{k+1} = project(y_k - lr * grad(y_k))
        # t_{k+1} = (1 + sqrt(1 + 4*t_k^2))/2
        
        # Actually let me just use heavy-ball momentum which is simpler:
        # v = momentum * v - lr * g
        # a_new = project(a + v)
        
        a_new = project_simplex(a + v)
        
        # Update
        a_prev = a
        g_prev = g
        a = a_new
        y = a + momentum * (a - a_prev)  # Nesterov lookahead
        
        # Evaluate
        conv = fast_convolve(a)
        cur_obj = objective_from_conv(n, conv)
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
        
        # Anneal tau
        tau = min(tau_max, tau * tau_growth)
        momentum = min(0.99, momentum + momentum_growth)
        
        step += 1
    
    return best_a, best_obj
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth approximation of max
(softmax with annealed temperature) combined with FISTA (Nesterov accelerated
projected gradient) on the probability simplex. Uses FFT-based cross-correlation
for efficient gradient computation. Multi-restart with diverse initializations
and mirror descent as complementary optimizer.
"""

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u * np.arange(1, n + 1) > (cssv - 1.0))
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_max_weights(conv, tau):
    """Compute softmax weights for smooth max approximation."""
    max_c = conv.max()
    scaled = tau * (conv - max_c)
    # Clip for numerical stability
    scaled = np.clip(scaled, -500, 500)
    exp_vals = np.exp(scaled)
    w_sum = exp_vals.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
        return w
    return exp_vals / w_sum


def compute_grad(a, weights):
    """Compute gradient of smooth_max(conv(a,a)) wrt a.
    grad[j] = 2 * sum_k weights[k] * a[k-j]
    Using FFT: ifft(FFT(w) * conj(FFT(a)))[:n]
    """
    n = len(a)
    corr_len = len(weights) + n - 1  # = 3n-2
    N = 1 << (corr_len - 1).bit_length()
    W = np.fft.fft(weights, N)
    A = np.fft.fft(a, N)
    C = np.fft.ifft(W * np.conj(A), N)
    return 2.0 * np.real(C[:n])


def obj_value(n, conv):
    """Compute 2*n*max(conv)."""
    return 2.0 * n * conv.max()


def optimize_fista(a0, time_limit, tau_start=5.0, tau_end=1000.0,
                   tau_steps=None, lr_scale=1.0):
    """
    FISTA (Nesterov accelerated projected gradient) on the simplex.
    Minimizes smooth_max(conv(a,a)) with annealed temperature.
    Returns (best_a, best_obj).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    if tau_steps is None:
        tau_steps = 200

    conv = fast_convolve(a)
    best_obj = obj_value(n, conv)
    best_a = a.copy()

    t0 = time.monotonic()
    a_prev = a.copy()
    t_mom = 1.0

    # Anneal tau linearly in log space
    log_tau = np.linspace(np.log(tau_start), np.log(tau_end), tau_steps)
    step = 0

    while time.monotonic() - t0 < time_limit:
        tau = log_tau[min(step, tau_steps - 1)]
        # Adaptive lr: smaller as tau grows
        lr = lr_scale / (1.0 + tau * 0.01)

        t_prev = t_mom
        t_mom = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        beta = (t_prev - 1.0) / t_mom

        # Nesterov extrapolation
        y = a + beta * (a - a_prev)
        y = project_simplex(y)

        # Gradient at y
        conv_y = fast_convolve(y)
        weights = smooth_max_weights(conv_y, tau)
        g = compute_grad(y, weights)

        # Gradient step + projection
        a_new = project_simplex(y - lr * g)

        a_prev = a
        a = a_new

        # Evaluate
        conv = fast_convolve(a)
        cur_obj = obj_value(n, conv)
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()

        step += 1
        # After tau schedule completes, stay at max tau
        if step >= tau_steps:
            tau = tau_end
            lr = lr_scale / (1.0 + tau * 0.01)

    return best_a, best_obj


def optimize_mirror(a0, time_limit, t_param=30.0, eta_init=0.5,
                    eta_decay=0.9995, t_growth=1.005, t_max=500.0):
    """
    Mirror descent (multiplicative weights) with softmax smoothing.
    Automatically maintains non-negativity and normalization.
    Returns (best_a, best_obj).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = obj_value(n, conv)
    best_a = a.copy()

    t0 = time.monotonic()
    eta = eta_init
    t_cur = t_param
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        weights = smooth_max_weights(conv, t_cur)
        g = compute_grad(a, weights)

        # Center gradient (removes uniform component that doesn't affect after normalization)
        g_centered = g - np.dot(g, a)

        # Multiplicative update
        log_update = -eta * g_centered
        log_update -= log_update.max()
        a_new = a * np.exp(np.clip(log_update, -50, 50))
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s_new
        a = a_new

        conv = fast_convolve(a)
        cur_obj = obj_value(n, conv)

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adapt parameters
        eta *= eta_decay
        t_cur = min(t_max, t_cur * t_growth)

        if no_improve > 200:
            # Restart eta
            eta = eta_init * 0.5
            t_cur = t_param
            no_improve = 0

    return best_a, best_obj


def optimize_softmax_lbfgs(a0, time_limit, tau=50.0, max_iter=500):
    """
    Optimize using L-BFGS-B on softmax parameterization.
    a = softmax(z), minimize smooth_max(conv(a,a)).
    Returns (best_a, best_obj).
    """
    n = len(a0)
    a_init = np.maximum(a0.astype(np.float64), 0.0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s

    # Initialize z in log space
    z0 = np.log(a_init + 1e-30)
    z0 -= z0.max()

    t0 = time.monotonic()
    best_a = a_init.copy()
    conv = fast_convolve(a_init)
    best_obj = obj_value(n, conv)
    iterations = [0]

    def objective(z):
        iterations[0] += 1
        if time.monotonic() - t0 > time_limit:
            return 1e10
        a = np.exp(z - z.max())
        a = a / a.sum()
        conv = fast_convolve(a)
        _, w = smooth_max_weights(conv, tau)
        val = np.log(w.sum()) / tau + conv.max()
        return val

    def gradient(z):
        a = np.exp(z - z.max())
        a = a / a.sum()
        conv = fast_convolve(a)
        _, w = smooth_max_weights(conv, tau)
        g = compute_grad(a, w)
        # Chain rule for softmax: dF/dz_i = a_i * (g_i - sum_j g_j * a_j)
        mean_ga = np.dot(g, a)
        return a * (g - mean_ga)

    result = scipy_minimize(objective, z0, jac=gradient, method='L-BFGS-B',
                            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-8})

    a_opt = np.exp(result.x - result.x.max())
    a_opt = a_opt / a_opt.sum()
    conv_opt = fast_convolve(a_opt)
    obj_opt = obj_value(n, conv_opt)

    if obj_opt < best_obj:
        best_obj = obj_opt
        best_a = a_opt

    return best_a, best_obj


def gen_initializations(n, parent=None, rng=None):
    """Generate diverse initializations for length n."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = {}
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits['uniform'] = np.ones(n) / n

    # Ramps
    r = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp_up'] = r / r.sum()
    inits['ramp_down'] = r[::-1] / r.sum()

    # Triangle
    tri = np.minimum(x, n - 1 - x) + 1.0
    inits['triangle'] = tri / tri.sum()

    # Bimodal (edges)
    bi = np.zeros(n)
    edge = max(1, n // 3)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    # Gaussian
    for sf in [0.1, 0.2, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (sf * n)) ** 2)
        inits[f'gau_{sf}'] = g / g.sum()

    # Random Dirichlet
    inits['dirichlet'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['dirichlet_sparse'] = rng.dirichlet(np.ones(n) * 0.1)

    # From parent
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            # Upsample parent to length n
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps

        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps

    return inits


def verify_gradient(a, tau, rtol=1e-5):
    """Verify gradient computation against finite differences."""
    n = len(a)
    conv = fast_convolve(a)
    weights = smooth_max_weights(conv, tau)
    g_analytic = compute_grad(a, weights)

    # Finite difference at a random coordinate
    eps = 1e-7
    idx = n // 2
    a_plus = a.copy()
    a_minus = a.copy()
    a_plus[idx] += eps
    a_minus[idx] -= eps
    # Renormalize
    a_plus = a_plus / a_plus.sum()
    a_minus = a_minus / a_minus.sum()

    def smooth_val(a_test):
        c = fast_convolve(a_test)
        w = smooth_max_weights(c, tau)
        return c.max() + np.log(w.sum()) / tau

    g_numeric = (smooth_val(a_plus) - smooth_val(a_minus)) / (2 * eps)

    # The analytic gradient is for the unnormalized version;
    # for the simplex, we need to check the directional derivative
    # This is just a sanity check
    diff = abs(g_analytic[idx] - g_numeric)
    scale = max(abs(g_analytic[idx]), abs(g_numeric), 1e-10)
    return diff / scale < rtol

        if key not in seen_keys and score < best_score + 0.05:
            deep_candidates.append((n, name, a_opt))
            seen_keys.add(key)
        if len(deep_candidates) >= 6:
            break

    # Add best found so far
    if best_sequence is not None:
        deep_candidates.append((len(best_sequence), 'best', best_sequence.copy()))

    # Add some fresh starts at promising lengths
    if parent is not None:
        best_n = len(parent)
    else:
        best_n = 256

    for extra_name in ['dirichlet_sparse', 'gau_0.1', 'bimodal']:
        inits_b = gen_initializations(best_n, parent, rng)
        if extra_name in inits_b:
            deep_candidates.append((best_n, extra_name, inits_b[extra_name]))

    time_per_deep = max(30.0, (phase2_deadline - time.monotonic()) / max(1, len(deep_candidates) * 2))

    for i, (n, name, a0) in enumerate(deep_candidates):
        if time.monotonic() > phase2_deadline - 30:
            break
        t_alloc = min(time_per_deep, phase2_deadline - time.monotonic() - 20)
        if t_alloc < 10:
            break

        # Try FISTA with different parameter settings
        for cfg_name, kwargs in [
            ('fista_A', dict(tau_start=3.0, tau_end=2000.0, tau_steps=300, lr_scale=2.0)),
            ('fista_B', dict(tau_start=10.0, tau_end=500.0, tau_steps=150, lr_scale=1.0)),
        ]:
            if time.monotonic() > phase2_deadline - 20:
                break
            t_single = t_alloc / 2.0
            if t_single < 5:
                break
            a_opt, obj = optimize_fista(a0, time_limit=t_single, **kwargs)
            try_candidate(a_opt, f"n={n} {name} {cfg_name}")

        # Try mirror descent
        if time.monotonic() < phase2_deadline - 20:
            a_opt, obj = optimize_mirror(a0, time_limit=t_alloc / 3.0,
                                         t_param=20.0, eta_init=0.3)
            try_candidate(a_opt, f"n={n} {name} mirror")

        # Try softmax L-BFGS
        if time.monotonic() < phase2_deadline - 20:
            a_opt, obj = optimize_softmax_lbfgs(a0, time_limit=t_alloc / 3.0,
                                                tau=30.0, max_iter=200)
            try_candidate(a_opt, f"n={n} {name} lbfgs")

    print(f"Phase 2 done. Best: {best_score:.6f}")

    # === Phase 3: Final polish ===
    print("=== Phase 3: Final Polish ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic()
        n_best = len(best_sequence)

        # Multiple FISTA polish runs with different params
        polish_configs = [
            dict(tau_start=1.0, tau_end=5000.0, tau_steps=500, lr_scale=2.5),
            dict(tau_start=5.0, tau_end=2000.0, tau_steps=300, lr_scale=1.5),
            dict(tau_start=20.0, tau_end=3000.0, tau_steps=200, lr_scale=0.8),
            dict(tau_start=0.5, tau_end=1000.0, tau_steps=400, lr_scale=3.0),
        ]

        t_per = remaining / (len(polish_configs) + 3)

        for idx, cfg in enumerate(polish_configs):
            if remaining < 10:
                break
            t_alloc = min(t_per, remaining - 10)
            a_opt, obj = optimize_fista(best_sequence, time_limit=t_alloc, **cfg)
            try_candidate(a_opt, f"polish_fista_{idx}")
            remaining = deadline - time.monotonic()

        # Mirror polish
        if remaining > 20:
            a_opt, obj = optimize_mirror(best_sequence, time_limit=remaining / 3,
                                         t_param=50.0, eta_init=0.2)
            try_candidate(a_opt, "polish_mirror")
            remaining = deadline - time.monotonic()

        # L-BFGS polish
        if remaining > 20:
            a_opt, obj = optimize_softmax_lbfgs(best_sequence, time_limit=remaining / 3,
                                                tau=100.0, max_iter=300)
            try_candidate(a_opt, "polish_lbfgs")
            remaining = deadline - time.monotonic()

        # Perturbation + re-optimize
        if remaining > 30:
            for _ in range(3):
                if remaining < 15:
                    break
                scale = rng.choice([0.005, 0.01, 0.02])
                noise = rng.standard_normal(n_best) * scale
                a_pert = project_simplex(best_sequence + noise)
                a_opt, obj = optimize_fista(a_pert, time_limit=remaining / 4,
                                            tau_start=10.0, tau_end=2000.0,
                                            tau_steps=150, lr_scale=1.5)
                try_candidate(a_opt, "perturb_reopt")
                remaining = deadline - time.monotonic()

    # Fallback
    if best_sequence is None:
        n_fb = 256
        best_sequence = np.ones(n_fb) / n_fb
        best_score = float(evaluate_sequence(best_sequence.tolist()))
        print(f"Fallback used: score={best_score:.6f}")

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
```
