"""
Hybrid LP + gradient optimization for minimizing peak autocorrelation
of non-negative sequences. Uses reduced-constraint LPs, FFT-based
gradients, and multi-scale search across sequence lengths.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import linprog, minimize

def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = next_fast_len(n + m - 1)
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]

def score_fast(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))

def score_and_grad(a, top_k=10):
    """Compute score and gradient w.r.t. a using top-k peak positions."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    S = float(np.sum(a))
    c = fft_convolve(a, a)
    max_c = np.max(c)
    score = 2.0 * n * max_c / (S * S)
    
    # Gradient of max(conv(a,a)) w.r.t. a
    # d/dc_k [a_j * a_{k-j}] = a_{k-j} (for fixed k)
    # For the max, we use the subgradient from top-k positions
    top_idx = np.argsort(c)[-top_k:]
    g = np.zeros(n)
    for k in top_idx:
        k = int(k)
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            other = k - j
            if 0 <= other < n:
                g[j] += a[other]
    g /= top_k
    
    # Gradient of full score = 2n * max_c / S^2
    # d(score)/da_j = (2n/S^2) * g_j - 2*(2n*max_c/S^3) * 1 = (2n/S^2)*g_j - 4n*max_c/S^3
    # But we want to minimize score, and we can fix S by normalization.
    # If we fix S, the gradient is just proportional to g.
    return score, g

def solve_lp_reduced(f, M, K=30, timeout=2.0):
    """Solve LP with only top-K binding constraints."""
    n = len(f)
    c = fft_convolve(f, f)
    
    # Find the K most "binding" constraints (highest conv values)
    top_k_idx = np.argsort(c)[-K:]
    
    # Build reduced constraint matrix
    A = np.zeros((len(top_k_idx), n))
    for i, k in enumerate(top_k_idx):
        k = int(k)
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            A[i, j] = f[k - j]
    
    b_ub = np.full(len(top_k_idx), M)
    c_obj = -np.ones(n)
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'time_limit': timeout})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def make_init(n, kind, rng):
    t = np.arange(n, dtype=np.float64)
    if kind == 'hann':
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / max(1, n - 1)))
        a += 0.05 * rng.standard_normal(n)
    elif kind == 'raised_cos':
        # Raised cosine with different parameters
        alpha = rng.uniform(0.5, 1.5)
        a = 0.5 * (1.0 + np.cos(np.pi * (2*t/(n-1) - 1) / alpha)) ** 2
        a = np.clip(a, 0, None)
        a += 0.03 * rng.standard_normal(n)
    elif kind == 'gaussian':
        center = n / 2
        sigma = rng.uniform(n/4, n/2)
        a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
        a += 0.03 * rng.standard_normal(n)
    elif kind == 'random_smooth':
        a = 0.3 + 0.7 * rng.random(n)
        # Smooth it
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
        a += 0.02 * rng.standard_normal(n)
    elif kind == 'bipolar_smooth':
        # Start with random signs, take absolute, smooth
        a = np.abs(rng.standard_normal(n))
        kernel = np.ones(7) / 7
        a = np.convolve(a, kernel, mode='same')
        a = 0.2 + 0.8 * a / a.max()
    else:
        a = np.ones(n) * 0.5 + 0.3 * rng.random(n)
    return np.clip(a, 0.01, None)

def optimize_hybrid(f_init, time_limit, verbose=False):
    """Hybrid LP + gradient optimization."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    it = 0
    stall = 0
    
    # Hyperparameters
    lp_K = max(20, n // 16)  # Number of constraints in LP
    lp_timeout = min(3.0, time_limit / 100)
    grad_lr = 0.01
    grad_momentum = 0.9
    
    momentum = np.zeros(n)
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 2.0:
            break
        
        c = fft_convolve(f, f)
        M = float(np.max(c))
        S = float(np.sum(f))
        current_score = 2.0 * n * M / (S * S)
        
        # Alternate between LP and gradient steps
        if it % 3 != 2:
            # LP step
            b = solve_lp_reduced(f, M, K=lp_K, timeout=lp_timeout)
            if b is not None:
                Sb = float(np.sum(b))
                if Sb > 1e-12:
                    b = b * (S / Sb)
                    # Line search
                    best_t = 0.1
                    best_s = np.inf
                    for t_try in [0.05, 0.1, 0.2, 0.3]:
                        f_cand = np.clip((1 - t_try) * f + t_try * b, 0.01, 1000.0)
                        s_cand = score_fast(f_cand)
                        if s_cand < best_s:
                            best_s = s_cand
                            best_t = t_try
                    f = np.clip((1 - best_t) * f + best_t * b, 0.01, 1000.0)
            else:
                # Fallback to gradient
                _, g = score_and_grad(f, top_k=min(10, n//8))
                g -= np.mean(g)
                momentum = grad_momentum * momentum + (1 - grad_momentum) * g
                f = np.clip(f - grad_lr * momentum, 0.01, 1000.0)
        else:
            # Gradient step
            _, g = score_and_grad(f, top_k=min(15, n//6))
            g -= np.mean(g)
            momentum = grad_momentum * momentum + (1 - grad_momentum) * g
            f = np.clip(f - grad_lr * momentum, 0.01, 1000.0)
        
        it += 1
        s = score_fast(f)
        if s < best_score - 1e-10:
            best_score = s
            best_f = f.copy()
            stall = 0
        else:
            stall += 1
        
        if stall >= 50:
            # Escape local minimum
            rng_esc = np.random.default_rng(1000 + it)
            f = f + 0.05 * rng_esc.standard_normal(n)
            f = np.clip(f, 0.01, 1000.0)
            stall = 0
        
        if verbose and it % 50 == 0:
            print(f"    [{time.monotonic()-t0:6.1f}s] it={it} score={s:.6f} best={best_score:.6f}")
    
    return best_f, best_score

    n = len(f_init)
    f = f_init.astype(np.float64).copy()

    t0 = time.monotonic()
    momentum = np.zeros(n)

    top_k = min(20, n // 8)

        top_idx = np.argsort(c)[-top_k:]
        
        g = np.zeros(n)
        for k in top_idx:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):

        g -= np.mean(g)

            candidate *= np.exp(rng.normal(0, 0.02, size=n))

    grad_lse = irfft(np.conj(A_fft) * W_fft, N)[:n] * 2.0

    # Full gradient of score
    # d(score)/da = (2n/S^2) * grad_lse - 2*(2n*lse/S^3) * ones
    grad_score = (2.0 * n / (S * S)) * grad_lse - 4.0 * n * lse / (S ** 3)

    return score, grad_score


def smooth_objective_wrapper(a_raw, T, target_sum):
    """Wrapper for scipy minimize: takes flat array, returns (f, grad)."""
    a = np.maximum(a_raw, 0.0)
    f, g = smooth_score_grad(a, T)
    # Add small penalty to fix scale (avoid scale invariance issue)
    S = float(np.sum(a))
    f += 0.01 * (S - target_sum) ** 2
    g = g - 0.02 * (S - target_sum) * np.ones_like(a)
    return f, g


def make_init(n, kind, rng):
    """Generate diverse initializations."""
    t = np.arange(n, dtype=np.float64)
    if kind == 'hann':
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / max(1, n - 1)))
        a += 0.03 * rng.standard_normal(n)
    elif kind == 'raised_cos':
        alpha = rng.uniform(0.5, 1.5)
        x = 2.0 * t / max(1, n - 1) - 1.0
        a = np.maximum(0.5 * (1.0 + np.cos(np.pi * x / alpha)), 0.0) ** 0.5
        a += 0.02 * rng.standard_normal(n)
    elif kind == 'gaussian':
        center = n / 2.0
        sigma = rng.uniform(n / 4.0, n / 2.5)
        a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
        a += 0.02 * rng.standard_normal(n)
    elif kind == 'random_smooth':
        a = 0.3 + 0.7 * rng.random(n)
        k = max(3, n // 30)
        kernel = np.exp(-0.5 * (np.arange(-k, k+1) / (k//2+1))**2)
        kernel /= kernel.sum()
        a = np.convolve(a, kernel, mode='same')
        a += 0.01 * rng.standard_normal(n)
    elif kind == 'bipolar':
        a = np.abs(rng.standard_normal(n))
        k = max(5, n // 20)
        kernel = np.exp(-0.5 * (np.arange(-k, k+1) / (k//3+1))**2)
        kernel /= kernel.sum()
        a = np.convolve(a, kernel, mode='same')
        a = 0.3 + 0.7 * a / (a.max() + 1e-10)
    elif kind == 'plateau':
        frac = rng.uniform(0.4, 0.8)
        width = int(frac * n)
        start = (n - width) // 2
        a = np.zeros(n)
        a[start:start+width] = 1.0
        # Smooth edges
        edge = max(3, n // 40)
        a[start:edge] = 0.5*(1-np.cos(np.pi*np.arange(edge)/edge)) if start >= edge else a[start:edge]
        a[n-edge:n] = 0.5*(1-np.cos(np.pi*np.arange(edge)/edge))
        a = np.convolve(a, np.ones(7)/7, mode='same')
        a += 0.01 * rng.standard_normal(n)
    else:
        a = 0.5 + 0.3 * rng.random(n)
    return np.clip(a, 0.005, None)


def optimize_smooth(f_init, time_limit, T_start=0.5, T_end=0.01, verbose=False):
    """
    Annealed smooth optimization using L-BFGS-B.
    Temperature T controls the sharpness of the log-sum-exp approximation.
    """
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    S = float(np.sum(f))
    target_sum = S  # Keep sum roughly constant
    f = f * (float(n) / S)  # Normalize so mean = 1
    target_sum = float(n)

    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()

    # Determine number of annealing stages
    n_stages = max(3, min(8, int(time_limit / 15)))
    temps = np.geomspace(T_start, T_end, n_stages)
    time_per_stage = time_limit / n_stages

    for s in range(n_stages):
        if time.monotonic() - t0 >= time_limit - 3.0:
            break
        T = temps[s]
        stage_end = t0 + (s + 1) * time_per_stage

        local_best = f.copy()
        local_best_score = score_fast(f)

        it_count = 0
        while time.monotonic() < stage_end - 1.0:
            f_val, f_grad = smooth_objective_wrapper(f, T, target_sum)
            it_count += 1

            result = minimize(
                smooth_objective_wrapper,
                f,
                args=(T, target_sum),
                jac=True,
                method='L-BFGS-B',
                bounds=[(0.001, 100.0)] * n,
                options={'maxiter': 50, 'maxfun': 200, 'gtol': 1e-6,
                         'ftol': 1e-10, 'iprint': -1}
            )

            f = np.maximum(result.x, 0.001)
            s_true = score_fast(f)

            if s_true < local_best_score:
                local_best_score = s_true
                local_best = f.copy()

            if s_true < best_score:
                best_score = s_true
                best_f = f.copy()

            if verbose:
                print(f"    T={T:.4f} it={it_count} smooth_f={f_val:.6f} true={s_true:.6f} best={best_score:.6f}")

            # Check for convergence
            if result.status == 0 and result.fun > f_val - 1e-8:
                break

            # If stuck, perturb
            if it_count % 5 == 0:
                rng_p = np.random.default_rng(it_count * 31 + s * 1000)
                f = f + 0.01 * rng_p.standard_normal(n)
                f = np.maximum(f, 0.001)

        f = local_best.copy()

    return best_f, best_score


def optimize_lp_fast(f_init, time_limit, K=25, verbose=False):
    """Fast LP-based optimization with reduced constraints."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 2.0:
            break

        c = fft_convolve(f, f)
        M = float(np.max(c))
        S = float(np.sum(f))

        # Build reduced constraint matrix (top K constraints)
        top_k_idx = np.argsort(c)[-K:]
        A = np.zeros((K, n))
        for i, k in enumerate(top_k_idx):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            for j in range(lo, hi):
                A[i, j] = f[k - j]

        b_ub = np.full(K, M)
        c_obj = -np.ones(n)
        timeout = min(2.0, remaining := time_limit - (time.monotonic() - t0))

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                          bounds=[(0, None)] * n,
                          method='highs',
                          options={'time_limit': timeout, 'presolve': True})
            if res.success:
                b = res.x
                Sb = float(np.sum(b))
                if Sb > 1e-12:
                    b = b * (S / Sb)
                    # Line search
                    best_t = 0.1
                    best_s = np.inf
                    for t_try in [0.05, 0.1, 0.2, 0.3]:
                        f_c = np.clip((1 - t_try) * f + t_try * b, 0.005, 1000.0)
                        s_c = score_fast(f_c)
                        if s_c < best_s:
                            best_s = s_c
                            best_t = t_try
                    f = np.clip((1 - best_t) * f + best_t * b, 0.005, 1000.0)
                    it += 1
                    s = best_s
                    if s < best_score - 1e-10:
                        best_score = s
                        best_f = f.copy()
                        stall = 0
                    else:
                        stall += 1
                else:
                    stall += 1
            else:
                stall += 1
        except Exception:
            stall += 1

        if stall % 20 == 0 and stall > 0:
            rng_esc = np.random.default_rng(777 + it * 13)
            f += 0.04 * rng_esc.standard_normal(n)
            f = np.clip(f, 0.005, 1000.0)
            stall = 0

        if verbose and it % 20 == 0:
            print(f"    [LP {time.monotonic()-t0:5.1f}s] it={it} best={best_score:.6f} stall={stall}")

    return best_f, best_score


def gradient_polish(f_init, time_limit, verbose=False):
    """Subgradient descent with momentum and adaptive step size."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    S = float(np.sum(f))
    f = f * (float(n) / S)
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    momentum = np.zeros(n)
    beta = 0.93
    lr = 0.002
    top_k = min(15, n // 10)
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 1.0:
            break

        c = fft_convolve(f, f)
        top_idx = np.argsort(c)[-top_k:]

        g = np.zeros(n)
        for k in top_idx:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                g[j] += f[k - j]
        g /= top_k
        g -= np.mean(g)

        momentum = beta * momentum + (1.0 - beta) * g
        f = f - lr * momentum
        f = np.clip(f, 0.001, 1000.0)
        f = f * (float(n) / float(np.sum(f)))

        it += 1
        if it % 50 == 0:
            s = score_fast(f)
            if s < best_score:
                best_score = s
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 10:
                    lr *= 0.7
                    no_improve = 0
            if verbose:
                print(f"    [grad {time.monotonic()-t0:5.1f}s] it={it} s={s:.6f} best={best_score:.6f} lr={lr:.5f}")

    s = score_fast(f)
    if s < best_score:
        best_score = s
        best_f = f.copy()
    return best_f, best_score


def local_search_final(f_init, time_limit, verbose=False):
    """Targeted local search for final polish."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    base_temp = 0.003
    it = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 1.0:
            break

        c = fft_convolve(f, f)
        peak_idx = int(np.argmax(c))
        candidate = f.copy()
        r = rng.random()

        if r < 0.45:
            # Targeted peak reduction
            contribs = np.zeros(n)
            for j in range(n):
                other = peak_idx - j
                if 0 <= other < n:
                    contribs[j] = f[j] * f[other]
            top_frac = max(3, n // 15)
            thr = np.sort(contribs)[-top_frac]
            mask = contribs >= thr
            candidate[mask] *= np.exp(rng.normal(-0.04, 0.025, size=int(mask.sum())))
            # Slight increase elsewhere
            other_idx = np.where(~mask)[0]
            if len(other_idx) > 0:
                inc_n = min(4, len(other_idx))
                chosen = rng.choice(other_idx, size=inc_n, replace=False)
                candidate[chosen] *= np.exp(rng.normal(0.03, 0.02, size=inc_n))
        elif r < 0.75:
            # Block perturbation
            width = max(2, int(rng.integers(2, max(3, n // 8))))
            start = int(rng.integers(0, max(1, n - width)))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.04, size=width))
        else:
            # Global
            candidate *= np.exp(rng.normal(0.0, 0.015, size=n))

        candidate = np.clip(candidate, 0.001, 1000.0)
        s = score_fast(candidate)

        if s < best_score:
            best_score = s
            best_f = candidate.copy()
            f = candidate
        else:
            current_s = score_fast(f)
            if rng.random() < np.exp(-(s - current_s) / max(base_temp, 1e-15)):
                f = candidate

        it += 1
        if verbose and it % 50 == 0:
            print(f"    [LS {time.monotonic()-t0:5.1f}s] it={it} best={best_score:.6f}")

    return best_f, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    Phase 1: Quick exploration with smooth L-BFGS-B across various lengths/inits
    Phase 2: Deep LP + gradient optimization on top candidates
    Phase 3: Final local search polish
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 25.0)
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Check for provided sequence
    provided = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.005, 1000.0)
        s = score_fast(provided)
        print(f"Provided: n={len(provided)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = provided.copy()

    candidates = []  # (score, sequence, n)

    # Phase 1: Smooth optimization exploration
    print(f"\n{'='*60}")
    print(f"Phase 1: Smooth L-BFGS-B exploration")
    print(f"{'='*60}")

    # Tasks: (n, init_kind, time_fraction)
    tasks = []
    if provided is not None:
        n_p = len(provided)
        tasks.append((n_p, 'provided', 0.12))

    # Focus on lengths where we can make progress
    for n in [256, 384, 512]:
        tasks.append((n, 'hann', 0.08))
        tasks.append((n, 'raised_cos', 0.06))
        tasks.append((n, 'gaussian', 0.05))
    for n in [768, 1024]:
        tasks.append((n, 'hann', 0.09))
        tasks.append((n, 'random_smooth', 0.06))
        tasks.append((n, 'bipolar', 0.05))
    tasks.append((1536, 'hann', 0.07))

    total_frac = sum(t[2] for t in tasks)
    p1_budget = (deadline - t_start) * 0.45

    for idx, (n, init_kind, frac) in enumerate(tasks):
        remaining_total = deadline - time.monotonic()
        if remaining_total < 60:
            break

        t_alloc = max(15.0, min(p1_budget * frac / total_frac, 120.0))
        t_alloc = min(t_alloc, remaining_total - 50)
        if t_alloc < 10:
            break

        # Generate init
        if init_kind == 'provided' and provided is not None:
            if len(provided) == n:
                f0 = provided.copy()
            else:
                x_old = np.linspace(0, 1, len(provided))
                x_new = np.linspace(0, 1, n)
                f0 = np.interp(x_new, x_old, provided)
        else:
            f0 = make_init(n, init_kind, rng)

        elapsed = time.monotonic() - t_start
        print(f"\n  [{elapsed:5.0f}s] n={n:4d} init={init_kind:14s} t={t_alloc:.0f}s")

        # Run smooth optimization
        f_best, s_best = optimize_smooth(f0, t_alloc, T_start=0.3, T_end=0.01, verbose=False)

        print(f"  => smooth score={s_best:.6f} (global best={best_score:.6f})")

        if s_best < best_score:
            best_score = s_best
            best_seq = f_best.copy()
            print(f"  *** NEW BEST: {best_score:.6f} ***")

        candidates.append((s_best, f_best.copy(), n))
        total_frac -= frac

    # Phase 2: Deep optimization on top candidates
    print(f"\n{'='*60}")
    print(f"Phase 2: Deep LP + gradient optimization")
    print(f"{'='*60}")

    candidates.sort(key=lambda x: x[0])
    top_n = min(6, len(candidates))
    top_candidates = candidates[:top_n]

    p2_budget = deadline - time.monotonic() - 80
    if p2_budget > 30:
        time_per = p2_budget / top_n

        for rank, (s_cand, f_cand, n) in enumerate(top_candidates):
            if time.monotonic() > deadline - 80:
                break
            t_alloc = max(10.0, min(time_per, 200.0))
            t_alloc = min(t_alloc, deadline - time.monotonic() - 40)
            if t_alloc < 10:
                break

            print(f"\n  [rank {rank}] n={n}, score={s_cand:.6f}, t={t_alloc:.0f}s")

            # LP optimization
            f_lp, s_lp = optimize_lp_fast(f_cand, t_alloc * 0.45, K=max(20, n // 12), verbose=False)
            current_best = f_lp if s_lp < s_cand else f_cand
            current_score = min(s_lp, s_cand)

            # Gradient polish
            if time.monotonic() < deadline - 50:
                t_grad = min(t_alloc * 0.3, deadline - time.monotonic() - 30)
                if t_grad > 8:
                    f_grad, s_grad = gradient_polish(current_best, t_grad, verbose=False)
                    if s_grad < current_score:
                        current_best = f_grad
                        current_score = s_grad

            if current_score < best_score:
                best_score = current_score
                best_seq = current_best.copy()
                print(f"  *** NEW BEST: {best_score:.6f} ***")

    # Phase 3: Final local search
    print(f"\n{'='*60}")
    print(f"Phase 3: Final local search polish")
    print(f"{'='*60}")

    if best_seq is not None and time.monotonic() < deadline - 8:
        t_ls = min(deadline - time.monotonic() - 5, 100)
        if t_ls > 10:
            print(f"  Local search with {t_ls:.0f}s budget")
            f_ls, s_ls = local_search_final(best_seq, t_ls, verbose=True)
            if s_ls < best_score:
                best_score = s_ls
                best_seq = f_ls.copy()
                print(f"  *** NEW BEST after LS: {best_score:.6f} ***")

    # Phase 4: Quick final gradient pass if time remains
    if best_seq is not None and time.monotonic() < deadline - 5:
        t_fin = deadline - time.monotonic() - 3
        if t_fin > 10:
            f_fin, s_fin = gradient_polish(best_seq, t_fin, verbose=False)
            if s_fin < best_score:
                best_score = s_fin
                best_seq = f_fin.copy()

    # Finalize
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'='*60}")
        print(f"FINAL RESULT: score={final_score:.6f}, n={len(final_list)}")
        print(f"{'='*60}")
        return final_list

    # Fallback
    return [1.0] * 256
