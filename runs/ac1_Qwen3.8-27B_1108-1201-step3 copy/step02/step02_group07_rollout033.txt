def fft_convolve(a, b):
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]

def score_fast(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))

def compute_smoothed_score_and_grad(a, T):
    """Compute smoothed score and its gradient."""
    n = len(a)
    c = fft_convolve(a, a)  # length 2n-1
    S = float(np.sum(a))
    
    # Log-sum-exp for numerical stability
    c_max = np.max(c)
    c_shifted = c - c_max
    exp_c = np.exp(c_shifted / T)
    log_sum_exp = c_max + T * np.log(np.sum(exp_c))
    w = exp_c / np.sum(exp_c)  # normalized weights
    
    smoothed_score = 2.0 * n * log_sum_exp / (S * S)
    
    # Gradient of log_sum_exp w.r.t. a[m]:
    # d log_sum_exp / da[m] = (1/T) * sum_k w[k] * d(c_k)/da[m]
    # d(c_k)/da[m] = a[k-m] (if 0<=k-m<n) + a[m-k] (if 0<=m-k<n)
    # 
    # Part 1: sum_k w[k] * a[k-m] = sum_j w[j+m] * a[j] (cross-correlation)
    # Part 2: sum_k w[k] * a[m-k] = sum_j w[m-j] * a[j] (convolution)
    
    N = 1
    while N < 3 * n:
        N <<= 1
    
    A = rfft(a, N)
    W = rfft(w, N)
    
    # Part 1: ifft(FFT(a) * conj(FFT(w))), take first n elements
    part1 = irfft(A * np.conj(W), N)[:n]
    # Part 2: ifft(FFT(a) * FFT(w)), take first n elements  
    part2 = irfft(A * W, N)[:n]
    
    grad_smoothed_max = (part1 + part2) / T
    grad = 2.0 * n * grad_smoothed_max / (S * S)
    
    return smoothed_score, grad

"""
Multi-phase optimizer for minimizing peak autoconvolution of non-negative sequences.
Combines log-sum-exp smoothed gradient descent (Adam), LP-based direction finding,
and local search. Uses FFT for fast convolution. Multi-start across lengths.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def fft_convolve(a, b=None):
    """Full linear convolution via FFT. If b is None, convolve a with itself."""
    if b is None:
        b = a
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]


def score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a)
    return float(2.0 * n * np.max(c) / (S * S))


def normalize(a, target_sum=None):
    """Normalize a to have the given target sum (default: len(a))."""
    n = len(a)
    if target_sum is None:
        target_sum = float(n)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.ones(n) * target_sum / n
    return a * (target_sum / S)


def smoothed_grad(a, T):
    """
    Compute gradient of LSE(conv(a,a)/T) w.r.t. a.
    Returns (lse_value, gradient).
    """
    n = len(a)
    c = fft_convolve(a)  # length 2n-1
    
    # Numerically stable log-sum-exp
    c_max = np.max(c)
    c_shifted = (c - c_max) / T
    # Clip to avoid overflow
    c_shifted = np.clip(c_shifted, -50, 50)
    exp_c = np.exp(c_shifted)
    lse = c_max + T * np.log(np.sum(exp_c))
    w = exp_c / np.sum(exp_c)
    
    # Gradient: (1/T) * sum_k w[k] * (a[k-m] + a[m-k]) for valid indices
    # Part 1: sum_j w[j+m] * a[j] (cross-correlation)
    # Part 2: sum_j w[m-j] * a[j] (convolution)
    
    N = 1
    while N < 3 * n:
        N <<= 1
    
    A = rfft(a, N)
    W = rfft(w, N)
    
    # Part 1: ifft(FFT(a) * conj(FFT(w)))[:n]
    part1 = irfft(A * np.conj(W), N)[:n]
    # Part 2: ifft(FFT(a) * FFT(w))[:n]
    part2 = irfft(A * W, N)[:n]
    
    grad = (part1 + part2) / T
    
    return lse, grad


def adam_smoothed_opt(a_init, T_start, T_end, n_steps, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """
    Minimize LSE(conv(a,a)/T) using Adam with temperature annealing.
    a is maintained with sum = n throughout.
    """
    a = a_init.astype(np.float64).copy()
    n = len(a)
    a = normalize(a)
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    best_a = a.copy()
    best_score = np.inf
    
    for t in range(n_steps):
        # Anneal temperature
        progress = t / max(1, n_steps - 1)
        T = T_start * (T_end / T_start) ** progress
        
        lse, grad = smoothed_grad(a, T)
        t += 1
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.01, None)
        a = normalize(a)
        
        if t % 50 == 0:
            s = score_fast(a)
            if s < best_score:
                best_score = s
                best_a = a.copy()
    
    s = score_fast(a)
    if s < best_score:
        best_score = s
        best_a = a.copy()
    
    return best_a, best_score


def build_A_conv(f):
    """Build matrix where A @ b = conv(f, b) (length 2n-1)."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_direction(f, M, timeout=5.0):
    """
    Solve: max sum(b) s.t. conv(f,b) <= M*1, b >= 0.
    Returns b (normalized to same sum as f) or None.
    """
    n = len(f)
    A = build_A_conv(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success and res.x is not None:
            b = np.maximum(res.x, 0)
            Sb = float(np.sum(b))
            if Sb > 1e-12:
                return b
    except Exception:
        pass
    return None


def lp_optimize(f_init, time_limit, verbose=False):
    """LP-based iterative optimization."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    f = normalize(f)
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 3.0:
            break
        
        c = fft_convolve(f)
        M = float(np.max(c))
        
        b = solve_lp_direction(f, M, timeout=3.0)
        if b is None:
            break
        
        Sf = float(np.sum(f))
        Sb = float(np.sum(b))
        if Sb < 1e-12:
            break
        b = b * (Sf / Sb)
        
        # Line search
        best_t = 0.1
        best_s = np.inf
        for t_try in [0.1, 0.05, 0.2, 0.02, 0.3]:
            f_cand = (1.0 - t_try) * f + t_try * b
            f_cand = np.clip(f_cand, 0.01, 1000.0)
            s_cand = score_fast(f_cand)
            if s_cand < best_s:
                best_s = s_cand
                best_t = t_try
        
        f = (1.0 - best_t) * f + best_t * b
        f = np.clip(f, 0.01, 1000.0)
        f = normalize(f)
        
        if best_s < best_score - 1e-10:
            best_score = best_s
            best_f = f.copy()
            stall = 0
        else:
            stall += 1
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"      [LP {elapsed:6.1f}s] score={best_s:.6f} best={best_score:.6f} stall={stall}")
        
        if stall >= 50:
            rng = np.random.default_rng(int(time.monotonic() * 1000))
            f += 0.02 * rng.standard_normal(n)
            f = np.clip(f, 0.01, None)
            f = normalize(f)
            stall = 0
    
    return best_f, best_score


def local_search(f_init, time_limit, verbose=False):
    """Perturbation-based local search."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    f = normalize(f)
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    temp = 0.001
    cooling = 0.999
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 >= time_limit - 1.0:
            break
        
        candidate = f.copy()
        r = rng.random()
        
        if r < 0.5:
            # Targeted: reduce peak contributions
            c = fft_convolve(f)
            peak_idx = int(np.argmax(c))
            contribs = np.zeros(n)
            for j in range(n):
                idx = peak_idx - j
                if 0 <= idx < n:
                    contribs[j] = f[j] * f[idx]
            top_k = max(3, n // 10)
            sorted_c = np.sort(contribs)[::-1]
            threshold = sorted_c[min(top_k - 1, n - 1)]
            top_mask = contribs >= threshold
            n_top = top_mask.sum()
            if n_top > 0:
                candidate[top_mask] *= np.exp(rng.normal(-0.02, 0.015, size=n_top))
            # Boost non-peak contributors
            other = ~top_mask
            if other.sum() > 0:
                idxs = np.where(other)[0]
                n_boost = min(8, len(idxs))
                chosen = rng.choice(idxs, size=n_boost, replace=False)
                candidate[chosen] *= np.exp(rng.normal(0.02, 0.015, size=n_boost))
        elif r < 0.7:
            # Local window perturbation
            width = max(1, int(rng.integers(1, max(2, n // 6))))
            start = int(rng.integers(0, max(1, n - width)))
            candidate[start:start + width] *= np.exp(rng.normal(0.0, 0.03, size=width))
        else:
            # Global small perturbation
            candidate *= np.exp(rng.normal(0.0, 0.01, size=n))
        
        candidate = np.clip(candidate, 0.01, 1000.0)
        candidate = normalize(candidate)
        new_score = score_fast(candidate)
        
        current_score = score_fast(f)
        if new_score < best_score:
            best_score = new_score
            best_f = candidate.copy()
            f = candidate
        elif rng.random() < np.exp(-(new_score - current_score) / max(temp, 1e-15)):
            f = candidate
        
        temp *= cooling
    
    return best_f, best_score


def make_init(n, init_type, rng):
    """Generate initial sequence."""
    t = np.arange(n, dtype=np.float64)
    if init_type == 'hann':
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / max(1, n - 1)))
        a += 0.05 * rng.standard_normal(n)
    elif init_type == 'flat':
        a = np.ones(n)
        a += 0.1 * rng.standard_normal(n)
    elif init_type == 'chirp':
        window = 0.5 * (1.0 + np.cos(np.pi * t / max(1, n - 1)))
        a = 0.5 + 0.5 * window + 0.1 * np.cos(2.0 * np.pi * t * t / (2.0 * n))
        a += 0.03 * rng.standard_normal(n)
    elif init_type == 'random':
        a = 0.5 + rng.random(n)
        w = 0.5 * (1.0 + np.cos(np.pi * t / max(1, n - 1)))
        a = a * (0.3 + 0.7 * w)
    else:
        a = np.ones(n)
    return np.clip(a, 0.01, None)

Let me also think about the problem from a different angle. The score is:
score = 2n * max_k c_k / S^2

where c_k = (a*a)[k] and S = sum(a).

If we define p_k = c_k / S^2 (normalized convolution), then sum(p_k) = 1 and score = 2n * max(p_k).

We want to make p as uniform as possible (each p_k ≈ 1/(2n-1)), which gives score ≈ 2n/(2n-1).

The constraint is that p comes from a non-negative a via p_k = (a*a)[k] / S^2.

This is a moment problem: given a probability distribution p on {0, ..., 2n-2}, find a non-negative a of length n such that (a*a)/(S^2) is close to p.

For the uniform target p_k = 1/(2n-1), we need (a*a)[k] = S^2/(2n-1) for all k. This is a system of 2n-1 quadratic equations in n unknowns (with the constraint sum(a) = S). For large n, this is overdetermined, so an exact solution likely doesn't exist. But we can find an approximate one.

This perspective suggests using **iterative proportional fitting** or **Sinkhorn-like** algorithms. The idea:
1. Start with a ≈ uniform (so a*a ≈ triangular, with peak at the center)
2. The convolution has a "triangular" shape (high in the middle, low at the edges)
3. We want to "flatten" it by adjusting a

One approach: **multiplicative updates inspired by NMF**.

Define the "residual" at position k: r_k = target_k - (a*a)[k], where target_k = S^2/(2n-1).

A natural update: increase a[j] for positions j that contribute "too little" to the peaks, and decrease a[j] for positions that contribute "too much."

Specifically, for a given j, a[j] contributes to (a*a)[k] for k in [j, j+n-1] (with weight 2*a[k-j] or a[j] for k=2j).

The "excess" at position j could be defined as:
excess[j] = sum_{k in [j, j+n-1]} w_k * (a*a)[k] - target

where w_k are weights that emphasize the peaks.
