Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. Smoothed L-BFGS optimization with log-sum-exp continuation
2. LP-based direction finding with improved line search
3. Nesterov-accelerated projected subgradient
4. Targeted local search with quadratic approximation
5. Multi-start with diverse initializations and sequence lengths

import numpy as np
import scipy.optimize
import time


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_value(a):
    """Objective: 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()


def softmax(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def smooth_objective(z, n, t):
    """Smoothed objective value."""
    a = softmax(z)
    conv = fast_convolve(a)
    conv_shifted = conv - conv.max()
    return 2.0 * n * conv.max() + (2.0 * n / t) * np.log(np.sum(np.exp(t * conv_shifted)))


def smooth_gradient(z, n, t):
    """Gradient of smoothed objective w.r.t. z (softmax parameterization)."""
    a = softmax(z)
    conv = fast_convolve(a)
    conv_shifted = conv - conv.max()
    exp_conv = np.exp(t * conv_shifted)
    w = exp_conv / exp_conv.sum()
    
    # Cross-correlation for grad_a
    size = 1
    while size < 3 * n:
        size *= 2
    W = np.fft.rfft(w, size)
    A_pad = np.zeros(size)
    A_pad[:n] = a
    A = np.fft.rfft(A_pad, size)
    cross_corr = np.fft.irfft(np.conj(W) * A, size)[:n]
    
    # grad_a[i] = 2 * cross_corr[i] - w[2i]*a[i] (diagonal correction)
    grad_a = 2.0 * cross_corr
    diag_idx = 2 * np.arange(n)
    valid = diag_idx < (2 * n - 1)
    grad_a[valid] -= w[diag_idx[valid]] * a[valid]
    grad_a *= 2.0 * n
    
    # Chain rule through softmax: grad_z = a * (grad_a - dot(grad_a, a))
    grad_z = a * (grad_a - np.dot(grad_a, a))
    return grad_z


def lbfgs_optimize(a0, n, time_limit, t_values=None):
    """Optimize using L-BFGS with log-sum-exp continuation."""
    if t_values is None:
        t_values = [0.5, 2.0, 10.0, 50.0, 200.0, 1000.0]
    
    a = np.maximum(a0.astype(np.float64), 1e-30).copy()
    a /= a.sum()
    z = np.log(a)
    
    t0 = time.monotonic()
    for t in t_values:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2.0:

        maxiter = max(10, int(remaining * 30))  # ~30 iterations per second
        try:
            res = scipy.optimize.minimize(
                smooth_objective, z, method='L-BFGS-B',
                jac=smooth_gradient, args=(n, t),
                options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10}

            z = res.x
        except Exception:
            break
        
        a = softmax(z)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:

        a /= s
    
    return a, obj_value(a)


def solve_lp(f, M):
    """LP: find g >= 0 maximizing sum(g) s.t. conv(f,g) <= M."""
    n = len(f)
    # Build constraint matrix A (2n-1) x n
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(2 * n - 1, j + n)
        A[j:end, j] = f[:end - j]
    
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return f.copy()

    no_improve = 0

        M = conv.max()
        if M < 1e-15:

        g = solve_lp(f, M)

            no_improve += 5

            best_t = 0.0
            best_o = best_obj
            for t in np.concatenate([np.linspace(0.01, 0.3, 10), np.linspace(0.3, 0.7, 8), np.linspace(0.7, 1.0, 6)]):
                a_new = (1.0 - t) * f + t * g
                a_new = np.maximum(a_new, 0.0)
                ns = a_new.sum()
                if ns < 1e-12:

def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)

        return np.ones(n) / n

    cssv = np.cumsum(u)
    rho = np.findlast((u - (cssv - 1.0) / np.arange(1, n + 1)) > 0)
    if rho is None:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w / w.sum()

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
1. Smoothed L-BFGS optimization with log-sum-exp continuation (new)
2. LP-based direction finding with adaptive line search
3. Nesterov-accelerated projected subgradient (new)

import numpy as np
import scipy.optimize
import time


def fast_convolve(a):
    """Linear auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_value(a):
    """Objective: 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def smooth_obj_val(z, n, t):
    """Smoothed objective value via log-sum-exp."""
    a = softmax_fn(z)
    conv = fast_convolve(a)
    conv_shifted = conv - conv.max()
    return 2.0 * n * conv.max() + (2.0 * n / t) * np.log(np.sum(np.exp(t * conv_shifted)))


def smooth_grad_fn(z, n, t):
    """Gradient of smoothed objective w.r.t. z (softmax parameterization)."""
    a = softmax_fn(z)
    conv = fast_convolve(a)
    conv_shifted = conv - conv.max()
    exp_conv = np.exp(t * conv_shifted)
    w = exp_conv / exp_conv.sum()

    # Cross-correlation: cross_corr[i] = sum_k w[k] * a[k-i]
    size = 1
    while size < 3 * n:
        size *= 2
    W = np.fft.rfft(w, size)
    A_pad = np.zeros(size)
    A_pad[:n] = a
    A = np.fft.rfft(A_pad, size)
    cross_corr = np.fft.irfft(np.conj(W) * A, size)[:n]

    # grad_a[i] = 2 * cross_corr[i] - w[2i]*a[i] (diagonal correction)
    grad_a = 2.0 * cross_corr
    diag_idx = 2 * np.arange(n)
    valid = diag_idx < (2 * n - 1)
    grad_a[valid] -= w[diag_idx[valid]] * a[valid]
    grad_a *= 2.0 * n

    # Chain rule through softmax
    grad_z = a * (grad_a - np.dot(grad_a, a))
    return grad_z


def lbfgs_optimize(a0, time_limit, t_values=None):
    """Optimize via L-BFGS with log-sum-exp continuation on softmax parameterization."""
    n = len(a0)

        t_values = [1.0, 5.0, 20.0, 100.0, 500.0]

            break

        try:
            res = scipy.optimize.minimize(
                smooth_obj_val, z, method='L-BFGS-B',
                jac=smooth_grad_fn, args=(n, t),
                options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10}

            if res.x is not None:
                z = res.x
        except Exception:

        a = softmax_fn(z)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            break
        a /= s

def solve_lp(f, M):
    """LP: find g >= 0 maximizing sum(g) s.t. conv(f,g)[k] <= M for all k."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        end = min(2 * n - 1, j + n)
        A[j:end, j] = f[:end - j]

    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if res.success:
            g = np.maximum(res.x, 0.0)
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return f.copy()

                    a_trial /= ts
                    o = obj_value(a_trial)
                    if o < best_obj - 1e-12:
                        best_obj = o
                        a = a_trial.copy()
                        found = True
                        break

        # Try three-way moves (reduce two peaks, increase one safe)
        if not found and len(top_danger) >= 2:
            d1, d2 = top_danger[0], top_danger[1]
            for si in bottom_safe[:5]:
                if d1 == si or d2 == si:
                    continue
                if a[d1] < 1e-15 or a[d2] < 1e-15:
                    continue
                for f1 in [0.2, 0.4]:
                    for f2 in [0.2, 0.4]:
                        amt1 = a[d1] * f1
                        amt2 = a[d2] * f2
                        a_trial = a.copy()
                        a_trial[d1] -= amt1
                        a_trial[d2] -= amt2
                        a_trial[si] += amt1 + amt2
                        a_trial = np.maximum(a_trial, 0.0)
                        ts = a_trial.sum()
                        if ts < 1e-12:
                            continue
                        a_trial /= ts
                        o = obj_value(a_trial)
                        if o < best_obj - 1e-12:
                            best_obj = o
                            a = a_trial.copy()
                            found = True
                            break
                    if found:
                        break
                if found:
                    break

        if not found:
            stuck += 1
            # Random perturbation to escape local minima
            if stuck % 30 == 0:
                for _ in range(3):
                    idx = srng.choice(n, size=3, replace=False)
                    if a[idx[0]] > 1e-10 and a[idx[1]] > 1e-10:
                        amt = min(a[idx[0]], a[idx[1]]) * 0.05
                        a_trial = a.copy()
                        a_trial[idx[0]] -= amt
                        a_trial[idx[1]] -= amt
                        a_trial[idx[2]] += 2.0 * amt
                        a_trial = np.maximum(a_trial, 0)
                        a_trial /= a_trial.sum()
                        o = obj_value(a_trial)
                        if o < best_obj:
                            a = a_trial
                            best_obj = o
                stuck = 0
        else:
            stuck = max(0, stuck - 2)

    return a, best_obj


def get_parent_resampled(n):
    """Resample the parent sequence to length n."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim != 1 or p.size == 0 or not np.isfinite(p).all():
            return None
        x_new = np.linspace(0, 1, n)
        x_old = np.linspace(0, 1, p.size)
        pc = np.interp(x_new, x_old, p)
        pc = np.maximum(pc, 0.0)
        s = pc.sum()
        if s < 1e-10:
            return None
        return pc / s
    except Exception:
        return None


def make_initializations(n, rng):
    """Generate diverse initializations for length n."""
    x = np.arange(n, dtype=np.float64)
    inits = {}

    inits['uniform'] = np.ones(n) / n

    for sf in [0.15, 0.25, 0.4, 0.6]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (max(1.0, sf * n))) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf}'] = g

    if n > 1:
        hann = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n - 1)))
    else:
        hann = np.ones(n)
    hann = np.maximum(hann, 0.0)
    inits['hann'] = hann / hann.sum()

    r = rng.dirichlet(np.ones(n) * 0.3)
    inits['random'] = r

    # Bimodal
    bim = np.exp(-0.5 * ((x - 0.2 * n) / (0.12 * n)) ** 2) + \
          np.exp(-0.5 * ((x - 0.8 * n) / (0.12 * n)) ** 2)
    bim /= bim.sum()
    inits['bimodal'] = bim

    # Triangle
    tri = np.minimum(x, n - 1 - x) + 1.0
    inits['triangle'] = tri / tri.sum()

    parent = get_parent_resampled(n)
    if parent is not None:
        inits['parent'] = parent
        noisy = parent + 0.05 * rng.standard_normal(n)
        noisy = np.maximum(noisy, 0.0)
        noisy /= noisy.sum()
        inits['parent_noisy'] = noisy

    return inits


def hybrid_optimize(a0, time_limit):
    """Alternate LP, L-BFGS, Nesterov, and local search."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    a /= a.sum()
    best_a = a.copy()
    best_obj = obj_value(a)
    t0 = time.monotonic()

    methods = ['lp', 'lbfgs', 'nesterov', 'local']
    method_time = max(5.0, time_limit / 20.0)
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3.0:
            break

        method = methods[iter_count % len(methods)]
        t_alloc = min(method_time, remaining - 2.0)
        if t_alloc < 2.0:
            break

        if method == 'lp':
            a_opt, obj_opt = lp_optimize(a, time_limit=t_alloc, max_no_improve=50)
        elif method == 'lbfgs':
            t_vals = [50.0, 200.0, 1000.0]
            a_opt, obj_opt = lbfgs_optimize(a, time_limit=t_alloc, t_values=t_vals)
        elif method == 'nesterov':
            a_opt, obj_opt = nesterov_optimize(a, time_limit=t_alloc)
        else:
            a_opt, obj_opt = local_search(a, time_limit=t_alloc, max_stuck=200,
                                          rng_seed=iter_count)

        if obj_opt < best_obj:
            best_obj = obj_opt
            best_a = a_opt.copy()
            a = a_opt.copy()
            print(f"    Hybrid [{method}] obj={best_obj:.8f}")

        iter_count += 1

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Uses hybrid optimization (LP + L-BFGS + Nesterov + local search)
    with multi-start to minimize 2n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    total_budget = float(budget_s)
    deadline = time.monotonic() + total_budget - 30.0

    best_sequence = None
    best_score = float('inf')

    lengths = [256, 512, 768, 1024]

    # --- Phase 1: Multi-start scan (50% of budget) ---
    phase1_end = time.monotonic() + total_budget * 0.50
    print(f"Phase 1: Multi-start scan (budget {total_budget * 0.50:.0f}s)...")

    scan_configs = []
    for n in lengths:
        inits = make_initializations(n, rng)
        for name, a0 in inits.items():
            priority = 0 if name == 'parent' else (1 if name == 'parent_noisy' else (2 if n == 512 else 3))
            scan_configs.append((priority, n, name, a0))

    scan_configs.sort(key=lambda x: (x[0], x[1]))

    phase1_time = phase1_end - time.monotonic()
    n_configs = len(scan_configs)
    time_per_config = max(5.0, phase1_time / max(1, n_configs))

    results = []
    for i, (prio, n, name, a0) in enumerate(scan_configs):
        remaining_dl = deadline - time.monotonic() - 60.0
        if remaining_dl < 30:
            break

        t_alloc = min(time_per_config, remaining_dl - 60.0)
        if t_alloc < 4.0:
            break

        # Quick LP
        a_lp, obj_lp = lp_optimize(a0, time_limit=min(t_alloc * 0.4, 30.0), max_no_improve=30)
        # Quick L-BFGS
        t_remain = min(t_alloc * 0.4, 30.0)
        a_lb, obj_lb = lbfgs_optimize(a_lp, time_limit=t_remain,
                                      t_values=[20.0, 100.0, 500.0])
        # Quick local
        a_ls, obj_ls = local_search(a_lb, time_limit=min(t_alloc * 0.2, 15.0),
                                    max_stuck=100, rng_seed=i)

        candidates = [(a_lp, obj_lp), (a_lb, obj_lb), (a_ls, obj_ls)]
        for a_cand, obj_cand in candidates:
            score = float(evaluate_sequence(a_cand.tolist()))
            if np.isfinite(score):
                results.append((score, n, name, a_cand))
                if score < best_score:
                    best_score = score
                    best_sequence = a_cand.copy()

    results.sort(key=lambda x: x[0])
    print(f"  Phase 1 best: {best_score:.6f}")

    # --- Phase 2: Deep optimization on top candidates (35% of budget) ---
    phase2_end = time.monotonic() + total_budget * 0.35
    print(f"Phase 2: Deep optimization (budget {total_budget * 0.35:.0f}s)...")

    top_configs = []
    seen = set()
    for score, n, name, a_opt in results:
        key = (n, name)
        if key not in seen and len(top_configs) < 6:
            top_configs.append((n, name, a_opt))
            seen.add(key)

    if best_sequence is not None:
        top_configs.append((len(best_sequence), 'best', best_sequence.copy()))

    phase2_time = phase2_end - time.monotonic()
    if phase2_time < 10:
        phase2_time = 10
    time_per_deep = max(30.0, phase2_time / max(1, len(top_configs)))

    for i, (n, name, a0) in enumerate(top_configs):
        remaining_dl = deadline - time.monotonic() - 60.0
        if remaining_dl < 30:
            break
        t_alloc = min(time_per_deep, remaining_dl - 30.0)
        if t_alloc < 10.0:
            break

        a_opt, obj_opt = hybrid_optimize(a0, time_limit=t_alloc)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Deep [{n},{name}]: score={score:.6f}")

    print(f"  Phase 2 best: {best_score:.6f}")

    # --- Phase 3: Intensive polish (remaining time) ---
    print(f"Phase 3: Intensive polish...")
    if best_sequence is not None:
        polish_end = deadline - 15.0
        no_improve_count = 0
        while time.monotonic() < polish_end and no_improve_count < 10:
            remaining = polish_end - time.monotonic()
            if remaining < 10:
                break

            # LP polish
            t_lp = min(remaining * 0.3, 120.0)
            a_lp, obj_lp = lp_optimize(best_sequence, time_limit=t_lp, max_no_improve=60)
            score = float(evaluate_sequence(a_lp.tolist()))
            improved = False
            if np.isfinite(score) and score < best_score - 1e-9:
                best_score = score
                best_sequence = a_lp.copy()
                improved = True
                print(f"  Polish LP: {score:.6f}")

            # L-BFGS polish with high t
            t_lb = min(remaining * 0.2, 60.0)
            a_lb, obj_lb = lbfgs_optimize(best_sequence, time_limit=t_lb,
                                          t_values=[200.0, 1000.0, 5000.0])
            score = float(evaluate_sequence(a_lb.tolist()))
            if np.isfinite(score) and score < best_score - 1e-9:
                best_score = score
                best_sequence = a_lb.copy()
                improved = True
                print(f"  Polish L-BFGS: {score:.6f}")

            # Local search polish
            t_ls = min(remaining * 0.3, 90.0)
            a_ls, obj_ls = local_search(best_sequence, time_limit=t_ls,
                                        max_stuck=2000,
                                        rng_seed=int(time.monotonic() * 1000) % 100000)
            score = float(evaluate_sequence(a_ls.tolist()))
            if np.isfinite(score) and score < best_score - 1e-9:
                best_score = score
                best_sequence = a_ls.copy()
                improved = True
                print(f"  Polish LS: {score:.6f}")

            # Nesterov polish
            t_ne = min(remaining * 0.15, 45.0)
            a_ne, obj_ne = nesterov_optimize(best_sequence, time_limit=t_ne)
            score = float(evaluate_sequence(a_ne.tolist()))
            if np.isfinite(score) and score < best_score - 1e-9:
                best_score = score
                best_sequence = a_ne.copy()
                improved = True
                print(f"  Polish Nesterov: {score:.6f}")

            if not improved:
                no_improve_count += 1

    # --- Final validation ---
    if best_sequence is None:
        best_sequence = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Try a few random perturbations of the best
    for _ in range(10):
        perturb = best_sequence + 0.01 * rng.standard_normal(len(best_sequence))
        perturb = np.maximum(perturb, 0)
        perturb /= perturb.sum()
        score = float(evaluate_sequence(perturb.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = perturb.copy()

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.6f}")
    print(f"Sequence length: {len(best_sequence)}")
    return [float(x) for x in best_sequence]
