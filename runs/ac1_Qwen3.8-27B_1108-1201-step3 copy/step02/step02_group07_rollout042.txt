"""
Multi-phase optimization for minimizing peak autocorrelation of non-negative sequences.
Uses log-sum-exp smoothed gradient, peak-targeted coordinate descent, and adaptive local search.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]


def score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def smooth_score(a, T=1.0):
    """Smooth approximation of the score using log-sum-exp."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.01, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    # Log-sum-exp for numerical stability
    c_max = np.max(c)
    lse = c_max + T * np.log(np.sum(np.exp((c - c_max) / T)))
    return float(2.0 * n * lse / (S * S))

    c = fft_convolve(a, a)
    c_max = np.max(c)
    p = np.exp((c - c_max) / T)
    p /= np.sum(p)
    lse = c_max + T * np.log(np.sum(np.exp((c - c_max) / T)))
    
    # Compute h_m = sum_k p_k * dc_k/da_m

def peak_targeted_descent(f_init, time_limit, verbose=True):
    """Coordinate descent that specifically targets the peak lag."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    S = float(np.sum(f))
    f = f / S * np.sqrt(2.0 * n)
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    
    while time.monotonic() - t0 < time_limit - 2:
        c = fft_convolve(f, f)
        peak_idx = int(np.argmax(c))
        M = c[peak_idx]
        S = float(np.sum(f))
        
        # Find which elements contribute most to the peak
        # c[peak_idx] = sum_j a[j] * a[peak_idx - j]
        contribs = np.zeros(n)
        for j in range(n):
            k = peak_idx - j
            if 0 <= k < n:
                contribs[j] = f[j] * f[k]
        
        # Perturb top contributors downward, others slightly upward
        top_k = max(5, n // 10)
        top_mask = contribs >= np.sort(contribs)[-top_k]
        
        # Randomly choose a subset of top contributors to reduce
        num_reduce = max(2, top_k // 4)
        top_indices = np.where(top_mask)[0]
        reduce_indices = rng.choice(top_indices, size=min(num_reduce, len(top_indices)), replace=False)
        
        # Amount to reduce (keep sum approximately constant)
        total_reduce = sum(f[i] for i in reduce_indices) * 0.05
        f[reduce_indices] -= total_reduce / len(reduce_indices)
        
        # Add to random non-contributing elements
        non_contrib = np.where(~top_mask)[0]
        if len(non_contrib) > 0:
            add_indices = rng.choice(non_contrib, size=min(5, len(non_contrib)), replace=False)
            f[add_indices] += total_reduce / len(add_indices)
        
        f = np.clip(f, 0.01, 1000.0)
        new_score = score_fast(f)
        
        if new_score < best_score:
            best_score = new_score
            best_f = f.copy()
        
        # Occasionally reset to best
        if rng.random() < 0.1:
            f = best_f.copy()
    
    return best_f, best_score


def coordinate_descent(f_init, time_limit, verbose=True):
    """Cyclic coordinate descent on the score."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    S = float(np.sum(f))
    f = f / S * np.sqrt(2.0 * n)
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    
    best_score_cd = best_score
    it = 0
    
    while time.monotonic() - t0 < time_limit - 2:
        # Random cyclic order
        order = rng.permutation(n)
        
        for m in order:
            if time.monotonic() - t0 >= time_limit - 2:
                    break
            
            # Try several perturbations of a_m
            base = f[m]
            S_other = float(np.sum(f)) - base
            
            # The score as a function of a_m (with others fixed):
            # c_k = c_k_base + 2*a_m*a_{k-m} (or a_m^2 if k=2m)
            # S = S_other + a_m
            # score = 2n * max_k(c_k_base + contrib_k(a_m)) / (S_other + a_m)^2
            
            # Try a few values
            candidates = [base * (1 + d) for d in [-0.1, -0.05, -0.02, 0.02, 0.05, 0.1]]
            candidates = [max(0.01, c) for c in candidates]
            
            best_val = score_fast(f)
            best_cand = base
            
            for cand in candidates:
                f[m] = cand
                val = score_fast(f)
                if val < best_val:
                    best_val = val
                    best_cand = cand
            
            f[m] = best_cand
            
        it += 1
        s = score_fast(f)
        if s < best_score:
            best_score = s
            best_f = f.copy()
        
        if verbose and it % 10 == 0:
            print(f"    [coord {time.monotonic()-t0:6.1f}s] epoch={it} score={s:.6f} best={best_score:.6f}")
        
        # Early stop if no improvement for a while
        if s >= best_score and it > 50:
            break
    
    return best_f, best_score


def local_search_v2(f_init, time_limit, verbose=True):
    """Improved local search with multiple move types and adaptive temperature."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    rng = np.random.default_rng(4242)
    temp = 0.001
    cooling = 0.9995
    current_score = best_score
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit - 2:
        candidate = f.copy()
        r = rng.random()
        
        if r < 0.35:
            # Peak-targeted: reduce contributors to peak, boost others
            c = fft_convolve(f, f)
            peak_idx = int(np.argmax(c))
            contribs = np.zeros(n)
            for j in range(n):
                k = peak_idx - j
                if 0 <= k < n:
                    contribs[j] = f[j] * f[k]
            
            top_k = max(3, n // 15)
            sorted_contribs = np.sort(contribs)
            threshold = sorted_contribs[-top_k]
            top_mask = contribs >= threshold
            
            # Reduce top contributors
            reduce_amt = 0.02 + 0.03 * rng.random()
            candidate[top_mask] *= (1.0 - reduce_amt)
            
            # Boost non-contributors
            other = ~top_mask
            if other.sum() > 0:
                boost_idx = rng.choice(np.where(other)[0], 
                                       size=min(8, other.sum()), replace=False)
                candidate[boost_idx] *= (1.0 + reduce_amt * top_mask.sum() / max(1, len(boost_idx)))
                
        elif r < 0.55:
            # Swap: take from one position, give to another
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                amt = f[i] * 0.05
                candidate[i] -= amt
                candidate[j] += amt
                
        elif r < 0.75:
            # Window perturbation
            width = int(rng.integers(2, max(3, n // 5)))
            start = int(rng.integers(0, max(1, n - width)))
            scale = np.exp(rng.normal(0, 0.03, width))
            candidate[start:start+width] *= scale
            
        elif r < 0.9:
            # Global scale noise
            candidate *= np.exp(rng.normal(0, 0.01, n))
            
        else:
            # Pair swap (move mass from high-contrib to low-contrib)
            c = fft_convolve(f, f)
            peak_idx = int(np.argmax(c))
            contribs = np.zeros(n)
            for j in range(n):
                k = peak_idx - j
                if 0 <= k < n:
                    contribs[j] = f[j] * f[k]
            
            # Take from high contributor, give to low contributor
            high_idx = np.argsort(contribs)[-5:]
            low_idx = np.argsort(contribs)[:5]
            i = high_idx[rng.integers(0, len(high_idx))]
            j = low_idx[rng.integers(0, len(low_idx))]
            amt = f[i] * 0.03
            candidate[i] -= amt
            candidate[j] += amt
        
        candidate = np.clip(candidate, 0.001, 1000.0)
        new_score = score_fast(candidate)
        
        if new_score < best_score:
            best_score = new_score
            best_f = candidate.copy()
            f = candidate
            current_score = new_score
            no_improve = 0
        elif rng.random() < np.exp(-(new_score - current_score) / max(temp, 1e-15)):
            f = candidate
            current_score = new_score
            no_improve += 1
        else:
            no_improve += 1
        
        temp *= cooling
        
        # Restart if stuck
        if no_improve > 200:
            f = best_f.copy()
            current_score = best_score
            no_improve = 0
            temp = 0.001
    
    return best_f, best_score


def lp_optimize(f_init, time_limit, verbose=True):
    """LP-based direction finding (similar to previous but improved)."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    S = float(np.sum(f))
    f = f / S * np.sqrt(2.0 * n)
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit - 5:
        c = fft_convolve(f, f)
        M = float(np.max(c))
        
        # Build constraint matrix
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for j in range(n):
            A[j:j+n, j] = f
        
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                         bounds=[(0, None)] * n,
                         method='highs',
                         options={'time_limit': 2.0, 'presolve': True})
            if not res.success:
                break
            b = res.x
        except Exception:
            break
        
        Sb = float(np.sum(b))
        if Sb < 1e-12:
            break
        b = b * (float(np.sum(f)) / Sb)
        
        # Line search
        best_t = 0.05
        best_s = np.inf
        for t_try in [0.1, 0.05, 0.02, 0.01]:
            f_cand = (1.0 - t_try) * f + t_try * b
            f_cand = np.clip(f_cand, 0.01, 1000.0)
            s_cand = score_fast(f_cand)
            if s_cand < best_s:
                best_s = s_cand
                best_t = t_try
        
        f = (1.0 - best_t) * f + best_t * b
        f = np.clip(f, 0.01, 1000.0)
        
        if best_s < best_score - 1e-10:
            best_score = best_s
            best_f = f.copy()
            stall = 0
        else:
            stall += 1
        
        if verbose and stall % 20 == 0:
            print(f"    [LP {time.monotonic()-t0:6.1f}s] score={best_s:.6f} best={best_score:.6f} stall={stall}")
        
        if stall >= 60:
            # Escape
            rng_esc = np.random.default_rng(int(time.monotonic() * 1000))
            f += 0.02 * rng_esc.standard_normal(n)
            f = np.clip(f, 0.01, 1000.0)
            stall = 0
    
    return best_f, best_score


def make_various_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    t = np.arange(n, dtype=np.float64)
    
    # Hann
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / max(1, n - 1)))
    a += 0.03 * rng.standard_normal(n)
    inits.append(np.clip(a, 0.01, None))
    
    # Raised cosine variant
    a = 0.3 + 0.7 * 0.5 * (1.0 + np.cos(2.0 * np.pi * t / max(1, n - 1)))
    a += 0.02 * rng.standard_normal(n)
    inits.append(np.clip(a, 0.01, None))
    
    # Flat top
    a = 0.2156 + 0.4166*np.cos(2*np.pi*t/n) + 0.2776*np.cos(4*np.pi*t/n) + 0.0835*np.cos(6*np.pi*t/n) + 0.0069*np.cos(8*np.pi*t/n)
    a = np.abs(a) + 0.01
    inits.append(np.clip(a, 0.01, None))
    
    # Gaussian
    center = n / 2.0
    sigma = n / 4.0
    a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
    a += 0.02 * rng.standard_normal(n)
    inits.append(np.clip(a, 0.01, None))
    
    # Sinc-like (windowed)
    a = np.sinc(t / (n/4.0) - 0.5)
    a = np.abs(a) * 0.5 * (1.0 + np.cos(2.0 * np.pi * t / max(1, n - 1)))
    a += 0.01
    inits.append(np.clip(a, 0.01, None))
    
    # Random
    a = 0.3 + 0.7 * rng.random(n)
    a *= 0.5 * (1.0 + np.cos(2.0 * np.pi * t / max(1, n - 1)))
    inits.append(np.clip(a, 0.01, None))
    
    # Triangular
    w = 1.0 - np.abs(2.0 * t / max(1, n - 1) - 1.0)
    a = 0.2 + 0.8 * w
    a += 0.02 * rng.standard_normal(n)
    inits.append(np.clip(a, 0.01, None))
    
    return inits

"""
Multi-phase optimization for minimizing peak autocorrelation of non-negative sequences.
Combines smooth gradient (log-sum-exp) optimization, analytic coordinate descent,
LP-based refinement, and adaptive local search. Exploits scale invariance of the score.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog


def fft_convolve(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1
    while size < n + m - 1:
        size <<= 1
    return irfft(rfft(a, size) * rfft(b, size), size)[:n + m - 1]


def score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fft_convolve(a, a)
    return float(2.0 * n * np.max(c) / (S * S))


def smooth_obj_and_grad(a, T=1.0):
    """Smooth log-sum-exp objective and its gradient (FFT-accelerated)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    S = float(np.sum(a))
    if S < 0.01:
        return 1e10, np.zeros(n)
    c = fft_convolve(a, a)
    c_max = np.max(c)
    p = np.exp((c - c_max) / T)
    p_sum = np.sum(p)
    lse = c_max + T * np.log(p_sum)
    # h[m] = 2 * sum_j a[j] * p[m+j] via cross-correlation trick
    size = 2 * n - 1
    p_rev = p[::-1]
    conv_ap = fft_convolve(a, p_rev)
    h = 2.0 * conv_ap[size - 1 - np.arange(n)]
    score_val = 2.0 * n * lse / (S * S)
    grad = 2.0 * n * (S * h - 2.0 * lse) / (S * S * S)
    return score_val, grad


def optimize_smoothed(f_init, time_limit, T_init=4.0, T_min=0.15, verbose=True):
    """Annealed L-BFGS-B on smoothed objective."""
    n = len(f_init)
    f = f_init.astype(np.float64).copy()
    S = float(np.sum(f))
    if S > 0:
        f = f / S * np.sqrt(2.0 * n)
    best_f = f.copy()
    best_score = score_fast(f)
    t0 = time.monotonic()

    T_values = np.geomspace(T_init, T_min, 10)
    iters_per_T = max(10, 200000 // n)

    for Ti in T_values:
        if time.monotonic() - t0 >= time_limit - 3:
            break

        def obj_grad(x):
            x = np.clip(x, 0.01, 1000.0)
            val, grad = smooth_obj_and_grad(x, Ti)
            if np.isinf(val) or np.isnan(val):
                return 1e10, np.zeros_like(x)
            if not np.all(np.isfinite(grad)):
                grad = np.nan_to_num(grad)
            return val, grad

        try:
            result = minimize(obj_grad, f, method='L-BFGS-B',
                             bounds=[(0.01, 1000.0)] * n,
                             options={'maxiter': iters_per_T, 'maxls': 30,
                                      'ftol': 1e-14, 'gtol': 1e-10})
            f = np.clip(result.x, 0.01, 1000.0)
        except Exception:
            pass

        s = score_fast(f)
        if s < best_score:

            best_f = f.copy()

def lp_refine(f_init, time_limit, verbose=True):

    n = len(f_init)
    f = f_init.astype(np.float64).copy()

        M = float(np.max(c))

        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for j in range(n):
            A[j:j + n, j] = f

        try:
            res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2 * n - 1, M),
                         bounds=[(0, None)] * n, method='highs',
                         options={'time_limit': 2.0, 'presolve': True})
            if not res.success:

            b = res.x
        except Exception:
            break

        Sb = float(np.sum(b))
        if Sb < 1e-12:
            break
        b = b * (float(np.sum(f)) / Sb)

        best_t = 0.05
        best_s = np.inf

            if s_cand < best_s:
                best_s = s_cand
                best_t = t_try

        f = (1.0 - best_t) * f + best_t * b
        f = np.clip(f, 0.01, 1000.0)

        if best_s < best_score - 1e-10:
            best_score = best_s
            best_f = f.copy()
            stall = 0
        else:
            stall += 1

        if verbose and stall % 20 == 0 and stall > 0:

        temp *= cooling

        if no_improve > 300:
            f = best_f.copy()
            current_score = best_score
            no_improve = 0
            temp = 0.0005

        if verbose and int(time.monotonic() - t0) % 30 < 1:
            print(f"    [LS {time.monotonic()-t0:6.1f}s] score={new_score:.6f} best={best_score:.6f}")

    return best_f, best_score


def make_diverse_inits(n, rng):
    """Generate diverse initializations."""
    t = np.arange(n, dtype=np.float64)
    inits = []

    # Hann window
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / max(1, n - 1)))
    a += 0.02 * rng.standard_normal(n)
    inits.append(np.clip(a, 0.01, None))

    # Flat-top window
    a = (0.2156 + 0.4166 * np.cos(2 * np.pi * t / n) + 0.2776 * np.cos(4 * np.pi * t / n)
         + 0.0835 * np.cos(6 * np.pi * t / n) + 0.0069 * np.cos(8 * np.pi * t / n))
    a = np.abs(a) + 0.05
    inits.append(np.clip(a, 0.01, None))

    # Gaussian
    center = n / 2.0
    sigma = n / 3.5
    a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
    a += 0.02 * rng.standard_normal(n)
    inits.append(np.clip(a, 0.01, None))

    # Raised cosine (wider)
    a = 0.3 + 0.7 * 0.5 * (1.0 + np.cos(np.pi * t / max(1, n - 1)))
    a += 0.02 * rng.standard_normal(n)
    inits.append(np.clip(a, 0.01, None))

    # Triangular
    w = 1.0 - np.abs(2.0 * t / max(1, n - 1) - 1.0)
    a = 0.15 + 0.85 * w
    a += 0.02 * rng.standard_normal(n)
    inits.append(np.clip(a, 0.01, None))

    # Random windowed
    a = (0.4 + 0.6 * rng.random(n)) * (0.3 + 0.7 * 0.5 * (1 + np.cos(2 * np.pi * t / max(1, n - 1))))
    inits.append(np.clip(a, 0.01, None))

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    1. Multi-start smooth optimization to find good basins
    2. Analytic coordinate descent for fine-tuning
    3. LP refinement
    4. Local search polish
    5. Final coordinate descent
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(seed)

    best_score = np.inf
    best_seq = None

    # Load provided sequence
    provided = None
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        provided = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.01, 1000.0)
        s = score_fast(provided)
        print(f"Provided: n={len(provided)}, score={s:.6f}")
        if s < best_score:
            best_score = s
            best_seq = provided.copy()

    # Phase 1: Multi-start smooth optimization
    print("\n=== Phase 1: Multi-start smooth optimization ===")
    p1_deadline = deadline - 900.0

    n_options = [len(provided)] if provided is not None else []
    for n in [192, 256, 320, 384, 512]:
        if n not in n_options:
            n_options.append(n)

    starts_done = 0
    max_starts = 12

    for n in n_options:
        if time.monotonic() > p1_deadline or starts_done >= max_starts:
            break
        inits = make_diverse_inits(n, rng)
        for ii, f0 in enumerate(inits):
            if time.monotonic() > p1_deadline or starts_done >= max_starts:
                break
            per_t = min(50.0, max(10.0, (p1_deadline - time.monotonic()) / max(1, max_starts - starts_done)))
            print(f"  n={n} init={ii} t={per_t:.0f}s")
            f_b, s_b = optimize_smoothed(f0, per_t, T_init=3.0, T_min=0.2, verbose=False)
            s_b = score_fast(f_b)
            if s_b < best_score:
                best_score = s_b
                best_seq = f_b.copy()
                print(f"  *** NEW BEST: {best_score:.6f} (n={n}) ***")
            starts_done += 1

    # Also smooth-optimize the provided sequence specifically
    if provided is not None and time.monotonic() < p1_deadline:
        f_b, s_b = optimize_smoothed(provided, min(60.0, p1_deadline - time.monotonic()),
                                     T_init=2.0, T_min=0.2, verbose=True)
        s_b = score_fast(f_b)
        if s_b < best_score:
            best_score = s_b
            best_seq = f_b.copy()
            print(f"  *** NEW BEST from provided: {best_score:.6f} ***")

    print(f"\nAfter Phase 1: best={best_score:.6f}")

    # Phase 2: Analytic coordinate descent (primary fine-tuner)
    if best_seq is not None and time.monotonic() < deadline - 500:
        print(f"\n=== Phase 2: Analytic coordinate descent ===")
        p2_time = min(500.0, deadline - time.monotonic() - 500)
        f_b, s_b = analytic_coord_descent(best_seq, p2_time, verbose=True)
        if s_b < best_score:
            best_score = s_b
            best_seq = f_b.copy()
            print(f"  *** NEW BEST: {best_score:.6f} ***")
        print(f"\nAfter Phase 2: best={best_score:.6f}")

    # Phase 3: LP refinement
    if best_seq is not None and time.monotonic() < deadline - 300:
        print(f"\n=== Phase 3: LP refinement ===")
        p3_time = min(300.0, deadline - time.monotonic() - 300)
        if p3_time > 20:
            f_b, s_b = lp_refine(best_seq, p3_time, verbose=True)
            if s_b < best_score:
                best_score = s_b
                best_seq = f_b.copy()
                print(f"  *** NEW BEST: {best_score:.6f} ***")
        print(f"\nAfter Phase 3: best={best_score:.6f}")

    # Phase 4: Local search
    if best_seq is not None and time.monotonic() < deadline - 150:
        print(f"\n=== Phase 4: Local search ===")
        p4_time = min(150.0, deadline - time.monotonic() - 150)
        if p4_time > 20:
            f_b, s_b = local_search(best_seq, p4_time, verbose=True)
            if s_b < best_score:
                best_score = s_b
                best_seq = f_b.copy()
                print(f"  *** NEW BEST: {best_score:.6f} ***")
        print(f"\nAfter Phase 4: best={best_score:.6f}")

    # Phase 5: Final coordinate descent polish
    if best_seq is not None and time.monotonic() < deadline - 30:
        print(f"\n=== Phase 5: Final coordinate descent ===")
        p5_time = min(120.0, deadline - time.monotonic() - 30)
        if p5_time > 15:
            f_b, s_b = analytic_coord_descent(best_seq, p5_time, verbose=True)
            if s_b < best_score:
                best_score = s_b
                best_seq = f_b.copy()
                print(f"  *** NEW BEST: {best_score:.6f} ***")
        print(f"\nAfter Phase 5: best={best_score:.6f}")

    # Finalize
    if best_seq is not None:
        final_list = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(final_list)
        print(f"\n{'=' * 50}")
        print(f"FINAL: score={final_score:.6f}, n={len(final_list)}")
        print(f"{'=' * 50}")
        return final_list

    return [1.0] * 256
