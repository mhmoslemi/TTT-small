def fft_convolve(a, b):
    n = len(a)
    m = len(b)

    while size < n + m - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n + m - 1]

def local_score(a):

    if S < 0.01:

"""
Coordinate descent with exact 1D subproblem solving for minimizing
peak autocorrelation of non-negative sequences. Combined with simulated
annealing for escape from local minima and multiple diverse starts.
"""

import numpy as np
from scipy.optimize import minimize_scalar
import time


def fft_convolve(a, b):
    """Compute linear convolution using FFT."""
    n = len(a)
    m = len(b)
    if n != m:
        b = np.zeros(n)
        b[:m] = b
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:2 * n - 1]


def compute_score(v):
    """Fast score computation."""
    n = len(v)
    v = np.maximum(v, 0.0)
    S = np.sum(v)
    if S < 0.01:
        return np.inf
    conv = fft_convolve(v, v)
    return 2.0 * n * np.max(conv) / (S * S)


def coordinate_descent(v_init, time_budget, rng, verbose=True):
    """
    Cyclic coordinate descent with exact 1D minimization.
    """
    n = len(v_init)
    v = np.maximum(v_init.astype(np.float64), 0.0)
    S = np.sum(v)
    if S < 1e-10:
        v = np.ones(n)
        S = n
    
    conv = fft_convolve(v, v)
    max_b = np.max(conv)
    current_score = 2.0 * n * max_b / (S * S)
    best_v = v.copy()
    best_score = current_score
    
    t0 = time.monotonic()
    sweep = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_budget:
        for j in range(n):
            x_old = v[j]
            S_prime = S - x_old
            if S_prime <= 1e-12:
                continue
            
            # Affected k range
            k_lo = j
            k_hi = min(2 * n - 2, j + n - 1)
            
            # Separate quadratic (k=2j) from linear terms
            has_quad = (2 * j >= k_lo and 2 * j <= k_hi)
            
            if has_quad:
                lin_indices = np.concatenate([np.arange(k_lo, 2 * j), np.arange(2 * j + 1, k_hi + 1)])
            else:
                lin_indices = np.arange(k_lo, k_hi + 1)
            
            # Precompute for fast evaluation
            if len(lin_indices) > 0:
                lin_slope = 2.0 * v[lin_indices - j]
                lin_base = conv[lin_indices].copy()  # current values (include x_old contribution)
            else:
                lin_slope = np.empty(0)
                lin_base = np.empty(0)
            
            # Unaffected max
            if k_lo == 0 and k_hi == 2 * n - 2:
                unaff_max = 0.0
            else:
                unaff_parts = []
                if k_lo > 0:
                    unaff_parts.append(np.max(conv[:k_lo]))
                if k_hi < 2 * n - 2:
                    unaff_parts.append(np.max(conv[k_hi + 1:]))
                unaff_max = max(unaff_parts) if unaff_parts else 0.0
            
            quad_val = (conv[2 * j] - x_old * x_old) if has_quad else -np.inf
            
            # 1D objective: minimize c(x) / (S_prime + x)^2
            # where c(x) = max(max_k(lin_base + lin_slope*(x-x_old)), quad_val + x^2, unaff_max)
            
            x_upper = max(2.0 * x_old + 0.01, 2.0 * S_prime / n + 0.01)
            
            # Use minimize_scalar
            def obj_1d(x):
                if len(lin_slope) > 0:
                    vals = lin_base + lin_slope * (x - x_old)
                    c = np.max(vals)
                else:
                    c = 0.0
                if has_quad:
                    q = quad_val + x * x
                    if q > c:
                        c = q
                if unaff_max > c:
                    c = unaff_max
                denom = S_prime + x
                return c / (denom * denom)
            
            res = minimize_scalar(obj_1d, bounds=(0.0, x_upper), method='bounded',
                                  options={'xatol': 1e-10, 'maxiter': 40})
            x_new = max(0.0, res.x) if res.success else x_old
            
            delta = x_new - x_old
            if abs(delta) > 1e-14:
                # Update conv
                if len(lin_indices) > 0:
                    conv[lin_indices] += 2.0 * delta * v[lin_indices - j]
                if has_quad:
                    conv[2 * j] += x_new * x_new - x_old * x_old
                v[j] = x_new
                S += delta
        
        sweep += 1
        max_b = np.max(conv)
        current_score = 2.0 * n * max_b / (S * S)
        
        if current_score < best_score - 1e-12:
            best_score = current_score
            best_v = v.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if verbose and (sweep % 50 == 0 or no_improve == 0):
            elapsed = time.monotonic() - t0
            print(f"      sweep={sweep}, score={current_score:.8f}, best={best_score:.8f}, no_improve={no_improve}, time={elapsed:.1f}s")

            # Perturb

            conv = fft_convolve(v, v)
            current_score = 2.0 * n * np.max(conv) / (S * S)
            no_improve = 0
    
    return best_v, best_score
