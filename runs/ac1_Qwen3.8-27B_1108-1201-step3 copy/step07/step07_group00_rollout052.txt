"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses Nesterov-accelerated subgradient with softmax peak weighting,
LP-based direction finding, peak-targeting perturbations, and coordinate descent.
Key innovations: adaptive temperature annealing, Barzilai-Borwein step sizes,
and a novel "peak-shifting" exploration to escape local minima.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient_fft(y, b, top_k, temp):
    """
    Compute weighted subgradient of max_k conv(y,y)[k] using softmax weights.
    sg[j] = 2 * sum_k w[k] * y[k-j] for valid k-j.
    This is 2 * cross_correlation(w, y) evaluated at appropriate indices.
    """
    n = len(y)
    size_b = len(b)  # 2n-1

    if top_k < size_b:
        idx = np.argpartition(b, -top_k)[-top_k:]
    else:
        idx = np.arange(size_b)

    b_sel = b[idx]
    b_max = b_sel.max()
    if temp < 1e-15:
        w = np.zeros(len(idx))
        w[b_sel == b_max] = 1.0 / (b_sel == b_max).sum()
    else:
        w = np.exp((b_sel - b_max) / temp)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(idx))
            w[b_sel == b_max] = 1.0 / (b_sel == b_max).sum()
        else:
            w /= w_sum

    # Build full weight vector
    w_full = np.zeros(size_b)
    w_full[idx] = w

    # sg[j] = 2 * sum_k w_full[k] * y[k-j] for 0 <= k-j < n
    # This is cross-correlation: sg = 2 * (w_full correlated with y)
    # cross_corr[m] = sum_k w_full[k] * y[k-m] for m in range
    # We want sg[j] = cross_corr[j] for j in [0, n-1]
    # cross_corr can be computed via FFT:
    # cross_corr[j] = sum_k w_full[k] * y[k-j] = sum_k y[k-j] * w_full[k]
    # Let m = k-j, then k = m+j: sum_m y[m] * w_full[m+j]
    # This is the cross-correlation of y and w_full, which is:
    # cross_corr = ifft(conj(fft(y)) * fft(w_full)) ... 
    # Actually, let's use a direct FFT approach:
    # sg[j] = sum_k w_full[k] * y[k-j] where 0<=k-j<n means 0<=m<n where m=k-j
    # = sum_{m=0}^{n-1} y[m] * w_full[m+j]
    # This is the cross-correlation (y ⋆ w_full)[j] = sum_m y[m]*w_full[m+j]
    # = ifft(fft(y) * conj(fft(w_full))) ... no
    # Cross-correlation R[j] = sum_m y[m] * w_full[m+j]
    # In frequency domain: R = ifft(conj(FFT(y)) * FFT(w_full)) ... 
    # Let me just use the direct formula with appropriate padding.
    
    # R[j] = sum_{m=0}^{n-1} y[m] * w_full[m+j]
    # This is (y * w_full_rev) where * is cross-correlation
    # Equivalently: R = ifft(FFT(y_padded) * conj(FFT(w_full_padded)))
    # where we zero-pad to size >= n + size_b - 1
    
    fft_size = 1 << (n + size_b - 1 - 1).bit_length()
    Y = rfft(y, fft_size)
    W = rfft(w_full, fft_size)
    # Cross-correlation: R[j] = sum_m y[m]*w_full[m+j]
    # = ifft(conj(FFT(y)) * FFT(w_full))[j] ... let me verify
    # ifft(FFT(y) * FFT(w))[k] = sum_m y[m]*w[k-m] (circular conv)
    # We want R[j] = sum_m y[m]*w_full[m+j] = sum_m y[m]*w_full[-(j-m)]
    # = (y * w_rev)[j] where w_rev[t] = w_full[-t]
    # In FFT: ifft(FFT(y) * FFT(w_rev))
    # w_rev in frequency: conj(FFT(w_full)) (for real signals)
    # So R = ifft(FFT(y) * conj(FFT(w_full)))
    
    R = irfft(Y * np.conj(W), fft_size)[:n]
    sg = 2.0 * R
    return sg


def nesterov_sg_phase(a_init, time_limit, top_k=200, temp_init=0.0005,
                      temp_max=0.5, gamma=0.01, label="nsg"):
    """
    Nesterov-accelerated subgradient with softmax peak weighting and
    adaptive temperature. Returns (best_a, best_obj).
    """
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a /= s

    y = a.copy()  # Nesterov lookahead
    a_prev = a.copy()
    t_nest = 1.0
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    temp = temp_init
    no_improve = 0
    it = 0
    rng = np.random.default_rng(999)

    # BB step size variables
    a_bb = a.copy()
    sg_bb = None

    while time.monotonic() - t0 < time_limit:
        it += 1

        b = fast_convolve(y)
        sg = compute_subgradient_fft(y, b, top_k, temp)

        # Adaptive step size: BB-like
        if sg_bb is not None:
            yd = y - a_bb
            sgd = sg - sg_bb
            ss = np.dot(sgd, sgd)
            if ss > 1e-20:
                step = np.dot(yd, sgd) / ss
                step = np.clip(step, 0.5 / t_nest, 10.0 / t_nest)
            else:
                step = 1.0 / t_nest
        else:
            step = 1.0 / t_nest

        a_bb = a.copy()
        sg_bb = sg.copy()

        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_new
        y_new = a + theta * (a - a_prev)
        # Clip to valid range
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y > 1e-15:
            y_new /= s_y

        # Projected gradient step
        a_new = project_simplex(y_new - step * sg / n)

        # Line search (backtracking)
        obj_new = fast_convolve(a_new).max()
        if obj_new > best_obj * 1.5:
            # Step too large, reduce
            for retry in range(5):
                step *= 0.5
                a_new = project_simplex(y_new - step * sg / n)
                obj_new = fast_convolve(a_new).max()
                if obj_new <= best_obj * 1.1:
                    break

        a_prev = a.copy()
        a = a_new
        t_nest = t_new

        # Anneal temperature
        if it % 200 == 0:
            temp = min(temp * 1.3, temp_max)

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                # Restart with perturbation
                perturb = rng.standard_normal(n) * (0.002 + 0.001 * np.sqrt(no_improve / 30.0))
                a = project_simplex(best_a + perturb)
                y = a.copy()
                a_prev = a.copy()
                t_nest = 1.0
                temp = max(temp * 0.5, temp_init)
                no_improve = 0
                sg_bb = None

        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_obj
            print(f"  {label} it={it} obj={best_obj:.10f} score={score:.6f} "
                  f"temp={temp:.6f} t={elapsed:.0f}s")

    return best_a, best_obj


def lp_direction(f, top_k, eps=1e-9):
    """
    LP: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for top_k peaks.
    Returns (g, M) or (None, M).
    """
    n = len(f)
    b = fast_convolve(f)
    M = b.max()
    if M < 1e-15:
        return None, M

    if top_k < len(b):
        top_idx = np.argpartition(b, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(b))
    top_idx = np.sort(top_idx)
    K = len(top_idx)

    A = np.zeros((K, n))
    for row_i in range(K):
        k = top_idx[row_i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=200000, eps=eps)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_iters=100000)
        except Exception:
            return None, M

    if g.value is None:
        return None, M
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None, M
    S = g_val.sum()
    if S < 1e-10:
        return None, M
    g_val /= S
    return g_val, M


def lp_phase(f_init, time_limit, top_k_init=50, top_k_max=300, label="lp"):
    """LP-based optimization with growing top_k."""
    t0 = time.monotonic()
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    top_k = top_k_init
    stuck = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 30 == 0 and top_k < top_k_max:
            top_k = min(top_k + 15, top_k_max)

        g, M = lp_direction(f, top_k)
        if g is None:
            g, M = lp_direction(f, min(top_k * 2, 2 * len(f) - 1))
            if g is None:
                stuck += 5
                if stuck > 20:
                    break
                continue

        S_g = g.sum()
        if S_g <= 1.0 + 1e-4:
            stuck += 2
            if stuck > 20:
                break
            top_k = min(top_k + 20, top_k_max)
            continue

        cur_obj = fast_convolve(f).max()
        best_t = 0.05
        best_t_obj = cur_obj
        for t in [0.02, 0.04, 0.07, 0.1, 0.15, 0.2, 0.3, 0.4, 0.6]:
            f_trial = (1.0 - t) * f + t * g
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t

        f = (1.0 - best_t) * f + best_t * g
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s

        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 40:
                break

        if it % 30 == 0:
            elapsed = time.monotonic() - t0
            n = len(f)
            print(f"    {label} it={it} obj={best_obj:.10f} "
                  f"score={2.0*n*best_obj:.6f} tk={top_k} t={elapsed:.0f}s")

    return best_f, best_obj


def peak_shift_explore(a_init, time_limit, n_trials=50, label="peak_shift"):
    """
    Novel exploration: identify worst peak, compute sensitivity,
    and apply targeted redistribution to reduce it.
    """
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    rng = np.random.default_rng(777)
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(a)
        k_star = b.argmax()
        peak_val = b[k_star]

        # Sensitivity: d(a*a)[k_star]/da[j] = 2*a[k_star-j] (valid range)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        sens = np.zeros(n)
        seg = a[k_star - hi:k_star - lo + 1][::-1]
        sens[lo:hi + 1] = 2.0 * seg

        # Targeted move: reduce high-sensitivity, increase low-sensitivity
        # Use softmax to get weights
        sens_norm = sens / (sens.max() + 1e-15)

        # Create perturbation: negative where sensitivity is high, positive where low
        # But we need to maintain sum(a) = 1
        noise_scale = 0.003 + 0.002 * np.sqrt(it / max(1, n_trials))
        perturb = -sens_norm * noise_scale
        # Add random component to escape
        perturb += rng.standard_normal(n) * noise_scale * 0.3
        # Zero out mean to preserve sum
        perturb -= perturb.mean()

        a_new = a + perturb
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a.copy()
        else:
            a_new /= s

        obj_new = fast_convolve(a_new).max()
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            # Try smaller step
            for frac in [0.5, 0.25, 0.1]:
                a_trial = a + frac * perturb
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-10:
                    continue
                a_trial /= s_t
                obj_t = fast_convolve(a_trial).max()
                if obj_t < best_obj - 1e-12:
                    best_obj = obj_t
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    break

        if it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} it={it} obj={best_obj:.10f} "
                  f"score={2.0*n*best_obj:.6f} t={elapsed:.0f}s")

    return best_a, best_obj


def coordinate_polish(f_init, time_limit):
    """Coordinate descent on individual elements with adaptive multipliers."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    multipliers = np.array([0.5, 0.7, 0.85, 1.0/0.85, 1.0/0.7, 1.4, 1.8, 2.5])
    no_improve_rounds = 0

    while no_improve_rounds < 3 and time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-14:
                continue
            best_j_val = f[j]
            best_j_obj = best_obj
            for mult in multipliers:
                new_val = f[j] * mult
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f /= s
                best_obj = best_j_obj
                improved = True

        if improved:
            no_improve_rounds = 0
        else:
            no_improve_rounds += 1

    return f, best_obj

            print(f"  NEW BEST [nsg1]: {score1:.6f}")

    # === Phase 2: LP refinement ===
    print(f"\n=== Phase 2: LP (n={len(best_seq)}, score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.25, 400.0)
    if t2 > 30:
        n_b = len(best_seq)
        print(f"  Budget: {t2:.0f}s")
        f2, obj2 = lp_phase(best_seq, time_limit=t2, top_k_init=60,
                            top_k_max=250, label="lp_main")
        score2 = float(evaluate_sequence(f2.tolist()))
        if np.isfinite(score2) and score2 < best_score:
            best_score = score2
            best_seq = f2.copy()
            print(f"  NEW BEST [lp]: {score2:.6f}")

    # === Phase 3: Peak-shift exploration ===
    print(f"\n=== Phase 3: Peak-shift (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.15, 200.0)
    if t3 > 20:
        print(f"  Budget: {t3:.0f}s")
        f3, obj3 = peak_shift_explore(best_seq, time_limit=t3, label="peak")
        score3 = float(evaluate_sequence(f3.tolist()))
        if np.isfinite(score3) and score3 < best_score:
            best_score = score3
            best_seq = f3.copy()
            print(f"  NEW BEST [peak]: {score3:.6f}")

    # === Phase 4: Nesterov SG round 2 (from best) ===
    print(f"\n=== Phase 4: Nesterov SG 2 (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.2, 250.0)
    if t4 > 20:
        print(f"  Budget: {t4:.0f}s")
        f4, obj4 = nesterov_sg_phase(best_seq, time_limit=t4, top_k=400,
                                     temp_init=0.0002, temp_max=0.2,
                                     label="nsg2")
        score4 = float(evaluate_sequence(f4.tolist()))
        if np.isfinite(score4) and score4 < best_score:
            best_score = score4
            best_seq = f4.copy()
            print(f"  NEW BEST [nsg2]: {score4:.6f}")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.15, 200.0)
    if t5 > 30:
        print(f"  Budget: {t5:.0f}s")
        f5, score5 = multi_length_explore(best_seq, time_limit=t5)
        if score5 < best_score:
            best_score = score5
            best_seq = f5.copy()
            print(f"  NEW BEST [multi]: {best_score:.6f}")

    # === Phase 6: Coordinate descent ===
    print(f"\n=== Phase 6: Coord descent (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.12, 100.0)
    if t6 > 10:
        print(f"  Budget: {t6:.0f}s")
        f6, obj6 = coordinate_polish(best_seq, time_limit=t6)
        score6 = float(evaluate_sequence(f6.tolist()))
        if np.isfinite(score6) and score6 < best_score:
            best_score = score6
            best_seq = f6.copy()
            print(f"  NEW BEST [cd]: {score6:.6f}")

    # === Phase 7: Final LP burst ===
    print(f"\n=== Phase 7: Final LP (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t7 = min(rem * 0.4, 120.0)
    if t7 > 15:
        n_b = len(best_seq)
        print(f"  Budget: {t7:.0f}s, n={n_b}")
        f7, obj7 = lp_phase(best_seq, time_limit=t7, top_k_init=100,
                            top_k_max=400, label="lp_final")
        score7 = float(evaluate_sequence(f7.tolist()))
        if np.isfinite(score7) and score7 < best_score:
            best_score = score7
            best_seq = f7.copy()
            print(f"  NEW BEST [lp_final]: {score7:.6f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
