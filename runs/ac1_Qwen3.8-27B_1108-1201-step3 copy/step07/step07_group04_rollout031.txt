from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from time import monotonic

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def smooth_max_grad(b, T):
    """Compute soft-max of b and its gradient weights."""
    b_shifted = (b - b.max()) / T
    exp_b = np.exp(b_shifted)
    w = exp_b / exp_b.sum()
    lse = b.max() + T * np.log(exp_b.sum())
    return lse, w

    """Compute gradient of 2n * soft_max(conv(a,a), T) wrt z, where a = softmax(z)."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)

    b = fast_convolve(a)
    lse, w = smooth_max_grad(b, T)

        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        if t % 1000 == 0:
            print(f"    Adam t={t} T={T:.4f} score={score:.6f} best={best_score:.6f}")
    
    return best_a, best_score

def cross_correlation_fft(w, a):
    """Compute C[j] = sum_k w[k] * a[k-j] using FFT.
    w has length 2n-1, a has length n, result has length n."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    # Pad w to fft_size
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    # Cross-correlation: C[j] = sum_k w[k] * conj(a[k-j]) = ifft(fft(w) * conj(fft(a)))[:n]
    # Wait, I need to be careful. Cross-correlation R_xy[j] = sum_k x[k] * y[k-j]
    # In FFT: R_xy = ifft(fft(x) * conj(fft(y)))
    # So C[j] = sum_k w[k] * a[k-j] = ifft(fft(w) * conj(fft(a)))[:n]
    C = irfft(W * np.conj(A), fft_size)
    return C[:n]


def compute_obj_and_grad(z, T, n):
    """Compute objective value and gradient in z-space.
    Returns (score, grad_z) where score = 2n * soft_max(conv(a,a), T)
    and grad_z = ∂score/∂z."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    
    # Soft max
    b_max = b.max()
    b_shifted = (b - b_max) / T
    exp_b = np.exp(b_shifted)
    w = exp_b / exp_b.sum()
    lse = b_max + T * np.log(exp_b.sum())
    
    score = 2.0 * n * lse
    
    # Gradient: grad_z = 4n * a * (C - dot(a, C))
    # where C[j] = sum_k w[k] * a[k-j]
    C = cross_correlation_fft(w, a)
    S_C = np.dot(a, C)
    grad_z = 4.0 * n * a * (C - S_C)
    
    return score, grad_z


def compute_obj_and_grad_a(a, T):
    """Compute objective and gradient in a-space (for projected gradient).
    Returns (score, grad_a) where grad_a = ∂(2n*soft_max(conv(a,a),T))/∂a."""
    n = len(a)
    b = fast_convolve(a)
    b_max = b.max()
    b_shifted = (b - b_max) / T
    exp_b = np.exp(b_shifted)
    w = exp_b / exp_b.sum()
    lse = b_max + T * np.log(exp_b.sum())
    score = 2.0 * n * lse
    
    # grad_a[j] = 4n * C[j] where C[j] = sum_k w[k] * a[k-j]
    C = cross_correlation_fft(w, a)
    grad_a = 4.0 * n * C
    
    return score, grad_a


def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_idx = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    return np.maximum(v - theta, 0.0)


def adam_logspace(a0, time_limit, T_init=None, T_min=None, lr_init=0.01, verbose=True):
    """Adam optimization in log-space with temperature annealing."""
    n = len(a0)
    a = np.maximum(a0, 1e-30)
    a = a / a.sum()
    z = np.log(a)
    z = z - z.mean()  # center for numerical stability
    
    if T_init is None:
        b = fast_convolve(a)
        T_init = b.max()
    if T_min is None:
        T_min = 0.001 * T_init
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0
    t0 = monotonic()
    best_score = float('inf')
    best_a = a.copy()
    lr = lr_init
    
    while monotonic() - t0 < time_limit:
        t += 1
        # Temperature schedule: exponential decay
        progress = min(1.0, (monotonic() - t0) / time_limit)
        T = T_init * (T_min / T_init) ** progress
        
        score, grad_z = compute_obj_and_grad(z, T, n)
        
        if score < best_score - 1e-15:
            best_score = score
            best_a = softmax_from_z(z).copy()
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * grad_z ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        if verbose and t % 5000 == 0:
            elapsed = monotonic() - t0
            print(f"    Adam t={t} ({elapsed:.0f}s) T={T:.6f} score={score:.6f} best={best_score:.6f}")
    
    return best_a, best_score


def lbgfs_logspace(a0, time_limit, T=1.0, verbose=True):
    """L-BFGS optimization in log-space."""
    n = len(a0)
    a = np.maximum(a0, 1e-30)
    a = a / a.sum()
    z0 = np.log(a)
    z0 = z0 - z0.mean()
    
    t0 = monotonic()
    
    def func(z):
        if monotonic() - t0 > time_limit:
            return 1e15
        score, _ = compute_obj_and_grad(z, T, n)
        return score
    
    def jac(z):
        if monotonic() - t0 > time_limit:
            return np.zeros(n)
        _, grad = compute_obj_and_grad(z, T, n)
        return grad
    
    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': 10000, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': 50000, 'disp': False}
        )
        z_final = res.x
        a_out = softmax_from_z(z_final)
    except Exception as e:
        print(f"    L-BFGS error: {e}")
        a_out = a0
    
    return a_out


def projected_gradient(a0, time_limit, T=0.01, lr_init=0.001, verbose=True):
    """Projected gradient descent in a-space."""
    n = len(a0)
    a = np.maximum(a0, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    lr = lr_init
    t = 0
    
    while monotonic() - t0 < time_limit:
        t += 1
        score, grad_a = compute_obj_and_grad_a(a, T)
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
        
        # Project gradient to tangent space (sum = 0)
        grad_proj = grad_a - np.mean(grad_a)
        
        a_new = a - lr * grad_proj
        a_new = project_simplex(a_new)
        a = a_new
        
        lr = lr_init / (1 + 0.01 * t)
        
        if verbose and t % 5000 == 0:
            elapsed = monotonic() - t0
            print(f"    ProjGrad t={t} ({elapsed:.0f}s) score={score:.6f} best={best_score:.6f}")
    
    return best_a, best_score


def peak_shave(a0, time_limit, verbose=True):
    """Local peak shaving: reduce elements contributing to the dominant peak."""
    n = len(a0)
    a = np.maximum(a0, 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    no_improve = 0
    
    while monotonic() - t0 < time_limit and no_improve < 200:
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]
        
        # Find top contributors to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        if contribs.max() < 1e-15:
            break
        
        top_m = min(10, len(idx))
        top_idx = idx[np.argsort(contribs)[::-1][:top_m]]
        
        improved = False
        for i in top_idx:
            if a[i] < 1e-14:
                continue
            # Try reducing a[i] and redistributing
            for frac in [0.8, 0.7, 0.5, 0.3]:
                old_val = a[i]
                new_val = old_val * frac
                diff = old_val - new_val
                if diff < 1e-15:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                a_trial[mask] = others + diff * others / others_sum
                a_trial = a_trial / a_trial.sum()
                new_score = 2.0 * n * fast_convolve(a_trial).max()
                if new_score < best_score - 1e-12:
                    a = a_trial
                    best_score = new_score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            no_improve += 1
            # Try random perturbation
            a_trial = a + np.random.standard_normal(n) * 0.001
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            new_score = 2.0 * n * fast_convolve(a_trial).max()
            if new_score < best_score - 1e-12:
                a = a_trial
                best_score = new_score
                best_a = a.copy()
                no_improve = 0
        
        if verbose and t % 1000 == 0:
            elapsed = monotonic() - t0
            print(f"    PeakShave ({elapsed:.0f}s) best={best_score:.6f}")
    
    return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using gradient-based optimization in log-space
with temperature annealing. Combines Adam, L-BFGS, projected gradient, and peak-shaving
local search. Key innovation: efficient FFT-based gradient computation for the softmax
parameterization enables ~10000x more iterations than LP-based approaches.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from time import monotonic


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax_from_z(z):
    """Compute stable softmax of z."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def cross_correlation_fft(w, a):
    """Compute C[j] = sum_k w[k+j] * a[k] for j=0,...,n-1 using FFT."""
    n = len(a)
    needed = 3 * n - 2
    fft_size = 1 << (needed - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    R = irfft(W * np.conj(A), fft_size)
    return R[:n]


def compute_obj_grad_z(z, T, n):
    """Compute smooth objective and gradient in z-space.
    score = 2n * soft_max(conv(a,a), T)
    grad_z = 4n * a * (C - dot(a,C))
    """
    a = softmax_from_z(z)
    b = fast_convolve(a)
    b_max = b.max()
    b_shifted = (b - b_max) / T
    exp_b = np.exp(b_shifted)
    w = exp_b / exp_b.sum()
    lse = b_max + T * np.log(exp_b.sum())
    score = 2.0 * n * lse
    C = cross_correlation_fft(w, a)
    S_C = np.dot(a, C)
    grad_z = 4.0 * n * a * (C - S_C)
    return score, grad_z


def compute_obj_grad_a(a, T):
    """Compute smooth objective and gradient in a-space."""
    n = len(a)
    b = fast_convolve(a)
    b_max = b.max()
    b_shifted = (b - b_max) / T
    exp_b = np.exp(b_shifted)
    w = exp_b / exp_b.sum()
    lse = b_max + T * np.log(exp_b.sum())
    score = 2.0 * n * lse
    C = cross_correlation_fft(w, a)
    grad_a = 4.0 * n * C
    return score, grad_a


def project_simplex(v):
    """Project v onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def adam_logspace(a0, time_limit, T_init=None, T_min=None, lr=0.02, restart_threshold=8000, verbose=True):
    """Adam optimizer in log-space with temperature annealing and restarts."""
    n = len(a0)
    a = np.maximum(a0, 1e-30)
    a = a / a.sum()
    z = np.log(a)
    z = z - z.mean()

    if T_init is None:
        b = fast_convolve(a)
        T_init = max(b.max(), 1e-10)
    if T_min is None:
        T_min = 0.002 * T_init

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0
    t0 = monotonic()
    best_score = float('inf')
    best_a = a.copy()
    best_z = z.copy()
    last_improve_t = 0
    n_restarts = 0

    while monotonic() - t0 < time_limit:
        # Temperature: exponential annealing
        progress = min(1.0, (monotonic() - t0) / time_limit)
        T = T_init * (T_min / T_init) ** progress
        T = max(T, T_min)

        score, grad_z = compute_obj_grad_z(z, T, n)

        if score < best_score - 1e-14:
            best_score = score
            a_tmp = softmax_from_z(z)
            best_a = a_tmp.copy()
            best_z = z.copy()
            last_improve_t = t

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * grad_z ** 2
        m_hat = m / (1 - beta1 ** (t + 1))
        v_hat = v / (1 - beta2 ** (t + 1))
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Restart if stuck
        if t - last_improve_t > restart_threshold and n_restarts < 5:
            n_restarts += 1
            z = best_z + np.random.standard_normal(n) * 0.05
            m = np.zeros(n)
            v = np.zeros(n)
            lr = max(0.005, lr * 1.5)
            last_improve_t = t
            if verbose:
                print(f"      Restart {n_restarts} at t={t}, best={best_score:.6f}")

        t += 1
        if verbose and t % 10000 == 0:
            elapsed = monotonic() - t0
            print(f"      Adam t={t} ({elapsed:.0f}s) T={T:.6f} score={score:.6f} best={best_score:.6f}")

    return best_a, best_score


def lbgfs_logspace(a0, time_limit, T, maxiter=20000, verbose=True):
    """L-BFGS-B optimization in log-space at fixed temperature."""
    n = len(a0)
    a = np.maximum(a0, 1e-30)
    a = a / a.sum()
    z0 = np.log(a)
    z0 = z0 - z0.mean()
    t0 = monotonic()
    call_count = [0]

    def func(z):
        call_count[0] += 1
        if monotonic() - t0 > time_limit:
            return 1e15
        score, _ = compute_obj_grad_z(z, T, n)
        return score

    def jac(z):
        call_count[0] += 1
        if monotonic() - t0 > time_limit:
            return np.zeros(n)
        _, grad = compute_obj_grad_z(z, T, n)
        return grad

    try:
        res = minimize(fun=func, x0=z0, jac=jac, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                                'maxfun': maxiter * 10, 'disp': False})
        z_final = res.x
        a_out = softmax_from_z(z_final)
        if verbose:
            print(f"      L-BFGS: {call_count[0]} evals, final={res.fun:.6f}")
    except Exception as e:
        if verbose:
            print(f"      L-BFGS error: {e}")
        a_out = a0
    return a_out


def projected_gradient_opt(a0, time_limit, T=0.05, lr_init=0.002, verbose=True):
    """Projected gradient descent in a-space with Nesterov acceleration."""
    n = len(a0)
    a = np.maximum(a0, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    a_mom = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    lr = lr_init
    beta = 0.9
    t = 0

    while monotonic() - t0 < time_limit:
        t += 1
        score, grad_a = compute_obj_grad_a(a_mom, T)

        # Nesterov momentum
        gamma = (1.0 + np.sqrt(1 + 4 * t * t * lr * lr)) / 2
        a_new = a + (t / (t + 1)) * (a - a_mom)
        a_mom = a_new

        # Gradient step
        grad_proj = grad_a - np.mean(grad_a)
        a = a_mom - lr / gamma * grad_proj
        a = project_simplex(a)

        actual_score = 2.0 * n * fast_convolve(a).max()
        if actual_score < best_score - 1e-13:
            best_score = actual_score
            best_a = a.copy()

        if verbose and t % 5000 == 0:
            elapsed = monotonic() - t0
            print(f"      ProjGrad t={t} ({elapsed:.0f}s) best={best_score:.6f}")

    return best_a, best_score


def peak_shave_opt(a0, time_limit, verbose=True):

    n = len(a0)
    a = np.maximum(a0, 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    no_improve = 0
    step = 0

    while monotonic() - t0 < time_limit and no_improve < 300:
        step += 1
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

        if contribs.max() < 1e-18:
            break

        top_m = min(15, len(idx))
        top_positions = idx[np.argsort(contribs)[::-1][:top_m]]

        improved = False

        # Strategy 1: Reduce single element, redistribute proportionally
        for i in top_positions:
            if a[i] < 1e-14:
                continue
            for frac in [0.85, 0.7, 0.5]:
                old_val = a[i]
                new_val = old_val * frac
                diff = old_val - new_val
                if diff < 1e-15:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                a_trial[mask] = others + diff * others / others_sum
                new_score = 2.0 * n * fast_convolve(a_trial).max()
                if new_score < best_score - 1e-12:
                    a = a_trial
                    best_score = new_score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        # Strategy 2: Reduce a pair of elements
        if not improved:
            for ci in range(min(5, len(top_positions))):
                i = top_positions[ci]
                j = peak_k - i
                if i == j or j < 0 or j >= n or a[j] < 1e-14:
                    continue
                for f1 in [0.8, 0.7]:
                    for f2 in [0.8, 0.7]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        s2 = a_trial.sum()
                        if s2 < 1e-15:
                            continue
                        a_trial = a_trial / s2
                        new_score = 2.0 * n * fast_convolve(a_trial).max()
                        if new_score < best_score - 1e-12:
                            a = a_trial
                            best_score = new_score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        # Strategy 3: Small random perturbation (focusing on peak region)
        if not improved:
            for _ in range(3):
                perturb = np.zeros(n)
                # Focus perturbation near the peak-contributing elements

                a_trial = a + perturb
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                new_score = 2.0 * n * fast_convolve(a_trial).max()
                if new_score < best_score - 1e-12:
                    a = a_trial
                    best_score = new_score
                    best_a = a.copy()
                    improved = True
                    break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                # Larger random restart
                a = best_a + np.random.standard_normal(n) * 0.005
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0

        if verbose and step % 2000 == 0:
            elapsed = monotonic() - t0
            print(f"      PeakShave step={step} ({elapsed:.0f}s) best={best_score:.6f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score, verbose=True):
    """Evaluate candidate and update best if improved."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s

        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def cosine_init(n):
    """Initialize with raised cosine shape."""
    t = np.linspace(0, np.pi, n)
    a = 0.5 * (1 + np.cos(t))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def gaussian_init(n, sigma_frac=0.3):
    """Initialize with Gaussian shape."""
    t = np.arange(n)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
    a = a / a.sum()
    return a

            a_ps = peak_shave_opt(best_seq, t_ps, verbose=False)
            best_seq, best_score = check_and_update(a_ps, "final_ps", best_seq, best_score)

    # === Phase 8: One more Adam round if time allows ===
    rem = deadline - monotonic() - 3
    if rem > 20 and best_seq is not None:
        n_best = len(best_seq)
        b_best = fast_convolve(best_seq)
        T_init = 0.05 * b_best.max()
        T_min = 0.002 * b_best.max()
        print(f"\n=== Phase 8: Final Adam (best={best_score:.6f}) ===")
        a_fa, _ = adam_logspace(best_seq, rem * 0.8, T_init=T_init, T_min=T_min,
                                 lr=0.005, restart_threshold=15000, verbose=False)
        best_seq, best_score = check_and_update(a_fa, "final_adam", best_seq, best_score)

    # Final verification
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512.0
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
