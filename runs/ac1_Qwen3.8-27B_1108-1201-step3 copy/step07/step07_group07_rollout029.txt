import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
from scipy.sparse import diags as sparse_diags

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix_sparse(a):
    """Build sparse matrix A such that A @ g = conv(a, g). A has shape (2n-1, n)."""
    n = len(a)
    m = 2 * n - 1
    diags_vals = []
    diags_offsets = []
    for d in range(n):
        length = min(n, 2 * n - 1 - d)
        if length > 0:
            diags_vals.append(np.full(length, a[d]))
            diags_offsets.append(d)
    A = sparse_diags(diags_vals, diags_offsets, shape=(m, n), format='csr')
    return A


def lp_solve_direction(a, M, top_k=None, time_limit=10.0):
    """
    Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0.
    Returns g (unnormalized) or None if failed.
    """
    t0 = time.monotonic()
    n = len(a)
    m = 2 * n - 1

    # Build sparse constraint matrix
    A_mat = build_conv_matrix_sparse(a)
    b_vec = np.full(m, M)
    c_vec = -np.ones(n)

    try:
        res = linprog(c_vec, A_ub=A_mat, b_ub=b_vec,
                      bounds=[(0.0, None)] * n, method='highs',
                      options={'time_limit': time_limit, 'dual_refine': True})
        if not res.success:
            return None
        return res.x
    except Exception:
        return None


def lp_solve_restricted(a, M, active_indices):
    """
    Solve LP with only active constraints (faster for selective constraint sets).
    """
    n = len(a)
    A_full = build_conv_matrix_sparse(a)
    A_sub = A_full[active_indices, :]
    b_sub = np.full(len(active_indices), M)
    c_vec = -np.ones(n)

    try:
        res = linprog(c_vec, A_ub=A_sub, b_ub=b_sub,
                      bounds=[(0.0, None)] * n, method='highs',
                      options={'time_limit': 5.0})
        if not res.success:
            return None
        return res.x
    except Exception:
        return None


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def eval_max_b(a):
    """Return max of auto-convolution for normalized sequence (sum=1)."""
    b = fast_convolve(a)
    return float(b.max())


def line_search_blend(a, g, best_obj, max_evals=20):
    """
    Find optimal blend t in (0,1) to minimize objective.
    Returns (best_t, best_obj).
    """
    g_norm = g / g.sum()
    best_t = 0.0
    best_bl = best_obj

    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)

    def obj_at_t(t):
        blend = (1.0 - t) * a + t * g_norm
        blend = np.maximum(blend, 0.0)
        s = blend.sum()
        if s < 1e-12:
            return best_obj
        blend = blend / s
        return eval_max_b(blend)

    fc = obj_at_t(c)
    fd = obj_at_t(d)
    evals = 2

    for _ in range(max_evals - 2):
        if fc < fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = obj_at_t(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_at_t(d)
        evals += 1

    best_t = (lo + hi) / 2.0
    best_bl = min(fc, fd)

    # Also check t=0 and t=1
    if obj_at_t(1.0) < best_bl:
        best_t = 1.0
        best_bl = obj_at_t(1.0)

    return best_t, best_bl


def subgradient_step(a, step_count, lr, momentum, n_top):
    """
    One subgradient step with FFT. Returns new a and velocity.
    """
    n = len(a)
    b = fast_convolve(a)
    b_max = b.max()

    # Select active peaks
    if n_top < len(b):
        top_idx = np.argpartition(b, -n_top)[-n_top:]
    else:
        top_idx = np.arange(len(b))

    threshold = b_max * 0.995
    active_mask = b[top_idx] >= threshold
    if not active_mask.any():
        active_mask[0] = True

    # Subgradient: g[i] = sum_{k active} a[k-i] (for valid k-i)
    w = np.zeros(len(b))
    active_indices = top_idx[active_mask]
    w[active_indices] = 1.0 / len(active_indices)

    # Compute gradient via FFT: grad[i] = sum_k w[k] * (sum_{j: i+j=k} a[j])
    # This is the cross-correlation of a with w
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    grad = irfft(np.conj(A) * W, fft_size)[:n]

    return grad, b


def subgradient_optimize(a0, time_limit, lr_init=0.003, momentum=0.9,
                         n_top=None, seed=42, restart_every=2000):
    """FFT-accelerated subgradient with Nesterov momentum."""
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    v = np.zeros(n)
    best_a = a.copy()
    best_obj = eval_max_b(a)
    cur_lr = lr_init
    no_improve = 0
    kick_size = 0.005

    if n_top is None:
        n_top = max(10, n // 10)

    while time.monotonic() - t0 < time_limit:
        grad, b = subgradient_step(a, 0, cur_lr, momentum, n_top)
        v = momentum * v - cur_lr * grad
        a_new = project_simplex(a + v)

        obj_new = eval_max_b(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        if no_improve >= restart_every:
            a = project_simplex(best_a + rng.standard_normal(n) * kick_size)
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.5, lr_init * 0.03)
            no_improve = 0
            kick_size *= 0.7
        elif no_improve >= restart_every // 3:
            cur_lr *= 0.95

    return best_a, best_obj


def iterative_lp_optimize(a0, time_limit, max_iters=500, verbose=False,
                          annealing=True, T_init=0.001, T_decay=0.95):
    """
    Main iterative LP optimization with simulated annealing.
    Alternates between full LP and restricted LP.
    """
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = eval_max_b(a)
    T = T_init
    iterations = 0
    lp_time_budget = 3.0  # seconds per LP

    while time.monotonic() - t0 < time_limit and iterations < max_iters:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2:
            break

        b = fast_convolve(a)
        M = b.max()
        iterations += 1

        # Adaptive active set size
        b_sorted = np.sort(b)[::-1]
        # Count peaks within 50% of max
        half_max_count = np.sum(b >= M * 0.5)
        # Use a range of active set sizes
        active_frac = max(0.05, min(0.5, half_max_count / len(b)))
        n_active = max(5, int(len(b) * active_frac))

        # Solve LP
        g = lp_solve_direction(a, M, time_limit=min(lp_time_budget, rem * 0.3))
        if g is None or g.sum() < 1e-10:
            # Try restricted LP
            top_idx = np.argsort(b)[::-1][:n_active]
            g = lp_solve_restricted(a, M, top_idx)
            if g is None or g.sum() < 1e-10:
                T *= T_decay
                continue

        g_norm = g / g.sum()

        # Line search
        best_t, blend_obj = line_search_blend(a, g_norm, best_obj, max_evals=15)

        if best_t <= 0:
            T *= T_decay
            continue

        a_new = (1.0 - best_t) * a + best_t * g_norm
        a_new = np.maximum(a_new, 0.0)
        ss = a_new.sum()
        if ss < 1e-12:
            T *= T_decay
            continue
        a_new = a_new / ss

        obj_new = eval_max_b(a_new)

        # Simulated annealing: accept if better or with probability exp(-delta/T)
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new.copy()
            T = max(T * T_decay, 1e-6)
            if verbose and iterations % 10 == 0:
                print(f"    LP iter {iterations}: obj={best_obj:.10f}, T={T:.2e}")
        else:
            delta = obj_new - best_obj
            if annealing and T > 1e-8:
                accept_prob = np.exp(-delta / T)
                if np.random.random() < accept_prob:
                    a = a_new.copy()
                    T *= T_decay
                else:
                    T *= T_decay
            else:
                T *= T_decay

    return best_a, best_obj


def coordinate_descent(a0, time_limit, multipliers=None):
    """Coordinate descent on simplex."""
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = eval_max_b(a)

    if multipliers is None:
        multipliers = np.array([0.5, 0.7, 0.8, 0.9, 1.1, 1.2, 1.3, 1.5, 2.0])

    improved_global = True
    while improved_global and time.monotonic() - t0 < time_limit:
        improved_global = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-15:
                continue
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0.0 or new_val > 0.999:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0.01 or scale > 50.0:
                    continue
                a_tr = a.copy()
                a_tr[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_tr[mask] *= scale
                ss = a_tr.sum()
                if ss < 1e-15:
                    continue
                a_tr = a_tr / ss
                val = eval_max_b(a_tr)
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved_global = True

    return best_a, best_obj

def make_bipolar_init(n, n_clusters=4):

    a = np.zeros(n)
    cluster_size = n // n_clusters
    for i in range(n_clusters):
        start = i * cluster_size
        end = min(start + cluster_size, n)
        a[start:end] = 1.0
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_random_init(n, seed=42):
    rng = np.random.default_rng(seed)
    a = rng.random(n) + 0.1
    return a / a.sum()


def interpolate_sequence(f, new_n):
    """Interpolate to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def combined_optimize(a_init, time_limit, n, seed, verbose=False):
    """
    Combined optimization: SG warmup -> LP refinement -> SG -> LP -> CD.
    """
    t0 = time.monotonic()
    a = a_init.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = eval_max_b(a)

    # Phase A: Quick subgradient warmup (20% of time)
    t_sg1 = time_limit * 0.2
    if t_sg1 > 3:
        a_sg, obj_sg = subgradient_optimize(
            a, time_limit=t_sg1, lr_init=0.004, momentum=0.9,
            n_top=max(10, n // 15), seed=seed
        )
        if obj_sg < best_obj:
            best_obj = obj_sg
            best_a = a_sg.copy()
            if verbose:
                print(f"    SG phase 1: obj={best_obj:.10f}")

    # Phase B: Main LP optimization (50% of time)
    rem = time_limit - (time.monotonic() - t0)
    t_lp = rem * 0.6
    if t_lp > 5:
        a_lp, obj_lp = iterative_lp_optimize(
            best_a, time_limit=t_lp, max_iters=300,
            verbose=verbose, annealing=True, T_init=0.0005, T_decay=0.97
        )
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()
            if verbose:
                print(f"    LP phase: obj={best_obj:.10f}")

    # Phase C: Second SG pass on LP result (15% of time)
    rem = time_limit - (time.monotonic() - t0)
    t_sg2 = rem * 0.4
    if t_sg2 > 5:
        a_sg2, obj_sg2 = subgradient_optimize(
            best_a, time_limit=t_sg2, lr_init=0.002, momentum=0.93,
            n_top=max(15, n // 10), seed=seed + 777
        )
        if obj_sg2 < best_obj:
            best_obj = obj_sg2
            best_a = a_sg2.copy()

    # Phase D: Final LP (remaining)
    rem = time_limit - (time.monotonic() - t0)
    if rem > 5:
        a_lp2, obj_lp2 = iterative_lp_optimize(
            best_a, time_limit=rem, max_iters=100,
            annealing=False, T_init=0.0, T_decay=1.0
        )
        if obj_lp2 < best_obj:
            best_obj = obj_lp2
            best_a = a_lp2.copy()
            if verbose:
                print(f"    LP final: obj={best_obj:.10f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses iterative LP with simulated annealing, subgradient descent,
    and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, budget_s - 20.0)

    best_seq = None
    best_score = float('inf')
    population = []  # List of (seq, score) for diversity

    def update_best(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}, max_b={eval_max_b(a_norm):.8f})")
        # Add to population
        if len(population) < 10:
            population.append((a_norm.copy(), score))
        elif score < population[-1][1]:
            population[-1] = (a_norm.copy(), score)
        population.sort(key=lambda x: x[1])

    # Load parent
    parent = None
    n_parent = 1152
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            n_parent = len(parent)
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={n_parent}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
            population.append((parent.copy(), p_score))
    except Exception:
        pass

    if parent is None:
        parent = make_uniform_init(1152)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"No parent, using uniform n=1152, score={best_score:.6f}")
        population.append((best_seq.copy(), best_score))

    print(f"\n{'='*60}")
    print(f"Starting optimization (budget={budget_s:.0f}s)")
    print(f"Initial best: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'='*60}")

    # === Phase 1: Deep LP optimization on parent ===
    print(f"\n--- Phase 1: Deep LP on parent (n={n_parent}) ---")
    rem = deadline - time.monotonic() - 300
    if rem > 60:
        t1 = min(rem * 0.4, 180.0)
        print(f"  Allocating {t1:.0f}s")
        a_opt, obj_opt = combined_optimize(parent, t1, n_parent, seed, verbose=True)
        update_best(a_opt, "phase1_parent")

    # === Phase 2: LP refinement with different hyperparameters ===
    print(f"\n--- Phase 2: LP refinement variants (best={best_score:.6f}) ---")
    if best_seq is not None:
        n_cur = len(best_seq)
        # Variant A: More annealing
        rem = deadline - time.monotonic() - 250
        if rem > 30:
            t2a = min(rem * 0.3, 80.0)
            print(f"  Variant A (strong annealing): {t2a:.0f}s")
            a_v, obj_v = iterative_lp_optimize(
                best_seq, time_limit=t2a, max_iters=200,
                annealing=True, T_init=0.002, T_decay=0.93, verbose=True
            )
            update_best(a_v, "lp_anneal")

        # Variant B: No annealing, more iterations
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t2b = min(rem * 0.3, 60.0)
            print(f"  Variant B (pure LP): {t2b:.0f}s")
            a_v, obj_v = iterative_lp_optimize(
                best_seq, time_limit=t2b, max_iters=400,
                annealing=False, T_init=0.0, T_decay=1.0, verbose=True
            )
            update_best(a_v, "lp_pure")

    # === Phase 3: Diverse initializations ===
    print(f"\n--- Phase 3: Diverse initializations (best={best_score:.6f}) ---")
    n_base = len(best_seq) if best_seq is not None else n_parent

    inits = {
        "gauss_0.2": make_gaussian_init(n_base, 0.2),
        "gauss_0.35": make_gaussian_init(n_base, 0.35),
        "cosine": make_cosine_init(n_base),
        "uniform": make_uniform_init(n_base),
        "bipolar_4": make_bipolar_init(n_base, 4),
        "bipolar_6": make_bipolar_init(n_base, 6),
        "random_1": make_random_init(n_base, seed=123),
        "random_2": make_random_init(n_base, seed=456),
    }

    total_time_div = min(200.0, (deadline - time.monotonic() - 250.0) * 0.35)
    time_per_init = total_time_div / len(inits)

    for name, init_a in inits.items():
        rem = deadline - time.monotonic() - 200
        if rem < 15:
            break
        t_alloc = min(time_per_init, rem * 0.25)
        if t_alloc < 8:
            continue
        print(f"  {name} (t={t_alloc:.0f}s)")
        a_div, obj_div = combined_optimize(
            init_a, t_alloc, n_base, seed + hash(name) % 100000
        )
        update_best(a_div, f"div_{name}")

    # === Phase 4: Multi-length exploration ===
    print(f"\n--- Phase 4: Multi-length exploration (best={best_score:.6f}) ---")
    if best_seq is not None:
        target_lengths = []
        for factor in [0.5, 0.67, 0.75, 1.0, 1.25, 1.5, 2.0]:
            tl = int(n_parent * factor)
            tl = max(256, min(4096, tl))
            if tl not in target_lengths:
                target_lengths.append(tl)
        target_lengths.sort(key=lambda x: abs(x - n_parent))

        for target_n in target_lengths[:6]:
            rem = deadline - time.monotonic() - 150
            if rem < 20:
                break
            if target_n == len(best_seq):
                continue
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.2, 50.0)
            if t_alloc < 10:
                continue
            print(f"  n={target_n}, t={t_alloc:.0f}s")
            a_len, obj_len = combined_optimize(
                f_interp, t_alloc, target_n, seed + target_n
            )
            update_best(a_len, f"len_n{target_n}")

    # === Phase 5: Crossover/diversity ===
    print(f"\n--- Phase 5: Population crossover (best={best_score:.6f}) ---")
    if len(population) >= 2:
        rem = deadline - time.monotonic() - 100
        if rem > 30:
            # Try interpolating between top sequences
            for i in range(min(3, len(population) - 1)):
                rem = deadline - time.monotonic() - 80
                if rem < 15:
                    break
                seq_a, _ = population[0]
                seq_b, _ = population[i + 1]
                if len(seq_a) != len(seq_b):
                    seq_b = interpolate_sequence(seq_b, len(seq_a))
                for t_mix in [0.3, 0.5, 0.7]:
                    cross = t_mix * seq_a + (1 - t_mix) * seq_b
                    cross = np.maximum(cross, 0.0)
                    ss = cross.sum()
                    if ss < 1e-12:
                        continue
                    cross = cross / ss
                    # Quick LP refinement
                    t_cross = min(20.0, rem * 0.2)
                    a_cross, obj_cross = iterative_lp_optimize(
                        cross, time_limit=t_cross, max_iters=50,
                        annealing=True, T_init=0.001, T_decay=0.9
                    )
                    update_best(a_cross, f"cross_{t_mix}")

    # === Phase 6: Deep optimization on current best ===
    print(f"\n--- Phase 6: Deep final optimization (best={best_score:.6f}) ---")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 60
        if rem > 30:
            t_deep = min(rem * 0.6, 120.0)
            print(f"  Deep LP: n={n_best}, t={t_deep:.0f}s")
            a_deep, obj_deep = iterative_lp_optimize(
                best_seq, time_limit=t_deep, max_iters=500,
                annealing=True, T_init=0.0005, T_decay=0.98, verbose=True
            )
            update_best(a_deep, "deep_lp")

            # Additional SG pass
            rem = deadline - time.monotonic() - 30
            if rem > 10:
                t_sg = min(rem * 0.5, 30.0)
                a_sg, obj_sg = subgradient_optimize(
                    best_seq, time_limit=t_sg, lr_init=0.001, momentum=0.95,
                    n_top=max(20, n_best // 8), seed=seed + 999
                )
                update_best(a_sg, "deep_sg")

    # === Phase 7: Coordinate descent polish ===
    print(f"\n--- Phase 7: Coordinate descent polish (best={best_score:.6f}) ---")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            t_cd = min(rem * 0.7, 40.0)
            n_cd = len(best_seq)
            print(f"  CD: n={n_cd}, t={t_cd:.0f}s")
            a_cd, obj_cd = coordinate_descent(best_seq, time_limit=t_cd)
            update_best(a_cd, "cd")

    # === Phase 8: Final micro-polish ===
    print(f"\n--- Phase 8: Final micro-polish (best={best_score:.6f}) ---")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            n_final = len(best_seq)
            # Quick LP with very small steps
            t_final = min(rem * 0.5, 15.0)
            a_fin, obj_fin = iterative_lp_optimize(
                best_seq, time_limit=t_final, max_iters=50,
                annealing=False, T_init=0.0, T_decay=1.0
            )
            update_best(a_fin, "final_lp")

            # Tiny CD
            rem = deadline - time.monotonic() - 2
            if rem > 3:
                a_cd2, obj_cd2 = coordinate_descent(
                    best_seq, time_limit=rem,
                    multipliers=np.array([0.8, 0.9, 0.95, 1.05, 1.1, 1.2])
                )
                update_best(a_cd2, "final_cd")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy() if parent is not None else make_uniform_init(256)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        # Reset to parent
        if parent is not None:
            best_seq = parent.copy()
        else:
            best_seq = make_uniform_init(256)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"  max_b = {eval_max_b(best_seq):.10f}")
    print(f"  n*max_b = {len(best_seq)*eval_max_b(best_seq):.10f}")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
