"""
Hybrid FISTA + LP optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

Uses FISTA with smooth max (log-sum-exp) as the main workhorse for fast
convergence, combined with LP-based direction finding for refinement,
adaptive temperature scheduling, and multi-length exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient_smooth_max(f, T, b=None):
    """
    Compute gradient of smooth max of auto-convolution.
    smooth_max_T(b) = (1/T) * log(sum_k exp(T * b_k))
    grad_j = sum_k w_k * f_{k-j}
    where w_k = exp(T*b_k) / sum_m exp(T*b_m)
    """
    n = len(f)
    if b is None:
        b = fast_convolve(f)
    scaled = T * b - T * b.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        # All weights equal (degenerate case)
        w = np.ones(len(b)) / len(b)
    else:
        w /= w_sum
    
    g = np.zeros(n)
    # Only process significant weights
    threshold = 1e-12
    for k in range(len(b)):
        if w[k] < threshold:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # g[lo:hi+1] += w[k] * f[k-hi : k-lo+1] reversed
        g[lo:hi+1] += w[k] * f[k-hi:k-lo+1][::-1]
    return g, b.max(), w


def fista_optimize(f_init, time_limit, T_init=50.0, T_max=50000.0, label="fista"):
    """
    FISTA optimization with smooth max of auto-convolution.
    Minimizes smooth_max_T(f*f) subject to f >= 0, sum(f) = 1.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    f_prev = f.copy()
    t_mom = 1.0
    lr = 1.0 / (4.0 * n)  # Conservative initial step size
    T = T_init
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    it = 0
    no_improve = 0
    stagnant = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # FISTA extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        momentum = (t_mom - 1.0) / t_new
        y = f + momentum * (f - f_prev)
        y = np.maximum(y, 0.0)
        s = y.sum()
        if s > 1e-10:
            y /= s
        else:
            y = f.copy()
        
        # Compute gradient
        g, bmax, w = compute_gradient_smooth_max(y, T)
        
        # Gradient step with projection
        f_new = project_simplex(y - lr * g)
        
        # Check if we made progress (objective decreased)
        obj_new = fast_convolve(f_new).max()
        obj_old = fast_convolve(f).max()
        
        # Restart if no progress (non-convex problem)
        if obj_new > obj_old * (1 + 1e-10):
            # Restart: go back and reduce step size
            f_new = f.copy()
            lr *= 0.7
            t_mom = 1.0
            stagnant += 1
            if stagnant > 20:
                # Big restart
                f = project_simplex(best_f + np.random.default_rng(it).standard_normal(n) * 0.01)
                f_prev = f.copy()
                t_mom = 1.0
                lr = 1.0 / (4.0 * n)
                stagnant = 0
                no_improve = 0
                continue
        else:
            stagnant = 0
        
        # Update
        f_prev = f
        f = f_new
        t_mom = t_new
        
        # Track best
        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adapt temperature
        if it % 50 == 0:
            T = min(T * 1.15, T_max)
        
        # Adapt learning rate

            no_improve = 0
            t_mom = 1.0  # Reset momentum

            print(f"    {label} iter={it} obj={best_obj:.10f} score={score:.6f} "

def lp_solve_direction(f, n, top_k, eps=1e-8):
    """
    Solve LP: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for top_k peaks.
    """
    b = fast_convolve(f)
    M = b.max()
    if M < 1e-15:
        return None, M
    
    if top_k < len(b):
        top_idx = np.argpartition(b, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(b))
    top_idx = np.sort(top_idx)
    K = len(top_idx)
    
    # Build constraint matrix
    A = np.zeros((K, n))
    for row_i in range(K):
        k = top_idx[row_i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi+1] = f[lo:hi+1]
    
    g = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [g >= 0, A @ g <= M])
    try:
        prob.solve(solver=cp.SCS, max_iters=200000, eps=eps)
    except Exception:
        return None, M
    
    if g.value is None:
        return None, M
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None, M
    return g_val, M

import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient_smooth_max(f, T, b=None):
    """
    Compute gradient of smooth max of auto-convolution.
    smooth_max_T(b) = (1/T) * log(sum_k exp(T * b_k))
    Returns (gradient, max_b, weights).
    """
    n = len(f)
    if b is None:
        b = fast_convolve(f)
    scaled = T * b
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.ones(len(b)) / len(b)
    else:
        w /= w_sum

    g = np.zeros(n)
    threshold = 1e-12
    for k in range(len(b)):
        if w[k] < threshold:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        g[lo:hi+1] += w[k] * f[k-hi:k-lo+1][::-1]
    return g, b.max(), w


def compute_gradient_topk(f, K, b=None):
    """
    Compute gradient of the maximum of top-K peaks (weighted average).
    Uses inverse-magnitude weighting for emphasis on worst peaks.
    """
    n = len(f)
    if b is None:
        b = fast_convolve(f)
    if K >= len(b):
        top_idx = np.arange(len(b))
    else:
        top_idx = np.argpartition(b, -K)[-K:]

    # Weight by magnitude (emphasize worst peaks)
    b_top = b[top_idx]
    w_top = b_top ** 2
    w_sum = w_top.sum()
    if w_sum < 1e-30:
        w_top = np.ones(K) / K
    else:
        w_top /= w_sum

    w = np.zeros(len(b))
    w[top_idx] = w_top

    g = np.zeros(n)
    for i, k in enumerate(top_idx):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        g[lo:hi+1] += w[k] * f[k-hi:k-lo+1][::-1]
    return g, b.max()


def fista_optimize(f_init, time_limit, T_init=50.0, T_max=50000.0,
                   use_topk=False, topk_K=50, label="fista"):
    """
    FISTA optimization with smooth max (or top-K) of auto-convolution.
    Minimizes subject to f >= 0, sum(f) = 1.
    Includes adaptive restarts and learning rate scheduling.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    f_prev = f.copy()
    f_obj = fast_convolve(f).max()
    t_mom = 1.0
    lr = 2.0 / (n * (1.0 + T_init * 0.1))
    T = T_init

    best_f = f.copy()
    best_obj = f_obj

    it = 0
    no_improve = 0
    restart_count = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        it += 1

        # FISTA extrapolation
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        momentum = (t_mom - 1.0) / t_new
        y = f + momentum * (f - f_prev)
        y = np.maximum(y, 0.0)
        s = y.sum()
        if s > 1e-10:
            y /= s
        else:
            y = f.copy()

        # Compute gradient
        if use_topk:
            g, _ = compute_gradient_topk(y, topk_K)
        else:
            g, _, _ = compute_gradient_smooth_max(y, T)

        # Gradient step with projection
        f_new = project_simplex(y - lr * g)

        # Evaluate new objective
        f_new_obj = fast_convolve(f_new).max()

        if f_new_obj <= f_obj * (1.0 + 1e-12):
            # Accept step
            f_prev = f
            f = f_new
            f_obj = f_new_obj
            t_mom = t_new
        else:
            # Reject step, reset momentum
            t_mom = 1.0
            f_prev = f.copy()
            no_improve += 1
            if no_improve > 60:
                # Big restart with perturbation
                lr = max(lr * 0.7, 1.0 / (100.0 * n))
                pert_scale = 0.01 / np.sqrt(n)
                f = project_simplex(best_f + rng.standard_normal(n) * pert_scale)
                f_prev = f.copy()
                f_obj = fast_convolve(f).max()
                t_mom = 1.0
                no_improve = 0
                restart_count += 1

        # Track best
        if f_obj < best_obj - 1e-15:
            best_obj = f_obj
            best_f = f.copy()
            no_improve = 0
        elif f_obj > best_obj:
            no_improve += 1

        # Adapt temperature
        if not use_topk and it % 80 == 0:
            T = min(T * 1.25, T_max)
            # Reduce lr as T increases (gradient becomes spikier)
            lr *= 0.95

        # Adapt learning rate
        if no_improve > 30:
            lr *= 0.85
            no_improve = 0
            t_mom = 1.0

        if it % 300 == 0:
            score = 2.0 * n * best_obj
            print(f"    {label} iter={it} score={score:.6f} "
                  f"T={T:.0f} lr={lr:.10f} restarts={restart_count} "
                  f"elapsed={time.monotonic()-t0:.0f}s")

    return best_f, best_obj


def lp_solve_direction(f, n, top_k, eps=1e-8):
    """
    Solve LP: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for top_k peaks.
    Returns (g, M) or (None, M) on failure.
    """
    b = fast_convolve(f)
    M = b.max()
    if M < 1e-15:
        return None, M

    if top_k < len(b):
        top_idx = np.argpartition(b, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(b))
    top_idx = np.sort(top_idx)
    K = len(top_idx)

    A = np.zeros((K, n))
    for row_i in range(K):
        k = top_idx[row_i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi+1] = f[lo:hi+1]

    g_var = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), [g_var >= 0, A @ g_var <= M])
    try:
        prob.solve(solver=cp.SCS, max_iters=300000, eps=eps)
    except Exception:
        return None, M

    if g_var.value is None:
        return None, M
    g_val = np.maximum(g_var.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None, M
    return g_val, M


def lp_optimize(f_init, time_limit, top_k_init=50, top_k_max=400, label="lp"):
    """LP-based optimization with adaptive top_k and line search."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    stuck = 0
    top_k = top_k_init

    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 25 == 0 and top_k < top_k_max:
            top_k = min(top_k + 15, top_k_max)

        g, M = lp_solve_direction(f, n, top_k)
        if g is None:
            g, M = lp_solve_direction(f, n, min(2 * top_k, 2 * n - 1))
            if g is None:
                break

        S_g = g.sum()
        if S_g <= 1.0 + 1e-5:
            stuck += 1
            if stuck > 15:
                break
            top_k = min(top_k * 2, top_k_max)
            continue

        h = g / S_g
        best_t = 0.1
        best_t_obj = best_obj

        for t in [0.02, 0.04, 0.06, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            f_trial = (1.0 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t

        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s

        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 40:
                break

        if it % 15 == 0:
            print(f"    {label} iter={it} score={2.0*n*best_obj:.6f} "
                  f"top_k={top_k} t={time.monotonic()-t0:.0f}s")

    return best_f, best_obj


def coordinate_polish(f_init, time_limit):
    """Coordinate descent with multipliers, focusing on peak reduction."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    multipliers = np.array([0.4, 0.6, 0.75, 0.9, 1.1, 1.25, 1.5, 2.0])

    improved = True
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-15:
                continue
            best_j_val = f[j]
            best_j_obj = best_obj
            for mult in multipliers:
                new_val = f[j] * mult
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-14:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f /= s
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    improved = True
    return f, best_obj


def peak_equalization(f_init, time_limit, label="pe"):
    """
    Iterative peak equalization: identify worst peak, reduce contributing
    coordinates, compensate by increasing less-important coordinates.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        peak_idx = b.argmax()
        peak_val = b[peak_idx]

        # Find contributors to the worst peak
        lo = max(0, peak_idx - n + 1)
        hi = min(n - 1, peak_idx)
        contributors = np.arange(lo, hi + 1)
        contrib_vals = f[contributors] * f[peak_idx - contributors]
        total_contrib = contrib_vals.sum()

        if total_contrib < 1e-15:
            break

        # Find coordinates with low contribution to ANY peak
        # (coordinates that are "safe" to increase)
        # Approximate: coordinates far from the peak
        peak_center = peak_idx - (n - 1) / 2.0
        dist = np.abs(np.arange(n) - peak_center)
        safe_mask = dist > n * 0.3

        if not safe_mask.any():
            safe_mask = np.ones(n, dtype=bool)

        # Reduce top contributors slightly, increase safe coordinates
        top_contrib_count = max(3, len(contributors) // 4)
        top_contrib_local = np.argsort(contrib_vals)[-top_contrib_count:]
        reduce_idx = contributors[top_contrib_local]

        # Reduce amount proportional to contribution
        reduce_amount = peak_val * 0.05
        per_reduce = reduce_amount / (f[reduce_idx] * f[peak_idx - reduce_idx]).sum() + 1e-15
        for idx in reduce_idx:
            f[idx] *= (1.0 - 0.02)

        # Increase safe coordinates
        safe_idx = np.where(safe_mask & (f > 1e-10))[0]
        if len(safe_idx) > 0:
            increase_total = f[reduce_idx].sum() * 0.02
            per_increase = increase_total / len(safe_idx)
            f[safe_idx] += per_increase

        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s

        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
        elif cur_obj > best_obj * 1.01:
            # Revert if too bad
            f = best_f.copy()

        if it % 50 == 0:
            print(f"    {label} iter={it} score={2.0*n*best_obj:.6f} "
                  f"t={time.monotonic()-t0:.0f}s")

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main optimization: FISTA (smooth max + top-K) + LP refinement +
    peak equalization + coordinate descent + multi-length exploration.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(1024) / 1024
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"No parent, using uniform n=1024, score={p_score:.6f}")

    best_seq = parent.copy()
    best_score = p_score

    # === Phase 1: FISTA with smooth max (main workhorse) ===
    n_parent = len(parent)
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.35, 600.0)
    if t1 > 30:
        print(f"\n=== Phase 1: FISTA smooth-max (n={n_parent}, budget={t1:.0f}s) ===")
        f1, obj1 = fista_optimize(parent, time_limit=t1, T_init=30.0,
                                   T_max=30000.0, use_topk=False, label="fista_sm")
        score1 = float(evaluate_sequence(f1.tolist()))
        if np.isfinite(score1) and score1 < best_score:
            best_score = score1
            best_seq = f1.copy()
            print(f"  NEW BEST [fista_sm]: {score1:.6f}")
        else:
            print(f"  fista_sm: {score1:.6f} (no improvement)")

    # === Phase 2: FISTA with top-K gradient ===
    print(f"\n=== Phase 2: FISTA top-K (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.20, 300.0)
    if t2 > 20:
        n_b = len(best_seq)
        K = max(30, n_b // 20)
        print(f"  n={n_b}, K={K}, budget={t2:.0f}s")
        f2, obj2 = fista_optimize(best_seq, time_limit=t2, T_init=100.0,
                                   T_max=10000.0, use_topk=True, topk_K=K,
                                   label="fista_topk")
        score2 = float(evaluate_sequence(f2.tolist()))
        if np.isfinite(score2) and score2 < best_score:
            best_score = score2
            best_seq = f2.copy()
            print(f"  NEW BEST [fista_topk]: {score2:.6f}")

    # === Phase 3: LP refinement ===
    print(f"\n=== Phase 3: LP refinement (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.20, 300.0)
    if t3 > 20:
        print(f"  Budget: {t3:.0f}s")
        f3, obj3 = lp_optimize(best_seq, time_limit=t3, top_k_init=50,
                                top_k_max=350, label="lp_refine")
        score3 = float(evaluate_sequence(f3.tolist()))
        if np.isfinite(score3) and score3 < best_score:
            best_score = score3
            best_seq = f3.copy()
            print(f"  NEW BEST [lp_refine]: {score3:.6f}")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length (best={best_score:.6f}) ===")
    n_base = len(best_seq)
    target_lengths = []
    for factor in [0.85, 1.15, 1.3, 1.5]:
        tl = int(n_base * factor)
        tl = max(512, min(4000, tl))
        if tl != n_base and tl not in target_lengths:
            target_lengths.append(tl)
    target_lengths.sort(key=lambda x: abs(x - n_base))

    for target_n in target_lengths[:2]:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t4 = min(rem * 0.18, 200.0)
        if t4 < 25:
            break
        old_x = np.linspace(0.0, 1.0, n_base)
        new_x = np.linspace(0.0, 1.0, target_n)
        f_interp = np.interp(new_x, old_x, best_seq)
        f_interp = np.maximum(f_interp, 0.0)
        s = f_interp.sum()
        if s > 1e-10:
            f_interp /= s
        else:
            f_interp = np.ones(target_n) / target_n

        print(f"  FISTA n={target_n}, budget={t4:.0f}s")
        f4, obj4 = fista_optimize(f_interp, time_limit=t4, T_init=30.0,
                                   T_max=30000.0, label=f"fista_n{target_n}")
        score4 = float(evaluate_sequence(f4.tolist()))
        if np.isfinite(score4) and score4 < best_score:
            best_score = score4
            best_seq = f4.copy()
            print(f"  NEW BEST [n{target_n}]: {score4:.6f}")

    # === Phase 5: Peak equalization ===
    print(f"\n=== Phase 5: Peak equalization (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.08, 80.0)
    if t5 > 10:
        print(f"  Budget: {t5:.0f}s")
        f5, obj5 = peak_equalization(best_seq, time_limit=t5)
        score5 = float(evaluate_sequence(f5.tolist()))
        if np.isfinite(score5) and score5 < best_score:
            best_score = score5
            best_seq = f5.copy()
            print(f"  NEW BEST [pe]: {score5:.6f}")

    # === Phase 6: FISTA high-temperature polish ===
    print(f"\n=== Phase 6: FISTA polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.15, 150.0)
    if t6 > 15:
        n_b = len(best_seq)
        print(f"  Budget: {t6:.0f}s, n={n_b}")
        f6, obj6 = fista_optimize(best_seq, time_limit=t6, T_init=2000.0,
                                   T_max=80000.0, label="fista_polish")
        score6 = float(evaluate_sequence(f6.tolist()))
        if np.isfinite(score6) and score6 < best_score:
            best_score = score6
            best_seq = f6.copy()
            print(f"  NEW BEST [fista_polish]: {score6:.6f}")

    # === Phase 7: Coordinate descent ===
    print(f"\n=== Phase 7: Coordinate descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t7 = min(rem * 0.12, 100.0)
    if t7 > 8:
        print(f"  Budget: {t7:.0f}s")
        f7, obj7 = coordinate_polish(best_seq, time_limit=t7)
        score7 = float(evaluate_sequence(f7.tolist()))
        if np.isfinite(score7) and score7 < best_score:
            best_score = score7
            best_seq = f7.copy()
            print(f"  NEW BEST [cd]: {score7:.6f}")

    # === Phase 8: Final LP burst ===
    print(f"\n=== Phase 8: Final LP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t8 = min(rem * 0.30, 150.0)
    if t8 > 10:
        n_b = len(best_seq)
        print(f"  Budget: {t8:.0f}s, n={n_b}")
        f8, obj8 = lp_optimize(best_seq, time_limit=t8, top_k_init=150,
                                top_k_max=500, label="lp_final")
        score8 = float(evaluate_sequence(f8.tolist()))
        if np.isfinite(score8) and score8 < best_score:
            best_score = score8
            best_seq = f8.copy()
            print(f"  NEW BEST [lp_final]: {score8:.6f}")

    # Final verification

    if np.isfinite(fs):
        best_score = fs
    else:

        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'='*50}")

    return [float(x) for x in best_seq]
