Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using coordinate descent with
bisection on 1D subproblems, simulated annealing restarts, and L-BFGS polish.
Key innovation: exact 1D coordinate optimization (convex 1D problem) is much
faster than LP-based best response, enabling millions of iterations.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from time import monotonic


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_unnormalized(a):
    """Compute the objective for a non-negative sequence (non-normalized)."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fast_convolve(a)
    return 2.0 * n * c.max() / (s * s)


def coordinate_descent_step(a, j, n):
    """
    Find optimal value of a[j] given all other elements fixed.
    Maintains sum(a) = 1 by scaling others.
    Returns the new a with optimized a[j].
    """
    # We want to find u = a[j] that minimizes max_k conv(a_new, a_new)[k]
    # where a_new has a[j] = u and others scaled to maintain sum=1.
    
    # If a[j] = u, others scaled by factor (1-u)/(1-a[j])
    # (assuming a[j] != 1)
    
    aj = a[j]
    others_sum = 1.0 - aj
    
    if others_sum < 1e-15:
        return a  # Can't change j meaningfully
    
    # Precompute the "base" autocorrelation without coordinate j
    # conv(a,a)[k] = base[k] + contributions from j
    # For k = 2*j: base + (u^2 - aj^2) + scaling effects
    # This is complex due to scaling. Let's just do a direct 1D search.
    
    # Use bisection/ternary search on u in [0, 0.999]
    # The function f(u) = max_k conv(a(u), a(u)) where a(u) has a[j]=u and rest scaled.
    
    def eval_u(u):
        if u < 0 or u > 1:
            return float('inf')
        a_new = a.copy()
        a_new[j] = u
        scale = (1.0 - u) / others_sum
        mask = np.ones(n, dtype=bool)
        mask[j] = False
        a_new[mask] = a[mask] * scale
        c = fast_convolve(a_new)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adam optimizer in log-space with softmax-smoothed gradient and temperature annealing
2. Multi-length exploration (512, 768, 1024, 1536, 2048)
3. Coordinate descent with direct evaluation
4. L-BFGS final polish
Key innovation: fast FFT-based gradient computation (O(n log n) per step) enables
millions of Adam steps, with temperature annealing to transition from broad
to peak-focused optimization.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize
from time import monotonic


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_weighted_grad(a, p, size_cache=None):
    """
    Compute gradient of sum_k p[k]*conv(a,a)[k] w.r.t. a.
    Returns array of length n.
    Uses FFT-based cross-correlation.
    """
    n = len(a)
    if size_cache is None:
        size = 1 << (3 * n - 2 - 1).bit_length()
    else:
        size = size_cache
    p_pad = np.zeros(size)
    p_pad[:len(p)] = p
    a_pad = np.zeros(size)
    a_pad[:n] = a
    P = fft(p_pad)
    A = fft(a_pad)
    corr = ifft(np.conj(P) * A).real[:n]
    return 2.0 * corr, size


def adam_step(z, m, v, t, lr, beta1, beta2, epsilon, grad_z):
    """Perform one Adam update step. Returns updated z, m, v."""
    m = beta1 * m + (1 - beta1) * grad_z
    v = beta2 * v + (1 - beta2) * (grad_z * grad_z)
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    z = z - lr * m_hat / (np.sqrt(v_hat) + epsilon)
    return z, m, v


def log_softmax(z):
    """Compute softmax of z (for converting log-space to a-space)."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def evaluate_score(a):
    """Compute the objective for a non-negative sequence."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fast_convolve(a)
    return 2.0 * n * c.max() / (s * s)

    Run Adam optimization in log-space with temperature annealing.
    initial_a: non-negative sequence (will be normalized)
    T_start, T_end: temperature range for softmax (linear annealing)
    Returns: (best_a, best_score)

    rng = np.random.RandomState(seed)
    
    a = np.maximum(np.asarray(initial_a, dtype=np.float64), 1e-30)
    a = a / a.sum()
    
    # Initialize in log-space
    z = np.log(a)
    z = z - z.mean()  # center for numerical stability
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    t0 = monotonic()
    total_planned = int(time_limit / 0.001)  # rough estimate
    no_improve = 0
    patience = 5000
    
    # Precompute FFT size
    size = 1 << (3 * n - 2 - 1).bit_length()
    
    while monotonic() - t0 < time_limit:
        t += 1
        elapsed = monotonic() - t0
        progress = min(1.0, elapsed / max(time_limit * 0.8, 1.0))
        
        # Anneal temperature
        T = T_start * (T_end / T_start) ** progress
        
        # Current a in normalized form
        a = log_softmax(z)
        
        # Compute convolution and softmax weights
        c = fast_convolve(a)
        Tc = T * c
        Tc_max = Tc.max()
        exp_Tc = np.exp(Tc - Tc_max)
        p = exp_Tc / exp_Tc.sum()
        
        # Compute gradient of weighted autocorrelation w.r.t. a
        grad_a, _ = compute_weighted_grad(a, p, size)
        
        # Transform to z-space gradient
        # df/dz[j] = a[j] * (grad_a[j] - sum_i a[i]*grad_a[i])
        alpha = np.dot(a, grad_a)
        grad_z = a * (grad_a - alpha)
        
        # Scale learning rate (decay over time)
        lr = lr_init / (1.0 + 0.001 * t)
        
        # Adam step
        z, m, v = adam_step(z, m, v, t, lr, 0.9, 0.999, 1e-8, grad_z)
        
        # Periodic evaluation of true objective
        if t % 100 == 0:
            a_check = log_softmax(z)
            score = 2.0 * n * fast_convolve(a_check).max()
            if score < best_score - 1e-10:
                best_score = score
                best_a = a_check.copy()
                no_improve = 0
            else:
                no_improve += 100
            
            if verbose and t % 10000 == 0:
                print(f"      Adam n={n} t={t} ({elapsed:.0f}s): T={T:.1f} score={best_score:.8f}")
        
        # Restart if stuck
        if no_improve > patience:
            z = np.log(best_a) + rng.standard_normal(n) * 0.05
            z = z - z.mean()
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            if verbose:
                print(f"      Adam restart at t={t}, score={best_score:.8f}")
    
    # Final evaluation
    a_final = log_softmax(z)
    score_final = 2.0 * n * fast_convolve(a_final).max()
    if score_final < best_score:
        best_score = score_final
        best_a = a_final.copy()
    
    return best_a, best_score


def coordinate_descent(a0, time_limit, verbose=True, seed=42):
    """
    Coordinate descent with direct evaluation.
    For each coordinate, do a 1D search (golden section) to minimize the max autocorrelation.
    """
    rng = np.random.RandomState(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    
    def eval_a(a_candidate):
        """Evaluate max autocorrelation of a_candidate (assumed sum=1, non-negative)."""
        c = fast_convolve(a_candidate)
        return c.max()
    
    current_max = eval_a(a)
    
    # Golden section search on a single coordinate
    golden_ratio = (np.sqrt(5) - 1) / 2  # ~0.618
    
    def optimize_coordinate(a_arr, j):
        """Find best value for a[j] keeping sum=1 by scaling others."""
        aj = a_arr[j]
        others_sum = 1.0 - aj
        if others_sum < 1e-12:
            return a_arr, current_max
        
        def obj(u):
            """Objective value when a[j]=u, others scaled to maintain sum=1."""
            if u < 0 or u >= 1.0:
                return float('inf')
            a_try = a_arr.copy()
            a_try[j] = u
            scale = (1.0 - u) / others_sum
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a_try[mask] *= scale
            if a_try.min() < -1e-15:
                return float('inf')
            return eval_a(a_try)
        
        # Golden section search on u in [0, 1-eps]
        lo = 0.0
        hi = 1.0 - 1e-6
        
        # First check if current value is already good
        obj_current = obj(aj)
        
        # Try a few exploratory points
        best_u = aj
        best_obj = obj_current
        
        for test_u in [0.01, 0.1, 0.3, 0.5, 0.7, 0.9, 0.99]:
            if abs(test_u - aj) > 0.05:  # don't waste time near current
                vo = obj(test_u)
                if vo < best_obj:
                    best_obj = vo
                    best_u = test_u
        
        # If we found a better point, do golden section around it
        if best_u != aj:
            # Narrow the search interval
            margin = 0.1
            lo = max(0.0, best_u - margin)
            hi = min(1.0 - 1e-6, best_u + margin)
            
            # Golden section
            c1 = hi - golden_ratio * (hi - lo)
            c2 = lo + golden_ratio * (hi - lo)
            f1 = obj(c1)
            f2 = obj(c2)
            
            for _ in range(20):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - golden_ratio * (hi - lo)
                    f1 = obj(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + golden_ratio * (hi - lo)
                    f2 = obj(c2)
            
            best_u = (lo + hi) / 2
            best_obj = obj(best_u)
        
        if best_obj < obj_current - 1e-12:
            a_new = a_arr.copy()
            a_new[j] = best_u
            scale = (1.0 - best_u) / others_sum
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a_new[mask] *= scale
            return a_new, best_obj
        
        return a_arr, obj_current
    
    # Main CD loop
    no_improve = 0
    total_steps = 0
    
    while monotonic() - t0 < time_limit:
        # Pick random coordinate
        j = rng.randint(n)
        
        a_new, new_max = optimize_coordinate(a, j)
        total_steps += 1
        
        if a_new is not a and new_max < current_max - 1e-12:
            a = a_new
            current_max = new_max
            score = 2.0 * n * current_max
            if score < best_score - 1e-10:
                best_score = score
                best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 3 * n:
            # Perturb and restart
            a = best_a + rng.standard_normal(n) * 0.002
            a = np.maximum(a, 0)
            a = a / a.sum()
            current_max = eval_a(a)
            no_improve = 0
        
        if verbose and total_steps % 5000 == 0:
            elapsed = monotonic() - t0
            print(f"    CD steps={total_steps} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score


def lbgfs_polish(a0, time_limit, T=50.0, maxiter=5000, seed=42):
    """L-BFGS polish in log-space with soft-max objective."""
    n = len(a0)
    a_safe = np.maximum(a0, 1e-30)
    z0 = np.log(a_safe)
    z0 = z0 - z0.mean()
    t0 = monotonic()
    
    size = 1 << (3 * n - 2 - 1).bit_length()
    
    def func(z):
        if monotonic() - t0 > time_limit:
            return 1e15, np.zeros(n)
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()
        c = fast_convolve(a)
        Tc = T * c
        Tc_max = Tc.max()
        exp_Tc = np.exp(Tc - Tc_max)
        sum_exp = exp_Tc.sum()
        g = (Tc_max + np.log(sum_exp)) / T
        p = exp_Tc / sum_exp
        
        # Gradient
        grad_a, _ = compute_weighted_grad(a, p, size)
        alpha = np.dot(a, grad_a)
        grad_z = a * (grad_a - alpha) * 2.0 * n
        
        return 2.0 * n * g, grad_z
    
    def func_jac(z):
        val, grad = func(z)
        return val
    
    def func_full(z):
        val, grad = func(z)
        return val
    
    def func_grad_full(z):
        val, grad = func(z)
        return grad
    
    try:
        # Use scipy minimize with custom gradient
        class FuncGrad:
            def __call__(self, z):
                return func(z)[0]
        
        fg = FuncGrad()
        
        res = minimize(
            fun=func_full,
            x0=z0,
            jac=func_grad_full,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': maxiter * 10, 'disp': False}
        )
        
        z_final = res.x
        a_out = log_softmax(z_final)
        a_out = np.maximum(a_out, 0)
        a_out = a_out / a_out.sum()
    except Exception:
        a_out = a0
    
    return a_out


def peak_shave(a0, time_limit, seed=42):
    """Targeted peak shaving: identify peak positions and reduce contributing elements."""
    rng = np.random.RandomState(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    
    for step in range(5000):
        if monotonic() - t0 > time_limit:
            break
        
        c = fast_convolve(a)
        peak_k = int(np.argmax(c))
        peak_val = c[peak_k]
        
        # Find contributing pairs
        # c[peak_k] = sum_i a[i]*a[peak_k-i] for valid i
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        if contribs.sum() < 1e-20:
            break
        
        # Try reducing top contributors
        top_k = np.argsort(contribs)[::-1][:8]
        improved = False
        
        for ci in top_k:
            i = int(idx[ci])
            old_val = a[i]
            if old_val < 1e-10:
                continue
            
            for factor in [0.95, 0.9, 0.85, 0.8, 0.7]:
                new_val = old_val * factor
                diff = old_val - new_val
                if diff < 1e-12:
                    continue
                
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a[mask]
                others_sum = others.sum()
                if others_sum < 1e-10:
                    continue
                
                a_trial = a.copy()
                a_trial[i] = new_val
                a_trial[mask] = others + diff * others / others_sum
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-12:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Try random perturbation
            a_trial = a + rng.standard_normal(n) * 0.0003
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-12:
                a = a_trial
                best_score = score
                best_a = a.copy()
            elif step > 200:
                break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def make_initialization(n, seed=42):
    """Create diverse initializations."""
    rng = np.random.RandomState(seed)
    inits = {}
    
    # Uniform
    inits['uniform'] = np.ones(n) / n
    
    # Linear ramp (asymmetric - important for non-palindromic optimization)
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = ramp / ramp.sum()
    
    # Exponential
    exp_seq = np.exp(np.linspace(0, 2, n))
    inits['exp'] = exp_seq / exp_seq.sum()
    
    # Sine-based asymmetric
    t = np.linspace(0, np.pi, n)
    sine = np.sin(t) ** 2 + 0.1 * np.sin(2 * t) + 0.5
    sine = np.maximum(sine, 0)
    inits['sine'] = sine / sine.sum()
    
    # Two-hump asymmetric
    x = np.linspace(0, 1, n)
    hump = np.exp(-((x - 0.3) ** 2) / 0.02) + 0.7 * np.exp(-((x - 0.8) ** 2) / 0.03)
    inits['twohump'] = hump / hump.sum()
    
    # Random
    rand = rng.random(n)
    inits['random'] = rand / rand.sum()
    
    # Cosine asymmetric (like half-Hann but shifted)
    cos_seq = 0.5 * (1 + np.cos(2 * np.pi * x / 1.5))
    cos_seq = np.maximum(cos_seq, 0)
    if cos_seq.sum() > 1e-10:
        inits['cos_asym'] = cos_seq / cos_seq.sum()
    
    return inits


def check_and_update(a, label, best_seq, best_score):
    """Evaluate and update best."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    Phase 1: Adam with T-annealing at multiple n values (main workhorse)
    Phase 2: Coordinate descent refinement
    Phase 3: L-BFGS polish
    Phase 4: Peak shaving
    Phase 5: Final verification
    """
    np.random.seed(seed)
    deadline = monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass
    
    if parent is not None:
        best_seq = parent.copy()
        best_score = p_score
    else:
        best_seq = np.ones(768) / 768.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Initial uniform: n=768, score={best_score:.8f}")
    
    n_parent = len(best_seq)
    
    # ============================================================
    # Phase 1: Adam optimization at multiple lengths
    # ============================================================
    print(f"\n=== Phase 1: Adam Optimization (best={best_score:.8f}) ===")
    
    # Define n values to try
    n_values = []
    
    # Always include parent length
    n_values.append(n_parent)
    
    # Add nearby and larger values
    for candidate_n in [512, 768, 1024, 1536, 2048, 3072]:
        if candidate_n not in n_values and 256 <= candidate_n <= 4096:
            n_values.append(candidate_n)
    
    n_values = sorted(set(n_values))
    
    # Allocate time: give parent length the most time
    rem = deadline - monotonic() - 400
    total_adam_time = max(100, min(900, rem))
    
    # Time allocation weights (parent gets most)
    num_n = len(n_values)
    weights = [1.0 / (i + 1) for i in range(num_n)]
    w_sum = sum(weights)
    weights = [w / w_sum for w in weights]
    
    # Try multiple initializations for the parent length
    init_types = ['ramp', 'exp', 'twohump', 'sine', 'random', 'uniform', 'cos_asym']
    
    for ni, n_try in enumerate(n_values):
        rem = deadline - monotonic() - 350
        if rem < 60:
            break
        
        t_alloc_total = total_adam_time * weights[ni]
        t_alloc_total = min(t_alloc_total, rem * 0.7)
        if t_alloc_total < 30:
            continue
        
        print(f"\n  --- n={n_try}, total budget={t_alloc_total:.0f}s ---")
        
        # Get initializations
        if len(best_seq) == n_try:
            inits = {'parent': best_seq.copy()}
        else:
            inits = {'interp': interpolate_sequence(best_seq, n_try)}
        
        # Add diverse initializations
        div_inits = make_initialization(n_try, seed=seed + n_try)
        inits.update(div_inits)
        
        # Brief screening: run each init for a short time, pick the best
        n_inits = len(inits)
        screen_time = min(20, t_alloc_total * 0.15 / max(n_inits, 1))
        screen_time = max(5, screen_time)
        
        best_init_score = float('inf')
        best_init_a = None
        
        for iname, ia in inits.items():
            if monotonic() + 30 > deadline:
                break
            t_screen = min(screen_time, deadline - monotonic() - 300)
            if t_screen < 3:
                break
            
            a_scr, scr_score = run_adam(
                n_try, ia, t_screen,
                T_start=200, T_end=5,
                lr_init=0.05, verbose=False, seed=seed + hash(iname) % 10000
            )
            if scr_score < best_init_score:
                best_init_score = scr_score
                best_init_a = a_scr
            if verbose:
                print(f"    Screen {iname}: {scr_score:.8f}")
        
        if best_init_a is None:
            best_init_a = inits[list(inits.keys())[0]].copy()
        
        # Main Adam run with the best initialization
        t_main = min(t_alloc_total - 30, deadline - monotonic() - 300)
        if t_main > 20:
            print(f"    Main Adam: init_score={best_init_score:.8f}, budget={t_main:.0f}s")
            
            # Multiple T-annealing runs
            T_configs = [
                (500, 3, 0.08),   # High start, low lr
                (200, 5, 0.05),   # Medium
                (100, 10, 0.03),  # Lower
            ]
            
            for ti, (T_s, T_e, lr) in enumerate(T_configs):
                rem = deadline - monotonic() - 280
                if rem < 20:
                    break
                t_run = min(t_main / (len(T_configs) - ti), rem * 0.8)
                if t_run < 10:
                    break
                
                a_adam, adam_score = run_adam(
                    n_try, best_init_a, t_run,
                    T_start=T_s, T_end=T_e,
                    lr_init=lr, verbose=True,
                    seed=seed + ni * 100 + ti
                )
                best_seq, best_score = check_and_update(a_adam, f"Adam_n{n_try}_T{T_s}-{T_e}", best_seq, best_score)
                best_init_a = a_adam  # use as init for next config
    
    # ============================================================
    # Phase 2: Coordinate Descent
    # ============================================================
    print(f"\n=== Phase 2: Coordinate Descent (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 150
        if rem > 60:
            t_cd = min(rem * 0.6, 400.0)
            print(f"  CD n={len(best_seq)}, budget={t_cd:.0f}s")
            a_cd, cd_score = coordinate_descent(best_seq, t_cd, verbose=True, seed=seed + 7)
            best_seq, best_score = check_and_update(a_cd, "coord_descent", best_seq, best_score)
    
    # ============================================================
    # Phase 3: More Adam with lower temperatures
    # ============================================================
    print(f"\n=== Phase 3: Low-T Adam (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 100
        if rem > 40:
            t_low = min(rem * 0.5, 200.0)
            n_best = len(best_seq)
            a_low, low_score = run_adam(
                n_best, best_seq, t_low,
                T_start=50, T_end=1,
                lr_init=0.02, verbose=True, seed=seed + 100
            )
            best_seq, best_score = check_and_update(a_low, "lowT_adam", best_seq, best_score)
    
    # ============================================================
    # Phase 4: L-BFGS Polish
    # ============================================================
    print(f"\n=== Phase 4: L-BFGS Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        for T in [30, 80, 200]:
            rem = deadline - monotonic() - 60
            if rem < 15:
                break
            t_lb = min(rem * 0.3, 60.0)
            print(f"  L-BFGS T={T}, budget={t_lb:.0f}s")
            a_lb = lbgfs_polish(best_seq, t_lb, T=T, maxiter=3000, seed=seed + 200 + T)
            best_seq, best_score = check_and_update(a_lb, f"lbgfs_T{T}", best_seq, best_score)
    
    # ============================================================
    # Phase 5: Peak Shaving
    # ============================================================
    print(f"\n=== Phase 5: Peak Shaving (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 30
        if rem > 15:
            t_ps = min(rem * 0.5, 80.0)
            a_ps, ps_score = peak_shave(best_seq, t_ps, seed=seed + 300)
            best_seq, best_score = check_and_update(a_ps, "peak_shave", best_seq, best_score)
    
    # ============================================================
    # Phase 6: Final Adam burst + verification
    # ============================================================
    print(f"\n=== Phase 6: Final (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 10
        if rem > 15:
            t_final = min(rem * 0.6, 40.0)
            n_best = len(best_seq)
            a_fin, fin_score = run_adam(
                n_best, best_seq, t_final,
                T_start=20, T_end=1,
                lr_init=0.01, verbose=False, seed=seed + 400
            )
            best_seq, best_score = check_and_update(a_fin, "final_adam", best_seq, best_score)
    
    # Final check
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
    
    # Verify with evaluate_sequence
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}")
    
    return [float(x) for x in best_seq]
