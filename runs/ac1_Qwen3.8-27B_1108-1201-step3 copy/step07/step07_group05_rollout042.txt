Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP-based fictitious play for global optimization
2. Exact pairwise coordinate descent with ternary search
3. Smoothed L-BFGS in log-space for fast local convergence
4. Multi-length exploration

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score_normalized(a):
    """Score for a normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def softmax_from_z(z):
    z = z - z.max()
    exp_z = np.exp(z)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def z_from_a(a):
    return np.log(np.maximum(a, 1e-30))

def smooth_max(conv, T):
    """Smooth approximation of max using log-sum-exp."""
    m = conv.max()
    return m + T * np.log(np.exp((conv - m) / T).sum() / len(conv))

def smooth_max_grad(conv, T):
    """Gradient of smooth max w.r.t. conv."""
    shifted = (conv - conv.max()) / T
    w = np.exp(shifted)
    w /= w.sum()
    return w

def lp_fictitious_play_step(f, M, upper_bound):
    """Solve the LP: max sum(g) s.t. (f*g)[k] <= M, 0 <= g[j] <= upper_bound."""
    n = len(f)
    # Constraint: (f*g)[k] = sum_j f[k-j] * g[j] <= M for k = 0, ..., 2n-2
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # Build A matrix
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        # g[j] contributes f[k-j] to constraint k, for j where 0 <= k-j < n
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = f[k - j]
    
    c = -np.ones(n)  # minimize -sum(g)
    b = np.full(2*n-1, M)
    bounds = [(0, upper_bound)] * n
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        return None

# Actually, building the A matrix with a double loop is slow. Let me use a different approach.
# The matrix A is Toeplitz-like. Let me construct it more efficiently.

def build_conv_matrix(f):
    """Build the matrix A where (A @ g)[k] = (f * g)[k]."""
    n = len(f)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        # g[j] contributes f[k-j] to row k, for k = j, j+1, ..., j+n-1
        # (since we need 0 <= k-j < n)
        A[j:j+n, j] = f
    return A

Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using a combination of:

import numpy as np

from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score_normalized(a):
    """Score = 2*n*max(conv) for sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

    a = np.maximum(a, 0.0)
    if a.sum() < 0.01:
        return np.inf
    return float(evaluate_sequence([float(x) for x in a]))

def softmax_from_z(z):
    z = z - z.max()

    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)

def interp_sequence(f, new_n):
    """Interpolate to new length, preserving non-negativity and normalization."""
    old_n = len(f)
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_new = np.interp(new_x, old_x, f_norm)
    f_new = np.maximum(f_new, 0)
    ss = f_new.sum()
    if ss > 1e-15:
        f_new = f_new / ss
    return f_new

def lp_fictitious_play(f, max_iters, time_limit, verbose=False):
    """LP-based fictitious play."""
    n = len(f)
    f = f.copy().astype(np.float64)
    s = f.sum()
    f = f / s
    best_f = f.copy()
    best_score = compute_score_normalized(f)
    t0 = time.monotonic()
    rng = np.random.default_rng(0)
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute current M
        conv_f = fast_convolve(f)
        M = conv_f.max()
        
        # Build constraint matrix A (2n-1 x n)
        A = np.zeros((2*n-1, n))
        for j in range(n):
            A[j:j+n, j] = f
        
        # LP: max sum(g) s.t. A@g <= M, 0 <= g <= ub
        ub = M / max(f.min(), 1e-10) * 10  # Upper bound
        ub = min(ub, 10.0)  # Cap the upper bound
        
        c = -np.ones(n)
        b = np.full(2*n-1, M)
        bounds = [(0, ub)] * n
        
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                     options={'presolve': True, 'time_limit': 5})
        
        if not res.success:
            # Fallback: use the max position as direction
            k_star = int(np.argmax(conv_f))
            g = np.zeros(n)
            js = max(0, k_star - n + 1)
            je = min(n - 1, k_star)
            g[js:je+1] = f[k_star - np.arange(js, je+1)]
            g = np.maximum(g, 0)
            if g.sum() < 1e-15:
                g = np.ones(n) / n
        else:
            g = res.x
        
        # Normalize g to match sum of f
        gs = g.sum()
        if gs < 1e-15:
            g = np.ones(n) / n
        else:
            g = g / gs
        
        # Update with adaptive step size
        alpha = 2.0 / (it + 2)
        alpha = min(alpha, 0.5)
        f_new = (1 - alpha) * f + alpha * g
        f_new = np.maximum(f_new, 0)
        fs = f_new.sum()
        if fs < 1e-15:
            f_new = best_f
        else:
            f_new = f_new / fs
        
        score = compute_score_normalized(f_new)
        if score < best_score:
            best_score = score
            best_f = f_new.copy()
        
        f = f_new
        
        if verbose and it % 100 == 0:
            print(f"    FP iter={it}, score={best_score:.8f}, M={M:.6f}")
    
    return best_f, best_score, time.monotonic() - t0

"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. LP-based fictitious play for global optimization
2. Exact pairwise coordinate descent with quadratic line search (vectorized)
3. Smoothed L-BFGS in log-space for fast local convergence

Key improvements over previous: exact line search in CD (vs random fractions),
LP fictitious play for global moves, and more aggressive multi-temperature L-BFGS.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Score = 2*n*max(conv) for a normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def softmax_from_z(z):
    """Convert log-domain to probability vector."""
    z = z - z.max()
    exp_z = np.exp(np.clip(z, -50, 50))
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-30))


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_new = np.interp(new_x, old_x, f_norm)
    f_new = np.maximum(f_new, 0)
    ss = f_new.sum()
    if ss > 1e-15:
        f_new = f_new / ss
    return f_new


def lp_fictitious_play(f0, time_limit, verbose=False):
    """LP-based fictitious play for global optimization."""
    n = len(f0)
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 0)
    f = f / f.sum()
    best_f = f.copy()
    best_score = compute_score(f)
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break

        conv_f = fast_convolve(f)
        M = conv_f.max()

        # Build constraint matrix: A[k,j] = f[k-j] if 0<=k-j<n
        A = np.zeros((2 * n - 1, n))
        for j in range(n):
            A[j:j + n, j] = f

        # Upper bound for g variables
        f_min_pos = f[f > 1e-12].min() if (f > 1e-12).any() else 1e-6
        ub = min(M / f_min_pos * 5, 50.0)

        c = -np.ones(n)
        b = np.full(2 * n - 1, M)
        bounds = [(0, ub)] * n

        try:
            res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 2.0})
            if res.success:
                g = np.maximum(res.x, 0)
            else:
                # Fallback: best response direction
                k_star = int(np.argmax(conv_f))
                g = np.zeros(n)
                js = max(0, k_star - n + 1)
                je = min(n - 1, k_star)
                if je >= js:
                    g[js:je + 1] = f[k_star - np.arange(js, je + 1)]
                if g.sum() < 1e-15:
                    g = np.ones(n) / n
        except Exception:
            k_star = int(np.argmax(conv_f))
            g = np.zeros(n)
            js = max(0, k_star - n + 1)
            je = min(n - 1, k_star)
            if je >= js:
                g[js:je + 1] = f[k_star - np.arange(js, je + 1)]
            if g.sum() < 1e-15:
                g = np.ones(n) / n

        gs = g.sum()
        if gs < 1e-15:
            g = np.ones(n) / n
        else:
            g = g / gs

        # Adaptive step size
        it += 1
        alpha = min(1.0 / np.sqrt(it + 1), 0.3)

        f_new = (1 - alpha) * f + alpha * g
        f_new = np.maximum(f_new, 0)
        fs = f_new.sum()
        if fs < 1e-15:
            f_new = best_f
        else:
            f_new = f_new / fs

        score = compute_score(f_new)
        if score < best_score:
            best_score = score
            best_f = f_new.copy()
        f = f_new

        if verbose and it % 500 == 0:
            print(f"    FP iter={it}, best_score={best_score:.8f}, cur={score:.8f}")

    return best_f, best_score, time.monotonic() - t0


def best_response_fp(f0, time_limit, top_k=5, verbose=False):
    """Simplified fictitious play using top-k peaks (no LP)."""
    n = len(f0)
    f = f0.copy().astype(np.float64)
    f = np.maximum(f, 0)
    f = f / f.sum()
    best_f = f.copy()
    best_score = compute_score(f)
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:

        conv_f = fast_convolve(f)
        M = conv_f.max()

        # Get top-k peak positions
        if top_k >= len(conv_f):
            peak_indices = np.arange(len(conv_f))
        else:
            peak_indices = np.argpartition(-conv_f, top_k)[:top_k]

        # Weighted average of response directions
        weights = conv_f[peak_indices]
        weights = np.maximum(weights, 1e-30)
        weights = weights / weights.sum()

        g = np.zeros(n)
        for w_k, k in zip(weights, peak_indices):
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je >= js:
                g[js:je + 1] += w_k * f[k - np.arange(js, je + 1)]

        gs = g.sum()
        if gs < 1e-15:
            g = np.ones(n) / n
        else:
            g = g / gs

        it += 1
        alpha = min(2.0 / (it + 2), 0.2)

        f_new = (1 - alpha) * f + alpha * g
        f_new = np.maximum(f_new, 0)
        fs = f_new.sum()
        if fs < 1e-15:
            f_new = best_f
        else:
            f_new = f_new / fs

        score = compute_score(f_new)
        if score < best_score:
            best_score = score
            best_f = f_new.copy()
        f = f_new

        if verbose and it % 1000 == 0:
            print(f"    BR-FP iter={it}, best={best_score:.8f}")

    return best_f, best_score, time.monotonic() - t0


def pairwise_cd_exact(a0, time_limit, n_pairs_per_iter=300, verbose=False):
    """
    Exact pairwise coordinate descent.
    For each pair (i,j), find optimal transfer t via ternary search
    on the (convex) function max_k (conv[k] + t*c_diff[k] + t^2*d_diff[k]).
    """
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 0)
    a = a / a.sum()
    conv = fast_convolve(a)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(7777)
    iters = 0
    size = 2 * n - 1

    while time.monotonic() - t0 < time_limit:
        iters += 1
        any_improved = False
        no_improve_count = 0

        for _ in range(n_pairs_per_iter):
            if time.monotonic() - t0 > time_limit:
                break

            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            if a[i] < 1e-14 and a[j] < 1e-14:
                continue

            # Compute c_diff and d_diff (vectorized)
            c_diff = np.zeros(size)
            d_diff = np.zeros(size)

            # +2*a[k-j] for k in [j, j+n-1]
            ks_j = np.arange(j, min(size, j + n))
            c_diff[ks_j] += 2.0 * a[ks_j - j]
            # -2*a[k-i] for k in [i, i+n-1]
            ks_i = np.arange(i, min(size, i + n))
            c_diff[ks_i] -= 2.0 * a[ks_i - i]
            # d_diff
            if 2 * j < size:
                d_diff[2 * j] += 1.0
            if 2 * i < size:
                d_diff[2 * i] -= 1.0

            t_lo = -a[j]
            t_hi = a[i]
            if t_lo >= t_hi - 1e-15:
                continue

            # Ternary search for optimal t
            lo = t_lo
            hi = t_hi
            # First, check if current max is already at the "flat" region
            # Do a coarse search first
            for _ in range(50):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                # f(t) = max(conv + t*c_diff + t^2*d_diff)
                f1 = np.max(conv + m1 * c_diff + m1 * m1 * d_diff)
                f2 = np.max(conv + m2 * c_diff + m2 * m2 * d_diff)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1

            t_opt = (lo + hi) / 2.0
            f_opt = np.max(conv + t_opt * c_diff + t_opt * t_opt * d_diff)

            current_max = conv.max()
            if f_opt < current_max - 1e-13:
                a[i] -= t_opt
                a[j] += t_opt
                a = np.maximum(a, 0)
                ss = a.sum()
                if ss < 1e-15:
                    a = best_a
                    conv = fast_convolve(a)
                    break
                a = a / ss
                conv = fast_convolve(a)
                new_score = compute_score(a)
                if new_score < best_score - 1e-14:
                    best_score = new_score
                    best_a = a.copy()
                any_improved = True

        if not any_improved:
            no_improve_count += 1
            if no_improve_count > 20:
                # Perturb to escape local minimum
                noise = rng.standard_normal(n) * 0.02
                a = best_a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                conv = fast_convolve(a)
                no_improve_count = 0

        if verbose and iters % 100 == 0:
            print(f"    CD iter={iters}, best={best_score:.8f}")

    return best_a, best_score, time.monotonic() - t0


def lbgfs_smooth(a0, time_limit, T, maxiter=100000):
    """L-BFGS with soft-max smoothed max in log-space."""
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    best_z = z0.copy()
    best_score = compute_score(a0)
    calls = [0]

    def func(z):
        calls[0] += 1
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = softmax_from_z(z)
        conv = fast_convolve(a)
        # Smoothed max
        m = conv.max()
        return (2.0 * n) * (m + T * np.log(np.exp((conv - m) / T).sum() / len(conv)))

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        a = softmax_from_z(z)
        conv = fast_convolve(a)
        # Softmax weights
        shifted = (conv - conv.max()) / T
        w = np.exp(shifted)
        w = w / w.sum()
        # d(conv[k])/da[i] = 2*a[k-i] for 0<=k-i<n
        # grad_a[i] = 2n * sum_k w[k] * 2*a[k-i] * I(0<=k-i<n)
        # This is 4n * cross_correlation(w, a)
        # cross_correlation: (w cross a)[i] = sum_k w[k] * a[k-i]
        # Use FFT
        fft_size = 1 << (2 * n - 1 - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        grad_a = 4.0 * n * irfft(np.conj(W) * A, fft_size)[:n]
        # Project to log-space
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)
        return grad_z

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 60, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
        score = compute_score(a_out)
        if score < best_score:
            best_score = score
            best_z = res.x.copy()
    except Exception as e:
        pass

    a_out = softmax_from_z(best_z)
    return a_out, best_score, time.monotonic() - t0


def peak_equalization_cd(a0, time_limit, n_peaks=10, verbose=False):

    Identifies which coordinates contribute most to each top peak,
    then optimizes those coordinates.
    """

            break

        conv = fast_convolve(a)
        M = conv.max()

        # Get top peaks
        k_peaks = min(n_peaks, size)
        peak_idx = np.argpartition(-conv, k_peaks)[:k_peaks]
        peak_vals = conv[peak_idx]

        # For each peak, find contributing coordinates
        target_coords = set()

            js = max(0, k - n + 1)

                contribs = a[idx] * a[k - idx]
                top_c = np.argsort(contribs)[::-1][:3]
                target_coords.update(idx[top_c].tolist())

        if not target_coords:

                    else:
                        lo = m1

                t_opt = (lo + hi) / 2.0
                f_opt = np.max(conv + t_opt * c_diff + t_opt * t_opt * d_diff)

                if f_opt < M - 1e-12:
                    a[i] -= t_opt
                    a[j] += t_opt
                    a = np.maximum(a, 0)
                    ss = a.sum()
                    if ss > 1e-15:

                        conv = fast_convolve(a)

            a = a / a.sum()

    return best_a, best_score, time.monotonic() - t0


def nesterov_smooth(a0, time_limit, T=500, lr_init=0.1, momentum=0.9):
    """Nesterov accelerated gradient in log-space."""
    n = len(a0)
    z = z_from_a(a0)
    v = np.zeros(n)
    best_z = z.copy()
    best_score = compute_score(a0)
    t0 = time.monotonic()
    lr = lr_init

    while time.monotonic() - t0 < time_limit:
        # Nesterov lookahead
        z_la = z - momentum * v
        a_la = softmax_from_z(z_la)
        conv_la = fast_convolve(a_la)

        # Gradient
        shifted = (conv_la - conv_la.max()) / T
        w = np.exp(shifted)
        w = w / w.sum()

        fft_size = 1 << (2 * n - 1 - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a_la, fft_size)
        grad_a = 4.0 * n * irfft(np.conj(W) * A, fft_size)[:n]

        S = np.dot(grad_a, a_la)
        grad_z = a_la * (grad_a - S)

        v = momentum * v + lr * grad_z
        z_new = z - v
        z_new = np.clip(z_new, -30, 30)

        a_new = softmax_from_z(z_new)
        score = compute_score(a_new)

        if score < best_score - 1e-14:
            best_score = score
            best_z = z_new.copy()

        z = z_new
        lr *= 0.99995

    a_out = softmax_from_z(best_z)
    return a_out, best_score, time.monotonic() - t0
