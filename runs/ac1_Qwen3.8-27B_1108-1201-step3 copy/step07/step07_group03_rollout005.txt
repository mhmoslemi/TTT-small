"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using differentiable LSE-smoothed
projected gradient with temperature annealing, diverse initialization,
and coordinate descent polishing.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_cross_corr(p, a):
    """Compute sum_k p[k]*a[k-j] for j=0..n-1 using FFT."""
    n = len(a)
    a_flip = a[::-1]
    # convolve p (length 2n-1) with a_flip (length n) -> length 3n-2
    # We need elements at indices n-1 to 2n-2
    # Use FFT
    fft_size = 1 << (len(p) + n - 1 - 1).bit_length()
    P = rfft(p, fft_size)
    Af = rfft(a_flip, fft_size)
    result = irfft(P * Af, fft_size)[:n + (2*n-1) - 1]  # length 3n-2
    return result[n-1 : n-1+n]

def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.max(np.nonzero(cond)[0])
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def compute_lse_gradient(a, T):
    """Compute gradient of LSE-smoothed max of auto-convolution."""
    n = len(a)
    b = fast_convolve(a)
    # Compute softmax with temperature T
    b_scaled = b / T
    b_max = b_scaled.max()
    exp_b = np.exp(b_scaled - b_max)
    p = exp_b / exp_b.sum()
    # Gradient: (2/T) * corr where corr[j] = sum_k p[k] * a[k-j]
    corr = fast_cross_corr(p, a)
    grad = (2.0 / T) * corr
    return grad, p, b

def lse_projected_grad_descent(a0, time_limit, T_init=0.05, T_final=0.005, 
                                 lr_init=0.01, momentum=0.9, verbose=False):
    """Projected gradient descent with LSE smoothing and temperature annealing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    v = np.zeros(n)  # momentum
    it = 0
    patience = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Anneal temperature
        progress = min(1.0, it / 500.0)
        T = T_init * (T_final / T_init) ** progress
        
        # Compute gradient
        grad, p, b = compute_lse_gradient(a, T)
        
        # Momentum step
        v = momentum * v + grad
        v_norm = np.linalg.norm(v)
        if v_norm > 1e-15:
            v = v / v_norm
        
        # Backtracking line search
        lr = lr_init
        score_current = 2.0 * n * b.max()
        for _ in range(20):
            a_trial = a - lr * v
            a_trial = simplex_project(a_trial)
            b_trial = fast_convolve(a_trial)
            score_trial = 2.0 * n * b_trial.max()
            if score_trial < score_current * (1 - 1e-4 * lr) - 1e-12:
                break
            lr *= 0.5
        else:
            lr = 0.01
            
        a = a_trial
        score = score_trial
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience > 200:
            # Restart with momentum reset
            v = np.zeros(n)
            lr_init = max(0.001, lr_init * 0.8)
            patience = 0
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-GD it={it} T={T:.4f} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score

import numpy as np

import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.max(np.nonzero(cond)[0])
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def compute_grad_lse(a, T):
    """
    Compute gradient of (2/T)*LSE(b/T) where b = conv(a,a).
    This is the gradient of the LSE-smoothed auto-convolution peak.
    Returns: grad (length n), b (length 2n-1)
    """
    n = len(a)
    b = fast_convolve(a)
    # Softmax with temperature
    b_scaled = b / T
    b_max = b_scaled.max()
    exp_b = np.exp(b_scaled - b_max)
    p = exp_b / exp_b.sum()
    
    # Gradient: (2/T) * corr where corr[j] = sum_k p[k] * a[k-j]
    # Compute via FFT: convolve p with reversed a
    a_flip = a[::-1]
    conv_size = len(p) + n - 1  # = 3n - 2
    fft_size = 1 << (conv_size - 1).bit_length()
    P = rfft(p, fft_size)
    Af = rfft(a_flip, fft_size)
    conv_result = irfft(P * Af, fft_size)[:conv_size]
    corr = conv_result[n-1 : n-1+n]
    
    grad = (2.0 / T) * corr
    return grad, b

def softmax_z(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

    a = a / a.sum()
    
    # Initialize z from a
    z = np.log(a)  # log to ensure positivity via softmax
    # Add small noise to break symmetry
    rng = np.random.default_rng(123)
    z += rng.standard_normal(n) * 0.01

    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    lr = lr_init
    it = 0
    no_improve = 0
    total_iters = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        total_iters += 1

        frac = min(1.0, it / 800.0)
        T = T_init * (T_end / T_init) ** frac

        grad_a, b = compute_grad_lse(a, T)

        dot_ga_a = np.dot(grad_a, a)

3. Nesterov-accelerated momentum with adaptive restarts

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.max(np.nonzero(cond)[0])
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def compute_lse_grad(a, T):
    """
    Compute gradient of (2/T)*LSE(b/T) where b = conv(a,a).
    Returns (grad of shape n, b of shape 2n-1, p of shape 2n-1).
    """
    n = len(a)
    b = fast_convolve(a)
    b_scaled = b / T
    b_max = b_scaled.max()
    exp_b = np.exp(b_scaled - b_max)
    p = exp_b / exp_b.sum()
    # grad[j] = (2/T) * sum_k p[k] * a[k-j] (cross-correlation)
    corr = _cross_corr_fft(p, a)
    grad = (2.0 / T) * corr
    return grad, b, p


def _cross_corr_fft(p, a):
    """Compute corr[j] = sum_k p[k]*a[k-j] for j=0..n-1 via FFT."""
    n = len(a)
    a_flip = a[::-1]
    conv_len = len(p) + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    P = rfft(p, fft_size)
    Af = rfft(a_flip, fft_size)
    conv_res = irfft(P * Af, fft_size)[:conv_len]
    return conv_res[n - 1:n - 1 + n]


def lse_projected_gd(a0, time_limit, T_init=0.04, T_end=0.004,
                     lr_init=0.008, use_momentum=True, verbose=False):
    """
    LSE-smoothed projected gradient on the simplex with temperature annealing.
    Uses Nesterov momentum and adaptive restarts.
    """

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)

    lr = lr_init
    it = 0
    no_improve = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Anneal temperature (smooth -> sharp)

        T = T_init * (T_end / T_init) ** frac

            v = beta_k * v + grad * lr
        else:
            v = grad

            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

            no_improve = 0

            lr = lr_init * np.sqrt(0.5)

def exponential_weights_opt(n, time_limit, eta=0.15, inner_steps=3,
                            verbose=False):
    """
    Exponential weights (dual) approach for the minimax problem.
    Alternates between:
      - Inner: minimize sum_k p_k (a*a)_k over simplex (projected gradient)
      - Outer: p <- softmax(eta * conv(a,a))
    """
    a = np.ones(n) / n
    p = np.ones(2 * n - 1) / (2 * n - 1)
    a_avg = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    lr_inner = 0.05

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Inner loop: minimize sum_k p_k * (a*a)_k
        for _ in range(inner_steps):
            # Gradient: 2 * cross_corr(p, a)
            corr = _cross_corr_fft(p, a)
            grad = 2.0 * corr
            a = simplex_project(a - lr_inner * grad)

        # Update dual variable p
        b = fast_convolve(a)
        b_max = b.max()
        p = np.exp(eta * (b - b_max))
        p = p / p.sum()

        #_running average_
        a_avg = (it * a_avg + a) / (it + 1)

        # Evaluate candidates
        for cand in [a, a_avg]:
            sc = 2.0 * n * fast_convolve(cand).max()
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = cand.copy()

        # Decay inner learning rate
        lr_inner *= 0.999

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      ExpW it={it} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score

    scale_factors_fine = [0.9, 0.95, 1.05, 1.1]

    for sweep in range(max_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        perm = rng.permutation(n)
        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            old = a[i]
            if old < 1e-12:
                continue
            factors = scale_factors_coarse if sweep < 50 else scale_factors_fine
            for f in factors:
                a[i] = old * f
                s = a.sum()
                if s < 1e-15:
                    a[i] = old
                    continue
                a_norm = a / s
                sc = 2.0 * n * fast_convolve(a_norm).max()
                a = a_norm
                if sc < best_score - 1e-13:
                    best_score = sc
                    best_a = a.copy()
                    old = a[i]
                    improved = True
                else:
                    a[i] = old
                    a = a / a.sum()
        if not improved and sweep > 20:
            break

    return best_a, best_score


def peak_shave(a0, time_limit):
    """Targeted shaving of the highest convolution peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for step in range(10000):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        # Top 3 peaks
        top_k = min(3, len(b))
        peaks = np.argpartition(b, -top_k)[-top_k:]
        peak_k = int(peaks[0])

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

        if contribs.max() < 1e-15:
            break

        # Try shaving top contributors
        top_contrib = np.argsort(contribs)[::-1][:8]
        improved = False

        for ci in top_contrib:
            i = int(idx[ci])
            j = peak_k - i
            if a[i] < 1e-14:
                continue
            for factor in [0.85, 0.7, 0.55]:
                a_trial = a.copy()
                a_trial[i] *= factor
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                sc = 2.0 * n * fast_convolve(a_trial).max()
                if sc < best_score - 1e-13:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try pair adjustments
            for ci in top_contrib[:4]:
                i = int(idx[ci])
                j = peak_k - i
                if i == j or a[j] < 1e-14:
                    continue
                for f1, f2 in [(0.8, 1.2), (0.7, 1.3), (0.9, 1.1)]:
                    a_trial = a.copy()
                    a_trial[i] *= f1
                    a_trial[j] *= f2
                    a_trial = np.maximum(a_trial, 0)
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial / s_t
                    sc = 2.0 * n * fast_convolve(a_trial).max()
                    if sc < best_score - 1e-13:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            break

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length, normalized to sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def init_fourier(n, rng):
    """Initialize with random Fourier cosine superposition."""
    n_harm = int(rng.integers(3, 20))
    amps = rng.random(n_harm) * 0.4
    phases = rng.uniform(0, 2 * np.pi, n_harm)
    freqs = rng.integers(1, n // 2, n_harm)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 1.0)
    for j in range(n_harm):
        a += amps[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_blocks(n, rng):
    """Initialize with random block structure."""
    n_blocks = int(rng.integers(4, 30))
    a = np.zeros(n)
    boundaries = np.sort(rng.choice(np.arange(1, n), size=n_blocks - 1, replace=False))
    starts = np.concatenate([[0], boundaries])
    ends = np.concatenate([boundaries, [n]])
    for b in range(n_blocks):
        a[starts[b]:ends[b]] = rng.random() + 0.05
    # Smooth
    ksize = min(5, n // 10 + 1)
    if ksize > 1:
        kernel = np.ones(ksize) / ksize
        a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_sink(n, rng, num_sinks=3):
    """Initialize with 'sink' structure: high values at sparse positions."""
    a = np.full(n, 0.3)
    positions = rng.choice(n, size=num_sinks, replace=False)
    for pos in positions:
        width = max(1, n // 20)
        lo = max(0, pos - width)
        hi = min(n, pos + width + 1)
        a[lo:hi] += rng.random() * 0.8
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate and track best sequence."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate():
    """
    Multi-phase search combining LSE-projected gradient, exponential weights,
    diverse initialization, and coordinate descent polishing.
    """
    deadline = time.monotonic() + 1790.0
    rng = np.random.default_rng(2024)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    # === Phase 1: Diverse initialization + short LSE-GD ===
    print("\n=== Phase 1: Diverse Init + Short LSE-GD ===")
    t1 = time.monotonic()
    t_phase1 = 200.0

    n_values = [512, 640, 768, 896, 1024, 1280]
    if parent is not None:
        np_parent = len(parent)
        if np_parent not in n_values:
            n_values.insert(0, np_parent)
        # Also try nearby lengths
        for factor in [0.85, 1.15]:
            alt_n = int(np_parent * factor)
            if 256 <= alt_n <= 2048 and alt_n not in n_values:
                n_values.append(alt_n)

    candidates = []
    for n in n_values:
        if time.monotonic() - t1 > t_phase1:
            break
        n_cands = 5
        for c in range(n_cands):
            if c == 0 and parent is not None:
                cand = interpolate_seq(parent, n)
            elif c == 1:
                cand = init_fourier(n, rng)
            elif c == 2:
                cand = init_blocks(n, rng)
            elif c == 3:
                base = interpolate_seq(parent, n) if parent is not None else np.ones(n) / n
                cand = np.maximum(base + rng.standard_normal(n) * 0.08, 0)
                cand = cand / cand.sum()
            else:
                cand = init_sink(n, rng)

            # Short LSE-GD
            t_quick = 15.0
            a_opt, s_opt = lse_projected_gd(cand, t_quick, T_init=0.05, T_end=0.01,
                                            lr_init=0.01, use_momentum=True)
            candidates.append((s_opt, a_opt, n, c))

    candidates.sort(key=lambda x: x[0])
    print(f"  {len(candidates)} candidates, best internal: {candidates[0][0]:.6f}")

    for s_c, a_c, n_c, c_c in candidates[:6]:
        best_seq, best_score = check_and_update(a_c, best_seq, best_score, f"p1_n{n_c}t{c_c}")

    # === Phase 2: Extended LSE-GD on top candidates ===
    print(f"\n=== Phase 2: Extended LSE-GD (best={best_score:.6f}) ===")
    t2 = time.monotonic()
    t_phase2 = 700.0
    time_per = t_phase2 / 4

    top_cands = candidates[:4]
    # Also include current best
    if best_seq is not None:
        n_best = len(best_seq)
        already = False
        for s_c, a_c, n_c, c_c in top_cands:
            if n_c == n_best:
                already = True
                break
        if not already:
            top_cands.append((best_score, best_seq, n_best, -1))

    for idx, (s_c, a_c, n_c, c_c) in enumerate(top_cands):
        rem = deadline - time.monotonic() - 400
        if rem < 30:
            break
        t_alloc = min(time_per, rem * 0.8)
        if t_alloc < 20:
            continue
        print(f"  LSE-GD n={n_c}, budget={t_alloc:.0f}s")

        # Main run with annealing
        a_opt, s_opt = lse_projected_gd(a_c, t_alloc * 0.6,
                                         T_init=0.03, T_end=0.003,
                                         lr_init=0.008, use_momentum=True, verbose=True)
        best_seq, best_score = check_and_update(a_opt, best_seq, best_score, f"p2_LSE_n{n_c}")

        # Exponential weights on the result
        rem = deadline - time.monotonic() - 300
        if rem > 20:
            t_ew = min(t_alloc * 0.3, rem * 0.5)
            a_ew, s_ew = exponential_weights_opt(n_c, t_ew, eta=0.2, inner_steps=2)
            best_seq, best_score = check_and_update(a_ew, best_seq, best_score, f"p2_EW_n{n_c}")

        # Coordinate descent polish
        rem = deadline - time.monotonic() - 250
        if rem > 15 and best_seq is not None:
            t_cd = min(60.0, rem * 0.4)
            a_cd, s_cd = coord_descent_polish(best_seq, t_cd, max_sweeps=200)
            best_seq, best_score = check_and_update(a_cd, best_seq, best_score, f"p2_CD_n{len(best_seq)}")

    # === Phase 3: Intensive polishing on best ===
    print(f"\n=== Phase 3: Intensive Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        # Peak shaving
        rem = deadline - time.monotonic() - 200
        if rem > 20:
            t_ps = min(80.0, rem * 0.4)
            print(f"  PeakShave budget={t_ps:.0f}s")
            a_ps, s_ps = peak_shave(best_seq, t_ps)
            best_seq, best_score = check_and_update(a_ps, best_seq, best_score, "p3_peakshave")

        # LSE-GD with very small T (near-hard max)
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t_lse = min(80.0, rem * 0.5)
            print(f"  LSE-GD fine T, budget={t_lse:.0f}s")
            a_fine, s_fine = lse_projected_gd(best_seq, t_lse,
                                               T_init=0.005, T_end=0.001,
                                               lr_init=0.003, use_momentum=True)
            best_seq, best_score = check_and_update(a_fine, best_seq, best_score, "p3_fineT")

        # More coordinate descent
        rem = deadline - time.monotonic() - 100
        if rem > 15:
            t_cd2 = min(60.0, rem * 0.5)
            a_cd2, s_cd2 = coord_descent_polish(best_seq, t_cd2, max_sweeps=400)
            best_seq, best_score = check_and_update(a_cd2, best_seq, best_score, "p3_cd2")

    # === Phase 4: Try different lengths from best ===
    print(f"\n=== Phase 4: Alt Lengths (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        alt_lengths = []
        for f in [0.8, 0.9, 1.1, 1.25, 1.5]:
            alt_n = int(n_best * f)
            if 256 <= alt_n <= 2560 and alt_n != n_best and alt_n not in alt_lengths:
                alt_lengths.append(alt_n)

        for alt_n in alt_lengths[:3]:
            rem = deadline - time.monotonic() - 80
            if rem < 30:
                break
            t_alt = min(80.0, rem * 0.5)
            init_alt = interpolate_seq(best_seq, alt_n)
            a_alt, s_alt = lse_projected_gd(init_alt, t_alt,
                                            T_init=0.03, T_end=0.004,
                                            lr_init=0.006, use_momentum=True)
            best_seq, best_score = check_and_update(a_alt, best_seq, best_score, f"p4_altn{alt_n}")
            if best_seq is not None and len(best_seq) == alt_n:
                t_cd3 = min(30.0, max(0, deadline - time.monotonic() - 60) * 0.5)
                if t_cd3 > 5:
                    a_cd3, s_cd3 = coord_descent_polish(best_seq, t_cd3, max_sweeps=150)
                    best_seq, best_score = check_and_update(a_cd3, best_seq, best_score, f"p4_cd_n{alt_n}")

    # === Phase 5: Final polish ===
    print(f"\n=== Phase 5: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            t_fin = min(40.0, rem * 0.5)
            a_fin, s_fin = peak_shave(best_seq, t_fin)
            best_seq, best_score = check_and_update(a_fin, best_seq, best_score, "final_ps")

        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_fin2, s_fin2 = coord_descent_polish(best_seq, min(20.0, rem), max_sweeps=100)
            best_seq, best_score = check_and_update(a_fin2, best_seq, best_score, "final_cd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
