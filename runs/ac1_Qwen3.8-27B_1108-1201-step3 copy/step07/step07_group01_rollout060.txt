Actually, one more important consideration: **the LP-based method as described might be slow to converge because it uses a fixed direction at each step (the LP solution).** A more efficient approach might be to:

1. Solve the LP to get g
2. Instead of a small step f <- (1-t)f + tg, do a **full line search**: find the best alpha in [0,1] such that f_alpha = (1-alpha)f + alpha*g minimizes max(f_alpha * f_alpha).

This is a 1D optimization along the line from f to g. It's non-convex, but we can use golden section search or just try a few values.

Actually, the maximum of (f_alpha * f_alpha) as a function of alpha is:
(f_alpha * f_alpha)_k = ((1-alpha)f + alpha*g) * ((1-alpha)f + alpha*g)
= (1-alpha)^2 (f*f)_k + 2*alpha*(1-alpha) (f*g)_k + alpha^2 (g*g)_k

The maximum over k of this expression is a piecewise quadratic function of alpha. We can find its minimum by:
1. For each k, find the alpha that minimizes the k-th component (a simple quadratic minimization)
2. The minimum of the max is at one of these critical points or at alpha=0 or alpha=1.

This is O(n^2) per step (n values of k, each giving a critical point, then evaluating the max at each).

"""
Multi-phase optimization using LP-based multiplicative weights iteration,
with adaptive step sizes, multiple starting points, multiple sequence lengths,
and coordinate descent / L-BFGS polishing.
Primary method: LP-based direction finding (solves max sum(g) s.t. (f*g)_k <= M, g >= 0)
at each step, providing the optimal "smoothing" direction.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    n_a, n_b = len(a), len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def score_from_a(a):
    """Compute 2*n*max(conv(a,a))."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())

def normalize_sum1(a):
    """Normalize to sum=1."""
    s = a.sum()
    if s < 1e-300:
        return np.ones(len(a)) / len(a)
    return a / s

def build_constraint_matrix(f):
    """Build the (2n-1) x n matrix C where (f*g)_k = (C@g)_k."""
    n = len(f)
    C = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        i_start = max(0, k - n + 1)
        i_end = min(n-1, k)
        idx = np.arange(i_start, i_end+1)
        C[k, idx] = f[k - idx]
    return C

def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0.
    Returns optimal g or None if infeasible/failed."""
    n = len(f)
    C = build_constraint_matrix(f)
    c = -np.ones(n)  # minimize -sum(g)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 10.0})
        if res.status == 0:
            return np.maximum(res.x, 0)
    except Exception as e:
        pass
    
    # Fallback to ECOS via cvxpy
    try:
        import cvxpy as cp
        g = cp.Variable(n)
        constraints = [C @ g <= M, g >= 0]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        prob.solve(solver=cp.ECOS, verbose=False)
        if g.value is not None:
            return np.maximum(g.value, 0)
    except Exception:
        pass
    
    return None

def lp_iteration(f, max_iters=500, time_limit=300.0, t_start=0.1, verbose=True, label=""):
    """LP-based multiplicative weights iteration.
    f: sequence with sum = sqrt(2n)
    Minimizes max(f*f).
    """
    n = len(f)
    f = f.copy().astype(np.float64)
    t0 = time.monotonic()
    sqrt_2n = np.sqrt(2.0 * n)
    
    # Normalize
    s = f.sum()
    if s < 1e-15:
        f = np.full(n, sqrt_2n / n)
    else:
        f = f * sqrt_2n / s
    
    b = fast_convolve(f)
    M = b.max()
    best_f = f.copy()
    best_M = M
    
    t = t_start
    stagnation = 0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if it % 50 == 0 and verbose:
            elapsed = time.monotonic() - t0
            print(f"    {label} LP iter={it} ({elapsed:.1f}s): M={M:.7f} best_M={best_M:.7f} t={t:.4f}")
        
        # Solve LP
        g = solve_lp_direction(f, M)
        if g is None:
            if verbose:
                print(f"    {label}: LP failed at iter {it}")
            break
        
        S = g.sum()
        if S < 1e-15:
            break
        
        # Normalize g to have sum = sqrt(2n)
        g_norm = g * sqrt_2n / S
        
        # Line search: try different step sizes
        improved = False
        for alpha in [t, t*0.5, t*0.25, 0.01]:
            f_trial = (1-alpha)*f + alpha*g_norm
            b_trial = fast_convolve(f_trial)
            M_trial = b_trial.max()
            if M_trial < M - 1e-14:
                f = f_trial
                M = M_trial
                improved = True
                break
        
        if improved:
            stagnation = 0
            # Occasionally increase step size
            if it % 100 == 0 and t < t_start:
                t = min(t * 1.5, t_start)
        else:
            stagnation += 1
            t *= 0.7
            if t < 0.001:
                if verbose:
                    print(f"    {label}: Stuck at iter {it}, M={M:.7f}")
                break
        
        # Update best
        if M < best_M:
            best_M = M
            best_f = f.copy()
    
    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    {label} LP done ({elapsed:.1f}s): M={M:.7f} best_M={best_M:.7f} its={it+1}")
    
    return best_f, best_M

def multiplicative_weights_smooth(f_init, time_limit=100.0, T_start=0.5, T_end=0.01,
                                   eta_start=0.05, eta_end=0.005, label=""):
    """Mirror descent with temperature annealing (faster per-step than LP)."""
    n = len(f_init)
    f = np.maximum(f_init.copy(), 1e-15).astype(np.float64)
    s = f.sum()
    f = f / s  # normalize to sum=1
    t0 = time.monotonic()
    best_f = f.copy()
    best_score = 2.0 * n * fast_convolve(f).max()
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        progress = min(1.0, it / 50000)
        T = T_start * (T_end / T_start) ** progress
        eta = eta_start * (eta_end / eta_start) ** progress
        
        b = fast_convolve(f)
        cur_score = 2.0 * n * b.max()
        
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_f = f.copy()
        
        # Softmax weights
        scaled = b / T
        m = scaled.max()
        exp_s = np.exp(scaled - m)
        w = exp_s / exp_s.sum()
        
        # Gradient
        # grad_a[i] = 2*n * sum_k w[k] * a[k-i] (valid range)
        fft_size = 1 << (3*n - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(f, fft_size)
        correl = irfft(np.conj(W) * A, fft_size)
        grad = 2.0 * n * correl[:n]
        
        # Mirror descent update
        f = f * np.exp(-eta * np.clip(grad, -5, 5))
        f = np.maximum(f, 1e-300)
        f = f / f.sum()
    
    return best_f, best_score

def coordinate_descent_polish(a_init, time_limit=30.0):
    """Fast coordinate descent polishing."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    factors = [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]
    
    no_improve = 0
    while time.monotonic() - t0 < time_limit and no_improve < 50:
        no_improve = 0
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            for fct in factors:
                a_trial = a.copy()
                old = a_trial[i]
                a_trial[i] = max(0.0, old * fct)
                diff = a_trial.sum() - 1.0
                if abs(diff) < 1e-15:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                os = others.sum()
                if os < 1e-15:
                    continue
                a_trial[mask] = others - diff * others / os
                if a_trial.min() < -1e-12:
                    continue
                a_trial = np.maximum(a_trial, 0)
                sc = score_from_a(a_trial)
                if sc < best_score - 1e-13:
                    best_score = sc
                    a = a_trial.copy()
                    best_a = a.copy()
                    improved = True
                    break
        if not improved:
            no_improve += 1
    
    return best_a

def lbgfs_polish(a_init, time_limit=60.0, T=5):
    """L-BFGS in log-space with small temperature."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).astype(np.float64)
    a = a / a.sum()
    z0 = np.log(a)
    z0 -= z0.mean()
    t0 = time.monotonic()
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = np.exp(z - z.max())
        a = a / a.sum()
        b = fast_convolve(a)
        scaled = b / T
        m = scaled.max()
        lse = m + np.log(np.exp(scaled - m).sum())
        return 2.0 * n * T * lse
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros(n)
        a = np.exp(z - z.max())
        a = a / a.sum()
        b = fast_convolve(a)
        scaled = b / T
        m = scaled.max()
        w = np.exp(scaled - m)
        w = w / w.sum()
        fft_size = 1 << (3*n - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        correl = irfft(np.conj(W) * A, fft_size)[:n]
        grad_a = 2.0 * n * correl
        mean_grad = np.dot(a, grad_a)
        return a * (grad_a - mean_grad)
    
    try:
        res = minimize(func, z0, jac=jac, method='L-BFGS-B',
                       options={'maxiter': 3000, 'ftol': 1e-16, 'gtol': 1e-12, 'disp': False})
        z_opt = res.x
        a_out = np.exp(z_opt - z_opt.max())
        a_out = a_out / a_out.sum()
        return a_out
    except Exception:
        return a_init.copy()

def interpolate_to_length(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_new = np.interp(new_x, old_x, f_norm)
    f_new = np.maximum(f_new, 0)
    ss = f_new.sum()
    if ss > 1e-15:
        f_new = f_new / ss
    return f_new

def check_and_update(a, label, best_seq, best_score):

        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.7f} (n={len(a_norm)})")
    return best_seq, best_score

"""
Multi-phase optimization using LP-based multiplicative weights (exact direction
finding via linear programming), combined with mirror descent, L-BFGS polishing,
and coordinate descent. The LP at each step finds the optimal "smoothing"
direction that keeps all cross-convolution values bounded, providing a
principled approach to peak reduction that outperforms single-peak subgradient

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    n_a, n_b = len(a), len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def score_from_a(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def build_constraint_matrix(f):
    """Build (2n-1) x n matrix C where (C@g)[k] = (f*g)[k]."""
    n = len(f)
    C = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1)
        C[k, idx] = f[k - idx]
    return C


def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0."""
    n = len(f)
    C = build_constraint_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5.0})
        if res.status == 0:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    try:
        import cvxpy as cp
        g = cp.Variable(n)
        constraints = [C @ g <= M, g >= 0]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        prob.solve(solver=cp.ECOS, verbose=False)
        if g.value is not None:
            return np.maximum(g.value, 0)
    except Exception:
        pass
    return None


def lp_iteration(f, max_iters=2000, time_limit=300.0, t_start=0.1,
                 verbose=True, label=""):
    """LP-based multiplicative weights iteration to minimize max(f*f).
    f should have sum = sqrt(2n)."""
    n = len(f)
    f = f.copy().astype(np.float64)
    t0 = time.monotonic()
    sqrt_2n = np.sqrt(2.0 * n)

    s = f.sum()
    if s < 1e-15:
        f = np.full(n, np.sqrt(2.0 / n))
    else:
        f = f * sqrt_2n / s

    b = fast_convolve(f)
    M = float(b.max())
    best_f = f.copy()
    best_M = M
    t = t_start
    stagnation = 0
    actual_iters = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        actual_iters = it + 1
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} LP it={it} ({elapsed:.1f}s): M={M:.7f} best={best_M:.7f} t={t:.4f}")

        g = solve_lp_direction(f, M)
        if g is None:
            if verbose:
                print(f"    {label}: LP failed at it={it}")
            break

        S = g.sum()
        if S < 1e-15:
            break

        g_norm = g * sqrt_2n / S

        improved = False
        for alpha in [t, t * 0.5, t * 0.25, 0.005]:
            f_trial = (1 - alpha) * f + alpha * g_norm
            b_trial = fast_convolve(f_trial)
            M_trial = float(b_trial.max())
            if M_trial < M - 1e-13:
                f = f_trial
                M = M_trial
                improved = True
                break

        if improved:
            stagnation = 0
            if stagnation == 0 and it % 50 == 0 and t < t_start * 0.8:
                t = min(t * 1.3, t_start)
        else:
            stagnation += 1
            t *= 0.6
            if t < 0.001:
                if verbose:
                    print(f"    {label}: Converged/stuck at it={it}, M={M:.7f}")
                break

        if M < best_M - 1e-14:
            best_M = M
            best_f = f.copy()

    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    {label} LP done ({elapsed:.1f}s): M={M:.7f} best={best_M:.7f} its={actual_iters}")

    return best_f, best_M


def multiplicative_weights_smooth(f_init, time_limit=100.0, T_start=0.5,
                                   T_end=0.02, eta_start=0.04, eta_end=0.004,
                                   label=""):
    """Mirror descent with temperature annealing (fast per-step)."""
    n = len(f_init)
    f = np.maximum(f_init.copy(), 1e-15).astype(np.float64)
    f = f / f.sum()
    t0 = time.monotonic()
    best_f = f.copy()
    best_score = 2.0 * n * fast_convolve(f).max()
    fft_size = 1 << (3 * n - 1).bit_length()

    for it in range(200000):
        if time.monotonic() - t0 > time_limit:
            break
        progress = min(1.0, it / 80000)
        T = T_start * (T_end / T_start) ** progress
        eta = eta_start * (eta_end / eta_start) ** progress

        b = fast_convolve(f)
        cur_score = 2.0 * n * b.max()
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_f = f.copy()

        scaled = b / T
        m = scaled.max()
        exp_s = np.exp(scaled - m)
        w = exp_s / exp_s.sum()

        W = rfft(w, fft_size)
        A = rfft(f, fft_size)
        correl = irfft(np.conj(W) * A, fft_size)
        grad = 2.0 * n * correl[:n]

        f = f * np.exp(-eta * np.clip(grad, -5.0, 5.0))
        f = np.maximum(f, 1e-300)
        s = f.sum()
        if s < 1e-15:
            f = best_f.copy()
        else:
            f = f / s

    return best_f, best_score


def coordinate_descent_polish(a_init, time_limit=30.0):
    """Fast coordinate descent with proportional redistribution."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    factors = [0.7, 0.8, 0.9, 1.1, 1.2, 1.4]
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 60:
        no_improve = 0
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            for fct in factors:
                a_trial = a.copy()
                old = a_trial[i]
                a_trial[i] = max(0.0, old * fct)
                diff = a_trial.sum() - 1.0
                if abs(diff) < 1e-15:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                os = others.sum()
                if os < 1e-15:
                    continue
                a_trial[mask] = others - diff * others / os
                if a_trial.min() < -1e-12:
                    continue
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = score_from_a(a_trial)
                if sc < best_score - 1e-13:
                    best_score = sc
                    a = a_trial.copy()
                    best_a = a.copy()
                    improved = True
                    break
        if not improved:
            no_improve += 1

    return best_a

def lbgfs_polish(a_init, time_limit=60.0, T=5):
    """L-BFGS in log-space with soft-max smoothing."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).astype(np.float64)
    a = a / a.sum()
    z0 = np.log(a)
    z0 -= z0.mean()
    t0 = time.monotonic()
    fft_size = 1 << (3 * n - 1).bit_length()

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = np.exp(z - z.max())
        a = a / a.sum()
        b = fast_convolve(a)
        scaled = b / T
        m = scaled.max()
        lse = m + np.log(np.exp(scaled - m).sum())
        return 2.0 * n * T * lse

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros(n)
        a = np.exp(z - z.max())
        a = a / a.sum()
        b = fast_convolve(a)
        scaled = b / T
        m = scaled.max()
        w = np.exp(scaled - m)
        w = w / w.sum()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        correl = irfft(np.conj(W) * A, fft_size)[:n]
        grad_a = 2.0 * n * correl
        mean_grad = np.dot(a, grad_a)
        return a * (grad_a - mean_grad)

    try:
        res = minimize(func, z0, jac=jac, method='L-BFGS-B',
                       options={'maxiter': 5000, 'ftol': 1e-16,
                                'gtol': 1e-12, 'disp': False})
        z_opt = res.x
        a_out = np.exp(z_opt - z_opt.max())
        a_out = a_out / a_out.sum()
        return a_out
    except Exception:
        return a_init.copy()


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_new = np.interp(new_x, old_x, f_norm)
    f_new = np.maximum(f_new, 0)
    ss = f_new.sum()
    if ss > 1e-15:
        f_new = f_new / ss
    return f_new
