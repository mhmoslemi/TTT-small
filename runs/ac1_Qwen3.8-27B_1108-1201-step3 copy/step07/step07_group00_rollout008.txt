def lp_solve_reduced(f, n, top_k, eps=1e-8):
    """
    Solve: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for top_k peaks.
    Returns (g, M) or (None, M) on failure.
    """
    b = fast_convolve(f)
    M = b.max()
    if M < 1e-15:
        return None, M

    if top_k < len(b):
        top_idx = np.argpartition(b, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(b))
    top_idx = np.sort(top_idx)
    K = len(top_idx)

    A_mat = np.zeros((K, n))
    for row_i in range(K):
        k = top_idx[row_i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A_mat[row_i, lo:hi + 1] = f[lo:hi + 1]

    g_var = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), [g_var >= 0, A_mat @ g_var <= M])
    try:
        prob.solve(solver=cp.SCS, max_iters=300000, eps=eps)
    except Exception:
        return None, M

    if g_var.value is None:
        return None, M
    g_val = np.maximum(g_var.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None, M
    return g_val, M

        g, M = lp_solve_reduced(f, n, top_k)

        if g is None:

            if g is None:

        S_g = g.sum()
        if S_g <= 1.0 + 1e-5:
            stuck += 1
            if stuck > 15:
                break
            top_k = min(top_k * 2, top_k_max)
            continue

        h = g / S_g
        best_t = 0.05
        best_t_obj = fast_convolve(f).max()

        for t in [0.02, 0.04, 0.06, 0.1, 0.15, 0.2, 0.3, 0.5]:
            f_trial = (1.0 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t

        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s

        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 60:
                break

        if it % 30 == 0:
            print(f"    {label} iter={it} obj={best_obj:.10f} "
                  f"score={2.0*n*best_obj:.6f} top_k={top_k} t={time.monotonic()-t0:.0f}s")

    return best_f, best_obj


def coordinate_polish(f_init, time_limit):
    """Coordinate descent on individual elements with multiplicative steps."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    multipliers = [0.5, 0.7, 0.85, 1.15, 1.3, 2.0, 0.3]

    improved = True
    passes = 0
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        passes += 1
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-15:
                continue
            best_j_val = f[j]
            best_j_obj = best_obj
            for mult in multipliers:
                new_val = f[j] * mult
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f /= s
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    improved = True
    return f, best_obj


def smoothed_gradient_phase(a_init, time_limit, tau_init=0.5, label="sg_smooth"):
    """
    Gradient descent on the log-sum-exp smoothed objective using
    projected gradient with momentum. This provides an alternative
    to the multiplicative approach.
    """
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    a /= a.sum()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    L = 1 << (3 * n - 1).bit_length()
    idx = (2 * n - 2 - np.arange(n)).astype(np.intp)

    tau = tau_init
    lr = 0.01
    momentum = 0.85
    v = np.zeros(n)
    it = 0
    no_improve = 0
    rng = np.random.default_rng(2024)

    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(a)

        # Softmax weights (same as multiplicative but used differently)
        c_shift = c - c.max()
        c_scaled = np.clip(c_shift * tau, -50, 50)
        w = np.exp(c_scaled)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(len(c))
            w[c.argmax()] = 1.0
        else:
            w /= w_sum

        g = compute_gradient_fft(a, w, L, idx)
        g = g / g.sum()  # Normalize gradient

        # Momentum + projected gradient step
        v = momentum * v - lr * g
        a_new = project_simplex(a + v)

        # Check improvement
        obj_new = fast_convolve(a_new).max()
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
            lr = min(lr * 1.02, 0.05)
        else:
            no_improve += 1
            if no_improve > 30:
                lr *= 0.95
                no_improve = 0
            if no_improve > 100:
                a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
                v = np.zeros(n)
                no_improve = 0

        if it % 500 == 0:
            print(f"    {label} iter={it} obj={best_obj:.10f} "
                  f"score={2.0*n*best_obj:.6f} t={time.monotonic()-t0:.0f}s")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main optimization combining multiplicative updates, LP, and coordinate descent.
    Starts from parent if available, explores multiple strategies and lengths.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(1024) / 1024
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"No parent, using uniform n=1024, score={p_score:.6f}")

    best_seq = parent.copy()
    best_score = p_score

    def try_update(seq, label):
        nonlocal best_seq, best_score
        s = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_seq = seq.copy()
            print(f"  *** NEW BEST [{label}]: score={best_score:.6f} (n={len(seq)})")
            return True
        return False

    # === Phase 1: Multiplicative on parent (fast, 35% time) ===
    print(f"\n=== Phase 1: Multiplicative main (n={len(parent)}, score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.35, 600.0)
    if t1 > 60:
        print(f"  Budget: {t1:.0f}s")
        f1, obj1 = multiplicative_phase(parent, time_limit=t1,
                                         eta_init=0.06, temp_init=3e-4,
                                         temp_decay=1.1, eta_decay=0.9997,
                                         label="mult_main")
        try_update(f1, "mult_main")

    # === Phase 2: LP refinement (20% time) ===
    print(f"\n=== Phase 2: LP refinement (n={len(best_seq)}, score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.20, 300.0)
    if t2 > 40:
        n_b = len(best_seq)
        print(f"  Budget: {t2:.0f}s, n={n_b}")
        f2, obj2 = run_lp_phase(best_seq, time_limit=t2, top_k_init=50,
                                 top_k_max=200, label="lp_refine")
        try_update(f2, "lp_refine")

    # === Phase 3: Multiplicative with different params (15% time) ===
    print(f"\n=== Phase 3: Multipulative alt (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.15, 200.0)
    if t3 > 30:
        print(f"  Budget: {t3:.0f}s, higher eta, lower temp")
        f3, obj3 = multiplicative_phase(best_seq, time_limit=t3,
                                         eta_init=0.15, temp_init=1e-4,
                                         temp_decay=1.15, eta_decay=0.999,
                                         label="mult_alt")
        try_update(f3, "mult_alt")

    # === Phase 4: Smoothed gradient (10% time) ===
    print(f"\n=== Phase 4: Smoothed gradient (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.10, 120.0)
    if t4 > 20:
        print(f"  Budget: {t4:.0f}s")
        f4, obj4 = smoothed_gradient_phase(best_seq, time_limit=t4,
                                            tau_init=2.0, label="sg_smooth")
        try_update(f4, "sg_smooth")

    # === Phase 5: Multi-length exploration (15% time) ===
    print(f"\n=== Phase 5: Multi-length (score={best_score:.6f}) ===")
    n_base = len(best_seq)
    target_lengths = []
    for factor in [0.7, 0.85, 1.15, 1.3, 1.5, 0.5]:
        tl = int(n_base * factor)
        tl = max(256, min(4000, tl))
        if tl != n_base and tl not in target_lengths:
            target_lengths.append(tl)
    target_lengths.sort(key=lambda x: abs(x - n_base))

    for target_n in target_lengths[:3]:
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t5 = min(rem * 0.25, 150.0)
        if t5 < 20:
            break
        # Interpolate best sequence to new length
        old_x = np.linspace(0.0, 1.0, n_base)
        new_x = np.linspace(0.0, 1.0, target_n)
        f_interp = np.interp(new_x, old_x, best_seq)
        f_interp = np.maximum(f_interp, 0.0)
        s = f_interp.sum()
        if s > 1e-10:
            f_interp /= s
        else:
            f_interp = np.ones(target_n) / target_n

        print(f"  Multiplicative n={target_n}, budget={t5:.0f}s")
        f5, obj5 = multiplicative_phase(f_interp, time_limit=t5,
                                         eta_init=0.05, temp_init=3e-4,
                                         temp_decay=1.1, eta_decay=0.9997,
                                         label=f"mult_n{target_n}")
        try_update(f5, f"n{target_n}")

    # === Phase 6: Coordinate descent + final multiplicative (15% time) ===
    print(f"\n=== Phase 6: CD + final (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()

    # Coordinate descent
    t6a = min(rem * 0.3, 80.0)
    if t6a > 10:
        print(f"  CD budget: {t6a:.0f}s")
        f6a, obj6a = coordinate_polish(best_seq, time_limit=t6a)
        try_update(f6a, "cd")

    # Final multiplicative burst with very low temperature
    rem = deadline - time.monotonic()
    t6b = min(rem * 0.5, 150.0)
    if t6b > 15:
        print(f"  Final mult budget: {t6b:.0f}s")
        f6b, obj6b = multiplicative_phase(best_seq, time_limit=t6b,
                                           eta_init=0.03, temp_init=5e-5,
                                           temp_decay=1.2, eta_decay=0.9995,
                                           label="mult_final")
        try_update(f6b, "mult_final")

    # Final LP if time remains
    rem = deadline - time.monotonic()
    t6c = min(rem * 0.6, 100.0)
    if t6c > 20:
        print(f"  Final LP budget: {t6c:.0f}s")
        f6c, obj6c = run_lp_phase(best_seq, time_limit=t6c, top_k_init=100,
                                   top_k_max=300, label="lp_final")
        try_update(f6c, "lp_final")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'='*55}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*55}")

    return [float(x) for x in best_seq]
