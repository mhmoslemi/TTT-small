The problem is: minimize $\max_k c_k$ subject to $\sum a_j = 1$, $a_j \geq 0$, where $c_k = \sum_j a_j a_{k-j}$.

This is a QCQP (quadratic constraint quadratic program) which is non-convex. But we can relax it:
- In the SDP relaxation, we introduce $X_{ij} = a_i a_j$ and relax $X = a a^T$ to $X \succeq 0$, $\text{diag}(X) = a$ (or just $X \succeq 0$).
- The objective becomes $\max_k \sum_j X_{j, k-j}$ (linear in X!)
- The constraint $\sum a_j = 1$ becomes $\sum_j X_{jj} = 1$... wait no, $\text{diag}(X) = a$ means $X_{jj} = a_j^2$, not $a_j$.

Actually, the SDP relaxation is:
- Variables: $a_j \geq 0$, $X \succeq 0$
- Constraints: $X_{ij} = a_i a_j$ ... no, this is the non-convex part.
- Relax: $X \succeq aa^T$ (which is equivalent to $X_{ii} = a_i^2$ and $X \succeq 0$... actually $X \succeq aa^T$ is a linear matrix inequality in (a, X) if we write it as $\begin{bmatrix} X & a \\ a^T & 1 \end{bmatrix} \succeq 0$)

Wait, $X \succeq aa^T$ is equivalent to $\begin{bmatrix} X & a \\ a^T & 1 \end{bmatrix} \succeq 0$ (by Schur complement, when the bottom-right is 1 > 0).

So the SDP relaxation is:
- Minimize $M$
- Subject to: $\sum_j X_{j, k-j} \leq M$ for all k (where indices are valid)
- $\sum_j a_j = 1$
- $a_j \geq 0$
- $\begin{bmatrix} X & a \\ a^T & 1 \end{bmatrix} \succeq 0$

This is a linear objective with linear constraints and one LMI constraint - it's an SDP! And it's a relaxation (the feasible set is larger than the original).

The SDP size: $a$ has n variables, $X$ is n×n symmetric (n(n+1)/2 variables), plus M. Total: about $n^2/2 + n + 1$ variables. The LMI is (n+1)×(n+1).

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Temperature-annealed soft-peak subgradient with Nesterov acceleration
2. Coordinated pair updates targeting peak contributors
3. Symmetry-regularized search (average with reverse)
4. Backtracking line search for adaptive step size
5. Multi-length exploration with interpolation
6. Iterative refinement with smart restarts
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Compute score for a normalized (sum=1) non-negative sequence."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def soft_subgrad(a, temperature=0.01):
    """
    Compute soft subgradient of max(conv(a,a)) using softmax weighting.
    Temperature controls the spread: low temp = hard max, high temp = average.
    """
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    shifted = (b - max_b) / temperature
    shifted = np.maximum(shifted, -50.0)  # avoid underflow
    weights = np.exp(shifted)
    wsum = weights.sum()
    if wsum < 1e-30:
        weights = np.zeros(len(b))
        weights[np.argmax(b)] = 1.0
    else:
        weights /= wsum

    sg = np.zeros(n)
    # Only process significant weights for efficiency
    sig_threshold = 1e-8
    k_indices = np.where(weights > sig_threshold)[0]
    for k in k_indices:
        w_k = weights[k]
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start > i_end:
            continue
        j_vals = k - np.arange(i_start, i_end + 1)
        sg[i_start:i_end + 1] += 2.0 * w_k * a[j_vals]
    return sg


def nesterov_soft_optimize(a0, time_limit, temp_start=0.05, temp_end=0.003,
                           lr_init=0.005, verbose=False):
    """
    Nesterov-accelerated projected subgradient with temperature annealing.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    v = np.zeros(n)
    lr = lr_init
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Temperature annealing
        progress = min(1.0, it / max(1, it * 0.995))
        temp = temp_start * (temp_end / temp_start) ** progress

        t_k = (1 + np.sqrt(1 + 4 * (it - 1) ** 2)) / 2.0
        beta_k = (t_k - 1) / t_k

        sg = soft_subgrad(y, temperature=temp)
        v = beta_k * v + sg
        norm_v = np.linalg.norm(v)
        if norm_v > 1e-15:
            v = v / norm_v

        # Backtracking line search (simplified)
        step = lr
        found = False
        for _ in range(5):
            y_new = simplex_project(y - step * v)
            sc = score_normalized(y_new)
            if sc < best_score + 0.01:  # allow slight increase
                found = True
                break
            step *= 0.7

        if not found:
            y_new = simplex_project(y - step * v)
            sc = score_normalized(y_new)

        y = y_new
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 50 == 0:
                lr *= 0.9

        # Periodic symmetry regularization
        if it % 100 == 0:
            sym = 0.5 * (y + y[::-1])
            sym = np.maximum(sym, 0)
            ss = sym.sum()
            if ss > 1e-15:
                sym = sym / ss
                sc_sym = score_normalized(sym)
                if sc_sym < best_score:
                    best_score = sc_sym
                    best_a = sym.copy()
                    y = sym.copy()
                    v = np.zeros(n)

        # Restart if stuck
        if no_improve > 300:
            noise = np.random.default_rng(it * 31 + 7).standard_normal(n) * 0.03
            y = simplex_project(best_a + noise)
            v = np.zeros(n)
            lr = max(lr_init * 0.5, lr * 1.5)
            no_improve = 0

        lr *= 0.9998

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Nesterov-soft it={it} ({elapsed:.0f}s): best={best_score:.6f} temp={temp:.5f} lr={lr:.6f}")

    return best_a, best_score


def coord_descent_fast(a0, time_limit, n_sweeps=500):
    """
    Fast coordinate descent with proper 1D optimization per coordinate.
    For each coordinate i, the score as a function of a_i is piecewise,
    and we find the optimum by ternary search over [0, upper_bound].
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        perm = rng.permutation(n)
        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[i]
            if old_val < 1e-12:
                # Try turning on
                for factor in [0.01, 0.05, 0.1]:
                    a_trial = a.copy()
                    a_trial[i] = factor / n
                    a_trial = a_trial / a_trial.sum()
                    sc = score_normalized(a_trial)
                    if sc < best_score - 1e-13:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                continue

            # Ternary search for optimal scaling
            lo = 0.0
            hi = old_val * 3.0
            for _ in range(12):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                a[i] = m1
                a = np.maximum(a, 0)
                a = a / a.sum()
                sc1 = score_normalized(a)
                a[i] = m2
                a = np.maximum(a, 0)
                a = a / a.sum()
                sc2 = score_normalized(a)
                if sc1 < sc2:
                    hi = m2
                else:
                    lo = m1
            opt_val = (lo + hi) / 2.0
            a[i] = opt_val
            a = np.maximum(a, 0)
            a = a / a.sum()
            sc = score_normalized(a)
            if sc < best_score - 1e-13:
                best_score = sc
                best_a = a.copy()
                improved = True
            else:
                a[i] = old_val
                a = a / a.sum()

        if not improved and sweep > 20:
            break

    return best_a, best_score


def pair_update(a0, time_limit):
    """
    Coordinated pair updates: find pairs of coordinates that both
    contribute to the same peak and adjust them together.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    for step in range(10000):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        # Find top 3 peaks
        top_k = min(3, len(b))
        peaks = np.argpartition(b, -top_k)[-top_k:]
        peak_k = int(peaks[0])

        # Find contributing pairs
        i_start = max(0, peak_k - n + 1)
        i_end = min(n - 1, peak_k)
        idx = np.arange(i_start, i_end + 1)
        contribs = a[idx] * a[peak_k - idx]

        # Get top 4 contributing indices
        top_contrib = np.argsort(contribs)[::-1][:4]

        improved = False
        for ci in top_contrib:
            i = int(idx[ci])
            j = peak_k - i
            if a[i] < 1e-12:
                continue

            # Try reducing i
            for f in [0.95, 0.9, 0.8, 0.7]:
                a_trial = a.copy()
                a_trial[i] *= f
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-13:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break

            # Try reducing both i and j (if different)
            if not improved and i != j and a[j] >= 1e-12:
                for f1 in [0.95, 0.9]:
                    for f2 in [0.95, 0.9]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        st = a_trial.sum()
                        if st < 1e-15:
                            continue
                        a_trial = a_trial / st
                        sc = score_normalized(a_trial)
                        if sc < best_score - 1e-13:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break

            # Try increasing a coordinate that doesn't contribute to this peak
            if not improved:
                # Pick a random coordinate far from the peak contributors
                candidates = list(set(range(n)) - set(idx.tolist()))
                if candidates:
                    ri = int(rng.integers(0, len(candidates)))
                    ri = candidates[ri]
                    for f in [1.1, 1.2, 1.5]:
                        a_trial = a.copy()
                        a_trial[ri] *= f
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        sc = score_normalized(a_trial)
                        if sc < best_score - 1e-13:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break

            if improved:
                break

        if not improved:
            # Try a random perturbation
            noise = rng.standard_normal(n) * 0.01
            a_trial = simplex_project(a + noise)
            sc = score_normalized(a_trial)
            if sc < best_score - 1e-14:
                a = a_trial
                best_score = sc
                best_a = a.copy()
            elif step > 200:
                break

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def init_fourier(n, rng):
    """Initialize using random Fourier superposition."""
    n_harmonics = int(rng.integers(3, 20))
    amplitudes = rng.random(n_harmonics) * 0.4
    phases = rng.uniform(0, 2 * np.pi, n_harmonics)
    freqs = np.arange(1, n_harmonics + 1)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harmonics):
        a += amplitudes[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_window(n, rng, window_type='hann'):
    """Initialize with a window function (reduces edge effects)."""
    t = np.arange(n, dtype=np.float64)
    if window_type == 'hann':
        a = 0.5 * (1 - np.cos(2 * np.pi * t / (n - 1)))
    elif window_type == 'hamming':
        a = 0.54 - 0.46 * np.cos(2 * np.pi * t / (n - 1))
    elif window_type == 'blackman':
        a = 0.42 - 0.5 * np.cos(2 * np.pi * t / (n - 1)) + 0.08 * np.cos(4 * np.pi * t / (n - 1))
    else:
        a = np.ones(n)
    # Add small perturbation
    a += rng.standard_normal(n) * 0.02
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate via evaluate_sequence and update best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score

            a_alt, s_alt = nesterov_soft_optimize(init_alt, t_alt, temp_start=0.03, temp_end=0.005, lr_init=0.005)
            best_seq, best_score = check_and_update(a_alt, best_seq, best_score, f"p4_n{alt_n}")

            # Quick CD
            rem = deadline - time.monotonic() - 60
            if rem > 15:
                a_cd_alt, _ = coord_descent_fast(a_alt, min(rem * 0.5, 30.0), n_sweeps=50)
                best_seq, best_score = check_and_update(a_cd_alt, best_seq, best_score, f"p4_cd_n{alt_n}")

    # === Phase 5: Final intensive polish (remaining time) ===
    print(f"\n=== Phase 5: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        # Symmetry pass
        sym = 0.5 * (best_seq + best_seq[::-1])
        sym = np.maximum(sym, 0)
        sym = sym / sym.sum()
        best_seq, best_score = check_and_update(sym, best_seq, best_score, "p5_sym")

        # Final Nesterov with very low temperature
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            t_fin = min(rem * 0.4, 50.0)
            a_fin, s_fin = nesterov_soft_optimize(best_seq, t_fin, temp_start=0.003, temp_end=0.0005, lr_init=0.001)
            best_seq, best_score = check_and_update(a_fin, best_seq, best_score, "p5_fin_nest")

        # Final CD
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            a_fin2, _ = coord_descent_fast(best_seq, min(rem * 0.6, 30.0), n_sweeps=100)
            best_seq, best_score = check_and_update(a_fin2, best_seq, best_score, "p5_fin_cd")

        # Final pair update
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            a_fin3, _ = pair_update(best_seq, min(rem * 0.6, 20.0))
            best_seq, best_score = check_and_update(a_fin3, best_seq, best_score, "p5_fin_pu")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
