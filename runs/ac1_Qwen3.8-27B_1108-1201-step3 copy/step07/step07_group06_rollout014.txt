    n = len(a)

    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_from_conv(conv, n):
    """Score assuming sum(a) = 1"""

- Let a = c + d where c is the "DC" component (uniform) and d is the "AC" component
- The power spectrum of a is |C + D(k)|^2 where C = S (the sum) and D(k) is the DFT of d
- For a flat power spectrum, we want |D(k)| to be constant (and much less than C)
- So we want d to have a "flat" power spectrum, i.e., d should be "white noise"-like
- But d = a - c must satisfy a >= 0, i.e., d >= -c everywhere

So the problem becomes: find d with flat spectrum, subject to d >= -1/n (if c = 1/n) and sum(d) = 0.

This is a constrained spectral design problem! We can solve it iteratively:
1. Start with some d_0
2. Compute its spectrum D(k)
3. Re-shape the spectrum to be flatter: D'(k) = D(k) / |D(k)|^alpha * target
4. Take inverse DFT
5. Clip to satisfy the constraint d >= -1/n
6. Repeat

This is similar to the Gerchberg-Saxton algorithm (used in optics for wavefront generation)!

"""
Hybrid optimization combining spectral flattening (Gerchberg-Saxton), top-K peak
gradient descent with Adam, peak migration, and multi-length exploration.
Key innovations over previous approach: (1) frequency-domain iterative compression
to flatten power spectrum, (2) exact top-K peak gradient instead of softmax
approximation, (3) asymmetry-aware search exploiting the insight that asymmetric
sequences can have center peak below S^2/n, (4) peak migration with multi-peak
awareness.

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute objective assuming a is non-negative."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    a_norm = a / s
    conv = fast_convolve(a_norm)
    return 2.0 * n * conv.max()


def topk_peak_grad(a, K=15):
    """
    Compute gradient of (4n/K) * sum_{k in topK} conv[k] w.r.t. a.
    Uses the approximation d(conv[k])/d(a[i]) ≈ 2*a[k-i].
    Returns (grad_array, top_indices, peak_values).
    """
    n = len(a)
    conv = fast_convolve(a)
    top_idx = np.argsort(conv)[::-1][:K]

    # Build peak indicator vector
    pe = np.zeros(2 * n - 1)
    pe[top_idx] = 1.0

    # Cross-correlation: corr[i] = sum_k pe[k] * a[k-i] for valid (k-i)
    # Use FFT: corr = ifft(fft(pe) * conj(fft(a)))
    fft_size = 1 << (2 * n - 2).bit_length()
    PE = rfft(pe, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(PE * np.conj(A), fft_size)[:n]

    grad = (4.0 * n / K) * corr
    return grad, top_idx, conv[top_idx]


def spectral_flatten(a_init, n_iters, beta_start=0.7, beta_end=0.3,
                     target_ratio=0.0, enforce_sym=False):
    """
    Iterative spectral flattening (Gerchberg-Saxton variant).
    Flattens the power spectrum of a while maintaining a >= 0 and sum(a) = 1.
    """
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-30:
        return a_init.copy()
    a = a / s

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters - 1)
        beta = beta_start + (beta_end - beta_start) * progress

        # Compute DFT
        A = rfft(a)
        mags = np.abs(A)
        phases = np.angle(A)

        # Target: flatten non-DC magnitudes
        # Use geometric mean as target
        non_dc_mags = mags[1:]
        if len(non_dc_mags) > 0 and np.all(non_dc_mags > 1e-30):
            target = np.exp(np.mean(np.log(non_dc_mags)))
        else:
            target = mags[0] / np.sqrt(n)

        # Iteratively compress magnitude toward target
        mags_new = mags.copy()
        for k in range(1, len(mags)):
            if mags[k] > 1e-30:
                mags_new[k] = mags[k] * (target / mags[k]) ** beta
            else:
                mags_new[k] = target

        # Reconstruct
        A_new = mags_new * np.exp(1j * phases)
        a_new = irfft(A_new, n)

        # Enforce non-negativity
        a_new = np.maximum(a_new, 0.0)
        ss = a_new.sum()
        if ss < 1e-30:
            break
        a_new = a_new / ss

        # Check if this is the real part only (for rfft reconstruction)
        # irfft already gives real output

        score_new = compute_score(a_new)
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()

        a = a_new

    return best_a


def spectral_flatten_with_phase(a_init, n_iters, beta_start=0.6, beta_end=0.2):
    """
    Extended spectral flattening that also optimizes phases.
    Uses the fact that for real a, the DFT has conjugate symmetry.
    """
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-30:
        return a_init.copy()
    a = a / s

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters - 1)
        beta = beta_start + (beta_end - beta_start) * progress

        # Full complex FFT for phase optimization
        A = fft(a)
        mags = np.abs(A)
        phases = np.angle(A)

        # Target magnitude (flat)
        target = mags[0] / np.sqrt(n) * 1.5  # Slightly above 1/sqrt(n)

        mags_new = mags.copy()
        for k in range(1, n):
            # Preserve conjugate symmetry
            mags_new[k] = mags[k] * (target / max(mags[k], 1e-30)) ** beta

        # Enforce conjugate symmetry
        for k in range(1, (n + 1) // 2):
            mags_new[n - k] = mags_new[k]

        # Try small phase perturbations at high-magnitude bins
        # (skip for now - just flatten magnitudes)

        A_new = mags_new * np.exp(1j * phases)
        # Enforce conjugate symmetry on full spectrum
        A_new[n - np.arange(1, (n + 1) // 2)] = np.conj(A_new[np.arange(1, (n + 1) // 2)])

        a_new = ifft(A_new).real
        a_new = np.maximum(a_new, 0.0)
        ss = a_new.sum()
        if ss < 1e-30:
            break
        a_new = a_new / ss

        score_new = compute_score(a_new)
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()

        a = a_new

    return best_a


def adam_topk(a0, time_limit, K=15, lr=0.01, beta1=0.9, beta2=0.999,
              eps=1e-8, K_schedule=None):
    """
    Adam optimizer using exact top-K peak gradient.
    Works in the space of a >= 0, sum(a) = 1 using log-parameterization.
    """
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    # Add offset so that softmax(z) is close to a0
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)
    no_improve = 0
    max_no_improve = 2000

    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)

        # Adaptive K: start with more peaks, focus on fewer later
        if K_schedule == 'decrease':
            K_curr = max(5, K - int(10 * progress))
        elif K_schedule == 'increase':
            K_curr = max(5, 5 + int(10 * (1 - progress)))
        else:
            K_curr = K

        # Current a from z (softmax)
        z_shifted = z - np.max(z)
        exp_z = np.exp(z_shifted)
        a_curr = exp_z / exp_z.sum()

        # Gradient
        grad, top_idx, peak_vals = topk_peak_grad(a_curr, K_curr)

        # Transform to z-space
        S = np.dot(grad, a_curr)
        grad_z = a_curr * (grad - S)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Evaluate
        a_new = softmax_from_z(z)
        score = compute_score(a_new)

        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Restart if stuck
        if no_improve > max_no_improve:
            z = np.log(np.maximum(best_a, 1e-300)) + np.random.standard_normal(n) * 0.01
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = max(0.001, lr * 0.7)

        # Slow LR decay
        lr = max(0.001, lr * 0.9999)

    return best_a


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - np.max(z)
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return exp_z / s


def peak_migration(a0, time_limit, n_trials=10000):
    """
    Peak migration: for each high peak, try moving mass from contributing
    positions to non-contributing positions, checking the effect on ALL peaks.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(a)
        max_val = conv.max()
        
        # Find top few peaks (exclude the very edges which are just a_i^2)
        # Focus on interior peaks
        search_conv = conv.copy()
        search_conv[:2] = -1
        search_conv[-2:] = -1
        peak_k = int(np.argmax(search_conv))
        
        if conv[peak_k] <= max_val * 0.95:
            # Peak is already at the boundary, that's fine
            peak_k = int(np.argmax(conv))

        # Contributing pairs: (i, peak_k - i) for valid i
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        # Try reducing top contributors
        top_n = min(8, len(idx))
        top_ci = np.argsort(contribs)[::-1][:top_n]

        improved = False
        for ci in top_ci:
            if a[idx[ci]] < 1e-8:
                continue
            i = int(idx[ci])
            j = int(peak_k - i)
            if i == j or a[j] < 1e-8:
                continue

            # Try reducing i and increasing a random non-contributing position
            for f_red in [0.7, 0.8, 0.9]:
                # Find non-contributing positions
                non_contrib_mask = np.ones(n, dtype=bool)
                non_contrib_mask[idx[ci]] = False
                # Also exclude positions that contribute to other top peaks
                non_contrib_mask[j] = False
                
                if non_contrib_mask.sum() < 10:
                    continue
                    
                # Pick a random non-contributing position
                candidates = np.where(non_contrib_mask)[0]
                # Prefer positions with low a values (less impact elsewhere)
                candidate_scores = a[candidates] + 0.1 * np.random.random(len(candidates))
                k_new = candidates[np.argmin(candidate_scores)]

                for f_red_val in [0.7, 0.8, 0.9]:
                    a_trial = a.copy()
                    delta = a_trial[i] * (1.0 - f_red_val)
                    a_trial[i] *= f_red_val
                    a_trial[k_new] += delta
                    ss = a_trial.sum()
                    a_trial = a_trial / ss
                    score = compute_score(a_trial)
                    if score < best_score - 1e-12:
                        a = a_trial
                        best_score = score
                        best_a = a_trial.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved and trial % 200 == 199:
            # Perturb to escape local minimum
            a = best_a + rng.standard_normal(n) * 0.002
            a = np.maximum(a, 0)
            a = a / a.sum()

    return best_a


def asymmetric_shift(a0, shift_fraction):
    """
    Create an asymmetric version by shifting mass from one side to the other.
    """
    n = len(a0)
    a = a0.copy()
    s = a.sum()
    a = a / s

    # Split into left and right halves
    mid = n // 2
    left_mass = a[:mid].sum()
    right_mass = a[mid:].sum()

    # Shift some mass
    delta = shift_fraction * min(left_mass, right_mass)
    if left_mass > right_mass:
        # Move mass from left to right
        a[:mid] -= delta * a[:mid] / left_mass
        a[mid:] += delta * a[mid:] / max(right_mass, 1e-15)
    else:
        a[mid:] -= delta * a[mid:] / right_mass
        a[:mid] += delta * a[:mid] / max(left_mass, 1e-15)

    a = np.maximum(a, 0)
    ss = a.sum()
    if ss > 1e-15:
        a = a / ss
    return a


def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0)
    ss = a_interp.sum()
    if ss > 1e-15:
        a_interp = a_interp / ss
    return a_interp


def generate_diverse_inits(parent, n, rng):
    """Generate diverse initializations."""
    candidates = []

    if parent is not None:
        p = np.maximum(parent, 0)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
            candidates.append(('parent', p.copy()))
            for scale in [0.01, 0.02, 0.05, 0.1, 0.2]:
                pert = p + rng.standard_normal(n) * scale
                pert = np.maximum(pert, 0)
                ps2 = pert.sum()
                if ps2 > 1e-10:
                    pert = pert / ps2
                    candidates.append((f'pert_{scale}', pert))
            # Asymmetric shifts
            for frac in [0.1, 0.2, 0.3, 0.5]:
                asym = asymmetric_shift(p, frac)
                candidates.append((f'asym_{frac}', asym))

    # Uniform
    candidates.append(('uniform', np.ones(n) / n))

    # Triangular
    tri = np.concatenate([np.linspace(0.1, 1, n // 2), np.linspace(1, 0.1, n - n // 2)])
    tri = tri / tri.sum()
    candidates.append(('triangular', tri))

    # Skewed (exponential)
    expo = np.exp(-np.linspace(0, 3, n))
    expo = expo / expo.sum()
    candidates.append(('expo_right', expo))
    expo2 = np.exp(-np.linspace(3, 0, n))
    expo2 = expo2 / expo2.sum()
    candidates.append(('expo_left', expo2))

    # Bimodal
    bi = np.zeros(n)
    bi[n // 4: 3 * n // 4] = 1.0
    bi = np.maximum(bi, 0.01)
    bi = bi / bi.sum()
    candidates.append(('bimodal', bi))

    # Gaussian
    x = np.linspace(-3, 3, n)
    gau = np.exp(-x ** 2 / 3)
    gau = np.maximum(gau, 1e-6)
    gau = gau / gau.sum()
    candidates.append(('gaussian', gau))

    # Random
    rand = np.abs(rng.standard_normal(n)) + 0.01
    rand = rand / rand.sum()
    candidates.append(('random', rand))

    # Random with structure
    t = np.linspace(0, 1, n)
    for freq in [1, 2, 3, 5]:
        struct = 0.5 + 0.3 * np.sin(2 * np.pi * freq * t + rng.random() * 2 * np.pi)
        struct = np.maximum(struct, 0.01)
        struct = struct / struct.sum()
        candidates.append((f'struct_{freq}', struct))

    return candidates


def evaluate_and_track(a, label, best_seq, best_score, verbose=True):
    """Evaluate and track best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    # Clip to valid range
    a_norm = np.minimum(a_norm, 1000.0)
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def adam_logspace_smooth(a0, time_limit, T_init=0.05, T_final=0.002, lr=0.008,
                         beta1=0.9, beta2=0.999, eps=1e-8, schedule='hold_anneal'):
    """Adam in log-space with softmax smoothing (kept from previous for compatibility)."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    s = a.sum()
    a = a / s
    z = np.log(np.maximum(a, 1e-300))
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    fft_size = 1 << (2 * n - 2).bit_length()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    no_improve = 0
    max_no_improve = 2000
    last_lr = lr

    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)

        if schedule == 'linear':
            T = T_init + (T_final - T_init) * progress
        elif schedule == 'exp':
            T = T_init * (T_final / T_init) ** progress
        elif schedule == 'hold_anneal':
            if progress < 0.3:
                T = T_init
            else:
                p2 = (progress - 0.3) / 0.7
                T = T_init * (T_final / T_init) ** p2
        else:
            T = T_init * (T_final / T_init) ** progress

        a_curr = softmax_from_z(z)
        b = fast_convolve(a_curr)
        max_b = b.max()
        exp_b = np.exp(b / T - max_b / T)
        w = exp_b / exp_b.sum()

        # Gradient
        W = rfft(w, fft_size)
        A = rfft(a_curr, fft_size)
        corr = irfft(W * np.conj(A), fft_size)[:n]
        grad_a = 2.0 * n * corr

        S = np.dot(grad_a, a_curr)
        grad_z = a_curr * (grad_a - S)

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = softmax_from_z(z)
        conv_new = fast_convolve(a_new)
        score = 2.0 * n * conv_new.max()

        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            z = np.log(np.maximum(best_a, 1e-300)) + np.random.standard_normal(n) * 0.005
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = last_lr * 0.7

        last_lr = lr * (1.0 - 0.0001)
        lr = max(0.0005, last_lr)

    return best_a, best_score


def coordinate_descent_fine(a0, time_limit, n_sweeps=200):
    """Fine coordinate descent with more factors."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    factors = [0.85, 0.9, 0.95, 1.05, 1.1, 1.15]

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        order = rng.permutation(n)
        improved = False

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-12:
                continue

            for f in factors:
                a_trial = a.copy()
                a_trial[i] *= f
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                score = compute_score(a_trial)
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            break

    return best_a


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization: spectral flattening, top-K gradient, peak migration,
    and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_default = 768
        uniform = np.ones(n_default) / n_default
        best_seq = uniform
        best_score = float(evaluate_sequence(uniform.tolist()))
        print(f"Default uniform: n={n_default}, score={best_score:.6f}")

    n_main = len(best_seq)
    print(f"Starting: n={n_main}, score={best_score:.6f}")

    # === Phase 1: Spectral flattening from diverse inits (25% time) ===
    print(f"\n=== Phase 1: Spectral Flattening (n={n_main}) ===")
    t_p1 = (deadline - time.monotonic()) * 0.25
    t_p1_start = time.monotonic()

    inits = generate_diverse_inits(parent if parent is not None else best_seq, n_main, rng)
    # Score and sort
    inits_scored = []
    for label, init_a in inits:
        s = compute_score(init_a)
        inits_scored.append((s, label, init_a))
    inits_scored.sort(key=lambda x: x[0])
    print(f"  Top init: {inits_scored[0][1]} ({inits_scored[0][0]:.6f})")

    # Run spectral flattening on top candidates
    n_spectral = min(8, len(inits_scored))
    time_per = t_p1 / n_spectral

    spectral_results = []
    for idx in range(n_spectral):
        rem = deadline - time.monotonic() - 400
        if rem < 60:
            break
        t_sub = min(time_per, rem * 0.4)
        if t_sub < 10:
            continue

        s_init, label, init_a = inits_scored[idx]
        n_iters = max(200, int(t_sub * 50))  # ~50 iters/sec estimate
        a_spec = spectral_flatten(init_a, n_iters, beta_start=0.7, beta_end=0.2)
        sc_spec = compute_score(a_spec)
        spectral_results.append((sc_spec, f'spec_{label}', a_spec))
        print(f"  Spectral [{label}]: {sc_spec:.6f}")

    # Also try phase optimization on best spectral result
    if spectral_results:
        spectral_results.sort(key=lambda x: x[0])
        best_spec = spectral_results[0][2]
        rem = deadline - time.monotonic() - 350
        if rem > 30:
            n_iters2 = max(100, int(rem * 30))
            a_spec2 = spectral_flatten_with_phase(best_spec, n_iters2)
            sc_spec2 = compute_score(a_spec2)
            spectral_results.append((sc_spec2, 'spec_phase', a_spec2))
            print(f"  Spectral+Phase: {sc_spec2:.6f}")

    # Track best from spectral
    for sc, label, a in spectral_results:
        best_seq, best_score = evaluate_and_track(a, label, best_seq, best_score, verbose=False)
    print(f"  After Phase 1: best={best_score:.6f}")

    # === Phase 2: Top-K gradient descent (30% time) ===
    print(f"\n=== Phase 2: Top-K Gradient (n={n_main}) ===")
    t_p2 = (deadline - time.monotonic()) * 0.30
    t_p2_start = time.monotonic()

    # Multiple K values and starting points
    start_points = [best_seq]
    if spectral_results:
        for sc, label, a in spectral_results[:3]:
            if len(a) == n_main:
                start_points.append(a)

    n_starts = min(5, len(start_points))
    time_per_start = t_p2 / n_starts

    for idx in range(n_starts):
        rem = deadline - time.monotonic() - 300
        if rem < 30:
            break
        t_sub = min(time_per_start, rem * 0.8)
        if t_sub < 20:
            continue

        sp = start_points[idx % len(start_points)]
        # Try different K values
        for K in [10, 20, 30]:
            rem = deadline - time.monotonic() - 250
            if rem < 15:
                break
            t_sub_k = min(t_sub / 3.0, rem * 0.6)
            if t_sub_k < 10:
                continue
            a_res = adam_topk(sp, t_sub_k, K=K, lr=0.01, K_schedule='decrease')
            best_seq, best_score = evaluate_and_track(a_res, f'topk_K{K}_s{idx}', best_seq, best_score, verbose=False)

    print(f"  After Phase 2: best={best_score:.6f}")

    # === Phase 3: Peak migration (20% time) ===
    print(f"\n=== Phase 3: Peak Migration (n={n_main}) ===")
    t_p3 = (deadline - time.monotonic()) * 0.20

    rem = deadline - time.monotonic() - 200
    if rem > 30:
        t_sub = min(t_p3 * 0.6, rem * 0.7)
        a_mig = peak_migration(best_seq, t_sub, n_trials=5000)
        best_seq, best_score = evaluate_and_track(a_mig, 'migration', best_seq, best_score, verbose=False)
        print(f"  After migration: {best_score:.6f}")

    # More gradient after migration
    rem = deadline - time.monotonic() - 180
    if rem > 20:
        t_sub = min(t_p3 * 0.4, rem * 0.7)
        a_g2 = adam_topk(best_seq, t_sub, K=15, lr=0.005, K_schedule='decrease')
        best_seq, best_score = evaluate_and_track(a_g2, 'topk_after_mig', best_seq, best_score, verbose=False)

    # === Phase 4: Smooth Adam + local polish (15% time) ===
    print(f"\n=== Phase 4: Polish (n={n_main}) ===")
    t_p4 = (deadline - time.monotonic()) * 0.15

    # Smooth Adam for fine-tuning
    rem = deadline - time.monotonic() - 100
    if rem > 20:
        t_sub = min(t_p4 * 0.5, rem * 0.7)
        a_smooth, _ = adam_logspace_smooth(best_seq, t_sub, T_init=0.01, T_final=0.0005,
                                           lr=0.003, schedule='exp')
        best_seq, best_score = evaluate_and_track(a_smooth, 'smooth_finish', best_seq, best_score, verbose=False)
        print(f"  After smooth Adam: {best_score:.6f}")

    # Coordinate descent
    rem = deadline - time.monotonic() - 80
    if rem > 15:
        t_sub = min(t_p4 * 0.5, rem * 0.7)
        a_cd = coordinate_descent_fine(best_seq, t_sub, n_sweeps=100)
        best_seq, best_score = evaluate_and_track(a_cd, 'coord_fine', best_seq, best_score, verbose=False)
        print(f"  After coord: {best_score:.6f}")

    # === Phase 5: Multi-length exploration (10% time) ===
    print(f"\n=== Phase 5: Multi-Length (best={best_score:.6f}) ===")
    t_p5 = (deadline - time.monotonic()) * 0.10
    rem = deadline - time.monotonic() - 60

    if rem > 40:
        alt_lengths = []
        for factor in [0.75, 1.25, 1.5, 0.5, 2.0]:
            alt_n = int(n_main * factor)
            if 128 <= alt_n <= 4096 and alt_n != n_main:
                if alt_n not in alt_lengths:
                    alt_lengths.append(alt_n)

        for alt_n in alt_lengths[:3]:
            rem = deadline - time.monotonic() - 30
            if rem < 30:
                break
            t_sub = min(t_p5 / 3.0, rem * 0.7)
            if t_sub < 10:
                continue
            init_alt = interpolate_to_length(best_seq, alt_n)
            # Spectral flatten at new length
            n_iters = max(100, int(t_sub * 40))
            a_alt_spec = spectral_flatten(init_alt, n_iters, beta_start=0.6, beta_end=0.2)
            # Then gradient
            rem2 = deadline - time.monotonic() - 20
            if rem2 > 10:
                a_alt = adam_topk(a_alt_spec, min(rem2 * 0.5, 30), K=15, lr=0.008)
            else:
                a_alt = a_alt_spec
            best_seq, best_score = evaluate_and_track(a_alt, f'alt_n{alt_n}', best_seq, best_score, verbose=False)
            print(f"  Alt n={alt_n}: {best_score:.6f}")

    # === Phase 6: Final intensive refinement ===
    print(f"\n=== Phase 6: Final Refinement (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 10

    if rem > 20:
        # Very low temperature Adam
        t_sub = min(rem * 0.4, 40)
        a_fin = adam_topk(best_seq, t_sub, K=10, lr=0.002, K_schedule=None)
        best_seq, best_score = evaluate_and_track(a_fin, 'final_topk', best_seq, best_score, verbose=False)

    rem = deadline - time.monotonic() - 5
    if rem > 10:
        t_sub = min(rem * 0.5, 20)
        a_mig_final = peak_migration(best_seq, t_sub, n_trials=2000)
        best_seq, best_score = evaluate_and_track(a_mig_final, 'final_mig', best_seq, best_score, verbose=False)

    # Final coordinate descent
    rem = deadline - time.monotonic() - 3
    if rem > 5:
        a_cd_final = coordinate_descent_fine(best_seq, rem, n_sweeps=50)
        best_seq, best_score = evaluate_and_track(a_cd_final, 'final_cd', best_seq, best_score, verbose=False)

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK: {best_score:.6f}")

    return [float(x) for x in best_seq]
