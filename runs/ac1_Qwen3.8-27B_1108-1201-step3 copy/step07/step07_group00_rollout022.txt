"""
Optimization of step function coefficients to minimize 2*n*max(conv(a,a))/(sum(a))^2.

Algorithm:
1. Temperature-annealed p-norm smoothing (smooth the max, gradually sharpen)
2. Mirror descent / multiplicative weight updates on the simplex
3. LP-based direction finding with PDLP solver for precise refinement
4. Coordinate descent polish
5. Multi-length exploration
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_pnorm_gradient(a, p):
    """
    Compute gradient of (sum_k b_k^p)^(1/p) where b = a*a (linear conv).
    Returns gradient vector of same length as a.
    """
    n = len(a)
    b = fast_convolve(a)
    # Ensure non-negative (should be for a >= 0, but numerical issues)
    b = np.maximum(b, 0.0)
    
    if p == np.inf:
        # Subgradient: 2 * a[b.argmax() - i] ... actually just return the subgradient
        k_max = b.argmax()
        w = np.zeros(len(b))
        w[k_max] = 1.0
    else:
        b_p = b ** p
        sum_bp = b_p.sum()
        if sum_bp < 1e-30:
            return np.zeros(n)
        coeff = sum_bp ** (1.0/p - 1.0)
        w = b ** (p - 1) * coeff  # This is (sum b^p)^(1/p - 1) * b^(p-1)
    
    # grad[i] = 2 * sum_j a[j] * w[i+j] (cross-correlation)
    a_reversed = a[::-1]
    fft_size = 1 << (n + len(b) - 1).bit_length()
    AR = rfft(a_reversed, fft_size)
    W = rfft(w, fft_size)
    conv_result = irfft(AR * W, fft_size)
    cross = conv_result[n-1:n-1+n]
    grad = 2.0 * cross
    return grad


def run_pnorm_phase(f_init, time_limit, p_start=4, p_end=64, p_gamma=1.05, 
                    eta_start=0.01, label="pnorm"):
    """
    Temperature-annealed p-norm optimization with mirror descent.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy().astype(np.float64)
    s = f.sum()
    if s > 1e-10:
        f /= s
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    p = p_start
    eta = eta_start
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Compute gradient
        grad = compute_pnorm_gradient(f, p)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Mirror descent update (multiplicative)
        # f_new[i] = f[i] * exp(-eta * grad[i] / grad_norm)
        # For stability, use: f_new[i] = f[i] * (1 - eta * grad[i] / grad_max)
        grad_max = np.max(np.abs(grad))
        if grad_max < 1e-15:
            break
        ratios = 1.0 - eta * grad / grad_max
        ratios = np.maximum(ratios, 1e-10)  # Keep positive
        f_new = f * ratios
        s_new = f_new.sum()
        if s_new < 1e-10:
            f_new = f.copy()
        else:
            f_new /= s_new
        
        # Check improvement
        obj_new = fast_convolve(f_new).max()
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            eta = min(eta * 1.01, 0.1)  # Slightly increase step
        else:
            no_improve += 1
            if no_improve > 20:
                eta *= 0.9
                no_improve = 0
            f = f_new  # Accept anyway for progress
        
        # Increase p (sharpen the objective)
        if it % 50 == 0:
            p = min(p * p_gamma, p_end)
            if p > 32:
                eta *= 0.95  # Reduce step as p increases
        
        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} p={p:.1f} eta={eta:.4f} "
                  f"obj={best_obj:.10f} score={2.0*n*best_obj:.6f} t={elapsed:.0f}s")
    
    return best_f, best_obj

def run_lp_phase(f_init, time_limit, top_k_init=50, top_k_max=200, label="lp"):
    """
    LP-based optimization: find direction g that maximizes sum(g)
    subject to conv(f,g) <= M and g >= 0. Take a step.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy().astype(np.float64)
    s = f.sum()
    if s > 1e-10:
        f /= s
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    stuck = 0
    top_k = top_k_init
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 30 == 0 and top_k < top_k_max:
            top_k = min(top_k + 15, top_k_max)
        
        b = fast_convolve(f)
        M = b.max()
        if M < 1e-15:
            break
        
        if top_k < len(b):
            top_idx = np.argpartition(b, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(b))
        top_idx = np.sort(top_idx)
        K = len(top_idx)
        
        # Build constraint matrix
        A = np.zeros((K, n))
        for row_i in range(K):
            k = top_idx[row_i]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            A[row_i, lo:hi+1] = f[lo:hi+1]
        
        g = cp.Variable(n)
        prob = cp.Problem(cp.Maximize(cp.sum(g)), [g >= 0, A @ g <= M])
        try:
            prob.solve(solver=cp.PDLP, max_iters=100000)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, max_iters=200000)
            except Exception:
                break
        
        if g.value is None:
            break
        g_val = np.maximum(g.value, 0.0)
        S_g = g_val.sum()
        if S_g <= 1.0 + 1e-5:
            stuck += 1
            if stuck > 15:
                break
            top_k = min(top_k * 2, 2*n-1)
            continue
        
        # Line search
        h = g_val / S_g
        f_curr = f.copy()
        best_t = 0.05
        best_t_obj = fast_convolve(f_curr).max()
        
        for t in [0.01, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3]:
            f_trial = (1.0 - t) * f_curr + t * h
            f_trial = np.maximum(f_trial, 0.0)
            ss = f_trial.sum()
            if ss < 1e-10:
                continue
            f_trial /= ss
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t
        
        f = (1.0 - best_t) * f_curr + best_t * h
        f = np.maximum(f, 0.0)
        ss = f.sum()
        if ss > 1e-10:
            f /= ss
        
        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 40:
                break
        
        if it % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} obj={best_obj:.10f} "
                  f"score={2.0*n*best_obj:.6f} top_k={top_k} t={elapsed:.0f}s")
    
    return best_f, best_obj

                # But when k=2m, the two terms are the same (j=m and k-j=m), so:

                k_start = m
                k_end = min(2*n-2, m + n - 1)
                for k in range(k_start, k_end + 1):
                    if k == 2 * m:
                        b_trial[k] += 2 * delta * f[m] + delta * delta
                    else:
                        b_trial[k] += 2 * delta * f[k - m]
                
                obj_trial = b_trial.max()
                if obj_trial < best_obj_local - 1e-14:
                    best_obj_local = obj_trial
                    best_mult = mult

                    if k == 2 * m:

                    else:

                if best_obj_local < best_obj - 1e-14:
                    best_obj = best_obj_local
                improved = True

            break

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_pnorm_gradient(a, p):
    """
    Compute gradient of (sum_k b_k^p)^(1/p) where b = a*a (linear conv).
    Uses FFT-based cross-correlation for efficiency.
    """
    n = len(a)
    b = fast_convolve(a)
    b = np.maximum(b, 0.0)

    if p > 1e6:
        # Subgradient at the max
        k_max = int(b.argmax())
        w = np.zeros(len(b))
        w[k_max] = 1.0
    else:
        b_p = b ** p
        sum_bp = b_p.sum()
        if sum_bp < 1e-30:
            return np.zeros(n)
        coeff = sum_bp ** (1.0 / p - 1.0)
        w = (b ** (p - 1.0)) * coeff

    # grad[i] = 2 * sum_j a[j] * w[i+j]
    # Computed via cross-correlation using FFT
    a_reversed = a[::-1]
    fft_size = 1 << (n + len(b) - 1).bit_length()
    AR = rfft(a_reversed, fft_size)
    W = rfft(w, fft_size)
    conv_result = irfft(AR * W, fft_size)
    cross = conv_result[n - 1:n - 1 + n]
    grad = 2.0 * cross
    return grad
