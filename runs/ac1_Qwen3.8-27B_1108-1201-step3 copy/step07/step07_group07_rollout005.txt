"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csc_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def eval_max_b(a):
    """Return max of auto-convolution for sequence a (assumed normalized)."""
    b = fast_convolve(a)
    return float(b.max())


def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_smoothed(a0, time_limit, lr_init=0.003, momentum=0.92,
                         n_top=None, seed=42, restart_interval=2000,
                         smooth_beta=50.0, kick_size=0.002):
    """
    Smoothed subgradient descent with adaptive peak weighting.
    Uses softmax weighting of peaks for smoother gradients.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / S

    v = np.zeros(n)
    best_a = a.copy()
    best_obj = eval_max_b(a)
    cur_lr = lr_init
    no_improve = 0
    it_count = 0

    if n_top is None:
        n_top = max(20, n // 8)

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        b_max = b.max()

        # Select top peaks
        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
        else:
            top_idx = np.arange(len(b))
        top_vals = b[top_idx]

        # Smoothed weighting (softmax)
        beta = smooth_beta / max(b_max, 1e-10)
        exp_vals = np.exp(-beta * (b_max - top_vals))
        weights = exp_vals / exp_vals.sum()

        w = np.zeros(2 * n - 1)
        w[top_idx] = weights

        # Compute subgradient: g[i] = sum_k w[k] * a[k-i]
        # This is a cross-correlation of a with w
        fft_size = 1 << (3 * n - 2).bit_length()
        A = rfft(a, fft_size)
        W = rfft(w, fft_size)
        g = irfft(np.conj(A) * W, fft_size)[:n]

        # Nesterov momentum
        v = momentum * v - cur_lr * g
        a_new = project_simplex(a + v)

        obj_new = eval_max_b(a_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        it_count += 1

        # Adaptive learning rate
        if no_improve >= restart_interval:
            a = project_simplex(best_a + rng.standard_normal(n) * kick_size)
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.5, lr_init * 0.03)
            no_improve = 0
            kick_size *= 0.7
        elif no_improve >= restart_interval // 2:
            cur_lr *= 0.93
        elif no_improve >= restart_interval // 4:
            cur_lr *= 0.97
        else:
            # Slowly anneal beta for smoother gradients over time
            smooth_beta *= 0.999

    return best_a, best_obj


def lp_refine_active(f, time_limit=60.0, max_iters=40, n_active_frac=None,
                     verbose=False):
    """
    LP-based refinement using active set (only constraints near the peak).
    Much faster than full LP because we only include a fraction of constraints.
    """
    t0 = time.monotonic()
    n = len(f)
    a = f.copy().astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = eval_max_b(a)

    if n_active_frac is None:
        n_active_frac = 0.1  # Use 10% of constraints (those closest to peak)

    for iteration in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        elapsed = time.monotonic() - t0
        # Budget per iteration
        iter_time = (time_limit - elapsed) / (max_iters - iteration)
        if iter_time < 1.0:
            break

        b = fast_convolve(a)
        M = b.max()

        # Select active constraints (peaks within threshold of max)
        threshold = M * 0.95
        active_mask = b >= threshold
        n_active = active_mask.sum()
        
        # Limit to a maximum number of constraints for speed
        max_constraints = max(100, int(n_active_frac * (2 * n - 1)))
        if n_active > max_constraints:
            # Keep only the top max_constraints
            top_idx = np.argpartition(b, -max_constraints)[-max_constraints:]
            active_mask = np.zeros(2 * n - 1, dtype=bool)
            active_mask[top_idx] = True

        active_indices = np.nonzero(active_mask)[0]
        n_cons = len(active_indices)

        if n_cons == 0:
            break

        # Build sparse constraint matrix
        # Row for constraint k: sum_{j} a[j] * g[k-j] <= M
        # where g[k-j] is only non-zero when 0 <= k-j <= n-1
        rows = []
        cols = []
        vals = []
        
        for idx, k in enumerate(active_indices):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                col = k - j
                if 0 <= col < n:
                    rows.append(idx)
                    cols.append(col)
                    vals.append(a[j])

        A_sparse = csc_matrix((vals, (rows, cols)), shape=(n_cons, n))
        b_ub = np.full(n_cons, M)
        c_vec = -np.ones(n)

        try:
            res = linprog(c_vec, A_ub=A_sparse, b_ub=b_ub,
                          bounds=[(0.0, None)] * n, method='highs')
            if not res.success:
                if verbose:
                    print(f"    LP failed at iter {iteration}: {res.message}")
                break

            g = res.x
            g_sum = g.sum()
            if g_sum < 1e-10:
                break

            g_norm = g / g_sum

            # Line search for blending
            best_t = 0.0
            best_blend_obj = best_obj
            t_candidates = [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
            for t in t_candidates:
                blend = (1.0 - t) * best_a + t * g_norm
                blend = np.maximum(blend, 0.0)
                s = blend.sum()
                if s < 1e-10:
                    continue
                blend = blend / s
                obj_blend = eval_max_b(blend)
                if obj_blend < best_blend_obj:
                    best_blend_obj = obj_blend
                    best_t = t

            if best_t > 0.0:
                a_new = (1.0 - best_t) * best_a + best_t * g_norm
                a_new = np.maximum(a_new, 0.0)
                a_new = a_new / a_new.sum()
                obj_new = eval_max_b(a_new)
                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    if verbose:
                        print(f"    LP iter {iteration}: t={best_t:.2f}, max_b={best_obj:.10f}")
                else:
                    a_new = best_a.copy()
            else:
                # Try even smaller t
                found = False
                for t in [0.005, 0.01, 0.015]:
                    blend = (1.0 - t) * best_a + t * g_norm
                    blend = np.maximum(blend, 0.0)
                    blend = blend / blend.sum()
                    obj_blend = eval_max_b(blend)
                    if obj_blend < best_obj - 1e-14:
                        best_t = t
                        best_obj = obj_blend
                        best_a = blend.copy()
                        found = True
                        break
                if not found:
                    break

        except Exception as e:
            if verbose:
                print(f"    LP error at iter {iteration}: {e}")
            break

    return best_a, best_obj


def coordinate_descent_fast(a0, time_limit, n_rounds=50):
    """
    Fast coordinate descent: for each coordinate, try a few multipliers.
    Uses FFT for fast evaluation.
    """
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = eval_max_b(a)

    multipliers = np.array([0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0])

    for _round in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-15:
                continue
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0.0 or new_val > 0.9999:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < -0.01 or scale > 20.0:
                    continue

                a_tr = a.copy()
                a_tr[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_tr[mask] *= scale
                ss = a_tr.sum()
                if ss < 1e-15:
                    continue
                a_tr = a_tr / ss
                val = eval_max_b(a_tr)
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved:
            break

    return best_a, best_obj


def make_blackman_harris(n, nterms=4):
    """Blackman-Harris window initialization."""
    x = np.linspace(0, 1, n)
    if nterms == 4:
        a = np.ones(n)
        a = 0.35875 - 0.48829 * np.cos(2 * np.pi * x) + 0.14128 * np.cos(4 * np.pi * x) - 0.01168 * np.cos(6 * np.pi * x)
    elif nterms == 5:
        a = 0.4103 - 0.5392 * np.cos(2 * np.pi * x) + 0.1430 * np.cos(4 * np.pi * x) - 0.0143 * np.cos(6 * np.pi * x) + 0.0006 * np.cos(8 * np.pi * x)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_nuttall(n):

    x = np.linspace(0, 1, n)

    a = np.maximum(a, 1e-12)

def make_gaussian_init(n, sigma=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cosine_init(n):
    x = np.linspace(0, 1, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_tukey_init(n, r=0.5):
    a = np.ones(n)
    taper = max(1, int(r * n // 2))
    if taper < n:
        t = np.linspace(0, 1, taper)
        a[:taper] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t / r - 1.0)))
        a[-taper:] = a[:taper][::-1]
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_uniform_init(n):
    return np.ones(n) / n


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def multiplicative_update(a0, time_limit, n_peaks=20, lr=0.01, seed=42):
    """
    Multiplicative (log-space) subgradient updates.
    Naturally preserves non-negativity.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = eval_max_b(a)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        b_max = b.max()

        # Get top peaks
        if n_peaks < len(b):
            top_idx = np.argpartition(b, -n_peaks)[-n_peaks:]
        else:
            top_idx = np.arange(len(b))

        # Subgradient for peak: g[i] = sum_k a[k-i] for k in top peaks
        g = np.zeros(n)
        for k in top_idx:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            # g[i] += a[k-i] for i where 0 <= k-i <= n-1
            # i ranges from k-(n-1) to k, intersected with [0, n-1]
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            # a[k-i] for i in [i_lo, i_hi]
            # Let j = k-i, so j ranges from k-i_hi to k-i_lo
            j_lo = k - i_hi
            j_hi = k - i_lo
            # g[i_lo:i_hi+1] += a[j_hi:j_lo-1:-1]
            g[i_lo:i_hi + 1] += a[j_hi:j_lo - 1: -1]

        g /= n_peaks

        # Multiplicative update in log space
        log_a = np.log(a + 1e-15)
        log_a -= lr * g / a  # This is like gradient descent in log space
        a_new = np.exp(log_a)
        a_new = a_new / a_new.sum()

        obj_new = eval_max_b(a_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 500:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Blackman-Harris/Nuttall window initializations (designed for peak suppression)
- Active-set LP refinement (sparse, only top-K constraints for speed)
- Smoothed subgradient descent with softmax peak weighting and Nesterov momentum
- Multiplicative (log-space) updates as complementary optimizer
- Multi-length exploration with interpolation
- Coordinate descent polish
- Adaptive time allocation across phases
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
from scipy.sparse import csc_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def eval_max_b(a):
    """Return max of auto-convolution for sequence a."""
    b = fast_convolve(a)
    return float(b.max())


def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_smoothed(a0, time_limit, lr_init=0.003, momentum=0.92,
                         n_top=None, seed=42, restart_interval=2000,
                         smooth_beta=50.0, kick_size=0.002):
    """
    Smoothed subgradient descent with softmax peak weighting and Nesterov momentum.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / S

    v = np.zeros(n)
    best_a = a.copy()
    best_obj = eval_max_b(a)
    cur_lr = lr_init
    no_improve = 0

    if n_top is None:
        n_top = max(20, n // 8)

    fft_size = 1 << (3 * n - 2).bit_length()

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        b_max = b.max()

        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
        else:
            top_idx = np.arange(len(b))
        top_vals = b[top_idx]

        # Softmax weighting
        beta = smooth_beta / max(b_max, 1e-10)
        exp_vals = np.exp(-beta * (b_max - top_vals))
        weights = exp_vals / exp_vals.sum()

        w = np.zeros(2 * n - 1)
        w[top_idx] = weights

        # Subgradient via FFT: g[i] = sum_k w[k] * a[k-i]
        A = rfft(a, fft_size)
        W = rfft(w, fft_size)
        g = irfft(np.conj(A) * W, fft_size)[:n]

        # Nesterov momentum step
        v = momentum * v - cur_lr * g
        a_new = project_simplex(a + v)

        obj_new = eval_max_b(a_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        if no_improve >= restart_interval:
            a = project_simplex(best_a + rng.standard_normal(n) * kick_size)
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.5, lr_init * 0.03)
            no_improve = 0
            kick_size *= 0.7
        elif no_improve >= restart_interval // 2:
            cur_lr *= 0.93
        elif no_improve >= restart_interval // 4:
            cur_lr *= 0.97
        else:
            smooth_beta *= 0.9995

    return best_a, best_obj


def lp_refine_active(f, time_limit=60.0, max_iters=40, n_active_frac=None):
    """
    LP-based refinement using active set (only constraints near the peak).
    """
    t0 = time.monotonic()
    n = len(f)
    a = f.copy().astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = eval_max_b(a)

    if n_active_frac is None:
        n_active_frac = 0.1

    for iteration in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        M = b.max()

        # Select active constraints
        max_constraints = max(50, int(n_active_frac * (2 * n - 1)))
        if max_constraints < len(b):
            top_idx = np.argpartition(b, -max_constraints)[-max_constraints:]
        else:
            top_idx = np.arange(len(b))

        # Further filter: only keep those within 5% of max
        thresh = M * 0.95
        valid = top_idx[b[top_idx] >= thresh]
        if len(valid) == 0:
            break

        active_indices = valid
        n_cons = len(active_indices)

        # Build sparse constraint matrix
        rows = []
        cols = []
        vals = []

        for idx, k in enumerate(active_indices):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                col = k - j
                if 0 <= col < n:
                    rows.append(idx)
                    cols.append(col)
                    vals.append(a[j])

        A_sparse = csc_matrix((vals, (rows, cols)), shape=(n_cons, n))
        b_ub = np.full(n_cons, M)
        c_vec = -np.ones(n)

        try:
            res = linprog(c_vec, A_ub=A_sparse, b_ub=b_ub,
                          bounds=[(0.0, None)] * n, method='highs')
            if not res.success:
                break

            g = res.x
            g_sum = g.sum()
            if g_sum < 1e-10:
                break

            g_norm = g / g_sum

            # Line search for blending
            best_t = 0.0
            best_blend_obj = best_obj
            t_candidates = [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
            for t in t_candidates:
                blend = (1.0 - t) * best_a + t * g_norm
                blend = np.maximum(blend, 0.0)
                s = blend.sum()
                if s < 1e-10:
                    continue
                blend = blend / s
                obj_blend = eval_max_b(blend)
                if obj_blend < best_blend_obj:
                    best_blend_obj = obj_blend
                    best_t = t

            if best_t > 0.0:
                a_new = (1.0 - best_t) * best_a + best_t * g_norm
                a_new = np.maximum(a_new, 0.0)
                a_new = a_new / a_new.sum()
                obj_new = eval_max_b(a_new)
                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_a = a_new.copy()
                else:
                    a_new = best_a.copy()
            else:
                found = False
                for t in [0.005, 0.01, 0.015]:
                    blend = (1.0 - t) * best_a + t * g_norm
                    blend = np.maximum(blend, 0.0)
                    blend = blend / blend.sum()
                    obj_blend = eval_max_b(blend)
                    if obj_blend < best_obj - 1e-14:
                        best_t = t
                        best_obj = obj_blend
                        best_a = blend.copy()
                        found = True
                        break
                if not found:
                    break

            a = a_new

        except Exception:
            break

    return best_a, best_obj


def coordinate_descent_fast(a0, time_limit, n_rounds=50):
    """
    Fast coordinate descent with FFT evaluation.
    """
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = eval_max_b(a)

    multipliers = np.array([0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0])

    for _round in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-15:
                continue
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0.0 or new_val > 0.9999:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < -0.01 or scale > 20.0:
                    continue

                a_tr = a.copy()
                a_tr[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_tr[mask] *= scale
                ss = a_tr.sum()
                if ss < 1e-15:
                    continue
                a_tr = a_tr / ss
                val = eval_max_b(a_tr)
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved:
            break

    return best_a, best_obj


def multiplicative_update(a0, time_limit, n_peaks=20, lr=0.01, seed=42):
    """
    Multiplicative (log-space) subgradient updates.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = eval_max_b(a)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        b_max = b.max()

        if n_peaks < len(b):
            top_idx = np.argpartition(b, -n_peaks)[-n_peaks:]
        else:
            top_idx = np.arange(len(b))

        g = np.zeros(n)
        for k in top_idx:
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            j_lo = k - i_hi
            j_hi = k - i_lo
            g[i_lo:i_hi + 1] += a[j_hi:j_lo - 1: -1]

        g /= len(top_idx)

        log_a = np.log(a + 1e-15)
        log_a -= lr * g / (a + 1e-15)
        a_new = np.exp(log_a)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        obj_new = eval_max_b(a_new)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 500:
            a_new = project_simplex(a + rng.standard_normal(n) * 0.001)
            no_improve = 0
            lr *= 0.8

        a = a_new

    return best_a, best_obj


def make_blackman_harris(n, nterms=4):
    """Blackman-Harris window."""
    x = np.linspace(0, 1, n, endpoint=False)
    if nterms == 4:
        a = 0.35875 - 0.48829 * np.cos(2 * np.pi * x) + 0.14128 * np.cos(4 * np.pi * x) - 0.01168 * np.cos(6 * np.pi * x)
    elif nterms == 5:
        a = 0.4103 - 0.5392 * np.cos(2 * np.pi * x) + 0.1430 * np.cos(4 * np.pi * x) - 0.0143 * np.cos(6 * np.pi * x) + 0.0006 * np.cos(8 * np.pi * x)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_nuttall(n):
    """Nuttall window."""
    x = np.linspace(0, 1, n, endpoint=False)
    a = 0.3635819 - 0.4891775 * np.cos(2 * np.pi * x) + 0.1365995 * np.cos(4 * np.pi * x) - 0.0106411 * np.cos(6 * np.pi * x)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_gaussian_init(n, sigma=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cosine_init(n):
    x = np.linspace(0, 1, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_tukey_init(n, r=0.5):
    a = np.ones(n)
    taper = max(1, int(r * n // 2))
    if taper < n:
        t = np.linspace(0, 1, taper)
        a[:taper] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t / r - 1.0)))
        a[-taper:] = a[:taper][::-1]
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_uniform_init(n):
    return np.ones(n) / n


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def combined_optimize(a_init, time_budget, n, seed):
    """
    Combined multi-phase optimization: SG -> LP -> SG -> LP -> multiplicative -> LP
    """
    t0 = time.monotonic()
    a = a_init.copy().astype(np.float64)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / S
    best_a = a.copy()
    best_obj = eval_max_b(a)

    # Phase 1: Subgradient
    rem = time_budget - (time.monotonic() - t0)
    if rem > 10:
        t_sg = rem * 0.3
        a_sg, obj_sg = subgradient_smoothed(
            a, time_limit=t_sg, lr_init=0.004, momentum=0.92,
            n_top=max(20, n // 8), seed=seed
        )
        if obj_sg < best_obj:
            best_obj = obj_sg
            best_a = a_sg.copy()

    # Phase 2: LP refinement
    rem = time_budget - (time.monotonic() - t0)
    if rem > 8:
        t_lp = min(rem * 0.3, 60.0)
        a_lp, obj_lp = lp_refine_active(
            best_a, time_limit=t_lp, max_iters=30, n_active_frac=0.12
        )
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()

    # Phase 3: Another SG with different params
    rem = time_budget - (time.monotonic() - t0)
    if rem > 10:
        t_sg2 = rem * 0.3
        a_sg2, obj_sg2 = subgradient_smoothed(
            best_a, time_limit=t_sg2, lr_init=0.001, momentum=0.94,
            n_top=max(40, n // 5), seed=seed + 77777,
            smooth_beta=100.0
        )
        if obj_sg2 < best_obj:
            best_obj = obj_sg2
            best_a = a_sg2.copy()

    # Phase 4: Multiplicative update
    rem = time_budget - (time.monotonic() - t0)
    if rem > 8:
        t_mult = min(rem * 0.2, 30.0)
        a_mult, obj_mult = multiplicative_update(
            best_a, time_limit=t_mult, n_peaks=max(10, n // 15), lr=0.005, seed=seed + 31337
        )
        if obj_mult < best_obj:
            best_obj = obj_mult
            best_a = a_mult.copy()

    # Phase 5: Final LP
    rem = time_budget - (time.monotonic() - t0)
    if rem > 5:
        a_lp2, obj_lp2 = lp_refine_active(
            best_a, time_limit=min(rem, 25.0), max_iters=15, n_active_frac=0.08
        )
        if obj_lp2 < best_obj:
            best_obj = obj_lp2
            best_a = a_lp2.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses Blackman-Harris initializations, active-set LP, smoothed subgradient,
    multiplicative updates, multi-length exploration, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, budget_s - 30.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = make_uniform_init(1152)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"No parent, using uniform n=1152, score={best_score:.6f}")

    n_parent = len(parent)

    # === Phase 0: Evaluate diverse initializations ===
    print(f"\n{'='*60}")
    print(f"Phase 0: Evaluate Initializations")
    print(f"{'='*60}")
    n_base = n_parent

    init_dict = {
        "blackman_harris_4": make_blackman_harris(n_base, 4),
        "blackman_harris_5": make_blackman_harris(n_base, 5),
        "nuttall": make_nuttall(n_base),
        "gauss_0.15": make_gaussian_init(n_base, 0.15),
        "gauss_0.25": make_gaussian_init(n_base, 0.25),
        "gauss_0.35": make_gaussian_init(n_base, 0.35),
        "cosine": make_cosine_init(n_base),
        "tukey_0.3": make_tukey_init(n_base, 0.3),
        "tukey_0.5": make_tukey_init(n_base, 0.5),
        "uniform": make_uniform_init(n_base),
    }

    init_scores = []
    for name, init_a in init_dict.items():
        s = float(evaluate_sequence(init_a.tolist()))
        init_scores.append((s, name, init_a))
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_seq = init_a.copy()
            print(f"  {name}: {s:.8f} - NEW BEST")
        else:
            print(f"  {name}: {s:.8f}")

    init_scores.sort()

    # === Phase 1: Deep optimization on parent ===
    print(f"\n{'='*60}")
    print(f"Phase 1: Deep optimization on parent (n={n_parent})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic() - 400
    if rem > 30:
        t_alloc = min(rem, 180.0)
        print(f"  Combined: n={n_parent}, t={t_alloc:.0f}s")
        a_opt, obj_opt = combined_optimize(parent, t_alloc, n_parent, seed)
        s = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_seq = a_opt.copy()
            print(f"  NEW BEST [phase1]: {s:.8f}")

    # === Phase 2: Optimize best initializations ===
    print(f"\n{'='*60}")
    print(f"Phase 2: Optimize Top Initializations (best={best_score:.6f})")
    print(f"{'='*60}")
    total_time_diverse = min(300.0, (deadline - time.monotonic() - 300.0) * 0.45)
    time_per_init = total_time_diverse / min(6, len(init_scores))

    for rank, (score_i, name, init_a) in enumerate(init_scores[:6]):
        rem = deadline - time.monotonic() - 250
        if rem < 15:
            break
        t_alloc = min(time_per_init, rem * 0.35)

        print(f"  [{rank}] {name} (init={score_i:.6f}, t={t_alloc:.0f}s)")
        a_div, obj_div = combined_optimize(init_a, t_alloc, n_base, seed + rank * 12345 + 100)
        s = float(evaluate_sequence(a_div.tolist()))

                    best_seq, time_limit=min(rem, 40.0), max_iters=20, n_active_frac=0.1

                s2 = float(evaluate_sequence(a_lp_extra.tolist()))
                if np.isfinite(s2) and s2 < best_score:
                    best_score = s2
                    best_seq = a_lp_extra.copy()

    if best_seq is not None:
