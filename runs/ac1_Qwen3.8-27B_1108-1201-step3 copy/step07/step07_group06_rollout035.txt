import numpy as np

from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_grad_for_lag(a, k):
    """Compute gradient of c[k] w.r.t. a. c[k] = sum_j a[j]*a[k-j]."""
    n = len(a)
    g = np.zeros(n)

    return exp_z / s


def compute_topk_grad(a, c, K=3):
    """Average gradient over top-K peaks."""
    n = len(a)
    top_idx = np.argsort(c)[::-1][:K]
    g = np.zeros(n)
    for k in top_idx:
        g += compute_grad_for_lag(a, int(k))
    g /= K
    return g


def adam_logspace_exact(a0, time_limit, lr=0.01, beta1=0.9, beta2=0.999,
                        eps=1e-8, K=3, restart_every=5000, verbose=False):
    """
    Adam optimizer in log-space using exact subgradient (top-K peak smoothing).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    a = a / a.sum()
    z = np.log(a)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0
    current_lr = lr
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit:
        t += 1
        a_curr = softmax_stable(z)

        c = fast_convolve(a_curr)
        g_a = compute_topk_grad(a_curr, c, K)

        # Transform to log-space gradient
        S = np.dot(g_a, a_curr)
        g_z = a_curr * (g_a - S)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_z
        v = beta2 * v + (1.0 - beta2) * (g_z * g_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - current_lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = softmax_stable(z)
        score = 2.0 * n * fast_convolve(a_new).max()

        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
            # Slightly increase lr on improvement (carefully)
            current_lr = min(0.05, current_lr * 1.001)
        else:
            no_improve += 1

        # Adaptive restart
        if no_improve > restart_every:
            z = np.log(np.maximum(best_a, 1e-300)) + rng.standard_normal(n) * 0.02
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            current_lr = max(0.001, current_lr * 0.6)

        if verbose and t % 5000 == 0:
            print(f"        t={t} score={best_score:.7f} lr={current_lr:.5f} K={K}")

    return best_a, best_score


def coordinate_descent(a0, time_limit, n_sweeps=200, factors=None):
    """
    Coordinate descent: for each element, try multiplicative updates and keep best.
    """
    if factors is None:
        factors = [0.6, 0.75, 0.85, 0.92, 1.0, 1.08, 1.15, 1.3, 1.5]
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0, 2.0 * n * fast_convolve(a0).max()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        order = rng.permutation(n)
        sweep_improved = False

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-18:
                continue

            best_local_score = best_score
            best_factor = 1.0

            for f in factors:
                a_trial = a.copy()
                a_trial[i] = a[i] * f
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_local_score - 1e-15:
                    best_local_score = score
                    best_factor = f

            if best_factor != 1.0:
                a[i] *= best_factor
                a = a / a.sum()
                if best_local_score < best_score:
                    best_score = best_local_score
                    best_a = a.copy()
                    sweep_improved = True

        if not sweep_improved and sweep > 5:
            break

    return best_a, best_score


def peak_pair_shave(a0, time_limit, n_trials=5000):
    """
    Peak-targeted pair adjustment: find the max peak and try to reduce it
    by adjusting pairs of elements that contribute to it.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), 2.0 * n * fast_convolve(a0).max()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        # Find the max peak (prefer interior)
        c_search = c.copy()
        c_search[0] = -np.inf
        c_search[-1] = -np.inf
        k_star = int(np.argmax(c_search))
        peak_val = c[k_star]
        target = best_score / (2.0 * n)
        if peak_val <= target * 1.0005:
            # All peaks are balanced; try random perturbation
            if trial % 100 == 0:
                a = best_a + rng.standard_normal(n) * 0.005
                a = np.maximum(a, 0)
                a = a / a.sum()
            continue

        # Elements contributing to this peak
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[k_star - idx]
        # Get top contributors
        top_n = min(8, len(idx))
        top_order = np.argsort(contribs)[::-1][:top_n]

        improved = False
        for ci in top_order:
            i = int(idx[ci])
            j = int(k_star - i)
            if j < 0 or j >= n or j == i:
                continue
            if a[i] < 1e-18 or a[j] < 1e-18:
                continue

            # Try reducing the pair product
            for f1 in [0.85, 0.75, 0.6]:
                for f2 in [1.0, 1.05, 1.1]:
                    a_trial = a.copy()
                    a_trial[i] *= f1
                    a_trial[j] *= f2
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a_trial.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Try single element reduction with redistribution
            for ci in top_order[:4]:
                i = int(idx[ci])
                if a[i] < 1e-18:
                    continue
                for factor in [0.7, 0.5, 0.3]:
                    a_trial = a.copy()
                    diff = a_trial[i] * (1.0 - factor)
                    a_trial[i] *= factor
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        continue
                    a_trial[mask] += diff * a_trial[mask] / others_sum
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a_trial.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved and trial % 200 == 0:
            # Stuck - random restart from best
            a = best_a + rng.standard_normal(n) * 0.01
            a = np.maximum(a, 0)
            a = a / a.sum()

    return best_a, best_score


def simulated_annealing(a0, time_limit, n_iter=10000):
    """
    Simulated annealing: random perturbations with temperature-based acceptance.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    T_init = 0.01
    T_final = 0.0001

    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break
        progress = it / n_iter
        T = T_init * (T_final / T_init) ** progress

        # Random perturbation
        scale = 0.02 * (1.0 - 0.5 * progress)
        a_trial = a + rng.standard_normal(n) * scale
        a_trial = np.maximum(a_trial, 0)
        ss = a_trial.sum()
        if ss < 1e-15:
            continue
        a_trial = a_trial / ss
        score = 2.0 * n * fast_convolve(a_trial).max()

        delta = score - best_score
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_trial
            if score < best_score:
                best_score = score
                best_a = a_trial.copy()

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def generate_difference_set_init(n, rng):
    """
    Generate an initialization inspired by difference sets.
    Creates a sequence with structured "blocks" that approximate low autocorrelation.
    """
    # Use a cosines-based construction that has good autocorrelation properties
    x = np.linspace(0, 1, n)
    # Multi-frequency cosine (Welch cost minimization inspired)
    a = np.ones(n)
    for freq in [2, 3, 5, 7, 11]:
        a += 0.3 * np.cos(2 * np.pi * freq * x + rng.standard_normal() * 0.5)
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def generate_diverse_inits(parent, n, rng):
    """Generate diverse initializations."""
    candidates = []

    if parent is not None:
        p = np.maximum(parent, 0)
        if len(p) != n:
            p = interpolate_sequence(p, n)
        p = p / p.sum()
        candidates.append(('parent', p.copy()))

        for scale in [0.02, 0.05, 0.1, 0.2]:
            pert = p + rng.standard_normal(n) * scale
            pert = np.maximum(pert, 0)
            pert = pert / pert.sum()
            candidates.append((f'pert_{scale}', pert))

    # Structured inits
    x = np.linspace(0, 1, n)
    # Uniform
    candidates.append(('uniform', np.ones(n) / n))

    # Cosine taper
    cos_t = 0.5 + 0.5 * np.cos(np.pi * x)
    cos_t = np.maximum(cos_t, 0.01)
    candidates.append(('cos_taper', cos_t / cos_t.sum()))

    # Multi-frequency
    mf = 1.0
    for f in [2, 3, 5, 8]:
        mf += 0.2 * np.cos(2 * np.pi * f * x)
    mf = np.maximum(mf, 0.01)
    candidates.append(('multifreq', mf / mf.sum()))

    # Difference-set-inspired
    candidates.append(('diffset', generate_difference_set_init(n, rng)))

    # Random
    rand = np.abs(rng.standard_normal(n)) + 0.01
    candidates.append(('random', rand / rand.sum()))

    # Bimodal
    mid = n // 2
    bi = np.exp(-((x - 0.3) ** 2) / 0.02) + np.exp(-((x - 0.7) ** 2) / 0.02)
    bi = np.maximum(bi, 0.01)
    candidates.append(('bimodal', bi / bi.sum()))

    return candidates


def evaluate_and_track(a, label, best_seq, best_score, print_flag=True):
    """Evaluate a candidate and return updated best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if print_flag:
            print(f"  NEW BEST [{label}]: {score:.7f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):

    parent = None

            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.7f}")

            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_default = 768
        uniform = np.ones(n_default) / n_default
        best_seq = uniform
        best_score = float(evaluate_sequence(uniform.tolist()))
        print(f"Default: n={n_default}, score={best_score:.7f}")

    print(f"\n=== Phase 1: Exact-Subgradient Adam ===")

        # Try multiple K values and learning rates

    # === Phase 2: Focused refinement (20% time) ===
    print(f"\n=== Phase 2: Focused Refinement (best={best_score:.7f}) ===")

                                         factors=[0.9, 0.95, 1.0, 1.05, 1.1])

        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.7f}")

    return [float(x) for x in best_seq]
