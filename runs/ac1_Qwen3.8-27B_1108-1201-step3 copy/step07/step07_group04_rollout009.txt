import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - s)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - s) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.full(n, s / n)


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def build_conv_matrix_symmetric(a_full):
    """Build matrix for palindromic b: T_sym @ b_half = conv(a_full, b_full)."""
    n = len(a_full)
    half = (n + 1) // 2
    T_full = build_conv_matrix(a_full)
    if n % 2 == 0:
        # b_full = [b_half, flip(b_half)]
        T_sym = T_full[:, :half] + T_full[:, half:][:, ::-1]
    else:
        # b_full = [b_half, flip(b_half[1:])]
        T_sym = T_full[:, :half].copy()
        for j in range(1, half):
            T_sym[:, j] += T_full[:, n - 1 - j]
    return T_sym

def nesterov_log_optimize(a0, time_limit, T_init=80.0, T_min=0.5, lr_init=0.01):
    """Nesterov-accelerated momentum in log-space with adaptive temperature."""
    n = len(a0)
    a_safe = np.maximum(a0, 1e-25)
    z = np.log(a_safe)
    z = z - z.mean()

    m = np.zeros(n)
    beta = 0.9
    lr = lr_init
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()
    t0 = monotonic()
    iters = 0
    no_improve = 0

    while monotonic() - t0 < time_limit:
        iters += 1
        T = max(T_min, T_init * np.exp(-iters / 800.0))

        # Nesterov lookahead
        z_y = z + beta * m

        # Convert to probability
        z_y_shifted = z_y - z_y.max()
        exp_z = np.exp(z_y_shifted)
        a_y = exp_z / exp_z.sum()

        # Compute smoothed gradient
        c = fast_convolve(a_y)
        max_c = c.max()
        if max_c < 1e-30:
            break
        scaled = (c - max_c) / T
        exp_c = np.exp(scaled)
        w = exp_c / exp_c.sum()

        # Gradient in a-space
        grad_a = compute_weighted_gradient(w, a_y, n)

        # Transform to z-space: grad_z = a * (grad_a - a * (grad_a . a))
        S = np.dot(grad_a, a_y)
        grad_z = a_y * (grad_a - a_y * S)

        # Momentum update
        m = beta * m - lr * grad_z
        z = z + m
        z = z - z.mean()

        # Convert back
        z_shifted = z - z.max()
        a_new = np.exp(z_shifted)
        a_new = a_new / a_new.sum()
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a0.copy()
        else:
            a_new = a_new / s_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive learning rate
        if no_improve > 200:
            lr *= 0.5
            beta = max(0.7, beta - 0.05)
            no_improve = 0
            # Perturb if stuck
            if no_improve > 500:
                z += np.random.standard_normal(n) * 0.01
                z = z - z.mean()
                no_improve = 0

        if iters % 500 == 0:
            elapsed = monotonic() - t0
            print(f"    Nesterov {elapsed:.0f}s: score={best_score:.6f} T={T:.2f} lr={lr:.5f}")

    return best_a, best_score

                for frac in [0.1, 0.2, 0.3]:
                    amount = a[i] * frac
                    # Choose target positions randomly or at low-contribution spots
                    targets = np.random.choice(n, size=3, replace=False)
                    for tgt in targets:
                        a_trial = a.copy()
                        a_trial[i] -= amount / 3
                        a_trial[tgt] += amount / 3
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        # Strategy 4: Small random perturbation
        if not improved:
            for attempt in range(3):
                a_trial = a + np.random.standard_normal(n) * 0.001
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            failed += 1
            if failed > 80:
                break
        else:
            failed = 0

    return best_a, best_score


def lbgfs_smooth(a0, time_limit, T=100, maxiter=5000):
    """L-BFGS refinement in log-space with soft-max smoothing."""
    n = len(a0)
    a_safe = np.maximum(a0, 1e-30)
    z0 = np.log(a_safe)
    z0 = z0 - z0.mean()
    t0 = monotonic()
    stopped = [False]

    def func_jac(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            f = 1e15
            g = np.zeros(n)
        else:
            z_shifted = z - z.max()
            exp_z = np.exp(z_shifted)
            a_local = exp_z / exp_z.sum()
            c = fast_convolve(a_local)
            max_c = c.max()
            scaled_c = (c - max_c) / T
            exp_c = np.exp(scaled_c)
            w = exp_c / exp_c.sum()
            lse = max_c + T * np.log(exp_c.sum())
            f = 2.0 * n * lse / T

            grad_a = compute_weighted_gradient(w, a_local, n) / (2.0 * n)
            S = np.dot(grad_a, a_local)
            g = a_local * (grad_a - a_local * S)

        return f, g

    try:
        def fun(z):
            f, g = func_jac(z)
            return f

        def jac(z):
            f, g = func_jac(z)
            return g

        res = minimize(
            fun=fun, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 10, 'disp': False}
        )
        if not stopped[0]:
            z_final = res.x
            z_shifted = z_final - z_final.max()
            a_out = np.exp(z_shifted)
            a_out = a_out / a_out.sum()
        else:
            a_out = a0.copy()
    except Exception:
        a_out = a0.copy()

    return a_out


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate and return updated best."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. LP fictitious play (symmetric + asymmetric)
    2. Nesterov-accelerated log-space optimization
    3. LP line search refinement
    4. Peak shaving
    5. L-BFGS polish
    6. Repeat with remaining time
    """
    np.random.seed(seed)
    deadline = monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = p_score
    else:
        best_seq = np.ones(768) / 768.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Initial: n=768, score={best_score:.6f}")

    n_parent = len(best_seq)

    # === Phase 1: LP Fictitious Play ===
    print(f"\n=== Phase 1: Fictitious Play (asymmetric) ===")
    n_try = n_parent
    rem = deadline - monotonic() - 600
    if rem > 60:
        t_alloc = min(rem * 0.6, 300.0)
        print(f"  FP asymmetric n={n_try}, budget={t_alloc:.0f}s")
        a_fp, score_fp, iters = fictitious_play(n_try, t_alloc, initial=best_seq, solver=solver, verbose=True)
        best_seq, best_score = check_and_update(a_fp, f"FP_asym_n{n_try}", best_seq, best_score)

    # FP with symmetry constraint
    print(f"\n=== Phase 1b: Fictitious Play (symmetric) ===")
    rem = deadline - monotonic() - 500
    if rem > 40:
        t_alloc = min(rem * 0.5, 200.0)
        # Make best_seq palindromic for symmetric LP
        n_cur = len(best_seq)
        a_sym = 0.5 * (best_seq + best_seq[::-1])
        a_sym = a_sym / a_sym.sum()
        print(f"  FP symmetric n={n_cur}, budget={t_alloc:.0f}s")
        a_fps, score_fps, iters_s = fictitious_play(n_cur, t_alloc, initial=a_sym, solver=solver, verbose=True, use_symmetry=True)
        best_seq, best_score = check_and_update(a_fps, f"FP_sym_n{n_cur}", best_seq, best_score)

    # === Phase 2: Nesterov log-space optimization ===
    print(f"\n=== Phase 2: Nesterov Log-Opt (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 400
        if rem > 30:
            t_alloc = min(rem * 0.6, 350.0)
            print(f"  Nesterov n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_ne, score_ne = nesterov_log_optimize(best_seq, t_alloc, T_init=60.0, T_min=0.3, lr_init=0.008)
            best_seq, best_score = check_and_update(a_ne, "nesterov", best_seq, best_score)

    # === Phase 3: LP Line Search Refinement ===
    print(f"\n=== Phase 3: LP Line Search (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 300
        if rem > 30:
            t_alloc = min(rem * 0.6, 200.0)
            print(f"  DampedLP n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_dl, _ = damped_lp_refine(best_seq, t_alloc, solver=solver)
            best_seq, best_score = check_and_update(a_dl, "damped_lp", best_seq, best_score)

    # === Phase 4: Nesterov again (second pass with lower T) ===
    print(f"\n=== Phase 4: Nesterov Pass 2 (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 200
        if rem > 20:
            t_alloc = min(rem * 0.6, 150.0)
            print(f"  Nesterov2 n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_ne2, score_ne2 = nesterov_log_optimize(best_seq, t_alloc, T_init=20.0, T_min=0.1, lr_init=0.004)
            best_seq, best_score = check_and_update(a_ne2, "nesterov2", best_seq, best_score)

    # === Phase 5: Peak Shaving ===
    print(f"\n=== Phase 5: Peak Shaving (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 150
        if rem > 10:
            t_alloc = min(rem * 0.6, 100.0)
            print(f"  PeakShave n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_ps, _ = peak_shave(best_seq, t_alloc)
            best_seq, best_score = check_and_update(a_ps, "peak_shave", best_seq, best_score)

    # === Phase 6: L-BFGS Polish ===
    print(f"\n=== Phase 6: L-BFGS Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        for T_val in [50, 150]:
            rem = deadline - monotonic() - 80
            if rem < 15:
                break
            t_alloc = min(rem * 0.4, 60.0)
            print(f"  L-BFGS T={T_val}, budget={t_alloc:.0f}s")
            a_lb = lbgfs_smooth(best_seq, t_alloc, T=T_val, maxiter=3000)
            best_seq, best_score = check_and_update(a_lb, f"lbgfs_T{T_val}", best_seq, best_score)

    # === Phase 7: Try slightly different length ===
    print(f"\n=== Phase 7: Length Exploration (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_cur = len(best_seq)
        rem = deadline - monotonic() - 60
        if rem > 50:
            # Try n+128
            alt_n = n_cur + 128
            t_alloc = min(rem * 0.5, 80.0)
            init_alt = interpolate_sequence(best_seq, alt_n)
            print(f"  FP alt n={alt_n}, budget={t_alloc:.0f}s")
            a_alt, _, _ = fictitious_play(alt_n, t_alloc, initial=init_alt, solver=solver, verbose=False)
            best_seq, best_score = check_and_update(a_alt, f"FP_alt_n{alt_n}", best_seq, best_score)

    # === Phase 8: Final rapid polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 10
        if rem > 10:
            t_ps = min(rem * 0.5, 20.0)
            a_ps2, _ = peak_shave(best_seq, t_ps)
            best_seq, best_score = check_and_update(a_ps2, "final_ps", best_seq, best_score)

        rem = deadline - monotonic() - 5
        if rem > 5:
            t_ne = min(rem * 0.8, 10.0)
            a_ne3, _ = nesterov_log_optimize(best_seq, t_ne, T_init=10.0, T_min=0.05, lr_init=0.002)
            best_seq, best_score = check_and_update(a_ne3, "final_nesterov", best_seq, best_score)

    # Final verification
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
