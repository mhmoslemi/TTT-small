    Iteratively solve the linearized LP:
    min t s.t. c + 2*T(f)*g <= t, g >= -f, sum(g) = 0
    where T(f)*g = conv(f,g) (the convolution constraint matrix times g).
    Take a step in direction g and re-linearize.
    """
    n = len(f)
    a = f / f.sum()
    best_a = a.copy()
    best_obj = max_conv(a)
    
    for it in range(max_iters):
        c = convolve(a)  # auto-convolution
        M0 = c.max()
        
        # Build constraint matrix
        A = np.zeros((2*n-1, n))
        for k in range(2*n-1):
            lo = max(0, k-n+1)
            hi = min(n-1, k)
            A[k, lo:hi+1] = a[lo:hi+1]
        
        # LP: min t

        #      g >= -a  (i.e., -g <= a)
        #      sum(g) = 0
        
        # In standard form for linprog:
        # min [0, ..., 0, 1] @ [g, t]
        # s.t. [[A, -I], [-I, 0]] @ [g, t] <= [-c, a]  (asymmetric constraints)
        #      [1, ..., 1, 0] @ [g, t] = 0
        
        c_vec = np.zeros(n+1)
        c_vec[-1] = 1.0  # minimize t
        
        # Inequality constraints: A @ g - t <= -c  =>  [A, -1] @ [g,t] <= -c

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. Linearized LP (sequential linear programming for peak reduction)
2. Smooth gradient descent (log-sum-exp smoothing of max)
3. FFT subgradient with momentum
4. Multi-length exploration
5. Coordinate descent polish

Key innovation: Direct linearized LP that minimizes the first-order
approximation of the peak, providing principled descent directions.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix T such that T @ g = conv(a, g)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        T[k, lo:hi + 1] = a[k - hi:k - lo + 1][::-1]
    return T


def linearized_lp_optimize(a0, time_limit=120.0, max_iters=200, verbose=False):
    """
    Sequential linearized LP for peak reduction.
    At each step, solve:
        min t
        s.t. c + 2*T(a)@g <= t
             g >= -a
             sum(g) = 0
    Then take a line search step in direction g.
    """
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = float(fast_convolve(a).max())
    
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_convolve(a)
        M0 = c.max()
        
        # Build constraint matrix
        T = build_conv_matrix(a)
        
        # LP: min t
        # s.t. 2*T@g - t <= -c    (2*(2n-1) constraints)
        #      -g <= a             (n constraints)
        #      sum(g) = 0          (1 equality)
        
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        
        # Inequality constraints
        A_ub = np.hstack([2.0 * T, -np.ones((2 * n - 1, 1))])
        b_ub_top = -c
        A_ub = np.vstack([A_ub, np.hstack([-np.eye(n), np.zeros((n, 1))])])
        b_ub = np.concatenate([b_ub_top, a])
        
        # Equality constraint
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([0.0])
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          method='highs', options={'presolve': True})
        except Exception:
            break
        
        if not res.success:
            break
        
        g = res.x[:n]
        t_star = res.x[-1]
        
        if t_star >= M0 - 1e-12:
            # No first-order improvement
            no_improve += 1
            if no_improve > 5:
                # Try random perturbation
                rng = np.random.default_rng(it * 7919 + 13)
                a = a + rng.standard_normal(n) * 1e-4
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                no_improve = 0
                continue
            break
        
        # Line search
        best_alpha = 0.0
        best_alpha_obj = best_obj
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_try = a + alpha * g
            a_try = np.maximum(a_try, 0.0)
            ss = a_try.sum()
            if ss < 1e-10:
                continue
            a_try = a_try / ss
            obj = float(fast_convolve(a_try).max())
            if obj < best_alpha_obj:
                best_alpha_obj = obj
                best_alpha = alpha
        
        if best_alpha > 0 and best_alpha_obj < best_obj - 1e-14:
            a = a + best_alpha * g
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            obj = float(fast_convolve(a).max())
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
                if verbose:
                    print(f"      LP it{it}: t*={t_star:.10f}, obj={best_obj:.10f}, alpha={best_alpha:.2f}")
        else:
            no_improve += 1
            if no_improve > 3:
                break
    
    return best_a, best_obj


def smooth_gradient_optimize(a0, time_limit=60.0, tau_init=0.01, decay=0.97,
                              lr=0.01, momentum=0.9, max_iters=5000, seed=42):
    """
    Smooth gradient descent using log-sum-exp approximation of max.
    Minimizes smooth_max(conv(a,a), tau) subject to a >= 0, sum(a) = 1.
    """
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = float(fast_convolve(a).max())
    tau = tau_init
    no_improve = 0
    
    # FFT size for gradient computation
    fft_size = 1 << (3 * n - 2).bit_length()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute convolution
        b = fast_convolve(a)
        
        # Log-sum-exp weights
        b_shifted = b - b.max()
        w_full = np.exp(b_shifted / tau)
        w_full /= w_full.sum()
        
        # Gradient: grad_i = 2 * sum_k w_full[k] * a_contribution
        # The gradient of conv(a,a)[k] w.r.t. a[i] is a[k-i] + a[i-k] (for valid indices)
        # So grad[i] = 2 * sum_k w_full[k] * (a[k-i] if valid) 
        # which is 2 * conv(a, w_full)[i]
        
        A = rfft(a, fft_size)
        W = rfft(w_full, fft_size)
        g = irfft(np.conj(A) * W, fft_size)[:n]
        g = 2.0 * g
        
        # Momentum step
        v = momentum * v - lr * g
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        
        obj = float(fast_convolve(a_new).max())
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        tau *= decay
        if no_improve > 500:
            lr *= 0.9
            if lr < 1e-6:
                break
    
    return best_a, best_obj


def subgradient_optimize_fft(a0, time_limit, lr_init=0.005, momentum=0.9,

        a = a / S

    best_a = a.copy()
    best_obj = float(fast_convolve(a).max())
    cur_lr = lr_init
    no_improve = 0
    
    if n_top is None:
        n_top = max(10, n // 10)
    
    fft_size = 1 << (3 * n - 2).bit_length()
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        b_max = b.max()
        
        if n_top < len(b):

            active_mask[0] = True

        W = rfft(w, fft_size)
        g = irfft(np.conj(A) * W, fft_size)[:n]

        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        
        obj = float(fast_convolve(a_new).max())
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        if no_improve >= 3000:
            a = best_a + rng.standard_normal(n) * 0.003
            a = np.maximum(a, 0.0)
