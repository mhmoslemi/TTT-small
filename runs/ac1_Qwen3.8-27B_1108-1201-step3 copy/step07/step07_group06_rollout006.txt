"""
Multi-phase optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Combines Adam log-space gradient descent with projected subgradient on hard max,
direct peak-targeted local search, and multi-length exploration.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def get_score(a):
    """Compute the objective score for a sequence."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - np.max(z)
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    return np.log(np.maximum(a, 1e-300))


def project_simplex(v, target_sum=1.0):
    """Project v onto {a >= 0, sum(a) = target_sum}."""
    a = np.maximum(v, 0)
    s = a.sum()
    if s > 1e-15:
        a = a * target_sum / s
    else:
        a = np.ones_like(v) * target_sum / len(v)
    return a


def adam_logspace_v2(a0, time_limit, T_init=0.05, T_final=0.0005, lr=0.01,
                     beta1=0.9, beta2=0.999, eps=1e-8, schedule='cosine',
                     verbose=False, verbose_interval=2000):
    """
    Adam optimizer in log-space with temperature annealing (v2 with cosine schedule).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    s = a.sum()
    a = a / s
    z = z_from_a(a)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    fft_size = 1 << (2 * n - 2).bit_length()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    no_improve = 0
    max_no_improve = 5000

    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)

        # Cosine temperature schedule
        if schedule == 'cosine':
            T = T_final + 0.5 * (T_init - T_final) * (1 + np.cos(np.pi * progress))
        elif schedule == 'exp':
            T = T_init * (T_final / T_init) ** progress
        elif schedule == 'linear':
            T = T_init + (T_final - T_init) * progress
        else:
            T = T_init * (T_final / T_init) ** progress

        # Cosine learning rate schedule
        lr_t = 0.01 * lr * (0.5 * (1 + np.cos(np.pi * progress))) + 0.01 * lr

        a_curr = softmax_from_z(z)
        b = fast_convolve(a_curr)
        max_b = b.max()
        exp_b = np.exp(b / T - max_b / T)
        w = exp_b / exp_b.sum()

        # Gradient
        W = rfft(w, fft_size)
        A = rfft(a_curr, fft_size)
        corr = irfft(W * np.conj(A), fft_size)[:n]
        grad_a = 2.0 * n * corr

        S = np.dot(grad_a, a_curr)
        grad_z = a_curr * (grad_a - S)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)

        # Evaluate
        a_new = softmax_from_z(z)
        conv_new = fast_convolve(a_new)

            v = np.zeros(n)
            no_improve = 0

        if verbose and t % verbose_interval == 0:
            print(f"      Adam t={t} T={T:.6f} lr={lr_t:.6f} score={best_score:.8f}")

    return best_a, best_score


def projected_subgradient(a0, time_limit, lr_init=0.001, lr_decay=0.9999,
                          n_inner=50, verbose=False):
    """
    Projected subgradient method on the hard max objective.
    Directly uses the peak lag gradient.
    """

    a = project_simplex(np.asarray(a0, dtype=np.float64), 1.0)
    best_a = a.copy()

    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        k_peak = int(np.argmax(b))
        
        # Gradient of b[k_peak] w.r.t. a
        # b[k_peak] = sum_i a[i] * a[k_peak - i] (for valid i)
        # d(b[k_peak])/da[i] = a[k_peak - i]
        i_start = max(0, k_peak - n + 1)
        i_end = min(n - 1, k_peak)
        
        grad = np.zeros(n)
        for i in range(i_start, i_end + 1):
            j = k_peak - i
            if 0 <= j < n:
                grad[i] += a[j]
                if j != i:
                    grad[j] += a[i]
        
        # Actually for the symmetric case, grad[i] = a[k_peak - i] (if valid)
        # Let me be more careful:
        # b[k] = sum_{i: 0<=i<n, 0<=k-i<n} a[i]*a[k-i]
        # db[k]/da[i] = a[k-i] (if 0 <= k-i < n)
        # But since b[k] = b[2(n-1)-k] by symmetry of autocorrelation,
        # and we're looking at the max which could be at k or 2(n-1)-k...
        
        # Wait, linear conv of a with a: (a*a)[k] = sum_i a[i]*a[k-i]
        # This is NOT symmetric in general! a*a[k] = a*a[2(n-1)-k] only if...
        # Actually, (a*a)[k] = sum_i a[i]*a[k-i].
        # (a*a)[2(n-1)-k] = sum_i a[i]*a[2(n-1)-k-i]
        # Let j = 2(n-1)-k-i, then i = 2(n-1)-k-j
        # = sum_j a[2(n-1)-k-j]*a[j] = sum_j a[j]*a[2(n-1)-k-j]
        # This equals (a*a)[k] only if a is symmetric. In general, (a*a) IS symmetric!
        # Because (a*a)[k] = sum_i a[i]*a[k-i] and (a*a)[m] = sum_i a[i]*a[m-i].
        # Let's check: is (a*a)[k] = (a*a)[2(n-1)-k]?
        # (a*a)[k] = sum_{i=0}^{n-1} a[i]*a[k-i] (where a[x]=0 if x<0 or x>=n)
        # (a*a)[2(n-1)-k] = sum_{i=0}^{n-1} a[i]*a[2(n-1)-k-i]
        # Let j = 2(n-1)-k-i, so i = 2(n-1)-k-j
        # = sum_j a[2(n-1)-k-j]*a[j] = sum_j a[j]*a[2(n-1)-k-j]
        # Now, a[2(n-1)-k-j] = a[2(n-1)-k-j]. This is a[k'+j'] where k' = 2(n-1)-k...
        # Actually this IS the same as (a*a)[k] by checking:
        # (a*a)[k] has terms a[i]*a[k-i] for i where both indices are in [0,n-1].
        # (a*a)[2(n-1)-k] has terms a[i]*a[2(n-1)-k-i] for i where both are in [0,n-1].
        # The set of valid (i, k-i) pairs for (a*a)[k] is the same as the set of valid
        # (i, 2(n-1)-k-i) pairs for (a*a)[2(n-1)-k], just with indices swapped.
        # So yes, (a*a)[k] = (a*a)[2(n-1)-k]. The autocorrelation IS symmetric.
        
        # So the peak is at k_peak, and also at 2(n-1)-k_peak.
        # The gradient is the same for both (by symmetry).
        
        # So: grad[i] = a[k_peak - i] if 0 <= k_peak - i < n, else 0.
        grad = np.zeros(n)

        else:
            no_improve += 1

def spectral_flatten(a0, time_limit, n_iter=10000, verbose=False):
    """
    Spectral approach: try to flatten the power spectrum |FFT(a)|^2.
    If the power spectrum is flat, the autocorrelation is a delta (ideal).
    We iteratively adjust a to make |FFT(a)|^2 more uniform, while
    maintaining a >= 0 and sum(a) = 1.
    """
    n = len(a0)
    a = project_simplex(np.asarray(a0, dtype=np.float64), 1.0)
    best_a = a.copy()
    best_score = get_score(a)
    t0 = time.monotonic()
    fft_size = 2 * n - 1
    fft_size_pow = 1 << (fft_size - 1).bit_length()
    
    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute power spectrum
        A = rfft(a, fft_size_pow)
        P = A * np.conj(A)  # Power spectrum
        abs_P = np.abs(A)
        
        # Target: uniform power spectrum
        # We want |A[k]| to be approximately constant
        # Adjust: A_new[k] = A[k] * (target / |A[k]|)^alpha
        # where target is the mean |A[k]|
        target = np.mean(abs_P)
        alpha = 0.1  # Blending factor
        
        # Adjust the amplitude of the spectrum
        scale = (target / (abs_P + 1e-30)) ** alpha
        A_new = A * scale
        
        # Inverse FFT and project
        a_trial = irfft(A_new, fft_size_pow)[:n]
        a_trial = project_simplex(a_trial, 1.0)
        
        new_score = get_score(a_trial)
        if new_score < best_score - 1e-14:
            a = a_trial
            best_a = a_trial.copy()
            best_score = new_score
        
        if verbose and it % 1000 == 0:
            print(f"      Spectral: it={it}, score={best_score:.8f}")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
