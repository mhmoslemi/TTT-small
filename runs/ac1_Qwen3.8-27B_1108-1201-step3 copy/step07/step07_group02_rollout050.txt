Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using focused L-BFGS with FFT-based
gradients, aggressive peak shaving, and random restarts.
Key innovation: FFT-accelerated gradient computation enables much faster

# Global state for objective/gradient functions

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fft_cross_corr(w, a):
    """Compute cross-correlation (w ⋆ a)[i] = sum_j w[i+j]*a[j] for i=0..n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size + n - 2 - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    result = irfft(np.conj(W) * A, fft_size)
    return result[:n]

def obj_fn(z):
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = _g_T * b
    m = Tb.max()
    e = np.exp(Tb - m)
    lse = m + np.log(e.sum())
    return (2.0 * _g_n / _g_T) * lse

def jac_fn(z):

    b = fast_convolve(a)
    Tb = _g_T * b
    m = Tb.max()
    e = np.exp(Tb - m)
    w = e / e.sum()
    # grad_a = 4n * cross_correlation(w, a)
    grad_a = 4.0 * _g_n * fft_cross_corr(w, a)
    G = 4.0 * _g_n * np.dot(w, b)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n

    old_x = np.linspace(0, 1, old_n)

    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:

    return f_interp

Minimizes 2*n*max(conv(a,a))/(sum(a))² using FFT-accelerated L-BFGS with
geometric temperature annealing, aggressive peak shaving with multiple
strategies, and random restarts for escaping local minima.
Key: FFT-based cross-correlation for O(N log N) gradient computation.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

_g_T = 100.0
_g_n = 768


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fft_cross_corr(w, a):
    n = len(a)
    needed = 3 * n - 2
    fft_size = 1 << (needed - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    result = irfft(np.conj(W) * A, fft_size)
    return result[:n]


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def obj_fn(z):
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = _g_T * b
    m = Tb.max()
    e = np.exp(Tb - m)
    lse = m + np.log(e.sum())
    return (2.0 * _g_n / _g_T) * lse


def jac_fn(z):
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = _g_T * b
    m = Tb.max()
    e = np.exp(Tb - m)
    w = e / e.sum()
    grad_a = 4.0 * _g_n * fft_cross_corr(w, a)
    G = 4.0 * _g_n * np.dot(w, b)
    return a * (grad_a - G)

def lbgfs_run(a0, T_start, T_end, time_limit, maxiter=5000, n_stages=5):
    global _g_T, _g_n
    _g_n = len(a0)
    z = z_from_a(a0)
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(10, maxiter // n_stages)

def peak_shave(a0, time_limit, n_steps=200000):

            break

        je = min(n - 1, rel_peak)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[rel_peak - idx]
        order = np.argsort(contribs)[::-1]

        improved = False

        for ci in order[:6]:
            i = int(idx[ci])
            if a[i] < 1e-12:
                continue
            for factor in [0.97, 0.93, 0.88, 0.80]:

                diff = a[i] - new_val

                    continue

                a_trial[i] = new_val
                mask = a_trial > 0
                mask[i] = False
                osum = a_trial[mask].sum()
                if osum < 1e-15:

                a_trial[mask] += diff * a_trial[mask] / osum

                    a_trial[i] *= f
