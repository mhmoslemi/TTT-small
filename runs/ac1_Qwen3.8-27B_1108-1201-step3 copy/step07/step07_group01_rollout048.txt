from scipy.optimize import linprog

def fast_convolve(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def lp_direction(a, M):
    """Solve LP to find best direction.
    max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0
    """
    n = len(a)
    # Build constraint matrix A_ub where A_ub @ b <= M (vector of M's)
    # (a*b)[k] = sum_j b[j] * a[k-j] (with appropriate bounds)
    # For each k, the constraint is: sum_j b[j] * a[k-j] <= M
    # where a[k-j] = 0 if k-j is out of [0, n-1]
    
    # A_ub has shape (2n-1, n)
    # A_ub[k, j] = a[k-j] if 0 <= k-j < n, else 0
    
    # This is a Toeplitz-like matrix
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        # A_ub[k, j] = a[k-j] for valid j
        js = max(0, k - n + 1)
        je = min(n-1, k)
        if js <= je:

    # Objective: minimize -sum(b)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    
    if res.success:

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding

simple subgradient methods.

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_from_a(a):
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())

def build_conv_matrix(n, a):
    """Build the (2n-1) x n matrix C where (C @ b)[k] = (a*b)[k]."""
    C = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        js = max(0, k - n + 1)
        je = min(n-1, k)
        if js <= je:
            C[k, js:je+1] = a[k - np.arange(js, je+1)]
    return C

    """Solve LP: max sum(b) s.t. a*b <= M, b >= 0."""
    n = len(a)
    C = build_conv_matrix(n, a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    res = linprog(c, A_ub=C, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if res.success and res.x is not None:
        b = np.maximum(res.x, 0)
        if b.sum() > 1e-15:
            return b / b.sum()
    return None

    best_a = a.copy()

def fast_convolve(a):

def build_conv_matrix(a):

    C = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        js = max(0, k - n + 1)
        je = min(n-1, k)
        if js <= je:
            C[k, js:je+1] = a[k - np.arange(js, je+1)]
    return C


def lp_solve(a, M):
    """Solve: max sum(b) s.t. C@b <= M*1, b >= 0. Returns normalized b."""
    n = len(a)
    C = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    try:
        res = linprog(c, A_ub=C, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
        if res.success and res.x is not None:
            b = np.maximum(res.x, 0.0)
            s = b.sum()
            if s > 1e-10:
                return b / s
    except Exception:
        pass
    return None

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using the LP-based direction finding
approach (solving a linear program at each step to find the optimal descent
direction), combined with smooth log-sum-exp gradient descent for escaping
local minima, and coordinate descent polishing. The LP approach is the key
novelty: at each iteration it solves max sum(b) s.t. a*b <= M, b>=0 to find
the best direction to reduce the peak auto-convolution, providing a much
stronger signal than simple subgradient methods.
"""
import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_from_a(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def build_conv_matrix(a):
    """Build (2n-1) x n matrix C where (C @ b)[k] = (a*b)[k]."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            C[k, js:je + 1] = a[k - np.arange(js, je + 1)]
    return C


def lp_solve(a, M):
    """Solve LP: max sum(b) s.t. C@b <= M*1, b >= 0. Returns normalized b."""
    n = len(a)
    C = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=C, b_ub=b_ub, bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if res.success and res.x is not None:
            b = np.maximum(res.x, 0.0)
            s = b.sum()
            if s > 1e-10:
                return b / s
    except Exception:
        pass
    return None


def lp_optimize(a_init, time_limit, verbose=True):
    """LP-based optimization with adaptive line search."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    it = 0
    no_improve_count = 0
    max_no_improve = 100

    while time.monotonic() - t0 < time_limit - 3:
        it += 1
        if time.monotonic() - t0 > time_limit - 3:
            break

        M = fast_convolve(a).max()
        b = lp_solve(a, M)
        if b is None:
            break

        # Line search over step sizes (try large steps first for speed)
        improved = False
        for t in [1.0, 0.7, 0.5, 0.4, 0.3, 0.2, 0.1]:
            a_trial = (1.0 - t) * a + t * b
            a_trial = np.maximum(a_trial, 0.0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            sc = score_from_a(a_trial)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a_trial.copy()
                a = a_trial.copy()
                improved = True
                no_improve_count = 0
                break

        if not improved:
            no_improve_count += 1
            if no_improve_count > max_no_improve:
                break
            # Small perturbation to escape stagnation
            if no_improve_count % 30 == 0:
                rng = np.random.default_rng(it * 104729)
                noise = rng.standard_normal(n) * 0.005
                a = best_a + noise
                a = np.maximum(a, 1e-15)
                a = a / a.sum()

        if verbose and it % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LP iter={it} ({elapsed:.1f}s): score={best_score:.7f}")

    return best_a, best_score


def lse_gradient_descent(a_init, time_limit, T_start=10.0, T_end=0.5,
                         lr_start=0.05, lr_end=0.001, max_iters=20000,
                         verbose=True):
    """Gradient descent on smooth log-sum-exp approximation with mirror updates."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, it / max_iters)
        T = T_start * (T_end / T_start) ** progress
        lr = lr_start * (lr_end / lr_start) ** progress

        b = fast_convolve(a)
        b_scaled = b / T
        b_max = b_scaled.max()
        w = np.exp(b_scaled - b_max)
        w = w / w.sum()

        # Compute gradient: grad_a[i] = 4n * sum_j a[j] * w[j+i]
        fft_size = 1 << (3 * n - 2).bit_length()
        A = rfft(a, fft_size)
        W = rfft(w, fft_size)
        corr = irfft(np.conj(A) * W, fft_size)
        grad_a = 4.0 * n * corr[:n]

        # Project to tangent space of simplex
        mean_grad = np.dot(a, grad_a)
        grad_a = grad_a - mean_grad

        # Multiplicative (mirror) update for simplex
        log_update = -lr * grad_a
        log_update = np.clip(log_update, -5.0, 5.0)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-300)
        s = a.sum()
        a = a / s

        sc = score_from_a(a)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a.copy()

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LSE-GD iter={it} ({elapsed:.1f}s): T={T:.3f} lr={lr:.5f} best={best_score:.7f}")

    return best_a, best_score


def coordinate_descent_polish(a_init, time_limit, verbose=True):
    """Targeted coordinate descent focusing on peak-contributing indices."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()

    factors = [0.7, 0.8, 0.85, 0.9, 0.95, 1.05, 1.1, 1.3, 1.5]
    stagnation = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        contrib_idx = np.arange(js, je + 1)
        contribs = a[contrib_idx] * a[peak_k - contrib_idx]
        top_n = min(30, len(contrib_idx))
        top_idx = np.argsort(contribs)[::-1][:top_n]

        for ci in top_idx:
            i = int(contrib_idx[ci])
            if a[i] < 1e-12:
                continue
            for f in factors:
                a_trial = a.copy()
                old = a_trial[i]
                a_trial[i] = old * f
                diff = old - a_trial[i]
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] = others + diff * others / others_sum
                if a_trial.min() < -1e-10:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                sc = score_from_a(a_trial)
                if sc < best_score - 1e-13:
                    best_score = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try paired reductions
            for ci1 in top_idx[:8]:
                for ci2 in top_idx[:8]:
                    i1 = int(contrib_idx[ci1])
                    i2 = int(contrib_idx[ci2])
                    if i1 == i2:
                        continue
                    a_trial = a.copy()
                    a_trial[i1] *= 0.85
                    a_trial[i2] *= 0.85
                    a_trial = np.maximum(a_trial, 0.0)
                    a_trial = a_trial / a_trial.sum()
                    sc = score_from_a(a_trial)
                    if sc < best_score - 1e-13:
                        best_score = sc
                        best_a = a_trial.copy()
                        a = a_trial.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            stagnation += 1
            if stagnation > 50:
                break
        else:
            stagnation = 0

        if verbose and (stagnation % 100 == 0 or improved):
            elapsed = time.monotonic() - t0
            if int(elapsed) % 5 == 0:
                print(f"  CD ({elapsed:.0f}s): best={best_score:.7f} stag={stagnation}")

    return best_a, best_score


def peak_shave(a_init, time_limit, verbose=True):
    """Aggressive peak shaving: iteratively reduce the largest peak contributors."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()

    for step in range(5000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        # Find the maximum peak (not at endpoints)
        b_work = b.copy()
        if n > 2:
            b_work[0] = -1
            b_work[-1] = -1
        peak_k = int(np.argmax(b_work))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_idx = np.argsort(contribs)[::-1][:10]

        improved = False
        for ci in top_idx:
            i = int(idx[ci])
            for factor in [0.9, 0.8, 0.7, 0.6]:
                a_trial = a.copy()
                old = a_trial[i]
                a_trial[i] = old * factor
                diff = old - a_trial[i]
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                osum = others.sum()
                if osum < 1e-15:
                    continue
                a_trial[mask] = others + diff * others / osum
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                sc = score_from_a(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate using evaluate_sequence."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.7f} (n={len(a_norm)})")
    return best_seq, best_score
