        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient_smooth(a, w):
    """
    Compute gradient of log-sum-exp approximation.
    
    a: sequence of length n (sum to 1)
    w: weights of length 2n-1 (softmax of T*b)
    
    Returns gradient of shape (n,).
    """
    n = len(a)
    # ∇f[i] = 2 * sum_m w[i+m] * a[m]
    # This is cross-correlation of w and a at lag i
    # Using FFT: IFFT(FFT(w_padded) * conj(FFT(a_padded)))
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    grad = irfft(W * np.conj(A), fft_size)[:n]
    return 2.0 * grad


def smooth_gradient_descent(f_init, time_limit, T_init=50.0, T_max=500.0, 
                             lr_init=0.01, momentum=0.9):
    """
    Smooth gradient descent with log-sum-exp approximation.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    v = np.zeros(n)  # momentum buffer
    lr = lr_init
    T = T_init
    stuck = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(f)
        T_b = T * b
        T_b -= T_b.max()  # numerical stability
        w = np.exp(T_b)
        w /= w.sum()
        
        grad = compute_gradient_smooth(f, w)
        
        # Momentum update
        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)
        
        # Check improvement
        obj_new = fast_convolve(f_new).max()
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            stuck = 0
            f = f_new
        else:
            f = f_new
            stuck += 1
        
        # Adaptive learning rate and temperature
        if stuck > 20:
            lr *= 0.9
            stuck = 0
        elif stuck == 0:
            lr = min(lr * 1.02, 0.05)
        
        # Anneal temperature
        T = min(T * 1.005, T_max)
    
    return best_f, best_obj


def lp_direction(f, top_k):
    """
    Find a good direction using LP relaxation.
    Returns (g_normalized, success).
    """
    n = len(f)
    b = fast_convolve(f)
    M = b.max()
    if M < 1e-15:
        return None, False
    
    if top_k < len(b):
        top_idx = np.argpartition(b, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(b))
    top_idx = np.sort(top_idx)
    K = len(top_idx)
    
    # Build constraint matrix
    A = np.zeros((K, n))
    for row_i in range(K):
        k = top_idx[row_i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]
    
    g = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [g >= 0, A @ g <= M])
    try:
        prob.solve(solver=cp.SCS, max_iters=200000, eps=1e-7)
    except Exception:
        return None, False
    
    if g.value is None:
        return None, False
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None, False
    
    S = g_val.sum()
    if S <= 1.0 + 1e-5:
        return None, False
    
    return g_val / S, True


def lp_phase(f_init, time_limit, top_k=50):
    """LP-based optimization phase."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    stuck = 0
    
    while time.monotonic() - t0 < time_limit:
        g, success = lp_direction(f, top_k)
        if not success:
            # Try with fewer constraints
            g, success = lp_direction(f, max(10, top_k // 2))
            if not success:
                break
        
        # Line search
        best_t = 0.05
        best_t_obj = best_obj
        for t in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
            f_trial = (1.0 - t) * f + t * g
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t
        
        f = (1.0 - best_t) * f + best_t * g
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s
        
        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 30:
                break
    
    return best_f, best_obj


def coordinate_descent(f_init, time_limit, step_sizes=None):
    """Coordinate descent polish."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    
    if step_sizes is None:
        step_sizes = [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]
    
    improved = True
    passes = 0
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        passes += 1
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-15:
                continue
            best_j_val = f[j]
            best_j_obj = best_obj
            for mult in step_sizes:
                new_val = f[j] * mult
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f /= s
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    improved = True
    
    return f, best_obj

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

Combines smooth gradient descent (log-sum-exp approximation) as the main engine,
LP-based direction finding for escaping local minima, mirror descent variants,
multi-length exploration, and coordinate descent for final polish.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_via_fft(w, a):
    """
    Compute gradient vector g[i] = 2 * sum_m w[i+m] * a[m].
    Uses FFT-based cross-correlation.
    w: shape (2n-1,), a: shape (n,)
    Returns: shape (n,)
    """
    n = len(a)
    needed = (2 * n - 1) + n - 1  # 3n - 2
    fft_size = 1 << (needed - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    result = irfft(W * np.conj(A), fft_size)[:n]
    return 2.0 * result


def gradient_descent_smooth(f_init, time_limit, T_init=50.0, T_max=500.0,
                            lr_init=0.008, momentum=0.85, use_mirror=False,
                            label="gd"):
    """
    Smooth gradient descent with log-sum-exp approximation.
    Adaptive temperature annealing, learning rate, and restarts.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    v = np.zeros(n)
    lr = lr_init
    T = T_init
    stuck = 0
    it = 0
    rng = np.random.default_rng(777)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        Tb = T * b
        Tb -= Tb.max()
        w = np.exp(Tb)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(len(b))
            w[b.argmax()] = 1.0
        else:
            w /= w_sum
        
        grad = compute_grad_via_fft(w, f)
        
        if use_mirror:
            # Mirror descent: a_new[j] = a[j] * exp(-lr * grad[j]) / Z
            log_a = np.where(f > 1e-30, np.log(f), -30.0)
            z = log_a - lr * grad
            z -= z.max()  # numerical stability
            a_new = np.exp(z)
            a_sum = a_new.sum()
            if a_sum < 1e-30:
                a_new = f.copy()
            else:
                a_new /= a_sum
            f = a_new
        else:
            # Euclidean projection with momentum
            v = momentum * v - lr * grad
            f = project_simplex(f + v)
        
        # Evaluate
        if it % 5 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f.copy()
                stuck = 0
                lr = min(lr * 1.01, 0.03)
            else:
                stuck += 1
                if stuck > 50:
                    lr *= 0.9
                    stuck = 0
                if stuck > 200:
                    # Restart with perturbation
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                    v = np.zeros(n)
                    lr = lr_init * 0.5
                    stuck = 0
        
        # Anneal temperature
        T = min(T * 1.003, T_max)
        
        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} it={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} "
                  f"T={T:.1f} lr={lr:.5f} t={elapsed:.0f}s")
    
    return best_f, best_obj


def lp_direction(f, top_k):
    """
    Find a good direction using LP relaxation.
    Returns (g_normalized, success).
    """
    n = len(f)
    b = fast_convolve(f)
    M = b.max()
    if M < 1e-15:
        return None, False
    
    if top_k < len(b):
        top_idx = np.argpartition(b, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(b))
    top_idx = np.sort(top_idx)
    K = len(top_idx)
    
    A = np.zeros((K, n))
    for row_i in range(K):
        k = top_idx[row_i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]
    
    g = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [g >= 0, A @ g <= M])
    try:
        prob.solve(solver=cp.SCS, max_iters=100000, eps=1e-7)
    except Exception:
        return None, False
    
    if g.value is None:
        return None, False
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None, False
    
    S = g_val.sum()
    if S <= 1.0 + 1e-5:
        return None, False
    
    return g_val / S, True


def lp_phase(f_init, time_limit, top_k=50):
    """LP-based optimization phase."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    stuck = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        g, success = lp_direction(f, top_k)
        if not success:
            g, success = lp_direction(f, max(20, top_k // 2))
            if not success:
                break
        
        # Line search
        best_t = 0.05
        best_t_obj = best_obj
        for t in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
            f_trial = (1.0 - t) * f + t * g
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t
        
        f = (1.0 - best_t) * f + best_t * g
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s
        
        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 20:
                break
        
        if it % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    lp it={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} "
                  f"t={elapsed:.0f}s")
    
    return best_f, best_obj


def coordinate_descent(f_init, time_limit):
    """Coordinate descent polish with adaptive step sizes."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    multipliers = [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]
    
    improved = True
    passes = 0
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        passes += 1
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-15:
                continue
            best_j_val = f[j]
            best_j_obj = best_obj
            for mult in multipliers:
                new_val = f[j] * mult
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f /= s
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    improved = True
    
    return f, best_obj


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main optimization: smooth gradient descent as primary engine,
    LP-based direction finding for escaping local minima,
    mirror descent variant, multi-length exploration, and coordinate descent.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass
    
    if parent is None:
        parent = np.ones(1024) / 1024
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"No parent, using uniform n=1024, score={p_score:.6f}")
    
    best_seq = parent.copy()
    best_score = p_score
    
    # === Phase 1: Smooth GD main (Euclidean projection) ===
    n_main = len(parent)
    print(f"\n=== Phase 1: Smooth GD main (n={n_main}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.40, 720.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        f1, obj1 = gradient_descent_smooth(parent, time_limit=t1,
                                            T_init=40.0, T_max=400.0,
                                            lr_init=0.01, momentum=0.85,
                                            label="gd_main")
        score1 = float(evaluate_sequence(f1.tolist()))
        if np.isfinite(score1) and score1 < best_score:
            best_score = score1
            best_seq = f1.copy()
            print(f"  NEW BEST [gd_main]: {score1:.6f}")
    
    # === Phase 2: Mirror descent variant ===
    print(f"\n=== Phase 2: Mirror descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.20, 350.0)
    if t2 > 30:
        print(f"  Budget: {t2:.0f}s")
        f2, obj2 = gradient_descent_smooth(best_seq, time_limit=t2,
                                            T_init=80.0, T_max=600.0,
                                            lr_init=0.004, momentum=0.0,
                                            use_mirror=True, label="mirror")
        score2 = float(evaluate_sequence(f2.tolist()))
        if np.isfinite(score2) and score2 < best_score:
            best_score = score2
            best_seq = f2.copy()
            print(f"  NEW BEST [mirror]: {score2:.6f}")
    
    # === Phase 3: LP-based refinement ===
    print(f"\n=== Phase 3: LP refinement (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.15, 250.0)
    if t3 > 30:
        print(f"  Budget: {t3:.0f}s")
        f3, obj3 = lp_phase(best_seq, time_limit=t3, top_k=60)
        score3 = float(evaluate_sequence(f3.tolist()))
        if np.isfinite(score3) and score3 < best_score:
            best_score = score3
            best_seq = f3.copy()
            print(f"  NEW BEST [lp]: {score3:.6f}")
    
    # === Phase 4: High-temperature GD (fine-tuning) ===
    print(f"\n=== Phase 4: High-T GD (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.15, 250.0)
    if t4 > 20:
        print(f"  Budget: {t4:.0f}s")
        f4, obj4 = gradient_descent_smooth(best_seq, time_limit=t4,
                                            T_init=300.0, T_max=2000.0,
                                            lr_init=0.003, momentum=0.90,
                                            label="gd_hT")
        score4 = float(evaluate_sequence(f4.tolist()))
        if np.isfinite(score4) and score4 < best_score:
            best_score = score4
            best_seq = f4.copy()
            print(f"  NEW BEST [gd_hT]: {score4:.6f}")
    
    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length (best={best_score:.6f}) ===")
    n_base = len(best_seq)
    target_lengths = []
    for factor in [0.9, 1.1, 1.2, 1.35]:
        tl = int(n_base * factor)
        tl = max(256, min(2500, tl))
        if abs(tl - n_base) > 10 and tl not in target_lengths:
            target_lengths.append(tl)
    
    for target_n in target_lengths[:3]:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t5 = min(rem * 0.3, 180.0)
        if t5 < 20:
            break
        
        old_x = np.linspace(0.0, 1.0, n_base)
        new_x = np.linspace(0.0, 1.0, target_n)
        f_interp = np.interp(new_x, old_x, best_seq)
        f_interp = np.maximum(f_interp, 0.0)
        s = f_interp.sum()
        if s > 1e-10:
            f_interp /= s
        
        print(f"  GD n={target_n}, budget={t5:.0f}s")
        f5, obj5 = gradient_descent_smooth(f_interp, time_limit=t5,
                                            T_init=50.0, T_max=500.0,
                                            lr_init=0.008, momentum=0.85,
                                            label=f"n{target_n}")
        score5 = float(evaluate_sequence(f5.tolist()))
        if np.isfinite(score5) and score5 < best_score:
            best_score = score5
            best_seq = f5.copy()
            print(f"  NEW BEST [n{target_n}]: {score5:.6f}")
    
    # === Phase 6: Coordinate descent ===
    print(f"\n=== Phase 6: Coordinate descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.15, 150.0)
    if t6 > 10:
        print(f"  Budget: {t6:.0f}s")
        f6, obj6 = coordinate_descent(best_seq, time_limit=t6)
        score6 = float(evaluate_sequence(f6.tolist()))
        if np.isfinite(score6) and score6 < best_score:
            best_score = score6
            best_seq = f6.copy()
            print(f"  NEW BEST [cd]: {score6:.6f}")
    
    # === Phase 7: Final burst ===
    print(f"\n=== Phase 7: Final burst (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t7 = min(rem * 0.8, 200.0)
    if t7 > 15:
        print(f"  Budget: {t7:.0f}s")
        f7, obj7 = gradient_descent_smooth(best_seq, time_limit=t7,
                                            T_init=200.0, T_max=1500.0,
                                            lr_init=0.002, momentum=0.92,
                                            label="final")
        score7 = float(evaluate_sequence(f7.tolist()))
        if np.isfinite(score7) and score7 < best_score:
            best_score = score7
            best_seq = f7.copy()
            print(f"  NEW BEST [final]: {score7:.6f}")
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
    
    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")
    
    return [float(x) for x in best_seq]
