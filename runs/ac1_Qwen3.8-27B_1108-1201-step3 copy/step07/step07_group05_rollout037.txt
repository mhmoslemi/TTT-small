```python
import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    """Compute 2*n*max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    conv = fast_convolve(a_norm)
    return 2.0 * n * conv.max()

def build_constraint_matrix(f):
    """Build the (2n-1) x n constraint matrix A where A[k,i] = f[k-i]."""
    n = len(f)
    A = np.zeros((2*n-1, n))
    for i in range(n):
        A[i:i+n, i] = f
    return A

def solve_best_response(f, M):
    """Solve the LP: max sum(b) s.t. A*b <= M*1, b >= 0."""
    n = len(f)
    A = build_constraint_matrix(f)
    c = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    b_ub = M * np.ones(2*n-1)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                     options={'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def fictitious_play(a0, time_limit, alpha=1.0, verbose=True):
    """Fictitious play with LP-based best response."""
    n = len(a0)
    a = a0.copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    # Running average
    a_avg = a.copy()
    t = 0
    
    t0 = time.monotonic()
    consecutive_no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        iter_start = time.monotonic()
        
        # Compute M
        conv = fast_convolve(a)
        M = conv.max()
        
        # Solve LP
        b = solve_best_response(a, M)
        if b is None:
            break
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        b = b / b_sum
        
        # Step size
        step = alpha / (t + 1)
        
        # Update
        a_new = (1 - step) * a + step * b
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        # Update running average
        a_avg = ((t * a_avg) + a_new) / (t + 1)
        
        # Evaluate
        score = compute_score(a_avg)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_avg.copy()
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1
        
        a = a_new
        
        if verbose and t % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"  FP iter {t}: M={M:.6f}, score={best_score:.8f}, "
                  f"LP_val={M*b_sum:.4f}, time={elapsed:.0f}s")
        
        # Early stopping
        if consecutive_no_improve > 100:
            break
    
    return best_a, best_score
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based fictitious play
combined with subgradient polishing and coordinate descent.

The core idea: at each iteration, solve the LP "max sum(b) s.t. (a*b)_k <= M, b>=0"
to find the best-response sequence, then update via convex combination. The running
average converges to the optimal sequence. Final polishing uses peak-focused
subgradient descent and coordinate moves.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max(conv(a,a)) for a (assumed normalized or will normalize)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    conv = fast_convolve(a_norm)
    return 2.0 * n * conv.max()


def cross_corr(a, w):
    """
    Compute g_i = sum_k w_k * a_{k-i} for i=0,...,n-1.
    w has length 2n-1, a has length n.
    Uses FFT for efficiency.
    """
    n = len(a)
    a_rev = a[::-1]
    total_len = len(w) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = rfft(w, fft_size)
    Ar = rfft(a_rev, fft_size)
    conv_r = irfft(W * Ar, fft_size)
    # g_i = conv_r[n-1+i]
    return conv_r[n - 1:n - 1 + n]


def build_constraint_matrix(f):
    """Build A (2n-1 x n) where A[k,i] = f[k-i] if valid."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        A[i:i + n, i] = f
    return A


def solve_best_response(f, M, use_lp=True):
    """
    Solve LP: max sum(b) s.t. A*f_conv b <= M, b >= 0.
    Returns the optimal b (normalized to sum 1) or None.
    """
    n = len(f)
    A = build_constraint_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    if use_lp:
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'presolve': True})
            if res.success:
                b = np.maximum(res.x, 0)
                s = b.sum()
                if s > 1e-15:
                    return b / s
        except Exception:
            pass

    # Fallback: subgradient method on dual
    # Dual: min M*sum(y) s.t. A^T y >= 1, y >= 0
    # Use projected gradient
    y = np.ones(2 * n - 1) / (2 * n - 1)
    eta = 1.0 / M
    for _ in range(20):
        # v_i = sum_k A[k,i] * y_k = cross-correlation
        v = cross_corr(f, y)
        # Subgradient of objective: M * ones
        # Projected gradient step on dual
        # The dual constraint is v_i >= 1
        # Penalty method: y_new = max(0, y - eta*(M - A^T * max(0, 1-v)))
        violation = np.maximum(0, 1 - v)
        # A^T * violation is cross-correlation of violation (padded) with... 
        # Actually (A^T v)_k = sum_i A[k,i] * v_i = sum_i f[k-i] * v_i
        # This is cross_corr of f and v (but v has length n, f has length n)
        # (A^T v)_k = sum_{i: 0<=k-i<=n-1} f[k-i] * v_i
        # = cross_corr(f, v_pad)_k where v_pad is v placed in [0, n-1]
        # Actually it's the same as: for each k, sum_i f[k-i]*v_i
        # This is (f * v) in the convolution sense... let me just compute directly
        # (A^T v)_k = sum_i A[k,i]*v_i where A[k,i] = f[k-i] if 0<=k-i<n
        # = sum_{i=max(0,k-n+1)}^{min(n-1,k)} f[k-i]*v_i
        # This is the cross-correlation of f and v at lag k (with appropriate indexing)
        # = irfft(conj(FFT(f)) * FFT(v))[k] ... hmm
        # Let's just use the direct formula via the cross_corr function
        # cross_corr(a, w) computes sum_k w_k * a_{k-i}
        # We want sum_i f[k-i] * v_i for each k
        # Let's define f_rev[j] = f[n-1-j]
        # Then f[k-i] = f_rev[n-1-k+i]
        # sum_i f_rev[n-1-k+i] * v_i = (f_rev * v)_{n-1+k}... 
        # This is getting complicated. Let me just do it directly for the fallback.
        dual_grad = np.zeros(2 * n - 1)
        for i in range(n):
            if violation[i] > 0:
                dual_grad[i:i+n] += violation[i] * f
        y = np.maximum(0, y - eta * (np.ones(2*n-1) * M - dual_grad))
        y_sum = y.sum()
        if y_sum > 1e-15:
            y = y / y_sum

    # Extract primal from dual: b_i is positive where (A^T y)_i is close to 1
    v = cross_corr(f, y)
    b = np.zeros(n)
    b[v < 1.01] = 1.0
    s = b.sum()
    if s > 1e-15:
        return b / s
    return None


def fictitious_play(a0, time_limit, verbose=True):
    """
    LP-based fictitious play.
    Returns (best_sequence, best_score).
    """
    n = len(a0)
    a = np.maximum(a0, 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    a_avg = a.copy()
    t = 0
    t0 = time.monotonic()
    stagnant = 0
    last_best = best_score
    lp_fail_count = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        iter_t0 = time.monotonic()

        conv = fast_convolve(a)
        M = float(conv.max())
        if M < 1e-15:
            break

        b = solve_best_response(a, M)
        if b is None:
            lp_fail_count += 1
            if lp_fail_count > 5:
                break
            # Fallback: just use a small random perturbation
            b = a + np.random.default_rng(t).standard_normal(n) * 0.01
            b = np.maximum(b, 0)
            bs = b.sum()
            if bs < 1e-15:
                break
            b = b / bs
        else:
            lp_fail_count = 0

        # Step size (Harmonically decreasing)
        step = 1.0 / (t + 1)

        # Update
        a_new = (1.0 - step) * a + step * b
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new = a_new / sn

        # Update running average
        a_avg = ((t * a_avg) + a_new) / (t + 1)

        # Evaluate running average score
        score = compute_score(a_avg)
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_avg.copy()
            stagnant = 0
        else:
            stagnant += 1

        a = a_new

        if verbose and t % 20 == 0:
            elapsed = time.monotonic() - t0
            iter_time = time.monotonic() - iter_t0
            print(f"  FP t={t}, M={M:.8f}, best={best_score:.8f}, "
                  f"iter_time={iter_time*1000:.1f}ms, elapsed={elapsed:.0f}s")

        # Early stopping if stagnant for too long
        if stagnant > 300 and t > 100:
            break

    return best_a, best_score


def subgradient_polish(a0, time_limit, K=30, T=5.0, lr_init=0.02,
                       verbose=True):
    """
    Subgradient polishing focused on top peaks of autocorrelation.
    Uses the exact subgradient with softmax smoothing over top K peaks.
    """
    n = len(a0)
    a = np.maximum(a0, 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    lr = lr_init
    momentum = 0.0
    v = np.zeros(n)
    stagnant = 0
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1

        conv = fast_convolve(a)

        # Find top K peaks (focus on the region near the center for auto-corr)
        top_k = min(K, len(conv))
        peak_idx = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peak_idx]

        # Softmax weights over top peaks
        shift = peak_vals.max()
        exp_vals = np.exp(T * (peak_vals - shift))
        w = np.zeros(len(conv))
        total_w = exp_vals.sum()
        if total_w < 1e-300:
            w[peak_idx] = 1.0 / top_k
        else:
            w[peak_idx] = exp_vals / total_w

        # Subgradient: s_i = sum_k w_k * a_{k-i}
        s_grad = cross_corr(a, w)
        # Project to tangent space of simplex
        s_grad = s_grad - s_grad.mean()

        # Momentum (Nesterov-like)
        momentum = 0.9 * momentum + 0.1 * lr
        v = 0.9 * v + momentum * s_grad
        a_new = a - v
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new = a_new / sn

        score = compute_score(a_new)
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 50:
                # Reset and try smaller step
                lr *= 0.8
                v = np.zeros(n)
                stagnant = 0
                if lr < 1e-6:
                    break

        # Slowly decay learning rate
        lr = max(lr * 0.9999, 1e-6)

        if verbose and iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"  SG iter={iters}, best={best_score:.8f}, lr={lr:.2e}, "
                  f"elapsed={elapsed:.0f}s")

    return best_a, best_score


def coordinate_descent(a0, time_limit, n_trials=500, verbose=True):
    """
    Coordinate descent with random pairwise moves.
    """
    n = len(a0)
    a = np.maximum(a0, 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(54321)
    iters = 0
    stagnant = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False

        for _ in range(n_trials):
            if time.monotonic() - t0 > time_limit:
                break

            # Random pairwise move
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue

            # Try several transfer amounts
            base = max(a[i], a[j], 1e-8)
            for frac in [0.1, 0.3, 0.5, 0.2, 0.7]:
                t_val = frac * base
                a_trial = a.copy()
                a_trial[i] -= t_val
                a_trial[j] += t_val
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                sc = compute_score(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a_trial.copy()
                    improved = True
                    break
            if improved:
                continue

        if not improved:
            stagnant += 1
            if stagnant > 20:
                # Add noise to escape
                a = best_a + rng.standard_normal(n) * 0.005
                a = np.maximum(a, 0)
                a = a / a.sum()
                stagnant = 0
                if best_score > 1.502:  # only bother if we're not already good enough
                    pass
                else:
                    break
        else:
            stagnant = 0

        if verbose and iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"  CD iter={iters}, best={best_score:.8f}, elapsed={elapsed:.0f}s")

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length, preserving non-negativity and sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    if old_n == 1:
        return np.ones(new_n) / new_n
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def fast_subgradient_one_step(a, K, T, lr):
    """Single subgradient step (inlined for speed)."""
    n = len(a)
    conv = fast_convolve(a)
    top_k = min(K, len(conv))
    peak_idx = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peak_idx]
    shift = peak_vals.max()
    exp_vals = np.exp(T * (peak_vals - shift))
    w = np.zeros(len(conv))
    total_w = exp_vals.sum()
    if total_w < 1e-300:
        w[peak_idx] = 1.0 / top_k
    else:
        w[peak_idx] = exp_vals / total_w
    s_grad = cross_corr(a, w)
    s_grad = s_grad - s_grad.mean()
    a_new = a - lr * s_grad
    a_new = np.maximum(a_new, 0)
    sn = a_new.sum()
    if sn < 1e-15:
        return a
    return a_new / sn


def aggsubgrad_poly(a0, time_limit, K=20, T=3.0, verbose=True):
    """
    Accelerated subgradient (AGD-style) with Polyak step size.
    More aggressive than plain subgradient.
    """
    n = len(a0)
    a = np.maximum(a0, 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    y = a.copy()  # extrapolation point
    gamma = 1.0  # AGD parameter

    while time.monotonic() - t0 < time_limit:
        iters += 1

        # Compute subgradient at y
        conv_y = fast_convolve(y)
        top_k = min(K, len(conv_y))
        peak_idx = np.argpartition(conv_y, -top_k)[-top_k:]
        peak_vals = conv_y[peak_idx]
        shift = peak_vals.max()
        exp_vals = np.exp(T * (peak_vals - shift))
        w = np.zeros(len(conv_y))
        total_w = exp_vals.sum()
        if total_w < 1e-300:
            w[peak_idx] = 1.0 / top_k
        else:
            w[peak_idx] = exp_vals / total_w
        g = cross_corr(y, w)
        g = g - g.mean()

        # Polyak step size
        score_y = 2.0 * n * conv_y.max()
        gap = score_y - best_score
        g_norm_sq = np.dot(g, g)
        if g_norm_sq < 1e-30:
            break
        if gap > 1e-12:
            step = gap / g_norm_sq
        else:
            step = 1.0 / (iters + 1)
        step = min(step, 0.5)

        # AGD update: x_new = y - step * g
        x_new = y - step * g
        x_new = np.maximum(x_new, 0)
        sn = x_new.sum()
        if sn < 1e-15:
            break
        x_new = x_new / sn

        # Evaluate
        score_x = compute_score(x_new)
        if score_x < best_score - 1e-14:
            best_score = score_x
            best_a = x_new.copy()
            gamma = 1.0
        else:
            gamma *= 0.95

        # AGD extrapolation: y_new = x_new + (gamma_new/gamma - 1) * (x_new - y)
        gamma_new = 0.5 * (1 + np.sqrt(1 + 4 * gamma * gamma))
        if gamma > 0:
            coef = gamma_new / gamma - 1
        else:
            coef = 0
        y = x_new + coef * (x_new - y)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy
        gamma = gamma_new

        if verbose and iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"  AGD iter={iters}, best={best_score:.8f}, "
                  f"gap={gap:.2e}, elapsed={elapsed:.0f}s")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. LP-based fictitious play (main workhorse)
    2. Accelerated subgradient polishing
    3. Coordinate descent for fine-tuning
    4. Try nearby lengths
    5. Final polish
    """
    deadline = time.monotonic() + max(10.0, budget_s - 30.0)
    rng = np.random.default_rng(seed)

    # --- Get initial sequence ---
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = compute_score(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is None:
        n_init = 768
        parent = np.ones(n_init) / n_init
        # Add some structure
        t_arr = np.linspace(0, 1, n_init)
        parent = parent * (1 + 0.1 * np.cos(2 * np.pi * 3 * t_arr))
        parent = np.maximum(parent, 0)
        parent = parent / parent.sum()
        print(f"Initialized random: n={n_init}, score={compute_score(parent):.8f}")

    best_seq = parent.copy()
    best_score = compute_score(parent)

    n_main = len(parent)

    # --- Phase 1: Fictitious Play (main optimization) ---
    rem = deadline - time.monotonic()
    fp_budget = min(900.0, rem * 0.45)
    print(f"\n=== Phase 1: Fictitious Play ===")
    print(f"  n={n_main}, budget={fp_budget:.0f}s, start_score={best_score:.8f}")

    a_fp, sc_fp = fictitious_play(best_seq, fp_budget, verbose=True)
    if sc_fp < best_score:
        best_score = sc_fp
        best_seq = a_fp
        print(f"  FP improved: {best_score:.8f}")
    else:
        print(f"  FP no improvement from parent: {best_score:.8f}")

    # --- Phase 2: Try a second FP round with different params ---
    rem = deadline - time.monotonic()
    if rem > 100:
        fp_budget2 = min(400.0, rem * 0.3)
        print(f"\n=== Phase 2: FP Round 2 ===")
        print(f"  n={n_main}, budget={fp_budget2:.0f}s, current={best_score:.8f}")
        # Slightly perturb to escape
        a_perturbed = best_seq + rng.standard_normal(n_main) * 0.002
        a_perturbed = np.maximum(a_perturbed, 0)
        a_perturbed = a_perturbed / a_perturbed.sum()
        a_fp2, sc_fp2 = fictitious_play(a_perturbed, fp_budget2, verbose=True)
        if sc_fp2 < best_score:
            best_score = sc_fp2
            best_seq = a_fp2
            print(f"  FP2 improved: {best_score:.8f}")

    # --- Phase 3: Accelerated Subgradient ---
    rem = deadline - time.monotonic()
    if rem > 60:
        sg_budget = min(300.0, rem * 0.25)
        print(f"\n=== Phase 3: Accelerated Subgradient ===")
        print(f"  n={n_main}, budget={sg_budget:.0f}s, current={best_score:.8f}")
        a_agd, sc_agd = aggsubgrad_poly(best_seq, sg_budget, K=25, T=4.0, verbose=True)
        if sc_agd < best_score:
            best_score = sc_agd
            best_seq = a_agd
            print(f"  AGD improved: {best_score:.8f}")

    # --- Phase 4: Coordinate Descent ---
    rem = deadline - time.monotonic()
    if rem > 40:
        cd_budget = min(200.0, rem * 0.25)
        print(f"\n=== Phase 4: Coordinate Descent ===")
        print(f"  n={n_main}, budget={cd_budget:.0f}s, current={best_score:.8f}")
        a_cd, sc_cd = coordinate_descent(best_seq, cd_budget, n_trials=300, verbose=True)
        if sc_cd < best_score:
            best_score = sc_cd
            best_seq = a_cd
            print(f"  CD improved: {best_score:.8f}")

    # --- Phase 5: Try nearby lengths ---
    rem = deadline - time.monotonic()
    alt_lengths = [n_main + 128, n_main + 256, n_main + 512, n_main - 128]
    alt_lengths = [x for x in alt_lengths if 128 <= x <= 4096]
    
    print(f"\n=== Phase 5: Nearby Lengths ===")
    for alt_n in alt_lengths[:3]:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        t_alloc = min(rem * 0.2, 150.0)
        print(f"  Trying n={alt_n}, budget={t_alloc:.0f}s")
        alt_init = interpolate_seq(best_seq, alt_n)
        # Quick FP at new length
        a_alt, sc_alt = fictitious_play(alt_init, t_alloc, verbose=False)
        if sc_alt < best_score:
            best_score = sc_alt
            best_seq = a_alt
            print(f"  n={alt_n} improved: {best_score:.8f}")
        else:
            print(f"  n={alt_n} score={sc_alt:.8f} (no improvement)")

    # --- Phase 6: Final Polish (AGD + CD) ---
    rem = deadline - time.monotonic()
    if rem > 30:
        polish_budget = min(150.0, rem * 0.5)
        print(f"\n=== Phase 6: Final Polish ===")
        print(f"  budget={polish_budget:.0f}s, current={best_score:.8f}")
        # AGD
        a_p1, sc_p1 = aggsubgrad_poly(best_seq, polish_budget * 0.5, K=30, T=3.0, verbose=False)
        if sc_p1 < best_score:
            best_score = sc_p1
            best_seq = a_p1
        # CD
        rem2 = deadline - time.monotonic() - 10
        if rem2 > 20:
            a_p2, sc_p2 = coordinate_descent(best_seq, min(rem2 * 0.7, 60.0),
                                             n_trials=500, verbose=False)
            if sc_p2 < best_score:
                best_score = sc_p2
                best_seq = a_p2

    # --- Final verification via evaluate_sequence ---
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
```
