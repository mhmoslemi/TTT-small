    # Let's compute via FFT properly:
    # We want cross[i] = sum_k p[k] * a[k-i] for i = 0, ..., n-1
    # This equals the i-th element of the linear cross-correlation of p and a
    # Linear cross-correlation can be computed as:
    #   ifft(fft(p, N) * conj(fft(a, N))) ... no
    #   Actually: cross_corr(p, a)[i] = sum_k p[k+i] * a[k]
    #   = (p shifted by i) dot a
    #   This is equivalent to the i-th element of ifft(conj(fft(p,N)) * fft(a,N))
    #   Wait, I need to be careful about conventions.
    
    # Let me just verify numerically for small n, but for now use the FFT approach:
    # The cross-correlation r[i] = sum_k p[k] * a[k+i] (or a[k-i] depending on convention)
    # Using numpy: np.correlate(p, a, mode='full') gives the full cross-correlation
    # But for large n, FFT is better.
    
    # I'll use: r = ifft(conj(FFT(p)) * FFT(a))
    # This gives r[i] = sum_k p[k] * a[k-i + something] ... let me just test
    
    # For safety, let me compute it as a convolution:
    # grad_i = 2 * sum_k p[k] * a[k-i]
    # Let a_rev = a[::-1] (reversed)
    # Then grad_i = 2 * sum_k p[k] * a[k-i] = 2 * sum_k p[k] * a_rev[n-1-(k-i)]
    # = 2 * sum_k p[k] * a_rev[n-1-k+i]
    # Let m = n-1+i: grad_i = 2 * (p * a_rev)[m] where * is linear convolution
    # So: conv_pa = np.convolve(p, a_rev, mode='full')
    # grad = 2 * conv_pa[n-1 : n-1+n]
    
    a_rev = a[::-1]
    # Use FFT for the convolution
    conv_size = len(p) + n - 1
    fft_sz = 1 << (conv_size - 1).bit_length()
    P = rfft(p, fft_sz)
    AR = rfft(a_rev, fft_sz)
    conv_pa = irfft(P * AR, fft_sz)[:conv_size]
    grad = 2.0 * conv_pa[n-1:n-1+n]
    return grad

def objective_and_grad(x, tau):
    """Objective and gradient for L-BFGS (sum constraint handled by elimination)."""
    n = x.shape[0] + 1  # we eliminated one variable
    a = np.zeros(n)
    a[:n-1] = x
    a[n-1] = max(0, 1.0 - x.sum())
    
    conv = fast_convolve(a)
    conv_clipped = np.clip(conv / tau, -500, 500)
    m = conv_clipped.max()
    obj = tau * (m + np.log(np.exp(conv_clipped - m).sum()))
    
    # Gradient w.r.t. a
    p = np.exp(conv_clipped - m)
    p = p / p.sum()
    a_rev = a[::-1]
    conv_size = len(p) + n - 1
    fft_sz = 1 << (conv_size - 1).bit_length()
    P = rfft(p, fft_sz)
    AR = rfft(a_rev, fft_sz)
    conv_pa = irfft(P * AR, fft_sz)[:conv_size]
    grad_a = 2.0 * conv_pa[n-1:n-1+n]
    
    # Gradient w.r.t. x (first n-1 components, last is 1-sum(x))
    # d(a_j)/dx_i = delta_{ij} for j < n-1
    # d(a_{n-1})/dx_i = -1
    grad_x = grad_a[:n-1] - grad_a[n-1]
    return obj, grad_x

def soft_max_continuation(a0, time_limit, verbose=False):
    """Continuation method: decrease tau from large to small."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Start with large tau and decrease
    tau_init = max(fast_convolve(a).max(), 1e-6) * 2
    n_stages = 12
    taus = np.geomspace(tau_init, tau_init * 0.01, n_stages)
    
    # Also try a few more stages at small tau
    tau_min = tau_init * 0.001
    taus = np.append(taus, np.geomspace(tau_init * 0.01, max(tau_min, 1e-8), 8))
    
    x0 = a[:-1].copy()
    
    for idx, tau in enumerate(taus):
        if time.monotonic() - t0 > time_limit:
            break
        
        def fun(x):
            val, g = objective_and_grad(x, tau)
            return val, g
        
        try:
            res = minimize(fun, x0, method='L-BFGS-B', 
                          bounds=[(0, 1)] * (n-1),
                          options={'maxiter': 200, 'ftol': 1e-10, 'gtol': 1e-8})
            x0 = np.maximum(res.x, 0)
            # Normalize to ensure sum = 1
            a_trial = np.zeros(n)
            a_trial[:-1] = x0
            a_trial[-1] = max(0, 1.0 - x0.sum())
            st = a_trial.sum()
            if st > 1e-10:
                a_trial = a_trial / st
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score:
                    best_score = score
                    best_a = a_trial.copy()
            x0 = a_trial[:-1].copy()
        except Exception as e:
            if verbose:
                print(f"  L-BFGS error at tau={tau:.6f}: {e}")
            # Try a smaller step
            continue
        
        if verbose and idx % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"  SM-cont tau={tau:.6f} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score

def multiplicative_grad_descent(a0, time_limit, verbose=False):
    """Mirror descent (multiplicative weights) on the simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.05
    momentum = np.zeros(n)
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute subgradient of max(conv(a,a))
        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))
        
        # Subgradient: sg_i = 2 * a_{k_peak - i} for valid indices
        sg = np.zeros(n)
        i_start = max(0, k_peak - n + 1)
        i_end = min(n - 1, k_peak)
        for i in range(i_start, i_end + 1):
            sg[i] = 2.0 * a[k_peak - i]
        
        # Momentum
        momentum = 0.9 * momentum + 0.1 * sg
        
        # Multiplicative update
        a_new = a * np.exp(-lr * momentum)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        lr *= 0.99995
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"  MGD it={it} ({elapsed:.0f}s): best={best_score:.6f} lr={lr:.6f}")
    
    return best_a, best_score

def block_coord_descent(a0, time_limit, block_size=8, verbose=False):
    """Block coordinate descent: optimize blocks of consecutive coordinates."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    
    n_blocks = n // block_size
    remainder = n % block_size
    
    for sweep in range(500):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        perm = rng.permutation(n_blocks)
        for bi in perm:
            if time.monotonic() - t0 > time_limit:
                break
            start = bi * block_size
            end = start + block_size
            
            # Try scaling the entire block
            block_sum = a[start:end].sum()
            if block_sum < 1e-15:
                continue
            
            for factor in [0.5, 0.7, 0.85, 0.92, 1.08, 1.15, 1.3, 1.5, 2.0]:
                a_trial = a.copy()
                a_trial[start:end] *= factor
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-13:
                    best_score = score
                    best_a = a_trial.copy()
                    a = a_trial
                    improved = True
                    break
                elif factor > 1.0:
                    break
            
            # Try redistributing within the block
            if not improved:
                for _ in range(3):
                    i1 = start + rng.integers(0, block_size)
                    i2 = start + rng.integers(0, block_size)
                    if i1 == i2:
                        continue
                    a_trial = a.copy()
                    mass = min(a[i1], a[i2]) * rng.random() * 0.5
                    if mass < 1e-15:
                        continue
                    # Move mass from the larger to the smaller
                    if a[i1] > a[i2]:
                        a_trial[i1] -= mass
                        a_trial[i2] += mass
                    else:
                        a_trial[i2] -= mass
                        a_trial[i1] += mass
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-13:
                        best_score = score
                        best_a = a_trial.copy()
                        a = a_trial
                        improved = True
                        break
        
        if not improved and sweep > 10:
            break
    
    return best_a, best_score

def simulated_annealing(a0, time_limit, verbose=False):
    """Simulated annealing with structured moves."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    cur_score = best_score
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    
    T_init = 0.01
    T_final = 1e-6
    n_iters = 20000
    cooling = (T_final / T_init) ** (1.0 / n_iters)
    T = T_init
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        T *= cooling
        
        # Generate a random move
        move_type = rng.random()
        
        if move_type < 0.3:
            # Scale a random block
            block_len = rng.integers(4, min(32, n // 4))
            start = rng.integers(0, n - block_len + 1)
            factor = rng.uniform(0.3, 2.5)
            a_trial = a.copy()
            a_trial[start:start+block_len] *= factor
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                a_trial = a.copy()
            else:
                a_trial = a_trial / st
        elif move_type < 0.5:
            # Swap two random blocks
            blen1 = rng.integers(4, min(20, n // 5))
            blen2 = rng.integers(4, min(20, n // 5))
            s1 = rng.integers(0, n - blen1 + 1)
            s2 = rng.integers(0, n - blen2 + 1)
            a_trial = a.copy()
            a_trial[s1:s1+blen1], a_trial[s2:s2+blen2] = a[s2:s2+blen2].copy(), a[s1:s1+blen1].copy()
            st = a_trial.sum()
            if st > 1e-15:
                a_trial = a_trial / st
        elif move_type < 0.7:
            # Add noise to a random segment
            seg_len = rng.integers(10, min(50, n // 5))
            start = rng.integers(0, n - seg_len + 1)
            noise = rng.standard_normal(seg_len) * 0.1
            a_trial = a.copy()
            a_trial[start:start+seg_len] = np.maximum(a_trial[start:start+seg_len] + noise, 0)
            st = a_trial.sum()
            if st < 1e-15:
                a_trial = a.copy()
            else:
                a_trial = a_trial / st
        elif move_type < 0.85:
            # Mix with a random Fourier pattern
            a_trial = a.copy()
            freq = rng.integers(1, n // 4)
            phase = rng.uniform(0, 2 * np.pi)
            amplitude = rng.uniform(0.05, 0.2)
            t = np.arange(n)
            pattern = amplitude * np.cos(2 * np.pi * freq * t / n + phase)
            a_trial = np.maximum(a_trial + pattern, 0)
            st = a_trial.sum()
            if st > 1e-15:
                a_trial = a_trial / st
        else:
            # Global perturbation
            a_trial = a + rng.standard_normal(n) * 0.05
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 1e-15:
                a_trial = a_trial / st
            else:
                a_trial = a.copy()
        
        trial_score = 2.0 * n * fast_convolve(a_trial).max()
        
        # Accept/reject
        if trial_score < cur_score or rng.random() < np.exp(-(trial_score - cur_score) / T):
            a = a_trial
            cur_score = trial_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
        
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  SA it={it} ({elapsed:.0f}s): T={T:.6f} best={best_score:.6f}")
    
    return best_a, best_score

def pairwise_exchange(a0, time_limit, verbose=False):
    """Try exchanging mass between pairs of positions."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(555)
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Pick two random positions
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        
        # Try moving some mass from i to j (or vice versa)
        for frac in [0.1, 0.25, 0.5, 0.75, 0.9]:
            a_trial = a.copy()
            if a[i] > a[j]:
                mass = a[i] * frac
                a_trial[i] -= mass
                a_trial[j] += mass
            else:
                mass = a[j] * frac
                a_trial[j] -= mass
                a_trial[i] += mass
            
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-13:
                best_score = score
                best_a = a_trial.copy()
                a = a_trial
                break
    
    return best_a, best_score

def soft_max_subgrad(a0, time_limit, verbose=False):
    """Subgradient method with soft-max smoothing (all peaks weighted)."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    lr = 0.01
    momentum = np.zeros(n)
    beta = 0.9
    tau = 0.005 * max(fast_convolve(a).max(), 1e-6)
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(a)
        conv_norm = conv / max(conv.max(), 1e-10)
        
        # Soft weights
        p = np.exp(conv_norm / tau - 1.0)
        p = p / p.sum()
        
        # Gradient
        a_rev = a[::-1]
        conv_size = len(p) + n - 1
        fft_sz = 1 << (conv_size - 1).bit_length()
        P = rfft(p, fft_sz)
        AR = rfft(a_rev, fft_sz)
        conv_pa = irfft(P * AR, fft_sz)[:conv_size]
        grad = 2.0 * conv_pa[n-1:n-1+n]
        
        # Project gradient to tangent space of simplex (remove mean)
        grad_tangent = grad - grad.mean()
        
        # Momentum
        momentum = beta * momentum + (1-beta) * grad_tangent
        
        # Update
        a_new = a - lr * momentum
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        lr *= 0.9999
        tau *= 0.9998
        
        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  SM-sg it={it} ({elapsed:.0f}s): best={best_score:.6f} tau={tau:.6f}")
    
    return best_a, best_score

def nesterov_optimize(a0, time_limit, top_k=20, lr_init=0.006, verbose=False):
    """Nesterov-accelerated projected subgradient method."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    y = a.copy()
    v = np.zeros(n)
    lr = lr_init
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        t_k = (1 + np.sqrt(1 + 4 * (it - 1) ** 2)) / 2
        beta_k = (t_k - 1) / t_k
        
        conv = fast_convolve(y)
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        temp = 0.01 * max(pv.max(), 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                j = np.arange(i_start, i_end + 1)
                sg[j] += 2.0 * w[idx] * a[k - j]
        
        v = beta_k * v + sg
        norm_v = np.linalg.norm(v)
        if norm_v > 1e-15:
            v = v / norm_v
        
        y_new = y - lr * v
        # Project to simplex
        y_projected = simplex_project(y_new)
        
        score = 2.0 * n * fast_convolve(y_projected).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = y_projected.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        y = y_projected
        
        if no_improve > 200:
            lr *= 0.8
            no_improve = 0
        
        if no_improve > 600:
            noise = np.random.default_rng(it * 7 + 13).standard_normal(n) * 0.02
            y = simplex_project(best_a + noise)
            v = np.zeros(n)
            lr = max(lr_init * 0.5, lr * 1.5)
            no_improve = 0
        
        lr *= 0.9998
    
    return best_a, best_score

def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:

        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score
