def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return 2.0 * n  # uniform gives score 2
    a_norm = a / s
    conv = fast_convolve(a_norm)

def project_simplex(v, target_sum=1.0):
    """Project v onto {x : x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if target_sum <= 0:
        return np.zeros(n)
    # We need sum(max(v_i - tau, 0)) = target_sum
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0][-1]
    tau = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - tau, 0)

    """
    Compute d/d a [sum_k weights[k] * (a*a)[k]]

    weights has length 2n-1, a has length n.
    Result has length n.

    # But we need to be careful about the definition

    # But with zero-padding to avoid circular effects

    A = rfft(a, fft_size)

def downsample(a, new_n):
    """Downsample by averaging blocks."""
    n = len(a)
    n_blocks = n // new_n
    if n_blocks == 0:
        return a[:new_n].copy()
    a_blocks = a[:n_blocks * new_n].reshape(new_n, n_blocks)
    result = a_blocks.mean(axis=1)
    result = result / result.sum()
    return result

def upsample(a, new_n):
    """Upsample by linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    result = np.interp(new_x, old_x, a)
    result = np.maximum(result, 0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    return result

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding
combined with projected gradient descent and L-BFGS refinement.

Key innovations over previous approach:
1. LP-based minimax perturbation: directly solves for the optimal direction
   that minimizes the max auto-correlation (linearized), providing a principled step
2. Projected gradient with adaptive peak weighting (no log/softmax parameterization)
3. Multi-constraint LP with adaptive active set selection
4. Combined LP + gradient + L-BFGS pipeline with structured restarts
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute score assuming a is normalized (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def project_simplex(v, target_sum=1.0):
    """Project v onto {x : x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if target_sum <= 0:
        return np.zeros(n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - target_sum))[0]
    if len(rho_candidates) == 0:
        tau = v.max()
    else:
        rho = rho_candidates[-1]
        tau = (cssv[rho] - target_sum) / (rho + 1.0)
    result = np.maximum(v - tau, 0)
    s = result.sum()
    if s > 0:
        result = result * target_sum / s
    return result


def compute_conv_gradient(a, weights):
    """
    Compute gradient of sum_k weights[k]*conv[k] w.r.t. a.
    grad[i] = sum_k weights[k] * (a[i+k] + a[i-k]) with boundary handling.
    Uses FFT for efficiency.
    """
    n = len(a)
    # grad[i] = sum_k w[k] * (a[i+k] + a[i-k])
    # = correlate(w, a)[i] + correlate(w_reversed, a)[n-1-i] ... 
    # Actually: sum_k w[k]*a[i+k] is a cross-correlation
    # Let's just do it via direct FFT
    fft_size = 1 << (3 * n - 2).bit_length()
    W = rfft(weights, fft_size)
    A = rfft(a, fft_size)
    # sum_k w[k]*a[i+k] = ifft(conj(FFT(a)) * FFT(w))[:n]
    part1 = irfft(np.conj(A) * W, fft_size)[:n]
    # sum_k w[k]*a[i-k] = ifft(FFT(a) * conj(FFT(w)))[:n] ... 
    # Actually: sum_k w[k]*a[i-k]. Let j = i-k, so k = i-j.
    # = sum_j w[i-j]*a[j] = (w * a)[i] (linear convolution of w and a)
    # = ifft(FFT(w) * FFT(a))[:n]
    part2 = irfft(W * A, fft_size)[:n]
    return part1 + part2


def top_k_weights(conv, K, temperature=None):
    """Compute weights emphasizing top K peaks."""
    if temperature is not None:
        shifted = conv / temperature
        shifted -= shifted.max()
        w = np.exp(shifted)
        s = w.sum()
        if s > 0:
            return w / s
        return np.ones(len(conv)) / len(conv)
    else:
        idx = np.argpartition(conv, -min(K, len(conv)))[-min(K, len(conv)):]
        w = np.zeros(len(conv))
        w[idx] = 1.0 / len(idx)
        return w


def solve_lp_direction(a, num_constraints=100):
    """
    Solve LP: minimize t
    Subject to: for top num_constraints lags k:
        sum_i d[i]*(a[i+k]+a[i-k]) + conv[k] <= t
        d[i] >= -a[i]
        sum(d) = 0
    Returns (d, t_opt) or (None, -1).
    """
    n = len(a)
    conv = fast_convolve(a)
    total_lags = 2 * n - 1
    
    if num_constraints >= total_lags:
        idx = np.arange(total_lags)
    else:
        idx = np.argpartition(conv, -num_constraints)[-num_constraints:]
    
    num_k = len(idx)
    
    # Variables: [d_0, ..., d_{n-1}, t] -> n+1 vars
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    # Convolution constraints: g_k(d) - t <= -conv[k]
    A_conv = np.zeros((num_k, n + 1))
    b_conv = np.zeros(num_k)
    for row, k in enumerate(idx):
        for i in range(n):
            val = 0.0
            if i + k < n:
                val += a[i + k]
            if i - k >= 0:
                val += a[i - k]
            A_conv[row, i] = val
        A_conv[row, -1] = -1.0
        b_conv[row] = -conv[k]
    
    # Non-negativity: -d[i] <= a[i]
    A_nn = np.zeros((n, n + 1))
    for i in range(n):
        A_nn[i, i] = -1.0
    b_nn = a.copy()
    
    A_ub = np.vstack([A_conv, A_nn])
    b_ub = np.concatenate([b_conv, b_nn])
    
    # Equality: sum(d) = 0

    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

    if res.status == 0 and res.x is not None:

def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-30))


def smoothed_objective_lbgfs(z, T, n):
    """Soft-max smoothed objective in log-space."""
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_s = shifted.max()
    log_sum_exp = max_s + np.log(np.exp(shifted - max_s).sum())
    return (2.0 * n / T) * np.exp(T * log_sum_exp)


def smoothed_gradient_lbgfs(z, T, n):
    """Gradient of smoothed objective in log-space."""
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_s = shifted.max()
    w = np.exp(shifted - max_s)
    w = w / w.sum()
    
    n_conv = len(conv)
    grad_a = np.zeros(n)
    for i in range(n):
        val = 0.0
        for k in range(n_conv):
            if i + k < n:
                val += w[k] * a[i + k]
            if i - k >= 0:
                val += w[k] * a[i - k]
        grad_a[i] = val
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return (2.0 * n / T) * grad_z


def lbgfs_optimize(a0, time_limit, T, maxiter=3000):
    """L-BFGS optimization in log-space."""
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        return smoothed_objective_lbgfs(z, T, n)
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        return smoothed_gradient_lbgfs(z, T, n)
    
    try:
        res = minimize(fun=func, x0=z0, jac=jac, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                                'maxfun': maxiter * 30, 'disp': False})
        a_out = softmax_from_z(res.x)
        a_out = project_simplex(a_out, 1.0)
        sc = compute_score(a_out)
        if sc < best_score:
            best_a = a_out
            best_score = sc
    except Exception:
        pass
    
    return best_a, best_score


def coordinate_descent_peaks(a_init, time_budget, verbose=True):
    """Targeted coordinate descent focusing on peak-contributing pairs."""
    n = len(a_init)
    a = a_init.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(555)
    
    iters = 0
    stagnant = 0
    
    while time.monotonic() - t0 < time_budget:
        iters += 1
        rem = time_budget - (time.monotonic() - t0)
        if rem < 1:
            break
        
        conv = fast_convolve(a)
        max_idx = int(np.argmax(conv))
        
        # Find contributing pairs for the max peak
        js = max(0, max_idx - n + 1)
        je = min(n - 1, max_idx)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[max_idx - idx]
        
        # Focus on top contributors
        top_c = np.argsort(contribs)[::-1][:10]
        
        improved = False
        for ci in top_c:
            i = idx[ci]
            j = max_idx - int(i)
            if i == j or a[i] < 1e-14:
                continue
            
            # Try moving mass from the larger to a random position
            donor = i if a[i] >= a[j] else j
            for frac in [0.1, 0.2, 0.3, 0.5]:
                delta = a[donor] * frac
                if delta < 1e-15:
                    continue
                
                # Try several target positions
                for target in [rng.integers(0, n) for _ in range(3)]:
                    a_trial = a.copy()
                    a_trial[donor] -= delta
                    a_trial[target] += delta
                    a_trial = np.maximum(a_trial, 0)
                    s = a_trial.sum()
                    if s < 1e-15:
                        continue
                    a_trial = a_trial / s
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                stagnant = 0
                break
        
        if not improved:
            stagnant += 1
            if stagnant > 20:
                # Larger perturbation
                idx_pert = rng.choice(n, size=min(10, n), replace=False)
                a = best_a.copy()
                for ip in idx_pert:
                    delta = a[ip] * 0.1 * rng.standard_normal()
                    a[ip] -= delta
                a = project_simplex(a, 1.0)
                stagnant = 0
        
        if verbose and iters % 50 == 0:
            print(f"    CD iter={iters}: best={best_score:.8f}, elapsed={time.monotonic()-t0:.0f}s")
    
    return best_a, best_score


def interpolate_to_length(f, new_n):
    """Interpolate sequence to new length, preserving non-negativity and sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score, verbose=True):
    """Evaluate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
        return a_norm.copy(), score
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-strategy search combining LP-based optimization, projected gradient,
    peak-targeted coordinate descent, and L-BFGS.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    n_parent = 768
    if parent is None:
        parent = np.ones(n_parent) / n_parent
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Initialized uniform: n={n_parent}, score={best_score:.8f}")
    else:
        n_parent = len(parent)
    
    # Phase 1: LP-based optimization (key new method)
    print(f"\n=== Phase 1: LP-based optimization (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(500.0, rem * 0.35)
    if t1 > 10:
        print(f"  Budget: {t1:.0f}s")
        a_lp, sc_lp = lp_based_optimize(best_seq, t1, verbose=True)
        best_seq, best_score = check_and_update(a_lp, "LP", best_seq, best_score)
    
    # Phase 2: Projected gradient with various settings
    print(f"\n=== Phase 2: Projected Gradient (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(300.0, rem * 0.25)
    if t2 > 10:
        for K_val in [10, 20, 50]:
            rem = deadline - time.monotonic()
            t_alloc = min(t2 / 3, rem * 0.4)
            if t_alloc < 5:
                break
            print(f"  PG K={K_val}, budget={t_alloc:.0f}s")
            a_pg, sc_pg = projected_gradient_optimize(best_seq, t_alloc, K=K_val, verbose=True)
            best_seq, best_score = check_and_update(a_pg, f"PG_K{K_val}", best_seq, best_score)
    
    # Phase 3: Peak-targeted coordinate descent
    print(f"\n=== Phase 3: Peak Coordinate Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(250.0, rem * 0.2)
    if t3 > 10:
        print(f"  Budget: {t3:.0f}s")
        a_cd, sc_cd = coordinate_descent_peaks(best_seq, t3, verbose=True)
        best_seq, best_score = check_and_update(a_cd, "CD_peaks", best_seq, best_score)
    
    # Phase 4: L-BFGS with multiple temperatures
    print(f"\n=== Phase 4: L-BFGS Multi-T (best={best_score:.8f}) ===")
    for T_val in [200, 500, 2000]:
        rem = deadline - time.monotonic()
        t_alloc = min(100.0, rem * 0.2)
        if t_alloc < 10:
            break
        print(f"  L-BFGS T={T_val}, budget={t_alloc:.0f}s")
        a_lb, sc_lb = lbgfs_optimize(best_seq, t_alloc, T=T_val, maxiter=2000)
        best_seq, best_score = check_and_update(a_lb, f"lbgfs_T{T_val}", best_seq, best_score)
    
    # Phase 5: Second round of LP (with improved starting point)
    print(f"\n=== Phase 5: LP Round 2 (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(200.0, rem * 0.25)
    if t5 > 10:
        print(f"  Budget: {t5:.0f}s")
        a_lp2, sc_lp2 = lp_based_optimize(best_seq, t5, verbose=True)
        best_seq, best_score = check_and_update(a_lp2, "LP2", best_seq, best_score)
    
    # Phase 6: Try nearby lengths
    print(f"\n=== Phase 6: Nearby Lengths (best={best_score:.8f}) ===")
    alt_lengths = [n_parent + 1, n_parent - 1, n_parent + 2, n_parent - 2, n_parent + 4]
    for alt_n in alt_lengths:
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        if alt_n < 64 or alt_n > 2048:
            continue
        t_alloc = min(80.0, rem * 0.2)
        if t_alloc < 10:
            break
        init_alt = interpolate_to_length(best_seq, alt_n)
        print(f"  Optimization at n={alt_n}, budget={t_alloc:.0f}s")
        # Quick LP + gradient
        a_alt, sc_alt = lp_based_optimize(init_alt, t_alloc * 0.6, verbose=False)
        a_alt, sc_alt2 = projected_gradient_optimize(a_alt, t_alloc * 0.4, K=20, verbose=False)
        best_seq, best_score = check_and_update(a_alt, f"n{alt_n}", best_seq, best_score)
    
    # Phase 7: Final polish - combined short runs
    print(f"\n=== Phase 7: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_alloc = min(60.0, rem * 0.3)
        a_f1, _ = lp_based_optimize(best_seq, t_alloc, verbose=False)
        best_seq, best_score = check_and_update(a_f1, "final_lp", best_seq, best_score)
    
    rem = deadline - time.monotonic()
    if rem > 20:
        t_alloc = min(40.0, rem * 0.4)
        a_f2, _ = projected_gradient_optimize(best_seq, t_alloc, K=30, verbose=False)
        best_seq, best_score = check_and_update(a_f2, "final_pg", best_seq, best_score)
    
    rem = deadline - time.monotonic()
    if rem > 15:
        t_alloc = min(30.0, rem * 0.5)
        a_f3, _ = coordinate_descent_peaks(best_seq, t_alloc, verbose=False)
        best_seq, best_score = check_and_update(a_f3, "final_cd", best_seq, best_score)
    
    # Phase 8: Final L-BFGS
    rem = deadline - time.monotonic()
    if rem > 15:
        t_alloc = min(30.0, rem * 0.7)
        a_f4, _ = lbgfs_optimize(best_seq, t_alloc, T=1000, maxiter=2000)
        best_seq, best_score = check_and_update(a_f4, "final_lbgfs", best_seq, best_score)
    
    # Fallback
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
