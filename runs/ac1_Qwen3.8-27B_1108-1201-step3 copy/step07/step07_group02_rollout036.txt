                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    # Geometric temperature schedule with more stages at low T
    n_stages = 8
    temps = np.geomspace(T_start, T_end, n_stages)
    z = z0.copy()
    iters_per_stage = maxiter // n_stages

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxfun': iters_per_stage * 100, 'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(z)
            sc = 2.0 * n * fast_convolve(a_check).max()
            print(f"    L-BFGS stage {si+1}/{n_stages} (T={stage_T:.3f}): score={sc:.8f}")
        except Exception as e:
            print(f"    L-BFGS stage {si+1} error: {e}")

    a_out = softmax_from_z(z)
    return a_out


def peak_shave_extended(a0, time_limit, n_steps=20000):
    """
    Extended peak shaving: target the top peaks and try pairwise/triple
    mass redistribution to reduce them.
    """
    n = len(a0)
    a = a0.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        # Find top peaks (excluding k=0 which is a[0]^2 and usually small)
        b_sorted = np.argsort(b)[::-1]
        top_peaks = b_sorted[:5]

        improved = False
        for peak_k in top_peaks:
            if improved:
                break
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contribs = a[idx] * a[peak_k - idx]
            top_ci = np.argsort(contribs)[::-1][:8]

            # Try reducing top contributor
            for ci in top_ci:
                if improved:
                    break
                i = idx[ci]
                j = peak_k - i
                if a[i] < 1e-14:
                    continue

                # Strategy 1: reduce a[i], redistribute to random other positions
                for factor in [0.98, 0.95, 0.92, 0.88]:
                    if improved:
                        break
                    new_val = a[i] * factor
                    diff = a[i] - new_val
                    if diff < 1e-16:
                        continue
                    # Redistribute to positions that don't contribute to this peak
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    # Avoid positions that would create new peaks
                    a_trial = a.copy()
                    a_trial[i] = new_val
                    # Redistribute proportionally to others
                    others_sum = a_trial[mask].sum()
                    if others_sum < 1e-15:
                        continue
                    a_trial[mask] += diff * a_trial[mask] / others_sum
                    a_trial = a_trial / a_trial.sum()
                    sc = 2.0 * n * fast_convolve(a_trial).max()
                    if sc < best_score - 1e-12:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True

                # Strategy 2: reduce both a[i] and a[j] (if i != j)
                if i != j and a[j] > 1e-14:
                    for f1 in [0.96, 0.93]:
                        for f2 in [0.96, 0.93]:
                            if improved:
                                break
                            a_trial = a.copy()
                            a_trial[i] *= f1
                            a_trial[j] *= f2
                            a_trial = np.maximum(a_trial, 0)
                            a_trial = a_trial / a_trial.sum()
                            sc = 2.0 * n * fast_convolve(a_trial).max()
                            if sc < best_score - 1e-12:
                                a = a_trial
                                best_score = sc
                                best_a = a.copy()
                                improved = True

            # Strategy 3: random perturbation near the peak contributors
            if not improved and len(top_ci) > 0:
                ci = top_ci[0]
                i = idx[ci]
                # Try moving mass from i to a random position far from the peak
                for _ in range(3):
                    target = rng.integers(0, n)
                    if target == i or abs(target - i) < 2:
                        continue
                    for frac in [0.02, 0.05, 0.1]:
                        a_trial = a.copy()
                        move = a[i] * frac
                        a_trial[i] -= move
                        a_trial[target] += move
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        sc = 2.0 * n * fast_convolve(a_trial).max()
                        if sc < best_score - 1e-12:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break

        if not improved and step % 1000 == 500:
            # Stuck: try a small random perturbation
            perturb = rng.standard_normal(n) * 0.01
            a_trial = simplex_project(a + perturb)
            sc = 2.0 * n * fast_convolve(a_trial).max()
            if sc < best_score:
                a = a_trial
                best_score = sc
                best_a = a.copy()

        if step % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave step={step} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def gradient_local_search(a0, time_limit, n_steps=50000):
    """
    Subgradient-based local search with backtracking line search.
    Uses the exact subgradient of max_k conv(a,a)[k].
    """
    n = len(a0)
    a = a0.copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    z = z_from_a(a)
    lr = 0.1

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        # Active set: peaks within 1% of max
        threshold = max_b * 0.99
        active = np.where(b >= threshold)[0]

        # Compute subgradient
        g = np.zeros(n)
        for k in active:
            for i in range(n):
                ki = k - i
                if 0 <= ki < n:
                    g[i] += 2.0 * a[ki]
        g /= len(active)

        # Mirror descent step
        z_new = z - lr * g
        a_new = softmax_from_z(z_new)
        sc = 2.0 * n * fast_convolve(a_new).max()

        if sc < best_score - 1e-12:
            z = z_new
            a = a_new
            best_score = sc
            best_a = a.copy()
        else:
            lr *= 0.95

        if step % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GradLS step={step} ({elapsed:.0f}s): best={best_score:.8f} lr={lr:.6f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization: extended LP-Nesterov, low-T L-BFGS,
    peak shaving, gradient local search, and multi-start.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Determine LP solver
    solver = None
    for s in [cp.PDLP, cp.ECOS, cp.SCS, cp.GLPK, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solver = s
                break
        except Exception:
            continue
    print(f"Using LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 768
        best_seq = np.ones(n_init) / n_init
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Extended LP-Nesterov (700s) ===
    print(f"\n=== Phase 1: Extended LP-Nesterov (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.40, 700.0)
    if t_alloc > 60:
        print(f"  LP-Nesterov n={len(best_seq)}, budget={t_alloc:.0f}s")
        a_lp, _ = extended_lp_nesterov(best_seq, t_alloc, solver=solver,
                                        kick_interval=300, n_kicks=30)
        try_update(a_lp, "lp_nesterov")

    # === Phase 2: Very low temperature L-BFGS (400s) ===
    print(f"\n=== Phase 2: Low-T L-BFGS (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.22, 400.0)
    if t_alloc > 30:
        print(f"  L-BFGS T: 10->0.05, n={len(best_seq)}, budget={t_alloc:.0f}s")
        a_lb = lbgfs_low_temp(best_seq, t_alloc, T_start=10.0, T_end=0.05, maxiter=12000)
        try_update(a_lb, "lbgfs_lowT")

    # === Phase 3: Extended peak shaving (200s) ===
    print(f"\n=== Phase 3: Peak Shaving (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 200.0)
    if t_alloc > 20:
        print(f"  PeakShave n={len(best_seq)}, budget={t_alloc:.0f}s")
        a_ps, _ = peak_shave_extended(best_seq, t_alloc, n_steps=30000)
        try_update(a_ps, "peak_shave")

    # === Phase 4: Gradient local search (200s) ===
    print(f"\n=== Phase 4: Gradient LS (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 200.0)
    if t_alloc > 20:
        print(f"  GradLS n={len(best_seq)}, budget={t_alloc:.0f}s")
        a_gls, _ = gradient_local_search(best_seq, t_alloc, n_steps=80000)
        try_update(a_gls, "grad_ls")

    # === Phase 5: Multi-start perturbations (200s) ===
    print(f"\n=== Phase 5: Multi-Start (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    n_starts = 5
    per_start_budget = min(rem / n_starts, 60.0)
    for si in range(n_starts):
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t_alloc = min(rem * 0.2, per_start_budget)
        if t_alloc < 15:
            break
        # Create perturbation
        rng_s = np.random.default_rng(seed + 1000 * si)
        perturb = rng_s.standard_normal(len(best_seq))
        a_pert = simplex_project(best_seq + 0.03 * perturb)
        sc_pert = evaluate_safe(a_pert)
        print(f"  Start {si+1}: perturbed score={sc_pert:.8f}")

        # Quick LP + L-BFGS
        t_lp = t_alloc * 0.5
        t_lb = t_alloc * 0.5
        a_lp2, _ = extended_lp_nesterov(a_pert, t_lp, solver=solver, kick_interval=100)
        try_update(a_lp2, f"multi_lp_{si}")

        rem = deadline - time.monotonic()
        if rem > 10:
            a_lb2 = lbgfs_low_temp(best_seq if best_score < sc_pert else a_pert,
                                    min(rem * 0.4, t_lb), T_start=5.0, T_end=0.1, maxiter=3000)
            try_update(a_lb2, f"multi_lb_{si}")

    # === Phase 6: Second LP round on best (150s) ===
    print(f"\n=== Phase 6: LP Round 2 (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 150.0)
    if t_alloc > 30:
        print(f"  LP Round 2, budget={t_alloc:.0f}s")
        a_lp3, _ = extended_lp_nesterov(best_seq, t_alloc, solver=solver,
                                         kick_interval=200, n_kicks=10)
        try_update(a_lp3, "lp_round2")

    # === Phase 7: Final ultra-low-T L-BFGS + peak shave (remainder) ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t1 = min(rem * 0.5, 80.0)
        if t1 > 10:
            a_final = lbgfs_low_temp(best_seq, t1, T_start=2.0, T_end=0.02, maxiter=5000)
            try_update(a_final, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 15:
        a_ps2, _ = peak_shave_extended(best_seq, min(rem * 0.5, 40.0), n_steps=10000)
        try_update(a_ps2, "final_ps")

    rem = deadline - time.monotonic()
    if rem > 10:
        a_gls2, _ = gradient_local_search(best_seq, min(rem * 0.5, 30.0), n_steps=30000)
        try_update(a_gls2, "final_gls")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(x) for x in best_seq]
