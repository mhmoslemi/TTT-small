"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based smoothed gradient descent in O(n log n) per step
2. Nesterov acceleration with adaptive temperature
3. Multiple structured starting points at various n
4. Fictitious play heuristic
5. Peak-shaving local refinement
Key: gradient of logsumexp(conv/T) = 2*conv(softmax(conv/T), flip(a))[:n], all via FFT.

import numpy as np
import time
from scipy.fft import rfft, irfft


def fft_convolve(a):
    """Linear convolution a*a via FFT, returns full result of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_max_conv(a):
    """Compute max of linear convolution a*a."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    conv = irfft(A * A, fft_size)[:size]
    return conv.max(), conv


def project(a):
    """Project to non-negative simplex (sum=1)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s


def smoothed_grad(a, T):
    """
    Compute gradient of T*logsumexp(conv(a,a)/T) w.r.t. a.
    Result: 2 * conv(softmax(conv/T), flip(a))[:n]
    All via FFT for O(n log n).
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    conv = irfft(A * A, fft_size)[:size]
    
    # Softmax weights
    shifted = conv / T
    max_s = shifted.max()
    w = np.exp(shifted - max_s)
    w = w / w.sum()
    
    # Gradient = 2 * conv(w, flip(a))[:n]
    W = rfft(w, fft_size)
    Af = rfft(a[::-1], fft_size)
    grad = 2.0 * irfft(W * Af, fft_size)[:n]
    
    return T * grad, conv.max()


def nesterov_optimize(a0, time_limit, T0=50.0, T_min=0.5, lr0=0.01, 
                      momentum=0.9, verbose=False):
    """
    Nesterov accelerated gradient descent on smoothed objective.
    a is maintained on the simplex (sum=1, a>=0).
    """
    n = len(a0)
    a = project(a0.copy())
    v = np.zeros(n)
    best_a = a.copy()
    best_score = a.size * 2.0 * compute_max_conv(a)[0]  # 2n*max_conv (sum=1)
    t0 = time.monotonic()
    lr = lr0
    T = T0
    iters = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        
        # Nesterov lookahead
        a_look = a - momentum * v
        a_look = project(a_look)
        
        # Compute gradient at lookahead point
        grad, _ = smoothed_grad(a_look, T)
        
        # Update velocity
        v = momentum * v + lr * grad
        
        # Step
        a_new = a - v
        
        # Project to simplex
        a_new = project(a_new)
        
        # Evaluate
        max_c, _ = compute_max_conv(a_new)
        score = 2.0 * n * max_c
        
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        
        # Adaptive learning rate (reduce on increase)
        if score > best_score * 1.01:
            lr *= 0.95
            v *= 0.5
        else:
            lr = min(lr * 1.001, lr0)
        
        # Cool temperature
        if iters % 10 == 0:
            T = max(T_min, T * 0.995)
        
        if verbose and iters % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NGD iter={iters}, T={T:.2f}, lr={lr:.6f}, score={score:.8f}, elapsed={elapsed:.1f}s")
    
    return best_a, best_score
