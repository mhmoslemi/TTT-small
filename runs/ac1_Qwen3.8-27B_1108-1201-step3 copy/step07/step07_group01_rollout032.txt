"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_from_a(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def softmax_from_z(z):
    """Convert z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def active_set_gradient(a, K=1):
    """Compute gradient using top K peaks (exact subgradient when K=1).
    Returns gradient vector and the max value."""
    n = len(a)
    b = fast_convolve(a)
    # Find top K peaks
    # For stability, exclude the very endpoints (they only depend on one a_i)
    top_k_indices = np.argsort(b)[::-1][:K]
    
    grad = np.zeros(n)
    max_b = b.max()
    
    for k in top_k_indices:
        k = int(k)
        # grad[i] += 2 * a[k-i] for valid i
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * a[k - idx]
    
    return grad, max_b, b


def subgradient_descent(a_init, time_limit, K=1, eta_init=0.01, eta_decay=0.9999,
                        max_iters=50000, verbose=True, label=""):
    """Exact subgradient descent on the active set with step decay."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    eta = eta_init
    stuck = 0
    stuck_threshold = 500
    
    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        grad, max_b, b = active_set_gradient(a, K=K)
        current_score = 2.0 * n * max_b
        
        if current_score < best_score - 1e-14:
            best_score = current_score
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
        
        # Project gradient to tangent space of simplex
        grad = grad - np.dot(a, grad)
        
        # Update
        a = a - eta * grad
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        eta *= eta_decay
        
        # Periodic restart if stuck
        if stuck > stuck_threshold:
            rng = np.random.default_rng(hash((it, n)) % (2**31))
            noise = rng.standard_normal(n) * 0.01
            a = best_a + noise
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            eta = eta_init * 0.5
            stuck = 0
        
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} SGD iter={it} ({elapsed:.0f}s): score={current_score:.7f} best={best_score:.7f} eta={eta:.6f}")
    
    return best_a, best_score


def mirror_descent_v2(a_init, time_limit, T_start=0.005, T_end=0.0003,
                      eta_start=0.05, eta_end=0.001, max_iters=30000,
                      verbose=True, label=""):
    """Improved mirror descent with better annealing and multi-peak awareness."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    stuck = 0
    stuck_threshold = 400
    
    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Annealing: use a "staircase" schedule
        progress = min(1.0, it / max_iters)
        T = T_start * (T_end / T_start) ** (progress ** 0.7)
        eta = eta_start * (eta_end / eta_start) ** progress
        
        b = fast_convolve(a)
        current_score = 2.0 * n * b.max()
        
        if current_score < best_score - 1e-14:
            best_score = current_score
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
        
        # Compute weights: use top K peaks with exponential weighting
        top_K = min(10, 2 * n - 1)
        sorted_idx = np.argsort(b)[::-1][:top_K]
        b_top = b[sorted_idx]
        # Weight: exponential decay from max
        w = np.exp(-(b_top - b_top[0]) / T)
        w = w / w.sum()
        
        # Compute gradient
        grad_a = np.zeros(n)
        for wi, ki in zip(w, sorted_idx):
            ki = int(ki)
            js = max(0, ki - n + 1)
            je = min(n - 1, ki)
            idx = np.arange(js, je + 1)
            grad_a[idx] += wi * 2.0 * a[ki - idx]
        
        # Project to tangent
        grad_a = grad_a - np.dot(a, grad_a)
        
        # Multiplicative update
        log_update = -eta * grad_a
        log_update = np.clip(log_update, -2.0, 2.0)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-300)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
        else:
            a = a / s
        
        # Stuck detection
        if stuck > stuck_threshold:
            rng = np.random.default_rng(it * 31 + 7)
            noise = rng.standard_normal(n) * 0.015
            a = best_a + noise
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            stuck = 0
            eta = min(eta * 2.0, eta_start * 0.3)
        
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} MD2 iter={it} ({elapsed:.0f}s): T={T:.5f} score={current_score:.7f} best={best_score:.7f}")
    
    return best_a, best_score


def lbfgs_smoothed(a_init, time_limit, T=10, maxiter=5000):
    """L-BFGS on the log-sum-exp smoothed objective."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).astype(np.float64)
    a = a / a.sum()
    z0 = np.log(a)
    z0 = z0 - z0.mean()
    t0 = time.monotonic()
    
    def obj(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15, None
        a_z = softmax_from_z(z)
        b = fast_convolve(a_z)
        scaled = b / T
        m = scaled.max()
        lse = m + np.log(np.exp(scaled - m).sum())
        val = 2.0 * n * T * lse
        return val, None
    
    def grad(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros(n)
        a_z = softmax_from_z(z)
        b = fast_convolve(a_z)
        scaled = b / T
        m = scaled.max()
        w = np.exp(scaled - m)
        w = w / w.sum()
        # Gradient of sum_k w_k * conv(a,a)[k] with respect to a
        grad_a = np.zeros(n)
        for ki in range(len(w)):
            if w[ki] < 1e-10:
                continue
            ki_int = ki
            js = max(0, ki_int - n + 1)
            je = min(n - 1, ki_int)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[ki] * a_z[ki_int - idx]
        # Convert to z-gradient (softmax parameterization)
        mean_grad = np.dot(a_z, grad_a)
        grad_z = a_z * (grad_a - mean_grad)
        return 2.0 * n * grad_z
    
    try:
        res = minimize(
            fun=obj, x0=z0, jac=grad,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 50, 'disp': False}
        )
        z_opt = res.x
        a_out = softmax_from_z(z_opt)
    except Exception:
        a_out = a_init.copy()
    
    return a_out

def coordinate_descent_v2(a_init, time_limit, verbose=True, label=""):

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def random_restart_optimize(a_init, time_limit, n_restarts=5, per_restart_time=None,
                            noise_scale=0.02, verbose=True):
    """Try random perturbations and optimize each, keep the best."""
    n = len(a_init)
    t0 = time.monotonic()
    if per_restart_time is None:
        per_restart_time = time_limit / n_restarts
    
    best_a = a_init.copy()
    best_score = score_from_a(a_init)
    rng = np.random.default_rng(42)
    
    for r in range(n_restarts):
        if time.monotonic() - t0 > time_limit:
            break
        
        noise = rng.standard_normal(n) * noise_scale
        a_perturbed = best_a + noise
        a_perturbed = np.maximum(a_perturbed, 1e-15)
        a_perturbed = a_perturbed / a_perturbed.sum()
        
        # Quick optimization
        rem = min(per_restart_time, time_limit - (time.monotonic() - t0))
        if rem < 5:
            break
        
        a_opt, sc = subgradient_descent(
            a_perturbed, rem, K=1, eta_init=0.02,
            eta_decay=0.9998, max_iters=20000, verbose=False,
            label=f"restart_{r}"
        )
        
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            if verbose:
                print(f"    Restart {r}: improved to {sc:.7f}")
    
    return best_a, best_score

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_from_a(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def softmax_from_z(z):
    """Convert z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def compute_peak_gradient(a, k):
    """Compute gradient of conv(a,a)[k] with respect to a."""
    n = len(a)
    grad = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    idx = np.arange(js, je + 1)
    grad[idx] = 2.0 * a[k - idx]
    return grad


def active_set_subgrad(a, K=1):
    """Compute subgradient using top K peaks. Returns (grad, max_val, b)."""
    n = len(a)
    b = fast_convolve(a)
    top_k = np.argsort(b)[::-1][:K]
    grad = np.zeros(n)
    for ki in top_k:
        grad += compute_peak_gradient(a, int(ki))
    max_b = b.max()
    return grad, max_b, b


def backtracking_line_search(a, grad, max_b, n, c1=1e-4, rho=0.5, eta_max=0.5):
    """Backtracking line search for subgradient step. Returns optimal eta."""
    # Project gradient to tangent space
    grad_proj = grad - np.dot(a, grad)
    grad_norm = np.linalg.norm(grad_proj)
    if grad_norm < 1e-15:
        return 0.0
    
    # Initial step size based on curvature estimate
    eta = min(eta_max, 0.01 / max(grad_norm, 0.01))
    
    for _ in range(20):
        a_trial = a - eta * grad_proj
        a_trial = np.maximum(a_trial, 1e-15)
        a_trial = a_trial / a_trial.sum()
        b_trial = fast_convolve(a_trial)
        new_max = b_trial.max()
        
        if new_max <= max_b - c1 * eta * grad_norm * max_b:
            return eta
        eta *= rho
    
    return eta

def lbfgs_smoothed_obj(z, T, n):
    """Objective and gradient for L-BFGS in log-space."""
    a_z = softmax_from_z(z)
    b = fast_convolve(a_z)
    scaled = b / T
    m = scaled.max()
    exp_scaled = np.exp(scaled - m)
    lse = m + np.log(exp_scaled.sum())
    val = 2.0 * n * T * lse
    
    # Gradient
    w = exp_scaled / exp_scaled.sum()
    grad_a = np.zeros(n)
    significant = np.where(w > 1e-8)[0]
    for ki in significant:
        ki_int = int(ki)
        grad_a += w[ki] * compute_peak_gradient(a_z, ki_int)
    
    mean_grad = np.dot(a_z, grad_a)
    grad_z = a_z * (grad_a - mean_grad)
    grad_z *= 2.0 * n
    
    return val, grad_z

        return val

    best_score = score_from_a(a)
    t0 = time.monotonic()
    
    factors = [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0]
    no_improve = 0
    max_no_improve = 80
    rng = np.random.default_rng(99991)
    
    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        no_improve = 0
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            for f in factors:
                a_trial = a.copy()
                old_val = a_trial[i]
                new_val = old_val * f
                if new_val < -1e-15:
                    continue
                a_trial[i] = new_val
                diff = a_trial.sum() - 1.0
                if abs(diff) < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] = others - diff * others / others_sum
                if a_trial.min() < -1e-12:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = score_from_a(a_trial)
                if sc < best_score - 1e-12:
                    best_score = sc
                    a = a_trial.copy()
                    best_a = a.copy()
                    no_improve = 0
                    break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate using the actual evaluate_sequence function."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.7f} (n={len(a_norm)})")
    return best_seq, best_score
