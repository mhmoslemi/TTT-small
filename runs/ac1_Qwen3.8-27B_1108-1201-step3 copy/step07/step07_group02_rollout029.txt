def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    if '_lp_solver_cached' in globals():
        return _lp_solver_cached
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return _lp_solver_cached


def solve_lp_best_response(a_fixed, solver=None):

    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

        prob.solve(solver=solver)
    except Exception:

                break

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()

    b_val = b_val / s

def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)


def compute_soft_max_obj_grad(a, T_val):
    """Compute LogSumExp smoothed objective and gradient."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp
    
    w = exp_Tb / exp_Tb.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    return obj, grad_a

def lbgfs_refine(a0, time_limit, T_start=5.0, T_end=0.5, maxiter=5000):
    """L-BFGS refinement with soft-max smoothing."""
    from scipy.optimize import minimize
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    
    def make_func(T_val):
        def func(z):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            return (2.0 * n / T_val) * (max_Tb + np.log(exp_Tb.sum()))
        return func
    
    def make_jac(T_val):
        def jac(z):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z)
            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac
    
    temps = np.geomspace(T_start, T_end, 4)
    z = z0.copy()
    per_stage = maxiter // len(temps)
    
    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': per_stage, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxfun': per_stage * 30, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    return softmax_from_z(z)

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combined approach:

2. L-BFGS-B refinement at low temperatures

import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    if '_lp_solver_cached' in globals():
        return _lp_solver_cached
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return _lp_solver_cached


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)


def compute_soft_max_grad(a, T_val):
    """Compute gradient of LogSumExp smoothed objective."""
    n = len(a)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    return grad_a


def score_of(a):
    """Compute the objective score for a probability vector."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()


def adam_logspace(a0, time_limit, T_start=200.0, T_end=0.3, lr=0.005):
    """Adam optimizer in log-space with LogSumExp smoothing."""

    n_temps = 10
    temps = np.geomspace(T_start, T_end, n_temps)

    while time.monotonic() - t0 < time_limit:
        progress = (time.monotonic() - t0) / time_limit
        temp_idx = min(int(progress * n_temps), n_temps - 1)

        a_cur = softmax_from_z(z)
        grad_a = compute_soft_max_grad(a_cur, T_val)
        S = np.dot(grad_a, a_cur)
        grad_z = a_cur * (grad_a - S)

        t += 1
        m = 0.9 * m + 0.1 * grad_z
        v = 0.999 * v + 0.001 * grad_z ** 2
        m_hat = m / (1 - 0.9 ** t)
        v_hat = v / (1 - 0.999 ** t)

        z_new = z - lr * m_hat / (np.sqrt(v_hat) + 1e-8)

    return best_a


def lbgfs_refine(a0, time_limit, T_start=5.0, T_end=0.2, maxiter=5000):
    """L-BFGS-B refinement with LogSumExp smoothing."""
    from scipy.optimize import minimize
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a_init = a_init / a_init.sum()
    z0 = z_from_a(a_init)
    t0 = time.monotonic()
    n_stages = 4
    temps = np.geomspace(T_start, T_end, n_stages)
    z = z0.copy()
    per_stage = maxiter // n_stages

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break

        def func(z_in, T_c=stage_T):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a_in = softmax_from_z(z_in)
            b_in = fast_convolve(a_in)
            Tb_in = T_c * b_in
            max_Tb = Tb_in.max()
            exp_Tb = np.exp(Tb_in - max_Tb)
            return (2.0 * n / T_c) * (max_Tb + np.log(exp_Tb.sum()))

        def jac(z_in, T_c=stage_T):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a_in = softmax_from_z(z_in)
            grad_a = compute_soft_max_grad(a_in, T_c)
            S_val = np.dot(grad_a, a_in)
            return a_in * (grad_a - S_val)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': per_stage, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxfun': per_stage * 40, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)

    t0 = time.monotonic()

    for it in range(n_max_iters):

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        alpha = 2.0 / (it + 2)

        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))

        sc_a = score_of(a)

            best_a = a.copy()
        if sc_y < best_score:

        if (it + 1) % 500 == 0:

def mirror_descent_refine(a0, time_limit, lr_init=0.02, n_steps=50000):

def interpolate_seq(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0, min(1000, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def try_update_state(a, label, best_seq, best_score):
    """Try to update best state. Returns (updated_best_seq, updated_best_score, improved)."""
    a = np.maximum(a, 0).astype(np.float64)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score, False
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < best_score:
        print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")
        return a.copy(), sc, True
    return best_seq, best_score, False


def propose_candidate():
    """
    Multi-phase search minimizing auto-convolution peak.
    Combines Adam, L-BFGS, LP Nesterov, mirror descent, and peak shaving.
    """
    deadline = time.monotonic() + 1780.0

    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    # Load parent sequence
    best_seq = None
    best_score = float('inf')
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_init = 768
        best_seq = np.ones(n_init) / n_init
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)
    print(f"Starting optimization, n={n_best}, score={best_score:.6f}")

    # === Phase 1: Adam (35%) ===
    print(f"\n=== Phase 1: Adam Log-Space ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.32, 500.0)
    if t_alloc > 30:
        print(f"  T: 300->0.2, budget={t_alloc:.0f}s")
        a_adam = adam_logspace(best_seq, t_alloc, T_start=300.0, T_end=0.2, lr=0.004)
        best_seq, best_score, _ = try_update_state(a_adam, "adam", best_seq, best_score)

    # === Phase 2: L-BFGS (22%) ===
    print(f"\n=== Phase 2: L-BFGS Refine ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.22, 300.0)
    if t_alloc > 30:
        print(f"  T: 10->0.1, budget={t_alloc:.0f}s")
        a_lb = lbgfs_refine(best_seq, t_alloc, T_start=10.0, T_end=0.1, maxiter=8000)
        best_seq, best_score, _ = try_update_state(a_lb, "lbfgs", best_seq, best_score)

    # === Phase 3: Nesterov LP (15%) ===
    print(f"\n=== Phase 3: Nesterov LP ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 30:
        print(f"  budget={t_alloc:.0f}s")
        a_np = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=5000)
        best_seq, best_score, _ = try_update_state(a_np, "lp", best_seq, best_score)

    # === Phase 4: Mirror Descent (8%) ===
    print(f"\n=== Phase 4: Mirror Descent ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 100.0)
    if t_alloc > 15:
        print(f"  budget={t_alloc:.0f}s")
        a_md = mirror_descent_refine(best_seq, t_alloc, lr_init=0.015, n_steps=80000)
        best_seq, best_score, _ = try_update_state(a_md, "mirror", best_seq, best_score)

    # === Phase 5: Peak Shaving (10%) ===
    print(f"\n=== Phase 5: Peak Shaving ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 10:
        print(f"  budget={t_alloc:.0f}s")
        a_ps = peak_shave_refine(best_seq, t_alloc, n_steps=50000)
        best_seq, best_score, _ = try_update_state(a_ps, "shave", best_seq, best_score)

    # === Phase 6: Second L-BFGS + Shave cycle (7%) ===
    print(f"\n=== Phase 6: Refinement Cycle ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.4, 50.0)
    if t1 > 10:
        a_r1 = lbgfs_refine(best_seq, t1, T_start=3.0, T_end=0.1, maxiter=4000)
        best_seq, best_score, _ = try_update_state(a_r1, "lbfgs2", best_seq, best_score)
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.5, 40.0)
    if t2 > 10:
        a_r2 = peak_shave_refine(best_seq, t2, n_steps=20000)
        best_seq, best_score, _ = try_update_state(a_r2, "shave2", best_seq, best_score)

    # === Phase 7: Alternative lengths (remaining) ===
    print(f"\n=== Phase 7: Alt Lengths ===")
    rem = deadline - time.monotonic()
    alt_lengths = sorted(set([
        max(256, int(n_best * 0.75)),
        min(2048, int(n_best * 1.25)),
        min(2048, int(n_best * 1.5)),
        max(256, int(n_best * 0.6)),
    ]))
    alt_lengths = [x for x in alt_lengths if x != n_best]

    for alt_n in alt_lengths[:3]:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        t_alloc = min(rem * 0.35, 100.0)
        if t_alloc < 20:
            break
        print(f"  n={alt_n}, budget={t_alloc:.0f}s")
        init_alt = interpolate_seq(best_seq, alt_n)

        a_a1 = adam_logspace(init_alt, t_alloc * 0.45, T_start=100.0, T_end=1.0, lr=0.004)
        best_seq, best_score, _ = try_update_state(a_a1, f"adam_n{alt_n}", best_seq, best_score)

        rem = deadline - time.monotonic()
        if rem > 20:
            cur = best_seq if len(best_seq) == alt_n else init_alt
            t2 = min(rem * 0.3, 40.0)
            if t2 > 10:
                a_a2 = lbgfs_refine(cur, t2, T_start=5.0, T_end=0.3, maxiter=3000)
                best_seq, best_score, _ = try_update_state(a_a2, f"lbfgs_n{alt_n}", best_seq, best_score)

    # === Final polish ===
    print(f"\n=== Final Polish ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t_f = min(rem * 0.6, 40.0)
        a_f = peak_shave_refine(best_seq, t_f, n_steps=15000)
        best_seq, best_score, _ = try_update_state(a_f, "final_shave", best_seq, best_score)

    rem = deadline - time.monotonic()
    if rem > 10:
        t_f2 = min(rem * 0.7, 20.0)
        a_f2 = lbgfs_refine(best_seq, t_f2, T_start=2.0, T_end=0.1, maxiter=2000)
        best_seq, best_score, _ = try_update_state(a_f2, "final_lbfgs", best_seq, best_score)

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}")
    return [float(max(0.0, x)) for x in best_seq]
