```python
"""
LP-based optimization for minimizing 2*n*max(conv(a,a))/(sum a)^2.
Uses linearized constraints to find descent directions via linear programming,
combined with mirror descent and coordinate descent for refinement.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def normalize(a):
    """Normalize to sum=1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def build_constraint_matrix(a):
    """Build the (2n-1) x n constraint matrix for the LP.
    A[k, j] = a[k-j] if 0 <= k-j < n, else 0."""
    n = len(a)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        js = max(0, k - n + 1)
        je = min(n-1, k)
        if js <= je:
            A[k, js:je+1] = a[k-je:k-js+1][::-1]
    return A


def solve_direction_lp(a, M, verbose=False):
    """Solve LP to find descent direction."""
    n = len(a)
    conv_aa = fast_convolve(a)
    
    A = build_constraint_matrix(a)
    rhs = (M + conv_aa) / 2.0
    c = -np.ones(n)
    
    try:
        result = linprog(c, A_ub=A, b_ub=rhs, bounds=[(0, None)]*n, method='highs')
        if result.success and result.x is not None:
            b = np.maximum(result.x, 0)
            if b.sum() > 1e-15:
                return b
    except Exception as e:
        if verbose:
            print(f"    LP error: {e}")
    
    return None


def line_search(a, g, max_evals=40):
    """Golden section search for best t in [0,1]."""
    def obj(t):
        x = (1.0 - t) * a + t * g
        return fast_convolve(x).max()
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = obj(c1)
    f2 = obj(c2)
    
    for _ in range(max_evals):
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = obj(c1)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = obj(c2)
    
    return (lo + hi) / 2.0

        T = T_start * (T_end / T_start) ** progress
        eta = eta_start * (eta_end / eta_start) ** progress
        
        b = fast_convolve(a)
        current_score = 2.0 * n * b.max()
        
        if current_score < best_score - 1e-15:
            best_score = current_score
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
        
        # Soft-max weights
        scaled = b / T
        m = scaled.max()
        exp_scaled = np.exp(scaled - m)
        w = exp_scaled / exp_scaled.sum()
        
        # Gradient
        fft_size = 1 << (3 * n - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        correl = irfft(np.conj(W) * A, fft_size)
        grad_a = 2.0 * n * correl[:n]
        
        # Mirror update
        log_update = -eta * grad_a
        log_update = np.clip(log_update, -3.0, 3.0)
        a = a * np.exp(log_update)
        a = normalize(a)
        
        if stuck > 500:
            rng = np.random.default_rng(it * 7919)
            noise = rng.standard_normal(n) * 0.05
            a = normalize(best_a + noise)
            stuck = 0
    
    return best_a, best_score


def peak_shave(a_init, time_limit, verbose=True, label=""):
    """Targeted peak shaving."""
    n = len(a_init)
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    for step in range(5000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        b_work = b.copy()
        b_work[0] = -np.inf
        b_work[-1] = -np.inf
        peak_k = int(np.argmax(b_work))
        peak_val = b[peak_k]
        
        if peak_val <= best_score / (2.0 * n) + 1e-16:
            break
        
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_idx = np.argsort(contribs)[::-1][:3]

                    best_a = a.copy()
                    improved = True
```

```python
"""
LP-based optimization for minimizing 2*n*max(conv(a,a))/(sum a)^2.
Primary method: Linear programming with linearized convolution constraints to find
optimal descent directions, combined with golden-section line search.
Secondary: Mirror descent with annealed temperature for refinement.
Tertiary: Peak shaving and coordinate descent for final polish.
Key insight: The auto-convolution (not auto-correlation) allows scores below 2,
and the LP approach finds the best "spreading" direction subject to all
linearized peak constraints simultaneously.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def normalize(a):
    """Clamp to non-negative and normalize to sum=1."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def build_A_matrix(a):
    """Build (2n-1) x n matrix where A[k,j] = a[k-j] if valid else 0."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            A[k, js:je + 1] = a[k - je:k - js + 1][::-1]
    return A


def solve_direction_lp(a, M, max_time=15.0):
    """Solve LP: max sum(b) s.t. 2*(b*a)_k <= M + (a*a)_k, b >= 0."""
    n = len(a)
    conv_aa = fast_convolve(a)
    A = build_A_matrix(a)
    rhs = (M + conv_aa) / 2.0
    c = -np.ones(n)
    t0 = time.monotonic()
    try:
        result = linprog(c, A_ub=A, b_ub=rhs,
                         bounds=[(0, None)] * n, method='highs')
        if time.monotonic() - t0 < max_time and result.success and result.x is not None:
            b = np.maximum(result.x, 0.0)
            if b.sum() > 1e-12:
                return b
    except Exception:
        pass
    return None


def line_search(a, g, max_evals=50):
    """Golden section search for best t in [0,1] minimizing conv peak."""
    def obj(t):
        x = (1.0 - t) * a + t * g
        return fast_convolve(x).max()

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo = 0.0
    hi = 1.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    f1 = obj(c1)
    f2 = obj(c2)
    for _ in range(max_evals):
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = obj(c1)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = obj(c2)
    t_best = (lo + hi) / 2.0
    return t_best


def lp_optimize(a_init, time_limit, verbose=True, label=""):
    """LP-based optimization with line search."""
    n = len(a_init)
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    no_improve_count = 0
    max_no_improve = 3

    for it in range(1, 10000):
        if time.monotonic() - t0 > time_limit:
            break

        M = fast_convolve(a).max()
        t_lp = time.monotonic()
        b = solve_direction_lp(a, M, max_time=max(2.0, (time_limit - (time.monotonic() - t0)) * 0.5))
        lp_time = time.monotonic() - t_lp

        if b is None:
            if verbose:
                print(f"    {label} LP iter={it}: solve failed")
            break

        g = normalize(b)
        cos_sim = float(np.dot(a, g))
        if cos_sim > 0.99995:
            if verbose:
                print(f"    {label} LP iter={it}: converged (cos={cos_sim:.8f})")
            break

        t = line_search(a, g)
        a_new = normalize((1.0 - t) * a + t * g)
        score_new = score_normalized(a_new)

        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            no_improve_count = 0
            if verbose and (it <= 5 or it % 10 == 0):
                elapsed = time.monotonic() - t0
                print(f"    {label} LP it={it} ({elapsed:.0f}s, lp={lp_time:.1f}s): "
                      f"t={t:.3f} score={score_new:.8f}")
        else:
            # Try smaller steps
            found = False
            for t_try in [0.5 * t, 0.25 * t, 0.1, 0.05]:
                if t_try <= 0:
                    continue
                a_try = normalize((1.0 - t_try) * a + t_try * g)
                s_try = score_normalized(a_try)
                if s_try < best_score - 1e-14:
                    a = a_try
                    best_score = s_try
                    best_a = a.copy()
                    found = True
                    break
            if not found:
                no_improve_count += 1
                if no_improve_count >= max_no_improve:
                    # Perturb and try again
                    rng = np.random.default_rng(it * 104729)
                    a = normalize(best_a + rng.standard_normal(n) * 0.01)
                    no_improve_count = 0
                    if verbose:
                        print(f"    {label} LP it={it}: perturbed restart")
                else:
                    continue
            else:
                if verbose and it % 10 == 0:
                    elapsed = time.monotonic() - t0
                    print(f"    {label} LP it={it} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def mirror_descent(a_init, time_limit, T_start=0.01, T_end=0.001,
                   eta_start=0.03, eta_end=0.003, max_iters=200000,
                   verbose=True, label=""):
    """Mirror descent with temperature annealing on simplex."""
    n = len(a_init)
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    stuck = 0

    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, it / max_iters)
        T = T_start * (T_end / T_start) ** progress
        eta = eta_start * (eta_end / eta_start) ** progress

        b = fast_convolve(a)
        cur = 2.0 * n * b.max()

        if cur < best_score - 1e-15:
            best_score = cur
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        scaled = b / T
        m = scaled.max()
        exp_s = np.exp(scaled - m)
        w = exp_s / exp_s.sum()

        fft_size = 1 << (3 * n - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        correl = irfft(np.conj(W) * A, fft_size)
        grad_a = 2.0 * n * correl[:n]

        log_update = -eta * grad_a
        log_update = np.clip(log_update, -4.0, 4.0)
        a = a * np.exp(log_update)
        a = normalize(a)

        if stuck > 1000:
            rng = np.random.default_rng(it * 7919)
            a = normalize(best_a + rng.standard_normal(n) * 0.03)
            stuck = 0

        if verbose and it % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} MD it={it} ({elapsed:.0f}s): T={T:.5f} "
                  f"score={cur:.8f} best={best_score:.8f}")

    return best_a, best_score


def subgradient_descent(a_init, time_limit, eta_start=0.1, decay=0.9999,
                        max_iters=100000, verbose=True, label=""):
    """Subgradient method with exact peak (no smoothing)."""
    n = len(a_init)
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = eta_start

    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        cur = 2.0 * n * b.max()
        if cur < best_score - 1e-15:
            best_score = cur
            best_a = a.copy()

        k_peak = int(np.argmax(b))
        grad = np.zeros(n)
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        grad[idx] = 2.0 * a[k_peak - idx]
        grad = grad - np.dot(a, grad)

        log_update = -eta * grad
        log_update = np.clip(log_update, -3.0, 3.0)
        a = a * np.exp(log_update)
        a = normalize(a)
        eta *= decay

        if verbose and it % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} SG it={it} ({elapsed:.0f}s): eta={eta:.6f} "
                  f"score={cur:.8f} best={best_score:.8f}")

    return best_a, best_score

    n = len(a_init)
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for step in range(10000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]
        if peak_val <= best_score / (2.0 * n) + 1e-15:
            break

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_idx = np.argsort(contribs)[::-1][:5]

        improved = False
        for ci in top_idx:
            i = int(idx[ci])
            j = peak_k - i
            if a[i] < 1e-14:
                continue
            for factor in [0.95, 0.9, 0.85, 0.8]:
                a_trial = a.copy()
                a_trial[i] *= factor
                a_trial = normalize(a_trial)
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try reducing both i and j
            for ci in top_idx[:3]:
                i = int(idx[ci])
                j = peak_k - i
                if i == j or a[j] < 1e-14:
                    continue
                for f1 in [0.92, 0.88]:
                    for f2 in [0.92, 0.88]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = normalize(a_trial)
                        sc = score_normalized(a_trial)
                        if sc < best_score - 1e-14:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
        if not improved:
            break

    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    {label} PS ({elapsed:.0f}s): best={best_score:.8f}")
    return best_a, best_score


def coordinate_descent(a_init, time_limit, verbose=True, label=""):
    """Coordinate descent with proportional redistribution."""
    n = len(a_init)
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    factors = [0.7, 0.85, 0.95, 1.05, 1.15, 1.4]
    no_improve = 0
    max_no_improve = 100

    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        no_improve = 0
        improved_pass = False
        order = np.random.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            for f in factors:
                a_trial = a.copy()
                old_val = a_trial[i]
                a_trial[i] = old_val * f
                if a_trial[i] < 0:
                    continue
                a_trial = normalize(a_trial)
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-13:
                    best_score = sc
                    a = a_trial
                    best_a = a.copy()
                    improved_pass = True
                    no_improve = 0
                    break
        if not improved_pass:
            no_improve += 1

    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    {label} CD ({elapsed:.0f}s): best={best_score:.8f}")
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    return normalize(f_interp)


def check_and_update(a, label, best_seq, best_score):
    """Evaluate candidate with actual evaluate_sequence."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a_norm)}) ***")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization: LP-based direction finding (primary),
    mirror descent (secondary), subgradient descent, peak shaving,
    and coordinate descent for refinement.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float('inf')

    # Get parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = float(evaluate_sequence(best_seq.tolist()))
            print(f"Parent: n={len(best_seq)}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(768) / 768.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default uniform: n={len(best_seq)}, score={best_score:.8f}")

    n = len(best_seq)

    # === Phase 1: LP Optimization (primary) ===
    print(f"\n{'='*60}")
    print(f"Phase 1: LP Optimization (n={n}, rem={deadline - time.monotonic():.0f}s)")
    print(f"{'='*60}")
    rem = deadline - time.monotonic() - 300
    if rem > 60:
        t_alloc = min(rem * 0.55, 700.0)
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, sc_lp = lp_optimize(best_seq, t_alloc, verbose=True, label="LP1")
        best_seq, best_score = check_and_update(a_lp, "LP1", best_seq, best_score)

    # === Phase 2: Mirror Descent ===
    print(f"\n{'='*60}")
    print(f"Phase 2: Mirror Descent (best={best_score:.8f})")
    print(f"{'='*60}")
    configs = [
        (0.008, 0.0008, 0.04, 0.004, "MD1"),
        (0.004, 0.0004, 0.025, 0.002, "MD2"),
    ]
    for T_s, T_e, et_s, et_e, lbl in configs:
        rem = deadline - time.monotonic() - 200
        if rem < 20:
            break
        t_alloc = min(rem * 0.45, 120.0)
        print(f"  {lbl}: T=[{T_s},{T_e}], eta=[{et_s},{et_e}], time={t_alloc:.0f}s")
        a_md, sc_md = mirror_descent(best_seq, t_alloc, T_start=T_s, T_end=T_e,
                                     eta_start=et_s, eta_end=et_e, label=lbl)
        best_seq, best_score = check_and_update(a_md, lbl, best_seq, best_score)

    # === Phase 3: LP again from current best ===
    print(f"\n{'='*60}")
    print(f"Phase 3: LP Refinement (best={best_score:.8f})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic() - 150
    if rem > 60:
        t_alloc = min(rem * 0.55, 400.0)
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, sc_lp2 = lp_optimize(best_seq, t_alloc, verbose=True, label="LP2")
        best_seq, best_score = check_and_update(a_lp2, "LP2", best_seq, best_score)

    # === Phase 4: Subgradient Descent ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Subgradient Descent (best={best_score:.8f})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic() - 100
    if rem > 15:
        t_alloc = min(rem * 0.5, 80.0)
        a_sg, sc_sg = subgradient_descent(best_seq, t_alloc, eta_start=0.05,
                                          decay=0.99995, label="SG")
        best_seq, best_score = check_and_update(a_sg, "SG", best_seq, best_score)

    # === Phase 5: Peak Shaving ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Peak Shaving (best={best_score:.8f})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic() - 60
    if rem > 10:
        t_alloc = min(rem * 0.5, 60.0)
        a_ps, sc_ps = peak_shave(best_seq, t_alloc, verbose=True, label="PS")
        best_seq, best_score = check_and_update(a_ps, "PS", best_seq, best_score)

    # === Phase 6: Coordinate Descent ===
    print(f"\n{'='*60}")
    print(f"Phase 6: Coordinate Descent (best={best_score:.8f})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic() - 50
    if rem > 10:
        t_alloc = min(rem * 0.5, 50.0)
        a_cd, sc_cd = coordinate_descent(best_seq, t_alloc, verbose=True, label="CD")
        best_seq, best_score = check_and_update(a_cd, "CD", best_seq, best_score)

    # === Phase 7: Final LP + MD polish ===
    print(f"\n{'='*60}")
    print(f"Phase 7: Final Polish (best={best_score:.8f})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic() - 30
    if rem > 40:
        t_alloc = min(rem * 0.5, 200.0)
        print(f"  Final LP: {t_alloc:.0f}s")
        a_fl, sc_fl = lp_optimize(best_seq, t_alloc, verbose=True, label="LP_final")
        best_seq, best_score = check_and_update(a_fl, "LP_final", best_seq, best_score)

    rem = deadline - time.monotonic() - 20
    if rem > 15:
        t_alloc = min(rem * 0.5, 50.0)
        a_fm, sc_fm = mirror_descent(best_seq, t_alloc, T_start=0.003, T_end=0.0003,
                                     eta_start=0.015, eta_end=0.001, label="MD_final")
        best_seq, best_score = check_and_update(a_fm, "MD_final", best_seq, best_score)

    rem = deadline - time.monotonic() - 10
    if rem > 8:
        a_fp, sc_fp = peak_shave(best_seq, min(rem * 0.6, 20.0), verbose=True, label="PS_final")
        best_seq, best_score = check_and_update(a_fp, "PS_final", best_seq, best_score)

    # === Phase 8: Try different length (if time permits) ===
    rem = deadline - time.monotonic() - 15
    if rem > 60:
        print(f"\n{'='*60}")
        print(f"Phase 8: Alternative Length (best={best_score:.8f})")
        print(f"{'='*60}")
        alt_n = int(n * 1.33)
        if 256 <= alt_n <= 2048:
            t_alloc = min(rem * 0.5, 100.0)
            init_alt = interpolate_sequence(best_seq, alt_n)
            a_alt, sc_alt = lp_optimize(init_alt, t_alloc, verbose=True, label=f"alt_{alt_n}")
            best_seq, best_score = check_and_update(a_alt, f"alt_{alt_n}", best_seq, best_score)

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'='*60}")
    return [float(x) for x in best_seq]
```
