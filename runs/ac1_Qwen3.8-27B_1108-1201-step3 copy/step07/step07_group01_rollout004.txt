"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding
(combined with mirror descent and coordinate descent polishing).
The LP approach finds the optimal non-negative direction g that keeps
cross-correlation conv(a,g) bounded, giving a principled descent direction.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(a, b):
    """Linear cross-convolution a * b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def score_from_a(a):
    """Compute 2*n*max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (S * S))


def score_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def lp_direction(a, M, solver='highs'):
    """
    Solve LP: maximize sum(g) subject to g >= 0 and conv(a,g)_k <= M for all k.
    Returns the optimal g or None if infeasible.
    """
    n = len(a)
    # Build constraint matrix: A_ub[k, i] = a[k-i] if 0 <= k-i < n else 0
    # This means A_ub @ g = conv(a, g)
    
    # Efficient construction using the Toeplitz structure
    # Column i of A_ub has a[0], a[1], ..., a[n-1] starting at row i
    A_ub = np.zeros((2 * n - 1, n))
    for i in range(n):
        A_ub[i:i+n, i] = a
    
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)  # minimize -sum(g) equivalent to maximize sum(g)
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                     bounds=[(0, None)] * n, method=solver)
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def mirror_descent_optimize(a_init, time_limit, T_start=0.01, T_end=0.0005,
                            eta_start=0.02, eta_end=0.0005, max_iters=20000,
                            verbose=True, label=""):
    """Adaptive mirror descent with temperature annealing on the simplex."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    stuck_count = 0
    stuck_threshold = 500

    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, it / (0.7 * max_iters))
        T = T_start * (T_end / T_start) ** progress
        eta = eta_start * (eta_end / eta_start) ** progress

        b = fast_convolve(a)
        current_score = 2.0 * n * b.max()

        if current_score < best_score - 1e-15:
            best_score = current_score
            best_a = a.copy()
            stuck_count = 0
        else:
            stuck_count += 1

        # Compute soft-max weights and gradient
        scaled = b / T
        m = scaled.max()
        exp_scaled = np.exp(scaled - m)
        w = exp_scaled / exp_scaled.sum()
        
        # Gradient: d/d a_i of sum_k w_k * b_k = 2 * sum_k w_k * (cross-conv contribution)
        # b_k = sum_j a_j * a_{k-j}
        # db_k / da_i = a_{k-i} + a_{i-k} (when valid), with special case k=i giving 2*a_i
        # sum_k w_k * db_k/da_i = 2 * sum_k w_k * a_{|k-i|} (approximately, for valid indices)
        # More precisely: sum_k w_k * (a_{k-i} + a_{i-k}) where valid
        # This is 2 * (w * a)_i in the appropriate sense
        
        # Using FFT: the gradient is 2n * cross_correlate(w, a)
        fft_size = 1 << (3 * n - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        # We need sum_k w_k * a_{k-i} for each i
        # This is the cross-correlation of w and a
        # cross_corr[i] = sum_k w[k] * a[k-i] (with appropriate bounds)
        correl = irfft(np.conj(W) * A, fft_size)
        grad_a = 2.0 * n * correl[:n]

        # Multiplicative (mirror) update
        log_update = -eta * grad_a
        log_update = np.clip(log_update, -3.0, 3.0)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-300)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
        else:
            a = a / s

        # Stuck detection
        if stuck_count > stuck_threshold:
            rng = np.random.default_rng(it * 7919)
            noise = rng.standard_normal(n) * 0.01
            a = best_a + noise
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            stuck_count = 0

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} MD iter={it} ({elapsed:.0f}s): T={T:.5f} "
                  f"eta={eta:.5f} score={current_score:.7f} best={best_score:.7f}")

    return best_a, best_score


def coordinate_descent_fine(a_init, time_limit, verbose=True, label=""):
    """Coordinate descent with fine factor grid and no redistribution (work in free space)."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0.0).astype(np.float64)
    S = a.sum()
    if S < 1e-15:
        return a, float('inf')
    
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    
    # Fine factor grid
    factors = np.array([0.5, 0.6, 0.7, 0.8, 0.9, 0.95, 1.05, 1.1, 1.2, 1.3, 1.5, 2.0])
    consecutive_no_improve = 0
    max_no_improve = 300
    
    while time.monotonic() - t0 < time_limit and consecutive_no_improve < max_no_improve:
        consecutive_no_improve = 0
        improved_pass = False
        order = np.random.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            for f in factors:
                a_trial = a.copy()
                a_trial[i] *= f
                if a_trial[i] < 0:
                    continue
                sc = score_from_a(a_trial)
                if sc < best_score - 1e-14:
                    best_score = sc
                    a = a_trial.copy()
                    best_a = a.copy()
                    improved_pass = True
                    # Continue with this i to try more factors
                    break
        
        if not improved_pass:
            consecutive_no_improve += 1
            
    # Normalize at the end
    a = best_a / best_a.sum()
    return a, score_normalized(a)

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
