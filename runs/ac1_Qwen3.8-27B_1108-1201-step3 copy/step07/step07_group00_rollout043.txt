import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_subgradient(a, w):
    """
    Compute g[j] = sum_k w[k] * a[k-j] for j = 0, ..., n-1.
    w has length 2n-1.

    size = 2 * n - 1

    A_flip = a[::-1]
    A = rfft(A_flip, fft_size)
    # Convolution of w and a_flipped gives cross-correlation
    corr = irfft(W * A, fft_size)[:size]

    g = corr[n-1 : 2*n-1]  # indices n-1 to 2n-2

def multiplicative_gradient_descent(a, time_limit, K=50, eta_init=0.01):
    """
    Multiplicative gradient descent to minimize max(conv(a,a)) subject to sum(a)=1, a>=0.
    """
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    eta = eta_init
    no_improve = 0
    
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        cur_obj = b.max()
        
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
            eta = min(eta * 1.02, 0.1)  # slightly increase step
        else:
            no_improve += 1
            if no_improve > 5:
                eta *= 0.8
                no_improve = 0
                if eta < 1e-6:
                    break
        
        # Compute top-K peaks
        if K < len(b):
            top_indices = np.argpartition(b, -K)[-K:]
            top_b = b[top_indices]
        else:
            top_indices = np.arange(len(b))
            top_b = b
        
        # Soft weights (softmax-like)
        # Use a temperature to control concentration
        temp = cur_obj * 0.01 + 1e-10
        w_alpha = (top_b - top_b.min()) / (top_b.max() - top_b.min() + 1e-15)
        weights = np.ones(K) * 0.5 + w_alpha  # base weight + emphasis on peaks
        # Or: weights = np.exp((top_b - top_b.max()) / temp)
        
        # Compute subgradient
        g = np.zeros(n)
        for idx_k in range(len(top_indices)):
            k = top_indices[idx_k]
            w_k = weights[idx_k]
            if w_k < 1e-12:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            g[lo:hi+1] += w_k * a[k-hi : k-lo+1][::-1]
        
        # Multiplicative update
        a *= np.exp(-eta * g)
        a = np.maximum(a, 1e-30)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a /= s
    
    return best_a, best_obj

def lp_direction_dual(f, top_k, M):
    """
    Solve the dual LP to find the optimal direction.
    Returns g (the primal optimum) and M.
    """
    n = len(f)
    b = fast_convolve(f)
    M = b.max()
    
    top_idx = np.argpartition(b, -top_k)[-top_k:]
    K = len(top_idx)
    
    # Build A^T: shape (n, K), where A^T[j, k] = f[j] if j is valid for peak top_idx[k]
    AT = np.zeros((n, K))
    for col in range(K):
        k = top_idx[col]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        AT[lo:hi+1, col] = f[lo:hi+1]
    
    w = cp.Variable(K)
    constraints = [w >= 0, AT @ w >= 1]
    prob = cp.Problem(cp.Minimize(M * cp.sum(w)), constraints)
    prob.solve(solver=cp.ECOS, max_iters=5000)
    
    if prob.status not in ['optimal', 'optimal_inaccurate']:
        return None, M
    
    # The primal solution can be recovered from the dual
    # Actually, for the LP, we can just get g from the KKT conditions
    # Or we can solve the primal directly using the dual solution
    # Simpler: just solve the primal
    g = cp.Variable(n)
    prob2 = cp.Problem(cp.Maximize(cp.sum(g)), [g >= 0, AT.T @ g <= M])
    prob2.solve(solver=cp.ECOS, max_iters=5000)
    
    if prob2.status not in ['optimal', 'optimal_inaccurate']:
        return None, M
    
    g_val = np.maximum(g.value, 0)
    return g_val, M

"""
Optimized approach combining multiplicative gradient descent, LP-based refinement,
and multi-length exploration to minimize 2*n*max(conv(a,a))/(sum(a))^2.

Key improvements over previous approach:
1. Better initialization using the "ideal" sequence from sqrt((1-x^{2n-1})/(1-x))
2. Fast multiplicative gradient descent phase (thousands of iterations)
3. LP refinement with dual formulation (efficiency for small K)
4. Adaptive strategy: increase K, add noise, try different lengths
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def ideal_init(n, seed=None):
    """
    Compute first n coefficients of sqrt((1-x^{2n-1})/(1-x)), normalized to sum=1.
    This gives approximately flat auto-convolution.
    """
    a = np.zeros(n)
    a[0] = 1.0
    for k in range(1, n):
        s = 0.0
        lo = max(1, k - (n - 1))
        hi = min(k - 1, n - 1)
        if lo <= hi:
            s = np.dot(a[lo:hi+1], a[k-hi:k-lo+1][::-1])
        a[k] = (1.0 - s) / 2.0
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def compute_subgradient(a, top_indices, weights):
    """
    Compute subgradient g[j] = sum_k w[k] * a[k-j] for j = 0, ..., n-1.
    """
    n = len(a)
    g = np.zeros(n)
    for idx in range(len(top_indices)):
        k = top_indices[idx]
        w_k = weights[idx]
        if w_k < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        g[lo:hi+1] += w_k * a[k-hi:k-lo+1][::-1]
    return g


def run_mgd(a_init, time_limit, K=50, eta_init=0.05, label="mgd"):
    """
    Multiplicative Gradient Descent (fast phase).
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a /= s
    
    b = fast_convolve(a)
    best_a = a.copy()
    best_obj = b.max()
    eta = eta_init
    no_improve = 0
    it = 0
    
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        it += 1
        cur_obj = b.max()
        
        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
            eta = min(eta * 1.01, 0.5)
        else:
            no_improve += 1
            if no_improve > 3:
                eta *= 0.7
                no_improve = 0
                if eta < 1e-7:
                    break
        
        # Compute top-K
        if K < len(b):
            top_indices = np.argpartition(b, -K)[-K:]
            top_b = b[top_indices]
        else:
            top_indices = np.arange(len(b))
            top_b = b
        
        # Softmax weights
        temp = max(cur_obj * 0.005, 1e-10)
        exp_vals = np.exp((top_b - top_b.max()) / temp)
        weights = exp_vals / exp_vals.sum()
        
        # Compute subgradient
        g = compute_subgradient(a, top_indices, weights)
        
        # Multiplicative update
        a *= np.exp(-eta * g)
        a = np.maximum(a, 1e-15)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a /= s
        
        b = fast_convolve(a)
        
        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} obj={best_obj:.10f} score={2*n*best_obj:.6f} eta={eta:.4f} t={elapsed:.0f}s")
    
    return best_a, best_obj


def run_lp(a_init, time_limit, top_k_init=20, top_k_max=150, label="lp"):
    """
    LP-based direction finding with line search.
    Solves: max sum(g) s.t. g >= 0, (f*g)[k] <= M for top_K peaks.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a /= s
    
    b = fast_convolve(a)
    best_a = a.copy()
    best_obj = b.max()
    top_k = top_k_init
    no_improve = 0
    it = 0
    
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        if it % 30 == 0 and top_k < top_k_max:
            top_k = min(top_k + 10, top_k_max)
        
        M = b.max()
        if M < 1e-15:
            break
        
        # Select top peaks
        if top_k < len(b):
            top_idx = np.argpartition(b, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(b))
        K = len(top_idx)
        
        # Build constraint matrix A (K x n)
        A = np.zeros((K, n))
        for row in range(K):
            k = top_idx[row]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            A[row, lo:hi+1] = a[lo:hi+1]
        
        # Solve LP
        g_var = cp.Variable(n)
        prob = cp.Problem(cp.Maximize(cp.sum(g_var)), [g_var >= 0, A @ g_var <= M])
        try:
            prob.solve(solver=cp.ECOS, max_iters=100000, eps=1e-7)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, max_iters=100000, eps=1e-7)
            except Exception:
                break
        
        if prob.status not in ['optimal', 'optimal_inaccurate']:
            break
        
        g_val = np.maximum(g_var.value, 0) if g_var.value is not None else None
        if g_val is None or np.any(np.isnan(g_val)):
            break
        
        S_g = g_val.sum()
        if S_g <= 1.0 + 1e-4:
            no_improve += 1
            if no_improve > 5:
                top_k = min(top_k * 2, top_k_max)
                continue
            if no_improve > 20:
                break
            continue
        
        # Line search
        h = g_val / S_g
        a_new = (1.0 - 0.1) * a + 0.1 * h
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new > 1e-10:
            a_new /= s_new
        new_obj = fast_convolve(a_new).max()
        
        best_t = 0.1
        best_obj_t = new_obj
        for t in [0.03, 0.05, 0.08, 0.15, 0.2, 0.3, 0.5]:
            a_trial = (1.0 - t) * a + t * h
            a_trial = np.maximum(a_trial, 0)
            s_t = a_trial.sum()
            if s_t < 1e-10:
                continue
            a_trial /= s_t
            obj_t = fast_convolve(a_trial).max()
            if obj_t < best_obj_t:
                best_obj_t = obj_t
                best_t = t
        
        a = (1.0 - best_t) * a + best_t * h
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            a /= s
        
        b = fast_convolve(a)
        cur_obj = b.max()
        
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                # Restart with noise
                rng = np.random.default_rng(it)
                a = best_a + rng.standard_normal(n) * 0.001
                a = np.maximum(a, 0)
                s = a.sum()
                if s > 1e-10:
                    a /= s
                b = fast_convolve(a)
                no_improve = 0
            
            if no_improve > 80:
                break
        
        if it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} obj={best_obj:.10f} score={2*n*best_obj:.6f} K={top_k} t={elapsed:.0f}s")
    
    return best_a, best_obj


def coordinate_polish(a_init, time_limit):
    """Coordinate descent on individual elements."""
    n = len(a_init)
    a = a_init.copy()
    b = fast_convolve(a)
    best_obj = b.max()
    
    t0 = time.monotonic()
    multipliers = [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]
    
    improved_any = True
    passes = 0
    while improved_any and time.monotonic() - t0 < time_limit:
        improved_any = False
        passes += 1
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[j] < 1e-14:
                continue
            best_j_obj = best_obj
            best_j_val = a[j]
            for mult in multipliers:
                new_val = a[j] * mult
                a[j] = new_val
                s = a.sum()
                if s < 1e-10:
                    a[j] = best_j_val
                    continue
                a /= s
                obj = fast_convolve(a).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = new_val
                a[j] = best_j_val
                s = a.sum()
                if s > 1e-10:
                    a /= s
            if best_j_val != a[j]:
                a[j] = best_j_val
                s = a.sum()
                if s > 1e-10:
                    a /= s
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    improved_any = True
                    b = fast_convolve(a)
    
    return a, best_obj

Wait, I have a performance concern with the coordinate descent. For n=2000, each coordinate involves computing the full convolution (O(n log n)). With 6 multipliers per coordinate, that's 6 * n * O(n log n) = O(n^2 log n) per pass. For n=2000, that's about 2000 * 6 * 2000 * 11 ≈ 265 million operations per pass. With numpy vectorization, each convolution takes maybe 1ms (FFT of size 4096), so 6 * 2000 = 12000 convolutions per pass, taking about 12 seconds per pass. That's too slow for many passes.

Let me optimize: I'll only try coordinates that have significant values (a[j] > threshold), and use fewer multipliers.

Actually, a better approach for coordinate descent: for each coordinate j, the optimal value (given all other coordinates fixed) can be found by solving a 1D optimization problem. The convolution changes in a structured way when we change a[j]:
- b[k] changes by: 2*a[j]*a[k-j] (for the cross terms) + a[j]^2 (for the self term at k=2j)

So the change in max_b when we change a[j] from its current value to a new value v is:
- For each k: b[k] + 2*(v - a[j])*a[k-j] + (v^2 - a[j]^2) * [k == 2j]
- (where a[k-j] is 0 if k-j is out of range)

The new max is max_k of these. We want to choose v to minimize the new max. This is a 1D problem that can be solved by bisection or by finding the minimum of the upper envelope of linear functions.

This is much faster than recomputing the full convolution each time! Let me implement this.

For a given j, changing a[j] from a_j to v affects:
- b[k] for k in [j, j+n-1] (where a[k-j] is non-zero)

Specifically, the cross-correlation term: 2*(v - a_j) * a[k-j] added to b[k] for k = j, j+1, ..., min(j+n-1, 2n-2).
And the self term: v^2 - a_j^2 added to b[2j] (if 2j < 2n-1).

So the new convolution at position k is:
b_new[k] = b[k] + 2*(v-a_j)*a[k-j] + (v^2-a_j^2)*[k==2j]

We want to minimize max_k b_new[k].

The function max_k b_new[k] as a function of v is the upper envelope of a set of linear (and one quadratic) functions. The minimum of this upper envelope can be found by:
1. Finding the k* that achieves the max at v = a_j (current)
2. At the optimum, either:
   a. The max-k is unchanged, and we just need the linear function to be minimized → set v such that the constraint is tight
   b. The max-k changes, and we're at the intersection of two constraint lines

This can be solved by bisection on v, or by finding the intersection points.

For simplicity, let me just use bisection:
1. Lower bound: v_low = 0
2. Upper bound: v_high = some large value (e.g., 2*a_j)
3. Bisect to find the v that minimizes max_k b_new[k]

Each evaluation of max_k b_new[k] takes O(n) time (just computing the modified convolution values). With ~50 bisection iterations, each coordinate takes O(n) * 50 = O(50n). For n=2000, that's 100,000 operations per coordinate, or 200 million for all coordinates. That's about 0.1-0.5 seconds per pass. Much better!

Let me implement this optimized coordinate descent:

"""
Optimized minimization of 2*n*max(conv(a,a))/(sum(a))^2 using:
1. "Ideal" initialization from sqrt((1-x^{2n-1})/(1-x)) power series
2. Fast multiplicative gradient descent (thousands of iterations)
3. LP-based direction finding with adaptive peak selection
4. Efficient coordinate descent with O(n) incremental convolution updates
5. Multi-length exploration and final refinement

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def ideal_init(n):
    """
    Compute first n coefficients of sqrt((1-x^{2n-1})/(1-x)), normalized to sum=1.
    This gives approximately flat auto-convolution.
    """
    a = np.zeros(n)
    a[0] = 1.0
    for k in range(1, n):
        lo = max(1, k - (n - 1))
        hi = min(k - 1, n - 1)
        s = 0.0
        if lo <= hi:
            s = np.dot(a[lo:hi+1], a[k-hi:k-lo+1][::-1])
        a[k] = (1.0 - s) / 2.0
    a = np.maximum(a, 0.0)
    ss = a.sum()
    if ss > 1e-10:
        a /= ss
    else:
        a = np.ones(n) / n
    return a


def compute_subgradient(a, top_indices, weights):
    """Compute g[j] = sum_k w_k * a[k-j] for j=0,...,n-1."""
    n = len(a)
    g = np.zeros(n)
    for idx in range(len(top_indices)):
        k = int(top_indices[idx])
        w_k = weights[idx]
        if w_k < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        g[lo:hi+1] += w_k * a[k-hi:k-lo+1][::-1]
    return g


def run_mgd(a_init, time_limit, K=50, eta_init=0.03, label="mgd"):
    """Multiplicative Gradient Descent for fast initial convergence."""
    n = len(a_init)
    a = np.maximum(a_init, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a /= s

    b = fast_convolve(a)
    best_a = a.copy()
    best_obj = b.max()
    eta = eta_init
    no_improve = 0
    it = 0

    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        it += 1
        cur_obj = b.max()

        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
            eta = min(eta * 1.005, 0.5)
        else:
            no_improve += 1
            if no_improve > 3:
                eta *= 0.75
                no_improve = 0
                if eta < 1e-8:
                    break

        # Top-K peaks
        if K < len(b):
            top_indices = np.argpartition(b, -K)[-K:]
            top_b = b[top_indices]
        else:
            top_indices = np.arange(len(b))
            top_b = b

        # Softmax weights with adaptive temperature
        temp = max(cur_obj * 0.01, 1e-12)
        exp_vals = np.exp((top_b - top_b.max()) / temp)
        w_sum = exp_vals.sum()
        if w_sum < 1e-30:
            weights = np.ones(len(top_indices)) / len(top_indices)
        else:
            weights = exp_vals / w_sum

        # Subgradient
        g = compute_subgradient(a, top_indices, weights)

        # Multiplicative update
        a *= np.exp(-eta * g)
        a = np.maximum(a, 1e-15)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a /= s

        b = fast_convolve(a)

        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} obj={best_obj:.10f} score={2*n*best_obj:.6f} eta={eta:.5f} t={elapsed:.0f}s")

    return best_a, best_obj
