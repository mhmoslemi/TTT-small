def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    result = irfft(A * A, fft_size)[:size]
    return result


def cross_convolve(a, g):
    """Cross-convolution a*g via FFT."""
    n = len(a)
    m = len(g)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    G = rfft(g, fft_size)
    return irfft(A * G, fft_size)[:size]

                a = best_a.copy()
                b_a = fast_convolve(a)
                no_improve += 1
            else:
                a = a / sm
                b_a = fast_convolve(a)
                cur_score = float(2.0 * n * b_a.max())
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
        else:
            no_improve += 1

        if no_improve > 80:
            rng = np.random.default_rng(it * 7919 + 13)
            noise = rng.standard_normal(n) * 0.015
            a = best_a + noise
            a = np.maximum(a, 1e-12)
            a = a / a.sum()
            b_a = fast_convolve(a)
            no_improve = 0

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} ({elapsed:.0f}s): "
                  f"best={best_score:.7f} no_improve={no_improve}")

    return best_a, best_score


def subgradient_optimize(a_init, time_limit, verbose=True, label=""):
    """Subgradient method with Nesterov-type acceleration."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0.0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()

    # Nesterov acceleration
    a_prev = a.copy()
    t_val = 1.0
    no_improve = 0

    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        t_new = 0.5 * (1 + np.sqrt(1 + 4 * t_val ** 2))
        beta = (t_val - 1) / t_new
        y = a + beta * (a - a_prev)
        y = np.maximum(y, 1e-15)
        y = y / y.sum()

        b_y = fast_convolve(y)
        k_peak = int(np.argmax(b_y))

        # Subgradient: gradient of b_y[k_peak] w.r.t. y
        grad = np.zeros(n)
        i_start = max(0, k_peak - n + 1)
        i_end = min(n - 1, k_peak)
        idx = np.arange(i_start, i_end + 1)
        grad[idx] = 2.0 * y[k_peak - idx]

        # Step size (diminishing)
        step = 0.1 / (1 + 0.01 * it)

        # Mirror descent update (multiplicative)
        log_update = -step * grad / y
        log_update = np.clip(log_update, -5.0, 5.0)
        a_new = y * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        score_new = score_from_a(a_new)
        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a_prev = a
        a = a_new
        t_val = t_new

        if no_improve > 200:
            rng = np.random.default_rng(it * 31)
            noise = rng.standard_normal(n) * 0.02
            a = best_a + noise
            a = np.maximum(a, 1e-12)
            a = a / a.sum()
            a_prev = a.copy()
            t_val = 1.0
            no_improve = 0

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} SG iter={it} ({elapsed:.0f}s): "
                  f"best={best_score:.7f}")

    return best_a, best_score


def coordinate_descent(a_init, time_limit, verbose=True, label=""):
    """Coordinate descent with proportional redistribution and line search."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()

    factors = [0.5, 0.7, 0.8, 0.85, 0.9, 0.95, 1.05, 1.1, 1.2, 1.3, 1.5]
    consecutive_no_improve = 0
    max_no_improve = 500

    while time.monotonic() - t0 < time_limit and consecutive_no_improve < max_no_improve:
        consecutive_no_improve = 0
        improved_pass = False
        order = np.random.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[i]
            if old_val < 1e-15:
                continue
            for f in factors:
                new_val = old_val * f
                if new_val < 0:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                diff = a_trial.sum() - 1.0
                if abs(diff) < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] = a_trial[mask] - diff * a_trial[mask] / others_sum
                if a_trial.min() < -1e-12:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                a_trial = a_trial / a_trial.sum()
                sc = score_from_a(a_trial)
                if sc < best_score - 1e-14:
                    best_score = sc
                    a = a_trial.copy()
                    best_a = a.copy()
                    improved_pass = True
                    break

        if not improved_pass:
            consecutive_no_improve += 1

        if verbose and consecutive_no_improve % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} CD ({elapsed:.0f}s): best={best_score:.7f} "
                  f"no_improve={consecutive_no_improve}")

    return best_a, best_score


def peak_shave(a_init, time_limit, verbose=True, label=""):
    """Targeted peak shaving: reduce contributions to the top peaks."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()

    for step in range(20000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        b_work = b.copy()
        if n > 1:
            b_work[0] = -np.inf
            b_work[-1] = -np.inf
        peak_k = int(np.argmax(b_work))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_count = min(10, len(idx))
        top_idx = np.argsort(contribs)[::-1][:top_count]

        improved = False
        for ci in top_idx:
            i = int(idx[ci])
            j = peak_k - i
            if a[i] < 1e-14:
                continue
            for factor in [0.95, 0.9, 0.85, 0.8, 0.7, 0.6]:
                a_trial = a.copy()
                old_val = a_trial[i]
                a_trial[i] = old_val * factor
                diff = old_val - a_trial[i]
                if diff < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] = others + diff * others / others_sum
                a_trial = np.maximum(a_trial, 0)
                sm = a_trial.sum()
                if sm < 1e-15:
                    continue
                a_trial = a_trial / sm
                sc = score_from_a(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try reducing both i and j simultaneously
            for ci in top_idx[:5]:
                i = int(idx[ci])
                j = peak_k - i
                if i == j or a[j] < 1e-14:
                    continue
                for f1 in [0.9, 0.8, 0.7]:
                    for f2 in [0.9, 0.8, 0.7]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        sm = a_trial.sum()
                        if sm < 1e-15:
                            continue
                        a_trial = a_trial / sm
                        sc = score_from_a(a_trial)
                        if sc < best_score - 1e-14:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            # Try multi-peak: target second highest peak
            b_sorted = np.sort(b)[::-1]
            if len(b_sorted) >= 2 and b_sorted[1] > best_score / (2.0 * n) * 0.95:
                # The second peak is also significant; continue anyway
                pass
            else:
                break

        if verbose and step % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} PS step={step} ({elapsed:.0f}s): best={best_score:.7f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate using the actual evaluate_sequence function."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.7f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase LP-guided optimization:
    1. LP-based optimization at parent length (primary)
    2. Subgradient method with acceleration
    3. Local refinement (peak shaving + coordinate descent)
    4. Alternative lengths exploration
    5. Final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.7f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = p_score
    else:
        # Fallback: uniform sequence
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Fallback uniform: n=768, score={best_score:.7f}")

    n_best = len(best_seq)

    # === Phase 1: LP Optimization (primary, ~800s) ===
    print(f"\n=== Phase 1: LP Optimization (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic() - 600
    if rem > 60:
        t_alloc = min(rem, 800.0)
        print(f"  LP: n={n_best}, budget={t_alloc:.0f}s")
        a_lp, _ = lp_optimize(best_seq, t_alloc, label="LP_main")
        best_seq, best_score = check_and_update(a_lp, "LP_main", best_seq, best_score)

    # === Phase 2: Second LP round with different parameters (~200s) ===
    print(f"\n=== Phase 2: LP Round 2 (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 40:
            t_alloc = min(rem, 250.0)
            # Perturb and run again
            n_cur = len(best_seq)
            noise = rng.standard_normal(n_cur) * 0.01
            a_pert = best_seq + noise
            a_pert = np.maximum(a_pert, 1e-12)
            a_pert = a_pert / a_pert.sum()
            print(f"  LP2: n={n_cur}, budget={t_alloc:.0f}s")
            a_lp2, _ = lp_optimize(a_pert, t_alloc, label="LP_pert")
            best_seq, best_score = check_and_update(a_lp2, "LP_pert", best_seq, best_score)

    # === Phase 3: Subgradient method (~150s) ===
    print(f"\n=== Phase 3: Subgradient (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 250
        if rem > 30:
            t_alloc = min(rem, 200.0)
            print(f"  SG: n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_sg, _ = subgradient_optimize(best_seq, t_alloc, label="SG")
            best_seq, best_score = check_and_update(a_sg, "SG", best_seq, best_score)

    # === Phase 4: Peak Shaving (~100s) ===
    print(f"\n=== Phase 4: Peak Shaving (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            t_alloc = min(rem, 120.0)
            print(f"  PS: n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_ps, _ = peak_shave(best_seq, t_alloc, label="PS")
            best_seq, best_score = check_and_update(a_ps, "PS", best_seq, best_score)

    # === Phase 5: Coordinate Descent (~100s) ===
    print(f"\n=== Phase 5: Coordinate Descent (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 15:
            t_alloc = min(rem, 120.0)
            print(f"  CD: n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_cd, _ = coordinate_descent(best_seq, t_alloc, label="CD")
            best_seq, best_score = check_and_update(a_cd, "CD", best_seq, best_score)

    # === Phase 6: LP on refined result (~100s) ===
    print(f"\n=== Phase 6: LP Refinement (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 30:
            t_alloc = min(rem, 100.0)
            print(f"  LP3: n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_lp3, _ = lp_optimize(best_seq, t_alloc, label="LP_refine")
            best_seq, best_score = check_and_update(a_lp3, "LP_refine", best_seq, best_score)

    # === Phase 7: Alternative lengths (~100s) ===
    print(f"\n=== Phase 7: Alternative Lengths (best={best_score:.7f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        alt_lengths = []
        for factor in [1.25, 1.5, 0.85, 2.0, 0.75]:
            alt_n = int(n_best * factor)
            if 128 <= alt_n <= 4096 and alt_n != n_best and alt_n not in alt_lengths:
                alt_lengths.append(alt_n)

        for alt_n in alt_lengths[:3]:
            rem = deadline - time.monotonic() - 50
            if rem < 30:
                break
            t_alloc = min(rem * 0.5, 60.0)
            init_alt = interpolate_sequence(best_seq, alt_n)
            a_alt, _ = lp_optimize(init_alt, t_alloc, label=f"alt_n{alt_n}")
            best_seq, best_score = check_and_update(a_alt, f"alt_n{alt_n}", best_seq, best_score)

    # === Phase 8: Final polish (~50s) ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            t_alloc = min(rem * 0.5, 30.0)
            a_ps_fin, _ = peak_shave(best_seq, t_alloc, label="final_PS")
            best_seq, best_score = check_and_update(a_ps_fin, "final_PS", best_seq, best_score)

        rem = deadline - time.monotonic() - 8
        if rem > 8:
            t_alloc = min(rem * 0.6, 20.0)
            a_cd_fin, _ = coordinate_descent(best_seq, t_alloc, label="final_CD")
            best_seq, best_score = check_and_update(a_cd_fin, "final_CD", best_seq, best_score)

        rem = deadline - time.monotonic() - 6
        if rem > 6:
            t_alloc = min(rem * 0.7, 15.0)
            a_lp_fin, _ = lp_optimize(best_seq, t_alloc, label="final_LP")
            best_seq, best_score = check_and_update(a_lp_fin, "final_LP", best_seq, best_score)

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.7f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
