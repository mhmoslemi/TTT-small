```python
def spectral_flatten(a, n_iter=1000):
    n = len(a)
    L = 1 << (2*n-1).bit_length()  # power of 2 >= 2n-1
    a_pad = np.zeros(L)
    a_pad[:n] = a
    
    for i in range(n_iter):
        A = np.fft.fft(a_pad)
        |A| = np.abs(A)
        # Target magnitude
        target = np.sqrt(np.mean(|A|**2))
        # Flatten
        A_new = np.zeros(L, dtype=complex)
        mask = |A| > 1e-10
        A_new[mask] = target * A[mask] / |A|[mask]
        # Inverse FFT
        a_new = np.fft.ifft(A_new).real
        a_new = a_new[:n]
        # Project to simplex
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s
        else:
            break
        a_pad[:n] = a_new
        a_pad[n:] = 0
    
    return a_new
```

```python
import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from time import monotonic


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def score_fn(a):
    """Compute the objective: 2*n*max(conv(a,a))/(sum(a))^2"""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    return 2.0 * n * fast_convolve(a_norm).max()


def spectral_flatten(a0, time_limit, n_iter_max=20000):
    """
    Gerchberg-Saxton style spectral flattening.
    Iteratively: FFT -> flatten magnitude -> IFFT -> project to simplex.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = score_fn(a)
    L = 1 << (2 * n - 1).bit_length()
    t0 = monotonic()
    stall = 0

    for it in range(n_iter_max):
        if monotonic() - t0 > time_limit:
            break

        a_pad = np.zeros(L)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        mag = np.abs(A)
        phase = A / np.maximum(mag, 1e-15)

        # Target: flat magnitude
        target_mag = np.sqrt(np.mean(mag ** 2))

        # Soft flattening: blend current and target
        alpha = 0.5  # blending factor
        mag_new = (1 - alpha) * mag + alpha * target_mag
        mag_new = np.maximum(mag_new, 1e-15)

        A_new = mag_new * phase
        a_new = np.fft.ifft(A_new).real[:n]

        # Project to simplex
        a_new = np.maximum(a_new, 0)
        sc = a_new.sum()
        if sc < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / sc

        sc = score_fn(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            # Try smaller step
            a_try = (1 - 0.5) * a + 0.5 * a_new
            a_try = np.maximum(a_try, 0)
            sc_try = a_try.sum()
            if sc_try > 1e-15:
                a_try = a_try / sc_try
                sc_try = score_fn(a_try)
                if sc_try < best_score - 1e-12:
                    best_score = sc_try
                    best_a = a_try.copy()
                    a = a_try
                    stall = 0

        if stall > 500:
            # Perturb to escape local minimum
            a = a + np.random.standard_normal(n) * 0.001
            a = np.maximum(a, 0)
            a = a / a.sum()
            stall = 0

        if it % 2000 == 0 and it > 0:
            print(f"    SpectralFlatten iter={it} ({monotonic()-t0:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def adaptive_lbgfs(a0, time_limit, T_start=50, T_end=5, maxiter=50000):
    """
    L-BFGS with geometric temperature cooling.
    Minimizes 2n * softmax_T(conv(a,a)) / (sum a)^2 in log-space.
    """
    n = len(a0)
    a_safe = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    z0 = np.log(a_safe)
    z0 = z0 - z0.mean()
    t0 = monotonic()
    stopped = [False]
    best_a = a0.copy()
    best_score = score_fn(a0)

    def z_to_a(z):
        z_s = z - z.max()
        e = np.exp(z_s)
        return e / e.sum()

    def make_func(jac):
        def func(z):
            if monotonic() - t0 > time_limit:
                stopped[0] = True
                return 1e15
            T = T_start * (T_end / T_start) ** (minimize._call_count / max(maxiter * 0.8, 1))
            T = max(T, T_end)
            a = z_to_a(z)
            b = fast_convolve(a)
            b_max = b.max()
            scaled = (b - b_max) / T
            exp_b = np.exp(scaled)
            b_sum = exp_b.sum()
            lse = b_max + T * np.log(b_sum)
            S = a.sum()
            val = 2.0 * n * lse / (T * S * S) * T
            # Simplify: 2n * (b_max + T*log(sum(exp((b-b_max)/T)))) / S^2
            val = 2.0 * n * (b_max + T * np.log(b_sum)) / (S * S)
            if jac:
                w = exp_b / b_sum
                grad_a = np.zeros(n)
                for k in range(len(b)):
                    if w[k] < 1e-12:
                        continue
                    js = max(0, k - n + 1)
                    je = min(n - 1, k)
                    idx = np.arange(js, je + 1)
                    grad_a[idx] += 2.0 * w[k] * a[k - idx]
                grad_a = (2.0 * n / (S * S)) * (grad_a - 2.0 * (b_max + T * np.log(b_sum)) / S)
                S2 = np.dot(grad_a, a)
                grad_z = a * (grad_a - a * S2)
                return val, grad_z
            return val
        return func

    # Run with cooling schedule
    n_phases = 4
    T_schedule = [T_start * (T_end / T_start) ** (i / n_phases) for i in range(n_phases)]
    T_schedule.append(T_end)

    minimize._call_count = 0
    z_current = z0.copy()

    for phase, T in enumerate(T_schedule):
        if monotonic() - t0 > time_limit - 5:
            break
        phase_time = (time_limit - (monotonic() - t0)) / (n_phases + 1 - phase)
        if phase_time < 5:
            break

        minimize._call_count = 0

        def func_phase(z, T_val=T):
            if monotonic() - t0 > time_limit:
                stopped[0] = True
                return 1e15
            a = z_to_a(z)
            b = fast_convolve(a)
            b_max = b.max()
            scaled = (b - b_max) / T_val
            exp_b = np.exp(scaled)
            b_sum = exp_b.sum()
            S = a.sum()
            val = 2.0 * n * (b_max + T_val * np.log(b_sum)) / (S * S)
            return val

        def jac_phase(z, T_val=T):
            if monotonic() - t0 > time_limit:
                stopped[0] = True
                return np.zeros(n)
            a = z_to_a(z)
            b = fast_convolve(a)
            b_max = b.max()
            scaled = (b - b_max) / T_val
            exp_b = np.exp(scaled)
            w = exp_b / exp_b.sum()
            S = a.sum()
            lse_val = b_max + T_val * np.log(exp_b.sum())

            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-12:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]

            grad_a = (2.0 * n / (S * S)) * (grad_a / 1.0) - (4.0 * n * lse_val / (S ** 3))
            # More carefully:
            # d/da_i [2n * lse / S^2] = 2n * (d(lse)/da_i * S^2 - lse * 2S) / S^4
            # = 2n/S^2 * d(lse)/da_i - 4n*lse/S^3
            # d(lse)/da_i = sum_k w_k * dm_k/da_i = sum_k w_k * a[k-i] (for valid k-i)
            # So d(lse)/da_i = (grad_a_raw)_i where grad_a_raw[i] = sum_k w_k * a[k-i]

            # Let me redo: lse = max + T*log(sum(exp((b-max)/T)))
            # d(lse)/da_i = sum_k exp((b_k - max)/T) / sum_j exp((b_j - max)/T) * d(b_k)/da_i * (1/T) * T
            # = sum_k w_k * d(b_k)/da_i
            # where w_k = exp((b_k - max)/T) / sum_j exp((b_j - max)/T)

            # d(b_k)/da_i = a[k-i] if 0 <= k-i < n, else 0
            # So d(lse)/da_i = sum_k w_k * a[k-i] (valid indices)

            # We already computed grad_a as sum_k 2*w_k * a[k-i] (the factor 2 is from b=a*a)
            # Actually d(a*a)_k / da_i = a[k-i] (counting both a[i]*a[k-i] and a[k-i]*a[i] gives 2*a[k-i] when i != k-i)
            # Wait no: (a*a)_k = sum_j a[j]*a[k-j]. d/da_i = a[k-i] + a[i]*0 ... 
            # Actually: (a*a)_k = sum_j a_j * a_{k-j}. If we differentiate wrt a_i:
            # = sum_j [delta_{ji} * a_{k-j} + a_j * delta_{k-j,i}]
            # = a_{k-i} + a_{k-i} (when i != k-i, both terms contribute)
            # = a_{k-i} (when i == k-i, only one term)
            # For cross-correlation: it's 2*a_{k-i} when i != k-i/2, and a_{k-i} when i = k-i (i.e., k=2i)
            # For simplicity and since n is large, the factor 2 vs 1 barely matters.
            # Let's just use the gradient we computed (which has the factor 2 built in from the sum)

            # Actually re-examining: (a*a)_k = sum_j a[j]*a[k-j]
            # d/da_i (a*a)_k = a[k-i] (from the a[j] term) + a[k-i] (from the a[k-j] term where k-j=i)
            # But these are the same term counted twice only when j=i AND k-j=i, i.e., k=2i.
            # In general: d/da_i (a*a)_k = a[k-i] if 0<=k-i<n
            # (each a_i appears in exactly one product a_i * a_{k-i} in the sum, unless k-i = i in which case it's a_i^2 and derivative is 2*a_i)

            # For our purposes, using the "2*w_k*a[k-i]" approximation is fine.

            grad_lse = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-12:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_lse[idx] += w[k] * a[k - idx]

            # d[obj]/da_i = 2n/S^2 * grad_lse[i] - 4n*lse_val/S^3
            obj_grad_a = (2.0 * n / (S * S)) * grad_lse - (4.0 * n * lse_val / (S ** 3))

            # Transform to z-space: a = softmax(z)
            # da_i/dz_j = a_i*(delta_ij - a_j)
            # dz_j = sum_i (da_i/dz_j) * obj_grad_a_i = a_j*obj_grad_a_j - a_j*sum_i a_i*obj_grad_a_i
            S3 = np.dot(obj_grad_a, a)
            grad_z = a * (obj_grad_a - a * S3)

            return val if False else grad_z

        # We need a combined function that returns both value and gradient
        class LbfgsFunc:
            def __init__(self, T_val):
                self.T = T_val
                self.val = 0
                self.grad = None
            def __call__(self, z):
                if monotonic() - t0 > time_limit:
                    stopped[0] = True
                    return 1e15
                a = z_to_a(z)
                b = fast_convolve(a)
                b_max = b.max()
                scaled = (b - b_max) / self.T
                exp_b = np.exp(scaled)
                b_sum = exp_b.sum()
                S = a.sum()
                lse_val = b_max + self.T * np.log(b_sum)
                self.val = 2.0 * n * lse_val / (S * S)

                w = exp_b / b_sum
                grad_lse = np.zeros(n)
                for k in range(len(b)):
                    if w[k] < 1e-12:
                        continue
                    js = max(0, k - n + 1)
                    je = min(n - 1, k)
                    idx = np.arange(js, je + 1)
                    grad_lse[idx] += w[k] * a[k - idx]

                obj_grad_a = (2.0 * n / (S * S)) * grad_lse - (4.0 * n * lse_val / (S ** 3))
                S3 = np.dot(obj_grad_a, a)
                self.grad = a * (obj_grad_a - a * S3)
                return self.val

        for temp in [T, T * 0.7, T * 0.5, T_end if phase == n_phases else T * 0.3]:
            if monotonic() - t0 > time_limit - 3:
                break
            remaining = time_limit - (monotonic() - t0)
            t_sub = min(remaining * 0.6, 80.0)
            if t_sub < 5:
                break
            lb_func = LbfgsFunc(temp)
            try:
                res = minimize(
                    fun=lambda z: lb_func(z),
                    x0=z_current,
                    jac=lambda z: (lb_func(z), lb_func.grad),
                    method='L-BFGS-B',
                    options={'maxiter': 3000, 'ftol': 1e-14, 'gtol': 1e-10,
                             'maxfun': 20000, 'disp': False}
                )
                if not stopped[0]:
                    z_current = res.x
                    a_check = z_to_a(z_current)
                    sc = score_fn(a_check)
                    if sc < best_score - 1e-12:
                        best_score = sc
                        best_a = a_check.copy()
            except Exception:
                pass

        elapsed = monotonic() - t0
        print(f"    L-BFGS phase={phase} T={T:.1f} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def nesterov_subgrad(a0, time_limit, lr_init=0.005, momentum=0.9, top_k=50):
    """Nesterov-accelerated subgradient method with peak focusing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = score_fn(a)
    x_prev = a.copy()
    v = np.zeros(n)
    lr = lr_init
    t0 = monotonic()
    k = 0
    no_improve = 0

    while monotonic() - t0 < time_limit:
        k += 1
        # Nesterov lookahead
        beta = (k - 1) / (k + 2.0)
        y = a + beta * (a - x_prev)
        x_prev = a.copy()
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            sy = s
        else:
            y = y / sy

        # Compute subgradient at y
        b = fast_convolve(y)
        M = b.max()
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]
        temp = 0.003 * max(pv.max(), 1e-10)
        scaled = (pv - pv.max()) / temp
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / ws

        # Subgradient: sum over peaks of w_k * a[k-i]
        sg = np.zeros(n)
        for ii in range(len(peaks)):
            if w[ii] < 1e-13:
                continue
            kk = int(peaks[ii])
            js = max(0, kk - n + 1)
            je = min(n - 1, kk)
            idx = np.arange(js, je + 1)
            sg[idx] += w[ii] * y[kk - idx]

        # Full gradient of 2n*M/S^2
        S_y = y.sum()
        obj_sg = (2.0 * n / (S_y * S_y)) * sg - (4.0 * n * M / (S_y ** 3))

        # Nesterov update
        v = momentum * v - lr * obj_sg
        a_candidate = a + v
        a_new = project_simplex(a_candidate)
        a_new = a_new / a_new.sum()

        sc = score_fn(a_new)
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive learning rate
        lr *= 0.9995
        if no_improve > 200:
            a = best_a + np.random.standard_normal(n) * 0.002
            a = project_simplex(a)
            a = a / a.sum()
            v = np.zeros(n)
            lr = max(lr * 1.5, 0.003)
            no_improve = 0

        if k % 5000 == 0:
            print(f"    Nesterov k={k} ({monotonic()-t0:.0f}s): score={best_score:.6f} lr={lr:.6f}")

    return best_a, best_score


def peak_shave_advanced(a0, time_limit, max_steps=20000):
    """Advanced peak shaving with swap operations and mass redistribution."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = score_fn(a)
    t0 = monotonic()
    failed = 0

    for step in range(max_steps):
        if monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]
        target = best_score / (2.0 * n)

        if peak_val <= target + 1e-15:
            break

        improved = False

        # Strategy 1: Reduce top contributors to peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        total_contrib = contribs.sum()

        if total_contrib > 1e-20:
            top_contrib = np.argsort(contribs)[::-1][:8]
            for ci in top_contrib:
                i = int(idx[ci])
                j = int(peak_k - i)
                old_i = a[i]
                if old_i < 1e-14:
                    continue

                # Try reducing i and redistributing
                for factor in [0.9, 0.85, 0.8, 0.75, 0.7]:
                    diff = old_i * (1 - factor)
                    if diff < 1e-15:
                        continue
                    # Find positions that don't contribute to peak
                    a_trial = a.copy()
                    a_trial[i] = old_i * factor
                    # Redistribute to random non-peak-contributing positions
                    non_peak = np.ones(n, dtype=bool)
                    non_peak[peak_k - idx] = False  # positions that pair with some idx to form peak_k
                    # Actually, position p contributes to peak_k if a[p]*a[peak_k-p] is part of the sum
                    # peak_k = sum_{p: valid} a[p]*a[peak_k-p]
                    # So position p contributes if 0 <= peak_k-p < n, i.e., p in [peak_k-n+1, peak_k]
                    # Which is exactly idx!
                    non_peak[idx] = False
                    non_peak[i] = True  # we reduced i, keep it
                    candidates = np.where(non_peak)[0]
                    if len(candidates) > 0:
                        # Distribute to top candidates by "safety" (low contribution to any peak)
                        np.random.seed(step % 10000)
                        picks = np.random.choice(candidates, size=min(10, len(candidates)), replace=False)
                        a_trial[picks] += diff / len(picks)
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    sc = score_fn(a_trial)
                    if sc < best_score - 1e-13:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        # Strategy 2: Swap mass between two positions
        if not improved:
            # Find two positions: one that helps the peak, one that doesn't
            js2 = max(0, peak_k - n + 1)
            je2 = min(n - 1, peak_k)
            idx2 = np.arange(js2, je2 + 1)
            contribs2 = a[idx2] * a[peak_k - idx2]
            # Position with highest contribution
            worst_pos = int(idx2[np.argmax(contribs2)])
            # Find a "safe" position (low contribution to multiple peaks)
            # Compute total contribution of each position to top few peaks
            top_peaks = np.argpartition(b, -10)[-10:]
            load = np.zeros(n)
            for pk in top_peaks:
                js3 = max(0, pk - n + 1)
                je3 = min(n - 1, pk)
                idx3 = np.arange(js3, je3 + 1)
                load[idx3] += b[pk] * a[idx3] * a[pk - idx3] / max(b[pk] * a[idx3].sum(), 1e-15)

            safe_positions = np.argsort(load)[:20]
            for sp in safe_positions:
                if sp == worst_pos or a[sp] > a[worst_pos] * 0.5:
                    continue
                for swap_frac in [0.1, 0.2, 0.3]:
                    amt = a[worst_pos] * swap_frac
                    if amt < 1e-14:
                        continue
                    a_trial = a.copy()
                    a_trial[worst_pos] -= amt
                    a_trial[sp] += amt
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    sc = score_fn(a_trial)
                    if sc < best_score - 1e-13:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        # Strategy 3: Random perturbation
        if not improved:
            for attempt in range(3):
                a_trial = a + np.random.standard_normal(n) * 0.0003
                a_trial = project_simplex(a_trial)
                a_trial = a_trial / a_trial.sum()
                sc = score_fn(a_trial)
                if sc < best_score - 1e-13:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break

        if improved:
            failed = 0
        else:
            failed += 1
            if failed > 100:
                break

    return best_a, best_score


def diverse_init(n, seed=0):
    """Generate diverse initial sequences."""
    np.random.seed(seed)
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine taper
    t = np.linspace(0, np.pi, n)
    c = 0.5 + 0.5 * np.cos(t)
    c = np.maximum(c, 0.01)
    inits.append(c / c.sum())

    # U-shape (high at edges)
    t = np.linspace(0, 1, n)
    u = 0.3 + 0.7 * (2 * t - 1) ** 2
    inits.append(u / u.sum())

    # Gaussian center
    t = np.linspace(0, 1, n)
    g = np.exp(-20 * (t - 0.5) ** 2)
    inits.append(g / g.sum())

    # Random Dirichlet
    r = np.random.dirichlet(np.ones(n) * 0.5)
    inits.append(r)

    # Bumps
    t = np.linspace(0, 1, n)
    b = 0.5 + 0.3 * np.sin(4 * np.pi * t) + 0.2 * np.sin(7 * np.pi * t)
    b = np.maximum(b, 0.01)
    inits.append(b / b.sum())

    return inits


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate and update best."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. Diverse initializations + spectral flattening
    2. Adaptive-temperature L-BFGS (geometric cooling)
    3. Nesterov-accelerated subgradient
    4. Advanced peak shaving
    5. Final L-BFGS polish
    6. Alternative length exploration
    """
    np.random.seed(seed)
    deadline = monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is not None and len(parent) >= 64:
        best_seq = parent.copy()
        best_score = p_score
        n_main = len(parent)
    else:
        n_main = 768
        best_seq = np.ones(n_main) / n_main
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Initial uniform: n={n_main}, score={best_score:.6f}")

    # === Phase 1: Spectral flattening from multiple inits ===
    print(f"\n=== Phase 1: Spectral Flattening (best={best_score:.6f}) ===")
    rem = deadline - monotonic() - 800
    if rem > 30:
        t_sf = min(rem * 0.6, 120.0)

        # Try parent + a few diverse inits
        inits = [best_seq]
        for i in range(4):
            inits.append(diverse_init(n_main, seed=seed + i))

        for i, init in enumerate(inits):
            rem = deadline - monotonic() - 700
            if rem < 10:
                break
            t_sub = min(t_sf / len(inits), rem * 0.8)
            if t_sub < 5:
                break
            print(f"  Spectral init={i} ({t_sub:.0f}s)...")
            a_sf, _ = spectral_flatten(init, t_sub)
            best_seq, best_score = check_and_update(a_sf, f"spectral_{i}", best_seq, best_score)

    # === Phase 2: Adaptive-temperature L-BFGS ===
    print(f"\n=== Phase 2: L-BFGS (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 400
        if rem > 30:
            t_lb = min(rem * 0.7, 350.0)
            print(f"  L-BFGS n={len(best_seq)}, budget={t_lb:.0f}s")
            a_lb, _ = adaptive_lbgfs(best_seq, t_lb, T_start=80, T_end=3, maxiter=30000)
            best_seq, best_score = check_and_update(a_lb, "lbgfs_main", best_seq, best_score)

    # === Phase 3: Nesterov subgradient ===
    print(f"\n=== Phase 3: Nesterov Subgrad (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 300
        if rem > 20:
            t_ns = min(rem * 0.6, 150.0)
            n_best = len(best_seq)
            top_k = max(30, n_best // 5)
            print(f"  Nesterov n={n_best}, budget={t_ns:.0f}s")
            a_ns, _ = nesterov_subgrad(best_seq, t_ns, lr_init=0.004, momentum=0.9, top_k=top_k)
            best_seq, best_score = check_and_update(a_ns, "nesterov", best_seq, best_score)

    # === Phase 4: Advanced peak shaving ===
    print(f"\n=== Phase 4: Peak Shaving (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 200
        if rem > 15:
            t_ps = min(rem * 0.5, 100.0)
            print(f"  PeakShave n={len(best_seq)}, budget={t_ps:.0f}s")
            a_ps, _ = peak_shave_advanced(best_seq, t_ps)
            best_seq, best_score = check_and_update(a_ps, "peak_shave", best_seq, best_score)

    # === Phase 5: Second L-BFGS pass (lower temps) ===
    print(f"\n=== Phase 5: L-BFGS Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 150
        if rem > 20:
            t_lb2 = min(rem * 0.6, 100.0)
            print(f"  L-BFGS polish n={len(best_seq)}, budget={t_lb2:.0f}s")
            a_lb2, _ = adaptive_lbgfs(best_seq, t_lb2, T_start=20, T_end=2, maxiter=10000)
            best_seq, best_score = check_and_update(a_lb2, "lbgfs_polish", best_seq, best_score)

    # === Phase 6: Another Nesterov + peak shave cycle ===
    print(f"\n=== Phase 6: Refinement Cycle (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 100
        if rem > 20:
            t_ns2 = min(rem * 0.4, 60.0)
            a_ns2, _ = nesterov_subgrad(best_seq, t_ns2, lr_init=0.002, momentum=0.9,
                                        top_k=max(20, len(best_seq) // 6))
            best_seq, best_score = check_and_update(a_ns2, "nesterov2", best_seq, best_score)

            rem = deadline - monotonic() - 50
            if rem > 10:
                t_ps2 = min(rem * 0.5, 40.0)
                a_ps2, _ = peak_shave_advanced(best_seq, t_ps2)
                best_seq, best_score = check_and_update(a_ps2, "peak_shave2", best_seq, best_score)

    # === Phase 7: Try alternative lengths ===
    print(f"\n=== Phase 7: Length Exploration (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        alt_lengths = []
        for factor in [0.85, 1.15, 1.3, 0.75]:
            alt_n = int(n_best * factor)
            if 256 <= alt_n <= 4096 and alt_n != n_best and alt_n not in alt_lengths:
                alt_lengths.append(alt_n)

        for alt_n in alt_lengths[:2]:
            rem = deadline - monotonic() - 50
            if rem < 30:
                break
            t_alt = min(rem * 0.5, 80.0)
            if t_alt < 15:
                break
            init_alt = interpolate_sequence(best_seq, alt_n)
            # Quick spectral + L-BFGS
            a_sf_alt, _ = spectral_flatten(init_alt, min(t_alt * 0.4, 30.0))
            best_seq, best_score = check_and_update(a_sf_alt, f"spec_alt{alt_n}", best_seq, best_score)

            rem = deadline - monotonic() - 30
            if rem > 15:
                a_lb_alt, _ = adaptive_lbgfs(a_sf_alt, min(rem * 0.6, 40.0), T_start=40, T_end=3)
                best_seq, best_score = check_and_update(a_lb_alt, f"lbgfs_alt{alt_n}", best_seq, best_score)

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 5
        if rem > 10:
            t_final = min(rem * 0.6, 30.0)
            a_fin, _ = adaptive_lbgfs(best_seq, t_final, T_start=10, T_end=1, maxiter=5000)
            best_seq, best_score = check_and_update(a_fin, "final_lbgfs", best_seq, best_score)

        rem = deadline - monotonic() - 2
        if rem > 5:
            a_fin2, _ = peak_shave_advanced(best_seq, min(rem * 0.8, 10.0))
            best_seq, best_score = check_and_update(a_fin2, "final_ps", best_seq, best_score)

    # Final verification
    if best_seq is None:
        best_seq = np.ones(768) / 768.0
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
