def cosine_window(n):
    t = np.linspace(0, np.pi, n)
    return np.cos(t / 2) ** 2

def gaussian_window(n, sigma_frac=0.25):
    x = (np.arange(n) - n / 2.0) / n
    return np.exp(-x ** 2 / (2.0 * sigma_frac ** 2))

def triangular_window(n):
    return 1.0 - 2.0 * np.abs(np.arange(n) - (n - 1) / 2.0) / n

def _lbgfs_func(z):
    n = _gs['n']
    T = _gs['T']
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    return (2.0 * n / T) * log_sum_exp

def _lbgfs_jac(z):
    n = _gs['n']
    T = _gs['T']
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    # Cross-correlation via FFT: c[j] = sum_k w[k] * a[k-j]
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(W * np.conj(A), fft_size)[:n]
    grad_a = 2.0 * n / T * corr
    S = np.dot(grad_a, a)
    return a * (grad_a - S)

def lbgfs_optimize(a0, time_limit, T_start=100.0, T_end=0.5, n_stages=6, maxiter_per_stage=2000):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)

    for stage_T in temps:
        elapsed = time.monotonic() - t0
        if elapsed > time_limit * 0.95:
            break
        remaining = time_limit - elapsed
        if remaining < 15:
            break

        _gs['T'] = stage_T
        _gs['n'] = n

        iters = maxiter_per_stage
        try:
            res = sp_minimize(
                fun=_lbgfs_func, x0=z, jac=_lbgfs_jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-15, 'gtol': 1e-9,
                         'maxfun': iters * 200, 'disp': False}
            )
            z = res.x
        except Exception as e:
            print(f"    L-BFGS T={stage_T:.2f} error: {e}")

    return softmax_from_z(z)

def peak_shave(a0, time_limit, n_steps=50000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

        improved = False
        top_cis = np.argsort(contribs)[::-1][:5]

        for ci in top_cis:
            i = idx[ci]
            if a[i] < 1e-14:
                continue
            for factor in [0.995, 0.99, 0.98, 0.96, 0.93, 0.9]:
                a_trial = a.copy()
                diff = a_trial[i] * (1.0 - factor)
                a_trial[i] *= factor
                if diff < 1e-16:
                    continue
                # Redistribute to non-contributing coordinates
                mask = np.ones(n, dtype=bool)
                mask[idx] = False
                if mask.sum() > 0:
                    other_sum = a_trial[mask].sum()
                    if other_sum > 1e-15:
                        a_trial[mask] += diff * a_trial[mask] / other_sum
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try multi-coordinate reduction
            if len(top_cis) >= 2:
                a_trial = a.copy()
                for ci in top_cis[:3]:
                    j_i = idx[ci]
                    a_trial[j_i] *= 0.98
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st < 1e-15:
                    break
                a_trial = a_trial / st
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True

        if not improved:
            # Stuck - try random perturbation
            if step % 200 == 0 and step > 0:
                a_trial = a.copy()
                pert_idx = np.random.randint(0, n, size=min(10, n))
                for pi in pert_idx:
                    a_trial[pi] *= (1.0 + 0.05 * np.random.randn())
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-15:
                    a_trial = a_trial / st
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()

    return best_a

def mirror_descent_peak(a0, time_limit, lr_init=0.1, n_steps=50000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    z = z_from_a(a)
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        threshold = max_b * 0.98
        active = np.where(b >= threshold)[0]

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        z_new = z - lr * g
        a_new = softmax_from_z(z_new)
        score_new = 2.0 * n * fast_convolve(a_new).max()

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            z = z_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 80:
            lr *= 0.85
            no_improve = 0
        lr = max(lr, 1e-8)

    return best_a

def coordinate_descent(a0, time_limit, n_passes=50):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for p in range(n_passes):
        if time.monotonic() - t0 > time_limit:
            break
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-15:
                continue
            base = a.copy()
            base[i] = 0
            base_sum = base.sum()
            if base_sum < 1e-15:
                continue
            best_t = a[i]
            best_local = best_score
            for factor in [0.8, 0.9, 1.0, 1.1, 1.25]:
                trial_t = a[i] * factor
                if trial_t < 0 or trial_t > 1.0:
                    continue
                a_trial = base * (1.0 - trial_t) + trial_t * np.eye(n)[i]
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                sc = 2.0 * n * fast_convolve(a_trial).max()
                if sc < best_local:
                    best_local = sc
                    best_t = trial_t
            if best_t != a[i]:
                a = base * (1.0 - best_t) + best_t * np.eye(n)[i]
                a = a / a.sum()
                sc = 2.0 * n * fast_convolve(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

    return best_a

def propose_candidate(seed=42, budget_s=1800.0):
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # Load parent
    print("=== Loading parent ===")
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            try_update(p, "parent")
            print(f"  Parent: n={len(p)}, score={best_score:.6f}")
    except Exception:
        pass

    # Phase 1: Multi-start with different lengths and initializations
    print(f"\n=== Phase 1: Multi-start (best={best_score:.6f}) ===")

    configs = [
        # (n, init_type, T_start, T_end, n_stages, maxiter, time_alloc_frac)
        (1024, 'cosine', 200, 1.0, 6, 3000, 0.18),
        (2048, 'cosine', 300, 2.0, 7, 2000, 0.22),
        (768, 'cosine', 100, 0.5, 6, 3000, 0.12),
        (1536, 'gauss', 200, 1.0, 6, 2500, 0.15),
        (3072, 'cosine', 400, 3.0, 7, 1500, 0.18),
    ]

    for n_init, init_type, T_s, T_e, n_st, maxit, t_frac in configs:
        remaining = deadline - time.monotonic()
        if remaining < 80:
            break
        t_alloc = min(remaining * t_frac, remaining * 0.35)
        if t_alloc < 40:
            continue

        print(f"\n  Config: n={n_init}, init={init_type}, T:{T_s}->{T_e}, time={t_alloc:.0f}s")

        if init_type == 'cosine':
            a_init = cosine_window(n_init)
        elif init_type == 'gauss':
            a_init = gaussian_window(n_init, 0.25)
        elif init_type == 'tri':
            a_init = triangular_window(n_init)
        else:
            a_init = np.ones(n_init)

        a_init = np.maximum(a_init, 0)
        a_init = a_init / a_init.sum()

        a_opt = lbgfs_optimize(a_init, t_alloc, T_start=T_s, T_end=T_e,
                               n_stages=n_st, maxiter_per_stage=maxit)
        try_update(a_opt, f"n{n_init}_{init_type}")

        # Peak shave
        remaining = deadline - time.monotonic()
        if remaining > 30:
            t_ps = min(remaining * 0.1, 80.0)
            if t_ps > 15:
                a_ps = peak_shave(a_opt, t_ps)
                try_update(a_ps, f"n{n_init}_{init_type}_peak")

    # Phase 2: Refine best found
    print(f"\n=== Phase 2: Refinement (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # L-BFGS at low temperature
        remaining = deadline - time.monotonic()
        if remaining > 80:
            t1 = min(remaining * 0.35, 350.0)
            print(f"  L-BFGS low-T: n={n_best}, T:5->0.1, time={t1:.0f}s")
            a_r = lbgfs_optimize(best_seq, t1, T_start=5.0, T_end=0.1,
                                 n_stages=5, maxiter_per_stage=4000)
            try_update(a_r, "refine_lowT")

        # Peak shaving
        remaining = deadline - time.monotonic()
        if remaining > 40:
            t2 = min(remaining * 0.25, 200.0)
            print(f"  Peak shave: time={t2:.0f}s")
            a_ps = peak_shave(best_seq, t2, n_steps=100000)
            try_update(a_ps, "refine_peak")

        # Mirror descent
        remaining = deadline - time.monotonic()
        if remaining > 30:
            t3 = min(remaining * 0.2, 120.0)
            print(f"  Mirror descent: time={t3:.0f}s")
            a_md = mirror_descent_peak(best_seq, t3, lr_init=0.05)
            try_update(a_md, "refine_mirror")

        # Very low temperature L-BFGS
        remaining = deadline - time.monotonic()
        if remaining > 40:
            t4 = min(remaining * 0.3, 150.0)
            print(f"  L-BFGS ultra-low-T: T:1->0.05, time={t4:.0f}s")
            a_ul = lbgfs_optimize(best_seq, t4, T_start=1.0, T_end=0.05,
                                  n_stages=4, maxiter_per_stage=6000)
            try_update(a_ul, "ultra_lowT")

    # Phase 3: Length extrapolation
    print(f"\n=== Phase 3: Length extrapolation (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        target_n = min(4096, max(2048, n_best * 2))
        if target_n > n_best:
            remaining = deadline - time.monotonic()
            if remaining > 80:
                old_x = np.linspace(0, 1, n_best)
                new_x = np.linspace(0, 1, target_n)
                a_interp = np.interp(new_x, old_x, best_seq)
                a_interp = np.maximum(a_interp, 0)
                a_interp = a_interp / a_interp.sum()

                t_alloc = min(remaining * 0.4, 250.0)
                print(f"  Interpolate to n={target_n}, time={t_alloc:.0f}s")
                a_large = lbgfs_optimize(a_interp, t_alloc, T_start=50.0, T_end=0.5,
                                         n_stages=6, maxiter_per_stage=2000)
                try_update(a_large, f"interp_n{target_n}")

                remaining = deadline - time.monotonic()
                if remaining > 30:
                    a_lps = peak_shave(a_large, min(remaining * 0.3, 80.0))
                    try_update(a_lps, f"interp_n{target_n}_peak")

    # Phase 4: Final micro-polish
    print(f"\n=== Phase 4: Final polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        remaining = deadline - time.monotonic()
        if remaining > 20:
            t_f = min(remaining * 0.6, 80.0)
            a_f = lbgfs_optimize(best_seq, t_f, T_start=0.5, T_end=0.01,
                                 n_stages=3, maxiter_per_stage=5000)
            try_update(a_f, "final_lbgfs")

        remaining = deadline - time.monotonic()
        if remaining > 15:
            a_fp = peak_shave(best_seq, min(remaining * 0.6, 40.0), n_steps=30000)
            try_update(a_fp, "final_peak")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(768) / 768
        best_score = evaluate_safe(best_seq)
        print(f"\nFALLBACK SCORE: {best_score:.6f}")

    return [float(x) for x in best_seq]
