import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    if '_lp_solver_cached_v2' in globals():
        return _lp_solver_cached_v2
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CBC, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):

                return solver_name
        except Exception:

    return _lp_solver_cached_v2


def solve_lp_best_response(a_fixed, prev_b=None, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    solve_kwargs = {}
    if prev_b is not None:
        try:
            b_var.value = prev_b
            M_prev = float(fast_convolve(a_fixed).max())
            M_var.value = M_prev

    try:

    except Exception:

                prob.solve(solver=alt)
                break
            except Exception:

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

def compute_gradient(a, weights):
    """Compute gradient of sum_k w[k]*c[k] where c = conv(a,a).
    grad[i] = 2 * sum_k w[k] * a[k-i] (where k-i is valid)
    """
    n = len(a)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * weights[k] * a[k - idx]
    return grad


def lse_bfgs_refine(a0, time_limit, p_start=50.0, p_end=500.0, n_stages=4):
    """Log-sum-exp smoothed BFGS with adaptive sharpening."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    t0 = time.monotonic()
    
    # Work in log-space for simplex constraint
    a_safe = np.maximum(a, 1e-300)
    z = np.log(a_safe)
    
    def make_func(p_val):
        def func(z_vec):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a_vec = np.exp(z_vec - z_vec.max())
            a_vec = a_vec / a_vec.sum()
            c = fast_convolve(a_vec)
            pc = p_val * c
            max_pc = pc.max()
            lse_val = max_pc + np.log(np.exp(pc - max_pc).sum())
            return 2.0 * n * lse_val / p_val
        return func

    def make_jac(p_val):
        def jac(z_vec):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_vec)
            a_vec = np.exp(z_vec - z_vec.max())
            a_vec = a_vec / a_vec.sum()
            c = fast_convolve(a_vec)
            pc = p_val * c
            max_pc = pc.max()
            exp_pc = np.exp(pc - max_pc)
            w = exp_pc / exp_pc.sum()
            grad_a = compute_gradient(a_vec, w)
            # Project to tangent space of simplex
            S = np.dot(grad_a, a_vec)
            grad_z = a_vec * (grad_a - S)
            return grad_z
        return jac

    # Geometric schedule for p
    ps = np.geomspace(p_start, p_end, n_stages)
    iters_total = 6000
    iters_per_stage = iters_total // n_stages
    
    z_curr = z.copy()
    for curr_p in ps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        try:
            res = scipy_minimize(
                fun=make_func(curr_p),
                x0=z_curr,
                jac=make_jac(curr_p),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-16, 'gtol': 1e-10,
                         'maxfun': iters_per_stage * 80, 'disp': False}
            )
            z_curr = res.x
        except Exception:
            pass
    
    a_out = np.exp(z_curr - z_curr.max())
    a_out = a_out / a_out.sum()
    return a_out


def lp_minimax_multi_avg(a0, time_limit, solver=None, n_max_iters=10000):
    """
    LP-based minimax with THREE averaging schemes:
    1. Simple running average (best worst-case guarantee)
    2. Exponential moving average
    3. Nesterov accelerated
    Returns the best of all three.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    t0 = time.monotonic()
    
    # Running average
    running_sum = a.copy()
    running_count = 1
    # EMA
    ema = a.copy()
    ema_alpha = 0.1
    # Nesterov
    y_nest = a.copy()
    nest_iter = 0
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    b_prev = None
    
    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP for best response to current iterate
        # Alternate between the three iterates for diversity
        if it % 3 == 0:
            x_curr = running_sum / running_count
            x_curr = np.maximum(x_curr, 0)
            if x_curr.sum() < 1e-15:
                x_curr = a / a.sum()
            else:
                x_curr = x_curr / x_curr.sum()
        elif it % 3 == 1:
            x_curr = ema
        else:
            x_curr = y_nest
        
        b_new, M_val = solve_lp_best_response(x_curr, prev_b=b_prev, solver=solver)
        if b_new is None:
            continue
        b_prev = b_new.copy()
        
        # Update all three averages
        running_sum += b_new
        running_count += 1
        
        ema = (1 - ema_alpha) * ema + ema_alpha * b_new
        # Decay alpha slowly
        ema_alpha = max(0.01, 0.1 / (1 + 0.01 * it))
        
        alpha_nest = 2.0 / (nest_iter + 2)
        y_nest = (1 - alpha_nest) * y_nest + alpha_nest * b_new
        y_nest = np.maximum(y_nest, 0)
        sy = y_nest.sum()
        if sy < 1e-15:
            y_nest = b_new.copy()
        else:
            y_nest = y_nest / sy
        nest_iter += 1
        
        # Evaluate all three
        for label, candidate in [("run", running_sum / running_count), 
                                  ("ema", ema),
                                  ("nest", y_nest)]:
            cand = np.maximum(candidate, 0)
            sc_sum = cand.sum()
            if sc_sum < 1e-15:
                continue
            cand = cand / sc_sum
            score = 2.0 * n * fast_convolve(cand).max()
            if score < best_score:
                best_score = score
                best_a = cand.copy()
        
        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LPMM iter={it+1} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate sequence safely."""

        if not a_list or sum(a_list) < 0.01:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

            a_lp2, _ = lp_minimax_multi_avg(best_seq, t1, solver=solver, n_max_iters=3000)
            try_update(a_lp2, "lpmm_R2")
        
        rem = deadline - time.monotonic()
        t2 = min(rem * 0.5, 100.0)
        if t2 > 15:
            a_lse2 = lse_bfgs_refine(best_seq, t2, p_start=100.0, p_end=1000.0, n_stages=3)
            try_update(a_lse2, "lse_R2")
    
    # === Phase 6: Try n=1024 (~300s) ===
    print(f"\n=== Phase 6: n=1024 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        n_alt = 1024
        t_alloc = min(rem * 0.45, 250.0)
        if t_alloc > 30:
            print(f"  Alt n={n_alt}, budget={t_alloc:.0f}s")
            init_alt = interpolate_sequence(best_seq, n_alt)
            a_alt, _ = lp_minimax_multi_avg(init_alt, t_alloc * 0.5, solver=solver, n_max_iters=3000)
            try_update(a_alt, f"lpmm_n{n_alt}")
            
            rem = deadline - time.monotonic()
            if rem > 30 and best_seq is not None and len(best_seq) == n_alt:
                a_alt2 = lse_bfgs_refine(best_seq, min(rem * 0.5, 100.0), p_start=50.0, p_end=500.0)
                try_update(a_alt2, f"lse_n{n_alt}")
    
    # === Phase 7: Try n=1536 if time remains (~150s) ===
    print(f"\n=== Phase 7: n=1536 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        n_alt2 = 1536
        t_alloc = min(rem * 0.6, 150.0)
        if t_alloc > 30:
            print(f"  Alt n={n_alt2}, budget={t_alloc:.0f}s")
            init_alt2 = interpolate_sequence(best_seq, n_alt2)
            a_alt2, _ = lp_minimax_multi_avg(init_alt2, t_alloc, solver=solver, n_max_iters=2000)
            try_update(a_alt2, f"lpmm_n{n_alt2}")
    
    # === Phase 8: Final polish (~100s) ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_final = min(rem * 0.5, 60.0)
        a_final = lse_bfgs_refine(best_seq, t_final, p_start=200.0, p_end=2000.0, n_stages=2)
        try_update(a_final, "final_lse")
        
        rem = deadline - time.monotonic()
        if rem > 15:
            a_ps_final, _ = peak_shave_v2(best_seq, min(rem * 0.6, 40.0), n_steps=10000)
            try_update(a_ps_final, "final_ps")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
