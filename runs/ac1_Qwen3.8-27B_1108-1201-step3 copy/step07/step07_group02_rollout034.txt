This is related to the problem of constructing "low cross-correlation" sequences, which has been studied extensively in radar/communications.

**Let me just implement a clean, efficient version with the following improvements over the existing code:**

1. **Better subgradient method**: 
   - Use the exact subgradient (not just the average over active set)
   - Use Polyak step size: `eta = (f(a) - f_lower_bound) / ||g||^2`
   - Keep a running average of all iterates
   - Use a better lower bound on f*

2. **QP-based local search**:
   - Identify the active set K
   - Formulate a QP: minimize t subject to (a*a)[k] <= t for k in K, sum a = 1, a >= 0
   - This is a convex QP (minimize a linear function subject to quadratic constraints... wait, that's not convex)
   
   Actually, minimizing t subject to quadratic equality/inequality constraints is non-convex. But if we linearize the constraints around the current point, we get a QP.

3. **Multiple restarts with perturbations**:
   - Take the best solution, add small structured noise, re-optimize
   - This can escape local minima

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(x):
    """Project x onto the probability simplex."""
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        r = np.max(np.where(cond)[0])
        theta = (cssv[r] - 1.0) / (r + 1)
        return np.maximum(x - theta, 0.0)
    return np.ones(n) / n


def get_convolution_gradient(a, k):
    """Gradient of (a*a)[k] w.r.t. a. Returns vector of length n."""
    n = len(a)
    g = np.zeros(n)
    # d/da[j] (a*a)[k] = 2*a[k-j] if 0 <= k-j < n, else 0
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        g[j] = 2.0 * a[k - j]
    return g


def compute_subgradient(a, active_indices):
    """Average subgradient over active set."""
    n = len(a)
    g = np.zeros(n)
    for k in active_indices:
        g += get_convolution_gradient(a, k)
    g /= len(active_indices)
    return g

def subgradient_mirror_descent(a0, time_limit, n_steps=200000, eta_init=0.01):
    """
    Subgradient mirror descent with Polyak step size and iterate averaging.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    z = np.log(np.maximum(a, 1e-300))
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    # Lower bound: 1/(2n-1) * 2n = 2n/(2n-1) > 1
    # Better lower bound: since sum conv = 1 and there are 2n-1 values,
    # max >= 1/(2n-1), so score >= 2n/(2n-1)
    f_lb = 2.0 * n / (2 * n - 1)
    
    avg_a = a.copy()
    t0 = time.monotonic()
    eta = eta_init
    no_improve = 0
    best_avg_score = float('inf')
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        f_val = 2.0 * n * max_b
        
        # Find active set (within 1% of max)
        threshold = max_b - 0.005 * max_b
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        
        # Compute subgradient
        g = compute_subgradient(a, active)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break
        
        # Polyak step size
        gap = f_val - f_lb
        if gap < 1e-10:
            break
        eta_polyak = gap / (g_norm * g_norm + 1e-15)
        # Use a damped version
        eta = min(eta_polyak, 1.0 / (step + 1) * 10)
        eta = max(eta, 1e-8)
        
        # Mirror descent step
        z_new = z - eta * g
        a_new = np.exp(z_new - z_new.max())
        ss = a_new.sum()
        a_new /= ss
        
        # Evaluate new point
        f_new = 2.0 * n * fast_convolve(a_new).max()
        
        # Update average
        avg_a = ((step + 1) * avg_a + a_new) / (step + 2)
        
        # Check average
        f_avg = 2.0 * n * fast_convolve(avg_a).max()
        if f_avg < best_avg_score:
            best_avg_score = f_avg
        
        # Update
        if f_new < f_val:
            z = z_new
            a = a_new
            if f_new < best_score - 1e-12:
                best_score = f_new
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            # Reduce step size if no improvement
            if no_improve > 50:
                eta *= 0.9
                no_improve = 0
        
        if (step + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGradMD step={step+1} ({elapsed:.0f}s): best={best_score:.6f}, avg={best_avg_score:.6f}, eta={eta:.2e}")
    
    # Also check the average
    if best_avg_score < best_score:
        best_score = best_avg_score
        # Re-optimize avg_a
        avg_a_norm = avg_a / avg_a.sum()
        best_a = avg_a_norm.copy()
    
    return best_a, best_score


def active_set_qp_refine(a0, time_limit, n_iters=5000):
    """
    Local refinement using linearized constraints on the active set.
    At each step, linearize the active constraints and solve a QP.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        f_val = 2.0 * n * max_b
        
        # Active set
        threshold = max_b - 0.002 * max_b
        active = np.where(b >= threshold)[0]
        
        if len(active) == 0:
            break
        
        # Try to find a direction d that reduces all active convolution values
        # Linearized: (a*a)[k] + 2*d^T a_shifted <= max_b - delta
        # where (a*a)[k] + sum_j d[j]*2*a[k-j] approximates new value
        
        # Set up: minimize ||d||^2 subject to:
        #   sum_j d[j] * 2*a[k-j] <= -delta for all k in active
        #   sum_j d[j] = 0 (maintain normalization)
        #   a + d >= 0 (maintain non-negativity, approximately)
        
        # This is a QP. Let's solve it with scipy.
        # Actually, let's use a simpler approach: find d that minimizes
        # the maximum of the linearized active constraints.
        
        # For each active k, the linearized change is: sum_j d[j] * 2*a[k-j]
        # We want this to be as negative as possible for all active k.
        
        # Simple approach: use the negative subgradient as the direction
        g = compute_subgradient(a, active)
        # Project out the mean (to maintain sum=1 to first order)
        g = g - np.mean(g)
        
        # Line search
        alpha = 0.01
        improved = False
        for _ in range(20):
            d = alpha * g
            a_trial = a + d
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-10:
                break
            a_trial = a_trial / st
            f_trial = 2.0 * n * fast_convolve(a_trial).max()
            if f_trial < best_score - 1e-12:
                a = a_trial
                best_score = f_trial
                best_a = a.copy()
                improved = True
                break
            alpha *= 0.7
        
        if not improved:
            # Try smaller steps or different directions
            alpha = 0.001
            for trial_factor in [0.5, 0.7, 1.0, 1.5]:
                d = -alpha * trial_factor * g
                d = d - np.mean(d)
                a_trial = a + d
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st < 1e-10:
                    continue
                a_trial = a_trial / st
                f_trial = 2.0 * n * fast_convolve(a_trial).max()
                if f_trial < best_score - 1e-12:
                    a = a_trial
                    best_score = f_trial
                    best_a = a.copy()
                    improved = True
                    break
            if not improved:
                break
    
    return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS with softmax parameterization and smoothed max (log-sum-exp) objective
2. Subgradient mirror descent with Polyak step size and iterate averaging
3. FFT-based cross-correlation for efficient gradient computation
4. Multiple perturbation-restart cycles
5. Coordinate descent peak shaving
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def compute_cross_corr(a, w):
    """
    Compute c[j] = sum_i a[i] * w[i+j] for j = 0,...,n-1
    This is the cross-correlation of a and w.
    Uses FFT for efficiency.
    """
    n = len(a)
    wlen = len(w)
    N = 1 << (max(n, wlen) - 1).bit_length()
    N = max(N, 2 * n)  # Ensure N >= 2n-1 for linear conv
    A = rfft(a, N)
    W = rfft(w, N)
    cross = irfft(np.conj(A) * W, N)[:n]
    return cross


def lbfgs_smoothed(a0, time_limit, T_values=None):
    """
    L-BFGS refinement using smoothed max objective in softmax parameterization.
    Gradient: grad_z[j] = (4n/T) * a[j] * (c[j] - dot(c,a))
    where c[j] = cross-correlation of a and w at lag j.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    z = np.log(np.maximum(a, 1e-300))
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    if T_values is None:
        T_values = [100.0, 50.0, 25.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.3]

    z_current = z.copy()

    for T in T_values:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func(z):
            a_local = softmax_from_z(z)
            b = fast_convolve(a_local)
            Tb = T * b
            max_Tb = Tb.max()
            lse = max_Tb + np.log(np.exp(Tb - max_Tb).sum())
            return (2.0 * n / T) * lse

        def jac(z):
            a_local = softmax_from_z(z)
            b = fast_convolve(a_local)
            Tb = T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            # c[j] = sum_i a[i] * w[i+j] (cross-correlation)
            c = compute_cross_corr(a_local, w)
            dot_ca = np.dot(c, a_local)
            grad_z = (4.0 * n / T) * a_local * (c - dot_ca)
            return grad_z

        iters = min(2000, int(remaining * 50))
        iters = max(iters, 100)

        try:
            res = minimize(
                fun=func,
                x0=z_current,
                jac=jac,
                method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxfun': iters * 100, 'disp': False}
            )
            z_candidate = res.x
            a_candidate = softmax_from_z(z_candidate)
            f_candidate = 2.0 * n * fast_convolve(a_candidate).max()
            if f_candidate < best_score:
                best_score = f_candidate
                best_a = a_candidate.copy()
                z_current = z_candidate
            elif f_candidate < best_score + 0.01:
                z_current = z_candidate  # Accept close solutions to continue
        except Exception:
            pass

        elapsed = time.monotonic() - t0
        print(f"    LBFGS T={T}: score={best_score:.7f} ({elapsed:.0f}s)")

    return best_a, best_score


def subgradient_average(a0, time_limit, n_steps=300000):

    Subgradient mirror descent with Polyak step size and iterate averaging.
    Averages all iterates (weighted by step) for convergence guarantee.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)

    best_score = 2.0 * n * fast_convolve(a).max()

    f_lb = 2.0 * n / (2 * n - 1)

    # Running average
    sum_a = np.zeros(n)
    t0 = time.monotonic()
    eta = 0.05
    best_avg_score = float('inf')
    best_avg_a = a.copy()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        f_val = 2.0 * n * max_b

        # Active set: indices within 0.1% of max
        threshold = max_b * 0.999
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        # Subgradient: average gradient over active set
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            g[js:je + 1] += 2.0 * a[k - np.arange(js, je + 1)]
        g /= len(active)

        g_norm_sq = np.dot(g, g)
        if g_norm_sq < 1e-20:
            break

        # Polyak step size
        gap = f_val - f_lb
        if gap < 1e-12:
            break
        eta_polyak = gap / (g_norm_sq + 1e-15)
        eta_use = min(eta_polyak, 2.0 / (step + 2))
        eta_use = max(eta_use, 1e-10)

        # Mirror descent update
        z_new = z - eta_use * g
        a_new = softmax_from_z(z_new)

        # Update running average
        sum_a += a_new

        # Check current and average
        f_new = 2.0 * n * fast_convolve(a_new).max()
        if f_new < best_score:
            best_score = f_new
            best_a = a_new.copy()
            z = z_new

        avg_a = sum_a / (step + 1)
        if (step + 1) % 10 == 0:
            f_avg = 2.0 * n * fast_convolve(avg_a).max()
            if f_avg < best_avg_score:
                best_avg_score = f_avg
                best_avg_a = avg_a.copy()

        if (step + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad step={step+1} ({elapsed:.0f}s): "
                  f"current={best_score:.7f}, avg={best_avg_score:.7f}")

    # Final check on average
    if best_avg_score < best_score:
        best_score = best_avg_score
        best_a = best_avg_a.copy()

    return best_a, best_score

def spectral_init(n):
    """
    Create an initial sequence based on spectral considerations.
    Aim for a sequence with relatively flat auto-convolution.
    Uses a raised-cosine-like shape.
    """
    a = np.zeros(n)
    for i in range(n):
        x = (i - n / 2 + 0.5) / (n / 2 + 0.5)
        a[i] = 0.5 * (1 + np.cos(np.pi * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def random_init(n, seed=0):
    """Random non-negative initialization."""
    rng = np.random.default_rng(seed)
    a = rng.random(n)
    a /= a.sum()
    return a


def perturb_sequence(a, eps, seed=0):
    """Add structured random perturbation to a sequence."""
    rng = np.random.default_rng(seed)
    n = len(a)
    noise = rng.standard_normal(n) * eps
    a_new = a + noise
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    a_new /= s
    return a_new
