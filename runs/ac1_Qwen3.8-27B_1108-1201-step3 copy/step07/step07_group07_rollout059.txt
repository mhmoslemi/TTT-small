    conv_aw = irfft(A * W, fft_size)  # length 3n-2 (we take first n)
    c2 = conv_aw[:n]
    
    A_rev = rfft(a_rev, fft_size)
    conv_arv_w = irfft(A_rev * W, fft_size)
    c1 = conv_arv_w[n - 1:2 * n - 1]
    
    grad = c1 + c2
    return lse_val, grad


def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def project_simplex_symmetric(v):
    """
    Project onto symmetric simplex: x[i] = x[n-1-i], x >= 0, sum(x) = 1.
    We work with the half-vector (first ceil(n/2) elements) and project that.
    """
    n = len(v)
    half = (n + 1) // 2
    
    # Extract half vector
    if n % 2 == 0:
        h = v[:half].copy()
    else:
        h = v[:half].copy()
    
    # The full sum constraint: 2*sum(h[:n//2]) + h[n//2] (if odd) = 1
    # For even n: 2*sum(h) = 1, so sum(h) = 0.5
    # For odd n: 2*sum(h[:-1]) + h[-1] = 1
    
    # Project h onto appropriate simplex
    if n % 2 == 0:
        # sum(h) = 0.5, h >= 0
        h_proj = project_simplex(2.0 * h) * 0.5
        result = np.zeros(n)
        result[:half] = h_proj
        result[half:] = h_proj[::-1]
    else:
        # 2*sum(h[:-1]) + h[-1] = 1
        # Let y = 2*h[:-1], z = h[-1]. Then sum(y) + z = 1, y >= 0, z >= 0
        y = 2.0 * h[:-1]
        z = h[-1]
        full = np.concatenate([y, [z]])
        full_proj = project_simplex(full)
        h_proj = np.zeros(half)
        h_proj[:-1] = full_proj[:-1] * 0.5
        h_proj[-1] = full_proj[-1]
        result = np.zeros(n)
        result[:half] = h_proj
        result[half:] = h_proj[:0:-1]
    
    return result


def eval_max_b(a):
    """Return max of auto-convolution for sequence (should have sum=1)."""
    b = fast_autoconv(a)
    return float(b.max())


def adam_optimize(a0, time_limit, beta_init=50.0, beta_max=500.0,
                  lr_init=0.001, symmetric=False, seed=42, verbose=False):
    """
    Adam optimizer on LSE-smoothed max auto-convolution.
    Subject to a >= 0, sum(a) = 1 (and optionally a symmetric).
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    if symmetric and not (np.allclose(a, a[::-1], atol=1e-10)):
        a = 0.5 * (a + a[::-1])
        a = np.maximum(a, 0)
        a = a / a.sum()
    
    # Adam state
    m = np.zeros_like(a)
    v_state = np.zeros_like(a)
    step_count = 0
    
    best_a = a.copy()
    best_obj = eval_max_b(a)
    
    beta = beta_init
    beta_growth = 1.05  # grow beta by 5% every 100 steps
    
    no_improve = 0
    eval_count = 0
    
    while time.monotonic() - t0 < time_limit:
        step_count += 1
        
        # Growth schedule for beta
        if step_count % 100 == 0:
            beta = min(beta * beta_growth, beta_max)
        
        # Adam learning rate schedule
        lr = lr_init / np.sqrt(1 + step_count * 1e-4)
        
        # Compute gradient
        lse_val, grad = compute_gradient_lse(a, beta)
        
        # Adam update
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m = beta1 * m + (1 - beta1) * grad
        v_state = beta2 * v_state + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**step_count)
        v_hat = v_state / (1 - beta2**step_count)
        update = m_hat / (np.sqrt(v_hat) + eps)
        
        a_new = a - lr * update
        
        # Project
        if symmetric:
            a_new = project_simplex_symmetric(a_new)
        else:
            a_new = project_simplex(a_new)
        
        # Evaluate
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        obj = eval_max_b(a_new)
        eval_count += 1
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
            if verbose and step_count % 200 == 0:
                print(f"      step {step_count}: obj={best_obj:.10f}, beta={beta:.1f}")
        else:
            no_improve += 1
        
        a = a_new
        
        # Restart logic
        if no_improve > 500:
            # Perturb and reset momentum
            noise_scale = 0.001
            a = a + rng.standard_normal(n) * noise_scale
            if symmetric:
                a = 0.5 * (a + a[::-1])
            a = np.maximum(a, 0)
            a = a / a.sum()
            m = np.zeros_like(a)
            v_state = np.zeros_like(a)
            no_improve = 0
            beta = max(beta_init, beta * 0.5)
    
    return best_a, best_obj


def lp_refine(f, time_limit=60.0, max_iters=30, verbose=False):
    """
    LP-based: max sum(g) s.t. conv(f,g) <= max(conv(f,f)), g >= 0.
    Blend result with f.
    """
    t0 = time.monotonic()
    n = len(f)
    a = f.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_obj = eval_max_b(a)
    
    for iteration in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_autoconv(a)
        M = b.max()
        
        # Build constraint matrix: A @ g <= M where A[k,j] = a[k-j] if valid
        num_constraints = 2 * n - 1
        # For efficiency, build the matrix
        A = np.zeros((num_constraints, n))
        for k in range(num_constraints):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                A[k, k - j] = a[j]
        
        b_lp = np.full(num_constraints, M)
        c = -np.ones(n)
        
        try:
            res = linprog(c, A_ub=A, b_ub=b_lp,
                          bounds=[(0.0, None)] * n, method='highs',
                          options={'time_limit': max(1, time_limit - (time.monotonic() - t0))})
            if not res.success:
                break
            
            g = np.maximum(res.x, 0)
            g_sum = g.sum()
            if g_sum < 1e-10:
                break
            
            g_norm = g / g_sum
            
            # Line search
            best_t = 0.0
            best_blend_obj = best_obj
            t_candidates = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0, 0.05, 0.15, 0.4, 0.6]
            for t in t_candidates:
                blend = (1.0 - t) * best_a + t * g_norm
                blend = np.maximum(blend, 0.0)
                ss = blend.sum()
                if ss < 1e-10:
                    continue
                blend = blend / ss
                obj_blend = eval_max_b(blend)
                if obj_blend < best_blend_obj:
                    best_blend_obj = obj_blend
                    best_t = t
            
            if best_t > 0.0:
                a = (1.0 - best_t) * best_a + best_t * g_norm
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                obj = eval_max_b(a)
                if obj < best_obj - 1e-13:
                    best_obj = obj
                    best_a = a.copy()
                    if verbose:
                        print(f"    LP iter {iteration}: t={best_t:.2f}, max_b={best_obj:.10f}")
                else:
                    a = best_a.copy()
            else:
                break
                
        except Exception as e:
            if verbose:
                print(f"    LP error: {e}")
            break
    
    return best_a, best_obj


def coordinate_descent(a0, time_limit, multipliers=None):
    """
    Coordinate descent on simplex.
    """
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_obj = eval_max_b(a)
    
    if multipliers is None:
        multipliers = np.array([0.3, 0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0, 3.0])
    
    improved = True
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-15:
                continue
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0.0 or new_val > 0.9999:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < -0.01 or scale > 20.0:
                    continue
                
                a_tr = a.copy()
                a_tr[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_tr[mask] *= scale
                ss = a_tr.sum()
                if ss < 1e-15:
                    continue
                a_tr = a_tr / ss
                val = eval_max_b(a_tr)
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
    
    return best_a, best_obj


def make_gaussian_init(n, sigma=0.25):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cosine_init(n):
    x = np.linspace(0, 1, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_tukey_init(n, r=0.5):
    a = np.ones(n)
    taper = max(1, int(r * n // 2))
    if taper < n:
        t = np.linspace(0, 1, taper)
        a[:taper] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t / r - 1.0)))
        a[-taper:] = a[:taper][::-1]
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_exp_decay_init(n):
    x = np.linspace(0, 1, n)
    a = np.exp(-3.0 * x)
    a = a / a.sum()
    # Make symmetric
    a = 0.5 * (a + a[::-1])
    a = a / a.sum()
    return a


def make_watershed_init(n):
    """Watershed-like shape: high in middle, with secondary peaks."""
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / 0.15) ** 2) + 0.3 * np.exp(-0.5 * ((x - 0.25) / 0.1) ** 2) + 0.3 * np.exp(-0.5 * ((x - 0.75) / 0.1) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LSE-smoothed true gradient
with Adam optimizer, LP-based refinement, symmetric sequence optimization,
parametric shape search, and multi-length exploration.

Key innovations:
- True gradient via log-sum-exp smoothing (not subgradient) via FFT
- Adam optimizer with per-coordinate adaptive learning rates
- Beta scheduling for progressive peak sharpening
- Symmetric sequence optimization (exploits structural symmetry)
- Parametric shape families for diverse initialization
- Efficient O(n log n) gradient computation
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fast_autoconv(a):
    """Fast auto-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_gradient_lse(a, beta):
    """
    Compute true gradient of LSE-smoothed max of auto-convolution.
    grad[i] = 2 * sum_m a[m] * w[i+m] computed via FFT.
    Returns (lse_value, gradient).
    """
    n = len(a)
    b = fast_autoconv(a)

    # LSE with shift for numerical stability
    b_max = b.max()
    exp_b = np.exp(beta * (b - b_max))
    lse_val = b_max + np.log(exp_b.sum()) / beta
    w = exp_b / exp_b.sum()  # weights length 2n-1

    # grad[i] = 2 * sum_m a[m] * w[i+m]
    # = 2 * (a_rev * w)[n-1+i] where a_rev = reversed a
    a_rev = a[::-1]
    fft_size = 1 << (3 * n - 1).bit_length()
    A_rev = rfft(a_rev, fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)
    grad = 2.0 * conv[n - 1:2 * n - 1]

    return lse_val, grad


def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def project_simplex_symmetric(v, n):
    """
    Project onto symmetric simplex: x[i] = x[n-1-i], x >= 0, sum(x) = 1.
    v is the full-length vector; return symmetric projection.
    """
    half = (n + 1) // 2
    if n % 2 == 0:
        h = v[:half].copy()
        h_proj = project_simplex(2.0 * h) * 0.5
        result = np.zeros(n)
        result[:half] = h_proj
        result[half:] = h_proj[::-1]
    else:
        h = v[:half].copy()
        y = 2.0 * h[:-1]
        z = h[-1]
        full = np.concatenate([y, [z]])
        full_proj = project_simplex(full)
        result = np.zeros(n)
        result[:half - 1] = full_proj[:-1] * 0.5
        result[half - 1] = full_proj[-1]
        result[half:] = result[:half - 1][::-1]
    return result


def eval_max_b(a):
    """Return max of auto-convolution for sequence with sum=1."""
    b = fast_autoconv(a)
    return float(b.max())


def adam_optimize(a0, time_limit, beta_init=30.0, beta_max=400.0,
                  lr_init=0.001, symmetric=False, seed=42, verbose=False):
    """
    Adam optimizer on LSE-smoothed max auto-convolution on the simplex.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)

    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    if symmetric:
        a = np.maximum(0.5 * (a + a[::-1]), 0)
        a = a / a.sum()

    m = np.zeros_like(a)
    v_state = np.zeros_like(a)
    step_count = 0

    best_a = a.copy()
    best_obj = eval_max_b(a)
    beta = beta_init
    beta_growth = 1.03
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        step_count += 1

        if step_count % 200 == 0:
            beta = min(beta * beta_growth, beta_max)

        lr = lr_init / (1.0 + 0.0001 * step_count)

        lse_val, grad = compute_gradient_lse(a, beta)

        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m = beta1 * m + (1.0 - beta1) * grad
        v_state = beta2 * v_state + (1.0 - beta2) * grad ** 2
        m_hat = m / (1.0 - beta1 ** step_count)
        v_hat = v_state / (1.0 - beta2 ** step_count)
        update = m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - lr * update

        if symmetric:
            a_new = project_simplex_symmetric(a_new, n)
        else:
            a_new = project_simplex(a_new)

        a_new = np.maximum(a_new, 0.0)
        ss = a_new.sum()
        if ss < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / ss

        obj = eval_max_b(a_new)

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
            if verbose and step_count % 500 == 0:
                print(f"      step {step_count}: obj={best_obj:.10f}, beta={beta:.1f}")
        else:
            no_improve += 1

        a = a_new

        if no_improve > 400:
            noise_scale = 0.002
            a = a + rng.standard_normal(n) * noise_scale
            if symmetric:
                a = np.maximum(0.5 * (a + a[::-1]), 0)
            else:
                a = np.maximum(a, 0)
            a = a / a.sum()
            m = np.zeros_like(a)
            v_state = np.zeros_like(a)
            no_improve = 0
            beta = max(beta_init, beta * 0.5)

    return best_a, best_obj


def lp_refine(f, time_limit=60.0, max_iters=30, verbose=False):
    """LP-based refinement: max sum(g) s.t. conv(f,g) <= M, g >= 0."""
    t0 = time.monotonic()
    n = len(f)
    a = f.copy().astype(np.float64)
    s = a.sum()
    a = a / s

    best_a = a.copy()
    best_obj = eval_max_b(a)

    for iteration in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autoconv(a)
        M = b.max()

        num_constraints = 2 * n - 1
        A = np.zeros((num_constraints, n))
        for k in range(num_constraints):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                A[k, k - j] = a[j]

        b_lp = np.full(num_constraints, M)
        c = -np.ones(n)

        try:
            remaining = time_limit - (time.monotonic() - t0)
            if remaining < 2:
                break
            res = linprog(c, A_ub=A, b_ub=b_lp,
                          bounds=[(0.0, None)] * n, method='highs',
                          options={'time_limit': max(1, int(remaining) - 1)})
            if not res.success:
                break

            g = np.maximum(res.x, 0)
            g_sum = g.sum()
            if g_sum < 1e-10:
                break

            g_norm = g / g_sum

            best_t = 0.0
            best_blend_obj = best_obj
            t_candidates = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0, 0.05, 0.15, 0.4, 0.6, 0.8]
            for t in t_candidates:
                blend = (1.0 - t) * best_a + t * g_norm
                blend = np.maximum(blend, 0.0)
                ss = blend.sum()
                if ss < 1e-10:
                    continue
                blend = blend / ss
                obj_blend = eval_max_b(blend)
                if obj_blend < best_blend_obj:
                    best_blend_obj = obj_blend
                    best_t = t

            if best_t > 0.0:
                a = (1.0 - best_t) * best_a + best_t * g_norm
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                obj = eval_max_b(a)
                if obj < best_obj - 1e-13:
                    best_obj = obj
                    best_a = a.copy()
                    if verbose:
                        print(f"    LP iter {iteration}: t={best_t:.2f}, max_b={best_obj:.10f}")
                else:
                    a = best_a.copy()
            else:
                break

        except Exception:
            break

    return best_a, best_obj


def coordinate_descent(a0, time_limit, multipliers=None):
    """Coordinate descent on simplex."""
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_obj = eval_max_b(a)

    if multipliers is None:
        multipliers = np.array([0.3, 0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0, 3.0])

    improved = True
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-15:
                continue
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0.0 or new_val > 0.9999:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < -0.01 or scale > 20.0:
                    continue

                a_tr = a.copy()
                a_tr[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_tr[mask] *= scale
                ss = a_tr.sum()
                if ss < 1e-15:
                    continue
                a_tr = a_tr / ss
                val = eval_max_b(a_tr)
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

    return best_a, best_obj


def make_gaussian_init(n, sigma=0.25):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cosine_init(n):
    x = np.linspace(0, 1, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_tukey_init(n, r=0.5):
    a = np.ones(n)
    taper = max(1, int(r * n // 2))
    if taper < n:
        t = np.linspace(0, 1, taper)
        a[:taper] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t / r - 1.0)))
        a[-taper:] = a[:taper][::-1]
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_watershed_init(n):
    x = np.linspace(0, 1, n)
    a = (np.exp(-0.5 * ((x - 0.5) / 0.15) ** 2) +
         0.3 * np.exp(-0.5 * ((x - 0.25) / 0.1) ** 2) +
         0.3 * np.exp(-0.5 * ((x - 0.75) / 0.1) ** 2))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_platesque_init(n):
    """Platesque-like: flat top with smooth edges."""
    x = np.linspace(0, 1, n)
    width = 0.4
    center = 0.5
    a = np.ones(n)
    edge = 0.15
    left = center - width / 2
    right = center + width / 2
    for i in range(n):
        if x[i] < left - edge:
            a[i] = 0.0
        elif x[i] < left:
            t = (x[i] - (left - edge)) / edge
            a[i] = 0.5 * (1.0 + np.cos(np.pi * (1.0 - t)))
        elif x[i] <= right:
            a[i] = 1.0
        elif x[i] < right + edge:
            t = (x[i] - right) / edge
            a[i] = 0.5 * (1.0 + np.cos(np.pi * t))
        else:
            a[i] = 0.0
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_multimodal_init(n, n_peaks=3, peak_width=0.1):
    x = np.linspace(0, 1, n)
    a = np.zeros(n)
    for k in range(n_peaks):
        center = (k + 1) / (n_peaks + 1)
        a += np.exp(-0.5 * ((x - center) / peak_width) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization: Adam (LSE gradient) + LP + symmetric +
    multi-length + diverse inits + coordinate descent polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, budget_s - 15.0)

    best_seq = None
    best_score = float('inf')

    def update_best(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  ** NEW BEST [{label}]: {score:.8f} (n={len(a)})")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_default = 1152
        parent = make_gaussian_init(n_default, 0.25)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"No parent. Gaussian n={n_default}, score={best_score:.6f}")

    n_parent = len(parent)

    # === Phase 1: Adam on parent (non-symmetric) ===
    print(f"\n{'='*60}")
    print(f"Phase 1: Adam (LSE) on parent n={n_parent}")
    print(f"{'='*60}")
    rem = deadline - time.monotonic()
    t1 = min(180.0, rem * 0.22)
    print(f"  Time: {t1:.0f}s")
    a1, obj1 = adam_optimize(
        parent, time_limit=t1,
        beta_init=30.0, beta_max=350.0,
        lr_init=0.0015, symmetric=False, seed=seed, verbose=True
    )
    update_best(a1, "adam_p1")

    # === Phase 2: LP on best ===
    print(f"\n{'='*60}")
    print(f"Phase 2: LP (best={best_score:.8f})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic()
    t2 = min(100.0, rem * 0.12)
    if t2 > 10:
        print(f"  Time: {t2:.0f}s, n={len(best_seq)}")
        a2, obj2 = lp_refine(best_seq, time_limit=t2, max_iters=25, verbose=True)
        update_best(a2, "lp_p2")

    # === Phase 3: Symmetric Adam ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Symmetric Adam (best={best_score:.8f})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic()
    t3 = min(140.0, rem * 0.18)
    if t3 > 20:
        a_sym_init = np.maximum(0.5 * (best_seq + best_seq[::-1]), 0)
        a_sym_init = a_sym_init / a_sym_init.sum()
        print(f"  Time: {t3:.0f}s, n={len(a_sym_init)}")
        a3, obj3 = adam_optimize(
            a_sym_init, time_limit=t3,
            beta_init=30.0, beta_max=400.0,
            lr_init=0.001, symmetric=True, seed=seed + 1, verbose=True
        )
        update_best(a3, "sym_p3")
        # LP on symmetric
        rem = deadline - time.monotonic()
        if rem > 25:
            t3b = min(50.0, rem * 0.08)
            a3b, obj3b = lp_refine(a3, time_limit=t3b, max_iters=15, verbose=False)
            update_best(a3b, "sym_lp_p3")

    # === Phase 4: High-beta Adam (fine peak shaping) ===
    print(f"\n{'='*60}")
    print(f"Phase 4: High-beta Adam (best={best_score:.8f})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic()
    t4 = min(150.0, rem * 0.18)
    if t4 > 20:
        print(f"  Time: {t4:.0f}s, n={len(best_seq)}")
        a4, obj4 = adam_optimize(
            best_seq, time_limit=t4,
            beta_init=80.0, beta_max=600.0,
            lr_init=0.0004, symmetric=False, seed=seed + 100, verbose=True
        )
        update_best(a4, "highbeta_p4")

    # === Phase 5: Multi-length exploration ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Multi-length (best={best_score:.8f})")
    print(f"{'='*60}")
    n_best = len(best_seq)
    target_lengths = []
    for factor in [0.5, 0.6, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0]:
        tl = int(n_best * factor)
        tl = max(256, min(4096, tl))
        if tl not in target_lengths and tl != n_best:
            target_lengths.append(tl)
    target_lengths.sort(key=lambda x: abs(x - n_best))

            f_interp, time_limit=t5 * 0.7,
            beta_init=30.0, beta_max=300.0,

            seed=seed + target_n, verbose=False
        )
        update_best(a5, f"len_{target_n}")
        rem = deadline - time.monotonic()
        if rem > 15:
            a5_lp, obj5_lp = lp_refine(a5, time_limit=min(rem * 0.3, 25.0), max_iters=8)

    # === Phase 6: Diverse initializations ===
    print(f"\n{'='*60}")
    print(f"Phase 6: Diverse Inits (best={best_score:.8f})")
    print(f"{'='*60}")
    n_cur = len(best_seq)
    inits = {
        "gauss_0.12": make_gaussian_init(n_cur, 0.12),
        "gauss_0.35": make_gaussian_init(n_cur, 0.35),
        "cosine": make_cosine_init(n_cur),
        "tukey_0.3": make_tukey_init(n_cur, 0.3),
        "tukey_0.7": make_tukey_init(n_cur, 0.7),
        "water": make_watershed_init(n_cur),
        "plate": make_platesque_init(n_cur),
        "multi3": make_multimodal_init(n_cur, 3, 0.12),
    }

    for name, init_a in inits.items():
        rem = deadline - time.monotonic()
        if rem < 25:
            break
        t6 = min(rem * 0.13, 50.0)
        if t6 < 12:

        print(f"  {name}: t={t6:.0f}s")
        a6, obj6 = adam_optimize(
            init_a, time_limit=t6,
            beta_init=25.0, beta_max=250.0,
            lr_init=0.001, symmetric=False,
            seed=seed + 50000 + hash(name) % 30000, verbose=False
        )
        update_best(a6, f"div_{name}")

    # === Phase 7: Deep optimization ===
    print(f"\n{'='*60}")
    print(f"Phase 7: Deep Opt (best={best_score:.8f})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic()
    if rem > 60:
        t7 = min(rem * 0.3, 160.0)
        print(f"  Adam: t={t7:.0f}s, n={len(best_seq)}")
        a7, obj7 = adam_optimize(
            best_seq, time_limit=t7 * 0.6,
            beta_init=50.0, beta_max=500.0,
            lr_init=0.0003, symmetric=False, seed=seed + 999, verbose=True

        update_best(a7, "deep_adam")

            update_best(a7b, "deep_lp")

    # === Phase 8: Coordinate descent ===
    print(f"\n{'='*60}")
    print(f"Phase 8: CD (best={best_score:.8f})")

    rem = deadline - time.monotonic()

    # === Phase 9: Final micro-polish ===

            best_seq, time_limit=t9,
