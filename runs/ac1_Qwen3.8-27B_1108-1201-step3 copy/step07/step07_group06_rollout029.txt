```python
    """Solve the LP: max sum(b) s.t. a*b[k] <= M for all k, b >= 0"""
    # This is a standard LP
    # Variables: b[0], ..., b[n-1]
    # Objective: maximize sum(b)
    # Constraints: for each k, sum_{valid j} a[j] * b[k-j] <= M
    #              b[j] >= 0
    n = len(a)
    # Build constraint matrix
    # For lag k, the constraint is: sum_{j=max(0,k-n+1)}^{min(n-1,k)} a[j] * b[k-j] <= M
    # Let i = k-j, so j = k-i, and b[i] appears with coefficient a[k-i]
    # So constraint k: sum_i a[k-i] * b[i] <= M (summed over valid i, k-i)
    
    # Using scipy.optimize.linprog
    # Minimize -sum(b)
    c = -np.ones(n)
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for i in range(max(0, k-n+1), min(n, k+1)):
            j = k - i
            A_ub[k, i] = a[j]
    b_ub = np.full(2*n-1, M)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    return res.x
```

```python
    A = rfft(a, fft_size)

def build_A_matrix(n):
    """Build the sparse constraint matrix for the LP."""
    # A[k, i] = 1 if i and k-i are both in [0, n-1]
    # The constraint k is: sum_i A[k,i] * w[i] <= M where w[i] = a[k-i] * b[i]

        i_end = min(n - 1, k)

            cols.append(i)
    
    return rows, cols

    A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

        return res.x
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based fictitious play (best-response
iteration), combined with Adam log-space gradient descent for warm-up and peak-targeted
local search for final polishing. The fictitious play solves an LP at each iteration to
find the most "spread-out" sequence compatible with the current peak level, which provides
a direct handling of the non-smooth max objective without gradient approximation.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(a, b):
    """Linear cross-convolution a*b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_lp_indices(n):
    """Pre-compute sparse indices for the LP constraint matrix."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_best_response_lp(a, M, rows, cols):
    """
    Solve: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0.
    Uses pre-computed sparsity pattern.
    """
    n = len(a)
    data = a[rows - cols]
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c, A_ub=A, b_ub=b_ub, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                           'primal_feasibility_tolerance': 1e-9,
                           'time_limit': 5.0})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def fictitious_play(a0, time_limit, step_mode='full', verbose=False, verbose_interval=500):
    """
    LP-based fictitious play for minimizing peak autocorrelation.
    At each iteration, solves an LP to find the best response, then updates.
    """
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy()
    a = a / s
    n = len(a)
    rows, cols = build_lp_indices(n)
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    M = fast_convolve(a).max()
    no_improve = 0
    max_no_improve = 5000
    t_current = 1.0
    momentum_buf = []
    buf_size = 50

    while time.monotonic() - t0 < time_limit:
        # Solve LP
        b = solve_best_response_lp(a, M, rows, cols)
        if b is None or b.sum() < 1e-15:
            break

        # Store in momentum buffer
        momentum_buf.append(b / b.sum())
        if len(momentum_buf) > buf_size:
            momentum_buf.pop(0)

        # Compute candidate
        if step_mode == 'full' and len(momentum_buf) <= 2:
            a_candidate = momentum_buf[-1]
        elif step_mode == 'momentum':
            a_candidate = np.mean(momentum_buf, axis=0)
            a_candidate = a_candidate / a_candidate.sum()
        else:
            a_candidate = momentum_buf[-1]

        # Evaluate
        conv_c = fast_convolve(a_candidate)
        score_c = 2.0 * n * conv_c.max()

        if score_c < best_score - 1e-14:
            best_score = score_c
            best_a = a_candidate.copy()
            no_improve = 0
            # Slightly increase step size if improving
            if t_current < 1.0:
                t_current = min(1.0, t_current * 1.1)
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                # Reset with perturbation
                a = best_a + np.random.standard_normal(n) * 0.002
                a = np.maximum(a, 0)
                a = a / a.sum()
                M = fast_convolve(a).max()
                momentum_buf = []
                no_improve = 0
                t_current = 0.5
                continue

        # Update a
        if t_current >= 0.999:
            a = a_candidate
        else:
            a = (1.0 - t_current) * a + t_current * a_candidate
            a = a / a.sum()

        M = fast_convolve(a).max()

        if verbose and (int((time.monotonic() - t0) * 100) % (verbose_interval * 100) == 0):
            print(f"    FP: score={best_score:.8f}, M={M:.2e}, it_no_improve={no_improve}")

    return best_a


def adam_logspace(a0, time_limit, T_init=0.05, T_final=0.002, lr=0.01,
                  beta1=0.9, beta2=0.999, eps=1e-8, schedule='exp',
                  verbose=False, verbose_interval=3000):
    """Adam optimizer in log-space with temperature annealing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    s = a.sum()
    a = a / s
    z = np.log(np.maximum(a, 1e-300))
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    fft_size = 1 << (2 * n - 2).bit_length()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    no_improve = 0
    max_no_improve = 5000

    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)

        if schedule == 'linear':
            T = T_init + (T_final - T_init) * progress
        elif schedule == 'exp':
            T = T_init * (T_final / T_init) ** progress
        elif schedule == 'hold_anneal':
            if progress < 0.3:
                T = T_init
            else:
                p2 = (progress - 0.3) / 0.7
                T = T_init * (T_final / T_init) ** p2
        else:
            T = T_init * (T_final / T_init) ** progress

        a_curr = _softmax_from_z(z)
        b = fast_convolve(a_curr)
        max_b = b.max()
        exp_b = np.exp(b / T - max_b / T)
        w = exp_b / exp_b.sum()

        # Gradient
        W = rfft(w, fft_size)
        A = rfft(a_curr, fft_size)
        corr = irfft(W * np.conj(A), fft_size)[:n]
        grad_a = 2.0 * n * corr

        S = np.dot(grad_a, a_curr)
        grad_z = a_curr * (grad_a - S)

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = _softmax_from_z(z)
        conv_new = fast_convolve(a_new)
        score = 2.0 * n * conv_new.max()

        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            z = np.log(np.maximum(best_a, 1e-300)) + np.random.standard_normal(n) * 0.003
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = max(0.001, lr * 0.7)

        if verbose and t % verbose_interval == 0:
            print(f"      Adam t={t} T={T:.5f} score={best_score:.6f}")

    return best_a, best_score


def _softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - np.max(z)
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return exp_z / s


def peak_shave(a0, time_limit, n_trials=5000):
    """Peak-targeted local search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        b_search = b.copy()
        b_search[0] = -np.inf
        b_search[-1] = -np.inf
        peak_k = int(np.argmax(b_search))
        peak_val = b[peak_k]

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_idx = np.argsort(contribs)[::-1][:5]

        improved = False
        for ci in top_idx:
            i = int(idx[ci])
            j = int(peak_k - i)
            if i == j or a[j] < 1e-15 or a[i] < 1e-15:
                continue
            for f1 in [0.95, 0.9, 0.85, 0.8]:
                for f2 in [1.0, 1.02, 1.05]:
                    a_trial = a.copy()
                    a_trial[i] *= f1
                    a_trial[j] *= f2
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a_trial.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            for ci in top_idx[:3]:
                i = int(idx[ci])
                if a[i] < 1e-15:
                    continue
                for factor in [0.9, 0.8, 0.7]:
                    a_trial = a.copy()
                    diff = a_trial[i] * (1.0 - factor)
                    a_trial[i] *= factor
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    others_sum = a_trial[mask].sum()
                    if others_sum < 1e-15:
                        continue
                    a_trial[mask] += diff * a_trial[mask] / others_sum
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a_trial.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved and trial > 200 and trial % 100 == 0:
            a = best_a + np.random.standard_normal(n) * 0.005
            a = np.maximum(a, 0)
            a = a / a.sum()

    return best_a, best_score


def coordinate_descent(a0, time_limit, n_sweeps=100):
    """Coordinate descent with multiplicative updates."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        order = rng.permutation(n)
        sweep_improved = False

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-15:
                continue
            best_local_score = best_score
            best_local_val = a[i]

            for factor in [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]:
                a_trial = a.copy()
                a_trial[i] = a[i] * factor
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_local_score:
                    best_local_score = score
                    best_local_val = a[i] * factor

            if best_local_val != a[i]:
                a[i] = best_local_val
                a = a / a.sum()
                if best_local_score < best_score:
                    best_score = best_local_score
                    best_a = a.copy()
                    sweep_improved = True

        if not sweep_improve:
            pass

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_and_track(a, label, best_seq, best_score):
    """Evaluate a candidate and return updated best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def generate_structured_inits(n, rng):
    """Generate diverse structured initializations."""
    candidates = []

    # Uniform
    candidates.append(('uniform', np.ones(n) / n))

    # Linear tilt (various strengths)
    for tilt in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        x = np.linspace(0, 1, n)
        a = 1.0 + tilt * (2 * x - 1)
        a = np.maximum(a, 0.01)
        a = a / a.sum()
        candidates.append((f'tilt_{tilt}', a))

    # Sinusoidal perturbations
    for freq in [1, 2, 3, 5]:
        for phase in [0, np.pi / 2, np.pi]:
            x = np.linspace(0, 1, n)
            a = 1.0 + 0.1 * np.sin(2 * np.pi * freq * x + phase)
            a = np.maximum(a, 0.01)
            a = a / a.sum()
            candidates.append((f'sin_{freq}_{phase:.2f}', a))

    # Gaussian-like
    x = np.linspace(-3, 3, n)
    for sigma in [1.0, 1.5, 2.0, 3.0]:
        gau = np.exp(-x ** 2 / (2 * sigma ** 2))
        gau = gau / gau.sum()
        candidates.append((f'gau_{sigma}', gau))

    # Bimodal
    mid = n // 2
    for width in [0.1, 0.2, 0.3]:
        bi = np.zeros(n)
        w = int(n * width)
        bi[max(0, mid - w):mid + w] = 1.0
        bi = np.maximum(bi, 0.001)
        bi = bi / bi.sum()
        candidates.append((f'bimodal_{width}', bi))

    # Exponential
    for rate in [0.5, 1.0, 2.0, 3.0]:
        expo = np.exp(-rate * np.linspace(0, 1, n))
        expo = expo / expo.sum()
        candidates.append((f'expo_{rate}', expo))

    # Random
    rand = np.abs(rng.standard_normal(n)) + 0.1
    rand = rand / rand.sum()
    candidates.append(('random', rand))

    return candidates


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    Phase 1: Fictitious play (LP-based) from multiple starts
    Phase 2: Adam gradient descent refinement
    Phase 3: Peak shaving and coordinate descent polishing
    Phase 4: Multi-length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_default = 768
        uniform = np.ones(n_default) / n_default
        best_seq = uniform
        best_score = float(evaluate_sequence(uniform.tolist()))
        print(f"Default: n={n_default}, score={best_score:.6f}")

    n_parent = len(best_seq)
    print(f"Starting best score: {best_score:.6f}")

    # === Phase 1: Fictitious Play (50% time) ===
    print(f"\n=== Phase 1: Fictitious Play (n={n_parent}) ===")
    rem = deadline - time.monotonic()
    t_p1 = rem * 0.50
    t_p1_deadline = time.monotonic() + t_p1

    # Pre-compute LP indices
    rows, cols = build_lp_indices(n_parent)
    print(f"  LP indices: {len(rows)} non-zeros")

    # Try from parent first
    print(f"  FP from parent (budget={t_p1 * 0.4:.0f}s)...")
    t_sub = t_p1 * 0.4
    a_fp = fictitious_play(best_seq, t_sub, step_mode='full', verbose=True, verbose_interval=2000)
    best_seq, best_score = evaluate_and_track(a_fp, "FP_parent", best_seq, best_score)
    print(f"  After FP from parent: {best_score:.6f}")

    # Try from structured inits
    inits = generate_structured_inits(n_parent, rng)
    # Score and sort
    inits_scored = []
    for label, init_a in inits:
        s = 2.0 * n_parent * fast_convolve(init_a).max()
        inits_scored.append((s, label, init_a))
    inits_scored.sort(key=lambda x: x[0])

    # Try top few inits with FP
    n_starts = min(5, len(inits_scored))
    rem = t_p1_deadline - time.monotonic()
    time_per = max(10.0, rem / n_starts)

    for idx in range(n_starts):
        rem = t_p1_deadline - time.monotonic()
        if rem < 30:
            break
        t_sub = min(time_per, rem * 0.8)
        s_init, label, init_a = inits_scored[idx]
        print(f"  FP init '{label}' (init_score={s_init:.6f}, budget={t_sub:.0f}s)...")
        a_fp = fictitious_play(init_a, t_sub, step_mode='momentum', verbose=False)
        best_seq, best_score = evaluate_and_track(a_fp, f"FP_{label}", best_seq, best_score)

    print(f"  Phase 1 best: {best_score:.6f}")

    # === Phase 2: Adam Refinement (25% time) ===
    print(f"\n=== Phase 2: Adam Refinement ===")
    rem = deadline - time.monotonic()
    t_p2 = rem * 0.25
    t_p2_deadline = time.monotonic() + t_p2

    # Multiple Adam runs with different parameters
    for sched, Ti, Tf, lr in [
        ('exp', 0.03, 0.002, 0.008),
        ('hold_anneal', 0.02, 0.001, 0.006),
        ('linear', 0.04, 0.003, 0.007),
    ]:
        rem = t_p2_deadline - time.monotonic()
        if rem < 20:
            break
        t_sub = min(rem * 0.5, 120.0)
        print(f"  Adam sched={sched} (budget={t_sub:.0f}s)...")
        a_adam, score_adam = adam_logspace(best_seq, t_sub, T_init=Ti, T_final=Tf,
                                           lr=lr, schedule=sched)
        best_seq, best_score = evaluate_and_track(a_adam, f"adam_{sched}", best_seq, best_score)

    # FP again after Adam (to exploit the improved starting point)
    rem = t_p2_deadline - time.monotonic()
    if rem > 30:
        t_sub = min(rem * 0.8, 120.0)
        print(f"  FP after Adam (budget={t_sub:.0f}s)...")
        a_fp2 = fictitious_play(best_seq, t_sub, step_mode='full', verbose=False)
        best_seq, best_score = evaluate_and_track(a_fp2, "FP_after_adam", best_seq, best_score)

    print(f"  Phase 2 best: {best_score:.6f}")

    # === Phase 3: Local Polishing (15% time) ===
    print(f"\n=== Phase 3: Local Polishing ===")
    rem = deadline - time.monotonic()
    t_p3 = rem * 0.15
    t_p3_deadline = time.monotonic() + t_p3

    # Peak shaving
    rem = t_p3_deadline - time.monotonic()
    if rem > 20:
        t_sub = min(rem * 0.5, 80.0)
        a_ps, _ = peak_shave(best_seq, t_sub, n_trials=5000)
        best_seq, best_score = evaluate_and_track(a_ps, "peak_shave", best_seq, best_score)

    # More FP (short bursts)
    rem = t_p3_deadline - time.monotonic()
    if rem > 15:
        t_sub = min(rem * 0.6, 60.0)
        a_fp3 = fictitious_play(best_seq, t_sub, step_mode='full', verbose=False)
        best_seq, best_score = evaluate_and_track(a_fp3, "FP_polish", best_seq, best_score)

    # Another peak shave
    rem = t_p3_deadline - time.monotonic()
    if rem > 10:
        t_sub = min(rem * 0.8, 40.0)
        a_ps2, _ = peak_shave(best_seq, t_sub, n_trials=3000)
        best_seq, best_score = evaluate_and_track(a_ps2, "peak_shave_2", best_seq, best_score)

    print(f"  Phase 3 best: {best_score:.6f}")

    # === Phase 4: Multi-Length Exploration (8% time) ===
    print(f"\n=== Phase 4: Multi-Length Exploration ===")
    rem = deadline - time.monotonic()
    t_p4 = rem * 0.08
    t_p4_deadline = time.monotonic() + t_p4

    n_best = len(best_seq)
    alt_lengths = []
    for factor in [1.25, 1.5, 0.75, 2.0]:
        alt_n = int(n_best * factor)
        if 256 <= alt_n <= 4096 and alt_n != n_best:
            if alt_n not in alt_lengths:
                alt_lengths.append(alt_n)

    for alt_n in alt_lengths[:3]:
        rem = t_p4_deadline - time.monotonic()
        if rem < 30:
            break
        t_sub = min(rem * 0.5, 80.0)
        init_alt = interpolate_sequence(best_seq, alt_n)

        # FP at new length
        rows_alt, cols_alt = build_lp_indices(alt_n)
        # Temporarily patch build_lp_indices for this call
        # (fictitious_play builds its own, so just call it)
        a_alt = fictitious_play(init_alt, t_sub, step_mode='full', verbose=False)
        best_seq, best_score = evaluate_and_track(a_alt, f"FP_n{alt_n}", best_seq, best_score)
        print(f"  Alt n={alt_n}: {best_score:.6f}")

    # === Phase 5: Final Intensive Polish (remaining time) ===
    print(f"\n=== Phase 5: Final Polish ===")
    rem = deadline - time.monotonic()

    if rem > 30:
        # Final FP
        t_sub = min(rem * 0.4, 60.0)
        a_fin = fictitious_play(best_seq, t_sub, step_mode='momentum', verbose=False)
        best_seq, best_score = evaluate_and_track(a_fin, "FP_final", best_seq, best_score)

    rem = deadline - time.monotonic()
    if rem > 20:
        # Final Adam with very low temperature
        t_sub = min(rem * 0.4, 40.0)
        a_fin2, _ = adam_logspace(best_seq, t_sub, T_init=0.01, T_final=0.0005,
                                  lr=0.003, schedule='exp')
        best_seq, best_score = evaluate_and_track(a_fin2, "adam_final", best_seq, best_score)

    rem = deadline - time.monotonic()
    if rem > 15:
        # Final peak shave
        t_sub = min(rem * 0.6, 30.0)
        a_fin3, _ = peak_shave(best_seq, t_sub, n_trials=2000)
        best_seq, best_score = evaluate_and_track(a_fin3, "ps_final", best_seq, best_score)

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.6f}")

    return [float(x) for x in best_seq]
```
