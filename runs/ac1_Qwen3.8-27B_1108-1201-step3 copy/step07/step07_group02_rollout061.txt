    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def z_from_a(a):
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)

def l2_conv_obj_and_grad_z(z, n):
    a = softmax_from_z(z)
    fft_size = 1 << (2 * n - 2).bit_length()
    A = rfft(a, fft_size)
    abs_A_sq = np.abs(A) ** 2
    obj = np.sum(abs_A_sq ** 2) / fft_size
    C = abs_A_sq * np.conj(A)
    grad_a = 4.0 * irfft(C, fft_size)[:n]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z

def softmax_obj_and_grad_z(z, T_val):
    a = softmax_from_z(z)
    n = len(a)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    exp_sum = exp_sum_total = exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_sum_total)
    obj = (2.0 * n / T_val) * log_sum_exp
    w = exp_Tb / exp_sum_total
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z

def lbgfs_l2_warmup(a0, time_limit, maxiter=3000):
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    stopped = [False]

    def func(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        return l2_conv_obj_and_grad_z(z, n)[0]

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return np.zeros_like(z)
        return l2_conv_obj_and_grad_z(z, n)[1]

    try:
        res = minimize(fun=func, x0=z0, jac=jac, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12,
                                'maxfun': maxiter * 50, 'disp': False})
        z = res.x
    except Exception:
        z = z0
    return softmax_from_z(z)

def lbgfs_softmax(a0, time_limit, T_start=100.0, T_end=0.3, n_stages=7, maxiter=10000):
    n = len(a0)
    z = z_from_a(a0)
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(maxiter // n_stages, 100)

    for stage_idx, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        def func(z_inner):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            return softmax_obj_and_grad_z(z_inner, stage_T)[0]

        def jac(z_inner):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_inner)
            return softmax_obj_and_grad_z(z_inner, stage_T)[1]

        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                                    'gtol': 1e-12, 'maxfun': iters_per_stage * 50,
                                    'disp': False})
            z = res.x
        except Exception:
            pass

        elapsed = time.monotonic() - t0
        a_check = softmax_from_z(z)
        score = 2.0 * n * fast_convolve(a_check).max()
        print(f"    SoftMax L-BFGS stage {stage_idx+1}/{n_stages} (T={stage_T:.1f}, {elapsed:.0f}s): score={score:.6f}")

    return softmax_from_z(z)

def solve_lp_best_response(a_fixed, solver):
    if solver is None:
        return None, None
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def lp_alternate(a0, time_limit, solver, n_max_iters=10000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        alpha = 2.0 / (it + 2.0)
        y = (1.0 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy
        a = ((it + 1) * a + (alpha * (it + 2.0)) * b_new) / (it + 1 + alpha * (it + 2.0))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        if (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Alt iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score

def peak_shave(a0, time_limit, n_steps=10000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stuck = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        b_inner = b.copy()
        if n > 1:
            b_inner[0] = -np.inf
            b_inner[-1] = -np.inf
        peak_k = int(np.argmax(b_inner))
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_idx = np.argsort(contribs)[::-1][:5]

        improved = False
        for ci in top_idx:
            i = idx[ci]
            j = peak_k - i
            if a[i] < 1e-14:
                continue
            for factor in [0.97, 0.94, 0.90, 0.85]:
                new_val = a[i] * factor
                diff = a[i] - new_val
                if diff < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a[mask]
                osum = others.sum()
                if osum < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                a_trial[mask] = others + diff * others / osum
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stuck = 0
                    break
            if improved:
                break

        if not improved:
            if len(top_idx) >= 2:
                for ci in top_idx[:3]:
                    i = idx[ci]
                    j = peak_k - i
                    if i == j or a[j] < 1e-14:
                        continue
                    for f1 in [0.95, 0.92]:
                        for f2 in [0.95, 0.92]:
                            a_trial = a.copy()
                            a_trial[i] *= f1
                            a_trial[j] *= f2
                            a_trial = np.maximum(a_trial, 0)
                            a_trial = a_trial / a_trial.sum()
                            score = 2.0 * n * fast_convolve(a_trial).max()
                            if score < best_score - 1e-14:
                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                improved = True
                                stuck = 0
                                break
                        if improved:
                            break
                    if improved:
                        break
            if not improved:
                top3 = top_idx[:3]
                a_trial = a.copy()
                for ci in top3:
                    i = idx[ci]
                    a_trial[i] *= 0.97
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stuck = 0
                else:
                    stuck += 1
                    if stuck > 500:
                        break

    return best_a, best_score

def mirror_descent(a0, time_limit, lr_init=0.05, n_steps=50000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    z = z_from_a(a)
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b - 0.03 * max_b
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        z_new = z - lr * g
        a_new = softmax_from_z(z_new)
        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            z = z_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 50:
            lr *= 0.7
            no_improve = 0
        lr = max(lr, 1e-8)

        if step % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MirrorDesc {elapsed:.0f}s: score={best_score:.6f} lr={lr:.6f}")

    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def propose_candidate(seed=42, budget_s=1800.0):

            parent = p / p.sum()
            p_score = evaluate_safe(parent)

        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:

            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # === Phase 1: L2 warm-up (150s) ===

    if t_alloc > 20:
        print(f"  L2 warm-up n={len(best_seq)}, budget={t_alloc:.0f}s")
        a_l2 = lbgfs_l2_warmup(best_seq, t_alloc, maxiter=2000)
        try_update(a_l2, "l2_warmup")

    # === Phase 2: Aggressive Soft-Max L-BFGS (500s) ===

            try_update(a_sm2, "softmax_lowT")

    # === Phase 3: LP Alternating (400s) ===
