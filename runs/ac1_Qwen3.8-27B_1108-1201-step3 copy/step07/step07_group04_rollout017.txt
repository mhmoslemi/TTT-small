Let me formalize this:

**Algorithm: Log-space multiplicative gradient descent**

1. Initialize a = uniform (or from parent), z = log(a)
2. For each iteration:
   a. Compute b = conv(a, a)
   b. Find the top-K peaks of b

   e. Set a = exp(z) / sum(exp(z)) (softmax normalization)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Parameterized family initialization (geometric, sinusoidal, polynomial)
2. Fast fictitious play with LP (PDLP)
3. Adam-based log-space gradient descent with softmax smoothing
4. L-BFGS refinement
5. Active-set peak shaving
6. Multi-length exploration with interpolation

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from time import monotonic


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    global _lp_solver
    if '_lp_solver' in globals():
        return _lp_solver
    for solver_name in [cp.PDLP, cp.SCS, cp.ECOS, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.SCS, cp.PDLP, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def fictitious_play(n, time_limit, initial=None, solver=None, verbose=True):
    t0 = monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

    b = a.copy()
    a_avg = a.copy().astype(np.float64)
    b_avg = b.copy().astype(np.float64)
    best_score = float('inf')
    best_a = a.copy()
    iters = 0
    stagnation = 0

    while monotonic() - t0 < time_limit:
        iters += 1

        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new

        a_new, M_a = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new

        a_avg = (iters * a_avg + a) / (iters + 1)
        b_avg = (iters * b_avg + b) / (iters + 1)

        candidates = [a, b, 0.5 * (a + b), a_avg, 0.5 * (a_avg + b_avg)]
        for cand in candidates:
            cand = np.maximum(cand, 0)
            s = cand.sum()
            if s < 1e-10:
                continue
            cand_norm = cand / s
            max_conv = fast_convolve(cand_norm).max()
            score = 2.0 * n * max_conv
            if score < best_score - 1e-12:
                best_score = score
                best_a = cand_norm.copy()
                stagnation = 0
            else:
                stagnation += 1

        if stagnation > 100:
            noise = np.random.standard_normal(n) * 0.002
            a = a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            b = b + np.random.standard_normal(n) * 0.002
            b = np.maximum(b, 0)
            b = b / b.sum()
            stagnation = 0

        if verbose and iters % 20 == 0:
            elapsed = monotonic() - t0
            print(f"  FP n={n} iter={iters} ({elapsed:.0f}s): best_score={best_score:.6f}")

    return best_a, best_score, iters


def adam_logspace(a0, time_limit, T=50, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer in log-space with softmax-smoothed max."""
    n = len(a0)
    a_safe = np.maximum(a0, 1e-30)
    z = np.log(a_safe)
    z = z - z.mean()
    
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t = 0
    t0 = monotonic()
    best_z = z.copy()
    best_a = a_safe / a_safe.sum()
    best_score = 2.0 * n * fast_convolve(best_a).max()
    no_improve = 0
    stopped = [False]

    def compute_grad(z_val):
        z_shifted = z_val - z_val.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()
        b = fast_convolve(a)
        max_b = b.max()
        scaled = (b - max_b) / T
        exp_b = np.exp(scaled)
        w = exp_b / exp_b.sum()
        
        # Gradient of sum_k w[k] * b[k] w.r.t. a
        grad_a = np.zeros(n)
        for k in range(len(w)):
            if w[k] < 1e-12:
                continue
            wk = w[k]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * wk * a[k - idx]
        
        grad_a *= 2.0 * n / T
        # Transform to z-space
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - a * S)
        return grad_z, a
    
    def compute_score(a_val):
        return 2.0 * n * fast_convolve(a_val).max()

    while monotonic() - t0 < time_limit:
        t += 1
        grad_z, a_current = compute_grad(z)
        
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * grad_z**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Convert back to a
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a_new = exp_z / exp_z.sum()
        
        score = compute_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            best_z = z.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                # Reset with perturbation
                z = best_z + np.random.standard_normal(n) * 0.01
                no_improve = 0
                lr = max(lr * 0.5, 1e-5)
        
        if t % 500 == 0:
            elapsed = monotonic() - t0
            print(f"    Adam {elapsed:.0f}s: score={best_score:.6f} lr={lr:.6f}")

    return best_a, best_score
