def project_simplex(v):

    n = len(v)

    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def spectral_peak_shave(a0, time_limit, alpha_init=0.3, alpha_min=0.01,
                        n_iter_max=50000):

    Spectral peak-shaving: modify the auto-convolution in frequency domain
    to make it flatter, then recover the time-domain sequence.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()

        if time.monotonic() - t0 > time_limit:
            break

        # Adaptive alpha: start aggressive, become conservative
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        alpha = alpha_init * (alpha_min / alpha_init) ** progress

        # Compute auto-convolution
        b = fast_convolve(a)

        # Target: approximately uniform convolution
        target = b.mean()

        # Modify b to be flatter: geometric interpolation with target
        b_pos = np.maximum(b, 1e-30)
        b_new = b_pos ** (1 - alpha) * (target ** alpha)

        # Convert back to power spectrum
        P_new = np.fft.fft(b_new, fft_size)
        # Enforce non-negativity (small violations expected)
        P_new = np.maximum(np.real(P_new), 0)

        # Recover magnitude spectrum
        A_old = rfft(a, fft_size)
        mag_new = np.sqrt(P_new[:fft_size // 2 + 1])

        # Preserve phase
        phase_old = np.angle(A_old)
        A_new = mag_new * np.exp(1j * phase_old)

        # Recover time domain
        a_new = np.fft.ifft(A_new, fft_size)[:n]
        a_new = np.real(a_new)

        # Enforce non-negativity and normalize
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()

        score = 2.0 * n * fast_convolve(a_new).max()

            best_a = a_new.copy()
            a = a_new
        else:

def topk_gradient_descent(a0, time_limit, K_init=20, K_min=3,
                          lr_init=0.01, lr_min=0.0001):
    """
    Exact gradient descent using top-K peaks with projected steps.
    Works in a-space (not log-space) with explicit simplex projection.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    v = np.zeros(n)  # momentum

    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)

        # Adaptive K and lr
        K = max(K_min, int(K_init * (1 - progress) + K_min * progress))
        lr = lr_init * (lr_min / lr_init) ** progress

        b = fast_convolve(a)
        max_b = b.max()

        # Find top-K peaks (exclude endpoints which are always small)
        b_search = b.copy()
        b_search[0] = -np.inf
        b_search[-1] = -np.inf
        top_idx = np.argsort(b_search)[::-1][:K]

        # Weight by proximity to max
        vals = b[top_idx]
        T_peaks = 0.05 * max_b if max_b > 0 else 0.05
        weights = np.exp(vals / T_peaks - vals.max() / T_peaks)
        weights = weights / weights.sum()

        # Compute combined gradient
        grad = np.zeros(n)
        for ki, wi in zip(top_idx, weights):
            k = int(ki)
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += wi * 2.0 * a[j]
        grad *= 2.0 * n

        # Project gradient onto tangent space of simplex (sum=0)
        grad = grad - grad.mean()

        # Nesterov-style momentum
        v = 0.9 * v + grad

        # Taking step with projection
        a_trial = project_simplex(a - lr * v)

        score = 2.0 * n * fast_convolve(a_trial).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
        else:
            # Backtracking: halve lr for this step
            v *= 0.9  # dampen momentum

    return best_a, best_score

        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        b_search = b.copy()
        b_search[0] = -np.inf
        b_search[-1] = -np.inf
        peak_k = int(np.argmax(b_search))
        peak_val = b[peak_k]

        # Check if peak is significantly above target
        target = best_score / (2.0 * n)
        if peak_val <= target * 1.005:
            # Try second peak too
            b_search[peak_k] = -np.inf
            peak_k2 = int(np.argmax(b_search))
            if b[peak_k2] <= target * 1.005:
                continue
            peak_k = peak_k2
            peak_val = b[peak_k]

        # Elements contributing to this peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_count = min(8, len(idx))
        top_idx = np.argsort(contribs)[::-1][:top_count]

        improved = False
        for ci in top_idx:
            i = int(idx[ci])
            j = int(peak_k - i)
            if i == j or a[j] < 1e-15 or a[i] < 1e-15:
                continue
            for f1 in factors_reduce:
                for f2 in factors_incre:
                    a_trial = a.copy()
                    a_trial[i] *= f1
                    a_trial[j] *= f2
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a_trial.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Single element reduction with redistribution
            for ci in top_idx[:4]:
                i = int(idx[ci])
                if a[i] < 1e-15:
                    continue
                for factor in factors_reduce:
                    a_trial = a.copy()
                    diff = a_trial[i] * (1.0 - factor)
                    a_trial[i] *= factor
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    others = a_trial[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        continue
                    # Redistribute proportionally to neighbor elements
                    a_trial[mask] += diff * others / others_sum
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a_trial.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved and trial > 200 and trial % 100 == 0:
            # Random perturbation to escape
            a = best_a + np.random.standard_normal(n) * 0.01
            a = np.maximum(a, 0)
            a = a / a.sum()

    return best_a, best_score


def coordinate_descent_v2(a0, time_limit, n_sweeps=300):
    """
    Coordinate descent with adaptive factor sets.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    fine_factors = [0.85, 0.9, 0.95, 1.05, 1.1, 1.15]
    coarse_factors = [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break

        order = rng.permutation(n)
        sweep_improved = False
        factors = fine_factors if sweep > 20 else coarse_factors

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-12:
                continue

            best_local = a[i]
            best_local_score = best_score

            for factor in factors:
                a_trial = a.copy()
                a_trial[i] = a[i] * factor
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_local_score:
                    best_local_score = score
                    best_local = a[i] * factor

            if best_local != a[i]:
                a[i] = best_local
                a = a / a.sum()
                if best_local_score < best_score:
                    best_score = best_local_score
                    best_a = a.copy()
                    sweep_improved = True

        if not sweep_improved:
            break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def diverse_initializations(parent, n, rng):
    """Generate diverse starting points."""
    candidates = []

    if parent is not None:
        if len(parent) == n:
            p = np.maximum(parent, 0)
            p = p / p.sum()
            candidates.append(('parent', p.copy()))
            for scale in [0.005, 0.01, 0.02, 0.05]:
                pert = p + rng.standard_normal(n) * scale
                pert = np.maximum(pert, 0)
                pert = pert / pert.sum()
                candidates.append((f'pert_{scale}', pert))
            for freq in [2, 3, 5, 7, 11]:
                t = np.linspace(0, 2 * np.pi * freq, n)
                pert = p + 0.02 * np.cos(t)
                pert = np.maximum(pert, 0)
                sp = pert.sum()
                if sp > 1e-10:
                    pert = pert / sp
                    candidates.append((f'cos_{freq}', pert))
        else:
            interp = interpolate_sequence(parent, n)
            candidates.append(('parent_interp', interp))

    candidates.append(('uniform', np.ones(n) / n))

    # Triangular
    tri = np.linspace(0, 1, n) + np.linspace(1, 0, n)
    tri = np.maximum(tri, 0.01)
    tri = tri / tri.sum()
    candidates.append(('triangular', tri))

    # Bimodal
    mid = n // 2
    bi = np.zeros(n)
    w1 = max(1, n // 8)
    w2 = max(1, n // 5)
    bi[max(0, mid - w1):mid + w1 + 1] = 1.0
    bi[max(0, mid - w2):mid + w2 + 1] = 0.6
    bi = np.maximum(bi, 0.001)
    bi = bi / bi.sum()
    candidates.append(('bimodal', bi))

    # Gaussian
    x = np.linspace(-3, 3, n)
    gau = np.exp(-x ** 2 / 3)
    gau = np.maximum(gau, 1e-6)
    gau = gau / gau.sum()
    candidates.append(('gaussian', gau))

    # Random
    rand = np.abs(rng.standard_normal(n)) + 0.01
    rand = rand / rand.sum()
    candidates.append(('random', rand))

    # Exponential
    expo = np.exp(-np.linspace(0, 2.5, n))
    expo = expo / expo.sum()
    candidates.append(('expo', expo))

    # Sine-modulated
    t = np.linspace(0, 2 * np.pi, n)
    sine = 1.0 + 0.3 * np.sin(t)
    sine = np.maximum(sine, 0.1)
    sine = sine / sine.sum()
    candidates.append(('sine_mod', sine))

    return candidates


def evaluate_and_track(a, label, best_seq, best_score):
    """Evaluate a candidate and return updated best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase hybrid optimization:
    Phase 1: Multi-start Adam log-space (35% time)
    Phase 2: Spectral peak shaving (20% time)
    Phase 3: Top-K gradient descent (15% time)
    Phase 4: Enhanced local search (20% time)
    Phase 5: Multi-length + final polish (10% time)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_default = 768
        uniform = np.ones(n_default) / n_default
        best_seq = uniform
        best_score = float(evaluate_sequence(uniform.tolist()))
        print(f"Default: n={n_default}, score={best_score:.6f}")

    n_parent = len(best_seq)
    print(f"Starting best score: {best_score:.6f}, n={n_parent}")

    # === Phase 1: Multi-start Adam (35% time) ===
    print(f"\n=== Phase 1: Multi-Start Adam (n={n_parent}) ===")
    t_p1 = min(630.0, (deadline - time.monotonic()) * 0.35)
    t_p1_start = time.monotonic()

    inits = diverse_initializations(
        parent if parent is not None else best_seq, n_parent, rng)

    # Score and sort initializations
    inits_scored = []
    for label, init_a in inits:
        s = 2.0 * n_parent * fast_convolve(init_a).max()
        inits_scored.append((s, label, init_a))
    inits_scored.sort(key=lambda x: x[0])

    n_starts = min(8, len(inits_scored))
    time_per_start = t_p1 / n_starts

    for idx in range(n_starts):
        rem = deadline - time.monotonic() - 300
        if rem < 60:
            break
        t_alloc = min(time_per_start, rem * 0.9)
        if t_alloc < 30:
            continue

        s_init, label, init_a = inits_scored[idx]
        print(f"  Adam Start {idx+1}/{n_starts}: {label} (init={s_init:.6f})")

        # Try 2 different configurations per start
        configs = [
            (0.04, 0.002, 0.008, 'hold_anneal'),
            (0.025, 0.001, 0.006, 'exp'),
        ]
        for T_i, T_f, lr, sched in configs:
            rem = deadline - time.monotonic() - 250
            if rem < 20:
                break
            t_sub = min(t_alloc / 2.0, rem * 0.8)
            if t_sub < 15:
                continue
            a_res, score_res = adam_logspace(
                init_a, t_sub, T_i, T_f, lr, sched
            )
            best_seq, best_score = evaluate_and_track(
                a_res, f"adam_{label[:8]}_{sched[:3]}", best_seq, best_score)

    print(f"  After Phase 1: best={best_score:.6f}")

    # === Phase 2: Spectral Peak Shaving (20% time) ===
    print(f"\n=== Phase 2: Spectral Peak Shaving ===")
    rem = deadline - time.monotonic() - 200
    if rem > 60:
        t_sub = min(rem * 0.2, 360.0)
        print(f"  Spectral peak shave, budget={t_sub:.0f}s")
        a_spec, score_spec = spectral_peak_shave(
            best_seq, t_sub, alpha_init=0.4, alpha_min=0.02
        )
        best_seq, best_score = evaluate_and_track(
            a_spec, "spectral_ps", best_seq, best_score)

        # Second pass with different parameters
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            t_sub2 = min(rem * 0.15, 180.0)
            a_spec2, _ = spectral_peak_shave(
                best_seq, t_sub2, alpha_init=0.15, alpha_min=0.005
            )
            best_seq, best_score = evaluate_and_track(
                a_spec2, "spectral_ps2", best_seq, best_score)

    print(f"  After Phase 2: best={best_score:.6f}")

    # === Phase 3: Top-K Gradient Descent (15% time) ===
    print(f"\n=== Phase 3: Top-K Gradient Descent ===")
    rem = deadline - time.monotonic() - 150
    if rem > 40:
        t_sub = min(rem * 0.15, 270.0)
        a_topk, _ = topk_gradient_descent(
            best_seq, t_sub, K_init=30, K_min=5,
            lr_init=0.005, lr_min=0.0005
        )
        best_seq, best_score = evaluate_and_track(
            a_topk, "topk_grad", best_seq, best_score)

        # Another pass with different params
        rem = deadline - time.monotonic() - 100
        if rem > 30:
            t_sub2 = min(rem * 0.1, 150.0)
            a_topk2, _ = topk_gradient_descent(
                best_seq, t_sub2, K_init=15, K_min=3,
                lr_init=0.003, lr_min=0.0003
            )
            best_seq, best_score = evaluate_and_track(
                a_topk2, "topk_grad2", best_seq, best_score)

    print(f"  After Phase 3: best={best_score:.6f}")

    # === Phase 4: Enhanced Local Search (20% time) ===
    print(f"\n=== Phase 4: Enhanced Local Search ===")
    rem = deadline - time.monotonic() - 100

    if rem > 30:
        t_sub = min(rem * 0.12, 216.0)
        a_ps, _ = enhanced_peak_shave(best_seq, t_sub, n_trials=15000)
        best_seq, best_score = evaluate_and_track(
            a_ps, "enhanced_ps", best_seq, best_score)

    rem = deadline - time.monotonic() - 80
    if rem > 20:
        t_sub = min(rem * 0.1, 180.0)
        a_cd, _ = coordinate_descent_v2(best_seq, t_sub, n_sweeps=200)
        best_seq, best_score = evaluate_and_track(
            a_cd, "coord_desc", best_seq, best_score)

    # More peak shaving after coord descent
    rem = deadline - time.monotonic() - 60
    if rem > 15:
        t_sub = min(rem * 0.1, 120.0)
        a_ps2, _ = enhanced_peak_shave(best_seq, t_sub, n_trials=10000)
        best_seq, best_score = evaluate_and_track(
            a_ps2, "enhanced_ps2", best_seq, best_score)

    print(f"  After Phase 4: best={best_score:.6f}")

    # === Phase 5: Multi-length + Final Polish (10% time) ===
    print(f"\n=== Phase 5: Multi-Length + Final Polish ===")
    rem = deadline - time.monotonic() - 30

    # Try a few nearby lengths
    if rem > 60:
        n_best = len(best_seq)
        alt_lengths = []
        for delta in [32, -32, 64, -64, 128, -128]:
            alt_n = n_best + delta
            if 256 <= alt_n <= 4096:
                alt_lengths.append(alt_n)

        for alt_n in alt_lengths[:3]:
            rem = deadline - time.monotonic() - 20
            if rem < 30:
                break
            t_sub = min(rem * 0.15, 60.0)
            init_alt = interpolate_sequence(best_seq, alt_n)
            a_alt, _ = adam_logspace(
                init_alt, t_sub, 0.03, 0.002, 0.007, 'hold_anneal'
            )
            best_seq, best_score = evaluate_and_track(
                a_alt, f"alt_n{alt_n}", best_seq, best_score)

    # Final intensive Adam polish
    rem = deadline - time.monotonic() - 10
    if rem > 20:
        t_sub = min(rem * 0.5, 60.0)
        a_fin, _ = adam_logspace(
            best_seq, t_sub, 0.01, 0.0005, 0.003, 'exp',
            max_no_improve=2000
        )
        best_seq, best_score = evaluate_and_track(
            a_fin, "final_adam", best_seq, best_score)

    # Final peak shave
    rem = deadline - time.monotonic() - 5
    if rem > 10:
        t_sub = min(rem * 0.5, 30.0)
        a_ps_fin, _ = enhanced_peak_shave(best_seq, t_sub, n_trials=5000)
        best_seq, best_score = evaluate_and_track(
            a_ps_fin, "final_ps", best_seq, best_score)

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.6f}")

    return [float(x) for x in best_seq]
