    a_rev = a[::-1]
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    A_rev = rfft(a_rev, fft_size)
    G = rfft(g, fft_size)
    conv_val = irfft(A_rev * G, fft_size)
    
    # We need conv_val[j+n-1] for j = 0, ..., n-1, i.e., indices n-1 to 2n-2
    grad_a = 4.0 * n * conv_val[n-1:2*n-1]
    
    return smooth_obj, grad_a, b, g


def adam_step(z, grad_z, m, v, t, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """In-place Adam update, returns updated z, m, v."""
    m = beta1 * m + (1.0 - beta1) * grad_z
    v = beta2 * v + (1.0 - beta2) * grad_z * grad_z
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
    return z, m, v


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    global _lp_solver_cached
    if '_lp_solver_cached' in globals():
        return _lp_solver_cached
    for solver_name in [cp.PDLP, cp.GUROBI, cp.MOSEK, cp.SCS, cp.ECOS, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b (non-negative, sum=1) minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.SCS, cp.PDLP, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def adam_optimize(a0, time_limit, T_start, T_end, lr, max_steps=None, verbose=True):
    """
    Adam optimization in log-space (softmax parameterization) with decreasing temperature.
    a0: initial sequence (non-negative, sum to 1)
    Returns best sequence found.
    """
    n = len(a0)
    a_safe = np.maximum(a0, 1e-30)
    z = np.log(a_safe)
    z = z - z.mean()  # center for numerical stability
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    a = a_safe
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    t0 = monotonic()
    total_time = time_limit
    # Number of steps to reach T_end from T_start geometrically
    if T_end > 0 and T_start > T_end:
        log_ratio = np.log(T_end / T_start)
    else:
        log_ratio = np.log(0.01)
    
    improved = 0
    stagnant = 0
    
    while monotonic() - t0 < total_time:
        t += 1
        # Temperature schedule: geometric from T_start to T_end
        frac = min(1.0, t / (total_time * 50))  # adapt rate
        T = T_start * (T_end / T_start) ** frac
        
        # Compute a from z
        a = softmax_vec(z)
        b = fast_convolve(a)
        current_score = 2.0 * n * b.max()
        
        # Check for improvement
        if current_score < best_score - 1e-12:
            best_score = current_score
            best_a = a.copy()
            improved += 1
            stagnant = 0
        else:
            stagnant += 1
        
        # If very stagnant, add perturbation
        if stagnant > 5000 and t % 1000 == 0:
            z = z + np.random.standard_normal(n) * 0.01
            stagnant = 0
        
        # Compute gradient
        _, grad_a, _, _ = compute_smoothed_objective_and_grad(a, T)
        
        # Transform to z-space: dz[j] = a[j] * (grad_a[j] - sum_i grad_a[i]*a[i])
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)
        
        # Adam step
        z, m, v = adam_step(z, grad_z, m, v, t, lr)
        
        if max_steps is not None and t >= max_steps:
            break
        
        if verbose and t % 5000 == 0:
            elapsed = monotonic() - t0
            print(f"    Adam t={t} ({elapsed:.0f}s): T={T:.6f} score={best_score:.6f} cur={current_score:.6f}")
    
    return best_a, best_score


def lp_fictitious_play_short(a0, time_limit, solver=None, max_iters=100):
    """Short run of LP fictitious play for a strong single move."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    
    for it in range(max_iters):
        if monotonic() - t0 > time_limit:
            break
        b_new, _ = solve_lp_best_response(a, solver)
        if b_new is None:
            continue
        # Damped update
        alpha = max(0.1, 2.0 / (it + 1))
        a_new = (1 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        a = a_new
        
        # Also try symmetric version
        b_sym, _ = solve_lp_best_response_symmetric(a, solver)
        if b_sym is not None:
            a_sym = (1 - alpha) * a + alpha * b_sym
            a_sym = np.maximum(a_sym, 0)
            s = a_sym.sum()
            if s > 1e-15:
                a_sym = a_sym / s
                score_sym = 2.0 * n * fast_convolve(a_sym).max()
                if score_sym < best_score - 1e-14:
                    best_score = score_sym
                    best_a = a_sym.copy()
                    a = a_sym
    
    return best_a, best_score


def solve_lp_best_response_symmetric(a_fixed, solver=None):
    """LP best response with palindromic constraint."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    half = (n + 1) // 2
    b_half = cp.Variable(half, nonneg=True)
    M_var = cp.Variable()
    if n % 2 == 0:
        b_full = cp.hstack([b_half, cp.flip(b_half)])
    else:
        b_full = cp.hstack([b_half, cp.flip(b_half[1:])])
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_full <= M_var,
        cp.sum(b_full) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.SCS, cp.PDLP, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_half_val = np.asarray(b_half.value, dtype=np.float64)
    b_half_val = np.maximum(b_half_val, 0)
    if n % 2 == 0:
        b_val = np.concatenate([b_half_val, b_half_val[::-1]])
    else:
        b_val = np.concatenate([b_half_val, b_half_val[1:][::-1]])
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def peak_shave_optimized(a0, time_limit):
    """Efficient peak shaving using FFT-based gradient targeting."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    
    for step in range(100000):
        if monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]
        if peak_val <= best_score / (2.0 * n) * (1 + 1e-10):
            break
        
        # Find top contributors
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_n = min(10, len(contribs))
        top_contrib = np.argsort(contribs)[::-1][:top_n]
        
        improved = False
        for ci in top_contrib:
            i = int(idx[ci])
            j = int(peak_k - i)
            if a[i] < 1e-14:
                continue
            
            # Try reducing a[i] (and redistributing)
            for factor in [0.95, 0.9, 0.85, 0.8]:
                old_val = a[i]
                new_val = old_val * factor
                diff = old_val - new_val
                if diff < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                # Redistribute to others proportionally
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] = others + diff * others / others_sum
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        # If not improved by single reduction, try pair reduction
        if not improved:
            for ci in top_contrib:
                i = int(idx[ci])
                j = int(peak_k - i)
                if i == j or a[j] < 1e-14:
                    continue
                for f1, f2 in [(0.9, 0.95), (0.85, 0.9), (0.95, 0.85)]:
                    aTrial = a.copy()
                    aTrial[i] *= f1
                    aTrial[j] *= f2
                    aTrial = np.maximum(aTrial, 0)
                    aTrial = aTrial / aTrial.sum()
                    score = 2.0 * n * fast_convolve(aTrial).max()
                    if score < best_score - 1e-13:
                        a = aTrial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        # Random perturbation as last resort
        if not improved:
            a_trial = a + np.random.standard_normal(n) * 0.001
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-13:
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True
        
        if not improved and step % 100 == 99:
            # Add larger perturbation
            a = best_a + np.random.standard_normal(n) * 0.005
            a = np.maximum(a, 0)
            a = a / a.sum()
    
    return best_a, best_score


def subgradient_refine(a0, time_limit, top_k=50):
    """Subgradient method with Nesterov acceleration on the simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    
    # Yinyang averaging
    a_hat = a.copy()
    v = np.zeros(n)
    momentum = 0.9
    lr = 0.005
    no_improve = 0
    
    while monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        k_actual = min(top_k, len(b))
        peaks = np.argpartition(b, -k_actual)[-k_actual:]
        pv = b[peaks]
        # Softmax weights over peaks
        temp = 0.001 * max(pv.max(), 1e-10)
        scaled = (pv - pv.max()) / temp
        w = np.exp(scaled)
        ws = w.sum()
        w = w / ws
        
        # Gradient computation
        grad = np.zeros(n)
        for ii in range(len(peaks)):
            if w[ii] < 1e-12:
                continue
            k = int(peaks[ii])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * w[ii] * a[k - idx]
        
        # Momentum step
        v = momentum * v - lr * grad
        a_candidate = a + v
        
        # Project onto simplex
        u = np.sort(a_candidate)[::-1]
        cssv = np.cumsum(u)
        cond = u * np.arange(1, n + 1) > (cssv - 1)
        if cond.any():
            rho = np.nonzero(cond)[0][-1]
            theta = (cssv[rho] - 1.0) / (rho + 1.0)
            a_new = np.maximum(a_candidate - theta, 0.0)
        else:
            a_new = np.ones(n) / n
        a_new = a_new / a_new.sum()
        
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            a_hat = a_hat * 0.95 + a_new * 0.05
            no_improve = 0
        else:
            no_improve += 1
        
        lr *= 0.9995
        if no_improve > 2000:
            a = best_a + np.random.standard_normal(n) * 0.003
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
            lr = max(lr * 3, 0.004)
            no_improve = 0
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    rem = deadline - monotonic() - 5
    if rem > 8:
        a_ps_final, _ = peak_shave_optimized(best_seq, min(rem * 0.8, 15.0))
        actual_score = float(evaluate_sequence(a_ps_final.tolist()))
        if np.isfinite(actual_score) and actual_score < best_score:
            best_score = actual_score
            best_seq = a_ps_final.copy()
            print(f"  Final PS improved: {best_score:.6f}")
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = min(best_score, fs)
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
