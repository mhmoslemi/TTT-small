def l4_gradient(a, b):
    """
    Compute gradient of sum(b[k]^2) where b = conv(a,a).
    Returns grad of shape (n,).
    """
    n = len(a)
    grad = np.zeros(n)
    for m in range(n):
        s = 0.0
        # Term 1: sum_k b[k] * a[k-m]
        # k-m in [0, n-1] => k in [m, m+n-1], and k in [0, 2n-2]
        k_lo = max(0, m)
        k_hi = min(2*n-2, m+n-1)
        for k in range(k_lo, k_hi+1):
            s += b[k] * a[k-m]
        # Term 2: sum_k b[k] * a[m-k]
        # m-k in [0, n-1] => k in [m-n+1, m], and k in [0, 2n-2]
        k_lo = max(0, m-n+1)
        k_hi = min(2*n-2, m)
        for k in range(k_lo, k_hi+1):
            s += b[k] * a[m-k]
        grad[m] = 2.0 * s
    return grad

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. L4 (L2-norm-of-convolution) gradient descent for initial flat convolution
2. Soft-max peak reduction with Adam optimizer and temperature annealing
3. Penalty-based peak constraint optimization
4. Peak shaving and coordinate descent for local refinement
5. Multi-length exploration and simulated annealing for diversity

Key innovations over previous approach:
- L4 minimization as a convex pre-optimization for better initialization
- Penalty-based alternative to soft-max for harder peaks
- Simulated annealing for escaping local minima
- More aggressive multi-length exploration

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - np.max(z)
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    return np.log(np.maximum(a, 1e-300))


def compute_l4_gradient(a, b, n, fft_size):
    """
    Compute gradient of sum(b[k]^2) where b = conv(a,a).
    grad[m] = 2 * (cross_corr(a,b)[m] + conv(a,b)[m])
    """
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    # cross_corr[m] = sum_j a[j] * b[j+m] = IDFT(conj(A) * B)
    cross_corr = irfft(np.conj(A) * B, fft_size)[:n]
    # conv_ab[m] = sum_j a[j] * b[m-j] = IDFT(A * B)
    conv_ab = irfft(A * B, fft_size)[:n]
    return 2.0 * (cross_corr + conv_ab)


def compute_softmax_grad_a(a, w, n, fft_size):
    """
    Compute gradient of 2n * softmax(conv(a,a), T) w.r.t. a.
    grad_a[i] = 2n * sum_{j} w[j+i] * a[j] (cross-correlation)
    """
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(W * np.conj(A), fft_size)[:n]
    return 2.0 * n * corr


def compute_penalty_gradient(a, b, t, n, fft_size, lam=1.0):
    """
    Compute gradient of penalty objective: sum max(0, b[k]-t)^2 + lam*(sum(a)-1)^2
    """
    excess = np.maximum(b - t, 0.0)
    # grad[m] = 2 * sum_k excess[k] * (a[k-m] + a[m-k]) + 2*lam*(sum(a)-1)
    E = rfft(excess, fft_size)
    A = rfft(a, fft_size)
    cross_corr = irfft(np.conj(A) * E, fft_size)[:n]
    conv_ab = irfft(A * E, fft_size)[:n]
    sum_term = 2.0 * lam * (a.sum() - 1.0)
    return 2.0 * (cross_corr + conv_ab) + sum_term


def get_score_local(a):
    """Compute the objective score for a sequence (local, no evaluate_sequence call)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def adam_logspace(a0, time_limit, T_init=0.05, T_final=0.002, lr=0.01,
                  beta1=0.9, beta2=0.999, eps=1e-8, schedule='exp',
                  mode='softmax', verbose=False, verbose_interval=5000):
    """
    Adam optimizer in log-space with temperature annealing.
    mode='softmax': minimize soft-max of convolution
    mode='penalty': minimize penalty for exceeding threshold t
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    a = a / a.sum()
    z = z_from_a(a)
    m = np.zeros(n)
    v = np.zeros(n)
    t_step = 0
    fft_size = 1 << (3 * n - 2).bit_length()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    no_improve = 0
    max_no_improve = 5000

    # For penalty mode, target threshold
    target_t = best_score / (2.0 * n) * 0.95

    while time.monotonic() - t0 < time_limit:
        t_step += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)

        # Temperature schedule
        if schedule == 'linear':
            T = T_init + (T_final - T_init) * progress
        elif schedule == 'exp':
            T = T_init * (T_final / T_init) ** progress
        elif schedule == 'hold_anneal':
            if progress < 0.3:
                T = T_init
            else:
                p2 = (progress - 0.3) / 0.7
                T = T_init * (T_final / T_init) ** p2
        else:
            T = T_init * (T_final / T_init) ** progress

        a_curr = softmax_from_z(z)
        b = fast_convolve(a_curr)
        max_b = b.max()

        if mode == 'softmax':
            exp_b = np.exp(b / T - max_b / T)
            w = exp_b / exp_b.sum()
            grad_a = compute_softmax_grad_a(a_curr, w, n, fft_size)
        elif mode == 'penalty':
            grad_a = compute_penalty_gradient(a_curr, b, target_t, n, fft_size)
        else:
            grad_a = compute_softmax_grad_a(a_curr, b / (b.max() + 1e-30), n, fft_size)

        # Transform to z-space
        S = np.dot(grad_a, a_curr)
        grad_z = a_curr * (grad_a - S)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t_step)
        v_hat = v / (1.0 - beta2 ** t_step)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Evaluate
        a_new = softmax_from_z(z)
        conv_new = fast_convolve(a_new)
        score = 2.0 * n * conv_new.max()

        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive restart
        if no_improve > max_no_improve:
            z = z_from_a(best_a) + np.random.standard_normal(n) * 0.01
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = max(0.001, lr * 0.8)

        # Slow LR decay
        lr = max(0.0005, lr * (1.0 - 0.00005))

        if verbose and t_step % verbose_interval == 0:
            print(f"      Adam t={t_step} T={T:.5f} lr={lr:.6f} score={best_score:.6f}")

    return best_a, best_score


def l4_gradient_descent(a0, time_limit, lr=0.01, momentum=0.9, verbose=False):
    """
    Gradient descent on the L4 objective (sum of squared convolution values).
    This is a convex problem and should converge to a global minimum.
    Works in log-space to maintain non-negativity.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    a = a / a.sum()
    z = z_from_a(a)
    vel = np.zeros(n)
    t0 = time.monotonic()
    best_a = a.copy()
    best_l4 = np.sum(fast_convolve(a) ** 2)
    best_score = 2.0 * n * fast_convolve(a).max()
    fft_size = 1 << (3 * n - 2).bit_length()
    step = 0

    while time.monotonic() - t0 < time_limit:
        step += 1
        a_curr = softmax_from_z(z)
        b = fast_convolve(a_curr)

        # L4 gradient in a-space
        grad_a = compute_l4_gradient(a_curr, b, n, fft_size)

        # Transform to z-space
        S = np.dot(grad_a, a_curr)
        grad_z = a_curr * (grad_a - S)

        # Momentum update
        vel = momentum * vel - lr * grad_z
        z = z + vel

        a_new = softmax_from_z(z)
        b_new = fast_convolve(a_new)
        l4_new = np.sum(b_new ** 2)

        if l4_new < best_l4:
            best_l4 = l4_new
            best_a = a_new.copy()
            score_new = 2.0 * n * b_new.max()
            if score_new < best_score:
                best_score = score_new

        # Adaptive LR
        if step % 1000 == 0:
            lr = max(0.001, lr * 0.9)

        if verbose and step % 5000 == 0:
            print(f"      L4 t={step} l4={best_l4:.6f} score={best_score:.6f}")

    return best_a, best_score


def peak_shave(a0, time_limit, n_trials=10000):
    """
    Peak-targeted local search: identify the autocorrelation peak and
    try to reduce it by adjusting contributing elements.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        # Find peak (prefer interior peaks with more contributions)
        b_search = b.copy()
        b_search[0] = -np.inf
        b_search[-1] = -np.inf
        peak_k = int(np.argmax(b_search))
        peak_val = b[peak_k]

        # Elements contributing to this peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

        # Try reducing top contributors
        top_idx = np.argsort(contribs)[::-1][:8]
        improved = False

        for ci in top_idx:
            i = idx[ci]
            j = int(peak_k - i)
            if i == j or a[j] < 1e-15 or a[i] < 1e-15:
                continue

            for f1 in [0.97, 0.93, 0.88, 0.82]:
                for f2 in [1.0, 1.01, 1.03, 1.06]:
                    a_trial = a.copy()
                    a_trial[i] *= f1
                    a_trial[j] *= f2
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a_trial.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Try single-element reduction with redistribution
            for ci in top_idx[:5]:
                i = int(idx[ci])
                if a[i] < 1e-15:
                    continue
                for factor in [0.92, 0.85, 0.75]:
                    a_trial = a.copy()
                    diff = a_trial[i] * (1.0 - factor)
                    a_trial[i] *= factor
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    others_sum = a_trial[mask].sum()
                    if others_sum < 1e-15:
                        continue
                    a_trial[mask] += diff * a_trial[mask] / others_sum
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a_trial.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved and trial > 200 and trial % 100 == 0:
            # Stuck - try random perturbation
            scale = 0.01 + 0.02 * min(1.0, trial / 5000.0)
            a = best_a + np.random.standard_normal(n) * scale
            a = np.maximum(a, 0)
            a = a / a.sum()

    return best_a, best_score


def coordinate_descent(a0, time_limit, n_sweeps=500):
    """
    Coordinate descent with multiplicative updates.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(9999)
    fft_size = 1 << (2 * n - 2).bit_length()

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break

        order = rng.permutation(n)
        sweep_improved = False

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-15:
                continue

            best_local = a[i]
            best_local_score = best_score

            for factor in [0.75, 0.85, 0.93, 1.07, 1.15, 1.3]:
                a_trial = a.copy()
                a_trial[i] = a[i] * factor
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_local_score:
                    best_local_score = score
                    best_local = a[i] * factor

            if best_local != a[i]:
                a[i] = best_local
                if best_local_score < best_score:
                    best_score = best_local_score
                    best_a = a.copy()
                    sweep_improved = True

        if not sweep_improved:
            break

    return best_a, best_score


def simulated_annealing(a0, time_limit, n_trials=20000):

    Tries random perturbations and accepts with temperature-dependent probability.
    """

    T_end = 0.0001

        if time.monotonic() - t0 > time_limit:
            break

        progress = trial / n_trials
        T = T_start * (T_end / T_start) ** progress

        # Propose perturbation: modify a few random elements
        n_mods = rng.integers(1, max(2, n // 100))
        a_trial = a.copy()
        for _ in range(n_mods):
            i = rng.integers(0, n)
            if a_trial[i] > 1e-15:
                factor = 1.0 + rng.standard_normal() * 0.1
                a_trial[i] = max(0.0, a_trial[i] * factor)

        s = a_trial.sum()
        if s < 1e-15:
            continue
        a_trial = a_trial / s

        score = 2.0 * n * fast_convolve(a_trial).max()

        # Accept if better or with probability exp(-delta/T)
        delta = score - current_score
        if delta < 0 or rng.random() < np.exp(-delta / T):
            a = a_trial
            current_score = score
            if score < best_score - 1e-14:
                best_score = score
                best_a = a_trial.copy()

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def generate_initializations(parent, n, rng):
    """Generate diverse starting points."""
    candidates = []

    if parent is not None:
        if len(parent) == n:
            p = np.maximum(parent, 0)
            p = p / p.sum()
            candidates.append(('parent', p.copy()))
            for scale in [0.01, 0.03, 0.05]:
                pert = p + rng.standard_normal(n) * scale
                pert = np.maximum(pert, 0)
                pert = pert / pert.sum()
                candidates.append((f'parent_pert_{scale}', pert))
            # Cosine modulation
            for freq in [1, 2, 3, 5]:
                t = np.linspace(0, 2 * np.pi * freq, n)
                pert = p * (1.0 + 0.1 * np.cos(t))
                pert = np.maximum(pert, 0)
                sp = pert.sum()
                if sp > 1e-10:
                    pert = pert / sp
                    candidates.append((f'parent_cos_{freq}', pert))
        else:
            interp = interpolate_sequence(parent, n)
            candidates.append(('parent_interp', interp))

    candidates.append(('uniform', np.ones(n) / n))

    # Triangular
    tri = np.linspace(0, 1, n)
    tri += np.linspace(1, 0, n)
    tri = np.maximum(tri, 0.01)
    tri = tri / tri.sum()
    candidates.append(('triangular', tri))

    # Bimodal
    mid = n // 2
    bi = np.zeros(n)
    bi[max(0, mid - n // 6):mid + n // 6] = 1.0
    bi[max(0, mid - n // 4):mid + n // 4] = 0.5
    bi = np.maximum(bi, 0.001)
    bi = bi / bi.sum()
    candidates.append(('bimodal', bi))

    # Sinusoidal modulation
    for freq in [2, 4, 8]:
        t = np.linspace(0, 2 * np.pi * freq, n)
        sin_seq = 1.0 + 0.3 * np.cos(t)
        sin_seq = np.maximum(sin_seq, 0.01)
        sin_seq = sin_seq / sin_seq.sum()
        candidates.append((f'sin_{freq}', sin_seq))

    # Random
    rand = np.abs(rng.standard_normal(n)) + 0.01
    rand = rand / rand.sum()
    candidates.append(('random', rand))

    return candidates


def evaluate_and_track(a, label, best_seq, best_score, print_flag=True):
    """Evaluate a candidate and return updated best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if print_flag:
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining L4 convex pre-optimization,
    soft-max/penalty peak reduction, simulated annealing, and local polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except NameError:
        pass
    except Exception as e:
        print(f"Parent loading failed: {e}")

    if parent is None:
        n_default = 768
        uniform = np.ones(n_default) / n_default
        best_seq = uniform
        best_score = float(evaluate_sequence(uniform.tolist()))
        print(f"Default: n={n_default}, score={best_score:.6f}")

    n_parent = len(best_seq)
    print(f"Starting best score: {best_score:.6f} (n={n_parent})")

    # === Phase 1: L4 Convex Pre-optimization (25% time) ===
    print(f"\n=== Phase 1: L4 Convex Optimization (n={n_parent}) ===")
    t_p1 = min((deadline - time.monotonic()) * 0.25, 450.0)
    t_start = time.monotonic()
    fft_size_p1 = 1 << (3 * n_parent - 2).bit_length()

    # Run L4 from parent
    a_l4, _ = l4_gradient_descent(best_seq, min(t_p1 * 0.5, 200.0), lr=0.005, momentum=0.9, verbose=True)
    best_seq, best_score = evaluate_and_track(a_l4, "l4_parent", best_seq, best_score)
    print(f"  After L4 (parent): {best_score:.6f}")

    # Run L4 from diverse inits
    inits = generate_initializations(parent, n_parent, rng)
    inits_scored = []
    for label, init_a in inits:
        s = 2.0 * n_parent * fast_convolve(init_a).max()
        inits_scored.append((s, label, init_a))
    inits_scored.sort(key=lambda x: x[0])

    rem = deadline - time.monotonic() - (budget_s - 15.0) * 0.75
    for idx in range(min(3, len(inits_scored))):
        if time.monotonic() > deadline - (budget_s - 15.0) * 0.75:
            break
        s_init, label, init_a = inits_scored[idx]
        t_alloc = min(60.0, (deadline - time.monotonic()) * 0.1)
        if t_alloc < 10:
            continue
        a_l4, _ = l4_gradient_descent(init_a, t_alloc, lr=0.005, momentum=0.9)
        best_seq, best_score = evaluate_and_track(a_l4, f"l4_{label}", best_seq, best_score)
    print(f"  After Phase 1: {best_score:.6f}")

    # === Phase 2: Soft-max Peak Reduction (30% time) ===
    print(f"\n=== Phase 2: Soft-max Peak Reduction (best={best_score:.6f}) ===")
    t_p2_budget = (deadline - time.monotonic()) * 0.30

    # Multiple Adam runs with different schedules and modes
    n_starts = 4
    time_per_start = t_p2_budget / n_starts

    for idx in range(n_starts):
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t_alloc = min(time_per_start, rem * 0.45)
        if t_alloc < 15:
            break

        if idx == 0:
            # Focused on best with soft-max
            init_a = best_seq
            label = "focused_softmax"
            T_i, T_f, lr_val, sched = 0.03, 0.001, 0.007, 'hold_anneal'
            mode = 'softmax'
        elif idx == 1:
            # Best with penalty mode
            init_a = best_seq
            label = "focused_penalty"
            T_i, T_f, lr_val, sched = 0.05, 0.005, 0.005, 'exp'
            mode = 'penalty'
        elif idx == 2:
            # L4 result with soft-max
            init_a = a_l4 if 'a_l4' in dir() else best_seq
            label = "l4_to_softmax"
            T_i, T_f, lr_val, sched = 0.04, 0.002, 0.008, 'exp'
            mode = 'softmax'
        else:
            # Diverse init with soft-max
            if len(inits_scored) > 3:
                init_a = inits_scored[1][2]
            else:
                init_a = best_seq
            label = "diverse_softmax"
            T_i, T_f, lr_val, sched = 0.05, 0.003, 0.01, 'linear'
            mode = 'softmax'

        a_res, _ = adam_logspace(init_a, t_alloc, T_init=T_i, T_final=T_f,
                                  lr=lr_val, schedule=sched, mode=mode, verbose=False)
        best_seq, best_score =evaluate_and_track(a_res, label, best_seq, best_score)
        print(f"  After {label}: {best_score:.6f}")

    # === Phase 3: Simulated Annealing + Local Search (25% time) ===
    print(f"\n=== Phase 3: Annealing + Local Search (best={best_score:.6f}) ===")
    t_p3 = min((deadline - time.monotonic()) * 0.25, 400.0)
    n_best = len(best_seq)

    # Simulated annealing for exploration
    t_sub = min(t_p3 * 0.3, 120.0)
    rem_check = deadline - time.monotonic()
    if t_sub < 10 or rem_check < 60:
        t_sub = max(5.0, rem_check * 0.1)
    a_sa, _ = simulated_annealing(best_seq, t_sub, n_trials=15000)
    best_seq, best_score = evaluate_and_track(a_sa, "sim_anneal", best_seq, best_score)
    print(f"  After simulated annealing: {best_score:.6f}")

    # Peak shaving
    rem = deadline - time.monotonic()
    t_sub = min(rem * 0.25, 100.0)
    if t_sub > 10:
        a_ps, _ = peak_shave(best_seq, t_sub, n_trials=5000)
        best_seq, best_score = evaluate_and_track(a_ps, "peak_shave", best_seq, best_score)
        print(f"  After peak shave: {best_score:.6f}")

    # Coordinate descent
    rem = deadline - time.monotonic()
    t_sub = min(rem * 0.2, 80.0)
    if t_sub > 10:
        a_cd, _ = coordinate_descent(best_seq, t_sub, n_sweeps=200)
        best_seq, best_score = evaluate_and_track(a_cd, "coord_desc", best_seq, best_score)
        print(f"  After coord descent: {best_score:.6f}")

    # Second round of peak shaving after coord descent
    rem = deadline - time.monotonic()
    t_sub = min(rem * 0.15, 50.0)
    if t_sub > 5:
        a_ps2, _ = peak_shave(best_seq, t_sub, n_trials=3000)
        best_seq, best_score = evaluate_and_track(a_ps2, "peak_shave_2", best_seq, best_score)

    # === Phase 4: Multi-Length Exploration (15% time) ===
    print(f"\n=== Phase 4: Multi-Length Exploration (best={best_score:.6f}) ===")
    n_best = len(best_seq)
    rem = deadline - time.monotonic()

    alt_lengths = []
    for factor in [1.5, 2.0, 0.75, 1.25, 3.0]:
        alt_n = int(n_best * factor)
        if 256 <= alt_n <= 4096 and alt_n != n_best:
            if alt_n not in alt_lengths:
                alt_lengths.append(alt_n)

    for alt_n in alt_lengths[:4]:
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t_sub = min(rem * 0.35, 80.0)
        init_alt = interpolate_sequence(best_seq, alt_n)

        # Quick Adam at new length
        a_alt, _ = adam_logspace(init_alt, t_sub, T_init=0.04, T_final=0.002,
                                  lr=0.007, schedule='hold_anneal', mode='softmax')
        best_seq, best_score = evaluate_and_track(a_alt, f"alt_n{alt_n}", best_seq, best_score)
        print(f"  Alt n={alt_n}: {best_score:.6f}")

    # === Phase 5: Final Intensive Polish (remaining time) ===
    print(f"\n=== Phase 5: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    n_best = len(best_seq)

    if rem > 30:
        # Final Adam with very low temperature
        t_sub = min(rem * 0.3, 60.0)
        a_fin, _ = adam_logspace(best_seq, t_sub, T_init=0.01, T_final=0.0003,
                                  lr=0.003, schedule='exp', mode='softmax')
        best_seq, best_score = evaluate_and_track(a_fin, "final_adam_sm", best_seq, best_score)

    rem = deadline - time.monotonic()
    if rem > 20:
        # Penalty mode final
        t_sub = min(rem * 0.3, 40.0)
        a_fin2, _ = adam_logspace(best_seq, t_sub, T_init=0.02, T_final=0.001,
                                   lr=0.004, schedule='exp', mode='penalty')
        best_seq, best_score = evaluate_and_track(a_fin2, "final_adam_pen", best_seq, best_score)

    rem = deadline - time.monotonic()
    if rem > 15:
        # Final peak shave
        t_sub = min(rem * 0.4, 30.0)
        a_ps_fin, _ = peak_shave(best_seq, t_sub, n_trials=3000)
        best_seq, best_score = evaluate_and_track(a_ps_fin, "final_ps", best_seq, best_score)

    rem = deadline - time.monotonic()
    if rem > 10:
        # Final coordinate descent
        t_sub = min(rem * 0.5, 20.0)
        a_cd_fin, _ = coordinate_descent(best_seq, t_sub, n_sweeps=100)
        best_seq, best_score = evaluate_and_track(a_cd_fin, "final_cd", best_seq, best_score)

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.6f}")

    return [float(x) for x in best_seq]
