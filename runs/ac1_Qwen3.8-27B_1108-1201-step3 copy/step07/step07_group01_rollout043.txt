"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. Spectral magnitude compression (novel)
2. Adaptive mirror descent with temperature annealing
3. Pair-wise local search for peak contributions
4. Coordinate descent polishing
5. Multi-length exploration

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_from_a(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def spectral_shaping(a_init, time_limit, alpha_schedule=None, max_iters=10000, verbose=True, label=""):
    """
    Iteratively compress the magnitude spectrum to flatten the auto-convolution.
    At each step: FFT -> compress |A| -> IFFT -> clip to non-negative.
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    
    fft_size = 1 << (2 * n - 1).bit_length()
    
    if alpha_schedule is None:
        # Start with strong compression, gradually reduce
        alpha_schedule = lambda it, max_it: 1.5 * (0.8 ** (it / max_it)) + 0.0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute FFT
        A = fft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Compress magnitude: mag_new = mag^(1/alpha) for alpha > 1
        # This reduces peaks and boosts valleys
        alpha = alpha_schedule(it, max_iters)
        if alpha < 1.01:
            # Very mild compression, just renormalize
            mag_new = mag / mag.max() * mag.mean()
        else:
            mag_shifted = mag - mag.min() + 1e-10
            mag_new = mag_shifted ** (1.0 / alpha)
            # Scale to maintain energy
            mag_new = mag_new * (mag.mean() / mag_new.mean())
        
        # Reconstruct
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, fft_size))[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        sc = score_from_a(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
        
        # Accept/reject (simple: always accept for now, could add annealing)
        a = a_new
        
        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} Spectral iter={it} ({elapsed:.0f}s): alpha={alpha:.3f} score={sc:.7f} best={best_score:.7f}")
    
    return best_a, best_score


def adaptive_mirror_descent(a_init, time_limit, T_start=0.01, T_end=0.0003,
                            max_iters=50000, verbose=True, label=""):
    """
    Mirror descent with adaptive step size and temperature.
    Uses soft-max over the convolution peaks.
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 1e-15)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    
    eta = 0.02
    T = T_start
    stuck_count = 0
    momentum = np.zeros(n)
    mu = 0.9  # momentum coefficient
    
    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Anneal temperature
        progress = min(1.0, it / max_iters)
        T = T_start * (T_end / T_start) ** progress
        
        # Compute convolution and soft-max weights
        b = fast_convolve(a)
        current_score = 2.0 * n * b.max()
        
        if current_score < best_score - 1e-14:
            best_score = current_score
            best_a = a.copy()
            stuck_count = 0
        else:
            stuck_count += 1
        
        # Soft-max weights
        scaled = b / T
        m = scaled.max()
        exp_scaled = np.exp(scaled - m)
        w = exp_scaled / exp_scaled.sum()
        
        # Gradient: cross-correlation of w with a
        fft_size = 1 << (3 * n - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        correl = irfft(np.conj(W) * A, fft_size)
        grad = 2.0 * n * correl[:n]
        
        # Project gradient to simplex tangent space
        grad = grad - np.dot(a, grad)
        
        # Momentum
        momentum = mu * momentum + (1 - mu) * grad
        direction = momentum
        
        # Adaptive step size: scale by current "sharpness"
        grad_norm = np.linalg.norm(direction)
        if grad_norm > 1e-15:
            direction = direction / grad_norm
        
        # Step size schedule
        eta_base = 0.015 * (0.5 ** progress)
        # If stuck, increase step size
        if stuck_count > 500:
            eta = eta_base * 2.0
        else:
            eta = eta_base
        
        # Mirror (multiplicative) update
        a = a * np.exp(-eta * direction)
        a = np.maximum(a, 1e-300)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
        else:
            a = a / s
        
        # Stuck detection: larger perturbation
        if stuck_count > 2000:
            rng = np.random.default_rng(it * 31 + 7)
            # Perturb a random subset of coordinates
            frac = 0.3
            n_perturb = int(n * frac)
            idx = rng.choice(n, size=n_perturb, replace=False)
            a[idx] *= rng.uniform(0.5, 2.0, size=n_perturb)
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            stuck_count = 0
            mu = 0.5  # Reset momentum for a while
        
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} MD iter={it} ({elapsed:.0f}s): T={T:.5f} eta={eta:.5f} "
                  f"score={current_score:.7f} best={best_score:.7f} stuck={stuck_count}")
    
    return best_a, best_score


def pairwise_local_search(a_init, time_limit, max_rounds=5000, verbose=True, label=""):
    """
    For each peak position, find the contributing pairs and try to adjust them.
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 1e-15)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    
    for rnd in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        current_score = 2.0 * n * b.max()
        
        if current_score < best_score - 1e-14:
            best_score = current_score
            best_a = a.copy()
        
        # Find the global peak
        k_peak = int(np.argmax(b))
        peak_val = b[k_peak]
        
        # Find contributing pairs
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[k_peak - idx]
        
        # Get top contributors
        top_n = min(10, len(idx))
        top_local = np.argsort(contribs)[::-1][:top_n]
        
        improved = False
        for ci in top_local:
            i = int(idx[ci])
            j = k_peak - i
            if i == j:
                # a[i]^2 contributes; reduce a[i]
                for factor in [0.95, 0.9, 0.85, 0.8]:
                    a_trial = a.copy()
                    old_val = a_trial[i]
                    a_trial[i] = old_val * factor
                    diff = old_val - a_trial[i]
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    others = a_trial[mask].copy()
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        continue
                    # Distribute diff to reduce other peaks
                    # Proportional to (1 - a[k_peak - m]) to avoid increasing the peak
                    weights = np.maximum(1.0 - a[np.maximum(0, np.minimum(n-1, k_peak - np.arange(n)[mask]))], 0.1)
                    weights = weights / weights.sum()
                    a_trial[mask] = others + diff * weights
                    a_trial = np.maximum(a_trial, 0)
                    s = a_trial.sum()
                    if s < 1e-15:
                        continue
                    a_trial = a_trial / s
                    sc = score_from_a(a_trial)
                    if sc < best_score - 1e-13:
                        a = a_trial.copy()
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
            else:
                # a[i]*a[j] contributes; try to reduce one of them
                # Reduce the larger one (more effective)
                if a[i] >= a[j]:
                    target = i
                    other = j
                else:
                    target = j
                    other = i
                
                for factor in [0.95, 0.9, 0.85]:
                    a_trial = a.copy()
                    old_val = a_trial[target]
                    a_trial[target] = old_val * factor
                    diff = old_val - a_trial[target]
                    mask = np.ones(n, dtype=bool)
                    mask[target] = False
                    others = a_trial[mask].copy()
                    others_sum = others.sum()
                    if others_sum < 1e-15:
                        continue
                    a_trial[mask] = others + diff * others / others_sum
                    a_trial = np.maximum(a_trial, 0)
                    s = a_trial.sum()
                    if s < 1e-15:
                        continue
                    a_trial = a_trial / s
                    sc = score_from_a(a_trial)
                    if sc < best_score - 1e-13:
                        a = a_trial.copy()
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
        
        if not improved:
            # Try a random perturbation to escape
            if rnd % 100 == 0:
                rng = np.random.default_rng(rnd * 17 + 3)
                n_perturb = max(5, n // 20)
                idx_pert = rng.choice(n, size=n_perturb, replace=False)
                a[idx_pert] *= rng.uniform(0.7, 1.3, size=n_perturb)
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
            else:
                # No improvement for a while, break
                if rnd > 50 and not improved:
                    break
    
    return best_a, best_score


def multi_peak_mirror_descent(a_init, time_limit, num_peaks=5, T=0.5,
                               max_iters=20000, verbose=True, label=""):
    """
    Mirror descent targeting the top-K peaks simultaneously.
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 1e-15)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    eta = 0.01
    
    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        current_score = 2.0 * n * b.max()
        
        if current_score < best_score - 1e-14:
            best_score = current_score
            best_a = a.copy()
        
        # Top-K peaks
        if num_peaks >= len(b):
            w = np.ones(len(b)) / len(b)
        else:
            top_indices = np.argsort(b)[::-1][:num_peaks]
            w = np.zeros(len(b))
            top_vals = b[top_indices]
            w[top_indices] = top_vals / top_vals.sum()
        
        # Gradient
        fft_size = 1 << (3 * n - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        correl = irfft(np.conj(W) * A, fft_size)
        grad = 2.0 * n * correl[:n]
        
        # Project to simplex
        grad = grad - np.dot(a, grad)
        
        # Multiplicative update
        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 1e-300)
        a = a / a.sum()
        
        # Decay eta
        eta *= 0.9999
        
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} MultiPeak iter={it} ({elapsed:.0f}s): "
                  f"eta={eta:.6f} score={current_score:.7f} best={best_score:.7f}")
    
    return best_a, best_score


def coordinate_descent_polish(a_init, time_limit, verbose=True, label=""):

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    s = a.sum()
    if s < 1e-15:

    a = a / s

        if trial == 0:

The problem: minimize max_k conv(a,a)[k] subject to sum(a) = 1, a >= 0.

This is a non-convex optimization problem (the max makes it non-smooth, and the product a[i]*a[j] makes it non-convex).

Actually, sum a[i]*a[k-i] <= M is the constraint that a quadratic form is bounded. For a fixed k, this is:
a^T Q_k a <= M

This is a quadratic inequality, which defines the interior of an ellipsoid (if Q_k is positive definite) or a hyperboloid (if Q_k is indefinite). In general, this is non-convex.

So the feasibility problem is non-convex, and we can't solve it exactly with standard convex optimization.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Spectral magnitude compression (iterative FFT domain equalization)
2. Adaptive mirror descent with Nesterov momentum and simulated annealing restarts
3. Targeted peak contribution reduction via smart redistribution
4. Phase randomized spectral search for global exploration
5. Multi-phase refinement with coordinate descent polishing

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_from_a(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def compute_conv_grad_weights(a, b, T):
    """Compute soft-max weights over convolution values."""
    scaled = b / T
    m = scaled.max()
    exp_scaled = np.exp(scaled - m)
    w = exp_scaled / exp_scaled.sum()
    return w


def compute_cross_corr(w, a):
    """Compute cross-correlation (w * a) using FFT. Result has length >= n."""
    n = len(a)
    fft_size = 1 << (3 * n - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    correl = irfft(np.conj(W) * A, fft_size)
    return correl[:n]


def spectral_magnitude_compression(a_init, time_limit, max_iters=20000,
                                    alpha_start=2.0, alpha_end=1.05,
                                    verbose=True, label=""):
    """
    Iteratively compress the magnitude spectrum to flatten the auto-convolution.
    At each step: FFT -> compress |A| -> IFFT -> clip to non-negative -> normalize.
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s

    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, it / max_iters)
        alpha = alpha_start * (alpha_end / alpha_start) ** progress

        A = fft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)

        mag_min = mag.min()
        mag_shifted = mag - mag_min + 1e-10
        mag_new = mag_shifted ** (1.0 / alpha)
        mag_new = mag_new * (mag.mean() / mag_new.mean())
        # Preserve DC component
        mag_new[0] = mag[0]

        A_new = mag_new * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, fft_size))[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        sc = score_from_a(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()

        a = a_new

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} Spec iter={it} ({elapsed:.0f}s): alpha={alpha:.3f} "
                  f"score={sc:.7f} best={best_score:.7f}")

    return best_a, best_score


def mirror_descent_annealed(a_init, time_limit, T_start=0.01, T_end=0.0001,
                            max_iters=50000, momentum_decay=0.85,
                            verbose=True, label=""):
    """
    Mirror descent with Nesterov momentum, adaptive step size,
    and simulated annealing restarts.
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()

    momentum = np.zeros(n)
    mu = momentum_decay
    stuck_count = 0
    ann_T = 1.0  # Annealing temperature for accepting worse solutions
    rng = np.random.default_rng(777)

    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, it / max_iters)
        T = T_start * (T_end / T_start) ** progress
        ann_T = max(0.01, 1.0 * (0.999 ** it))

        b = fast_convolve(a)
        current_score = 2.0 * n * b.max()

        if current_score < best_score - 1e-14:
            best_score = current_score
            best_a = a.copy()
            stuck_count = 0
        else:
            stuck_count += 1

        # Soft-max weights
        w = compute_conv_grad_weights(a, b, T)

        # Gradient via cross-correlation
        grad = 2.0 * n * compute_cross_corr(w, a)

        # Project to simplex tangent space
        grad = grad - np.dot(a, grad)

        # Nesterov momentum
        momentum = mu * momentum + (1.0 - mu) * grad

        # Adaptive step size
        grad_norm = np.linalg.norm(momentum)
        if grad_norm < 1e-15:
            break
        direction = momentum / grad_norm

        eta_base = 0.02 * (1.0 - 0.5 * progress)
        if stuck_count > 1000:
            eta = eta_base * 1.5
        elif stuck_count > 500:
            eta = eta_base * 1.2
        else:
            eta = eta_base

        # Mirror update
        a_new = a * np.exp(-eta * direction)
        a_new = np.maximum(a_new, 1e-300)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a = best_a.copy()
        else:
            a_new = a_new / s_new
            sc_new = score_from_a(a_new)
            # Simulated annealing: accept worse solutions with some probability
            if sc_new > best_score:
                delta = sc_new - best_score
                if rng.random() < np.exp(-delta / (ann_T * 0.01)):
                    a = a_new
                else:
                    a = a_new if sc_new < current_score else a
            else:
                a = a_new

        # Restart logic
        if stuck_count > 3000:
            # Large perturbation
            frac = 0.2 + 0.3 * (stuck_count / 10000.0)
            n_perturb = max(10, int(n * frac))
            idx = rng.choice(n, size=n_perturb, replace=False)
            a[idx] *= rng.uniform(0.3, 3.0, size=n_perturb)
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            momentum = np.zeros(n)
            stuck_count = 0

        if verbose and it % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} MD iter={it} ({elapsed:.0f}s): T={T:.5f} "
                  f"score={current_score:.7f} best={best_score:.7f} stuck={stuck_count}")

    return best_a, best_score

def spectral_phase_search(a_init, time_limit, n_trials=2000, verbose=True, label=""):
    """
    Search over random/modulated phases while keeping magnitude spectrum.
    Each trial: random phase -> IFFT -> clip to non-negative -> evaluate.
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s

    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(555)

    fft_size = 1 << (2 * n - 1).bit_length()
    A = fft(a, fft_size)
    mag = np.abs(A).copy()
    orig_phase = np.angle(A).copy()

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break

        if trial == 0:
            phase = orig_phase.copy()
        elif trial < 20:
            # Small random perturbation to original phase
            noise = rng.standard_normal(fft_size) * 0.1
            phase = orig_phase + noise
        else:
            # Blend of original and random phase
            frac_random = min(0.8, 0.3 + 0.5 * (trial / n_trials))
            random_phase = rng.uniform(-np.pi, np.pi, size=fft_size)
            phase = (1 - frac_random) * orig_phase + frac_random * random_phase

        A_new = mag * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, fft_size))[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            continue
        a_new = a_new / s_new

        sc = score_from_a(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            # Update target spectrum from improved solution
            A = fft(a_new, fft_size)
            mag = np.abs(A).copy()
            orig_phase = np.angle(A).copy()

        if verbose and trial % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} Phase trial={trial} ({elapsed:.0f}s): "
                  f"score={sc:.7f} best={best_score:.7f}")

    return best_a, best_score

        other_coords = np.setdiff1d(np.arange(n), peak_coords)
        rng.shuffle(other_coords)
        order = np.concatenate([peak_coords, other_coords])

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break

            for f in factors:
                old_val = a[i]
                new_val = old_val * f
                if new_val < 0:
                    continue
                diff = new_val - old_val
                if abs(diff) < 1e-16:
                    continue

                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a[mask].copy()
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue

                a_trial = a.copy()
                a_trial[i] = new_val
                a_trial[mask] = others - diff * others / others_sum
                if a_trial.min() < -1e-12:
                    continue
                a_trial = np.maximum(a_trial, 0)
                s_trial = a_trial.sum()
                if s_trial < 1e-15:
                    continue
                a_trial = a_trial / s_trial

                sc = score_from_a(a_trial)
                if sc < best_score - 1e-13:
                    best_score = sc
                    a = a_trial.copy()
                    best_a = a.copy()
                    improved_pass = True
                    break

        if not improved_pass:
            no_improve_rounds += 1

    return best_a, best_score


def multi_temperature_sweep(a_init, time_limit, verbose=True, label=""):
    """Run mirror descent at multiple temperatures and keep the best."""
    n = len(a_init)
    best_overall = a_init.copy()
    best_score = score_from_a(a_init)
    t0 = time.monotonic()

    temperatures = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0003]
    n_temps = len(temperatures)
    per_temp_budget = (time_limit - 5) / n_temps

    for T in temperatures:
        rem = deadline_func(time_limit, t0)
        if rem < 10:
            break
        t_alloc = min(per_temp_budget, rem - 5)
        if t_alloc < 5:
            break

        a_res, sc_res = mirror_descent_annealed(
            best_overall, t_alloc, T_start=T, T_end=T * 0.05,
            label=f"{label}_T{T}"
        )
        if sc_res < best_score - 1e-12:
            best_score = sc_res
            best_overall = a_res.copy()

    return best_overall, best_score


def deadline_func(budget, t0):
    """Safe remaining time."""
    return max(0.0, budget - (time.monotonic() - t0))


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, lbl, best_seq, best_score):
    """Evaluate a candidate using evaluate_sequence."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{lbl}]: {score:.7f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase search for the optimal sequence:
    Phase 1: Spectral magnitude compression
    Phase 2: Mirror descent with momentum (multiple temperature configs)
    Phase 3: Targeted peak reduction
    Phase 4: Spectral phase search (global exploration)
    Phase 5: Coordinate descent polishing
    Phase 6: Refinement rounds
    Phase 7: Length variation
    Phase 8: Final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.7f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = p_score
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default init: n={len(best_seq)}, score={best_score:.7f}")

    # === Phase 1: Spectral Magnitude Compression ===
    print(f"\n=== Phase 1: Spectral Compression (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        t_alloc = min(rem * 0.15, 120.0)
        print(f"  n={len(best_seq)}, budget={t_alloc:.0f}s")
        a_spec, _ = spectral_magnitude_compression(best_seq, t_alloc, label="spec1")
        best_seq, best_score = check_and_update(a_spec, "spec1", best_seq, best_score)

    # === Phase 2: Mirror Descent (multiple configs) ===
    print(f"\n=== Phase 2: Mirror Descent (best={best_score:.7f}) ===")
    configs = [
        (0.015, 0.0001, "MD_h"),
        (0.008, 0.00005, "MD_m"),
        (0.003, 0.00001, "MD_l"),
    ]
    for T_s, T_e, lbl in configs:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        t_alloc = min(rem * 0.25, 200.0)
        print(f"  {lbl}: T=[{T_s},{T_e}], budget={t_alloc:.0f}s")
        a_md, _ = mirror_descent_annealed(best_seq, t_alloc, T_start=T_s, T_end=T_e, label=lbl)
        best_seq, best_score = check_and_update(a_md, lbl, best_seq, best_score)

        # Follow up with spectral compression
        rem = deadline - time.monotonic()
        if rem > 60:
            a_spec2, _ = spectral_magnitude_compression(
                best_seq, min(rem * 0.15, 60.0), label=f"spec_{lbl}"
            )
            best_seq, best_score = check_and_update(a_spec2, f"spec_{lbl}", best_seq, best_score)

    # === Phase 3: Targeted Peak Reduction ===
    print(f"\n=== Phase 3: Targeted Peak Reduction (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t_alloc = min(rem * 0.2, 150.0)
        print(f"  budget={t_alloc:.0f}s")
        a_tpr, _ = targeted_peak_reduction(best_seq, t_alloc, label="tpr")
        best_seq, best_score = check_and_update(a_tpr, "tpr", best_seq, best_score)

        # Mirror descent after TPR
        rem = deadline - time.monotonic()
        if rem > 40:
            a_md2, _ = mirror_descent_annealed(
                best_seq, min(rem * 0.3, 80.0), T_start=0.005, T_end=0.00005, label="MD_after_tpr"
            )
            best_seq, best_score = check_and_update(a_md2, "MD_after_tpr", best_seq, best_score)

    # === Phase 4: Spectral Phase Search ===
    print(f"\n=== Phase 4: Spectral Phase Search (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    if rem > 50:
        t_alloc = min(rem * 0.2, 80.0)
        print(f"  budget={t_alloc:.0f}s")
        a_ps, _ = spectral_phase_search(best_seq, t_alloc, n_trials=3000, label="phase_search")
        best_seq, best_score = check_and_update(a_ps, "phase_search", best_seq, best_score)

        # MD after phase search
        rem = deadline - time.monotonic()
        if rem > 30:
            a_md3, _ = mirror_descent_annealed(
                best_seq, min(rem * 0.4, 60.0), T_start=0.003, T_end=0.00002, label="MD_after_ps"
            )
            best_seq, best_score = check_and_update(a_md3, "MD_after_ps", best_seq, best_score)

    # === Phase 5: Coordinate Descent ===
    print(f"\n=== Phase 5: Coordinate Descent (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_alloc = min(rem * 0.2, 80.0)
        print(f"  budget={t_alloc:.0f}s")
        a_cd, _ = coordinate_descent_fast(best_seq, t_alloc, label="cd1")
        best_seq, best_score = check_and_update(a_cd, "cd1", best_seq, best_score)

    # === Phase 6: Refinement Round ===
    print(f"\n=== Phase 6: Refinement (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        # More MD
        t_alloc = min(rem * 0.25, 100.0)
        a_md4, _ = mirror_descent_annealed(
            best_seq, t_alloc, T_start=0.002, T_end=0.00001, label="MD_refine"
        )
        best_seq, best_score = check_and_update(a_md4, "MD_refine", best_seq, best_score)

        # Spectral
        rem = deadline - time.monotonic()
        if rem > 30:
            a_spec3, _ = spectral_magnitude_compression(
                best_seq, min(rem * 0.2, 50.0), alpha_start=1.5, alpha_end=1.02, label="spec_refine"
            )
            best_seq, best_score = check_and_update(a_spec3, "spec_refine", best_seq, best_score)

        # TPR
        rem = deadline - time.monotonic()
        if rem > 30:
            a_tpr2, _ = targeted_peak_reduction(
                best_seq, min(rem * 0.25, 60.0), label="tpr2"
            )
            best_seq, best_score = check_and_update(a_tpr2, "tpr2", best_seq, best_score)

    # === Phase 7: Length Variation ===
    print(f"\n=== Phase 7: Length Variation (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    if rem > 80:
        n_best = len(best_seq)
        alt_lengths = []
        for factor in [1.33, 1.5, 0.75, 2.0]:
            alt_n = int(n_best * factor)
            if 256 <= alt_n <= 4096 and alt_n != n_best and alt_n not in alt_lengths:
                alt_lengths.append(alt_n)

        for alt_n in alt_lengths[:2]:
            rem = deadline - time.monotonic()
            if rem < 50:
                break
            t_alloc = min(rem * 0.3, 80.0)
            init_alt = interpolate_sequence(best_seq, alt_n)
            a_alt, _ = mirror_descent_annealed(
                init_alt, t_alloc, T_start=0.01, T_end=0.0005, label=f"alt_{alt_n}"
            )
            best_seq, best_score = check_and_update(a_alt, f"alt_{alt_n}", best_seq, best_score)

    # === Phase 8: Final Polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_alloc = min(rem * 0.4, 30.0)
        a_fin, _ = coordinate_descent_fast(best_seq, t_alloc, label="final_cd")
        best_seq, best_score = check_and_update(a_fin, "final_cd", best_seq, best_score)

    rem = deadline - time.monotonic()
    if rem > 10:
        a_fin2, _ = spectral_magnitude_compression(
            best_seq, min(rem * 0.5, 15.0), alpha_start=1.3, alpha_end=1.01, label="final_spec"
        )
        best_seq, best_score = check_and_update(a_fin2, "final_spec", best_seq, best_score)

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.7f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
