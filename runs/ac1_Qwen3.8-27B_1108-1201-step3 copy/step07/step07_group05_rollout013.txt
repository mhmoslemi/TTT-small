```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based fictitious play
combined with gradient polishing. The LP step finds sequences that
minimize interference with the current solution's peaks, driving the
autocorrelation towards flatness. Polishing with L-BFGS in log-space
finishes the optimization.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score_normalized(a):
    """Compute score for a sequence (will be normalized)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    conv = fast_convolve(a_norm)
    return 2.0 * n * conv.max()

def solve_lp(f, M):
    """
    Solve LP: maximize sum(g) s.t. (f*g)(k) <= M for all k, g >= 0.
    Returns normalized g (sum=1) or None if infeasible.
    """
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    
    # Build constraint matrix A where row k gives (f*g)(k)
    # Use sparse for efficiency
    from scipy.sparse import csr_matrix
    
    rows = []
    cols = []
    vals = []
    
    for k in range(2 * n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            j = k - i
            if 0 <= j < n:
                rows.append(k)
                cols.append(j)
                vals.append(f[i])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    constraints = [A @ g <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    
    try:
        # Try multiple solvers
        for solver in [cp.ECOS, cp.SCS, cp.CLARABEL]:
            try:
                prob.solve(solver=solver)
                if g.value is not None and prob.status in ['optimal', 'optimal_inaccurate']:
                    g_val = np.maximum(np.asarray(g.value).flatten(), 0)
                    if g_val.sum() > 1e-10:
                        return g_val / g_val.sum()
            except:
                continue
    except:
        pass
    
    return None

def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-30)
    return np.log(a_safe)

def smoothed_objective(z, T, n):
    """Soft-max smoothed objective in log-space."""
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    log_sum_exp = max_shifted + np.log(np.exp(shifted - max_shifted).sum())
    return (2.0 * n / T) * np.exp(T * log_sum_exp)

def smoothed_gradient(z, T, n):
    """Gradient of the soft-max smoothed objective in log-space."""
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    w = np.exp(shifted - max_shifted)
    w = w / w.sum()
    # Gradient w.r.t. a: 2 * cross-correlation of w with a
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
    # Project to log-space
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return (2.0 * n / T) * grad_z

def lbgfs_optimize(a0, time_limit, T, maxiter=5000):
    """L-BFGS optimization in log-space with soft-max smoothing."""
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score_normalized(a0)

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        return smoothed_objective(z, T, n)

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        return smoothed_gradient(z, T, n)

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 50, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
        score = compute_score_normalized(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except:
        pass

    return best_a, best_score, time.monotonic() - t0

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based fictitious play
combined with multi-temperature L-BFGS polishing, gradient descent, and
coordinate descent. The LP step (solved via HiGHS) finds sequences that
minimize interference with the current solution's autocorrelation peaks,
driving the sequence towards a flat autocorrelation profile. Multiple
sequence lengths are explored to avoid local minima.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score_normalized(a):
    """Compute score 2*n*max(conv) for a sequence with sum=1."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    conv = fast_convolve(a_norm)
    return 2.0 * n * conv.max()


def solve_lp_fictitious(f, M):
    """
    Solve LP: maximize sum(g) s.t. (f*g)(k) <= M for all k, g >= 0.
    Returns normalized g (sum=1) or None.
    """
    n = len(f)
    c = -np.ones(n)

    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            j = k - i
            if 0 <= j < n:
                rows.append(k)
                cols.append(j)
                vals.append(f[i])

    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.success and res.x is not None:
            g_val = np.maximum(np.asarray(res.x).flatten(), 0)
            if g_val.sum() > 1e-10:
                return g_val / g_val.sum()
    except Exception:
        pass
    return None


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-30)
    return np.log(a_safe)


def smoothed_objective(z, T, n):
    """Soft-max smoothed objective in log-space."""
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    log_sum_exp = max_shifted + np.log(np.exp(shifted - max_shifted).sum())
    return (2.0 * n / T) * np.exp(T * log_sum_exp)


def smoothed_gradient(z, T, n):
    """Gradient of the soft-max smoothed objective in log-space."""
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    w = np.exp(shifted - max_shifted)
    w = w / w.sum()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return (2.0 * n / T) * grad_z


def lbgfs_optimize(a0, time_limit, T, maxiter=5000):
    """L-BFGS optimization in log-space with soft-max smoothing."""
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score_normalized(a0)

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        return smoothed_objective(z, T, n)

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        return smoothed_gradient(z, T, n)

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 50, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
        score = compute_score_normalized(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except Exception:
        pass

    return best_a, best_score


def momentum_gradient_descent(a0, time_limit, T=500, lr_init=0.005, momentum=0.92):
    """Gradient descent with momentum in log-space."""
    n = len(a0)
    z = z_from_a(a0)
    v = np.zeros(n)
    best_a = a0.copy()
    best_score = compute_score_normalized(a0)
    t0 = time.monotonic()
    lr = lr_init
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        z_la = z - momentum * v
        a_la = softmax_from_z(z_la)
        grad = smoothed_gradient(z_la, T, n)
        v = momentum * v + lr * grad
        z_new = np.clip(z - v, -25, 25)
        a_new = softmax_from_z(z_new)
        score = compute_score_normalized(a_new)
        if score < best_score:
            best_a = a_new.copy()
            best_score = score
        z = z_new
        if iters % 1000 == 0:
            lr *= 0.9

    return best_a, best_score


def coordinate_descent(a0, time_limit, n_pairs=500):
    """Coordinate descent with pairwise transfers."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score_normalized(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    stagnant = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or a[i] < 1e-14:
                continue
            base = min(a[i], a[j]) if a[j] > 1e-14 else a[i]
            best_t = 0.0
            best_local = best_score
            for frac in [0.05, 0.1, 0.2, 0.4, 0.6, 0.8]:
                t = frac * base
                if t >= a[i] - 1e-15:
                    t = a[i] * 0.5
                a_trial = a.copy()
                a_trial[i] -= t
                a_trial[j] += t
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score_normalized(a_trial)
                if sc < best_local - 1e-15:
                    best_local = sc
                    best_t = t
                if a[j] > 1e-14:
                    t2 = frac * a[j]
                    a_trial2 = a.copy()
                    a_trial2[j] -= t2
                    a_trial2[i] += t2
                    a_trial2 = np.maximum(a_trial2, 0)
                    ss2 = a_trial2.sum()
                    if ss2 < 1e-15:
                        continue
                    a_trial2 = a_trial2 / ss2
                    sc2 = compute_score_normalized(a_trial2)
                    if sc2 < best_local - 1e-15:
                        best_local = sc2
                        best_t = -t2
            if best_t != 0.0:
                a[i] -= best_t
                a[j] += best_t
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = best_local
                best_a = a.copy()
                improved = True
        if not improved:
            stagnant += 1
            if stagnant > 3:
                noise = rng.standard_normal(n) * 0.003
                a = best_a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                stagnant = 0
        else:
            stagnant = 0

    return best_a, best_score


def peak_flatten(a0, time_limit, n_top=20):
    """Targeted peak equalization: reduce the highest autocorrelation peaks."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        top_idx = np.argpartition(conv, -n_top)[-n_top:]
        max_peak = conv.max()
        min_top = conv[top_idx].min()

        if max_peak - min_top < 1e-9:
            break

        max_k = int(np.argmax(conv))
        i_start = max(0, max_k - n + 1)
        i_end = min(n - 1, max_k)
        idx = np.arange(i_start, i_end + 1)
        contribs = a[idx] * a[max_k - idx]
        top_c = np.argsort(contribs)[::-1][:8]

        improved = False
        for ci in top_c:
            i = idx[ci]
            if a[i] < 1e-14:
                continue
            for frac in [0.05, 0.1, 0.15, 0.25]:
                delta = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= delta
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                os = others.sum()
                if os < 1e-15:
                    continue
                a_trial[mask] = others + delta * others / os
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = compute_score_normalized(a_trial)
                if sc < best_score - 1e-15:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Random perturbation on peak-contributing positions
            for _ in range(5):
                p = idx[rng.integers(0, len(idx))]
                a_trial = a.copy()
                delta = a[p] * 0.08 * rng.standard_normal()
                a_trial[p] -= delta
                mask2 = np.ones(n, dtype=bool)
                mask2[p] = False
                oth2 = a_trial[mask2]
                o2s = oth2.sum()
                if o2s > 1e-15:
                    a_trial[mask2] = oth2 + delta * oth2 / o2s
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = compute_score_normalized(a_trial)
                if sc < best_score - 1e-15:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if not improved and iters > 50:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving non-negativity and sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def lp_fictitious_play(f0, time_budget):
    """Run LP-based fictitious play iteration."""
    n = len(f0)
    f = f0.copy()
    best_a = f0.copy()
    best_score = compute_score_normalized(f0)
    t0 = time.monotonic()
    iters = 0
    no_improve = 0

    while time.monotonic() - t0 < time_budget:
        iters += 1
        conv = fast_convolve(f)
        M = conv.max()

        g = solve_lp_fictitious(f, M)
        if g is None:
            no_improve += 1
            if no_improve > 10:
                break
            continue

        no_improve = 0

        # Check if g is too similar to f (stagnation)
        cos_sim = np.dot(f, g) / (np.linalg.norm(f) * np.linalg.norm(g) + 1e-30)
        if cos_sim > 0.9999:
            no_improve += 3
            if no_improve > 15:
                break

        # Adaptive step size
        progress = min(1.0, (time.monotonic() - t0) / time_budget)
        t_step = max(0.01, 0.4 * (1.0 - progress) ** 1.5)

        f_new = (1 - t_step) * f + t_step * g
        f_new = np.maximum(f_new, 0)
        ss = f_new.sum()
        if ss < 1e-15:
            break
        f_new = f_new / ss

        score = compute_score_normalized(f_new)
        if score < best_score:
            best_score = score
            best_a = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                break

        f = f_new
        if iters % 100 == 0:
            print(f"    LP iter={iters}, score={score:.8f}, best={best_score:.8f}")

    return best_a, best_score, iters


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-strategy optimization:
    1. LP fictitious play (primary novel approach)
    2. L-BFGS with soft-max smoothing (multiple temperatures)
    3. Momentum gradient descent
    4. Peak flattening
    5. Coordinate descent
    6. Multi-length exploration
    7. Final polishing
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 768
        parent = np.ones(n_init) / n_init
        t = np.linspace(0, 2 * np.pi * 5, n_init)
        parent = parent + 0.03 * np.cos(t) + 0.02 * np.sin(2 * t)
        parent = np.maximum(parent, 0)
        parent = parent / parent.sum()
        best_seq = parent.copy()
        best_score = compute_score_normalized(parent)
        print(f"Initialized: n={n_init}, score={best_score:.8f}")

    n_parent = len(parent)

    # === Phase 1: LP Fictitious Play (primary) ===
    print(f"\n=== Phase 1: LP Fictitious Play (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        lp_budget = min(rem * 0.3, 400.0)
        print(f"  Budget: {lp_budget:.0f}s, n={n_parent}")
        a_lp, sc_lp, iters = lp_fictitious_play(best_seq, lp_budget)
        if sc_lp < best_score:
            best_score = sc_lp
            best_seq = a_lp.copy()
            print(f"  LP improved to {best_score:.8f} ({iters} iters)")
        else:
            print(f"  LP: no improvement ({iters} iters), best={best_score:.8f}")

    # === Phase 2: L-BFGS Multi-Temperature ===
    print(f"\n=== Phase 2: L-BFGS Polish (best={best_score:.8f}) ===")
    for T_val in [150, 400, 1500, 5000]:
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t_alloc = min(rem * 0.15, 100.0)
        if t_alloc < 15:
            continue
        print(f"  T={T_val}, n={n_parent}, budget={t_alloc:.0f}s")
        a_out, sc = lbgfs_optimize(best_seq, t_alloc, T=T_val, maxiter=5000)
        if sc < best_score:
            best_score = sc
            best_seq = a_out.copy()
            print(f"    improved to {best_score:.8f}")

    # === Phase 3: Momentum Gradient Descent ===
    print(f"\n=== Phase 3: Gradient Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t_alloc = min(rem * 0.12, 80.0)
        if t_alloc >= 10:
            print(f"  budget={t_alloc:.0f}s")
            a_out, sc = momentum_gradient_descent(best_seq, t_alloc, T=300, lr_init=0.003, momentum=0.93)
            if sc < best_score:
                best_score = sc
                best_seq = a_out.copy()
                print(f"    improved to {best_score:.8f}")

    # === Phase 4: Second LP Round ===
    print(f"\n=== Phase 4: LP Round 2 (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 50:
        lp_budget2 = min(rem * 0.2, 200.0)
        print(f"  Budget: {lp_budget2:.0f}s")
        a_lp2, sc_lp2, iters2 = lp_fictitious_play(best_seq, lp_budget2)
        if sc_lp2 < best_score:
            best_score = sc_lp2
            best_seq = a_lp2.copy()
            print(f"  LP2 improved to {best_score:.8f} ({iters2} iters)")
        else:
            print(f"  LP2: no improvement ({iters2} iters)")

    # === Phase 5: Peak Flattening ===
    print(f"\n=== Phase 5: Peak Flattening (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_alloc = min(rem * 0.15, 100.0)
        if t_alloc >= 10:
            print(f"  budget={t_alloc:.0f}s")
            a_pf, sc_pf = peak_flatten(best_seq, t_alloc, n_top=25)
            if sc_pf < best_score:
                best_score = sc_pf
                best_seq = a_pf.copy()
                print(f"    improved to {best_score:.8f}")

    # === Phase 6: Coordinate Descent ===
    print(f"\n=== Phase 6: Coordinate Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_alloc = min(rem * 0.15, 100.0)
        if t_alloc >= 10:
            print(f"  budget={t_alloc:.0f}s")
            a_cd, sc_cd = coordinate_descent(best_seq, t_alloc, n_pairs=600)
            if sc_cd < best_score:
                best_score = sc_cd
                best_seq = a_cd.copy()
                print(f"    improved to {best_score:.8f}")

    # === Phase 7: Multi-Length Exploration ===
    print(f"\n=== Phase 7: Length Exploration (best={best_score:.8f}) ===")
    candidate_ns = [n_parent + 1, n_parent - 1, n_parent + 2, n_parent - 2,
                    n_parent + 4, n_parent - 4, n_parent + 8, n_parent - 8]
    tried_n = set()
    for alt_n in candidate_ns:
        if alt_n < 256 or alt_n > 2048 or alt_n in tried_n:
            continue
        tried_n.add(alt_n)
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        t_alloc = min(rem * 0.15, 80.0)
        if t_alloc < 20:
            continue
        init_alt = interpolate_sequence(best_seq, alt_n)
        print(f"  n={alt_n}, budget={t_alloc:.0f}s")
        a_out, sc = lbgfs_optimize(init_alt, t_alloc, T=500, maxiter=4000)
        # Verify with actual evaluator
        eval_sc = float(evaluate_sequence(a_out.tolist()))
        if np.isfinite(eval_sc) and eval_sc < best_score:
            best_score = eval_sc
            best_seq = a_out.copy()
            n_parent = alt_n
            print(f"    improved to {best_score:.8f} at n={alt_n}")

    # === Phase 8: Random Restart + Polish ===
    print(f"\n=== Phase 8: Random Restart (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        n_restarts = min(3, int(rem // 80))
        for r in range(n_restarts):
            rem = deadline - time.monotonic()
            if rem < 50:
                break
            # Create perturbed version
            noise_scale = [0.01, 0.02, 0.03][r % 3]
            a_pert = best_seq + rng.standard_normal(n_parent) * noise_scale
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()

            # Quick LP + L-BFGS
            t_lp = min(rem * 0.3, 60.0)
            t_lb = min(rem * 0.4, 80.0)
            if t_lp > 15:
                a_r, sc_r, _ = lp_fictitious_play(a_pert, t_lp)
            else:
                a_r, sc_r = a_pert, compute_score_normalized(a_pert)
            rem2 = deadline - time.monotonic()
            if rem2 > 20 and t_lb > 10:
                a_r2, sc_r2 = lbgfs_optimize(a_r, min(t_lb, rem2 - 10), T=500, maxiter=3000)
                if sc_r2 < sc_r:
                    a_r, sc_r = a_r2, sc_r2

            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
                print(f"  Restart {r}: improved to {best_score:.8f}")
            else:
                print(f"  Restart {r}: no improvement (got {sc_r:.8f})")

    # === Phase 9: Final L-BFGS + Peak Flatten ===
    print(f"\n=== Phase 9: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_alloc = min(rem * 0.4, 50.0)
        a_out, sc = lbgfs_optimize(best_seq, t_alloc, T=1000, maxiter=5000)
        if sc < best_score:
            best_score = sc
            best_seq = a_out.copy()

    rem = deadline - time.monotonic()
    if rem > 15:
        t_alloc = min(rem * 0.4, 30.0)
        a_pf, sc_pf = peak_flatten(best_seq, t_alloc, n_top=30)
        if sc_pf < best_score:
            best_score = sc_pf
            best_seq = a_pf.copy()

    # Final verification with the actual evaluator
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
