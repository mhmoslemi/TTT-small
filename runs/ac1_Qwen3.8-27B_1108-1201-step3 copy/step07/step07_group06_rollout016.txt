def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, n=fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def softmax_from_z(z):
    """Stable softmax."""
    z_shifted = z - np.max(z)
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return exp_z / s


def z_from_a(a):
    """Log of a (for log-space parameterization)."""

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase strategy:
(1) Soft-max gradient descent in log-space with Nesterov-accelerated Adam,
(2) Aggressive peak-targeted local search with pair moves and coordinate descent,
(3) Multi-length exploration with interpolation,
(4) Final fine-tuning with very low temperature and targeted shaving.
Key improvements over prior approach: better temperature schedule, pair-based
local moves, Nesterov momentum, more aggressive exploration of sequence lengths.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, n=fft_size)
    return irfft(A * A, n=fft_size)[:size]


def softmax_stable(z):
    z_shifted = z - np.max(z)
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return exp_z / s


def log_of_a(a):
    return np.log(np.maximum(a, 1e-300))


def cross_corr_w_a(w, a, fft_size):
    """Compute c[j] = sum_k w[k]*a[k-j] for j=0..n-1 using FFT."""
    n = len(a)
    W = rfft(w, n=fft_size)
    A = rfft(a, n=fft_size)
    # (w cross-correlate a)[j] = sum_k w[k]*a[k-j]
    # FFT: ifft(conj(FFT(w)) * FFT(a))
    corr = irfft(np.conj(W) * A, n=fft_size)[:n]
    return corr


def compute_grad_z(a, T, n, fft_size):
    """Compute gradient of soft-max objective w.r.t. z (log-space)."""
    B = fast_convolve(a)
    max_B = B.max()
    exp_B = np.exp(B / T - max_B / T)
    w = exp_B / exp_B.sum()
    c = cross_corr_w_a(w, a, fft_size)
    W = np.dot(a, c)
    grad_z = 4.0 * n * a * (c - W)
    return grad_z, B.max()


def score_normalized(a):
    """Score assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    if a.sum() < 1e-15:
        return np.inf
    a = a / a.sum()
    return 2.0 * n * fast_convolve(a).max()


def adam_nesterov(a0, time_limit, T_init=0.03, T_final=0.001, lr=0.01,
                  beta1=0.9, beta2=0.999, mu=0.9, eps=1e-8,
                  schedule='hold_anneal', restart_interval=4000):
    """
    Adam with Nesterov momentum in log-space.
    Returns (best_a, best_score).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    a = a / a.sum()
    z = log_of_a(a)
    m = np.zeros(n)
    v = np.zeros(n)
    z_prev = z.copy()
    t = 0
    fft_size = 1 << (2 * n - 2).bit_length()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)

        if schedule == 'linear':
            T = T_init + (T_final - T_init) * progress
        elif schedule == 'exp':
            T = T_init * (T_final / T_init) ** progress
        elif schedule == 'hold_anneal':
            if progress < 0.4:
                T = T_init
            else:
                p2 = (progress - 0.4) / 0.6
                T = T_init * (T_final / T_init) ** (p2 * 1.5)
        elif schedule == 'slow_anneal':
            if progress < 0.6:
                T = T_init
            else:
                p2 = (progress - 0.6) / 0.4
                T = T_init * (T_final / T_init) ** (p2 * 2.0)
        else:
            T = T_init * (T_final / T_init) ** progress

        a_curr = softmax_stable(z)

        # Nesterov: evaluate gradient at z + mu*(z - z_prev)
        z_nest = z + mu * (z - z_prev)
        a_nest = softmax_stable(z_nest)
        grad_z, _ = compute_grad_z(a_nest, T, n, fft_size)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        z_prev = z.copy()
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Evaluate
        a_new = softmax_stable(z)
        conv_new = fast_convolve(a_new)
        score = 2.0 * n * conv_new.max()

        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive restart
        if no_improve > restart_interval:
            z = log_of_a(best_a) + np.random.standard_normal(n) * 0.01
            m = np.zeros(n)
            v = np.zeros(n)
            z_prev = z.copy()
            no_improve = 0

        # Slow LR decay
        lr = max(0.0005, lr * 0.99995)

    return best_a, best_score


def peak_shave_aggressive(a0, time_limit, n_trials=10000):
    """
    Aggressive peak shaving: identify peak, try pair moves and single moves
    to reduce it. More aggressive than previous version.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break

        # Find peak (avoid edges sometimes)
        B_search = B.copy()
        peak_k = int(np.argmax(B_search))
        peak_val = B[peak_k]
        target = best_score / (2.0 * n)

            found = False
            for pk in B_sorted[:5]:
                pk = int(pk)
                if B[pk] <= target * 1.0005:
                    break
                peak_k = pk
                peak_val = B[pk]
                found = True
                break
            if not found:
                # Random perturbation to escape

                    a = best_a + rng.standard_normal(n) * 0.003
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                continue

        # Get contributing pairs
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_count = min(8, len(idx))
        top_idx = np.argsort(contribs)[::-1][:top_count]

        for ci in top_idx:

                    a_trial[i] *= f1
                    a_trial[j] *= f2
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a_trial.copy()
                        improved = True
                        break

                    break
            if improved:
                break

        if not improved:

                    continue

                    a_trial = a.copy()
                    diff = a_trial[i] * (1.0 - factor)
                    a_trial[i] *= factor
                    # Redistribute to random non-peak elements
                    mask = np.zeros(n, dtype=bool)
                    mask[:] = True
                    # Avoid elements that contribute to the peak
                    mask[i] = False
                    j_comp = peak_k - i
                    if 0 <= j_comp < n:
                        mask[j_comp] = False
                    others = np.where(mask)[0]
                    if len(others) == 0:
                        continue
                    # Distribute proportionally
                    a_trial[others] += diff * a_trial[others] / a_trial[others].sum()
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a_trial.copy()
                        improved = True
                        break
                if improved:
                    break

        # Try reducing a peak-contributing element and adding to a low element
        if not improved:
            # Find a low element far from the peak
            low_positions = np.argsort(B)[:n // 4]
            for ci in top_idx[:3]:
                i = int(idx[ci])
                if a[i] < 1e-15:
                    continue
                for lp in low_positions[:3]:
                    lp = int(lp)
                    if lp == i or a[lp] > 0.5 * a[i]:
                        continue
                    for factor in [0.80, 0.70, 0.60]:
                        a_trial = a.copy()
                        diff = a_trial[i] * (1.0 - factor)
                        a_trial[i] *= factor
                        # Add to low element (spread to neighbors too)

                        lo = max(0, lp - add_range)
                        hi = min(n, lp + add_range + 1)
                        a_trial[lo:hi] += diff * 0.5
                        # Redistribute remaining
                        ss = a_trial.sum()
                        a_trial = a_trial / ss
                        a_trial = np.maximum(a_trial, 0)
                        ss = a_trial.sum()
                        if ss < 1e-15:
                            continue
                        a_trial = a_trial / ss
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a_trial.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        # Stuck: random restart
        if not improved and trial > 200 and trial % 200 == 0:
            a = best_a + rng.standard_normal(n) * (0.01 + 0.01 * (trial // 1000))
            a = np.maximum(a, 0)
            a = a / a.sum()

    return best_a, best_score


def coordinate_descent_fast(a0, time_limit, n_sweeps=100):
    """Fast coordinate descent with multiplicative updates."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        order = rng.permutation(n)
        sweep_improved = False

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-15:
                continue

            best_local = a[i]
            best_local_score = best_score

            for factor in [0.75, 0.85, 0.95, 1.05, 1.15, 1.25]:
                a_trial = a.copy()
                a_trial[i] = a[i] * factor
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_local_score:
                    best_local_score = score
                    best_local = a[i] * factor

            if abs(best_local - a[i]) > 1e-15:
                a[i] = best_local
                a = a / a.sum()
                if best_local_score < best_score:
                    best_score = best_local_score
                    best_a = a.copy()
                    sweep_improved = True

        if not sweep_improved:
            break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_and_track(a, label, best_seq, best_score, verbose=True):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def lbgfs_polish(a0, time_limit, T=50):
    """L-BFGS polishing with soft-max."""
    n = len(a0)
    z0 = log_of_a(a0)
    fft_size = 1 << (2 * n - 2).bit_length()
    t0 = time.monotonic()
    stopped = [False]

    def obj(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        a = softmax_stable(z)
        B = fast_convolve(a)
        max_B = B.max()
        exp_B = np.exp(B / T - max_B / T)
        return 2.0 * n * T * (max_B / T + np.log(exp_B.sum()))

    def grad(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return np.zeros_like(z)
        a = softmax_stable(z)
        return compute_grad_z(a, T, n, fft_size)[0]

    try:
        res = minimize(fun=obj, x0=z0, jac=grad, method='L-BFGS-B',
                       options={'maxiter': 3000, 'ftol': 1e-15, 'gtol': 1e-10,
                                'maxfun': 100000, 'disp': False})
        a_out = softmax_stable(res.x)
    except Exception:
        a_out = a0.copy()
    return a_out


def generate_diverse_inits(parent, n, rng, count=8):
    """Generate diverse initializations."""
    candidates = []

    if parent is not None:
        if len(parent) == n:
            p = np.maximum(parent, 0)
            p = p / p.sum()
            candidates.append(p.copy())
            for scale in [0.02, 0.05, 0.1]:
                pert = p + rng.standard_normal(n) * scale
                pert = np.maximum(pert, 0)
                pert = pert / pert.sum()
                candidates.append(pert)
        else:
            interp = interpolate_sequence(parent, n)
            candidates.append(interp)

    # Uniform
    candidates.append(np.ones(n) / n)

    # Gaussian-like
    x = np.linspace(-3, 3, n)
    gau = np.exp(-x ** 2 / 3)
    gau = np.maximum(gau, 1e-6)
    candidates.append(gau / gau.sum())

    # Bimodal
    mid = n // 2
    bi = np.zeros(n)
    bi[max(0, mid - n // 5):mid + n // 5] = 1.0
    bi[max(0, mid - n // 3):mid + n // 3] = 0.3
    bi = np.maximum(bi, 0.001)
    candidates.append(bi / bi.sum())

    # Linear ramp
    ramp = np.linspace(0.1, 1.0, n)
    candidates.append(ramp / ramp.sum())

    # Random
    rand = np.abs(rng.standard_normal(n)) + 0.01
    candidates.append(rand / rand.sum())

    # Cosine modulation
    t = np.linspace(0, 2 * np.pi * 3, n)
    cos_mod = 1.0 + 0.3 * np.cos(t)
    cos_mod = np.maximum(cos_mod, 0.1)
    candidates.append(cos_mod / cos_mod.sum())

    return candidates[:count]

    print(f"\n=== Phase 4: Local Refinement (best={best_score:.8f}) ===")
    n_best = len(best_seq)
    rem = deadline - time.monotonic() - 200

    if rem > 20:
        t_sub = min(rem * 0.3, 80.0)
        a_ps2, _ = peak_shave_aggressive(best_seq, t_sub, n_trials=5000)
        best_seq, best_score = evaluate_and_track(a_ps2, "peak_shave_2", best_seq, best_score)
        print(f"  After peak shave 2: {best_score:.8f}")

    rem = deadline - time.monotonic() - 100
    if rem > 15:
        t_sub = min(rem * 0.4, 60.0)
        a_cd, _ = coordinate_descent_fast(best_seq, t_sub, n_sweeps=80)
        best_seq, best_score = evaluate_and_track(a_cd, "coord_desc", best_seq, best_score)
        print(f"  After coord descent: {best_score:.8f}")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-Length (best={best_score:.8f}) ===")
    n_best = len(best_seq)
    rem = deadline - time.monotonic() - 100

    if rem > 60:
        alt_lengths = []
        for factor in [1.25, 1.5, 2.0, 0.75, 1.33]:
            alt_n = int(n_best * factor)
            if 384 <= alt_n <= 4096 and alt_n != n_best:
                if alt_n not in alt_lengths:
                    alt_lengths.append(alt_n)

        for alt_n in alt_lengths[:3]:
            rem = deadline - time.monotonic() - 50
            if rem < 40:
                break
            t_sub = min(rem * 0.35, 80.0)
            init_alt = interpolate_sequence(best_seq, alt_n)
            a_alt, _ = adam_nesterov(
                init_alt, t_sub, T_init=0.03, T_final=0.002,
                lr=0.008, schedule='hold_anneal', mu=0.9
            )
            best_seq, best_score = evaluate_and_track(
                a_alt, f"alt_n{alt_n}", best_seq, best_score
            )
            # Quick peak shave on the new length
            rem2 = deadline - time.monotonic() - 30
            if rem2 > 20 and len(a_alt) == len(best_seq):
                a_ps_alt, _ = peak_shave_aggressive(a_alt, min(rem2 * 0.5, 40.0), n_trials=3000)
                best_seq, best_score = evaluate_and_track(
                    a_ps_alt, f"alt_n{alt_n}_ps", best_seq, best_score
                )
            print(f"  Alt n={alt_n}: best={best_score:.8f}")

    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.8f}) ===")
    n_best = len(best_seq)
    rem = deadline - time.monotonic() - 10

    if rem > 20:
        # Very low temperature Adam
        t_sub = min(rem * 0.35, 60.0)
        a_fin, _ = adam_nesterov(
            best_seq, t_sub, T_init=0.01, T_final=0.0005,
            lr=0.004, schedule='exp', mu=0.9
        )
        best_seq, best_score = evaluate_and_track(a_fin, "final_adam", best_seq, best_score)

    rem = deadline - time.monotonic() - 5
    if rem > 10:
        t_sub = min(rem * 0.6, 30.0)
        a_fin_ps, _ = peak_shave_aggressive(best_seq, t_sub, n_trials=3000)
        best_seq, best_score = evaluate_and_track(a_fin_ps, "final_ps", best_seq, best_score)

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.8f}")

    return [float(x) for x in best_seq]
