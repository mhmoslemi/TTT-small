import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize as sci_minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(a, b):
    """Linear cross-convolution of a and b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def build_conv_matrix_rows(a, k_indices):
    """Build rows of the convolution matrix for specific output indices."""
    n = len(a)
    rows = []
    for k in k_indices:
        row = np.zeros(n)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        row[idx] = a[k - idx]
        rows.append(row)
    return np.array(rows)


def get_lp_solver():
    """Find a working LP solver."""
    if '_lp_solver_g' in globals():
        return _lp_solver_g
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CBC]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_g = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_g = cp.ECOS
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix_rows(a_fixed, range(2 * n - 1))
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CBC]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def symmetrize(a):
    """Symmetrize a sequence: (a + a[::-1]) / 2, renormalize."""
    a_sym = (a + a[::-1]) / 2.0
    s = a_sym.sum()
    if s > 1e-15:
        a_sym = a_sym / s
    return a_sym


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)


def lbgfs_soft_max(a0, time_limit, T_start=100.0, T_end=5.0, maxiter=5000):
    """L-BFGS refinement with soft-max smoothing and geometric temperature."""
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    z = z0.copy()

    n_stages = 5
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(50, maxiter // n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func(z_val):
            a = softmax_from_z(z_val)
            b = fast_convolve(a)
            Tb = stage_T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n * b.max() * 0 + (2.0 * n / stage_T)) * log_sum_exp

        def jac(z_val):
            a = softmax_from_z(z_val)
            b = fast_convolve(a)
            Tb = stage_T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)

        try:
            res = sci_minimize(
                fun=func, x0=z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-15,
                         'gtol': 1e-10, 'maxfun': iters_per_stage * 60, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    a_out = softmax_from_z(z)
    return a_out


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=10000,
                       lbgfs_every=200, lbgfs_time=10.0):
    """
    Nesterov dual averaging LP refinement with periodic L-BFGS interleaving.
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    a = symmetrize(a)
    y = a.copy()
    x_prev = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        b_new = symmetrize(b_new)

        alpha = 2.0 / (it + 2.0)
        y_new = (1.0 - alpha) * y + alpha * b_new
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            break
        y_new = y_new / sy

        x_new = ((it + 1) * x_prev + (alpha * (it + 2)) * b_new) / ((it + 1) + alpha * (it + 2))
        x_new = np.maximum(x_new, 0)
        sx = x_new.sum()
        if sx < 1e-15:
            x_new = y_new.copy()
            sx = 1.0
        x_new = x_new / sx
        x_new = symmetrize(x_new)

        score_x = 2.0 * n * fast_convolve(x_new).max()
        score_y = 2.0 * n * fast_convolve(y_new).max()

        if score_x < best_score:
            best_score = score_x
            best_a = x_new.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y_new.copy()

        x_prev = x_new
        y = y_new

        # Periodic L-BFGS polish
        if (it + 1) % lbgfs_every == 0:
            elapsed = time.monotonic() - t0
            remaining = time_limit - elapsed
            if remaining > lbgfs_time + 5:
                a_lb = lbgfs_soft_max(best_a, lbgfs_time, T_start=5.0, T_end=1.0, maxiter=1000)
                sc_lb = 2.0 * n * fast_convolve(a_lb).max()
                if sc_lb < best_score:
                    best_score = sc_lb
                    best_a = a_lb.copy()
                    x_prev = best_a.copy()
                    y = best_a.copy()

        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NL iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def peak_shave_aggressive(a0, time_limit, n_steps=20000):
    """Aggressive multi-scale peak shaving with paired and multi-coordinate moves."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        # Exclude the trivial endpoints
        b_work = b.copy()
        b_work[0] = -np.inf
        b_work[-1] = -np.inf
        # Also exclude the very center if it's the global max (we want to shave secondary peaks too)
        peak_k = int(np.argmax(b_work))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_n = min(8, len(idx))
        top_contrib = np.argsort(contribs)[::-1][:top_n]

        improved = False

        # Try single coordinate reduction
        for ci in top_contrib[:5]:
            i = int(idx[ci])
            if a[i] < 1e-14:
                continue
            for factor in [0.98, 0.95, 0.92, 0.88, 0.82]:
                new_val = a[i] * factor
                diff = a[i] - new_val
                if diff < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                a_trial[mask] = others + diff * others / others_sum
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial.copy()
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        # Try paired reduction
        if not improved:
            for ci in top_contrib[:4]:
                i = int(idx[ci])
                j = int(peak_k - i)
                if i == j or a[j] < 1e-14 or a[i] < 1e-14:
                    continue
                for f1 in [0.95, 0.90, 0.85]:
                    for f2 in [0.95, 0.90, 0.85]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        ss = a_trial.sum()
                        if ss < 1e-15:
                            continue
                        a_trial = a_trial / ss
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial.copy()
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        # Try reducing top 3 simultaneously
        if not improved:
            a_trial = a.copy()
            for ci in top_contrib[:3]:
                i = int(idx[ci])
                a_trial[i] *= 0.96
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss > 1e-15:
                a_trial = a_trial / ss
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial.copy()
                    best_score = score
                    best_a = a.copy()
                    improved = True

        if not improved:
            # Try a random perturbation to escape
            if step % 50 == 49:
                rng = np.random.default_rng(step)
                a_trial = a * (1.0 + 0.01 * rng.standard_normal(n))
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss > 1e-15:
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial.copy()
                        best_score = score
                        best_a = a.copy()

    return best_a, best_score


def coordinate_descent_fast(a0, time_limit):
    """Fast cyclic coordinate descent on the simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    n_passes = 0

    while time.monotonic() - t0 < time_limit:
        n_passes += 1
        pass_improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-15:
                continue
            base_sum = 1.0 - a[i]
            if base_sum < 1e-15:
                continue
            best_t = a[i]
            best_local = best_score
            for factor in [0.8, 0.9, 1.1, 1.25]:
                t_trial = a[i] * factor
                if t_trial < -1e-15 or t_trial > 1.0 + 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] = t_trial
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = 2.0 * n * fast_convolve(a_trial).max()
                if sc < best_local:
                    best_local = sc
                    best_t = t_trial
            if best_t != a[i]:
                a[i] = best_t
                a = np.maximum(a, 0)
                ss = a.sum()
                a = a / ss
                sc = 2.0 * n * fast_convolve(a).max()
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    pass_improved = True

        if not pass_improved and n_passes > 3:
            break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving symmetry."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    # Ensure symmetric interpolation
    f_sym = (f_norm + f_norm[::-1]) / 2.0
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_sym)
    f_interp = np.maximum(f_interp, 0)
    # Enforce symmetry
    if new_n % 2 == 0:
        f_interp = (f_interp + f_interp[::-1]) / 2.0
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    print(f"\n=== Phase 3: Low-T L-BFGS (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 200.0)
    if t_alloc > 20:
        a_lb2 = lbgfs_soft_max(best_seq, t_alloc, T_start=5.0, T_end=0.5, maxiter=4000)
        try_update(a_lb2, "lbgfs_lowT")

    # === Phase 4: Aggressive peak shaving (200s) ===
    print(f"\n=== Phase 4: Peak Shaving (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 200.0)
    if t_alloc > 15:
        a_ps, _ = peak_shave_aggressive(best_seq, t_alloc, n_steps=50000)
        try_update(a_ps, "peak_shave")

    # === Phase 5: Coordinate descent (100s) ===
    print(f"\n=== Phase 5: Coord Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.06, 100.0)
    if t_alloc > 10:
        a_cd, _ = coordinate_descent_fast(best_seq, t_alloc)
        try_update(a_cd, "coord_desc")

    # === Phase 6: Second LP round (200s) ===
    print(f"\n=== Phase 6: Second LP (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 200.0)
    if t_alloc > 20:
        a_np2, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver,
                                       n_max_iters=20000, lbgfs_every=500, lbgfs_time=5.0)
        try_update(a_np2, "lp_round2")

    # === Phase 7: Alternative lengths (250s) ===
    print(f"\n=== Phase 7: Alternative Lengths (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    alt_lengths = sorted(set([
        int(n_best * 1.25),
        int(n_best * 1.5),
        int(n_best * 2.0),
    ]), key=lambda x: abs(x - n_best))
    alt_lengths = [x for x in alt_lengths if 256 <= x <= 4096 and x != n_best]

    for alt_n in alt_lengths[:3]:
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t_alloc = min(rem * 0.3, 80.0)
        if t_alloc < 20:
            break
        print(f"  Trying n={alt_n}, budget={t_alloc:.0f}s")
        init_alt = interpolate_sequence(best_seq, alt_n)
        a_alt = lbgfs_soft_max(init_alt, t_alloc * 0.5, T_start=50.0, T_end=2.0, maxiter=2000)
        try_update(a_alt, f"alt_n{alt_n}_lbgfs")

        rem = deadline - time.monotonic()
        if rem > 20 and len(best_seq) == alt_n:
            t_lp = min(rem * 0.3, 50.0)
            if t_lp > 15:
                a_al, _ = nesterov_lp_refine(best_seq, t_lp, solver=solver, n_max_iters=3000)
                try_update(a_al, f"alt_n{alt_n}_lp")

    # === Phase 8: Final polish (remaining time) ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t_f1 = min(rem * 0.3, 40.0)
        a_f1 = lbgfs_soft_max(best_seq, t_f1, T_start=2.0, T_end=0.2, maxiter=2000)
        try_update(a_f1, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 10:
        t_f2 = min(rem * 0.3, 30.0)
        a_f2, _ = peak_shave_aggressive(best_seq, t_f2, n_steps=20000)
        try_update(a_f2, "final_ps")

    rem = deadline - time.monotonic()
    if rem > 10:
        t_f3 = min(rem * 0.5, 30.0)
        a_f3, _ = nesterov_lp_refine(best_seq, t_f3, solver=solver, n_max_iters=2000)
        try_update(a_f3, "final_lp")

    rem = deadline - time.monotonic()
    if rem > 5:
        a_f4, _ = peak_shave_aggressive(best_seq, min(rem * 0.8, 20.0), n_steps=10000)
        try_update(a_f4, "final_ps2")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
