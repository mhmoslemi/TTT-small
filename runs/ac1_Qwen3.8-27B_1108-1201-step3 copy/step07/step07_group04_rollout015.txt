def fast_convolve(a):

def lbgfs_logspace(a0, time_limit, T_schedule=None, verbose=True):
    """
    L-BFGS-B in log-space with log-sum-exp smoothing.
    Variable: z = log(a), a = exp(z) / sum(exp(z))
    """
    n = len(a0)
    a_safe = np.maximum(a0, 1e-30)
    z0 = np.log(a_safe / a_safe.sum())
    z0 = z0 - z0.mean()  # center to avoid large values
    
    t0 = monotonic()
    stopped = [False]
    
    def func_and_grad(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()
        
        # Compute smoothed objective and its gradient wrt a
        b = fast_convolve(a)
        T = T_schedule  # fixed for now
        b_max = b.max()
        scaled = (b - b_max) / T
        exp_scaled = np.exp(scaled)
        w = exp_scaled / exp_scaled.sum()
        
        # Objective: 2*n*(b_max + T*log(sum(exp_scaled)))
        # = 2*n*(b_max + T*(T*log(sum) / T))  -- just compute it
        obj = 2.0 * n * (b_max + T * np.log(exp_scaled.sum()))
        
        # Gradient wrt a: 2*n * 2 * (w cross-correlated with a)
        N = 1 << (len(w) + n - 2).bit_length()
        W = rfft(w, N)
        A = rfft(a, N)
        corr = irfft(W * np.conj(A), N)
        grad_a = 4.0 * n * corr[:n]
        
        # Transform to z-space: a = exp(z)/sum(exp(z))
        # da_j/dz_k = a_j*(delta_jk - a_k)
        # dz_k: d(obj)/dz_k = sum_j grad_a_j * da_j/dz_k
        # = sum_j grad_a_j * a_j * (delta_jk - a_k)
        # = grad_a_k * a_k - a_k * sum_j grad_a_j * a_j
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - a * S)
        
        return obj, grad_z
    
    best_z = z0.copy()
    best_f = func_and_grad(z0)[0]
    last_T = T_schedule
    
    # Run L-BFGS with current T
    try:
        res = minimize(
            fun=lambda z: func_and_grad(z)[0],
            x0=z0,
            jac=lambda z: func_and_grad(z)[1],
            method='L-BFGS-B',
            options={'maxiter': 5000, 'ftol': 1e-15, 'gtol': 1e-10,
                     'maxfun': 50000, 'disp': False}
        )
        if not stopped[0]:

            print(f"    L-BFGS error: {e}")
    
    if stopped[0] or verbose:
        z_shifted = best_z - best_z.max()
        a_out = np.exp(z_shifted)
        a_out = a_out / a_out.sum()
        actual_score = score_of(a_out)
        if verbose:
            print(f"    L-BFGS T={T_schedule}: actual_score={actual_score:.6f}")
    
    z_shifted = best_z - best_z.max()
    a_out = np.exp(z_shifted)
    a_out = a_out / a_out.sum()
    return a_out, score_of(a_out)

    best_score = score_of(a)
    t0 = monotonic()
    it = 0
    lr = 0.1
    
    while monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        b_max = b.max()
        # Find all "near-peak" indices (within 1% of max)
        threshold = b_max * 0.99
        peak_indices = np.where(b >= threshold)[0]
        
        # Compute subgradient: average of subgradients at peak indices
        # subgrad[j] = (2/|P|) * sum_{k in P} a[k-j] (with bounds)
        subgrad = np.zeros(n)
        for k in peak_indices:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                subgrad[j] += 2.0 * a[k - j]
        subgrad /= len(peak_indices)
        
        # Projected gradient step onto simplex
        # a_new = proj_simplex(a - lr * subgrad)
        candidate = a - lr * subgrad
        
        # Project onto simplex {x: sum(x)=1, x>=0}
        u = np.sort(candidate)[::-1]
        cssv = np.cumsum(u)
        cond = u * np.arange(1, n + 1) > (cssv - 1)
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        a_new = np.maximum(candidate - theta, 0.0)
        a_new = a_new / a_new.sum()
        
        sc = score_of(a_new)
        it += 1
        
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.05, 0.5)  # increase step size slightly
        else:
            lr *= 0.9
            if lr < 1e-6:
                lr = 0.1
                # Random restart locally
                a = best_a + np.random.standard_normal(n) * 0.01
                a = np.maximum(a, 0)
                a = a / a.sum()
        
        if verbose and it % 200 == 0:
            print(f"    SubGrad iter={it} ({monotonic()-t0:.0f}s): score={best_score:.6f} lr={lr:.6f}")
    
    return best_a, best_score


def multiplicative_weight(a0, time_limit, verbose=True):
    """
    Multiplicative weight (mirror descent) method.
    a_new = a * exp(-eta * subgrad) / Z
    """
    n = len(a0)
    a = np.maximum(a0, 1e-30).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_of(a)
    t0 = monotonic()
    it = 0
    eta = 0.5
    no_improve = 0
    
    while monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        b_max = b.max()
        threshold = b_max * 0.995
        peak_indices = np.where(b >= threshold)[0]
        
        # Subgradient (averaged over near-peaks)
        subgrad = np.zeros(n)
        for k in peak_indices:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                subgrad[j] += a[k - j]
        subgrad /= len(peak_indices)
        
        # Multiplicative update
        a_new = a * np.exp(-eta * subgrad)
        s = a_new.sum()
        if s < 1e-30:
            break
        a_new = a_new / s
        
        sc = score_of(a_new)
        it += 1
        
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            eta = min(eta * 1.01, 5.0)
        else:
            no_improve += 1
            eta *= 0.95
            if no_improve > 200:
                eta = max(eta * 3, 0.1)
                a = best_a + np.random.standard_normal(n) * 0.005
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
        
        if verbose and it % 500 == 0:
            print(f"    MW iter={it} ({monotonic()-t0:.0f}s): score={best_score:.6f} eta={eta:.4f}")
    
    return best_a, best_score


def coordinate_descent(a0, time_limit, verbose=True):
    """
    Coordinate descent on the exact objective.
    For each coordinate j, find the optimal value given all others fixed.
    """
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_of(a)
    t0 = monotonic()
    it = 0
    
    while monotonic() - t0 < time_limit:
        improved = False
        b = fast_convolve(a)
        b_max = b.max()
        
        # Find coordinates that contribute to peaks
        peak_k = int(np.argmax(b))
        lo = max(0, peak_k - n + 1)
        hi = min(n - 1, peak_k)
        active_coords = list(range(lo, hi + 1))
        # Add some random coordinates too
        active_coords.extend(np.random.choice(n, min(10, n), replace=False).tolist())
        active_coords = list(set(active_coords))
        np.random.shuffle(active_coords)
        
        for j in active_coords:
            if monotonic() - t0 > time_limit:
                break
            
            old_val = a[j]
            if old_val < 1e-15:
                # Try increasing from zero
                for trial in [0.001, 0.005, 0.01]:
                    a[j] = trial
                    a[j] = 0  # reset
                    # Redistribute
                    others_sum = 1.0 - trial
                    if others_sum <= 0:
                        continue
                    a_trial = a.copy()
                    a_trial[j] = trial
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a_trial[mask] = a[mask] * (others_sum / a[mask].sum()) if a[mask].sum() > 0 else 0
                    sc = score_of(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                continue
            
            # Try different values for a[j]
            # The contribution to conv at peak_k is a[j]*a[peak_k-j] + a[j]*a[j-peak_k] (if valid)
            # More precisely, a[j] appears in conv_k for all k where 0<=k-j<n or 0<=j-k<n
            # For the peak, the main contribution is a[j]*a[peak_k-j]
            
            for factor in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                new_val = old_val * factor
                if new_val < 0:
                    continue
                diff = new_val - old_val
                if abs(diff) < 1e-15:
                    continue
                
                a_trial = a.copy()
                a_trial[j] = new_val
                # Redistribute the difference to other coordinates proportionally
                if abs(diff) > 1e-15:
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    others = a_trials - a_trial[j] if False else a[mask]
                    others_sum = others.sum()
                    if others_sum > 1e-15:
                        a_trial[mask] = others - diff * others / (others.sum() + diff)
                        a_trial = np.maximum(a_trial, 0)
                        s = a_trial.sum()
                        if s > 1e-15:
                            a_trial = a_trial / s
                            sc = score_of(a_trial)
                            if sc < best_score - 1e-14:
                                a = a_trial
                                best_score = sc
                                best_a = a.copy()
                                improved = True
                                break
        
        it += 1
        if not improved and it > 50:
            break
    
    return best_a, best_score
