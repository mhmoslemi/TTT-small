    """
    Compute the gradient of (1/beta)*log(sum exp(beta*b[k])) w.r.t. a.
    Returns gradient vector g of length n.
    g[i] = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
    where w[k] = exp(beta*(b[k]-max_b)) / sum exp(beta*(b-l-max_b))

    b_shifted = beta * (b - b.max())
    w = np.exp(b_shifted)
    w_sum = w.sum()

        w = np.zeros(len(b))
        w[b.argmax()] = 1.0
    else:
        w /= w_sum

            # Perturb from best
            a = project_simplex(best_a + rng.standard_normal(n) * perturb_mag)
            perturb_mag *= 0.7
        elif no_improve > 400:
            beta = min(beta * 1.3, beta_end)
            lr *= 0.8

        # Gentle lr decay
        if step % 3000 == 0:
            lr *= 0.97

    return best_a, best_obj


def smooth_gd_additive(a_init, time_limit, beta_start=5.0, beta_end=2000.0,
                       lr_init=0.004, momentum=0.9, seed=42):
    """
    Smooth gradient descent with additive updates and Nesterov momentum.
    Projects onto simplex after each step.
    Returns (best_a, best_max_b).
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s

    v = np.zeros(n)
    best_a = a.copy()
    best_obj = eval_max_b(a)
    beta = beta_start
    lr = lr_init
    no_improve = 0
    step = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        g = compute_smooth_gradient(a, b, beta)

        # Nesterov momentum
        v = momentum * v - lr * g
        a_new = project_simplex(a + v)

        obj = eval_max_b(a_new)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        step += 1

        if no_improve > 1000:
            beta = min(beta * 1.5, beta_end)
            lr *= 0.5
            v = np.zeros(n)
            no_improve = 0
            a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
        elif no_improve > 500:
            lr *= 0.9
        if step % 5000 == 0:
            lr *= 0.95

    return best_a, best_obj


def lp_refine_sparse(f, K=80, max_iters=25, time_limit=60.0, verbose=False):
    """
    LP-based refinement using only top-K constraining positions.
    Solves: max sum(g) s.t. (f*g)[k] <= M for top-K k, g >= 0.
    Blends solution with current f to reduce max convolution.
    Returns (best_a, best_max_b).
    """
    t0 = time.monotonic()
    n = len(f)
    a = f.copy().astype(np.float64)
    a /= a.sum()
    best_a = a.copy()
    best_obj = eval_max_b(a)

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        M = b.max()

        # Top-K positions
        if K < len(b):
            top_idx = np.argpartition(b, -K)[-K:]
        else:
            top_idx = np.arange(len(b))

        # Build constraint matrix (K x n)
        A_mat = np.zeros((len(top_idx), n))
        for i, k in enumerate(top_idx):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                A_mat[i, k - j] = a[j]

        b_vec = np.full(len(top_idx), M)
        c_vec = -np.ones(n)

        try:
            res = linprog(c_vec, A_ub=A_mat, b_ub=b_vec,
                          bounds=[(0.0, None)] * n, method='highs')
            if not res.success:
                break
            g = res.x
            g_sum = g.sum()
            if g_sum < 1e-10:
                break
            g_norm = g / g_sum

            # Line search for optimal blend
            best_t = 0.0
            best_blend_obj = best_obj
            for t in [0.03, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
                blend = (1.0 - t) * best_a + t * g_norm
                blend = np.maximum(blend, 0.0)
                ss = blend.sum()
                if ss < 1e-10:
                    continue
                blend /= ss
                obj = eval_max_b(blend)
                if obj < best_blend_obj:
                    best_blend_obj = obj
                    best_t = t

            if best_t > 0.0:
                a = (1.0 - best_t) * best_a + best_t * g_norm
                a = np.maximum(a, 0.0)
                a /= a.sum()
                obj = eval_max_b(a)
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                    if verbose:
                        print(f"    LP iter {it}: t={best_t:.2f}, max_b={best_obj:.10f}")
            else:
                break
        except Exception:
            break

    return best_a, best_obj


def coordinate_descent(a0, time_limit, multipliers=None):
    """Coordinate descent polish on simplex."""
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a /= a.sum()
    best_a = a.copy()
    best_obj = eval_max_b(a)

    if multipliers is None:
        multipliers = np.array([0.3, 0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0, 3.0, 5.0])

    improved = True
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-15:
                continue
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0.0 or new_val > 0.9999:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < -0.01 or scale > 50.0:
                    continue
                a_tr = a.copy()
                a_tr[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_tr[mask] *= scale
                ss = a_tr.sum()
                if ss < 1e-15:
                    continue
                a_tr /= ss
                val = eval_max_b(a_tr)
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a /= a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

    return best_a, best_obj


def make_sigmoid_init(n, sharpness=5.0, center=0.5):
    """Sigmoid-shaped initialization (lopsided to reduce center peak)."""
    x = np.linspace(0, 1, n)
    a = 1.0 / (1.0 + np.exp(-sharpness * (x - center)))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_power_init(n, p=1.0):
    """Power-law shape: a[j] proportional to (j+1)^p."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** p
    return a / a.sum()


def make_tukey_init(n, r=0.5):
    a = np.ones(n)
    taper = max(1, int(r * n // 2))
    if taper < n:
        t = np.linspace(0, 1, taper)
        a[:taper] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t / r - 1.0)))
        a[-taper:] = a[:taper][::-1]
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_gaussian_init(n, sigma=0.3, center=0.5):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def interpolate_sequence(f, new_n):
    """Interpolate to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def combined_optimize(a_init, t_sg, t_lp, t_sg2, n, seed, verbose=False):
    """
    Combined optimization pipeline:
    1. Smooth GD (multiplicative)
    2. LP refinement
    3. Smooth GD (additive)
    4. LP refinement
    Returns (best_a, best_max_b).
    """
    t0 = time.monotonic()
    a = a_init.copy().astype(np.float64)
    a /= a.sum()
    best_a = a.copy()
    best_obj = eval_max_b(a)

    # Phase 1: Smooth GD multiplicative
    if t_sg > 5:
        a1, obj1 = smooth_gd_multiplicative(
            a, time_limit=t_sg, beta_start=1.0, beta_end=500.0,
            lr_base=2.0, damp=0.4, seed=seed
        )
        if obj1 < best_obj:
            best_obj = obj1
            best_a = a1.copy()
            if verbose:
                print(f"    SmoothGD-M: max_b={best_obj:.10f}")

    # Phase 2: LP
    rem = t_sg + t_lp - (time.monotonic() - t0)
    if rem > 5:
        a2, obj2 = lp_refine_sparse(
            best_a, K=80, max_iters=20,
            time_limit=min(rem, t_lp), verbose=verbose
        )
        if obj2 < best_obj:
            best_obj = obj2
            best_a = a2.copy()
            if verbose:
                print(f"    LP: max_b={best_obj:.10f}")

    # Phase 3: Smooth GD additive (fine)
    rem = t_sg + t_lp + t_sg2 - (time.monotonic() - t0)
    if rem > 5:
        a3, obj3 = smooth_gd_additive(
            best_a, time_limit=rem, beta_start=50.0, beta_end=3000.0,
            lr_init=0.002, momentum=0.92, seed=seed + 777
        )
        if obj3 < best_obj:
            best_obj = obj3
            best_a = a3.copy()
            if verbose:
                print(f"    SmoothGD-A: max_b={best_obj:.10f}")

    # Phase 4: Final LP
    rem = t_sg + t_lp + t_sg2 - (time.monotonic() - t0)
    if rem > 3:
        a4, obj4 = lp_refine_sparse(
            best_a, K=100, max_iters=10, time_limit=min(rem, 30.0)
        )
        if obj4 < best_obj:
            best_obj = obj4
            best_a = a4.copy()

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses smooth gradient descent (log-sum-exp), sparse LP, and multi-length
    exploration with adaptive strategy.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, budget_s - 30.0)

    best_seq = None
    best_score = float('inf')

    def update_best(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        # Good initial shape: slightly lopsided sigmoid
        parent = make_sigmoid_init(1152, sharpness=3.0, center=0.5)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"No parent, using sigmoid n=1152, score={best_score:.6f}")

    n_parent = len(parent)

    # === Phase 1: Deep smooth GD on parent ===
    print(f"\n{'='*60}")
    print(f"Phase 1: Deep Smooth GD on parent (n={n_parent})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic() - 600
    if rem > 30:
        t_sg = min(rem * 0.55, 180.0)
        t_lp = min(rem * 0.3, 60.0)
        t_sg2 = min(rem * 0.15, 40.0)
        print(f"  mul-GD t={t_sg:.0f}s, LP t={t_lp:.0f}s, add-GD t={t_sg2:.0f}s")
        a_opt, obj_opt = combined_optimize(
            parent, t_sg, t_lp, t_sg2, n_parent, seed, verbose=True
        )
        update_best(a_opt, "phase1")

    # === Phase 2: Diverse initializations ===
    print(f"\n{'='*60}")
    print(f"Phase 2: Diverse Initializations (best={best_score:.6f})")
    print(f"{'='*60}")
    n_base = len(best_seq) if best_seq is not None else n_parent

    inits = {
        "sigmoid_3": make_sigmoid_init(n_base, 3.0, 0.5),
        "sigmoid_5": make_sigmoid_init(n_base, 5.0, 0.5),
        "sigmoid_8": make_sigmoid_init(n_base, 8.0, 0.5),
        "power_0.5": make_power_init(n_base, 0.5),
        "power_1.0": make_power_init(n_base, 1.0),
        "power_2.0": make_power_init(n_base, 2.0),
        "tukey_0.3": make_tukey_init(n_base, 0.3),
        "gauss_0.25": make_gaussian_init(n_base, 0.25),
        "gauss_0.4": make_gaussian_init(n_base, 0.4),
    }

    total_time_diverse = min(300.0, (deadline - time.monotonic() - 400.0) * 0.35)
    time_per_init = total_time_diverse / len(inits)

    for name, init_a in inits.items():
        rem = deadline - time.monotonic() - 300
        if rem < 15:
            break
        t_alloc = min(time_per_init, rem * 0.3)
        if t_alloc < 8:
            continue
        t_sg_i = t_alloc * 0.5
        t_lp_i = t_alloc * 0.3
        t_sg2_i = t_alloc * 0.2
        print(f"  {name} (t={t_alloc:.0f}s)")
        a_div, obj_div = combined_optimize(
            init_a, t_sg_i, t_lp_i, t_sg2_i, n_base,
            seed + hash(name) % 100000, verbose=False
        )
        update_best(a_div, f"div_{name}")

    # === Phase 3: Multi-length exploration ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Multi-length Exploration (best={best_score:.6f})")
    print(f"{'='*60}")
    if best_seq is not None:
        target_lengths = []
        for factor in [0.75, 1.25, 1.5, 0.5, 2.0, 1.33, 0.88]:
            tl = int(n_parent * factor)
            tl = max(256, min(4096, tl))
            if tl not in target_lengths:
                target_lengths.append(tl)
        target_lengths.sort(key=lambda x: abs(x - n_parent))

        for target_n in target_lengths[:5]:
            rem = deadline - time.monotonic() - 200
            if rem < 25:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.2, 50.0)
            if t_alloc < 10:
                continue
            t_sg_l = t_alloc * 0.5
            t_lp_l = t_alloc * 0.3
            t_sg2_l = t_alloc * 0.2
            print(f"  n={target_n}, t={t_alloc:.0f}s")
            a_len, obj_len = combined_optimize(
                f_interp, t_sg_l, t_lp_l, t_sg2_l, target_n,
                seed + target_n, verbose=False
            )
            update_best(a_len, f"len_n{target_n}")

    # === Phase 4: Deep optimization on best ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Deep Optimization (best={best_score:.6f})")
    print(f"{'='*60}")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            t_sg_d = min(rem * 0.4, 100.0)
            t_lp_d = min(rem * 0.3, 60.0)
            t_sg2_d = min(rem * 0.3, 50.0)
            print(f"  Deep: n={n_best}, mul-GD t={t_sg_d:.0f}s, LP t={t_lp_d:.0f}s, add-GD t={t_sg2_d:.0f}s")
            a_deep, obj_deep = combined_optimize(
                best_seq, t_sg_d, t_lp_d, t_sg2_d, n_best,
                seed + 55555, verbose=True
            )
            update_best(a_deep, "deep")

            # Extra LP pass
            rem = deadline - time.monotonic() - 80
            if rem > 10 and best_seq is not None:
                a_lp2, obj_lp2 = lp_refine_sparse(
                    best_seq, K=120, max_iters=15,
                    time_limit=min(rem, 40.0), verbose=True
                )
                update_best(a_lp2, "deep_lp")

    # === Phase 5: Coordinate descent polish ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Coordinate Descent (best={best_score:.6f})")
    print(f"{'='*60}")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 50
        if rem > 10:
            t_cd = min(rem, 50.0)
            n_cd = len(best_seq)
            print(f"  CD: n={n_cd}, t={t_cd:.0f}s")
            a_cd, obj_cd = coordinate_descent(
                best_seq, time_limit=t_cd,
                multipliers=np.array([0.4, 0.6, 0.8, 0.9, 0.95, 1.05, 1.1, 1.2, 1.5, 2.0, 3.0])
            )
            update_best(a_cd, "cd")

    # === Phase 6: Final micro-polish ===
    print(f"\n{'='*60}")
    print(f"Phase 6: Final Micro-polish (best={best_score:.6f})")
    print(f"{'='*60}")
    if best_seq is not None:
        n_final = len(best_seq)
        rem = deadline - time.monotonic() - 20
        if rem > 8:
            # High-beta smooth GD
            t_micro = min(rem * 0.4, 20.0)
            a_micro, obj_micro = smooth_gd_additive(
                best_seq, time_limit=t_micro,
                beta_start=100.0, beta_end=5000.0,
                lr_init=0.0005, momentum=0.95, seed=seed + 999999
            )
            update_best(a_micro, "micro_gd")

            # Quick LP
            rem = deadline - time.monotonic() - 10
            if rem > 3:
                a_micro_lp, obj_micro_lp = lp_refine_sparse(
                    best_seq, K=150, max_iters=8, time_limit=min(rem, 12.0)
                )
                update_best(a_micro_lp, "micro_lp")

            # Tiny CD
            rem = deadline - time.monotonic() - 5
            if rem > 3:
                a_micro_cd, obj_micro_cd = coordinate_descent(
                    best_seq, time_limit=rem,
                    multipliers=np.array([0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5])
                )
                update_best(a_micro_cd, "micro_cd")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
