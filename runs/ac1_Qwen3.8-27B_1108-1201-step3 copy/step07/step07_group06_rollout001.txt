        elif schedule == 'hold_anneal':
            if progress < 0.2:
                T = T_init
            elif progress < 0.5:
                T = T_init * 0.5
            else:
                p2 = (progress - 0.5) / 0.5
                T = T_init * 0.5 * (T_final / (T_init * 0.5)) ** p2
        else:
            T = T_init * (T_final / T_init) ** progress

        a_curr = softmax_from_z(z)
        b = fast_convolve(a_curr)
        max_b = b.max()
        exp_b = np.exp(b / T - max_b / T)
        sw = exp_b.sum()
        if sw < 1e-300:
            w = np.ones(len(b)) / len(b)
        else:
            w = exp_b / sw

        grad_a = compute_grad_a(a_curr, w, n, fft_size)
        S = np.dot(grad_a, a_curr)
        grad_z = a_curr * (grad_a - S)

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr_current * m_hat / (np.sqrt(v_hat) + eps)

        a_new = softmax_from_z(z)
        score = get_score(a_new)

        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            z = z_from_a(best_a) + np.random.standard_normal(n) * 0.003
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            no_improve = 0
            lr_current = max(0.001, lr_current * 0.5)

        lr_current = max(0.0005, lr_current * (1.0 - 0.00005))

        if verbose and t % verbose_interval == 0:
            print(f"      Adam t={t} T={T:.5f} lr={lr_current:.6f} score={best_score:.6f}")

    return best_a, best_score


def multiplicative_grad(a0, time_limit, n_iter=50000, lr_init=0.1, lr_final=0.001):
    """
    Multiplicative weight / natural gradient method.
    a_i <- a_i * exp(-lr * grad_i) / normalization
    This is mirror descent with KL divergence, keeps a_i > 0 automatically.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    a = a / a.sum()
    best_a = a.copy()
    best_score = get_score(a)
    fft_size = 1 << (2 * n - 2).bit_length()
    t0 = time.monotonic()

    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break

        progress = it / n_iter
        lr = lr_init * (lr_final / lr_init) ** progress

        b = fast_convolve(a)
        # Use top-k weights for the max (sparse approximation)
        max_b = b.max()
        # Soft weights
        T = max(0.01, 0.1 * (1 - progress) + 0.01 * progress)
        exp_b = np.exp(b / T - max_b / T)
        w = exp_b / exp_b.sum()

        grad_a = compute_grad_a(a, w, n, fft_size)
        # Normalize gradient: subtract mean to preserve sum
        S = np.dot(grad_a, a)
        grad_centered = grad_a - S

        # Multiplicative update
        a = a * np.exp(-lr * grad_centered)
        a = np.maximum(a, 1e-300)
        a = a / a.sum()

        score = get_score(a)
        if score < best_score - 1e-15:
            best_score = score
            best_a = a.copy()

    return best_a, best_score


def coordinate_descent(a0, time_limit, n_sweeps=500):
    """
    Coordinate descent with multiplicative updates.
    For each element, try several scaling factors and keep the best.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = get_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    factors = [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5]

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        order = rng.permutation(n)
        improved = False

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-18:
                continue

            best_f = 1.0
            best_s = best_score
            for f in factors:
                a_trial = a.copy()
                a_trial[i] *= f
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial /= ss
                sc = get_score(a_trial)
                if sc < best_s:
                    best_s = sc
                    best_f = f

            if best_f != 1.0:
                a[i] *= best_f
                a /= a.sum()
                if best_s < best_score:
                    best_score = best_s
                    best_a = a.copy()
                    improved = True

        if not improved:
            break

    return best_a, best_score


def peak_targeted_shave(a0, time_limit, n_trials=10000):
    """
    Identify the conv peak and try to reduce it by adjusting top contributors.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = get_score(a)
    t0 = time.monotonic()

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        # Find the max
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]
        target = best_score / (2.0 * n)
        if peak_val <= target * 1.0001:
            break

        # Find top contributors to this peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top = np.argsort(contribs)[::-1][:8]

        improved = False
        for ci in top:
            i = int(idx[ci])
            j = int(peak_k - i)
            if j < 0 or j >= n or i == j:
                continue
            if a[i] < 1e-15 or a[j] < 1e-15:
                continue

            for f1 in [0.9, 0.8, 0.7]:
                for f2 in [1.0, 1.05, 1.1]:
                    a_trial = a.copy()
                    a_trial[i] *= f1
                    a_trial[j] *= f2
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial /= ss
                    score = get_score(a_trial)
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a_trial.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        # Also try single element reduction with redistribution
        if not improved:
            for ci in top[:4]:
                i = int(idx[ci])
                if a[i] < 1e-15:
                    continue
                for factor in [0.8, 0.6, 0.4]:
                    a_trial = a.copy()
                    diff = a_trial[i] * (1.0 - factor)
                    a_trial[i] *= factor
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    others_sum = a_trial[mask].sum()
                    if others_sum < 1e-15:
                        continue
                    a_trial[mask] += diff * a_trial[mask] / others_sum
                    score = get_score(a_trial)
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a_trial.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved and trial > 200 and trial % 100 == 0:
            a = best_a.copy()
            noise = np.random.standard_normal(n) * 0.01
            a = a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def generate_candidates(parent, n, rng):
    """Generate diverse candidate initializations."""
    candidates = []

    if parent is not None:
        if len(parent) == n:
            p = np.maximum(parent, 0)
            p = p / p.sum()
            candidates.append(('parent', p.copy()))
            for scale in [0.005, 0.01, 0.02, 0.05]:
                pert = p + rng.standard_normal(n) * scale
                pert = np.maximum(pert, 0)
                pert = pert / pert.sum()
                candidates.append((f'pert_{scale}', pert))
            # Structured perturbations
            for freq in [1, 2, 3, 5]:
                t = np.linspace(0, 2 * np.pi * freq, n)
                for amp in [0.01, 0.03]:
                    pert = p + amp * np.cos(t)
                    pert = np.maximum(pert, 0)
                    sp = pert.sum()
                    if sp > 1e-10:
                        pert /= sp
                        candidates.append((f'cos_{freq}_{amp}', pert))
        else:
            interp = interpolate_sequence(parent, n)
            candidates.append(('interp', interp))

    # Uniform
    candidates.append(('uniform', np.ones(n) / n))

    # Triangular
    tri = np.linspace(0, 1, n) + np.linspace(1, 0, n)
    tri = np.maximum(tri, 0.01)
    tri /= tri.sum()
    candidates.append(('tri', tri))

    # Gaussian
    x = np.linspace(-3, 3, n)
    gau = np.exp(-x**2 / 3)
    gau = np.maximum(gau, 1e-6)
    gau /= gau.sum()
    candidates.append(('gau', gau))

    # Exponential
    expo = np.exp(-np.linspace(0, 2, n))
    expo /= expo.sum()
    candidates.append(('expo', expo))

    # Random
    rand = np.abs(rng.standard_normal(n)) + 0.01
    rand /= rand.sum()
    candidates.append(('rand', rand))

    return candidates


def evaluate_seq(a):
    """Safely evaluate a sequence."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return np.inf
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return np.inf
    return score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining Adam log-space, multiplicative gradient,
    coordinate descent, and peak shaving.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_seq(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_default = 768
        uniform = np.ones(n_default) / n_default
        best_seq = uniform.copy()
        best_score = evaluate_seq(uniform)
        print(f"Default uniform: n={n_default}, score={best_score:.6f}")
        parent = uniform

    n_parent = len(best_seq)
    print(f"Starting best: {best_score:.6f} (n={n_parent})")

    # === Phase 0: Quick evaluation of structured candidates ===
    print("\n=== Phase 0: Structured candidates ===")
    t0 = time.monotonic()
    n_use = n_parent
    inits = generate_candidates(parent, n_use, rng)
    inits_scored = []
    for label, init_a in inits:
        sc = get_score(init_a)
        inits_scored.append((sc, label, init_a))
    inits_scored.sort(key=lambda x: x[0])
    print(f"  Best init: {inits_scored[0][1]} = {inits_scored[0][0]:.6f}")

    # Update best from initializations
    for sc, label, init_a in inits_scored[:5]:
        if sc < best_score:
            best_score = sc
            best_seq = init_a.copy()
            print(f"  Updated best: {label} = {sc:.6f}")

    # === Phase 1: Adam optimization on top candidates (35% time) ===
    print(f"\n=== Phase 1: Adam Multi-Start (n={n_use}) ===")
    t_p1 = min(630.0, (deadline - time.monotonic()) * 0.35)
    n_starts = 4
    time_per_start = t_p1 / n_starts

    for idx in range(n_starts):
        rem = deadline - time.monotonic() - 400
        if rem < 60:
            break
        t_alloc = min(time_per_start, rem * 0.85)
        if t_alloc < 30:
            continue

        _, label, init_a = inits_scored[idx]
        print(f"  Start {idx+1}/{n_starts}: {label}, budget={t_alloc:.0f}s")

        # Try two different schedules
        for sched, T_i, T_f, lr in [
            ('hold_anneal', 0.05, 0.002, 0.008),
            ('exp', 0.03, 0.001, 0.006),
        ]:
            rem = deadline - time.monotonic() - 350
            if rem < 20:
                break
            t_sub = min(t_alloc / 2.0, rem * 0.85)
            if t_sub < 15:
                continue

            a_res, score_res = adam_logspace(
                init_a, t_sub, T_init=T_i, T_final=T_f,
                lr=lr, schedule=sched
            )
            sc = evaluate_seq(a_res)
            if sc < best_score:
                best_score = sc
                best_seq = a_res.copy()
                print(f"    NEW BEST adam_{label}_{sched}: {sc:.6f}")

    # === Phase 2: Multiplicative gradient on best (15% time) ===
    print(f"\n=== Phase 2: Multiplicative Gradient (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 300
    if rem > 30:
        t_sub = min(rem * 0.6, 180.0)
        a_mg, _ = multiplicative_grad(best_seq, t_sub,
                                      n_iter=20000, lr_init=0.05, lr_final=0.003)
        sc = evaluate_seq(a_mg)
        if sc < best_score:
            best_score = sc
            best_seq = a_mg.copy()
            print(f"  NEW BEST multiplicative: {sc:.6f}")
        else:
            print(f"  multiplicative: {sc:.6f} (no improvement)")

    # === Phase 3: Focused Adam refinement (20% time) ===
    print(f"\n=== Phase 3: Focused Adam (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 200
    if rem > 30:
        t_sub = min(rem * 0.7, 300.0)

        sc = evaluate_seq(a_fin)
        if sc < best_score:
            best_score = sc
            best_seq = a_fin.copy()
            print(f"  NEW BEST focused_adam: {sc:.6f}")
        else:
            print(f"  focused_adam: {sc:.6f}")

        a_ps, _ = peak_targeted_shave(best_seq, t_sub, n_trials=5000)

            print(f"  peak_shave: {sc:.6f}")

    print(f"\n=== Phase 5: Coordinate Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic() - 50
    if rem > 15:
        t_sub = min(rem * 0.7, 90.0)
        a_cd, _ = coordinate_descent(best_seq, t_sub, n_sweeps=200)

    if rem > 60:

        for alt_n in alt_lengths[:2]:

            t_sub = min(rem * 0.5, 60.0)
            init_alt = interpolate_sequence(best_seq, alt_n)

                init_alt, t_sub, T_init=0.04, T_final=0.002,

    # === Phase 7: Final polish ===

    rem = deadline - time.monotonic() - 5

    # Final verification
    if best_seq is not None:
        fs = evaluate_seq(best_seq)
        if np.isfinite(fs):
            best_score = fs
