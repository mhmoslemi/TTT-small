def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, n=fft_size)
    conv = irfft(A * A, n=fft_size)[:size]
    return conv

def compute_lse_gradient(a, b, tau):
    """
    Compute gradient of log-sum-exp(b/tau) w.r.t. a.
    Returns gradient of length n.
    """
    # p_i = exp(b_i/tau) / sum(exp(b_k/tau))
    max_b = b.max()
    exp_b = np.exp((b - max_b) / tau)
    p = exp_b / exp_b.sum()
    
    # grad[j] = (2/tau) * sum_{m=0}^{n-1} a[m] * p[j+m]
    # = (2/tau) * (a_rev * p)[j+n-1 : 2n-1]
    n = len(a)

    conv_ap = irfft(A_rev * P, n=fft_size)
    grad = (2.0 / tau) * conv_ap[n-1 : 2*n-1]
    return grad

def softmax(v):
    """Numerically stable softmax."""
    v_max = v.max()
    e = np.exp(v - v_max)
    return e / e.sum()

def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def max_b_eval(a):

    a = a / a.sum()
    b = fast_convolve(a)

def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():

    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_safe(v):
    """Numerically stable softmax."""
    v_max = v.max()
    e = np.exp(v - v_max)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return e / s

    s = a.sum()

def lse_value_and_grad(a, tau):
    """
    Compute log-sum-exp(conv(a,a)/tau) and its gradient w.r.t. a.
    a is normalized (sum=1), a>=0.
    Returns (scalar_value, gradient_vector_of_length_n).
    """
    b = fast_convolve(a)
    n = len(a)
    max_b = b.max()
    shifted = (b - max_b) / tau
    exp_b = np.exp(shifted)
    total = exp_b.sum()
    lse_val = (max_b + tau * np.log(total)) / tau
    p = exp_b / total

def subgradient_optimize(a0, time_limit, lr_init=0.003, momentum=0.93,
                         n_top=None, seed=42, kick_size=0.002):
    """
    Subgradient optimization on simplex with Nesterov momentum.
    Optimizes max(conv(a,a)) subject to a>=0, sum(a)=1.
    Returns (best_a, best_max_b).

    n = len(a0)
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    v_mom = np.zeros(n)
    best_a = a.copy()
    best_obj = max_b_eval(a)
    cur_lr = lr_init
    no_improve = 0

    if n_top is None:
        n_top = max(10, n // 10)

    rng = np.random.default_rng(seed)

    # Precompute FFT size
    fft_size = 1 << (3 * n - 2).bit_length()

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        b_max = b.max()

        # Select active peaks
        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
        else:
            top_idx = np.arange(len(b))
        threshold = b_max * 0.999
        active_mask = b[top_idx] >= threshold
        if not active_mask.any():
            active_mask[0] = True
        active_indices = top_idx[active_mask]
        n_active = len(active_indices)

        # Subgradient: g[j] = (2/n_active) * sum_{i in active} a[i-j]
        w = np.zeros(2 * n - 1)
        w[active_indices] = 1.0 / n_active

        # g[j] = 2 * sum_i w[i] * a[i-j] for valid ranges
        # = 2 * conv(a_rev, w)[j+n-1 : 2n-1]...

        # = sum_{m=0}^{n-1} w[j+m]*a[m] = conv(a_rev, w)[j+n-1]
        a_rev = a[::-1]
        A_rev = rfft(a_rev, n=fft_size)
        W = rfft(w, n=fft_size)
        conv_aw = irfft(A_rev * W, n=fft_size)
        g = 2.0 * conv_aw[n - 1:2 * n - 1]

        # Nesterov momentum step
        v_mom = momentum * v_mom - cur_lr * g
        a_new = project_simplex(a + v_mom)

        obj_new = max_b_eval(a_new)

            best_obj = obj_new
            best_a = a_new.copy()

            v_mom = np.zeros(n)
            cur_lr = max(cur_lr * 0.5, lr_init * 0.05)
            no_improve = 0
            kick_size *= 0.7
        elif no_improve >= 1500:
            cur_lr *= 0.95

    return best_a, best_obj

    best_a = a.copy()
    best_obj = max_b_eval(a)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        M = max_b_eval(a)
        if M < best_obj * 1e-10:
            break

        # Build LP matrix: A_ub[k, j] = a[j] if 0 <= k-j <= n-1 else 0
        # Constraint k: sum_j a[j] * g[k-j] <= M  (for valid j)
        num_constraints = 2 * n - 1
        A_ub = np.zeros((num_constraints, n))
        for k in range(num_constraints):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            A_ub[k, lo:hi + 1] = a[hi - lo + 1:hi + 1][::-1]

        b_ub = np.full(num_constraints, M)
        c = -np.ones(n)

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                          bounds=[(0.0, None)] * n, method='highs',
                          options={'time_limit': 5, 'presolve': True})
            if not res.success or res.x is None:
                no_improve += 1
                if no_improve > 5:
                    break
                continue
        except Exception:
            break

        g = res.x
        g_sum = g.sum()
        if g_sum < 1e-10:
            break

        g_norm = g / g_sum

        # Line search for best blend
        best_t = 0.0
        best_blend_obj = best_obj
        t_candidates = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
        for t in t_candidates:
            blend = (1.0 - t) * best_a + t * g_norm
            blend = np.maximum(blend, 0.0)
            ss = blend.sum()
            if ss < 1e-10:
                continue
            blend = blend / ss
            obj_blend = max_b_eval(blend)
            if obj_blend < best_blend_obj:
                best_blend_obj = obj_blend
                best_t = t

        if best_t > 0.0:
            a = (1.0 - best_t) * best_a + best_t * g_norm
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            obj = max_b_eval(a)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
                if verbose and it % 20 == 0:
                    print(f"      LP iter {it}: t={best_t:.2f}, max_b={best_obj:.10f}")
            else:
                a = best_a.copy()
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 10:
            break

    return best_a, best_obj


def coordinate_descent_exact(a0, time_limit, verbose=False):
    """
    Coordinate descent with golden section 1D minimization.
    For each coordinate j, fix all others and find optimal a[j].
    Returns (best_a, best_max_b).
    """
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = max_b_eval(a)

    # Golden ratio
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    # Precompute for fast conv: we'll use the full conv each time
    # but the coordinate fix means we only need to evaluate 1D function

    improved_global = True
    pass_num = 0
    while improved_global and time.monotonic() - t0 < time_limit:
        improved_global = False
        pass_num += 1
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj

            old_val = a[j]
            if old_val < 1e-15:
                continue
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue

            # Define 1D function: f(x) = max_b when a[j] = x, others scaled
            # to keep sum = 1
            # a[j] = x, others = a[k] * (1-x)/denom for k != j

            # Search range for x: [0, 0.99]
            # Use golden section search
            lo = 0.0
            hi = 0.99

            def obj_1d(x):
                if x < 0 or x > 0.999:
                    return float('inf')
                scale = (1.0 - x) / denom
                a_tr = a.copy()
                others = a_tr * scale
                others[j] = 0.0
                a_tr = others
                a_tr[j] = x
                # a_tr should sum to 1 already (by construction)
                b = fast_convolve(a_tr)
                return float(b.max())

            # Golden section search (about 25 iterations for good precision)
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            f_c1 = obj_1d(c1)
            f_c2 = obj_1d(c2)

            for _ in range(25):
                if time.monotonic() - t0 > time_limit:
                    return best_a, best_obj
                if f_c1 < f_c2:
                    hi = c2
                    c2 = c1
                    f_c2 = f_c1
                    c1 = hi - gr * (hi - lo)
                    f_c1 = obj_1d(c1)
                else:
                    lo = c1
                    c1 = c2
                    f_c1 = f_c2
                    c2 = lo + gr * (hi - lo)
                    f_c2 = obj_1d(c2)

            x_opt = (lo + hi) / 2.0
            f_opt = min(f_c1, f_c2)

            # Compare with current
            if f_opt < best_obj - 1e-14:
                scale = (1.0 - x_opt) / denom
                a = a * scale
                a[j] = x_opt
                a = a / a.sum()
                cur_obj = max_b_eval(a)
                if cur_obj < best_obj:
                    best_obj = cur_obj
                    best_a = a.copy()
                    improved_global = True

    return best_a, best_obj


def make_gaussian_init(n, sigma):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cosine_init(n):
    x = np.linspace(0, 1, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_tukey_init(n, r):
    a = np.ones(n)
    taper = max(1, int(r * n // 2))
    if taper < n:
        t = np.linspace(0, 1, taper)
        a[:taper] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t / r - 1.0)))
        a[-taper:] = a[:taper][::-1]
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_power_init(n, alpha):
    """Power-law window: a_i proportional to (i+1)^alpha * (n-i)^alpha."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** alpha * (n - i + 1) ** alpha
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def run_optimization_pipeline(a_init, time_limit, seed=42, verbose=False):
    """
    Full optimization pipeline: LSE → SG → LP → CD → LP.
    Returns (best_a, best_max_b).
    """
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = max_b_eval(a)

    # Phase A: Log-sum-exp optimization
    t_remaining = time_limit - (time.monotonic() - t0)
    t_lse = min(t_remaining * 0.25, 120.0)
    if t_lse > 10:
        if verbose:
            print(f"    LSE phase: {t_lse:.0f}s")
        a_lse, obj_lse = adam_lse_optimize(
            a, time_limit=t_lse, tau_decay=0.7,
            min_tau_frac=1e-4, lr=0.004, seed=seed
        )
        if obj_lse < best_obj:
            best_obj = obj_lse
            best_a = a_lse.copy()

    # Phase B: Subgradient refinement
    t_remaining = time_limit - (time.monotonic() - t0)
    t_sg = min(t_remaining * 0.25, 100.0)
    if t_sg > 10:
        if verbose:
            print(f"    SG phase: {t_sg:.0f}s")
        a_sg, obj_sg = subgradient_optimize(
            best_a, time_limit=t_sg, lr_init=0.002, momentum=0.93,
            n_top=max(15, n // 12), seed=seed
        )
        if obj_sg < best_obj:
            best_obj = obj_sg
            best_a = a_sg.copy()

    # Phase C: LP refinement (aggressive)
    t_remaining = time_limit - (time.monotonic() - t0)
    t_lp = min(t_remaining * 0.35, 200.0)
    if t_lp > 10:
        if verbose:
            print(f"    LP phase: {t_lp:.0f}s")
        a_lp, obj_lp = lp_refine_aggressive(
            best_a, time_limit=t_lp, max_iters=500, verbose=verbose
        )
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()

    # Phase D: Coordinate descent
    t_remaining = time_limit - (time.monotonic() - t0)
    t_cd = min(t_remaining * 0.25, 80.0)
    if t_cd > 10:
        if verbose:
            print(f"    CD phase: {t_cd:.0f}s")
        a_cd, obj_cd = coordinate_descent_exact(
            best_a, time_limit=t_cd, verbose=verbose
        )
        if obj_cd < best_obj:
            best_obj = obj_cd
            best_a = a_cd.copy()

    # Phase E: Final LP
    t_remaining = time_limit - (time.monotonic() - t0)
    t_lp2 = min(t_remaining * 0.4, 60.0)
    if t_lp2 > 5 and best_seq_global is not None:
        pass
    if t_lp2 > 5:
        a_lp2, obj_lp2 = lp_refine_aggressive(
            best_a, time_limit=t_lp2, max_iters=200
        )
        if obj_lp2 < best_obj:
            best_obj = obj_lp2
            best_a = a_lp2.copy()

    return best_a, best_obj

def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses log-sum-exp continuation, aggressive LP, exact coordinate descent,
    and multi-length exploration.
    """
    global best_seq_global
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, budget_s - 30.0)

    best_seq = None
    best_score = float('inf')

    def update_best(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_list = (a / s).tolist()
        try:
            score = float(evaluate_sequence(a_list))
        except Exception:
            return
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a / s
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")

    # Load parent
    parent = None
    n_parent = 1152
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
            n_parent = len(parent)
    except Exception:
        pass

    if parent is None:
        parent = make_cosine_init(1152)
        n_parent = 1152
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"No parent, using cosine n={n_parent}, score={best_score:.6f}")

    # === Phase 1: Deep optimization on parent ===
    print(f"\n{'='*60}")
    print(f"Phase 1: Full pipeline on parent (n={n_parent})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic() - 600
    if rem > 50:
        t_alloc = min(rem, 400.0)
        print(f"  Budget: {t_alloc:.0f}s")
        a_opt, obj_opt = run_optimization_pipeline(
            parent, time_limit=t_alloc, seed=seed, verbose=True
        )
        update_best(a_opt, "phase1_parent")

    # === Phase 2: Diverse initializations ===
    print(f"\n{'='*60}")
    print(f"Phase 2: Diverse Initializations (best={best_score:.6f})")
    print(f"{'='*60}")
    n_base = len(best_seq) if best_seq is not None else n_parent

    inits = {}
    for sig in [0.2, 0.3, 0.4]:
        inits[f"gauss_{sig}"] = make_gaussian_init(n_base, sig)
    inits["cosine"] = make_cosine_init(n_base)
    for r in [0.3, 0.5, 0.7]:
        inits[f"tukey_{r}"] = make_tukey_init(n_base, r)
    for alpha in [0.5, 1.0, 1.5, 2.0]:
        inits[f"power_{alpha}"] = make_power_init(n_base, alpha)

    total_time_diverse = min(300.0, (deadline - time.monotonic() - 400.0) * 0.45)
    n_inits = len(inits)
    time_per_init = total_time_diverse / n_inits

    for name, init_a in inits.items():
        rem = deadline - time.monotonic() - 350
        if rem < 20:
            break
        t_alloc = min(time_per_init, rem * 0.25)
        if t_alloc < 15:
            continue
        print(f"  {name} (t={t_alloc:.0f}s)")
        a_div, obj_div = run_optimization_pipeline(
            init_a, time_limit=t_alloc,
            seed=seed + hash(name) % 100000, verbose=False
        )
        update_best(a_div, f"div_{name}")

    # === Phase 3: Multi-length exploration ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Multi-length Exploration (best={best_score:.6f})")
    print(f"{'='*60}")
    if best_seq is not None:
        target_lengths = []
        for factor in [0.75, 1.25, 1.5, 0.5, 2.0, 0.67, 1.33, 0.8, 1.2]:
            tl = int(n_parent * factor)
            tl = max(256, min(4096, tl))
            if tl not in target_lengths and tl != n_parent:
                target_lengths.append(tl)
        target_lengths.sort(key=lambda x: abs(x - n_parent))

        for target_n in target_lengths[:6]:
            rem = deadline - time.monotonic() - 200
            if rem < 30:
                break

            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.2, 60.0)
            if t_alloc < 15:
                continue
            print(f"  n={target_n}, t={t_alloc:.0f}s")
            a_len, obj_len = run_optimization_pipeline(
                f_interp, time_limit=t_alloc,
                seed=seed + target_n, verbose=False
            )
            update_best(a_len, f"len_n{target_n}")

    # === Phase 4: Deep optimization on current best ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Deep Optimization on Best (best={best_score:.6f})")

    if best_seq is not None:

    print(f"Phase 5: Final Polish (best={best_score:.6f})")

    if best_seq is not None:
        n_final = len(best_seq)
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            # Quick CD with exact search
            t_cd = min(rem * 0.4, 60.0)
            if t_cd > 10:
                a_cd, obj_cd = coordinate_descent_exact(
                    best_seq, time_limit=t_cd
                )
                update_best(a_cd, "final_cd")

            # Quick LP
            rem = deadline - time.monotonic() - 10
            if rem > 10:
                a_lp, obj_lp = lp_refine_aggressive(
                    best_seq, time_limit=min(rem, 40.0), max_iters=100
                )
                update_best(a_lp, "final_lp")

            # Tiny subgradient
            rem = deadline - time.monotonic() - 5
            if rem > 8:
                a_sg, obj_sg = subgradient_optimize(
                    best_seq, time_limit=min(rem, 30.0),
                    lr_init=0.0005, momentum=0.95,
                    n_top=max(20, n_final // 8), seed=seed + 999999
                )
                update_best(a_sg, "final_sg")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        else:
            best_seq = parent.copy()
            best_score = float(evaluate_sequence(best_seq.tolist()))
    except Exception:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
