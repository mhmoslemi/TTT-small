import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_from_a(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def compute_gradient_at_peak(a, b):
    """Compute exact subgradient of max(conv(a,a)) at the peak."""
    n = len(a)
    k_peak = int(np.argmax(b))
    grad = np.zeros(n)
    js = max(0, k_peak - n + 1)
    je = min(n - 1, k_peak)
    idx = np.arange(js, je + 1)
    grad[idx] = 2.0 * a[k_peak - idx]
    return grad, k_peak


def compute_smoothed_weights(b, T):
    """Softmax weights with temperature T."""
    scaled = b / T
    m = scaled.max()
    exp_scaled = np.exp(scaled - m)
    w = exp_scaled / exp_scaled.sum()
    return w


def compute_pressure_weights(b, T, flatten_alpha=0.0):
    """
    Combined peak + flatten weights.
    At flatten_alpha=0: pure peak (softmax on b).
    At flatten_alpha->1: uniform (flatten).
    """
    n_b = len(b)
    b_norm = b / b.max() if b.max() > 0 else np.ones(n_b) / n_b
    
    # Peak weights (concentrated on max)
    w_peak = compute_smoothed_weights(b, T)
    
    # Flatten weights (inverse - favor low values)
    # Actually, for flattening, we want to REDUCE high values and INCREASE low values
    # The gradient of var(b) w.r.t. a involves (b[k] - mean(b)) at each k
    
    if flatten_alpha > 0:
        mean_b = b.mean()
        # Flatten gradient: for each k, the contribution to increase var is (b[k] - mean)
        # We want to reduce this, so weight proportional to (b[k] - mean) if positive
        b_centered = b - mean_b
        b_centered = np.maximum(b_centered, 0)  # Only penalize above-mean
        if b_centered.max() > 0:
            b_centered_norm = b_centered / b_centered.max()
            scaled_fc = b_centered_norm / (T * 0.5)
            m_fc = scaled_fc.max()
            exp_fc = np.exp(scaled_fc - m_fc)
            w_flat = exp_fc / exp_fc.sum()
        else:
            w_flat = np.ones(n_b) / n_b
        
        w = (1 - flatten_alpha) * w_peak + flatten_alpha * w_flat
    else:
        w = w_peak
    
    return w


def compute_grad_a_from_weights(a, w):
    """Compute gradient 2 * cross_corr(w, a) where w has length 2n-1."""
    n = len(a)
    fft_size = 1 << (3 * n - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    correl = irfft(np.conj(W) * A, fft_size)
    return correl[:n]

            break

        progress = min(1.0, it / (0.8 * max_iters))
        T = T_start * (T_end / T_start) ** progress
        eta = eta_start * (eta_end / eta_start) ** progress
        flatten = flatten_start + (flatten_end - flatten_start) * progress

        b = fast_convolve(a)
        current_score = 2.0 * n * b.max()

        if current_score < best_score - 1e-15:
            best_score = current_score
            best_a = a.copy()
            stuck_count = 0
        else:
            stuck_count += 1

        w = compute_pressure_weights(b, T, flatten)
        grad_a = compute_grad_a_from_weights(a, w) * 2.0 * n

        # Project to tangent space
        grad_a = grad_a - np.dot(a, grad_a)

        # Mirror (multiplicative) update
        log_update = -eta * grad_a
        log_update = np.clip(log_update, -3.0, 3.0)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-300)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
        else:
            a = a / s

        # Stuck detection
        if stuck_count > 500:
            rng = np.random.default_rng(it * 104729)
            noise_scale = rng.uniform(0.01, 0.05)
            noise = rng.standard_normal(n) * noise_scale
            a = best_a + noise
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            stuck_count = 0

        if label and it % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} VMD iter={it} ({elapsed:.0f}s): T={T:.5f} eta={eta:.5f} "
                  f"flat={flatten:.3f} score={current_score:.7f} best={best_score:.7f}")

    return best_a, best_score

def iterative_proportional_fitting(a_init, time_limit, max_iters=50000, label=""):
    """
    IPF-like algorithm: iteratively adjust a to flatten conv(a,a).
    At each step, compute the "excess" at each convolution position and
    use it to multiplicatively update a.
    """
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()

    eta = 0.05
    decay = 0.9999

    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        current_score = 2.0 * n * b.max()

        if current_score < best_score - 1e-15:
            best_score = current_score
            best_a = a.copy()

        # Target: uniform convolution
        target = 1.0 / (2 * n - 1)
        excess = b - target  # Positive where b is too high

        # For each position i in a, compute its "pressure"
        # a[i] contributes to b[k] for k = i, i+1, ..., i+n-1 (via a[i]*a[k-i])
        # More precisely: a[i] contributes a[i]*a[j] to b[i+j] for j = 0, ..., n-1
        # So the pressure on a[i] is sum over k of excess[k] * a[k-i] (for valid k-i)
        # This is: pressure[i] = sum_j excess[i+j] * a[j] = (excess * a)[i] in convolution sense
        # Actually it's a cross-correlation

        fft_size = 1 << (3 * n - 1).bit_length()
        E = rfft(excess, fft_size)
        A = rfft(a, fft_size)
        # pressure[i] = sum_j excess[i+j] * a[j] = (excess correlated with a)[i]
        # = irfft(conj(E) * A)[i] ... let me think
        # excess_conv_a[k] = sum_i excess[i] * a[k-i]
        # We want: pressure[i] = sum_j excess[i+j] * a[j]
        # Let m = i+j, so j = m-i: pressure[i] = sum_m excess[m] * a[m-i]
        # This is (excess * a)[2i] ... no.
        # Actually: (conv(excess, a))[k] = sum_i excess[i] * a[k-i]
        # We want pressure[i] = sum_j excess[i+j] * a[j]
        # Let's use: pressure[i] = sum_j excess[i+j] * a[j] = sum_j a[j] * excess[i+j]
        # = (a * excess) reversed... 
        # Actually this is the cross-correlation of excess and a:
        # xcorr[i] = sum_j excess[i+j] * a[j] (with appropriate bounds)
        # In FFT terms: xcorr = irfft(conj(FFT(a)) * FFT(excess)) gives:
        # xcorr[l] = sum_j a[j] * excess[l+j] (circular, need to be careful with sizes)

        correl = irfft(np.conj(A) * E, fft_size)[:n]
        pressure = correl

        # Normalize pressure
        p_max = np.abs(pressure).max()
        if p_max < 1e-15:
            break
        pressure = pressure / p_max

        # Multiplicative update: reduce a where pressure is positive, increase where negative
        log_update = -eta * pressure
        log_update = np.clip(log_update, -2.0, 2.0)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-300)
        a = a / a.sum()

        eta *= decay

        if label and it % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} IPF iter={it} ({elapsed:.0f}s): eta={eta:.6f} "
                  f"score={current_score:.7f} best={best_score:.7f}")

    return best_a, best_score


def multi_scale_optimize(time_limit, initial_n=64, final_n=768, num_scales=5):
    """
    Multi-scale optimization: start with small n, optimize, upscale, repeat.
    """
    t0 = time.monotonic()
    time_per_scale = time_limit / num_scales

    # Generate scale factors
    scales = np.geomspace(initial_n, final_n, num_scales).astype(int)
    scales = np.maximum(scales, 16)

    a = np.ones(scales[0]) / scales[0]

    for s, n in enumerate(scales):
        t_start = time.monotonic()
        t_alloc = min(time_per_scale * (1 + 0.5 * s), time_limit - (time.monotonic() - t0) - 10)
        if t_alloc < 5:
            break

        if s > 0:
            a = interpolate_sequence(a, n)

        print(f"  Scale {s}: n={n}, budget={t_alloc:.0f}s")

        # Optimize at this scale
        a, sc = variance_regularized_mirror_descent(
            a, t_alloc * 0.5, T_start=0.02, T_end=0.001,
            eta_start=0.03, eta_end=0.001, flatten_start=0.0, flatten_end=0.2,
            label=f"scale{s}"
        )
        a, sc = iterative_proportional_fitting(a, t_alloc * 0.3, label=f"ipf_s{s}")

    return a


def perturb_and_optimize(a_init, time_limit, num_perturbations=10, label=""):
    """
    Create diverse perturbations of a_init and optimize each one.
    """
    n = len(a_init)
    t0 = time.monotonic()
    best_a = a_init.copy()
    best_score = score_from_a(a_init)
    time_per = time_limit / num_perturbations

    for p in range(num_perturbations):
        if time.monotonic() - t0 > time_limit - 5:
            break

        rng = np.random.default_rng(p * 7919 + 1)

        # Different perturbation strategies
        strategy = p % 4
        if strategy == 0:
            # Gaussian noise
            noise = rng.standard_normal(n) * 0.03
            a_pert = a_init + noise
        elif strategy == 1:
            # Random scaling of a subset
            a_pert = a_init.copy()
            mask = rng.random(n) < 0.3
            a_pert[mask] *= rng.uniform(0.5, 1.5, mask.sum())
        elif strategy == 2:
            # Smooth perturbation (low-frequency)
            freq = np.fft.rfftfreq(n)
            noise_freq = rng.standard_normal(len(freq)) * np.exp(-5 * freq)
            noise = np.fft.irfft(noise_freq * 2 * n) * 0.05
            a_pert = a_init + noise
        else:
            # Structured perturbation (phase shift in frequency domain)
            A = rfft(a_init, 2 * n)
            A[1:-1] *= np.exp(1j * rng.uniform(-0.1, 0.1, len(A[1:-1])))
            a_pert = irfft(A, 2 * n)[:n]

        a_pert = np.maximum(a_pert, 1e-15)
        a_pert = a_pert / a_pert.sum()

        # Quick optimization from perturbed point
        t_rem = time.monotonic() - t0
        t_alloc = min(time_per, time_limit - t_rem - 5)
        if t_alloc < 1:
            continue

        a_opt, sc = variance_regularized_mirror_descent(
            a_pert, t_alloc, T_start=0.005, T_end=0.0003,
            eta_start=0.01, eta_end=0.0005, flatten_start=0.1, flatten_end=0.3,
            max_iters=5000, label=""
        )

        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"    {label} Perturb {p} (strat={strategy}): new best {sc:.7f}")

    return best_a, best_score


def gradient_descent_momentum(a_init, time_limit, lr_start=0.001, lr_end=0.0001,
                               momentum=0.9, T=0.001, max_iters=30000, label=""):
    """
    Gradient descent with momentum on the probability simplex.
    Uses projected gradient: project back to simplex after each step.
    """
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    v = np.zeros(n)  # Momentum
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()

    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, it / (0.8 * max_iters))
        lr = lr_start * (lr_end / lr_start) ** progress

        b = fast_convolve(a)
        current_score = 2.0 * n * b.max()

        if current_score < best_score - 1e-15:
            best_score = current_score
            best_a = a.copy()

        # Compute gradient
        w = compute_smoothed_weights(b, T)
        grad = compute_grad_a_from_weights(a, w) * 2.0 * n

        # Project to tangent space
        grad = grad - np.dot(a, grad)

        # Momentum update
        v = momentum * v + (1 - momentum) * grad
        a = a - lr * v
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        if label and it % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} GD-M iter={it} ({elapsed:.0f}s): lr={lr:.6f} "
                  f"score={current_score:.7f} best={best_score:.7f}")

    return best_a, best_score


def subgradient_momentum(a_init, time_limit, lr_start=0.0005, lr_end=0.00001,
                          momentum=0.9, max_iters=50000, label=""):
    """
    Subgradient method with momentum using exact peak subgradient.
    """
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    v = np.zeros(n)
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()

    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, it / (0.8 * max_iters))
        lr = lr_start * (lr_end / lr_start) ** progress

        b = fast_convolve(a)
        current_score = 2.0 * n * b.max()

        if current_score < best_score - 1e-15:
            best_score = current_score
            best_a = a.copy()

        # Exact subgradient at peak
        k_peak = int(np.argmax(b))
        grad = np.zeros(n)
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        grad[idx] = 2.0 * a[k_peak - idx]

        # Project to tangent space
        grad = grad - np.dot(a, grad)

        # Momentum
        v = momentum * v + (1 - momentum) * grad
        a = a - lr * v
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

    return best_a, best_score


def topp_k_subgradient(a_init, time_limit, top_k=10, lr_start=0.001, lr_end=0.0001,
                        momentum=0.9, max_iters=30000, label=""):
    """
    Subgradient method using the top-k peaks (not just the single peak).
    This gives a richer gradient signal.
    """
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    v = np.zeros(n)
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()

    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, it / (0.8 * max_iters))
        lr = lr_start * (lr_end / lr_start) ** progress

        b = fast_convolve(a)
        current_score = 2.0 * n * b.max()

        if current_score < best_score - 1e-15:
            best_score = current_score
            best_a = a.copy()

        # Top-k peaks
        b_sorted = np.sort(b)[::-1]
        threshold = b_sorted[min(top_k, len(b_sorted)) - 1]
        peak_indices = np.where(b >= threshold * 0.99)[0]

        # Weighted subgradient from top peaks
        b_peaks = b[peak_indices]
        weights = b_peaks / b_peaks.sum()

        grad = np.zeros(n)
        for k_p in peak_indices:
            ws = weights[peak_indices == k_p][0]
            js = max(0, int(k_p) - n + 1)
            je = min(n - 1, int(k_p))
            idx = np.arange(js, je + 1)
            grad[idx] += ws * 2.0 * a[int(k_p) - idx]

        # Project to tangent space
        grad = grad - np.dot(a, grad)

        # Momentum
        v = momentum * v + (1 - momentum) * grad
        a = a - lr * v
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        if label and it % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} TopK iter={it} ({elapsed:.0f}s): lr={lr:.6f} "
                  f"score={current_score:.7f} best={best_score:.7f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate using the actual evaluate_sequence function."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.7f} (n={len(a_norm)})")
    return best_seq, best_score


def final_polish(a_init, time_limit, label=""):
    """
    Final polish: alternating between very low temperature mirror descent
    and exact subgradient with small steps.
    """
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()

    # Phase A: Very low temperature MD
    t_alloc_a = time_limit * 0.4
    a, sc = variance_regularized_mirror_descent(
        a, t_alloc_a, T_start=0.001, T_end=0.00005,
        eta_start=0.003, eta_end=0.0001, flatten_start=0.2, flatten_end=0.4,
        max_iters=20000, label="polish_A"
    )
    best_a = a.copy() if sc < best_score else best_a
    best_score = min(sc, best_score)

    # Phase B: Exact subgradient with tiny steps
    t_alloc_b = min(time_limit * 0.35, time_limit - (time.monotonic() - t0) - 5)
    a, sc = subgradient_momentum(
        best_a, t_alloc_b, lr_start=0.0001, lr_end=0.000001,
        momentum=0.95, max_iters=50000, label="polish_B"
    )
    if sc < best_score:
        best_a = a.copy()
        best_score = sc

    # Phase C: Top-k subgradient
    t_alloc_c = min(time_limit * 0.2, time_limit - (time.monotonic() - t0) - 3)
    if t_alloc_c > 2:
        a, sc = topp_k_subgradient(
            best_a, t_alloc_c, top_k=5, lr_start=0.0002, lr_end=0.00001,
            momentum=0.9, max_iters=20000, label="polish_C"
        )
        if sc < best_score:
            best_a = a.copy()
            best_score = sc

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization with novel techniques:
    1. Variance-regularized mirror descent (convolution flattening)
    2. Iterative proportional fitting
    3. Multi-scale optimization
    4. Diverse perturbation + local search
    5. Top-k subgradient with momentum
    6. Final polish with ultra-low temperatures
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 12.0)
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.7f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = p_score

    # === Phase 1: Variance-Regularized Mirror Descent ===
    print(f"\n=== Phase 1: Variance-Regularized MD (best={best_score:.7f}) ===")
    if best_seq is not None:
        configs = [
            (0.01, 0.0005, 0.02, 0.001, 0.0, 0.2, "VMD1"),
            (0.005, 0.0002, 0.01, 0.0005, 0.1, 0.3, "VMD2"),
            (0.003, 0.0001, 0.005, 0.0002, 0.2, 0.4, "VMD3"),
        ]
        for T_s, T_e, eta_s, eta_e, fl_s, fl_e, lbl in configs:
            rem = deadline - time.monotonic() - 300
            if rem < 30:
                break
            t_alloc = min(rem * 0.3, 180.0)
            print(f"  {lbl}: T=[{T_s},{T_e}], eta=[{eta_s},{eta_e}], flat=[{fl_s},{fl_e}], budget={t_alloc:.0f}s")
            a_vmd, _ = variance_regularized_mirror_descent(

                eta_start=eta_s, eta_end=eta_e,
                flatten_start=fl_s, flatten_end=fl_e, label=lbl
            )
            best_seq, best_score = check_and_update(a_vmd, lbl, best_seq, best_score)

    # === Phase 2: Iterative Proportional Fitting ===
    print(f"\n=== Phase 2: IPF (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 250

            t_alloc = min(rem * 0.4, 150.0)
            print(f"  IPF: budget={t_alloc:.0f}s")

    # === Phase 3: Top-k Subgradient with Momentum ===
    print(f"\n=== Phase 3: Top-K Subgradient (best={best_score:.7f}) ===")
    if best_seq is not None:
        for top_k in [5, 10, 20]:
            rem = deadline - time.monotonic() - 200
            if rem < 20:
                break
            t_alloc = min(rem * 0.3, 100.0)
            print(f"  TopK={top_k}: budget={t_alloc:.0f}s")
            a_tk, _ = topp_k_subgradient(
                best_seq, t_alloc, top_k=top_k,
                lr_start=0.001, lr_end=0.0001, momentum=0.9
            )
            best_seq, best_score = check_and_update(a_tk, f"topk{top_k}", best_seq, best_score)

    # === Phase 4: Diverse Perturbation + Local Search ===
    print(f"\n=== Phase 4: Perturbation + Local Search (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 30:

    print(f"\n=== Phase 6: Final Polish (best={best_score:.7f}) ===")

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Variance-regularized mirror descent (convolution flattening via softmax + flatten blend)
2. Iterative Proportional Fitting (IPF) targeting uniform convolution
3. Top-k peak subgradient methods with momentum
4. Diverse structured perturbations + local search
5. Multi-resolution optimization (small-n to large-n)
6. Window-function initializations for alternative starting points
7. Ultra-low-temperature final polish
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_from_a(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def compute_smoothed_weights(b, T):
    """Softmax weights with temperature T."""
    scaled = b / T
    m = scaled.max()
    exp_scaled = np.exp(scaled - m)
    s = exp_scaled.sum()
    if s < 1e-300:
        return np.ones(len(b)) / len(b)
    return exp_scaled / s


def compute_pressure_weights(b, T, flatten_alpha=0.0):
    """
    Combined peak + flatten weights.
    At flatten_alpha=0: pure peak softmax.
    At flatten_alpha->1: emphasizes above-mean values (flattening).
    """
    n_b = len(b)
    w_peak = compute_smoothed_weights(b, T)

    if flatten_alpha > 0.001:
        mean_b = b.mean()
        b_centered = np.maximum(b - mean_b, 0)
        if b_centered.max() > 1e-15:
            T_flat = max(T * 0.5, 1e-10)
            scaled_fc = b_centered / T_flat
            m_fc = scaled_fc.max()
            exp_fc = np.exp(scaled_fc - m_fc)
            s_fc = exp_fc.sum()
            if s_fc > 1e-300:
                w_flat = exp_fc / s_fc
            else:
                w_flat = np.ones(n_b) / n_b
        else:
            w_flat = np.ones(n_b) / n_b
        w = (1.0 - flatten_alpha) * w_peak + flatten_alpha * w_flat
    else:
        w = w_peak

    return w


def compute_grad_a_from_weights(a, w):
    """Compute cross-correlation gradient: grad[i] = sum_j w[i+j]*a[j]."""
    n = len(a)
    fft_size = 1 << (3 * n - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    correl = irfft(np.conj(A) * W, fft_size)
    return correl[:n]


def variance_regularized_md(a_init, time_limit, T_start=0.01, T_end=0.0001,
                            eta_start=0.02, eta_end=0.0003,
                            flatten_start=0.0, flatten_end=0.3,
                            max_iters=50000, label=""):
    """Mirror descent with variance regularization for convolution flattening."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    stuck_count = 0

    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, it / (0.8 * max_iters))
        T = T_start * (T_end / T_start) ** progress
        eta = eta_start * (eta_end / eta_start) ** progress
        flatten = flatten_start + (flatten_end - flatten_start) * progress

        b = fast_convolve(a)
        current_score = 2.0 * n * b.max()

        if current_score < best_score - 1e-15:
            best_score = current_score
            best_a = a.copy()
            stuck_count = 0
        else:
            stuck_count += 1

        w = compute_pressure_weights(b, T, flatten)
        grad_a = compute_grad_a_from_weights(a, w) * 2.0 * n
        grad_a = grad_a - np.dot(a, grad_a)

        log_update = -eta * grad_a
        log_update = np.clip(log_update, -3.0, 3.0)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-300)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
        else:
            a = a / s

        if stuck_count > 500:
            rng = np.random.default_rng(it * 104729 + 7)
            noise_scale = rng.uniform(0.01, 0.05)
            noise = rng.standard_normal(n) * noise_scale
            a = best_a + noise
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            stuck_count = 0

        if label and it % 15000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} VMD iter={it} ({elapsed:.0f}s): T={T:.5f} eta={eta:.5f} "
                  f"flat={flatten:.3f} score={current_score:.7f} best={best_score:.7f}")

    return best_a, best_score


def iterative_proportional_fitting(a_init, time_limit, max_iters=50000, label=""):
    """
    IPF: iteratively adjust a to flatten conv(a,a) toward uniform.
    Uses cross-correlation to compute pressure on each a[i].
    """
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()

    eta = 0.06
    decay = 0.9998
    fft_size = 1 << (3 * n - 1).bit_length()

    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        current_score = 2.0 * n * b.max()

        if current_score < best_score - 1e-15:
            best_score = current_score
            best_a = a.copy()

        target = 1.0 / (2 * n - 1)
        excess = b - target

        E = rfft(excess, fft_size)
        A = rfft(a, fft_size)
        pressure = irfft(np.conj(A) * E, fft_size)[:n]

        p_max = np.abs(pressure).max()
        if p_max < 1e-15:
            break
        pressure = pressure / p_max

        log_update = -eta * pressure
        log_update = np.clip(log_update, -2.5, 2.5)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-300)
        a = a / a.sum()

        eta *= decay

        if label and it % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} IPF iter={it} ({elapsed:.0f}s): eta={eta:.6f} "
                  f"score={current_score:.7f} best={best_score:.7f}")

    return best_a, best_score


def topp_k_subgradient(a_init, time_limit, top_k=10, lr_start=0.001, lr_end=0.00005,
                       momentum=0.9, max_iters=30000, label=""):
    """Subgradient method using top-k peaks with momentum."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    v = np.zeros(n)
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()

    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, it / (0.8 * max_iters))
        lr = lr_start * (lr_end / lr_start) ** progress

        b = fast_convolve(a)
        current_score = 2.0 * n * b.max()

        if current_score < best_score - 1e-15:
            best_score = current_score
            best_a = a.copy()

        # Get top-k peak indices
        b_copy = b.copy()
        n_peaks = min(top_k, len(b_copy))
        peak_indices = np.argsort(b_copy)[::-1][:n_peaks]

        # Weight by peak value
        b_peaks = b[peak_indices]
        w_sum = b_peaks.sum()
        if w_sum < 1e-15:
            break
        weights = b_peaks / w_sum

        grad = np.zeros(n)
        for idx_i, k_p in enumerate(peak_indices):
            k_p = int(k_p)
            w_i = weights[idx_i]
            js = max(0, k_p - n + 1)
            je = min(n - 1, k_p)
            idx = np.arange(js, je + 1)
            grad[idx] += w_i * 2.0 * a[k_p - idx]

        grad = grad - np.dot(a, grad)
        v = momentum * v + (1.0 - momentum) * grad
        a = a - lr * v
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        if label and it % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} TopK iter={it} ({elapsed:.0f}s): lr={lr:.7f} "
                  f"score={current_score:.7f} best={best_score:.7f}")

    return best_a, best_score


def exact_subgradient_momentum(a_init, time_limit, lr_start=0.0005, lr_end=0.000005,
                               momentum=0.92, max_iters=50000, label=""):
    """Exact subgradient at single peak with momentum."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    v = np.zeros(n)
    best_a = a.copy()

        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, it / (0.8 * max_iters))
        lr = lr_start * (lr_end / lr_start) ** progress

        b = fast_convolve(a)
        current_score = 2.0 * n * b.max()

        if current_score < best_score - 1e-15:
            best_score = current_score
            best_a = a.copy()

        k_peak = int(np.argmax(b))
        grad = np.zeros(n)
        js = max(0, k_peak - n + 1)

        idx = np.arange(js, je + 1)

        grad = grad - np.dot(a, grad)
        v = momentum * v + (1.0 - momentum) * grad

def window_initializations(n, seed=42):
    """Generate various window function initializations."""
    rng = np.random.default_rng(seed)
    candidates = []

    # Uniform
    candidates.append(np.ones(n))

    # Hann
    x = np.linspace(0, 2 * np.pi, n)
    candidates.append(0.5 - 0.5 * np.cos(x))

    # Raised cosine (different)
    t = np.linspace(0, 1, n)
    candidates.append(0.5 + 0.5 * np.cos(2 * np.pi * t))

    # Triangular
    candidates.append(1.0 - 2.0 * np.abs(t - 0.5))

    # Exponential decay
    candidates.append(np.exp(-3.0 * t))

    # Symmetric exponential
    candidates.append(np.exp(-3.0 * np.abs(t - 0.5)))

    # Gaussian
    candidates.append(np.exp(-8.0 * (t - 0.5) ** 2))

    # Bump (compact support in middle)
    candidates.append(np.exp(-20.0 * (t - 0.5) ** 4))

    # Random smooth
    freq = np.fft.rfftfreq(n)
    noise_freq = rng.standard_normal(len(freq)) * np.exp(-4.0 * freq * n / 10)
    smooth = np.fft.irfft(noise_freq * 2 * n)
    candidates.append(np.abs(smooth) + 0.1)

    # Two-lobe
    candidates.append(np.exp(-15.0 * (t - 0.3) ** 2) + np.exp(-15.0 * (t - 0.7) ** 2))

    # Three-lobe
    candidates.append(np.exp(-20.0 * (t - 0.2) ** 2) + np.exp(-20.0 * (t - 0.5) ** 2) + np.exp(-20.0 * (t - 0.8) ** 2))

    # x^(1/3) shaped
    candidates.append(t ** 0.3 + (1 - t) ** 0.3)

    # 1/(1+t^4) shaped
    candidates.append(1.0 / (1.0 + (4.0 * t - 2.0) ** 4))

    results = []
    for c in candidates:
        c = np.maximum(c, 0)
        s = c.sum()
        if s > 1e-10:
            results.append(c / s)
    return results

    t_rem = time_limit - (time.monotonic() - t0) - 2

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate using the actual evaluate_sequence function."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.7f} (n={len(a_norm)})")
    return best_seq, best_score
