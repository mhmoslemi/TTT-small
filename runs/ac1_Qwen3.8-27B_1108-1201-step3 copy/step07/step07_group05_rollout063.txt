def compute_smooth_and_H(a, include_zero, include_edge, T):
    n=len(a)
    sel=make_selector(n, include_zero, include_edge)
    H=np.zeros(2*n-1, dtype=np.float64) if n>0 else np.zeros(0)
    if sel.size == 0:
        return 0.0, H, 0.0
    c=autocorr_lags(a)
    v=c[sel]
    vmax=float(v.max())
    if vmax <= 0:
        return 0.0, H, 0.0
    shifted = T * (v - vmax)
    shifted = np.minimum(shifted, 0.0)
    w = np.exp(shifted)
    ws = float(w.sum())
    if ws <= 0 or not np.isfinite(ws):
        # uniform weights
        w = np.ones(v.size, dtype=np.float64) / v.size
        ws = 1.0
    else:
        w = w / ws
    smooth = vmax + np.log(ws) / T if T > 0 else vmax
    score = 2.0 * n * smooth
    exact = 2.0 * n * vmax
    for i, m in enumerate(sel):
        wm = float(w[i])
        if wm <= 0: continue
        if m == 0:
            H[n-1] += 2.0 * wm
        else:
            H[n-1 - m] += wm
            H[n-1 + m] += wm
    return score, H, exact
```

```python
"""
Search algorithm for minimizing a peak-convolution objective.

The program assumes the true scoring metric is a peak of the (possibly
center-excluded and/or edge-excluded) linear auto-convolution of a nonnegative
nonzero-measure-vector sequence.  It begins from the supplied parent sequence,
detects which lags matter via a few black-box evaluations, and then combines:
  - L-BFGS-B optimization of a soft-max smoothed peak objective in log/softmax space
  - linear-program best-response / fictitious-play style moves
  - exact top-peak descent
  - interpolation to nearby lengths
  - short direct local search using the actual evaluator

All helper functions are top-level and no lambda functions are used.
"""

import time
import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog

RUN_DEADLINE = 0.0
RUN_BEST_Z = None
RUN_BEST_SCORE = float('inf')
SELECTOR_CACHE = {}


def infer_metric_flags():
    """Detect whether the hidden evaluator appears to include lag 0 and/or the outer lag."""
    test_n = 96
    include_zero = False
    include_edge = True
    try:
        sp = np.zeros(test_n, dtype=np.float64)
        sp[test_n // 2] = 1.0
        sp_score = float(evaluate_sequence(sp.tolist()))

        adj = np.zeros(test_n, dtype=np.float64)
        adj[test_n // 2 - 1] = 0.5
        adj[test_n // 2] = 0.5
        adj_score = float(evaluate_sequence(adj.tolist()))

        edge = np.zeros(test_n, dtype=np.float64)
        edge[0] = 0.5
        edge[test_n - 1] = 0.5
        edge_score = float(evaluate_sequence(edge.tolist()))

        print("Metric probe: spike=%.6f adjacent=%.6f edge=%.6f" % (sp_score, adj_score, edge_score))

        full_spike = 2.0 * test_n
        if np.isfinite(sp_score):
            if sp_score > 0.40 * full_spike:
                include_zero = True
            else:
                include_zero = False
                if np.isfinite(adj_score) and adj_score > 1e-3:
                    if np.isfinite(edge_score):
                        include_edge = float(edge_score) > 0.25 * float(adj_score)
                    else:
                        include_edge = True
                else:
                    include_edge = True
        else:
            include_zero = False
            include_edge = True
    except Exception as exc:
        print("Metric probe failed:", exc)
        include_zero = False
        include_edge = True

    return include_zero, include_edge


def make_selector(n, include_zero, include_edge):
    """Return the lags m in [0, n-1] counted by the chosen surrogate metric."""
    key = (int(n), bool(include_zero), bool(include_edge))
    if key in SELECTOR_CACHE:
        return SELECTOR_CACHE[key]

    if n <= 0:
        sel = np.array([], dtype=int)
    elif include_zero:
        sel = np.arange(n, dtype=int)
    else:
        if include_edge:
            if n == 1:
                sel = np.array([], dtype=int)
            else:
                sel = np.arange(1, n, dtype=int)
        else:
            if n <= 2:
                sel = np.array([], dtype=int)
            else:
                sel = np.arange(1, n - 1, dtype=int)

    SELECTOR_CACHE[key] = sel
    return sel


def softmax_from_z(z):
    """Stable softmax."""
    z = np.asarray(z, dtype=np.float64)
    if z.size == 0:
        return z.copy()
    zmax = float(np.max(z))
    if not np.isfinite(zmax):
        return np.ones(z.size, dtype=np.float64) / float(z.size)
    ez = np.exp(z - zmax)
    s = float(np.sum(ez))
    if s <= 1e-300 or not np.isfinite(s):
        return np.ones(z.size, dtype=np.float64) / float(z.size)
    return ez / s


def z_from_a(a):
    """Log-domain coordinates for a nonnegative vector."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 1e-30)
    return np.log(a)


def fast_autoconv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def autocorr_lags(a):
    """
    Return c[m] for m=0..n-1, where
    c[0] = sum a_i^2 and c[m] = sum_i a_i a_{i-m} for m>0.
    """
    n = len(a)
    c = np.zeros(n, dtype=np.float64)
    if n == 0:
        return c

    conv = fast_autoconv(a)
    c[0] = conv[n - 1]
    if n > 1:
        idx = np.arange(1, n, dtype=int)
        c[1:] = 0.5 * (conv[n - 1 - idx] + conv[n - 1 + idx])
    np.maximum(c, 0.0, out=c)
    return c


def score_from_c(c, n, include_zero, include_edge):
    """Peak surrogate score from an autocorr_lags array."""
    if n <= 0:
        return 0.0
    sel = make_selector(n, include_zero, include_edge)
    if sel.size == 0:
        return 0.0
    vals = c[sel]
    if vals.size == 0:
        return 0.0
    mx = float(np.max(vals))
    if not np.isfinite(mx) or mx < 0.0:
        mx = 0.0
    return 2.0 * float(n) * mx


def exact_conv_score(a, include_zero, include_edge):
    """Evaluate the internal surrogate objective for a normalized vector."""
    return score_from_c(autocorr_lags(a), len(a), include_zero, include_edge)


def compute_smooth_and_H(a, include_zero, include_edge, T):
    """
    Compute the soft-max smoothed peak objective and the gradient kernel H.
    H is arranged centered at index n-1 so that
        G_i = sum_d H_{center+d} a_{i-d}.
    """
    n = len(a)
    if n == 0:
        return 0.0, np.zeros(0, dtype=np.float64), 0.0

    sel = make_selector(n, include_zero, include_edge)
    H = np.zeros(2 * n - 1, dtype=np.float64)

    if sel.size == 0:
        return 0.0, H, 0.0

    c = autocorr_lags(a)
    v = c[sel]
    if v.size == 0:
        return 0.0, H, 0.0

    vmax = float(np.max(v))
    if not np.isfinite(vmax) or vmax <= 0.0:
        return 0.0, H, 0.0

    if T <= 0.0:
        idxmax = int(np.argmax(v))
        w = np.zeros(v.size, dtype=np.float64)
        w[idxmax] = 1.0
        smooth = vmax
    else:
        shifted = T * (v - vmax)
        shifted = np.minimum(shifted, 0.0)
        w = np.exp(shifted)
        ws = float(np.sum(w))
        if ws <= 0.0 or not np.isfinite(ws):
            w = np.full(v.size, 1.0 / float(v.size), dtype=np.float64)
            ws = 1.0
        else:
            w = w / ws
        smooth = vmax + float(np.log(ws)) / float(T)

    score = 2.0 * float(n) * float(smooth)
    exact = 2.0 * float(n) * vmax

    zero_mask = sel == 0
    if np.any(zero_mask):
        H[n - 1] = 2.0 * float(np.sum(w[zero_mask]))

    pos_mask = sel > 0
    if np.any(pos_mask):
        pm = sel[pos_mask]
        pw = w[pos_mask]
        H[n - 1 + pm] = pw
        H[n - 1 - pm] = pw

    return float(score), H, float(exact)


def convolve_kernel(a, H):
    """Linear convolution a * H via FFT."""
    n = len(a)
    hlen = len(H)
    if n == 0 or hlen == 0:
        return np.array([], dtype=np.float64)
    size = n + hlen - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    Hr = rfft(H, fft_size)
    return irfft(A * Hr, fft_size)[:size]


def objective_lbgfs(z, T, n, include_zero, include_edge):
    """Soft-max smoothed objective for L-BFGS in log-space."""
    global RUN_BEST_Z, RUN_BEST_SCORE

    if time.monotonic() > RUN_DEADLINE:
        raise TimeoutError

    a = softmax_from_z(z)
    smooth_score, H, exact_score = compute_smooth_and_H(a, include_zero, include_edge, T)

    if np.isfinite(exact_score) and exact_score < RUN_BEST_SCORE:
        RUN_BEST_SCORE = float(exact_score)
        RUN_BEST_Z = z.copy()

    if not np.isfinite(smooth_score):
        return 1e30
    return float(smooth_score)


def gradient_lbgfs(z, T, n, include_zero, include_edge):
    """Gradient of the soft-max smoothed objective in log-space."""
    if time.monotonic() > RUN_DEADLINE:
        raise TimeoutError

    a = softmax_from_z(z)
    _, H, _ = compute_smooth_and_H(a, include_zero, include_edge, T)

    if H.size == 0:
        return np.zeros(n, dtype=np.float64)

    full = convolve_kernel(a, H)
    G = full[n - 1:n - 1 + n]

    if G.size != n:
        return np.zeros(n, dtype=np.float64)

    S = float(np.dot(G, a))
    grad = a * (G - S)
    grad[~np.isfinite(grad)] = 0.0
    return grad


def run_lbgfs(a, include_zero, include_edge, T, time_limit, maxiter=None, maxfun=None):
    """Run L-BFGS-B on a soft-max smoothed peak objective."""
    n = len(a)
    if n <= 1:
        return a.copy(), exact_conv_score(a, include_zero, include_edge)

    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s <= 1e-15:
        return a, float('inf')
    a = a / s

    z = z_from_a(a)
    z = np.clip(z, -45.0, 45.0)

    global RUN_BEST_Z, RUN_BEST_SCORE, RUN_DEADLINE
    RUN_BEST_Z = z.copy()
    RUN_BEST_SCORE = exact_conv_score(a, include_zero, include_edge)
    RUN_DEADLINE = time.monotonic() + max(0.1, float(time_limit))

    if maxiter is None:
        maxiter = max(300, int(float(time_limit) * 20.0))
        maxiter = min(5000, maxiter)
    if maxfun is None:
        maxfun = max(1000, int(float(time_limit) * 300.0) + 10 * maxiter)
        maxfun = min(200000, maxfun)

    bounds = [(-45.0, 45.0)] * n
    options = {
        'maxiter': maxiter,
        'maxfun': maxfun,
        'ftol': 1e-18,
        'gtol': 1e-10,
        'disp': False,
    }

    out_z = None
    try:
        res = minimize(
            fun=objective_lbgfs,
            x0=z,
            jac=gradient_lbgfs,
            method='L-BFGS-B',
            args=(T, n, include_zero, include_edge),
            bounds=bounds,
            options=options,
        )
        if res is not None and res.x is not None:
            out_z = res.x
    except TimeoutError:
        out_z = None
    except Exception:
        out_z = None

    best_z = RUN_BEST_Z if RUN_BEST_Z is not None else z.copy()
    best_score = RUN_BEST_SCORE

    if out_z is not None:
        a_out = softmax_from_z(out_z)
        sc = exact_conv_score(a_out, include_zero, include_edge)
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_z = out_z.copy()

    a_best = softmax_from_z(best_z)
    a_best = np.maximum(a_best, 0.0)
    ss = float(np.sum(a_best))
    if ss > 1e-15:
        a_best = a_best / ss

    return a_best, best_score


def polish_lbfgs(a, include_zero, include_edge, time_budget):
    """Polish a sequence using several L-BFGS temperatures."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))

        return a, float('inf')

    best_score = exact_conv_score(a, include_zero, include_edge)

    elif time_budget >= 100.0:
        temps = [5000.0, 50000.0, 200000.0, 1000000.0]
    else:
        temps = [20000.0, 100000.0, 500000.0]

    nt = len(temps)

        rem = float(time_budget) - elapsed
        if rem < 5.0:
            break
        alloc = rem / float(nt - i)
        if alloc > 130.0:
            alloc = max(5.0, min(130.0, 0.5 * rem))

        a2, s2 = run_lbgfs(best, include_zero, include_edge, T, alloc)
        if np.isfinite(s2) and s2 < best_score - 1e-13:
            best = a2
            best_score = s2

    return best, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length while preserving nonnegativity and unit sum."""
    f = np.asarray(f, dtype=np.float64)
    old_n = len(f)
    if old_n <= 0:
        return np.ones(new_n, dtype=np.float64) / float(max(1, new_n))

    s = float(np.sum(f))
    if s <= 1e-15:
        return np.ones(new_n, dtype=np.float64) / float(max(1, new_n))

    if old_n == new_n:
        return f / s

    fn = f / s
    if old_n == 1:
        out = np.full(new_n, float(fn[0]), dtype=np.float64)
    else:
        old_x = np.linspace(0.0, 1.0, old_n)
        new_x = np.linspace(0.0, 1.0, new_n)
        out = np.interp(new_x, old_x, fn)

    out = np.maximum(out, 0.0)
    ss = float(np.sum(out))
    if ss > 1e-15:
        out = out / ss
    else:
        out = np.ones(new_n, dtype=np.float64) / float(max(1, new_n))
    return out


def build_uniform_H_for_lags(n, m_list):
    """Build gradient kernel for a uniform set of selected lags."""
    if n <= 0:
        return np.zeros(0, dtype=np.float64)
    H = np.zeros(2 * n - 1, dtype=np.float64)
    if len(m_list) == 0:
        return H
    w = 1.0 / float(len(m_list))
    for m in m_list:
        m = int(m)
        if m == 0:
            H[n - 1] += 2.0 * w
        else:
            H[n - 1 - m] += w
            H[n - 1 + m] += w
    return H


def peak_refine(a0, include_zero, include_edge, time_budget, K=8):
    """Steepest/log-direction descent focused on current top-peak lags."""
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s <= 1e-15:
        return a, float('inf')
    a = a / s

    sel = make_selector(n, include_zero, include_edge)
    if sel.size == 0:
        return a, 0.0

    best = a.copy()
    best_score = exact_conv_score(a, include_zero, include_edge)
    if not np.isfinite(best_score):
        best_score = 0.0

    z = z_from_a(a)
    z = np.clip(z, -45.0, 45.0)
    rng = np.random.default_rng(123456)
    etas = [0.005, 0.02, 0.05, 0.1, 0.3, 1.0, 3.0, 10.0]
    deadline = time.monotonic() + max(1.0, float(time_budget))
    no_improve = 0
    k_cur = max(2, int(K))

    while time.monotonic() < deadline:
        c = autocorr_lags(a)
        v = c[sel]
        if v.size == 0:
            break

        if k_cur >= v.size:
            top_idx = np.arange(v.size, dtype=int)
        else:
            top_idx = np.argpartition(-v, k_cur)[:k_cur]

        m_list = sel[top_idx]
        H = build_uniform_H_for_lags(n, m_list)
        full = convolve_kernel(a, H)
        G = full[n - 1:n - 1 + n]

        if G.size != n:
            break

        meanG = float(np.dot(G, a))
        d = -(G - meanG)
        scale = float(np.linalg.norm(d))
        if scale < 1e-14:
            z = np.clip(z + 0.01 * rng.standard_normal(n), -45.0, 45.0)
            a = softmax_from_z(z)
            no_improve += 1
            continue

        d = d / scale
        best_eta = 0.0
        best_local = best_score
        best_b = None
        best_c = None

        for eta in etas:
            z_new = z + eta * d
            b = softmax_from_z(z_new)
            cb = autocorr_lags(b)
            sc = score_from_c(cb, n, include_zero, include_edge)
            if np.isfinite(sc) and sc < best_local - 1e-13:
                best_local = sc
                best_eta = float(eta)
                best_b = b
                best_c = cb

        if best_eta > 0.0 and best_b is not None:
            z = z + best_eta * d
            a = best_b
            best = a.copy()
            best_score = best_local
            no_improve = 0
        else:
            no_improve += 1
            z = np.clip(z + 0.005 * rng.standard_normal(n), -45.0, 45.0)
            a = softmax_from_z(z)

            if no_improve % 20 == 0:
                k_cur = max(2, k_cur // 2)
            if k_cur <= 2 and no_improve > 60:
                break

    return best, best_score


def build_lp_matrix(a, sel):
    """Build A[g, m] = sum_j (a_{j+m}g_j + a_{j-m}g_j) for selected lags."""
    n = len(a)
    rows = len(sel)
    A = np.zeros((rows, n), dtype=np.float64)
    for r, m in enumerate(sel):
        m = int(m)
        if m == 0:
            A[r, :] = 2.0 * a
        else:
            A[r, :n - m] += a[m:]
            A[r, m:] += a[:n - m]
    return A


def fallback_best_response(a, include_zero, include_edge):
    """Soft fallback best-response if LP is unavailable or fails."""
    n = len(a)
    sel = make_selector(n, include_zero, include_edge)
    if sel.size == 0:
        return a.copy(), 0.0

    H = np.zeros(2 * n - 1, dtype=np.float64)
    for m in sel:
        m = int(m)
        if m == 0:
            H[n - 1] += 2.0
        else:
            H[n - 1 - m] += 1.0
            H[n - 1 + m] += 1.0

    full = convolve_kernel(a, H)
    G = full[n - 1:n - 1 + n]
    G = np.maximum(G, 0.0)

    gmin = float(np.min(G))
    gmax = float(np.max(G))
    if gmax - gmin > 1e-12:
        z = -5.0 * (G - gmin) / (gmax - gmin)
    else:
        z = np.zeros(n, dtype=np.float64)

    g = softmax_from_z(z)
    return g, 0.0


def solve_best_response_lp(a, include_zero, include_edge, per_solve_time=5.0):
    """
    Solve the linear best-response problem:
        min t
        s.t. A g <= t, sum g = 1, g >= 0.
    """
    n = len(a)
    sel = make_selector(n, include_zero, include_edge)
    if sel.size == 0:
        return a.copy(), 0.0

    if n > 1800:
        return fallback_best_response(a, include_zero, include_edge)

    A = build_lp_matrix(a, sel)
    rows = len(sel)

    c = np.zeros(rows + n, dtype=np.float64)
    c[0] = 1.0

    A_ub = np.zeros((rows, rows + n), dtype=np.float64)
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = A
    b_ub = np.zeros(rows, dtype=np.float64)

    A_eq = np.zeros((1, rows + n), dtype=np.float64)
    A_eq[0, 1:] = 1.0
    b_eq = np.array([1.0], dtype=np.float64)

    bounds = [(0.0, None)] * (rows + n)
    options = {'presolve': True}
    if per_solve_time > 0:
        options['time_limit'] = float(per_solve_time)

    try:
        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method='highs',
            options=options,
        )
        if res is not None and res.success and res.x is not None:
            g = res.x[1:n + 1].copy()
            g = np.maximum(g, 0.0)
            gs = float(np.sum(g))
            if gs > 1e-12:
                g = g / gs
                return g, float(res.x[0])
    except Exception:
        pass

    return fallback_best_response(a, include_zero, include_edge)


def mix_vectors(x, y, alpha):
    """Normalize a nonnegative convex mix."""
    z = (1.0 - alpha) * x + alpha * y
    z = np.maximum(z, 0.0)
    s = float(np.sum(z))
    if s <= 1e-15:
        z = y.copy()
        s = float(np.sum(z))
        if s <= 1e-15:
            z = np.ones(len(y), dtype=np.float64) / float(max(1, len(y)))
        else:
            z = z / s
        return z
    return z / s


def run_fp_lp(a0, include_zero, include_edge, time_budget, max_iters=50000):
    """
    Fictitious-play style iteration using LP best responses.
    Works on the internal surrogate objective.
    """
    n = len(a0)
    if n <= 1:
        s = float(np.sum(a0))
        a = a0.copy() if s <= 0 else a0 / s
        return a, exact_conv_score(a, include_zero, include_edge)

    a = np.asarray(a0, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s <= 1e-15:
        a = np.ones(n, dtype=np.float64) / float(n)
        s = 1.0
    a = a / s

    sel = make_selector(n, include_zero, include_edge)
    best = a.copy()
    best_score = exact_conv_score(a, include_zero, include_edge)
    if not np.isfinite(best_score):
        best_score = 0.0

    c_a = autocorr_lags(a)
    current_score = score_from_c(c_a, n, include_zero, include_edge)

    emp = a.copy()
    emp_score = current_score

    alphas = np.array(
        [0.0, 0.005, 0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0],
        dtype=np.float64
    )

    start = time.monotonic()
    deadline = start + max(1.0, float(time_budget))
    rng = np.random.default_rng(98765)
    stall = 0
    no_improve = 0
    it = 0

    temp_start = 1e-3
    temp_end = 1e-7

    while it < max_iters and time.monotonic() < deadline:
        it += 1
        remaining = deadline - time.monotonic()
        if remaining <= 0:
            break

        per_solve_time = max(0.5, min(5.0, 0.1 * remaining))
        g, _ = solve_best_response_lp(a, include_zero, include_edge, per_solve_time)

        c_g = autocorr_lags(g)
        c_sum = autocorr_lags(a + g)
        cross = c_sum - c_a - c_g

        best_alpha = float(alphas[0])
        best_alpha_score = float('inf')
        best_alpha_idx = 0
        for i, al in enumerate(alphas):
            al = float(al)
            c_mix = (1.0 - al) * (1.0 - al) * c_a + al * (1.0 - al) * cross + al * al * c_g
            sc = score_from_c(c_mix, n, include_zero, include_edge)
            if sc < best_alpha_score - 1e-15:
                best_alpha_score = sc
                best_alpha = al
                best_alpha_idx = i

        mix = mix_vectors(a, g, best_alpha)
        mix_score = best_alpha_score

        gamma = min(0.1, 3.0 / float(it + 20.0))
        emp_new = (1.0 - gamma) * emp + gamma * g
        emp_new = np.maximum(emp_new, 0.0)
        es = float(np.sum(emp_new))
        if es <= 1e-15:
            emp_new = a.copy()
            es = 1.0
        else:
            emp_new = emp_new / es
        emp_new_score = exact_conv_score(emp_new, include_zero, include_edge)
        if not np.isfinite(emp_new_score):
            emp_new_score = current_score

        if emp_new_score < mix_score:
            cand = emp_new
            cand_score = emp_new_score
            cand_source = 'emp'
        else:
            cand = mix
            cand_score = mix_score
            cand_source = 'mix'

        if cand_source == 'mix' and best_alpha <= 1e-12 and abs(cand_score - current_score) < 1e-12:
            force_idx = 1
            for j in range(alphas.size):
                if alphas[j] > 1e-8:
                    force_idx = j
                    break
            al = float(alphas[force_idx])
            c_force = (1.0 - al) * (1.0 - al) * c_a + al * (1.0 - al) * cross + al * al * c_g
            force_score = score_from_c(c_force, n, include_zero, include_edge)
            cand = mix_vectors(a, g, al)
            cand_score = force_score
            cand_source = 'force'

        delta = cand_score - current_score
        progress = min(1.0, (time.monotonic() - start) / max(1e-9, float(time_budget)))
        temp = temp_start * (temp_end / temp_start) ** progress

        if delta <= 0.0:
            accept = True
        else:
            if delta > 700.0 * max(1e-300, temp):
                accept = False
            else:
                accept = bool(rng.random() < float(np.exp(-delta / max(1e-300, temp))))

        if accept:
            a = np.maximum(cand, 0.0)
            ss = float(np.sum(a))
            if ss <= 1e-15:
                a = a0.copy()
                ss = 1.0
            else:
                a = a / ss
            c_a = autocorr_lags(a)
            current_score = score_from_c(c_a, n, include_zero, include_edge)
            stall = 0

            if current_score < best_score - 1e-13:
                best_score = current_score
                best = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            stall += 1
            no_improve += 1
            if stall >= 20 or no_improve >= 80:
                z = z_from_a(a)
                z = np.clip(z + 0.02 * rng.standard_normal(n), -45.0, 45.0)
                a = softmax_from_z(z)
                c_a = autocorr_lags(a)
                current_score = score_from_c(c_a, n, include_zero, include_edge)
                if current_score < best_score:
                    best_score = current_score
                    best = a.copy()
                stall = 0

        emp = emp_new
        if emp_new_score < best_score - 1e-13:
            best_score = emp_new_score
            best = emp_new.copy()

        if it % 50 == 0 and best_score < current_score - 1e-10:
            a = best.copy()
            c_a = autocorr_lags(a)
            current_score = best_score
            stall = 0

        if it % 200 == 0:
            print("  FP iter=%d current=%.8f best=%.8f" % (it, current_score, best_score))

    return best, best_score


def random_start(n, base, rng):
    """Generate a random nonnegative unit-sum starting vector."""
    if base is not None:
        b = np.asarray(base, dtype=np.float64)
        if b.size == n:
            z = np.log(np.maximum(b, 1e-20)) + 0.3 * rng.standard_normal(n)
            return softmax_from_z(z)

    t = np.linspace(0.0, 2.0 * np.pi, n, dtype=np.float64)
    x = np.ones(n, dtype=np.float64)
    for k in range(1, 6):
        amp = float(rng.uniform(0.05, 0.5))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        x = x + amp * np.cos(k * t + ph)
    x = np.maximum(x, 1e-8)
    z = np.log(x)
    return softmax_from_z(z)


def update_best_scores(
    seq,
    include_zero,
    include_edge,
    best_actual_seq,
    best_actual_score,
    best_actual_internal,
    best_internal_seq,
    best_internal_score,
    label
):
    """Update actual and internal best sequences."""
    if seq is None:
        return best_actual_seq, best_actual_score, best_actual_internal, best_internal_seq, best_internal_score

    arr = np.asarray(seq, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        return best_actual_seq, best_actual_score, best_actual_internal, best_internal_seq, best_internal_score

    arr = np.maximum(arr, 0.0)
    arr = np.minimum(arr, 1000.0)
    s = float(np.sum(arr))
    if not np.isfinite(s) or s <= 1e-15:
        return best_actual_seq, best_actual_score, best_actual_internal, best_internal_seq, best_internal_score

    arr = arr / s

    internal = exact_conv_score(arr, include_zero, include_edge)
    if not np.isfinite(internal):
        internal = float('inf')

    actual = float('inf')
    try:
        actual = float(evaluate_sequence(arr.tolist()))
    except Exception:
        actual = float('inf')

    improved = False
    if np.isfinite(actual) and actual < best_actual_score - 1e-12:
        best_actual_seq = arr.copy()
        best_actual_score = actual
        best_actual_internal = internal
        improved = True

    if np.isfinite(internal) and internal < best_internal_score - 1e-12:
        best_internal_seq = arr.copy()
        best_internal_score = internal
        improved = True

    if improved:
        print(
            "UPDATE %s: actual_best=%.8f internal_best=%.8f n=%d"
            % (label, best_actual_score, best_internal_score, len(arr))
        )

    return best_actual_seq, best_actual_score, best_actual_internal, best_internal_seq, best_internal_score


def actual_local_search(a0, time_budget, rng):
    """Short direct local search using the actual evaluator."""
    a = np.asarray(a0, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s <= 1e-15:
        return a, float('inf')
    a = a / s

    try:
        cur = float(evaluate_sequence(a.tolist()))
    except Exception:
        return a, float('inf')

    if not np.isfinite(cur):
        return a, float('inf')

    best = a.copy()
    best_score = cur
    z = z_from_a(a)
    start = time.monotonic()
    deadline = start + max(1.0, float(time_budget))
    it = 0

    while time.monotonic() < deadline and it < 20000:
        it += 1
        progress = min(1.0, (time.monotonic() - start) / max(1e-9, float(time_budget)))
        sigma = 0.02 * max(0.02, 1.0 - progress)
        z_new = np.clip(z + sigma * rng.standard_standard if False else sigma * rng.standard_normal(len(a)), -45.0, 45.0)
        cand = softmax_from_z(z_new)

        try:
            sc = float(evaluate_sequence(cand.tolist()))
        except Exception:
            continue

        if np.isfinite(sc) and sc < cur - 1e-12:
            z = z_new
            a = cand
            cur = sc
            if sc < best_score:
                best_score = sc
                best = cand.copy()
        elif np.isfinite(sc) and sc < cur + 1e-4 and rng.random() < 0.02:
            z = z_new
            a = cand
            cur = sc

    return best, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search entry point."""
    total_start = time.monotonic()
    deadline = total_start + max(60.0, float(budget_s) - 20.0)

    include_zero, include_edge = infer_metric_flags()
    print("Selected metric flags: include_zero=%s include_edge=%s" % (include_zero, include_edge))

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        if p.ndim == 1 and p.size > 0:
            p = np.clip(p, 0.0, 1000.0)
            if np.isfinite(p).all() and float(np.sum(p)) > 1e-12:
                parent = p / float(np.sum(p))
    except Exception:
        parent = None

    if parent is None:
        n_init = 768
        t = np.linspace(0.0, 4.0 * np.pi, n_init)
        parent = np.ones(n_init, dtype=np.float64) * (1.0 + 0.2 * np.cos(t))
        parent = np.maximum(parent, 1e-8)
        parent = parent / float(np.sum(parent))

    if len(parent) > 2048:
        parent = interpolate_sequence(parent, 1536)

    best_actual_seq = parent.copy()
    best_actual_score = float('inf')
    best_actual_internal = float('inf')
    best_internal_seq = parent.copy()
    best_internal_score = exact_conv_score(parent, include_zero, include_edge)

    best_actual_seq, best_actual_score, best_actual_internal, best_internal_seq, best_internal_score = update_best_scores(
        parent, include_zero, include_edge,
        best_actual_seq, best_actual_score, best_actual_internal,
        best_internal_seq, best_internal_score,
        'parent'
    )

    work_seq = best_internal_seq.copy()
    rng = np.random.default_rng(seed)

    # Phase 1: initial LBFGS polish
    rem = deadline - time.monotonic()
    alloc = min(500.0, rem * 0.32)
    if alloc > 20.0:
        seq1, score1 = polish_lbfgs(work_seq, include_zero, include_edge, alloc)
        best_actual_seq, best_actual_score, best_actual_internal, best_internal_seq, best_internal_score = update_best_scores(
            seq1, include_zero, include_edge,
            best_actual_seq, best_actual_score, best_actual_internal,
            best_internal_seq, best_internal_score,
            'polish1'
        )
        work_seq = best_internal_seq.copy()

    # Phase 2: LP fictitious play
    rem = deadline - time.monotonic()
    if len(work_seq) <= 1536:
        alloc = min(600.0, rem * 0.35)
        if alloc > 30.0:
            seq2, score2 = run_fp_lp(work_seq, include_zero, include_edge, alloc)
            best_actual_seq, best_actual_score, best_actual_internal, best_internal_seq, best_internal_score = update_best_scores(
                seq2, include_zero, include_edge,
                best_actual_seq, best_actual_score, best_actual_internal,
                best_internal_seq, best_internal_score,
                'fp1'
            )
            work_seq = best_internal_seq.copy()

    # Phase 3: LBFGS polish after play
    rem = deadline - time.monotonic()
    alloc = min(400.0, rem * 0.30)
    if alloc > 20.0:
        seq3, score3 = polish_lbfgs(work_seq, include_zero, include_edge, alloc)
        best_actual_seq, best_actual_score, best_actual_internal, best_internal_seq, best_internal_score = update_best_scores(
            seq3, include_zero, include_edge,
            best_actual_seq, best_actual_score, best_actual_internal,
            best_internal_seq, best_internal_score,
            'polish2'
        )
        work_seq = best_internal_seq.copy()

    # Phase 4: exact top-peak refinement
    rem = deadline - time.monotonic()
    alloc = min(180.0, rem * 0.14)
    if alloc > 15.0:
        seq4, score4 = peak_refine(work_seq, include_zero, include_edge, alloc, K=10)
        best_actual_seq, best_actual_score, best_actual_internal, best_internal_seq, best_internal_score = update_best_scores(
            seq4, include_zero, include_edge,
            best_actual_seq, best_actual_score, best_actual_internal,
            best_internal_seq, best_internal_score,
            'peak1'
        )
        work_seq = best_internal_seq.copy()

    # Phase 5: another round of LP play if time remains
    rem = deadline - time.monotonic()
    if len(work_seq) <= 1536:
        alloc = min(300.0, rem * 0.25)
        if alloc > 30.0:
            seq5, score5 = run_fp_lp(work_seq, include_zero, include_edge, alloc)
            best_actual_seq, best_actual_score, best_actual_internal, best_internal_seq, best_internal_score = update_best_scores(
                seq5, include_zero, include_edge,
                best_actual_seq, best_actual_score, best_actual_internal,
                best_internal_seq, best_internal_score,
                'fp2'
            )
            work_seq = best_internal_seq.copy()

    # Phase 6: randomized restarts / perturbations
    rem = deadline - time.monotonic()
    for trial in range(3):
        rem = deadline - time.monotonic()
        if rem < 40.0:
            break
        base = work_seq if trial == 0 else None
        init = random_start(len(work_seq), base, rng)
        alloc = min(45.0, rem * 0.22)
        seqr, scor = run_lbgfs(init, include_zero, include_edge, 100000.0, alloc)
        best_actual_seq, best_actual_score, best_actual_internal, best_internal_seq, best_internal_score = update_best_scores(
            seqr, include_zero, include_edge,
            best_actual_seq, best_actual_score, best_actual_internal,
            best_internal_seq, best_internal_score,
            'random%d' % trial
        )
        work_seq = best_internal_seq.copy()

    # Phase 7: nearby length search
    rem = deadline - time.monotonic()
    if rem > 60.0:
        n_cur = len(work_seq)
        candidates = []

        for d in [1, 2, 3, 4, 6, 8, 12, 16, 24, 32]:
            for sgn in (1, -1):
                n1 = n_cur + sgn * d
                if 128 <= n1 <= 1536 and n1 != n_cur:
                    candidates.append(n1)

        for f in (2, 3):
            for n1 in (n_cur // f, n_cur * f):
                if 128 <= n1 <= 1536 and n1 != n_cur:
                    candidates.append(n1)

        # Unique and sort by distance without lambda
        uniq = []
        for x in candidates:
            if x not in uniq:
                uniq.append(x)
        candidates = uniq

        ordered = []
        while len(candidates) > 0 and len(ordered) < 4:
            best_idx = 0
            best_dist = 10 ** 12
            for i in range(len(candidates)):
                dist = abs(candidates[i] - n_cur)
                if dist < best_dist:
                    best_dist = dist
                    best_idx = i
            ordered.append(candidates.pop(best_idx))

        for n1 in ordered:
            rem = deadline - time.monotonic()
            if rem < 30.0:
                break
            init = interpolate_sequence(work_seq, n1)
            alloc = min(40.0, rem * 0.3)
            seqL, scoreL = run_lbgfs(init, include_zero, include_edge, 100000.0, alloc)
            best_actual_seq, best_actual_score, best_actual_internal, best_internal_seq, best_internal_score = update_best_scores(
                seqL, include_zero, include_edge,
                best_actual_seq, best_actual_score, best_actual_internal,
                best_internal_seq, best_internal_score,
                'len%d' % n1
            )
            work_seq = best_internal_seq.copy()

    # Phase 8: direct actual local search
    rem = deadline - time.monotonic()
    if rem > 40.0:
        alloc = min(90.0, rem * 0.25)
        seqA, scoreA = actual_local_search(work_seq, alloc, rng)
        best_actual_seq, best_actual_score, best_actual_internal, best_internal_seq, best_internal_score = update_best_scores(
            seqA, include_zero, include_edge,
            best_actual_seq, best_actual_score, best_actual_internal,
            best_internal_seq, best_internal_score,
            'actual_ls'
        )
        work_seq = best_actual_seq.copy()

    # Phase 9: final polish
    rem = deadline - time.monotonic()
    alloc = min(120.0, rem * 0.45)
    if alloc > 10.0:
        seqF, scoreF = polish_lbfgs(work_seq, include_zero, include_edge, alloc)
        best_actual_seq, best_actual_score, best_actual_internal, best_internal_seq, best_internal_score = update_best_scores(
            seqF, include_zero, include_edge,
            best_actual_seq, best_actual_score, best_actual_internal,
            best_internal_seq, best_internal_score,
            'final_polish'
        )

    # Final verification and selection by actual evaluator
    final_seq = None
    final_actual = float('inf')

    candidates_final = []
    for seq in (best_actual_seq, best_internal_seq, parent):
        if seq is not None:
            candidates_final.append(seq)

    for seq in candidates_final:
        arr = np.asarray(seq, dtype=np.float64)
        if arr.ndim != 1 or arr.size == 0:
            continue
        arr = np.maximum(arr, 0.0)
        arr = np.minimum(arr, 1000.0)
        s = float(np.sum(arr))
        if not np.isfinite(s) or s < 0.01:
            continue
        arr = arr / s
        try:
            sc = float(evaluate_sequence(arr.tolist()))
        except Exception:
            sc = float('inf')
        if np.isfinite(sc) and sc < final_actual:
            final_actual = sc
            final_seq = arr

    if final_seq is None or len(final_seq) == 0:
        final_seq = parent.copy()

    final_seq = np.maximum(final_seq, 0.0)
    final_seq = np.minimum(final_seq, 1000.0)
    fs = float(np.sum(final_seq))
    if not np.isfinite(fs) or fs < 0.01:
        final_seq = parent.copy()
        fs = float(np.sum(final_seq))
        if fs > 1e-15:
            final_seq = final_seq / fs

    else:
        final_seq = final_seq / fs

    try:
        final_score = float(evaluate_sequence(final_seq.tolist()))
    except Exception:
        final_score = float('inf')

    print("FINAL: n=%d actual_final=%.8f internal_parent_metric=%s edge=%s time=%.1fs"
          % (len(final_seq), final_score, include_zero, include_edge, time.monotonic() - total_start))

    return [float(x) for x in final_seq]
