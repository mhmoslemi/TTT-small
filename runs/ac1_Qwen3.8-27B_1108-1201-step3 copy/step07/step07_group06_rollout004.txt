import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def get_score_normalized(a):
    """Score assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def lp_step(a, M, max_iter=1):
    """
    One step of LP-based alternating minimization.
    Given current a (sum=1) and peak M, find g that maximizes sum(g)
    subject to (a*g)[k] <= M for all k, g >= 0.
    Returns the new sequence after line search.
    """
    n = len(a)
    # Build constraint matrix A where A[k,j] = a[k-j] if 0 <= k-j < n
    # This is a (2n-1) x n matrix
    # A[k] is a shifted version of a
    # For efficiency, we can construct it row by row
    
    # Actually, let's think about this more carefully.
    # (a*g)[k] = sum_{j=0}^{n-1} g[j] * a[k-j] (where a is 0 outside [0,n-1])
    # So A[k,j] = a[k-j] if max(0, k-(n-1)) <= j <= min(n-1, k), else 0
    
    # For the LP: minimize -sum(g) subject to A@g <= M, g >= 0
    # In scipy form: c = -ones(n), A_ub = A, b_ub = M*ones(2n-1), bounds = (0, None)
    
    c = -np.ones(n)
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        # A_ub[k, j] = a[k-j] if 0 <= k-j < n
        j_min = max(0, k - (n-1))
        j_max = min(n-1, k)
        A_ub[k, j_min:j_max+1] = a[k - np.arange(j_min, j_max+1)]
    
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.status != 0:
        return a, M
    
    g = res.x
    g_sum = g.sum()
    if g_sum < 1e-15:
        return a, M
    
    g = g / g_sum  # Normalize to sum 1
    
    # Line search: find best t in [0,1]
    best_t = 0.0
    best_new_M = M
    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:

        new_M = fast_convolve(a_new).max()
        if new_M < best_new_M:
            best_new_M = new_M
            best_t = t
    
    a_new = (1-best_t) * a + best_t * g
    return a_new, best_new_M

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based alternating minimization
combined with gradient polishing and multi-length exploration.
Key innovations: (1) LP-based peak reduction via linear programming,
(2) multi-length search to find optimal n, (3) hybrid LP + gradient approach.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def get_score_local(a):
    """Compute score for normalized sequence (sum=1)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    a_norm = a / s
    conv = fast_convolve(a_norm)
    return 2.0 * n * conv.max()


def lp_step(a, M, verbose=False):
    """
    One LP-based step: find g maximizing sum(g) s.t. (a*g)[k] <= M, g >= 0.
    Then line search to find best convex combination.
    Returns (new_a, new_M).
    """
    n = len(a)
    
    # Build constraint matrix
    # A_ub[k, j] = a[k-j] if 0 <= k-j < n, else 0
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(n-1, k)
        if j_start <= j_end:
            idx = np.arange(j_start, j_end+1)
            A_ub[k, idx] = a[k - idx]
    
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, 
                     method='highs', options={'presolve': True})
    except Exception:
        return a, M
    
    if res.status != 0 or res.x is None:
        return a, M
    
    g = np.maximum(res.x, 0)
    g_sum = g.sum()
    if g_sum < 1e-15:
        return a, M
    
    g = g / g_sum
    
    # Line search
    best_t = 0.0
    best_M = M
    for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
        a_new = (1-t) * a + t * g
        new_M = fast_convolve(a_new).max()
        if new_M < best_M:
            best_M = new_M
            best_t = t
    
    if best_t == 0.0:
        return a, M
    
    a_new = (1-best_t) * a + best_t * g
    return a_new, best_M


def multiplicative_weight_step(a, M, eta=0.5):
    """
    Multiplicative weight update: reduce elements that contribute to peaks.
    """
    n = len(a)
    conv = fast_convolve(a)
    
    # Weight lags by how much they exceed the target
    target = conv.mean()  # average level
    w = np.maximum(conv - target, 0)
    w_sum = w.sum()
    if w_sum < 1e-15:
        return a, M
    
    w = w / w_sum
    
    # Compute pressure on each position
    # pressure[j] = sum_k w[k] * a[k-j] (valid indices only)
    # This is a cross-correlation
    # pressure = conv(w, a) but with appropriate alignment
    # Actually: pressure[j] = sum_{k} w[k] * a[k-j] where 0 <= k-j < n
    # Let's compute via FFT
    fft_size = 1 << (2*n - 2).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    # We want sum_k w[k] * a[k-j] = cross-correlation
    # This is equivalent to conv(w, flip(a)) shifted
    # Or: it's the "anti-convolution"
    # Actually, sum_k w[k] * a[k-j] is the cross-correlation of w and a at lag j
    # corr[j] = sum_k w[k] * a[k+j] (standard definition)
    # But we want sum_k w[k] * a[k-j] = corr at lag -j
    
    # Let's just use the formula directly with FFT
    # sum_k w[k] * a[k-j]: let m = k-j, so k = m+j
    # = sum_m w[m+j] * a[m] = (w_shifted * a) where w_shifted[m] = w[m+j]
    # This is a convolution of a with a shifted w
    
    # Simpler: just compute it as a filtered version
    # pressure = ifft(conj(fft(w)) * fft(a)) (circular cross-correlation)
    corr = irfft(np.conj(W) * A, fft_size)
    # corr[j] = sum_k w[(k-j) mod fft_size] * a[k]
    # For our purposes (with zero-padding), this gives the linear cross-correlation
    pressure = corr[:n]  # Take the first n values
    
    # Actually, let me think again. We want:
    # pressure[j] = sum_{k: 0<=k<2n-1} w[k] * a[k-j] (where a[k-j]=0 if not in [0,n-1])
    # = sum_{k: max(0,j) <= k <= min(2n-2, j+n-1)} w[k] * a[k-j]
    # Let i = k-j: sum_{i: 0<=i<n} w[i+j] * a[i] (where w[i+j]=0 if i+j >= 2n-1)
    # This is the inner product of w[j:j+n] (padded) with a
    # Which is the cross-correlation of w and a at lag j
    
    # Using FFT: cross-correlation of w and a at lag j = ifft(conj(fft(w)) * fft(a))[j]
    # (with appropriate zero-padding for linear correlation)
    
    # Actually the standard formula: (w ⋆ a)[j] = sum_k w[k] a[k-j]
    # In frequency domain: FFT(w ⋆ a) = conj(FFT(w)) * FFT(a)
    # But we need linear (not circular) correlation, so zero-pad.
    
    # With our fft_size >= 2n-1 + n - 1 = 3n-2, we get linear correlation.
    # Hmm, actually for linear cross-correlation of two sequences of lengths L1 and L2,
    # we need FFT size >= L1 + L2 - 1.
    # w has length 2n-1, a has length n. So we need size >= 3n-2.
    
    # Let me redo with proper FFT size
    fft_size2 = 1 << (3*n - 3).bit_length()
    W2 = rfft(w, fft_size2)
    A2 = rfft(a, fft_size2)
    corr_linear = irfft(np.conj(W2) * A2, fft_size2)
    # corr_linear[j] = sum_k w[k] * a[k-j] for j = 0, 1, ...
    pressure = corr_linear[:n]
    pressure = np.maximum(pressure, 0)
    
    # Update
    a_new = a * np.exp(-eta * pressure / max(pressure.max(), 1e-15))
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a, M
    a_new = a_new / s
    new_M = fast_convolve(a_new).max()
    
    return a_new, new_M


def softmax_gradient_step(a, T, lr, eta=1.0):
    """
    One step of soft-max gradient descent.
    """
    n = len(a)
    conv = fast_convolve(a)
    max_b = conv.max()
    
    # Soft-max weights
    exp_b = np.exp(conv / T - max_b / T)
    w = exp_b / exp_b.sum()
    
    # Gradient of 2n * sum(w * conv) w.r.t. a
    # d(conv[k])/d(a[j]) = a[k-j] (if valid) + a[j-k] (if valid)
    # So grad[j] = 2n * sum_k w[k] * (a[k-j] + a[j-k]) (valid terms)
    # = 2n * 2 * cross_correlation(w, a)[j] (approximately, for large n)
    
    fft_size = 1 << (2*n - 2).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(W * np.conj(A), fft_size)[:n]
    # corr[j] = sum_k w[k] * a[k-j] (approximately, for the linear part)
    # Actually this gives the cross-correlation
    grad = 2.0 * n * 2.0 * corr
    
    # Normalize gradient (subtract mean to maintain sum constraint)
    grad = grad - grad.mean()
    
    # Update
    a_new = a * np.exp(eta * lr * grad / (np.abs(grad).max() + 1e-15))
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a, max_b
    a_new = a_new / s
    new_M = fast_convolve(a_new).max()
    
    return a_new, new_M


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp
