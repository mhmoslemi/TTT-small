```python
def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() == 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.argmax(1 - u * np.arange(1, n+1) > 0)
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)
```

```python
3. LP-based fictitious play (best response iteration)

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog


def next_pow2(n):
    return 1 << (n - 1).bit_length()


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = next_pow2(size)
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute score for normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def compute_gradient(a, w):
    """
    Compute gradient of sum_k w[k] * conv[a][k] w.r.t. a.
    w has length 2n-1 (weights on conv positions), a has length n.
    grad[m] = 2 * sum_k w[k] * a[k-m] (a[j]=0 outside [0,n-1])
    """
    n = len(a)
    # conv length = 2n-1, a_rev length = n
    # linear conv result length = (2n-1) + n - 1 = 3n-2
    fft_size = next_pow2(3 * n - 2)
    W = rfft(w, fft_size)
    A_rev = rfft(a[::-1], fft_size)
    G = irfft(W * A_rev, fft_size)
    grad = 2.0 * G[n - 1: n - 1 + n]
    return grad


def softmax_weights(conv, T):
    """Compute softmax weights on conv values."""
    shifted = conv / T
    max_sh = shifted.max()
    exp_sh = np.exp(shifted - max_sh)
    s = exp_sh.sum()
    if s < 1e-300:
        return np.ones(len(conv)) / len(conv)
    return exp_sh / s


def top_k_weights(conv, k=5):
    """Uniform weights on top-k peaks (T=0 limit)."""
    w = np.zeros(len(conv))
    top_idx = np.argpartition(conv, -k)[-k:]
    w[top_idx] = 1.0 / k
    return w


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rng = np.arange(1, n + 1)
    cond = u - (cssv - 1.0) / rng > 0
    if not cond.any():
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def softmax_from_z(z):
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-30))


def subgradient_descent(a0, time_limit, T_start=500.0, T_end=1.0,
                        lr_init=0.001, top_k=5, verbose=True):
    """
    Subgradient descent on max(conv) with adaptive temperature.
    Uses projective subgradient method with backtracking.
    """
    n = len(a0)
    a = a0.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    total_time = time_limit
    iters = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / total_time)
        # Exponential cooling of temperature
        T = T_start * (T_end / T_start) ** (progress ** 0.7)

        iters += 1
        conv = fast_convolve(a)
        max_conv = conv.max()

        # Choose weights: blend between top-k and softmax
        if T < 10.0:
            w = top_k_weights(conv, top_k)
        elif T < 100.0:
            w_soft = softmax_weights(conv, T)
            w_top = top_k_weights(conv, top_k)
            alpha = min(1.0, (100.0 - T) / 90.0)
            w = alpha * w_top + (1 - alpha) * w_soft
        else:
            w = softmax_weights(conv, T)

        # Compute gradient
        grad = compute_gradient(a, w)

        # Project gradient to tangent space of simplex
        grad_tan = grad - np.dot(grad, a)

        # Adaptive step size: lr = c / (iter + 10) * scale
        scale = max_conv / max(a.max(), 1e-10)
        lr = lr_init * scale / (1.0 + 0.01 * iters)

        # Backtracking line search
        accepted = False
        for backtrack in range(10):
            trial = a - lr * grad_tan
            trial = project_simplex(trial)
            trial_score = compute_score(trial)
            if trial_score < best_score + 1e-12:
                a = trial
                if trial_score < best_score:
                    best_score = trial_score
                    best_a = trial.copy()
                accepted = True
                break
            lr *= 0.5

        if not accepted:
            # Try a smaller fixed step or skip
            lr2 = lr_init * scale * 0.01
            trial = a - lr2 * grad_tan
            trial = project_simplex(trial)
            if compute_score(trial) < compute_score(a):
                a = trial

        if verbose and iters % 100 == 0:
            print(f"  SD iter={iters}, T={T:.1f}, score={best_score:.8f}")

    return best_a, best_score


def fourier_smooth_optimize(a0, time_limit, verbose=True):
    """
    Minimize ||conv||_2^2 as a smooth proxy for max(conv).
    gradient[m] = 4 * sum_k conv[k] * a[k-m]
    """
    n = len(a0)
    a = a0.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    lr = 0.01

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)

        # Gradient of ||conv||^2 w.r.t. a: 4 * (conv * a_rev)[m+n-1]
        fft_size = next_pow2(3 * n - 2)
        C = rfft(conv, fft_size)
        A_rev = rfft(a[::-1], fft_size)
        G = irfft(C * A_rev, fft_size)
        grad = 4.0 * G[n - 1: n - 1 + n]

        # Project to simplex tangent
        grad_tan = grad - np.dot(grad, a)

        # Normalized step
        grad_norm = np.linalg.norm(grad_tan)
        if grad_norm < 1e-15:
            break

        trial = a - lr * grad_tan / grad_norm
        trial = project_simplex(trial)
        trial_score = compute_score(trial)

        if trial_score < best_score:
            best_score = trial_score
            best_a = trial.copy()
            a = trial
            # Increase lr slightly
            lr = min(lr * 1.05, 0.1)
        else:
            lr *= 0.7

        if verbose and iters % 200 == 0:
            print(f"  FourOpt iter={iters}, lr={lr:.4f}, score={best_score:.8f}")

    return best_a, best_score


def lbgfs_smoothed(a0, time_limit, T=500, maxiter=3000):
    """L-BFGS-B on smoothed objective in log-space."""
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = softmax_from_z(z)
        conv = fast_convolve(a)
        shifted = conv / T
        max_sh = shifted.max()
        lse = max_sh + np.log(np.exp(shifted - max_sh).sum())
        return (2.0 * n / T) * np.exp(T * lse)

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        a = softmax_from_z(z)
        conv = fast_convolve(a)
        w = softmax_weights(conv, T)
        grad_a = compute_gradient(a, w)
        # Project to log-space
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)
        return (2.0 * n / T) * grad_z

    try:
        res = minimize(fun=func, x0=z0, jac=jac, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-15,
                                'gtol': 1e-10, 'maxfun': maxiter * 30,
                                'disp': False})
        a_out = softmax_from_z(res.x)
        sc = compute_score(a_out)
        if sc < best_score:
            best_score = sc
            best_a = a_out.copy()
    except Exception:
        pass

    return best_a, best_score


def lp_best_response(a, M):
    """
    Solve LP: maximize sum(b) subject to (a*b)[k] <= M, b >= 0.
    Returns b normalized to sum=1.
    """
    n = len(a)
    # Variables: b[0], ..., b[n-1]
    # Objective: maximize sum(b) -> minimize -sum(b)
    c = -np.ones(n)

    # Constraints: (a*b)[k] <= M for k = 0, ..., 2n-2
    # (a*b)[k] = sum_i a[i] * b[k-i] where 0<=i<n and 0<=k-i<n
    # Build constraint matrix A_ub (2n-1 x n)
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            A_ub[k, i] = a[i]

    b_ub = np.full(2 * n - 1, M)

    # Bounds: b >= 0
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs',
                      options={'time_limit': 30, 'presolve': True})
        if res.success:
            b = res.x
            s = b.sum()
            if s > 1e-15:
                return b / s
    except Exception:
        pass
    return None


def fictitious_play(a0, time_limit, n_iters_max=200, verbose=True):
    """
    LP-based fictitious play: alternate best responses.
    """
    n = len(a0)
    a = a0.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0

    while time.monotonic() - t0 < time_limit and iters < n_iters_max:
        iters += 1
        conv = fast_convolve(a)
        M = conv.max()

        # Find best response b to a
        b = lp_best_response(a, M)
        if b is None:
            break

        # Mix
        t = 0.3 / (1.0 + 0.05 * iters)
        a_new = (1 - t) * a + t * b
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()

        a = a_new

        if verbose and iters % 10 == 0:
            print(f"  FP iter={iters}, score={best_score:.8f}")

    return best_a, best_score


def coordinate_descent(a0, time_limit, n_pairs=200, verbose=False):
    """Coordinate descent with pairwise transfers."""
    n = len(a0)
    a = a0.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False

        for _ in range(n_pairs):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or a[i] < 1e-12:
                continue

            base = a[i]
            for frac in [0.05, 0.1, 0.2, 0.4, 0.7]:
                t = frac * base
                if t >= base - 1e-15:
                    t = base * 0.5
                trial = a.copy()
                trial[i] -= t
                trial[j] += t
                trial = np.maximum(trial, 0)
                s = trial.sum()
                if s < 1e-15:
                    continue
                trial = trial / s
                sc = compute_score(trial)
                if sc < best_score - 1e-14:
                    a = trial
                    best_score = sc
                    best_a = trial.copy()
                    improved = True

            if not improved and iters > 20:
                # Perturb
                noise = rng.standard_normal(n) * 0.002
                a = best_a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()

    return best_a, best_score


def simulated_annealing(a0, time_limit, n_moves=60, t_start=0.01, t_end=0.0001):
    """Simulated annealing with diverse moves."""
    n = len(a0)
    a = a0.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    curr_score = best_score
    t0 = time.monotonic()
    rng = np.random.default_rng(77)
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        temp = t_start * (t_end / t_start) ** progress

        for _ in range(n_moves):
            trial = a.copy()
            move = rng.random()

            if move < 0.35:
                # Pairwise transfer
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i != j and a[i] > 1e-12:
                    t = rng.standard_normal() * 0.05 * a[i]
                    trial[i] -= t
                    trial[j] += t
            elif move < 0.6:
                # Scale one coordinate, redistribute
                i = rng.integers(0, n)
                if a[i] > 1e-12:
                    factor = 1.0 + rng.standard_normal() * 0.2
                    new_val = max(0.0, a[i] * factor)
                    diff = a[i] - new_val
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    others = trial[mask]
                    os = others.sum()
                    if os > 1e-15:
                        trial[i] = new_val
                        trial[mask] = others + diff * others / os
            elif move < 0.8:
                # Block noise
                blen = min(n, rng.integers(3, max(4, n // 40)))
                start = rng.integers(0, n - blen + 1)
                idx = np.arange(start, start + blen)
                trial[idx] += rng.standard_normal(blen) * 0.02
            else:
                # Two-point adjust
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i != j:
                    t1 = rng.standard_normal() * 0.03 * a[i]
                    t2 = rng.standard_normal() * 0.03 * a[j] if a[j] > 0 else 0
                    trial[i] -= t1
                    trial[j] -= t2
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    mask[j] = False
                    others = trial[mask]
                    os = others.sum()
                    if os > 1e-15:
                        redist = t1 + t2
                        trial[mask] = others + redist * others / os

            trial = np.maximum(trial, 0)
            s = trial.sum()
            if s < 1e-15:
                continue
            trial = trial / s
            trial_sc = compute_score(trial)
            delta = trial_sc - curr_score
            if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
                a = trial
                curr_score = trial_sc
                if trial_sc < best_score:
                    best_score = trial_sc
                    best_a = trial.copy()

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_update(a, label, best_seq, best_score, verbose=True):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def peak_equalization_refined(a0, time_limit, top_k=7, verbose=True):
    """
    Refined peak equalization:
    At each step, identify top-k peaks, compute average subgradient,
    and take a projected step to reduce the max while equalizing peaks.
    """
    n = len(a0)
    a = a0.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        max_val = conv.max()

        # Find top-k peaks
        top_idx = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[top_idx]
        peak_spread = peak_vals.max() - peak_vals.min()

        # Weight: emphasize the largest peaks more
        # Use inverse-gap weighting: peaks further above the mean get more weight
        mean_peak = peak_vals.mean()
        if peak_spread > 1e-12:
            weights_local = (peak_vals - peak_vals.min() + 1e-6)
            weights_local = weights_local / weights_local.sum()
        else:
            weights_local = np.ones(top_k) / top_k

        w = np.zeros(len(conv))
        w[top_idx] = weights_local

        # Compute subgradient
        grad = compute_gradient(a, w)
        grad_tan = grad - np.dot(grad, a)

        # Step size: adaptive
        grad_norm = np.linalg.norm(grad_tan)
        if grad_norm < 1e-15:
            break

        # Target: reduce max by a small amount
        lr = 0.5 * max_val / (grad_norm * grad_norm + 1e-10)
        lr = min(lr, 0.01)

        # Backtracking
        accepted = False
        for bt in range(8):
            trial = a - lr * grad_tan
            trial = project_simplex(trial)
            trial_sc = compute_score(trial)
            if trial_sc < best_score:
                a = trial
                best_score = trial_sc
                best_a = trial.copy()
                accepted = True
                break
            lr *= 0.5

        if not accepted and iters % 50 == 0:
            # Perturbation to escape
            noise = np.random.default_rng(iters).standard_normal(n) * 0.001
            a = best_a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()

        if verbose and iters % 50 == 0:
            print(f"  PE iter={iters}, spread={peak_spread:.2e}, score={best_score:.8f}")

        if peak_spread < 1e-9 and iters > 20:
            break

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-strategy optimization to minimize 2*n*max(conv(a,a))/sum(a)^2.
    Combines: L-BFGS, subgradient descent, Fourier smooth optimization,
    LP fictitious play, peak equalization, coordinate descent, SA.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is None:
        n_init = 768
        parent = np.ones(n_init) / n_init
        t = np.linspace(0, 2 * np.pi * 5, n_init)
        parent = parent + 0.03 * np.cos(t) + 0.02 * np.cos(2 * t + 0.5)
        parent = np.maximum(parent, 0)
        parent = parent / parent.sum()
        p_score = compute_score(parent)
        print(f"Initialized: n={n_init}, score={p_score:.8f}")

    best_seq = parent.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    n_parent = len(parent)
    print(f"Starting best: {best_score:.8f} (n={n_parent})")

    # === Phase 1: L-BFGS with multiple temperatures (initial refinement) ===
    print(f"\n=== Phase 1: L-BFGS Multi-T (best={best_score:.8f}) ===")
    t1_budget = min(300.0, (deadline - time.monotonic()) * 0.2)
    t_per_T = t1_budget / 4

    for T_val in [200, 500, 2000, 8000]:
        rem = deadline - time.monotonic() - (budget_s * 0.65)
        if rem < 15:
            break
        t_alloc = min(t_per_T, rem * 0.3)
        if t_alloc < 5:
            continue
        print(f"  L-BFGS T={T_val}, budget={t_alloc:.0f}s")
        res_a, res_sc = lbgfs_smoothed(best_seq, t_alloc, T=T_val, maxiter=2000)
        best_seq, best_score = check_update(res_a, f"lbgfs_T{T_val}", best_seq, best_score)

    # === Phase 2: Fourier smooth optimization ===
    print(f"\n=== Phase 2: Fourier Smooth (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - (budget_s * 0.55)
    if rem > 20:
        t_alloc = min(rem * 0.3, 120.0)
        print(f"  FourierOpt budget={t_alloc:.0f}s")
        res_a, res_sc = fourier_smooth_optimize(best_seq, t_alloc, verbose=True)
        best_seq, best_score = check_update(res_a, "fourier", best_seq, best_score)

    # === Phase 3: Subgradient descent with adaptive T ===
    print(f"\n=== Phase 3: Subgradient Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - (budget_s * 0.4)
    if rem > 20:
        t_alloc = min(rem * 0.4, 200.0)
        print(f"  SGD budget={t_alloc:.0f}s")
        res_a, res_sc = subgradient_descent(best_seq, t_alloc,
                                            T_start=1000.0, T_end=0.5,
                                            lr_init=0.002, top_k=7, verbose=True)
        best_seq, best_score = check_update(res_a, "subgrad", best_seq, best_score)

    # === Phase 4: Peak equalization (refined) ===
    print(f"\n=== Phase 4: Peak Equalization (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - (budget_s * 0.25)
    if rem > 20:
        t_alloc = min(rem * 0.4, 180.0)
        print(f"  PE budget={t_alloc:.0f}s")
        res_a, res_sc = peak_equalization_refined(best_seq, t_alloc, top_k=7, verbose=True)
        best_seq, best_score = check_update(res_a, "peak_eq", best_seq, best_score)

    # === Phase 5: LP Fictitious Play ===
    print(f"\n=== Phase 5: Fictitious Play (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - (budget_s * 0.15)
    if rem > 30:
        t_alloc = min(rem * 0.4, 150.0)
        print(f"  FP budget={t_alloc:.0f}s")
        try:
            res_a, res_sc = fictitious_play(best_seq, t_alloc, n_iters_max=100, verbose=True)
            best_seq, best_score = check_update(res_a, "fict_play", best_seq, best_score)
        except Exception as e:
            print(f"  FP failed: {e}")

    # === Phase 6: Second round of subgradient + peak eq ===
    print(f"\n=== Phase 6: Refinement (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - (budget_s * 0.08)
    if rem > 20:
        t_alloc = min(rem * 0.35, 100.0)
        print(f"  SGD2 budget={t_alloc:.0f}s")
        res_a, res_sc = subgradient_descent(best_seq, t_alloc,
                                            T_start=200.0, T_end=0.1,
                                            lr_init=0.001, top_k=5, verbose=True)
        best_seq, best_score = check_update(res_a, "subgrad2", best_seq, best_score)

    rem = deadline - time.monotonic() - (budget_s * 0.04)
    if rem > 15:
        t_alloc = min(rem * 0.5, 60.0)
        print(f"  PE2 budget={t_alloc:.0f}s")
        res_a, res_sc = peak_equalization_refined(best_seq, t_alloc, top_k=5, verbose=True)
        best_seq, best_score = check_update(res_a, "peak_eq2", best_seq, best_score)

    # === Phase 7: Coordinate descent ===
    print(f"\n=== Phase 7: Coord Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 30
    if rem > 15:
        t_alloc = min(rem * 0.4, 80.0)
        print(f"  CD budget={t_alloc:.0f}s")
        res_a, res_sc = coordinate_descent(best_seq, t_alloc, n_pairs=300, verbose=False)
        best_seq, best_score = check_update(res_a, "coord_desc", best_seq, best_score)

    # === Phase 8: Simulated annealing ===
    print(f"\n=== Phase 8: Sim Annealing (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 20
    if rem > 15:
        t_alloc = min(rem * 0.4, 60.0)
        print(f"  SA budget={t_alloc:.0f}s")
        res_a, res_sc = simulated_annealing(best_seq, t_alloc, n_moves=50,
                                            t_start=0.005, t_end=0.00005)
        best_seq, best_score = check_update(res_a, "sim_anneal", best_seq, best_score)

    # === Phase 9: Final polish ===
    print(f"\n=== Phase 9: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 10
    if rem > 10:
        t_alloc = min(rem * 0.5, 30.0)
        res_a, res_sc = peak_equalization_refined(best_seq, t_alloc, top_k=10, verbose=False)
        best_seq, best_score = check_update(res_a, "final_pe", best_seq, best_score)

    rem = deadline - time.monotonic() - 5
    if rem > 8:
        t_alloc = min(rem * 0.6, 20.0)
        res_a, res_sc = subgradient_descent(best_seq, t_alloc,
                                            T_start=50.0, T_end=0.01,
                                            lr_init=0.0005, top_k=3, verbose=False)
        best_seq, best_score = check_update(res_a, "final_sg", best_seq, best_score)

    # === Phase 10: Try nearby lengths ===
    print(f"\n=== Phase 10: Nearby Lengths (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 10
    if rem > 30:
        for alt_n in [n_parent - 2, n_parent + 2]:
            if alt_n < 64 or alt_n > 2048:
                continue
            rem2 = deadline - time.monotonic() - 5
            if rem2 < 15:
                break
            t_alloc = min(rem2 * 0.3, 20.0)
            init_alt = interpolate_sequence(best_seq, alt_n)
            res_a, res_sc = lbgfs_smoothed(init_alt, t_alloc, T=500, maxiter=1000)
            best_seq, best_score = check_update(res_a, f"n{alt_n}", best_seq, best_score)

    # Final verification
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))

    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
