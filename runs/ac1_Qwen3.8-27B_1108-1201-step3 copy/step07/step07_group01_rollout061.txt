def interpolated_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    n = len(a_init)
    a = np.maximum(a_init, 1e-15).astype(np.float64)
    a = a / a.sum()

    fft_size = 1 << (2 * n - 1).bit_length()

        if time.monotonic() - t0 > time_limit:
            break

        A = fft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Target: flat magnitude (all equal to mean)
        target_mag = np.full_like(mag, mag.mean())
        
        # Blend (to avoid oscillation)
        alpha = 0.1  # blending parameter
        new_mag = mag + alpha * (target_mag - mag)

        new_A = new_mag * np.exp(1j * phase)

            new_a = a.copy()
        else:
            new_a = new_a / s
        
        a = new_a
        sc = score_normalized(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

                t = t_start * 0.5

    return best_a, best_score


def multi_peak_gradient(a_init, time_limit, lr_start=0.003):
    """Multi-peak soft gradient descent with momentum."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    vel = np.zeros(n)
    momentum = 0.9
    lr = lr_start
    fft_size = 1 << (3 * n - 1).bit_length()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(a)

        # Soft-max weights over peaks
        T = 0.003 + 0.03 * np.exp(-0.0005 * it)
        scaled = b / T
        m = scaled.max()
        exp_scaled = np.exp(scaled - m)
        w = exp_scaled / exp_scaled.sum()

        # Gradient via FFT: grad[j] = 2 * cross_corr(w, a)[j]
        Wf = fft(w.astype(np.complex128), fft_size)
        Af = fft(a.astype(np.complex128), fft_size)
        corr = np.real(ifft(np.conj(Wf) * Af, fft_size))
        grad = 2.0 * corr[:n]

        # Project to tangent space of simplex
        grad = grad - np.dot(a, grad)

        # Momentum update
        vel = momentum * vel + grad
        a_new = a - lr * vel
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
            vel = np.zeros(n)
        else:
            a_new = a_new / s

        a = a_new
        sc = score_normalized(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        # Decay learning rate
        lr = lr_start / (1.0 + 0.002 * it)

        if it % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      MPG iter={it} ({elapsed:.0f}s): score={sc:.7f} best={best_score:.7f}")

    return best_a, best_score


def exact_peak_descent(a_init, time_limit, lr_start=0.002):
    """Descent using exact peak subgradient with momentum and line search."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    vel = np.zeros(n)
    momentum = 0.85
    lr = lr_start
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))

        # Exact subgradient
        grad = np.zeros(n)
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        grad[idx] = 2.0 * a[peak_k - idx]

        # Project to tangent space
        grad = grad - np.dot(a, grad)

        # Momentum
        vel = momentum * vel + grad

        # Line search
        improved = False
        for scale in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            a_new = a - lr * scale * vel
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                break
            a_new = a_new / s
            sc = score_normalized(a_new)
            if sc < best_score - 1e-13:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
                improved = True
                break
            elif sc < score_normalized(a) - 1e-13:
                a = a_new
                improved = True
                break

        if not improved:
            vel *= 0.5
            lr *= 0.95

        lr = max(lr_start / (1.0 + 0.001 * it), 1e-6)

        if it % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      EPD iter={it} ({elapsed:.0f}s): score={score_normalized(a):.7f} best={best_score:.7f}")

    return best_a, best_score


def coordinate_descent(a_init, time_limit, n_factors=6):
    """Coordinate descent with proportional redistribution."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    factors = np.array([0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5][:n_factors])
    no_improve = 0
    max_no_improve = 500
    rng = np.random.default_rng(777)
    iter_count = 0

    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        no_improve = 0
        iter_count += 1
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                return best_a, best_score
            old_val = a[i]
            if old_val < 1e-12:
                continue
            for f in factors:
                new_val = old_val * f
                a[i] = new_val
                a = np.maximum(a, 0)
                s = a.sum()
                if s < 1e-15:
                    a[i] = old_val
                    break
                a = a / s
                sc = score_normalized(a)
                if sc < best_score - 1e-13:
                    best_score = sc
                    best_a = a.copy()
                    no_improve = 0
                    break
                else:
                    a[i] = old_val
                    s = a.sum()
                    if s > 0:
                        a = a / s

        if no_improve > 100:
            # Perturb
            a = best_a.copy()
            idx = rng.integers(0, n, size=min(10, n))
            for ii in idx:
                a[ii] *= rng.uniform(0.7, 1.3)
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            no_improve = 0

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. LP-alternating direction finding (primary)
    2. Exact peak gradient descent with line search
    3. Multi-peak soft gradient with momentum
    4. Coordinate descent polishing
    5. Alternative length exploration
    6. Final refinement rounds
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.7f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = p_score

    # === Phase 1: LP Alternating (primary, ~40% time) ===
    print(f"\n=== Phase 1: LP Alternating (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 60:
            t_alloc = min(rem * 0.45, 500.0)
            print(f"  LP-Alt: n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_lp, _ = lp_alternating(best_seq, t_alloc, t_start=0.3)
            best_seq, best_score = check_eval(a_lp, "lp_alt1", best_seq, best_score)

    # Second LP run from updated best
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 60:
            t_alloc = min(rem * 0.35, 300.0)
            print(f"  LP-Alt 2: budget={t_alloc:.0f}s")
            a_lp2, _ = lp_alternating(best_seq, t_alloc, t_start=0.2)
            best_seq, best_score = check_eval(a_lp2, "lp_alt2", best_seq, best_score)

    # === Phase 2: Exact Peak Descent (~20% time) ===
    print(f"\n=== Phase 2: Exact Peak Descent (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 250
        if rem > 40:
            t_alloc = min(rem * 0.35, 250.0)
            print(f"  EPD: n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_ep, _ = exact_peak_descent(best_seq, t_alloc, lr_start=0.002)
            best_seq, best_score = check_eval(a_ep, "epd1", best_seq, best_score)

    # === Phase 3: Multi-Peak Gradient (~15% time) ===
    print(f"\n=== Phase 3: Multi-Peak Gradient (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 40:
            t_alloc = min(rem * 0.35, 200.0)
            print(f"  MPG: n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_mp, _ = multi_peak_gradient(best_seq, t_alloc, lr_start=0.002)
            best_seq, best_score = check_eval(a_mp, "mpg1", best_seq, best_score)

    # === Phase 4: Coordinate Descent (~15% time) ===
    print(f"\n=== Phase 4: Coordinate Descent (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            t_alloc = min(rem * 0.4, 150.0)
            print(f"  CD: n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_cd, _ = coordinate_descent(best_seq, t_alloc)
            best_seq, best_score = check_eval(a_cd, "cd1", best_seq, best_score)

    # === Phase 5: Alternative lengths (~10% time) ===
    print(f"\n=== Phase 5: Alternative Lengths (best={best_score:.7f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        alt_lengths = []
        for factor in [1.5, 2.0, 0.75, 1.33, 3.0]:
            alt_n = int(n_best * factor)
            if 128 <= alt_n <= 5120 and alt_n != n_best and alt_n not in alt_lengths:
                alt_lengths.append(alt_n)

        for alt_n in alt_lengths[:2]:
            rem = deadline - time.monotonic() - 80
            if rem < 30:
                break
            t_alloc = min(rem * 0.35, 80.0)
            init_alt = interpolate_sequence(best_seq, alt_n)
            a_alt, _ = multi_peak_gradient(init_alt, t_alloc, lr_start=0.003)
            best_seq, best_score = check_eval(a_alt, f"alt_n{alt_n}", best_seq, best_score)

    # === Phase 6: Final refinement rounds ===
    print(f"\n=== Phase 6: Final Refinement (best={best_score:.7f}) ===")
    if best_seq is not None:
        # LP from current best
        rem = deadline - time.monotonic() - 40
        if rem > 20:
            t_alloc = min(rem * 0.4, 50.0)
            print(f"  Final LP: budget={t_alloc:.0f}s")
            a_fl, _ = lp_alternating(best_seq, t_alloc, t_start=0.1)
            best_seq, best_score = check_eval(a_fl, "final_lp", best_seq, best_score)

        # Coord descent
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            t_alloc = min(rem * 0.5, 25.0)
            print(f"  Final CD: budget={t_alloc:.0f}s")
            a_fc, _ = coordinate_descent(best_seq, t_alloc, n_factors=4)
            best_seq, best_score = check_eval(a_fc, "final_cd", best_seq, best_score)

        # Exact peak (very small steps)
        rem = deadline - time.monotonic() - 10
        if rem > 8:
            t_alloc = min(rem * 0.6, 15.0)
            a_fp, _ = exact_peak_descent(best_seq, t_alloc, lr_start=0.0005)
            best_seq, best_score = check_eval(a_fp, "final_epd", best_seq, best_score)

    # Fallback
    if best_seq is None:
        if parent is not None:
            best_seq = parent
        else:
            best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.7f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
