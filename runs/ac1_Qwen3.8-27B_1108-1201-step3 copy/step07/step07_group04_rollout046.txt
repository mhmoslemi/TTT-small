        return np.ones(len(a)) / len(a)
    return a / s


def score_norm(a):
    """Score for sum=1 non-negative sequence."""
    n = len(a)
    return 2.0 * n * fast_convolve(a).max()


def soft_max_obj_grad(z, n, beta):
    """
    Smooth objective and gradient in log-space.
    a = exp(z) / sum(exp(z)), objective = 2n * soft_max(conv(a,a), beta)
    """
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    S = exp_z.sum()
    a = exp_z / S

    b = fast_convolve(a)

    # Soft-max: (1/beta) * log(sum(exp(beta*b)))
    beta_b = beta * b
    bb_max = beta_b.max()
    exp_bb = np.exp(beta_b - bb_max)
    exp_sum = exp_bb.sum()
    lse = (bb_max + np.log(max(exp_sum, 1e-300))) / beta
    obj = 2.0 * n * lse

    # Weights (softmax of beta*b)
    w = exp_bb / exp_sum

    # Gradient wrt a: grad_a[i] = 2n * sum_k w[k] * 2 * a[k-i] (approx)
    # Use cross-correlation via FFT for speed
    # sum_k w[k] * a[k-i] is cross-correlation of w and a
    # Pad to same length for FFT
    fft_len = 1 << (2 * n - 1 - 1).bit_length()
    W = rfft(w, fft_len)
    A = rfft(a, fft_len)
    # Cross-correlation: corr[i] = sum_k w[k+i] * a[i] = sum_k w[k] * a[k-i]
    # This is ifft(conj(W) * A) shifted... actually let's just use direct for correctness
    # For n=768, O(n*2n) = ~1.2M ops is fine
    grad_a = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * wk * a[k - idx]
    grad_a *= 2.0 * n

    # Transform to z-space
    S_grad = np.dot(grad_a, a)
    grad_z = a * (grad_a - S_grad * a)

    return obj, grad_z


def lbgfs_phase(a0, time_limit, beta_schedule, max_iter_per_beta=2000):
    """
    Multi-temperature L-BFGS optimization in log-space.
    beta_schedule: list of beta values (increasing = sharper)
    """
    n = len(a0)
    a_safe = np.maximum(a0, 1e-30)
    z0 = np.log(a_safe)
    z0 = z0 - z0.mean()
    t0 = monotonic()
    a_current = a0.copy()
    z_current = z0.copy()

    for beta in beta_schedule:
        rem = time_limit - (monotonic() - t0)
        if rem < 5:
            break
        print(f"    L-BFGS beta={beta}, remaining={rem:.0f}s")

        obj_list = []

        def func(z):
            obj, _ = soft_max_obj_grad(z, n, beta)
            return obj

        def jac(z):
            _, g = soft_max_obj_grad(z, n, beta)
            return g

        try:
            res = minimize(
                fun=func, x0=z_current, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': max_iter_per_beta, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': max_iter_per_beta * 20,
                         'maxcor': 30, 'disp': False}
            )
            z_current = res.x
        except Exception as e:
            print(f"    L-BFGS failed at beta={beta}: {e}")
            break

        # Convert back to a
        z_sh = z_current - z_current.max()
        a_new = np.exp(z_sh)
        a_new = a_new / a_new.sum()
        a_new = np.maximum(a_new, 0)
        a_new = normalize(a_new)

        s_new = score_norm(a_new)
        s_old = score_norm(a_current)
        if s_new < s_old:
            a_current = a_new
        print(f"    beta={beta}: score={score_norm(a_current):.10f} (iter={res.nit if hasattr(res,'nit') else '?'})")

    return a_current


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    global _lp_solver
    if '_lp_solver' in globals():
        return _lp_solver
    for solver_name in [cp.PDLP, cp.SCS, cp.ECOS, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return cp.ECOS


def lp_best_response(a_fixed, solver=None, max_iters_hint=500):
    """LP: minimize M subject to conv(a,b) <= M, b>=0, sum(b)=1."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, max_iter=max_iters_hint)
    except Exception:
        try:
            prob.solve(solver=cp.SCS)
        except Exception:
            return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def damped_lp_refine(a0, time_limit, solver=None, n_alpha_decay=300):
    """Damped LP refinement."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = score_norm(a)
    t0 = monotonic()
    it = 0

    while monotonic() - t0 < time_limit:
        b_new, M_val = lp_best_response(a, solver)
        if b_new is None:
            break
        alpha = min(1.0, 2.0 / (it + 50))
        a_new = (1 - alpha) * a + alpha * b_new
        a_new = normalize(a_new)
        sc = score_norm(a_new)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
        a = a_new
        it += 1
        if it % 100 == 0:
            print(f"    DLP it={it}: score={best_score:.10f}")

    return best_a, best_score


def peak_coordinate_descent(a0, time_limit, n_attempts=5000):
    """
    Aggressive coordinate descent: identify peak, reduce contributing pairs,
    redistribute mass to positions that don't contribute to peaks.
    """
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = score_norm(a)
    t0 = monotonic()
    stall = 0

    for step in range(n_attempts):
        rem = time_limit - (monotonic() - t0)
        if rem < 2:
            break

        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]

        # Find contributing pairs for the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        total_contrib = contribs.sum()
        if total_contrib < 1e-20:
            break

        # Find positions NOT contributing to any top-5 peak
        top_peaks = np.argpartition(b, -5)[-5:]
        peak_positions = set(int(p) for p in top_peaks)

        # For each significant contributor, try to reduce it
        top_c = np.argsort(contribs)[::-1][:8]
        improved = False

        for ci in top_c:
            i = int(idx[ci])
            j = int(peak_k - i)
            if a[i] < 1e-12:
                continue
            # Try reducing a[i] and redistributing
            for frac in [0.7, 0.5, 0.3]:
                reduce_amt = a[i] * (1 - frac)
                if reduce_amt < 1e-15:
                    continue
                # Find best redistribution target: position that minimizes new peak
                # Try positions far from peak
                a_trial = a.copy()
                a_trial[i] *= frac
                # Redistribute: add proportionally to low-contribution positions
                # Simple: add uniformly to positions not in peak contributing set
                mask = np.ones(n, dtype=bool)
                for pk in peak_positions:
                    ms = max(0, pk - n + 1)
                    me = min(n - 1, pk)
                    mask[ms:me + 1] = False
                mask[i] = False
                if mask.any():
                    a_trial[mask] += reduce_amt / mask.sum()
                    a_trial = normalize(a_trial)
                    sc = score_norm(a_trial)
                    if sc < best_score - 1e-13:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break

            if not improved and i != j and a[j] > 1e-12:
                # Try reducing the pair
                for f in [0.8, 0.6]:
                    a_trial = a.copy()
                    a_trial[i] *= f
                    a_trial[j] *= f
                    reduce_amt = a[i] * (1 - f) + a[j] * (1 - f)
                    a_trial = normalize(a_trial)
                    sc = score_norm(a_trial)
                    if sc < best_score - 1e-13:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break

            if improved:
                break

        if not improved:
            # Try small random perturbation
            for attempt in range(3):
                a_trial = a + np.random.standard_normal(n) * 0.001
                a_trial = normalize(a_trial)
                sc = score_norm(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break

        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 100:
                break

    return best_a, best_score


def random_restarts_local(a0, time_limit, n_restarts=5):
    """Try small random perturbations and do local optimization from each."""
    n = len(a0)
    best_a = normalize(a0.copy())
    best_score = score_norm(best_a)
    t0 = monotonic()
    time_per_restart = time_limit / n_restarts

    for r in range(n_restarts):
        rem = time_limit - (monotonic() - t0)
        if rem < 5:
            break
        # Perturb
        a_pert = best_a + np.random.standard_normal(n) * 0.005
        a_pert = normalize(a_pert)

        # Quick L-BFGS at moderate beta
        beta = 50
        z_safe = np.log(np.maximum(a_pert, 1e-30))
        z_safe = z_safe - z_safe.mean()

        def func_r(z):
            o, _ = soft_max_obj_grad(z, n, beta)
            return o

        def jac_r(z):
            _, g = soft_max_obj_grad(z, n, beta)
            return g

        try:
            res = minimize(fun=func_r, x0=z_safe, jac=jac_r, method='L-BFGS-B',
                           options={'maxiter': 500, 'ftol': 1e-16, 'gtol': 1e-13,
                                    'maxfun': 5000, 'maxcor': 20, 'disp': False})
            z_new = res.x
            z_sh = z_new - z_new.max()
            a_new = normalize(np.exp(z_sh))
            sc = score_norm(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                print(f"    Restart {r}: improved to {best_score:.10f}")
        except Exception:
            pass

    return best_a, best_score


def structural_init(n, scheme='raised_cosine'):
    """Generate structured initial sequences."""
    k = np.arange(n)
    if scheme == 'raised_cosine':
        a = np.cos(np.pi * k / n) ** 2
    elif scheme == 'tukey':
        alpha = 0.5
        a = np.ones(n)
        m = int(alpha * n / 2)
        a[:m] = 0.5 * (1 + np.cos(np.pi * (np.arange(m) + 0.5) / m))
        a[-m:] = a[:m][::-1]
    elif scheme == 'sinc_sq':
        center = (n - 1) / 2.0
        x = (k - center) / (n / 4.0)
        a = np.sinc(x) ** 2
    elif scheme == 'gabor':
        center = (n - 1) / 2.0
        sigma = n / 4.0
        a = np.exp(-0.5 * ((k - center) / sigma) ** 2)
    elif scheme == 'uniform':
        a = np.ones(n)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def check_update(a, label, best_seq, best_score):
    """Evaluate and potentially update best."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for auto-convolution minimization:
    1. Multi-temperature L-BFGS (beta from 1 to 500)
    2. Peak-directed coordinate descent
    3. LP-based damped refinement
    4. Random restarts with local optimization
    5. Final ultra-low-temperature L-BFGS polish
    """
    np.random.seed(seed)
    deadline = monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        best_seq = np.ones(768) / 768.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Initial (uniform): n=768, score={best_score:.8f}")

    n_parent = len(best_seq)
    a = best_seq.copy()  # Working sequence (sum=1)

    # === Phase 1: Multi-temperature L-BFGS on parent ===
    print(f"\n=== Phase 1: Multi-T L-BFGS (n={n_parent}, best={best_score:.8f}) ===")
    rem = deadline - monotonic() - 600
    if rem > 30:
        t1 = min(rem, 400.0)
        beta_schedule = [1, 2, 5, 10, 20, 50, 100, 200, 400]
        a_new = lbgfs_phase(a, t1, beta_schedule, max_iter_per_beta=3000)
        best_seq, best_score = check_update(a_new, "L-BFGS_multiT", best_seq, best_score)
        a = best_seq.copy() if best_seq is not None else a

    # === Phase 2: Coordinate descent ===
    print(f"\n=== Phase 2: Peak Coord Descent (best={best_score:.8f}) ===")
    rem = deadline - monotonic() - 400
    if rem > 20:
        t2 = min(rem * 0.5, 150.0)
        a_cd, _ = peak_coordinate_descent(a, t2, n_attempts=20000)
        best_seq, best_score = check_update(a_cd, "coord_descent", best_seq, best_score)
        a = best_seq.copy() if best_seq is not None else a

    # === Phase 3: LP-based refinement ===
    print(f"\n=== Phase 3: Damped LP (best={best_score:.8f}) ===")
    rem = deadline - monotonic() - 300
    if rem > 30:
        t3 = min(rem * 0.5, 180.0)
        solver = get_lp_solver()
        print(f"  LP solver: {solver}")
        a_lp, _ = damped_lp_refine(a, t3, solver)
        best_seq, best_score = check_update(a_lp, "damped_lp", best_seq, best_score)
        a = best_seq.copy() if best_seq is not None else a

    # === Phase 4: Second L-BFGS pass (lower temps) ===
    print(f"\n=== Phase 4: L-BFGS Pass 2 (best={best_score:.8f}) ===")
    rem = deadline - monotonic() - 250
    if rem > 20:
        t4 = min(rem * 0.6, 200.0)
        beta_schedule2 = [50, 100, 200, 400, 800]
        a_new2 = lbgfs_phase(a, t4, beta_schedule2, max_iter_per_beta=3000)
        best_seq, best_score = check_update(a_new2, "L-BFGS_P2", best_seq, best_score)
        a = best_seq.copy() if best_seq is not None else a

    # === Phase 5: Coordinate descent again ===
    print(f"\n=== Phase 5: Coord Descent 2 (best={best_score:.8f}) ===")
    rem = deadline - monotonic() - 200
    if rem > 15:
        t5 = min(rem * 0.5, 100.0)
        a_cd2, _ = peak_coordinate_descent(a, t5, n_attempts=20000)
        best_seq, best_score = check_update(a_cd2, "coord_descent_2", best_seq, best_score)
        a = best_seq.copy() if best_seq is not None else a

    # === Phase 6: Random restarts ===
    print(f"\n=== Phase 6: Random Restarts (best={best_score:.8f}) ===")
    rem = deadline - monotonic() - 150
    if rem > 20:
        t6 = min(rem * 0.6, 150.0)
        a_rr, _ = random_restarts_local(a, t6, n_restarts=5)
        best_seq, best_score = check_update(a_rr, "restarts", best_seq, best_score)
        a = best_seq.copy() if best_seq is not None else a

    # === Phase 7: Try different structural initializations ===
    print(f"\n=== Phase 7: Structural Inits (best={best_score:.8f}) ===")
    rem = deadline - monotonic() - 100
    if rem > 30:
        for scheme in ['raised_cosine', 'gabor', 'tukey']:
            rem = deadline - monotonic() - 80
            if rem < 20:
                break
            a_struct = structural_init(n_parent, scheme)
            # Quick L-BFGS
            t_struct = min(rem * 0.3, 40.0)
            a_s_result = lbgfs_phase(a_struct, t_struct, [10, 50, 200], max_iter_per_beta=1000)
            best_seq, best_score = check_update(a_s_result, f"struct_{scheme}", best_seq, best_score)

    # === Phase 8: Try nearby lengths ===
    print(f"\n=== Phase 8: Length Exploration (best={best_score:.8f}) ===")
    rem = deadline - monotonic() - 60
    if rem > 30:
        for alt_n in [n_parent - 16, n_parent + 16, n_parent + 32]:
            if alt_n < 256:
                continue
            rem = deadline - monotonic() - 50
            if rem < 20:
                break
            a_interp = interpolate_sequence(a, alt_n)
            t_alt = min(rem * 0.4, 40.0)
            a_alt = lbgfs_phase(a_interp, t_alt, [10, 50, 200], max_iter_per_beta=1500)
            best_seq, best_score = check_update(a_alt, f"alt_n{alt_n}", best_seq, best_score)

    # === Phase 9: Final ultra-low temperature L-BFGS ===
    print(f"\n=== Phase 9: Final Ultra-Low T Polish (best={best_score:.8f}) ===")
    rem = deadline - monotonic() - 10
    if rem > 10:
        t9 = min(rem * 0.7, 60.0)
        a_final = lbgfs_phase(a, t9, [200, 500, 1000], max_iter_per_beta=5000)
        best_seq, best_score = check_update(a_final, "final_ultraT", best_seq, best_score)
        a = best_seq.copy() if best_seq is not None else a

    # === Phase 10: Final coordinate descent ===
    print(f"\n=== Phase 10: Final Coord Descent (best={best_score:.8f}) ===")
    rem = deadline - monotonic() - 5
    if rem > 5:
        t10 = min(rem * 0.8, 20.0)
        a_fc, _ = peak_coordinate_descent(a, t10, n_attempts=10000)
        best_seq, best_score = check_update(a_fc, "final_coord", best_seq, best_score)

    # Final verification
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
