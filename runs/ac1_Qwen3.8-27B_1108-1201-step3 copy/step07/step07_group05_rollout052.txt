dC[k]/da[j] = d/d a[j] (sum_i a[i] * a[k-i])

Case 2: i=k-j. The term is a[k-j]*a[j]. This is the same as a[j]*a[k-j]. Derivative w.r.t. a[j] is a[k-j] (if k-j != j). If k = 2j, this is the same as Case 1.

If k = 2j: dC[k]/da[j] = 2*a[j]

In both cases: dC[k]/da[j] = 2*a[k-j] (with the understanding that a[k-j] = 0 if k-j is out of range).

import numpy as np

from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the optimization score for a normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-300:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def subgradient_at_peak(a, k_peak):
    """
    Compute subgradient of max_k C[k] at lag k_peak.
    grad[j] = 2 * a[k_peak - j] if k_peak - j in [0, n-1], else 0.
    """
    n = len(a)
    grad = np.zeros(n)
    # For each j in [0, n-1], grad[j] = 2*a[k_peak - j] if 0 <= k_peak - j < n
    js = max(0, k_peak - n + 1)
    je = min(n - 1, k_peak)
    if js <= je:
        indices = np.arange(js, je + 1)
        grad[indices] = 2.0 * a[k_peak - indices][::-1]
    return grad

def smoothed_lbgfs(a0, time_limit, T=100.0, maxiter=3000):
    """L-BFGS on the smooth log-sum-exp approximation in probability space."""
    from scipy.optimize import minimize
    n = len(a0)
    t0 = time.monotonic()

    def objective(a):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = project_simplex(a)
        conv = fast_convolve(a)
        shifted = conv / T
        max_s = shifted.max()
        lse = max_s + np.log(np.exp(shifted - max_s).sum())
        return T * lse

    def gradient(a):
        if time.monotonic() - t0 > time_limit:
            return np.zeros(n)
        a = project_simplex(a)
        conv = fast_convolve(a)
        shifted = conv / T
        max_s = shifted.max()
        w = np.exp(shifted - max_s)
        w = w / w.sum()
        # Gradient: 2 * (a correlated with w)
        fft_size = 1 << (2 * n - 1 - 1).bit_length()
        A = rfft(a, fft_size)
        W = rfft(w, fft_size)
        grad_corr = irfft(np.conj(A) * W, fft_size)
        idx = np.array([(fft_size - j) % fft_size for j in range(n)])
        grad = 2.0 * grad_corr[idx]
        return grad

    try:
        res = minimize(
            fun=objective, x0=a0, jac=gradient,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 30, 'disp': False}
        )
        a_out = project_simplex(res.x)
        score = compute_score(a_out)
        if score < compute_score(a0):
            return a_out
    except Exception:
        pass
    return a0


def targeted_peak_reduction(a0, time_limit, n_top_peaks=8, verbose=False):
    """
    Directly target the top peaks. For each top peak, find the pair of positions
    (i, k-i) that contributes most, and try to reduce it by moving mass to a
    position that doesn't contribute to that peak.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    iters = 0
    stagnation = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        top_k = min(n_top_peaks, len(conv))
        peak_indices = np.argpartition(conv, -top_k)[-top_k:]
        peak_indices = peak_indices[np.argsort(conv[peak_indices])[::-1]]

        improved = False
        for pk in peak_indices[:5]:
            pk = int(pk)
            # Contributing pairs: (i, pk-i) for valid i
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[pk - idx]
            # Find top contributors
            top_c = np.argsort(contribs)[::-1][:3]
            for ci in top_c:
                i = idx[ci]
                j = pk - i
                if a[i] < 1e-14:
                    continue
                # Try moving mass from i to a random position r that doesn't contribute to peak pk
                for _ in range(5):
                    r = rng.integers(0, n)
                    if r == i or r == j:
                        continue
                    # Check if r would contribute to peak pk
                    if 0 <= pk - r < n:
                        continue  # skip if it would also contribute
                    for frac in [0.02, 0.05, 0.1, 0.15, 0.2]:
                        delta = a[i] * frac
                        a_trial = a.copy()
                        a_trial[i] -= delta
                        a_trial[r] += delta
                        if a_trial[r] < 0 or a_trial[i] < -1e-15:
                            continue
                        a_trial = np.maximum(a_trial, 0)
                        ss = a_trial.sum()
                        if ss < 1e-15:
                            continue
                        a_trial = a_trial / ss
                        sc = compute_score(a_trial)
                        if sc < best_score - 1e-14:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
            if improved:
                break

        if not improved:
            stagnation += 1
            if stagnation > 10:
                # Random perturbation
                noise = rng.standard_normal(n) * 0.005
                a = best_a + noise
                a = project_simplex(a)
                stagnation = 0
        else:
            stagnation = 0

        if verbose and iters % 50 == 0:
            print(f"    TPR iter={iters}, best={best_score:.8f}")

    return best_a, best_score


def pairwise_line_search(a0, time_limit, n_pairs=200, verbose=False):
    """
    For random pairs (i,j), find the best fraction to transfer using ternary search.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(77)
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False

        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            if a[i] < 1e-14:
                continue

            # Ternary search for optimal transfer from i to j
            # f(t) = score(a - t*e_i + t*e_j)
            lo = 0.0
            hi = a[i] * 0.99

            # Quick check: try a few fixed fractions first
            best_t = 0.0
            best_local = best_score
            for frac in [0.01, 0.05, 0.1, 0.2, 0.3, 0.5]:
                t = frac * a[i]
                a_trial = a.copy()
                a_trial[i] -= t
                a_trial[j] += t
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score(a_trial)
                if sc < best_local - 1e-14:
                    best_local = sc
                    best_t = t

            # Also try reverse direction
            if a[j] > 1e-14:
                for frac in [0.01, 0.05, 0.1, 0.2, 0.3]:
                    t = frac * a[j]
                    a_trial = a.copy()
                    a_trial[j] -= t
                    a_trial[i] += t
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    sc = compute_score(a_trial)
                    if sc < best_local - 1e-14:
                        best_local = sc
                        best_t = -t

            if best_t != 0.0:
                if best_t > 0:
                    a[i] -= best_t
                    a[j] += best_t
                else:
                    a[j] -= (-best_t)
                    a[i] += (-best_t)
                a = np.maximum(a, 0)
                ss = a.sum()
                a = a / ss
                best_score = best_local
                best_a = a.copy()
                improved = True

        if not improved and iters > 20:
            # Perturb
            noise = rng.standard_normal(n) * 0.003
            a = best_a + noise
            a = project_simplex(a)

        if verbose and iters % 100 == 0:
            print(f"    PLS iter={iters}, best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def generate_inits(n, parent, rng):
    """Generate diverse initial sequences of length n."""
    inits = []

    # Parent interpolated
    if parent is not None:
        inits.append(interpolate_sequence(parent, n))

    # Uniform
    inits.append(np.ones(n) / n)

    # Triangular
    t = np.linspace(0, 1, n)
    tri = np.sin(np.pi * t)
    tri = tri / tri.sum()
    inits.append(tri)

    # Cosine bump
    cos = np.cos(np.pi * (t - 0.5)) ** 2
    cos = np.maximum(cos, 0)
    cos = cos / cos.sum()
    inits.append(cos)

    # Random
    r = rng.random(n)
    r = r / r.sum()
    inits.append(r)

    # Two peaks
    t = np.linspace(0, 1, n)
    two = np.exp(-100 * (t - 0.25) ** 2) + np.exp(-100 * (t - 0.75) ** 2)
    two = two / two.sum()
    inits.append(two)

    # Linear ramp
    ramp = np.arange(1, n + 1, dtype=float)
    ramp = ramp / ramp.sum()
    inits.append(ramp)

    # Alternating high-low (smoothed)
    alt = np.where(np.arange(n) % 2 == 0, 1.5, 0.5)
    alt = alt / alt.sum()
    inits.append(alt)

    return inits


def quick_nesterov(a0, time_limit):
    """Quick Nesterov for initial exploration."""
    n = len(a0)
    a = a0.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    y = a.copy()
    lr = 0.5 / np.sqrt(n)
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(y)
        k_peak = int(np.argmax(conv))
        grad = subgradient_at_peak(y, k_peak)
        momentum = 2.0 / (iters + 2.0)
        y = y - lr * grad
        y = project_simplex(y)
        score = 2.0 * n * fast_convolve(y).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = y.copy()
        a = y
        lr *= 0.9995

    return best_a, best_score


def run_full_optimization(a0, time_budget, verbose=False):
    """Run a full optimization pipeline on a single starting point."""
    n = len(a0)
    a = a0.copy()
    best_score = compute_score(a)
    t_start = time.monotonic()
    deadline = t_start + time_budget

    if verbose:
        print(f"  Full opt: n={n}, initial={best_score:.8f}, budget={time_budget:.0f}s")

    # Stage 1: Nesterov subgradient (adaptive T)
    rem = deadline - time.monotonic()
    if rem > 10:
        t_alloc = min(rem * 0.35, 120.0)
        a, sc = nesterov_subgradient(a, t_alloc, T_start=100, T_end=0.5, verbose=verbose)
        if sc < best_score:
            best_score = sc

    # Stage 2: Smoothed L-BFGS
    rem = deadline - time.monotonic()
    if rem > 10:
        t_alloc = min(rem * 0.25, 80.0)
        a2 = smoothed_lbgfs(a, t_alloc, T=200, maxiter=2000)
        sc2 = compute_score(a2)
        if sc2 < best_score:
            best_score = sc2
            a = a2

    # Stage 3: Targeted peak reduction
    rem = deadline - time.monotonic()
    if rem > 10:
        t_alloc = min(rem * 0.25, 80.0)
        a3, sc3 = targeted_peak_reduction(a, t_alloc, n_top_peaks=10, verbose=verbose)
        if sc3 < best_score:
            best_score = sc3
            a = a3

    # Stage 4: Pairwise line search
    rem = deadline - time.monotonic()
    if rem > 10:
        t_alloc = min(rem * 0.25, 60.0)
        a4, sc4 = pairwise_line_search(a, t_alloc, n_pairs=300, verbose=verbose)
        if sc4 < best_score:
            best_score = sc4
            a = a4

    # Stage 5: Final Nesterov polish
    rem = deadline - time.monotonic()
    if rem > 5:
        t_alloc = min(rem * 0.5, 30.0)
        a5, sc5 = nesterov_subgradient(a, t_alloc, T_start=10, T_end=0.01, verbose=False)
        if sc5 < best_score:
            best_score = sc5
            a = a5

    return a, best_score


def check_and_update(a, label, best_seq, best_score, verbose=True):
    """Evaluate and update best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-strategy optimization:
    1. Get parent sequence, evaluate
    2. Aggressive length sweep with quick Nesterov
    3. Full optimization on top candidates
    4. Final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    t_global_start = time.monotonic()

    best_seq = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 512
        parent = np.ones(n_init) / n_init
        t = np.linspace(0, 1, n_init)
        parent = parent + 0.1 * np.sin(2 * np.pi * 2 * t)
        parent = np.maximum(parent, 0)
        parent = parent / parent.sum()
        best_seq = parent.copy()
        best_score = compute_score(parent)
        print(f"Initialized: n={n_init}, score={best_score:.8f}")

    n_parent = len(parent)

    # === Phase 1: Aggressive length sweep ===
    print(f"\n=== Phase 1: Length Sweep (best={best_score:.8f}) ===")
    lengths_to_try = sorted(set([
        n_parent, n_parent - 2, n_parent + 2,
        n_parent // 2, n_parent * 2,
        256, 384, 512, 640, 768, 1024, 1280, 1536, 2048
    ]))
    lengths_to_try = [n for n in lengths_to_try if 128 <= n <= 2560]

    candidates = []  # (length, sequence, score)

    for n_try in lengths_to_try:
        rem = deadline - time.monotonic()
        if rem < 30:
            break

        inits = generate_inits(n_try, parent, rng)
        best_for_n = None
        best_sc_for_n = float('inf')

        for ii, init in enumerate(inits[:4]):
            rem = deadline - time.monotonic()
            if rem < 10:
                break
            t_alloc = min(rem * 0.08, 15.0)
            a_q, sc_q = quick_nesterov(init, t_alloc)
            if sc_q < best_sc_for_n:
                best_sc_for_n = sc_q
                best_for_n = a_q

        if best_for_n is not None:
            candidates.append((n_try, best_for_n, best_sc_for_n))
            print(f"  n={n_try}: score={best_sc_for_n:.8f}")

    # Sort by score
    candidates.sort(key=lambda x: x[2])

    # === Phase 2: Full optimization on top candidates ===
    print(f"\n=== Phase 2: Full Optimization (best={best_score:.8f}) ===")
    top_k = min(4, len(candidates))
    for idx in range(top_k):
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        n_cand, a_cand, sc_cand = candidates[idx]
        t_alloc = min(rem * 0.25, 200.0)
        print(f"  Candidate {idx}: n={n_cand}, score={sc_cand:.8f}, budget={t_alloc:.0f}s")
        a_opt, sc_opt = run_full_optimization(a_cand, t_alloc, verbose=True)
        best_seq, best_score = check_and_update(a_opt, f"full_{idx}_n{n_cand}", best_seq, best_score)

    # === Phase 3: Re-optimize best with more time ===
    print(f"\n=== Phase 3: Deep Optimize Best (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t_alloc = min(rem * 0.4, 300.0)
        print(f"  Deep optimize: n={len(best_seq)}, budget={t_alloc:.0f}s")
        a_deep, sc_deep = run_full_optimization(best_seq, t_alloc, verbose=True)
        best_seq, best_score = check_and_update(a_deep, "deep", best_seq, best_score)

    # === Phase 4: Length refinement around best ===
    print(f"\n=== Phase 4: Length Refinement (best={best_score:.8f}) ===")
    n_best = len(best_seq)
    for delta in [-3, -2, -1, 1, 2, 3, 4, -4]:
        alt_n = n_best + delta
        if alt_n < 64 or alt_n > 3000:
            continue
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t_alloc = min(rem * 0.15, 60.0)
        alt_seq = interpolate_sequence(best_seq, alt_n)
        a_alt, sc_alt = run_full_optimization(alt_seq, t_alloc, verbose=False)
        best_seq, best_score = check_and_update(a_alt, f"refine_n{alt_n}", best_seq, best_score)

    # === Phase 5: Final polish with multiple methods ===
    print(f"\n=== Phase 5: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_alloc = min(rem * 0.3, 60.0)
        a_tpr, _ = targeted_peak_reduction(best_seq, t_alloc, n_top_peaks=15)
        best_seq, best_score = check_and_update(a_tpr, "tpr_final", best_seq, best_score)

    rem = deadline - time.monotonic()
    if rem > 20:
        t_alloc = min(rem * 0.3, 50.0)
        a_pls, _ = pairwise_line_search(best_seq, t_alloc, n_pairs=500)
        best_seq, best_score = check_and_update(a_pls, "pls_final", best_seq, best_score)

    rem = deadline - time.monotonic()
    if rem > 10:
        t_alloc = min(rem * 0.4, 30.0)
        a_ns, _ = nesterov_subgradient(best_seq, t_alloc, T_start=5, T_end=0.01)
        best_seq, best_score = check_and_update(a_ns, "ns_final", best_seq, best_score)

    # === Phase 6: Random restarts from best ===
    print(f"\n=== Phase 6: Random Restarts (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    n_restarts = max(1, int(rem // 60))
    for ri in range(min(n_restarts, 5)):
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t_alloc = min(rem * 0.2, 50.0)
        # Perturb best
        noise = rng.standard_normal(len(best_seq)) * 0.01
        a_pert = best_seq + noise
        a_pert = project_simplex(a_pert)
        a_rs, sc_rs = run_full_optimization(a_pert, t_alloc, verbose=False)
        best_seq, best_score = check_and_update(a_rs, f"restart_{ri}", best_seq, best_score)

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))

    return [float(x) for x in best_seq]
