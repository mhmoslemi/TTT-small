import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog
import time

def fast_autoconv(a):
    """Compute linear auto-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def lse_gradient(a, t):
    """
    Compute gradient of (1/t)*log(sum(exp(t*b_k))) where b = a*a (linear conv).
    Returns gradient vector of length n.
    """
    n = len(a)
    b = fast_autoconv(a)
    # Numerically stable softmax
    b_shifted = t * (b - b.max())
    exp_b = np.exp(b_shifted)
    p = exp_b / exp_b.sum()
    
    # Cross-correlation: grad_j = sum_m a_m * p_{j+m}
    # Using FFT: ifft(conj(fft(a,L)) * fft(p,L))[:n]
    fft_size = 1 << (3 * n - 1).bit_length()
    A = fft(a, fft_size)
    P = fft(p, fft_size)
    grad = ifft(np.conj(A) * P, fft_size)[:n].real
    return grad

def adam_smooth_max(a_init, time_limit, t_start=5.0, t_end=500.0, 
                     n_steps=5000, lr=0.01, betas=(0.9, 0.999), eps=1e-8,
                     t_anneal_frac=0.5, seed=42):
    """
    Adam optimizer on the log-sum-exp smooth max of auto-convolution.
    Constraint: a >= 0, sum(a) = 1 (projected).
    """
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    
    best_a = a.copy()
    best_obj = fast_autoconv(a).max()
    
    n_anneal = int(n_steps * t_anneal_frac)
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Anneal t
        if step < n_anneal:
            t = t_start * (t_end / t_start) ** (step / max(1, n_anneal - 1))
        else:
            t = t_end
        
        # Compute gradient
        grad = lse_gradient(a, t)
        
        # Adam update
        m = betas[0] * m + (1 - betas[0]) * grad
        v = betas[1] * v + (1 - betas[1]) * grad ** 2
        m_hat = m / (1 - betas[0] ** (step + 1))
        v_hat = v / (1 - betas[1] ** (step + 1))
        
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0)
        
        # Normalize
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s
        
        # Evaluate
        obj = fast_autoconv(a).max()
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Adam optimization with 
log-sum-exp smoothing, spectral initialization, LP-based refinement,
and multi-scale exploration.

Key innovations:
- Log-sum-exp smooth objective with annealing for differentiable optimization
- Adam optimizer for adaptive per-coordinate learning rates
- FFT-accelerated gradient computation via cross-correlation
- Spectral initialization (flat power spectrum heuristic)
- Aggressive LP refinement with line search
- Multi-length and multi-initialization exploration
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import linprog
import time


def fast_autoconv(a):
    """Compute linear auto-convolution of a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def lse_gradient(a, t):
    """
    Compute gradient of (1/t)*log(sum(exp(t*b_k))) where b = a*a (linear conv).
    Uses FFT cross-correlation for O(n log n) computation.
    """
    n = len(a)
    b = fast_autoconv(a)
    # Numerically stable softmax
    b_shifted = t * (b - b.max())
    exp_b = np.exp(b_shifted)
    p = exp_b / exp_b.sum()
    
    # Cross-correlation: grad_j = sum_m a_m * p_{j+m}
    # FFT: ifft(conj(fft(a,L)) * fft(p, L))[:n]
    fft_size = 1 << (3 * n).bit_length()
    A = fft(a, fft_size)
    P = fft(p, fft_size)
    grad = ifft(np.conj(A) * P, fft_size)[:n].real
    return grad


def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def adam_smooth_max(a_init, time_limit, t_start=5.0, t_schedule=None,
                    lr=0.005, betas=(0.9, 0.999), eps=1e-8,
                    anneal_frac=0.7, plateau_frac=0.3, seed=42):
    """
    Adam optimizer on log-sum-exp smooth max of auto-convolution.
    Constraint: a >= 0, normalized to sum=1.
    """
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    m = np.zeros(n)
    v = np.zeros(n)
    
    best_a = a.copy()
    best_obj = fast_autoconv(a).max()
    step = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Anneal t
        progress = min(1.0, step / max(1, int(10000 * anneal_frac)))
        if progress < 1.0:
            # Exponential annealing
            t_end = 200.0
            t = t_start * (t_end / t_start) ** progress
        else:
            t = 200.0 + (step - int(10000 * anneal_frac)) * 0.01
            t = min(t, 1000.0)
        
        # Compute gradient
        grad = lse_gradient(a, t)
        
        # Adam update
        step += 1
        m = betas[0] * m + (1 - betas[0]) * grad
        v = betas[1] * v + (1 - betas[1]) * grad ** 2
        m_hat = m / (1 - betas[0] ** step)
        v_hat = v / (1 - betas[1] ** step)
        
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0)
        
        # Normalize
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s
        
        # Evaluate (every few steps to save time)
        if step % 5 == 0 or step < 10:
            obj = fast_autoconv(a).max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        # Adaptive learning rate
        if no_improve > 200:
            lr *= 0.8
            no_improve = 0
        if no_improve > 100:
            lr *= 0.95
    
    return best_a, best_obj


def lp_refine(f, max_iters=20, time_limit=60.0):
    """
    LP-based refinement: solve max sum(g) s.t. conv(f,g) <= M, g >= 0.
    Blend with current f to reduce max convolution.
    """
    t0 = time.monotonic()
    n = len(f)
    a = f.copy().astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = fast_autoconv(a).max()
    
    for iteration in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_autoconv(a)
        M = b.max()
        
        # Build LP constraint matrix (sparse construction)
        num_constraints = 2 * n - 1
        # A[k, j] = a[k-j] if 0 <= k-j <= n-1
        A = np.zeros((num_constraints, n))
        for k in range(num_constraints):
            # j ranges from max(0, k-n+1) to min(n-1, k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            js = np.arange(lo, hi + 1)
            A[k, js] = a[k - js]
        
        b_lp = np.full(num_constraints, M)
        c = -np.ones(n)
        
        try:
            res = linprog(c, A_ub=A, b_ub=b_lp,
                          bounds=[(0.0, None)] * n, method='highs')
            if not res.success:
                break
            
            g = res.x
            g_sum = g.sum()
            if g_sum <= 1.0 + 1e-6:
                break
            
            g_norm = g / g_sum
            
            # Line search for blending
            best_t = 0.0
            best_blend_obj = best_obj
            t_candidates = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
            for t in t_candidates:
                blend = (1.0 - t) * best_a + t * g_norm
                blend = np.maximum(blend, 0.0)
                ss = blend.sum()
                if ss < 1e-10:
                    continue
                blend = blend / ss
                obj_blend = fast_autoconv(blend).max()
                if obj_blend < best_blend_obj:
                    best_blend_obj = obj_blend
                    best_t = t
            
            if best_t > 0.0:
                a = (1.0 - best_t) * best_a + best_t * g_norm
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                obj = fast_autoconv(a).max()
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
                else:
                    a = best_a.copy()
            else:
                break
        except Exception:
            break
    
    return best_a, best_obj


def spectral_init(n, seed=42):
    """
    Initialize with approximately flat power spectrum.
    Generate random phases, take IFFT, take absolute value, clip.
    """
    rng = np.random.default_rng(seed)
    # Generate random complex spectrum with unit magnitude
    phases = rng.uniform(0, 2 * np.pi, n)
    spec = np.exp(1j * phases)
    # Make it Hermitian for real output
    if n % 2 == 0:
        spec[n//2] = 1.0  # Nyquist
    else:
        spec[n//2] = 1.0
    # IFFT
    a = ifft(spec, n).real
    # Take absolute value and clip
    a = np.abs(a)
    a = np.maximum(a, 0.01 * a.max())
    a = a / a.sum()
    return a


def make_gaussian_init(n, sigma=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_uniform_init(n):
    return np.ones(n) / n


def make_tukey_init(n, r=0.5):
    a = np.ones(n)
    taper = max(1, int(r * n // 2))
    if taper < n:
        t = np.linspace(0, 1, taper)
        a[:taper] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t / r - 1.0)))
        a[-taper:] = a[:taper][::-1]
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def run_optimization(a_init, time_limit, n, seed=42):
    """Combined Adam + LP optimization."""
    t0 = time.monotonic()
    
    # Phase A: Adam with log-sum-exp
    t_adam = time_limit * 0.5
    a_adam, obj_adam = adam_smooth_max(a_init, time_limit=t_adam, seed=seed)
    
    best_a = a_adam.copy()
    best_obj = obj_adam
    
    # Phase B: LP refinement
    rem = time_limit - (time.monotonic() - t0)
    if rem > 10:
        a_lp, obj_lp = lp_refine(best_a, max_iters=15, time_limit=rem * 0.6)
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()
    
    # Phase C: More Adam from LP result
    rem = time_limit - (time.monotonic() - t0)
    if rem > 10:
        a_adam2, obj_adam2 = adam_smooth_max(
            best_a, time_limit=rem * 0.7,
            t_start=50.0, lr=0.002, seed=seed + 12345
        )
        if obj_adam2 < best_obj:
            best_obj = obj_adam2
            best_a = a_adam2.copy()
    
    # Phase D: Final LP
    rem = time_limit - (time.monotonic() - t0)
    if rem > 5:
        a_lp2, obj_lp2 = lp_refine(best_a, max_iters=10, time_limit=rem)
        if obj_lp2 < best_obj:
            best_obj = obj_lp2
            best_a = a_lp2.copy()
    
    return best_a, best_obj
