```python
def simplex_project(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        rho_val = np.nonzero(cond)[0][-1] + 1
        theta = (cssv[rho_val - 1] - 1.0) / rho_val
    else:
        theta = v.mean()
    return np.maximum(v - theta, 0.0)
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adam optimizer on log-sum-exp smoothed auto-convolution peak
2. Adaptive smoothing parameter (annealing omega)
3. Multi-start with diverse initializations at multiple sequence lengths
4. Sequential coordinate descent with golden section search
5. 2D pairwise golden section moves to escape 1D local minima
6. Peak-targeted variable reduction as a final heuristic

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        rho_val = np.nonzero(cond)[0][-1] + 1
        theta = (cssv[rho_val - 1] - 1.0) / rho_val
    else:
        theta = v.mean()
    return np.maximum(v - theta, 0.0)


def compute_smooth_gradient(a, p):
    """
    Compute gradient of smoothed max of conv(a,a).
    grad_j = 2 * sum_{m=0}^{n-1} p[j+m] * a[m]
    where p are the softmax weights over convolution values.
    """
    n = len(a)
    grad = np.zeros(n)
    for j in range(n):
        grad[j] = 2.0 * np.dot(p[j:j + n], a)
    return grad


def adam_optimize(a0, time_limit, verbose=False):
    """
    Adam optimizer on log-sum-exp smoothed auto-convolution peak.
    a0 is assumed to have sum=1 and non-negative.
    """
    n = len(a0)
    a = a0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.008
    omega_start = 1.5 * np.sqrt(n)
    omega_decay = 0.9997  # multiplicative increase per step
    omega = omega_start

    t0 = time.monotonic()
    no_improve = 0
    n_evals = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        # Compute convolution
        b = fast_convolve(a)
        b_max = b.max()

        # Compute softmax weights
        log_w = omega * (b - b_max)
        log_w -= log_w.max()
        p = np.exp(log_w)
        p_sum = p.sum()
        if p_sum < 1e-30:
            p = np.zeros(len(b))
            p[int(np.argmax(b))] = 1.0
        else:
            p /= p_sum

        # Compute gradient
        grad = compute_smooth_gradient(a, p)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = simplex_project(a_new)

        # Check score periodically (every 20 iterations)
        if t % 20 == 0:
            a_new = np.maximum(a_new, 0)
            a_new = simplex_project(a_new)
            b_new = fast_convolve(a_new)
            score = 2.0 * n * b_new.max()
            if score < best_score - 1e-12:
                best_score = score
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1

        a = a_new
        omega *= omega_decay
        lr *= 0.99995

        # Adaptive restart if stuck
        if no_improve > 200:
            if no_improve > 600:

                a = simplex_project(best_a + noise)

                noise = rng.standard_normal(n) * 0.02
                a = simplex_project(a + noise)
                lr = lr * 1.5
                no_improve = 0

def golden_section_1d(a0, i, time_per_eval=0.002):
    """
    Golden section search on coordinate i of a0 (keeping sum=1).
    Returns the optimized sequence and improvement.
    """
    n = len(a0)
    a = a0.copy()
    a[i] = 0.0
    s_rest = a.sum()
    if s_rest < 1e-15:
        return a0, 0.0

    base = a.copy()
    base_sum = s_rest

    def eval_at(x):
        trial = base.copy()
        trial[i] = x
        trial = trial / trial.sum()
        return 2.0 * n * fast_convolve(trial).max()

    # Search range: [0, 1] for the fraction x_i / (x_i + rest)
    golden = (np.sqrt(5) - 1) / 2
    lo, hi = 0.0, 1.0
    x1 = lo + (1 - golden) * (hi - lo)
    x2 = lo + golden * (hi - lo)
    f1 = eval_at(x1 * base_sum)
    f2 = eval_at(x2 * base_sum)

    best_x = x1 if f1 < f2 else x2
    best_f = min(f1, f2)
    base_score = 2.0 * n * fast_convolve(a0 / a0.sum()).max()

    for _ in range(40):
        if f1 < f2:
            hi = x2
            x2 = x1
            f2 = f1
            x1 = lo + (1 - golden) * (hi - lo)
            f1 = eval_at(x1 * base_sum)
        else:
            lo = x1
            x1 = x2
            f1 = f2
            x2 = lo + golden * (hi - lo)
            f2 = eval_at(x2 * base_sum)

        if f1 < best_f:
            best_f = f1
            best_x = x1
        if f2 < best_f:
            best_f = f2
            best_x = x2

        if hi - lo < 1e-10:
            break

    trial = base.copy()
    trial[i] = best_x * base_sum
    trial = trial / trial.sum()
    final_score = 2.0 * n * fast_convolve(trial).max()

    if final_score < base_score - 1e-14:
        return trial, base_score - final_score
    return a0, 0.0


def golden_section_2d(a0, i, j, time_limit=5.0):
    """
    2D golden section: vary ratio between coordinates i and j keeping their sum fixed.
    """
    n = len(a0)
    a = a0.copy()
    s_ij = a[i] + a[j]
    if s_ij < 1e-14:
        return a0, 0.0
    s_rest = a.sum() - s_ij

    base = a.copy()
    base[i] = 0.0
    base[j] = 0.0

    def eval_at(frac):
        """frac is the fraction of s_ij going to coordinate i."""
        trial = base.copy()
        trial[i] = frac * s_ij
        trial[j] = (1 - frac) * s_ij
        trial_sum = trial.sum()
        if trial_sum < 1e-15:
            return 1e10
        trial = trial / trial_sum
        return 2.0 * n * fast_convolve(trial).max()

    base_score = 2.0 * n * fast_convolve(a).max()

    # Golden section on frac in [0, 1]
    golden = (np.sqrt(5) - 1) / 2
    lo, hi = 0.0, 1.0
    x1 = lo + (1 - golden) * (hi - lo)
    x2 = lo + golden * (hi - lo)
    f1 = eval_at(x1)
    f2 = eval_at(x2)

    t0 = time.monotonic()
    for _ in range(60):
        if time.monotonic() - t0 > time_limit:
            break
        if f1 < f2:
            hi = x2
            x2 = x1
            f2 = f1
            x1 = lo + (1 - golden) * (hi - lo)
            f1 = eval_at(x1)
        else:
            lo = x1
            x1 = x2
            f1 = f2
            x2 = lo + golden * (hi - lo)
            f2 = eval_at(x2)

        if hi - lo < 1e-10:
            break

    best_frac = x1 if f1 < f2 else x2
    best_f = min(f1, f2)

    trial = base.copy()
    trial[i] = best_frac * s_ij
    trial[j] = (1 - best_frac) * s_ij
    trial_sum = trial.sum()
    if trial_sum < 1e-15:
        return a0, 0.0
    trial = trial / trial_sum
    final_score = 2.0 * n * fast_convolve(trial).max()

    if final_score < base_score - 1e-14:
        return trial, base_score - final_score
    return a0, 0.0


def coord_descent_golden(a0, time_limit, n_sweeps=100, use_2d=True):
    """
    Sequential coordinate descent with golden section search.
    Periodically attempts 2D moves.
    """
    n = len(a0)
    a = a0.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        perm = rng.permutation(n)

        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-12:
                continue

            # 1D golden section on coordinate i
            new_a, imp = golden_section_1d(a, i)
            if imp > 1e-14:
                a = new_a
                if 2.0 * n * fast_convolve(a).max() < best_score:
                    best_score = 2.0 * n * fast_convolve(a).max()
                    best_a = a.copy()
                improved = True

        # 2D moves every 3 sweeps
        if use_2d and sweep % 3 == 2:
            if time.monotonic() - t0 > time_limit:
                break
            # Find pairs that both contribute to the peak
            b = fast_convolve(a)
            peak_k = int(np.argmax(b))
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_pairs = np.argsort(contribs)[::-1][:6]

            for ci in range(len(top_pairs) - 1):
                if time.monotonic() - t0 > time_limit:
                    break
                i1 = int(idx[top_pairs[ci]])
                i2 = int(idx[top_pairs[ci + 1]])
                if i1 == i2:
                    continue
                if a[i1] < 1e-12 or a[i2] < 1e-12:
                    continue
                new_a, imp = golden_section_2d(a, i1, i2, time_limit=3.0)
                if imp > 1e-14:
                    a = new_a
                    sc = 2.0 * n * fast_convolve(a).max()
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
                    improved = True

        if not improved and sweep > 10:
            break

    return best_a, best_score

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length, preserving sum=1."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n, endpoint=False)
    new_x = np.linspace(0, 1, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm, period=1)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def init_fourier(n, rng, n_harmonics=None):
    """Initialize using random Fourier superposition."""
    if n_harmonics is None:
        n_harmonics = int(rng.integers(3, min(20, n // 4)))
    amplitudes = rng.random(n_harmonics) * 0.4
    phases = rng.uniform(0, 2 * np.pi, n_harmonics)
    freqs = np.arange(1, n_harmonics + 1)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harmonics):
        a += amplitudes[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def init_snap_to_grid(n, base, rng, n_grids=None):
    """Initialize by snapping a base to a low-resolution grid then upscaling."""
    if n_grids is None:
        n_grids = int(rng.integers(3, max(4, n // 30)))
    grid = np.zeros(n_grids)
    for b in range(n_grids):
        start = int(b * n / n_grids)
        end = int((b + 1) * n / n_grids)
        if end <= start:
            end = start + 1
        grid[b] = base[start:end].mean() + rng.standard_normal() * 0.1
    grid = np.maximum(grid, 0.01)
    grid = grid / grid.sum()
    # Upscale
    a = np.zeros(n)
    for b in range(n_grids):
        start = int(b * n / n_grids)
        end = int((b + 1) * n / n_grids)
        if end <= start:
            end = start + 1
        a[start:end] = grid[b]
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

    a = a0.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    def remaining():
        return max(0, time_budget - (time.monotonic() - t0))

    # Stage 1: Adam (60% of budget)
    t_adam = remaining() * 0.5
    if t_adam > 5:
        if verbose:
            print(f"    Adam: budget={t_adam:.0f}s")
        a, sc, _ = adam_optimize(a, t_adam)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    # Stage 2: Coordinate descent (25% of remaining)
    t_cd = remaining() * 0.35
    if t_cd > 5:
        if verbose:
            print(f"    CoordDescent: budget={t_cd:.0f}s (current best={best_score:.6f})")
        a, sc = coord_descent_golden(best_a, t_cd, n_sweeps=50, use_2d=True)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    # Stage 3: Peak shave (10% of remaining)
    t_ps = remaining() * 0.2
    if t_ps > 3:
        if verbose:
            print(f"    PeakShave: budget={t_ps:.0f}s")
        a, sc = peak_shave_aggressive(best_a, t_ps)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    # Stage 4: Fine Adam (10% of remaining)
    t_fine = remaining() * 0.3
    if t_fine > 3:
        if verbose:
            print(f"    FineAdam: budget={t_fine:.0f}s")
        a, sc, _ = adam_optimize(best_a, t_fine)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    # Stage 5: Final quick CD
    t_final = remaining() * 0.5
    if t_final > 3:
        a, sc = coord_descent_golden(best_a, t_final, n_sweeps=30, use_2d=True)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def propose_candidate():
    """
    Multi-phase optimization:
    Phase 1: Diverse multi-start with quick Adam at multiple lengths
    Phase 2: Intensive full optimization on top candidates
    Phase 3: Length exploration and cross-pollination
    Phase 4: Final intensive polish on the best
    """
    deadline = time.monotonic() + 1790.0
    rng = np.random.default_rng(2024)

    best_seq = None
    best_score = float('inf')

    # Try to use parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = 2.0 * len(parent) * fast_convolve(parent).max()
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    # ===================== PHASE 1: Multi-start =====================
    print("\n=== Phase 1: Multi-Start ===")
    t1_start = time.monotonic()
    t_phase1 = 200.0

    n_values = [640, 700, 768, 832, 896, 960, 1120]
    if parent is not None:
        np_ = len(parent)
        if np_ not in n_values:
            n_values.insert(0, np_)

    candidates = []

    for n in n_values:
        rem = deadline - time.monotonic() - (1800 - t1_start - t_phase1 + 100)
        if time.monotonic() - t1_start > t_phase1:
            break
        n_inits = 4
        per_init_time = max(3.0, (t_phase1 - (time.monotonic() - t1_start)) / (n_values.index(n) + 1) / n_inits - 2)

        for c in range(n_inits):
            if parent is not None and c == 0:
                cand = interpolate_seq(parent, n)
            elif c == 1:
                cand = init_fourier(n, rng)
            elif c == 2:
                base = interpolate_seq(parent, n) if parent is not None else np.ones(n) / n
                cand = init_snap_to_grid(n, base, rng)
            else:
                cand = init_fourier(n, rng, n_harmonics=int(rng.integers(5, 15)))

            t_quick = min(15.0, max(3.0, per_init_time))
            a_opt, s_opt, _ = adam_optimize(cand, t_quick)
            candidates.append((s_opt, a_opt, n, c))

    candidates.sort(key=lambda x: x[0])
    top5 = candidates[:5]
    print(f"  Generated {len(candidates)} candidates")
    for s_c, a_c, n_c, c_c in top5:
        print(f"    n={n_c} type={c_c}: {s_c:.6f}")
        if s_c < best_score:
            best_score = s_c
            best_seq = a_c.copy()

    # ===================== PHASE 2: Intensive Optimization =====================
    print(f"\n=== Phase 2: Intensive Optimization (best={best_score:.6f}) ===")
    t2_start = time.monotonic()
    t_phase2 = 650.0

    # Optimize top 3 candidates
    for rank, (s_c, a_c, n_c, c_c) in enumerate(top3 := candidates[:3]):
        rem = deadline - time.monotonic() - 900
        if rem < 60:
            break
        t_alloc = min(t_phase2 / 3, rem * 0.8)
        if t_alloc < 30:
            break
        print(f"  Optimize candidate {rank} (n={n_c}, init_score={s_c:.6f}), budget={t_alloc:.0f}s")
        a_opt, s_opt = full_optimize(a_c, t_alloc, verbose=True)
        if s_opt < best_score:
            best_score = s_opt
            best_seq = a_opt.copy()
            print(f"  NEW BEST: {best_score:.6f} (n={len(best_seq)})")

    # ===================== PHASE 3: Length Exploration =====================
    print(f"\n=== Phase 3: Length Exploration (best={best_score:.6f}) ===")
    t3_start = time.monotonic()
    t_phase3 = 450.0

    if best_seq is not None:
        n_best = len(best_seq)

        # Try nearby lengths
        alt_lengths = []
        for delta in [-64, -32, 32, 64, -96, 96, 128]:
            alt_n = n_best + delta
            if 300 <= alt_n <= 2000:
                alt_lengths.append(alt_n)

        # Also try some fixed good lengths
        for fixed_n in [700, 800, 850, 900]:
            if fixed_n != n_best and fixed_n not in alt_lengths:
                alt_lengths.append(fixed_n)

        alt_lengths = alt_lengths[:5]

        for alt_n in alt_lengths:
            rem = deadline - time.monotonic() - 300
            if rem < 40:
                break
            t_alloc = min(t_phase3 / len(alt_lengths), rem * 0.7)
            if t_alloc < 20:
                break

            init_alt = interpolate_seq(best_seq, alt_n)
            print(f"  Alt length n={alt_n}, budget={t_alloc:.0f}s")
            a_alt, s_alt = full_optimize(init_alt, t_alloc, verbose=False)
            if s_alt < best_score:
                best_score = s_alt
                best_seq = a_alt.copy()
                print(f"  NEW BEST at n={alt_n}: {best_score:.6f}")

    # ===================== PHASE 4: Final Polish =====================
    print(f"\n=== Phase 4: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Intensive coordinate descent
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            t_cd = min(rem * 0.4, 120.0)
            print(f"  Final CD, budget={t_cd:.0f}s")
            a_fin, s_fin = coord_descent_golden(best_seq, t_cd, n_sweeps=100, use_2d=True)
            if s_fin < best_score:
                best_score = s_fin
                best_seq = a_fin.copy()
                print(f"  After CD: {best_score:.6f}")

        # Peak shave
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            t_ps = min(rem * 0.4, 60.0)
            print(f"  Final PeakShave, budget={t_ps:.0f}s")
            a_fin, s_fin = peak_shave_aggressive(best_seq, t_ps)
            if s_fin < best_score:
                best_score = s_fin
                best_seq = a_fin.copy()
                print(f"  After PeakShave: {best_score:.6f}")

        # One more Adam with very small lr
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            t_adam = min(rem * 0.5, 40.0)
            a_fin, s_fin, _ = adam_optimize(best_seq, t_adam)
            if s_fin < best_score:
                best_score = s_fin
                best_seq = a_fin.copy()
                print(f"  After FineAdam: {best_score:.6f}")

        # Final micro CD
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_fin, s_fin = coord_descent_golden(best_seq, rem, n_sweeps=50, use_2d=False)
            if s_fin < best_score:
                best_score = s_fin
                best_seq = a_fin.copy()

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = 2.0

    # Final verification with evaluate_sequence
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
