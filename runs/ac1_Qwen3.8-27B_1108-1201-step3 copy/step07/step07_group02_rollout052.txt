Multi-strategy optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses: (1) L-BFGS with adaptive log-sum-exp smoothing, (2) LP-based alternating

import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def conv_matrix_apply(a, b):
    """Compute conv(a, b) efficiently."""
    return np.convolve(a, b, mode='full')


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def solve_lp_best_response(a_fixed, solver=None):

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)


def lbgfs_optimize(a0, time_limit, T_start, T_end, n_stages, maxiter_per_stage):
    """L-BFGS with log-sum-exp smoothing and geometric temperature schedule."""
    n = len(a0)
    z = z_from_a(a0)
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)

            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:

            log_sum_exp = max_Tb + np.log(exp_Tb.sum())

                    continue

            S = np.dot(grad_a, a)
            return a * (grad_a - S)

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n + 1)
    new_x = np.linspace(0, 1, new_n + 1)
    f_interp = np.interp(new_x[:-1], old_x, np.concatenate([f_norm, [f_norm[-1]]]))
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

Let me also think about a key insight I might be missing: **the structure of the optimal solution**.

For the zero-sum game interpretation, at equilibrium, both players are playing mixed strategies. Player 1's strategy is the optimal `a`, and Player 2's strategy is a distribution `w` over the convolution positions `k` such that `sum_k w_k * b_k = max_k b_k` (i.e., `w` is concentrated on the argmax positions).

At equilibrium, the "cost" to Player 1 of playing pure strategy `e_i` (putting all mass at position i) against Player 2's mixed strategy `w` should be at most the value:
`(Gw)_i = sum_k w_k * (conv(e_i, e_i))_k <= V` for all `i`.

Where `V` is the game value (= `max_k b_k` at equilibrium, with `sum a = 1`).

This gives us a certificate: if we can find `w` such that `(Gw)_i <= V` for all `i`, then `V` is an upper bound on the optimal value.

This suggests a dual approach: directly optimize `w` to maximize the minimum of `(Gw)_i`. But `(Gw)_i = sum_k w_k * a_{k-i}` (where `a` is the equilibrium sequence)... this is circular.

Actually, the correct dual is: minimize `V` subject to `sum_k w_k * T_{k,i} <= V` for all `i`, `sum w_k = 1`, `w_k >= 0`, where `T_{k,i} = (conv(a, e_i))_k`. But `T` depends on `a`, so it's not a standard LP.

**Key improvement for LP: Use the solution as warm start.**

If I solve the LP once and get `b*`, then in the next iteration (with a slightly updated `y`), the optimal `b` should be close to `b*`. Many LP solvers benefit from warm starts, but cvxpy doesn't directly support this. Instead, I can:

**Another key idea: Use the symmetric structure.**

The convolution `conv(a, a)` is symmetric: `b_k = b_{2n-2-k}`. So we only need to worry about `k = 0, 1, ..., n-1` (the first half). The peak is likely at `k = n-1` (the center).

This means the effective number of constraints is `n`, not `2n-1`. The LP becomes:
`min M` s.t. `(T @ b)_k <= M` for `k = 0, ..., n-1`, `sum(b) = 1`, `b >= 0`.

This is a smaller LP and should solve faster.

Let me also think about **using a different inner loop method instead of LP**. For the inner problem:
`min_b max_k (T @ b)_k` s.t. `sum(b) = 1, b >= 0`

This is a convex problem. We can solve it with:
- Mirror descent (exponentiated gradient) on the simplex
- Or projected subgradient

Mirror descent would be:
```
b_{t+1} = softmax(z_t - lr * g_t)
```
where `g_t` is the subgradient of `max_k (T @ b)_k` at `b_t`, and `lr` is the learning rate.

"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines: L-BFGS with log-sum-exp smoothing, LP-based Nesterov alternating
(zero-sum game equilibrium), targeted peak shaving, and coordinate descent.
Uses adaptive time allocation and multi-length exploration.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size_max = 2 * n - 1
    fft_size = 1 << (size_max - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size_max]


def build_conv_matrix_half(a):
    """Build n x n matrix T_half such that (T_half @ b)[:n] = conv(a,b)[:n].
    We only need the first n coefficients since conv is symmetric."""
    n = len(a)
    T = np.zeros((n, n))
    for j in range(n):
        # Row k, col j: contribution of a[j] to b[k] is a[k-j] if valid
        # b[k] = sum_i a[i]*a[k-i], so T[k,j] = a[k-j] if 0<=k-j<n else 0
        # For k in 0..n-1, j in 0..n-1: valid if k-j >= 0, i.e., j <= k
        # Also need k-j < n which is always true since k < n and j >= 0
        for k in range(j, n):
            T[k, j] = a[k - j]
    return T


def build_conv_matrix_full_fast(a):
    """Build (2n-1) x n matrix T such that T @ b = conv(a, b). Vectorized."""
    n = len(a)
    # Use the structure: T[k,j] = a[k-j] if 0<=k-j<n, else 0
    # This is a banded matrix
    idx_k = np.repeat(np.arange(2 * n - 1), n)
    idx_j = np.tile(np.arange(n), 2 * n - 1)
    diff = idx_k - idx_j
    valid = (diff >= 0) & (diff < n)
    T = np.zeros((2 * n - 1, n), dtype=np.float64)
    T[valid] = a[diff[valid]]
    return T


def get_lp_solver():
    """Find a working LP solver."""
    global _lp_solver_cached
    if '_lp_solver_cached' in globals():
        return _lp_solver_cached
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return _lp_solver_cached


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (nonneg, sum=1), find b>=0, sum(b)=1 minimizing max_k conv(a,b)_k."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    # Only use first n constraints (by symmetry, max is in first half or at center)
    # Actually use all 2n-1 for correctness
    T = build_conv_matrix_full_fast(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)


def lbgfs_optimize(a0, time_limit, T_start, T_end, n_stages, maxiter_per_stage):
    """L-BFGS with log-sum-exp smoothing and geometric temperature schedule."""
    n = len(a0)
    z = z_from_a(a0)
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        def func(z_val):
            a = softmax_from_z(z_val)
            b = fast_convolve(a)
            Tb = stage_T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / stage_T) * log_sum_exp

        def jac(z_val):
            a = softmax_from_z(z_val)
            b = fast_convolve(a)
            Tb = stage_T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                wk = w[k]
                if wk < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * wk * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-12, 'maxfun': maxiter_per_stage * 100,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass

        a_current = softmax_from_z(z)
        score = 2.0 * n * fast_convolve(a_current).max()
        print(f"    L-BFGS T={stage_T:.2f}: score={score:.8f} ({time.monotonic()-t0:.0f}s)")

    return softmax_from_z(z)


def nesterov_lp(a0, time_limit, solver=None, n_max_iters=50000):
    """Nesterov-accelerated LP alternating for zero-sum game equilibrium."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    a = a / a.sum()
    y = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stale = 0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            stale += 1
            if stale > 100:
                break
            continue

        alpha = 2.0 / (it + 2)
        y_new = (1 - alpha) * y + alpha * b_new
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            break
        y_new = y_new / sy

        # Weighted average for actual iterate
        a = ((it + 1) * a + b_new) / (it + 2)
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y_new.copy()
            sa = 1.0
        a = a / sa
        y = y_new

        # Evaluate both
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stale = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stale = 0
        else:
            stale += 1

        if stale > 2000:
            break

        if (it + 1) % 500 == 0:
            print(f"    NLP iter={it+1} ({time.monotonic()-t0:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def two_point_cd(a0, time_limit, n_trials=100000, seed=99):
    """Random two-point coordinate descent with mass transfer."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0

    for step in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > 5000:
            break

        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        if a[i] < 1e-14:
            src, dst = j, i
        elif a[j] < 1e-14:
            src, dst = i, j
        elif a[i] >= a[j]:
            src, dst = i, j
        else:
            src, dst = j, i

        for frac in [0.01, 0.03, 0.05]:
            transfer = a[src] * frac
            if transfer <= 0:
                continue
            a_trial = a.copy()
            a_trial[src] -= transfer
            a_trial[dst] += transfer
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-15:
                a = a_trial
                best_score = score
                best_a = a.copy()
                no_improve = 0
                break
        else:
            no_improve += 1

        if step % 10000 == 0:
            print(f"    2PtCD {time.monotonic()-t0:.0f}s: score={best_score:.8f}")

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

                try_update(a7c, f"PS-n{alt_n}")

    # === PHASE 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish (n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    if rem > 10:
        t8a = min(rem * 0.35, 50.0)
        if t8a > 8:
            a8 = lbgfs_optimize(best_seq, t8a, T_start=1.0, T_end=0.1,
                                n_stages=3, maxiter_per_stage=4000)
            try_update(a8, "Final-LBFGS")

    rem = deadline - time.monotonic()
    if rem > 8:
        t8b = min(rem * 0.35, 35.0)
        if t8b > 5:
            a8b, _ = peak_shave_topk(best_seq, t8b, top_k=7, n_steps=20000)
            try_update(a8b, "Final-PS")

    rem = deadline - time.monotonic()
    if rem > 5:
        t8c = min(rem * 0.30, 25.0)
        if t8c > 3:
            a8c, _ = two_point_cd(best_seq, t8c, n_trials=30000, seed=314)
            try_update(a8c, "Final-CD")

    # Final verification
    if best_seq is not None:
        fs = eval_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
