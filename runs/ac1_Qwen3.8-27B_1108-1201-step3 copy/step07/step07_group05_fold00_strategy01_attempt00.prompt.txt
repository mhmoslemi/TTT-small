def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the optimization score for a normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-30)
    return np.log(a_safe)


def smoothed_objective(z, T, n):
    """Soft-max smoothed objective in log-space."""
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    log_sum_exp = max_shifted + np.log(np.exp(shifted - max_shifted).sum())
    return (2.0 * n / T) * np.exp(T * log_sum_exp)


def smoothed_gradient(z, T, n):
    """Gradient of the soft-max smoothed objective in log-space."""
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    w = np.exp(shifted - max_shifted)
    w = w / w.sum()
    # Gradient w.r.t. a: 2n * (w * a) convolved (cross-correlation)
    # Actually: d(max_k c_k)/da_i = 2 * sum_k w_k * a_{k-i}
    # Cross-correlation of w with a
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
    # Project gradient to log-space (variance-preserving)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return (2.0 * n / T) * grad_z

def coordinate_descent(a0, time_limit, n_pairs_per_iter=200, verbose=False):
    """
    Coordinate descent using pairwise moves.
    For each pair (i,j), find the optimal transfer t that minimizes the score.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0, compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    iters = 0

    # Precompute random pairs (biased towards center where action happens)
    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False

        # Generate random pairs
        pairs = []
        for _ in range(n_pairs_per_iter):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 1e-12 and a[j] > 1e-12:
                pairs.append((i, j))
            elif i != j and a[i] > 1e-12:
                pairs.append((i, j))

        for (i, j) in pairs:
            if a[i] < 1e-14:
                continue

            # Try transferring a fraction of a[i] to a[j]
            # Binary/ternary search for optimal t
            # t > 0: move from i to j; t < 0: move from j to i
            best_t = 0.0
            best_local_score = best_score

            # Try several transfer sizes
            base = min(a[i], a[j]) if a[j] > 0 else a[i]
            for frac in [0.1, 0.3, 0.5, 0.7, 0.9, 0.01, 0.05]:
                t = frac * base
                if t >= a[i] - 1e-15:
                    t = a[i] * 0.5

                a_trial = a.copy()
                a_trial[i] -= t
                a_trial[j] += t
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score(a_trial)
                if sc < best_local_score - 1e-14:
                    best_local_score = sc
                    best_t = t

                # Also try negative transfer
                if a[j] > 1e-15:
                    t2 = frac * a[j]
                    a_trial2 = a.copy()
                    a_trial2[j] -= t2
                    a_trial2[i] += t2
                    a_trial2 = np.maximum(a_trial2, 0)
                    ss2 = a_trial2.sum()
                    if ss2 < 1e-15:
                        continue
                    a_trial2 = a_trial2 / ss2
                    sc2 = compute_score(a_trial2)
                    if sc2 < best_local_score - 1e-14:
                        best_local_score = sc2
                        best_t = -t2

            if best_t != 0.0:
                a[i] -= best_t
                a[j] += best_t
                a = np.maximum(a, 0)
                ss = a.sum()
                a = a / ss
                best_score = best_local_score
                best_a = a.copy()
                improved = True

        if not improved and iters > 50:
            # Perturb to escape local minimum
            noise = rng.standard_normal(n) * 0.001
            a = best_a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()

        if verbose and iters % 50 == 0:
            print(f"  CD iter={iters}, score={best_score:.8f}")

    return best_a, best_score


def simulated_annealing(a0, time_limit, n_moves_per_iter=100,
                        t_start=0.01, t_end=0.0001, verbose=False):
    """Simulated annealing with local moves."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    current_score = best_score
    t0 = time.monotonic()
    total_time = time_limit
    rng = np.random.default_rng(999)
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        # Temperature schedule (exponential cooling)
        progress = min(1.0, (time.monotonic() - t0) / total_time)
        temp = t_start * (t_end / t_start) ** progress

        improved_this_iter = False
        for _ in range(n_moves_per_iter):
            if time.monotonic() - t0 > time_limit:
                break

            # Choose move type
            move_type = rng.random()
            a_trial = a.copy()

            if move_type < 0.4:
                # Single coordinate: scale a[i] and redistribute
                i = rng.integers(0, n)
                if a[i] < 1e-14:
                    continue
                factor = 1.0 + rng.standard_normal() * 0.1
                new_val = max(0, a[i] * factor)
                diff = a[i] - new_val
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others_sum = a[mask].sum()
                if others_sum < 1e-15:
                    continue
                a_trial[i] = new_val
                a_trial[mask] = a[mask] + diff * a[mask] / others_sum

            elif move_type < 0.7:
                # Pairwise swap
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue
                t = rng.standard_normal() * 0.01 * max(a[i], a[j], 1e-6)
                a_trial[i] -= t
                a_trial[j] += t

            elif move_type < 0.85:
                # Block move: modify a contiguous block
                block_len = min(n, rng.integers(3, max(4, n // 50)))
                start = rng.integers(0, n - block_len + 1)
                idx = np.arange(start, start + block_len)
                noise = rng.standard_normal(block_len) * 0.01
                a_trial[idx] += noise
            else:
                # Two-point: adjust two coordinates
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue
                t1 = rng.standard_normal() * 0.02 * a[i]
                t2 = rng.standard_normal() * 0.02 * a[j] if a[j] > 0 else 0
                a_trial[i] -= t1
                a_trial[j] -= t2
                # Redistribute to keep sum
                redist = t1 + t2
                if abs(redist) > 1e-18:
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    mask[j] = False
                    others = a[mask]
                    os = others.sum()
                    if os > 1e-15:
                        a_trial[mask] = others + redist * others / os

            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-15:
                continue
            a_trial = a_trial / ss
            trial_score = compute_score(a_trial)

            # Accept or reject
            delta = trial_score - current_score
            if delta < 0 or rng.random() < np.exp(-delta / temp):
                a = a_trial
                current_score = trial_score
                if trial_score < best_score - 1e-14:
                    best_score = trial_score
                    best_a = a_trial.copy()
                    improved_this_iter = True

        if verbose and iters % 100 == 0:
            print(f"  SA iter={iters}, temp={temp:.6f}, best={best_score:.8f}")

    return best_a, best_score


def peak_equalization(a0, time_limit, n_peaks=10, verbose=False):
    """
    Iteratively try to equalize the top peaks of the auto-correlation.
    At the optimum, the top peaks should be approximately equal.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        # Find top peaks
        top_k = min(n_peaks, len(conv))
        peak_indices = np.argpartition(conv, -top_k)[-top_k:]
        peak_values = conv[peak_indices]
        max_peak = peak_values.max()
        min_peak = peak_values.min()

        if max_peak - min_peak < 1e-8:
            # Peaks are well equalized
            break

        # Try to reduce the maximum peak
        max_idx = int(np.argmax(conv))
        js = max(0, max_idx - n + 1)
        je = min(n - 1, max_idx)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[max_idx - idx]

        # Find which pairs contribute most to the peak
        top_contrib = np.argsort(contribs)[::-1][:5]

        improved = False
        for ci in top_contrib:
            i = idx[ci]
            j = max_idx - i
            if a[i] < 1e-14:
                continue

            # Try reducing a[i] and redistributing
            for frac in [0.05, 0.1, 0.2, 0.3]:
                delta = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= delta
                taski = np.ones(n, dtype=bool)
                taski[i] = False
                others = a_trial[taski]
                os = others.sum()
                if os < 1e-15:
                    continue
                a_trial[taski] = others + delta * others / os
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = compute_score(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try a random perturbation focused on peak-contributing positions
            peak_positions = set()
            for pi in peak_indices[:5]:
                si = max(0, int(pi) - n + 1)
                ei = min(n - 1, int(pi))
                peak_positions.update(range(si, ei + 1))
            if peak_positions:
                pos_list = list(peak_positions)
                for _ in range(10):
                    p = pos_list[rng.integers(0, len(pos_list))]
                    a_trial = a.copy()
                    delta = a[p] * 0.1 * rng.standard_normal()
                    a_trial[p] -= delta
                    taski2 = np.ones(n, dtype=bool)
                    taski2[p] = False
                    oth2 = a_trial[taski2]
                    o2s = oth2.sum()
                    if o2s > 1e-15:
                        a_trial[taski2] = oth2 + delta * oth2 / o2s
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break

        if verbose and iters % 20 == 0:
            print(f"  PE iter={iters}, best={best_score:.8f}, peak_spread={max_peak-min_peak:.2e}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving non-negativity and sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def nesterov_gradient(a0, time_limit, T=300, lr_init=0.05, momentum=0.9):
    """Nesterov accelerated gradient descent in log-space."""
    n = len(a0)
    z = z_from_a(a0)
    v = np.zeros(n)
    best_z = z.copy()
    best_score = compute_score(a0)
    t0 = time.monotonic()
    lr = lr_init
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1

        # Nesterov lookahead
        z_lookahead = z - momentum * v
        a_la = softmax_from_z(z_lookahead)
        grad = smoothed_gradient(z_lookahead, T, n)

        v = momentum * v + lr * grad
        z_new = z - v

        # Project: ensure z doesn't go to extreme values
        z_new = np.clip(z_new, -20, 20)

        a_new = softmax_from_z(z_new)
        score = compute_score(a_new)

        if score < best_score - 1e-14:
            best_score = score
            best_z = z_new.copy()

        z = z_new
        lr *= 0.9998

    a_out = softmax_from_z(best_z)
    return a_out, best_score
