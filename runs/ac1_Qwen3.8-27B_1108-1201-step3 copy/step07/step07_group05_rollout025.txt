def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the optimization score for a sequence (sum normalized to 1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-30)
    return np.log(a_safe)

def fictitious_play_step(a, q, n_vars, n_inner_iters=50, lr=None):
    """
    One step of fictitious play: solve min sum_k q_k * (f*f)_k s.t. sum(f)=1, f>=0
    using projected gradient in the simplex.
    """
    n = n_vars
    Q_f = q  # The "weights" on each lag
    
    def quad_obj(f):
        """Compute sum_k Q[k] * (f*f)[k]"""
        conv = fast_convolve(f)
        return np.dot(Q_f, conv)
    
    def quad_grad(f):
        """Gradient of quad_obj: 2 * cross-correlation of (Q * conv) structure"""
        conv = fast_convolve(f)
        weighted = Q_f * conv  # This gives sum_k Q[k] * conv[k] * ... 
        # d/da_j [sum_k Q[k] conv[k]] = sum_k Q[k] * d(conv_k)/da_j = sum_k Q[k] * 2*a[k-j]
        # = 2 * cross-correlation of (Q * elementwise?) 
        # Actually: sum_k Q[k] * conv[k] = sum_k Q[k] * sum_i a_i * a_{k-i}
        # d/da_j = sum_k Q[k] * 2 * a_{k-j} (for valid k-j)
        # = 2 * sum_k Q[k] * a_{k-j}
        # This is 2 * (a cross-correlated with Q, but Q has length 2n-1)
        # = 2 * sum_{k=j}^{j+2n-2} Q[k] * a_{k-j}
        # But a_{k-j} is only valid for k-j in [0,n-1], so k in [j, j+n-1]
        # = 2 * sum_{m=0}^{n-1} Q[j+m] * a[m]
        # This is the cross-correlation of Q and a at lag j
        
        fft_size = 1 << (3 * n - 2 - 1).bit_length()
        Qft = rfft(Q_f, fft_size)
        Aft = rfft(f, fft_size)
        corr = irfft(Qft * np.conj(Aft), fft_size)[:n]
        return 2.0 * corr
    
    # Projected gradient on simplex
    f = a.copy()
    if lr is None:
        lr = 1.0 / (2.0 * n)
    
    for _ in range(n_inner_iters):
        g = quad_grad(f)
        # Mirror descent step (exponentiated gradient)
        z = np.log(np.maximum(f, 1e-15)) - lr * g
        f_new = softmax_from_z(z)
        # Check improvement
        if quad_obj(f_new) < quad_obj(f):
            f = f_new
        else:
            lr *= 0.5
            z = np.log(np.maximum(f, 1e-15)) - lr * g
            f = softmax_from_z(z)
    
    return f


def fictitious_play_optimize(a0, time_limit, n_inner_iters=30, verbose=False):

    """

        # Solve inner problem
        f_new = fictitious_play_step(a, q, n, n_inner_iters)
        
        # Update weights based on current peaks

        max_val = conv.max()
        if max_val > 0:
            # Temperature-based weighting: peaks get more weight
            temp = 0.01
            q_new = np.exp(conv / (temp * max_val))

def power_sum_lbgfs(a0, time_limit, p=4, maxiter=5000):
    """L-BFGS optimization of power-sum smoothed objective in log-space."""
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    result = {'a': a0.copy(), 'score': compute_score(a0)}
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = softmax_from_z(z)
        conv = fast_convolve(a)
        return np.power(np.sum(conv ** p), 1.0 / p)
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        a = softmax_from_z(z)
        grad_a = grad_power_sum(a, p)
        # Project to log-space
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)
        return grad_z
    
    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                     'maxfun': maxiter * 40, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
        score = compute_score(a_out)
        if score < result['score']:
            result['a'] = a_out
            result['score'] = score
    except Exception:
        pass
    
    result['seconds'] = time.monotonic() - t0
    return result

def active_set_equalization(a0, time_limit, n_peaks=10, verbose=False):

    Iteratively equalize the top peaks of the convolution.
    At the optimum, top peaks should be approximately equal.
    Uses bisection to find the target level and projected gradient to achieve it.

        max_peak = peak_values.max()
        min_peak = peak_values.min()
        
        if max_peak - min_peak < max_peak * 1e-6:
            # Well equalized, break
            break

        max_idx = int(np.argmax(conv))
        js = max(0, max_idx - n + 1)
        je = min(n - 1, max_idx)
        
        if js > je:
            break
            
        idx = np.arange(js, je + 1)

        total_contrib = contribs.sum()
        if total_contrib < 1e-15:

                os = others.sum()

                a_trial[mask] = others + delta * others / os
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-10:
                    continue
                a_trial = a_trial / ss
                sc = compute_score(a_trial)

                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:

            n_tries = 20
            for _ in range(n_tries):
                p = idx[rng.integers(0, len(idx))]
                a_trial = a.copy()
                delta = a[p] * 0.05 * rng.standard_normal()
                a_trial[p] -= delta
                mask = np.ones(n, dtype=bool)
                mask[p] = False
                oth = a_trial[mask]
                o2s = oth.sum()
                if o2s > 1e-15:
                    a_trial[mask] = oth + delta * oth / o2s
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss > 1e-10:

    return best_a, best_score


def softmax_smoothed_lbgfs(a0, time_limit, T, maxiter=5000):
    """L-BFGS with soft-max smoothing."""
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    result = {'a': a0.copy(), 'score': compute_score(a0)}
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = softmax_from_z(z)
        conv = fast_convolve(a)
        shifted = conv / T
        max_s = shifted.max()
        lse = max_s + np.log(np.exp(shifted - max_s).sum())
        return (2.0 * n / T) * np.exp(T * lse)
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        a = softmax_from_z(z)
        conv = fast_convolve(a)
        shifted = conv / T
        max_s = shifted.max()
        w = np.exp(shifted - max_s)
        w = w / w.sum()
        # Gradient of max(conv) approx: 2 * cross-corr(w, a)
        fft_size = 1 << (3 * n - 2 - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        grad_a = 2.0 * irfft(W * np.conj(A), fft_size)[:n]
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)
        return (2.0 * n / T) * grad_z
    
    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                     'maxfun': maxiter * 40, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
        score = compute_score(a_out)
        if score < result['score']:
            result['a'] = a_out
            result['score'] = score
    except Exception:
        pass
    
    result['seconds'] = time.monotonic() - t0
    return result

    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    current_score = best_score
    t0 = time.monotonic()
    total_time = time_limit
    rng = np.random.default_rng(31337)
    
    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / total_time)
        temp = t_start * (t_end / t_start) ** progress
        
        for _ in range(n_moves):
            if time.monotonic() - t0 > time_limit:
                break
            
            a_trial = a.copy()
            move_type = rng.random()
            
            if move_type < 0.3:
                # Pairwise transfer
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue
                t = rng.standard_normal() * 0.02 * max(a[i], a[j], 1e-6)
                a_trial[i] -= t
                a_trial[j] += t
            elif move_type < 0.6:
                # Single coordinate scale + redistribute
                i = rng.integers(0, n)
                if a[i] < 1e-12:
                    continue
                factor = 1.0 + rng.standard_normal() * 0.15
                new_val = max(0, a[i] * factor)
                diff = a[i] - new_val
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a[mask]
                os = others.sum()
                if os < 1e-15:
                    continue
                a_trial[i] = new_val
                a_trial[mask] = others + diff * others / os
            else:
                # Block modification
                block_len = min(n, max(3, n // 100))
                start = rng.integers(0, max(1, n - block_len))
                idx = np.arange(start, start + block_len)
                a_trial[idx] += rng.standard_normal(block_len) * 0.01
            
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial = a_trial / ss
            trial_score = compute_score(a_trial)
            
            delta = trial_score - current_score
            if delta < 0 or rng.random() < np.exp(-delta / (temp + 1e-10)):
                a = a_trial
                current_score = trial_score
                if trial_score < best_score - 1e-12:
                    best_score = trial_score
                    best_a = a_trial.copy()
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score, verbose=True):
    """Evaluate candidate and update best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def mirrored_optimize(a0, time_limit):
    """
    Exploit the fact that if a is optimal, reverse(a) is also optimal.
    Optimize on the "symmetrized" half and mirror back.
    For odd n, the middle element is free; for even n, all elements come in pairs.
    Actually, this doesn't directly help since the sequence need not be symmetric.
    Instead, try optimizing a + reverse(a) / 2 as a starting point.
    """
    n = len(a0)
    a_sym = (a0 + a0[::-1]) / 2.0
    a_sym = np.maximum(a_sym, 0)
    a_sym = a_sym / a_sym.sum()
    
    # Quick optimization on the symmetrized version
    result = power_sum_lbgfs(a_sym, min(30, time_limit), p=4, maxiter=2000)
    return result['a'], result['score']


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-strategy optimization:
    1. Fictitious play (peak balancing via weighted QP)
    2. Power-sum L-BFGS (p=2,4,8,16,32)
    3. Soft-max L-BFGS (multiple temperatures)
    4. Active-set peak equalization
    5. Coordinate descent with golden-section line search
    6. Simulated annealing
    7. Length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if parent is None:
        n_init = 768
        parent = np.ones(n_init) / n_init
        best_seq = parent.copy()
        best_score = compute_score(parent)
        print(f"Initialized: n={n_init}, score={best_score:.8f}")
    
    n_parent = len(parent)
    
    # === Phase 1: Power-sum L-BFGS with multiple p values ===
    print(f"\n=== Phase 1: Power-sum L-BFGS (best={best_score:.8f}) ===")
    for p_val in [2, 4, 8, 16, 32]:
        rem = deadline - time.monotonic() - 800
        if rem < 30:
            break
        t_alloc = min(rem * 0.2, 120.0)
        if t_alloc < 10:
            break
        print(f"  PS-LBFGS p={p_val}, n={n_parent}, budget={t_alloc:.0f}s")
        res = power_sum_lbgfs(best_seq, t_alloc, p=p_val, maxiter=3000)
        best_seq, best_score = check_and_update(res['a'], f"ps_p{p_val}", best_seq, best_score)
        print(f"    score={best_score:.8f}")
    
    # === Phase 2: Fictitious play ===
    print(f"\n=== Phase 2: Fictitious Play (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 500
    if rem > 30:
        t_alloc = min(rem * 0.35, 250.0)
        print(f"  FP n={n_parent}, budget={t_alloc:.0f}s")
        a_fp, sc_fp = fictitious_play_optimize(best_seq, t_alloc, n_inner_iters=20, verbose=True)
        best_seq, best_score = check_and_update(a_fp, "fict_play", best_seq, best_score)
        print(f"    score={best_score:.8f}")
    
    # === Phase 3: Soft-max L-BFGS (multiple temperatures) ===
    print(f"\n=== Phase 3: Soft-max L-BFGS (best={best_score:.8f}) ===")
    for T_val in [200, 500, 2000, 5000]:
        rem = deadline - time.monotonic() - 400
        if rem < 20:
            break
        t_alloc = min(rem * 0.25, 100.0)
        if t_alloc < 10:
            break
        print(f"  SM-LBFGS T={T_val}, budget={t_alloc:.0f}s")
        res = softmax_smoothed_lbgfs(best_seq, t_alloc, T=T_val, maxiter=3000)
        best_seq, best_score = check_and_update(res['a'], f"sm_T{T_val}", best_seq, best_score)
        print(f"    score={best_score:.8f}")
    
    # === Phase 4: Active-set peak equalization ===
    print(f"\n=== Phase 4: Peak Equalization (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 300
    if rem > 20:
        t_alloc = min(rem * 0.4, 200.0)
        print(f"  PE n={n_parent}, budget={t_alloc:.0f}s")
        a_pe, sc_pe = active_set_equalization(best_seq, t_alloc, n_peaks=15, verbose=True)
        best_seq, best_score = check_and_update(a_pe, "peak_eq", best_seq, best_score)
        print(f"    score={best_score:.8f}")
    
    # === Phase 5: Coordinate descent with line search ===
    print(f"\n=== Phase 5: CD Line Search (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 200
    if rem > 20:
        t_alloc = min(rem * 0.5, 150.0)
        print(f"  CD-LS n={n_parent}, budget={t_alloc:.0f}s")
        a_cd, sc_cd = coordinate_descent_line_search(best_seq, t_alloc, verbose=True)
        best_seq, best_score = check_and_update(a_cd, "cd_ls", best_seq, best_score)
        print(f"    score={best_score:.8f}")
    
    # === Phase 6: Another round of fictitious play (refined) ===
    print(f"\n=== Phase 6: FP Refinement (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 150
    if rem > 20:
        t_alloc = min(rem * 0.5, 100.0)
        print(f"  FP2 n={n_parent}, budget={t_alloc:.0f}s")

        print(f"    score={best_score:.8f}")
    
    # === Phase 7: Simulated annealing ===
    print(f"\n=== Phase 7: Simulated Annealing (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 100
    if rem > 20:
        t_alloc = min(rem * 0.5, 80.0)
        print(f"  SA n={n_parent}, budget={t_alloc:.0f}s")
        a_sa, sc_sa = simulate_anneal(best_seq, t_alloc, n_moves=60, t_start=0.003, t_end=0.00005)
        best_seq, best_score = check_and_update(a_sa, "sim_anneal", best_seq, best_score)
        print(f"    score={best_score:.8f}")

                   n_parent + 8, n_parent - 8]
    for alt_n in alt_lengths:
        if alt_n < 64 or alt_n > 4096:
            continue
        rem = deadline - time.monotonic() - 60
        if rem < 20:
            break
        t_alloc = min(rem * 0.25, 60.0)
        if t_alloc < 10:
            break
        init_alt = interpolate_sequence(best_seq, alt_n)
        print(f"  Length n={alt_n}, budget={t_alloc:.0f}s")
        # Quick power-sum optimization
        res = power_sum_lbgfs(init_alt, t_alloc, p=8, maxiter=2000)
        best_seq, best_score = check_and_update(res['a'], f"len_{alt_n}", best_seq, best_score)
        # Also try CD on the new length
        rem2 = deadline - time.monotonic() - 50
        if rem2 > 15:
            t2 = min(rem2 * 0.3, 30.0)
            a_cd_alt, _ = coordinate_descent_line_search(res['a'], t2)
            best_seq, best_score = check_and_update(a_cd_alt, f"cd_len_{alt_n}", best_seq, best_score)
    
    # === Phase 9: Final polish ===
    print(f"\n=== Phase 9: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 30
    if rem > 15:
        t_alloc = min(rem * 0.4, 50.0)
        a_pe2, _ = active_set_equalization(best_seq, t_alloc, n_peaks=20)
        best_seq, best_score = check_and_update(a_pe2, "final_pe", best_seq, best_score)
    
    rem = deadline - time.monotonic() - 20
    if rem > 10:
        t_alloc = min(rem * 0.5, 40.0)
        res = power_sum_lbgfs(best_seq, t_alloc, p=16, maxiter=3000)
        best_seq, best_score = check_and_update(res['a'], "final_ps16", best_seq, best_score)
    
    rem = deadline - time.monotonic() - 15
    if rem > 8:
        t_alloc = min(rem * 0.6, 30.0)
        a_cd_final, _ = coordinate_descent_line_search(best_seq, t_alloc)
        best_seq, best_score = check_and_update(a_cd_final, "final_cd", best_seq, best_score)
    
    # Fallback
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
