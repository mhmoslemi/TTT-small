```python
def fast_convolve(a):

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.max(np.nonzero(cond)[0]) if cond.any() else n - 1
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    for j in range(n):
        T[j:j+n, j] = a_fixed
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.GLPK, cp.SCS, cp.PDLP, cp.GUROBI, cp.MOSEK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    return b_val / s, float(M_var.value)
```

Now for the smooth max optimization:

```python
def smooth_max_grad(x, tau, n):
    """Compute smooth max objective and its gradient w.r.t. x (log-space parameterization).
    a = exp(x) / sum(exp(x)) (softmax)
    Objective: 2*n * smooth_max(conv(a,a), tau)
    """
    # Softmax
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    
    # Convolution
    conv = fast_convolve(a)
    
    # Smooth max: (1/tau) * log(sum(exp(tau * conv)))
    max_c = conv.max()
    exp_c = np.exp(tau * (conv - max_c))
    log_sum = np.log(exp_c.sum())
    obj = 2.0 * n * (max_c + log_sum / tau)
    
    # Probabilities
    p = exp_c / exp_c.sum()
    
    # Gradient w.r.t. a: grad_a[i] = 4*n * sum_m a[m] * p[m+i]
    a_flipped = a[::-1]
    size = 1 << (3 * n - 2).bit_length()
    A = rfft(a_flipped, size)
    P = rfft(p, size)
    conv_full = irfft(A * P, size)
    grad_a = 4.0 * n * conv_full[n-1 : 2*n-1]
    
    # Gradient w.r.t. x: grad_x[i] = a[i] * (grad_a[i] - dot(a, grad_a))
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)
    
    return obj, grad_x
```

Wait, I realize there might be an issue. The softmax parameterization means a[i] > 0 for all i, which prevents zeros. For this problem, I think that's OK (the optimal should have all positive entries for n ≥ 2).

But there's another issue: the L-BFGS optimizer might struggle with the softmax parameterization for large n because the condition number of the softmax Jacobian can be poor. Let me instead use the simplex-projected approach:

```python
def smooth_max_projected_grad(a, tau, n):
    """Compute smooth max and gradient, with simplex projection."""
    conv = fast_convolve(a)
    max_c = conv.max()
    exp_c = np.exp(tau * (conv - max_c))
    log_sum = np.log(exp_c.sum())
    obj = 2.0 * n * (max_c + log_sum / tau)
    
    p = exp_c / exp_c.sum()
    
    # Gradient w.r.t. a
    a_flipped = a[::-1]
    size = 1 << (3 * n - 2).bit_length()
    A = rfft(a_flipped, size)
    P = rfft(p, size)
    conv_full = irfft(A * P, size)
    grad_a = 4.0 * n * conv_full[n-1 : 2*n-1]
    
    # Project gradient onto the tangent space of the simplex
    # The tangent space is {v : sum(v) = 0}
    grad_projected = grad_a - np.mean(grad_a)
    
    return obj, grad_projected
```

Then the update is:

a_new = simplex_project(a_new)

```python
    t0 = time.monotonic()

    tau = 50.0

        obj, grad = smooth_max_projected_grad(a, tau, n)

        lr_try = lr
        for _ in range(20):
            a_new = simplex_project(a - lr_try * grad)
            new_score = 2.0 * n * fast_convolve(a_new).max()
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
            if new_score < obj - 0.5 * lr_try * np.dot(grad, a_new - a):
                a = a_new
                lr = min(lr_try * 2, 0.1)

            lr_try *= 0.5

            a = simplex_project(a - lr_try * grad)
            new_score = 2.0 * n * fast_convolve(a_new).max()
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()

        tau = max(tau * 0.98, tau_min)
    
    return best_a, best_score
```
