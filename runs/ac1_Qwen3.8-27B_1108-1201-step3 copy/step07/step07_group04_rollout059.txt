import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver, preferring PDLP for speed."""
    if '_lp_solver' in globals():
        return _lp_solver
    for solver_name in [cp.PDLP, cp.SCS, cp.ECOS, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.SCS
    return cp.SCS


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b (non-negative, sum=1) minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.SCS, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def lbgfs_adaptive(a0, time_limit, T_start=50, T_end=5, maxiter=10000):
    """L-BFGS with adaptive temperature (geometric decrease)."""

    z0 = z0 - z0.mean()
    t0 = monotonic()

    temps = []
    T = T_start
    while T >= T_end:
        temps.append(T)
        T *= 0.7
    
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()
    z = z0.copy()
    
    for T in temps:
        remaining = time_limit - (monotonic() - t0)
        if remaining < 5:
            break
        
        def func(z):
            z_shifted = z - z.max()
            exp_z = np.exp(z_shifted)
            a = exp_z / exp_z.sum()
            b = fast_convolve(a)
            max_b = b.max()
            scaled_b = (b - max_b) / T
            exp_b = np.exp(scaled_b)
            lse = max_b + T * np.log(exp_b.sum())
            return 2.0 * n * lse / T
        
        def jac(z):
            z_shifted = z - z.max()
            exp_z = np.exp(z_shifted)
            a = exp_z / exp_z.sum()
            b = fast_convolve(a)
            max_b = b.max()
            scaled_b = (b - max_b) / T
            exp_b = np.exp(scaled_b)
            w = exp_b / exp_b.sum()
            
            grad_a = np.zeros(n)
            for k in range(len(w)):
                if w[k] < 1e-10:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            
            grad_a *= 2.0 * n
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - a * S)
            return grad_z
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': 2000, 'ftol': 1e-15, 'gtol': 1e-10,
                         'maxfun': 10000, 'disp': False}
            )
            z = res.x
            z_shifted = z - z.max()
            a_out = np.exp(z_shifted)
            a_out = a_out / a_out.sum()
            score = 2.0 * n * fast_convolve(a_out).max()
            if score < best_score:
                best_score = score
                best_a = a_out.copy()
        except Exception:
            pass
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def fast_convolve(a):

    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver = None


def get_lp_solver():
    """Find a working LP solver, preferring PDLP for speed."""
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.PDLP, cp.SCS, cp.ECOS, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.SCS
    return cp.SCS


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b (non-negative, sum=1) minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.SCS, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def fictitious_play(n, time_limit, initial=None, solver=None, verbose=True, perturb_interval=80):
    """Run fictitious play for auto-correlation minimization with periodic perturbations."""
    if solver is None:
        solver = get_lp_solver()
    t0 = monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

    b = a.copy()
    a_avg = a.copy().astype(np.float64)
    b_avg = b.copy().astype(np.float64)
    best_score = float('inf')
    best_a = a.copy()
    iters = 0
    lbfgs_interval = 150
    last_lbfgs = 0

    while monotonic() - t0 < time_limit:
        iters += 1
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new
        a_new, M_a = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new
        a_avg = (iters * a_avg + a) / (iters + 1)
        b_avg = (iters * b_avg + b) / (iters + 1)

        candidates = [a, b, 0.5 * (a + b), a_avg, 0.5 * (a_avg + b_avg),
                      0.7 * a_avg + 0.3 * a, 0.3 * a_avg + 0.7 * a]
        for cand in candidates:
            cand = np.maximum(cand, 0)
            s = cand.sum()
            if s < 1e-10:
                continue
            cand_norm = cand / s
            max_conv = fast_convolve(cand_norm).max()
            score = 2.0 * n * max_conv
            if score < best_score - 1e-12:
                best_score = score
                best_a = cand_norm.copy()

        # Periodic perturbation to escape local structure
        if iters % perturb_interval == 0:
            noise_scale = 0.002 * max(1.0, 1.0 - iters / 5000.0)
            a = a + np.random.standard_normal(n) * noise_scale
            a = np.maximum(a, 0)
            a = a / a.sum()
            b = b + np.random.standard_normal(n) * noise_scale
            b = np.maximum(b, 0)
            b = b / b.sum()
            # Also perturb the averages slightly
            a_avg = a_avg + np.random.standard_normal(n) * noise_scale * 0.5
            a_avg = np.maximum(a_avg, 0)
            a_avg = a_avg / a_avg.sum()

        # Periodic quick L-BFGS polish
        if iters - last_lbfgs >= lbfgs_interval and monotonic() - t0 > 10:
            last_lbfgs = iters
            a_polished = quick_lbgfs(best_a, maxiter=200, T=20.0)
            score_p = 2.0 * n * fast_convolve(a_polished).max()
            if score_p < best_score - 1e-10:
                best_score = score_p
                best_a = a_polished.copy()
                # Use polished version to restart FP
                a = best_a + np.random.standard_normal(n) * 0.001
                a = np.maximum(a, 0)
                a = a / a.sum()
                b = best_a.copy()
                a_avg = best_a.copy()
                b_avg = best_a.copy()

        if verbose and iters % 40 == 0:
            elapsed = monotonic() - t0
            print(f"  FP n={n} iter={iters} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score, iters


def quick_lbgfs(a0, maxiter=500, T=20.0):
    """Quick L-BFGS polish with fixed temperature."""
    n = len(a0)
    a_safe = np.maximum(a0, 1e-30)
    z0 = np.log(a_safe)
    z0 = z0 - z0.mean()

    def func(z):
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()
        b = fast_convolve(a)
        max_b = b.max()
        scaled_b = (b - max_b) / T
        exp_b = np.exp(scaled_b)
        lse = max_b + T * np.log(exp_b.sum())
        return 2.0 * n * lse / T

    def jac(z):
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()
        b = fast_convolve(a)
        max_b = b.max()
        scaled_b = (b - max_b) / T
        exp_b = np.exp(scaled_b)
        w = exp_b / exp_b.sum()

        grad_a = np.zeros(n)
        for k in range(len(w)):
            if w[k] < 1e-10:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]

        grad_a *= 2.0 * n
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - a * S)
        return grad_z

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10, 'disp': False}
        )
        z_final = res.x
        z_shifted = z_final - z_final.max()
        a_out = np.exp(z_shifted)
        a_out = a_out / a_out.sum()
        return a_out
    except Exception:
        return a0.copy()


def lbgfs_adaptive(a0, time_limit, T_start=60, T_end=2.0, decay=0.65):
    """L-BFGS with geometric temperature decrease schedule."""
    n = len(a0)
    a = a0.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()

    temps = []
    T = T_start
    while T >= T_end:
        temps.append(T)
        T *= decay

    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()

    for i, T in enumerate(temps):
        remaining = time_limit - (monotonic() - t0)
        if remaining < 5:
            break
        maxiter = min(3000, int(remaining * 100))
        if maxiter < 50:
            maxiter = 50

        def func(z, T=T):
            z_shifted = z - z.max()
            exp_z = np.exp(z_shifted)
            a_local = exp_z / exp_z.sum()
            b = fast_convolve(a_local)
            max_b = b.max()
            scaled_b = (b - max_b) / T
            exp_b = np.exp(scaled_b)
            lse = max_b + T * np.log(exp_b.sum())
            return 2.0 * n * lse / T

        def jac(z, T=T):
            z_shifted = z - z.max()
            exp_z = np.exp(z_shifted)
            a_local = exp_z / exp_z.sum()
            b = fast_convolve(a_local)
            max_b = b.max()
            scaled_b = (b - max_b) / T
            exp_b = np.exp(scaled_b)
            w = exp_b / exp_b.sum()

            grad_a = np.zeros(n)
            for k in range(len(w)):
                if w[k] < 1e-10:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_local[k - idx]

            grad_a *= 2.0 * n
            S = np.dot(grad_a, a_local)
            grad_z = a_local * (grad_a - a_local * S)
            return grad_z

        try:
            res = minimize(
                fun=func, x0=z, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12, 'disp': False}
            )
            z = res.x
            z_shifted = z - z.max()
            a_out = np.exp(z_shifted)
            a_out = a_out / a_out.sum()
            score = 2.0 * n * fast_convolve(a_out).max()
            if score < best_score - 1e-14:
                best_score = score
                best_a = a_out.copy()
                a = a_out
            print(f"    L-BFGS T={T:.1f}: score={best_score:.6f}")
        except Exception:
            pass

    return best_a, best_score


def peak_redistribute(a0, time_limit, max_iters=10000):
    """Peak redistribution: transfer mass to reduce convex peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    stuck = 0

    for it in range(max_iters):
        if monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        # Get top few peaks (not just the max)
        top_k = min(5, len(b))
        peak_indices = np.argpartition(b, -top_k)[-top_k:]
        peak_indices = peak_indices[np.argsort(b[peak_indices])[::-1]]

        improved = False
        for peak_k in peak_indices[:3]:
            if improved:
                break
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]

            if contribs.sum() < 1e-20:
                continue

            order = np.argsort(contribs)[::-1]

            for ci in order[:4]:
                if improved:
                    break
                i = int(idx[ci])
                j = int(peak_k - i)
                if a[i] < 1e-12:
                    continue

                # Try moving mass from i to various targets
                fracs = [0.15, 0.1, 0.05, 0.03]
                delta = a[i] * 0.1  # will be adjusted

                for frac in fracs:
                    delta = a[i] * frac
                    if delta < 1e-15:
                        continue

                    # Candidate targets: positions far from the peak
                    target_candidates = []
                    if i > 0:
                        target_candidates.append(0)
                    if i < n - 1:
                        target_candidates.append(n - 1)
                    # Position that minimizes contribution to current peak
                    if j != i:
                        target_candidates.append(j)
                    # Random positions
                    for _ in range(3):
                        target_candidates.append(np.random.randint(n))

                    for target in sorted(set(target_candidates)):
                        if target == i:
                            continue
                        a_trial = a.copy()
                        a_trial[i] -= delta
                        a_trial[target] += delta
                        a_trial = np.maximum(a_trial, 0)
                        ss = a_trial.sum()
                        if ss < 1e-15:
                            continue
                        a_trial = a_trial / ss
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-13:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break

                    if improved:
                        break

                # Also try reducing both i and j
                if not improved and i != j and a[j] > 1e-12:
                    for f in [0.92, 0.88, 0.85]:
                        a_trial = a.copy()
                        a_trial[i] *= f
                        a_trial[j] *= f
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-13:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break

        # Random perturbation if stuck
        if not improved:
            for _ in range(15):
                a_trial = a + np.random.standard_normal(n) * 0.0004
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

        if improved:
            stuck = 0
        else:
            stuck += 1
            if stuck > 200:
                break

    return best_a, best_score


def subgradient_accelerated(a0, time_limit, lr_init=0.004, max_iters=30000):
    """Accelerated subgradient method with adaptive restarts."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    y = a.copy()
    t_val = 1.0
    t0 = monotonic()
    no_improve = 0
    lr = lr_init

    for it in range(max_iters):
        if monotonic() - t0 > time_limit:
            break

        # Compute subgradient at y
        b_y = fast_convolve(y)
        # Use top-k peaks for a smoother subgradient
        k_sel = min(10, len(b_y))
        peaks = np.argpartition(b_y, -k_sel)[-k_sel:]
        pv = b_y[peaks]
        temp_sg = 0.005 * max(pv.max(), 1e-10)
        scaled = (pv - pv.max()) / temp_sg
        w = np.exp(scaled)
        w = w / w.sum()

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            for j in range(n):
                k_minus_j = k - j
                if 0 <= k_minus_j < n:
                    sg[j] += 2.0 * w[i] * y[k_minus_j]

        sg -= np.mean(sg)

        # Accelerated step
        lr = lr_init / (1.0 + 0.02 * it)
        a_new = y - lr * sg

        # Project to simplex
        u = np.sort(a_new)[::-1]
        cssv = np.cumsum(u)
        cond = u * np.arange(1, n + 1) > (cssv - 1)
        if cond.any():
            rho = np.nonzero(cond)[0][-1]
            theta = (cssv[rho] - 1.0) / (rho + 1.0)
            a_new = np.maximum(a_new - theta, 0.0)
        else:
            a_new = np.ones(n) / n
        a_new = a_new / a_new.sum()

        # Nesterov update
        t_new = 0.5 * (1.0 + math.sqrt(1.0 + 4.0 * t_val * t_val))
        y = a_new + ((t_val - 1.0) / t_new) * (a_new - y)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy
        else:
            y = a_new.copy()
        t_val = t_new

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300:
            y = best_a + np.random.standard_normal(n) * 0.002
            y = np.maximum(y, 0)
            y = y / y.sum()
            t_val = 1.0
            no_improve = 0

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate and return updated best."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for auto-correlation minimization.
    Combines LP fictitious play, adaptive L-BFGS, peak redistribution,
    and accelerated subgradient with multi-length exploration.
    """
    np.random.seed(seed)
    deadline = monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = p_score
    else:
        best_seq = np.ones(768) / 768.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Initial: n=768, score={best_score:.6f}")

    n_parent = len(best_seq)

    # === Phase 1: LP Fictitious Play (extended) ===
    print(f"\n=== Phase 1: LP Fictitious Play ===")
    rem = deadline - monotonic()
    t_fp = min(rem * 0.30, 480.0)
    print(f"  FP n={n_parent}, budget={t_fp:.0f}s")
    a_fp, score_fp, iters = fictitious_play(n_parent, t_fp, initial=best_seq, solver=solver, verbose=True)
    best_seq, best_score = check_and_update(a_fp, f"FP_n{n_parent}_it{iters}", best_seq, best_score)

    # === Phase 2: Damped LP refinement ===
    print(f"\n=== Phase 2: Damped LP (best={best_score:.6f}) ===")
    rem = deadline - monotonic()
    if rem > 40:
        t_dl = min(rem * 0.20, 300.0)
        print(f"  DampedLP n={len(best_seq)}, budget={t_dl:.0f}s")
        a = best_seq.copy()
        a = a / a.sum()
        best_a_local = a.copy()
        best_score_local = 2.0 * len(a) * fast_convolve(a).max()
        it = 0
        t0_dl = monotonic()
        while monotonic() - t0_dl < t_dl:
            it += 1
            b_new, M_b = solve_lp_best_response(a, solver)
            if b_new is None:
                continue
            alpha = 1.0 / (1.0 + 0.1 * it)
            a_new = (1.0 - alpha) * a + alpha * b_new
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                break
            a_new = a_new / s
            score = 2.0 * len(a) * fast_convolve(a_new).max()
            if score < best_score_local - 1e-14:
                best_score_local = score
                best_a_local = a_new.copy()
            a = a_new
            if it % 100 == 0:
                print(f"    DampedLP iter={it}: score={best_score_local:.6f}")
        best_seq, best_score = check_and_update(best_a_local, "damped_lp", best_seq, best_score)

    # === Phase 3: L-BFGS with adaptive temperature ===
    print(f"\n=== Phase 3: L-BFGS Adaptive (best={best_score:.6f}) ===")
    rem = deadline - monotonic()
    if rem > 30:
        t_lb = min(rem * 0.22, 360.0)
        print(f"  L-BFGS n={len(best_seq)}, budget={t_lb:.0f}s")
        a_lb, score_lb = lbgfs_adaptive(best_seq, t_lb, T_start=80, T_end=1.5, decay=0.6)
        best_seq, best_score = check_and_update(a_lb, "lbgfs_adaptive", best_seq, best_score)

    # === Phase 4: Peak Redistribution ===
    print(f"\n=== Phase 4: Peak Redistribution (best={best_score:.6f}) ===")
    rem = deadline - monotonic()
    if rem > 20:
        t_pr = min(rem * 0.15, 180.0)
        print(f"  PeakRedist n={len(best_seq)}, budget={t_pr:.0f}s")
        a_pr, score_pr = peak_redistribute(best_seq, t_pr)
        best_seq, best_score = check_and_update(a_pr, "peak_redist", best_seq, best_score)

    # === Phase 5: Accelerated Subgradient ===
    print(f"\n=== Phase 5: Accel Subgradient (best={best_score:.6f}) ===")
    rem = deadline - monotonic()
    if rem > 20:
        t_sg = min(rem * 0.15, 180.0)
        print(f"  SubGrad n={len(best_seq)}, budget={t_sg:.0f}s")
        a_sg, score_sg = subgradient_accelerated(best_seq, t_sg, lr_init=0.003)
        best_seq, best_score = check_and_update(a_sg, "accel_subgrad", best_seq, best_score)

    # === Phase 6: Second L-BFGS pass with low temperature ===
    print(f"\n=== Phase 6: Low-T L-BFGS (best={best_score:.6f}) ===")
    rem = deadline - monotonic()
    if rem > 20:
        t_lb2 = min(rem * 0.15, 180.0)
        print(f"  L-BFGS low-T n={len(best_seq)}, budget={t_lb2:.0f}s")
        a_lb2, score_lb2 = lbgfs_adaptive(best_seq, t_lb2, T_start=15, T_end=1.0, decay=0.5)
        best_seq, best_score = check_and_update(a_lb2, "lbgfs_lowT", best_seq, best_score)

    # === Phase 7: Alternative lengths ===
    print(f"\n=== Phase 7: Length Exploration (best={best_score:.6f}) ===")
    rem = deadline - monotonic()
    if rem > 80:
        n_best = len(best_seq)
        alt_lengths = []
        for factor in [1.3, 1.5, 0.75, 1.7]:
            alt_n = int(n_best * factor)
            if 256 <= alt_n <= 4096 and alt_n != n_best and alt_n not in alt_lengths:
                alt_lengths.append(alt_n)

        for alt_n in alt_lengths[:2]:
            rem = deadline - monotonic()
            if rem < 80:
                break
            t_alt = min(rem * 0.45, 250.0)
            print(f"  Trying n={alt_n}, budget={t_alt:.0f}s")
            init_alt = interpolate_sequence(best_seq, alt_n)
            a_alt, _, _ = fictitious_play(alt_n, t_alt * 0.5, initial=init_alt, solver=solver, verbose=False)
            best_seq, best_score = check_and_update(a_alt, f"alt_fp_n{alt_n}", best_seq, best_score)
            rem2 = deadline - monotonic()
            if rem2 > 30:
                a_alt2, _ = lbgfs_adaptive(a_alt, min(rem2 * 0.3, 80.0), T_start=40, T_end=3, decay=0.6)
                best_seq, best_score = check_and_update(a_alt2, f"alt_lb_n{alt_n}", best_seq, best_score)

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - monotonic()
    if rem > 15:
        t_f1 = min(rem * 0.35, 80.0)
        a_f1, _ = lbgfs_adaptive(best_seq, t_f1, T_start=10, T_end=0.8, decay=0.5)
        best_seq, best_score = check_and_update(a_f1, "final_lb", best_seq, best_score)

    rem = deadline - monotonic()
    if rem > 10:
        t_f2 = min(rem * 0.4, 50.0)
        a_f2, _ = peak_redistribute(best_seq, t_f2, max_iters=5000)
        best_seq, best_score = check_and_update(a_f2, "final_peak", best_seq, best_score)

    rem = deadline - monotonic()
    if rem > 10:
        t_f3 = min(rem * 0.5, 30.0)
        a_f3, _ = peak_redistribute(best_seq, t_f3, max_iters=3000)
        best_seq, best_score = check_and_update(a_f3, "final_peak2", best_seq, best_score)

    # Final verification
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
