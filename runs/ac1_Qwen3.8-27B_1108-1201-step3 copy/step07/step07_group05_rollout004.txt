import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def softmax_from_z(z):
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def z_from_a(a):
    return np.log(np.maximum(a, 1e-30))

def smoothed_objective(z, T, n):
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    log_sum_exp = max_shifted + np.log(np.exp(shifted - max_shifted).sum())
    return (2.0 * n / T) * np.exp(T * log_sum_exp)

def smoothed_gradient(z, T, n):
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    w = np.exp(shifted - max_shifted)
    w = w / w.sum()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return (2.0 * n / T) * grad_z

import numpy as np

from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def softmax_from_z(z):
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-30))


def smoothed_objective(z, T, n):
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    log_sum_exp = max_shifted + np.log(np.exp(shifted - max_shifted).sum())
    return (2.0 * n / T) * np.exp(T * log_sum_exp)


def smoothed_gradient(z, T, n):
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    w = np.exp(shifted - max_shifted)
    w = w / w.sum()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return (2.0 * n / T) * grad_z


def lbgfs_optimize(a0, time_limit, T, maxiter=5000):

    z0 = z_from_a(a0)

            result['score'] = score

    return result

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        best_seq, best_score = check_and_update(a3, "pairwise", best_seq, best_score)

    # === Phase 4: Multiplicative Weights ===
    print(f"\n=== Phase 4: Multiplicative Weights (best={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(200.0, rem * 0.13)
    if rem > 30:
        a4, s4 = multiplicative_weights(best_seq, t4, n_active=10, verbose=True)
        best_seq, best_score = check_and_update(a4, "mult_weight", best_seq, best_score)

    # === Phase 5: Second round of L-BFGS + Subgradient ===
    print(f"\n=== Phase 5: Refinement Round (best={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(250.0, rem * 0.15)
    if rem > 30:
        a5, s5 = adaptive_lbfgs(best_seq, t5 * 0.5, n_parent)
        best_seq, best_score = check_and_update(a5, "lbfgs2", best_seq, best_score)
        rem = deadline - time.monotonic()
        if rem > 30:
            a5b, s5b = subgradient_descent(best_seq, min(t5 * 0.5, rem * 0.4), n_active=2)
            best_seq, best_score = check_and_update(a5b, "subgrad2", best_seq, best_score)

    # === Phase 6: Multi-length trials ===
    print(f"\n=== Phase 6: Multi-Length Trials (best={best_score:.10f}) ===")
    nearby_ns = [n_parent - 2, n_parent - 1, n_parent + 1, n_parent + 2]
    for alt_n in nearby_ns:
        if alt_n < 128 or alt_n > 2048:
            continue
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t_alloc = min(rem * 0.12, 80.0)
        init_alt = interpolate_sequence(best_seq, alt_n)
        print(f"  Trying n={alt_n}, budget={t_alloc:.0f}s")
        # Quick L-BFGS then pairwise
        res = lbgfs_optimize(init_alt, t_alloc * 0.5, T=500, maxiter=2000)
        a_alt = res['a']
        a_alt, s_alt = focused_pairwise_descent(a_alt, t_alloc * 0.5, n_top_pairs=15)
        best_seq, best_score = check_and_update(a_alt, f"n{alt_n}", best_seq, best_score)
        print(f"    n={alt_n} result: {s_alt:.10f}")

    # === Phase 7: Final Polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.10f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t7 = min(300.0, rem * 0.35)
        a7, s7 = focused_pairwise_descent(best_seq, t7 * 0.5, n_top_pairs=30, verbose=True)
        best_seq, best_score = check_and_update(a7, "polish_pw", best_seq, best_score)
        rem = deadline - time.monotonic()
        if rem > 20:
            a7b, s7b = subgradient_descent(best_seq, min(t7 * 0.3, rem * 0.3), n_active=2)
            best_seq, best_score = check_and_update(a7b, "polish_sg", best_seq, best_score)

    # Final L-BFGS with low temperature
    rem = deadline - time.monotonic()
    if rem > 15:
        t_final = min(rem * 0.6, 60.0)
        res = lbgfs_optimize(best_seq, t_final, T=5, maxiter=3000)
        best_seq, best_score = check_and_update(res['a'], "final_lbfgs", best_seq, best_score)

    # Fallback
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
