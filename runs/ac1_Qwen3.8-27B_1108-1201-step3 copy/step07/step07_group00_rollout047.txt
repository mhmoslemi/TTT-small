"""
LP-based optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses MOSEK solver for fast LPs, adaptive peak selection,
bisection line search, and multi-phase optimization strategy.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_constraint_matrix(f, top_idx):
    """Build dense constraint matrix for convolution constraints."""
    n = len(f)
    K = len(top_idx)
    A = np.zeros((K, n))
    for row_i in range(K):
        k = top_idx[row_i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]
    return A


def lp_solve(f, top_k, solver='MOSEK'):
    """
    Solve: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for top_k peaks.
    Returns (g, M) or (None, M).
    """
    n = len(f)
    b = fast_convolve(f)
    M = b.max()
    if M < 1e-15:
        return None, M

    total_peaks = len(b)
    if top_k < total_peaks:
        top_idx = np.argpartition(b, -top_k)[-top_k:]
    else:
        top_idx = np.arange(total_peaks)
    top_idx = np.sort(top_idx)
    K = len(top_idx)

    A = build_constraint_matrix(f, top_idx)
    
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    
    try:
        if solver == 'MOSEK':
            prob.solve(solver=cp.MOSEK)
        elif solver == 'SCS':
            prob.solve(solver=cp.SCS, max_iters=100000, eps=1e-7)
        else:
            prob.solve(solver=cp.ECOS)
    except Exception as e:
        # Fallback to SCS
        try:
            prob.solve(solver=cp.SCS, max_iters=100000, eps=1e-7)
        except Exception:
            return None, M

    if g.value is None:
        return None, M
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
        return None, M
    return g_val, M


def bisection_line_search(f, h, lo=0.0, hi=1.0, max_iter=20):
    """Bisection to find smallest t in [lo, hi] such that obj(f + t*(h-f)) is minimized."""
    n = len(f)
    f_curr = project_simplex(f)
    cur_obj = fast_convolve(f_curr).max()
    
    best_t = lo
    best_obj = cur_obj
    
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        f_trial = (1.0 - mid) * f_curr + mid * h
        f_trial = project_simplex(f_trial)
        obj = fast_convolve(f_trial).max()
        
        if obj < best_obj:
            best_obj = obj
            best_t = mid
        
        # We want to find where the objective starts increasing
        f_at_lo = project_simplex((1.0 - lo) * f_curr + lo * h)
        obj_lo = fast_convolve(f_at_lo).max()
        
        if obj > obj_lo:
            hi = mid
        else:
            lo = mid
    
    return best_t, best_obj

def subgradient_polish(f_init, time_limit, n_top=100, label="sg"):
    """Subgradient with soft peak weighting."""
    t0 = time.monotonic()
    n = len(f_init)
    f = project_simplex(f_init.copy())
    v = np.zeros(n)
    lr = 0.003
    momentum = 0.9
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(999)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()
        
        # Soft weighting: w_k = max(0, b_k - 0.8*M)^2
        threshold = 0.8 * M
        w = np.maximum(0, b - threshold) ** 2
        w_sum = w.sum()
        if w_sum > 1e-30:
            w /= w_sum
        else:
            w = np.zeros(len(b))
            w[b.argmax()] = 1.0
        
        # Compute subgradient
        g = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-12:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            g[lo:hi+1] += w[k] * f[k - (lo:hi+1)]
        
        v = momentum * v - lr * g
        f_new = project_simplex(f + v)
        f = f_new
        
        if it % 50 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
                lr = min(lr * 1.02, 0.01)
            else:
                no_improve += 1
                if no_improve == 3:
                    lr *= 0.9
                    no_improve = 0
        
        if it % 200 == 0:
            print(f"    {label} iter={it} obj={best_obj:.10f} "
                  f"score={2.0*n*best_obj:.8f} lr={lr:.5f} t={time.monotonic()-t0:.0f}s")
        
        if no_improve > 30:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            no_improve = 0
    
    return best_f, best_obj


def coordinate_polish(f_init, time_limit, label="cd"):
    """Coordinate descent on individual elements."""
    t0 = time.monotonic()
    n = len(f_init)
    f = project_simplex(f_init.copy())
    best_obj = fast_convolve(f).max()
    multipliers = [0.7, 0.85, 1.15, 1.3]
    
    passes = 0
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-14:
                continue
            for mult in multipliers:
                f[j] *= mult
                f_trial = project_simplex(f)
                obj = fast_convolve(f_trial).max()
                if obj < best_obj - 1e-13:
                    best_obj = obj
                    improved = True
                else:
                    f[j] /= mult
                    # Try the other direction

                    f_trial = project_simplex(f)
                    obj = fast_convolve(f_trial).max()
                    if obj < best_obj - 1e-13:
                        best_obj = obj
                        improved = True
                    else:
                        f[j] *= mult  # restore
            if not improved and j % 500 == 0:
                # periodic check
                pass
        
        if not improved:
            break
    
    return f, best_obj

"""
Multi-phase LP optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses MOSEK solver with adaptive full-constraint LP, bisection line search,
occasional perturbation for escape, and multi-length exploration.

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_constraint_matrix(f, top_idx):
    """Build constraint matrix A where A[k,j] = f component in conv constraint k."""
    n = len(f)
    K = len(top_idx)
    A = np.zeros((K, n))
    for row_i in range(K):
        k = int(top_idx[row_i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]
    return A


def lp_solve_full(f, solver_name):
    """Solve LP with ALL convolution constraints. Returns (g, M, solve_time)."""
    n = len(f)
    b = fast_convolve(f)
    M = float(b.max())
    if M < 1e-15:
        return None, M, 0.0
    total_peaks = len(b)
    top_idx = np.arange(total_peaks)
    A = build_constraint_matrix(f, top_idx)
    g = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [g >= 0, A @ g <= M])
    t_start = time.monotonic()
    try:
        if solver_name == 'MOSEK':
            prob.solve(solver=cp.MOSEK, msdk_params={"MSK_IPAR_INTPNT_COVER_CONSTRAINTS": 1})
        elif solver_name == 'SCS':
            prob.solve(solver=cp.SCS, max_iters=200000, eps=1e-8)
        else:
            prob.solve(solver=cp.MOSEK)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=200000, eps=1e-8)
        except Exception:
            return None, M, time.monotonic() - t_start
    solve_time = time.monotonic() - t_start
    if g.value is None:
        return None, M, solve_time
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
        return None, M, solve_time
    return g_val, M, solve_time

    g = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [g >= 0, A @ g <= M])
    t_start = time.monotonic()

            return None, M, time.monotonic() - t_start
    solve_time = time.monotonic() - t_start

        return None, M, solve_time

def line_search(f, h, n_evals=8):
    """
    Try multiple step sizes and return the best (t, obj).
    Uses a mix of small and large steps.
    """
    n = len(f)
    f_norm = project_simplex(f)
    cur_obj = fast_convolve(f_norm).max()
    best_t = 0.0
    best_obj = cur_obj
    candidates = [0.02, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7]
    for t in candidates:
        if t > 1.0:
            continue
        f_trial = project_simplex((1.0 - t) * f_norm + t * h)
        obj = fast_convolve(f_trial).max()
        if obj < best_obj:
            best_obj = obj
            best_t = t
    return best_t, best_obj

def subgradient_polish(f_init, time_limit, label="sg"):
    """Subgradient method with adaptive soft-threshold weighting on peaks."""
    t0 = time.monotonic()
    n = len(f_init)
    f = project_simplex(f_init.copy())
    v = np.zeros(n)
    lr = 0.004
    momentum = 0.9
    best_f = f.copy()
    best_obj = float(fast_convolve(f).max())
    no_improve = 0
    it = 0
    rng = np.random.default_rng(77777)
    threshold_frac = 0.7

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = float(b.max())
        if M < 1e-15:
            break

        # Soft peak weighting
        threshold = threshold_frac * M
        w = np.maximum(0.0, b - threshold)
        w = w ** 2
        w_sum = w.sum()
        if w_sum > 1e-30:
            w = w / w_sum
        else:
            w = np.zeros(len(b))
            w[int(np.argmax(b))] = 1.0

        # Compute subgradient: g_j = sum_k w_k * f_{k-j}
        g = np.zeros(n)
        # Only iterate over peaks with significant weight
        active = np.where(w > 1e-10)[0]
        for k in active:
            lo = max(0, int(k) - n + 1)
            hi = min(n - 1, int(k))
            g[lo:hi + 1] += w[int(k)] * f[int(k) - lo:int(k) - lo + (hi - lo + 1)]

        v = momentum * v - lr * g
        f = project_simplex(f + v)

        if it % 100 == 0:
            obj = float(fast_convolve(f).max())
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
                lr = min(lr * 1.03, 0.015)
            else:
                no_improve += 1
                if no_improve % 5 == 0:
                    lr *= 0.92

            if it % 500 == 0:
                print(f"    {label} it={it} obj={best_obj:.10f} "
                      f"score={2.0*n*best_obj:.8f} lr={lr:.5f} t={time.monotonic()-t0:.0f}s")

            if no_improve > 40:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
                v = np.zeros(n)
                no_improve = 0
                threshold_frac = max(0.3, threshold_frac - 0.05)

    return best_f, best_obj


def coordinate_polish(f_init, time_limit, label="cd"):
    """Cycle through coordinates, trying multiplicative adjustments."""
    t0 = time.monotonic()
    n = len(f_init)
    f = project_simplex(f_init.copy())
    best_obj = float(fast_convolve(f).max())
    multipliers = [0.75, 0.9, 1.1, 1.25]
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        any_improve = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 5e-15:
                continue
            for mult in multipliers:
                original = f[j]
                f[j] = original * mult
                f_safe = project_simplex(f)
                obj = float(fast_convolve(f_safe).max())
                if obj < best_obj - 1e-13:
                    f = f_safe
                    best_obj = obj
                    any_improve = True
                else:
                    f[j] = original
                    # Try opposite direction
                    f[j] = original / mult
                    f_safe = project_simplex(f)
                    obj = float(fast_convolve(f_safe).max())
                    if obj < best_obj - 1e-13:
                        f = f_safe
                        best_obj = obj
                        any_improve = True
                    else:
                        f[j] = original
        if not any_improve:
            break
        if passes % 3 == 0:
            print(f"    {label} pass={passes} obj={best_obj:.10f} "
                  f"score={2.0*n*best_obj:.8f} t={time.monotonic()-t0:.0f}s")

    return f, best_obj


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining full-constraint LP, adaptive top-K LP,
    subgradient polish, and coordinate descent.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(1024) / 1024
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default uniform n=1024, score={p_score:.8f}")

    best_seq = parent.copy()
    best_score = p_score

    # === Phase 1: Adaptive LP with full-constraint checks ===
    n_p = len(parent)
    print(f"\n=== Phase 1: Adaptive LP (n={n_p}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.45, 700.0)
    if t1 > 60:
        print(f"  Budget: {t1:.0f}s")
        f1, obj1 = run_adaptive_lp(parent, time_limit=t1, label="lp1")
        score1 = float(evaluate_sequence(f1.tolist()))
        if np.isfinite(score1) and score1 < best_score:
            best_score = score1
            best_seq = f1.copy()
            print(f"  NEW BEST [phase1]: {score1:.8f}")

    # === Phase 2: Continue LP with larger K ===
    print(f"\n=== Phase 2: LP deep (n={len(best_seq)}, score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.20, 300.0)
    if t2 > 40:
        print(f"  Budget: {t2:.0f}s")
        f2, obj2 = run_lp_phase(best_seq, time_limit=t2, top_k_init=200,
                                top_k_max=600, use_full=True, label="lp2")
        score2 = float(evaluate_sequence(f2.tolist()))
        if np.isfinite(score2) and score2 < best_score:
            best_score = score2
            best_seq = f2.copy()
            print(f"  NEW BEST [phase2]: {score2:.8f}")

    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-length (best={best_score:.8f}) ===")
    n_base = len(best_seq)
    target_lengths = set()
    for factor in [0.85, 1.15, 1.3, 1.5]:
        tl = int(n_base * factor)
        tl = max(512, min(3000, tl))
        if tl != n_base:
            target_lengths.add(tl)
    target_lengths = sorted(target_lengths, key=lambda x: abs(x - n_base))

    for target_n in target_lengths[:3]:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        t3 = min(rem * 0.12, 120.0)
        if t3 < 25:
            break
        old_x = np.linspace(0.0, 1.0, n_base)
        new_x = np.linspace(0.0, 1.0, target_n)
        f_interp = np.interp(new_x, old_x, best_seq)
        f_interp = np.maximum(f_interp, 0.0)
        s = f_interp.sum()
        if s > 1e-10:
            f_interp = f_interp / s
        else:
            f_interp = np.ones(target_n) / target_n

        print(f"  LP n={target_n}, budget={t3:.0f}s")
        f3, obj3 = run_lp_phase(f_interp, time_limit=t3, top_k_init=100,
                                top_k_max=400, label=f"n{target_n}")
        score3 = float(evaluate_sequence(f3.tolist()))
        if np.isfinite(score3) and score3 < best_score:
            best_score = score3
            best_seq = f3.copy()
            print(f"  NEW BEST [n{target_n}]: {score3:.8f}")

    # === Phase 4: Subgradient polish ===
    print(f"\n=== Phase 4: Subgradient (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.12, 120.0)
    if t4 > 20:
        print(f"  Budget: {t4:.0f}s, n={len(best_seq)}")
        f4, obj4 = subgradient_polish(best_seq, time_limit=t4)
        score4 = float(evaluate_sequence(f4.tolist()))
        if np.isfinite(score4) and score4 < best_score:
            best_score = score4
            best_seq = f4.copy()
            print(f"  NEW BEST [sg]: {score4:.8f}")

    # === Phase 5: Coordinate descent ===
    print(f"\n=== Phase 5: Coord descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.08, 60.0)
    if t5 > 10:
        print(f"  Budget: {t5:.0f}s")
        f5, obj5 = coordinate_polish(best_seq, time_limit=t5)
        score5 = float(evaluate_sequence(f5.tolist()))
        if np.isfinite(score5) and score5 < best_score:
            best_score = score5
            best_seq = f5.copy()
            print(f"  NEW BEST [cd]: {score5:.8f}")

    # === Phase 6: Final LP burst ===
    print(f"\n=== Phase 6: Final LP (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.20, 120.0)
    if t6 > 15:
        print(f"  Budget: {t6:.0f}s, n={len(best_seq)}")
        f6, obj6 = run_adaptive_lp(best_seq, time_limit=t6, label="lp_final")
        score6 = float(evaluate_sequence(f6.tolist()))
        if np.isfinite(score6) and score6 < best_score:
            best_score = score6
            best_seq = f6.copy()
            print(f"  NEW BEST [final]: {score6:.8f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
