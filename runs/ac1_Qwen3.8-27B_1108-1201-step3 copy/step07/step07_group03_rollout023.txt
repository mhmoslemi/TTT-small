**Adaptive coordinate descent:**
- At each step, identify the "peak positions" k* where `(a*a)[k*]` is largest.
- For each peak k*, the contributing coordinates are those `i` where `i ∈ I_{k*}`.
- Among these, the most impactful coordinates are those with large `a[i] * a[k*-i]`.
- Try to reduce these by:
  1. Reducing `a[i]` (or `a[k*-i]`)
  2. Compensating by increasing other coordinates (to maintain sum=1)

The compensation step is key: if we reduce `a[i]`, we need to add that mass somewhere. The best place to add it is where it has minimal impact on the peak.

This is essentially a **swap** operation: move mass from `a[i]` to `a[j]`.

**Swap-based local search:**
```
for each pair (i, j) where a[i] > 0:
    for each small epsilon:
        a'[i] = a[i] - epsilon
        a'[j] = a[j] + epsilon
        if score(a') < score(a):
            accept
```

This is O(n^2) per sweep, which for n=768 is ~600K pairs. Each evaluation is O(n log n) for the FFT. Total per sweep: ~600K * 768 * log(768) ≈ 600K * 768 * 10 ≈ 4.6 billion operations. That's too slow for a single sweep.

Instead, we can be smarter:
- Only try swaps involving the top contributing coordinates (top 20-50)
- Use a precomputed "impact matrix" to estimate the score change without full FFT

**Impact estimation:**
If we change `a[i]` by `delta`, the change in `(a*a)[k]` is approximately:
`d(a*a)[k] ≈ 2 * delta * a[k-i]` (for the linear term, ignoring `delta^2`)

So the change in the max is approximately:
`d(max) ≈ max_k 2 * delta * a[k-i]` (if the peak doesn't shift)

More precisely, if the current peak is at position `k*`, then:
`d(a*a)[k*] ≈ 2 * delta * a[k*-i]`

If `delta < 0` (reducing `a[i]`) and `a[k*-i] > 0`, this reduces the peak.
The "cost" is that we need to add `|delta|` somewhere else.

The best place to add it is at a position `j` where `a[k*-j]` is small (so it doesn't increase the peak much).

This gives us an O(n) heuristic for choosing good swap partners.

Key improvements over prior approach:

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T such that T @ b = conv(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        T[k, i_start:i_end + 1] = a_fixed[k - i_end:k - i_start + 1][::-1]
    return T


def solve_lp_best_response(a_fixed):
    """Solve min M s.t. (a*b)[k] <= M for all k, sum(b)=1, b>=0.
    Returns (b, M) or (None, None) on failure."""
    from scipy.optimize import linprog
    n = len(a_fixed)
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_ub = np.zeros((2 * n - 1, n + 1))
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        A_ub[k, i_start:i_end + 1] = a_fixed[k - i_end:k - i_start + 1][::-1]
        A_ub[k, n] = -1.0
    b_ub = np.zeros(2 * n - 1)
    
    A_eq = np.ones((1, n))
    A_eq_full = np.hstack([A_eq, np.zeros((1, 1))])
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub,
                         A_eq=A_eq_full, b_eq=b_eq,
                         bounds=bounds, method='highs')
        if result.success:
            b_val = np.maximum(result.x[:n], 0)
            s = b_val.sum()
            if s > 1e-15:
                b_val = b_val / s
            return b_val, float(result.x[n])
    except Exception:
        pass
    return None, None


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    v = np.maximum(v, 0)
    s = v.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return v / s

    Decay increases from decay_init to decay_end over the run."""

    b_ema = b.copy()

    iters = 0
    failed = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        
        # Adaptive decay: start low (fast tracking), end high (convergence)

        decay = decay_init + (decay_end - decay_init) * (1 - np.exp(-iters / 200.0))

        b_new, _ = solve_lp_best_response(a)
        if b_new is not None:

        a_new, _ = solve_lp_best_response(b)

            a = simplex_project((1 - decay) * a_new + decay * a)
        
        # Update EMA

        a_ema = simplex_project(ema_decay * a_ema + (1 - ema_decay) * a)
        b_ema = simplex_project(ema_decay * b_ema + (1 - ema_decay) * b)

            elapsed = time.monotonic() - t0
            print(f"    FP-EMA n={n} it={iters} ({elapsed:.0f}s) decay={decay:.4f}: best={best_score:.6f}")
    
    return best_a, best_score, iters


def fp_standard(n, time_limit, initial=None, verbose=False):
    """Standard fictitious play with uniform averaging."""
    t0 = time.monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:
        a = simplex_project(np.asarray(initial, dtype=np.float64).copy())
    
    b = a.copy()
    a_avg = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    iters = 0
    while time.monotonic() - t0 < time_limit:
        iters += 1
        b_new, _ = solve_lp_best_response(a)
        if b_new is not None:
            b = b_new
        a_new, _ = solve_lp_best_response(b)
        if a_new is not None:
            a = a_new
        
        a_avg = (iters * a_avg + a) / (iters + 1)
        
        candidates = [a, a_avg, 0.5 * (a + a_avg)]
        for cand in candidates:
            cand = simplex_project(cand)
            score = 2.0 * n * fast_convolve(cand).max()
            if score < best_score - 1e-14:
                best_score = score
                best_a = cand.copy()
        
        if verbose and iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP-Std n={n} it={iters} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score, iters


def swap_polish(a0, time_limit, seed=42):
    """Peak-aware swap-based local search."""
    n = len(a0)
    a = simplex_project(np.asarray(a0, dtype=np.float64).copy())
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        # Find top peak positions
        top_k = min(10, len(b))
        peaks = np.argpartition(b, -top_k)[-top_k:]
        
        improved = False
        for k_peak in peaks[:5]:
            k_peak = int(k_peak)
            i_start = max(0, k_peak - n + 1)
            i_end = min(n - 1, k_peak)
            idx = np.arange(i_start, i_end + 1)
            contribs = a[idx] * a[k_peak - idx]
            
            if contribs.max() < 1e-12:
                continue
            
            # Get top contributors
            top_idx = idx[np.argsort(contribs)[::-1][:15]]
            
            for i in top_idx:
                if a[i] < 1e-10:
                    continue
                # Try moving mass to random positions
                for trial in range(8):
                    j = rng.integers(n)
                    if j == i:
                        continue
                    eps_frac = rng.choice([0.05, 0.1, 0.15, 0.2, 0.3])
                    epsilon = a[i] * eps_frac
                    a_try = a.copy()
                    a_try[i] -= epsilon
                    a_try[j] += epsilon
                    a_try = np.maximum(a_try, 0)
                    s = a_try.sum()
                    if s < 1e-10:
                        continue
                    a_try = a_try / s
                    score = 2.0 * n * fast_convolve(a_try).max()
                    if score < best_score - 1e-14:
                        a = a_try
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        no_improve = 0
                        break
                if improved:
                    break
            
            if improved:
                break
        
        if not improved:
            no_improve += 1
            # Try smaller perturbations
            for trial in range(30):
                i = rng.integers(n)
                if a[i] < 1e-10:
                    continue
                j = rng.integers(n)
                if j == i:
                    continue
                epsilon = a[i] * rng.uniform(0.01, 0.06)
                a_try = a.copy()
                a_try[i] -= epsilon
                a_try[j] += epsilon
                a_try = np.maximum(a_try, 0)
                s = a_try.sum()
                if s < 1e-10:
                    continue
                a_try = a_try / s
                score = 2.0 * n * fast_convolve(a_try).max()
                if score < best_score - 1e-15:
                    a = a_try
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    no_improve = 0
                    break
            
            if no_improve > 20:
                break
        else:
            no_improve = 0
    
    return best_a, best_score


def coord_descent_fast(a0, time_limit, seed=123):
    """Fast coordinate descent focusing on peak-contributing coordinates."""
    n = len(a0)
    a = simplex_project(np.asarray(a0, dtype=np.float64).copy())
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve_streak = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        k_peak = int(np.argmax(b))
        i_start = max(0, k_peak - n + 1)
        i_end = min(n - 1, k_peak)
        idx = np.arange(i_start, i_end + 1)
        contribs = a[idx] * a[k_peak - idx]
        
        # Focus on top contributors
        top_count = min(30, len(idx))
        top_local = np.argsort(contribs)[::-1][:top_count]
        focus_idx = idx[top_local]
        
        improved = False
        for i in focus_idx:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-12:
                continue
            old_val = a[i]
            for factor in [0.95, 0.9, 0.85, 0.8, 0.7, 1.1, 1.15, 1.2]:
                a[i] = old_val * factor
                a = np.maximum(a, 0)
                s = a.sum()
                if s < 1e-10:
                    a[i] = old_val
                    a = a / a.sum()
                    continue
                a = a / s
                score = 2.0 * n * fast_convolve(a).max()
                if score < best_score - 1e-14:
                    best_score = score
                    best_a = a.copy()
                    old_val = a[i]
                    improved = True
                else:
                    a[i] = old_val
                    a = a / a.sum()
            
            if improved:
                break
        
        if not improved:
            no_improve_streak += 1
            # Random coordinate tries
            any_improve = False
            for _ in range(20):
                i = rng.integers(n)
                if a[i] < 1e-12:
                    continue
                old_val = a[i]
                for factor in [0.97, 0.98, 1.02, 1.03]:
                    a[i] = old_val * factor
                    a = np.maximum(a, 0)
                    s = a.sum()
                    if s < 1e-10:
                        a[i] = old_val
                        continue
                    a = a / s
                    score = 2.0 * n * fast_convolve(a).max()
                    if score < best_score - 1e-15:
                        best_score = score
                        best_a = a.copy()
                        old_val = a[i]
                        improved = True
                        any_improve = True
                        break
                    else:
                        a[i] = old_val
                        a = a / a.sum()
                if improved:
                    break
            if not any_improve and no_improve_streak > 15:
                break
        else:
            no_improve_streak = 0
    
    return best_a, best_score


def init_fourier(n, rng):
    """Initialize using random Fourier superposition."""
    n_harmonics = int(rng.integers(3, 20))
    amplitudes = rng.random(n_harmonics) * 0.4
    phases = rng.uniform(0, 2 * np.pi, n_harmonics)
    freqs = np.arange(1, n_harmonics + 1)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harmonics):
        a += amplitudes[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_smooth(n, rng):
    """Initialize with a smooth random profile (interpolated random control points)."""
    n_pts = int(rng.integers(5, 20))
    pts = np.sort(rng.random(n_pts))
    vals = rng.random(n_pts) + 0.05
    # Add a bias towards center
    center = 0.5
    vals += 0.3 * np.exp(-((pts - center) ** 2) / 0.1)
    x = np.linspace(0, 1, n)
    a = np.interp(x, pts, vals)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_two_hump(n, rng):
    """Two-hump initialization."""
    c1 = rng.uniform(0.2, 0.45)
    c2 = 1.0 - c1
    w1 = rng.uniform(0.05, 0.2)
    w2 = rng.uniform(0.05, 0.2)
    h1 = rng.uniform(0.5, 1.5)
    h2 = rng.uniform(0.5, 1.5)
    t = np.linspace(0, 1, n)
    a = h1 * np.exp(-((t - c1) ** 2) / (2 * w1 ** 2)) + \
        h2 * np.exp(-((t - c2) ** 2) / (2 * w2 ** 2)) + 0.05
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_and_track(a, best_seq, best_score, label=""):
    """Evaluate candidate using the actual evaluate_sequence function."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score
