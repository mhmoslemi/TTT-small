import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum x = s}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive-temperature gradient
descent on the simplex with FFT-based gradient computation, active-set peak
equalization, and L-BFGS polishing. Key improvements over previous approach:
1. Simplex projection instead of log-space softmax (allows exact zeros)
2. Backtracking line search for adaptive step sizes
3. Direct peak equalization via pair manipulation
4. Multiple starting points including designed initializations

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_general(x, y):
    """Linear convolution of two sequences via FFT."""
    n = len(x) + len(y) - 1
    fft_size = 1 << (n - 1).bit_length()
    X = rfft(x, fft_size)
    Y = rfft(y, fft_size)
    return irfft(X * Y, fft_size)[:n]


def project_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s} using the efficient algorithm."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - s)
    if not cond.any():
        return np.full(n, s / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_score_normalized(a):
    """Compute 2*n*max(conv) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def compute_gradient_smoothed(conv, a, T):
    """
    Compute the smoothed gradient of T*log(sum(exp(conv/T))) w.r.t. a.
    Uses the formula: g[j] = 2 * sum_k w[k] * a[k-j] * 1_{0<=k-j<n}
    where w = softmax(conv/T).
    Computed via FFT: g = 2 * flip(linear_conv(rev(w), a))[n-1:2n-1]
    """
    n = len(a)
    size = 2 * n - 1
    # Softmax weights
    shifted = conv / T
    max_s = shifted.max()
    w = np.exp(shifted - max_s)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(size) / size
    else:
        w = w / w_sum
    
    # g[j] = 2 * sum_k w[k] * a[k-j]
    # = 2 * linear_conv(rev(w), a)[size-1-j]
    w_rev = w[::-1]
    conv_wa = fast_convolve_general(w_rev, a)
    # conv_wa has length size + n - 1 = 3n - 2
    # g[j] = 2 * conv_wa[size-1-j] = 2 * conv_wa[2n-2-j]
    # For j=0,...,n-1: indices 2n-2, 2n-3, ..., n-1
    g = 2.0 * conv_wa[n-1:2*n-1][::-1]
    return g

    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score_normalized(a0)
    a = a / s
    
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    
    T = T_init
    lr = lr_init
    momentum = momentum_init
    v = np.zeros(n)
    t0 = time.monotonic()
    iters = 0
    stagnation_count = 0
    best_stagnation_reset = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        g = compute_gradient_smoothed(conv, a, T)
        
        # Momentum update
        v = momentum * v + lr * g
        a_trial = a - v
        
        # Project onto simplex
        a_trial = project_simplex(a_trial, 1.0)
        
        # Check improvement
        score = compute_score_normalized(a_trial)
        
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_trial.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        a = a_trial
        
        # Anneal temperature
        T = max(T_min, T * 0.9997)
        
        # Decay learning rate slowly
        lr = max(0.01, lr * 0.99995)
        
        # Momentum decay
        momentum = max(0.8, momentum * 0.99999)
        
        # Stagnation recovery
        if stagnation_count > 500:
            T = min(T_init, T * 3.0)
            lr = lr_init * 0.3
            v = np.zeros(n)
            stagnation_count = 0
            # Perturb from best
            rng = np.random.default_rng(hash((iters, n)) % 2**32)
            noise_scale = 0.005 * (stagnation_count // 500)
            a = best_a + rng.standard_normal(n) * noise_scale
            a = project_simplex(a, 1.0)
        
        if verbose and iters % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GD iter={iters}, T={T:.4f}, lr={lr:.4f}, "
                  f"score={best_score:.8f}, elapsed={elapsed:.1f}s")
    
    return best_a, best_score


def active_set_equalize(a0, time_limit, K=10, verbose=False):
    """
    Direct peak equalization: iteratively reduce the top peaks by
    manipulating the pairs that contribute to them.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score_normalized(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    iters = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        max_val = conv.max()
        
        # Find top K peaks
        K_actual = min(K, len(conv))
        top_indices = np.argpartition(conv, -K_actual)[-K_actual:]
        top_indices = top_indices[np.argsort(conv[top_indices])[::-1]]
        
        improved = False
        for k in top_indices:
            if time.monotonic() - t0 > time_limit:
                break
            
            # Find contributing pairs: conv[k] = sum_i a[i]*a[k-i]
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start > i_end:
                continue
            
            idx = np.arange(i_start, i_end + 1)
            contribs = a[idx] * a[k - idx]
            
            # Find the top contributing pair
            if contribs.sum() < 1e-15:
                continue
            
            top_c = np.argsort(contribs)[::-1][:3]
            for ci in top_c:
                i = int(idx[ci])
                j = int(k - i)
                if i >= n or j >= n or i < 0 or j < 0:
                    continue
                
                # Try to reduce min(a[i], a[j]) by moving mass to a random position
                # that doesn't contribute to the top peaks
                src = i if a[i] >= a[j] else j
                if a[src] < 1e-10:
                    continue
                
                for frac in [0.05, 0.1, 0.15, 0.2]:
                    delta = a[src] * frac
                    # Choose destination: a position not in the top peak contributing sets
                    peak_positions = set()
                    for pk in top_indices[:5]:
                        ps = max(0, int(pk) - n + 1)
                        pe = min(n - 1, int(pk))
                        peak_positions.update(range(ps, pe + 1))
                    
                    # Try a few random destinations outside peak positions
                    for _ in range(5):
                        dst = int(rng.integers(0, n))
                        if dst in peak_positions and rng.random() < 0.7:
                            continue
                        
                        a_trial = a.copy()
                        a_trial[src] -= delta
                        a_trial[dst] += delta
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        sc = compute_score_normalized(a_trial)
                        if sc < best_score - 1e-13:
                            a = a_trial
                            best_score = sc
                            best_a = a_trial.copy()
                            improved = True
                            break
                    if improved:
                        break
            if improved:
                break
        
        if not improved:
            # Perturb
            noise = rng.standard_normal(n) * 0.002
            a = best_a + noise
            a = project_simplex(a, 1.0)
        
        if verbose and iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ASE iter={iters}, score={best_score:.8f}, elapsed={elapsed:.1f}s")
    
    return best_a, best_score


def lbgfs_polish(a0, time_limit, T=100.0, maxiter=5000):
    """L-BFGS polish using smoothed objective."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        return a0, compute_score_normalized(a0)
    a = a / s
    
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    t0 = time.monotonic()
    
    # Use log-space parameterization for L-BFGS (unconstrained)
    a_safe = np.maximum(a, 1e-15)
    z0 = np.log(a_safe)
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e10
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_local = exp_z / exp_z.sum()
        conv = fast_convolve(a_local)
        # Smoothed max
        shifted = conv / T
        ms = shifted.max()
        lse = ms + np.log(np.exp(shifted - ms).sum())
        return T * lse
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        a_local = exp_z / exp_z.sum()
        conv = fast_convolve(a_local)
        shifted = conv / T
        ms = shifted.max()
        w = np.exp(shifted - ms)
        w = w / w.sum()
        # Gradient w.r.t. a
        g = compute_gradient_smoothed(conv, a_local, T)
        # Chain rule: dz -> da
        # da/dz = diag(a) - a*a^T (Jacobian of softmax)
        # grad_z = a * (g - a @ g)
        S = np.dot(a_local, g)
        grad_z = a_local * (g - S)
        return grad_z
    
    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 30, 'disp': False}
        )
        z_max = res.x.max()
        exp_z = np.exp(res.x - z_max)
        a_out = exp_z / exp_z.sum()
        a_out = np.maximum(a_out, 0)
        a_out = a_out / a_out.sum()
        score = compute_score_normalized(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except Exception:
        pass
    
    return best_a, best_score


def pnorm_descent(a0, time_limit, p=50.0, lr=0.1, n_iterations=None):
    """
    Gradient descent using p-norm smoothing of the max.
    f(a) = (sum_k conv[k]^p)^(1/p)
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score_normalized(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    t0 = time.monotonic()
    v = np.zeros(n)
    momentum = 0.9
    iters = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        # p-norm gradient
        conv_p = np.maximum(conv, 1e-30)
        pnorm = np.sum(conv_p ** p) ** (1.0 / p)
        weights = (conv_p ** (p - 1)) / (np.sum(conv_p ** p) ** ((p - 1) / p))
        # g[j] = 2 * sum_k weights[k] * a[k-j]
        w_rev = weights[::-1]
        size = 2 * n - 1
        conv_wa = fast_convolve_general(w_rev, a)
        g = 2.0 * conv_wa[n-1:2*n-1][::-1]
        
        v = momentum * v + lr * g
        a_trial = a - v
        a_trial = project_simplex(a_trial, 1.0)
        
        score = compute_score_normalized(a_trial)
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_trial.copy()
        
        a = a_trial
        lr *= 0.9999
        momentum = min(0.95, momentum + 0.0001)
        
        if iters % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    pnorm iter={iters}, score={best_score:.8f}, elapsed={elapsed:.1f}s")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate and update best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score
