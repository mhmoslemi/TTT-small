```python
def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
```

Near a local minimum `a*`, the objective `f(a) = max_k conv(a,a)[k]` is approximately:
`f(a* + d) ≈ f(a*) + sum_{k in S} w_k * (2 * d^T T_k a* + d^T T_k d)`

where `T_k` is the matrix such that `conv(a,a)[k] = a^T T_k a` and `S` is the set of active peaks.

If only one peak is active (say k*), then:
`f(a* + d) ≈ f(a*) + 2*d^T T_{k*} a* + d^T T_{k*} d`

The minimum of this quadratic in d (subject to sum(d) = 0 and a* + d >= 0) can be found by solving a QP.

If multiple peaks are active, we use the weighted combination.

This is essentially a Gauss-Newton step, which should converge quadratically near the optimum.

```python
def gauss_newton_step(a, top_k=3):
    """
    Gauss-Newton step for minimizing max_k conv(a,a)[k]
    subject to sum(a) = 1, a >= 0.
    """
    n = len(a)
    b = fast_convolve(a)
    peaks = np.argsort(b)[::-1][:top_k]
    peak_vals = b[peaks]
    
    # Weights: softmax of peak values
    temp = 0.01 * max(peak_vals[0], 1e-10)
    w = np.exp((peak_vals - peak_vals[0]) / temp)
    w /= w.sum()
    
    # Compute the "Hessian" H = sum_k w_k * T_k (symmetric matrix)
    # H[i,j] = sum_k w_k * (a[k-i] if j==... ) 
    # Actually, T_k[i,j] = 1 if i+j==k (for the auto-convolution)
    # Wait, conv(a,a)[k] = sum_i a[i] * a[k-i]
    # So T_k[i,j] = 1 if j = k-i, i.e., T_k[i, k-i] = 1
    # The quadratic form a^T T_k a = sum_i a[i] * a[k-i] (counting each pair once)
    # Actually a^T T_k a = sum_i sum_j a[i] T_k[i,j] a[j] = sum_i a[i] * a[k-i]
    # So T_k is the matrix with 1's on the "anti-diagonal" corresponding to k.
    
    # The gradient of conv(a,a)[k] is: 2*a[k-i] for coordinate i (as derived earlier)
    # The Hessian of conv(a,a)[k] is: 2*T_k (the anti-diagonal matrix)
    
    # For the Gauss-Newton step, we want to minimize:
    # f(a* + d) ≈ sum_k w_k * (2*d^T T_k a* + d^T T_k d)
    # = d^T (sum_k w_k * T_k) d + 2*(sum_k w_k * T_k a*)^T d
    
    # Let H = sum_k w_k * T_k and g = 2 * sum_k w_k * T_k a*
    # Then we minimize d^T H d + g^T d subject to 1^T d = 0, a* + d >= 0
    
    # H[i,j] = sum_k w_k * [j == k-i] = sum_k w_k * [i+j == k]
    # g[i] = 2 * sum_k w_k * a[k-i]
    
    # Let's compute H and g:
    H = np.zeros((n, n))
    g = np.zeros(n)
    
    for ki, k in enumerate(peaks):
        wk = w[ki]
        # T_k[i,j] = 1 if i+j == k
        # H contribution: wk * T_k
        # g contribution: 2*wk * T_k @ a = 2*wk * a[k-i] (with bounds)
        
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            if 0 <= j < n:
                H[i, j] += wk
                H[j, i] += wk
                # Wait, T_k is symmetric? T_k[i,j] = 1 if i+j=k. Yes, symmetric.
                # But a^T T_k a = sum_i a[i] * (T_k a)[i] = sum_i a[i] * a[k-i]
                # And (T_k a)[i] = a[k-i]. So the gradient is T_k a + T_k^T a = 2 T_k a (since T_k is symmetric)
                # So g[i] += 2*wk*a[k-i]
    
    for i in range(n):
        for ki, k in enumerate(peaks):
            j = k - i
            if 0 <= j < n:
                g[i] += 2.0 * w[ki] * a[j]
    
    # Now solve: minimize d^T H d + g^T d subject to 1^T d = 0, a + d >= 0
    # Lagrangian: d^T H d + g^T d + lambda * 1^T d
    # KKT: 2*H*d + g + lambda*1 = 0, 1^T d = 0
    # From first: d = -(1/2) * H^{-1} * (g + lambda*1)
    # From second: 1^T H^{-1} (g + lambda*1) = -2 * 1^T d = 0... 
    
    # Actually, let me just use a simpler approach: project the gradient step onto the simplex tangent.
    # Direction: -g (steepest descent of the linearization)
    # Project onto sum(d) = 0: d = -g + (sum(g)/n) * 1
    # Step size: line search
    
    d = -g
    d = d - np.mean(d)  # Project onto sum=0
    
    # Line search
    best_alpha = 0
    best_f = 2.0 * n * b.max()
    for alpha in [0.5, 0.3, 0.2, 0.15, 0.1, 0.05, 0.03, 0.02, 0.01]:
        trial = a + alpha * d
        if trial.min() < -1e-10:
            continue
        trial = np.maximum(trial, 0)
        trial = trial / trial.sum()
        f_trial = 2.0 * len(trial) * fast_convolve(trial).max()
        if f_trial < best_f - 1e-14:
            best_f = f_trial
            best_alpha = alpha
    
    if best_alpha > 0:
        new_a = a + best_alpha * d
        new_a = np.maximum(new_a, 0)
        new_a = new_a / new_a.sum()
        return new_a
    return a
```

```python
def cyclic_feedback(a, threshold_factor=1.0, n_iters=1000):
    n = len(a)
    a = a / a.sum()
    for it in range(n_iters):
        b = fast_convolve(a)
        threshold = threshold_factor * b.max()
        # For each position i, compute how much it "contributes" to the peaks
        contribution = np.zeros(n)
        for k in range(len(b)):
            if b[k] > threshold:
                excess = b[k] - threshold
```

```python
def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

            T[k, j] = a_fixed[k - j]
    return T

        # Softmax weights for active peaks
        temp = 0.02 * max(peak_vals[0], 1e-10)
        w = np.exp((peak_vals - peak_vals[0]) / temp)
        w = w / w.sum()

        # Compute subgradient: g[i] = 2 * sum_k w_k * a[k-i] (with bounds)
        g = np.zeros(n)
        for ki, k in enumerate(peak_idx):
            wk = w[ki]
            if wk < 1e-12:
                continue

            i_end = min(n - 1, k)

                if 0 <= j < n:
                    g[i] += 2.0 * wk * a[j]

        # Projected gradient step: move in direction -g, projected onto simplex tangent

        best_alpha = 0.0

            sc = score_func(trial)
            if sc < best_trial_score - 1e-15:
                best_trial_score = sc
                best_alpha = alpha

        if best_alpha > 0:
            a = a + best_alpha * d
            a = np.maximum(a, 0)
            a = a / a.sum()
            if best_trial_score < best_score - 1e-14:
                best_score = best_trial_score
                best_a = a.copy()

            print(f"    SubGrad it={it} ({elapsed:.0f}s): best={best_score:.6f} lr={lr:.6f}")

def coord_descent_line_search(a0, time_interval):
    """Coordinate descent with golden-section line search on each coordinate."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()

                continue

            # Parameterize: a[i] = t, rest scaled to maintain sum = 1
            # New a: a'[i] = t, a'[j] = a[j] * (1-t)/(1-a[i]) for j != i
            lo = 0.0
            hi = min(a[i] * 3.0, 2.0)
            golden = (np.sqrt(5) - 1) / 2

            def eval_at_t(t):
                if t < 0 or t > hi:
                    return float('inf')
                rest_sum = 1.0 - a[i]
                if rest_sum < 1e-15:
                    return float('inf')
                scale = (1.0 - t) / rest_sum
                if scale < -1e-10:
                    return float('inf')
                trial = a.copy() * max(scale, 0)
                trial[i] = t
                if trial.min() < -1e-12:
                    return float('inf')
                trial = np.maximum(trial, 0)
                trial_sum = trial.sum()
                if trial_sum < 1e-15:
                    return float('inf')
                trial = trial / trial_sum
                return score_func(trial)

            # Golden section
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            c = lo + (1.0 - gr) * (hi - lo)
            d_pt = lo + gr * (hi - lo)
            fc = eval_at_t(c)
            fd = eval_at_t(d_pt)
            for _ in range(25):
                if fc < fd:
                    hi = d_pt
                    d_pt = c
                    fd = fc
                    c = lo + (1.0 - gr) * (hi - lo)
                    fc = eval_at_t(c)
                else:
                    lo = c
                    c = d_pt
                    fc = fd
                    d_pt = lo + gr * (hi - lo)
                    fd = eval_at_t(d_pt)

            t_opt = (lo + hi) / 2.0
            f_opt = eval_at_t(t_opt)
            if f_opt < best_score - 1e-14:
                # Apply the optimal value

                        best_a = a.copy()
                        improved_sweep = True

        if not improved_sweep:
            break

    return best_a, best_score

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def check_and_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    sc = evaluate_local(a_norm)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a_norm)})")
    return best_seq, best_score
```
