"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a novel fictitious play
(min-max) algorithm with FFT-based Hankel-transform gradients, combined
with Adam log-space refinement and peak-targeted local search.

Key innovation: The min-max reformulation min_a max_k conv(a,a)[k] is
solved via alternating optimization (fictitious play), where the inner
problem (minimize p^T conv given dual p) has gradients computable in
O(n log n) via FFT-based Hankel transforms. This provides a more
principled optimization landscape than pure gradient descent on the
non-smooth max, and the dual variable p naturally identifies which
autocorrelation lags are "active" (binding constraints).
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def hankele_transform(a, p, n, fft_size=None):
    """
    Compute (P@a)[i] = sum_j p[i+j] * a[j] for i=0,...,n-1
    where p has length 2n-1.
    Uses FFT: (P@a) = conv(flip(a), p)[n-1 : 2n-2]
    """
    if fft_size is None:
        # conv(flip(a), p) has length n + (2n-1) - 1 = 3n-1
        total = 3 * n - 1
        fft_size = 1 << (total - 1).bit_length()
    
    r = a[::-1]  # flip
    R = rfft(r, fft_size)
    P = rfft(p, fft_size)
    conv = irfft(R * P, fft_size)
    return conv[n-1 : 2*n-2]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    c = fast_convolve(a)
    return 2.0 * n * c.max()


def simplex_project(x):
    """Project x onto the probability simplex {y >= 0, sum(y) = 1}."""
    n = len(x)
    # Simple projection: clip and normalize (not exact but fast)
    x = np.maximum(x, 0)
    s = x.sum()
    if s < 1e-300:
        return np.ones(n) / n
    return x / s


def simplex_project_exact(x):
    """Exact projection onto probability simplex."""
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(x - theta, 0)


def fictitious_play(a_init, time_limit, n_inner=20, lr=0.001, alpha=0.1,
                    use_fista=True, verbose=False):
    """
    Fictitious play (min-max) algorithm for minimizing max autocorrelation.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    # Dual variable: probability over lags
    n_lags = 2 * n - 1
    p = np.zeros(n_lags)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a fictitious play (min-max)
algorithm with FFT-based Hankel-transform gradients for the dual-to-primal
mapping, combined with Adam log-space refinement, peak-targeted local search,
and multi-length exploration.

The key innovation is the min-max reformulation:
  min_{a in simplex} max_k conv(a,a)[k]
solved via alternating optimization where the primal step uses multiplicative
weight updates (mirror descent) with O(n log n) gradient computation via
FFT-based Hankel transforms, and the dual step uses softmax-weighted updates.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def hankele_grad(a, p, n, fft_size):
    """
    Compute grad[i] = 2 * sum_j p[i+j] * a[j] for i=0,...,n-1.
    This is the gradient of f(a) = sum_k p[k] * conv(a,a)[k] w.r.t. a.
    Uses FFT: sum_j p[i+j]*a[j] = conv(flip(a), p)[i+n-1]
    """
    r = a[::-1]
    R = rfft(r, fft_size)
    P = rfft(p, fft_size)
    conv_full = irfft(R * P, fft_size)
    return 2.0 * conv_full[n - 1:2 * n - 1]


def compute_score_direct(a):
    """Compute 2*n*max(conv(a,a)) for a with sum=1."""
    n = len(a)
    c = fast_convolve(a)
    return 2.0 * n * c.max()


def softmax_stable(z, T=1.0):
    """Numerically stable softmax."""
    z_shifted = z / T - np.max(z) / T
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return exp_z / s


def mu_update(a, grad, lr):
    """
    Multiplicative weight update (mirror descent on simplex).
    a_new[i] = a[i] * exp(-lr * grad[i]) / Z
    """
    log_a = np.log(np.maximum(a, 1e-300))
    log_a_new = log_a - lr * grad
    log_a_new -= np.max(log_a_new)
    exp_a = np.exp(log_a_new)
    s = exp_a.sum()
    if s < 1e-300:
        return np.ones_like(a) / len(a)
    return exp_a / s


def fictitious_play(a_init, time_limit, n_inner=50, lr_init=0.05,
                    alpha=0.05, dual_T=0.01, verbose=False):
    """
    Fictitious play (min-max) for minimizing max autocorrelation.
    
    Alternates between:
    - Primal: multiplicative weight updates to minimize p^T conv(a,a)
    - Dual: update p towards the current peak lag
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-300).astype(np.float64)
    a = a / a.sum()
    
    n_lags = 2 * n - 1
    c_init = fast_convolve(a)
    
    # Initialize p at current peak
    p = np.zeros(n_lags)
    # Use soft initialization: put more weight near the peak
    peak_k = np.argmax(c_init)
    logits = c_init / dual_T
    logits -= logits.max()
    p = np.exp(logits)
    p /= p.sum()
    
    # FFT size for Hankel transform: conv(flip(a), p) has length n + n_lags - 1 = 3n-1
    total = 3 * n - 1
    fft_size = 1 << (total - 1).bit_length()
    
    best_a = a.copy()
    best_score = 2.0 * n * c_init.max()
    
    t0 = time.monotonic()
    outer_iter = 0
    lr = lr_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # === Primal step: optimize a given p (MWU) ===
        for _ in range(n_inner):
            grad = hankele_grad(a, p, n, fft_size)
            a = mu_update(a, grad, lr)
        
        # === Dual step: update p ===
        c = fast_convolve(a)
        # Soft update using log-sum-exp
        logits = c / dual_T
        logits -= logits.max()
        p_soft = np.exp(logits)
        p_soft /= p_soft.sum()
        p = (1.0 - alpha) * p + alpha * p_soft
        
        # Track best
        score = 2.0 * n * c.max()
        if score < best_score - 1e-15:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive: reduce lr if no improvement
        if no_improve > 200:
            lr *= 0.7
            no_improve = 0
        
        outer_iter += 1
        if verbose and outer_iter % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP iter={outer_iter} t={elapsed:.1f}s "
                  f"lr={lr:.5f} best={best_score:.6f}")
    
    return best_a, best_score


def adam_logspace(a0, time_limit, T_init=0.05, T_final=0.002, lr=0.01,
                  beta1=0.9, beta2=0.999, eps=1e-8, schedule='hold_anneal',
                  verbose=False, verbose_interval=5000):
    """
    Adam optimizer in log-space with temperature annealing.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    a = a / a.sum()
    z = np.log(a)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    fft_size = 1 << (2 * n - 2).bit_length()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    no_improve = 0
    max_no_improve = 5000
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        
        if schedule == 'linear':
            T = T_init + (T_final - T_init) * progress
        elif schedule == 'exp':
            T = T_init * (T_final / T_init) ** progress
        elif schedule == 'hold_anneal':
            if progress < 0.3:
                T = T_init
            else:
                p2 = (progress - 0.3) / 0.7
                T = T_init * (T_final / T_init) ** p2
        else:
            T = T_init * (T_final / T_init) ** progress
        
        a_curr = np.exp(z - z.max())
        a_curr = a_curr / a_curr.sum()
        
        b = fast_convolve(a_curr)
        max_b = b.max()
        exp_b = np.exp(b / T - max_b / T)
        w = exp_b / exp_b.sum()
        
        # Gradient via cross-correlation
        W = rfft(w, fft_size)
        A = rfft(a_curr, fft_size)
        corr = irfft(W * np.conj(A), fft_size)[:n]
        grad_a = 2.0 * n * corr
        
        S = np.dot(grad_a, a_curr)
        grad_z = a_curr * (grad_a - S)
        
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        a_new = np.exp(z - z.max())
        a_new = a_new / a_new.sum()
        conv_new = fast_convolve(a_new)
        score = 2.0 * n * conv_new.max()
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > max_no_improve:
            z = np.log(np.maximum(best_a, 1e-300)) + np.random.standard_normal(n) * 0.003
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = max(0.001, lr * 0.7)
        
        if verbose and t % verbose_interval == 0:
            print(f"      Adam t={t} T={T:.5f} lr={lr:.6f} score={best_score:.6f}")
    
    return best_a, best_score


def peak_shave(a0, time_limit, n_trials=5000):
    """
    Peak-targeted local search.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()
    rng = np.random.default_rng(42)
    
    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        # Find peak (prefer interior)
        b_search = b.copy()
        b_search[0] = -np.inf
        b_search[-1] = -np.inf
        peak_k = int(np.argmax(b_search))
        
        # Elements contributing to this peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        if len(idx) == 0:
            continue
        contribs = a[idx] * a[peak_k - idx]
        top_count = min(8, len(idx))
        top_idx = np.argsort(contribs)[::-1][:top_count]
        
        improved = False
        for ci in top_idx:
            i = int(idx[ci])
            j = int(peak_k - i)
            if j < 0 or j >= n or i == j or a[j] < 1e-15 or a[i] < 1e-15:
                continue
            
            for f1 in [0.92, 0.85, 0.78]:
                for f2 in [1.0, 1.03, 1.06]:
                    a_trial = a.copy()
                    a_trial[i] *= f1
                    a_trial[j] *= f2
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a_trial.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Try single-element reduction with redistribution
            for ci in top_idx[:4]:
                i = int(idx[ci])
                if a[i] < 1e-15:
                    continue
                for factor in [0.85, 0.75, 0.65]:
                    a_trial = a.copy()
                    diff = a_trial[i] * (1.0 - factor)
                    a_trial[i] *= factor
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    others_sum = a_trial[mask].sum()
                    if others_sum < 1e-15:
                        continue
                    a_trial[mask] += diff * a_trial[mask] / others_sum
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a_trial.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if not improved and trial > 200 and trial % 100 == 0:
            # Random perturbation to escape
            a = best_a + rng.standard_normal(n) * 0.004
            a = np.maximum(a, 0)
            a = a / a.sum()
    
    return best_a, best_score


def coordinate_descent(a0, time_limit, n_sweeps=100):
    """Coordinate descent with multiplicative updates."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(99)
    
    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        
        order = rng.permutation(n)
        sweep_improved = False
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-15:
                continue
            
            best_local = a[i]
            best_local_score = best_score
            
            for factor in [0.75, 0.85, 0.92, 1.08, 1.15, 1.25]:
                a_trial = a.copy()
                a_trial[i] = a[i] * factor
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_local_score:
                    best_local_score = score
                    best_local = a[i] * factor
            
            if best_local != a[i]:
                a[i] = best_local
                a = a / a.sum()
                if best_local_score < best_score:
                    best_score = best_local_score
                    best_a = a.copy()
                    sweep_improved = True
        
        if not sweep_improved:
            break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
