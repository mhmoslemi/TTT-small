    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def central_binomial_init(n):
    """Initialize with central binomial coefficients c_i = binom(2i,i)/4^i."""
    c = np.ones(n)
    for i in range(1, n):
        c[i] = c[i-1] * (2*i - 1) / (2*i)
    s = c.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return c / s


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver, preferring PDLP."""
    if '_lp_solver' in globals():
        return _lp_solver
    for solver_name in [cv.PDLP, cv.SCS, cv.ECOS, cv.GLPK]:
        try:
            x = cv.Variable(3, nonneg=True)
            prob = cv.Problem(cv.Minimize(cv.sum(x)), [cv.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cv.SCS
    return cv.SCS


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (sum=1), find b (sum=1, b>=0) minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cv.Variable(n, nonneg=True)
    M_var = cv.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cv.sum(b_var) == 1]
    prob = cv.Problem(cv.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cv.PDLP, cv.SCS, cv.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def fictitious_play(n, time_limit, initial=None, solver=None, verbose=True):
    """LP-based fictitious play for auto-correlation minimization."""
    if solver is None:
        solver = get_lp_solver()
    t0 = monotonic()
    if initial is None:
        a = central_binomial_init(n)
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        a = a / s if s > 1e-15 else np.ones(n) / n

    b = a.copy()
    a_avg = a.copy().astype(np.float64)
    b_avg = b.copy().astype(np.float64)
    best_score = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    iters = 0
    stagnation = 0

    while monotonic() - t0 < time_limit:
        iters += 1

        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new

        a_new, M_a = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new

        a_avg = (iters * a_avg + a) / (iters + 1)
        b_avg = (iters * b_avg + b) / (iters + 1)

        candidates = [a, b, 0.5 * (a + b), a_avg, 0.5 * (a_avg + b_avg)]
        improved = False
        for cand in candidates:
            cand = np.maximum(cand, 0)
            s = cand.sum()
            if s < 1e-10:
                continue
            cand_norm = cand / s
            score = 2.0 * n * fast_convolve(cand_norm).max()
            if score < best_score - 1e-12:
                best_score = score
                best_a = cand_norm.copy()
                improved = True
                stagnation = 0
        if not improved:
            stagnation += 1

        if stagnation > 150:
            noise = np.random.standard_normal(n) * 0.002
            a = a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            b = b + np.random.standard_normal(n) * 0.002
            b = np.maximum(b, 0)
            b = b / b.sum()
            a_avg = a.copy()
            b_avg = b.copy()
            stagnation = 0

        if verbose and iters % 50 == 0:
            elapsed = monotonic() - t0
            print(f"    FP n={n} iter={iters} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score, iters


def lbgfs_polish(a0, time_limit, T=50, maxiter=3000):
    """L-BFGS refinement in log-space with soft temperature."""
    n = len(a0)
    a_safe = np.maximum(a0, 1e-30)
    z0 = np.log(a_safe)
    z0 = z0 - z0.mean()
    t0 = monotonic()
    stopped = [False]

    def func(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()
        b = fast_convolve(a)
        max_b = b.max()
        scaled_b = (b - max_b) / T
        exp_b = np.exp(scaled_b)
        lse = max_b + T * np.log(exp_b.sum())
        return 2.0 * n * lse / T

    def jac(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return np.zeros(n)
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()
        b = fast_convolve(a)
        max_b = b.max()
        scaled_b = (b - max_b) / T
        exp_b = np.exp(scaled_b)
        w = exp_b / exp_b.sum()

        grad_a = np.zeros(n)
        for k in range(len(w)):
            if w[k] < 1e-10:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[k] * a[k - idx]

        grad_a *= 2.0 * n
        S = np.dot(grad_a, a)
        return a * (grad_a - a * S)

    try:
        res = minimize(fun=func, x0=z0, jac=jac, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-16,
                                'gtol': 1e-12, 'maxfun': maxiter * 15, 'disp': False})
        if not stopped[0]:
            z_final = res.x
            z_shifted = z_final - z_final.max()
            a_out = np.exp(z_shifted)
            a_out = a_out / a_out.sum()
        else:
            a_out = a0.copy()
    except Exception:
        a_out = a0.copy()
    return a_out


def peak_shave(a0, time_limit):
    """Local peak shaving optimization."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    failed = 0

    while monotonic() - t0 < time_limit and failed < 100:
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

        if contribs.sum() < 1e-20:
            break

        top_contrib = np.argsort(contribs)[::-1][:8]
        improved = False

        for ci in top_contrib:
            i = int(idx[ci])
            old_val = a[i]
            if old_val < 1e-14:
                continue
            for factor in [0.95, 0.9, 0.8, 0.7, 0.6]:
                new_val = old_val * factor
                diff = old_val - new_val
                if diff < 1e-15:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                a_trial[mask] = others + diff * others / others_sum
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try pair reduction
            for ci in top_contrib[:4]:
                i = int(idx[ci])
                j = int(peak_k - i)
                if i == j or a[j] < 1e-14:
                    continue
                for f1 in [0.9, 0.8]:
                    for f2 in [0.9, 0.8]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-13:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            # Random perturbation
            for _ in range(3):
                a_trial = a + np.random.standard_normal(n) * 0.001
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            failed += 1
        else:
            failed = 0

    return best_a, best_score


def half_quadratic_smooth(a0, time_limit, lr=0.1, max_iters=5000):
    """Half-quadratic / proximal-style optimization."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    mu = 1.0 / n  # regularization parameter

    for it in range(max_iters):
        if monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        k_peak = int(np.argmax(b))

        # Subgradient direction for the peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        grad = 2.0 * a[k_peak - idx]

        # Add regularization toward best_a
        direction = -grad + mu * (a - best_a)
        a_trial = a - lr * direction
        # Project onto simplex
        a_trial = np.maximum(a_trial, 0)
        ts = a_trial.sum()
        if ts < 1e-15:
            a_trial = best_a.copy()
        else:
            a_trial = a_trial / ts

        score = 2.0 * n * fast_convolve(a_trial).max()
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
        else:
            lr *= 0.95
            if lr < 1e-6:
                lr = 0.1
                mu *= 0.5

        if (it + 1) % 500 == 0:
            elapsed = monotonic() - t0
            print(f"    HQ {elapsed:.0f}s: score={best_score:.6f} lr={lr:.4f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate and return updated best."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. Central binomial init + LP fictitious play at multiple lengths
    2. L-BFGS polish
    3. Peak shaving local search
    4. Half-quadratic refinement
    5. Final verification
    """
    np.random.seed(seed)
    deadline = monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Try to use parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = p_score

    # Central binomial init at n=768 as fallback/alternative
    cb_init = central_binomial_init(768)
    cb_score = float(evaluate_sequence(cb_init.tolist()))
    print(f"Central binomial n=768: score={cb_score:.6f}")
    if best_seq is None or cb_score < best_score:
        best_seq = cb_init.copy()
        best_score = cb_score

    n_parent = len(best_seq)

    # === Phase 1: LP Fictitious Play ===
    print(f"\n=== Phase 1: LP Fictitious Play ===")

    # Primary: n=768 with central binomial start
    n1 = 768
    init1 = central_binomial_init(n1)
    if best_seq is not None and len(best_seq) == n1:
        # Blend: start from best of parent and CB
        init1 = 0.5 * best_seq + 0.5 * init1
        init1 = np.maximum(init1, 0)
        init1 = init1 / init1.sum()

    rem = deadline - monotonic()
    t1 = max(30, min(500, rem * 0.35))
    print(f"  FP n={n1}, budget={t1:.0f}s")
    a1, sc1, it1 = fictitious_play(n1, t1, initial=init1, solver=solver, verbose=True)
    best_seq, best_score = check_and_update(a1, f"FP1_n{n1}_it{it1}", best_seq, best_score)

    # Try parent-based init if different
    if parent is not None and len(parent) == n1:
        rem = deadline - monotonic()
        t1b = max(20, min(200, rem * 0.15))
        print(f"  FP n={n1} (parent init), budget={t1b:.0f}s")
        a1b, sc1b, it1b = fictitious_play(n1, t1b, initial=parent, solver=solver, verbose=False)
        best_seq, best_score = check_and_update(a1b, f"FP1b_n{n1}_it{it1b}", best_seq, best_score)

    # Secondary: n=1024
    n2 = 1024
    rem = deadline - monotonic()
    t2 = max(20, min(250, rem * 0.2))
    if t2 > 20:
        print(f"  FP n={n2}, budget={t2:.0f}s")
        init2 = interpolate_sequence(best_seq, n2) if best_seq is not None else central_binomial_init(n2)
        a2, sc2, it2 = fictitious_play(n2, t2, initial=init2, solver=solver, verbose=True)
        best_seq, best_score = check_and_update(a2, f"FP2_n{n2}_it{it2}", best_seq, best_score)

    # Tertiary: n=2048
    n3 = 2048
    rem = deadline - monotonic()
    t3 = max(20, min(200, rem * 0.15))
    if t3 > 20:
        print(f"  FP n={n3}, budget={t3:.0f}s")
        init3 = interpolate_sequence(best_seq, n3) if best_seq is not None else central_binomial_init(n3)
        a3, sc3, it3 = fictitious_play(n3, t3, initial=init3, solver=solver, verbose=True)
        best_seq, best_score = check_and_update(a3, f"FP3_n{n3}_it{it3}", best_seq, best_score)

    # === Phase 2: L-BFGS Polish ===
    print(f"\n=== Phase 2: L-BFGS Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        for T in [30, 80, 200]:
            rem = deadline - monotonic()
            if rem < 20:
                break
            t_alloc = min(rem * 0.15, 60.0)
            print(f"  L-BFGS T={T}, n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_lb = lbgfs_polish(best_seq, t_alloc, T=T)
            best_seq, best_score = check_and_update(a_lb, f"lbgfs_T{T}", best_seq, best_score)

    # === Phase 3: Peak Shaving ===
    print(f"\n=== Phase 3: Peak Shaving (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic()
        if rem > 15:
            t_ps = min(rem * 0.2, 80.0)
            print(f"  PeakShave n={len(best_seq)}, budget={t_ps:.0f}s")
            a_ps, _ = peak_shave(best_seq, t_ps)
            best_seq, best_score = check_and_update(a_ps, "peak_shave", best_seq, best_score)

    # === Phase 4: Half-Quadratic ===
    print(f"\n=== Phase 4: Half-Quadratic (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic()
        if rem > 20:
            t_hq = min(rem * 0.15, 60.0)
            print(f"  HQ n={len(best_seq)}, budget={t_hq:.0f}s")
            a_hq, _ = half_quadratic_smooth(best_seq, t_hq)
            best_seq, best_score = check_and_update(a_hq, "half_quad", best_seq, best_score)

    # === Phase 5: Second FP round on best ===
    print(f"\n=== Phase 5: FP Round 2 (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - monotonic()
        if rem > 40:
            t5 = min(rem * 0.25, 150.0)
            print(f"  FP round 2: n={n_best}, budget={t5:.0f}s")
            a5, sc5, it5 = fictitious_play(n_best, t5, initial=best_seq, solver=solver, verbose=False)
            best_seq, best_score = check_and_update(a5, f"FP2_n{n_best}_it{it5}", best_seq, best_score)

    # === Phase 6: Final L-BFGS + Peak Shave ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic()
        if rem > 15:
            t_f = min(rem * 0.3, 40.0)
            a_f = lbgfs_polish(best_seq, t_f, T=50, maxiter=2000)
            best_seq, best_score = check_and_update(a_f, "final_lbgfs", best_seq, best_score)

        rem = deadline - monotonic()
        if rem > 10:
            t_ps2 = min(rem * 0.4, 30.0)
            a_ps2, _ = peak_shave(best_seq, t_ps2)
            best_seq, best_score = check_and_update(a_ps2, "final_ps", best_seq, best_score)

    # Final verification
    if best_seq is None:
        best_seq = central_binomial_init(768)
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
