        try:
            res = linprog(c_vec, A_ub=A_mat, b_ub=b_lp,
                          bounds=[(0.0, None)] * n, method='highs',
                          options={'presolve': True, 'time_limit': max(1, time_limit - (time.monotonic() - t0))})
            if not res.success or res.x is None:
                break

            g = np.maximum(res.x, 0.0)
            g_sum = g.sum()
            if g_sum < 1e-10:
                break
            g_norm = g / g_sum

            # Line search for best blend
            best_t = 0.0
            best_blend_obj = best_obj
            for t_val in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0, 0.15, 0.25, 0.4, 0.6]:
                blend = (1.0 - t_val) * best_a + t_val * g_norm
                blend = np.maximum(blend, 0.0)
                ss = blend.sum()
                if ss < 1e-10:
                    continue
                blend = blend / ss
                obj_blend = eval_max_b(blend)
                if obj_blend < best_blend_obj:
                    best_blend_obj = obj_blend
                    best_t = t_val

            if best_t > 0.0:
                a_new = (1.0 - best_t) * best_a + best_t * g_norm
                a_new = np.maximum(a_new, 0.0)
                a_new = a_new / a_new.sum()
                obj_new = eval_max_b(a_new)
                if obj_new < best_obj - 1e-14:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    if verbose:
                        print(f"    LP iter {it}: t={best_t:.2f}, max_b={best_obj:.12f}")
                else:
                    a_new = best_a.copy()
            else:
                break
        except Exception:
            break

    return best_a, best_obj


def peak_coordinate_descent(a0, time_limit=60.0, verbose=False):
    """
    Peak-aware coordinate descent: focus on reducing elements that
    contribute most to current peaks.
    """
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = eval_max_b(a)

    multipliers = np.array([0.1, 0.2, 0.3, 0.5, 0.7, 0.85, 0.95,
                            1.05, 1.15, 1.3, 1.5, 2.0, 3.0])

    converged = False
    while not converged and time.monotonic() - t0 < time_limit:
        converged = True
        b = fast_convolve(a)
        b_max = b.max()
        peak_idx = np.argmax(b)

        # Find elements contributing to the peak
        lo = max(0, peak_idx - n + 1)
        hi = min(n - 1, peak_idx)
        contributors = np.arange(lo, hi + 1)

        # Sort contributors by contribution (a[j] * a[peak_idx - j])
        contrib_vals = np.array([a[j] * a[peak_idx - j] for j in contributors])
        order = np.argsort(contrib_vals)[::-1]
        top_contributors = contributors[order[:min(20, len(contributors))]]

        # Try reducing top contributors
        for j in top_contributors:
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-15:
                continue
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0.0 or new_val > 0.999:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < -0.01 or scale > 50.0:
                    continue
                a_tr = a.copy()
                a_tr[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_tr[mask] *= scale
                ss = a_tr.sum()
                if ss < 1e-15:
                    continue
                a_tr = a_tr / ss
                val = eval_max_b(a_tr)
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / (1.0 - old_val)
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    converged = False

    return best_a, best_obj


def make_init(n, method="tukey", **kwargs):
    """Generate various window initializations."""
    if method == "uniform":
        return np.ones(n) / n
    elif method == "gaussian":
        sigma = kwargs.get("sigma", 0.3)
        x = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif method == "cosine":
        x = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif method == "tukey":
        r = kwargs.get("r", 0.5)
        a = np.ones(n)
        taper = max(1, int(r * n // 2))
        if taper < n:
            t = np.linspace(0, 1, taper)
            a[:taper] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t / r - 1.0)))
            a[-taper:] = a[:taper][::-1]
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    elif method == "triangular":
        half = n // 2
        a = np.concatenate([np.linspace(0.1, 1.0, half),
                            np.linspace(1.0, 0.1, n - half)])
        return a / a.sum()
    elif method == "flat_top":
        # Flat top: minimize peak of auto-convolution
        # Use a distribution that makes the convolution as flat as possible
        x = np.linspace(0, 1, n)
        a = 1.0 - 0.6 * np.cos(2 * np.pi * x) + 0.08 * np.cos(4 * np.pi * x)
        a = np.maximum(a, 1e-12)
        return a / a.sum()
    else:
        return np.ones(n) / n


def interpolate_sequence(f, new_n):
    """Interpolate to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses log-sum-exp Adam, aggressive LP, peak-aware CD, and multi-restart.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, budget_s - 30.0)

    best_seq = None
    best_score = float('inf')

    def update_best(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        try:
            score = float(evaluate_sequence(a_norm.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = make_init(1152, "tukey", r=0.5)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"No parent, using tukey n=1152, score={best_score:.6f}")

    n_parent = len(parent)

    # === Phase 1: Log-sum-exp Adam on parent ===
    rem = deadline - time.monotonic() - 600
    if rem > 30:
        print(f"\nPhase 1: LSE-Adam on parent (n={n_parent}), t={min(rem, 200):.0f}s")
        t_alloc = min(rem, 200.0)
        a1, obj1 = logsumexp_optimize(parent, time_limit=t_alloc,
                                       tau_init=100, tau_max=3000,
                                       lr_init=0.008, seed=seed, verbose=True)
        update_best(a1, "phase1_lse")

    # === Phase 2: Aggressive LP ===
    rem = deadline - time.monotonic() - 500
    if rem > 20:
        print(f"\nPhase 2: Aggressive LP (n={len(best_seq)}), t={min(rem, 150):.0f}s")
        t_alloc = min(rem, 150.0)
        a2, obj2 = lp_refine_aggressive(best_seq, time_limit=t_alloc,
                                         max_iters=40, verbose=True)
        update_best(a2, "phase2_lp")

    # === Phase 3: LSE-Adam from LP result ===
    rem = deadline - time.monotonic() - 350
    if rem > 20:
        print(f"\nPhase 3: LSE-Adam round 2, t={min(rem, 150):.0f}s")
        t_alloc = min(rem, 150.0)
        a3, obj3 = logsumexp_optimize(best_seq, time_limit=t_alloc,
                                       tau_init=200, tau_max=8000,
                                       lr_init=0.004, seed=seed + 1000, verbose=True)
        update_best(a3, "phase3_lse2")

    # === Phase 4: More LP ===
    rem = deadline - time.monotonic() - 250
    if rem > 15:
        print(f"\nPhase 4: LP round 2, t={min(rem, 120):.0f}s")
        t_alloc = min(rem, 120.0)
        a4, obj4 = lp_refine_aggressive(best_seq, time_limit=t_alloc,
                                         max_iters=35, verbose=True)
        update_best(a4, "phase4_lp2")

    # === Phase 5: Peak-aware coordinate descent ===
    rem = deadline - time.monotonic() - 100
    if rem > 10:
        print(f"\nPhase 5: Peak CD, t={min(rem, 80):.0f}s")
        t_alloc = min(rem, 80.0)
        a5, obj5 = peak_coordinate_descent(best_seq, time_limit=t_alloc, verbose=True)
        update_best(a5, "phase5_cd")

    # === Phase 6: Multi-restart with diverse inits ===
    rem = deadline - time.monotonic() - 80
    if rem > 30:
        print(f"\nPhase 6: Multi-restart exploration, total t={rem:.0f}s")
        n_best = len(best_seq)
        inits = [
            make_init(n_best, "tukey", r=0.3),
            make_init(n_best, "tukey", r=0.7),
            make_init(n_best, "gaussian", sigma=0.2),
            make_init(n_best, "gaussian", sigma=0.4),
            make_init(n_best, "flat_top"),
            make_init(n_best, "cosine"),
        ]

        # Add perturbed best
        perturbed = best_seq + rng.standard_normal(n_best) * 0.02
        perturbed = np.maximum(perturbed, 0.0)
        perturbed = perturbed / perturbed.sum()
        inits.append(perturbed)

        time_per = max(5.0, rem / len(inits) * 0.8)
        for idx, init_a in enumerate(inits):
            rem = deadline - time.monotonic() - 60
            if rem < 10:
                break
            t_alloc = min(time_per, rem * 0.4)
            if t_alloc < 8:
                continue
            print(f"  Restart {idx}: t={t_alloc:.0f}s")
            # LSE then LP
            a_r, obj_r = logsumexp_optimize(init_a, time_limit=t_alloc * 0.6,
                                             tau_init=150, tau_max=5000,
                                             lr_init=0.006, seed=seed + idx * 777)
            rem_inner = deadline - time.monotonic() - 50
            if rem_inner > 10:
                a_r2, obj_r2 = lp_refine_aggressive(a_r,
                                                     time_limit=min(rem_inner, t_alloc * 0.35),
                                                     max_iters=15)
                if obj_r2 < obj_r:
                    a_r = a_r2
            update_best(a_r, f"restart_{idx}")

    # === Phase 7: Final polish ===
    rem = deadline - time.monotonic() - 20
    if rem > 10 and best_seq is not None:
        print(f"\nPhase 7: Final polish, t={rem:.0f}s")
        n_final = len(best_seq)
        # Quick LSE
        t1 = min(rem * 0.4, 20.0)
        if t1 > 3:
            a_f1, _ = logsumexp_optimize(best_seq, time_limit=t1,
                                          tau_init=500, tau_max=10000,
                                          lr_init=0.002, seed=seed + 99999)
            update_best(a_f1, "final_lse")
        # Quick LP
        rem = deadline - time.monotonic() - 15
        if rem > 5:
            a_f2, _ = lp_refine_aggressive(best_seq, time_limit=min(rem, 15.0),
                                            max_iters=8)
            update_best(a_f2, "final_lp")
        # Quick CD
        rem = deadline - time.monotonic() - 10
        if rem > 3:
            a_f3, _ = peak_coordinate_descent(best_seq, time_limit=min(rem, 8.0))
            update_best(a_f3, "final_cd")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        else:
            best_seq = parent.copy()
            best_score = float(evaluate_sequence(best_seq.tolist()))
    except Exception:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'=' * 60}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'=' * 60}")

    return [float(x) for x in best_seq]
