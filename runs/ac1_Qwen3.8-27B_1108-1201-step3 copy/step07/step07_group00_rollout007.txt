def mirror_descent_update(a, grad, eta):
    """Mirror descent update with entropy: a_new = a * exp(-eta * grad) / Z"""
    # Use log-space for numerical stability
    log_a = np.log(np.maximum(a, 1e-300))
    log_a_shifted = log_a - log_a.max()
    scores = log_a_shifted - eta * grad
    scores -= scores.max()
    a_new = np.exp(scores)
    s = a_new.sum()
    if s < 1e-300:
        return a.copy()
    return a_new / s

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_solve_reduced(f, n, top_k, eps=1e-8, solver='CLARABEL'):
    """
    Solve: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for top_k peaks.
    """
    b = fast_convolve(f)
    M = b.max()
    if M < 1e-15:
        return None, M
    
    if top_k < len(b):
        top_idx = np.argpartition(b, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(b))
    top_idx = np.sort(top_idx)
    K = len(top_idx)
    
    # Build constraint matrix using sparse structure
    A = np.zeros((K, n))
    for row_i in range(K):
        k = top_idx[row_i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]
    
    g = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [g >= 0, A @ g <= M])
    
    # Try different solvers
    for s in [solver, 'SCS', 'ECOS']:
        try:
            prob.solve(solver=s, max_iters=300000, eps=eps)
            break
        except Exception:
            continue
    
    if g.value is None:
        return None, M
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None, M
    return g_val, M

def coordinate_polish(f_init, time_limit):
    """Coordinate descent on individual elements."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    multipliers = np.array([0.5, 0.7, 0.85, 1.15, 1.3, 1.8])
    
    improved = True
    rounds = 0
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        rounds += 1
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-14:
                continue
            best_j_val = f[j]
            best_j_obj = best_obj
            for mult in multipliers:
                new_val = f[j] * mult
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-12:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f /= s
                if best_j_obj < best_obj - 1e-12:
                    best_obj = best_j_obj
                    improved = True
    return f, best_obj
