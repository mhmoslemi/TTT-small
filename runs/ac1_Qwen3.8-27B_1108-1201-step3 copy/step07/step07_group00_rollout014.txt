"""
LP-based optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Improvements over previous: full constraint set, ECOS solver, 
optimized coordinate descent polish, multi-length exploration, 
and random restarts.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, peak_indices, solver='ECOS'):
    """
    Solve: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for k in peak_indices.
    Returns (g, M) or (None, M).
    """
    n = len(f)
    b = fast_convolve(f)
    M = float(b.max())
    if M < 1e-15:
        return None, M

    K = len(peak_indices)
    A = np.zeros((K, n))
    for row_i in range(K):
        k = int(peak_indices[row_i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # (f*g)[k] = sum_{j=lo}^{hi} f[j] * g[k-j]
        # Column index for g is (k-j), row value is f[j]
        for j in range(lo, hi + 1):
            col = k - j
            A[row_i, col] = f[j]

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    
    try:
        prob.solve(solver=solver, eps=1e-8)
    except Exception:
        try:
            prob.solve(solver='SCS', max_iters=200000)
        except Exception:
            return None, M

    if g.value is None or np.any(np.isnan(g.value)):
        return None, M
    g_val = np.maximum(g.value, 0.0)
    return g_val, M

        if S_g <= 1.0 + 1e-5:

                    top_k = min(top_k * 2, top_k_max)
                else:

        h = g / S_g
        best_t = 0.05

        for t in [0.01, 0.02, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.5]:

        if s > 1e-10:

        if cur_obj < best_obj - 1e-15:

            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 40:

"""
Optimized LP-based approach for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements: full constraint set in LP, ECOS solver, optimized
coordinate descent polish, multi-length exploration, random restarts.

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp(f, peak_indices, solver='ECOS'):
    n = len(f)
    b = fast_convolve(f)
    M = float(b.max())
    if M < 1e-15:
        return None, M
    K = len(peak_indices)
    A = np.zeros((K, n))
    for ri in range(K):
        k = int(peak_indices[ri])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A[ri, k - j] = f[j]
    g = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [g >= 0, A @ g <= M])
    try:
        prob.solve(solver=solver, eps=1e-8)
    except Exception:
        try:
            prob.solve(solver='SCS', max_iters=200000)
        except Exception:
            return None, M
    if g.value is None or np.any(np.isnan(g.value)):
        return None, M
    return np.maximum(g.value, 0.0), M
