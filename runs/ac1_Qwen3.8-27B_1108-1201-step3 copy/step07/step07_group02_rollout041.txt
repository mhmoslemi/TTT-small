"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-accelerated L-BFGS with
soft-max smoothing, LP-based alternating initialization, and targeted peak shaving.
Key improvements over previous approach:
1. FFT-based gradient computation (O(n log n) per gradient eval)
2. Multi-stage temperature annealing (1000 -> 0.1)
3. LP alternating for strong initialization
4. Adaptive peak shaving targeting the top convolution peaks
5. Multiple sequence length exploration
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time
import cvxpy as cp


def fft_next(n):
    """Next power of 2 >= n."""
    return 1 << (n - 1).bit_length()


def fast_convolve_auto(a):
    """Linear auto-convolution of a with itself via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = fft_next(size)
    A = rfft(a, fsize)
    return irfft(A * A, fsize)[:size]


def cross_corr_fft(w, a):
    """
    Compute cross-correlation (w ★ a)[i] = sum_k w[k] * a[k-i]
    for i = 0, ..., n-1, where w has length 2n-1 and a has length n.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    mw = len(w)
    fsize = fft_next(mw + n - 1)
    W = rfft(w, fsize)
    A = rfft(a, fsize)
    corr = irfft(W * np.conj(A), fsize)
    return corr[:n]


def soft_max_obj_and_grad_z(z, T):
    """
    Compute soft-max approximated score and its gradient w.r.t. z (log-space).
    a = softmax(z), score ≈ (2n/T) * logsumexp(T * conv(a,a))
    Returns (score, grad_z).
    """
    n = len(z)
    # Softmax
    z_shift = z - z.max()
    exp_z = np.exp(z_shift)
    a = exp_z / exp_z.sum()
    
    # Convolution
    b = fast_convolve_auto(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    score = (2.0 * n / T) * log_sum_exp
    
    # Weights
    w = exp_Tb / exp_Tb.sum()
    
    # Gradient w.r.t. a: (2/T) * cross_corr(w, a)
    grad_a = (2.0 / T) * cross_corr_fft(w, a)
    
    # Gradient w.r.t. z: a * (grad_a - <grad_a, a>)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return score, grad_z


def lbgfs_optimize(a0, time_limit, T_stages=None, maxiter_total=10000):
    """
    L-BFGS optimization with multi-stage temperature annealing.
    a0: initial probability vector (sum=1, non-negative)
    time_limit: seconds
    T_stages: list of temperatures to try (high to low)
    """
    if T_stages is None:
        T_stages = [500.0, 100.0, 20.0, 5.0, 1.0, 0.3, 0.1]
    
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-300)
    a = a / a.sum()
    z = np.log(a)
    t0 = time.monotonic()
    best_score = 2.0 * n * fast_convolve_auto(a).max()
    best_a = a.copy()
    
    n_stages = len(T_stages)
    iters_per_stage = max(100, maxiter_total // n_stages)
    
    for stage_idx, T_val in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        # Define objective and gradient for this temperature
        def func(z_in):
            sc, _ = soft_max_obj_and_grad_z(z_in, T_val)
            return sc
        
        def jac(z_in):
            _, gz = soft_max_obj_and_grad_z(z_in, T_val)
            return gz
        
        try:
            res = minimize(
                fun=func,
                x0=z,
                jac=jac,
                method='L-BFGS-B',
                options={
                    'maxiter': iters_per_stage,
                    'ftol': 1e-18,
                    'gtol': 1e-12,
                    'maxfun': iters_per_stage * 100,
                    'disp': False
                }
            )
            z = res.x
        except Exception as e:
            print(f"    L-BFGS stage T={T_val} error: {e}")
            break
        
        # Evaluate true objective
        a_new = softmax_vec(z)
        sc = 2.0 * n * fast_convolve_auto(a_new).max()
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        print(f"    L-BFGS T={T_val:.1f}: score={best_score:.6f} (iter {stage_idx+1}/{n_stages})")
    
    return best_a, best_score


def softmax_vec(z):
    """Compute softmax of vector z."""
    z_shift = z - z.max()
    exp_z = np.exp(z_shift)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def build_conv_matrix(a):
    """Build matrix T such that T @ b = conv(a, b). T has shape (2n-1, n)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T


def solve_lp_best_response(a_fixed, solver=None):
    """
    Given fixed a (non-negative, sum=1), find b >= 0, sum(b)=1
    that minimizes max_k (conv(a,b))[k].
    Returns (b, M) or (None, None) on failure.
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def get_lp_solver():
    """Find a working LP solver."""
    if '_lp_solver_cached' in globals():
        return _lp_solver_cached
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return _lp_solver_cached


def lp_alternating(a0, time_limit, solver=None, n_iters=2000):
    """
    Alternating LP: fix a, optimize b via LP; fix b, optimize a via LP.
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    a = a / a.sum()
    b = a.copy()
    t0 = time.monotonic()
    best_score = 2.0 * n * fast_convolve_auto(a).max()
    best_a = a.copy()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Optimize b given a
        b_new, M_ab = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new
        
        # Optimize a given b
        a_new, M_ba = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new
        
        # Evaluate
        sc = 2.0 * n * fast_convolve_auto(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        if (it + 1) % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Alt iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score


def peak_shave(a0, time_limit, n_steps=50000):
    """
    Targeted peak shaving: identify top peaks in auto-convolution
    and reduce them via careful coordinate adjustments.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve_auto(a).max()
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve_auto(a)
        # Find the peak position (skip endpoints which are always smaller)
        peak_k = int(np.argmax(b))
        
        # Identify contributing indices
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        # Sort by contribution (descending)
        order = np.argsort(contribs)[::-1]
        
        improved = False
        # Try reducing top contributors
        for ci in order[:8]:
            i = idx[ci]
            if a[i] < 1e-15:
                continue
            # Try different reduction factors
            for factor in [0.99, 0.98, 0.97, 0.95, 0.93, 0.90]:
                new_val = a[i] * factor
                diff = a[i] - new_val
                if diff < 1e-16:
                    continue
                # Redistribute mass to other coordinates proportionally
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a[mask].copy()
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                a_trial[mask] = others + diff * others / others_sum
                a_trial = a_trial / a_trial.sum()
                sc = 2.0 * n * fast_convolve_auto(a_trial).max()
                if sc < best_score - 1e-12:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        # Try reducing pairs of contributors
        if not improved and len(order) >= 2:
            for ci1 in order[:5]:
                for ci2 in order[:5]:
                    if ci1 == ci2:
                        continue
                    i = idx[ci1]
                    j = idx[ci2]
                    if a[i] < 1e-15 or a[j] < 1e-15:
                        continue
                    for f1 in [0.98, 0.95]:
                        for f2 in [0.98, 0.95]:
                            a_trial = a.copy()
                            a_trial[i] *= f1
                            a_trial[j] *= f2
                            a_trial = a_trial / a_trial.sum()
                            sc = 2.0 * n * fast_convolve_auto(a_trial).max()
                            if sc < best_score - 1e-12:
                                a = a_trial
                                best_score = sc
                                best_a = a.copy()
                                improved = True
                                break
                        if improved:
                            break
                    if improved:
                        break
                if improved:
                    break
        
        if not improved:
            # Try reducing top-3 simultaneously
            a_trial = a.copy()
            for ci in order[:3]:
                a_trial[idx[ci]] *= 0.97
            a_trial = a_trial / a_trial.sum()
            sc = 2.0 * n * fast_convolve_auto(a_trial).max()
            if sc < best_score - 1e-12:
                a = a_trial
                best_score = sc
                best_a = a.copy()
            else:
                # No improvement, break to avoid infinite loop
                break
    
    return best_a, best_score

def interpolate_to_length(a, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0)
    ss = a_interp.sum()
    if ss > 1e-15:
        a_interp = a_interp / ss

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-accelerated L-BFGS with
soft-max smoothing, LP-based alternating initialization, and targeted peak shaving.
Key improvements: FFT-based O(n log n) gradients, multi-stage temperature annealing,
efficient LP warm-start, adaptive peak shaving, and multi-length exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time
import cvxpy as cp


def fft_next(n):
    """Next power of 2 >= n."""
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def fast_convolve_auto(a):
    """Linear auto-convolution of a with itself via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fsize = fft_next(size)
    A = rfft(a, fsize)
    return irfft(A * A, fsize)[:size]


def cross_corr_fft(w, a):
    """
    Compute (w * a)[i] = sum_k w[k] * a[k-i] for i=0..n-1
    where w has length 2n-1 and a has length n. Uses FFT.
    """
    n = len(a)
    mw = len(w)
    fsize = fft_next(mw + n - 1)
    W = rfft(w, fsize)
    A = rfft(a, fsize)
    corr = irfft(W * np.conj(A), fsize)
    return corr[:n]


def softmax_vec(z):
    """Compute softmax of vector z."""
    z_shift = z - z.max()
    exp_z = np.exp(z_shift)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def obj_func(z, T):
    """Soft-max approximated score in log-space. score ~= (2n/T)*logsumexp(T*conv(softmax(z),softmax(z)))"""
    n = len(z)
    a = softmax_vec(z)
    b = fast_convolve_auto(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    return (2.0 * n / T) * log_sum_exp


def obj_jac(z, T):
    """Gradient of soft-max score w.r.t. log-space variables z."""
    n = len(z)
    a = softmax_vec(z)
    b = fast_convolve_auto(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    grad_a = (2.0 / T) * cross_corr_fft(w, a)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return grad_z


def true_score(a):
    """Compute the true score 2n*max(conv(a,a)) for sum(a)=1."""
    n = len(a)
    b = fast_convolve_auto(a)
    return 2.0 * n * b.max()

def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    if '_lp_solver_cached' in globals():
        return _lp_solver_cached
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = 'ECOS'
    return _lp_solver_cached


def solve_lp_best_response(a_fixed, solver):
    """
    Given fixed a (non-negative, sum=1), find b>=0, sum(b)=1 minimizing max conv(a,b).
    Returns (b, M) or (None, None).
    """
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def peak_shave(a0, time_limit, n_steps):
    """Targeted peak shaving: reduce top convolution peaks via coordinate moves."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = true_score(a)
    t0 = time.monotonic()
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > 200:
            break

        b = fast_convolve_auto(a)
        peak_k = int(np.argmax(b))
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        order = np.argsort(contribs)[::-1]

        improved = False
        for ci in order[:10]:
            i = idx[ci]
            if a[i] < 1e-15:
                continue
            for factor in [0.995, 0.99, 0.985, 0.98, 0.97, 0.96, 0.94]:
                new_val = a[i] * factor
                diff = a[i] - new_val
                if diff < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a[mask].copy()
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                a_trial[mask] = others + diff * others / others_sum
                a_trial = a_trial / a_trial.sum()
                sc = true_score(a_trial)
                if sc < best_score - 1e-12:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved and len(order) >= 2:
            for ci1 in order[:4]:
                for ci2 in order[:4]:
                    if ci1 == ci2:
                        continue
                    i = idx[ci1]
                    j = idx[ci2]
                    if i == j or a[i] < 1e-15 or a[j] < 1e-15:
                        continue
                    for f1 in [0.99, 0.98]:
                        for f2 in [0.99, 0.98]:
                            a_trial = a.copy()
                            a_trial[i] *= f1
                            a_trial[j] *= f2
                            a_trial = a_trial / a_trial.sum()
                            sc = true_score(a_trial)
                            if sc < best_score - 1e-12:
                                a = a_trial
                                best_score = sc
                                best_a = a.copy()
                                improved = True
                                break
                        if improved:
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            a_trial = a.copy()
            for ci in order[:3]:
                a_trial[idx[ci]] *= 0.98
            a_trial = a_trial / a_trial.sum()
            sc = true_score(a_trial)
            if sc < best_score - 1e-12:
                a = a_trial
                best_score = sc
                best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
        else:
            no_improve = 0

    return best_a, best_score

def interpolate_to_length(a, new_n):
    """Interpolate a sequence to a new length, preserving sum=1."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0)
    ss = a_interp.sum()
    if ss > 1e-15:
        a_interp = a_interp / ss
    return a_interp

            init_alt = interpolate_to_length(best_seq, alt_n)
            T_stages_alt = [100.0, 20.0, 5.0, 1.0, 0.3]
            a_alt, _ = lbgfs_optimize(init_alt, t_alloc, T_stages_alt, maxiter_total=8000)
            try_update(a_alt, f"alt_n{alt_n}")

    # === Phase 7: Final polish cycle ===
    rem = deadline - time.monotonic()
    if rem > 30:
        print(f"\n=== Phase 7: Final Polish ===")
        t1 = min(rem * 0.4, 80.0)
        if t1 > 10:
            T_stages_final = [0.5, 0.2, 0.1, 0.05]
            a_fin, _ = lbgfs_optimize(best_seq, t1, T_stages_final, maxiter_total=8000)
            try_update(a_fin, "final_lbgfs")

        rem = deadline - time.monotonic()
        t2 = min(rem * 0.4, 60.0)
        if t2 > 10:
            a_ps2, _ = peak_shave(best_seq, t2, n_steps=50000)
            try_update(a_ps2, "final_shave")

        rem = deadline - time.monotonic()
        t3 = min(rem * 0.5, 40.0)
        if t3 > 5:
            a_final, _ = lbgfs_optimize(best_seq, t3, [0.1, 0.05, 0.03], maxiter_total=5000)
            try_update(a_final, "ultra_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL: score={best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
