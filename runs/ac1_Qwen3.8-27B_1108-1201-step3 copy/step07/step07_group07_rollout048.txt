"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based peak shaving with
quadratic line search, FFT-accelerated subgradient descent, and coordinate
descent polish. Key innovation: proper quadratic line search after each LP
direction, exploiting the fact that conv((1-t)a+tg, (1-t)a+tg) is a set of
quadratics in t whose max can be evaluated in O(n) after O(n log n) setup.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
import time


def fast_auto_convolve(a):
    """Linear auto-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def conv_cross(a, b):
    """Cross-convolution of a and b via FFT. Returns array of length len(a)+len(b)-1."""
    na, nb = len(a), len(b)
    size = na + nb - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def eval_max_b(a):
    """Return max of auto-convolution for a (assumed normalized to sum=1)."""
    return float(fast_auto_convolve(a).max())


def quadratic_line_search(a, g, nugget=0.0):
    """
    Find best t in [0,1] minimizing max_k conv((1-t)a + t*g, (1-t)a + t*g).
    Uses the fact that each component is a quadratic in t.
    Returns (best_t, best_val).
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (3 * n - 2).bit_length()

    A = rfft(a, fft_size)
    G = rfft(g, fft_size)

    # conv(a,a), conv(a,g), conv(g,g)
    conv_aa = irfft(A * A, fft_size)[:size]
    conv_ag = irfft(A * G, fft_size)[:size]
    conv_gg = irfft(G * G, fft_size)[:size]

    # q_k(t) = (1-t)^2 * conv_aa[k] + 2*t*(1-t) * conv_ag[k] + t^2 * conv_gg[k]
    #        = conv_aa[k] + t*(2*conv_ag[k] - 2*conv_aa[k]) + t^2*(conv_aa[k] - 2*conv_ag[k] + conv_gg[k])
    c0 = conv_aa  # constant
    c1 = 2.0 * (conv_ag - conv_aa)  # linear coeff
    c2 = conv_aa - 2.0 * conv_ag + conv_gg  # quadratic coeff

    # Evaluate max of quadratics at candidate t values
    def f_of_t(t):
        return np.max(c0 + t * c1 + t * t * c2)

    # First try: uniform grid
    t_candidates = np.linspace(0, 1, 41)
    vals = np.array([f_of_t(t) for t in t_candidates])
    best_t = t_candidates[np.argmin(vals)]
    best_val = vals.min()

    # Refine around best with golden section
    if 0 < best_t < 1:
        # Determine bracket
        idx = np.argmin(vals)
        if idx > 0 and idx < len(t_candidates) - 1:
            lo = t_candidates[idx - 1]
            hi = t_candidates[idx + 1]
            # Golden section search
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            for _ in range(50):
                m1 = hi - gr * (hi - lo)
                m2 = lo + gr * (hi - lo)
                f1 = f_of_t(m1)
                f2 = f_of_t(m2)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
                if hi - lo < 1e-10:
                    break
            t_mid = (lo + hi) / 2.0
            f_mid = f_of_t(t_mid)
            if f_mid < best_val:
                best_t = t_mid
                best_val = f_mid

    return best_t, best_val


def lp_direction(f, M, warp=None, method='highs'):
    """
    Solve LP: max sum(g) s.t. conv(f,g) <= M*warp[k], g >= 0.
    If warp is None, uses flat constraint conv(f,g) <= M.
    Returns g (the LP solution) or None if infeasible.
    """
    n = len(f)
    num_constraints = 2 * n - 1

    # Build constraint matrix (sparse)
    row_indices = []
    col_indices = []
    data = []

    for k in range(num_constraints):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            row_indices.append(k)
            col_indices.append(k - j)
            data.append(f[j])

    A_ub = lil_matrix((num_constraints, n))
    for idx in range(len(row_indices)):
        A_ub[row_indices[idx], col_indices[idx]] = data[idx]

    if warp is None:
        b_ub = np.full(num_constraints, M)
    else:
        b_ub = M * np.array(warp, dtype=np.float64)

    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      bounds=[(0.0, None)] * n, method=method)
        if res.success:
            return res.x
        # Try with slightly relaxed M
        res = linprog(c, A_ub=A_ub, b_ub=b_ub * 1.001,
                      bounds=[(0.0, None)] * n, method=method)
        if res.success and res.x is not None and res.x.min() >= -1e-10:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_optimize_with_les(f0, time_limit, max_iters=200, verbose=False):
    """
    LP-based optimization with quadratic line search.
    Iterates: solve LP direction, do line search, update.
    Returns (best_a, best_max_b) where a is normalized to sum=1.
    """
    t0 = time.monotonic()
    n = len(f0)
    a = f0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = eval_max_b(a)
    no_improve = 0

    for iteration in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_current = fast_auto_convolve(a)
        M = b_current.max()

        # Solve LP
        g = lp_direction(a, M)
        if g is None or g.sum() < 1e-15:
            # Try with slightly relaxed M
            g = lp_direction(a, M * 1.01)
            if g is None or g.sum() < 1e-15:
                break
            g = np.maximum(g, 0.0)

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum  # Normalize g to sum=1

        # Quadratic line search
        best_t, best_val = quadratic_line_search(a, g)

        if best_t > 1e-8:
            new_a = (1.0 - best_t) * a + best_t * g
            new_a = np.maximum(new_a, 0.0)
            ns = new_a.sum()
            if ns > 1e-15:
                new_a = new_a / ns
                new_obj = eval_max_b(new_a)
                if new_obj < best_obj - 1e-14:
                    a = new_a
                    if new_obj < best_obj:
                        best_obj = new_obj
                        best_a = a.copy()
                        no_improve = 0
                        if verbose and iteration % 10 == 0:
                            print(f"    LP-LS iter {iteration}: t={best_t:.4f}, max_b={best_obj:.10f}")
                else:
                    no_improve += 1
            else:
                break
        else:
            break

        if no_improve >= 20:
            # Try with warped constraints
            g = lp_direction(a, M, warp=_make_warp(b_current, M))
            if g is not None and g.sum() > 1e-15:
                g = g / g.sum()
                best_t, best_val = quadratic_line_search(a, g)
                if best_t > 1e-8:
                    new_a = (1.0 - best_t) * a + best_t * g
                    new_a = np.maximum(new_a, 0.0)
                    ns = new_a.sum()
                    if ns > 1e-15:
                        new_a = new_a / ns
                        new_obj = eval_max_b(new_a)
                        if new_obj < best_obj - 1e-14:
                            a = new_a
                            if new_obj < best_obj:
                                best_obj = new_obj
                                best_a = a.copy()
                                no_improve = 0
                                if verbose:
                                    print(f"    LP-WARP iter {iteration}: t={best_t:.4f}, max_b={best_obj:.10f}")
            else:
                break
            if no_improve >= 30:
                break

    return best_a, best_obj


def _make_warp(b_current, M):
    """Create a warp vector that gives more room at non-peak positions."""
    # Give positions close to full room, peak positions tighter
    ratio = b_current / M if M > 0 else np.ones_like(b_current)
    # Warp: positions with ratio < 1 get more budget (1.0), peak gets exactly M
    # This encourages the LP to use non-peak budget
    warp = np.ones_like(ratio)
    warp[ratio < 0.95] = 1.0 / ratio[ratio < 0.95] * 0.5 + 0.5  # more room
    # Actually, let's just use a simple approach: uniform warp = 1
    # But give the top few peaks slightly less than M
    top_k = max(5, len(ratio) // 100)
    top_idx = np.argsort(ratio)[-top_k:]
    warp[top_idx] = 0.99
    return warp


def subgradient_optimize(a0, time_limit, lr_init=0.003, momentum=0.93,
                         n_top=None, seed=42):
    """
    FFT-accelerated projected subgradient with momentum.
    Minimizes max(conv(a,a)) subject to a >= 0, sum(a) = 1.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    v = np.zeros(n)
    best_a = a.copy()
    best_obj = eval_max_b(a)
    cur_lr = lr_init
    no_improve = 0
    it_count = 0

    if n_top is None:
        n_top = max(10, n // 15)

    while time.monotonic() - t0 < time_limit:
        b = fast_auto_convolve(a)
        b_max = b.max()

        # Select active peaks
        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
        else:
            top_idx = np.arange(len(b))

        threshold = b_max * 0.995
        top_vals = b[top_idx]
        active_mask = top_vals >= threshold
        if not active_mask.any():
            active_mask[0] = True

        w = np.zeros(len(b))
        active_indices = top_idx[active_mask]
        w[active_indices] = 1.0 / len(active_indices)

        # Subgradient: g[i] = sum_j a[j] * w[i+j]
        # Compute via FFT
        fft_size = 1 << (3 * n - 2).bit_length()
        A = rfft(a, fft_size)
        W = rfft(w, fft_size)
        g = irfft(np.conj(A) * W, fft_size)[:n]

        # Nesterov momentum
        v = momentum * v - cur_lr * g
        a_new = project_simplex(a + v)

        obj_new = eval_max_b(a_new)
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        it_count += 1

        if no_improve >= 2000:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.5, lr_init * 0.05)
            no_improve = 0
        elif no_improve >= 1000:
            cur_lr *= 0.95
        elif no_improve >= 500:
            cur_lr *= 0.98

    return best_a, best_obj


def coordinate_descent(a0, time_limit, multipliers=None):
    """
    Coordinate descent on the simplex.
    """
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = eval_max_b(a)

    if multipliers is None:
        multipliers = np.array([0.3, 0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0, 3.0])

    improved = True
    passes = 0
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        passes += 1
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-15:
                continue
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0.0 or new_val > 0.9999:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < -0.01 or scale > 50.0:
                    continue
                a_tr = a.copy()
                a_tr[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_tr[mask] *= scale
                ss = a_tr.sum()
                if ss < 1e-15:
                    continue
                a_tr = a_tr / ss
                val = eval_max_b(a_tr)
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

    return best_a, best_obj


def make_initialization(n, init_type='gaussian', sigma=0.25):
    """Create various initializations."""
    x = np.linspace(0, 1, n)
    if init_type == 'gaussian':
        a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    elif init_type == 'cosine':
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    elif init_type == 'tukey':
        a = np.ones(n)
        taper = max(1, int(0.4 * n // 2))
        if taper < n:
            t = np.linspace(0, 1, taper)
            a[:taper] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t / 0.4 - 1.0)))
            a[-taper:] = a[:taper][::-1]
    elif init_type == 'triangular':
        half = n // 2
        a = np.concatenate([np.linspace(0.05, 1.0, half), np.linspace(1.0, 0.05, n - half)])
    elif init_type == 'ramp':
        a = np.linspace(0.05, 1.0, n)
    elif init_type == 'bump':
        a = np.exp(-5.0 * (x - 0.4) ** 2) + 0.3 * np.exp(-5.0 * (x - 0.7) ** 2)
    else:
        a = np.ones(n)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length, preserving normalization."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp

        rem = deadline - time.monotonic() - 200
        if rem < 30:
            break
        f_interp = interpolate_to_length(best_seq, target_n)
        t_alloc = min(rem * 0.3, 80.0)
        if t_alloc < 15:
            continue
        print(f"  n={target_n}, t={t_alloc:.0f}s")

        # LP on interpolated
        a_len, obj_len = lp_optimize_with_les(f_interp, time_limit=t_alloc, max_iters=100, verbose=False)
        # Brief SG
        rem_sub = deadline - time.monotonic() - 150
        if rem_sub > 15:
            a_len_sg, _ = subgradient_optimize(
                a_len, time_limit=min(rem_sub * 0.5, 30.0),
                lr_init=0.002, momentum=0.94,
                n_top=max(10, target_n // 15), seed=seed + target_n
            )
            a_len = a_len_sg
        update_best(a_len, f"len_{target_n}")

    # === Phase 5: Deep LP on best ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Deep LP (n={len(best_seq)})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic() - 150
    if rem > 30:
        t_deep = min(rem * 0.6, 250.0)
        print(f"  Deep LP time: {t_deep:.0f}s")
        a_deep, obj_deep = lp_optimize_with_les(best_seq, time_limit=t_deep, max_iters=300, verbose=True)
        update_best(a_deep, "deep_lp")

    # === Phase 6: Coordinate descent ===
    print(f"\n{'='*60}")
    print(f"Phase 6: Coordinate Descent")
    print(f"{'='*60}")
    rem = deadline - time.monotonic() - 50
    if rem > 10:
        t_cd = min(rem, 80.0)
        t_cd = max(t_cd, 5.0)
        print(f"  CD time: {t_cd:.0f}s")
        a_cd, obj_cd = coordinate_descent(best_seq, time_limit=t_cd)
        update_best(a_cd, "cd")

    # === Phase 7: Final micro-polish ===
    print(f"\n{'='*60}")
    print(f"Phase 7: Final polish")
    print(f"{'='*60}")
    rem = deadline - time.monotonic() - 10
    if rem > 10:
        # Tiny LP
        t_final = min(rem * 0.6, 30.0)
        a_fin, obj_fin = lp_optimize_with_les(best_seq, time_limit=t_final, max_iters=50, verbose=False)
        update_best(a_fin, "final_lp")

        rem = deadline - time.monotonic() - 5
        if rem > 5:
            # Tiny CD
            a_fin_cd, obj_fin_cd = coordinate_descent(
                best_seq, time_limit=min(rem, 15.0),
                multipliers=np.array([0.8, 0.9, 0.95, 1.05, 1.1, 1.2])
            )
            update_best(a_fin_cd, "final_cd")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
