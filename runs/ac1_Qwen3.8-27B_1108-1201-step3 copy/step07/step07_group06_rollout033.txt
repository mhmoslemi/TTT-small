    a = a / a.sum()
    eta = eta_init
    best_a = a.copy()
    best_score = 2*n*fast_convolve(a).max()
    
    while time remaining:
        b = fast_convolve(a)
        k = argmax(b)
        
        # Gradient of b[k] w.r.t. a
        # grad[i] = 2*a[k-i] for valid i
        grad = np.zeros(n)
        # For i in range(n): if 0 <= k-i < n: grad[i] = 2*a[k-i]
        lo = max(0, k-n+1)
        hi = min(n-1, k)
        grad[lo:hi+1] = 2*a[k-lo:k-hi-1:-1]  # a[k-i] for i from lo to hi
        
        # Multiplicative update

        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        eta *= eta_decay
    
    return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Variance reduction via smooth gradient descent on ||conv(a,a)||_2^2
2. Peak-targeting subgradient method for direct max reduction
3. Local search polishing with coordinate descent
Uses FFT-based convolution for efficiency and multiplicative updates
for natural non-negativity maintenance.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def grad_variance(a):
    """
    Gradient of sum_k (conv(a,a)[k])^2 w.r.t. a.
    Result: 4 * ifft(conj(A) * A^2)
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = A * A
    G = irfft(np.conj(A) * B, fft_size)[:n]
    return 4.0 * G


def peak_gradient(a, k, n, fft_size):
    """
    Gradient of b[k] = conv(a,a)[k] w.r.t. a.
    grad[i] = 2*a[k-i] if 0 <= k-i < n, else 0.
    """
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    # For i in [lo, hi]: grad[i] = 2*a[k-i]
    # k-i ranges from k-lo to k-hi
    grad[lo:hi+1] = 2.0 * a[k-hi:k-lo+1][::-1]
    return grad


def compute_score(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def variance_phase(a0, time_limit, eta_init=0.005, eta_decay=0.99995):
    """
    Phase 1: Reduce variance of autocorrelation via smooth gradient descent.
    Minimizes ||conv(a,a)||_2^2 using multiplicative updates.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    a = a / a.sum()
    eta = eta_init
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 3).bit_length()
    no_improve = 0
    patience = 5000
    
    while time.monotonic() - t0 < time_limit:
        # Compute gradient of variance
        G = grad_variance(a)
        
        # Normalize gradient for stable updates
        G_norm = np.linalg.norm(G)
        if G_norm < 1e-15:
            break
        G = G / G_norm
        
        # Multiplicative update
        a = a * np.exp(-eta * G * G_norm)  # Scale back for consistent step
        a = a / a.sum()
        
        # Evaluate
        score = compute_score(a)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        eta *= eta_decay
        
        if no_improve > patience:
            # Reset with small perturbation
            a = best_a * (1.0 + 0.001 * np.random.standard_normal(n))
            a = np.maximum(a, 1e-300)
            a = a / a.sum()
            eta = eta_init * 0.5
            no_improve = 0
    
    return best_a, best_score


def peak_phase(a0, time_limit, top_k=5, eta_init=0.002):
    """
    Phase 2: Direct peak reduction using subgradient method.
    Targets the top-k peaks with multiplicative updates.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    a = a / a.sum()
    eta = eta_init
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    no_improve = 0
    patience = 3000
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        max_b = b.max()
        
        # Find top-k peaks
        top_indices = np.argsort(b)[::-1][:top_k]
        
        # Weight peaks by their value (soft-max with small T)
        T = max_b * 0.1
        weights = np.exp(b[top_indices] / T - 1.0)
        weights = weights / weights.sum()
        
        # Compute weighted subgradient
        grad = np.zeros(n)
        for w, k in zip(weights, top_indices):
            grad += w * peak_gradient(a, int(k), n, None)
        
        # Multiplicative update
        a = a * np.exp(-eta * grad)
        a = np.maximum(a, 1e-300)
        a = a / a.sum()
        
        # Evaluate
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        eta *= 0.99999
        
        if no_improve > patience:
            a = best_a * (1.0 + 0.005 * np.random.standard_normal(n))
            a = np.maximum(a, 1e-300)
            a = a / a.sum()
            eta = eta_init * 0.7
            no_improve = 0
    
    return best_a, best_score


def local_search(a0, time_limit, n_trials=10000):
    """
    Phase 3: Local search - try individual element modifications.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        # Find the peak
        k_peak = int(np.argmax(b))
        
        # Find top contributing elements to this peak
        lo = max(0, k_peak - n + 1)
        hi = min(n - 1, k_peak)
        idx_range = np.arange(lo, hi + 1)
        contribs = a[idx_range] * a[k_peak - idx_range]
        
        # Pick a random element from the top contributors
        top_n = min(10, len(idx_range))
        top_idx = np.argsort(contribs)[::-1][:top_n]
        
        if len(top_idx) == 0:
            continue
            
        ci = top_idx[rng.integers(len(top_idx))]
        i = int(idx_range[ci])
        j = int(k_peak - i)
        
        # Try to reduce a[i] and redistribute
        for factor in [0.95, 0.9, 0.85, 0.8]:
            a_trial = a.copy()
            diff = a_trial[i] * (1.0 - factor)
            a_trial[i] *= factor
            
            # Redistribute to random other element
            j_redist = int(rng.integers(n))
            if j_redist == i:
                j_redist = (j_redist + 1) % n
            a_trial[j_redist] += diff
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            
            score = compute_score(a_trial)
            if score < best_score - 1e-14:
                a = a_trial
                best_score = score
                best_a = a_trial.copy()
                break
        else:
            # Try paired modification: reduce a[i], increase a[j] (the partner)
            if j != i and 0 <= j < n:
                for f1 in [0.9, 0.85]:
                    for f2 in [1.01, 1.02, 1.05]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        s = a_trial.sum()
                        if s < 1e-15:
                            continue
                        a_trial = a_trial / s
                        score = compute_score(a_trial)
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a_trial.copy()
                            break
                    else:
                        continue
                    break
    
    return best_a, best_score


def coordinate_descent(a0, time_limit):
    """
    Coordinate descent: for each element, find the best multiplicative factor.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    
    sweep = 0
    while time.monotonic() - t0 < time_limit:
        sweep += 1
        order = rng.permutation(n)
        improved = False
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                return best_a, best_score
            if a[i] < 1e-20:
                continue
            
            best_factor = 1.0
            for factor in [0.8, 0.9, 0.95, 1.05, 1.1, 1.2]:
                a_trial = a.copy()
                a_trial[i] *= factor
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                score = compute_score(a_trial)
                if score < best_score - 1e-14:
                    best_score = score
                    best_factor = factor
                    best_a = a_trial.copy()
            
            if best_factor != 1.0:
                a = best_a.copy()
                improved = True
        
        if not improved and sweep > 3:
            break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
