"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive mirror descent
with Nesterov acceleration on the simplex, diverse structured initializations
exploiting sequence asymmetry (monotone, sawtooth), adaptive temperature
softmax smoothing, and L-BFGS + coordinate descent refinement.

2. Adaptive temperature based on peak concentration (Hartley entropy)
3. Structured asymmetry initializations that reduce center peak
4. Nesterov momentum with adaptive step size
5. Peak-aware gradient: weights peaks by their excess over the mean
6. Systematic multi-length search with interpolation

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the optimization score for a normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-30)
    return np.log(a_safe)


def peak_aware_gradient(a, T, n):
    """
    Compute gradient of T * logsumexp(conv(a,a)/T) w.r.t. a.
    Uses peak-aware weighting: peaks that exceed the mean get extra weight.
    """
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    w = np.exp(shifted - max_shifted)
    w = w / w.sum()
    
    # Gradient: d/da_i [sum_k w_k * conv[k]] = 2 * sum_k w_k * a[k-i] (cross-corr)
    # cross-correlation of w with a: grad[i] = 2 * sum_k w[k] * a[k-i]
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
    return (2.0 * n / T) * grad_a, conv, w


def mirror_descent_nesterov(a0, time_limit, T_init=200, T_min=10,
                            eta_init=0.5, momentum=0.9, verbose=False):
    """
    Mirror descent with Nesterov momentum and adaptive temperature.
    Works directly on the simplex (a >= 0, sum(a) = 1).
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score(a0)
    a = a / s
    
    best_a = a.copy()
    best_score = compute_score(a)
    v = np.zeros(n)  # momentum velocity in log-space
    t0 = time.monotonic()
    total_time = time_limit
    T = T_init
    eta = eta_init
    iters = 0
    stall_count = 0
    last_improve = 0.0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        elapsed = time.monotonic() - t0
        progress = elapsed / total_time
        
        # Adaptive temperature: decrease over time, but keep some
        T = max(T_min, T_init * (T_min / T_init) ** (progress ** 0.5))
        
        # Nesterov lookahead in log-space
        z = np.log(np.maximum(a, 1e-30))
        z_look = z - momentum * v
        a_look = softmax_from_z(z_look)
        
        # Gradient at lookahead point
        grad, conv, w = peak_aware_gradient(a_look, T, n)
        
        # Update velocity
        v = momentum * v + grad
        
        # Mirror descent step in log-space (equivalent to multiplicative updates)
        z_new = z - eta * v
        
        # Clip to prevent explosion
        z_new = np.clip(z_new, -30, 30)
        
        a_new = softmax_from_z(z_new)
        score = compute_score(a_new)
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            last_improve = iters
        else:
            # Adaptive step size: reduce if no improvement
            a = a_new  # still move (helps escape)
            if iters - last_improve > 500:
                stall_count += 1
                eta *= 0.95  # reduce step size
                if stall_count > 10:
                    # Perturb to escape
                    z_pert = z + np.random.standard_normal(n) * 0.1
                    a = softmax_from_z(z_pert)
                    stall_count = 0
                    last_improve = iters
        
        # Adaptive eta: based on gradient magnitude
        grad_mag = np.mean(np.abs(grad))
        if grad_mag > 1e-10:
            eta_target = np.sqrt(n) / (T * grad_mag * n) * 0.5
            eta = 0.9 * eta + 0.1 * min(eta_target, 2.0)
        
        if verbose and iters % 5000 == 0:
            print(f"    MD iter={iters}, T={T:.1f}, eta={eta:.4f}, score={best_score:.8f}")

    return best_a, best_score


def lbgfs_optimize(a0, time_limit, T, maxiter=3000):
    """L-BFGS optimization in log-space with soft-max smoothing."""
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    result = {'a': a0.copy(), 'score': compute_score(a0)}

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = softmax_from_z(z)
        conv = fast_convolve(a)
        shifted = conv / T
        max_shifted = shifted.max()
        log_sum_exp = max_shifted + np.log(np.exp(shifted - max_shifted).sum())
        return (2.0 * n / T) * np.exp(T * log_sum_exp)

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        a = softmax_from_z(z)
        conv = fast_convolve(a)
        shifted = conv / T
        max_shifted = shifted.max()
        w = np.exp(shifted - max_shifted)
        w = w / w.sum()
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)
        return (2.0 * n / T) * grad_z

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-13,
                     'maxfun': maxiter * 40, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
        score = compute_score(a_out)
        if score < result['score']:
            result['a'] = a_out
            result['score'] = score
    except Exception:
        pass

    result['seconds'] = time.monotonic() - t0
    return result

    """Fast coordinate descent with pairwise moves and local search."""
    n = len(a0)

        improved = False

            if time.monotonic() - t0 > time_limit:
                break
            
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue

                a = best_a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
        else:

    return best_a, best_score


def generate_monotone(n, direction=1, curvature=1.0, rng=None):
    """Generate a monotone sequence with optional curvature."""
    if rng is None:
        rng = np.random.default_rng(0)
    x = np.linspace(0, 1, n)
    if direction > 0:
        a = x ** curvature
    else:
        a = (1 - x) ** curvature
    a = a + 0.01 * rng.standard_normal(n)
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def generate_sawtooth(n, periods=1, amplitude=0.3, rng=None):
    """Generate a sawtooth-like sequence."""
    if rng is None:
        rng = np.random.default_rng(1)
    x = np.linspace(0, 1, n)
    a = np.ones(n)
    a += amplitude * (2 * (periods * x - np.floor(periods * x + 0.5)))
    a = a + 0.01 * rng.standard_normal(n)
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def generate_asymmetric(n, skew=0.5, rng=None):
    """Generate a sequence with controlled asymmetry."""
    if rng is None:
        rng = np.random.default_rng(2)
    x = np.linspace(0, 1, n)
    # Skewed distribution
    a = (x + skew) ** 2
    a = a + 0.02 * rng.standard_normal(n)
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        t_alloc = min(rem * 0.7, 40.0)
        a_md2, _ = mirror_descent_nesterov(best_seq, t_alloc, T_init=100, T_min=5,
                                           eta_init=0.2, momentum=0.95)
        best_seq, best_score = check_and_update(a_md2, "final_md", best_seq, best_score)

    # Fallback
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
