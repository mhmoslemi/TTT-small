```python
import numpy as np

from scipy.sparse import csr_matrix
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    A[k, j] = f[j] if max(0,k-n+1) <= j <= min(n-1,k), else 0.

    A = np.zeros((K, n))

        hi = min(K - 1, j + n - 1)
        A[j:hi+1, j] = f[j]
    return A
```

```python
def subgradient_max_conv(a, n):
    """Compute a subgradient of max_k (a*a)[k] at a."""
    c = fast_convolve(a)
    M = c.max()
    # Find active peaks
    active = np.where(np.isclose(c, M, atol=1e-12))[0]
    
    # Subgradient: average of active subgradients
    # d c[k] / d a[j] = 2 * a[k-j] (with a[m]=0 for out of range)
    grad = np.zeros(n)
    for k in active:
        for j in range(n):
            m = k - j
            if 0 <= m < n:
                grad[j] += 2.0 * a[m]
    grad /= len(active)
    return grad
```
