from scipy.optimize import linprog

def solve_lp_direction(p, K_max=100):
    n = len(p)
    b = fast_convolve(p)
    M = b.max()
    
    # Select top K peaks
    K = min(K_max, len(b))
    top_idx = np.argpartition(b, -K)[-K:]
    top_idx = np.sort(top_idx)
    
    # Build constraint matrix A (K x n)
    A_ub = np.zeros((K, n))
    for i, k in enumerate(top_idx):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A_ub[i, lo:hi+1] = p[lo:hi+1]
    
    # max sum(g) s.t. A_ub @ g <= M, g >= 0
    c = -np.ones(n)  # minimize -sum(g)
    res = linprog(c, A_ub=A_ub, b_ub=np.full(K, M), bounds=[(0, None)]*n, method='highs')
    
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0)
        return g
    return None

Optimized search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses multiplicative weight updates + LP-guided direction finding with scipy HiGHS.
"""

from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_weighted_gradient(p, w):
    """Compute gradient of sum_k w[k]*conv(p,p)[k] w.r.t p."""
    n = len(p)
    g = np.zeros(n)
    for k in range(len(w)):
        wk = w[k]
        if wk < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        js = np.arange(lo, hi + 1)
        g[js] += wk * p[k - js]
    return g


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho_idx = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_idx] - 1.0) / (rho_idx + 1.0)
    return np.maximum(v - theta, 0.0)


def multiplicative_descend(p, temperature, lr, n_steps=1):
    """Run n_steps of exponentiated gradient descent on the simplex."""
    for _ in range(n_steps):
        b = fast_convolve(p)
        scaled = b / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()

            w = np.zeros(len(b))
            w[b.argmax()] = 1.0
        else:
            w /= w_sum
        g = compute_weighted_gradient(p, w)
        p_new = p * np.exp(-lr * g)
        s = p_new.sum()
        if s < 1e-30:

        p_new /= s
        p = p_new

    b_ub = np.full(K, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, 
                  method='highs', options={'presolve': True})
    
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        if g.sum() > 1.0 + 1e-5:
            return g
    return None


def line_search(p, direction, n_evals=8):
    """Find best step size for moving from p toward direction."""
    s_dir = direction.sum()
    if s_dir < 1e-15:
        return 0.0, p.copy()
    h = direction / s_dir
    best_t = 0.05
    best_obj = fast_convolve(p).max()
    
    for t in np.linspace(0.01, 0.5, n_evals):
        p_trial = (1.0 - t) * p + t * h
        p_trial = np.maximum(p_trial, 0.0)
        s = p_trial.sum()
        if s < 1e-10:
            continue
        p_trial /= s
        obj = fast_convolve(p_trial).max()
        if obj < best_obj:
            best_obj = obj
            best_t = t
    
    return best_t, best_obj


def run_lp_phase(p_init, time_limit, K_init=50, K_max=300, label="lp"):
    """LP-guided optimization."""

    n = len(p_init)
    p = p_init.copy()
    best_p = p.copy()
    best_obj = fast_convolve(p).max()
    K = K_init
    stuck = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 30 == 0 and K < K_max:
            K = min(K + 20, K_max)
        
        g = solve_lp_direction(p, K)
        if g is None:
            if stuck > 5:
                break
            stuck += 1
            continue
        
        S_g = g.sum()
        h = g / S_g
        
        # Line search
        best_t = 0.05
        best_t_obj = fast_convolve(p).max()
        for t in [0.01, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.5]:
            p_trial = (1.0 - t) * p + t * h
            p_trial = np.maximum(p_trial, 0.0)

            if s < 1e-10:

            p_trial /= s
            obj = fast_convolve(p_trial).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t
        
        p = (1.0 - best_t) * p + best_t * h

        if s > 1e-10:

            stuck = 0
        else:

                break
        
        if it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} obj={best_obj:.12f} "

    return best_p, best_obj


def run_mwp_phase(p_init, time_limit, T_init=0.005, label="mwp"):
    """Multiplicative weight / exponentiated gradient phase."""
    t0 = time.monotonic()
    n = len(p_init)
    p = p_init.copy()
    best_p = p.copy()
    best_obj = fast_convolve(p).max()
    T = T_init
    lr = 0.1
    it = 0
    no_improve = 0
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        for _ in range(5):  # 5 sub-steps per iteration
            b = fast_convolve(p)
            scaled = b / T
            scaled -= scaled.max()
            w = np.exp(scaled)
            ws = w.sum()
            if ws < 1e-30:
                break
            w /= ws
            g = compute_weighted_gradient(p, w)
            p = p * np.exp(-lr * g)
            s = p.sum()
            if s < 1e-30:
                break
            p /= s
        
        if it % 10 == 0:
            obj = fast_convolve(p).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_p = p.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 10:
                    lr *= 0.9
                    T *= 1.1
                    no_improve = 0
                if no_improve > 40:
                    # Restart with perturbation
                    p = project_simplex(best_p + rng.standard_normal(n) * 0.001)
                    lr = 0.05
                    no_improve = 0
        
        if it % 100 == 0:
            T = max(T * 0.995, 0.001)
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} obj={best_obj:.12f} "
                  f"score={2.0*n*best_obj:.8f} T={T:.5f} lr={lr:.4f} t={elapsed:.0f}s")
    
    return best_p, best_obj


def coordinate_polish(p_init, time_limit):
    """Coordinate descent refinement."""
    t0 = time.monotonic()
    n = len(p_init)
    p = p_init.copy()
    best_obj = fast_convolve(p).max()
    multipliers = np.array([0.5, 0.7, 0.85, 1.15, 1.3, 2.0])
    
    improved = True
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if p[j] < 1e-14:
                continue
            for mult in multipliers:
                new_val = p[j] * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                p_trial = p.copy()
                p_trial[j] = new_val
                s = p_trial.sum()
                if s < 1e-10:
                    continue
                p_trial = p_trial / s
                obj = fast_convolve(p_trial).max()
                if obj < best_obj - 1e-13:
                    best_obj = obj
                    p = p_trial
                    improved = True
                    break
    
    return p, best_obj
