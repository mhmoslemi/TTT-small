```
    active = positions where b >= max_b - tol (near-max positions)
    w = uniform over active (or softmax-weighted)
    g = 2n * cross_corr(w, y) (the subgradient, via FFT)
    # Projected gradient step
    a_new = project_simplex(y - lr * g / (2n * max_b))
    # Nesterov update
    y = a_new + ((sqrt(1 + 4*lr^2) - 1) / 2) * (a_new - a)
    a = a_new
    # Adaptive lr
    if max_b decreased: increase lr slightly
    else: decrease lr
```

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Nesterov-accelerated subgradient
method with adaptive soft-max temperature, LP-based active set refinement,
and peak shaving local search. Explores multiple sequence lengths.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_subgrad(w, a):
    """Compute g[i] = sum_j w[j+i] * a[j] for i=0..n-1.
    This is the (unnormalized) subgradient of max_k (a*a)_k when w is
    a probability distribution over the active positions k."""
    n = len(a)
    g = np.zeros(n)
    for i in range(n):
        # w[i:i+n] has length n (w has length 2n-1, so i+n <= 2n-1 for i <= n-1)
        g[i] = np.dot(w[i:i+n], a)
    return g


def simplex_project(x):
    """Project x onto the probability simplex {p >= 0, sum(p) = 1}."""
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(x - theta, 0.0)


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    global _lp_solver
    if '_lp_solver' in globals() and _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return _lp_solver
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_subgradient(a0, time_limit, tau_start=0.1, tau_end=0.01, tau_decay=0.9995):
    """
    Nesterov-accelerated subgradient method with adaptive soft-max temperature.
    a0: initial point on the simplex (sum=1, non-negative)
    Returns the best point found.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy()
    a = a / s
    
    y = a.copy()  # Nesterov virtual iterate
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    tau = tau_start
    lr = 1.0 / np.sqrt(n)  # Initial learning rate
    t0 = time.monotonic()
    
    for it in range(500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute convolution of y
        b = fast_convolve(y)
        max_b = b.max()
        score_y = 2.0 * n * max_b
        
        # Soft-max weights
        beta = 1.0 / tau
        b_scaled = beta * b
        b_max = b_scaled.max()
        exp_b = np.exp(b_scaled - b_max)
        w = exp_b / exp_b.sum()
        
        # Only keep weights above threshold for efficiency
        w_threshold = 1e-10
        w[w < w_threshold] = 0
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros_like(b)
            w[np.argmax(b)] = 1.0
        else:
            w /= w_sum
        
        # Compute subgradient
        g = compute_subgrad(w, y)
        
        # Nesterov step
        # Step size: for f-L smooth objective, lr_t = 1/(L*sqrt(t))
        # Here L is related to 2n * sum(a)^2 * max stuff... use adaptive
        step = lr / (2.0 * n * max_b + 1e-10)
        
        # Projected gradient step
        a_new = y - step * 2.0 * g
        a_new = simplex_project(a_new)
        
        # Nesterov momentum
        t_curr = it + 1
        momentum = (1.0 + np.sqrt(1.0 + 4.0 * step * step)) / 2.0
        y = a_new + momentum * (a_new - a)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy
        else:
            y = a_new.copy()
        
        a = a_new
        
        # Evaluate and update best
        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        # Decay temperature
        tau = max(tau_decay * tau, tau_end)
        
        # Adaptive lr
        if it % 100 == 0:
            lr = 0.5 / np.sqrt(it + 1)
        
        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NS iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} tau={tau:.4f}")
    
    return best_a, best_score


def lp_refine(a0, time_limit, solver=None, n_iters=5000):
    """LP-based alternating refinement."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, M_val = solve_lp_best_response(a, solver)
        if b_new is None:
            continue
        
        # Averaged iterate
        alpha = 2.0 / (it + 2)
        a_avg = (1 - alpha) * a + alpha * b_new
        a_avg = np.maximum(a_avg, 0)
        s_avg = a_avg.sum()
        if s_avg > 1e-15:
            a_avg = a_avg / s_avg
        
        score_avg = 2.0 * n * fast_convolve(a_avg).max()
        if score_avg < best_score:
            best_score = score_avg
            best_a = a_avg.copy()
        
        a = a_avg
        
        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score


def peak_shave(a0, time_limit, n_steps=10000):
    """Aggressive peak shaving via coordinate adjustments."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        # Find the peak (excluding boundaries which are naturally lower)
        peak_k = int(np.argmax(b))
        
        # Find contributing indices
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        # Try reducing top contributors
        order = np.argsort(contribs)[::-1][:8]
        improved = False
        
        for ci in order:
            i = idx[ci]
            j = peak_k - i
            if a[i] < 1e-14:
                continue
            
            # Try several reduction factors
            for factor in [0.98, 0.95, 0.9, 0.8]:
                new_val = a[i] * factor
                diff = a[i] - new_val
                if diff < 1e-15:
                    continue
                
                a_trial = a.copy()
                a_trial[i] = new_val
                # Redistribute the removed mass proportionally to others
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                o_sum = others.sum()
                if o_sum < 1e-15:
                    continue
                a_trial[mask] = others + diff * others / o_sum
                a_trial = a_trial / a_trial.sum()
                
                sc = 2.0 * n * fast_convolve(a_trial).max()
                if sc < best_score - 1e-12:
                    a = a_trial

    """L-BFGS refinement with soft-max smoothing."""
    from scipy.optimize import minimize
    n = len(a0)

    s = a.sum()
    if s < 1e-15:
        return a0.copy()
    a = a / s
    
    # Work in log-space for simplex constraint
    a_safe = np.maximum(a, 1e-30)
    z = np.log(a_safe)
    t0 = time.monotonic()
    
    def make_func(tau):
        def func(z_val):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a_val = np.exp(z_val - z_val.max())
            a_val = a_val / a_val.sum()
            b = fast_convolve(a_val)
            Tb = tau * b
            return (2.0 * n / tau) * (Tb.max() + np.log(np.exp(Tb - Tb.max()).sum()))
        return func
    
    def make_jac(tau):
        def jac(z_val):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_val)
            a_val = np.exp(z_val - z_val.max())
            a_val = a_val / a_val.sum()
            b = fast_convolve(a_val)
            Tb = tau * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            # Gradient w.r.t. a
            g_a = compute_subgrad(w, a_val)
            g_a *= 2.0
            # Project to tangent space of simplex
            S = np.dot(g_a, a_val)
            grad_z = a_val * (g_a - S)
            return grad_z
        return jac
    
    best_z = z.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    # Multi-stage temperature
    n_stages = 4
    temps = np.geomspace(tau_start, tau_end, n_stages)
    
    for tau in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        maxiter = max(50, int(remaining * 100))
        try:
            res = minimize(
                fun=make_func(tau),
                x0=z,
                jac=make_jac(tau),
                method='L-BFGS-B',

            a_check = a_check / a_check.sum()
            sc = 2.0 * n * fast_convolve(a_check).max()
            if sc < best_score:
                best_score = sc
                best_z = z.copy()
        except Exception:
            pass
    
    a_out = np.exp(best_z - best_z.max())
    a_out = a_out / a_out.sum()
    return a_out

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-strategy approach:
1. Nesterov-accelerated subgradient with adaptive soft-max temperature (fast, many iterations)
2. L-BFGS refinement in log-space with geometric temperature annealing
3. LP-based best response averaging
4. Aggressive peak shaving local search
5. Exploration of multiple sequence lengths around the parent size
Key differences from prior: faster per-iteration method (Nesterov subgrad vs L-BFGS),
more aggressive exploration of n values, and better time budgeting.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_subgrad(w, a):
    """Compute g[i] = sum_j w[j+i] * a[j] for i=0..n-1 (subgradient of max conv)."""
    n = len(a)
    g = np.zeros(n)
    for i in range(n):
        g[i] = np.dot(w[i:i+n], a)
    return g


def simplex_project(x):
    """Project x onto the probability simplex."""
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(x - theta, 0.0)


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver_cached = None


def get_lp_solver():
    global _lp_solver_cached
    if _lp_solver_cached is not None:
        return _lp_solver_cached
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return _lp_solver_cached
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return _lp_solver_cached


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (sum=1), find b minimizing max_k conv(a,b)[k] s.t. sum(b)=1, b>=0."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_subgradient(a0, time_limit, tau_start=0.05, tau_end=0.005, tau_decay=0.9997):
    """
    Nesterov-accelerated projected subgradient with adaptive soft-max temperature.
    Works directly on the simplex (no log-parameterization).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), 2.0 * n * fast_convolve(a0).max()
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    tau = tau_start
    t0 = time.monotonic()
    stuck_count = 0
    current_lr = 1.0

    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute convolution
        b = fast_convolve(y)
        max_b = b.max()
        if max_b < 1e-15:
            break

        # Soft-max weights on convolution
        beta = 1.0 / tau
        b_scaled = beta * b
        b_max_val = b_scaled.max()
        exp_b = np.exp(b_scaled - b_max_val)
        w = exp_b / exp_b.sum()

        # Threshold for sparsity
        w_thresh = 1e-12
        mask = w > w_thresh
        if mask.any():
            w = w * mask
            w = w / w.sum()
        else:
            w = np.zeros_like(b)
            w[np.argmax(b)] = 1.0

        # Compute subgradient (unnormalized)
        g = compute_subgrad(w, y)

        # Adaptive step size: use Omega = Lipschitz estimate
        # f is smooth with L ~ 2n * max_b (roughly)
        # Use step = 1/sqrt(t) for subgradient, scaled
        t_val = float(it + 1)
        step = current_lr / (np.sqrt(t_val) * (2.0 * n * max_b + 1e-10))

        # Projected gradient step
        a_new = y - step * 2.0 * g
        a_new = simplex_project(a_new)

        # Nesterov momentum update
        mu = (1.0 + np.sqrt(1.0 + 4.0 * step * step)) / 2.0
        y = a_new + (mu - 1.0) * (a_new - a)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy
        else:
            y = a_new.copy()

        a = a_new

        # Track best
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stuck_count = 0
        elif score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stuck_count = 0
        else:
            stuck_count += 1

        # Decay temperature
        tau = max(tau_decay * tau, tau_end)

        # Adjust lr if stuck
        if stuck_count > 5000:
            current_lr *= 0.7
            stuck_count = 0

        if (it + 1) % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NS: iter={it+1} ({elapsed:.0f}s) score={best_score:.8f} tau={tau:.5f} lr={current_lr:.4f}")

    return best_a, best_score


def lbgfs_refine(a0, time_limit, tau_start=10.0, tau_end=1.0):
    """L-BFGS in log-space with soft-max smoothing and geometric temperature."""
    from scipy.optimize import minimize
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy()
    a = a / a / s if s > 0 else a
    a = a / a.sum()

    a_safe = np.maximum(a, 1e-25)
    z = np.log(a_safe) / np.log(a_safe).sum() * 0  # just log
    z = np.log(a_safe)
    t0 = time.monotonic()
    best_score = 2.0 * n * fast_convolve(a).max()
    best_z = z.copy()

    n_stages = 5
    temps = np.geomspace(tau_start, tau_end, n_stages)

    for tau in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func(z_val, tau=tau):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a_val = np.exp(z_val - z_val.max())
            a_val = a_val / a_val.sum()
            b = fast_convolve(a_val)
            Tb = tau * b
            m = Tb.max()
            return (2.0 * n / tau) * (m + np.log(np.exp(Tb - m).sum()))

        def jac(z_val, tau=tau):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_val)
            a_val = np.exp(z_val - z_val.max())
            a_val = a_val / a_val.sum()
            b = fast_convolve(a_val)
            Tb = tau * b
            m = Tb.max()
            exp_Tb = np.exp(Tb - m)
            w = exp_Tb / exp_Tb.sum()
            g_a = compute_subgrad(w, a_val) * 2.0
            S = np.dot(g_a, a_val)
            return a_val * (g_a - S)

        maxiter = max(30, int(remaining * 80))
        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                                    'maxfun': maxiter * 30, 'disp': False})
            z = res.x
            a_check = np.exp(z - z.max())
            a_check = a_check / a_check.sum()
            sc = 2.0 * n * fast_convolve(a_check).max()
            if sc < best_score:
                best_score = sc
                best_z = z.copy()
        except Exception:
            pass

    a_out = np.exp(best_z - best_z.max())
    a_out = a_out / a_out.sum()
    return a_out


def lp_refine(a0, time_limit, solver=None, n_iters=2000):
    """LP-based alternating with Nesterov-like averaging."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_val = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        a = ((it + 1) * a + alpha * (it + 2) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        sc_a = 2.0 * n * fast_convolve(a).max()
        sc_y = 2.0 * n * fast_convolve(y).max()
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        if (it + 1) % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LP: iter={it+1} ({elapsed:.0f}s) score={best_score:.8f}")

    return best_a, best_score


def peak_shave(a0, time_limit, n_steps=20000):
    """Aggressive peak shaving with multiple strategies per peak."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        peak_k = int(np.argmax(b))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        order = np.argsort(contribs)[::-1][:6]
        improved = False

        # Strategy 1: reduce single top contributor
        for ci in order[:4]:
            i = idx[ci]
            if a[i] < 1e-14:
                continue
            for factor in [0.99, 0.97, 0.94, 0.9, 0.85]:
                new_val = a[i] * factor
                diff = a[i] - new_val
                if diff < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                o_sum = others.sum()
                if o_sum < 1e-15:
                    continue
                a_trial[mask] = others + diff * others / o_sum
                a_trial = a_trial / a_trial.sum()
                sc = 2.0 * n * fast_convolve(a_trial).max()
                if sc < best_score - 1e-13:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        # Strategy 2: reduce symmetric pair
        if not improved:
            for ci in order[:3]:
                i = idx[ci]
                j = peak_k - i
                if i == j or a[j] < 1e-14 or a[i] < 1e-14:
                    continue
                for f in [0.96, 0.92, 0.88]:
                    a_trial = a.copy()
                    a_trial[i] *= f
                    a_trial[j] *= f
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    sc = 2.0 * n * fast_convolve(a_trial).max()
                    if sc < best_score - 1e-13:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        # Strategy 3: reduce all top 3 slightly
        if not improved:
            a_trial = a.copy()
            for ci in order[:3]:
                a_trial[idx[ci]] *= 0.98
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            sc = 2.0 * n * fast_convolve(a_trial).max()
            if sc < best_score - 1e-13:
                a = a_trial
                best_score = sc
                best_a = a.copy()
            else:
                break

    return best_a, best_score

def optimize_at_length(a_init, deadline, solver, time_frac=1.0):
    """Run full optimization pipeline at a fixed length. Returns best sequence and score."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a, 2.0 * n
    a = a / s

    def get_score(x):
        x = np.maximum(x, 0)
        xs = x.sum()
        if xs < 1e-15:
            return 2.0 * n
        x = x / xs
        return 2.0 * n * fast_convolve(x).max()

    cur_score = get_score(a)

    # Step 1: Nesterov subgradient (most of the time)
    rem = deadline - time.monotonic()
    t_ns = min(rem * 0.45, 500.0)
    if t_ns > 10:
        a, ns_score = nesterov_subgradient(a, t_ns, tau_start=0.03, tau_end=0.003)
        cur_score = min(cur_score, get_score(a))

    # Step 2: L-BFGS refinement
    rem = deadline - time.monotonic()
    t_lb = min(rem * 0.25, 200.0)
    if t_lb > 10:
        a_lb = lbgfs_refine(a, t_lb, tau_start=5.0, tau_end=0.5)
        lb_score = get_score(a_lb)
        if lb_score < cur_score:
            a = a_lb
            cur_score = lb_score

    # Step 3: LP refinement
    rem = deadline - time.monotonic()
    t_lp = min(rem * 0.15, 120.0)
    if t_lp > 10:
        a_lp, lp_score = lp_refine(a, t_lp, solver=solver, n_iters=1000)
        if get_score(a_lp) < cur_score:
            a = a_lp
            cur_score = get_score(a)

    # Step 4: Peak shaving
    rem = deadline - time.monotonic()
    t_ps = min(rem * 0.15, 100.0)
    if t_ps > 5:
        a_ps, ps_score = peak_shave(a, t_ps, n_steps=5000)
        if get_score(a_ps) < cur_score:
            a = a_ps
            cur_score = get_score(a)

    # Step 5: Final Nesterov polish
    rem = deadline - time.monotonic()
    t_final = min(rem * 0.20, 80.0)
    if t_final > 5:
        a2, fs_score = nesterov_subgradient(a, t_final, tau_start=0.01, tau_end=0.002, tau_decay=0.9999)
        if get_score(a2) < cur_score:
            a = a2
            cur_score = get_score(a)

    return a, cur_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization exploring multiple lengths:
    Nesterov subgradient + L-BFGS + LP + peak shaving pipeline.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 768
        best_seq = np.ones(n_init) / n_init
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.8f}")

    n_parent = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a_norm = a / s
        sc = evaluate_safe(a_norm)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Main optimization at parent length ===
    print(f"\n=== Main optimization at n={n_parent} (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_main = min(rem * 0.55, 900.0)
    if t_main > 30:
        print(f"  Budget: {t_main:.0f}s")
        a_main, sc_main = optimize_at_length(best_seq, time.monotonic() + t_main, solver)
        try_update(a_main, "main")

    # === Try nearby lengths ===
    print(f"\n=== Length exploration (best={best_score:.8f}) ===")
    alt_lengths = []
    for factor in [0.85, 0.9, 1.05, 1.1, 1.15, 1.25, 1.4, 0.75]:
        alt_n = int(n_parent * factor)
        if 256 <= alt_n <= 3000 and alt_n != n_parent and alt_n not in alt_lengths:
            alt_lengths.append(alt_n)

    # Sort by how close to parent (prioritize nearby lengths)
    alt_lengths.sort(key=lambda x: abs(x - n_parent))

    for alt_n in alt_lengths[:4]:
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t_alt = min(rem * 0.3, 150.0)
        if t_alt < 20:
            break
        print(f"  Trying n={alt_n} (budget={t_alt:.0f}s)")
        init_alt = interpolate_sequence(best_seq, alt_n)
        # Add small noise for diversity
        noise = rng.random(alt_n) * 0.01
        init_alt = init_alt + noise
        init_alt = np.maximum(init_alt, 0)
        init_alt = init_alt / init_alt.sum()
        a_alt, sc_alt = optimize_at_length(init_alt, time.monotonic() + t_alt, solver)
        try_update(a_alt, f"n{alt_n}")

    # === Final polish on best ===
    print(f"\n=== Final polish (best={best_score:.8f}, n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_polish = min(rem * 0.6, 100.0)
        if t_polish > 10:
            a_ps, _ = peak_shave(best_seq, t_polish, n_steps=10000)
            try_update(a_ps, "polish_ps")

        rem = deadline - time.monotonic()
        if rem > 15:
            t_ns = min(rem * 0.6, 60.0)
            if t_ns > 5:
                a_ns, _ = nesterov_subgradient(best_seq, t_ns, tau_start=0.005, tau_end=0.001, tau_decay=0.9999)
                try_update(a_ns, "polish_ns")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
