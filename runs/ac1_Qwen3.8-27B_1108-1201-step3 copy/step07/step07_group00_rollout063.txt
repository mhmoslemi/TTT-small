"""
Enhanced optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses: (1) LP with quadratic-corrected constraints, (2) Nesterov-accelerated
smoothed gradient, (3) mirror descent in log-space, (4) multi-length search.

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_from_conv(b, n):
    return 2.0 * n * b.max()  # assumes sum(a) = 1


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_solve_direction(f, top_k, t_est=0.1):
    """LP with quadratic correction for direction finding."""
    n = len(f)
    b = fast_convolve(f)
    M = b.max()
    if M < 1e-15:
        return None

    # Select peaks: all peaks above a fraction of M, plus top_k
    threshold = M * 0.7
    peak_idx = np.where(b >= threshold)[0]
    if len(peak_idx) < top_k:
        top_idx = np.argpartition(b, -top_k)[-top_k:]
        peak_idx = np.union1d(peak_idx, top_idx)
    else:
        # Use top_k highest peaks
        top_idx = np.argpartition(b, -top_k)[-top_k:]
        peak_idx = np.sort(top_idx)
    
    peak_idx = np.sort(peak_idx)
    K = len(peak_idx)

    A = np.zeros((K, n))
    for row_i in range(K):
        k = peak_idx[row_i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]

    # Quadratic correction: constraint is (f*g)[k] + (t_est/2)*(g*g)[k] <= M
    # Approximate g*g[k] by using current f as proxy: f*f[k]
    # So constraint becomes: (f*g)[k] + (t_est/2)*f*f[k] <= M
    # For peaks where f*f[k] = M: (f*g)[k] <= M*(1 - t_est/2)
    # For other peaks: (f*g)[k] <= M - (t_est/2)*f*f[k]
    rhs = M - (t_est / 2.0) * b[peak_idx]
    rhs = np.maximum(rhs, M * 0.1)  # safety: don't make rhs too small

    g = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [g >= 0, A @ g <= rhs])
    try:
        prob.solve(solver=cp.ECOS, max_iters=100000, abstol=1e-8, reltol=1e-8)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=200000, eps=1e-8)
        except Exception:
            return None

    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None
    return g_val


def golden_section_search(f, h, a=0.0, b_val=1.0, tol=0.01, max_iter=20):
    """Find t in [a, b] that minimizes conv((1-t)f + t*h).max() after normalization."""
    gr = (np.sqrt(5) - 1) / 2
    c = b_val - gr * (b_val - a)
    d = a + gr * (b_val - a)
    fc = _eval_direction(f, h, c)
    fd = _eval_direction(f, h, d)
    for _ in range(max_iter):
        if b_val - a < tol:
            break
        if fc < fd:
            b_val = d
            d = c
            fd = fc
            c = b_val - gr * (b_val - a)
            fc = _eval_direction(f, h, c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b_val - a)
            fd = _eval_direction(f, h, d)
    return (a + b_val) / 2


def _eval_direction(f, h, t):
    f_trial = (1.0 - t) * f + t * h
    f_trial = np.maximum(f_trial, 0.0)
    s = f_trial.sum()
    if s < 1e-10:
        return 1e10
    f_trial = f_trial / s
    return fast_convolve(f_trial).max()


def run_lp_optimize(f_init, time_limit, top_k=50, label="lp"):
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    stuck = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        g = lp_solve_direction(f, top_k, t_est=0.1)
        if g is None:
            g = lp_solve_direction(f, top_k, t_est=0.0)
            if g is None:
                break

        S_g = g.sum()
        if S_g <= 1.0 + 1e-5:
            stuck += 1
            if stuck > 15:
                break
            top_k = min(top_k + 10, 2 * n - 1)
            continue

        h = g / S_g

        # Golden section line search
        t_opt = golden_section_search(f, h, a=0.0, b_val=1.0, tol=0.005, max_iter=15)
        t_opt = max(0.01, min(0.99, t_opt))

        f_new = (1.0 - t_opt) * f + t_opt * h
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-10:
            break
        f_new /= s

        cur_obj = fast_convolve(f_new).max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f_new.copy()
            f = f_new.copy()
            stuck = 0
        else:
            # Try smaller step
            for t_try in [0.5 * t_opt, 0.3 * t_opt, 0.1 * t_opt]:
                f_try = (1.0 - t_try) * f + t_try * h
                f_try = np.maximum(f_try, 0.0)
                s_try = f_try.sum()
                if s_try < 1e-10:
                    continue
                f_try /= s_try
                obj_try = fast_convolve(f_try).max()
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_f = f_try.copy()
                    f = f_try.copy()
                    stuck = 0
                    break
            else:
                stuck += 1
                if stuck > 20:
                    break

        if it % 30 == 0:
            print(f"    {label} iter={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} t={time.monotonic()-t0:.0f}s")

    return best_f, best_obj


def nesterov_smoothed_gradient(f_init, time_limit, lr_init=0.001, smoothing=0.001):
    """Nesterov-accelerated smoothed gradient descent on the simplex."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    f_momentum = f.copy()
    v = np.zeros(n)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Compute smoothed gradient
        b = fast_convolve(f_momentum)
        # Softmax weighting
        scaled = b / smoothing
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(b))
            w[b.argmax()] = 1.0
        else:
            w /= w_sum

        # Compute subgradient: g[j] = sum_k w[k] * f[k-j] (valid range)
        g = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-10:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            j_start = k - hi
            j_end = k - lo
            g[j_start:j_end + 1] += w[k] * f[lo:hi + 1][::-1]

        # Nesterov step
        gamma = 2.0 / (it + 3.0)
        f_new = project_simplex(f_momentum - lr_init * g)
        f_momentum = f_new + (gamma - 1.0) / gamma * (f_new - f)
        f_momentum = np.maximum(f_momentum, 0.0)
        s = f_momentum.sum()
        if s > 1e-10:
            f_momentum /= s
        f = f_new

        if it % 50 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 10:
                    break

        if it % 200 == 0:
            # Restart from best
            if no_improve > 5:
                f = best_f.copy()
                f_momentum = best_f.copy()
                v = np.zeros(n)
                no_improve = 0

    return best_f, best_obj


def mirror_descent(f_init, time_limit, lr=0.01):
    """Mirror descent in log-space (exponential weighting)."""
    t0 = time.monotonic()
    n = len(f_init)
    # Work in log-space: a_i = exp(x_i) / sum(exp(x_j))
    # Initialize from f_init
    f = f_init.copy()
    eps = 1e-10
    x = np.log(f + eps)
    best_f = f_init.copy()
    best_obj = fast_convolve(f_init).max()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        f = np.exp(x - x.max())
        f = f / f.sum()

        b = fast_convolve(f)
        # Top peaks
        top_idx = np.argpartition(b, -20)[-20:]
        top_vals = b[top_idx]
        # Weight proportional to how much each peak exceeds the mean
        mean_b = b.mean()
        excess = np.maximum(b - mean_b, 0)
        excess[top_idx] += b[top_idx]  # emphasize top peaks
        w = excess / excess.sum()

        # Gradient in log-space
        # d/dx_i log(a_i) = 1/a_i, so gradient of conv w.r.t. x is grad_conv / a
        g_x = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-12:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g_x[idx] += w[k] * f[k - idx]

        # Update: x -= lr * g_x / f (Bregman step for entropy)
        x -= lr * g_x / (f + eps)

        if it % 20 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 20:
                    break

    return best_f, best_obj

"""
Enhanced optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses: (1) LP with quadratic-corrected constraints for direction finding,
(2) Nesterov-accelerated smoothed gradient on the simplex,
(3) Mirror descent in log-space, (4) golden-section line search,
(5) multi-length exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.signal import correlate as sp_correlate
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, w):
    """Compute weighted subgradient of auto-convolution.
    g[j] = 2 * sum_{m} w[j+m] * a[m]
    Uses scipy correlate for efficiency.
    """
    # correlate(w, a, mode='valid')[j] = sum_n w[j+n] * a[n]
    # This gives exactly sum_m w[j+m] * a[m] for j = 0,...,len(w)-len(a)
    grad = 2.0 * sp_correlate(w, a, mode='valid')
    return grad


def lp_solve_direction(f, top_k, t_est=0.05):
    """LP with quadratic correction for direction finding.
    Maximizes sum(g) s.t. g>=0 and (f@g)[k] + (t_est/2)*f*f[k] <= M for selected peaks.
    """
    n = len(f)
    b = fast_convolve(f)
    M = b.max()
    if M < 1e-15:
        return None

    # Select peaks: union of all peaks above 0.6*M and top_k peaks
    threshold = M * 0.6
    above = np.where(b >= threshold)[0]
    top_idx = np.argpartition(b, -top_k)[-top_k:]
    peak_idx = np.union1d(above, top_idx)
    peak_idx = np.sort(peak_idx)
    K = len(peak_idx)

    # Build constraint matrix
    A = np.zeros((K, n))
    for row_i in range(K):
        k = peak_idx[row_i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]

    # Quadratic correction: rhs = M - (t_est/2)*b[peak_idx]
    rhs = M - (t_est / 2.0) * b[peak_idx]
    rhs = np.maximum(rhs, 0.01 * M)  # safety floor

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= rhs]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    solved = False
    try:
        prob.solve(solver=cp.ECOS, max_iters=150000, abstol=1e-9, reltol=1e-9)
        solved = True
    except Exception:
        pass
    if not solved:
        try:
            prob.solve(solver=cp.SCS, max_iters=300000, eps=1e-8)
            solved = True
        except Exception:
            pass
    if not solved:
        try:
            prob.solve(solver=cp.SCS, max_iters=300000, eps=1e-7)
            solved = True
        except Exception:
            return None

    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
        return None
    return g_val


def eval_direction(f, h, t):
    """Evaluate objective at (1-t)*f + t*h (normalized)."""
    f_trial = (1.0 - t) * f + t * h
    f_trial = np.maximum(f_trial, 0.0)
    s = f_trial.sum()
    if s < 1e-10:
        return 1e10
    f_trial = f_trial / s
    return fast_convolve(f_trial).max()


def golden_section_search(f, h, a=0.01, b=0.95, max_iter=12):
    """Golden section search for optimal step size."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = eval_direction(f, h, c)
    fd = eval_direction(f, h, d)
    for _ in range(max_iter):
        b - a < 0.005
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = eval_direction(f, h, c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = eval_direction(f, h, d)
        if b - a < 0.005:
            break
    return (a + b) / 2.0


def run_lp_optimize(f_init, time_limit, top_k_init=50, top_k_max=200, label="lp"):
    """LP-based optimization with adaptive peak selection and golden section line search."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    stuck = 0
    top_k = top_k_init

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Adaptive top_k
        if it % 40 == 0 and top_k < top_k_max:
            top_k = min(top_k + 15, top_k_max)

        g = lp_solve_direction(f, top_k, t_est=0.05)
        if g is None:
            g = lp_solve_direction(f, top_k, t_est=0.0)
        if g is None:
            g = lp_solve_direction(f, min(top_k * 2, 2 * n - 1), t_est=0.0)
        if g is None:
            break

        S_g = g.sum()
        if S_g <= 1.0 + 1e-4:
            stuck += 1
            if stuck > 15:
                # Try without correction
                g = lp_solve_direction(f, top_k, t_est=0.0)
                if g is None or g.sum() <= 1.0 + 1e-4:
                    break
                S_g = g.sum()
            else:
                top_k = min(top_k * 2, top_k_max)
                continue

        h = g / S_g

        # Golden section line search
        t_opt = golden_section_search(f, h)
        t_opt = max(0.02, min(0.95, t_opt))

        f_new = (1.0 - t_opt) * f + t_opt * h
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-10:
            break
        f_new /= s

        cur_obj = fast_convolve(f_new).max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f_new.copy()
            f = f_new.copy()
            stuck = 0
        else:
            # Try progressively smaller steps
            improved = False
            for t_try in [0.5 * t_opt, 0.3 * t_opt, 0.15 * t_opt, 0.05 * t_opt]:
                f_try = (1.0 - t_try) * f + t_try * h
                f_try = np.maximum(f_try, 0.0)
                s_try = f_try.sum()
                if s_try < 1e-10:
                    continue
                f_try /= s_try
                obj_try = fast_convolve(f_try).max()
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_f = f_try.copy()
                    f = f_try.copy()
                    stuck = 0
                    improved = True
                    break
            if not improved:
                stuck += 1
                if stuck > 25:
                    break

        if it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} it={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} "
                  f"topk={top_k} t={elapsed:.0f}s")

    return best_f, best_obj


def nesterov_fista(f_init, time_limit, lr_init=0.01, smoothing_init=0.001):
    """FISTA (Nesterov accelerated) on the smoothed objective over the simplex."""
    t0 = time.monotonic()
    n = len(f_init)
    x = f_init.copy()
    y = f_init.copy()
    t_nesterov = 1.0
    best_f = f_init.copy()
    best_obj = fast_convolve(f_init).max()
    no_improve = 0
    it = 0
    smoothing = smoothing_init

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Compute smoothed gradient at y
        b = fast_convolve(y)
        scaled = b / smoothing
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(b))
            w[b.argmax()] = 1.0
        else:
            w /= w_sum

        grad = compute_subgradient(y, w)

        # FISTA step
        t_old = t_nesterov
        t_nesterov = (1.0 + np.sqrt(1.0 + 4.0 * t_old * t_old)) / 2.0
        momentum = (t_old - 1.0) / t_nesterov

        x_new = project_simplex(y - lr_init * grad)
        y_new = x_new + momentum * (x_new - x)

        x = x_new
        y = np.maximum(y_new, 0.0)
        s = y.sum()
        if s > 1e-10:
            y /= s

        if it % 30 == 0:
            obj = fast_convolve(x).max()
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_f = x.copy()
                no_improve = 0
                # Slightly increase lr if making good progress
                lr_init = min(lr_init * 1.02, 0.05)
            else:
                no_improve += 1
                if no_improve > 5:
                    lr_init *= 0.9
                    smoothing = min(smoothing * 1.3, 0.1)
                    no_improve = 0
                if no_improve > 20:
                    # Restart from best
                    x = best_f.copy()
                    y = best_f.copy()
                    t_nesterov = 1.0
                    no_improve = 0
                    if it % 100 == 0:
                        print(f"    fista restart at it={it}, best_obj={best_obj:.10f}")

        if it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    fista it={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} "
                  f"lr={lr_init:.5f} smooth={smoothing:.5f} t={elapsed:.0f}s")

    return best_f, best_obj


def mirror_descent_log(f_init, time_limit, lr=0.005):
    """Mirror descent in log-space (exponential weights / multiplicative weights)."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy().copy()
    eps = 1e-12
    log_f = np.log(f + eps)
    best_f = f_init.copy()
    best_obj = fast_convolve(f_init).max()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Current probability
        f = np.exp(log_f - log_f.max())
        f /= f.sum()

        b = fast_convolve(f)
        # Weight peaks by excess over mean
        mean_b = b.mean()
        excess = np.maximum(b - mean_b, 1e-15)
        w = excess / excess.sum()

        # Gradient in log-space: d/dx_i [sum_k w_k * b_k] where b_k = sum_j a_j * a_{k-j}
        # da_i/dx_i = a_i, so d b_k / dx_i = 2 * a_i * a_{k-i} (valid range) * a_i / a_i = ...
        # Actually: x_i = log(a_i), so a_i = exp(x_i), da_i/dx_i = a_i
        # d b_k / dx_i = sum_j (da_j/dx_i * a_{k-j} + a_j * da_{k-j}/dx_i)
        #             = a_i * a_{k-i} + a_i * a_{k-i} = 2 * a_i * a_{k-i} (when valid)
        # So d phi / dx_i = 2 * a_i * sum_k w_k * a_{k-i} (valid range)
        # = 2 * a_i * (correlate(w, a)[i] with appropriate indexing)

        # Using our subgradient: g[i] = 2 * sum_m w[i+m] * a[m]
        # But for mirror descent in log-space, we need d phi / d x_i = g[i] * a_i (chain rule)
        # Wait: d phi / d x_i = (d phi / d a_i) * (d a_i / d x_i) = g[i] * a_i
        # But g[i] already includes the a_i factor in the sum... let me recheck.

        # Actually from our derivation: d(b[k])/da[j] = 2*a[k-j] (valid range)
        # d(b[k])/dx[j] = d(b[k])/da[j] * da[j]/dx[j] = 2*a[k-j] * a[j]
        # So gradient in log-space: grad_x[j] = sum_k w[k] * 2*a[j]*a[k-j] (valid)
        # = 2*a[j] * sum_k w[k]*a[k-j] (valid)
        # = 2*a[j] * (1/2) * g_direct[j] where g_direct is the "unweighted by a" version

        # Let me just compute directly:
        grad_log = np.zeros(n)
        for j in range(n):
            # sum over valid k: w[k] * a[k-j]
            # valid: 0 <= k-j <= n-1, i.e., j <= k <= j+n-1, and k <= 2n-2
            k_lo = j
            k_hi = min(2 * n - 1, j + n)
            if k_lo < k_hi:
                grad_log[j] = 2.0 * f[j] * np.dot(w[k_lo:k_hi], f[0:k_hi - k_lo])

        # Multiplicative weights update: x_i -= lr * grad_x[i]
        log_f -= lr * grad_log
        # Clip to prevent overflow
        log_f = np.clip(log_f, -50.0, 50.0)

        if it % 20 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 30:
                    break

    return best_f, best_obj


def random_perturb(f, sigma=0.01):
    """Add random perturbation and project to simplex."""
    n = len(f)
    rng = np.random.default_rng()
    perturbed = f + rng.standard_normal(n) * sigma
    return project_simplex(perturbed)


def propose_candidate(seed=42, budget_s=1800.0):

            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(1024) / 1024.0
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"No parent, using uniform n=1024, score={p_score:.6f}")

    best_seq = parent.copy()
    best_score = p_score

    # === Phase 1: LP with quadratic correction (main) ===
    n_p = len(parent)
    print(f"\n=== Phase 1: LP quad-correct (n={n_p}, score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.30, 500.0)
    if t1 > 30:

    # === Phase 3: Multi-length exploration ===

    n_base = len(best_seq)
    tried_lengths = [n_base]

            tried_lengths.append(tl)
    tried_lengths = tried_lengths[1:3]  # take at most 2

    for target_n in tried_lengths:

            break
        t3 = min(rem * 0.22, 250.0)

        old_x = np.linspace(0.0, 1.0, n_base)

        f_interp = np.interp(new_x, old_x, best_seq)
