"""
Hybrid optimization for minimum peak autocorrelation problem.
Combines improved coordinate descent with line search, LP-based fictitious play,
and simulated annealing to find sequences minimizing 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_score(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    a_norm = a / s
    return 2.0 * n * fast_convolve(a_norm).max()

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def compute_subgrad_full(a):
    """Compute exact subgradient of max(conv(a,a)) w.r.t. a."""
    n = len(a)
    b = fast_convolve(a)
    k_star = int(np.argmax(b))
    # subgradient is 2 * a shifted appropriately
    sg = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(n - 1, k_star)
    for j in range(js, je + 1):
        sg[j] += 2.0 * a[k_star - j]
    return sg

def compute_subgrad_smooth(a, top_k=10, temp=0.005):
    """Smoothed subgradient using softmax over top peaks."""
    n = len(a)
    b = fast_convolve(a)
    k_actual = min(top_k, len(b))
    peaks = np.argpartition(b, -k_actual)[-k_actual:]
    pv = b[peaks]
    w = np.exp((pv - pv.max()) / temp)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[int(np.argmax(pv))] = 1.0
    else:
        w /= ws
    sg = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-14:
            continue
        k = int(peaks[idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        j = np.arange(js, je + 1)
        sg[j] += 2.0 * w[idx] * a[k - j]
    return sg

def nesterov_v2(a0, time_limit, top_k=20, lr_init=0.005, verbose=False):

def simulated_annealing(a0, time_limit, verbose=False):
    """Simulated annealing with multiple perturbation scales."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    T = 0.005  # Initial temperature
    T_min = 0.00001
    cooling = 0.9995
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Choose perturbation scale
        scale = rng.choice([0.001, 0.005, 0.01, 0.02, 0.05], p=[0.3, 0.3, 0.2, 0.1, 0.1])
        
        # Perturb
        if rng.random() < 0.7:
            # Gaussian perturbation
            a_trial = a + rng.standard_normal(n) * scale
        else:
            # Single coordinate perturbation
            i = int(rng.integers(n))
            a_trial = a.copy()
            a_trial[i] *= (1 + rng.standard_normal() * scale * 5)
        
        a_trial = np.maximum(a_trial, 0)
        s_t = a_trial.sum()
        if s_t < 1e-15:
            continue
        a_trial = a_trial / s_t
        
        score_trial = fast_score(a_trial)
        delta = score_trial - fast_score(a)
        
        if delta < 0 or rng.random() < np.exp(-delta / T):
            a = a_trial
            if score_trial < best_score:
                best_score = score_trial
                best_a = a.copy()
        
        T = max(T * cooling, T_min)
        
        if verbose and it % 1000 == 0:
            print(f"      SA it={it}: best={best_score:.8f} T={T:.6f}")
    
    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Hybrid optimization for minimum peak autocorrelation:
minimizes 2*n*max(conv(a,a))/(sum(a))^2 using peak-directed search,
improved coordinate descent with adaptive 1D search, LP-based fictitious play,
Nesterov subgradient, and simulated annealing.

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_score(a):
    s = a.sum()
    if s < 1e-15:
        return np.inf
    a_norm = a / s
    n = len(a_norm)
    return 2.0 * n * fast_convolve(a_norm).max()


def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def get_lp_solver():
    import cvxpy as cp
    global _lp_solver_cache
    if '_lp_solver_cache' in globals() and _lp_solver_cache is not None:
        return _lp_solver_cache
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.GLPK, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cache = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cache = cp.ECOS
    return cp.ECOS

        prob.solve(solver=solver)

def compute_subgrad_smooth(a, top_k=15, temp=0.004):
    n = len(a)
    b = fast_convolve(a)
    k_actual = min(top_k, len(b))
    peaks = np.argpartition(b, -k_actual)[-k_actual:]
    pv = b[peaks]
    w = np.exp((pv - pv.max()) / temp)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[int(np.argmax(pv))] = 1.0
    else:
        w /= ws
    sg = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-14:
            continue
        k = int(peaks[idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        j = np.arange(js, je + 1)
        sg[j] += 2.0 * w[idx] * a[k - j]
    return sg

        order = np.argsort(contribs)[::-1]
        improved = False

            j = k_peak - i

            for factor in [0.97, 0.94, 0.9, 0.85, 0.78, 0.7]:

                a = a / a.sum()
                score = 2.0 * n * fast_convolve(a).max()
                if score < best_score - 1e-14:
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a[i] /= factor
                    a = a / a.sum()

            if improved:
                break

            if j != i and a[j] > 1e-12:
                for factor in [0.97, 0.94, 0.9, 0.85]:
                    a[j] *= factor
                    a = a / a.sum()
                    score = 2.0 * n * fast_convolve(a).max()
                    if score < best_score - 1e-14:
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[j] /= factor
                        a = a / a.sum()
                if improved:
                    break

        if not improved:
            # Try increasing low-contribution coordinates
            peak_contrib = np.zeros(n)
            top_peaks = np.argpartition(b, -5)[-5:]
            for k in top_peaks:
                js2 = max(0, int(k) - n + 1)
                je2 = min(n - 1, int(k))
                idx2 = np.arange(js2, je2 + 1)
                c2 = a[idx2] * a[int(k) - idx2]
                for ii in range(len(idx2)):
                    peak_contrib[idx2[ii]] += c2[ii]

            safe = np.argsort(peak_contrib)[:30]
            for i in safe:
                if a[i] < 1e-12:
                    continue
                for factor in [1.03, 1.06, 1.1, 1.15]:
                    a[i] *= factor
                    a = a / a.sum()
                    score = 2.0 * n * fast_convolve(a).max()
                    if score < best_score - 1e-14:
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[i] /= factor
                        a = a / a.sum()
                if improved:
                    break

        if not improved:
            # Random pair perturbation
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i != j and a[i] > 1e-12 and a[j] > 1e-12:
                for di, dj in [(0.92, 1.08), (0.88, 1.12), (0.95, 1.05), (1.05, 0.95)]:
                    a[i] *= di
                    a[j] *= dj
                    a = a / a.sum()
                    score = 2.0 * n * fast_convolve(a).max()
                    if score < best_score - 1e-14:
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                    else:
                        a[i] /= di
                        a[j] /= dj
                        a = a / a.sum()

        if not improved and step > 200:
            # Try a bolder random perturbation
            if step % 100 == 0:
                mask = np.random.random(n) < 0.1
                a_trial = a.copy()
                a_trial[mask] *= (1 + np.random.standard_normal(mask.sum()) * 0.05)
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-12:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()

        if not improved and step > 500:
            break

    return best_a, best_score


def coordinate_descent_v2(a0, time_limit):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    sweep = 0
    no_improve_total = 0
    while time.monotonic() - t0 < time_limit:
        sweep += 1
        any_improve = False

        sg = compute_subgrad_smooth(a, top_k=12, temp=0.003)
        order = np.argsort(np.abs(sg))[::-1]

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[i]
            if old_val < 1e-14:
                continue

            factors = [0.85, 0.92, 0.96, 1.04, 1.08, 1.15]
            best_factor = 1.0
            best_factor_score = best_score

            for f in factors:
                a[i] = old_val * f
                a = a / a.sum()
                score = 2.0 * n * fast_convolve(a).max()
                if score < best_factor_score:
                    best_factor_score = score
                    best_factor = f
                a[i] = old_val
                a = a / a.sum()

            if best_factor != 1.0:
                a[i] = old_val * best_factor
                a = a / a.sum()
                if best_factor_score < best_score - 1e-14:
                    best_score = best_factor_score
                    best_a = a.copy()
                    any_improve = True
                    no_improve_total = 0
                else:
                    no_improve_total += 1
            else:
                no_improve_total += 1

        # Pair swap moves every few sweeps
        if sweep % 2 == 0:
            for _ in range(30):
                if time.monotonic() - t0 > time_limit:
                    break
                i = np.random.randint(n)
                j = np.random.randint(n)
                if i == j or a[i] < 1e-12 or a[j] < 1e-12:
                    continue
                for frac in [0.1, 0.2, 0.3]:
                    delta = a[i] * frac
                    a[i] -= delta
                    a[j] += delta
                    if a[i] < 0:
                        a[i] += delta
                        a[j] -= delta
                        continue
                    a = a / a.sum()
                    score = 2.0 * n * fast_convolve(a).max()
                    if score < best_score - 1e-14:
                        best_score = score
                        best_a = a.copy()
                        any_improve = True
                        no_improve_total = 0
                        break
                    else:
                        a[i] += delta
                        a[j] -= delta
                        a = a / a.sum()

        if not any_improve and no_improve_total > 300:
            break

    return best_a, best_score


def nesterov_v2(a0, time_limit, top_k=20, lr_init=0.005):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s

    y = a.copy()
    v = np.zeros(n)
    lr = lr_init
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        t_k = (1 + np.sqrt(1 + 4 * (it - 1) ** 2)) / 2
        beta_k = (t_k - 1) / t_k

        sg = compute_subgrad_smooth(y, top_k=top_k, temp=0.004)
        v = beta_k * v + sg

        norm_v = np.linalg.norm(v)
        if norm_v > 1e-15:
            v = v / norm_v

        y_new = y - lr * v
        y_projected = simplex_project(y_new)

        score = 2.0 * n * fast_convolve(y_projected).max()
        if score < best_score - 1e-13:
            best_score = score
            best_a = y_projected.copy()
            no_improve = 0
        else:
            no_improve += 1

        y = y_projected

        if no_improve > 150:
            lr *= 0.8
            no_improve = 0

        if no_improve > 500:
            noise = np.random.standard_normal(n) * 0.005
            y = simplex_project(best_a + noise)
            v = np.zeros(n)
            lr = max(lr_init * 0.5, lr * 2)
            no_improve = 0

        lr *= 0.9996

    return best_a, best_score


def fictitious_play_v2(n, time_limit, initial=None, solver=None):
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    t0 = time.monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        a = a / s

    b = a.copy()
    a_ema = a.copy().astype(np.float64)

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    iters = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is None:
            continue
        a_new, M_a = solve_lp_best_response(b_new, solver)
        if a_new is None:
            continue

        ema_rate = max(0.01, 0.08 / (1 + iters * 0.005))
        a_ema = (1 - ema_rate) * a_ema + ema_rate * a_new

        candidates = [a_new, b_new, 0.5 * (a_new + b_new), a_ema,
                      (2 * a_ema + a_new) / 3]

        for cand in candidates:
            cand = np.maximum(cand, 0)
            s = cand.sum()
            if s < 1e-10:
                continue
            cand = cand / s
            score = 2.0 * n * fast_convolve(cand).max()
            if score < best_score - 1e-14:
                best_score = score
                best_a = cand.copy()
                no_improve = 0
            else:
                no_improve += 1

        a = a_new
        b = b_new

        if no_improve > 400:
            break

    return best_a, best_score, iters


def simulated_annealing(a0, time_limit):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    T = 0.003
    T_min = 0.000005
    cooling = 0.9997
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        scale = np.random.choice([0.001, 0.003, 0.008, 0.015, 0.03],
                                  p=[0.25, 0.3, 0.2, 0.15, 0.1])

        if np.random.random() < 0.7:
            a_trial = a + np.random.standard_normal(n) * scale
        else:
            a_trial = a.copy()
            idx = np.random.choice(n, size=max(1, n // 50), replace=False)
            a_trial[idx] *= (1 + np.random.standard_normal(len(idx)) * scale * 3)
            a_trial[idx] = np.maximum(a_trial[idx], 0)

        a_trial = np.maximum(a_trial, 0)
        s_t = a_trial.sum()
        if s_t < 1e-15:
            continue
        a_trial = a_trial / s_t

        score_trial = 2.0 * n * fast_convolve(a_trial).max()
        cur_score = 2.0 * n * fast_convolve(a).max()
        delta = score_trial - cur_score

        if delta < 0 or np.random.random() < np.exp(-delta / max(T, T_min)):
            a = a_trial
            if score_trial < best_score:
                best_score = score_trial
                best_a = a.copy()

        T = max(T * cooling, T_min)

    return best_a, best_score


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def verify_and_update(a, best_seq, best_score, label=""):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score
