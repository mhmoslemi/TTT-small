"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
L-BFGS with multi-temperature soft-max, adaptive Adam, direct peak-shaving,
simulated annealing, and multi-length exploration. Key improvements:
(1) L-BFGS superlinear convergence for smooth phases,
(2) adaptive temperature based on improvement rate,
(3) targeted peak manipulation with pair adjustments,
(4) simulated annealing for fine-tuning,
(5) efficient multi-start with diverse structural initializations.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via real FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def get_score(a):
    """Compute 2n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - np.max(z)
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    return np.log(np.maximum(a, 1e-300))


def compute_grad_soft(a, w, n, fft_size):
    """Gradient of 2n * sum(w * conv(a,a)) w.r.t. a."""
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(W * np.conj(A), fft_size)[:n]
    return 2.0 * n * corr


def lbfgs_softmax(a0, time_limit, T=50.0, maxiter=3000):
    """L-BFGS optimization with soft-max smoothing in log-space."""
    n = len(a0)
    z0 = z_from_a(a0)
    fft_size = 1 << (2 * n - 2).bit_length()
    t0 = time.monotonic()
    stopped = [False]

    def obj(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e30
        a = softmax_from_z(z)
        b = fast_convolve(a)
        max_b = b.max()
        exp_b = np.exp(b / T - max_b / T)
        return 2.0 * n * (max_b + T * np.log(exp_b.sum()))

    def grad(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return np.zeros_like(z)
        a = softmax_from_z(z)
        b = fast_convolve(a)
        max_b = b.max()
        exp_b = np.exp(b / T - max_b / T)
        w = exp_b / exp_b.sum()
        grad_a = compute_grad_soft(a, w, n, fft_size)
        S = np.dot(grad_a, a)
        return a * (grad_a - S)

    try:
        res = minimize(
            fun=obj, jac=grad, x0=z0,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12,
                     'maxfun': maxiter * 80, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
    except Exception:
        a_out = a0.copy()
    return a_out


def adam_adaptive(a0, time_limit, T_init=0.05, T_final=0.001, lr=0.01,
                  beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam with adaptive temperature based on improvement rate."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    a = a / a.sum()
    z = z_from_a(a)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    fft_size = 1 << (2 * n - 2).bit_length()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = get_score(a)
    no_improve = 0
    patience = 2000
    T = T_init
    last_best_time = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)

        # Adaptive temperature: decrease slowly, increase if stuck
        if no_improve < patience // 2:
            T = max(T_final, T * 0.9995)
        elif no_improve < patience:
            T = max(T_final, T * 0.999)
        else:
            T = min(T_init, T * 1.01)

        a_curr = softmax_from_z(z)
        b = fast_convolve(a_curr)
        max_b = b.max()
        exp_b = np.exp(b / T - max_b / T)
        w = exp_b / exp_b.sum()

        grad_a = compute_grad_soft(a_curr, w, n, fft_size)
        S = np.dot(grad_a, a_curr)
        grad_z = a_curr * (grad_a - S)

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        lr_t = lr * (1.0 - 0.00005 * t)
        lr_t = max(0.0005, lr_t)
        z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)

        a_new = softmax_from_z(z)
        score = get_score(a_new)

        if score < best_score - 1e-15:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
            last_best_time = t
        else:
            no_improve += 1

        # Adaptive restart
        if no_improve > patience:
            z = z_from_a(best_a) + np.random.standard_normal(n) * 0.01
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            T = max(T, T_init * 0.3)

    return best_a, best_score


def peak_shave_direct(a0, time_limit, n_trials=5000):
    """
    Direct peak shaving: identify top contributing pairs for each peak
    and try to reduce the peak by adjusting them.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = get_score(a)
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        # Find the peak (prefer interior)
        b_search = b.copy()
        b_search[0] = -np.inf
        b_search[-1] = -np.inf
        peak_k = int(np.argmax(b_search))
        peak_val = b[peak_k]

        if peak_val * 2.0 * n < best_score * 0.99:
            # Peak is not the binding one, try the actual max
            peak_k = int(np.argmax(b))
            peak_val = b[peak_k]

        target = best_score / (2.0 * n)
        if peak_val <= target * 1.0001:
            continue

        # Find contributing pairs
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_count = min(8, len(idx))
        top_pos = np.argsort(contribs)[::-1][:top_count]

        improved = False
        for ci in top_pos:
            i = int(idx[ci])
            j = int(peak_k - i)
            if i == j or i >= n or j >= n or a[j] < 1e-15 or a[i] < 1e-15:
                continue

            # Try reducing the larger element and redistributing
            for f1 in [0.92, 0.85, 0.75, 0.6]:
                a_trial = a.copy()
                diff = a_trial[i] * (1.0 - f1)
                a_trial[i] *= f1
                # Redistribute to elements far from peak
                mask = np.ones(n, dtype=bool)
                mask[max(0, peak_k - n // 4):peak_k + n // 4] = False
                mask[i] = False
                if mask.sum() < 2:
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                others = a_trial[mask]
                if others.sum() < 1e-15:
                    continue
                a_trial[mask] += diff * others / others.sum()
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                score = get_score(a_trial)
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a_trial.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try single element reduction with redistribution to random positions
            if trial % 3 == 0:
                for ci in top_pos[:3]:
                    i = int(idx[ci])
                    if a[i] < 1e-15:
                        continue
                    for f1 in [0.85, 0.7]:
                        a_trial = a.copy()
                        diff = a_trial[i] * (1.0 - f1)
                        a_trial[i] *= f1
                        # Random redistribution
                        targets = np.random.choice(n - 1, size=min(5, n - 1), replace=False)
                        targets = targets[targets != i]
                        if len(targets) == 0:
                            continue
                        weights = np.random.dirichlet(np.ones(len(targets)))
                        for ti, wi in zip(targets, weights):
                            a_trial[ti] += diff * wi
                        a_trial = np.maximum(a_trial, 0)
                        ss = a_trial.sum()
                        if ss < 1e-15:
                            continue
                        a_trial = a_trial / ss
                        score = get_score(a_trial)
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a_trial.copy()
                            improved = True
                            break
                    if improved:
                        break

        # Periodic perturbation when stuck
        if not improved and trial % 200 == 199:
            a = best_a.copy()
            mask = np.random.random(n) < 0.1
            a[mask] *= (1.0 + np.random.standard_normal(mask.sum()) * 0.02)
            a = np.maximum(a, 0)
            ss = a.sum()
            if ss > 1e-15:
                a = a / ss

    return best_a, best_score


def simulated_annealing(a0, time_limit, T_init=0.001, T_final=0.00001,
                        move_scale=0.02):
    """Simulated annealing for fine-tuning."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = get_score(a)
    t0 = time.monotonic()
    current_score = best_score
    T = T_init

    step = 0
    while time.monotonic() - t0 < time_limit:
        step += 1
        elapsed = (time.monotonic() - t0) / time_limit
        T = T_init * (T_final / T_init) ** elapsed

        # Generate a move
        move_type = np.random.random()
        if move_type < 0.5:
            # Pair swap: move mass from i to j
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i == j:
                continue
            amount = a[i] * move_scale * np.random.exponential(1.0)
            amount = min(amount, a[i] * 0.5)
            a_trial = a.copy()
            a_trial[i] -= amount
            a_trial[j] += amount
        elif move_type < 0.8:
            # Gaussian perturbation on random subset
            mask = np.random.random(n) < 0.05
            count = max(1, mask.sum())
            a_trial = a.copy()
            a_trial[mask] += np.random.standard_normal(count) * move_scale
        else:
            # Multiplicative perturbation
            i = np.random.randint(n)
            factor = 1.0 + np.random.standard_normal() * move_scale
            a_trial = a.copy()
            a_trial[i] *= max(0.1, factor)

        a_trial = np.maximum(a_trial, 0)
        ss = a_trial.sum()
        if ss < 1e-15:
            continue
        a_trial = a_trial / ss
        new_score = get_score(a_trial)

        # Accept/reject
        delta = new_score - current_score
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-20)):
            a = a_trial
            current_score = new_score
            if new_score < best_score - 1e-15:
                best_score = new_score
                best_a = a_trial.copy()

    return best_a, best_score


def coordinate_refine(a0, time_limit, n_sweeps=50):
    """Coordinate descent with multiplicative updates."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = get_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        order = rng.permutation(n)
        sweep_improved = False

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-15:
                continue

            best_local = a[i]
            best_local_score = best_score

            for factor in [0.8, 0.9, 0.95, 1.05, 1.1, 1.2]:
                a_trial = a.copy()
                a_trial[i] = a[i] * factor
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                score = get_score(a_trial)
                if score < best_local_score:
                    best_local_score = score
                    best_local = a[i] * factor

            if best_local != a[i]:
                a[i] = best_local
                a = a / a.sum()
                if best_local_score < best_score:
                    best_score = best_local_score
                    best_a = a.copy()
                    sweep_improved = True

        if not sweep_improved:
            break

    return best_a, best_score


def make_diverse_inits(parent, n, rng, n_inits=10):
    """Generate diverse initializations."""
    inits = []
    if parent is not None and len(parent) == n:
        p = np.maximum(parent, 0)
        p = p / p.sum()
        inits.append(p.copy())
        for scale in [0.02, 0.05, 0.1, 0.2]:
            pert = p + rng.standard_normal(n) * scale
            pert = np.maximum(pert, 0)
            ps = pert.sum()
            if ps > 1e-10:
                inits.append(pert / ps)
        # Structured perturbations
        for freq in [2, 3, 5, 7, 11]:
            t = np.linspace(0, 2 * np.pi * freq, n)
            for phase in [0, np.pi / 3, np.pi / 2]:
                pert = p + 0.03 * np.cos(t + phase)
                pert = np.maximum(pert, 0)
                ps = pert.sum()
                if ps > 1e-10:
                    inits.append(pert / ps)
    else:
        # Random initializations
        for _ in range(n_inits):
            r = np.abs(rng.standard_normal(n)) + 0.01
            inits.append(r / r.sum())
        # Uniform
        inits.append(np.ones(n) / n)
        # Triangular
        tri = np.abs(np.linspace(-1, 1, n))
        tri = np.maximum(tri, 0.01)
        inits.append(tri / tri.sum())

    # Deduplicate and limit
    unique = []
    for init in inits:
        if len(unique) < 20:
            unique.append(init)
    return unique


def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
