import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_xy(x, y):
    """Linear convolution of x and y via FFT."""
    n = len(x) + len(y) - 1
    fft_size = 1 << (n - 1).bit_length()
    X = rfft(x, fft_size)
    Y = rfft(y, fft_size)
    return irfft(X * Y, fft_size)[:n]


def build_conv_matrix(a):
    """Build matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    m = 2 * n - 1
    T = np.zeros((m, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    global _lp_solver_cached
    if '_lp_solver_cached' in globals():
        return _lp_solver_cached
    for solver_name in [cp.ECOS, cp.SCS, cp.SCS, cp.ECOS, cp.PDLK if hasattr(cp, 'PDLK') else cp.ECOS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver_cached = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver_cached = cp.ECOS
    return _lp_solver_cached


def solve_lp_best_response(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.SCS)
        except Exception:
            return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)


def compute_score(a):
    """Compute the objective score for a normalized sequence."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def lbgfs_multistage(a0, time_limit, T_list, maxiter_per_stage=3000):
    """
    Multi-stage L-BFGS with soft-max objective on the simplex (log-parameterized).
    T_list: list of temperatures to use in sequence.
    """
    n = len(a0)
    z = z_from_a(np.maximum(a0, 1e-300))
    t0 = time.monotonic()
    best_a = np.maximum(a0, 0).copy()
    best_a = best_a / best_a.sum()
    best_score = compute_score(best_a)

    for stage_i, T_val in enumerate(T_list):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func(z):
            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp

        def jac(z):
            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            # Gradient of sum_k w[k] * conv(a,a)[k] w.r.t. a
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-18:
                    continue
                ws = w[k]
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * ws * a[k - idx]
            # Project gradient to tangent space of simplex (in log-space)
            S = np.dot(grad_a, a)
            return a * (grad_a - S)

        try:
            res = minimize(
                fun=func,
                x0=z,
                jac=jac,
                method='L-BFGS-B',
                options={'maxiter': maxiter_per_stage, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxfun': maxiter_per_stage * 30, 'disp': False}
            )
            z_new = res.x
            a_new = softmax_from_z(z_new)
            sc = compute_score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            z = z_new
        except Exception:
            pass

        elapsed = time.monotonic() - t0
        print(f"    L-BFGS stage T={T_val:.2f} ({elapsed:.0f}s): best_score={best_score:.8f}")

    return best_a


def lp_best_response_dynamics(a0, time_limit, solver, max_iters=50000):
    """
    LP-based best response dynamics with Nesterov-type averaging.
    Alternates: solve LP for b*, then use gradient info to update a.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    # Maintain running average for Nesterov
    avg_b = a.copy()
    y = a.copy()  # virtual iterate

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_val = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        # Nesterov step size
        alpha = 2.0 / (it + 2.0)

        # Update virtual iterate
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        # Update actual iterate via averaging
        a = y.copy()  # Use the averaged iterate directly (FTRL-style)

        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Dyn iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} cur={sc:.8f}")

    return best_a, best_score


def peak_equalize(a0, time_limit, n_steps=20000, lr_init=0.01):
    """
    Subgradient method with peak equalization: penalize both the max and
    the variance among top peaks to encourage flatness.
    """
    n = len(a0)
    z = z_from_a(np.maximum(a0, 1e-300))
    best_a = np.maximum(a0, 0).copy()
    best_a = best_a / best_a.sum()
    best_score = compute_score(best_a)
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        a = softmax_from_z(z)
        b = fast_convolve(a)
        max_b = b.max()
        score = 2.0 * n * max_b

        # Find top peaks
        threshold = max_b - 0.05 * max_b
        top_mask = b >= threshold
        top_indices = np.where(top_mask)[0]
        if len(top_indices) == 0:
            top_indices = np.array([np.argmax(b)])

        # Subgradient: average of gradients at active peaks
        g = np.zeros(n)
        for k in top_indices:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(top_indices)

        # Additional: penalize variance among top peaks (encourage equalization)
        # Gradient of variance of top b values
        top_vals = b[top_indices]
        mean_top = top_vals.mean()
        var_grad = np.zeros(n)
        for k in top_indices:
            diff = top_vals[list(top_indices).index(k)] - mean_top
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            var_grad[idx] += 2.0 * diff * a[k - idx]
        g = 0.9 * g + 0.1 * var_grad / max(1e-10, var_grad.max())

        # Mirror descent step
        S = np.dot(g, a)
        grad_z = a * (g - S)
        z_new = z - lr * grad_z
        a_new = softmax_from_z(z_new)

        sc = compute_score(a_new)
        if sc < best_score - 1e-15:
            best_score = sc
            best_a = a_new.copy()
            z = z_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            lr *= 0.7
            no_improve = 0
        lr = max(lr, 1e-7)

        if step % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakEq {elapsed:.0f}s: score={best_score:.8f} lr={lr:.2e}")

    return best_a, best_score


def coordinate_descent_fast(a0, time_limit, n_passes=50):
    """Fast coordinate descent trying multiple perturbations per coordinate."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    for p in range(n_passes):
        if time.monotonic() - t0 > time_limit:
            break
        # Process coordinates in random order for diversity
        order = np.random.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue
            base = a.copy()
            base[i] = 0
            base_sum = base.sum()
            if base_sum < 1e-15:
                continue
            best_t = a[i]
            best_local = best_score
            for factor in [0.7, 0.85, 1.0, 1.15, 1.3, 1.5]:
                trial_t = a[i] * factor
                if trial_t < 0 or trial_t > 1.0:
                    continue
                a_trial = base * (1.0 - trial_t) + trial_t * np.eye(n)[i]
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score(a_trial)
                if sc < best_local:
                    best_local = sc
                    best_t = trial_t
            if best_t != a[i]:
                a = base * (1.0 - best_t) + best_t * np.eye(n)[i]
                a = a / a.sum()
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

        if p % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc pass={p} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def peak_shave_aggressive(a0, time_limit, n_steps=10000):
    """Aggressive local peak shaving targeting the global max peak."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        # Find the peak (exclude edges which are naturally small)
        b_internal = b.copy()
        b_internal[:max(1, n // 10)] = -np.inf
        b_internal[-max(1, n // 10):] = -np.inf
        peak_k = int(np.argmax(b_internal))
        if peak_k < 0 or b[peak_k] <= 0:
            peak_k = int(np.argmax(b))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        n_top = min(8, len(contribs))
        top_contrib = np.argsort(contribs)[::-1][:n_top]

        improved = False
        for ci in top_contrib:
            i = idx[ci]
            j = peak_k - i
            if a[i] < 1e-14:
                continue
            for factor in [0.98, 0.95, 0.90, 0.85]:
                new_val = a[i] * factor
                diff = a[i] - new_val
                if diff < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                a_trial[mask] = others + diff * others / others_sum
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score(a_trial)
                if sc < best_score - 1e-15:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try simultaneous reduction of top contributors
            if len(top_contrib) >= 2:
                for n_sim in [2, 3]:
                    a_trial = a.copy()
                    for ci in top_contrib[:n_sim]:
                        i = idx[ci]
                        a_trial[i] *= 0.95
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        break
                    a_trial = a_trial / ss
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-15:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
            if not improved:
                # Try adding mass to a random low-contribution position
                rand_pos = np.random.randint(0, n)
                if a[rand_pos] < 0.1 / n:
                    a_trial = a.copy()
                    a_trial[rand_pos] += 0.001 / n
                    # Reduce randomly
                    reduce_pos = np.random.randint(0, n)
                    if reduce_pos != rand_pos and a_trial[reduce_pos] > 0.01 / n:
                        a_trial[reduce_pos] -= 0.001 / n
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss > 1e-15:
                        a_trial = a_trial / ss
                        sc = compute_score(a_trial)
                        if sc < best_score - 1e-15:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if not a_list:
            return float('inf')
        if sum(a_list) < 0.01:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining LP best-response, multi-stage L-BFGS,
    peak equalization, and aggressive local search.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = compute_score(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 768
        best_seq = np.ones(n_init) / n_init
        best_score = compute_score(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = compute_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: LP Best Response Dynamics (350s) ===
    print(f"\n=== Phase 1: LP Best Response (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 350.0)
    if t_alloc > 30:
        print(f"  Budget={t_alloc:.0f}s, n={n_best}")
        a_lp, _ = lp_best_response_dynamics(best_seq, t_alloc, solver)
        try_update(a_lp, "lp_dyn")

    # === Phase 2: Multi-stage L-BFGS (500s) ===
    print(f"\n=== Phase 2: L-BFGS Multi-Stage (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 500.0)
    if t_alloc > 30:
        T_stages = [100.0, 50.0, 20.0, 10.0, 5.0, 2.0, 1.0, 0.5]
        maxiter = max(500, int(40000 * t_alloc / 500))
        print(f"  Temps={T_stages}, maxiter/stage~{maxiter}, budget={t_alloc:.0f}s")
        a_lb = lbgfs_multistage(best_seq, t_alloc, T_stages, maxiter_per_stage=maxiter)
        try_update(a_lb, "lbgfs_main")

    # === Phase 3: LP Refinement on L-BFGS result (200s) ===
    print(f"\n=== Phase 3: LP Refine (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 200.0)
    if t_alloc > 30:
        print(f"  Budget={t_alloc:.0f}s")
        a_lp2, _ = lp_best_response_dynamics(best_seq, t_alloc, solver, max_iters=30000)
        try_update(a_lp2, "lp_refine")

    # === Phase 4: Ultra-low-T L-BFGS (400s) ===
    print(f"\n=== Phase 4: Ultra-Low-T L-BFGS (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 30:
        T_stages_low = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01]
        maxiter_low = max(1000, int(80000 * t_alloc / 400))
        print(f"  Temps={T_stages_low}, maxiter/stage~{maxiter_low}, budget={t_alloc:.0f}s")

    # === Phase 6: Peak Shaving (150s) ===

        t_alloc = min(rem * 0.35, 80.0)
        if t_alloc < 15:
            break
        print(f"  Alt n={alt_n}, budget={t_alloc:.0f}s")
        init_alt = interpolate_sequence(best_seq, alt_n)
        a_alt = lbgfs_multistage(init_alt, t_alloc, [20.0, 5.0, 1.0, 0.2, 0.05], maxiter_per_stage=3000)
        try_update(a_alt, f"alt_n{alt_n}")

    # === Final: one more peak shave on best ===
    rem = deadline - time.monotonic()
    if rem > 10:

        b = fast_convolve(best_seq)

        print(f"  Top-5 conv values: {sorted(b, reverse=True)[:5]}")
    return [float(x) for x in best_seq]
