"""
Hybrid convex optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses parametric initialization, LP-based direction finding with ECOS solver,
Nesterov-accelerated subgradient descent, coordinate descent, and
multi-length exploration to improve upon previous best of 1.504861.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    # Scale to find projection onto sum=1, then scale back
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def evaluate_local(a):
    """Compute the objective for a normalized sequence (sum=1)."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def lp_direction(f, top_k, solver='ECOS', eps=1e-7):
    """
    Solve: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for top_k peaks.
    Returns (g, M) or (None, M) on failure.
    """
    n = len(f)
    b = fast_convolve(f)
    M = b.max()
    if M < 1e-15:
        return None, M

    if top_k < len(b):
        top_idx = np.argpartition(b, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(b))
    top_idx = np.sort(top_idx)
    K = len(top_idx)

    # Build constraint matrix (K x n)
    A_con = np.zeros((K, n))
    for row_i in range(K):
        k = top_idx[row_i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A_con[row_i, lo:hi + 1] = f[lo:hi + 1]

    g = cp.Variable(n)
    constraints = [g >= 0, A_con @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=solver, **{'eps': eps} if solver in ('SCS', 'ECOS') else {})
    except Exception:
        try:
            prob.solve(solver='SCS', max_iters=100000, eps=1e-7)
        except Exception:
            return None, M

    if g.value is None:
        return None, M
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None, M
    return g_val, M


def run_lp_optimize(f_init, time_limit, top_k_start=20, top_k_end=80, label="lp"):
    """LP-based optimization with adaptive top_k."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    f = f / s if s > 1e-10 else f
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    stuck = 0
    top_k = top_k_start

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Gradually increase top_k
        progress = min(1.0, it / 200.0)
        top_k = int(top_k_start + (top_k_end - top_k_start) * progress)

        g, M = lp_direction(f, top_k)

        if g is None:
            g, M = lp_direction(f, top_k * 2)
            if g is None:
                if stuck > 5:
                    break
                stuck += 1
                continue

        S_g = g.sum()
        if S_g <= 1.0 + 1e-4:
            stuck += 1
            if stuck > 20:
                break
            continue

        # Line search with more points
        h = g / S_g
        best_t = 0.0
        best_t_obj = fast_convolve(f).max()

        # Coarse then fine line search
        for t in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7]:
            f_trial = (1.0 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0.0)
            s_trial = f_trial.sum()
            if s_trial < 1e-10:
                continue
            f_trial = f_trial / s_trial
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t

        if best_t == 0.0:
            stuck += 1
            if stuck > 30:
                break
            continue

        # Update
        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        f = f / s if s > 1e-10 else f

        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 40:
                break

        if it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} it={it} obj={best_obj:.10f} "
                  f"score={2.0*n*best_obj:.6f} tk={top_k} t={elapsed:.0f}s")

    return best_f, best_obj


def subgradient_optimize(f_init, time_limit, lr=0.005, momentum=0.9, temp=0.001, n_top=50, label="sg"):
    """Subgradient optimization with momentum and restarts."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    f = f / s if s > 1e-10 else f
    v = np.zeros(n)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        
        # Compute subgradient direction
        # Gradient of max_k conv(f,f)[k] w.r.t. f is conv(e_k, f) at the maximizing k
        # But we want to minimize, so we use a "soft" version with all peaks weighted
        scaled = b / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(b))
            w[b.argmax()] = 1.0
        else:
            w = w / w_sum

        # Focus on top peaks
        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
            w_sel = w[top_idx].copy()
            ws = w_sel.sum()
            if ws > 1e-30:
                w_sel = w_sel / ws
            else:
                w_sel = np.ones(len(top_idx)) / len(top_idx)
            w_new = np.zeros(len(b))
            w_new[top_idx] = w_sel
            w = w_new

        # Compute gradient: grad_j = sum_k w[k] * f[k-j] (with bounds)
        g = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-12:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += w[k] * f[k - idx]

        v = momentum * v - lr * g
        f_new = project_simplex(f + v, target_sum=1.0)
        f = f_new

        if it % 50 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
                lr = min(lr * 1.02, 0.02)
            else:
                no_improve += 1
                if no_improve > 10:
                    lr *= 0.9
                    no_improve = 0

            if it % 200 == 0:
                temp = min(temp * 1.2, 0.5)
                print(f"    {label} it={it} obj={best_obj:.10f} "
                      f"score={2.0*n*best_obj:.6f} t={time.monotonic()-t0:.0f}s")

            if no_improve > 30:
                # Restart with perturbation
                f = project_simplex(best_f + rng.standard_normal(n) * 0.01, target_sum=1.0)
                v = np.zeros(n)
                no_improve = 0
                lr = 0.005

    return best_f, best_obj


def coordinate_descent(f_init, time_limit, label="cd"):
    """Coordinate descent with multiple trial values."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    f = f / s if s > 1e-10 else f
    best_obj = fast_convolve(f).max()
    
    # Use multiplicative updates
    mults = [0.5, 0.7, 0.85, 1.0/0.85, 1.0/0.7, 1.0/0.5]
    
    improved_any = True
    rounds = 0
    while improved_any and time.monotonic() - t0 < time_limit:
        improved_any = False
        rounds += 1
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-15:
                continue
            best_j = f[j]
            best_j_obj = best_obj
            for m in mults:
                new_val = f[j] * m
                if new_val > 0.9999:
                    continue
                f[j] = new_val
                s = f.sum()
                if s < 1e-10:
                    f[j] = best_j
                    continue
                f_norm = f / s
                obj = fast_convolve(f_norm).max()
                f[j] = best_j
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j = new_val
            if best_j != f[j]:
                f[j] = best_j
                s = f.sum()
                f = f / s if s > 1e-10 else f
                best_obj = best_j_obj
                improved_any = True
        if rounds % 5 == 0:
            print(f"    {label} round={rounds} obj={best_obj:.10f} "
                  f"score={2.0*n*best_obj:.6f} t={time.monotonic()-t0:.0f}s")

    return f, best_obj


def parametric_init(n, family='gaussian', alpha=1.0):
    """Generate parametric initial sequences."""
    x = np.linspace(0, 1, n)
    if family == 'gaussian':
        a = np.exp(-alpha * (x - 0.5)**2 / 0.05)
    elif family == 'lorentzian':
        a = 1.0 / (1.0 + alpha * (x - 0.5)**2 / 0.05)
    elif family == 'exponential':
        a = np.exp(-alpha * np.abs(x - 0.5) / 0.2)
    elif family == 'flat':
        a = np.ones(n)
    elif family == 'power':
        a = np.exp(-alpha * np.abs(x - 0.5)**3 / 0.05)
    elif family == 'bimodal':
        a = np.exp(-alpha * (x - 0.25)**2 / 0.03) + np.exp(-alpha * (x - 0.75)**2 / 0.03)
    elif family == 'cosine':
        a = np.cos(np.pi * (x - 0.5))**2
        a = np.maximum(a, 0.0)
    else:
        a = np.ones(n)
    
    s = a.sum()
    if s > 1e-10:
        a = a / s
    return a

"""

Improves upon previous state-of-the-art (1.504861) by combining:

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def sym_to_full(half, n):
    """Expand symmetric half-sequence to full sequence."""
    if n % 2 == 0:
        return np.concatenate([half[::-1], half])
    else:
        return np.concatenate([half[:-1][::-1], half])


def full_to_sym(a, n):
    """Extract symmetric half from full sequence (average with reverse)."""
    if n % 2 == 0:
        return (a[:n//2] + a[n//2:][::-1]) / 2.0
    else:
        mid = n // 2
        h = (a[:mid] + a[mid+1:][::-1]) / 2.0
        h = np.concatenate([h, [a[mid]]])
        return h


def make_symmetric(a):
    """Make a sequence symmetric (palindrome)."""
    n = len(a)
    return (a + a[::-1]) / 2.0


def compute_score(a):
    """Compute the objective value for a non-negative sequence."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (s ** 2))


def lp_direction(f, top_k, use_sym=False, solver='ECOS'):
    """
    Solve LP to find best direction.
    If use_sym, f is the half-sequence and we optimize over half variables.
    Returns (g, M) where g is in the same space as f.
    """
    n = len(f)
    full_n = 2 * n if use_sym else n
    full_f = sym_to_full(f, full_n) if use_sym else f
    b = fast_convolve(full_f)
    M = b.max()
    if M < 1e-15:
        return None, M

    if top_k < len(b):
        top_idx = np.argpartition(b, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(b))
    top_idx = np.sort(top_idx)
    K = len(top_idx)

    # Build constraint matrix (K x full_n)
    A_con = np.zeros((K, full_n))
    for row_i in range(K):
        k = top_idx[row_i]
        lo = max(0, k - full_n + 1)
        hi = min(full_n - 1, k)
        A_con[row_i, lo:hi + 1] = full_f[lo:hi + 1]

    if use_sym:
        # Convert constraints to half-space
        # (A @ g_full)_i = (A @ [g_rev; g])_i
        # For symmetric g_full = [g_rev; g], we need A_half such that A_half @ g = A @ [g_rev; g]
        # (A @ [g_rev; g])[i] = sum_j A[i,j] * g_full[j]
        # g_full[j] = g[min(j, full_n-1-j)] for the symmetric case
        # Actually let's just build it directly
        A_half = np.zeros((K, n))
        for row_i in range(K):
            for j in range(n):
                # j in half maps to j in second half and full_n-1-j in first half
                A_half[row_i, j] = A_con[row_i, j] + (A_con[row_i, full_n - 1 - j] if full_n - 1 - j != j else 0.0)

        g_var = cp.Variable(n)
        g_nonneg = g_var >= 0
        constraints = [g_nonneg, A_half @ g_var <= M]
    else:
        A_half = A_con
        g_var = cp.Variable(n)
        constraints = [g_var >= 0, A_half @ g_var <= M]

    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
    try:
        prob.solve(solver=solver, eps=1e-8)
    except Exception:
        try:
            prob.solve(solver='SCS', max_iters=100000, eps=1e-8)
        except Exception:
            return None, M

    if g_var.value is None:
        return None, M
    g_val = np.maximum(g_var.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None, M
    return g_val, M

def coordinate_polish(f_init, time_limit, label="cd"):
    """Coordinate descent with multiplicative updates."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-10:
        f = f / s
    best_obj = fast_convolve(f).max()
    mults = [0.3, 0.5, 0.7, 0.85, 1.18, 1.43, 2.0, 3.33]
    rounds = 0
    improved_any = True

    while improved_any and time.monotonic() - t0 < time_limit:
        improved_any = False
        rounds += 1
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-15:
                continue
            orig = f[j]
            best_j = orig
            best_j_obj = best_obj
            for m in mults:
                new_val = orig * m
                if new_val > 0.9999 or new_val < 1e-15:
                    continue
                f[j] = new_val
                s = f.sum()
                if s < 1e-10:

                    continue

                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j = new_val
            if best_j != orig:
                f[j] = best_j
                s = f.sum()
                f = f / s if s > 1e-10 else f
                best_obj = best_j_obj
                improved_any = True

        if rounds % 3 == 0:
            print(f"    {label} r={rounds} obj={best_obj:.10f} "
                  f"score={2.0*n*best_obj:.6f} t={time.monotonic()-t0:.0f}s")

    return f, best_obj


def parametric_init(n, family, alpha=1.0):
    """Generate parametric initial sequences."""
    x = np.linspace(0, 1, n)
    xc = x - 0.5
    if family == 'gaussian':
        a = np.exp(-alpha * xc**2 / 0.05)
    elif family == 'lorentzian':
        a = 1.0 / (1.0 + alpha * xc**2 / 0.03)
    elif family == 'exponential':
        a = np.exp(-alpha * np.abs(xc) / 0.15)
    elif family == 'flat':
        a = np.ones(n)
    elif family == 'power3':
        a = np.exp(-alpha * np.abs(xc)**3 / 0.03)
    elif family == 'bimodal':
        a = np.exp(-alpha * (x - 0.25)**2 / 0.02) + np.exp(-alpha * (x - 0.75)**2 / 0.02)
    elif family == 'cosine':
        a = np.maximum(np.cos(np.pi * xc), 0.0)**2
    elif family == 'sinc':
        a = np.sinc(alpha * xc)
        a = np.abs(a)
    elif family == 'plateau':
        width = 1.0 / (1.0 + alpha * 0.1)
        a = np.where(np.abs(xc) < width / 2, 1.0, 0.1)
    else:
        a = np.ones(n)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-10:
        a = a / s
    return a

                if s > 1e-10:

                    score_i = compute_score(p_interp)
                    if score_i < best_score:
                        best_score = score_i
                        best_a = p_interp.copy()
                        best_n = target_n
    except Exception:
        pass

    return best_a, best_score, best_n


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main optimization: multi-phase search combining LP, subgradient,
    coordinate descent, and multi-length exploration.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(seed)

    # Phase 0: Find best starting point
    print("=== Phase 0: Initialization ===")
    candidate_ns = [512, 1024, 1152, 1536, 2048]
    best_init, init_score, init_n = find_best_start(candidate_ns, time_budget=30.0)

    if best_init is None:
        best_init = np.ones(1024) / 1024
        init_n = 1024
        init_score = compute_score(best_init)

    best_seq = best_init.copy()
    best_score = init_score
    print(f"  Best init: n={init_n}, score={best_score:.6f}")

    # Phase 1: LP optimization on best init (asymmetric)
    print(f"\n=== Phase 1: LP asymmetric (n={init_n}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.35, 550.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        f1, obj1 = run_lp_phase(best_init, time_limit=t1, top_k_start=15,
                                top_k_end=70, use_sym=False, label="lp_asym")
        score1 = compute_score(f1)
        if score1 < best_score:
            best_score = score1
            best_seq = f1.copy()
            print(f"  NEW BEST [lp_asym]: {score1:.6f}")

    # Phase 2: Symmetric LP from best
    print(f"\n=== Phase 2: LP symmetric (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.25, 350.0)
    if t2 > 30:
        n_b = len(best_seq)
        # Make symmetric and extract half
        sym_best = make_symmetric(best_seq)
        sym_best = sym_best / sym_best.sum()
        half = full_to_sym(sym_best, n_b)
        half = half / half.sum()
        print(f"  Budget: {t2:.0f}s, n_half={len(half)}")
        f2_half, obj2 = run_lp_phase(half, time_limit=t2, top_k_start=15,
                                     top_k_end=60, use_sym=True, label="lp_sym")
        full_2 = sym_to_full(f2_half, n_b)
        full_2 = full_2 / full_2.sum()
        score2 = compute_score(full_2)
        if score2 < best_score:
            best_score = score2
            best_seq = full_2.copy()
            print(f"  NEW BEST [lp_sym]: {score2:.6f}")

    # Phase 3: Subgradient polish
    print(f"\n=== Phase 3: Subgradient (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.2, 250.0)
    if t3 > 20:
        n_b = len(best_seq)
        print(f"  Budget: {t3:.0f}s, n={n_b}")
        f3, obj3 = subgradient_phase(best_seq, time_limit=t3, lr=0.003,
                                     momentum=0.9, temp=0.003,
                                     n_top=max(30, n_b // 15), label="sg")
        score3 = compute_score(f3)
        if score3 < best_score:
            best_score = score3
            best_seq = f3.copy()
            print(f"  NEW BEST [sg]: {score3:.6f}")

    # Phase 4: Try a different length (interpolate and LP)
    print(f"\n=== Phase 4: Multi-length (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    n_b = len(best_seq)
    alt_lengths = []
    for factor in [0.75, 1.25, 1.5]:
        tl = int(n_b * factor)
        tl = max(256, min(4096, tl))
        if tl != n_b and tl not in alt_lengths:
            alt_lengths.append(tl)

    for alt_n in alt_lengths[:2]:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        t4 = min(rem * 0.25, 200.0)
        if t4 < 25:
            break
        old_x = np.linspace(0, 1, n_b)
        new_x = np.linspace(0, 1, alt_n)
        f_interp = np.interp(new_x, old_x, best_seq)
        f_interp = np.maximum(f_interp, 0.0)
        s = f_interp.sum()
        f_interp = f_interp / s if s > 1e-10 else f_interp

        print(f"  LP n={alt_n}, budget={t4:.0f}s")
        f4, obj4 = run_lp_phase(f_interp, time_limit=t4, top_k_start=20,
                                top_k_end=60, use_sym=False, label=f"lp_n{alt_n}")
        score4 = compute_score(f4)
        if score4 < best_score:
            best_score = score4
            best_seq = f4.copy()
            print(f"  NEW BEST [n{alt_n}]: {score4:.6f}")

    # Phase 5: Coordinate descent polish
    print(f"\n=== Phase 5: Coordinate descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.15, 120.0)
    if t5 > 15:
        print(f"  Budget: {t5:.0f}s")
        f5, obj5 = coordinate_polish(best_seq, time_limit=t5, label="cd")
        score5 = compute_score(f5)
        if score5 < best_score:
            best_score = score5
            best_seq = f5.copy()
            print(f"  NEW BEST [cd]: {score5:.6f}")

    # Phase 6: Final LP burst
    print(f"\n=== Phase 6: Final LP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.5, 150.0)
    if t6 > 20:
        n_b = len(best_seq)
        print(f"  Budget: {t6:.0f}s, n={n_b}")
        f6, obj6 = run_lp_phase(best_seq, time_limit=t6, top_k_start=30,
                                top_k_end=100, use_sym=False, label="lp_final")
        score6 = compute_score(f6)
        if score6 < best_score:
            best_score = score6
            best_seq = f6.copy()
            print(f"  NEW BEST [lp_final]: {score6:.6f}")

    # Phase 7: Final coordinate descent
    rem = deadline - time.monotonic()
    t7 = min(rem * 0.3, 60.0)
    if t7 > 10:
        f7, obj7 = coordinate_polish(best_seq, time_limit=t7, label="cd_final")
        score7 = compute_score(f7)
        if score7 < best_score:
            best_score = score7
            best_seq = f7.copy()
            print(f"  NEW BEST [cd_final]: {score7:.6f}")

    # Final verification with evaluate_sequence
    fs = evaluate_sequence(best_seq.tolist())
    if np.isfinite(fs):
        best_score = fs
    else:
        # Fallback to parent
        try:
            p = np.asarray(height_sequence_1, dtype=np.float64)
            p = np.clip(p, 0.0, 1000.0)
            if p.ndim == 1 and p.size > 0:
                best_seq = p
                best_score = evaluate_sequence(best_seq.tolist())
        except Exception:
            best_seq = np.ones(1024) / 1024
            best_score = evaluate_sequence(best_seq.tolist())

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
