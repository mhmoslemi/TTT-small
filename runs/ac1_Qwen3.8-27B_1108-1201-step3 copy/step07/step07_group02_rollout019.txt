"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Nesterov-accelerated gradient descent on soft-max smoothed objective
   with FFT-based O(n log n) gradient computation
2. LP-based best response for game-theoretic refinement  
3. Aggressive local peak shaving with multi-coordinate adjustments
4. Temperature annealing and multiple restarts
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def softmax_from_z(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def z_from_a(a):
    """Log of a (for log-space parameterization)."""
    return np.log(np.maximum(a, 1e-300))

def cross_corr_for_grad(a, w):
    """
    Compute c[i] = sum_{m=0}^{n-1} a[m] * w[i+m] for i=0..n-1.
    This is the cross-correlation needed for the gradient.
    """
    n = len(a)
    fft_size = 1 << (3 * n - 2).bit_length()
    a_flip = a[::-1]
    c_full = irfft(rfft(a_flip, fft_size) * rfft(w, fft_size), fft_size)
    return c_full[n-1:n-1+n]

def compute_grad_z(a, T):
    """
    Compute gradient of soft-max objective w.r.t. z = log(a).
    Returns (objective_value, gradient).
    """
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    c = cross_corr_for_grad(a, w)
    grad_a = 4.0 * n * c
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z

def exact_score(a):
    """Compute the exact objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    a_norm = a / s
    return 2.0 * n * fast_convolve(a_norm).max()

def nesterov_optimize(a0, time_budget, T=30.0, lr_init=0.05, beta=0.9):
    """
    Nesterov-accederated gradient descent in log-space.
    Uses soft-max gradient direction but evaluates exact objective for line search.
    """
    n = len(a0)
    z = z_from_a(a0)
    v = np.zeros(n)
    lr = lr_init
    best_a = a0.copy()
    best_score = exact_score(a0)
    t0 = time.monotonic()
    iter_count = 0
    
    while time.monotonic() - t0 < time_budget:
        # Compute gradient at current point
        a = softmax_from_z(z)
        obj_soft, grad_z = compute_grad_z(a, T)
        
        # Nesterov momentum update

        z_new = z - lr * v
        a_new = softmax_from_z(z_new)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated gradient descent in log-space with FFT-based O(n log n) gradients
2. Soft-max smoothed objective with geometric temperature annealing
3. Aggressive multi-coordinate peak shaving for final refinement
4. Adaptive learning rate with improvement tracking
5. Multiple phases with escalating precision
"""

import numpy as np
import time
from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def softmax_from_z(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def z_from_a(a):
    """Convert probability vector to log-space."""
    return np.log(np.maximum(a, 1e-300))

def cross_corr_for_grad(a, w):
    """
    Compute c[i] = sum_{m=0}^{n-1} a[m] * w[i+m] for i=0..n-1.
    Used in gradient computation. O(n log n) via FFT.
    """
    n = len(a)
    # Linear conv of a_flip (len n) and w (len 2n-1) has length 3n-2
    fft_size = 1 << (3 * n - 2).bit_length()
    a_flip = a[::-1]
    c_full = irfft(rfft(a_flip, fft_size) * rfft(w, fft_size), fft_size)
    return c_full[n - 1:n - 1 + n]

def compute_grad_z(a, T):
    """
    Compute soft-max objective value and gradient w.r.t. z=log(a).
    Returns (objective, gradient_vector).
    """
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    c = cross_corr_for_grad(a, w)
    grad_a = 4.0 * n * c
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z

def exact_score(a):
    """Compute exact objective: 2n * max(conv(a,a)) / (sum a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    a_norm = a / s
    return 2.0 * n * fast_convolve(a_norm).max()

    """

    v = np.zeros(n)
    lr = lr_init
    best_a = a0.copy()

    step_count = 0
    improve_count = 0
    total_steps = 0

    if max_steps is None:
        max_steps = 1000000

    while time.monotonic() - t0 < time_budget and total_steps < max_steps:

        soft_obj, grad_z = compute_grad_z(a, T)

        a_new = softmax_from_z(z_new)

        improved_this_sweep = False
        for i in range(n):
            if time.monotonic() - t0 > time_budget:
                return best_a, best_score
            if a[i] < 1e-14:
                continue
            # Try a few multiplicative adjustments
            base_sum_others = a.sum() - a[i]
            if base_sum_others < 1e-14:
                continue
            for factor in [0.7, 0.85, 1.0, 1.15, 1.35]:
                new_val = a[i] * factor
                if new_val < 0:
                    continue
                a_new = a.copy()
                a_new[i] = new_val
                a_new = a_new / a_new.sum()
                score = exact_score(a_new)
                if score < best_score - 1e-13:
                    a = a_new
                    best_score = score
                    best_a = a_new.copy()
                    improved_this_sweep = True
        sweep += 1
        if not improved_this_sweep and sweep > 3:
            break
        if sweep % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordSweep sweep={sweep} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
