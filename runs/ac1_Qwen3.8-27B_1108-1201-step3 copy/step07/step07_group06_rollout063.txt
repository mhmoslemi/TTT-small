Let me think about the peak-equalization more carefully:

The key insight is: at the optimum, multiple peaks should be "active" (at the maximum). If only one peak is active, we can usually reduce it further by adjusting the sequence.

So the algorithm should:
1. Find all peaks that are within epsilon of the maximum
2. For each such peak, identify contributing pairs
3. Try to reduce all active peaks simultaneously
4. The way to do this is to find a direction in which all active peaks decrease

This is like solving a system of inequalities:
`conv(a + t*d, a + t*d)[k] <= current_max` for all active k

For small t, this is approximately:
`2 * sum_i (a[i]*d[k-i] + d[i]*a[k-i]) * t <= 0` for all active k

This is a linear constraint on d. We can find a direction d that satisfies all these constraints (if one exists) using linear programming or a simple heuristic.

But this might be overcomplicating things. Let me stick with the simpler approach of trying individual pair adjustments and see if that's sufficient.

Actually, here's a simpler but effective approach for the local search:

**Pairwise adjustment search:**
1. Compute conv(a,a) and find the top K peaks
2. For each peak at lag k, the top contributing pairs are (i, k-i) where a[i]*a[k-i] is large
3. For each such pair, try reducing one and increasing the other (or both)
4. Also try: reduce one element and distribute its mass to random other elements
5. Accept any move that reduces the overall max

The key is to make the search space large enough. With n=768, there are ~768*767/2 ≈ 294k pairs to consider, but we only need to look at the top contributing ones for each peak.

def peak_equalize(a0, time_limit, n_trials=100000):
    """
    Intensive local search: identify peaks and try pairwise adjustments
    to reduce the maximum convolution value.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    trial = 0
    stall = 0
    max_stall = 2000
    
    while trial < n_trials and time.monotonic() - t0 < time_limit:
        trial += 1
        b = fast_convolve(a)
        peak_val = b.max()
        
        if peak_val * 2 * n <= best_score * 0.999:
            # Shouldn't happen but safety check
            pass
        
        # Find peaks that are close to the maximum
        threshold = peak_val * 0.999
        peak_indices = np.where(b >= threshold)[0]
        
        # For stochastic selection, pick a random peak
        if len(peak_indices) > 0:
            if rng.random() < 0.7:
                # Pick the global peak
                k = int(np.argmax(b))
            else:
                # Pick a random near-peak
                k = int(peak_indices[rng.integers(len(peak_indices))])
        else:
            k = int(np.argmax(b))
        
        # Elements contributing to this peak
        # b[k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[i]*a[k-i]
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        
        if i_start > i_end:
            # No valid pairs, skip
            stall += 1
            continue
        
        indices = np.arange(i_start, i_end + 1)
        partners = k - indices
        contribs = a[indices] * a[partners]
        
        # Sort by contribution (largest first)
        order = np.argsort(contribs)[::-1]
        
        improved = False
        n_top = min(8, len(order))
        
        for ci in range(n_top):
            if improved:
                break
            i = int(indices[order[ci]])
            j = int(partners[order[ci]])
            if i == j:
                continue
            if a[i] < 1e-12 or a[j] < 1e-12:
                continue
            
            # Try reducing the larger of the two and increasing the smaller
            # Or try various adjustments
            if a[i] > a[j]:
                big, small = i, j
            else:
                big, small = j, i
            
            # Various adjustment strategies
            strategies = [
                # Reduce big, increase small
                (big, small, 0.95, 1.02),
                (big, small, 0.9, 1.05),
                (big, small, 0.85, 1.1),
                (big, small, 0.8, 1.15),
                # Reduce big only (redistribute to others)
                (big, -1, 0.9, 1.0),
                (big, -1, 0.85, 1.0),
                # Reduce both
                (big, small, 0.95, 0.95),
            ]
            
            for big_i, small_i, f_big, f_small in strategies:
                a_trial = a.copy()
                a_trial[big_i] *= f_big
                diff = a[big_i] - a_trial[big_i]
                if small_i >= 0:
                    a_trial[small_i] += diff
                else:
                    # Redistribute to random elements
                    if diff > 0:
                        # Add to random element
                        target = int(rng.integers(n))
                        a_trial[target] += diff
                
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                
                conv_trial = fast_convolve(a_trial)
                score_trial = 2.0 * n * conv_trial.max()
                
                if score_trial < best_score - 1e-13:
                    a = a_trial
                    best_score = score_trial
                    best_a = a_trial.copy()
                    improved = True
                    break
        
        if not improved:
            stall += 1
            # Periodic random kick
            if stall > max_stall and stall % 500 == 0:
                a = best_a + rng.standard_normal(n) * 0.002
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
            elif stall > max_stall * 2:
                # Bigger kick
                a = best_a + rng.standard_normal(n) * 0.01
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
        else:
            stall = 0
    
    return best_a, best_score


def penalty_method(a0, time_limit, M_init=None, M_final=None):
    """
    Penalty method: minimize sum_k max(0, conv(a,a)[k] - M)^2
    with M decreasing over time.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    
    if M_init is None:
        M_init = 1.1 * fast_convolve(a).max()
    if M_final is None:
        M_final = 0.95 * fast_convolve(a).max()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(99)
    lr = 0.01
    epoch = 0
    max_epochs = 200
    
    while epoch < max_epochs and time.monotonic() - t0 < time_limit:
        epoch += 1
        progress = epoch / max_epochs
        M = M_init + (M_final - M_init) * progress
        
        # Sub-gradient descent on penalty function
        for inner in range(50):
            if time.monotonic() - t0 > time_limit:
                break
            
            conv = fast_convolve(a)

def coord_descent_fine(a0, time_limit, n_sweeps=500):
    """Fine-grained coordinate descent with adaptive step sizes."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    
    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        
        order = rng.permutation(n)
        improved_this_sweep = False
        steps_no_improve = 0
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue
            
            # Try a range of multiplicative factors
            best_factor = 1.0
            best_local_score = best_score
            
            # Adaptive: try finer steps if close to optimal
            factors = [0.95, 0.97, 0.99, 1.01, 1.03, 1.05]
            
            for f in factors:
                a_trial = a.copy()
                a_trial[i] *= f
                diff = a[i] * (f - 1.0)
                if f < 1.0:
                    # Redistribute removed mass
                    others = np.ones(n)
                    others[i] = 0
                    others = others * a_trial
                    others_sum = others.sum()
                    if others_sum > 1e-15:
                        a_trial += diff * others / others_sum
                # If f > 1, sum increases, we'll normalize
                
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_local_score - 1e-14:
                    best_local_score = score
                    best_factor = f
            
            if best_factor != 1.0:
                a[i] *= best_factor
                diff = a[i] * (best_factor - 1.0) - (a[i] / best_factor) * (best_factor - 1.0)
                # Just re-normalize
                a = np.maximum(a, 0)
                a = a / a.sum()
                if best_local_score < best_score:
                    best_score = best_local_score
                    best_a = a.copy()
                    improved_this_sweep = True
                    steps_no_improve = 0
                else:
                    steps_no_improve += 1
            else:
                steps_no_improve += 1
        
        if not improved_this_sweep and sweep > 10:
            # Try a random perturbation
            if sweep % 20 == 0:
                idx = rng.choice(n, size=min(5, n), replace=False)
                a_pert = a.copy()

                if score < best_score - 1e-13:
                    a = a_pert
                    best_score = score
                    best_a = a.copy()
    
    return best_a, best_score


def simulate_anneal(a0, time_limit, T_init=0.01, T_final=0.0001, n_iter=50000):
    """Simulated annealing for final micro-polish."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        progress = it / max(n_iter, 1)
        T = T_init * (T_final / T_init) ** progress
        
        # Random move: pick two elements, transfer mass
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        if a[i] < 1e-12:
            continue
        
        # Transfer a small fraction from i to j
        frac = rng.uniform(0.001, 0.01)
        amount = a[i] * frac
        
        a_trial = a.copy()
        a_trial[i] -= amount
        a_trial[j] += amount
        a_trial = np.maximum(a_trial, 0)
        a_trial = a_trial / a_trial.sum()
        
        conv_trial = fast_convolve(a_trial)
        score_trial = 2.0 * n * conv_trial.max()
        delta = score_trial - best_score
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_trial
            if score_trial < best_score - 1e-14:
                best_score = score_trial
                best_a = a_trial.copy()
    
    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""

Phase 1: Diverse multi-start with softmax-Adam gradient descent (correct FFT-based gradients)
Phase 2: Focused refinement with adaptive temperature

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Fast linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_norm(a):
    """Compute score for sequence with sum=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def softmax_from_z(z):
    """Convert log-domain to probability vector."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def compute_gradient(a, w, fft_size):
    """
    Compute gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a.
    grad_a[j] = 2 * sum_{m=0}^{n-1} w[j+m] * a[m]
    Computed via FFT cross-correlation.
    """
    n = len(a)
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    corr = irfft(W * np.conj(A), fft_size)[:n]
    return 2.0 * corr


def gradient_descent_softmax(a0, time_limit, T_init=0.03, T_final=0.001,
                             lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8,
                             schedule='exp', restart_interval=5000,
                             restart_scale=0.003):
    """Adam in log-space with softmax temperature annealing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-300)
    a = a / a.sum()
    z = z_from_a(a)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    no_improve = 0
    max_no_improve = restart_interval
    current_lr = lr

    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / max(time_limit, 1e-9))

        if schedule == 'exp':
            T = T_init * (T_final / T_init) ** progress
        elif schedule == 'hold_anneal':
            if progress < 0.25:
                T = T_init
            else:
                p2 = (progress - 0.25) / 0.75
                T = T_init * (T_final / T_init) ** p2
        elif schedule == 'aggressive':
            if progress < 0.1:
                T = T_init
            elif progress < 0.5:
                p2 = (progress - 0.1) / 0.4
                T = T_init * (T_final / T_init) ** (p2 * 0.5)
            else:
                p3 = (progress - 0.5) / 0.5
                T = (T_init * (T_final / T_init) ** 0.5) * (T_final / (T_init * (T_final / T_init) ** 0.5)) ** p3
        else:
            T = T_init * (T_final / T_init) ** progress

        a_curr = softmax_from_z(z)
        b = fast_convolve(a_curr)
        max_b = b.max()
        exp_b = np.exp(b / T - max_b / T)
        w = exp_b / exp_b.sum()

        grad_a = compute_gradient(a_curr, w, fft_size)
        S = np.dot(grad_a, a_curr)
        grad_z = a_curr * (grad_a - S)

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z = z - current_lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = softmax_from_z(z)
        conv_new = fast_convolve(a_new)
        score = 2.0 * n * conv_new.max()

        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            z = z_from_a(best_a) + np.random.standard_normal(n) * restart_scale
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            current_lr = max(0.001, current_lr * 0.6)
            restart_scale *= 0.7

        current_lr = max(0.0003, current_lr * 0.99995)

    return best_a, best_score


def structured_peak_search(a0, time_limit, n_trials=100000, top_k_peaks=5):
    """
    Structured local search exploiting active-set structure.
    Key idea: find elements contributing to multiple active peaks and reduce them.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    trial = 0
    stall = 0
    max_stall = 3000

    while trial < n_trials and time.monotonic() - t0 < time_limit:
        trial += 1
        if trial % 1000 == 0 and time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        peak_val = b.max()
        threshold = peak_val * (1.0 - 0.001)
        active_peaks = np.where(b >= threshold)[0]

        improved = False

        if len(active_peaks) >= 2:
            # Find elements contributing to multiple active peaks
            peak_contrib = {}
            for k in active_peaks[:top_k_peaks]:
                i_start = max(0, k - n + 1)
                i_end = min(n - 1, k)
                for i in range(i_start, i_end + 1):
                    j = k - i
                    contrib = a[i] * a[j]
                    if contrib > 0:
                        if i not in peak_contrib:
                            peak_contrib[i] = 0.0
                        peak_contrib[i] += contrib
                        if j != i and j not in peak_contrib:
                            peak_contrib[j] = 0.0
                        if j != i:
                            peak_contrib[j] += contrib

            # Sort elements by total contribution to active peaks
            if peak_contrib:
                sorted_elems = sorted(peak_contrib.keys(), key=lambda x: peak_contrib[x], reverse=True)
                n_try = min(5, len(sorted_elems))
                for ci in range(n_try):
                    if improved:
                        break
                    elem = sorted_elems[ci]
                    if a[elem] < 1e-12:
                        continue

                    # Try reducing this element and redistributing
                    for factor in [0.95, 0.9, 0.85]:
                        if improved:
                            break
                        a_trial = a.copy()
                        diff = a_trial[elem] * (1.0 - factor)
                        a_trial[elem] *= factor

                        # Redistribute to least-contributing elements
                        others = np.ones(n)
                        others[elem] = 0
                        # Weight redistribution by inverse contribution
                        weights = np.ones(n)
                        for pk in active_peaks[:3]:
                            i_s = max(0, pk - n + 1)
                            i_e = min(n - 1, pk)
                            for ii in range(i_s, i_e + 1):
                                weights[ii] += a[ii] * a[pk - ii]
                        weights[elem] = 0
                        weights = others * np.maximum(1.0 / (weights + 1e-10), 0)
                        w_sum = weights.sum()
                        if w_sum < 1e-15:
                            weights = others
                            w_sum = weights.sum()
                        if w_sum < 1e-15:
                            continue
                        a_trial += diff * weights / w_sum

                        a_trial = np.maximum(a_trial, 0)
                        ss = a_trial.sum()
                        if ss < 1e-15:
                            continue
                        a_trial = a_trial / ss
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-13:
                            a = a_trial
                            best_score = score
                            best_a = a_trial.copy()
                            improved = True

        # Also try direct pairwise adjustments on the global peak
        if not improved:
            k = int(np.argmax(b))
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            indices = np.arange(i_start, i_end + 1)
            partners = k - indices
            contribs = a[indices] * a[partners]
            order = np.argsort(contribs)[::-1]

            n_top = min(6, len(order))
            for ci in range(n_top):
                if improved:
                    break
                i = int(indices[order[ci]])
                j = int(partners[order[ci]])
                if i == j or a[i] < 1e-12 or a[j] < 1e-12:
                    continue

                # Various moves
                moves = []
                if a[i] > a[j]:
                    moves.extend([(i, j, 0.95, 1.01), (i, j, 0.9, 1.03),
                                  (i, j, 0.85, 1.05), (i, -1, 0.92, 1.0),
                                  (i, -1, 0.87, 1.0)])
                else:
                    moves.extend([(j, i, 0.95, 1.01), (j, i, 0.9, 1.03),
                                  (j, i, 0.85, 1.05), (j, -1, 0.92, 1.0),
                                  (j, -1, 0.87, 1.0)])

                for big, small, f1, f2 in moves:
                    if improved:
                        break
                    a_trial = a.copy()
                    a_trial[big] *= f1
                    diff = a[big] - a_trial[big]
                    if small >= 0:
                        a_trial[small] += diff
                    else:
                        target = int(rng.integers(n))
                        while target == big:
                            target = int(rng.integers(n))
                        a_trial[target] += diff

                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a_trial.copy()
                        improved = True

        # Skip lag-0 and lag-(2n-2) peaks (they're just a[0]^2 and a[n-1]^2)
        # These are hard to reduce without increasing other peaks

        if not improved:
            stall += 1
            if stall > max_stall:
                # Random restart
                scale = 0.002 + 0.003 * min(1.0, stall / (max_stall * 4))
                a = best_a + rng.standard_normal(n) * scale
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
        else:
            stall = 0

    return best_a, best_score


def penalty_descent(a0, time_limit, n_epochs=100, inner_iters=30):
    """
    Penalty method: minimize sum_k max(0, conv[k]-M)^2 with decreasing M.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (3 * n - 2 - 1).bit_length()

    initial_peak = fast_convolve(a).max()
    M_start = initial_peak * 1.01
    M_end = initial_peak * 0.985
    lr = 0.02

    for epoch in range(n_epochs):
        if time.monotonic() - t0 > time_limit:
            break
        progress = epoch / max(n_epochs - 1, 1)
        M = M_start + (M_end - M_start) * progress

        for _ in range(inner_iters):
            if time.monotonic() - t0 > time_limit:
                break

            conv = fast_convolve(a)
            excess = np.maximum(conv - M, 0)
            active = excess  # (conv[k]-M) where conv[k]>M, else 0

            # Gradient: 4 * cross(active, a)
            A = rfft(a, fft_size)
            Act = rfft(active, fft_size)
            cross = irfft(Act * np.conj(A), fft_size)[:n]
            grad = 4.0 * cross

            # Normalization gradient (to maintain sum=1 after clipping)
            a = a - lr * grad
            a = np.maximum(a, 0)
            ss = a.sum()
            if ss < 1e-15:
                a = np.ones(n) / n
            else:
                a = a / ss

            score = 2.0 * n * fast_convolve(a).max()
            if score < best_score - 1e-14:
                best_score = score
                best_a = a.copy()

        lr *= 0.92
        if lr < 0.0005:
            break

    return best_a, best_score


def fine_coord_descent(a0, time_limit, n_sweeps=300):
    """Adaptive coordinate descent with fine steps."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()

    rng = np.random.default_rng(31415)

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        order = rng.permutation(n)
        sweep_best = best_score

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-13:
                continue

            # Try multiplicative factors (finer near convergence)
            if best_score < 1.55:
                factors = [0.97, 0.985, 0.995, 1.005, 1.015, 1.03]
            else:
                factors = [0.9, 0.95, 1.05, 1.1]

            best_f = 1.0
            best_sc = sweep_best

            for f in factors:
                a_trial = a.copy()
                a_trial[i] *= f
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = 2.0 * n * fast_convolve(a_trial).max()
                if sc < best_sc - 1e-14:
                    best_sc = sc
                    best_f = f

            if best_f != 1.0:
                a[i] *= best_f
                a = np.maximum(a, 0)
                a = a / a.sum()
                if best_sc < best_score:
                    best_score = best_sc
                    best_a = a.copy()
                    sweep_best = best_score

    return best_a, best_score


def simulated_anneal(a0, time_limit, T_init=0.005, T_final=0.0001, n_iter=30000):
    """Simulated annealing for micro-polish."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(27182)

    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break

        progress = it / max(n_iter, 1)
        T = T_init * (T_final / T_init) ** progress

        # Move: transfer mass between two random elements
        i = int(rng.integers(n))
        if a[i] < 1e-12:
            continue
        j = int(rng.integers(n))
        if i == j:
            continue

        frac = rng.uniform(0.002, 0.02)
        amount = a[i] * frac

        a_trial = a.copy()
        a_trial[i] -= amount
        a_trial[j] += amount
        a_trial = np.maximum(a_trial, 0)
        ss = a_trial.sum()
        if ss < 1e-15:
            continue
        a_trial = a_trial / ss

        sc = 2.0 * n * fast_convolve(a_trial).max()
        delta = sc - best_score

        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            a = a_trial
            if sc < best_score - 1e-14:
                best_score = sc
                best_a = a_trial.copy()

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def try_update(a, best_seq, best_score, label):
    """Evaluate candidate and return updated best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        sc = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {sc:.6f}")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining gradient descent, structured local search,
    penalty methods, and simulated annealing to minimize the peak convolution score.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"No parent: {e}")

    if best_seq is None:
        n_default = 768
        best_seq = np.ones(n_default) / n_default
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Default: n={n_default}, score={best_score:.6f}")

    n = len(best_seq)
    print(f"Starting: score={best_score:.6f}, n={n}")

    # === Phase 1: Multi-start gradient descent (30% time) ===
    print("\n=== Phase 1: Multi-Start Gradient Descent ===")
    t_p1 = min((deadline - time.monotonic()) * 0.30, 540.0)

    # Generate diverse starts
    starts = []
    starts.append(("parent", best_seq.copy()))

    for scale in [0.03, 0.06, 0.1]:
        pert = best_seq + rng.standard_normal(n) * scale
        pert = np.maximum(pert, 0.0)
        pert = pert / pert.sum()
        starts.append((f"pert{scale}", pert))

    # Fourier modulation
    for freq in [2, 3]:
        t_arr = np.linspace(0, 2 * np.pi * freq, n)
        f_mod = 1.0 + 0.2 * np.cos(t_arr)
        f_mod = np.maximum(f_mod, 0.01)
        f_mod = f_mod / f_mod.sum()
        starts.append((f"fourier{freq}", f_mod))

    # Cosine window
    t_arr = np.linspace(0, np.pi, n)
    cos_w = np.cos(t_arr) ** 2 + 0.1
    cos_w = cos_w / cos_w.sum()
    starts.append(("cosine", cos_w))

    # Random
    r = np.abs(rng.standard_normal(n)) + 0.05
    r = r / r.sum()
    starts.append(("random", r))

    # Sort by score
    scored = []
    for label, s_arr in starts:
        sc = 2.0 * n * fast_convolve(s_arr).max()
        scored.append((sc, label, s_arr))
    scored.sort(key=lambda x: x[0])

    for sc, label, _ in scored:
        print(f"  Init {label}: {sc:.6f}")

    time_per_start = t_p1 / 5
    configs = [
        (0.04, 0.001, 0.01, 'exp'),
        (0.03, 0.0008, 0.008, 'hold_anneal'),
        (0.025, 0.001, 0.007, 'aggressive'),
        (0.05, 0.002, 0.012, 'exp'),
        (0.02, 0.0005, 0.006, 'hold_anneal'),
    ]

    for idx in range(min(5, len(scored))):
        rem = deadline - time.monotonic() - 400
        if rem < 45:
            break
        t_alloc = min(time_per_start, rem * 0.9)
        if t_alloc < 25:
            continue

        sc_init, label, init_a = scored[idx]
        T_i, T_f, lr_c, sched = configs[idx % len(configs)]
        print(f"  GD {label}: budget={t_alloc:.0f}s, T={T_i}->{T_f}, lr={lr_c}")

        a_res, s_res = gradient_descent_softmax(
            init_a, t_alloc, T_init=T_i, T_final=T_f, lr=lr_c, schedule=sched
        )
        best_seq, best_score = try_update(a_res, best_seq, best_score, f"gd_{label}")

    print(f"  After Phase 1: {best_score:.6f}")

    # === Phase 2: Focused refinement (20% time) ===
    print(f"\n=== Phase 2: Focused Refinement ===")
    rem = deadline - time.monotonic() - 500
    if rem > 50:
        t_alloc = min(rem * 0.6, 300.0)
        print(f"  Long GD run, budget={t_alloc:.0f}s")
        a_res, s_res = gradient_descent_softmax(
            best_seq, t_alloc, T_init=0.015, T_final=0.0005,
            lr=0.005, schedule='hold_anneal'
        )
        best_seq, best_score = try_update(a_res, best_seq, best_score, "gd_focused")
        print(f"  After Phase 2: {best_score:.6f}")

    # === Phase 3: Structured local search (30% time) ===
    print(f"\n=== Phase 3: Structured Peak Search ===")
    rem = deadline - time.monotonic() - 250
    if rem > 40:
        t_alloc = min(rem * 0.65, 450.0)
        print(f"  Peak search, budget={t_alloc:.0f}s")
        a_res, s_res = structured_peak_search(best_seq, t_alloc, n_trials=80000)
        best_seq, best_score = try_update(a_res, best_seq, best_score, "peak_search")
        print(f"  After Phase 3: {best_score:.6f}")

    # === Phase 4: Penalty method (10% time) ===
    print(f"\n=== Phase 4: Penalty Method ===")
    rem = deadline - time.monotonic() - 120
    if rem > 30:
        t_alloc = min(rem * 0.6, 150.0)
        print(f"  Penalty, budget={t_alloc:.0f}s")
        a_res, s_res = penalty_descent(best_seq, t_alloc, n_epochs=80, inner_iters=25)
        best_seq, best_score = try_update(a_res, best_seq, best_score, "penalty")
        print(f"  After Phase 4: {best_score:.6f}")

    # === Phase 5: Intensive final polish (10% time) ===
    print(f"\n=== Phase 5: Final Polish ===")
    rem = deadline - time.monotonic() - 20

    # Alternate between different polishers
    for round_num in range(4):
        rem = deadline - time.monotonic() - 15
        if rem < 8:
            break

        # Peak search
        t_alloc = rem * 0.3
        a_res, _ = structured_peak_search(best_seq, t_alloc, n_trials=15000)
        best_seq, best_score = try_update(a_res, best_seq, best_score, f"polish_pe{round_num}")

        rem = deadline - time.monotonic() - 15
        if rem < 8:
            break

        # Coord descent
        t_alloc = rem * 0.3
        a_res, _ = fine_coord_descent(best_seq, t_alloc, n_sweeps=150)
        best_seq, best_score = try_update(a_res, best_seq, best_score, f"polish_cd{round_num}")

        rem = deadline - time.monotonic() - 10
        if rem < 5:
            break

        # Simulated annealing
        t_alloc = rem * 0.5
        a_res, _ = simulated_anneal(best_seq, t_alloc, T_init=0.003, T_final=0.0001, n_iter=5000)
        best_seq, best_score = try_update(a_res, best_seq, best_score, f"polish_sa{round_num}")

        print(f"  Round {round_num}: {best_score:.6f}")

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: {best_score:.6f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK: {best_score:.6f}")

    return [float(x) for x in best_seq]
