"""
Convex optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

Uses adaptive threshold-based LP directions with momentum, careful line search,
multi-length exploration, and polishing phases.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    # Scale to sum=1 first
    u = v.copy()
    # Find theta such that sum(max(u - theta, 0)) = 1
    u_sorted = np.sort(u)[::-1]
    cssv = np.cumsum(u_sorted)
    rho_vals = np.arange(1, n + 1)
    cond = u_sorted * rho_vals > (cssv - 1.0)
    if not cond.any():
        result = np.ones(n) / n
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        result = np.maximum(u - theta, 0.0)
    return result * target_sum


def lp_direction(f, n, threshold_frac=0.95, max_constraints=300, min_constraints=10):
    """
    Find LP direction: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for peaks above threshold.
    Returns (g, M, num_constraints) or (None, M, 0).
    """
    b = fast_convolve(f)
    M = b.max()
    if M < 1e-15:
        return None, M, 0

    # Select peaks above threshold
    threshold = threshold_frac * M
    sel_idx = np.where(b >= threshold)[0]
    
    if len(sel_idx) < min_constraints:
        # Fall back to top-K
        k = min_constraints
        sel_idx = np.argpartition(b, -k)[-k:]
    
    if len(sel_idx) > max_constraints:
        # Take the top max_constraints
        sel_idx = np.argpartition(b, -max_constraints)[-max_constraints:]
    
    sel_idx = np.sort(sel_idx)
    K = len(sel_idx)

    # Build constraint matrix
    A = np.zeros((K, n))
    for row_i in range(K):
        k = sel_idx[row_i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]

    g = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [g >= 0, A @ g <= M])
    try:
        prob.solve(solver=cp.SCS, max_iters=200000, eps=1e-8)
    except Exception:
        return None, M, K

    if g.value is None:
        return None, M, K
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None, M, K
    return g_val, M, K


def line_search(f, direction, n, max_obj, max_iter=30):
    """
    Find best step size t in [0, 1] such that obj((1-t)*f + t*direction) is minimized.
    Uses golden section search.
    """
    invphi = (np.sqrt(5.0) - 1.0) / 2.0
    
    def obj_at_t(t):
        f_trial = (1.0 - t) * f + t * direction
        f_trial = np.maximum(f_trial, 0.0)
        s = f_trial.sum()
        if s < 1e-10:
            return max_obj
        f_trial = f_trial / s
        return fast_convolve(f_trial).max()
    
    # First try a few fixed points to bracket
    candidates = [0.01, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7]
    best_t = 0.0
    best_val = obj_at_t(0.0)
    for t in candidates:
        v = obj_at_t(t)
        if v < best_val:
            best_val = v
            best_t = t
    
    # Refine with golden section around best_t
    lo = max(0.0, best_t - 0.1)
    hi = min(1.0, best_t + 0.1)
    
    for _ in range(max_iter):
        c = hi - invphi * (hi - lo)
        d = lo + invphi * (hi - lo)
        fc = obj_at_t(c)
        fd = obj_at_t(d)
        if fc < fd:
            hi = d
        else:
            lo = c
        if hi - lo < 1e-5:
            break
    
    best_t = (lo + hi) / 2.0
    best_val = obj_at_t(best_t)
    return best_t, best_val

    it = 0
    stuck = 0

        it += 1

        if g is None:

            g, M, K = lp_direction(f, n, threshold_frac=1.0 - 1e-6, max_constraints=1, min_constraints=1)
            if g is None:
                break
        
        S_g = g.sum()
        if S_g <= 1.0 + 1e-5:
            stuck += 1

                stuck = 0
                continue
            continue

            direction = h

                break
            continue
        
        # Update

def multiplicative_update(f, n_peaks=5, lr=0.1, time_limit=None, t0=None):
    """
    Multiplicative weight update: reduce elements that contribute to peak convolutions.
    """
    t0 = t0 or time.monotonic()
    n = len(f)
    f = f.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    it = 0
    stuck = 0
    
    while (time_limit is None or time.monotonic() - t0 < time_limit):
        it += 1
        
        b = fast_convolve(f)
        
        # Get top n_peaks indices
        if n_peaks < len(b):
            top_idx = np.argpartition(b, -n_peaks)[-n_peaks:]
        else:
            top_idx = np.arange(len(b))
        
        # Compute gradient contribution
        grad = np.zeros(n)
        for k in top_idx:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                # (a*a)[k] = sum_j a[j] * a[k-j], gradient wrt a[j] is a[k-j]
                grad[j] += f[k - j]
        
        # Normalize gradient
        grad_max = grad.max()
        if grad_max < 1e-15:
            break
        
        # Multiplicative update: a_new[j] = a[j] * (1 - lr * grad[j] / grad_max)
        multipliers = 1.0 - lr * grad / grad_max
        multipliers = np.maximum(multipliers, 0.1)  # Don't reduce too much
        f_new = f * multipliers
        
        # Project to simplex
        s = f_new.sum()
        if s < 1e-10:
            break
        f = f_new / s
        
        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 20:
                break
        
        if it % 50 == 0:
            print(f"    mu-update iter={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f}")
    
    return best_f, best_obj


def coordinate_descent_polish(f_init, time_limit):
    """Coordinate descent with binary search on each coordinate."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    
    improved = True
    passes = 0
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        passes += 1
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-14:
                continue
            
            # Binary search for optimal value of f[j]
            lo, hi = 0.0, min(f[j] * 3.0, 0.99)
            best_j_val = f[j]
            best_j_obj = best_obj
            best_trial = f.copy()
            
            # Try current value
            for mult in [0.3, 0.5, 0.7, 1.0, 1.3, 1.7, 2.5]:
                new_val = f[j] * mult
                if new_val > 0.999:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial = f_trial / s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = new_val
                    best_trial = f_trial
            
            if best_j_val != f[j]:
                f = best_trial
                if best_j_obj < best_obj - 1e-12:
                    best_obj = best_j_obj
                    improved = True
    
    return f, best_obj

Let me think about this differently. The problem is equivalent to: minimize max_k (a*a)[k] subject to a >= 0, sum(a) = 1.

This is a convex optimization problem (minimize a convex function over a convex set). The optimal solution exists and is unique (up to symmetry).

By duality, the KKT conditions give:
- There exist Lagrange multipliers lambda_k >= 0 for each k such that:
  2*sum_k lambda_k * a_{k-j} + nu = 0 for all j with a_j > 0 (complementary slackness: lambda_k * ((a*a)[k] - M) = 0)
  where M is the optimal value of max_k (a*a)[k].

So at the optimum, for each j with a_j > 0:
sum_{k: lambda_k > 0} a_{k-j} = constant (independent of j)

This means: the "weighted sum" of a, weighted by the active Lagrange multipliers, is constant on the support of a.

This is a very useful characterization! It tells us that at the optimum, the sequence a is "eigen-like" with respect to a certain weighted autocorrelation operator.

This suggests an iterative algorithm:
1. Start with a guess for a.
2. Compute the convolution b = (a*a).
3. Identify the "active" set of peaks (those at the maximum).
4. Solve for a such that the weighted sum condition holds.
5. Iterate.

This is essentially what the LP method does, but we can make it more direct.

Actually, the condition "sum_{k: lambda_k > 0} a_{k-j} = constant" is exactly what the LP is finding! The LP direction g satisfies (f*g)[k] = M for the active peaks, which means the "gradient" is aligned.

So the LP method is essentially implementing the KKT conditions. The question is convergence speed.
