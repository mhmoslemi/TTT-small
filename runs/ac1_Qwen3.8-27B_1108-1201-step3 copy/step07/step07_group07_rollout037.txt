import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax(x):
    """Numerically stable softmax."""
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()


def compute_grad_smoothed(a, w):
    """
    Compute gradient of logsumexp(beta*b)/beta with respect to a,
    given weights w (normalized exp(beta*b)/sum(exp(beta*b))).
    Result: g[i] = sum_j w[i+j] * a[j]
    """
    n = len(a)
    w_rev = w[::-1]
    # g[i] = conv(w_rev, a)[2n-2-i]
    # Use FFT
    fft_size = 1 << (3 * n - 2).bit_length()
    W = rfft(w_rev, fft_size)
    A = rfft(a, fft_size)
    conv_full = irfft(W * A, fft_size)
    # g[i] = conv_full[2n-2-i] for i=0,...,n-1
    # i.e., g = conv_full[n-1:2n-1][::-1]
    g = conv_full[n-1:2n-1][::-1]
    return g


def smooth_obj_and_grad(x, n, beta):
    """
    Compute smoothed objective and gradient with respect to x (log-space).
    a = softmax(x), objective = 2n * logsumexp(beta*conv(a,a))/beta
    """
    a = softmax(x)
    b = fast_convolve(a)
    # Smoothed max
    beta_b = beta * b
    logsum = beta_b.max() + np.log(np.sum(np.exp(beta_b - beta_b.max())))
    obj = 2 * n * logsum / beta
    
    # Weights
    w = np.exp(beta_b - logsum)  # w[k] = exp(beta*b[k]) / sum(exp(beta*b))
    
    # Gradient with respect to a
    g_a = compute_grad_smoothed(a, w)
    
    # Chain rule through softmax: dL/dx_j = a_j * (g_a_j - sum_i a_i * g_a_i)
    dot = np.dot(a, g_a)
    g_x = a * (g_a - dot)
    
    return obj, g_x


def lp_refine(f, max_iters=50, time_limit=120.0):

    LP-based refinement: solve max sum(g) s.t. conv(f,g) <= M, g >= 0.

def coordinate_descent(a0, time_limit, multipliers=None):
    """Coordinate descent on the simplex."""
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    if multipliers is None:
        multipliers = np.array([0.3, 0.5, 0.7, 0.8, 0.9, 1.0, 1.1, 1.2, 1.4, 1.7, 2.0, 3.0])

    improved_global = True
    while improved_global and time.monotonic() - t0 < time_limit:
        improved_global = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-15:
                continue
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0.0 or new_val > 0.9999:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < -0.01 or scale > 50.0:
                    continue

                a_tr = a.copy()
                a_tr[j] = new_val
                mask = np.arange(n) != j
                a_tr[mask] *= scale
                ss = a_tr.sum()
                if ss < 1e-15:
                    continue
                a_tr = a_tr / ss
                val = fast_convolve(a_tr).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.arange(n) != j
                a[mask] *= scale
                a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved_global = True

    return best_a, best_obj


def make_chirp_init(n):
    """Chirp-based initialization with flat autocorrelation."""
    x = np.arange(n)
    # Quadratic chirp
    phase = 2.0 * np.pi * x * x / (4.0 * n)
    a = np.cos(phase)
    a = np.maximum(a, 0.0)
    # Add small positive floor
    a = a + 0.01
    return a / a.sum()


def make_gaussian_init(n, sigma=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cosine_init(n):

    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    a = np.maximum(a, 1e-12)

def make_tukey_init(n, r=0.5):

        a[-taper:] = a[:taper][::-1]

def make_uniform_init(n):

def make_bimodal_init(n):
    """Bimodal: high at edges, low in middle."""
    x = np.linspace(0, 1, n)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * x))
    a = np.maximum(a, 0.05)

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def lbfgs_optimize(a_init, time_limit, beta_start=20.0, beta_end=500.0, n_restarts=3):
    """
    L-BFGS-B optimization with adaptive log-sum-exp smoothing.
    a_init: initial point (sum=1, non-negative)
    """
    t0 = time.monotonic()
    n = len(a_init)
    best_a = a_init.copy()
    best_obj = fast_convolve(a_init).max()
    
    rng = np.random.default_rng(42)
    
    for restart in range(n_restarts):
        a = best_a.copy()
        # Add small perturbation for later restarts
        if restart > 0:
            perturbation = rng.standard_normal(n) * 0.01 / (restart + 1)
            a = a + perturbation
            a = np.maximum(a, 0)
            a = a / a.sum()
        
        # Initialize in log-space
        x = np.log(np.maximum(a, 1e-30))
        
        # Anneal beta
        n_anneal_steps = 5
        time_per_step = (time_limit - (time.monotonic() - t0)) / (n_restarts - restart)
        time_per_beta = time_per_step / n_anneal_steps
        
        for beta_step in range(n_anneal_steps):
            if time.monotonic() - t0 > time_limit - 5:
                break
            
            beta = beta_start * (beta_end / beta_step ** (n_anneal_steps - 1)) if n_anneal_steps > 1 else beta_start
            # Better annealing: geometric
            if n_anneal_steps > 1:
                beta = beta_start * (beta_end / beta_start) ** (beta_step / (n_anneal_steps - 1))
            else:
                beta = beta_start
            
            t_beta_start = time.monotonic()
            t_beta_limit = min(time_per_beta, time_limit - (time.monotonic() - t0) - 5)
            
            if t_beta_limit < 2:
                continue
            
            # Run L-BFGS-B
            try:
                res = minimize(
                    smooth_obj_and_grad,
                    x,
                    args=(n, beta),
                    jac=True,
                    method='L-BFGS-B',
                    options={'maxiter': 200, 'maxfun': 500,
                             'iprint': 0,
                             'gtol': 1e-6}
                )
                
                a_new = softmax(res.x)
                a_new = np.maximum(a_new, 0)
                a_new = a_new / a_new.sum()
                
                obj_new = fast_convolve(a_new).max()
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_a = a_new.copy()
                a = a_new
                x = np.log(np.maximum(a, 1e-30))
                
            except Exception as e:
                print(f"    L-BFGS error: {e}")
                break
        
        # Check time
        if time.monotonic() - t0 > time_limit - 10:
            break
    
    return best_a, best_obj


def subgradient_optimize(a0, time_limit, lr_init=0.004, momentum=0.92, seed=42):
    """FFT-accelerated projected subgradient with Nesterov momentum."""
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    cur_lr = lr_init
    no_improve = 0
    n_top = max(15, n // 10)
    kick_size = 0.003
    
    fft_size = 1 << (3 * n - 2).bit_length()
    
    def project_simplex(v):
        n_local = len(v)
        if n_local == 1:
            return np.array([max(v[0], 0.0)])
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u)
        rho_vals = np.arange(1, n_local + 1)
        cond = u * rho_vals > (cssv - 1.0)
        if not cond.any():
            return np.ones(n_local) / n_local
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    
    it_count = 0
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        b_max = b.max()
        
        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
        else:
            top_idx = np.arange(len(b))
        
        threshold = b_max * 0.999
        active_mask = b[top_idx] >= threshold
        if not active_mask.any():
            active_mask[0] = True
        
        w = np.zeros(2 * n - 1)
        active_indices = top_idx[active_mask]
        w[active_indices] = 1.0 / len(active_indices)
        
        # Compute subgradient
        g = compute_grad_smoothed(a, w)
        
        # Nesterov momentum step
        v = momentum * v - cur_lr * g
        a_new = project_simplex(a + v)
        
        obj_new = fast_convolve(a_new).max()
        
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        it_count += 1
        
        if no_improve >= 3000:
            a = project_simplex(best_a + np.random.default_rng(it_count).standard_normal(n) * kick_size)
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.5, lr_init * 0.05)
            no_improve = 0
            kick_size *= 0.7
        elif no_improve >= 1500:
            cur_lr *= 0.92
        elif no_improve >= 750:
            cur_lr *= 0.97
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS-B on log-sum-exp smoothed objective with adaptive beta annealing
2. FFT-accelerated subgradient with Nesterov momentum
3. LP-based peak reduction refinement
4. Coordinate descent with multiplier-based line search
5. Multi-length exploration and diverse initializations (chirp, Gaussian, etc.)
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax(x):
    """Numerically stable softmax."""
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()


def compute_grad_smoothed(a, w):
    """
    Compute sum_j w[i+j]*a[j] for i=0..n-1 via FFT.
    This is the gradient of logsumexp(beta*b) wrt a, given normalized weights w.
    """
    n = len(a)
    w_rev = w[::-1]
    fft_size = 1 << (3 * n - 2).bit_length()
    W = rfft(w_rev, fft_size)
    A = rfft(a, fft_size)
    conv_full = irfft(W * A, fft_size)
    g = conv_full[n-1:2*n-1][::-1]
    return g


def smooth_obj_and_grad(x, n, beta):
    """
    Smoothed objective and gradient in log-space (softmax parameterization).
    Objective: 2n * logsumexp(beta*conv(softmax(x), softmax(x))) / beta
    """
    a = softmax(x)
    b = fast_convolve(a)
    beta_b = beta * b
    max_bb = beta_b.max()
    logsum = max_bb + np.log(np.sum(np.exp(beta_b - max_bb)))
    obj = 2.0 * n * logsum / beta

    w = np.exp(beta_b - logsum)
    g_a = compute_grad_smoothed(a, w)
    dot = np.dot(a, g_a)
    g_x = a * (g_a - dot)

    return obj, g_x


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_refine(f, max_iters=50, time_limit=120.0):
    """
    LP-based refinement: maximize sum(g) s.t. conv(f,g) <= M, g >= 0.
    Blend result with current f to reduce peak convolution.
    """
    t0 = time.monotonic()
    n = len(f)
    a = f.copy().astype(np.float64)
    a = a / a.sum()

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    # Pre-build the structure (column i has a[0..n-1] in rows i..i+n-1)
    for iteration in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        M = b.max()

        num_constraints = 2 * n - 1
        A_mat = np.zeros((num_constraints, n))
        for i in range(n):
            A_mat[i:i+n, i] = a

        b_lp = np.full(num_constraints, M)
        c = -np.ones(n)

        try:
            res = linprog(c, A_ub=A_mat, b_ub=b_lp,
                          bounds=[(0.0, None)] * n, method='highs')
            if not res.success:
                break

            g = res.x
            g_sum = g.sum()
            if g_sum < 1e-10:
                break

            g_norm = g / g_sum

            best_t = 0.0
            best_blend_obj = best_obj
            t_candidates = [0.01, 0.02, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
            for t in t_candidates:
                blend = (1.0 - t) * best_a + t * g_norm
                blend = np.maximum(blend, 0.0)
                s = blend.sum()
                if s < 1e-10:
                    continue
                blend = blend / s
                obj_blend = fast_convolve(blend).max()
                if obj_blend < best_blend_obj:
                    best_blend_obj = obj_blend
                    best_t = t

            if best_t > 0.0:
                a = (1.0 - best_t) * best_a + best_t * g_norm
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                obj = fast_convolve(a).max()
                if obj < best_obj:
                    best_obj = obj
                    best_a = a.copy()
            else:
                break

        except Exception:
            break

    return best_a, best_obj


def coordinate_descent(a0, time_limit):
    """Coordinate descent on the simplex with multiplier line search."""
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    multipliers = np.array([0.3, 0.5, 0.7, 0.8, 0.9, 1.1, 1.2, 1.4, 1.7, 2.0, 3.0])

    improved_global = True
    while improved_global and time.monotonic() - t0 < time_limit:
        improved_global = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-15:
                continue
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue

            best_j_obj = best_obj
            best_j_val = old_val

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0.0 or new_val > 0.9999:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < -0.01 or scale > 50.0:
                    continue

                a_tr = a.copy()
                a_tr[j] = new_val
                mask = np.arange(n) != j
                a_tr[mask] *= scale
                ss = a_tr.sum()
                if ss < 1e-15:
                    continue
                a_tr = a_tr / ss
                val = fast_convolve(a_tr).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.arange(n) != j
                a[mask] *= scale
                a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved_global = True

    return best_a, best_obj


def subgradient_optimize(a0, time_limit, lr_init=0.004, momentum=0.92, seed=42):
    """FFT-accelerated projected subgradient with Nesterov momentum."""
    t0 = time.monotonic()
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()

    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    cur_lr = lr_init
    no_improve = 0
    n_top = max(15, n // 10)
    kick_size = 0.003

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        b_max = b.max()

        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
        else:
            top_idx = np.arange(len(b))

        threshold = b_max * 0.999
        active_mask = b[top_idx] >= threshold
        if not active_mask.any():
            active_mask[0] = True

        w = np.zeros(2 * n - 1)
        active_indices = top_idx[active_mask]
        w[active_indices] = 1.0 / len(active_indices)

        g = compute_grad_smoothed(a, w)

        v = momentum * v - cur_lr * g
        a_new = project_simplex(a + v)

        obj_new = fast_convolve(a_new).max()

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        if no_improve >= 3000:
            a = project_simplex(best_a + rng.standard_normal(n) * kick_size)
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.5, lr_init * 0.05)
            no_improve = 0
            kick_size *= 0.7
        elif no_improve >= 1500:
            cur_lr *= 0.92
        elif no_improve >= 750:
            cur_lr *= 0.97

    return best_a, best_obj


def lbfgs_optimize(a_init, time_limit, beta_start=20.0, beta_end=500.0, n_restarts=3, seed=42):
    """
    L-BFGS-B optimization with adaptive log-sum-exp smoothing and beta annealing.
    Uses softmax parameterization: a = softmax(x), optimize over x.
    """
    t0 = time.monotonic()
    n = len(a_init)
    rng = np.random.default_rng(seed)
    best_a = a_init.copy()
    best_obj = fast_convolve(a_init).max()

    for restart in range(n_restarts):
        if time.monotonic() - t0 > time_limit - 10:
            break

        a = best_a.copy()
        if restart > 0:
            pert = rng.standard_normal(n) * 0.005 / (restart + 1)
            a = a + pert
            a = np.maximum(a, 0)
            a = a / a.sum()

        x = np.log(np.maximum(a, 1e-30))

        n_anneal = 5
        time_per_restart = (time_limit - (time.monotonic() - t0)) / max(1, (n_restarts - restart))
        time_per_beta = time_per_restart / n_anneal

        for beta_step in range(n_anneal):
            if time.monotonic() - t0 > time_limit - 5:
                break

            if n_anneal > 1:
                frac = float(beta_step) / float(n_anneal - 1)
                beta = beta_start * (beta_end / beta_start) ** frac
            else:
                beta = beta_start

            t_beta_limit = min(time_per_beta, time_limit - (time.monotonic() - t0) - 5)
            if t_beta_limit < 3.0:
                continue

            maxiter = min(300, max(50, int(t_beta_limit * 30)))

            try:
                res = minimize(
                    smooth_obj_and_grad,
                    x,
                    args=(n, beta),
                    jac=True,
                    method='L-BFGS-B',
                    options={'maxiter': maxiter, 'maxfun': maxiter * 3,
                             'iprint': 0, 'gtol': 1e-8}
                )

                a_new = softmax(res.x)
                a_new = np.maximum(a_new, 0)
                s = a_new.sum()
                if s > 1e-15:
                    a_new = a_new / s
                    obj_new = fast_convolve(a_new).max()
                    if obj_new < best_obj:
                        best_obj = obj_new
                        best_a = a_new.copy()
                    a = a_new
                    x = np.log(np.maximum(a, 1e-30))

            except Exception:
                break

    return best_a, best_obj


def make_chirp_init(n):
    """Chirp-based initialization (quadratic phase cosine, rectified)."""
    x = np.arange(n, dtype=np.float64)
    phase = 2.0 * np.pi * x * x / (4.0 * n)
    a = np.cos(phase)
    a = np.maximum(a, 0.0)
    a = a + 0.01
    return a / a.sum()


def make_gaussian_init(n, sigma=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_cosine_init(n):
    x = np.linspace(0, 1, n)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_tukey_init(n, r=0.5):
    a = np.ones(n)
    taper = max(1, int(r * n // 2))
    if taper < n:
        t = np.linspace(0, 1, taper)
        a[:taper] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t / r - 1.0)))
        a[-taper:] = a[:taper][::-1]
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def make_uniform_init(n):
    return np.ones(n) / n


def make_bimodal_init(n):
    """U-shaped: high at edges, low in middle."""
    x = np.linspace(0, 1, n)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * x))
    a = np.maximum(a, 0.05)
    return a / a.sum()


def make_triangular_init(n):
    a = np.ones(n)
    half = n // 2
    a[:half] = np.linspace(0.1, 1.0, half)
    a[half:] = np.linspace(1.0, 0.1, n - half)
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def eval_score(a):
    """Evaluate a candidate sequence (returns inf if invalid)."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a_norm = a / s
    return float(evaluate_sequence(a_norm.tolist()))
