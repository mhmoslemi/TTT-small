Let me think about another approach that might be different:

**Approach: Binary/ternary search on M + LP feasibility**

We want to find the minimum M such that there exists a non-negative sequence with sum=1 and max(conv(a,a)) <= M.

This is: find min M such that the feasible set {a >= 0, sum(a) = 1, max_k (a*a)_k <= M} is non-empty.

For a given M, checking feasibility is: does there exist a >= 0 with sum(a) = 1 and (a*a)_k <= M for all k?

This is NOT an LP because of the quadratic terms (a*a)_k. So we can't directly binary search on M.

However, the iterative approach (fix a, solve LP for g, update a) is a form of "successive linearization" that should converge to a local optimum.

**Another idea: Use the fact that the problem is convex in disguise**

Actually, the problem `min max_k (a*a)_k s.t. a >= 0, sum(a) = 1` is NOT convex in general because (a*a)_k is quadratic in a. But if we fix one variable at a time, it becomes easier.

**Coordinate descent with exact line search:**
For each coordinate i, find the optimal value of a_i (given all other a_j fixed) that minimizes max_k (a*a)_k.

For fixed a_{-i}, the convolution values are:
`(a*a)_k = sum_{j != i} a_j * a_{k-j} + a_i * (a_{k-i} + a_{k+n-1-i} - ...)` 

Hmm, this is getting complicated because changing a_i affects multiple convolution values.

Let me think of yet another approach:

**Approach: Water-filling / Proportional fair optimization**

The idea is to iteratively "flatten" the convolution by redistributing mass. At each step:
1. Find the position k* where conv is maximized
2. Reduce the contribution of a_{k*} (or a_0 if k*=0) to lower the peak
3. Redistribute the removed mass to positions that have low convolution contribution

This is similar to the subgradient approach but with a more explicit redistribution.

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using iterative LP-based descent
with active constraint identification, FISTA subgradient refinement,
and multi-length exploration.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build the matrix B such that (B @ g)[k] = (a * g)[k] for all k."""
    n = len(a)
    num_rows = 2 * n - 1
    B = np.zeros((num_rows, n))
    for k in range(num_rows):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            B[k, j] = a[k - j]
    return B


def build_conv_matrix_sparse(a):
    """Build the banded convolution matrix more efficiently."""
    n = len(a)
    num_rows = 2 * n - 1
    B = np.zeros((num_rows, n))
    # Row k has non-zeros at columns j where max(0,k-n+1) <= j <= min(n-1,k)
    # B[k,j] = a[k-j]
    for k in range(num_rows):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        j_range = np.arange(lo, hi + 1)
        B[k, j_range] = a[k - j_range]
    return B


def solve_lp_for_g(a, M):
    """
    Solve: max sum(g) s.t. conv(a,g) <= M, g >= 0
    Returns g (or None if infeasible).
    """
    n = len(a)
    B = build_conv_matrix(a)
    c = -np.ones(n)  # minimize -sum(g)
    try:
        res = linprog(c, A_ub=B, b_ub=np.full(2*n-1, M),
                      bounds=[(0, None)] * n, method='highs')
        if res.success:
            return res.x
        # Check if infeasible
        if res.status == 2:  # infeasible
            return None
        return res.x if res.x is not None else None
    except Exception:
        return None


def lp_descent(a_init, time_limit, max_iters=100, verbose=True):
    """
    Iterative LP descent: repeatedly solve LP and update.
    Returns (best_a, best_max_b).
    """
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_M = fast_convolve(a).max()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_a = fast_convolve(a)
        M = conv_a.max()
        
        # Solve LP
        g = solve_lp_for_g(a, M)
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        
        g_norm = g / g_sum
        g_conv = fast_convolve(g_norm)
        g_max = g_conv.max()
        
        if verbose:
            print(f"  LP iter {it}: M_a={M:.10f}, M_g={g_max:.10f}, sum_g={g_sum:.6f}")
        
        if g_max < best_M:
            # Full replacement works
            best_M = g_max
            best_a = g_norm.copy()
            a = g_norm.copy()
        else:
            # Try blending
            improved = False
            for t in [0.3, 0.5, 0.7, 0.1, 0.2, 0.9]:
                blend = (1 - t) * a + t * g_norm

                    best_M = blend_M
                    best_a = blend.copy()
                    a = blend.copy()
                    improved = True
                    break
            if not improved:
                break

    # For simplicity, use a conservative initial value and backtracking

            active_mask = np.argmax(conv_y)
            active_mask = np.zeros_like(conv_y, dtype=bool)
            active_mask[np.argmax(conv_y)] = True
            n_active = 1
        
        # Subgradient: for each active position k, grad_k[i] = y[k-i] (if valid)
        # Average over active positions
        grad = np.zeros(n)
        active_indices = np.where(active_mask)[0]
        for k in active_indices:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi+1] += y[k - hi:k - lo + 1]
        
        grad /= n_active
        
        # Backtracking line search
        while True:
            next_a_raw = y - step * grad
            next_a = np.maximum(next_a_raw, 0)
            next_a_sum = next_a.sum()
            if next_a_sum < 1e-15:

                next_a = next_a / next_a_sum

                break

                break

        t_new = (1 + np.sqrt(1 + 4 * t_k * t_k)) / 2

        a = next_a
        t_k = t_new
        
        if a_M < best_M - 1e-12 if False else next_M < best_M - 1e-12:
            ...
    
    return best_a, best_M

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using iterative LP-based descent
with full constraint set, aggressive updates, FISTA refinement,
and multi-length exploration. Key improvements: full LP replacement,
tighter M, diverse restarts, and focused time allocation.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    n = len(a)
    num_rows = 2 * n - 1
    B = np.zeros((num_rows, n))
    for k in range(num_rows):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        j_idx = np.arange(lo, hi + 1)
        B[k, j_idx] = a[k - j_idx]
    return B


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_descent(a_init, time_limit, verbose=False):
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_M = float(fast_convolve(a).max())
    
    for it in range(5000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv_a = fast_convolve(a)
        M = float(conv_a.max())
        
        B = build_conv_matrix(a)
        c = -np.ones(n)

            res = linprog(c, A_ub=B, b_ub=np.full(2 * n - 1, M),
                          bounds=[(0.0, None)] * n, method='highs',
                          options={'presolve': True})

            break

        if g_sum < 1e-10:
            break

        M_g = float(fast_convolve(g_n).max())
        
        if M_g < best_M * (1.0 - 1e-12):
            best_M = M_g
            best_a = g_n.copy()
            a = g_n.copy()
            if verbose and (it < 10 or it % 50 == 0):
                print(f"    LP {it}: M={M_g:.12f}")
        else:

def subgradient_fista(a_init, time_limit, seed=42, verbose=False):
    t0 = time.monotonic()
    n = len(a_init)
    rng = np.random.default_rng(seed)
    
    a = a_init.astype(np.float64).copy()
    a = a / a.sum()
    
    v = np.zeros(n)
    best_a = a.copy()
    best_M = float(fast_convolve(a).max())
    
    lr = 0.01
    momentum = 0.9
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        conv_a = fast_convolve(a)
        M = conv_a.max()
        
        # Active peaks (top k or within 1% of max)
        k = max(10, n // 20)
        top_idx = np.argpartition(conv_a, -k)[-k:]
        threshold = M * 0.999
        active = top_idx[conv_a[top_idx] >= threshold]
        if len(active) == 0:
            active = np.array([np.argmax(conv_a)])
        
        # Subgradient
        grad = np.zeros(n)
        for ki in active:
            lo = max(0, ki - n + 1)
            hi = min(n - 1, ki)
            grad[lo:hi+1] += a[ki - hi:ki - lo + 1]
        grad /= len(active)
        
        # Momentum step
        v = momentum * v - lr * grad
        a_new = project_simplex(a + v)
        
        M_new = float(fast_convolve(a_new).max())
        
        if M_new < best_M - 1e-13:
            best_M = M_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        it += 1
        
        if no_improve >= 500:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.01)
            v = np.zeros(n)
            lr *= 0.5
            no_improve = 0
        elif no_improve >= 200:
            lr *= 0.95
        elif no_improve >= 100:
            lr *= 0.99
    
    return best_a, best_M


def coordinate_descent(a_init, time_limit, verbose=False):
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_M = float(fast_convolve(a).max())
    
    multipliers = np.array([0.3, 0.5, 0.7, 0.8, 0.9, 1.1, 1.3, 1.5, 2.0, 3.0])
    
    improved = True
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_M
            old = a[j]
            if old < 1e-15:
                continue
            denom = 1.0 - old
            if denom < 1e-10:
                continue
            
            best_j_val = old
            best_j_M = best_M
            
            for m in multipliers:
                new_val = old * m
                if new_val < 0 or new_val > 0.9999:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0 or scale > 50:
                    continue
                
                a_tr = a.copy()
                a_tr[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_tr[mask] *= scale
                ss = a_tr.sum()
                if ss < 1e-15:
                    continue
                a_tr = a_tr / ss
                M_tr = float(fast_convolve(a_tr).max())
                if M_tr < best_j_M - 1e-14:
                    best_j_M = M_tr
                    best_j_val = new_val
            
            if best_j_val != old:
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using iterative LP-based descent
with aggressive full-replacement updates, FISTA subgradient refinement,
perturbation-based restarts, and multi-length exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix B where (B@g)[k] = (a*g)[k]."""
    n = len(a)
    B = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        jj = np.arange(lo, hi + 1)
        B[k, jj] = a[k - jj]
    return B


def project_simplex(v):
    """Projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_descent(a_init, time_limit, verbose=False):
    """Iterative LP: max sum(g) s.t. conv(a,g)<=M, g>=0, then update a."""
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_M = float(fast_convolve(a).max())
    
    for it in range(10000):
        if time.monotonic() - t0 > time_limit:
            break
        M = float(fast_convolve(a).max())
        B = build_conv_matrix(a)
        try:
            res = linprog(-np.ones(n), A_ub=B, b_ub=np.full(2 * n - 1, M),
                          bounds=[(0.0, None)] * n, method='highs')
        except Exception:
            break
        if not res.success:
            break
        g = res.x
        gs = g.sum()
        if gs < 1e-10:
            break
        g_n = g / gs
        M_g = float(fast_convolve(g_n).max())
        if M_g < best_M - 1e-14:
            best_M = M_g
            best_a = g_n.copy()
            a = g_n.copy()
            if verbose:
                print(f"    LP[{it}] M={M_g:.14f}")
        else:
            # Try blends
            ok = False
            for t in [0.5, 0.3, 0.7, 0.1, 0.2, 0.9, 0.05]:
                bl = np.maximum((1 - t) * a + t * g_n, 0.0)
                bs = bl.sum()
                if bs < 1e-10:

                Mb = float(fast_convolve(bl).max())
                if Mb < best_M - 1e-14:
                    best_M = Mb
                    best_a = bl.copy()
                    a = bl.copy()
                    ok = True
                    break
            if not ok:
                break
    return best_a, best_M
