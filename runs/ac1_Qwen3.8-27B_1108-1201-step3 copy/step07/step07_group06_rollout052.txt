Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using direct peak-targeted local search
with sensitivity-aware redistribution, simulated annealing escape, multi-start
diverse initialization, and multi-length exploration. Key differences from the
previous Adam/soft-max approach: (1) direct peak manipulation instead of smooth
approximation, (2) sensitivity-aware redistribution to minimize side effects,
(3) fine-grained step sizes for the near-optimal regime, (4) multi-peak

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def normalize(a):
    """Normalize to sum=1, ensuring non-negativity."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / len(a)
    return a / s


def local_score(a):
    """Compute 2n*max(conv(a,a)) assuming sum(a) ~ 1."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def compute_PEAKS(a, top_k=5):
    """Return top_k peak positions and their values."""
    b = fast_convolve(a)
    # Exclude position 0 (always the largest)
    b_search = b.copy()
    b_search[0] = -np.inf
    if len(b_search) > 1:
        b_search[-1] = -np.inf
    idx = np.argpartition(-b_search, top_k)[:top_k]
    idx = idx[np.argsort(-b_search[idx])]
    return idx, b[idx]


def find_contributors(a, k, n):
    """Find indices i where a[i]*a[k-i] contributes to b[k]."""
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    if js > je:
        return np.array([]), np.array([])
    idx = np.arange(js, je + 1)
    contribs = a[idx] * a[k - idx]
    return idx, contribs


def direct_peak_search(a_init, time_limit, rng, min_step=0.001, max_trials=50000):
    """
    Direct peak-targeted local search with simulated annealing.
    Works with sum(a)=1 constraint.
    """
    n = len(a_init)
    a = normalize(a_init.copy())
    best_a = a.copy()
    best_score = local_score(a)
    t0 = time.monotonic()
    
    # Track best
    current_score = best_score
    no_improve = 0
    max_no_improve = 2000
    
    # SA parameters
    T_sa = 0.001
    T_min = 1e-7
    
    for trial in range(max_trials):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        b_search = b.copy()
        b_search[0] = -np.inf
        if len(b_search) > 1:
            b_search[-1] = -np.inf
        peak_k = int(np.argmax(b_search))
        peak_val = b[peak_k]
        
        # Get contributors to the peak
        idx, contribs = find_contributors(a, peak_k, n)
        if len(idx) == 0:
            break
        
        # Sort by contribution (largest first)
        order = np.argsort(-contribs)
        
        improved = False
        step_size = min_step + 0.01 * max(0, 1 - trial / max_trials)
        
        # Try reducing top contributors
        for ci in order[:min(10, len(order))]:
            i = int(idx[ci])
            j_pos = int(peak_k - i)
            if a[i] < 1e-15:
                continue
            
            # Try different step sizes
            for eps in [step_size, step_size * 0.5, step_size * 2]:
                if eps < 1e-5:
                    continue
                delta = a[i] * eps
                if delta < 1e-15:
                    continue
                
                a_trial = a.copy()
                a_trial[i] -= delta
                
                # Redistribute: add to position with minimum value (excluding i and j_pos)
                # Try a few candidates for redistribution
                min_mask = np.ones(n, dtype=bool)
                min_mask[i] = False
                if 0 <= j_pos < n:
                    min_mask[j_pos] = False
                # Also exclude neighbors of i and j_pos to reduce interaction
                for dx in [-1, 1]:
                    for pos in [i, j_pos]:
                        if 0 <= pos + dx < n:
                            min_mask[pos + dx] = False
                
                candidates = np.where(min_mask)[0]
                if len(candidates) == 0:
                    continue
                
                # Pick the candidate with minimum a value
                c_val = a[candidates]
                best_cand = candidates[np.argmin(c_val)]
                
                a_trial[best_cand] += delta
                a_trial = normalize(a_trial)
                
                score = local_score(a_trial)
                if score < current_score - 1e-15:
                    a = a_trial
                    current_score = score
                    if score < best_score:
                        best_score = score
                        best_a = a.copy()
                    no_improve = 0
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            no_improve += 1
            
            # SA escape: try a random perturbation
            if no_improve % 50 == 0:
                scale = 0.005 * max(1.0, min(10.0, T_sa / T_min))
                a_trial = a + rng.standard_normal(n) * scale
                a_trial = normalize(a_trial)
                score = local_score(a_trial)
                # Accept if better or with SA probability
                if score < current_score:
                    a = a_trial
                    current_score = score
                elif rng.random() < np.exp(-(score - current_score) / max(T_sa, 1e-10)):
                    a = a_trial
                    current_score = score
                # Cool down
                T_sa *= 0.95
                
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
            
            # Adaptive: if stuck for too long, try larger perturbation
            if no_improve > max_no_improve:
                scale = 0.02
                a_trial = a + rng.standard_normal(n) * scale
                a_trial = normalize(a_trial)
                score = local_score(a_trial)
                if score < current_score * 1.01:  # Allow slightly worse
                    a = a_trial
                    current_score = score
                no_improve = 0
                T_sa = 0.001
    
    return best_a, best_score

    T = 0.05
    no_improve = 0
    
    fft_size = 1 << (2 * n - 2).bit_length()
    
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Compute soft-max weights
        exp_b = np.exp((b - max_b) / T)
        w = exp_b / exp_b.sum()
        
        # Gradient: grad[i] = 2n * sum_{m} w[m] * (a[m-i] + a[i-m])
        # Use FFT for efficiency
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        # Cross-correlation: sum_m w[m] * a[m-i]
        corr = irfft(W * np.conj(A), fft_size)[:n]
        # Also need sum_m w[m] * a[i-m] which is the same as corr[n-1-i] for circular
        # But we have linear correlation, so let's compute both
        # Actually for linear: sum_{m} w[m]*a[m-i] where valid m: i <= m <= i+n-1
        # And sum_{m} w[m]*a[i-m] where valid m: 0 <= i-m < n, i.e., i-n+1 <= m <= i
        
        # Simpler: just use the FFT-based approach
        # grad[i] = 2n * (cross_corr_w_a[i])
        # where cross_corr is the linear cross-correlation of w and a
        
        # Actually, let's just compute the gradient directly for the peak
        # The subgradient of max(b) is w (the soft-max weights)
        # grad of b[m] w.r.t. a[i] = a[m-i] + a[i-m] (for valid indices)
        
        # Efficient computation using FFT:
        # sum_m w[m] * a[m-i] = (w cross-correlate a)[i]
        W_full = rfft(w, fft_size)
        A_full = rfft(a, fft_size)
        # Cross-correlation of w and a: sum_m w[m+i]*a[m] = ifft(W * conj(A))
        xcorr1 = irfft(W_full * np.conj(A_full), fft_size)
        # This gives sum_m w[m+i]*conj(a[m]) = sum_m w[m+i]*a[m] for real a
        # We want sum_m w[m]*a[m-i] = sum_m w[m+i]*a[m] (substituting m->m+i)
        # So xcorr1[i] = sum_m w[m+i]*a[m] which is what we want
        # But this is circular, we need linear
        
        # For linear correlation, we need to pad:
        # Let's use the direct approach since n is not too large
        # grad[i] = 2n * sum_m w[m] * (a[m-i] if 0<=m-i<n else 0 + a[i-m] if 0<=i-m<n else 0)
        
        # Using FFT with proper padding:
        # sum_m w[m]*a[m-i] for m=0..2n-2, a[m-i] valid when 0<=m-i<n
        # = sum_{j=0}^{n-1} w[j+i]*a[j] (where j = m-i, so m = j+i)
        # This is the cross-correlation of w and a at lag i
        
        # Let me just compute it directly for clarity (n=768, this is fine)
        grad = np.zeros(n)
        for i in range(n):
            # sum_m w[m] * a[m-i] where 0 <= m-i < n
            j_start = max(0, i)
            j_end = min(2*n-1, i + n - 1)
            if j_start <= j_end:
                m_vals = np.arange(j_start, j_end + 1)
                j_vals = m_vals - i
                grad[i] += np.dot(w[m_vals], a[j_vals])
            # sum_m w[m] * a[i-m] where 0 <= i-m < n
            # m range: i-n+1 to i, and 0 to 2n-2
            m_start = max(0, i - n + 1)
            m_end = min(2*n - 2, i)
            if m_start <= m_end:
                m_vals = np.arange(m_start, m_end + 1)
                j_vals = i - m_vals
                grad[i] += np.dot(w[m_vals], a[j_vals])
        
        grad *= 2.0 * n
        
        # Project gradient: we want to move in -grad direction while maintaining
        # sum(a)=1 and a>=0
        # Simple approach: a <- a - lr*grad, then clip and renormalize
        a_trial = a - lr * grad
        a_trial = np.maximum(a_trial, 0)
        a_trial = normalize(a_trial)
        
        score = local_score(a_trial)
        if score < current_score - 1e-15:
            a = a_trial
            current_score = score
            if score < best_score:
                best_score = score
                best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            
        # Adapt learning rate
        if no_improve > 200:
            lr *= 0.8
            no_improve = 0
            if lr < 1e-6:
                # Reset with perturbation
                a = best_a + rng.standard_normal(n) * 0.005
                a = normalize(a)
                current_score = local_score(a)
                lr = 0.005
        
        # Cool temperature
        T = max(0.001, T * 0.9995)
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize(f_interp)


def diverse_initializations(parent, n, rng):
    """Generate diverse starting points."""
    candidates = []
    
    # Parent
    p = normalize(parent)
    candidates.append(p.copy())
    
    # Perturbed parent
    for scale in [0.005, 0.01, 0.03, 0.05]:
        pert = p + rng.standard_normal(n) * scale
        candidates.append(normalize(pert))
    
    # Structured: cosine perturbation
    for freq in [2, 3, 5]:
        t = np.linspace(0, 2 * np.pi * freq, n)
        pert = p + 0.02 * np.cos(t)
        candidates.append(normalize(pert))
    
    # Gaussian
    x = np.linspace(-3, 3, n)
    gau = np.exp(-x**2 / 3)
    candidates.append(normalize(gau))
    
    # Uniform
    candidates.append(np.ones(n) / n)
    
    # Triangular
    tri = np.minimum(np.arange(1, n+1), n - np.arange(n))
    candidates.append(normalize(tri.astype(float)))
    
    # Power-law
    x = np.linspace(0.01, 1, n)
    for alpha in [0.5, 1.0, 2.0]:
        candidates.append(normalize(x ** alpha))
    
    return candidates


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining direct peak search, pair search,
    multi-peak balancing, gradient steps, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, budget_s - 20.0)
    
    # Load parent
    best_a = None
    best_score = float('inf')
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = normalize(p)
            s = local_score(parent)
            es = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(es):
                s = es
            best_a = parent.copy()
            best_score = s
            print(f"Parent loaded: n={len(parent)}, score={best_score:.6f}")
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    if parent is None:
        n_default = 768
        x = np.linspace(-3, 3, n_default)
        parent = normalize(np.exp(-x**2 / 3))
        best_a = parent.copy()
        best_score = local_score(parent)
        print(f"Default init: n={n_default}, score={best_score:.6f}")
    
    n = len(best_a)
    
    def try_update(a_candidate, label):
        nonlocal best_a, best_score
        a_candidate = normalize(a_candidate)
        s_local = local_score(a_candidate)
        s_eval = float(evaluate_sequence(a_candidate.tolist()))
        if np.isfinite(s_eval) and s_eval < best_score:
            best_score = s_eval
            best_a = a_candidate.copy()
            print(f"  NEW BEST [{label}]: {best_score:.6f} (n={len(best_a)})")
        elif s_local < best_score * 0.999:
            # Trust local if very different
            if np.isfinite(s_eval) and s_eval < best_score + 0.01:
                best_score = min(best_score, s_eval) if np.isfinite(s_eval) else best_score
                best_a = a_candidate.copy()
                print(f"  Updated [{label}]: local={s_local:.6f}, eval={s_eval:.6f}")
    
    # === Phase 1: Direct peak search from parent (35% time) ===
    print(f"\n=== Phase 1: Direct Peak Search (n={n}) ===")
    t_p1 = (deadline - time.monotonic()) * 0.35
    t_avail = max(30.0, min(t_p1, 600.0))
    
    a_res, s_res = direct_peak_search(best_a, t_avail * 0.5, rng)
    try_update(a_res, "direct_peak")
    
    rem = deadline - time.monotonic() - 0.3 * (deadline - time.monotonic())
    t_avail2 = max(30.0, min(t_p1 - t_avail * 0.5, (deadline - time.monotonic()) * 0.3))
    a_res2, s_res2 = pair_peak_search(best_a, t_avail2, rng)
    try_update(a_res2, "pair_peak")
    
    # === Phase 2: Multi-peak balance (20% time) ===
    print(f"\n=== Phase 2: Multi-Peak Balance (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_p2 = min(rem * 0.2, 300.0)
    if t_p2 > 20:
        a_res3, s_res3 = multi_peak_balance(best_a, t_p2, rng)
        try_update(a_res3, "multi_peak")
    
    # === Phase 3: Gradient-based refinement (20% time) ===
    print(f"\n=== Phase 3: Gradient Refinement (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_p3 = min(rem * 0.25, 300.0)
    if t_p3 > 20:
        a_res4, s_res4 = gradient_step_search(best_a, t_p3, rng, lr_init=0.005)
        try_update(a_res4, "gradient")
    
    # === Phase 4: Multi-start exploration (15% time) ===
    print(f"\n=== Phase 4: Multi-Start (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_p4 = min(rem * 0.2, 250.0)
    if t_p4 > 30:
        inits = diverse_initializations(best_a, n, rng)
        # Sort by initial score
        inits_scored = []
        for i, init_a in enumerate(inits):
            s_init = local_score(init_a)
            inits_scored.append((s_init, i, init_a))
        inits_scored.sort(key=lambda x: x[0])
        
        n_starts = min(5, len(inits_scored))
        t_per_start = t_p4 / n_starts
        
        for si in range(n_starts):
            rem = deadline - time.monotonic() - 60
            if rem < 20:
                break
            _, _, init_a = inits_scored[si]
            t_sub = min(t_per_start, rem * 0.8)
            if t_sub < 15:
                continue
            print(f"  Multi-start {si+1}/{n_starts} (init score={inits_scored[si][0]:.6f})")
            a_ms, s_ms = direct_peak_search(init_a, t_sub, rng)
            try_update(a_ms, f"multi_start_{si}")
    
    # === Phase 5: Multi-length exploration (10% time) ===
    print(f"\n=== Phase 5: Multi-Length (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_p5 = min(rem * 0.15, 200.0)
    if t_p5 > 30:
        n_best = len(best_a)
        alt_lengths = []
        for factor in [1.25, 0.8, 1.5, 1.1, 0.7]:
            alt_n = int(n_best * factor)
            if 256 <= alt_n <= 2048 and alt_n != n_best:
                if alt_n not in alt_lengths:
                    alt_lengths.append(alt_n)
        
        for alt_n in alt_lengths[:3]:
            rem = deadline - time.monotonic() - 30
            if rem < 20:
                break
            t_sub = min(t_p5 / len(alt_lengths[:3]), rem * 0.7)
            if t_sub < 10:
                continue
            print(f"  Alt length n={alt_n}")
            init_alt = interpolate_sequence(best_a, alt_n)
            a_alt, s_alt = direct_peak_search(init_alt, t_sub, rng)
            try_update(a_alt, f"alt_n{alt_n}")
    
    # === Phase 6: Final intensive polish (remaining time) ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        # Very fine-grained direct search
        t_sub = min(rem * 0.6, 120.0)
        a_fin, s_fin = direct_peak_search(best_a, t_sub, rng, min_step=0.0005)
        try_update(a_fin, "final_direct")
        
        rem = deadline - time.monotonic()
        if rem > 15:
            t_sub2 = min(rem * 0.5, 60.0)
            a_fin2, s_fin2 = pair_peak_search(best_a, t_sub2, rng)
            try_update(a_fin2, "final_pair")
    
    # Final verification
    if best_a is not None:
        fs = float(evaluate_sequence(best_a.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_a)})")
    else:
        best_a = normalize(np.ones(768))
        best_score = float(evaluate_sequence(best_a.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.6f}")
    
    return [float(x) for x in best_a]
