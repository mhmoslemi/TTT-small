```python
"""
Multi-strategy optimizer for step function auto-correlation minimization.
Combines LP fictitious play, stochastic coordinate descent, simulated annealing,
and L-BFGS refinement across multiple sequence lengths and starting points.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from time import monotonic
import cvxpy as cp

# Global solver cache
_lp_solver = None

def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.PDLP, cp.GUROBI, cp.MOSEK, cp.SCS, cp.ECOS, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return _lp_solver
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.SCS, cp.PDLP, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)
```

Actually, the most efficient way is:
$g = 2 \cdot \text{FFT}^{-1}(\text{FFT}(w_{padded}) \cdot \text{FFT}(a_{flipped}))$

where $w_{padded}$ is $w$ of length $2n-1$ zero-padded to FFT size, and $a_{flipped}$ is $a$ reversed and zero-padded.

Let me verify: I want $g_m = 2 \sum_{k=0}^{2n-2} w_k a_{k-m}$ where $a_j = 0$ for $j < 0$ or $j \geq n$.

Let $a_{ext}$ be $a$ zero-padded to length $2n-1$ (same as the convolution output length).
Then $g_m = 2 \sum_k w_k a_{ext}[k-m]$ where $k-m$ ranges over valid indices.

This is the cross-correlation of $w$ and $a_{ext}$, which equals the convolution of $w$ and $\text{flip}(a_{ext})$:
$g = 2 \cdot \text{convolve}(w, \text{flip}(a_{ext}))$

In the FFT domain: $\text{FFT}(g) = 2 \cdot \text{FFT}(w) \cdot \text{FFT}(\text{flip}(a_{ext}))$

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-strategy approach:
1. LP-based fictitious play for basin finding
2. Stochastic coordinate descent with pair swaps for local optimization
3. L-BFGS with soft-max smoothing for smooth refinement
4. Multiple diverse starting points and adaptive sequence lengths
5. Simulated annealing restarts to escape local minima
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from time import monotonic
import cvxpy as cp

_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.PDLP, cp.GUROBI, cp.MOSEK, cp.SCS, cp.ECOS, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return _lp_solver
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.SCS, cp.PDLP, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def fictitious_play(n, time_limit, initial=None, solver=None, verbose=False):
    if solver is None:
        solver = get_lp_solver()
    t0 = monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        a = a / s if s > 1e-15 else np.ones(n) / n

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    iters = 0

    while monotonic() - t0 < time_limit:
        iters += 1
        b_new, _ = solve_lp_best_response(a, solver)
        if b_new is not None:
            a_new, _ = solve_lp_best_response(b_new, solver)
            if a_new is not None:
                a = 0.6 * a + 0.4 * a_new
                a = np.maximum(a, 0)
                a = a / a.sum()

            sc = 2.0 * n * fast_convolve(a).max()
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a.copy()

            mid = 0.5 * (a + b_new)
            mid = np.maximum(mid, 0)
            mid_s = mid.sum()
            if mid_s > 1e-15:
                mid = mid / mid_s
                sc_mid = 2.0 * n * fast_convolve(mid).max()
                if sc_mid < best_score - 1e-12:
                    best_score = sc_mid
                    best_a = mid.copy()

        if verbose and iters % 50 == 0:
            elapsed = monotonic() - t0
            print(f"    FP n={n} iter={iters} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def coordinate_descent(a, time_limit, rng=None, verbose=False):
    n = len(a)
    a = np.array(a, dtype=np.float64).copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    if rng is None:
        rng = np.random.default_rng(12345)

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    fracs = np.array([0.05, 0.2, 0.4, 0.6, 0.8, 0.95])

    iters = 0
    no_improve = 0
    stagnation_count = 0

    while monotonic() - t0 < time_limit:
        iters += 1
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        u = a[i]
        v = a[j]
        s_pair = u + v
        if s_pair < 1e-12:
            continue

        improved = False
        for frac in fracs:
            t = s_pair * frac
            if t < 1e-14 or (s_pair - t) < 1e-14:
                continue
            if abs(t - u) < s_pair * 1e-12:
                continue
            a[i] = t
            a[j] = s_pair - t
            A = rfft(a, fft_size)
            conv = irfft(A * A, fft_size)[:size]
            sc = 2.0 * n * conv.max()
            if sc < best_score - 1e-13:
                best_score = sc
                best_a = a.copy()
                improved = True
                break

        if not improved:
            a[i] = u
            a[j] = v
            no_improve += 1
        else:
            no_improve = 0

        if no_improve > 5000:
            pertur

b = 0.003 * (iters // 5000)
            a = best_a + rng.standard_normal(n) * pertur
            a = np.maximum(a, 0)
            a = a / a.sum()
            no_improve = 0

        if verbose and iters % 20000 == 0:
            elapsed = monotonic() - t0
            print(f"    CD iter={iters} ({elapsed:.0f}s): score={best_score:.6f} no_improve={no_improve}")

    return best_a, best_score


def lbgfs_refine(a0, time_limit, T=100, maxiter=5000):
    n = len(a0)
    a_safe = np.maximum(a0, 1e-30)
    z0 = np.log(a_safe)
    z0 = z0 - z0.mean()
    t0 = monotonic()
    stopped = [False]

    def func(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()
        b = fast_convolve(a)
        max_b = b.max()
        scaled_b = (b - max_b) / T
        exp_b = np.exp(scaled_b)
        lse = max_b + T * np.log(exp_b.sum())
        return 2.0 * n * lse / T

    def jac(z):
        if monotonic() - t0 > time_limit:
            stopped[0] = True
            return np.zeros(n)
        z_shifted = z - z.max()
        exp_z = np.exp(z_shifted)
        a = exp_z / exp_z.sum()
        b = fast_convolve(a)
        max_b = b.max()
        scaled_b = (b - max_b) / T
        exp_b = np.exp(scaled_b)
        w = exp_b / exp_b.sum()

        grad_a = np.zeros(n)
        nonzero_w = w > 1e-8
        ks = np.nonzero(nonzero_w)[0]
        for k in ks:
            wk = w[k]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                idx = np.arange(lo, hi + 1)
                grad_a[idx] += 2.0 * wk * a[k - idx]

        grad_a *= 2.0 * n
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - a * S)
        return grad_z

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                     'maxfun': maxiter * 15, 'disp': False}
        )
        if not stopped[0]:
            z_final = res.x
            z_shifted = z_final - z_final.max()
            a_out = np.exp(z_shifted)
            a_out = a_out / a_out.sum()
        else:
            a_out = a0.copy()
    except Exception:
        a_out = a0.copy()

    return a_out


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def gaussian_init(n, rng):
    x = np.linspace(-1, 1, n)
    a = np.exp(-2.0 * x * x)
    a += rng.standard_normal(n) * 0.1
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def random_init(n, rng):
    a = rng.random(n)
    a = a / a.sum()
    return a


def structured_init(n, rng):
    t = np.linspace(0, 2 * np.pi, n, endpoint=False)
    a = np.ones(n)
    n_harmonics = rng.integers(2, 6)
    for _ in range(n_harmonics):
        freq = rng.integers(1, n // 2)
        amp = rng.uniform(0.05, 0.3)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(freq * t + phase)
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def full_pipeline(a0, time_limit, solver, rng, verbose=False):
    n = len(a0)
    a = np.array(a0, dtype=np.float64).copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    t0 = monotonic()
    rem = time_limit - (monotonic() - t0)

    # Step 1: Short FP
    fp_time = min(rem * 0.3, 60.0)
    if fp_time > 5:
        a_fp, sc_fp = fictitious_play(n, fp_time, initial=a, solver=solver, verbose=verbose)
        if sc_fp < best_score:

            best_a = a_fp.copy()
        a = best_a.copy()

    rem = time_limit - (monotonic() - t0)

        if sc_cd < best_score:
            best_score = sc_cd
            best_a = a_cd.copy()
        a = best_a.copy()

    # Step 3: L-BFGS
    rem = time_limit - (monotonic() - t0)
    lb_time = min(rem * 0.4, 40.0)

    if rem > 20:
        # Final CD
        cd_final_time = min(rem * 0.4, 80.0)
        if cd_final_time > 5:
            rng_final = np.random.default_rng(seed + 99999)
            print(f"  Final CD, budget={cd_final_time:.0f}s")
            a_f1, _ = coordinate_descent(best_seq, cd_final_time, rng=rng_final, verbose=False)
            best_seq, best_score = check_and_update(a_f1, "final_cd", best_seq, best_score)

        # Final L-BFGS
        rem = deadline - monotonic() - 5
        if rem > 8:
            lb_final_time = min(rem * 0.5, 30.0)
            print(f"  Final L-BFGS, budget={lb_final_time:.0f}s")
            a_f2 = lbgfs_refine(best_seq, lb_final_time, T=60, maxiter=5000)
            best_seq, best_score = check_and_update(a_f2, "final_lbgfs", best_seq, best_score)

        # One more quick CD
        rem = deadline - monotonic() - 2
        if rem > 5:
            a_f3, _ = coordinate_descent(best_seq, rem * 0.7, rng=np.random.default_rng(seed + 111111), verbose=False)
            best_seq, best_score = check_and_update(a_f3, "final_cd2", best_seq, best_score)

    # Final verification
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = min(best_score, fs)

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
