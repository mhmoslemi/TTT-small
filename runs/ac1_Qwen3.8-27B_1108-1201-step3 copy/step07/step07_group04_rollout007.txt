import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver, preferring PDLP for speed."""
    if '_lp_solver' in globals():
        return _lp_solver
    for solver_name in [cp.PDLP, cp.GUROBI, cp.MOSEK, cp.SCS, cp.ECOS, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None, top_k_constraints=None):
    """Given fixed a (non-negative, sum=1), find b (non-negative, sum=1) minimizing max conv(a,b)."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.SCS, cp.PDLP, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def fictitious_play_enhanced(n, time_limit, initial=None, solver=None, verbose=False):
    """Enhanced fictitious play with adaptive damping and warm-start awareness."""
    if solver is None:
        solver = get_lp_solver()
    t0 = monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s

    a_avg = a.copy().astype(np.float64)
    best_score = float('inf')
    best_a = a.copy()
    iters = 0
    stagnation = 0
    last_best = float('inf')
    patience_counter = 0

    while monotonic() - t0 < time_limit:
        iters += 1

        # Solve LP best response for b given a
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is None:
            b_new = a_avg.copy()
        # Solve LP best response for a given b
        a_new, M_a = solve_lp_best_response(b_new, solver)
        if a_new is None:
            a_new = a_avg.copy()

        # Adaptive averaging: use smaller weight on recent iterates after convergence
        if iters <= 50:
            alpha = 1.0 / (iters + 1)
        elif iters <= 200:
            alpha = 1.0 / (50 + (iters - 50) * 2)
        else:
            alpha = 0.005 + 0.01 / (1 + (iters - 200) * 0.01)

        a_avg = (1 - alpha) * a_avg + alpha * a_new

        # Periodically use the raw iterate too
        candidates = [a_avg, a_new, 0.5 * (a_avg + a_new), 0.7 * a_avg + 0.3 * a_new]
        for cand in candidates:
            cand = np.maximum(cand, 0)
            s = cand.sum()

                continue
            cand_norm = cand / s
            max_conv = fast_convolve(cand_norm).max()

                best_score = score
                best_a = cand_norm.copy()

                    patience_counter = 0
                else:
                    patience_counter += 1
                last_best = best_score
            else:

        # Restart with perturbation if stuck

            noise_scale = 0.003 / (1 + iters * 0.001)
            a_avg = a_avg + np.random.standard_normal(n) * noise_scale
            a_avg = np.maximum(a_avg, 0)
            a_avg = a_avg / a_avg.sum()
            patience_counter = 0
            stagnation = 0

        if verbose and iters % 50 == 0:
            elapsed = monotonic() - t0
            print(f"    FP n={n} iter={iters} ({elapsed:.0f}s): M_a={M_a:.6f} M_b={M_b:.6f} best={best_score:.6f}")

def active_set_refine(a0, time_limit, verbose=False):
    """
    Active-set method: identify the set of peaks, then solve
    min_a max_{k in S} conv(a,a)[k] using softened L-BFGS with 
    progressively tightening active set.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()

    # Start with a loose active set and tighten
    tol_schedule = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0001]
    temp_schedule = [50, 30, 20, 10, 5, 3, 1, 0.5]

    for phase in range(len(tol_schedule)):
        if monotonic() - t0 > time_limit:
            break
        tol = tol_schedule[phase]
        T = temp_schedule[phase]

        b = fast_convolve(a)
        max_b = b.max()
        active = np.where(b >= max_b * (1 - tol))[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        # Phase time allocation
        phase_time = min(
            (time_limit - (monotonic() - t0)) * 0.4,
            max(20, 100 / (phase + 1))
        )
        phase_deadline = monotonic() + phase_time

        # L-BFGS on the active set with soft-max
        z0 = np.log(np.maximum(a, 1e-30))
        z0 = z0 - z0.mean()

        def func(z):
            z_shift = z - z.max()
            exp_z = np.exp(z_shift)
            a_curr = exp_z / exp_z.sum()
            b_curr = fast_convolve(a_curr)
            b_active = b_curr[active]
            m = b_active.max()
            return m + T * np.log(np.exp((b_active - m) / T).sum())

        def jac(z):
            z_shift = z - z.max()
            exp_z = np.exp(z_shift)
            a_curr = exp_z / exp_z.sum()
            b_curr = fast_convolve(a_curr)
            b_active = b_curr[active]
            m = b_active.max()
            scaled = (b_active - m) / T
            exp_b = np.exp(scaled)
            w = exp_b / exp_b.sum()

            # Gradient of sum_k w[k] * conv(a,a)[k] wrt a
            grad_a = np.zeros(n)
            for idx in range(len(active)):
                if w[idx] < 1e-12:
                    continue
                k = int(active[idx])
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idxs = np.arange(js, je + 1)
                grad_a[idxs] += 2.0 * w[idx] * a_curr[k - idxs]

            # Soft max gradient adds a constant
            grad_a += 1.0 / len(active)  # approximation for the log-sum term

            # Transform to z-space (softmax parameterization)
            S = np.dot(grad_a, a_curr)
            grad_z = a_curr * (grad_a - a_curr * S)
            return grad_z

        try:
            res = minimize(
                func, z0, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': 200, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': 3000, 'disp': False}
            )
            z_final = res.x
            z_shift = z_final - z_final.max()
            a_new = np.exp(z_shift)
            a_new = a_new / a_new.sum()
            a_new = np.maximum(a_new, 0)

            score = 2.0 * n * fast_convolve(a_new).max()
            if score < best_score - 1e-14:
                best_score = score
                best_a = a_new.copy()
                a = a_new
        except Exception:
            pass

        if verbose:
            print(f"    ActiveSet phase={phase} tol={tol} T={T}: score={best_score:.6f} active={len(active)}")

    return best_a, best_score


def peak_redistribute(a0, time_limit, verbose=False):
    """
    Direct peak reduction: identify top contributing pairs to each peak,
    and try to reduce their contribution by redistributing mass.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    no_improve = 0

    while monotonic() - t0 < time_limit and no_improve < 200:
        b = fast_convolve(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        total_contrib = contribs.sum()

        if total_contrib < 1e-20:
            break

        improved = False

        # Try reducing top contributors and redistributing to non-peak areas
        top_m = min(8, len(contribs))
        top_idx_sorted = np.argsort(contribs)[::-1][:top_m]

        # Find "safe" positions (low contribution to all peaks)
        # Compute contribution of each position to the top 3 peaks
        top_peaks = np.argpartition(b, -3)[-3:]
        contrib_matrix = np.zeros((n, len(top_peaks)))
        for pi, pk in enumerate(top_peaks):
            js2 = max(0, pk - n + 1)
            je2 = min(n - 1, pk)
            idx2 = np.arange(js2, je2 + 1)
            contrib_matrix[idx2, pi] = a[idx2] * a[pk - idx2]

        # Positions with low total contribution are "safe" targets
        total_contrib_per_pos = contrib_matrix.sum(axis=1)
        safe_mask = total_contrib_per_pos < np.median(total_contrib_per_pos)

        # Only consider safe positions that are not already contributing to the peak
        peak_mask = np.zeros(n, dtype=bool)
        peak_mask[idx] = True
        safe_mask = safe_mask & ~peak_mask

        safe_positions = np.where(safe_mask)[0]
        if len(safe_positions) == 0:
            safe_positions = np.where(total_contrib_per_pos < 0.1 * max_b)[0]

        for ci in top_idx_sorted:
            i = int(idx[ci])
            j = int(peak_k - i)
            if a[i] < 1e-14:
                continue

            # Try reducing a[i] (and a[j] if different) and adding to safe positions
            for frac in [0.3, 0.5, 0.7]:
                amount = a[i] * frac
                if amount < 1e-15:
                    continue

                a_trial = a.copy()
                a_trial[i] -= amount
                if j != i:
                    a_trial[j] -= amount * 0.5
                    amount_j = amount * 0.5
                    total_redist = amount + amount_j
                else:
                    total_redist = amount

                a_trial = np.maximum(a_trial, 0)

                # Redistribute to safe positions
                if len(safe_positions) > 0:
                    n_safe = len(safe_positions)
                    add_per = total_redist / n_safe
                    a_trial[safe_positions] += add_per

                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_a = a_trial.copy()
                    best_score = score
                    improved = True
                    break

            if improved:
                break

        if not improved:
            # Try small random perturbations
            for _ in range(10):
                a_trial = a + np.random.standard_normal(n) * 0.001
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_a = a_trial.copy()
                    best_score = score
                    improved = True
                    break

        if improved:
            no_improve = 0
        else:
            no_improve += 1

    return best_a, best_score


def lbgfs_adaptive(a0, time_limit, verbose=False):
    """L-BFGS with adaptive temperature annealing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()

    # Temperature schedule: start high, decrease geometrically
    T_start = 80.0
    T_end = 0.5
    n_phases = 5
    T_values = np.geomspace(T_start, T_end, n_phases)

    for T in T_values:
        if monotonic() - t0 > time_limit:
            break
        phase_time = (time_limit - (monotonic() - t0)) / (n_phases - int(np.where(T_values == T)[0][0]))
        phase_time = min(phase_time, 80.0)

        z0 = np.log(np.maximum(a, 1e-30))
        z0 = z0 - z0.mean()
        stopped = [False]
        phase_deadline = monotonic() + phase_time

        def func(z):
            if monotonic() > phase_deadline:
                stopped[0] = True
                return 1e15
            z_shift = z - z.max()
            exp_z = np.exp(z_shift)
            a_curr = exp_z / exp_z.sum()
            b_curr = fast_convolve(a_curr)
            m = b_curr.max()
            return 2.0 * n * (m + T * np.log(np.exp((b_curr - m) / T).sum()))

        def jac(z):
            if monotonic() > phase_deadline:
                stopped[0] = True
                return np.zeros(n)
            z_shift = z - z.max()
            exp_z = np.exp(z_shift)
            a_curr = exp_z / exp_z.sum()
            b_curr = fast_convolve(a_curr)
            m = b_curr.max()
            scaled = (b_curr - m) / T
            exp_b = np.exp(scaled)
            w = exp_b / exp_b.sum()

            grad_a = np.zeros(n)
            for k in range(len(b_curr)):
                if w[k] < 1e-10:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idxs = np.arange(js, je + 1)
                grad_a[idxs] += 2.0 * w[k] * a_curr[k - idxs]

            grad_a *= 2.0 * n
            S = np.dot(grad_a, a_curr)
            grad_z = a_curr * (grad_a - a_curr * S)
            return grad_z

        try:
            res = minimize(
                func, z0, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': 500, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': 7000, 'disp': False}
            )
            if not stopped[0]:
                z_final = res.x
                z_shift = z_final - z_final.max()
                a_new = np.exp(z_shift)
                a_new = np.maximum(a_new, 0)
                a_new = a_new / a_new.sum()
                score = 2.0 * n * fast_convolve(a_new).max()
                if score < best_score - 1e-14:
                    best_score = score
                    best_a = a_new.copy()
                    a = a_new
        except Exception:
            pass

        if verbose:
            print(f"    L-BFGS T={T:.1f}: score={best_score:.6f}")

    return best_a, best_score


def subgradient_momentum(a0, time_limit, verbose=False):
    """Momentum-based subgradient with adaptive learning rate and peak focusing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    v = np.zeros(n)
    lr = 0.002
    momentum = 0.92
    t0 = monotonic()
    no_improve = 0
    best_lr = lr

    while monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        max_b = b.max()

        # Focus on top peaks with soft weighting
        k_focus = min(30, len(b))
        peaks = np.argpartition(b, -k_focus)[-k_focus:]
        pv = b[peaks]
        temp = 0.001 * max(pv.max(), 1e-10)
        scaled = (pv - pv.max()) / temp
        w = np.exp(scaled)
        w = w / w.sum()

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idxs = np.arange(js, je + 1)
            sg[idxs] += 2.0 * w[i] * a[k - idxs]

        # Update velocity
        v = momentum * v - lr * sg

        # Candidate
        a_candidate = a + v

        # Project onto simplex
        u = np.sort(a_candidate)[::-1]
        cssv = np.cumsum(u)
        cond = u * np.arange(1, n + 1) > (cssv - 1)
        if cond.any():
            rho = int(np.nonzero(cond)[0][-1])
            theta = (cssv[rho] - 1.0) / (rho + 1.0)
            a_new = np.maximum(a_candidate - theta, 0.0)
        else:
            a_new = np.ones(n) / n
        a_new = a_new / a_new.sum()

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            best_lr = lr
        else:
            no_improve += 1

        # Adaptive learning rate
        if no_improve > 50:
            lr *= 0.8
        elif no_improve > 200:
            # Restart from best with perturbation
            a = best_a + np.random.standard_normal(n) * 0.001
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
            lr = best_lr
            no_improve = 0

        if verbose and (monotonic() - t0) % 30 < 0.1:
            elapsed = monotonic() - t0
            print(f"    SubGrad ({elapsed:.0f}s): score={best_score:.6f} lr={lr:.6f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def make_structured_init(n, style='mixed'):
    """Create structured initializations."""
    if style == 'uniform':
        return np.ones(n) / n
    elif style == 'u_shape':
        t = np.linspace(0, 1, n)
        f = (t - 0.5) ** 2 + 0.1
        return f / f.sum()
    elif style == 'bimodal':
        t = np.linspace(0, 1, n)
        f = np.exp(-20 * (t - 0.2) ** 2) + np.exp(-20 * (t - 0.8) ** 2) + 0.01
        return f / f.sum()
    elif style == 'wave':
        t = np.linspace(0, 2 * np.pi, n, endpoint=False)
        f = 0.5 + 0.3 * np.cos(4 * t) + 0.1 * np.cos(7 * t)
        f = np.maximum(f, 0.01)
        return f / f.sum()
    elif style == 'mixed':
        t = np.linspace(0, 1, n)
        f = 0.3 + 0.2 * np.cos(6 * np.pi * t) + 0.1 * np.sin(10 * np.pi * t)
        f = np.maximum(f, 0.01)
        return f / f.sum()
    else:
        return np.ones(n) / n


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate and return updated best."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining enhanced LP fictitious play,
    active-set refinement, L-BFGS annealing, subgradient momentum,
    and peak redistribution. Explores multiple initializations and lengths.
    """
    np.random.seed(seed)
    deadline = monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = p_score
    else:
        best_seq = np.ones(768) / 768.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Initial: n=768, score={best_score:.6f}")

    n_parent = len(best_seq)

    # === Phase 1: Multi-initialization short FP ===
    print(f"\n=== Phase 1: Multi-init FP (best={best_score:.6f}) ===")
    n_try = n_parent
    rem = deadline - monotonic() - 800
    t_per_init = max(20, min(80, rem / 5))

    inits = [
        ('parent', best_seq if len(best_seq) == n_try else interpolate_sequence(best_seq, n_try)),
        ('u_shape', make_structured_init(n_try, 'u_shape')),
        ('bimodal', make_structured_init(n_try, 'bimodal')),
        ('wave', make_structured_init(n_try, 'wave')),
    ]

    for name, init_seq in inits:
        rem = deadline - monotonic() - 700
        if rem < t_per_init:
            break
        print(f"  FP init={name}, t={t_per_init:.0f}s")
        a_fp, score_fp, _ = fictitious_play_enhanced(n_try, t_per_init, initial=init_seq, solver=solver)
        best_seq, best_score = check_and_update(a_fp, f"FP_{name}", best_seq, best_score)

    # === Phase 2: Long FP from best ===
    print(f"\n=== Phase 2: Long FP (best={best_score:.6f}) ===")
    rem = deadline - monotonic() - 500
    if rem > 60:
        t_alloc = min(rem * 0.6, 400.0)
        print(f"  Long FP n={n_parent}, budget={t_alloc:.0f}s")
        a_fp, score_fp, iters = fictitious_play_enhanced(n_parent, t_alloc, initial=best_seq, solver=solver, verbose=True)
        best_seq, best_score = check_and_update(a_fp, f"FP_long_it{iters}", best_seq, best_score)

    # === Phase 3: Active-set refinement ===
    print(f"\n=== Phase 3: Active-Set Refine (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 300
        if rem > 30:
            t_alloc = min(rem * 0.5, 200.0)
            print(f"  ActiveSet n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_as, _ = active_set_refine(best_seq, t_alloc, verbose=True)
            best_seq, best_score = check_and_update(a_as, "active_set", best_seq, best_score)

    # === Phase 4: L-BFGS adaptive annealing ===
    print(f"\n=== Phase 4: L-BFGS Anneal (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 300
        if rem > 30:
            t_alloc = min(rem * 0.5, 150.0)
            print(f"  L-BFGS n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_lb, _ = lbgfs_adaptive(best_seq, t_alloc, verbose=True)
            best_seq, best_score = check_and_update(a_lb, "lbgfs_adapt", best_seq, best_score)

    # === Phase 5: Subgradient momentum ===
    print(f"\n=== Phase 5: Subgradient (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 200
        if rem > 20:
            t_alloc = min(rem * 0.5, 100.0)
            print(f"  SubGrad n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_sg, _ = subgradient_momentum(best_seq, t_alloc)
            best_seq, best_score = check_and_update(a_sg, "subgrad", best_seq, best_score)

    # === Phase 6: Peak redistribution ===
    print(f"\n=== Phase 6: Peak Redistribute (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 150
        if rem > 15:
            t_alloc = min(rem * 0.5, 80.0)
            print(f"  PeakRedist budget={t_alloc:.0f}s")
            a_pr, _ = peak_redistribute(best_seq, t_alloc)
            best_seq, best_score = check_and_update(a_pr, "peak_redist", best_seq, best_score)

    # === Phase 7: Second round - FP + Active Set ===
    print(f"\n=== Phase 7: FP Round 2 (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 100
        if rem > 40:
            t_alloc = min(rem * 0.4, 100.0)
            print(f"  FP2 budget={t_alloc:.0f}s")
            a_fp2, _, _ = fictitious_play_enhanced(len(best_seq), t_alloc, initial=best_seq, solver=solver)
            best_seq, best_score = check_and_update(a_fp2, "FP2", best_seq, best_score)

            rem = deadline - monotonic() - 60
            if rem > 20:
                t_as = min(rem * 0.5, 60.0)
                a_as2, _ = active_set_refine(best_seq, t_as)
                best_seq, best_score = check_and_update(a_as2, "active_set_2", best_seq, best_score)

    # === Phase 8: Try alternative lengths ===
    print(f"\n=== Phase 8: Alt Lengths (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        alt_lengths = [512, 1024]
        for alt_n in alt_lengths:
            if alt_n == n_best:
                continue
            rem = deadline - monotonic() - 50
            if rem < 30:
                break
            t_alloc = min(rem * 0.4, 80.0)
            init_alt = interpolate_sequence(best_seq, alt_n)
            a_alt, _, _ = fictitious_play_enhanced(alt_n, t_alloc, initial=init_alt, solver=solver)
            best_seq, best_score = check_and_update(a_alt, f"alt_n{alt_n}", best_seq, best_score)

            # Quick polish on alt length
            rem = deadline - monotonic() - 30
            if rem > 15 and len(best_seq) == alt_n:
                a_pol, _ = active_set_refine(best_seq, min(rem * 0.5, 30.0))
                best_seq, best_score = check_and_update(a_pol, f"alt_n{alt_n}_pol", best_seq, best_score)

    # === Phase 9: Final fine-grained polish ===
    print(f"\n=== Phase 9: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 15
        if rem > 10:
            t_final = min(rem * 0.4, 30.0)
            a_fin, _ = active_set_refine(best_seq, t_final)
            best_seq, best_score = check_and_update(a_fin, "final_as", best_seq, best_score)

        rem = deadline - monotonic() - 5
        if rem > 8:
            t_sm = min(rem * 0.5, 20.0)
            a_sm, _ = lbgfs_adaptive(best_seq, t_sm)
            best_seq, best_score = check_and_update(a_sm, "final_lb", best_seq, best_score)

        rem = deadline - monotonic() - 3
        if rem > 5:
            a_prf, _ = peak_redistribute(best_seq, min(rem * 0.6, 10.0))
            best_seq, best_score = check_and_update(a_prf, "final_pr", best_seq, best_score)

    # Final verification
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
