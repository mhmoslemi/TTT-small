Where a is a non-negative sequence of length n.

Key observations:
1. The objective is scale-invariant (scaling a doesn't change the score), so we can normalize sum(a) = 1 or any other convenient value.
2. The convolution `conv(a,a)` has length 2n-1.
3. We want to minimize the maximum of the auto-convolution relative to the total mass.

For a uniform sequence of length n: conv(a,a) = [1, 2, 3, ..., n-1, n, n-1, ..., 2, 1] * (1/n)^2
max = n/n^2 = 1/n
score = 2*n*(1/n)/1 = 2

So uniform gives score = 2. We need to do better.

The theoretical minimum for this type of problem relates to the "flatness" of the autocorrelation. This is related to:
- Sidon sets
- Costas arrays
- M-sequences
- Difference sets

For the specific form `2*n*max(conv(a,a))/(sum(a))^2`, if we think of `a` as a probability distribution, then `conv(a,a)` is the distribution of the sum of two independent random variables with that distribution. We want to minimize the maximum probability of the sum.

For a uniform distribution on {0,...,n-1}, the convolution is triangular with max = n/n^2 = 1/n.

If we could make the convolution perfectly flat (which is impossible for a non-negative sequence), we'd get max = 1/(2n-1), giving score = 2n/(2n-1) ≈ 1.

So the theoretical lower bound is approximately 1.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based fictitious play for initial refinement
2. Smoothed L-BFGS in log-softmax parameterization (key innovation)
3. Active-set peak-targeted coordinate descent
4. Multi-start exploration

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def cross_corr(a, b):
    """Compute cross-correlation: result[i] = sum_j a[j] * b[i+j] (valid range)."""
    # We want c[i] = sum_j a[j] * b[i+j] for i = 0, ..., n_a-1
    # where j ranges over valid indices (0 <= j < n_a, 0 <= i+j < n_b)
    n_a = len(a)
    n_b = len(b)
    # Use FFT: cross-correlation via convolution
    # sum_j a[j] * b[i+j] = (a_flipped * b)[n_a-1+i] / ... 
    # Actually: let a_flipped[k] = a[n_a-1-k]
    # Then sum_j a[j] * b[i+j] = sum_j a_flipped[n_a-1-j] * b[i+j]
    # Let m = n_a-1-j, j = n_a-1-m:
    # = sum_m a_flipped[m] * b[i+n_a-1-m]
    # = (conv(a_flipped, b))[i+n_a-1]
    a_flipped = a[::-1]
    conv = fast_convolve_pair(a_flipped, b)
    return conv[n_a-1:n_a-1+n_a]

def fast_convolve_pair(a, b):
    """Linear convolution of two sequences via FFT."""
    n = len(a) + len(b) - 1
    fft_size = 1 << (n - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:n]

def compute_smoothed_grad(a, tau):
    """
    Compute gradient of smoothed objective f_s(a) = (2n/tau)*log(sum exp(tau*b_k))
    w.r.t. a. Returns gradient vector of length n.
    """
    n = len(a)
    b = fast_convolve(a)
    # Compute lambda = softmax(tau * b)
    tb = tau * b
    tb_max = tb.max()
    exp_tb = np.exp(tb - tb_max)
    lambda_ = exp_tb / exp_tb.sum()
    
    # grad_a[i] = 4n * sum_k lambda_k * a[k-i] for valid k-i
    # This is 4n * cross_corr(a, lambda)
    # cross_corr(a, lambda)[i] = sum_j a[j] * lambda[i+j]
    # Wait, I need: sum_k lambda_k * a[k-i] where k-i is in [0, n-1]
    # Let m = k-i, k = i+m:
    # = sum_m lambda[i+m] * a[m] (for m in [0, n-1] and i+m in [0, 2n-2])
    # This is cross_corr in the sense: sum_m a[m] * lambda[i+m]
    # = (conv(a_flipped, lambda))[n-1+i]
    a_flipped = a[::-1]
    conv_al = fast_convolve_pair(a_flipped, lambda_)
    grad = 4.0 * n * conv_al[n-1:n-1+n]
    return grad, lambda_

def softmax_obj_and_grad(theta, tau):
    """
    Objective and gradient for L-BFGS in log-softmax parametrization.
    a = softmax(theta)
    f = (2n/tau) * log(sum_k exp(tau * b_k))
    """
    # Compute a from theta (stable softmax)
    theta_max = theta.max()
    exp_theta = np.exp(theta - theta_max)
    a = exp_theta / exp_theta.sum()
    
    # Compute smoothed objective and gradient in a-space
    n = len(a)
    b = fast_convolve(a)
    tb = tau * b
    tb_max = tb.max()
    exp_tb = np.exp(tb - tb_max)
    log_sum = tb_max + np.log(exp_tb.sum())
    obj = 2.0 * n / tau * log_sum
    
    # Gradient in a-space
    lambda_ = exp_tb / exp_tb.sum()
    a_flipped = a[::-1]
    conv_al = fast_convolve_pair(a_flipped, lambda_)
    grad_a = 4.0 * n * conv_al[n-1:n-1+n]
    
    # Transform to theta-space: grad_theta = D(a) * (I - a a^T) * grad_a
    # = a * (grad_a - (a^T grad_a))
    a_dot_grad = np.dot(a, grad_a)
    grad_theta = a * (grad_a - a_dot_grad)
    
    return obj, grad_theta

def lbfgs_smoothed(a_init, time_limit, tau_schedule=None, max_iter=None):
    """
    Run L-BFGS on the smoothed objective in log-softmax parameterization.
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-15).copy()
    a = a / a.sum()
    
    # Initial theta
    theta = np.log(a)  # a = softmax(theta) when a is the softmax of log(a)
    # Actually softmax(log(a)) = a / sum(a) = a (since sum(a)=1)
    # Good, so theta = log(a) gives the right initial point.
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    # Tau schedule: start high (smooth), decrease over time
    if tau_schedule is None:
        tau_schedule = [(100, 0.3), (50, 0.3), (25, 0.2), (10, 0.2), (5, 0.15), (2, 0.1)]
    # Each entry: (tau, fraction_of_time)
    total_frac = sum(f for _, f in tau_schedule)
    tau_schedule = [(t, f/total_frac) for t, f in tau_schedule]
    
    current_theta = theta.copy()
    
    for tau_frac, time_frac in tau_schedule:
        t_alloc = time_frac * time_limit
        if time.monotonic() - t0 > time_limit - 10:
            break
        
        # Run L-BFGS for this tau
        result = minimize(
            fun=lambda th: softmax_obj_and_grad(th, tau_frac)[0],
            x0=current_theta,
            jac=lambda th: softmax_obj_and_grad(th, tau_frac)[1],
            method='L-BFGS-B',
            options={'maxiter': 500, 'maxfun': 20000, 
                     'maxls': 50, 'ftol': 1e-12, 'gtol': 1e-8}
        )
        
        current_theta = result.x
        
        # Convert back to a
        theta_max = current_theta.max()
        a_new = np.exp(current_theta - theta_max)
        a_new = a_new / a_new.sum()
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        # Evaluate true objective
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        print(f"    L-BFGS tau={tau_frac}: score={score:.6f} (iters={result.nit})")
    
    return best_a, best_score

def lp_best_response(a_fixed, n):
    """LP: minimize max_k conv(a_fixed, b) subject to b >= 0, sum(b) = 1."""
    from scipy.optimize import linprog
    
    # Build constraint matrix: T @ b <= t
    # T is (2n-1) x n, T[k,j] = a_fixed[k-j] if 0 <= k-j < n
    # This is too large for linprog. Use a different approach.
    # 
    # Alternative: use the fact that we can solve this with a custom approach.
    # Actually, for n up to ~1000, the LP has 2n-1 + 1 = 2n constraints and n+1 variables.
    # That's about 2000 constraints and 1000 variables. Should be feasible.
    
    # Actually, let's use a different approach: iterative bisection on t.
    # For a given t, check if there exists b >= 0, sum(b) = 1, T@b <= t.
    # This is a feasibility LP, which is equivalent to checking if the polytope is non-empty.
    
    # Even simpler: use the dual. The dual of:
    # min t
    # s.t. T@b <= t*1, sum(b) = 1, b >= 0
    # is:
    # max 1 * sum(lambda) - nu  ... hmm this is getting complicated.
    
    # Let me just use a simple approach: projected gradient on the LP.
    # Or better: use cvxpy if available.
    
    try:
        import cvxpy as cp
        b_var = cp.Variable(n, nonneg=True)
        t_var = cp.Variable()
        # Build T efficiently
        T_rows = []
        for k in range(2*n-1):
            row = np.zeros(n)
            j_start = max(0, k-n+1)
            j_end = min(n-1, k)
            for j in range(j_start, j_end+1):
                row[j] = a_fixed[k-j]
            T_rows.append(row)
        T = np.array(T_rows)
        
        constraints = [
            T @ b_var <= t_var,
            cp.sum(b_var) == 1
        ]
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        prob.solve(solver=cp.ECOS, verbose=False)
        
        if prob.status in ('optimal', 'optimal_inaccurate'):
            b_val = np.asarray(b_var.value, dtype=np.float64)
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s > 1e-15:
                b_val = b_val / s
            return b_val, float(t_var.value)
    except Exception as e:
        print(f"    LP solver error: {e}")
    
    return None, None

def fictitious_play_steps(a0, n_steps, solver=None):
    """Run a few steps of fictitious play."""
    n = len(a0)
    a = np.maximum(a0, 0).copy()

    b = a.copy()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    
    for step in range(n_steps):
        # Best response to a -> new b
        b_new, M_b = lp_best_response(a, n)
        if b_new is not None:
            b = b_new
        # Best response to b -> new a  
        a_new, M_a = lp_best_response(b, n)
        if a_new is not None:
            a = a_new
        
        # Check score
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
            
        # Also check the average
        avg = (a + b) / 2
        score_avg = 2.0 * n * fast_convolve(avg).max()
        if score_avg < best_score:
            best_score = score_avg
            best_a = avg.copy()
    
    return best_a, best_score

def simplex_project(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def nesterov_smoothed(a0, time_limit, top_k=20, lr_init=0.01):
    """Nesterov-accelerated projected subgradient on the true (max) objective."""
    n = len(a0)
    a = np.maximum(a0, 1e-15).copy()
    a = a / a.sum()
    
    y = a.copy()
    v = np.zeros(n)
    lr = lr_init
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        t_k = (1 + np.sqrt(1 + 4 * (it - 1) ** 2)) / 2
        beta_k = (t_k - 1) / t_k
        
        # Compute subgradient of max(conv(y,y))
        b = fast_convolve(y)
        n_pk = min(top_k, len(b))
        peaks = np.argpartition(b, -n_pk)[-n_pk:]
        pv = b[peaks]
        temp = 0.005 * max(pv.max(), 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        w /= ws
        
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-14:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            sg[j] += 2.0 * w[idx] * y[k - j]
        
        v = beta_k * v + sg
        norm_v = np.linalg.norm(v)
        if norm_v > 1e-15:
            v = v / norm_v
        
        y_new = y - lr * v
        y_projected = simplex_project(y_new)
        
        score = 2.0 * n * fast_convolve(y_projected).max()
        if score < best_score - 1e-13:
            best_score = score
            best_a = y_projected.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        y = y_projected
        
        if no_improve > 100:
            lr *= 0.8
            no_improve = 0
        
        if no_improve > 400:
            noise = np.random.default_rng(it * 31 + 7).standard_normal(n) * 0.005
            y = simplex_project(best_a + noise)
            v = np.zeros(n)
            lr = max(lr_init * 0.5, lr * 2)
            no_improve = 0
        
        lr *= 0.9998
    
    return best_a, best_score

def coord_descent_fast(a0, time_limit, n_sweeps=100):
    """Fast coordinate descent trying to reduce the max peak."""
    n = len(a0)

        perm = rng.permutation(n)

            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[i]
            if old_val < 1e-12:

                a_trial = a.copy()
