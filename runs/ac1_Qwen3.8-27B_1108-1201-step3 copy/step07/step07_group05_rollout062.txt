import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix, hstack as sp_hstack


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the optimization score for a normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def normalize(a):
    """Clip to non-negative and normalize sum to 1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix(a):
    """Build sparse matrix C such that C @ g = full_convolve(a, g)."""
    n = len(a)
    j_idx = np.arange(n * n) // n
    offset = np.arange(n * n) % n
    rows = j_idx + offset
    cols = np.repeat(np.arange(n), n)
    vals = np.tile(a, n)
    C = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return C


def solve_lp_direction(a, M):
    """
    Solve LP: min t s.t. (a*g)_k <= t*M for all k, sum(g)=1, g>=0.
    Returns optimal g vector.
    """
    n = len(a)
    C = build_conv_matrix(a)
    # A_ub: C@g - M*t <= 0
    A_ub = sp_hstack([C, csr_matrix(-M * np.ones((2 * n - 1, 1)))])
    b_ub = np.zeros(2 * n - 1)
    # A_eq: sum(g) = 1
    A_eq = np.hstack([np.ones((1, n)), [0.0]])
    b_eq = np.ones(1)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    bounds = [(0, None)] * (n + 1)

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            g = np.maximum(res.x[:n], 0)
            s = g.sum()
            if s > 1e-15:
                return g / s
    except Exception:
        pass
    return a.copy()


def line_search_alpha(a, g, n_evals=15):
    """Find alpha in [0,1] minimizing score((1-alpha)*a + alpha*g)."""
    best_alpha = 0.0
    best_score = compute_score(a)
    alphas = np.linspace(0.1, 1.0, n_evals)
    for alpha in alphas:

        trial = normalize(trial)
        sc = compute_score(trial)
        if sc < best_score:
            best_score = sc
            best_alpha = alpha
    return best_alpha, best_score

def build_conv_matrix(a):

    n = len(a)
