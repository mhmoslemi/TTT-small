import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft

def fast_convolve(a, b=None):
    """Full linear convolution."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def solve_lp(f, M):

    Solve the LP: maximize sum(g) subject to (f*g)_k <= M for all k, g >= 0.
    f is the current sequence, M is the peak bound.
    Returns the optimal g.

    n = len(f)

    # (f*g)_k = sum_i f_i * g_{k-i} where g_j = 0 for j outside [0, n-1]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based fictitious play.
The key algorithm iteratively solves an LP to find the "best response"
sequence against the current one, using the empirical average as the
fictitious play update. Combined with local refinement using coordinate
descent and smoothed gradient methods.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_convolve(a, b=None):
    """Full linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def solve_lp_best_response(f, M):
    """
    Solve LP: maximize sum(g) subject to (f*g)_k <= M for all k, g >= 0.
    
    Parameters:
        f: current sequence (length n)
        M: peak bound
    
    Returns:
        (g, V) where g is the optimal solution and V = sum(g), 
        or (None, None) if infeasible.
    """
    n = len(f)
    num_rows = 2 * n - 1
    
    # Build A_ub: A_ub[k, j] = f[k-j] if 0 <= k-j < n
    A_ub = np.zeros((num_rows, n))
    for j in range(n):
        A_ub[j:j+n, j] = f
    
    c = -np.ones(n)
    b_ub = np.full(num_rows, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.status == 0:
        return res.x, -res.fun  # res.fun is min of -sum(g), so V = -res.fun
    return None, None


def fictitious_play(f0, max_iters, time_limit, use_average=True, verbose=True):
    """
    LP-based fictitious play to minimize max auto-correlation peak.
    
    Returns the best sequence found.
    """
    n = len(f0)
    f = f0.copy()
    best_f = f0.copy()
    best_score = compute_score(f0)
    
    avg_f = np.zeros(n)
    t = 0
    t0 = time.monotonic()
    
    for iteration in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute current peak
        conv_f = fast_convolve(f)
        M = conv_f.max()
        
        # Solve LP for best response
        g, V = solve_lp_best_response(f, M)
        if g is None:
            # LP failed, try with slightly larger M
            g, V = solve_lp_best_response(f, M * 1.001)
            if g is None:
                break
        
        # Rescale g to sum = 1
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        # Update
        t += 1
        if use_average:
            avg_f = avg_f * (1 - 1.0/t) + g * (1.0/t)
            candidate = avg_f
        else:
            candidate = g
        
        candidate = np.maximum(candidate, 0)
        cand_sum = candidate.sum()
        if cand_sum < 1e-15:
            break
        candidate = candidate / cand_sum
        
        # Check score
        sc = compute_score(candidate)
        if sc < best_score:
            best_score = sc
            best_f = candidate.copy()
        
        # Also check g directly
        sc_g = compute_score(g)
        if sc_g < best_score:
            best_score = sc_g
            best_f = g.copy()
        
        # Update f for next iteration
        f = g
        
        if verbose and iteration % 100 == 0:
            print(f"  FP iter={iteration}, V={V:.6f}, best_score={best_score:.8f}")
    
    return best_f, best_score


def coordinate_descent(a0, time_limit, n_pairs=500, verbose=False):
    """Coordinate descent with pairwise moves for local refinement."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            if a[i] < 1e-14:
                continue
            
            best_local = best_score
            best_t = 0.0
            
            base = a[i]
            for frac in [0.05, 0.1, 0.2, 0.5, 0.8, 0.01, 0.02]:
                delta = frac * base
                if delta >= a[i] - 1e-15:
                    delta = 0.5 * a[i]
                
                a_trial = a.copy()
                a_trial[i] -= delta
                a_trial[j] += delta
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score(a_trial)
                if sc < best_local - 1e-14:
                    best_local = sc
                    best_t = delta
            
            # Also try reverse
            if a[j] > 1e-14:
                for frac in [0.05, 0.1, 0.2, 0.5]:
                    delta = frac * a[j]
                    a_trial = a.copy()
                    a_trial[j] -= delta
                    a_trial[i] += delta
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    sc = compute_score(a_trial)
                    if sc < best_local - 1e-14:
                        best_local = sc
                        best_t = -delta
            
            if best_t != 0.0:
                a[i] -= best_t
                a[j] += best_t
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = best_local
                best_a = a.copy()
                improved = True
        
        if not improved:
            # Perturb
            noise = rng.standard_normal(n) * 0.0005
            a = best_a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
    
    return best_a, best_score


def smoothed_grad_descent(a0, time_limit, T=500, lr_init=0.1, momentum=0.9, verbose=False):
    """Smoothed gradient descent with Nesterov momentum."""
    n = len(a0)
    # Work in log-space
    a_safe = np.maximum(a0, 1e-30)
    z = np.log(a_safe / a_safe.sum())
    v = np.zeros(n)
    best_z = z.copy()
    best_score = compute_score(a0)
    t0 = time.monotonic()
    lr = lr_init
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov lookahead
        z_la = z - momentum * v
        # Softmax
        z_la_max = z_la.max()
        a_la = np.exp(z_la - z_la_max)
        a_la = a_la / a_la.sum()
        
        # Compute gradient of smoothed objective
        conv = fast_convolve(a_la)
        shifted = conv / T
        max_sh = shifted.max()
        w = np.exp(shifted - max_sh)
        w = w / w.sum()
        
        # grad_a = 2 * cross_correlation(w, a)
        # Actually: d/d a_i [sum_k w_k * (conv(a))_k] = 2 * sum_k w_k * a_{|k-i|}... 
        # More precisely: (conv(a))_k = sum_j a_j * a_{k-j}
        # d/d a_i = sum_j [delta(j,i)*a_{k-j} + a_j*delta(k-j,i)]
        #          = a_{k-i} + a_{k-i} = 2*a_{k-i} (for valid indices)
        # So grad_a_i = 2 * sum_k w_k * a_{k-i} (with proper index handling)
        # This is the cross-correlation of w with a, times 2
        
        # Cross-correlation: (w * a)_i = sum_k w_k * a_{k-i}
        # Using FFT: cross_corr = ifft(conj(FFT(w)) * FFT(a))
        fft_size = 1 << (2*n-2).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a_la, fft_size)
        grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
        
        # Project to log-space: grad_z = a * (grad_a - sum(grad_a * a))
        S = np.dot(grad_a, a_la)
        grad_z = a_la * (grad_a - S)
        grad_z = (2.0 * n / T) * grad_z
        
        # Update
        v = momentum * v + lr * grad_z
        z_new = z - v
        z_new = np.clip(z_new, -30, 30)
        
        a_new = np.exp(z_new - z_new.max())
        a_new = a_new / a_new.sum()
        
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_z = z_new.copy()
        
        z = z_new
        lr = lr_init * (0.9999 ** it)
    
    a_out = np.exp(best_z - best_z.max())
    a_out = a_out / a_out.sum()
    return a_out, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
