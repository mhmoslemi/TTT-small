"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FISTA with log-sum-exp annealing,
active-set subgradient methods, and exact coordinate descent.
Key innovations: FISTA with acceleration, FFT-based smooth gradient,
temperature annealing schedule, and peak-targeted coordinate moves.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max(conv(a,a)) for a with sum=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return np.array([1.0])
    v = np.asarray(v, dtype=np.float64)
    if v.sum() < 1e-18:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_gradient(a, T):
    """
    Compute gradient of (2n/T)*log(sum_k exp(T*conv(a,a)[k])).
    Uses FFT for O(n log n) complexity.
    """
    n = len(a)
    conv = fast_convolve(a)
    Tconv = T * conv
    max_Tconv = Tconv.max()
    exp_Tconv = np.exp(Tconv - max_Tconv)
    w = exp_Tconv / exp_Tconv.sum()  # length 2n-1

    # grad[j] = 4n * sum_m a[m] * w[m+j] (with w out of bounds = 0)
    # = 4n * conv(a, w_flip)[2n-2-j] where w_flip = w[::-1]
    w_flip = w[::-1]
    fft_size = 1 << (3 * n - 1).bit_length()
    A = rfft(a, fft_size)
    Wf = rfft(w_flip, fft_size)
    conv_aw = irfft(A * Wf, fft_size)[:3 * n - 2]

    indices = 2 * n - 2 - np.arange(n)
    grad = 4.0 * n * conv_aw[indices]
    return grad


def smooth_objective(a, T):
    """Compute (2n/T)*log(sum_k exp(T*conv_k)) for monitoring."""
    n = len(a)
    conv = fast_convolve(a)
    Tconv = T * conv
    max_val = Tconv.max()
    lse = max_val + np.log(np.exp(Tconv - max_val).sum())
    return (2.0 * n / T) * np.exp(T * (lse - max_val)) * np.exp(max_val)


def subgradient_active_set(a, eps_rel=0.01):
    """
    Compute a subgradient of 2n*max(conv(a,a)) using the active set.
    Active set = {k : conv[k] >= max*(1-eps_rel)}
    """
    n = len(a)
    conv = fast_convolve(a)
    max_conv = conv.max()
    if max_conv < 1e-30:
        return np.zeros(n)
    active = conv >= max_conv * (1.0 - eps_rel)
    if not active.any():
        active = np.zeros_like(conv, dtype=bool)
        active[np.argmax(conv)] = True
    w = active.astype(np.float64)
    w = w / w.sum()

    # Same formula as smooth gradient but with 0/1 weights (subgradient)
    w_flip = w[::-1]
    fft_size = 1 << (3 * n - 1).bit_length()
    A = rfft(a, fft_size)
    Wf = rfft(w_flip, fft_size)
    conv_aw = irfft(A * Wf, fft_size)[:3 * n - 2]
    indices = 2 * n - 2 - np.arange(n)
    grad = 4.0 * n * conv_aw[indices]
    return grad


def fista_smoothed(a0, T, time_limit, max_iter=200000, verbose=True):

    z = a.copy()
    t_val = 1.0
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    # Estimate Lipschitz constant

    alpha = 1.0 / L

    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
        z = a + (t_val - 1.0) / t_new * (a - z)

        else:

def fista_anneal(a0, time_limit, temperatures=None, verbose=True):
    """FISTA with temperature annealing schedule."""
    if temperatures is None:
        temperatures = [1000, 500, 200, 100, 50, 25, 10]

    a = np.asarray(a0, dtype=np.float64)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    total_time = time_limit

    for i, T in enumerate(temperatures):
        remaining = total_time - (time.monotonic() - t0)
        if remaining < 10:
            break
        # Allocate more time to lower temperatures
        frac = 1.0 / len(temperatures)
        alloc = min(remaining * 0.7, total_time * frac * 1.5)
        alloc = max(alloc, 5.0)

        if verbose:
            print(f"  Annealing T={T}, budget={alloc:.0f}s, current_best={best_score:.8f}")

        res_a, res_score = fista_smoothed(best_a, T, alloc, max_iter=100000, verbose=False)
        if res_score < best_score:
            best_score = res_score
            best_a = res_a.copy()

    return best_a, best_score


def subgradient_method(a0, time_limit, eps_rel=0.01, verbose=True):
    """Subgradient method with diminishing step size on the exact max objective."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    # Step size: alpha_k = c / sqrt(k)
    c_init = 0.1
    k = 0

    while time.monotonic() - t0 < time_limit:
        k += 1
        alpha = c_init / np.sqrt(k)
        alpha = min(alpha, 0.5)  # cap step size

        grad = subgradient_active_set(a, eps_rel)
        a_new = project_simplex(a - alpha * grad)

        score = compute_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()

        a = a_new

        if verbose and k % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad iter={k} alpha={alpha:.6f} best={best_score:.8f} elapsed={elapsed:.1f}s")

    return best_a, best_score

    Coordinate descent with exact 1D optimization for pair moves.
    For each pair (i,j), find optimal transfer t minimizing the score.
    Uses ternary search on the 1D function.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    def eval_pair_move(a_curr, i, j, t):
        """Evaluate score after moving t from i to j."""
        a_temp = a_curr.copy()
        a_temp[i] -= t
        a_temp[j] += t
        if a_temp[i] < -1e-15 or a_temp[j] < -1e-15:
            return np.inf
        a_temp = np.maximum(a_temp, 0)
        s = a_temp.sum()
        if s < 1e-15:
            return np.inf
        a_temp = a_temp / s
        return compute_score(a_temp)

    iters = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False

        # Generate candidate pairs (biased toward peak-contributing positions)
        conv = fast_convolve(a)
        top_k = min(5, 2 * n - 1)
        peak_idxs = np.argsort(conv)[::-1][:top_k]

        # Collect important positions from peaks
        important_positions = set()
        for pk in peak_idxs:
            pk = int(pk)
            start = max(0, pk - n + 1)
            end = min(n - 1, pk)
            for pos in range(start, end + 1):
                important_positions.add(pos)
        important_list = list(important_positions)

        # Try moves from important positions
        n_trials = 50
        for _ in range(n_trials):
            if time.monotonic() - t0 > time_limit:
                break

            if important_list and rng.random() < 0.7:
                i = important_list[rng.integers(0, len(important_list))]
            else:
                i = rng.integers(0, n)

            j = rng.integers(0, n)
            if i == j:
                continue
            if a[i] < 1e-12:
                continue

            # Ternary search for optimal t in [0, a[i]*0.9]
            lo = 0.0
            hi = a[i] * 0.9
            for _ in range(20):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = eval_pair_move(a, i, j, m1)
                f2 = eval_pair_move(a, i, j, m2)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1

            t_opt = (lo + hi) / 2.0
            if t_opt < 1e-14:
                continue

            a_trial = a.copy()
            a_trial[i] -= t_opt
            a_trial[j] += t_opt
            a_trial = np.maximum(a_trial, 0)
            s = a_trial.sum()
            if s < 1e-15:
                continue
            a_trial = a_trial / s
            sc = compute_score(a_trial)

            if sc < best_score - 1e-14:
                a = a_trial
                best_score = sc
                best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 10:
                # Perturb to escape local minimum
                noise = rng.standard_normal(n) * 0.002
                a = best_a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0

    return best_a, best_score


def peak_reduction(a0, time_limit, verbose=True):
    """
    Targeted peak reduction: identify the max peak and try to reduce it
    by moving mass from the most contributing positions to "safe" positions.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        max_idx = int(np.argmax(conv))
        max_val = conv[max_idx]

        # Find contributing pairs to the max peak
        start = max(0, max_idx - n + 1)
        end = min(n - 1, max_idx)
        idx = np.arange(start, end + 1)
        contribs = a[idx] * a[max_idx - idx]
        top_contrib = np.argsort(contribs)[::-1][:10]

        improved = False
        for ci in top_contrib:
            if time.monotonic() - t0 > time_limit:
                break
            i = int(idx[ci])
            if a[i] < 1e-10:
                continue

            # Try moving a fraction of a[i] to random "safe" positions
            for _ in range(5):
                j = rng.integers(0, n)
                if j == i:
                    continue
                for frac in [0.05, 0.1, 0.2, 0.3]:
                    t = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= t
                    a_trial[j] += t
                    a_trial = np.maximum(a_trial, 0)
                    s = a_trial.sum()
                    if s < 1e-15:
                        continue
                    a_trial = a_trial / s
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            # Try small random perturbations near the peak contributing positions
            for _ in range(20):
                i = int(rng.choice(idx))
                j = rng.integers(0, n)
                if j == i:
                    continue
                t = rng.random() * a[i] * 0.1
                a_trial = a.copy()
                a_trial[i] -= t
                a_trial[j] += t
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s > 1e-15:
                    a_trial = a_trial / s
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break

        if verbose and (iters := iters + 1) % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakRed iter={iters} best={best_score:.8f} elapsed={elapsed:.1f}s")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def raised_cosine_init(n):
    """Create a raised cosine shaped initialization."""
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a


def random_init(n, seed=0):
    """Random non-negative initialization."""
    rng = np.random.default_rng(seed)
    a = rng.random(n)
    a = a / a.sum()
    return a


def lbgfs_optimize(a0, time_limit, T=300, maxiter=3000):
    """L-BFGS optimization in log-space with soft-max smoothing (from previous program, adapted)."""
    from scipy.optimize import minimize
    n = len(a0)

    def softmax_from_z(z):
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        s = exp_z.sum()
        if s < 1e-300:
            return np.ones(len(z)) / len(z)
        return exp_z / s

    def z_from_a(a_vec):
        a_safe = np.maximum(a_vec, 1e-30)
        return np.log(a_safe)

    def smoothed_obj(z):
        a_vec = softmax_from_z(z)
        conv = fast_convolve(a_vec)
        shifted = conv / T
        max_shifted = shifted.max()
        lse = max_shifted + np.log(np.exp(shifted - max_shifted).sum())
        return (2.0 * n / T) * np.exp(T * lse - T * max_shifted) * np.exp(T * max_shifted)

    def smoothed_grad(z):
        a_vec = softmax_from_z(z)
        conv = fast_convolve(a_vec)
        shifted = conv / T
        max_shifted = shifted.max()
        w = np.exp(shifted - max_shifted)
        w = w / w.sum()
        # Gradient w.r.t. z: a * (grad_a - sum(grad_a * a))
        # grad_a from our formula
        w_flip = w[::-1]
        fft_size = 1 << (3 * n - 1).bit_length()
        A = rfft(a_vec, fft_size)
        Wf = rfft(w_flip, fft_size)
        conv_aw = irfft(A * Wf, fft_size)[:3 * n - 2]
        indices = 2 * n - 2 - np.arange(n)
        grad_a = 4.0 * n * conv_aw[indices]
        S = np.dot(grad_a, a_vec)
        grad_z = a_vec * (grad_a - S)
        return (2.0 * n / T) * grad_z

    z0 = z_from_a(a0)
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        return smoothed_obj(z)

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        return smoothed_grad(z)

    try:
        res = minimize(fun=func, x0=z0, jac=jac, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                                'maxfun': maxiter * 50, 'disp': False})
        a_out = softmax_from_z(res.x)
        score = compute_score(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except Exception:
        pass

    return best_a, best_score
