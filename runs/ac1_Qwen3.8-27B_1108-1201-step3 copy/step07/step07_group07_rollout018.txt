import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_subgradient(a, w):
    """
    Compute subgradient of max_k (w[k] * conv(a,a)[k]) w.r.t. a.
    grad[i] = sum_k w[k] * 2 * a[k-i] (zero-padded)
    Computed via FFT: grad = 2 * irfft(conj(FFT(a)) * FFT(w))[:n]
    """
    n = len(a)
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    g = irfft(np.conj(A) * W, fft_size)[:n]
    return 2.0 * g


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def eval_max_b(a):
    """Return max of auto-convolution for normalized sequence."""
    b = fast_convolve(a)
    return float(b.max())


def normalize_a(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def subgradient_optimize(a0, time_limit, lr_init=0.003, momentum=0.92,
                         n_top=None, seed=42, verbose=False):
    """
    FFT-accelerated projected subgradient with Nesterov momentum.
    Optimizes max(conv(a,a)) subject to a >= 0, sum(a) = 1.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = normalize_a(a0)

    v = np.zeros(n)
    best_a = a.copy()
    best_obj = eval_max_b(a)
    cur_lr = lr_init
    no_improve = 0
    it_count = 0

    if n_top is None:
        n_top = max(20, n // 8)

    peak_fraction = 0.999

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        b_max = b.max()

        # Adaptive peak selection: all peaks within tolerance of max
        threshold = b_max * peak_fraction
        active = np.where(b >= threshold)[0]

        # Ensure minimum number of peaks
        if len(active) < max(5, n_top // 4):
            active = np.argpartition(b, -n_top)[-n_top:]

        # Limit number of peaks
        if len(active) > n_top:
            # Keep the most prominent
            idx = np.argsort(b[active])[::-1][:n_top]
            active = active[idx]

        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / len(active)

        # Compute subgradient
        g = compute_subgradient(a, w)

        # Nesterov momentum step
        v = momentum * v - cur_lr * g
        a_new = project_simplex(a + v)

        obj_new = eval_max_b(a_new)

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        it_count += 1

        # Adaptive strategy
        if no_improve >= 2000:
            a = normalize_a(best_a + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.5, lr_init * 0.05)
            no_improve = 0
        elif no_improve >= 1000:
            cur_lr *= 0.93
        elif no_improve >= 500:
            cur_lr *= 0.97

        if verbose and it_count % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG iter {it_count} ({elapsed:.0f}s): obj={best_obj:.10f}")

    return best_a, best_obj


def kkt_refine(a0, time_limit, verbose=False):
    """
    KKT-based active-set refinement.
    Solves the stationary point conditions:
      For active peaks P with uniform weights, and support S:
      M[i,j] = 2 * w[i+j] for i,j in S
      [M  -1] [a_S] = [0]
      [1'   0] [mu ]   [1]
    
    Uses damped iterations with dynamic active set and support updates.
    """
    t0 = time.monotonic()
    n = len(a0)
    a = normalize_a(a0)

    best_a = a.copy()
    best_obj = eval_max_b(a)

    n_top = max(30, n // 5)
    support_threshold = 1e-10
    damping_start = 0.5
    damping = damping_start

    stalled = 0

    for iteration in range(500):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute convolution and identify active peaks
        b = fast_convolve(a)
        b_max = b.max()

        # Active peaks: within 0.05% of max
        threshold = b_max * 0.9995
        P = np.where(b >= threshold)[0]

        # Ensure sufficient peaks
        if len(P) < 10:
            P = np.argpartition(b, -min(n_top, len(b)))[-min(n_top, len(b)):]
        # Limit peaks
        if len(P) > n_top:
            P = P[np.argsort(b[P])[::-1][:n_top]]

        p_len = len(P)

        # Determine support
        S = np.where(a > support_threshold)[0]
        if len(S) < 3:
            S = np.arange(n)
        s_len = len(S)

        if s_len > 2500:
            # For very large support, subsample
            S = S[np.linspace(0, s_len - 1, 2500).astype(int)]
            s_len = len(S)

        # Build peak weight vector
        w = np.zeros(2 * n - 1)
        w[P] = 1.0 / p_len

        # Build matrix M[si, sj] = 2 * w[S[si] + S[sj]]
        S_col = S[:, None]
        S_row = S[None, :]
        S_sum = S_col + S_row
        M = 2.0 * w[S_sum]

        # Build augmented system
        # [M  -1] [a_S] = [0 ]
        # [1'   0] [mu ]   [1 ]
        A_aug = np.zeros((s_len + 1, s_len + 1))
        A_aug[:s_len, :s_len] = M
        A_aug[:s_len, s_len] = -1.0
        A_aug[s_len, :s_len] = 1.0

        b_aug = np.zeros(s_len + 1)
        b_aug[s_len] = 1.0

        # Solve
        try:
            # Add small regularization for numerical stability
            reg = 1e-12
            A_reg = A_aug.copy()
            A_reg[:s_len, :s_len] += reg * np.eye(s_len)
            sol = np.linalg.solve(A_reg, b_aug)
            a_S_sol = sol[:s_len]
        except np.linalg.LinAlgError:
            try:
                sol, _, _, _ = np.linalg.lstsq(A_aug, b_aug, rcond=None)
                a_S_sol = sol[:s_len]
            except Exception:
                stalled += 1
                if stalled > 10:
                    break
                damping *= 0.7
                continue

        # Check and handle negative components
        min_sol = a_S_sol.min()
        if min_sol < -1e-8:
            # Clip negatives
            a_S_sol = np.maximum(a_S_sol, 0.0)
            sol_sum = a_S_sol.sum()
            if sol_sum < 1e-10:
                stalled += 1
                if stalled > 10:
                    break
                damping *= 0.7
                # Try wider support
                S = np.where(a > support_threshold * 0.1)[0]
                continue
            a_S_sol = a_S_sol / sol_sum
        else:
            a_S_sol = a_S_sol / a_S_sol.sum()

        # Construct full solution
        a_new = np.zeros(n)
        a_new[S] = a_S_sol
        a_new = a_new / a_new.sum()

        # Damped update
        a_damped = (1.0 - damping) * a + damping * a_new
        a_damped = np.maximum(a_damped, 0.0)
        dsum = a_damped.sum()
        if dsum < 1e-15:
            a_damped = best_a.copy()
        else:
            a_damped = a_damped / dsum

        obj_new = eval_max_b(a_damped)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_damped.copy()
            a = a_damped.copy()
            stalled = 0
            # Maintain or slightly increase damping
            damping = min(1.0, damping * 1.02 + 0.01)
        else:
            stalled += 1
            damping = max(0.05, damping * 0.85)
            a = best_a.copy()
            if damping < 0.05:
                break

        if verbose and iteration % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    KKT iter {iteration} ({elapsed:.0f}s): obj={best_obj:.10f}, "
                  f"dam={damping:.3f}, |P|={p_len}, |S|={s_len}")

        # Early convergence check
        if stalled > 30:
            break

        # Check if we've essentially converged
        if iteration > 20:
            # Compare b from best_a
            b_check = fast_convolve(best_a)
            if b_check.max() < best_obj * 1.00001:
                if iteration > 50:
                    break

    return best_a, best_obj


def kkt_refine_v2(a0, time_limit, verbose=False):
    """
    Enhanced KKT refinement with:
    - Variable peak weighting (weight by excess above median)
    - Support adaptation
    - Fallback to subgradient if KKT stalls
    """
    t0 = time.monotonic()
    n = len(a0)
    a = normalize_a(a0)

    best_a = a.copy()
    best_obj = eval_max_b(a)

    n_top = max(40, n // 4)
    support_threshold = 1e-8

    for iteration in range(300):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        b_max = b.max()

        # Identify active peaks with weighted selection
        # Use peaks above 95th percentile
        sorted_b = np.sort(b)[::-1]
        p95_idx = max(10, min(len(b), len(b) // 10))
        threshold = max(b[p95_idx] if p95_idx < len(b) else b.max() * 0.9, b_max * 0.999)

        P = np.where(b >= threshold)[0]
        if len(P) < 10:
            P = np.argpartition(b, -n_top)[-n_top:]
        if len(P) > n_top:
            P = P[np.argsort(b[P])[::-1][:n_top]]

        p_len = len(P)

        # Use weighted peaks: weight proportional to excess above median peak value
        P_vals = b[P]
        P_median = np.median(P_vals)
        weights = np.maximum(P_vals - P_median, 0.01 * b_max)
        weights = weights / weights.sum()

        # Build peak weight vector
        w = np.zeros(2 * n - 1)
        w[P] = weights

        # Determine support (be more inclusive)
        S = np.where(a > support_threshold)[0]
        if len(S) < 5:
            S = np.arange(n)
        s_len = len(S)

        if s_len > 3000:
            # Subsample support for very large problems
            step = s_len // 2500 + 1
            S = S[::step]
            s_len = len(S)

        # Build matrix M[si, sj] = 2 * w[S[si] + S[sj]]
        S_sum = S[:, None] + S[None, :]
        M = 2.0 * w[S_sum]

        # Build augmented system
        A_aug = np.zeros((s_len + 1, s_len + 1))
        A_aug[:s_len, :s_len] = M
        A_aug[:s_len, s_len] = -1.0
        A_aug[s_len, :s_len] = 1.0
        b_aug = np.zeros(s_len + 1)
        b_aug[s_len] = 1.0

        # Solve with regularization
        reg = 1e-10
        A_reg = A_aug.copy()
        A_reg[:s_len, :s_len] += reg * np.eye(s_len)

        try:
            sol = np.linalg.solve(A_reg, b_aug)
            a_S_sol = sol[:s_len]
        except np.linalg.LinAlgError:
            try:
                sol, _, _, _ = np.linalg.lstsq(A_aug, b_aug, rcond=None)
                a_S_sol = sol[:s_len]
            except Exception:
                break

        # Handle negatives
        n_neg = np.sum(a_S_sol < -1e-6)
        if n_neg > s_len * 0.05:
            # Too many negatives - fall back to lighter approach
            a_S_sol = np.maximum(a_S_sol, 0.0)
            ss = a_S_sol.sum()
            if ss < 1e-10:
                break
            a_S_sol /= ss

        a_S_sol = np.maximum(a_S_sol, 0.0)
        ss = a_S_sol.sum()
        if ss < 1e-10:
            break
        a_S_sol /= ss

        # Construct and evaluate
        a_new = np.zeros(n)
        a_new[S] = a_S_sol
        a_new = a_new / a_new.sum()

        # Try full step first, then damped
        obj_full = eval_max_b(a_new)
        if obj_full < best_obj - 1e-14:
            best_obj = obj_full
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            # Try damped steps
            improved = False
            for damp in [0.8, 0.5, 0.3, 0.15, 0.05]:
                a_d = (1.0 - damp) * best_a + damp * a_new
                a_d = normalize_a(a_d)
                obj_d = eval_max_b(a_d)
                if obj_d < best
