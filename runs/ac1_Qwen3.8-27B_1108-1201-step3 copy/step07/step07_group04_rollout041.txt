Let me also think about whether there's a smarter way to do the peak-directed search. The idea is:
- Find the lag k* that achieves the maximum in the convolution
- The value is (a*a)_{k*} = sum_j a_j * a_{k*-j}
- To reduce this, we want to decrease the product a_j * a_{k*-j} for the largest terms
- But we need to maintain sum(a) = 1 and a >= 0

A good local move:
- Pick the pair (j, k*-j) with the largest product
- Decrease a_j by ε and increase a_{k*-j} by ε (or vice versa) → this doesn't change the sum
- Or: decrease both a_j and a_{k*-j} and distribute the reduction to other positions

Actually, the simplest effective local search is:
- For the peak at k*, try reducing each a_j (for j contributing to the peak) and compensating by increasing other a_m (that don't contribute to the peak as much)
- This is essentially what the "peak_shave" function does in the previous code

Let me think of something different. What about **simulated annealing with structured moves**?

Moves:
1. Swap: pick two positions i, j and swap a_i and a_j (preserves sum)
2. Transfer: pick i, j and move δ from a_i to a_j (a_i -= δ, a_j += δ)
3. Blend: a_new = (1-t)*a + t*prot, where prot is a prototype (uniform, random, or other good sequence)

With simulated annealing, we accept worse moves with probability exp(-ΔE/T) where T decreases over time.

```python
"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Multi-length exploration with structured initialization
2. Smooth gradient descent (log-sum-exp) with adaptive temperature cooling
3. Peak-directed coordinate descent for final polishing
4. Iterated local search with structured perturbations
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from time import monotonic
import sys

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_from_conv(n, max_conv, sum_a):
    """Compute score from components."""
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_conv / (sum_a ** 2)

def z_to_a(z):
    """Convert log-space parameters to probability simplex."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()

def a_to_z(a):
    """Convert simplex vector to log-space."""
    a_safe = np.maximum(a, 1e-30)
    z = np.log(a_safe)
    return z - z.mean()

def smooth_obj_grad(z, T, n):
    """
    Compute log-sum-exp smooth approximation of 2n*max(conv(a,a))
    and its gradient with respect to z (log-softmax parameterization).
    """
    a = z_to_a(z)
    b = fast_convolve(a)  # length 2n-1
    
    # Log-sum-exp of b
    b_max = b.max()
    scaled = (b - b_max) / T
    exp_scaled = np.exp(scaled)
    lse_val = b_max + T * np.log(exp_scaled.sum())
    w = exp_scaled / exp_scaled.sum()
    
    # Objective
    obj = 2.0 * n * lse_val
    
    # Gradient wrt a: grad_a[i] = 2 * sum_{m=0}^{n-1} a[m] * w[m+i]
    # Using matrix formulation: grad_a = 2 * W @ a where W[i,m] = w[m+i]
    W = np.lib.stride_tricks.sliding_window_view(w, n)  # shape (n, n)
    grad_a = 2.0 * (W @ a)
    
    # Transform to z-space
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    
    return obj, grad_z

def smooth_lbfgs(a0, T, time_budget, maxiter=50000):
    """Run L-BFGS-B optimization in log-space with fixed temperature T."""
    n = len(a0)
    z0 = a_to_z(a0)
    t0 = monotonic()
    stopped = [False]
    
    def fun(z):
        if monotonic() - t0 > time_budget:
            stopped[0] = True
        return smooth_obj_grad(z, T, n)[0]
    
    def fun_and_grad(z):
        if monotonic() - t0 > time_budget:
            stopped[0] = True
            return 1e15, np.zeros(n)
        return smooth_obj_grad(z, T, n)
    
    res = minimize(
        fun_and_grad, z0, jac=True,
        method='L-BFGS-B',
        options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                 'maxfun': maxiter * 10, 'disp': False}
    )
    
    if not stopped[0]:
        a_out = z_to_a(res.x)
    else:
        a_out = a0
    return a_out

def adaptive_temperature_optimize(a0, time_budget, T_init=None, T_final=1e-6, verbose=True):
    """
    Multi-stage smooth optimization with decreasing temperature.
    At each stage, run L-BFGS with the current T, then decrease T.
    """
    n = len(a0)
    a = a0.copy()
    t0 = monotonic()
    
    # Initial temperature based on the current max convolution
    if T_init is None:
        b = fast_convolve(a)
        T_init = max(b.max() * 0.5, 1e-4)
    
    T = T_init
    stage = 0
    best_score = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    
    while monotonic() - t0 < time_budget:
        stage += 1
        remaining = time_budget - (monotonic() - t0)
        
        # Allocate time for this stage
        if remaining < 5:
            break
        stage_time = min(remaining * 0.7, 60.0)  # Cap at 60s per stage
        
        # Run L-BFGS at current temperature
        a = smooth_lbfgs(a, T, stage_time)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            a = a0
            break
        a = a / s
        
        # Evaluate
        max_conv = fast_convolve(a).max()
        score = 2.0 * n * max_conv
        if score < best_score:
            best_score = score
            best_a = a.copy()
            if verbose:
                print(f"  Stage {stage}: T={T:.6f}, score={score:.6f} (improved)")
        else:
            if verbose:
                print(f"  Stage {stage}: T={T:.6f}, score={score:.6f}")
        
        # Decrease temperature
        T *= 0.7
        if T < T_final:
            T = T_final
    
    return best_a, best_score

def peak_directed_search(a0, time_budget, verbose=True):
    """
    Local search that specifically targets reducing the maximum convolution value.
    Strategy: identify peak lags, find contributing pairs, try reducing them.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    no_improve = 0
    
    while monotonic() - t0 < time_budget:
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]
        
        # Find contributing indices for this peak
        # (a*a)[peak_k] = sum_j a[j] * a[peak_k - j] where 0 <= j < n and 0 <= peak_k-j < n
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        indices = np.arange(js, je + 1)
        partners = peak_k - indices
        products = a[indices] * a[partners]
        
        # Sort by contribution (largest first)
        order = np.argsort(products)[::-1]
        
        improved = False
        
        # Strategy 1: Reduce a single large contributor
        for idx_rank in range(min(10, len(order))):
            pos = indices[order[idx_rank]]
            old_val = a[pos]
            if old_val < 1e-12:
                continue
            
            for factor in [0.9, 0.8, 0.7, 0.5]:
                new_val = old_val * factor
                diff = old_val - new_val
                if diff < 1e-14:
                    continue
                
                # Redistribute to other positions (not contributing to this peak)
                a_trial = a.copy()
                a_trial[pos] = new_val
                # Spread diff to positions that don't contribute to peak_k
                mask = np.ones(n, dtype=bool)
                mask[indices] = False  # Don't add to positions contributing to this peak
                if mask.sum() == 0:
                    # Alternative: spread to all non-peak positions
                    mask[indices] = True  # Undo
                    mask[pos] = False
                
                other_positions = np.where(mask)[0]
                if len(other_positions) == 0:
                    continue
                
                # Add proportionally to existing values (to avoid creating new peaks)
                other_vals = a_trial[other_positions]
                other_sum = other_vals.sum()
                if other_sum < 1e-15:
                    a_trial[other_positions] += diff / len(other_positions)
                else:
                    a_trial[other_positions] += diff * other_vals / other_sum
                
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                
                trial_score = 2.0 * n * fast_convolve(a_trial).max()
                if trial_score < best_score - 1e-12:
                    a = a_trial
                    best_a = a.copy()
                    best_score = trial_score
                    improved = True
                    break
            
            if improved:
                break
        
        # Strategy 2: Reduce a pair that contributes to the peak
        if not improved:
            for idx_rank in range(min(5, len(order))):
                pos_i = indices[order[idx_rank]]
                pos_j = partners[order[idx_rank]]
                if pos_i == pos_j:
                    continue
                if a[pos_j] < 1e-12:
                    continue
                
                for f1, f2 in [(0.85, 0.85), (0.8, 0.9), (0.9, 0.8), (0.7, 0.8)]:
                    a_trial = a.copy()
                    a_trial[pos_i] *= f1
                    a_trial[pos_j] *= f2
                    diff = (1 - f1) * a[pos_i] + (1 - f2) * a[pos_j]
                    
                    # Redistribute
                    mask = np.ones(n, dtype=bool)
                    mask[pos_i] = False
                    mask[pos_j] = False
                    other_positions = np.where(mask)[0]
                    if len(other_positions) == 0:
                        continue
                    other_vals = a_trial[other_positions]
                    other_sum = other_vals.sum()
                    if other_sum < 1e-15:
                        a_trial[other_positions] += diff / len(other_positions)
                    else:
                        a_trial[other_positions] += diff * other_vals / other_sum
                    
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    trial_score = 2.0 * n * fast_convolve(a_trial).max()
                    if trial_score < best_score - 1e-12:
                        a = a_trial
                        best_a = a.copy()
                        best_score = trial_score
                        improved = True
                        break
            
            if improved:
                no_improve = 0
        
        # Strategy 3: Random perturbation
        if not improved:
            for _ in range(10):
                perturbation = np.random.standard_normal(n) * 0.001
                a_trial = a + perturbation
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                trial_score = 2.0 * n * fast_convolve(a_trial).max()
                if trial_score < best_score - 1e-12:
                    a = a_trial
                    best_a = a.copy()
                    best_score = trial_score
                    improved = True
                    break
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                # Restart with perturbation
                a = best_a + np.random.standard_normal(n) * 0.005
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
        
        if verbose and (stage := int((monotonic() - t0) // 10)) % 3 == 0:
            pass  # Will add proper logging below
    
    return best_a, best_score


def iterated_local_search(a0, time_budget, n_restarts=None, verbose=True):
    """
    Iterated local search: optimize, perturb, repeat.
    Combines smooth optimization with peak-directed search.
    """
    n = len(a0)
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()
    t0 = monotonic()
    
    if n_restarts is None:
        n_restarts = max(1, int(time_budget / 60))
    
    for r in range(n_restarts):
        if monotonic() - t0 > time_budget:
            break
        
        # Phase A: Smooth optimization
        t_a = min(30.0, (time_budget - (monotonic() - t0)) * 0.5)
        if t_a > 5:
            a_smooth, _ = adaptive_temperature_optimize(
                best_a, t_a, T_init=None, T_final=1e-8, verbose=False
            )
            sc = 2.0 * n * fast_convolve(a_smooth).max()
            if sc < best_score:
                best_score = sc
                best_a = a_smooth
                if verbose:
                    print(f"    ILS restart {r}: smooth → {sc:.6f}")
        
        # Phase B: Peak-directed search
        t_b = min(20.0, (time_budget - (monotonic() - t0)) * 0.3)
        if t_b > 3:
            a_peak, _ = peak_directed_search(best_a, t_b, verbose=False)
            sc = 2.0 * n * fast_convolve(a_peak).max()
            if sc < best_score:
                best_score = sc
                best_a = a_peak
                if verbose:
                    print(f"    ILS restart {r}: peak → {sc:.6f}")
        
        # Phase C: Perturb for next restart
        if r < n_restarts - 1:
            perturbation_scale = 0.01 * (1 + r * 0.5)
            a_perturbed = best_a + np.random.standard_normal(n) * perturbation_scale
            a_perturbed = np.maximum(a_perturbed, 0)
            a_perturbed = a_perturbed / a_perturbed.sum()
            # Use perturbed as starting point for next smooth opt
            # (but keep best_a as the incumbent)
    
    return best_a, best_score

def initialize_sequence(n, parent=None, seed=42):
    """Create an initial sequence of length n."""
    rng = np.random.RandomState(seed)
    
    if parent is not None and len(parent) == n:
        # Slightly perturb the parent
        a = parent + rng.standard_normal(n) * 0.005
        a = np.maximum(a, 0)
        a = a / a.sum()
        return a
    
    if parent is not None:
        # Interpolate parent to length n
        old_n = len(parent)
        old_x = np.linspace(0, 1, old_n)
        new_x = np.linspace(0, 1, n)
        a = np.interp(new_x, old_x, parent)
        a = np.maximum(a, 0)
        a = a / a.sum()
        return a
    
    # Default: uniform with small perturbation
    a = np.ones(n) / n + rng.standard_normal(n) * 0.001
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a
```

```python
"""
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 using:
1. Adaptive log-sum-exp smoothing with L-BFGS in log-softmax parameterization
2. Peak-directed local search targeting top convolution peaks
3. Iterated local search with structured perturbations

Key: correct gradient via sliding window matrix-vector product,
geometric temperature cooling, and aggressive final polishing.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
from time import monotonic


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def z_to_a(z):
    """Convert log-softmax parameters to probability simplex."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def a_to_z(a):
    """Convert simplex vector to centered log-space."""
    a_safe = np.maximum(a, 1e-30)
    z = np.log(a_safe)
    return z - z.mean()


def smooth_obj_grad(z, T, n, cache=None):
    """
    Compute log-sum-exp smooth approximation of 2n*max(conv(a,a))
    and its gradient w.r.t. z (log-softmax parameterization).
    
    Gradient derivation:
      grad_a[i] = 2 * sum_m a[m] * w[m+i]  where w = softmax(b/T)
      grad_z[i] = a[i] * (grad_a[i] - S)    where S = dot(a, grad_a)
    """
    a = z_to_a(z)
    b = fast_convolve(a)  # length 2n-1

    # Log-sum-exp
    b_max = b.max()
    scaled = (b - b_max) / T
    exp_scaled = np.exp(scaled)
    w_sum = exp_scaled.sum()
    lse_val = b_max + T * np.log(w_sum)
    w = exp_scaled / w_sum

    # Objective
    obj = 2.0 * n * lse_val

    # Gradient wrt a using sliding window
    W = np.lib.stride_tricks.sliding_window_view(w, n)  # shape (n, n)
    grad_a = 2.0 * (W @ a)

    # Transform to z-space: grad_z[i] = a[i] * (grad_a[i] - S)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)

    return obj, grad_z


def smooth_lbfgs_run(a0, T, time_budget, maxiter=100000):
    """Single L-BFGS-B run in log-space with fixed temperature T."""
    n = len(a0)
    z0 = a_to_z(a0)
    t0 = monotonic()

    def fun_and_grad(z):
        if monotonic() - t0 > time_budget:
            return 1e15, np.zeros(n)
        return smooth_obj_grad(z, T, n)

    try:
        res = minimize(
            fun_and_grad, z0, jac=True,
            method='L-BFGS-B',
            options={
                'maxiter': maxiter,
                'ftol': 1e-20,
                'gtol': 1e-16,
                'maxfun': maxiter * 12,
                'disp': False
            }
        )
        a_out = z_to_a(res.x)
    except Exception:
        a_out = a0.copy()

    a_out = np.maximum(a_out, 0)
    s = a_out.sum()
    if s < 1e-15:
        a_out = a0.copy()
    else:
        a_out = a_out / s
    return a_out

    Multi-stage smooth optimization with geometrically cooling temperature.
    Returns (best_a, best_score).
    """

    s = a.sum()
    a = a / s

    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()

        remaining = time_budget - elapsed

            best_a = a.copy()
        if verbose:

        T *= cool_rate
        if T < T_min:

                    a_trial = a.copy()
                    a_trial[pos] = new_val

                    # Redistribute to positions far from this peak
                    # (positions where adding won't create new peaks)
                    mask = np.ones(n, dtype=bool)
                    mask[indices] = False  # Avoid adding to peak contributors
                    other_pos = np.where(mask)[0]
                    if len(other_pos) == 0:
                        mask[indices] = True
                        mask[pos] = False
                        other_pos = np.where(mask)[0]
                    if len(other_pos) == 0:
                        continue

                    other_vals = a_trial[other_pos]
                    other_sum = other_vals.sum()
                    if other_sum < 1e-15:
                        a_trial[other_pos] += diff / len(other_pos)
                    else:
                        # Weighted redistribution (prefer positions with larger values)
                        a_trial[other_pos] += diff * other_vals / other_sum

                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    trial_score = 2.0 * n * fast_convolve(a_trial).max()

                    if trial_score < best_score - 1e-13:
                        a = a_trial
                        best_a = a.copy()
                        best_score = trial_score
                        improved = True
                        break

            if not improved:
                # Try pair reduction
                for rank in order[:5]:
                    if improved:
                        break
                    pi = int(indices[rank])
                    pj = int(partners[rank])
                    if pi == pj or a[pj] < 1e-13:
                        continue

                    for f1, f2 in [(0.9, 0.9), (0.85, 0.9), (0.9, 0.85), (0.8, 0.85)]:
                        a_trial = a.copy()
                        a_trial[pi] *= f1
                        a_trial[pj] *= f2
                        diff = (1 - f1) * a[pi] + (1 - f2) * a[pj]

                        mask = np.ones(n, dtype=bool)
                        mask[pi] = False
                        mask[pj] = False
                        other_pos = np.where(mask)[0]
                        if len(other_pos) == 0:
                            continue
                        other_vals = a_trial[other_pos]
                        other_sum = other_vals.sum()
                        if other_sum < 1e-15:
                            a_trial[other_pos] += diff / len(other_pos)
                        else:
                            a_trial[other_pos] += diff * other_vals / other_sum

                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        trial_score = 2.0 * n * fast_convolve(a_trial).max()

                        if trial_score < best_score - 1e-13:
                            a = a_trial
                            best_a = a.copy()
                            best_score = trial_score
                            improved = True
                            break

        # Random perturbation escape
        if not improved:
            no_improve += 1
            if no_improve > 50:
                scale = 0.003 + 0.002 * (no_improve // 100)
                a = best_a + np.random.standard_normal(n) * scale
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
        else:
            no_improve = 0

        if verbose and (monotonic() - t0) % 15 < 0.1:
            print(f"    {label} peak_search: score={best_score:.8f}")

    return best_a, best_score


def iterated_local_search(a0, time_budget, verbose=True, label="ILS"):
    """
    Iterated local search: alternate smooth optimization and peak search,
    with structured perturbations between iterations.
    """
    n = len(a0)
    best_a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = best_a.sum()
    best_a = best_a / s
    best_score = 2.0 * n * fast_convolve(best_a).max()
    t0 = monotonic()
    iteration = 0

    while True:
        remaining = time_budget - (monotonic() - t0)
        if remaining < 10:
            break

        iteration += 1

        # Phase A: Smooth optimization
        t_smooth = min(remaining * 0.45, 50.0)
        if t_smooth > 5:
            a_s, sc_s = adaptive_smooth_optimize(
                best_a, t_smooth, T_init=None, T_min=1e-9,
                cool_rate=0.55, verbose=False, label=label
            )
            if sc_s < best_score:
                best_score = sc_s
                best_a = a_s.copy()
                if verbose:
                    print(f"  {label} iter {iteration} smooth: {best_score:.8f}")

        # Phase B: Peak-directed search
        t_peak = min(remaining * 0.3, 30.0)
        if t_peak > 3 and monotonic() - t0 < time_budget:
            a_p, sc_p = peak_local_search(
                best_a, t_peak, verbose=False, label=label
            )
            if sc_p < best_score:
                best_score = sc_p
                best_a = a_p.copy()
                if verbose:
                    print(f"  {label} iter {iteration} peak:   {best_score:.8f}")

        # Perturb for diversity (except last iteration)
        remaining = time_budget - (monotonic() - t0)
        if remaining > 30:
            scale = 0.004 + 0.003 * min(iteration * 0.3, 1.0)
            # Structured perturbation: blend with a slightly different profile
            a_pert = best_a * (1 + np.random.standard_normal(n) * scale)
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            # Only use perturbed start if it's not too far from best
            # (just begin next iteration from best_a, not perturbed)
            # Actually, let's keep best_a as the working point
            # The perturbation is implicitly handled by the smooth optimization
            # starting from a slightly different point via the L-BFGS trajectory

    return best_a, best_score


def get_parent_sequence():
    """Access the parent sequence if available."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            return p / p.sum()
    except Exception:
        pass
    return None


def initialize_sequence(n, parent=None, seed=0):
    """Create initial sequence of length n."""
    rng = np.random.RandomState(seed)

    if parent is not None and len(parent) == n:
        a = parent + rng.standard_normal(n) * 0.003
        a = np.maximum(a, 0)
        return a / a.sum()

    if parent is not None:
        old_n = len(parent)
        old_x = np.linspace(0, 1, old_n)
        new_x = np.linspace(0, 1, n)
        a = np.interp(new_x, old_x, parent)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a = a / s
        return a

    # Default: uniform with tiny random perturbation
    a = np.ones(n) / n + rng.standard_normal(n) * 0.001
    a = np.maximum(a, 0)
    return a / a.sum()


def full_pipeline(a0, time_budget, verbose=True, label=""):
    """
    Full optimization pipeline: adaptive smooth + peak search + ILS + final polish.
    """
    n = len(a0)
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()
    t0 = monotonic()

    # Stage 1: Adaptive smooth optimization (medium budget)
    t1 = min(time_budget * 0.35, 200.0)
    if t1 > 10:
        remaining_check = time_budget - (monotonic() - t0)
        if remaining_check > 10:
            a1, sc1 = adaptive_smooth_optimize(
                best_a, t1, T_init=None, T_min=1e-8,
                cool_rate=0.55, verbose=verbose, label=label + " smooth1"
            )
            if sc1 < best_score:
                best_score = sc1
                best_a = a1
                if verbose:
                    print(f"  {label} after smooth1: {best_score:.8f}")

    # Stage 2: Peak-directed local search
    t2 = min(time_budget * 0.25, 120.0)
    remaining = time_budget - (monotonic() - t0)
    if t2 > 5 and remaining > 30:
        a2, sc2 = peak_local_search(best_a, min(t2, remaining - 30),
                                    verbose=verbose, label=label + " peak")
        if sc2 < best_score:
            best_score = sc2
            best_a = a2
            if verbose:
                print(f"  {label} after peak: {best_score:.8f}")

    # Stage 3: Iterated local search
    remaining = time_budget - (monotonic() - t0)
    t3 = min(remaining * 0.4, 200.0)
    if t3 > 30:
        a3, sc3 = iterated_local_search(best_a, t3, verbose=verbose,
                                        label=label + " ILS")
        if sc3 < best_score:
            best_score = sc3
            best_a = a3
            if verbose:
                print(f"  {label} after ILS: {best_score:.8f}")

    # Stage 4: Final ultra-low-T polish
    remaining = time_budget - (monotonic() - t0)
    t4 = min(remaining * 0.5, 60.0)
    if t4 > 5:
        a4, sc4 = adaptive_smooth_optimize(
            best_a, t4, T_init=None, T_min=1e-12,
            cool_rate=0.4, verbose=verbose, label=label + " final"
        )
        if sc4 < best_score:
            best_score = sc4
            best_a = a4
            if verbose:
                print(f"  {label} after final: {best_score:.8f}")

    # Stage 5: One last peak search
    remaining = time_budget - (monotonic() - t0)
    if remaining > 5:
        a5, sc5 = peak_local_search(best_a, min(remaining * 0.6, 30.0),
                                    verbose=False, label=label + " final_peak")
        if sc5 < best_score:
            best_score = sc5
            best_a = a5

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main entry point: multi-phase optimization to minimize
    2*n*max(convolve(a,a))/(sum(a))^2.

    Strategy:
    1. Get parent sequence or start from uniform
    2. Quick multi-length exploration
    3. Deep optimization at best length (adaptive smooth + ILS + peak search)
    4. Final polishing
    """
    np.random.seed(seed)
    deadline = monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    # Get parent
    parent = get_parent_sequence()
    if parent is not None:
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Parent: n={len(parent)}, score={p_score:.6f}")
        best_seq = parent.copy()
        best_score = p_score
    else:
        n_init = 768
        best_seq = np.ones(n_init) / n_init
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Initial uniform n={n_init}: score={best_score:.6f}")

    # === Phase 1: Quick multi-length exploration ===
    print("\n=== Phase 1: Multi-length Exploration ===")
    n_parent = len(best_seq)
    n_candidates = sorted(set([n_parent, 512, 768, 1024, 1536, 2048]))
    # Focus on lengths near parent
    n_candidates.sort(key=lambda x: abs(x - n_parent) + 0.01 * x / 1000)
    n_candidates = n_candidates[:5]

    rem = deadline - monotonic() - 800  # Reserve 800s for deep optimization
    if rem > 60:
        t_per = min(rem / len(n_candidates), 50.0)
        for n_try in n_candidates:
            rem = deadline - monotonic() - 800
            if rem < 20:
                break
            t_alloc = min(t_per, rem * 0.85)
            if t_alloc < 10:
                continue
            # Skip if same as parent and we already have a parent score
            if n_try == n_parent and parent is not None:
                # Still do a quick optimization from parent
                pass

            print(f"  Exploring n={n_try} (budget {t_alloc:.0f}s)")
            a_init = initialize_sequence(n_try, parent, seed=seed + n_try)

            # Quick optimization: just adaptive smooth
            a_opt, sc = adaptive_smooth_optimize(
                a_init, t_alloc, T_init=None, T_min=1e-6,
                cool_rate=0.6, verbose=False, label=f"n{n_try}"
            )
            actual_score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(actual_score) and actual_score < best_score:
                best_score = actual_score
                best_seq = a_opt.copy()
                print(f"    ** NEW BEST: n={n_try}, score={actual_score:.6f}")
            elif np.isfinite(actual_score):
                print(f"    n={n_try}: score={actual_score:.6f}")

    # === Phase 2: Deep optimization at best length ===
    print(f"\n=== Phase 2: Deep Optimization (current best: {best_score:.6f}) ===")
    rem = deadline - monotonic() - 200
    if rem > 60:
        t_deep = rem * 0.7
        n_best = len(best_seq)
        print(f"  Deep pipeline: n={n_best}, budget={t_deep:.0f}s")

        a_deep, sc_deep = full_pipeline(
            best_seq, t_deep, verbose=True, label=f"deep_n{n_best}"
        )
        actual_score = float(evaluate_sequence(a_deep.tolist()))
        if np.isfinite(actual_score) and actual_score < best_score:
            best_score = actual_score
            best_seq = a_deep.copy()
            print(f"  ** NEW BEST after deep: {actual_score:.6f}")
        else:
            print(f"  Deep result: {actual_score:.6f} (no improvement)")

    # === Phase 3: Try one more length if time allows ===
    rem = deadline - monotonic() - 100
    if rem > 60:
        n_best = len(best_seq)
        # Try a slightly different length
        candidates = [int(n_best * 1.2), int(n_best * 0.85), n_best + 128]
        for alt_n in candidates[:2]:
            rem = deadline - monotonic() - 80
            if rem < 40:
                break
            if not (128 <= alt_n <= 4096) or alt_n == n_best:
                continue
            t_alt = min(rem * 0.5, 80.0)
            print(f"  Trying alt n={alt_n} (budget {t_alt:.0f}s)")
            a_alt = initialize_sequence(alt_n, best_seq, seed=seed + alt_n)
            a_opt, sc = adaptive_smooth_optimize(
                a_alt, t_alt, T_init=None, T_min=1e-7,
                cool_rate=0.55, verbose=False, label=f"alt_n{alt_n}"
            )
            # Quick peak search
            remaining = t_alt - (monotonic() - (monotonic() - t_alt))
            a_opt, sc = peak_local_search(a_opt, min(20, max(5, t_alt * 0.2)),
                                          verbose=False, label=f"alt_n{alt_n}")
            actual_score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(actual_score) and actual_score < best_score:
                best_score = actual_score
                best_seq = a_opt.copy()
                print(f"    ** NEW BEST: n={alt_n}, score={actual_score:.6f}")

    # === Phase 4: Final intensive polish ===
    print(f"\n=== Phase 4: Final Polish (best: {best_score:.6f}) ===")
    rem = deadline - monotonic() - 10
    if rem > 20:
        t_final = rem * 0.8
        print(f"  Final pipeline: budget={t_final:.0f}s")
        a_final, sc_final = full_pipeline(
            best_seq, t_final, verbose=True, label="FINAL"
        )
        actual_score = float(evaluate_sequence(a_final.tolist()))
        if np.isfinite(actual_score) and actual_score < best_score:
            best_score = actual_score
            best_seq = a_final.copy()
            print(f"  ** NEW BEST final: {actual_score:.6f}")

    # === Verification ===
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        # Fallback
        best_seq = np.ones(768) / 768.0
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
```
