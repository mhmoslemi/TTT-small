"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding as the primary
method, combined with multi-peak mirror descent, L-BFGS polishing, and coordinate descent.
The LP approach finds the optimal direction to move at each step by solving a linear

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_from_a(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def build_lp_matrix(f):
    """Build the constraint matrix A for the LP.
    A[k, i] = f[i] if max(0, k-n+1) <= i <= min(n-1, k), else 0
    Shape: (2n-1, n)
    """
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = f[lo:hi + 1]
    return A


def lp_step(f, M):
    """Solve the LP: max sum(g) s.t. f*g <= M, g >= 0.
    Returns the optimal g (normalized to sum=1) or None if LP fails.
    """
    n = len(f)
    A = build_lp_matrix(f)
    b = np.full(2 * n - 1, M)
    c = -np.ones(n)  # minimize -sum(g)
    
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-10})
        if res.success:
            g = np.maximum(res.x, 0)
            s = g.sum()
            if s > 1e-10:
                return g / s
    except Exception:
        pass
    return None

        F = rfft(f, fft_size)
        correl = irfft(np.conj(W) * F, fft_size)
        grad = 4.0 * n * correl[:n]
        
        # Project to tangent space
        grad = grad - np.dot(f, grad)
        
        # Mirror (multiplicative) update
        log_update = -eta * grad
        log_update = np.clip(log_update, -3.0, 3.0)
        f = f * np.exp(log_update)
        f = np.maximum(f, 1e-300)
        s = f.sum()
        if s > 1e-15:
            f = f / s
        else:
            f = best_f.copy()
        
        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} ({elapsed:.0f}s): score={current_score:.7f} best={best_score:.7f}")
    
    return best_f, best_score


def lbgfs_polish(f_init, time_limit, T=10, maxiter=5000):
    """L-BFGS in log-space with softmax smoothing."""
    n = len(f_init)
    f = np.maximum(f_init, 1e-15).astype(np.float64)
    f = f / f.sum()
    z0 = np.log(f)
    z0 = z0 - z0.mean()
    t0 = time.monotonic()
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        f_s = softmax_from_z(z)
        b = fast_convolve(f_s)
        scaled = b / T
        m = scaled.max()
        lse = m + np.log(np.exp(scaled - m).sum())
        return 2.0 * n * T * lse
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros(n)
        f_s = softmax_from_z(z)
        b = fast_convolve(f_s)
        scaled = b / T
        m = scaled.max()
        w = np.exp(scaled - m) / np.exp(scaled - m).sum()
        fft_size = 1 << (3 * n - 1).bit_length()
        W = rfft(w, fft_size)
        F = rfft(f_s, fft_size)
        correl = irfft(np.conj(W) * F, fft_size)
        grad_a = 4.0 * n * correl[:n]
        mean_grad = np.dot(f_s, grad_a)
        return f_s * (grad_a - mean_grad)
    
    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                     'maxfun': maxiter * 20, 'disp': False}
        )
        return softmax_from_z(res.x)
    except Exception:
        return f_init.copy()


def softmax_from_z(z):
    """Convert z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def coordinate_descent(f_init, time_limit, verbose=False, label="CD"):
    """Coordinate descent with proportional redistribution."""
    n = len(f_init)
    f = np.maximum(f_init.copy(), 0.0).astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        return f, score_from_a(f)
    f = f / s
    best_f = f.copy()
    best_score = score_from_a(f)
    t0 = time.monotonic()
    
    factors = [0.8, 0.9, 0.95, 1.05, 1.1, 1.2, 1.5]
    no_improve = 0
    max_no_improve = 50
    
    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        no_improve = 0
        improved = False
        order = np.random.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            for fac in factors:
                f_trial = f.copy()
                old_val = f_trial[i]
                f_trial[i] = old_val * fac
                diff = f_trial.sum() - 1.0
                if abs(diff) < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others_sum = f_trial[mask].sum()
                if others_sum < 1e-15:
                    continue
                f_trial[mask] -= diff * f_trial[mask] / others_sum
                if f_trial.min() < -1e-10:
                    continue
                f_trial = np.maximum(f_trial, 0)
                f_trial = f_trial / f_trial.sum()
                sc = score_from_a(f_trial)
                if sc < best_score - 1e-13:
                    best_score = sc
                    f = f_trial.copy()
                    best_f = f.copy()
                    improved = True
                    break
        if not improved:
            no_improve += 1
    
    return best_f, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based direction finding as the primary
method, combined with multi-peak mirror descent, L-BFGS polishing, and coordinate descent.
The LP approach finds the optimal direction at each step by solving a linear program that
maximizes the spread of the new sequence while keeping the cross-convolution bounded,
leading to faster and more reliable convergence than gradient-based methods alone.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_from_a(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def build_lp_matrix(f):
    """Build the constraint matrix A for the LP.
    A[k, i] = f[i] if max(0, k-n+1) <= i <= min(n-1, k), else 0.
    Shape: (2n-1, n)
    """
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = f[lo:hi + 1]
    return A


def lp_solve(f, M):
    """Solve the LP: max sum(g) s.t. f*g <= M (elementwise), g >= 0.
    Returns optimal g (unnormalized) or None if LP fails.
    """
    n = len(f)
    A = build_lp_matrix(f)
    b = np.full(2 * n - 1, M)
    c = -np.ones(n)
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
        if res.success:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def lp_optimize(f_init, time_limit, M_relax=1.0, verbose=True, label="LP"):
    """LP-based optimization with adaptive line search."""
    n = len(f_init)
    f = np.maximum(f_init.copy(), 1e-15).astype(np.float64)
    f = f / f.sum()
    best_f = f.copy()
    best_score = score_from_a(f)
    t0 = time.monotonic()
    stuck = 0
    stuck_limit = 150
    step_sizes = [0.5, 0.3, 0.2, 0.1, 0.07, 0.04, 0.02, 0.01, 0.005]

    for it in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(f)
        M = b.max()
        current_score = 2.0 * n * M

        if current_score < best_score - 1e-12:
            best_score = current_score
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1

        if stuck > stuck_limit:
            break

        M_lp = M * M_relax
        g = lp_solve(f, M_lp)

        if g is None or g.sum() < 1e-12:
            rng = np.random.default_rng(it * 31 + 7)
            f = best_f + rng.standard_normal(n) * 0.005
            f = np.maximum(f, 1e-15)
            f = f / f.sum()
            stuck += 5
            continue

        g = g / g.sum()

        best_t = None
        best_trial_score = current_score
        best_f_trial = None
        for t in step_sizes:
            f_trial = (1.0 - t) * f + t * g
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-12:
                continue
            f_trial = f_trial / s
            tr_score = score_from_a(f_trial)
            if tr_score < best_trial_score - 1e-14:
                best_trial_score = tr_score
                best_t = t
                best_f_trial = f_trial.copy()

        if best_t is not None:
            f = best_f_trial
        else:
            stuck += 8

        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} ({elapsed:.0f}s): score={current_score:.7f} best={best_score:.7f}")

    return best_f, best_score


def multi_peak_mirror_descent(f_init, time_limit, K=5, T_start=0.01, T_end=0.0005,
                               eta_start=0.02, eta_end=0.001, max_iters=50000,
                               verbose=True, label="MPMD"):
    """Mirror descent with top-K peak weighting for smoother gradients."""
    n = len(f_init)
    f = np.maximum(f_init.copy(), 1e-15).astype(np.float64)
    f = f / f.sum()
    best_f = f.copy()
    best_score = score_from_a(f)
    t0 = time.monotonic()
    stuck = 0

    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, it / (0.8 * max_iters))
        T = T_start * (T_end / T_start) ** progress
        eta = eta_start * (eta_end / eta_start) ** progress

        b = fast_convolve(f)
        current_score = 2.0 * n * b.max()

        if current_score < best_score - 1e-12:
            best_score = current_score
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1

        if stuck > 600:
            rng = np.random.default_rng(it * 7919)
            f = best_f + rng.standard_normal(n) * 0.008
            f = np.maximum(f, 1e-15)
            f = f / f.sum()
            stuck = 0
            continue

        top_k_indices = np.argsort(b)[::-1][:K]
        w = np.zeros_like(b)
        top_k_vals = b[top_k_indices]
        w[top_k_indices] = np.exp(top_k_vals / T - top_k_vals.max() / T)
        w = w / w.sum()

        fft_size = 1 << (3 * n - 1).bit_length()
        W = rfft(w, fft_size)
        F = rfft(f, fft_size)
        correl = irfft(np.conj(W) * F, fft_size)
        grad = 4.0 * n * correl[:n]
        grad = grad - np.dot(f, grad)

        log_update = -eta * grad
        log_update = np.clip(log_update, -3.0, 3.0)
        f = f * np.exp(log_update)
        f = np.maximum(f, 1e-300)
        s = f.sum()
        if s > 1e-15:
            f = f / s
        else:
            f = best_f.copy()

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} ({elapsed:.0f}s): score={current_score:.7f} best={best_score:.7f}")

    return best_f, best_score


def softmax_from_z(z):
    """Convert z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def lbgfs_polish(f_init, time_limit, T=10, maxiter=5000):
    """L-BFGS in log-space with softmax smoothing of the max."""
    n = len(f_init)
    f = np.maximum(f_init, 1e-15).astype(np.float64)
    f = f / f.sum()
    z0 = np.log(f)
    z0 = z0 - z0.mean()
    t0 = time.monotonic()

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        f_s = softmax_from_z(z)
        b = fast_convolve(f_s)
        scaled = b / T
        m = scaled.max()
        lse = m + np.log(np.exp(scaled - m).sum())
        return 2.0 * n * T * lse

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros(n)
        f_s = softmax_from_z(z)
        b = fast_convolve(f_s)
        scaled = b / T
        m = scaled.max()
        w = np.exp(scaled - m) / np.exp(scaled - m).sum()
        fft_size = 1 << (3 * n - 1).bit_length()
        W = rfft(w, fft_size)
        F = rfft(f_s, fft_size)
        correl = irfft(np.conj(W) * F, fft_size)
        grad_a = 4.0 * n * correl[:n]
        mean_grad = np.dot(f_s, grad_a)
        return f_s * (grad_a - mean_grad)

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                     'maxfun': maxiter * 20, 'disp': False}
        )
        return softmax_from_z(res.x)
    except Exception:
        return f_init.copy()


def peak_shave(f_init, time_limit, verbose=False, label="PS"):
    """Targeted peak shaving: reduce top contributing terms to the max peak."""
    n = len(f_init)
    f = np.maximum(f_init.copy(), 1e-15).astype(np.float64)
    f = f / f.sum()
    best_f = f.copy()
    best_score = score_from_a(f)
    t0 = time.monotonic()

    for step in range(10000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(f)
        b_work = b.copy()
        b_work[0] = -np.inf
        b_work[-1] = -np.inf
        peak_k = int(np.argmax(b_work))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = f[idx] * f[peak_k - idx]
        top_idx = np.argsort(contribs)[::-1][:5]

        improved = False
        for ci in top_idx:
            i = int(idx[ci])
            if f[i] < 1e-14:
                continue
            for factor in [0.92, 0.88, 0.83, 0.78, 0.7]:
                f_trial = f.copy()
                old_val = f_trial[i]
                f_trial[i] = old_val * factor
                diff = old_val - f_trial[i]
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others_sum = f_trial[mask].sum()
                if others_sum < 1e-15:
                    continue
                f_trial[mask] += diff * f_trial[mask] / others_sum
                f_trial = np.maximum(f_trial, 0)
                f_trial = f_trial / f_trial.sum()
                sc = score_from_a(f_trial)
                if sc < best_score - 1e-13:
                    f = f_trial
                    best_score = sc
                    best_f = f.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            for ci in top_idx[:4]:
                i = int(idx[ci])
                j = peak_k - i
                if i == j or f[j] < 1e-14:
                    continue
                for factor in [0.92, 0.87, 0.8]:
                    f_trial = f.copy()
                    f_trial[j] *= factor
                    f_trial = np.maximum(f_trial, 0)
                    f_trial = f_trial / f_trial.sum()
                    sc = score_from_a(f_trial)
                    if sc < best_score - 1e-13:
                        f = f_trial
                        best_score = sc
                        best_f = f.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            break

    return best_f, best_score


def coordinate_descent(f_init, time_limit, verbose=False, label="CD"):
    """Coordinate descent with proportional redistribution."""
    n = len(f_init)
    f = np.maximum(f_init.copy(), 0.0).astype(np.float64)
    s = f.sum()
    if s < 1e-15:
        return f, score_from_a(f)
    f = f / s
    best_f = f.copy()
    best_score = score_from_a(f)
    t0 = time.monotonic()

    factors = [0.85, 0.92, 0.96, 1.04, 1.08, 1.15, 1.3]
    no_improve = 0
    max_no_improve = 60
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit and no_improve < max_no_improve:
        no_improve = 0
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            for fac in factors:
                f_trial = f.copy()
                old_val = f_trial[i]
                f_trial[i] = old_val * fac
                diff = f_trial.sum() - 1.0
                if abs(diff) < 1e-16:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others_sum = f_trial[mask].sum()
                if others_sum < 1e-15:
                    continue
                f_trial[mask] -= diff * f_trial[mask] / others_sum
                if f_trial.min() < -1e-10:
                    continue
                f_trial = np.maximum(f_trial, 0)
                f_trial = f_trial / f_trial.sum()
                sc = score_from_a(f_trial)
                if sc < best_score - 1e-13:
                    best_score = sc
                    f = f_trial.copy()
                    best_f = f.copy()
                    improved = True
                    break
        if not improved:
            no_improve += 1

    return best_f, best_score


def check_and_update(f, label, best_seq, best_score):
    """Evaluate a candidate using evaluate_sequence."""
    if f is None:
        return best_seq, best_score
    f = np.maximum(np.asarray(f, dtype=np.float64), 0.0)
    s = f.sum()
    if s < 0.01:
        return best_seq, best_score
    f_norm = f / s
    score = float(evaluate_sequence(f_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = f_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.7f} (n={len(f_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. LP-based direction finding (primary, most principled)
    2. Multi-peak mirror descent (experiments with different K)
    3. L-BFGS polishing at decreasing temperatures
    4. Peak shaving and coordinate descent for final refinement
    5. Perturbation-based escape from local minima
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.7f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = p_score

    n = len(best_seq) if best_seq is not None else 768

    # === Phase 1: LP-based optimization (primary method) ===
    print(f"\n=== Phase 1: LP Optimization (best={best_score:.7f}) ===")
    if best_seq is not None:
        # LP round 1: tight constraint
        rem = deadline - time.monotonic() - 300
        if rem > 60:
            t_alloc = min(rem * 0.45, 400.0)
            print(f"  LP round 1: n={n}, M_relax=1.0, budget={t_alloc:.0f}s")
            a_lp, _ = lp_optimize(best_seq, t_alloc, M_relax=1.0, label="LP1")
            best_seq, best_score = check_and_update(a_lp, "LP1", best_seq, best_score)

        # LP round 2: slightly relaxed
        rem = deadline - time.monotonic() - 250
        if rem > 40 and best_seq is not None:
            t_alloc = min(rem * 0.35, 250.0)
            print(f"  LP round 2: n={n}, M_relax=1.005, budget={t_alloc:.0f}s")
            a_lp2, _ = lp_optimize(best_seq, t_alloc, M_relax=1.005, label="LP2")
            best_seq, best_score = check_and_update(a_lp2, "LP2", best_seq, best_score)

        # LP round 3: tight again for refinement
        rem = deadline - time.monotonic() - 200
        if rem > 30 and best_seq is not None:
            t_alloc = min(rem * 0.35, 200.0)
            print(f"  LP round 3: n={n}, M_relax=1.0, budget={t_alloc:.0f}s")
            a_lp3, _ = lp_optimize(best_seq, t_alloc, M_relax=1.0, label="LP3")
            best_seq, best_score = check_and_update(a_lp3, "LP3", best_seq, best_score)

    # === Phase 2: Multi-peak Mirror Descent ===
    print(f"\n=== Phase 2: Multi-Peak Mirror Descent (best={best_score:.7f}) ===")
    if best_seq is not None:
        for K in [3, 5, 8]:
            rem = deadline - time.monotonic() - 180
            if rem < 30:
                break
            t_alloc = min(rem * 0.3, 100.0)
            print(f"  MPMD K={K}: n={n}, budget={t_alloc:.0f}s")
            a_mp, _ = multi_peak_mirror_descent(
                best_seq, t_alloc, K=K, T_start=0.008, T_end=0.0005,
                eta_start=0.012, eta_end=0.0008, label=f"MPMD_K{K}"
            )
            best_seq, best_score = check_and_update(a_mp, f"MPMD_K{K}", best_seq, best_score)

    # === Phase 3: L-BFGS polishing ===
    print(f"\n=== Phase 3: L-BFGS Polish (best={best_score:.7f}) ===")
    if best_seq is not None:
        for T in [30, 10, 5, 2]:
            rem = deadline - time.monotonic() - 150
            if rem < 20:
                break
            t_alloc = min(rem * 0.2, 50.0)
            print(f"  L-BFGS T={T}: budget={t_alloc:.0f}s")
            a_lb = lbgfs_polish(best_seq, t_alloc, T=T, maxiter=4000)
            best_seq, best_score = check_and_update(a_lb, f"lbfs_T{T}", best_seq, best_score)

    # === Phase 4: Peak Shaving ===
    print(f"\n=== Phase 4: Peak Shaving (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_alloc = min(rem * 0.4, 70.0)
            print(f"  Peak Shave: budget={t_alloc:.0f}s")
            a_ps, _ = peak_shave(best_seq, t_alloc)
            best_seq, best_score = check_and_update(a_ps, "peak_shave", best_seq, best_score)

    # === Phase 5: Coordinate Descent ===
    print(f"\n=== Phase 5: Coordinate Descent (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 20:
            t_alloc = min(rem * 0.5, 70.0)
            print(f"  CD: budget={t_alloc:.0f}s")
            a_cd, _ = coordinate_descent(best_seq, t_alloc)
            best_seq, best_score = check_and_update(a_cd, "coord_descent", best_seq, best_score)

    # === Phase 6: Combined refinement ===
    print(f"\n=== Phase 6: Combined Refinement (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 30:
            t_alloc = min(rem * 0.35, 60.0)
            print(f"  LP refine: budget={t_alloc:.0f}s")
            a_r1, _ = lp_optimize(best_seq, t_alloc, M_relax=1.0, label="RefLP")
            best_seq, best_score = check_and_update(a_r1, "RefLP", best_seq, best_score)

        rem = deadline - time.monotonic() - 40
        if rem > 20:
            t_alloc = min(rem * 0.35, 40.0)
            a_r2 = lbgfs_polish(best_seq, t_alloc, T=3, maxiter=4000)
            best_seq, best_score = check_and_update(a_r2, "RefLB", best_seq, best_score)

        rem = deadline - time.monotonic() - 25
        if rem > 15:
            a_r3, _ = peak_shave(best_seq, min(rem * 0.5, 25.0))
            best_seq, best_score = check_and_update(a_r3, "RefPS", best_seq, best_score)

        rem = deadline - time.monotonic() - 15
        if rem > 10:
            a_r4, _ = coordinate_descent(best_seq, min(rem * 0.6, 15.0))
            best_seq, best_score = check_and_update(a_r4, "RefCD", best_seq, best_score)

    # === Phase 7: Perturbation-based escape ===
    print(f"\n=== Phase 7: Perturbation Escape (best={best_score:.7f}) ===")
    if best_seq is not None:
        for psize in [0.03, 0.015, 0.008]:
            rem = deadline - time.monotonic() - 30
            if rem < 20:
                break
            t_alloc = min(rem * 0.3, 35.0)
            perturbed = best_seq + rng.standard_normal(n) * psize
            perturbed = np.maximum(perturbed, 1e-12)
            perturbed = perturbed / perturbed.sum()
            a_p, _ = lp_optimize(perturbed, t_alloc, M_relax=1.0, label=f"pert{psize}")
            best_seq, best_score = check_and_update(a_p, f"pert{psize}", best_seq, best_score)

            rem = deadline - time.monotonic() - 25
            if rem > 15 and best_seq is not None:
                t_alloc2 = min(rem * 0.4, 20.0)
                a_p2, _ = peak_shave(best_seq, t_alloc2)
                best_seq, best_score = check_and_update(a_p2, f"pert{psize}_ps", best_seq, best_score)

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 8:
            a_f1 = lbgfs_polish(best_seq, min(rem * 0.4, 15.0), T=2, maxiter=3000)
            best_seq, best_score = check_and_update(a_f1, "FinLB", best_seq, best_score)

        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_f2, _ = peak_shave(best_seq, min(rem * 0.6, 10.0))
            best_seq, best_score = check_and_update(a_f2, "FinPS", best_seq, best_score)

        rem = deadline - time.monotonic() - 3
        if rem > 3:
            a_f3, _ = coordinate_descent(best_seq, min(rem * 0.7, 8.0))
            best_seq, best_score = check_and_update(a_f3, "FinCD", best_seq, best_score)

    # Fallback
    if best_seq is None:
        if parent is not None:
            best_seq = parent
        else:
            best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.7f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
