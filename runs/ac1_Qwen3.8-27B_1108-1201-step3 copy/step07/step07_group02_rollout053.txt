            if m > 500:
                return np.zeros_like(z_vec)
            exp_Tb = np.exp(Tb - m)
            w = exp_Tb / exp_Tb.sum()
            # grad_a[j] = (4n/T) * sum_k w[k] * a[k-j]
            grad_a = np.zeros(n)
            for k in range(2 * n - 1):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 4.0 * w[k] * a[k - idx]
            # Convert to log-space
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac
    
    # Geometric temperature schedule
    n_stages = 6
    temps = np.geomspace(T_start, T_end, n_stages)
    z_current = z
    iters_per_stage = max(100, maxiter // n_stages)
    
    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z_current,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxfun': iters_per_stage * 30, 'disp': False}
            )
            z_current = res.x
        except Exception:
            break
    
    a_out = softmax_from_z(z_current)
    return a_out

def subgradient_optimize(a0, time_limit, lr_init=0.5, active_frac=0.1):
    """Active-set subgradient method on the simplex."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    
    for step in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        score = 2.0 * n * max_b
        
        # Active set: top active_frac fraction of peaks
        threshold = max_b - active_frac * max_b
        active = np.where(b >= threshold)[0]
        
        # Compute subgradient
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)
        
        # Projected gradient step
        a_new = a - lr * g
        # Project onto simplex
        a_new = _simplex_project(a_new)
        
        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive lr
        if no_improve > 50:
            lr *= 0.9
            no_improve = 0
        lr = max(lr, 1e-8)
        
        if step % 5000 == 0:
            print(f"  SubGrad {time.monotonic()-t0:.0f}s: score={best_score:.6f} lr={lr:.6f} |A|={len(active)}")
    
    return best_a, best_score

def _simplex_project(x):
    n = len(x)
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
        return np.maximum(x - theta, 0.0)
    return np.ones(n) / n

def peak_shave(a0, time_limit, n_steps=20000):
    """Targeted peak shaving with redistribution."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        # Find the global peak (excluding endpoints which are always small)
        b_search = b.copy()
        b_search[0] = -1
        b_search[-1] = -1
        peak_k = int(np.argmax(b_search))
        peak_val = b[peak_k]
        
        if peak_val * n * 2 <= best_score + 1e-10:
            # The peak at peak_k is not the limiting one, try next
            b_search[peak_k] = -1
            peak_k = int(np.argmax(b_search))
            if b_search[peak_k] < 0:
                break
        
        # Find top contributors to this peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_n = min(5, len(idx))
        top_contrib = np.argsort(contribs)[::-1][:top_n]
        
        improved = False
        for ci in top_contrib:
            i = int(idx[ci])
            j = peak_k - i
            if a[i] < 1e-12:
                continue

            for factor in [0.95, 0.9, 0.85, 0.8]:
                diff = a[i] * (1.0 - factor)
                if diff < 1e-15:
                    continue

                break
