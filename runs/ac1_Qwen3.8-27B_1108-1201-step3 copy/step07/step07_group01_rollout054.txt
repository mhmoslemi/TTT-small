"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a novel Hankel fixed-point 
iteration combined with adaptive multiplicative updates, L-BFGS refinement,
and multi-resolution exploration.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_from_a(a):
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())

def build_hankel(w, n):
    """Build Hankel matrix W where W[i,j] = w[i+j] if i+j < len(w), else 0."""
    # W[i,j] = w[i+j] for i+j < 2n-1
    # Use broadcasting
    i_idx = np.arange(n)[:, None]
    j_idx = np.arange(n)[None, :]
    indices = i_idx + j_idx
    W = np.zeros((n, n))
    valid = indices < len(w)
    W[valid] = w[indices[valid]]
    return W

def hankel_solve(w, n):
    """Solve W(a) @ a_new = lambda * 1, sum(a_new) = 1."""
    W = build_hankel(w, n)
    # Augmented system: [W  -1] [a]   = [0]
    #                    [1^T 0] [λ]    = [1]
    A = np.zeros((n + 1, n + 1))
    A[:n, :n] = W
    A[:n, n] = -1.0
    A[n, :n] = 1.0
    rhs = np.zeros(n + 1)
    rhs[n] = 1.0
    
    try:
        sol = np.linalg.solve(A, rhs)
        a_new = sol[:n]
    except np.linalg.LinAlgError:
        # Fallback to least squares
        sol, _, _, _ = np.linalg.lstsq(A, rhs, rcond=None)
        a_new = sol[:n]
    
    return a_new

def hankel_fixed_point(a_init, T, max_iters=100, blend=0.5):
    """Fixed-point iteration using Hankel matrix solve."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    
    for it in range(max_iters):
        b = fast_convolve(a)
        current_score = 2.0 * n * b.max()
        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()
        
        # Compute subgradient weights
        scaled = b / T
        m = scaled.max()
        w = np.exp(scaled - m)
        w = w / w.sum()
        
        # Solve Hankel system
        a_sol = hankel_solve(w, n)
        
        # Project to simplex
        if a_sol.min() < -0.1:
            # Solution is far from feasible, use smaller blend
            a_sol = np.maximum(a_sol, 0)
            s = a_sol.sum()
            if s > 1e-15:
                a_sol = a_sol / s
            else:
                break
        
        # Blend
        a = (1 - blend) * a + blend * np.maximum(a_sol, 0)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
        else:
            a = a / s
    
    return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1) Multi-initialization exploration (parent, structured, random)
2) Top-k multiplicative weights with FFT-based gradient
3) Hankel fixed-point iteration (novel: solves linear system from subgradient condition)

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_from_a(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return float(2.0 * n * b.max())


def compute_grad_fft(a, w):
    """Compute g[i] = sum_k w[k]*a[k-i] using FFT. w has length 2n-1."""
    n = len(a)
    N = 1 << (3 * n - 2).bit_length()
    WF = rfft(w, N)
    AF = rfft(a, N)
    C = irfft(np.conj(WF) * AF, N)
    return C[:n]


def peak_grad(a, k_peak):
    """Subgradient of b[k_peak] wrt a."""
    n = len(a)
    grad = np.zeros(n)
    js = max(0, k_peak - n + 1)
    je = min(n - 1, k_peak)
    idx = np.arange(js, je + 1)
    grad[idx] = a[k_peak - idx]
    return grad


def topk_grad(a, k_list):
    """Average subgradient over multiple peaks."""
    n = len(a)
    grad = np.zeros(n)
    for k in k_list:
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += a[k - idx]
    grad /= len(k_list)
    return grad


def softmax_vec(x, T):
    """Stable softmax with temperature."""
    x = np.asarray(x, dtype=np.float64) / T
    m = x.max()
    e = np.exp(x - m)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(x) / len(x)
    return e / s


def logsumexp_smoothed_obj(a, T, n):
    """Smoothed objective: 2n * T * logsumexp(b/T)."""
    b = fast_convolve(a)
    scaled = b / T
    m = scaled.max()
    lse = m + np.log(np.exp(scaled - m).sum())
    return float(2.0 * n * T * lse)


def logsumexp_smoothed_grad(a, T, n):
    """Gradient of smoothed objective."""
    b = fast_convolve(a)
    w = softmax_vec(b, T)
    grad = compute_grad_fft(a, w)
    return grad


def multiplicative_weights_topk(a_init, time_limit, k=5, eta_start=0.01,
                                decay=0.9997, max_iters=200000,
                                exprange=3.0, verbose=True):
    """Multiplicative weights using top-k peak subgradient."""
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    eta = eta_start
    stuck = 0
    stuck_thresh = 500

    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        cur = 2.0 * n * b.max()

        if cur < best_score - 1e-14:
            best_score = cur
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1

        # Determine active peaks
        b_max = b.max()
        if k == 1:
            k_peak = int(np.argmax(b))
            grad = peak_grad(a, k_peak)
        else:
            # Top-k peaks
            sorted_b = np.sort(b)[::-1]
            threshold = sorted_b[min(k, len(b)) - 1]
            active = np.where(b >= threshold * 0.99)[0]
            if len(active) > k:
                top_k_indices = np.argsort(b)[::-1][:k]
            else:
                top_k_indices = active
            grad = topk_grad(a, top_k_indices)

        # Project to tangent space of simplex
        grad = grad - np.dot(a, grad)

        # Multiplicative update
        log_upd = -eta * grad
        log_upd = np.clip(log_upd, -exprange, exprange)
        a = a * np.exp(log_upd)
        a = np.maximum(a, 1e-300)
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
        else:
            a = a / s

        eta *= decay

        # Stuck recovery
        if stuck > stuck_thresh:
            rng = np.random.default_rng(it * 104729)
            noise = rng.standard_normal(n) * 0.03
            a = best_a + noise
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            stuck = 0
            eta = min(eta * 5.0, eta_start * 0.3)

        if verbose and it % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MW-topk iter={it} ({elapsed:.0f}s): k={k} eta={eta:.6f} "
                  f"score={cur:.7f} best={best_score:.7f}")

    return best_a, best_score


def hankel_fixed_point(a_init, time_limit, T_start=1.0, T_end=0.01,
                       blend_start=0.3, blend_end=0.8, max_iters=2000,
                       verbose=True):
    """
    Fixed-point iteration: at each step, solve the linear system
    W(w) @ a_new = lambda * 1, sum(a_new) = 1
    where W is the Hankel matrix built from the subgradient weights w.
    """
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()

    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        progress = min(1.0, it / (0.8 * max_iters))
        T = T_start * (T_end / T_start) ** progress
        blend = blend_start + (blend_end - blend_start) * progress

        b = fast_convolve(a)
        cur = 2.0 * n * b.max()

        if cur < best_score - 1e-14:
            best_score = cur
            best_a = a.copy()

        # Compute subgradient weights
        w = softmax_vec(b, T)

        # Build augmented Hankel system
        # W[i,j] = w[i+j] for i+j < 2n-1
        i_idx = np.arange(n)[:, None].astype(np.int64)
        j_idx = np.arange(n)[None, :].astype(np.int64)
        indices = i_idx + j_idx
        valid = indices < (2 * n - 1)

        W = np.zeros((n, n))
        W[valid] = w[indices[valid]]

        # Augmented: [W  -1] [a]   = [0]
        #             [1^T 0] [lam]  = [1]
        A = np.zeros((n + 1, n + 1))
        A[:n, :n] = W
        A[:n, n] = -1.0
        A[n, :n] = 1.0
        rhs = np.zeros(n + 1)
        rhs[n] = 1.0

        try:
            # Check condition number
            cond = np.linalg.cond(A)
            if cond > 1e12:
                # Ill-conditioned, use regularized solve
                A_reg = A + 1e-6 * np.eye(n + 1)
                sol = np.linalg.solve(A_reg, rhs)
            else:
                sol = np.linalg.solve(A, rhs)
            a_sol = sol[:n]
        except np.linalg.LinAlgError:
            a_sol = a.copy()

        # Project to simplex
        a_sol = np.maximum(a_sol, 0.0)
        s_sol = a_sol.sum()
        if s_sol < 1e-15:
            a_sol = a.copy()
        else:
            a_sol = a_sol / s_sol

        # Blend
        a = (1.0 - blend) * a + blend * a_sol
        s = a.sum()
        if s < 1e-15:
            a = best_a.copy()
        else:
            a = a / s

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Hankel FP iter={it} ({elapsed:.0f}s): T={T:.4f} "
                  f"blend={blend:.2f} score={cur:.7f} best={best_score:.7f}")

    return best_a, best_score


def lbfgs_continuation(a_init, time_limit, T_schedule=None, maxiter_per_T=3000,
                       verbose=True):
    """L-BFGS with geometric temperature continuation."""
    n = len(a_init)
    if T_schedule is None:
        T_schedule = [200, 100, 50, 25, 12, 6, 3, 1.5, 0.8, 0.4]

    a = np.maximum(np.asarray(a_init, dtype=np.float64), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()

    def softmax_from_z(z):
        z_s = z - z.max()
        e = np.exp(z_s)
        s = e.sum()
        if s < 1e-300:
            return np.ones_like(z) / len(z)
        return e / s

    for Ti, T in enumerate(T_schedule):
        if time.monotonic() - t0 > time_limit:
            break
        T_remaining = time_limit - (time.monotonic() - t0)
        if T_remaining < 5:
            break

        z0 = np.log(np.maximum(a, 1e-15))
        z0 = z0 - z0.mean()

        calls = [0]

        def func(z):
            calls[0] += 1
            a_z = softmax_from_z(z)
            b = fast_convolve(a_z)
            scaled = b / T
            m = scaled.max()
            lse = m + np.log(np.exp(scaled - m).sum())
            return float(2.0 * n * T * lse)

        def jac(z):
            calls[0] += 1
            a_z = softmax_from_z(z)
            b = fast_convolve(a_z)
            w = softmax_vec(b, T)
            grad_a = compute_grad_fft(a_z, w)
            mean_g = np.dot(a_z, grad_a)
            return a_z * (grad_a - mean_g)

        try:
            res = minimize(
                fun=func, x0=z0, jac=jac,
                method='L-BFGS-B',
                options={'maxiter': maxiter_per_T, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': maxiter_per_T * 40,
                         'disp': False}
            )
            z_opt = res.x
            a_new = softmax_from_z(z_opt)
        except Exception:
            a_new = a.copy()

        # Evaluate
        sc = score_from_a(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    L-BFGS T={T} ({elapsed:.0f}s): score={sc:.7f} "
                  f"best={best_score:.7f} calls={calls[0]}")

    return best_a, best_score


def fista_smoothed(a_init, time_limit, T=2.0, max_iters=50000,
                   lr=None, verbose=True):
    """FISTA (accelerated proximal gradient) on the smoothed problem."""
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()

    if lr is None:
        # Estimate Lipschitz constant
        b = fast_convolve(a)
        w = softmax_vec(b, T)
        g = compute_grad_fft(a, w)
        lip = np.linalg.norm(g) / T + 1.0
        lr = 1.0 / lip

    a_curr = a.copy()
    a_prev = a.copy()
    t_moment = 1.0

    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        # FISTA momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_moment * t_moment)) / 2.0
        a_y = a_curr + ((t_moment - 1.0) / t_new) * (a_curr - a_prev)
        a_y = np.maximum(a_y, 1e-300)
        a_y = a_y / a_y.sum()
        t_moment = t_new

        # Gradient step
        b = fast_convolve(a_y)
        w = softmax_vec(b, T)
        grad = compute_grad_fft(a_y, w)

        # Project gradient to tangent space
        grad = grad - np.dot(a_y, grad)

        a_new = a_y - lr * grad
        # Project to simplex
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a_curr
        else:
            a_new = a_new / s

        a_prev = a_curr
        a_curr = a_new

        sc = 2.0 * n * fast_convolve(a_curr).max()
        if sc < best_score:
            best_score = sc
            best_a = a_curr.copy()

        if verbose and it % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FISTA iter={it} ({elapsed:.0f}s): T={T} lr={lr:.6f} "
                  f"score={sc:.7f} best={best_score:.7f}")

    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def generate_structured_init(n, seed=0):
    """Generate various structured initializations."""
    rng = np.random.default_rng(seed)
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Sine window
    i = np.arange(n)
    a_sin = np.abs(np.sin(np.pi * (i + 1) / n))
    inits.append(a_sin / a_sin.sum())

    # Raised cosine
    a_rc = np.cos(np.pi * i / (2 * n)) ** 2
    inits.append(a_rc / a_rc.sum())

    # Gaussian
    mu = n / 2.0
    sigma = n / 4.0
    a_g = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    inits.append(a_g / a_g.sum())

    # Random with smoothing
    a_r = rng.random(n)
    # Simple boxcar smoothing
    kernel = np.ones(5) / 5
    a_r_smooth = np.convolve(a_r, kernel, mode='same')
    a_r_smooth = np.maximum(a_r_smooth, 0)
    inits.append(a_r_smooth / a_r_smooth.sum())

    # Power-law
    a_p = 1.0 / (i + 1) ** 0.3
    inits.append(a_p / a_p.sum())

    return inits


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 12.0)
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.7f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default init: score={best_score:.7f}")

    n_parent = len(parent)

    # === Phase 0: Multi-resolution warmup ===
    print(f"\n=== Phase 0: Multi-Resolution Warmup ===")
    resolutions = [128, 256, 512, n_parent]
    resolutions = sorted(set([r for r in resolutions if 64 <= r <= 4096]))

    a_warm = np.ones(64) / 64
    if n_parent >= 256:
        a_warm = interpolate_sequence(parent, 128)
    elif n_parent <= 256:
        a_warm = interpolate_sequence(parent, max(64, n_parent // 2))

    for res_n in resolutions:
        rem = deadline - time.monotonic() - 400
        if rem < 30:
            break
        t_alloc = min(rem * 0.25, 60.0)

        a_warm = interpolate_sequence(a_warm, res_n)
        a_warm, _ = multiplicative_weights_topk(
            a_warm, t_alloc, k=3, eta_start=0.03, decay=0.9999,
            max_iters=50000, verbose=False
        )
        best_seq, best_score = check_and_update(a_warm, f"warmup_n{res_n}", best_seq, best_score)
        if best_seq is not None:
            a_warm = interpolate_sequence(best_seq, res_n)

    print(f"  After warmup: best={best_score:.7f}")

    # === Phase 1: Core optimization on best length ===
    n_work = len(best_seq)
    print(f"\n=== Phase 1: Core Optimization (n={n_work}) ===")

    # 1a: Top-k multiplicative weights (k=1, then k=5, then k=10)
    for k_val in [1, 5, 10]:
        rem = deadline - time.monotonic() - 500
        if rem < 30:
            break
        t_alloc = min(rem * 0.3, 120.0)
        print(f"  MW topk={k_val}, budget={t_alloc:.0f}s")
        a_mw, _ = multiplicative_weights_topk(
            best_seq, t_alloc, k=k_val,
            eta_start=0.008, decay=0.9998, max_iters=100000
        )
        best_seq, best_score = check_and_update(a_mw, f"MW_k{k_val}", best_seq, best_score)

    # 1b: Hankel fixed-point
    rem = deadline - time.monotonic() - 400
    if rem > 30:
        t_alloc = min(rem * 0.35, 150.0)
        print(f"  Hankel FP, budget={t_alloc:.0f}s")
        a_hf, _ = hankel_fixed_point(
            best_seq, t_alloc, T_start=2.0, T_end=0.05,
            blend_start=0.2, blend_end=0.7, max_iters=2000
        )
        best_seq, best_score = check_and_update(a_hf, "HankelFP", best_seq, best_score)

    # 1c: More MW with different parameters
    rem = deadline - time.monotonic() - 350
    if rem > 30:
        t_alloc = min(rem * 0.3, 100.0)
        a_mw2, _ = multiplicative_weights_topk(
            best_seq, t_alloc, k=3,
            eta_start=0.005, decay=0.9999, max_iters=100000, exprange=2.0
        )
        best_seq, best_score = check_and_update(a_mw2, "MW2", best_seq, best_score)

    # === Phase 2: L-BFGS Continuation ===
    print(f"\n=== Phase 2: L-BFGS Continuation (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic() - 250
    if rem > 30:
        T_sched = [100, 50, 25, 12, 6, 3, 1.5, 0.8, 0.4, 0.2, 0.1]
        t_alloc = min(rem * 0.7, 600.0)
        a_lb, _ = lbfgs_continuation(
            best_seq, t_alloc, T_schedule=T_sched, maxiter_per_T=2000
        )
        best_seq, best_score = check_and_update(a_lb, "L-BFGS_cont", best_seq, best_score)

    # === Phase 3: FISTA on smoothed problem ===
    print(f"\n=== Phase 3: FISTA (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic() - 150
    if rem > 20:
        for T_fista in [5.0, 2.0, 1.0]:
            rem = deadline - time.monotonic() - 100
            if rem < 15:
                break
            t_alloc = min(rem * 0.3, 60.0)
            a_fista, _ = fista_smoothed(
                best_seq, t_alloc, T=T_fista, max_iters=30000
            )
            best_seq, best_score = check_and_update(a_fista, f"FISTA_T{T_fista}", best_seq, best_score)

    # === Phase 4: Refinement round ===
    print(f"\n=== Phase 4: Refinement (best={best_score:.7f}) ===")
    # More MW
    rem = deadline - time.monotonic() - 100
    if rem > 20:
        t_alloc = min(rem * 0.4, 80.0)
        a_mw3, _ = multiplicative_weights_topk(
            best_seq, t_alloc, k=1, eta_start=0.003, decay=0.99995,
            max_iters=200000, exprange=1.5
        )
        best_seq, best_score = check_and_update(a_mw3, "MW3_fine", best_seq, best_score)

    # Low-T L-BFGS
    rem = deadline - time.monotonic() - 80
    if rem > 15:
        t_alloc = min(rem * 0.4, 60.0)
        a_lb2, _ = lbfgs_continuation(
            best_seq, t_alloc, T_schedule=[1.0, 0.5, 0.2, 0.1], maxiter_per_T=3000
        )
        best_seq, best_score = check_and_update(a_lb2, "L-BFGS_fine", best_seq, best_score)

    # === Phase 5: Peak shaving and coordinate descent ===
    print(f"\n=== Phase 5: Peak Shaving + CD (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic() - 40
    if rem > 15:
        t_alloc = min(rem * 0.4, 60.0)
        a_ps, _ = peak_shave_refined(best_seq, t_alloc)
        best_seq, best_score = check_and_update(a_ps, "peak_shave", best_seq, best_score)

    rem = deadline - time.monotonic() - 20
    if rem > 10:
        t_alloc = min(rem * 0.5, 40.0)
        a_cd, _ = coordinate_descent_fast(best_seq, t_alloc)
        best_seq, best_score = check_and_update(a_cd, "coord_descent", best_seq, best_score)

    # === Phase 6: Alternative lengths ===
    print(f"\n=== Phase 6: Alternative Lengths (best={best_score:.7f}) ===")
    n_best = len(best_seq)
    alt_lengths = []
    for factor in [1.5, 2.0, 0.75, 1.33]:
        alt_n = int(n_best * factor)
        if 128 <= alt_n <= 4096 and alt_n != n_best and alt_n not in alt_lengths:
            alt_lengths.append(alt_n)

    for alt_n in alt_lengths[:2]:
        rem = deadline - time.monotonic() - 30
        if rem < 40:
            break
        t_alloc = min(rem * 0.4, 80.0)
        init_alt = interpolate_sequence(best_seq, alt_n)

        # Quick MW + L-BFGS
        a_alt_mw, _ = multiplicative_weights_topk(
            init_alt, t_alloc * 0.5, k=3, eta_start=0.02,
            decay=0.9998, max_iters=50000, verbose=False
        )
        best_seq, best_score = check_and_update(a_alt_mw, f"MW_alt_n{alt_n}", best_seq, best_score)

        if len(best_seq) == alt_n:
            rem = deadline - time.monotonic() - 20
            if rem > 15:
                a_alt_lb, _ = lbfgs_continuation(
                    best_seq, min(rem * 0.5, 40.0),
                    T_schedule=[50, 10, 3, 1], maxiter_per_T=1500
                )
                best_seq, best_score = check_and_update(a_alt_lb, f"LBFGS_alt_n{alt_n}", best_seq, best_score)

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic() - 5
    if rem > 8:
        # Very fine MW
        t_alloc = min(rem * 0.4, 30.0)
        a_fin, _ = multiplicative_weights_topk(
            best_seq, t_alloc, k=1, eta_start=0.001,
            decay=0.99998, max_iters=100000, exprange=1.0
        )
        best_seq, best_score = check_and_update(a_fin, "final_MW", best_seq, best_score)

    rem = deadline - time.monotonic() - 3
    if rem > 5:
        a_fin2, _ = peak_shave_refined(best_seq, min(rem * 0.7, 15.0))
        best_seq, best_score = check_and_update(a_fin2, "final_PS", best_seq, best_score)

    # === Final verification ===
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs

    total_time = time.monotonic() - t_start
    print(f"\nTOTAL TIME: {total_time:.1f}s")
    print(f"FINAL SCORE: {best_score:.7f} (n={len(best_seq) if best_seq is not None else 0})")

    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    return [float(x) for x in best_seq]
