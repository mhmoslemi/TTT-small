def compute_smoothed_grad(a, top_k=30, temperature=0.005):
    n = len(a)
    b = fast_convolve(a)
    k_actual = min(top_k, len(b))
    peaks = np.argpartition(b, -k_actual)[-k_actual:]
    pv = b[peaks]
    temp = temperature * max(pv.max(), 1e-10)
    w = np.exp((pv - pv.max()) / temp)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-14:
            continue
        k = int(peaks[idx])
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            j = np.arange(i_start, i_end + 1)
            partners = k - j
            if k == 2 * int(j[0]) and len(j) == 1:
                grad[j] += w[idx] * a[k - j]
            else:
                grad[j] += 2.0 * w[idx] * a[partners]
    return grad

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        if rem > 30:
            t_dl = min(rem * 0.3, 80.0)
            a_dl, _ = damped_lp_refine(best_seq, t_dl, solver=solver)
            best_seq, best_score = check_and_update(a_dl, best_seq, best_score, "damped_lp")
    print(f"  After Phase 1: best={best_score:.6f} ({time.monotonic()-t1_start:.0f}s)")

    # Phase 2: Try nearby lengths
    print(f"\n=== Phase 2: Nearby Lengths (best={best_score:.6f}) ===")
    t2_start = time.monotonic()
    t_phase2 = 300.0
    if best_seq is not None:
        n_best = len(best_seq)
        alt_lengths = []
        for delta in [-64, -32, 32, 64, 96, 128]:
            alt_n = n_best + delta
            if 256 <= alt_n <= 2560:
                alt_lengths.append(alt_n)
        for alt_n in alt_lengths[:4]:
            rem = deadline - time.monotonic() - 700
            if rem < 40:
                break
            t_alt = min(rem * 0.4, 60.0)
            init_alt = interpolate_seq(best_seq, alt_n)
            a_alt, s_alt, _ = fictitious_play(alt_n, t_alt, initial=init_alt, solver=solver, verbose=False)
            best_seq, best_score = check_and_update(a_alt, best_seq, best_score, f"alt_n{alt_n}")
            if best_seq is not None and len(best_seq) != n_best:
                # Try local search on the new length
                rem = deadline - time.monotonic() - 600
                if rem > 20:
                    a_cd, _ = coord_descent_zoom(best_seq, min(rem * 0.3, 30.0), rng)
                    best_seq, best_score = check_and_update(a_cd, best_seq, best_score, f"cd_n{len(best_seq)}")
    print(f"  After Phase 2: best={best_score:.6f} ({time.monotonic()-t2_start:.0f}s)")

    # Phase 3: Intensive coordinate descent with zoom
    print(f"\n=== Phase 3: Coordinate Descent (best={best_score:.6f}) ===")
    t3_start = time.monotonic()
    if best_seq is not None:
        rem = deadline - time.monotonic() - 500
        if rem > 30:
            t_cd = min(rem * 0.5, 500.0)
            print(f"  CoordDescent n={len(best_seq)}, budget={t_cd:.0f}s")
            a_cd, s_cd = coord_descent_zoom(best_seq, t_cd, rng)
            best_seq, best_score = check_and_update(a_cd, best_seq, best_score, "coord_descent")
    print(f"  After Phase 3: best={best_score:.6f} ({time.monotonic()-t3_start:.0f}s)")

    # Phase 4: Pair optimization targeting peaks
    print(f"\n=== Phase 4: Pair Optimization (best={best_score:.6f}) ===")
    t4_start = time.monotonic()
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 20:
            t_pair = min(rem * 0.5, 200.0)
            print(f"  PairOpt budget={t_pair:.0f}s")
            a_pair, s_pair = pair_optimize(best_seq, t_pair, rng)
            best_seq, best_score = check_and_update(a_pair, best_seq, best_score, "pair_opt")
            # Follow with coord descent
            rem = deadline - time.monotonic() - 250
            if rem > 15:
                a_cd2, _ = coord_descent_zoom(best_seq, min(rem * 0.4, 60.0), rng)
                best_seq, best_score = check_and_update(a_cd2, best_seq, best_score, "cd_after_pair")
    print(f"  After Phase 4: best={best_score:.6f} ({time.monotonic()-t4_start:.0f}s)")

    # Phase 5: Nesterov with smoothed objective
    print(f"\n=== Phase 5: Smoothed Nesterov (best={best_score:.6f}) ===")
    t5_start = time.monotonic()
    if best_seq is not None:
        for temp_try in [0.003, 0.008, 0.015]:
            rem = deadline - time.monotonic() - 200
            if rem < 20:
                break
            t_nest = min(rem * 0.35, 100.0)
            print(f"  Nesterov temp={temp_try}, budget={t_nest:.0f}s")
            a_nt, s_nt = nesterov_smoothed(best_seq, t_nest, top_k=30, temperature=temp_try, lr_init=0.004)
            best_seq, best_score = check_and_update(a_nt, best_seq, best_score, f"nest_t{temp_try}")
            # Quick coord descent follow-up
            rem = deadline - time.monotonic() - 150
            if rem > 15:
                a_cd3, _ = coord_descent_zoom(best_seq, min(rem * 0.3, 40.0), rng)
                best_seq, best_score = check_and_update(a_cd3, best_seq, best_score, "cd_after_nest")
    print(f"  After Phase 5: best={best_score:.6f} ({time.monotonic()-t5_start:.0f}s)")

    # Phase 6: Second round of pair + coord descent (different focus)
    print(f"\n=== Phase 6: Refined Pair+CD (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            a_p2, _ = pair_optimize(best_seq, min(rem * 0.3, 80.0), rng)
            best_seq, best_score = check_and_update(a_p2, best_seq, best_score, "pair2")
            rem = deadline - time.monotonic() - 100
            if rem > 15:
                a_c2, _ = coord_descent_zoom(best_seq, min(rem * 0.5, 80.0), rng)
                best_seq, best_score = check_and_update(a_c2, best_seq, best_score, "cd2")
        # Pair again
        rem = deadline - time.monotonic() - 80
        if rem > 15:
            a_p3, _ = pair_optimize(best_seq, min(rem * 0.5, 50.0), rng)
            best_seq, best_score = check_and_update(a_p3, best_seq, best_score, "pair3")

    # Phase 7: Final FP burst + polish
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 50
        if rem > 30:
            t_fp_final = min(rem * 0.3, 60.0)
            a_ff, _, _ = fictitious_play(len(best_seq), t_fp_final, initial=best_seq, solver=solver, verbose=False)
            best_seq, best_score = check_and_update(a_ff, best_seq, best_score, "fp_final")
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            a_cf, _ = coord_descent_zoom(best_seq, min(rem * 0.5, 40.0), rng)
            best_seq, best_score = check_and_update(a_cf, best_seq, best_score, "final_cd")
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            a_pf, _ = pair_optimize(best_seq, min(rem * 0.5, 20.0), rng)
            best_seq, best_score = check_and_update(a_pf, best_seq, best_score, "final_pair")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = fast_score(best_seq)

    # Final verification with evaluate_sequence
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
