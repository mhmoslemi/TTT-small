    Compute gradient of sum_i w[i] * b_i w.r.t. a, where b = autoconv(a).

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS with log-sum-exp smoothing
and softmax parameterization for simplex constraint handling.
Uses annealed temperature for progressive peak reduction.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_autoconv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    return exp_z / exp_z.sum()


def lse_objective_and_grad(a, T):
    """
    Compute (1/T)*log(sum(exp(T*b))) and its gradient w.r.t. a,
    where b = autoconv(a).
    """
    n = len(a)
    b = fast_autoconv(a)
    
    # Shift for numerical stability
    b_max = b.max()
    b_shifted = T * (b - b_max)
    exp_b = np.exp(b_shifted)
    sum_exp = exp_b.sum()
    obj = b_max + np.log(sum_exp) / T
    
    # Weights for the gradient
    weights = exp_b / sum_exp  # length 2n-1
    
    # Gradient: grad_a[k] = (2/T) * sum_i weights[i] * a[i-k]
    # = (2/T) * cross_corr(a, weights)[k]
    N = 1 << (3 * n - 2).bit_length()
    A = rfft(a, N)
    W = rfft(weights, N)
    corr = irfft(np.conj(A) * W, N)
    grad_a = (2.0 / T) * corr[:n]
    
    return obj, grad_a


def evaluate_with_lse(a, T):
    """Objective function for L-BFGS (in z-space)."""
    z = a  # This is actually z, not a
    a_s = softmax(z)
    obj, _ = lse_objective_and_grad(a_s, T)
    return obj


def evaluate_grad_with_lse(z, T):
    """Gradient for L-BFGS (in z-space)."""
    a = softmax(z)
    _, grad_a = lse_objective_and_grad(a, T)
    # Transform gradient from a-space to z-space
    # grad_z[j] = a[j] * (grad_a[j] - dot(a, grad_a))
    dot = np.dot(a, grad_a)
    grad_z = a * (grad_a - dot)
    return grad_z

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS with log-sum-exp 
smoothing and softmax parameterization. Uses temperature annealing for 
progressive peak reduction. Includes multi-length exploration and 
coordinate descent polishing.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_autoconv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax_stable(z):
    """Numerically stable softmax."""
    z_shifted = z - np.max(z)
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def lse_objective_and_grad(a, T):
    """
    Compute (1/T)*log(sum(exp(T*b))) and gradient w.r.t. a,
    where b = autoconv(a).
    Returns (objective_value, gradient_vector).
    """
    n = len(a)
    b = fast_autoconv(a)
    b_max = b.max()
    b_shifted = T * (b - b_max)
    exp_b = np.exp(b_shifted)
    sum_exp = exp_b.sum()
    obj = b_max + np.log(sum_exp) / T
    weights = exp_b / sum_exp
    
    # Gradient: (2/T) * cross_corr(a, weights)
    N = 1 << (3 * n - 2).bit_length()
    A = rfft(a, N)
    W = rfft(weights, N)
    corr = irfft(np.conj(A) * W, N)
    grad_a = (2.0 / T) * corr[:n]
    
    return obj, grad_a

    a = softmax_stable(z)
    obj, _ = lse_objective_and_grad(a, T)

    _, grad_a = lse_objective_and_grad(a, T)
    dot = np.dot(a, grad_a)

    return exp_z / s

    W = rfft(weights, N)

    w[active] = 1.0 / len(active)

    # Subgradient: 2 * cross_corr(a, w)
    N = 1 << (3 * n - 2).bit_length()
    A = rfft(a, N)
    W = rfft(w, N)
    corr = irfft(np.conj(A) * W, N)
    grad = 2.0 * corr[:n]

    # Projected step (simple gradient step with projection)
    a_new = a - lr * grad
    # Project onto simplex
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    else:
        a_new = a.copy()
    return a_new, grad


def subgradient_descent(a_init, time_budget, lr_init=0.01, T_top=20, verbose=False):
    """Subgradient descent with adaptive step size. Returns (best_a, best_max_conv)."""
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_obj = fast_autoconv(a).max()
    v = np.zeros(n)
    lr = lr_init
    no_improve = 0
    it = 0

    while (time.monotonic() - t0) < time_budget:
        b = fast_autoconv(a)
        b_max = b.max()
        threshold = b_max * 0.95
        active = np.where(b >= threshold)[0]
        if len(active) > T_top:
            top_idx = np.argpartition(b, -T_top)[-T_top:]
            active = top_idx
        w = np.zeros(2 * n - 1)
        w[active] = 1.0 / len(active)

        N = 1 << (3 * n - 2).bit_length()
        A = rfft(a, N)
        W = rfft(w, N)
        corr = irfft(np.conj(A) * W, N)
        grad = 2.0 * corr[:n]

        # Nesterov momentum
        v = momentum_decay * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s

        obj_new = fast_autoconv(a_new).max()
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        it += 1

        # Adaptive step size
        if no_improve >= 200:
            lr *= 0.7
            v *= 0.5
            no_improve = 0
            if lr < 1e-6:
                break
        elif no_improve >= 100:
            lr *= 0.9

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining L-BFGS with log-sum-exp annealing,
    subgradient descent, multi-length exploration, and coordinate descent.
    """
    t_start = time.monotonic()
    deadline = t_start + max(1.0, budget_s - 45.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"Parent load error: {e}")

    if parent is None:
        n_init = 2048
        parent = make_uniform(n_init)
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"No parent, uniform n={n_init}, score={best_score:.6f}")

    n_parent = len(parent)

    # === Phase 1: L-BFGS anneal on parent ===
    print(f"\n{'='*60}")
    print(f"Phase 1: L-BFGS anneal on parent (n={n_parent})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_alloc = min(rem * 0.35, 350.0)
        print(f"  Time budget: {t_alloc:.0f}s")
        a_opt, obj = lbfgs_anneal(parent, t_alloc, verbose=True)
        s = a_opt.sum()
        if s > 0.01:
            a_norm = a_opt / s
            score = float(evaluate_sequence(a_norm.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = a_norm.copy()
                print(f"  NEW BEST [lbfgs_parent]: {score:.8f}")

    # === Phase 2: Subgradient from L-BFGS result ===
    print(f"\n{'='*60}")
    print(f"Phase 2: Subgradient descent (n={len(best_seq) if best_seq is not None else n_parent})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic()
    if rem > 20 and best_seq is not None:
        t_alloc = min(rem * 0.15, 100.0)
        n_sg = len(best_seq)
        print(f"  n={n_sg}, time={t_alloc:.0f}s")
        a_sg, obj_sg = subgradient_descent(best_seq, t_alloc, lr_init=0.005, T_top=30)
        s = a_sg.sum()
        if s > 0.01:
            a_norm = a_sg / s
            score = float(evaluate_sequence(a_norm.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = a_norm.copy()
                print(f"  NEW BEST [subgrad]: {score:.8f}")

    # === Phase 3: Multi-length exploration ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Multi-length exploration (best={best_score:.6f})")
    print(f"{'='*60}")
    if best_seq is not None:
        n_best = len(best_seq)
        target_ns = []
        for factor in [0.75, 1.0, 1.25, 1.5, 2.0, 0.6, 1.33]:
            tn = int(n_best * factor)
            tn = max(512, min(8192, tn))
            if tn not in target_ns:
                target_ns.append(tn)
        target_ns.sort(key=lambda x: abs(x - n_best))

        for tn in target_ns[:4]:
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            t_alloc = min(rem * 0.12, 80.0)
            if t_alloc < 10:
                continue

            # Interpolated initial
            init_interp = interpolate_seq(best_seq, tn)
            print(f"  n={tn} (interp), t={t_alloc:.0f}s")
            a_opt, obj = lbfgs_anneal(init_interp, t_alloc, verbose=False)
            s = a_opt.sum()
            if s > 0.01:
                a_norm = a_opt / s
                score = float(evaluate_sequence(a_norm.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_seq = a_norm.copy()
                    print(f"    NEW BEST [len_n{tn}]: {score:.8f}")

            # Different initializations
            rem = deadline - time.monotonic()
            if rem < 10:
                break

            inits_list = [
                (make_gaussian(tn, 0.25), "gauss"),
                (make_cosine(tn), "cos"),
                (make_tukey(tn, 0.4), "tukey"),
            ]
            for init_arr, init_name in inits_list:
                rem = deadline - time.monotonic()
                if rem < 10:
                    break
                t_alloc2 = min(rem * 0.06, 30.0)
                if t_alloc2 < 5:
                    continue
                a_opt2, obj2 = lbfgs_anneal(init_arr, t_alloc2, verbose=False)
                s2 = a_opt2.sum()
                if s2 > 0.01:
                    a_norm2 = a_opt2 / s2
                    score2 = float(evaluate_sequence(a_norm2.tolist()))
                    if np.isfinite(score2) and score2 < best_score:
                        best_score = score2
                        best_seq = a_norm2.copy()
                        print(f"    NEW BEST [n{tn}_{init_name}]: {score2:.8f}")

    # === Phase 4: Deep L-BFGS on best ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Deep L-BFGS (best={best_score:.6f})")
    print(f"{'='*60}")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 30:
            n_deep = len(best_seq)
            t_alloc = min(rem * 0.30, 250.0)
            print(f"  n={n_deep}, t={t_alloc:.0f}s")
            # Use higher T schedule for deep optimization
            T_deep = [50, 100, 200, 500, 1000, 2000, 5000, 10000]
            a_deep, obj_deep = lbfgs_anneal(best_seq, t_alloc, T_schedule=T_deep, verbose=True)
            s = a_deep.sum()
            if s > 0.01:
                a_norm = a_deep / s
                score = float(evaluate_sequence(a_norm.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_seq = a_norm.copy()
                    print(f"  NEW BEST [deep]: {score:.8f}")

    # === Phase 5: Coordinate descent polish ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Coordinate descent polish (best={best_score:.6f})")
    print(f"{'='*60}")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 10:
            t_cd = min(rem * 0.25, 150.0)
            n_cd = len(best_seq)
            print(f"  n={n_cd}, t={t_cd:.0f}s")
            a_cd, obj_cd = coordinate_polish(best_seq, t_cd)
            s = a_cd.sum()
            if s > 0.01:
                a_norm = a_cd / s
                score = float(evaluate_sequence(a_norm.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_seq = a_norm.copy()
                    print(f"  NEW BEST [cd]: {score:.8f}")

    # === Phase 6: Final micro-polish ===
    print(f"\n{'='*60}")
    print(f"Phase 6: Final micro-polish (best={best_score:.6f})")
    print(f"{'='*60}")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 5:
            # Very fine L-BFGS with high T
            t_micro = min(rem * 0.4, 30.0)
            T_final = [2000, 5000, 10000, 20000]
            a_micro, obj_micro = lbfgs_anneal(best_seq, t_micro, T_schedule=T_final, verbose=False)
            s = a_micro.sum()
            if s > 0.01:
                a_norm = a_micro / s
                score = float(evaluate_sequence(a_norm.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_seq = a_norm.copy()
                    print(f"  NEW BEST [micro_lbfgs]: {score:.8f}")

        rem = deadline - time.monotonic()
        if rem > 5:
            # Fine coordinate descent
            a_fine, obj_fine = coordinate_polish(best_seq, min(rem, 20.0), num_multipliers=15)
            s = a_fine.sum()
            if s > 0.01:
                a_norm = a_fine / s
                score = float(evaluate_sequence(a_norm.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_seq = a_norm.copy()
                    print(f"  NEW BEST [micro_cd]: {score:.8f}")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if not np.isfinite(fs):
        best_seq = parent.copy()
        fs = float(evaluate_sequence(best_seq.tolist()))
        print(f"WARNING: final check failed, reverting to parent. score={fs:.6f}")

    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: score={fs:.8f}, n={len(best_seq)}, elapsed={elapsed:.0f}s")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
