def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

db[k]/da[j] = 2*a[k-j] if max(0, k-n+1) <= j <= min(n-1, k)
db[k]/da[j] = 0 otherwise

This is much simpler than what I had before! The constraint matrix is:
A[row_i, j] = 2*f[k-row_i - j] if max(0, k-row_i-n+1) <= j <= min(n-1, k-row_i)

3. **Smarter step size**: The optimal step size for the LP direction can be found by solving a 1D minimization:
   minimize max_k b_k((1-t)f + t*h)
   where h = g / sum(g) (normalized direction).
   This is a 1D problem that can be solved by bisection or ternary search.

4. **Combined direction**: Instead of using a single LP direction, combine multiple directions (from different K values) and find the best convex combination.

Let me think about the 1D step size optimization more carefully.

Given f and direction h (both on the simplex, i.e., non-negative with sum 1), we want:
  find t in [0, 1] that minimizes max_k b_k((1-t)f + t*h)

The function b_k((1-t)f + t*h) is a quadratic in t:
  b_k(t) = sum_i ((1-t)f[i] + t*h[i]) * ((1-t)f[k-i] + t*h[k-i])
  = (1-t)^2 * b_f[k] + (1-t)*t * (cross(k)) + t^2 * b_h[k]
  where cross(k) = sum_i (f[i]*h[k-i] + h[i]*f[k-i])

So max_k b_k(t) is the maximum of quadratics in t, which is a convex function of t. We can minimize it using golden section search or bisection on the derivative.

Actually, for a maximum of convex functions, the result is convex but not necessarily differentiable. We can use a simple grid search + refinement.

Let me implement a practical approach: try t in a grid of values, then refine around the best.

"""
Optimized step function search: minimize 2*n*max(conv(a,a))/(sum(a))^2.

Strategy:
1. Multiple smart initializations (exponential, two-blob, parent)
2. LP-based direction finding with adaptive peak selection
3. Bisection line search for optimal step size
4. Spike-chasing: alternate between top-K and single-peak LPs
5. Coordinate descent polish
6. Multi-length exploration
7. Subgradient refinement
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_A(f, n, peak_indices):
    """Build constraint matrix for LP.
    Row i corresponds to peak at index k_i.
    A[i, j] = 2*f[k_i - j] if max(0, k_i-n+1) <= j <= min(n-1, k_i)
    """
    K = len(peak_indices)
    A = np.zeros((K, n))
    for row_i in range(K):
        k = peak_indices[row_i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            A[row_i, lo:hi + 1] = 2.0 * f[k - hi:k - lo + 1][::-1]
    return A


def lp_solve(f, top_k, solver=cp.SCS):
    """Solve direction-finding LP. Returns (g, b) or (None, b)."""
    n = len(f)
    b = fast_convolve(f)
    M = b.max()
    
    if top_k < len(b):
        top_idx = np.argpartition(b, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(b))
    top_idx = np.sort(top_idx)
    
    A = build_A(f, n, top_idx)
    
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= 1e-10]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    
    try:
        prob.solve(solver=solver, max_iters=300000, eps=1e-10)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=300000, eps=1e-10)
        except Exception:
            return None, b
    
    if g.value is None:
        return None, b
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
        return None, b
    return g_val, b


def lp_solve_peak(f, k_idx):
    """Solve LP targeting a single peak. Returns g or None."""
    n = len(f)
    A = build_A(f, n, [k_idx])
    
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= 1e-10]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    
    try:
        prob.solve(solver=cp.SCS, max_iters=300000, eps=1e-10)
    except Exception:
        return None
    
    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None
    return g_val


def find_best_step(f, h):
    """Find optimal step size t for f + t*(h - f) (h and f both on simplex)."""
    best_t = 0.1
    best_obj = fast_convolve(f).max()
    
    # Coarse search
    for t in [0.02, 0.03, 0.05, 0.07, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
        f_t = (1.0 - t) * f + t * h
        f_t = np.maximum(f_t, 0.0)
        s = f_t.sum()
        if s < 1e-10:
            continue
        f_t /= s
        obj = fast_convolve(f_t).max()
        if obj < best_obj:
            best_obj = obj
            best_t = t
    
    # Fine search around best
    t_lo = max(0.005, best_t * 0.7)
    t_hi = min(0.6, best_t * 1.3 + 0.02)
    for t in np.linspace(t_lo, t_hi, 15):
        f_t = (1.0 - t) * f + t * h
        f_t = np.maximum(f_t, 0.0)
        s = f_t.sum()
        if s < 1e-10:
            continue
        f_t /= s
        obj = fast_convolve(f_t).max()
        if obj < best_obj:
            best_obj = obj
            best_t = t
    
    return best_t


def run_lp_optimization(f_init, time_limit, top_k_init=20, top_k_max=150, label="lp"):
    """Main LP-based optimization loop."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 0:
        f /= s
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    stuck = 0
    top_k = top_k_init
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Adaptive top_k: grow every 30 iterations
        if it % 30 == 0 and top_k < top_k_max:
            top_k = min(top_k + 10, top_k_max)
        
        # Choose strategy: top-K LP or single-peak LP (alternate)
        if it % 3 == 0:
            # Single peak chasing
            b = fast_convolve(f)
            worst_k = int(b.argmax())
            g = lp_solve_peak(f, worst_k)
        else:
            # Top-K LP
            g, b = lp_solve(f, top_k)
        
        if g is None:
            # Try larger K
            g, b = lp_solve(f, min(2 * top_k + 20, 2 * n - 1))
            if g is None:
                stuck += 2
                if stuck > 20:
                    break
                continue
        
        S_g = g.sum()
        if S_g <= 1e-5:
            stuck += 1
            if stuck > 15:
                break
            top_k = min(top_k + 20, top_k_max)
            continue
        
        # Normalize direction
        h = g / S_g
        
        # Find best step
        best_t = find_best_step(f, h)
        
        if best_t < 0.005:
            stuck += 1
            if stuck > 15:
                break
            continue
        
        # Update
        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s
        else:
            break
        
        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 40:
                break
        
        if it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it} obj={best_obj:.10f} "
                  f"score={2.0*n*best_obj:.6f} top_k={top_k} stuck={stuck} t={elapsed:.0f}s")
    
    return best_f, best_obj


def coordinate_polish(f_init, time_limit):
    """Coordinate descent: try scaling each element."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    multipliers = [0.5, 0.7, 0.85, 1.15, 1.3, 2.0]
    
    improved = True
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-15:
                continue
            best_j_val = f[j]
            best_j_obj = best_obj
            for mult in multipliers:
                new_val = f[j] * mult
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f /= s
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    improved = True
    
    return f, best_obj


def make_exp_decay(n, decay=2.0):
    """Create exponential decay sequence."""
    x = np.linspace(0, 1, n)
    a = np.exp(-decay * x)
    a = np.maximum(a, 0)
    a /= a.sum()
    return a


def make_two_blob(n):
    """Create two-blob sequence."""
    c1 = n * 0.25
    c2 = n * 0.7
    s = n * 0.08
    a = np.exp(-((np.arange(n) - c1) / s) ** 2) + np.exp(-((np.arange(n) - c2) / s) ** 2)
    a = np.maximum(a, 0)
    a /= a.sum()
    return a


def make_ramp(n, power=1.5):
    """Create decreasing ramp."""
    x = np.linspace(0, 1, n)
    a = (1.0 - x) ** power + 0.01
    a = np.maximum(a, 0)
    a /= a.sum()
    return a


def make_cosine_window(n, alpha=0.3):
    """Cosine-tapered sequence."""
    x = np.linspace(0, 1, n)
    a = 0.5 + 0.5 * np.cos(np.pi * (x - 0.5))
    a = a ** (1/alpha if alpha > 0 else 1)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    a /= s
    return a
