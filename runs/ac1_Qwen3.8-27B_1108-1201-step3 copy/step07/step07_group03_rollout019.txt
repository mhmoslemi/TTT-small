"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using soft-max smoothed
projected gradient descent with momentum, adaptive temperature,
FFT-based gradients, and coordinate descent polishing.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    """Project vector v onto the probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def compute_soft_gradient(a, conv_vals, T, n):
    """
    Compute gradient of T*log(sum_k exp(conv_k/T)) w.r.t. a.
    Uses FFT for efficiency.
    
    gradient_m = (1/T) * sum_k exp(c_k/T) / Z * 2*a_{k-m} (if valid)
               = (2/T) * sum_{k: 0<=k-m<n} p_k * a_{k-m}
               = (2/T) * sum_j p_{m+j} * a_j
    
    where p_k = exp(c_k/T) / Z
    """
    Z = np.sum(np.exp(conv_vals / T))
    p = np.exp(conv_vals / T) / Z  # length 2n-1
    
    # We need: grad_m = (2/T) * sum_{j=0}^{n-1} p_{m+j} * a_j
    # This is a cross-correlation.
    
    # Use FFT:
    N = 1 << (3 * n - 1).bit_length()
    A = rfft(a, N)
    P = rfft(p, N)
    
    # Cross-correlation: (a ⋆ p)[k] = sum_j a[j] * p[j+k]
    # DFT: DFT(a⋆p) = conj(DFT(a)) * DFT(p)
    # For real signals, use rfft
    cross = irfft(conj(A) * P, N)
    
    # cross[k] = sum_j a[j] * p[j+k] for k=0,...,N-1 (with wrap-around)
    # We need cross[m] for m=0,...,n-1
    grad = (2.0 / T) * cross[:n]
    
    return grad

def compute_soft_gradient_direct(a, conv_vals, T):
    """Direct O(n^2) computation of soft-max gradient."""
    n = len(a)
    Z = np.sum(np.exp(conv_vals / T))
    p = np.exp(conv_vals / T) / Z
    
    grad = np.zeros(n)
    for m in range(n):
        # grad_m = (2/T) * sum_j p_{m+j} * a_j
        # m+j ranges from m to m+n-1
        idx = np.arange(m, m + n)
        grad[m] = (2.0 / T) * np.dot(p[idx], a)
    
    return grad

def soft_max_objective(a, T):
    """Compute T*log(sum exp(conv/T)) and the gradient."""
    n = len(a)
    conv_vals = fast_convolve(a)
    cmax = conv_vals.max()
    log_sum = T * np.log(np.sum(np.exp((conv_vals - cmax) / T))) + cmax
    # gradient
    p = np.exp((conv_vals - cmax) / T)
    p /= p.sum()
    
    # FFT-based gradient
    N = 1 << (3 * n - 1).bit_length()
    A = rfft(a, N)
    P = rfft(p, N)
    cross = irfft(conj(A) * P, N)
    grad = (2.0 / T) * cross[:n]
    
    return log_sum, grad, conv_vals

"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    """Project vector v onto the probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def compute_soft_gradient(a, conv_vals, T):
    """Compute gradient of T*log(sum_k exp(conv_k/T)) w.r.t. a using O(n^2) method."""
    n = len(a)
    cmax = conv_vals.max()
    p = np.exp((conv_vals - cmax) / T)
    p_sum = p.sum()
    p = p / p_sum
    # grad[m] = (2/T) * sum_j p[m+j] * a[j] for m=0,...,n-1
    # Build the n x n matrix H where H[m,j] = p[m+j]
    idx = np.arange(n)[:, None] + np.arange(n)[None, :]
    H = p[idx]
    grad = (2.0 / T) * (H @ a)
    return grad

def soft_max_value(conv_vals, T):
    """Compute T*log(sum exp(conv/T))."""
    cmax = conv_vals.max()
    return T * np.log(np.sum(np.exp((conv_vals - cmax) / T))) + cmax

def adam_simplex_optimize(a0, time_limit, T_init=0.008, T_min=0.0005,
                          lr_init=0.005, beta1=0.9, beta2=0.999, eps=1e-8,
                          verbose=True, tag=""):
    """
    Adam optimizer with simplex projection for minimizing 2*n*max(conv(a,a)).
    Uses soft-max smoothing with temperature annealing.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    lr = lr_init
    T = T_init
    t = 0
    t0 = time.monotonic()
    no_improve = 0
    total_iters = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        total_iters += 1

        # Compute gradient
        conv_vals = fast_convolve(a)
        cmax = conv_vals.max()
        p = np.exp((conv_vals - cmax) / T)
        p_sum = p.sum()
        p = p / p_sum

        # Gradient: grad[m] = (2/T) * sum_j p[m+j] * a[j]
        idx = np.arange(n)[:, None] + np.arange(n)[None, :]
        H = p[idx]
        grad = (2.0 / T) * (H @ a)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = simplex_project(a_new)

        # Evaluate
        conv_new = fast_convolve(a_new)
        score = 2.0 * n * conv_new.max()

        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Temperature annealing
        frac = min(1.0, t / max(1, int(time_limit / 0.02)))
        T = T_init * (T_min / T_init) ** frac
        T = max(T, T_min)

        # Adaptive learning rate
        if no_improve > 200:
            lr *= 0.85
            m *= 0.5
            v *= 0.5
            no_improve = 0
        elif no_improve > 100:
            lr *= 0.95
            no_improve = 0

        # Restart if stuck for too long
        if no_improve > 500:
            noise = np.random.default_rng(t * 17 + 3).standard_normal(n) * 0.015
            a = simplex_project(best_a + noise)
            m = np.zeros(n)
            v = np.zeros(n)
            lr = lr_init * 0.7
            T = T_init * 0.5
            no_improve = 0
            if verbose:
                print(f"    [{tag}] Restart at t={t}, best={best_score:.6f}")

        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    [{tag}] Adam t={t} ({elapsed:.0f}s): score={best_score:.6f} lr={lr:.5f} T={T:.5f}")

    return best_a, best_score

def coord_descent_polish(a0, time_limit, n_sweeps=500):
    """Coordinate descent: try scaling each coordinate to reduce the peak."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(99)

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        perm = rng.permutation(n)
        for i in perm:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[i]
            if old_val < 1e-15:
                continue
            for factor in [0.7, 0.85, 0.93, 1.07, 1.15, 1.4]:
                a[i] = old_val * factor
                a = np.maximum(a, 0)
                s = a.sum()
                if s < 1e-15:
                    a[i] = old_val
                    continue
                a_norm = a / s
                score = 2.0 * n * fast_convolve(a_norm).max()
                if score < best_score - 1e-13:
                    best_score = score
                    best_a = a_norm.copy()
                    a = a_norm.copy()
                    old_val = a[i]
                    improved = True
                else:
                    a[i] = old_val
                    a = a / a.sum()
            # Small perturbations
            if not improved and old_val > 1e-8:
                for factor in [0.97, 0.985, 1.015, 1.03]:
                    a[i] = old_val * factor
                    a = np.maximum(a, 0)
                    s = a.sum()
                    if s < 1e-15:
                        a[i] = old_val
                        continue
                    a_norm = a / s
                    score = 2.0 * n * fast_convolve(a_norm).max()
                    if score < best_score - 1e-14:
                        best_score = score
                        best_a = a_norm.copy()
                        a = a_norm.copy()
                        old_val = a[i]
                        improved = True
                    else:
                        a[i] = old_val
                        a = a / a.sum()

        if not improved and sweep > 10:
            break

    return best_a, best_score

def peak_directed_optimize(a0, time_limit):
    """Focus on reducing the highest peaks by adjusting their contributing coordinates."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    for step in range(10000):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        top_k = min(3, len(b))
        peaks = np.argpartition(b, -top_k)[-top_k:]

        # For each peak, try reducing its contribution
        improved_any = False
        for pk in peaks:
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[pk - idx]
            if contribs.max() < 1e-15:
                continue

            # Try reducing the top contributors
            top_c = np.argsort(contribs)[::-1][:3]
            for ci in top_c:
                i = int(idx[ci])
                j = pk - i
                if a[i] < 1e-14:
                    continue
                for factor in [0.9, 0.8, 0.7]:
                    a_trial = a.copy()
                    a_trial[i] *= factor
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        break
                    a_trial = a_trial / st
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved_any = True
                        break
                if improved_any:
                    break

        if not improved_any:
            # Try small random perturbations on peak contributors
            pk = int(peaks[0])
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[pk - idx]
            top_c = np.argsort(contribs)[::-1][:5]
            ci = top_c[rng.integers(len(top_c))]
            i = int(idx[ci])
            a_trial = a.copy()
            a_trial[i] *= (1.0 + rng.standard_normal() * 0.05)
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 1e-15:
                a_trial = a_trial / st
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()

    return best_a, best_score

def nesterov_subgrad(a0, time_limit, top_k=20, lr_init=0.006):
    """Nesterov-accelerated projected subgradient on the max peak."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    v = np.zeros(n)
    lr = lr_init
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        t_k = (1 + np.sqrt(1 + 4 * (it - 1) ** 2)) / 2
        beta_k = (t_k - 1) / t_k

        # Compute subgradient of max(conv(y,y))
        b = fast_convolve(y)
        k_pk = min(top_k, len(b))
        peaks = np.argpartition(b, -k_pk)[-k_pk:]
        pv = b[peaks]
        temp = 0.005 * max(pv.max(), 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        w /= w.sum()

        sg = np.zeros(n)
        for x in range(len(peaks)):
            if w[x] < 1e-14:
                continue
            k = int(peaks[x])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            sg[j] += 2.0 * w[x] * y[k - j]

        v = beta_k * v + sg
        nv = np.linalg.norm(v)
        if nv > 1e-15:
            v = v / nv

        y_new = y - lr * v
        y_proj = simplex_project(y_new)

        score = 2.0 * n * fast_convolve(y_proj).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = y_proj.copy()
            no_improve = 0
        else:
            no_improve += 1

        y = y_proj

        if no_improve > 200:
            lr *= 0.8
            no_improve = 0
        if no_improve > 600:
            noise = np.random.default_rng(it * 11 + 5).standard_normal(n) * 0.01
            y = simplex_project(best_a + noise)
            v = np.zeros(n)
            lr = max(lr_init * 0.5, lr * 2)
            no_improve = 0

        lr *= 0.9995

    return best_a, best_score

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def init_fourier(n, rng):
    """Initialize using random Fourier superposition."""
    n_h = int(rng.integers(4, 20))
    amps = rng.random(n_h) * 0.4
    phases = rng.uniform(0, 2 * np.pi, n_h)
    freqs = np.arange(1, n_h + 1)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_h):
        a += amps[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def init_variation(n, base, rng, sigma=0.08):
    """Add noise to base."""
    a = base.copy() + rng.standard_normal(n) * sigma
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

                best_seq, best_score = check_and_update(a_alt2, best_seq, best_score, f"alt_cd_{alt_n}")

    # === Phase 6: Final intensive polish ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            t_fin = min(rem * 0.3, 40.0)
            a_fin, _ = coord_descent_polish(best_seq, t_fin, n_sweeps=300)
            best_seq, best_score = check_and_update(a_fin, best_seq, best_score, "final_cd")

        rem = deadline - time.monotonic() - 15
        if rem > 10:
            a_fin2, _ = peak_directed_optimize(best_seq, min(rem * 0.4, 25.0))
            best_seq, best_score = check_and_update(a_fin2, best_seq, best_score, "final_peak")

        rem = deadline - time.monotonic() - 10
        if rem > 8:
            a_fin3, _ = adam_simplex_optimize(
                best_seq, min(rem * 0.5, 15.0),
                T_init=0.001, T_min=0.0005,
                lr_init=0.002, verbose=False, tag="final_adam"
            )
            best_seq, best_score = check_and_update(a_fin3, best_seq, best_score, "final_adam")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
