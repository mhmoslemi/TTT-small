This is essentially a von Neumann minimax / zero-sum game approach. The sequence a is trying to minimize its peak, while b is trying to "challenge" a by finding a direction that would increase the peak.

A key improvement: instead of using the current peak M as the constraint, we could use a target peak M_target < M and check if there exists a b with ||a*b||_inf <= M_target. If not, increase M_target. This gives a more meaningful update.

Actually, the LP approach as described is: given fixed a, find b that maximizes sum(b) subject to a*b <= M everywhere. The optimal M is the minimum value such that there exists a non-negative b with sum(b) = 1 and a*b <= M everywhere. This is:

M* = min { M : exists b >= 0, sum(b) = 1, T(a) @ b <= M * 1 }

where T(a) is the convolution matrix. This is a linear program, and the optimal M* is the "best response" value.

If M* < current peak, then replacing a with b (or blending) should reduce the peak.

The function f(a) = max_k (a*a)[k] is convex (as a function of a, since each (a*a)[k] is... wait, is it convex? (a*a)[k] = sum_i a[i]*a[k-i] is a quadratic form. It's not necessarily convex or concave in a.)

Actually, (a*a)[k] = a^T T_k a where T_k is a symmetric matrix (the k-th shift). This quadratic form is positive semi-definite (since it's a sum of products of non-negative terms... wait, no. For general a, a^T T_k a can be negative. But for a >= 0, all terms a[i]*a[k-i] >= 0, so (a*a)[k] >= 0.)

The Hessian of (a*a)[k] with respect to a is 2*T_k (where T_k is the convolution matrix for lag k). This is not positive semi-definite in general, so the function is not convex.

This means we can't use simple convex optimization. We need heuristics.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-stage soft-max L-BFGS with geometric cooling
2. Active-set subgradient descent with backtracking
3. LP-based best response (von Neumann minimax)

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize as scipy_minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

def evaluate_score(a):
    """Compute the score for a normalized (sum=1) non-negative sequence."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()

def evaluate_safe(a):
    """Evaluate sequence using the actual evaluation function."""
    try:
        a_list = [float(max(0, min(1000, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)

def lbgfs_soft_max(a0, time_limit, T_start=200.0, T_end=1.0, n_stages=8, maxiter_per_stage=3000):
    """L-BFGS refinement with soft-max smoothing and geometric temperature."""
    n = len(a0)
    z = z_from_a(a0)
    t0 = time.monotonic()
    
    def func_and_jac(T_val):
        def func(z):
            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        
        def jac(z):
            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            n_b = len(b)
            grad_a = np.zeros(n)
            for k in range(n_b):
                if w[k] < 1e-15:
                    continue
                wt = w[k]
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * wt * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        
        return func, jac

    temps = np.geomspace(T_start, T_end, n_stages)
    
    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        
        func, jac = func_and_jac(stage_T)
        maxiter = min(maxiter_per_stage, int(remaining * 100))
        
        try:
            res = scipy_minimize(
                fun=func,
                x0=z,
                jac=jac,
                method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12,
                         'maxfun': maxiter * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
        
        a_current = softmax_from_z(z)
        sc = evaluate_score(a_current)
        elapsed = time.monotonic() - t0
        print(f"    L-BFGS T={stage_T:.1f} ({elapsed:.0f}s): score={sc:.6f}")
    
    return softmax_from_z(z)

def active_set_gradient_descent(a0, time_limit, lr_init=0.01, n_steps=100000):
    """
    Gradient descent using the active set (subgradient of max).
    Works in log-domain to maintain non-negativity.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a0
    a = a / s
    best_a = a.copy()
    best_score = evaluate_score(a)
    z = z_from_a(a)
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        
        # Active set: top 5% of convolution values
        threshold = max_b - 0.05 * max_b
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        
        # Subgradient of max at the active set
        # d/d a[i] of (a*a)[k] = sum over valid j: a[k-i] (symmetrized)
        # For the max, subgradient = average over active set
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)
        
        # Update in log-domain
        z_new = z - lr * g
        a_new = softmax_from_z(z_new)
        
        score_new = evaluate_score(a_new)
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            z = z_new
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive learning rate
        if no_improve > 200:
            lr *= 0.7
            no_improve = 0
        if no_improve > 1000:
            lr *= 0.5
            no_improve = 500
        lr = max(lr, 1e-8)
        
        if step % 5000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    ActiveSet {elapsed:.0f}s: score={best_score:.6f} lr={lr:.2e}")
    
    return best_a

def lp_best_response(a_fixed, solver=None):
    """Solve LP: minimize M subject to conv(a_fixed, b) <= M, b >= 0, sum(b) = 1."""
    if solver is None:
        solver = _find_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s

_lp_solver = None

def _find_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return _lp_solver
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver

def nesterov_lp(a0, time_limit, solver=None, n_iters=2000):
    """Nesterov-accelerated LP-based iteration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    a = a / s
    y = a.copy()  # virtual iterate
    best_a = a.copy()
    best_score = evaluate_score(a)
    t0 = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new = lp_best_response(y, solver)
        if b_new is None:
            continue
        
        alpha = 2.0 / (it + 2)
        
        # Nesterov update
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy
        
        # Weighted average for actual iterate
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        score_a = evaluate_score(a)
        score_y = evaluate_score(y)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a

def peak_shave_anneal(a0, time_limit, T_start=0.01, T_end=0.0001, n_steps=50000):
    """
    Peak shaving with simulated annealing.
    Tries to reduce the peak by adjusting coordinates.
    """
    n = len(a0)
    rng = np.random.default_rng(12345)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = evaluate_score(a)
    t0 = time.monotonic()
    
    n_total_steps = n_steps
    for step in range(n_total_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Temperature schedule
        frac = step / n_total_steps
        T = T_start * (T_end / T_start) ** frac
        
        b = fast_convolve(a)
        max_b = b.max()
        peak_k = int(np.argmax(b))
        
        # Find top contributors to peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        # Strategy: reduce top contributor, redistribute to random other position
        top_ci = np.argmax(contribs)
        i = idx[top_ci]
        
        # Also pick a random position to receive the mass
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)
        
        if a[i] < 1e-14:
            continue
        
        # Reduce a[i] by small fraction, add to a[j]
        delta = a[i] * rng.uniform(0.01, 0.1)
        a_trial = a.copy()
        a_trial[i] -= delta
        a_trial[j] += delta
        if a_trial[i] < 0:
            continue
        a_trial = a_trial / a_trial.sum()
        
        score_trial = evaluate_score(a_trial)
        delta_score = score_trial - evaluate_score(a)
        
        if delta_score < 0 or rng.random() < np.exp(-delta_score / max(T, 1e-10)):
            a = a_trial
            if score_trial < best_score:
                best_score = score_trial
                best_a = a.copy()
        
        if step % 10000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakAnneal {elapsed:.0f}s: score={best_score:.6f} T={T:.6f}")
    
    return best_a

def pairwise_peak_shave(a0, time_limit, n_steps=20000):

    n = len(a0)

        b_copy[0] = -1
        b_copy[-1] = -1
        top_peaks = np.argsort(b_copy)[::-1][:3]
        
        improved = False
        for peak_k in top_peaks:
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_ci = np.argsort(contribs)[::-1][:3]
            
            for ci in top_ci:
                i = int(idx[ci])
                if a[i] < 1e-14:
                    continue
                # Try reducing a[i] and spreading to random positions
                for f in [0.95, 0.9, 0.85]:
                    a_trial = a.copy()
                    a_trial[i] *= f
                    diff = a[i] - a_trial[i]

                    a_trial[targets] += per
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    sc = evaluate_score(a_trial)
                    if sc < best_score - 1e-12:
                        a = a_trial.copy()
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved and step % 5000 == 0:
            # Try a random perturbation
            idx = rng.integers(0, n)
            a_trial = a.copy()
            a_trial[idx] *= rng.uniform(0.7, 1.3)

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    n = len(a)
    size = 2 * n - 1

    try:
        a_list = [float(max(0, min(1000, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    a_safe = np.maximum(a, 1e-300)
    return np.log(a_safe)


def _lp_solver_global():
    global _cached_lp_solver
    if '_cached_lp_solver' not in globals():
        for sn in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            try:
                x = cp.Variable(3, nonneg=True)
                prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
                prob.solve(solver=sn)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    _cached_lp_solver = sn
                    return _cached_lp_solver
            except Exception:
                continue
        _cached_lp_solver = cp.ECOS
    return _cached_lp_solver


def lbgfs_softmax(a0, time_limit, T_start_mult=60.0, T_end_mult=3.0,
                  n_stages=10, maxiter_per_stage=2500):
    """
    L-BFGS with soft-max smoothing. T scaled proportionally to n.
    For n=768: T ranges from ~46000 to ~2300, giving T*b_max ~ O(1-10).
    """
    n = len(a0)
    z = z_from_a(a0)
    t0 = time.monotonic()

    T_start = T_start_mult * n
    T_end = T_end_mult * n
    temps = np.geomspace(T_start, T_end, n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 15:
            break

        def _func(z_vec):
            a_cur = softmax_from_z(z_vec)
            b_cur = fast_convolve(a_cur)
            Tb = stage_T * b_cur
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / stage_T) * log_sum_exp

        def _jac(z_vec):
            a_cur = softmax_from_z(z_vec)
            b_cur = fast_convolve(a_cur)
            Tb = stage_T * b_cur
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b_cur)):
                if w[k] < 1e-14:
                    continue
                wt = 2.0 * w[k]
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += wt * a_cur[k - idx]
            S = np.dot(grad_a, a_cur)
            return a_cur * (grad_a - S)

        maxiter = min(maxiter_per_stage, max(200, int(remaining * 80)))
        try:
            res = scipy_minimize(
                fun=_func, x0=z, jac=_jac, method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12,
                         'maxfun': maxiter * 80, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

        a_cur = softmax_from_z(z)
        sc = evaluate_score(a_cur)
        elapsed = time.monotonic() - t0
        print(f"    LBF T={stage_T:.0f} ({elapsed:.0f}s): score={sc:.7f}")

    return softmax_from_z(z)


def active_set_descend(a0, time_limit, lr_init=0.003, n_steps=100000,
                       top_frac=0.03):
    """
    Subgradient descent using active set (top fraction of convolution peaks).
    Works in log-domain to maintain non-negativity.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a0
    a = a / s
    best_a = a.copy()
    best_score = evaluate_score(a)
    z = z_from_a(a)
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Active set: top fraction of values
        n_active = max(1, int(len(b) * top_frac))
        active = np.argsort(b)[::-1][:n_active]

        # Subgradient
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        # Update in log-domain
        z_new = z - lr * g
        a_new = softmax_from_z(z_new)
        score_new = evaluate_score(a_new)

        if score_new < best_score - 1e-13:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            z = z_new
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive LR
        if no_improve > 300:
            lr *= 0.75
            no_improve = 0
        if no_improve > 1500:
            lr *= 0.5
            no_improve = 750
        lr = max(lr, 1e-10)

        if step % 20000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    ActSet {elapsed:.0f}s: score={best_score:.7f} lr={lr:.2e}")

    return best_a


def lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. conv(a_fixed,b) <= M, b >= 0, sum(b)=1."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s


def nesterov_lp_refine(a0, time_limit, solver, n_iters=1000):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = evaluate_score(a)
    t0 = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new = lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        sc_a = evaluate_score(a)
        sc_y = evaluate_score(y)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        if (it + 1) % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLp it={it+1} ({elapsed:.0f}s): score={best_score:.7f}")

    return best_a


def peak_shave_refine(a0, time_limit, seed=99):
    """Aggressive peak shaving: reduce top contributors, redistribute mass."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = evaluate_score(a)
    t0 = time.monotonic()
    step = 0

    while time.monotonic() - t0 < time_limit:
        step += 1
        b = fast_convolve(a)
        # Find peaks excluding trivial endpoints
        b_work = b.copy()
        b_work[0] = -1
        b_work[-1] = -1
        peak_k = int(np.argmax(b_work))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_ci = np.argsort(contribs)[::-1][:4]

        improved_this_iter = False
        for ci in top_ci:
            i = int(idx[ci])
            if a[i] < 1e-14:
                continue
            for f in [0.97, 0.94, 0.9]:
                a_trial = a.copy()
                a_trial[i] *= f
                diff = a[i] - a_trial[i]
                # Redistribute to random positions (away from peak)
                n_redist = 3
                targets = rng.choice(n, size=n_redist, replace=False)
                per = diff / n_redist
                a_trial[targets] += per
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = evaluate_score(a_trial)
                if sc < best_score - 1e-13:
                    a = a_trial.copy()
                    best_score = sc
                    best_a = a.copy()
                    improved_this_iter = True
                    break
            if improved_this_iter:
                break

        # Also try reducing two coordinates simultaneously
        if not improved_this_iter and len(top_ci) >= 2:
            i1 = int(idx[top_ci[0]])
            i2 = int(idx[top_ci[1]])
            if i1 != i2 and a[i1] > 1e-14 and a[i2] > 1e-14:
                for f1 in [0.95, 0.9]:
                    for f2 in [0.95, 0.9]:
                        a_trial = a.copy()
                        diff1 = a_trial[i1] * (1 - f1)
                        diff2 = a_trial[i2] * (1 - f2)
                        a_trial[i1] *= f1
                        a_trial[i2] *= f2
                        targets = rng.choice(n, size=2, replace=False)
                        a_trial[targets[0]] += diff1
                        a_trial[targets[1]] += diff2
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        sc = evaluate_score(a_trial)
                        if sc < best_score - 1e-13:
                            a = a_trial.copy()
                            best_score = sc
                            best_a = a.copy()
                            improved_this_iter = True
                            break
                    if improved_this_iter:
                        break

        # Random perturbation every 5000 steps
        if not improved_this_iter and step % 5000 == 0:
            for _ in range(5):
                i = rng.integers(0, n)
                a_trial = a.copy()
                a_trial[i] *= rng.uniform(0.5, 1.5)
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss > 1e-10:
                    a_trial = a_trial / ss
                    sc = evaluate_score(a_trial)
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        break

        if step % 10000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave {elapsed:.0f}s: score={best_score:.7f}")

    return best_a


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Uses properly-scaled soft-max L-BFGS, active-set descent, LP, and peak shaving.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    solver = _lp_solver_global()
    print(f"LP solver: {solver}")

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.7f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n=768, score={best_score:.7f}")

    n_best = len(best_seq)

    def do_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.7f} (n={len(a)})")

    # === Phase 1: Main L-BFGS with properly scaled T (38% time) ===
    print(f"\n=== Phase 1: L-BFGS Main (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.38, 580.0)
    if t1 > 30:
        print(f"  n={n_best}, budget={t1:.0f}s, T: {60*n_best:.0f}->{3*n_best:.0f}")
        a_res = lbgfs_softmax(best_seq, t1, T_start_mult=60.0, T_end_mult=3.0,
                              n_stages=10, maxiter_per_stage=2500)
        do_update(a_res, "lbfgs_main")

    # === Phase 2: Active-set gradient (18% time) ===
    print(f"\n=== Phase 2: ActiveSet (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.18, 300.0)
    if t2 > 20:
        print(f"  n={len(best_seq)}, budget={t2:.0f}s")
        a_res = active_set_descend(best_seq, t2, lr_init=0.002, n_steps=80000)
        do_update(a_res, "active_set")

    # === Phase 3: Nesterov LP (15% time) ===
    print(f"\n=== Phase 3: NesterovLP (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.15, 250.0)
    if t3 > 20:
        print(f"  n={len(best_seq)}, budget={t3:.0f}s")
        a_res = nesterov_lp_refine(best_seq, t3, solver, n_iters=800)
        do_update(a_res, "nesterov_lp")

    # === Phase 4: Low-T L-BFGS refinement (13% time) ===
    print(f"\n=== Phase 4: L-BFGS LowT (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.13, 220.0)
    if t4 > 20:
        n_cur = len(best_seq)
        print(f"  n={n_cur}, budget={t4:.0f}s, T: {10*n_cur:.0f}->{1*n_cur:.0f}")
        a_res = lbgfs_softmax(best_seq, t4, T_start_mult=10.0, T_end_mult=1.0,
                              n_stages=6, maxiter_per_stage=3000)
        do_update(a_res, "lbfgs_lowT")

    # === Phase 5: Peak shaving (8% time) ===
    print(f"\n=== Phase 5: PeakShave (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.08, 140.0)
    if t5 > 15:
        print(f"  n={len(best_seq)}, budget={t5:.0f}s")
        a_res = peak_shave_refine(best_seq, t5, seed=123)
        do_update(a_res, "peak_shave")

    # === Phase 6: Alternative lengths (5% time) ===
    print(f"\n=== Phase 6: Alt Lengths (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    n_cur = len(best_seq)
    alt_lengths = [int(n_cur * 1.4), int(n_cur * 0.7), int(n_cur * 1.2)]
    alt_lengths = sorted(set([x for x in alt_lengths if 256 <= x <= 3000 and x != n_cur]))
    for alt_n in alt_lengths[:2]:
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t_alt = min(rem * 0.45, 90.0)
        if t_alt < 20:
            break
        print(f"  Alt n={alt_n}, budget={t_alt:.0f}s")
        init_alt = interpolate_seq(best_seq, alt_n)
        a_alt = lbgfs_softmax(init_alt, t_alt, T_start_mult=30.0, T_end_mult=2.0,
                              n_stages=6, maxiter_per_stage=2000)
        do_update(a_alt, f"alt_{alt_n}")

        rem = deadline - time.monotonic()
        if rem > 25 and best_seq is not None and len(best_seq) == alt_n:
            t_lp = min(rem * 0.4, 50.0)
            if t_lp > 15:
                a_lp = nesterov_lp_refine(best_seq, t_lp, solver, n_iters=300)
                do_update(a_lp, f"lp_alt_{alt_n}")

    # === Phase 7: Final polish (remaining time) ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        n_cur = len(best_seq)
        t7 = min(rem * 0.5, 60.0)
        if t7 > 15:
            print(f"  Final L-BFGS n={n_cur}, budget={t7:.0f}s")
            a_res = lbgfs_softmax(best_seq, t7, T_start_mult=5.0, T_end_mult=0.5,
                                  n_stages=4, maxiter_per_stage=4000)
            do_update(a_res, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 15:
        t8 = min(rem * 0.5, 30.0)
        if t8 > 10:
            print(f"  Final PeakShave budget={t8:.0f}s")
            a_res = peak_shave_refine(best_seq, t8, seed=456)
            do_update(a_res, "final_ps")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.7f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
