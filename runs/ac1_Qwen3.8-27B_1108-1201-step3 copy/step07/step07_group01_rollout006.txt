    n = len(f)
    m = len(g)
    size = n + m - 1

    F = rfft(f, fft_size)
    G = rfft(g, fft_size)

    k ranges over 0..2n-2, i ranges over 0..n-1

def build_conv_matrix(f):

def mirror_descent_refine(f_init, time_budget, T_start=0.005, T_end=0.0003,
                          eta_start=0.01, eta_end=0.0002, max_iters=50000,
                          verbose=True, label=""):
    """Mirror descent with temperature annealing for refinement."""
    n = len(f_init)
    a = np.maximum(f_init, 1e-15).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = score_from_a(a)
    t0 = time.monotonic()
    stuck_count = 0

    for it in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_budget:
            break
        progress = min(1.0, it / (0.7 * max_iters))
        T = T_start * (T_end / T_start) ** progress
        eta = eta_start * (eta_end / eta_start) ** progress

        b = fast_convolve(a)
        current_score = 2.0 * n * b.max()
        if current_score < best_score - 1e-14:
            best_score = current_score
            best_a = a.copy()
            stuck_count = 0
        else:
            stuck_count += 1

        # Softmax weights
        scaled = b / T
        m = scaled.max()
        exp_s = np.exp(scaled - m)
        w = exp_s / exp_s.sum()

        # Gradient via FFT cross-correlation
        fft_size = 1 << (3 * n - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        correl = irfft(np.conj(W) * A, fft_size)
        grad_a = 4.0 * n * correl[:n]

        # Mirror update
        log_update = -eta * grad_a
        log_update = np.clip(log_update, -3.0, 3.0)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-300)
        ss = a.sum()
        if ss < 1e-15:
            a = best_a.copy()
        else:
            a = a / ss

        if stuck_count > 200:
            rng = np.random.default_rng(it * 104729)
            noise = rng.standard_normal(n) * 0.01
            a = best_a + noise
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            stuck_count = 0

            elapsed = time.monotonic() - t0
            print(f"    {label} MD iter={it} ({elapsed:.0f}s): "

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    6. Final combined polish (~5% time)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.7f}")
    except Exception as e:
        print(f"Warning: could not load parent: {e}")

    if parent is not None:
        best_seq = parent.copy()
        best_score = p_score

    if best_seq is None:
        n = 768
        best_seq = np.ones(n) / n
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Using uniform init: n={n}, score={best_score:.7f}")

    # === Phase 1: LP-based optimization (primary) ===
    print(f"\n=== Phase 1: LP Optimization (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic() - 300
    if rem > 30:
        t_alloc = min(rem * 0.5, 1200.0)
        print(f"  LP main: n={len(best_seq)}, budget={t_alloc:.0f}s")
        f_lp, lp_score, n_iters = lp_main_loop(best_seq, t_alloc, verbose=True,
                                                label="LP_main")
        best_seq, best_score = evaluate_and_update(f_lp, f"LP_main({n_iters}it)",
                                                    best_seq, best_score)
        print(f"  After LP: best={best_score:.7f}")

    # === Phase 2: Mirror Descent refinement ===
    print(f"\n=== Phase 2: Mirror Descent (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 20:
            t_alloc = min(rem * 0.3, 200.0)
            print(f"  MD: n={len(best_seq)}, budget={t_alloc:.0f}s")
            f_md, _ = mirror_descent_refine(best_seq, t_alloc,
                                            T_start=0.003, T_end=0.0002,
                                            eta_start=0.008, eta_end=0.0001,
                                            label="MD")
            best_seq, best_score = evaluate_and_update(f_md, "MD", best_seq, best_score)

            # Second MD with different params
            rem = deadline - time.monotonic() - 150
            if rem > 15:
                t_alloc = min(rem * 0.3, 120.0)
                f_md2, _ = mirror_descent_refine(best_seq, t_alloc,
                                                 T_start=0.001, T_end=0.0001,
                                                 eta_start=0.005, eta_end=0.0001,
                                                 label="MD2")
                best_seq, best_score = evaluate_and_update(f_md2, "MD2",
                                                           best_seq, best_score)

    # === Phase 3: Peak Shaving ===
    print(f"\n=== Phase 3: Peak Shaving (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 120
        if rem > 15:
            t_alloc = min(rem * 0.4, 120.0)
            print(f"  PS: budget={t_alloc:.0f}s")
            f_ps, _ = peak_shave_refine(best_seq, t_alloc, label="PS")
            best_seq, best_score = evaluate_and_update(f_ps, "peak_shave",
                                                       best_seq, best_score)

    # === Phase 4: Coordinate Descent ===
    print(f"\n=== Phase 4: Coordinate Descent (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 10:
            t_alloc = min(rem * 0.5, 100.0)
            print(f"  CD: budget={t_alloc:.0f}s")
            f_cd, _ = coordinate_descent_refine(best_seq, t_alloc, label="CD")
            best_seq, best_score = evaluate_and_update(f_cd, "coord_descent",
                                                       best_seq, best_score)

    # === Phase 5: LP again from current best ===
    print(f"\n=== Phase 5: LP Refinement (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t_alloc = min(rem * 0.5, 100.0)
            print(f"  LP2: budget={t_alloc:.0f}s")
            f_lp2, _, n_it2 = lp_main_loop(best_seq, t_alloc, verbose=False,
                                           label="LP2")
            best_seq, best_score = evaluate_and_update(f_lp2, f"LP2({n_it2}it)",
                                                       best_seq, best_score)

    # === Phase 6: Alternative lengths ===
    print(f"\n=== Phase 6: Alternative Lengths (best={best_score:.7f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        alt_lengths = []
        for factor in [1.25, 1.5, 0.8, 2.0, 0.75]:
            alt_n = int(n_best * factor)
            if 128 <= alt_n <= 4096 and alt_n != n_best and alt_n not in alt_lengths:
                alt_lengths.append(alt_n)

        for alt_n in alt_lengths[:3]:
            rem = deadline - time.monotonic() - 40
            if rem < 30:
                break
            t_alloc = min(rem * 0.4, 60.0)
            init_alt = interpolate_sequence(best_seq, alt_n)
            f_alt, _, _ = lp_main_loop(init_alt, t_alloc, verbose=False,
                                       label=f"alt{alt_n}")
            best_seq, best_score = evaluate_and_update(f_alt, f"alt_n{alt_n}",
                                                       best_seq, best_score)
            if len(best_seq) != alt_n:
                continue
            # Quick MD polish on alt length
            rem = deadline - time.monotonic() - 30
            if rem > 10:
                f_alt_md, _ = mirror_descent_refine(best_seq, min(rem * 0.5, 30.0),
                                                    T_start=0.002, T_end=0.0002,
                                                    eta_start=0.005, eta_end=0.0001,
                                                    verbose=False, label=f"alt{alt_n}_md")
                best_seq, best_score = evaluate_and_update(f_alt_md,
                                                           f"alt_n{alt_n}_md",
                                                           best_seq, best_score)

    # === Phase 7: Final combined polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 8:
            t_ps = min(rem * 0.4, 15.0)
            f_fin, _ = peak_shave_refine(best_seq, t_ps, verbose=False, label="finPS")
            best_seq, best_score = evaluate_and_update(f_fin, "final_PS",
                                                       best_seq, best_score)

        rem = deadline - time.monotonic() - 5
        if rem > 5:
            t_cd = min(rem * 0.5, 10.0)
            f_fin2, _ = coordinate_descent_refine(best_seq, t_cd, verbose=False,
                                                  label="finCD")
            best_seq, best_score = evaluate_and_update(f_fin2, "final_CD",
                                                       best_seq, best_score)

        rem = deadline - time.monotonic() - 3
        if rem > 3:
            t_md = min(rem * 0.6, 8.0)
            f_fin3, _ = mirror_descent_refine(best_seq, t_md,
                                              T_start=0.001, T_end=0.0001,
                                              eta_start=0.003, eta_end=0.00005,
                                              verbose=False, label="finMD")
            best_seq, best_score = evaluate_and_update(f_fin3, "final_MD",
                                                       best_seq, best_score)

    # Fallback
    if best_seq is None:
        if parent is not None:
            best_seq = parent
        else:
            best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    total_time = time.monotonic() - t_start
    print(f"\nFINAL SCORE: {best_score:.7f} (n={len(best_seq)}, time={total_time:.1f}s)")
    return [float(x) for x in best_seq]
