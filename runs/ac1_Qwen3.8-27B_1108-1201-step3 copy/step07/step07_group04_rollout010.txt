def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    rho_final = np.nonzero(cond)[0][-1] if cond.any() else n - 1
    theta = (cssv[rho_final] - 1.0) / (rho_final + 1.0)
    return np.maximum(v - theta, 0.0)

"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize
from time import monotonic
import cvxpy as cp

_lp_solver = None


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return cp.ECOS


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if v.sum() <= 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient_fft(a, top_k=None, temperature=None):
    """Compute subgradient of max_k (a*a)[k] using FFT-accelerated approach."""
    n = len(a)
    b = fast_convolve(a)
    m = len(b)

    if top_k is not None:
        top_k = min(top_k, m)
        peaks = np.argpartition(b, -top_k)[-top_k:]
    else:
        peaks = np.arange(m)

    pv = b[peaks]
    if temperature is not None and temperature > 0:
        w = np.exp((pv - pv.max()) / temperature)
        w = w / w.sum()
    else:
        max_val = pv.max()
        active = pv >= max_val - 1e-10
        w = np.zeros(len(peaks))
        w[active] = 1.0 / active.sum()

    # Build weight vector W of length 2n-1
    W = np.zeros(m)
    for i in range(len(peaks)):
        W[int(peaks[i])] = w[i]

    # Gradient: grad[i] = 2 * sum_{k} W[k] * a[k-i] for valid k-i
    # = 2 * cross_correlation(W, a) evaluated appropriately
    # Using FFT: cross-correlation at lag i = sum_m W[m+i] * a[m]
    # This is ifft(fft(W) * conj(fft(a))) with appropriate padding

    fft_size = 1 << (m + n - 1 - 1).bit_length()
    W_pad = np.zeros(fft_size)
    W_pad[:m] = W
    a_pad = np.zeros(fft_size)
    a_pad[:n] = a

    # Cross-correlation: (W * a_flipped)[i] gives sum_m W[m] * a[m-i]
    # We want sum_m W[m+i] * a[m] = sum_j W[j] * a[j-i] where j=m+i
    # This is the cross-correlation of W and a at lag -i (or W and flip(a) at lag i)

    a_flip = np.zeros(fft_size)
    a_flip[:n] = a[::-1]
    corr = np.real(ifft(fft(W_pad) * fft(a_flip)))
    # corr[i] should give sum_j W[j] * a[i-j]... let me just use direct for small top_k

    # Direct computation is faster for small top_k
    if top_k is not None and top_k < 200:
        grad = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * w[i] * a[k - idx]
    else:
        # Use FFT-based
        # grad[i] = 2 * sum_{j=0}^{n-1} W[i+j] * a[j]
        # = 2 * sum_{j=0}^{n-1} W[i+j] * a[j]
        # This is like a "cross-correlation" but with a non-standard definition
        # Let's use direct for now (n=768 is manageable)
        grad = np.zeros(n)
        for k_val in range(m):
            if W[k_val] < 1e-12:
                continue
            js = max(0, k_val - n + 1)
            je = min(n - 1, k_val)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * W[k_val] * a[k_val - idx]

    return grad, b, peaks, w


def solve_lp_best_response(a_fixed, solver=None):
    """LP: find b >= 0, sum(b)=1 minimizing max_k (a*b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    # Build T as (2n-1) x n matrix where T[k,j] = a[k-j] if valid
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed

    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.PDLP, cp.SCS, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    M_val = float(M_var.value) if M_var.value is not None else fast_convolve(a_fixed).max()
    return b_val, M_val


def fictitious_play(n, time_limit, initial=None, solver=None, verbose=True):
    """Run fictitious play for auto-correlation minimization."""
    t0 = monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        a = a / s if s > 1e-15 else np.ones(n) / n

    b = a.copy()
    a_avg = a.copy().astype(np.float64)
    b_avg = b.copy().astype(np.float64)
    best_score = float('inf')
    best_a = a.copy()
    iters = 0
    stagnation = 0

    while monotonic() - t0 < time_limit:
        iters += 1

        b_new, _ = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new

        a_new, _ = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new

        a_avg = (iters * a_avg + a) / (iters + 1)
        b_avg = (iters * b_avg + b) / (iters + 1)

        candidates = [a, b, 0.5 * (a + b), a_avg, 0.5 * (a_avg + b_avg),
                      0.7 * a_avg + 0.3 * a, 0.3 * a_avg + 0.7 * a]
        local_best = None
        local_score = float('inf')
        for cand in candidates:
            cand = np.maximum(cand, 0)
            s = cand.sum()
            if s < 1e-10:
                continue
            cand = cand / s
            score = 2.0 * n * fast_convolve(cand).max()
            if score < local_score:
                local_score = score
                local_best = cand.copy()

        if local_best is not None and local_score < best_score - 1e-12:
            best_score = local_score
            best_a = local_best
            stagnation = 0
        else:
            stagnation += 1

        if stagnation > 150:
            noise_scale = 0.003 * max(1.0 / n, 0.001)
            a = a + np.random.standard_normal(n) * noise_scale
            a = np.maximum(a, 0)
            a = a / a.sum()
            b = b + np.random.standard_normal(n) * noise_scale
            b = np.maximum(b, 0)
            b = b / b.sum()
            stagnation = 0

        if verbose and iters % 50 == 0:
            print(f"    FP iter={iters} ({monotonic()-t0:.0f}s): best={best_score:.6f}")

    return best_a, best_score, iters


def subgradient_descent(a0, time_limit, n_top=50, lr_init=0.02):
    """Projected subgradient descent with adaptive step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    lr = lr_init
    v = np.zeros(n)  # momentum
    momentum = 0.85
    t0 = monotonic()
    no_improve = 0
    step = 0

    while monotonic() - t0 < time_limit:
        step += 1
        grad, b, peaks, w = compute_subgradient_fft(a, top_k=n_top, temperature=None)

        # Nesterov-like momentum
        v = momentum * v - lr * grad
        scale = n / (np.abs(v).sum() + 1e-15)
        v = v * scale

        a_trial = project_simplex(a - v)
        curr_score = 2.0 * n * fast_convolve(a).max()
        trial_score = 2.0 * n * fast_convolve(a_trial).max()

        # Backtracking if no improvement
        alpha = 1.0
        while trial_score > curr_score and alpha > 0.01:
            alpha *= 0.5
            a_trial = project_simplex(a - alpha * v)
            trial_score = 2.0 * n * fast_convolve(a_trial).max()

        if trial_score < best_score - 1e-14:
            best_score = trial_score
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
        else:
            no_improve += 1

        # Adapt learning rate
        lr *= 0.999
        if no_improve > 200:
            a = best_a + np.random.standard_normal(n) * 0.001
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
            lr = max(lr * 3, 0.01)
            no_improve = 0

        if step % 500 == 0:
            print(f"    SubGrad step={step} ({monotonic()-t0:.0f}s): score={best_score:.6f} lr={lr:.6f}")

    return best_a, best_score


def softmax_lbgfs(a0, time_limit, T_start=50.0, T_end=1.0, n_temps=12):
    """L-BFGS on softmax-smoothed objective with decreasing temperature."""
    n = len(a0)
    a_current = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    a_current = a_current / a_current.sum()
    best_a = a_current.copy()
    best_score = 2.0 * n * fast_convolve(a_current).max()
    t0 = monotonic()

    # Geometric temperature schedule
    temperatures = np.geomspace(T_start, T_end, n_temps)

    for T in temperatures:
        if monotonic() - t0 > time_limit:
            break
        time_per_T = (time_limit - (monotonic() - t0)) / (n_temps - list(temperatures).index(T))
        time_per_T = max(5.0, time_per_T)

        z0 = np.log(np.maximum(a_current, 1e-30))
        z0 = z0 - z0.mean()
        stopped = [False]
        t_sub = monotonic()

        def func(z):
            if monotonic() - t_sub > time_per_T:
                stopped[0] = True
                return 1e10
            z_shifted = z - z.max()
            exp_z = np.exp(z_shifted)
            a = exp_z / exp_z.sum()
            b = fast_convolve(a)
            max_b = b.max()
            scaled = (b - max_b) / T
            exp_b = np.exp(np.minimum(scaled, 50))
            lse = max_b + T * np.log(exp_b.sum())
            return 2.0 * n * lse / T

        def jac(z):
            if monotonic() - t_sub > time_per_T:
                stopped[0] = True
                return np.zeros(n)
            z_shifted = z - z.max()
            exp_z = np.exp(z_shifted)
            a = exp_z / exp_z.sum()
            b = fast_convolve(a)
            max_b = b.max()
            scaled = (b - max_b) / T
            exp_b = np.exp(np.minimum(scaled, 50))
            w = exp_b / exp_b.sum()

            # Gradient of smoothed objective wrt a
            # d/d a[i] sum_k w_k * (a*a)[k] = 2 * sum_k w_k * a[k-i]
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-10:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]

            grad_a *= 2.0 * n
            # Transform to z-space: d/dz = (I - a a^T) * a * .
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - a * S)
            return grad_z

        try:
            res = minimize(fun=func, x0=z0, jac=jac, method='L-BFGS-B',
                           options={'maxiter': 2000, 'ftol': 1e-18, 'gtol': 1e-10})
            if not stopped[0]:
                z_final = res.x
                z_shifted = z_final - z_final.max()
                a_out = np.exp(z_shifted)
                a_out = a_out / a_out.sum()
                score = 2.0 * n * fast_convolve(a_out).max()
                if score < best_score - 1e-12:
                    best_score = score
                    best_a = a_out.copy()
                else:
                    a_out = a_current
            else:
                a_out = a_current
        except Exception:
            a_out = a_current

        a_current = a_out
        print(f"    L-BFGS T={T:.2f}: score={best_score:.6f}")

    return best_a, best_score


def peak_shave_advanced(a0, time_limit, max_steps=5000):
    """Advanced peak shaving with multiple strategies including pair and triple redistribution."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    failed = 0
    steps = 0

    while monotonic() - t0 < time_limit and failed < 100:
        steps += 1
        b = fast_convolve(a)
        # Find all peaks near the maximum
        max_b = b.max()
        peak_mask = b >= max_b * 0.999
        peak_indices = np.where(peak_mask)[0]

        if len(peak_indices) == 0:
            break

        improved = False

        for pk in peak_indices:
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[pk - idx]
            total = contribs.sum()
            if total < 1e-20:
                continue

            # Strategy 1: Reduce top single contributor
            top_contrib = np.argsort(contribs)[::-1]
            for ci in top_contrib[:10]:
                i = int(idx[ci])
                old_val = a[i]
                if old_val < 1e-12:
                    continue
                for factor in [0.9, 0.85, 0.8, 0.7, 0.6]:
                    new_val = old_val * factor
                    diff = old_val - new_val
                    if diff < 1e-14:
                        continue
                    mask = np.ones(n, dtype=bool)
                    mask[i] = False
                    others = a[mask]
                    others_sum = others.sum()
                    if others_sum < 1e-14:
                        continue
                    a_trial = a.copy()
                    a_trial[i] = new_val
                    a_trial[mask] = others + diff * others / others_sum
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

            if improved:
                break

            # Strategy 2: Reduce a pair of contributors
            for ci in top_contrib[:5]:
                i = int(idx[ci])
                j = int(pk - i)
                if i == j or j < 0 or j >= n or a[j] < 1e-12:
                    continue
                for f1 in [0.8, 0.7]:
                    for f2 in [0.8, 0.7]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        # Redistribute to random positions
                        diff = (a[i] - a_trial[i]) + (a[j] - a_trial[j])
                        if diff < 1e-14:
                            continue
                        # Add to positions far from peak
                        far_mask = np.ones(n, dtype=bool)
                        for fpi in peak_indices:
                            fpi = int(fpi)
                            lo = max(0, fpi - n + 1)
                            hi = min(n - 1, fpi)
                            far_mask[lo:hi + 1] = False
                        if far_mask.sum() > 0:
                            far_positions = np.where(far_mask)[0]
                            a_trial[far_positions] += diff * a[far_positions] / (a[far_positions].sum() + 1e-15)
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-13:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

            if improved:
                break

            # Strategy 3: Random perturbation focused on reducing peak
            for attempt in range(10):
                perturbation = np.zeros(n)
                # Perturb positions that contribute to the peak
                n_perturb = max(5, n // 20)
                perturb_idx = np.random.choice(len(idx), size=min(n_perturb, len(idx)), replace=False)
                for pi in perturb_idx:
                    ii = int(idx[pi])
                    perturbation[ii] = -np.random.uniform(0.001, 0.01)
                # Add positive perturbation elsewhere
                pos_perturb = max(5, n // 20)
                pos_idx = np.random.choice(n, size=pos_perturb, replace=False)
                total_neg = -perturbation.sum() * 0.5
                for pi in pos_idx:
                    perturbation[pi] += total_neg / pos_perturb
                a_trial = a + perturbation
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

        if improved:
            failed = 0
        else:
            failed += 1

        if steps % 500 == 0:
            print(f"    PeakShave step={steps} ({monotonic()-t0:.0f}s): score={best_score:.6f}")

    return best_a, best_score, steps


def damped_lp_refine(a0, time_limit, solver=None, max_iters=3000):
    """Damped LP refinement with adaptive step size."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()

    for it in range(max_iters):
        if monotonic() - t0 > time_limit:
            break
        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is None:
            continue

        alpha = min(1.0, 3.0 / (it + 1))
        a_new = (1 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        if (it + 1) % 200 == 0:
            print(f"    DampedLP iter={it+1} ({monotonic()-t0:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def alternating_optimize(a0, time_limit, solver=None):
    """Alternating optimization with multiple damping strategies."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = monotonic()
    it = 0

    damping_schedule = [0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.08, 0.05, 0.03, 0.02, 0.01]
    di = 0

    while monotonic() - t0 < time_limit:
        it += 1
        b_new, _ = solve_lp_best_response(a, solver)
        if b_new is None:
            break

        alpha = damping_schedule[min(di, len(damping_schedule) - 1)]
        a_new = (1 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = 2.0 * n * fast_convolve(a_new).max()

        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()

        a = a_new

        # Advance damping
        if it % 100 == 0:
            di += 1

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate candidate using evaluate_sequence and update best."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. LP fictitious play for global structure
    2. Subgradient descent with momentum
    3. L-BFGS with adaptive temperature smoothing
    4. Advanced peak shaving
    5. Alternating LP with decreasing damping
    6. Final polish loop
    """
    np.random.seed(seed)
    deadline = monotonic() + max(1.0, float(budget_s) - 30.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is not None:
        best_seq = parent.copy()
        best_score = p_score
    else:
        best_seq = np.ones(768) / 768.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Initial uniform: n=768, score={best_score:.6f}")

    n_best = len(best_seq)

    # === Phase 1: LP Fictitious Play ===
    print(f"\n=== Phase 1: Fictitious Play (n={n_best}) ===")
    rem = deadline - monotonic() - 600
    if rem > 60:
        t_fp = min(rem * 0.5, 350.0)
        print(f"  FP budget={t_fp:.0f}s")
        a_fp, score_fp, iters = fictitious_play(n_best, t_fp, initial=best_seq, solver=solver, verbose=True)
        best_seq, best_score = check_and_update(a_fp, f"FP_n{n_best}", best_seq, best_score)

    # === Phase 2: Subgradient Descent ===
    print(f"\n=== Phase 2: Subgradient (best={best_score:.6f}) ===")
    rem = deadline - monotonic() - 500
    if rem > 40:
        t_sg = min(rem * 0.5, 250.0)
        print(f"  SubGrad budget={t_sg:.0f}s")
        n_top = max(30, n_best // 15)
        a_sg, _ = subgradient_descent(best_seq, t_sg, n_top=n_top, lr_init=0.015)
        best_seq, best_score = check_and_update(a_sg, "subgrad", best_seq, best_score)

    # === Phase 3: L-BFGS with adaptive temperature ===
    print(f"\n=== Phase 3: L-BFGS (best={best_score:.6f}) ===")
    rem = deadline - monotonic() - 400
    if rem > 30:
        t_lb = min(rem * 0.5, 250.0)
        print(f"  L-BFGS budget={t_lb:.0f}s")
        # Use multiple temperature ranges
        a_lb1, _ = softmax_lbgfs(best_seq, t_lb * 0.6, T_start=40.0, T_end=5.0, n_temps=8)
        best_seq, best_score = check_and_update(a_lb1, "lbgfs_warm", best_seq, best_score)

        rem = deadline - monotonic() - 300
        if rem > 20:
            a_lb2, _ = softmax_lbgfs(best_seq, min(rem * 0.6, 120.0), T_start=5.0, T_end=0.5, n_temps=8)
            best_seq, best_score = check_and_update(a_lb2, "lbgfs_cold", best_seq, best_score)

    # === Phase 4: Advanced Peak Shaving ===
    print(f"\n=== Phase 4: Peak Shaving (best={best_score:.6f}) ===")
    rem = deadline - monotonic() - 300
    if rem > 20:
        t_ps = min(rem * 0.5, 150.0)
        print(f"  PeakShave budget={t_ps:.0f}s")
        a_ps, _, steps = peak_shave_advanced(best_seq, t_ps)
        best_seq, best_score = check_and_update(a_ps, f"peak_shave_{steps}steps", best_seq, best_score)

    # === Phase 5: Alternating LP with decreasing damping ===
    print(f"\n=== Phase 5: Alternating LP (best={best_score:.6f}) ===")
    rem = deadline - monotonic() - 250
    if rem > 30:
        t_alt = min(rem * 0.5, 150.0)
        print(f"  AltLP budget={t_alt:.0f}s")
        a_alt, _ = alternating_optimize(best_seq, t_alt, solver=solver)
        best_seq, best_score = check_and_update(a_alt, "alt_lp", best_seq, best_score)

    # === Phase 6: Second round of refinement ===
    print(f"\n=== Phase 6: Refinement Round 2 (best={best_score:.6f}) ===")
    if best_seq is not None:
        # Subgradient again
        rem = deadline - monotonic() - 200
        if rem > 20:
            t_sg2 = min(rem * 0.4, 60.0)
            a_sg2, _ = subgradient_descent(best_seq, t_sg2, n_top=max(20, n_best // 20), lr_init=0.008)
            best_seq, best_score = check_and_update(a_sg2, "subgrad_2", best_seq, best_score)

        # L-BFGS cold
        rem = deadline - monotonic() - 150
        if rem > 15:
            t_lb3 = min(rem * 0.5, 60.0)
            a_lb3, _ = softmax_lbgfs(best_seq, t_lb3, T_start=3.0, T_end=0.3, n_temps=8)
            best_seq, best_score = check_and_update(a_lb3, "lbgfs_cold2", best_seq, best_score)

        # Peak shave again
        rem = deadline - monotonic() - 100
        if rem > 15:
            t_ps2 = min(rem * 0.5, 50.0)
            a_ps2, _, _ = peak_shave_advanced(best_seq, t_ps2)
            best_seq, best_score = check_and_update(a_ps2, "peak_shave_2", best_seq, best_score)

    # === Phase 7: Try different length (if time permits) ===
    print(f"\n=== Phase 7: Length Exploration (best={best_score:.6f}) ===")
    rem = deadline - monotonic() - 80
    if rem > 60:
        # Try a slightly different length
        alt_n = n_best + 64 if n_best % 128 != 0 else n_best - 64
        if 256 <= alt_n <= 2048:
            t_alt_n = min(rem * 0.4, 60.0)
            init_alt = interpolate_sequence(best_seq, alt_n)
            a_alt_n, _, _ = fictitious_play(alt_n, t_alt_n, initial=init_alt, solver=solver, verbose=False)
            best_seq, best_score = check_and_update(a_alt_n, f"FP_n{alt_n}", best_seq, best_score)

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - monotonic() - 30
        if rem > 10:
            # Quick subgradient
            t_final_sg = min(rem * 0.4, 20.0)
            a_f1, _ = subgradient_descent(best_seq, t_final_sg, n_top=max(20, n_best // 25), lr_init=0.005)
            best_seq, best_score = check_and_update(a_f1, "final_sg", best_seq, best_score)

        rem = deadline - monotonic() - 20
        if rem > 8:
            # Quick peak shave
            t_final_ps = min(rem * 0.5, 15.0)
            a_f2, _, _ = peak_shave_advanced(best_seq, t_final_ps)
            best_seq, best_score = check_and_update(a_f2, "final_ps", best_seq, best_score)

    # Final verification
    if best_seq is None:
        best_seq = np.ones(512) / 512.0
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
