    # In frequency domain: FFT(w ⋆ a) = conj(FFT(a)) * FFT(w)
    # Wait no. Let me use the definition:
    # (f ⋆ g)[n] = sum_m f[m] * conj(g[m-n])
    # FFT(f ⋆ g) = conj(FFT(f)) * FFT(g) ... I'm getting confused.
    
    # Let me just use the direct relationship:
    # cross[i] = sum_{j=0}^{n-1} a[j] * w[i+j]
    # Let me define v as the reversed w: v[j] = w[2n-2-j] for j=0,...,2n-2
    # Then cross[i] = sum_j a[j] * w[i+j]
    # Hmm, this isn't a standard convolution or correlation directly.
    
    # Alternative: cross[i] = sum_j a[j] * w[i+j]
    # Let's define u of length 2n-1: u[k] = w[k]
    # Then cross[i] = sum_{j=0}^{n-1} a[j] * u[i+j]
    # This is like a "sliding dot product" - the dot product of a with a window of u starting at position i.
    
    # We can compute this via FFT:
    # Define a_ext of length 4n: a_ext[0:n] = a, rest = 0
    # Define u of length 2n-1, extended to 4n
    # Then cross[i] = sum_j a_ext[j] * u[i+j]
    # = (a_ext * u_reversed)[i + (2n-2)] where u_reversed = u[::-1]
    # Actually this is getting complicated. Let me use a different approach.
    
    # Simple FFT approach:
    # We want cross[i] = sum_{j=0}^{n-1} a[j] * w[i+j] for i = 0, ..., n-1
    # Let's define b of length 3n-2: b[j] = a[j] for j < n, 0 otherwise
    # And w of length 2n-1.
    # cross[i] = sum_j b[j] * w[i+j]
    # If we reverse w: wr[j] = w[2n-2-j], then:
    # cross[i] = sum_j b[j] * w[i+j] = sum_j b[j] * wr[2n-2-i-j]
    # = (b * wr)[2n-2-i] (convolution)
    # So cross = convolve(b, wr)[2n-2-i] for i=0,...,n-1
    # = convolve(b, wr)[2n-2 : 2n-2+n-1 : -1]... 
    
    # This is getting messy. Let me just use the direct FFT correlation formula.
    # 
    # Actually, the simplest way: 
    # cross[i] = sum_{j=0}^{n-1} a[j] * w[i+j]
    # = <a, shift_i(w[:n+...])>  -- not helpful
    #
    # Let me pad and use FFT:
    # Define A of length L (pad a to length L): A[0:n] = a
    # Define W of length L: W[0:2n-1] = w, W[2n-1:L] = 0
    # Then IFT(FFT(A) * FFT(W)) gives the circular convolution, which for our purposes
    # (with sufficient padding) gives the linear convolution.
    # 
    # (A * W)[k] = sum_j A[j] * W[k-j]
    # I want: cross[i] = sum_j A[j] * W[i+j]
    # Note: (A * W)[i + something]... 
    # Let me try: (A * rev_W)[k] where rev_W[m] = W[L-1-m]
    # (A * rev_W)[k] = sum_j A[j] * rev_W[k-j] = sum_j A[j] * W[L-1-k+j]
    # If L = 3n-2 (sufficient to avoid wrap-around):
    # (A * rev_W)[k] = sum_j a[j] * w[L-1-k+j] (for valid indices)
    # I want sum_j a[j] * w[i+j], so set L-1-k = i, i.e., k = L-1-i
    # (A * rev_W)[L-1-i] = sum_j a[j] * w[i+j] = cross[i] ✓
    
    # So: cross[i] = (A * rev_W)[L-1-i] for i=0,...,n-1
    # Where L >= n + (2n-1) - 1 = 3n-2
    
    L = 1 << (3 * n - 2 - 1).bit_length()
    A = rfft(a, L)
    W = rfft(w, L)
    # rev_W in frequency domain: ifft(conj(fft(W)))... no
    # rev_W[m] = W[L-1-m], so FFT(rev_W) = conj(FFT(W)) * exp(-2πi*... )
    # Actually for real signals: FFT(rev_W)[k] = conj(FFT(W)[-k mod L])... 
    # This is getting too complicated. Let me just do it the other way.
    
    # Simpler approach: use the fact that cross[i] = sum_j a[j] * w[i+j]
    # = sum_{k=i}^{i+n-1} a[k-i] * w[k]  (substituting j = k-i)
    # = <a, w[i:i+n]> (dot product of a with the slice w[i:i+n])
    # This is a sliding window dot product, which IS a cross-correlation:
    # cross = correlate(w, a) in the "non-standard" sense
    # 
    # Standard numpy correlate(w, a, mode='valid') gives:
    # corr[i] = sum_j w[i+j] * a[j] for i=0,...,len(w)-len(a) = 2n-1-n = n-1
    # That's exactly what we want!
    # 
    # But numpy correlate is O(n^2). We need FFT.
    #
    # numpy correlate(w, a) = convolve(w, a[::-1])
    # So cross = convolve(w, a[::-1])[:n] (for the "valid" part, first n elements)
    # Wait: convolve(w, a_rev)[k] = sum_j w[j] * a_rev[k-j]
    # = sum_j w[j] * a[n-1-k+j] (since a_rev[m] = a[n-1-m])
    # For k=0: sum_j w[j] * a[n-1+j]... that doesn't seem right.
    
    # Let me just use the direct FFT approach with proper padding:
    # cross[i] = sum_{j=0}^{n-1} a[j] * w[i+j] for i = 0,...,n-1
    # 
    # Method: 
    # Pad a to length L: a_pad[0:n] = a, a_pad[n:L] = 0
    # Pad w to length L: w_pad[0:2n-1] = w, w_pad[2n-1:L] = 0
    # 
    # We want cross[i] = sum_j a_pad[j] * w_pad[i+j]
    # 
    # Define w_rev[m] = w_pad[L-1-m]
    # conv(a_pad, w_rev)[k] = sum_j a_pad[j] * w_rev[k-j] = sum_j a_pad[j] * w_pad[L-1-k+j]
    # Set L-1-k = i: k = L-1-i
    # conv(a_pad, w_rev)[L-1-i] = sum_j a_pad[j] * w_pad[i+j] = cross[i] ✓
    
    w_rev = np.zeros(L)
    w_rev[:2*n-1] = w[::-1]
    # Wait, w_rev[m] = w_pad[L-1-m], so w_rev[0] = w_pad[L-1] = 0 (since w_pad is zero after 2n-2)
    # w_rev[L-1-(2n-2)] = w_pad[2n-2] = w[2n-2]
    # So w_rev[L-1-(2n-2) : L] = w[::-1]
    # i.e., w_rev[L-2n+2 : L] = w[::-1]
    
    # Hmm, this is getting complicated with the indexing. Let me just use a simpler method.
    
    # ALTERNATIVE SIMPLE METHOD:
    # cross[i] = sum_{j=0}^{n-1} a[j] * w[i+j]
    # This is the same as: for the vector a, compute the "upper triangular" part of a * w^T
    # 
    # Actually, the simplest FFT-based method:
    # Define u of length 2n-1 where u[k] = w[k]
    # Define v of length 2n-1 where v[k] = a[k] for k < n, 0 otherwise  
    # 
    # The "anti-convolution" or "correlation" we want is:
    # cross[i] = sum_j v[j] * u[i+j]
    #
    # This equals: ifft(fft(v) * conj(fft(u)))[i] ... let me verify.
    # 
    # Ifft(FFT(v) * conj(FFT(u)))[i] = sum_j v[j] * u[j-i] (circular correlation, positive lag)
    # That gives sum_j v[j] * u[j-i], not sum_j v[j] * u[i+j].
    #
    # Ifft(FFT(v) * FFT(u))[i] = sum_j v[j] * u[i-j] (circular convolution)
    # 
    # Neither gives exactly what we want directly. But:
    # cross[i] = sum_j v[j] * u[i+j] = sum_j v[j] * u[(i+j) mod L] (circular, with L large enough)
    # 
    # Let me define u_ext where u_ext[k] = u[k] for k < 2n-1, and u_ext[k] = u[k - (2n-1)] for 2n-1 <= k < L
    # No, that's circular wrapping which we don't want.
    
    # OK I'll just use a different approach. Let me note that:
    # cross[i] = sum_{j=0}^{n-1} a[j] * w[i+j]
    # 
    # If I define a_rev[j] = a[n-1-j], then:
    # cross[i] = sum_j a[n-1-(n-1-j)] * w[i+j] = sum_j a_rev[n-1-j] * w[i+j]
    # Let m = n-1-j, j = n-1-m:
    # cross[i] = sum_{m=0}^{n-1} a_rev[m] * w[i+n-1-m]
    # = (a_rev * w)[i+n-1] (standard convolution!)
    # 
    # YES! So cross[i] = convolve(a_rev, w)[i + n - 1] where a_rev = a[::-1]
    # 
    # And convolve(a_rev, w) can be computed with FFT!
    
    a_rev = a[::-1]
    conv_aw = fast_convolve_general(a_rev, w)  # linear convolution
    cross = conv_aw[n-1 : n-1+n]  # extract the right part
    
    return p * cross  # multiply by p for the gradient


def fast_convolve_general(a, b):
    """Linear convolution of two sequences via FFT."""
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def pnorm_adam(a0, time_limit, p=20, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """
    Adam optimization minimizing ||conv(a)||_p^p (equivalently ||conv(a)||_p)
    subject to a >= 0, sum(a) = 1.
    Uses log-space parameterization for non-negativity.
    """
    n = len(a0)
    a = normalize(a0)
    z = np.log(np.maximum(a, 1e-30))
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = get_score_normalized(a)
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        a_curr = softmax_from_z(z)
        
        # Compute p-norm gradient
        conv = fast_convolve(a_curr)
        w = np.power(np.maximum(conv, 1e-30), p - 1)
        
        # cross[i] = sum_j a[j] * w[i+j] = convolve(a[::-1], w)[i+n-1]
        a_rev = a_curr[::-1]
        conv_aw = fast_convolve_general(a_rev, w)
        cross = conv_aw[n-1 : 2*n-1]
        
        # Gradient: g_i = 2p * cross_i + p * conv[2i] * delta(2i < 2n-1)
        # (the diagonal terms are small, ignore for now)
        grad_a = 2.0 * p * cross
        
        # Project to tangent space of simplex: remove mean
        S = np.dot(grad_a, a_curr)
        grad_z = a_curr * (grad_a - S)
        
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        
        # Learning rate decay
        lr_t = lr / (1.0 + 0.001 * t)
        z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        
        a_new = softmax_from_z(z)
        score = get_score_normalized(a_new)
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 2000:
            z = np.log(np.maximum(best_a, 1e-30)) + np.random.standard_normal(n) * 0.01
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
    
    return best_a, best_score


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - np.max(z)
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return exp_z / s


def peak_reductions(a0, time_limit, n_trials=10000):
    """
    Peak-targeted local search with intelligent mass redistribution.
    For each peak, identify contributing elements and try to redirect mass
    to positions that don't contribute to any peak.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = get_score_normalized(a)
    t0 = time.monotonic()
    
    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(a)
        n_conv = len(conv)
        peak_val = conv.max()
        peak_positions = np.where(conv > 0.95 * peak_val)[0]
        
        # Find "safe" positions: those not contributing to any peak
        # Position j contributes to peak at lag k if 0 <= k-j <= n-1
        # i.e., j <= k and k-j < n, i.e., k-n+1 <= j <= k
        unsafe = np.zeros(n, dtype=bool)
        for k in peak_positions:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            unsafe[lo:hi+1] = True
        
        safe_indices = np.where(~unsafe)[0]
        if len(safe_indices) == 0:
            safe_indices = np.where(unsafe)[0]  # fallback
        
        # Find the most "overweight" elements (appear in most peak contributions)
        peak_count = np.zeros(n)
        for k in peak_positions:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            peak_count[lo:hi+1] += 1
        
        # Try to reduce elements with high peak_count and high value
        candidates = np.argsort(peak_count * a)[::-1][:10]
        
        improved = False
        for idx in candidates:
            if a[idx] < 1e-10:
                continue
            # Try moving some mass to safe positions
            for frac in [0.1, 0.2, 0.3, 0.5]:
                delta = a[idx] * frac
                for dest in safe_indices[:5]:
                    a_trial = a.copy()
                    a_trial[idx] -= delta
                    a_trial[dest] += delta
                    a_trial = np.maximum(a_trial, 0)
                    s = a_trial.sum()
                    if s < 1e-15:
                        continue
                    a_trial = a_trial / s
                    score = get_score_normalized(a_trial)
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a_trial.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved and trial % 200 == 0 and trial > 100:
            # Random perturbation
            a = best_a + np.random.standard_normal(n) * 0.02
            a = normalize(a)
    
    return best_a, best_score


def pairwise_swap_search(a0, time_limit, n_sweeps=50):
    """
    Pairwise swap: try moving mass from position i to position j.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = get_score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        
        order = rng.permutation(n)
        sweep_improved = False
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-12:
                continue
            
            # Try several fractions to move
            for frac in [0.05, 0.1, 0.2, 0.3]:
                delta = a[i] * frac
                # Try several destinations (random subset)
                for _ in range(10):
                    j = rng.integers(0, n)
                    if j == i:
                        continue
                    a_trial = a.copy()
                    a_trial[i] -= delta
                    a_trial[j] += delta
                    a_trial = np.maximum(a_trial, 0)
                    s = a_trial.sum()
                    if s < 1e-15:
                        continue
                    a_trial = a_trial / s
                    score = get_score_normalized(a_trial)
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a_trial.copy()
                        sweep_improved = True
                        break
                if sweep_improved:
                    break
        
        if not sweep_improved and sweep > 5:
            break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def adam_softmax(a0, time_limit, T_init=0.05, T_final=0.002, lr=0.01):
    """Adam in log-space with softmax max smoothing (similar to previous but standalone)."""
    n = len(a0)
    a = normalize(a0)
    z = np.log(np.maximum(a, 1e-30))
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = get_score_normalized(a)
    no_improve = 0
    fft_size = 1 << (2 * n - 2).bit_length()
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        T = T_init * (T_final / T_init) ** progress
        
        a_curr = softmax_from_z(z)
        b = fast_convolve(a_curr)
        max_b = b.max()
        exp_b = np.exp(b / T - max_b / T)
        w = exp_b / exp_b.sum()
        
        # Gradient: g_i = 2n * sum_j w[j+i] * a[j] (cross-correlation)
        # = 2n * convolve(a[::-1], w)[i+n-1]... 
        # Actually: g_i = 2n * (w * a_rev)[i + n - 1] where a_rev = a[::-1]
        # Wait, let me recompute.
        # g_i = 2n * d/da_i [sum_k w_k * conv_k]
        # = 2n * sum_k w_k * d(conv_k)/da_i
        # = 2n * sum_k w_k * 2*a_{k-i} (for valid k-i)
        # = 4n * sum_k w_k * a_{k-i}
        # Let j = k-i: = 4n * sum_j w_{j+i} * a_j
        # = 4n * cross[i] where cross[i] = sum_j a[j] * w[i+j]
        # = 4n * convolve(a[::-1], w)[i+n-1]
        
        a_rev = a_curr[::-1]
        conv_aw = fast_convolve_general(a_rev, w)
        cross = conv_aw[n-1 : 2*n-1]
        grad_a = 4.0 * n * cross
        
        S = np.dot(grad_a, a_curr)
        grad_z = a_curr * (grad_a - S)
        
        m = 0.9 * m + 0.1 * grad_z
        v = 0.999 * v + 0.001 * (grad_z * grad_z)
        m_hat = m / (1.0 - 0.9 ** t)
        v_hat = v / (1.0 - 0.999 ** t)
        lr_t = lr / (1.0 + 0.0005 * t)
        z = z - lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)
        
        a_new = softmax_from_z(z)
        score = get_score_normalized(a_new)
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 3000:
            z = np.log(np.maximum(best_a, 1e-30)) + np.random.standard_normal(n) * 0.005
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
    
    return best_a, best_score

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_general(a, b):
    """Linear convolution of two sequences via FFT."""
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def get_score_normalized(a):
    """Compute score assuming sum(a) = 1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - np.max(z)
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones_like(z) / len(z)
    return exp_z / s

    """

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def peak_reductions(a0, time_limit, n_trials=5000):

        peak_threshold = 0.9 * peak_val

        unsafe = np.zeros(n, dtype=bool)
        for k in peak_positions:
            lo = max(0, int(k) - n + 1)
            hi = min(n - 1, int(k))
            unsafe[lo:hi+1] = True

                if others_sum < 1e-15:
                    continue
                a_trial[mask] -= diff * a_trial[mask] / others_sum
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                score = get_score_normalized(a_trial)
                if score < best_local_score:
                    best_local_score = score
                    best_factor = factor
            
            if best_factor != 1.0:
                a[i] *= best_factor
                diff = a[i] * (1.0 - 1.0/best_factor)  # approximate
                # Proper redistribution
                a[i] = a[i]  # already set
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others_sum = a[mask].sum()
                if others_sum > 1e-15:
                    target_sum = 1.0 - a[i]
                    a[mask] *= target_sum / others_sum
                if best_local_score < best_score:
                    best_score = best_local_score
                    best_a = a.copy()
                    sweep_improved = True
        
        if not sweep_improved and sweep > 3:
            break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate sequence and return updated best."""
    a = normalize(a)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score

def parametric_search_best(n, time_limit):
    """
    Improved parametric search: random restarts in Fourier space
    with local refinement in full space.
    """
    t0 = time.monotonic()
    best_a = np.ones(n) / n
    best_score = get_score_normalized(best_a)
    best_coeffs = np.zeros(12)
    rng = np.random.default_rng(54321)
    
    n_fourier = 10
    
    # Generate diverse starting points
    starts = [np.zeros(n_fourier)]
    # Cosine bump
    c = np.zeros(n_fourier)
    c[1] = 0.15
    c[2] = -0.1
    starts.append(c)
    # Different structures
    for _ in range(10):
        c = rng.standard_normal(n_fourier) * 0.15
        starts.append(c)
    
    for si, start_c in enumerate(starts):
        if time.monotonic() - t0 > time_limit:
            break
        t_per_start = (time_limit - (time.monotonic() - t0)) / max(1, len(starts) - si)
        
        # Local search in Fourier space
        curr_c = start_c.copy()
        curr_score = fourier_score(curr_c, n)
        if curr_score < best_score:
            best_score = curr_score
            best_a = fourier_to_sequence(curr_c, n)
            best_coeffs = curr_c.copy()
        
        for step in range(200):
            if time.monotonic() - t0 > time_limit:
                break
            # Try random perturbation
            perturb_scale = 0.1 * max(0.2, 1.0 - step / 200)
            trial_c = curr_c + rng.standard_normal(n_fourier) * perturb_scale
            trial_score = fourier_score(trial_c, n)
            
            if trial_score < curr_score:
                curr_c = trial_c
                curr_score = trial_score
                if curr_score < best_score:
                    best_score = curr_score
                    best_a = fourier_to_sequence(curr_c, n)
                    best_coeffs = curr_c.copy()
                    print(f"    Fourier start {si} step {step}: {curr_score:.6f}")
            elif rng.random() < 0.1:
                # Occasional larger jump
                trial_c = curr_c + rng.standard_normal(n_fourier) * 0.3
                trial_score = fourier_score(trial_c, n)
                if trial_score < curr_score:
                    curr_c = trial_c
                    curr_score = trial_score
        
        # After each Fourier search, do a quick full-space refinement
        if time.monotonic() - t0 < time_limit * 0.8:
            t_refine = min(20.0, (deadline_global - time.monotonic()) if False else 20.0)
            a_fin, _ = adam_softmax(best_a, min(15.0, time_limit * 0.05), T_init=0.05, T_final=0.005, lr=0.01)
            s_fin = get_score_normalized(a_fin)
            if s_fin < best_score:
                best_score = s_fin
                best_a = a_fin
    
    return best_a, best_score
