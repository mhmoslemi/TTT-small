```python
from scipy.fft import rfft, irfft, fft, ifft
import time

def fast_autoconv(a):
    """Compute full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_gradient(a, w):
    """Compute gradient of sum_k w[k]*conv(a,a)[k] wrt a.
    
    grad_a[i] = 2 * sum_{m=0}^{n-1} a[m] * w[i+m]
    
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    # We need: grad_a[i] = 2 * sum_m a[m] * w[i+m] for i=0..n-1
    # This equals 2 * flip(convolve(a, flip(w)))[n-1+i]
    
    w_flipped = w[::-1]
    # Compute convolution via FFT
    conv_size = n + len(w_flipped) - 1
    fft_size = 1 << (conv_size - 1).bit_length()
    
    A = fft(a, fft_size)
    W = fft(w_flipped, fft_size)
    C = ifft(A * W)[:conv_size]  # C = convolve(a, w_flipped)
    
    C_flipped = C[::-1]
    # grad_a[i] = 2 * C_flipped[n-1+i]
    grad = 2.0 * C_flipped[n-1:n-1+n]
    return grad

def simplex_project(v):
    """Project vector v onto the probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v

        theta = (cssv[rho] - 1.0) / (rho + 1.0)
```

```python
def compute_grad_fft(a, w):
    n = len(a)
    # We need: grad[i] = 2 * sum_m a[m] * w[i+m] for i=0..n-1
    # Equivalently: grad[i] = 2 * c_flipped[n-1+i] where c = conv(a, w[::-1])
    
    # Using FFT for convolution
    w_rev = w[::-1]  # length 2n-1
    conv_len = n + (2*n-1) - 1  # = 3n-2
    fft_size = 1 << (conv_len - 1).bit_length()
    
    A = fft(a, fft_size)
    W = fft(w_rev, fft_size)
    C = ifft(A * W).real[:conv_len]  # linear convolution of a and w_rev
    
    C_flipped = C[::-1]
    grad = 2.0 * C_flipped[n-1:n-1+n]
    return grad
```

```python
from scipy.fft import fft, ifft
import time


def fast_autoconv(a):
    """Compute full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    C = ifft(A * A).real[:size]
    return C


def compute_grad_fft(a, w):
    """Compute gradient of sum_k w[k]*conv(a,a)[k] wrt a, using FFT.
    
    grad_a[i] = 2 * sum_{m=0}^{n-1} a[m] * w[i+m]
    Computed as 2 * flip(convolve(a, flip(w)))[n-1+i]
    """
    n = len(a)
    w_rev = w[::-1]
    conv_len = n + len(w_rev) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    A = fft(a, fft_size)
    W = fft(w_rev, fft_size)
    C = ifft(A * W).real[:conv_len]
    C_flipped = C[::-1]
    grad = 2.0 * C_flipped[n - 1:n - 1 + n]
    return grad


def simplex_project(v):
    """Project vector v onto the probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def score_normalized(a):
    """Compute evaluation score assuming sum(a)=1, a>=0."""
    n = len(a)
    b = fast_autoconv(a)
    return 2.0 * n * b.max()


def compute_weights(b, T, top_k):
    """Compute softmax weights on convolution peaks."""
    n_b = len(b)
    k_actual = min(top_k, n_b)
    peaks = np.argpartition(b, -k_actual)[-k_actual:]
    pv = b[peaks]
    w_full = np.zeros(n_b)
    w_peaks = np.exp((pv - pv.max()) / T)
    ws = w_peaks.sum()
    if ws < 1e-30:
        idx = np.argmax(pv)
        w_full[peaks[idx]] = 1.0
    else:
        w_peaks = w_peaks / ws
        w_full[peaks] = w_peaks
    return w_full


def grad_optimize(a0, time_budget, T_init, T_min, top_k, lr_init, rng, label):
    """FFT-based projected gradient with Nesterov momentum and adaptive params."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = score_normalized(a)
    v = np.zeros(n)
    T = T_init
    lr = lr_init
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_budget:
        it += 1
        b = fast_autoconv(a)
        w = compute_weights(b, T, top_k)
        grad = compute_grad_fft(a, w)

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * (it - 1) ** 2)) / 2.0
        beta = (t_k - 1.0) / t_k
        v = beta * v + grad

        a_new = a - lr * v
        a_new = simplex_project(a_new)

        sc = score_normalized(a_new)
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        if no_improve > 200:
            lr *= 0.88
            no_improve = 0
            T = max(T * 0.75, T_min)

        if no_improve > 800:
            # Perturb from best
            pert = rng.standard_normal(n) * 0.04
            a = simplex_project(best_a + pert)
            v = np.zeros(n)
            lr = max(lr_init * 0.4, lr * 1.5)
            T = max(T_init * 0.4, T_min)
            no_improve = 0

        if it % 500 == 0:
            T = max(T * 0.93, T_min)

        if it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} it={it} ({elapsed:.0f}/{time_budget:.0f}s): "
                  f"score={best_score:.7f} T={T:.5f} lr={lr:.6f}")

    return best_a, best_score


def coord_descent(a0, time_budget, rng, label):
    """Coordinate descent: try scaling individual coordinates."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    coarse_factors = [0.5, 0.7, 0.85, 0.93, 1.07, 1.15, 1.3, 1.6]
    fine_factors = [0.97, 0.99, 1.01, 1.03]

    sweep = 0
    while time.monotonic() - t0 < time_budget:
        sweep += 1
        improved = False
        perm = rng.permutation(n)
        for i in perm:
            if time.monotonic() - t0 > time_budget:
                break
            old_val = a[i]
            if old_val < 1e-14:
                continue
            # Coarse
            for factor in coarse_factors:
                a[i] = old_val * factor
                a = np.maximum(a, 0)
                s = a.sum()
                if s < 1e-15:
                    a[i] = old_val
                    continue
                a_norm = a / s
                sc = score_normalized(a_norm)
                if sc < best_score - 1e-13:
                    best_score = sc
                    best_a = a_norm.copy()
                    a = a_norm.copy()
                    old_val = a[i]
                    improved = True
                else:
                    a[i] = old_val
                    a = a / a.sum()
            # Fine
            if not improved or old_val > 0:
                for factor in fine_factors:
                    if time.monotonic() - t0 > time_budget:
                        break
                    a[i] = old_val * factor
                    a = np.maximum(a, 0)
                    s = a.sum()
                    if s < 1e-15:
                        a[i] = old_val
                        continue
                    a_norm = a / s
                    sc = score_normalized(a_norm)
                    if sc < best_score - 1e-14:
                        best_score = sc
                        best_a = a_norm.copy()
                        a = a_norm.copy()
                        old_val = a[i]
                        improved = True
                    else:
                        a[i] = old_val
                        a = a / a.sum()

        if not improved and sweep > 10:
            break
        if sweep % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} CD sweep={sweep} ({elapsed:.0f}s): score={best_score:.7f}")

    return best_a, best_score


def peak_shave(a0, time_budget, rng):
    """Targeted peak reduction: find top contributing pairs to the peak, reduce them."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for step in range(20000):
        if time.monotonic() - t0 > time_budget:
            break
        b = fast_autoconv(a)
        peak_k = int(np.argmax(b))

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        if js > je:
            break
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

        if contribs.max() < 1e-15:
            break

        top_c = np.argsort(contribs)[::-1][:8]
        improved = False

        for ci in top_c:
            if time.monotonic() - t0 > time_budget:
                break
            i = int(idx[ci])
            j = peak_k - i
            if a[i] < 1e-14:
                continue
            for factor in [0.92, 0.85, 0.75, 0.65]:
                a_trial = a.copy()
                a_trial[i] *= factor
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-13:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
            if not improved and i != j and a[j] >= 1e-14:
                for f1 in [0.9, 0.82]:
                    for f2 in [0.9, 0.82]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        st = a_trial.sum()
                        if st < 1e-15:
                            continue
                        a_trial = a_trial / st
                        sc = score_normalized(a_trial)
                        if sc < best_score - 1e-13:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
            if improved:
                break

        if not improved:
            # Try random small perturbations
            for _ in range(10):
                if time.monotonic() - t0 > time_budget:
                    break
                ri = rng.integers(n)
                old_v = a[ri]
                if old_v < 1e-14:
                    continue
                a_trial = a.copy()
                a_trial[ri] *= (1.0 + rng.standard_normal() * 0.1)
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
            if not improved:
                break

    return best_a, best_score


def simulated_annealing(a0, time_budget, rng):
    """SA-like perturbation: randomly modify a few coords, accept/reject with temperature."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = score_normalized(a)
    T_sa = 0.002
    t0 = time.monotonic()
    step = 0

    while time.monotonic() - t0 < time_budget:
        step += 1
        a_trial = a.copy()
        n_pert = int(rng.integers(2, 8))
        for _ in range(n_pert):
            ri = int(rng.integers(n))
            a_trial[ri] *= (1.0 + rng.standard_normal() * 0.15)
        a_trial = np.maximum(a_trial, 0)
        st = a_trial.sum()
        if st < 1e-15:
            continue
        a_trial = a_trial / st
        sc_trial = score_normalized(a_trial)
        delta = sc_trial - best_score
        if delta < 0:
            a = a_trial
            if sc_trial < best_score:
                best_score = sc_trial
                best_a = a_trial.copy()
        else:
            accept_prob = np.exp(-delta / T_sa)
            if rng.random() < accept_prob:
                a = a_trial

        if step % 500 == 0:
            T_sa = max(T_sa * 0.97, 0.0001)

        if step % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SA step={step} ({elapsed:.0f}s): best={best_score:.7f} T={T_sa:.6f}")

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def init_fourier(n, rng):
    """Fourier superposition initialization."""
    n_h = int(rng.integers(3, 20))
    amps = rng.random(n_h) * 0.35
    phases = rng.uniform(0, 2 * np.pi, n_h)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_h):
        a += amps[j] * np.cos(2 * np.pi * (j + 1) * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_blocks(n, rng):
    """Random block initialization."""
    n_blocks = int(rng.integers(4, 30))
    a = np.zeros(n)
    for b in range(n_blocks):
        s_start = b * n // n_blocks
        s_end = (b + 1) * n // n_blocks
        a[s_start:s_end] = rng.random() + 0.1
    ksize = min(5, max(2, n // n_blocks))
    if ksize > 1:
        kernel = np.ones(ksize) / ksize
        a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def propose_candidate():
    """Main search function: FFT-gradient + CD + SA + multi-length exploration."""
    deadline = time.monotonic() + 1790.0
    rng = np.random.default_rng(777)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            best_seq = parent.copy()
            best_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={best_score:.7f}")
    except Exception as e:
        print(f"No parent available: {e}")

    def record(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a_norm = a / s
        sc = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_norm.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.7f} (n={len(a_norm)})")

    # === Phase 1: Main gradient optimization from parent ===
    print("\n=== Phase 1: FFT Gradient from parent ===")
    if parent is not None:
        n_p = len(parent)
        rem = deadline - time.monotonic() - 400
        t1 = min(rem * 0.35, 400.0)
        print(f"  n={n_p}, budget={t1:.0f}s")
        a_opt, sc_opt = grad_optimize(parent, t1, T_init=0.015, T_min=0.001,
                                      top_k=20, lr_init=0.04, rng=rng, label="p1")
        record(a_opt, f"p1_grad_n{n_p}")

    # === Phase 2: Try multiple temperatures and configs ===
    print(f"\n=== Phase 2: Multi-config optimization (best={best_score:.7f}) ===")
    if best_seq is not None:
        configs = [
            (0.008, 0.0005, 10, 0.03),
            (0.02, 0.001, 40, 0.05),
            (0.005, 0.0003, 5, 0.02),
            (0.012, 0.0008, 15, 0.035),
        ]
        for ci, (T_i, T_m, tk, lr_i) in enumerate(configs):
            rem = deadline - time.monotonic() - 300
            if rem < 60:
                break
            t_alloc = min(rem * 0.2, 200.0)
            print(f"  Config {ci}: T={T_i}, top_k={tk}, lr={lr_i}, budget={t_alloc:.0f}s")
            a_opt, sc_opt = grad_optimize(best_seq, t_alloc, T_init=T_i, T_min=T_m,
                                          top_k=tk, lr_init=lr_i, rng=rng, label=f"p2_c{ci}")
            record(a_opt, f"p2_c{ci}")

    # === Phase 3: Peak shaving + coordinate descent ===
    print(f"\n=== Phase 3: Peak shaving + CD (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t_ps = min(rem * 0.35, 150.0)
            print(f"  PeakShave budget={t_ps:.0f}s")
            a_opt, sc_opt = peak_shave(best_seq, t_ps, rng)
            record(a_opt, "p3_peakshave")

        rem = deadline - time.monotonic() - 150
        if rem > 30:
            t_cd = min(rem * 0.4, 150.0)
            print(f"  CoordDescent budget={t_cd:.0f}s")
            a_opt, sc_opt = coord_descent(best_seq, t_cd, rng, "p3_cd")
            record(a_opt, "p3_cd")

    # === Phase 4: Simulated annealing ===
    print(f"\n=== Phase 4: Simulated Annealing (best={best_score:.7f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            t_sa = min(rem * 0.4, 150.0)
            print(f"  SA budget={t_sa:.0f}s")
            a_opt, sc_opt = simulated_annealing(best_seq, t_sa, rng)
            record(a_opt, "p4_sa")

    # === Phase 5: Different lengths ===
    print(f"\n=== Phase 5: Alternative lengths (best={best_score:.7f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        alt_lengths = []
        for f in [0.8, 1.25, 1.5, 0.67]:
            alt_n = int(n_best * f)
            if 256 <= alt_n <= 2048 and alt_n != n_best:
                alt_lengths.append(alt_n)
        alt_lengths = list(dict.fromkeys(alt_lengths))[:3]

        for alt_n in alt_lengths:
            rem = deadline - time.monotonic() - 100
            if rem < 60:
                break
            t_alloc = min(rem * 0.3, 150.0)
            print(f"  n={alt_n}, budget={t_alloc:.0f}s")
            a_init = interpolate_seq(best_seq, alt_n)
            a_opt, sc_opt = grad_optimize(a_init, t_alloc, T_init=0.012, T_min=0.001,
                                          top_k=15, lr_init=0.035, rng=rng, label=f"p5_n{alt_n}")
            record(a_opt, f"p5_n{alt_n}")

            rem = deadline - time.monotonic() - 80
            if rem > 20:
                t_cd2 = min(rem * 0.3, 60.0)
                a_opt2, sc2 = coord_descent(a_opt, t_cd2, rng, f"p5_cd_n{alt_n}")
                record(a_opt2, f"p5_cd_n{alt_n}")

    # === Phase 6: Diverse random restarts ===
    print(f"\n=== Phase 6: Random restarts (best={best_score:.7f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for r in range(4):
            rem = deadline - time.monotonic() - 50
            if rem < 40:
                break
            t_alloc = min(rem * 0.2, 100.0)

            if r == 0:
                a_init = best_seq + rng.standard_normal(n_best) * 0.08

            elif r == 1:
                a_init = init_fourier(n_best, rng)
            elif r == 2:
                a_init = init_blocks(n_best, rng)
            else:
                a_rand = rng.random(n_best) + 0.05
                a_rand = a_rand / a_rand.sum()
                a_init = 0.6 * best_seq + 0.4 * a_rand
                a_init = a_init / a_init.sum()

            a_opt, sc_opt = grad_optimize(a_init, t_alloc, T_init=0.01, T_min=0.0008,
                                          top_k=12, lr_init=0.03, rng=rng, label=f"p6_r{r}")
            record(a_opt, f"p6_r{r}")

    # === Phase 7: Final intensive polish ===
    print(f"\n=== Phase 7: Final polish (best={best_score:.7f}) ===")
    if best_seq is not None:
        # Final gradient with low temp
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            t_fin = min(rem * 0.4, 100.0)
            print(f"  Final grad budget={t_fin:.0f}s")
            a_opt, sc_opt = grad_optimize(best_seq, t_fin, T_init=0.004, T_min=0.0003,
                                          top_k=8, lr_init=0.015, rng=rng, label="p7_grad")
            record(a_opt, "p7_grad")

        # Final CD
        rem = deadline - time.monotonic() - 15
        if rem > 15:
            t_fin_cd = min(rem * 0.5, 60.0)
            print(f"  Final CD budget={t_fin_cd:.0f}s")
            a_opt, sc_opt = coord_descent(best_seq, t_fin_cd, rng, "p7_cd")
            record(a_opt, "p7_cd")

        # Final peak shave

    # Fallback

        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL: score={best_score:.7f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
```
