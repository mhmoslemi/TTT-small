Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based peak minimization,

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    result = irfft(A * A, fft_size)[:size]
    return result

def fast_cross_convolve(a, g):
    """Linear cross-convolution a * g via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    G = rfft(g, fft_size)
    return irfft(A * G, fft_size)[:size]

def eval_max_b(a):
    """Return max of auto-convolution."""
    return float(fast_convolve(a).max())

def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lp_peak_minimize(a, time_limit=30.0, verbose=False):

    Solve: min t s.t. (a*g)_k <= t for all k, sum(g)=1, g>=0.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_conv(a, g):
    """Linear cross-convolution a*g via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    G = rfft(g, fft_size)
    return irfft(A * G, fft_size)[:size]


def eval_max_b(a):
    """Return max of auto-convolution of a (assumed normalized to sum=1)."""
    return float(fast_convolve(a).max())


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix M (shape (2n-1, n)) such that M @ g = cross_conv(a, g)."""
    n = len(a)
    M = np.zeros((2 * n - 1, n))
    # For column i, M[k, i] = a[k-i] for k = i, i+1, ..., min(i+n-1, 2n-2)
    for i in range(n):
        end = min(i + n, 2 * n - 1)
        M[i:end, i] = a[:end - i]
    return M


def lp_solve_min_peak(a, time_per_lp=10.0):
    """
    Solve LP: min t s.t. (a*g)_k <= t for all k, sum(g)=1, g>=0.
    Returns (g, t_opt) or (None, None) on failure.
    """
    n = len(a)
    M = build_conv_matrix(a)  # shape (2n-1, n)
    
    # Variables: [g_0, ..., g_{n-1}, t]
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize t
    
    # Inequality: M @ g - t <= 0 => [M, -1] @ [g; t] <= 0
    A_ub = np.column_stack([M, -np.ones(2 * n - 1)])
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0.0, None)] * n + [(None, None)]
    
    t0 = time.monotonic()
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': max(1.0, time_per_lp)})
        if res.success:
            g = res.x[:n]
            t_opt = res.x[-1]
            # Ensure g is valid
            g = np.maximum(g, 0)
            s = g.sum()
            if s > 1e-15:
                g = g / s
            return g, t_opt
        return None, None
    except Exception:
        return None, None


def lp_optimize(a_init, time_limit, blending_strategy='adaptive', verbose=False):
    """
    Iteratively improve a by solving LPs and blending.
    Returns (best_a, best_max_b).
    """
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = eval_max_b(a)
    
    iteration = 0
    while time.monotonic() - t0 < time_limit:
        # Solve LP
        g, t_opt = lp_solve_min_peak(a, time_per_lp=5.0)
        if g is None:
            break
        
        current_peak = eval_max_b(a)
        
        if t_opt >= current_peak:
            # LP says we can't improve (or barely can)
            break
        
        # Try different blending strategies
        best_t = 0.0
        best_blend_obj = current_peak
        
        if blending_strategy == 'adaptive':
            candidates = [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.1]
        else:
            candidates = [1.0, 0.5, 0.3]
        
        for alpha in candidates:
            blend = (1 - alpha) * best_a + alpha * g
            blend = np.maximum(blend, 0)
            bs = blend.sum()
            if bs < 1e-15:
                continue
            blend = blend / bs
            obj = eval_max_b(blend)
            if obj < best_blend_obj - 1e-14:
                best_blend_obj = obj
                best_t = alpha
        
        if best_t <= 0:
            # Try very small alpha
            for alpha in [0.05, 0.02, 0.01]:
                blend = (1 - alpha) * best_a + alpha * g
                blend = np.maximum(blend, 0)
                blend = blend / blend.sum()
                obj = eval_max_b(blend)
                if obj < best_obj - 1e-14:
                    best_t = alpha
                    best_blend_obj = obj
                    break
        
        if best_t > 0:
            a = (1 - best_t) * best_a + best_t * g
            a = np.maximum(a, 0)
            a = a / a.sum()
            obj = eval_max_b(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                if verbose and iteration % 5 == 0:
                    elapsed = time.monotonic() - t0
                    print(f"    LP iter {iteration}: peak={best_obj:.10f}, t={best_t:.2f}, t_opt={t_opt:.6f}, time={elapsed:.1f}s")
            else:
                a = best_a.copy()
        else:
            break
        
        iteration += 1
    
    return best_a, best_obj

        # Create a matrix where row i is s[i:i+n] (n rows, n columns)
        # Then grad = 4 * (that_matrix @ a)
        # But that's O(n^2) memory (1152^2 * 8 = 10MB) which is fine!
        
        # Using stride tricks:
        if n <= 4096:
            # Create sliding window matrix
            s_mat = np.lib.stride_tricks.sliding_window_view(s, n)  # shape (n, n)
            grad = 4.0 * (s_mat @ a)
        else:
            # For larger n, do it in chunks
            grad = np.zeros(n)
            chunk = 256
            for start in range(0, n, chunk):
                end = min(start + chunk, n)
                s_chunk = s[start:end]
                # For each i in [start, end), grad[i] = 4 * sum_j s[i+j] * a[j]
                for i in range(start, end):
                    valid = s[i:i+n]
                    if len(valid) == n:
                        grad[i] = 4.0 * np.dot(valid, a)
        
        # Gradient step with simplex projection
        a_new = project_simplex(a - lr * grad)
        
        obj_new = eval_max_b(a_new)
        
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            stuck_count = 0
        else:
            stuck_count += 1
        
        a = a_new
        total_iters += 1
        
        if verbose and total_iters % 100 == 0:
            print(f"    Grad iter {total_iters}: peak={best_obj:.10f}, tau={tau:.1f}, lr={lr:.6f}")
        
        if stuck_count > 5000:
            break
    
    return best_a, best_obj


def multiplicative_weight_update(a_init, time_limit, eta_init=0.1, eta_min=0.001, verbose=False):
    """
    Multiplicative weight update: penalize elements that contribute to peak.
    a_i <- a_i * exp(-eta * contribution_i)
    """
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = eval_max_b(a)
    
    iteration = 0
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        eta = eta_init * (eta_min / eta_init) ** progress
        
        # Compute auto-convolution
        b = fast_convolve(a)
        b_max = b.max()
        max_idx = np.argmax(b)
        
        # The peak is at max_idx. Find which elements contribute most.
        # b[max_idx] = sum_j a[j] * a[max_idx - j]
        # Each pair (j, max_idx-j) contributes a[j]*a[max_idx-j]
        # We want to reduce the peak, so penalize the elements in the peak
        
        # Contribution of a[i] to b[max_idx]: a[i] * a[max_idx - i] (if valid)
        # + a[max_idx - i] * a[i] (same thing)
        # So total contribution = 2 * a[i] * a[max_idx - i] (for i != max_idx - i)
        # or a[i]^2 (for i = max_idx - i, i.e., i = max_idx/2)
        
        contrib = np.zeros(n)
        for i in range(n):
            j = max_idx - i
            if 0 <= j < n:
                if i == j:
                    contrib[i] = a[i] ** 2
                else:
                    contrib[i] = 2 * a[i] * a[j]
        
        # Update: a_i <- a_i * exp(-eta * contrib_i / b_max)
        a_new = a * np.exp(-eta * contrib / max(b_max, 1e-15))
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        obj_new = eval_max_b(a_new)
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            a = best_a.copy()
            # Try smaller eta
            eta *= 0.9
        
        iteration += 1
    
    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def make_window_init(n, window_type='gaussian', sigma=0.25):
    """Create various window initializations."""
    if window_type == 'uniform':
        return np.ones(n) / n
    elif window_type == 'gaussian':
        x = np.linspace(0, 1, n)
        a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    elif window_type == 'cosine':
        x = np.linspace(0, 1, n)
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
    elif window_type == 'tukey':
        r = 0.5
        a = np.ones(n)
        taper = max(1, int(r * n // 2))
        if taper < n:
            t = np.linspace(0, 1, taper)
            a[:taper] = 0.5 * (1.0 + np.cos(np.pi * (2.0 * t / r - 1.0)))
            a[-taper:] = a[:taper][::-1]
    else:
        return np.ones(n) / n
    a = np.maximum(a, 1e-12)
    return a / a.sum()


def coordinate_descent_polish(a0, time_limit, n_multipliers=5):
    """
    Coordinate descent: for each coordinate, try scaling it and rescaling others.
    """
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = eval_max_b(a)
    
    multipliers = np.array([0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0])
    
    improved = True
    passes = 0
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        passes += 1
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-15:
                continue
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue
            
            best_j_val = old_val
            best_j_obj = best_obj + 1e-10  # slight bias toward accepting
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                scale = (1.0 - new_val) / denom
                a_tr = a.copy()
                a_tr[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_tr[mask] *= scale
                ss = a_tr.sum()
                if ss < 1e-15:
                    continue
                a_tr = a_tr / ss
                val = eval_max_b(a_tr)
                if val < best_j_obj:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                if best_j_obj < best_obj:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
    
    return best_a, best_obj
