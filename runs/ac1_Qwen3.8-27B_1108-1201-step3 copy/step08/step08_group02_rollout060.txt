"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def objective(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2"""
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fast_convolve(a)
    return 2.0 * len(a) * c.max() / (s * s)


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, n, peak_indices, M):
    """
    Solve: maximize sum(g) s.t. g >= 0, sum(g) = 1, (f*g)[k] <= M for k in peaks.
    Actually: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for k in peaks.
    """
    K = len(peak_indices)
    A = np.zeros((K, n))
    for row_i in range(K):
        k = int(peak_indices[row_i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    for solver in [cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            prob.solve(solver=solver, max_iters=1000000)
            break
        except Exception:
            continue

    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
        return None
    s = g_val.sum()
    if s < 1e-10:
        return None
    return g_val / s

def lp_optimize(f_init, time_limit, label="lp", peak_frac_start=0.95, peak_frac_end=0.85, max_peaks=300):
    """LP-based optimization with adaptive peaks."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    f = f / f.sum()
    best_f = f.copy()
    best_b = fast_convolve(f).max()
    it = 0
    stuck = 0
    peak_frac = peak_frac_start
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        B = fast_convolve(f)
        M = B.max()
        
        threshold = peak_frac * M
        peaks = np.where(B >= threshold)[0]
        if len(peaks) > max_peaks:
            peaks = np.argpartition(B, -max_peaks)[-max_peaks:]
        if len(peaks) < 5:
            peaks = np.argpartition(B, -max(10, len(B)//10))[-max(10, len(B)//10):]
        
        g = solve_lp_direction(f, n, peaks, M)
        
        if g is None or g.sum() <= 1.0 + 1e-5:
            stuck += 1
            if stuck > 20:
                break
            # Try with all peaks
            if stuck == 10:
                peaks_all = np.argpartition(B, -min(max_peaks*2, len(B)))[-min(max_peaks*2, len(B)):]
                g = solve_lp_direction(f, n, peaks_all, M)
                if g is not None and g.sum() > 1.0 + 1e-5:
                    stuck = 0
                else:
                    continue
            continue
        
        # Line search
        best_t = 0.0
        best_t_obj = M
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            f_trial = (1-t)*f + t*g
            f_trial = np.maximum(f_trial, 0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj - 1e-15:
                best_t_obj = obj
                best_t = t
        
        if best_t <= 0:
            stuck += 1
            if stuck > 20:
                break
            continue
        
        f = (1-best_t)*f + best_t*g
        f = np.maximum(f, 0)
        f /= f.sum()
        
        cur_obj = fast_convolve(f).max()
        if cur_obj < best_b - 1e-14:

    return best_f, best_b


def interpolate_to_length(f, new_n):
    """Interpolate sequence to new length."""
    n = len(f)
    if n == new_n:
        return f.copy()
    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    return f_new

"""
Optimized sequence search combining LP-based direction finding,
fast continuous coordinate descent with O(n) conv updates,
and adaptive subgradient methods. Targets score below 1.5027.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    if size <= 0:
        return np.array([0.0])
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, n, peak_indices, M):
    """
    Solve: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for k in peak_indices.
    Returns normalized g or None.
    """
    K = len(peak_indices)
    if K == 0:
        return None
    A = np.zeros((K, n))
    for row_i in range(K):
        k = int(peak_indices[row_i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    solved = False
    for solver in [cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            prob.solve(solver=solver, max_iters=1000000)
            solved = True
            break
        except Exception:
            continue

    if not solved or g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
        return None
    s = g_val.sum()
    if s < 1e-10:
        return None
    return g_val / s


def lp_optimize(f_init, time_limit, label="lp", peak_frac_start=0.95,
                peak_frac_end=0.85, max_peaks=300):
    """LP-based optimization with adaptive peaks and line search."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    f = f / f.sum()
    best_f = f.copy()
    best_b = fast_convolve(f).max()
    it = 0
    stuck = 0
    peak_frac = peak_frac_start
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit:
        it += 1
        B = fast_convolve(f)
        M = B.max()

        threshold = peak_frac * M
        peaks = np.where(B >= threshold)[0]
        if len(peaks) > max_peaks:
            peaks = np.argpartition(B, -max_peaks)[-max_peaks:]
        if len(peaks) < 5:
            peaks = np.argpartition(B, -max(10, len(B) // 10))[
                -max(10, len(B) // 10):]

        g = solve_lp_direction(f, n, peaks, M)

        if g is None or g.sum() <= 1.0 + 1e-5:
            stuck += 1
            if stuck > 25:

            if stuck == 12:
                npk = min(max_peaks * 2, len(B))
                more_peaks = np.argpartition(B, -npk)[-npk:]
                g2 = solve_lp_direction(f, n, more_peaks, M)
                if g2 is not None and g2.sum() > 1.0 + 1e-5:
                    g = g2
                    stuck = 0
                else:
                    continue
            else:
                continue

        # Line search with golden section refinement

        best_t_obj = M

            if s < 1e-10:

            f_trial /= s

            if obj < best_t_obj - 1e-15:

                if dt <= 0 or dt > 0.9:

                f_trial = (1.0 - dt) * f + dt * g

                if s < 1e-10:

                if obj < best_t_obj - 1e-15:

            if stuck > 25:
                break
            continue

        if cur_obj < best_b - 1e-14:

            stuck = 0

            f = f + rng.standard_normal(n) * 0.001 * f
            f = np.maximum(f, 0.0)
            f /= f.sum()

    return best_f, best_b


def coordinate_descent_fast(f_init, time_limit, label="cd"):
    """
    Fast coordinate descent with O(n) convolution updates per coordinate.
    Maintains B = conv(f,f) incrementally.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    S = f.sum()
    if S < 1e-15:
        f = np.ones(n) / n
        S = 1.0
    f = f / S
    S = 1.0
    B = fast_convolve(f)
    best_obj_val = 2.0 * n * B.max()
    best_f = f.copy()
    rng = np.random.default_rng(999)

    no_improve_sweeps = 0
    sweep = 0

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        B = fast_convolve(f)
        S = f.sum()
        current_obj = 2.0 * n * B.max() / (S * S)
        sweep_improved = False

        # Determine order
        if sweep % 7 == 1:
            order = rng.permutation(n)
        else:
            order = np.arange(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break

            v = f[j]
            S = f.sum()
            
            # Generate candidate deltas
            if v < 1e-12:
                candidates = [0.001, 0.005, 0.01, 0.02]
            else:
                candidates = [v * m - v for m in
                              [0.3, 0.5, 0.7, 0.85, 1.0, 1.2, 1.5, 2.0, 3.0]]
                # Also try small perturbations
                candidates.append(0.01 * v)
                candidates.append(-0.01 * v)

            best_delta = 0.0
            best_local = 2.0 * n * B.max() / (S * S)

            for delta in candidates:
                new_v = v + delta
                if new_v < 0:
                    continue
                new_S = S + delta
                if new_S < 1e-10:
                    continue

                # Affected region: [j, j+n-1]
                lo = j
                hi = min(j + n - 1, 2 * n - 2)
                length = hi - lo + 1

                # New B in affected region
                # B_new[k] = B[k] + 2*delta*f[k-j] + delta^2*(k==2j)
                f_slice = f[0:length]
                affected = B[lo:hi + 1] + 2.0 * delta * f_slice
                if 2 * j >= lo and 2 * j <= hi:
                    affected[2 * j - lo] += delta * delta

                max_affected = affected.max()
                
                # Unaffected max
                max_unaffected = 0.0
                if lo > 0:
                    max_unaffected = max(max_unaffected, B[:lo].max())
                if hi + 1 < 2 * n - 1:
                    max_unaffected = max(max_unaffected, B[hi + 1:].max())

                new_max = max(max_affected, max_unaffected)
                obj = 2.0 * n * new_max / (new_S * new_S)

                if obj < best_local - 1e-15:
                    best_local = obj
                    best_delta = delta

            if best_delta != 0.0:
                # Update B incrementally
                lo = j
                hi = min(j + n - 1, 2 * n - 2)
                length = hi - lo + 1
                f_slice = f[0:length]
                B[lo:hi + 1] += 2.0 * best_delta * f_slice
                if 2 * j >= lo and 2 * j <= hi:
                    B[2 * j] += best_delta * best_delta
                
                f[j] += best_delta
                S += best_delta
                sweep_improved = True

        # End of sweep
        if sweep_improved:
            no_improve_sweeps = 0
        else:
            no_improve_sweeps += 1
            if no_improve_sweeps > 30:
                # Perturb and restart
                f = best_f.copy()
                f = f + rng.standard_normal(n) * 0.005 * f
                f = np.maximum(f, 0)
                S = f.sum()
                f /= S
                no_improve_sweeps = 0
                if sweep > 200:
                    break
            elif no_improve_sweeps > 10:
                f = best_f.copy()
                f = f + rng.standard_normal(n) * 0.002 * f
                f = np.maximum(f, 0)
                f /= f.sum()

        if sweep % 10 == 0:
            B_check = fast_convolve(f)
            S_check = f.sum()
            obj_check = 2.0 * n * B_check.max() / (S_check * S_check)
            if obj_check < best_obj_val - 1e-14:
                best_obj_val = obj_check
                best_f = f.copy()

        if sweep % 50 == 0:
            score = best_obj_val
            print(f"    {label} sweep={sweep} obj={best_obj_val:.6f} "

        if time.monotonic() - t0 > time_limit - 5:
            break

    # Final check
    B_final = fast_convolve(f)
    S_final = f.sum()
    obj_final = 2.0 * n * B_final.max() / (S_final * S_final)

def subgradient_optimize(f_init, time_limit, lr_init=0.008, label="sg"):
    """Adaptive subgradient with momentum."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    f = f / f.sum()
    v = np.zeros(n)
    momentum = 0.85
    lr = lr_init
    best_f = f.copy()
    best_b = fast_convolve(f).max()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(31337)

    while time.monotonic() - t0 < time_limit:
        it += 1
        B = fast_convolve(f)
        M = B.max()

        # Softmax weights on B
        temp = 0.002 * M
        scaled = B / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        # Compute gradient using cross-correlation via FFT
        N_fft = 1 << (3 * n).bit_length()
        F = rfft(f, N_fft)
        W = rfft(w, N_fft)
        R = irfft(np.conj(F) * W, N_fft)
        grad = 2.0 * R[:n]

        # Momentum update
        v = momentum * v - lr * grad
        f_new = f + v
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-15:
            break
        f_new /= s
        f = f_new

        if it % 20 == 0:
            B = fast_convolve(f)
            M = B.max()
            if M < best_b - 1e-14:
                best_b = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 5:
                    lr *= 0.92
                    no_improve = 0

            if it % 200 == 0:
                score = 2.0 * n * best_b
                print(f"    {label} it={it} obj={best_b:.12f} score={score:.6f} "

                v = np.zeros(n)

def interpolate_to_length(f, new_n):
    """Interpolate sequence to new length."""
    n = len(f)
    if n == new_n:
        return f.copy()
    if n < 2:
        return np.ones(new_n) / new_n
    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

    n_base = len(state['best_seq'])
    target_lengths = []
    for delta in [-128, -64, 64, 128, 192, 256, -192, 320]:
        tl = n_base + delta
        if 512 <= tl <= 3000 and tl != n_base:
            target_lengths.append(tl)
    target_lengths = list(dict.fromkeys(target_lengths))
    target_lengths.sort(key=lambda x: abs(x - n_base))

    for target_n in target_lengths[:4]:
        rem = deadline - time.monotonic()
        if rem < 100:
            break
        t5 = min(rem * 0.15, 180.0)
        if t5 < 40:
            break
        f_interp = interpolate_to_length(state['best_seq'], target_n)
        print(f"\n=== Phase 5: n={target_n} (budget={t5:.0f}s) ===")
        # LP then CD
        f5a, _ = lp_optimize(f_interp, time_limit=t5 * 0.4,
                             label=f"lp_n{target_n}",
                             peak_frac_start=0.93, peak_frac_end=0.83,
                             max_peaks=250)
        f5b, _ = coordinate_descent_fast(f5a, time_limit=t5 * 0.6,
                                         label=f"cd_n{target_n}")
        update_best(f5b, f"n{target_n}", state)

    # === Phase 6: Final LP ===
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.20, 200.0)
    if t6 > 40:
        n_b = len(state['best_seq'])
        print(f"\n=== Phase 6: Final LP (n={n_b}, budget={t6:.0f}s) ===")
        f6, _ = lp_optimize(state['best_seq'], time_limit=t6, label="lp_fin",
                            peak_frac_start=0.93, peak_frac_end=0.80,
                            max_peaks=400)
        update_best(f6, "lp_fin", state)

    # === Phase 7: Final CD burst ===
    rem = deadline - time.monotonic()
    t7 = min(rem * 0.40, 300.0)
    if t7 > 30:
        n_b = len(state['best_seq'])
        print(f"\n=== Phase 7: Final CD (n={n_b}, budget={t7:.0f}s) ===")
        f7, _ = coordinate_descent_fast(state['best_seq'], time_limit=t7, label="cd_fin")
        update_best(f7, "cd_fin", state)

    # === Phase 8: One more CD if time remains ===
    rem = deadline - time.monotonic()
    if rem > 40:
        t8 = rem * 0.85
        print(f"\n=== Phase 8: Final CD2 (budget={t8:.0f}s) ===")
        f8, _ = coordinate_descent_fast(state['best_seq'], time_limit=t8, label="cd_last")
        update_best(f8, "cd_last", state)

    # Final verification
    fs = float(evaluate_sequence(state['best_seq'].tolist()))
    if np.isfinite(fs):
        state['best_score'] = fs
    else:
        state['best_seq'] = parent.copy()
        state['best_score'] = float(evaluate_sequence(parent.tolist()))

    elapsed = time.monotonic() - start_time
    print(f"\n{'=' * 50}")
    print(f"FINAL SCORE: {state['best_score']:.6f} "
          f"(n={len(state['best_seq'])}, time={elapsed:.0f}s)")
    print(f"{'=' * 50}")

    return [float(x) for x in state['best_seq']]
