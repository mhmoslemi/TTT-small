import numpy as np

from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def get_lp_solver():
    for solver_name in ['CLARABEL', 'ECOS', 'SCS', 'PDLP']:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return 'ECOS'

def solve_lp(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in ['ECOS', 'SCS', 'PDLP']:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def nesterov_lp(a0, time_limit, solver, max_iters=200000):
    """Nesterov accelerated LP alternating optimization."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, M_b = solve_lp(y, solver)
        if b_new is None:
            break
        
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy
        
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score

def lbgfs_refine(a0, time_limit, T_start=100.0, T_end=0.5, maxiter=8000):
    """L-BFGS with LogSumExp smoothing."""
    n = len(a0)
    z = np.log(np.maximum(np.asarray(a0, dtype=np.float64), 1e-300))
    
    def softmax_from_z(zz):
        zz_shifted = zz - zz.max()
        exp_z = np.exp(zz_shifted)
        s = exp_z.sum()
        if s < 1e-300:
            return np.ones(len(zz)) / len(zz)
        return exp_z / s
    
    t0 = time.monotonic()
    
    def make_func(T_val):
        def func(zz):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            aa = softmax_from_z(zz)
            bb = fast_convolve(aa)
            Tb = T_val * bb
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func
    
    def make_jac(T_val):
        def jac(zz):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(zz)
            aa = softmax_from_z(zz)
            bb = fast_convolve(aa)
            Tb = T_val * bb
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(bb)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * aa[k - idx]
            S = np.dot(grad_a, aa)
            return aa * (grad_a - S)
        return jac

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in ['CLARABEL', 'ECOS', 'SCS', 'PDLP']:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return 'ECOS'


def solve_lp_given_a(a_fixed, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in ['ECOS', 'SCS', 'PDLP', 'CLARABEL']:

                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break

                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def softmax_from_z_func(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a_func(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def lbgfs_refine_func(a0, time_limit, T_start=100.0, T_end=0.5, maxiter=10000):
    """L-BFGS with LogSumExp smoothing and geometric temperature annealing."""
    n = len(a0)
    z = z_from_a_func(a0).copy()
    t0 = time.monotonic()

    def func_wrapper(T_val):
        def func(zz):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            aa = softmax_from_z_func(zz)
            bb = fast_convolve(aa)
            Tb = T_val * bb
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def jac_wrapper(T_val):
        def jac(zz):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(zz)
            aa = softmax_from_z_func(zz)
            bb = fast_convolve(aa)
            Tb = T_val * bb
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(bb)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * aa[k - idx]
            S = np.dot(grad_a, aa)
            return aa * (grad_a - S)
        return jac

    n_stages = 8
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(50, maxiter // n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        try:
            res = minimize(
                fun=func_wrapper(stage_T),
                x0=z,
                jac=jac_wrapper(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z_func(z)


def coordinate_descent_func(a0, time_limit, n_trials=80000):
    """Aggressive coordinate descent with pair moves and redistribution."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for trial in range(n_trials):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()

        # Identify near-peak positions
        threshold = max_b * 0.998
        peaks = np.where(b >= threshold)[0]

        improved = False

        # Try single-coordinate reduction with redistribution
        for k in peaks[:8]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contribs = a[idx] * a[k - idx]
            top = np.argsort(contribs)[::-1][:6]
            for ci in top:
                i = idx[ci]
                if a[i] < 1e-12:
                    continue
                for factor in [0.995, 0.99, 0.985, 0.98, 0.97]:
                    new_val = a[i] * factor
                    diff = a[i] - new_val
                    if diff < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[i] = new_val
                    pos_mask = (a > 1e-12).copy()
                    pos_mask[i] = False
                    if pos_mask.sum() == 0:
                        break
                    redist = diff * a[pos_mask] / a[pos_mask].sum()
                    a_trial[pos_mask] += redist
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        # Try pair reduction
        if not improved and len(peaks) > 0:
            k_main = int(np.argmax(b))
            js = max(0, k_main - n + 1)
            je = min(n - 1, k_main)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[k_main - idx]
            top4 = np.argsort(contribs)[::-1][:4]
            for ci1 in range(len(top4)):
                for ci2 in range(ci1 + 1, len(top4)):
                    i1, i2 = int(idx[top4[ci1]]), int(idx[top4[ci2]])
                    if i1 == i2:
                        continue
                    for f1 in [0.98, 0.96]:
                        for f2 in [0.98, 0.96]:
                            a_trial = a.copy()
                            a_trial[i1] *= f1
                            a_trial[i2] *= f2
                            a_trial = np.maximum(a_trial, 0)
                            a_trial = a_trial / a_trial.sum()
                            score = 2.0 * n * fast_convolve(a_trial).max()
                            if score < best_score - 1e-14:
                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                improved = True
                                break
                        if improved:
                            break
                    if improved:
                        break
                if improved:
                    break

        # If stuck, occasional perturbation to escape
        if not improved and trial > 200 and trial % 500 == 0:
            noise = rng.standard_normal(n) * 0.005
            a_trial = a + noise
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score * 1.005:
                a = a_trial

    return best_a, best_score


def interpolate_to_length(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe_func(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def update_best(best_seq, best_score, candidate, label):
    """Try to update best if candidate is better. Returns (best_seq, best_score)."""
    candidate = np.maximum(candidate, 0).astype(np.float64)
    s = candidate.sum()
    if s < 0.01:
        return best_seq, best_score
    candidate = candidate / s
    sc = evaluate_safe_func(candidate)
    if np.isfinite(sc) and sc < best_score:
        print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(candidate)})")
        return candidate.copy(), sc
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    Phase 1: Nesterov LP (long, main workhorse) - 40%
    Phase 2: L-BFGS refinement - 10%
    Phase 3: Coordinate descent - 10%
    Phase 4: Second LP round - 10%
    Phase 5: Multi-start perturbations - 10%
    Phase 6: Alternative lengths - 10%
    Phase 7: Final polish - 10%
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe_func(parent)
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if best_seq is None:
        n_init = 960
        x = np.linspace(0, 1, n_init)
        best_seq = np.sin(np.pi * x) ** 2
        best_seq = np.maximum(best_seq, 0)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe_func(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.8f}")

    n_best = len(best_seq)
    print(f"Starting: n={n_best}, score={best_score:.8f}")

    # === Phase 1: Long Nesterov LP ===
    print(f"\n=== Phase 1: Nesterov LP (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.40, 650.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a1, _ = nesterov_lp_optimize(best_seq, t_alloc, solver, max_iters=300000)
        best_seq, best_score = update_best(best_seq, best_score, a1, "lp1")

    # === Phase 2: L-BFGS refinement ===
    print(f"\n=== Phase 2: L-BFGS (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a2 = lbgfs_refine_func(best_seq, t_alloc, T_start=50.0, T_end=0.3, maxiter=8000)
        best_seq, best_score = update_best(best_seq, best_score, a2, "lbgfs1")

    # === Phase 3: Coordinate descent ===
    print(f"\n=== Phase 3: Coord Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 10:
        print(f"  Budget: {t_alloc:.0f}s")
        a3, _ = coordinate_descent_func(best_seq, t_alloc, n_trials=100000)
        best_seq, best_score = update_best(best_seq, best_score, a3, "cd1")

    # === Phase 4: Second LP round ===
    print(f"\n=== Phase 4: LP Round 2 (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a4, _ = nesterov_lp_optimize(best_seq, t_alloc, solver, max_iters=100000)
        best_seq, best_score = update_best(best_seq, best_score, a4, "lp2")

    # === Phase 5: Multi-start perturbations ===
    print(f"\n=== Phase 5: Multi-start (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    n_perturb = 3
    t_per = min(rem / (n_perturb + 2), 120.0)
    for pi in range(n_perturb):
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        noise_scales = [0.015, 0.03, 0.05]
        noisy = best_seq + noise_scales[pi] * rng.standard_normal(len(best_seq))
        noisy = np.maximum(noisy, 0)
        noisy = noisy / noisy.sum()

        t_lp = t_per * 0.65
        if t_lp > 10:
            a_p, _ = nesterov_lp_optimize(noisy, t_lp, solver, max_iters=50000)
            best_seq, best_score = update_best(best_seq, best_score, a_p, f"pert{pi}_lp")

        rem = deadline - time.monotonic()
        if rem > 15:
            t_cd = min(rem * 0.3, 25.0)
            if t_cd > 5:
                a_p2, _ = coordinate_descent_func(best_seq, t_cd, n_trials=30000)
                best_seq, best_score = update_best(best_seq, best_score, a_p2, f"pert{pi}_cd")

    # === Phase 6: Alternative lengths ===
    print(f"\n=== Phase 6: Alt Lengths (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    n_best = len(best_seq)
    alt_lengths = []
    for mult in [1.25, 1.5, 0.8, 2.0, 0.67]:
        alt_n = int(n_best * mult)
        if 300 <= alt_n <= 2500 and alt_n != n_best:
            alt_lengths.append(alt_n)
    alt_lengths = sorted(set(alt_lengths))

    for alt_n in alt_lengths[:3]:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        t_total = min(rem * 0.22, 130.0)
        if t_total < 25:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")
        init_alt = interpolate_to_length(best_seq, alt_n)

        t_lp = t_total * 0.7
        a_alt, _ = nesterov_lp_optimize(init_alt, t_lp, solver, max_iters=40000)
        best_seq, best_score = update_best(best_seq, best_score, a_alt, f"alt_n{alt_n}_lp")

        rem = deadline - time.monotonic()
        if rem > 10 and best_seq is not None and len(best_seq) == alt_n:
            t_cd = min(rem * 0.3, 25.0)
            if t_cd > 5:
                a_alt2, _ = coordinate_descent_func(best_seq, t_cd, n_trials=20000)
                best_seq, best_score = update_best(best_seq, best_score, a_alt2, f"alt_n{alt_n}_cd")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.35, 60.0)
        a_fin, _ = coordinate_descent_func(best_seq, t_fin, n_trials=60000)
        best_seq, best_score = update_best(best_seq, best_score, a_fin, "final_cd")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_fin2 = min(rem * 0.3, 40.0)
        a_fin2 = lbgfs_refine_func(best_seq, t_fin2, T_start=5.0, T_end=0.1, maxiter=3000)
        best_seq, best_score = update_best(best_seq, best_score, a_fin2, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 10 and best_seq is not None:
        t_fin3 = min(rem * 0.4, 25.0)
        a_fin3, _ = nesterov_lp_optimize(best_seq, t_fin3, solver, max_iters=5000)
        best_seq, best_score = update_best(best_seq, best_score, a_fin3, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe_func(best_seq)
        if np.isfinite(fs):
            best_score = fs

    total_time = time.monotonic() - (deadline - float(budget_s) + 30.0)
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, time={total_time:.0f}s)")
    return [float(x) for x in best_seq]
