def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_lse_gradient(f, T):
    n = len(f)
    b = fast_convolve(f)
    b_scaled = b / T
    b_scaled -= b_scaled.max()
    w = np.exp(b_scaled)
    w /= w.sum()
    w_rev = w[::-1]
    size = 3 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w_rev, fft_size)
    F = rfft(f, fft_size)
    full_conv = irfft(W * F, fft_size)[:size]
    h = full_conv[n - 1:2 * n - 1][::-1]
    correction = w[0:2 * n - 1:2] * f
    return (2.0 / T) * h - (1.0 / T) * correction


def normalize_nonneg(f):
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-12:
        return np.ones_like(f) / len(f)
    return f / s


def adam_lse_optimize(f_init, time_limit, T_init=None, T_min=0.002, lr_init=0.05,

def subgradient_optimize(f_init, time_limit, lr_init=0.003, n_top=60,
                         temp_init=0.005, label="sg"):
    t0 = time.monotonic()
    n = len(f_init)
    f = normalize_nonneg(f_init.copy())
    v = np.zeros(n)
    lr = lr_init
    temp = temp_init
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(777)
    momentum = 0.8

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        scaled = b / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
            w_sel = w[top_idx].copy()
            ws = w_sel.sum()
            if ws > 1e-30:
                w_sel /= ws
            else:
                w_sel = np.ones(len(top_idx)) / len(top_idx)
            w = np.zeros(len(b))
            w[top_idx] = w_sel
        g = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-13:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            contrib = 2.0 * w[k] * f[k - idx]
            if k < 2 * n - 2:
                for i in range(len(idx)):
                    if k == 2 * idx[i]:
                        contrib[i] = w[k] * f[k - idx[i]]
            g += contrib
        v = momentum * v - lr * g
        f = np.maximum(f + v, 0.0)
        s = f.sum()
        if s > 1e-12:
            f /= s
        else:
            f = best_f.copy()
        if it % 100 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 100
                if no_improve > 500:
                    lr *= 0.9
                    temp *= 1.15
                    no_improve = 0
                if no_improve > 3000:
                    f = normalize_nonneg(best_f + 0.002 * rng.standard_normal(n))
                    v = np.zeros(n)
                    no_improve = 0
            if it % 2000 == 0:
                print(f"    {label} it={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} "

    return best_f, best_obj


def build_constraint_matrix(f, indices, n):
    K = len(indices)
    A = np.zeros((K, n))
    for i, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    return A


def solve_lp(f, n, M, active_idx, solver=cp.PDLP):
    K = len(active_idx)
    if K == 0:
        return None
    A = build_constraint_matrix(f, active_idx, n)
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=solver, max_iters=500000)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=300000)
        except Exception:
            return None
    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None
    return g_val


def lp_polish(f_init, time_limit, initial_k=50, max_k=300, label="lp"):

    it = 0

        it += 1

        if g is None:

            if stuck > 10:

        if S_g <= 1.0 + 1e-6:

            if len(active_idx) < max_k:
                b_cur = fast_convolve(f)
                new_top = np.argpartition(b_cur, -min(max_k, len(b_cur)))[-min(max_k, len(b_cur)):]
                active_idx = np.sort(np.unique(np.concatenate([active_idx, new_top])))
            continue
        fg = irfft(rfft(f, 1 << (4 * n - 3).bit_length()) *
                   rfft(g, 1 << (4 * n - 3).bit_length()),
                   1 << (4 * n - 3).bit_length())[:2 * n - 1]

                to_add = not_active[np.argsort(fg[not_active])[-n_add:]]
                active_idx = np.sort(np.unique(np.concatenate([active_idx, to_add])))
                g = solve_lp(f, n, M, active_idx)
                if g is None:

                    if stuck > 5:

                S_g = g.sum()
                if S_g <= 1.0 + 1e-6:
                    stuck += 1
                    if stuck > 25:
                        break
                    continue
        stuck = 0
        h = g / g.sum()
        best_t = 0.1
        best_t_obj = M
        for t in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:

                active_idx = np.sort(active_idx[:max_k])
            if it % 100 == 0:

    return best_f, best_obj

        if np.isfinite(score1) and score1 < best_score:
            best_score = score1
            best_seq = f1.copy()
            print(f"  NEW BEST: {best_score:.6f}")

    # === Phase 2: Second Adam run with different params ===
    print(f"\n=== Phase 2: Adam+LSE v2 (n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.20, 300.0)
    if t2 > 30:
        n_b = len(best_seq)
        b_b = fast_convolve(best_seq)
        T2 = 0.05 * (b_b.max() - b_b.min() + 1e-10)
        print(f"  Budget: {t2:.0f}s, T_init={T2:.4f}, lr=0.02")
        f2, obj2 = adam_lse_optimize(best_seq, time_limit=t2, T_init=T2, T_min=0.0005,
                                     lr_init=0.02, restart_patience=400, label="adam2")
        score2 = float(evaluate_sequence(f2.tolist()))
        print(f"  Adam2 result: score={score2:.6f}")
        if np.isfinite(score2) and score2 < best_score:
            best_score = score2
            best_seq = f2.copy()
            print(f"  NEW BEST: {best_score:.6f}")

    # === Phase 3: Subgradient refinement ===
    print(f"\n=== Phase 3: Subgradient (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.15, 200.0)
    if t3 > 20:
        n_b = len(best_seq)
        print(f"  Budget: {t3:.0f}s, n={n_b}")
        f3, obj3 = subgradient_optimize(best_seq, time_limit=t3, lr_init=0.003,
                                        n_top=max(40, n_b // 10), temp_init=0.003,
                                        label="sg")
        score3 = float(evaluate_sequence(f3.tolist()))
        print(f"  SG result: score={score3:.6f}")
        if np.isfinite(score3) and score3 < best_score:
            best_score = score3
            best_seq = f3.copy()
            print(f"  NEW BEST: {best_score:.6f}")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-length (best={best_score:.6f}) ===")
    n_base = len(best_seq)
    target_lengths = sorted(set([
        max(512, int(n_base * 0.85)),
        max(512, int(n_base * 0.95)),
        max(512, int(n_base * 1.10)),
        max(512, int(n_base * 1.25)),
        max(512, int(n_base * 1.50)),
    ]))
    for target_n in target_lengths:
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        if target_n == n_base:
            continue
        t4 = min(rem * 0.10, 80.0)
        if t4 < 15:
            break
        f_interp = interpolate_sequence(best_seq, target_n)
        print(f"  Exploring n={target_n}, budget={t4:.0f}s")
        f4, _ = adam_lse_optimize(f_interp, time_limit=t4, T_min=0.002,
                                  lr_init=0.04, restart_patience=150,
                                  label=f"n{target_n}")
        s4 = float(evaluate_sequence(f4.tolist()))
        if np.isfinite(s4) and s4 < best_score:
            best_score = s4
            best_seq = f4.copy()
            print(f"  NEW BEST [n{target_n}]: {best_score:.6f}")

    # === Phase 5: LP polish on best ===
    print(f"\n=== Phase 5: LP polish (best={best_score:.6f}, n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.20, 250.0)
    if t5 > 20:
        n_b = len(best_seq)
        print(f"  Budget: {t5:.0f}s, k=50->300")
        f5, obj5 = lp_polish(best_seq, time_limit=t5, initial_k=50, max_k=300, label="lp5")
        score5 = float(evaluate_sequence(f5.tolist()))
        print(f"  LP5 result: score={score5:.6f}")
        if np.isfinite(score5) and score5 < best_score:
            best_score = score5
            best_seq = f5.copy()
            print(f"  NEW BEST: {best_score:.6f}")

    # === Phase 6: Final Adam burst ===
    print(f"\n=== Phase 6: Final Adam (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.15, 150.0)
    if t6 > 20:
        n_b = len(best_seq)
        b_b = fast_convolve(best_seq)
        T6 = 0.03 * (b_b.max() - b_b.min() + 1e-10)
        print(f"  Budget: {t6:.0f}s, T_init={T6:.5f}")
        f6, obj6 = adam_lse_optimize(best_seq, time_limit=t6, T_init=T6, T_min=0.0003,
                                     lr_init=0.015, restart_patience=500, label="adam_final")
        score6 = float(evaluate_sequence(f6.tolist()))
        if np.isfinite(score6) and score6 < best_score:
            best_score = score6
            best_seq = f6.copy()
            print(f"  NEW BEST: {best_score:.6f}")

    # === Phase 7: Final LP ===
    rem = deadline - time.monotonic()
    if rem > 30:
        print(f"\n=== Phase 7: Final LP (best={best_score:.6f}) ===")
        t7 = min(rem * 0.6, 120.0)
        n_b = len(best_seq)
        print(f"  Budget: {t7:.0f}s, k=80->400")
        f7, obj7 = lp_polish(best_seq, time_limit=t7, initial_k=80, max_k=400, label="lp7")
        score7 = float(evaluate_sequence(f7.tolist()))
        if np.isfinite(score7) and score7 < best_score:
            best_score = score7
            best_seq = f7.copy()
            print(f"  NEW BEST: {best_score:.6f}")

    # Final validation
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*60}")

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return result
