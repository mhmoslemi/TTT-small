    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def get_lp_solver():
    import cvxpy as cp
    global _lp_solver
    if '_lp_solver' in globals() and _lp_solver is not None:
        return _lp_solver
    _lp_solver = None
    for solver_name in [cp.MOSEK, cp.GUROBI, cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def lp_fictitious_play(n, time_limit, initial=None, solver=None, verbose=False):
    if solver is None:
        solver = get_lp_solver()
    t0 = time.monotonic()
    if initial is None:
        a = np.ones(n) / n
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        a = a / s if s > 1e-15 else np.ones(n) / n

    b = a.copy()
    a_avg = a.copy().astype(np.float64)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    iters = 0
    last_improve = 0
    patience = 300

    while time.monotonic() - t0 < time_limit:
        iters += 1
        b_new, _ = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new
        a_new, _ = solve_lp_best_response(b, solver)
        if a_new is not None:
            alpha = min(1.0, 5.0 / (iters + 1))
            a = (1 - alpha) * a + alpha * a_new
            a = np.maximum(a, 0)
            s = a.sum()
            a = a / s if s > 1e-15 else np.ones(n) / n

        a_avg = (iters * a_avg + a) / (iters + 1)

        for cand in [a, b, 0.5 * (a + b), a_avg]:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-10:
                continue
            cand_norm = cand / sc
            score = 2.0 * n * fast_convolve(cand_norm).max()
            if score < best_score - 1e-14:
                best_score = score
                best_a = cand_norm.copy()
                last_improve = iters

        if verbose and iters % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FP n={n} it={iters} ({elapsed:.0f}s): best={best_score:.6f}")

        if iters - last_improve > patience:
            break

    return best_a, best_score, iters


def spectral_flatten(a0, time_limit, alpha=0.7, n_inner=50, verbose=False):
    """
    Iteratively compress the power spectrum to reduce autocorrelation peak.
    Uses geometric interpolation between current magnitude and flat magnitude.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s if s > 1e-15 else np.ones(n) / n

    N = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)

        # Current autocorrelation peak
        ac = np.fft.irfft(mag * mag, N)[:2 * n - 1]
        peak = ac.max()

        # Geometric interpolation toward flat spectrum
        target_mag = np.ones_like(mag) * (mag.sum() / len(mag))
        # Preserve DC: target_mag[0] = mag[0]
        target_mag[0] = mag[0]

        log_mag = np.log(np.maximum(mag, 1e-30))
        log_target = np.log(np.maximum(target_mag, 1e-30))
        log_new = alpha * log_mag + (1 - alpha) * log_target
        mag_new = np.exp(log_new)

        # Preserve DC
        mag_new[0] = mag[0]

        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, N)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
        else:
            break

        a = a_new
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Spectral it={it} ({elapsed:.0f}s): best={best_score:.6f} peak={peak:.8f}")

    return best_a, best_score


def softmax_subgradient(a0, time_limit, lr_init=0.015, verbose=False):
    """Subgradient descent with softmax smoothing of the max."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s if s > 1e-15 else np.ones(n) / n

    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * conv.max()
    t0 = time.monotonic()
    it = 0
    lr = lr_init
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = float(conv.max())
        score = 2.0 * n * M

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 150:
            lr *= 0.85
            no_improve = 0
            if lr < 1e-9:
                break

        # Adaptive beta
        beta = min(50.0 + it * 0.5, 500.0)

        # Softmax weights over the convolution values
        shifted = conv - M
        w = np.exp(beta * shifted / max(M, 1e-10))
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros_like(conv)
            w[conv.argmax()] = 1.0
        else:
            w = w / w_sum

        # Gradient: grad[i] = 2 * sum_k w[k] * a[k-i] for valid k-i
        T_a = build_conv_matrix(a)
        grad = 2.0 * (T_a.T @ w)

        # Mirror descent step
        scaled = a * np.exp(-lr * grad)
        a = simplex_project(scaled)

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD it={it} ({elapsed:.0f}s): best={best_score:.6f} lr={lr:.6f}")

    return best_a, best_score


def coord_descent_precise(a0, time_limit, verbose=False, label=""):
    """
    Coordinate descent using analytical optimal step for each coordinate.
    For coordinate i, the optimal a[i] minimizes max_k conv_new[k] / (sum_new)^2.
    We use a combination of analytical candidates and binary search.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s if s > 1e-15 else np.ones(n) / n

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            length_aff = k_end - k_start + 1

            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            a_shifted = a[0:length_aff]
            conv_aff = conv[k_start:k_end + 1]

            # Compute candidate delta values
            # Analytical optimal for each active peak:
            # For peak at relative position r (k = k_start + r):
            #   d_opt = (a_shifted[r] - conv_aff[r]) / a_shifted[r] if a_shifted[r] > 0
            # But we also need to consider (1+d)^2 in the denominator.
            # The exact optimum of (c + 2d*a_r) / (1+d)^2 is at d = (a_r - c)/(c + 2*a_r) ... 
            # actually let's just use binary search on the unimodal function.

            # Generate candidates
            candidates = [0.0]
            # Current value and small perturbations
            candidates.extend([-0.5 * a[i], -0.3 * a[i], -0.1 * a[i],
                               0.1 * a[i], 0.3 * a[i], 0.5 * a[i]])
            # Analytical candidates from top peaks
            top_r = np.argsort(conv_aff)[-min(5, length_aff):]
            for r in top_r:
                if a_shifted[r] > 1e-15 and conv_aff[r] > 0:
                    d_cand = (a_shifted[r] - conv_aff[r]) / (conv_aff[r] + 2 * a_shifted[r])
                    candidates.append(d_cand)
                    candidates.append(d_cand * 0.8)
                    candidates.append(d_cand * 1.2)

            # Evaluate each candidate
            best_d = 0.0
            best_val = 2.0 * n * max(unaff_max, float(conv_aff.max()))

            for d in candidates:
                new_val_i = a[i] + d
                if new_val_i < -1e-15:
                    continue
                new_sum = 1.0 + d
                if new_sum < 0.01:
                    continue

                new_aff = conv_aff + 2.0 * d * a_shifted
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    new_aff = new_aff.copy()
                    new_aff[idx_2i] += d * d

                new_max = max(unaff_max, float(new_aff.max()))
                val = 2.0 * n * new_max / (new_sum * new_sum)

                if val < best_val - 1e-15:
                    best_val = val
                    best_d = d

            if best_d != 0.0:
                # Binary search refinement around best_d
                lo = best_d - 0.1 * a[i]
                hi = best_d + 0.1 * a[i]
                lo = max(lo, -a[i] + 1e-15)

                for _ in range(30):
                    m1 = lo + (hi - lo) / 3.0
                    m2 = hi - (hi - lo) / 3.0

                    def eval_d(d):
                        new_sum = 1.0 + d
                        if new_sum < 0.01 or a[i] + d < -1e-15:
                            return 1e30
                        new_aff = conv_aff + 2.0 * d * a_shifted
                        if 2 * i >= k_start and 2 * i <= k_end:
                            idx_2i = 2 * i - k_start
                            new_aff = new_aff.copy()
                            new_aff[idx_2i] += d * d
                        new_max = max(unaff_max, float(new_aff.max()))
                        return 2.0 * n * new_max / (new_sum * new_sum)

                    v1 = eval_d(m1)
                    v2 = eval_d(m2)
                    if v1 < v2:
                        hi = m2
                    else:
                        lo = m1

                d_final = (lo + hi) / 2.0
                v_final = eval_d(d_final)

                if v_final < best_val - 1e-15:
                    best_d = d_final
                    best_val = v_final

                # Apply update
                a[i] = max(a[i] + best_d, 0.0)

                # Check if actually improved
                new_sum = a.sum()
                a_norm = a / new_sum
                cur_score = 2.0 * n * fast_convolve(a_norm).max()

                if cur_score < best_score - 1e-15:
                    best_score = cur_score
                    best_a = a_norm.copy()
                    conv = fast_convolve(a_norm)
                    M = float(conv.max())
                    a = a_norm
                    improved = True
                else:
                    # Revert
                    a[i] = max(a[i] - best_d, 0.0)
                    # Don't change conv (it's still correct for the reverted a)

        if not improved and sweeps > 10:
            break
        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score, sweeps


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def init_fourier(n, rng):
    n_harmonics = int(rng.integers(5, 25))
    amplitudes = rng.random(n_harmonics) * 0.3
    phases = rng.uniform(0, 2 * np.pi, n_harmonics)
    freqs = np.arange(1, n_harmonics + 1)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harmonics):
        a += amplitudes[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_cosine(n, rng):
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    width = n * rng.uniform(0.2, 0.6)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_u_shape(n, rng):
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    alpha = rng.uniform(0.5, 3.0)
    a = 1.0 + alpha * np.cos(np.pi * t) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_ricatto(n, rng):
    """Initialize with a shape inspired by optimal transport / Riccati flow."""
    t = np.linspace(0, 1, n)
    # Product of cosines
    a = np.ones(n)
    for k in range(1, 6):
        a *= (1.0 + 0.3 * np.cos(2 * np.pi * k * t + rng.uniform(0, 2 * np.pi)))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def check_and_update(a, best_seq, best_score, label=""):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate():
    """
    Multi-phase optimization:
    1. Diverse initialization + quick screening
    2. LP fictitious play for global optimization
    3. Spectral flattening for diverse local optima
    4. Softmax subgradient for smooth refinement
    5. Precise coordinate descent for local optimum
    6. Final polishing
    """
    deadline = time.monotonic() + 1780.0
    rng = np.random.default_rng(42)
    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    # === Phase 1: Diverse initialization + quick optimization ===
    print("\n=== Phase 1: Diverse Init + Quick Screen ===")
    t1_start = time.monotonic()
    n_values = [768, 896, 1024, 1152, 1280]
    if parent is not None:
        np_ = len(parent)
        if np_ not in n_values:
            n_values.insert(0, np_)
    n_values = n_values[:5]

    candidates = []
    for n in n_values:
        if time.monotonic() - t1_start > 150:
            break
        inits = []
        if parent is not None:
            inits.append(interpolate_seq(parent, n))
        inits.append(init_fourier(n, rng))
        inits.append(init_cosine(n, rng))
        inits.append(init_u_shape(n, rng))
        inits.append(init_ricatto(n, rng))

        for a_init in inits:
            # Quick coordinate descent
            a_opt, s_opt, _ = coord_descent_precise(a_init, 5.0, label=f"p1_n{n}")
            candidates.append((s_opt, a_opt, n))

    candidates.sort(key=lambda x: x[0])
    print(f"  {len(candidates)} candidates, top score: {candidates[0][0]:.6f}")
    for s_c, a_c, n_c in candidates[:3]:
        best_seq, best_score = check_and_update(a_c, best_seq, best_score, f"p1_n{n_c}")

    # === Phase 2: LP Fictitious Play ===
    print(f"\n=== Phase 2: LP Fictitious Play (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        fp_lengths = [n_best]
        for alt_n in [1024, 896, 1152]:
            if alt_n != n_best and 512 <= alt_n <= 1536:
                fp_lengths.append(alt_n)
        fp_lengths = fp_lengths[:3]

        for n_fp in fp_lengths:
            rem = deadline - time.monotonic() - 600
            if rem < 60:
                break
            t_alloc = min(rem * 0.45, 350.0)
            print(f"  LP-FP n={n_fp}, budget={t_alloc:.0f}s")

            if best_seq is not None:
                initial = interpolate_seq(best_seq, n_fp)
            else:
                initial = None

            a_fp, score_fp, iters = lp_fictitious_play(
                n_fp, t_alloc, initial=initial, solver=solver, verbose=True
            )
            best_seq, best_score = check_and_update(a_fp, best_seq, best_score, f"fp_n{n_fp}")

    # === Phase 3: Spectral Flattening ===
    print(f"\n=== Phase 3: Spectral Flattening (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for alpha_val in [0.7, 0.5, 0.8]:
            rem = deadline - time.monotonic() - 500
            if rem < 30:
                break
            t_alloc = min(rem * 0.3, 100.0)
            print(f"  Spectral alpha={alpha_val}, n={n_best}, budget={t_alloc:.0f}s")
            a_spec, s_spec = spectral_flatten(best_seq, t_alloc, alpha=alpha_val, verbose=True)
            best_seq, best_score = check_and_update(a_spec, best_seq, best_score, f"spec_a{alpha_val}")

    # === Phase 4: Softmax Subgradient ===
    print(f"\n=== Phase 4: Softmax SGD (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 350
        if rem > 30:
            t_alloc = min(rem * 0.5, 150.0)
            print(f"  SGD n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_sgd, s_sgd = softmax_subgradient(best_seq, t_alloc, lr_init=0.012, verbose=True)
            best_seq, best_score = check_and_update(a_sgd, best_seq, best_score, "sgd")

    # === Phase 5: Precise Coordinate Descent ===
    print(f"\n=== Phase 5: Precise CD (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t_alloc = min(rem * 0.6, 250.0)
            print(f"  CD n={len(best_seq)}, budget={t_alloc:.0f}s")
            a_cd, s_cd, sweeps = coord_descent_precise(best_seq, t_alloc, verbose=True, label="ph5")
            best_seq, best_score = check_and_update(a_cd, best_seq, best_score, f"cd_{sweeps}s")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-length (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        alt_lengths = [n_best - 128, n_best + 128, n_best - 64, n_best + 64]
        alt_lengths = [x for x in alt_lengths if 384 <= x <= 2048]

        for alt_n in alt_lengths[:3]:
            rem = deadline - time.monotonic() - 150
            if rem < 30:
                break
            t_alloc = min(rem * 0.4, 80.0)
            init_alt = interpolate_seq(best_seq, alt_n)
            a_alt, s_alt, _ = coord_descent_precise(init_alt, t_alloc, label=f"alt{alt_n}")
            best_seq, best_score = check_and_update(a_alt, best_seq, best_score, f"alt_n{alt_n}")

    # === Phase 7: Final Polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        # Another round of CD
        rem = deadline - time.monotonic() - 30
        if rem > 20:
            t_alloc = min(rem * 0.5, 100.0)
            a_fin, s_fin, _ = coord_descent_precise(best_seq, t_alloc, label="final")
            best_seq, best_score = check_and_update(a_fin, best_seq, best_score, "final_cd")

        # Quick spectral
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            a_sp, s_sp = spectral_flatten(best_seq, min(rem * 0.5, 30.0), alpha=0.6)
            best_seq, best_score = check_and_update(a_sp, best_seq, best_score, "final_spec")

        # Final SGD
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_sg, s_sg = softmax_subgradient(best_seq, min(rem, 20.0), lr_init=0.005)
            best_seq, best_score = check_and_update(a_sg, best_seq, best_score, "final_sgd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
