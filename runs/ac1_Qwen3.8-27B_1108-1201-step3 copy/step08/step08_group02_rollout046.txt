"""
Improved optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses LP-based Frank-Wolfe with exact quadratic line search,
adaptive constraint sets, multiple restarts, and local search.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def exact_line_search(f, h, B_f, C, B_h, t_min=0.0, t_max=1.0, n_iter=50):
    """
    Exact line search: minimize max_k B_t[k] where
    B_t[k] = (1-t)^2 * B_f[k] + 2*t*(1-t) * C[k] + t^2 * B_h[k].
    Uses golden section search since the objective is convex in t.
    """
    def obj(t):
        val = (1-t)**2 * B_f + 2*t*(1-t) * C + t**2 * B_h
        return val.max()
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2
    a_t = t_min
    b_t = t_max
    c_t = b_t - gr * (b_t - a_t)
    d_t = a_t + gr * (b_t - a_t)
    f_c = obj(c_t)
    f_d = obj(d_t)
    
    for _ in range(n_iter):
        if f_c < f_d:
            b_t = d_t
            d_t = c_t
            f_d = f_c
            c_t = b_t - gr * (b_t - a_t)
            f_c = obj(c_t)
        else:
            a_t = c_t
            c_t = d_t
            f_c = f_d
            d_t = a_t + gr * (b_t - a_t)
            f_d = obj(d_t)
    
    t_opt = (a_t + b_t) / 2
    return t_opt, obj(t_opt)


def solve_lp_direction(f, n, peak_indices, M, eps=1e-9):
    """
    Solve: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for k in peak_indices.
    Returns g vector or None.
    """
    K = len(peak_indices)
    A = np.zeros((K, n))
    for row_i in range(K):
        k = int(peak_indices[row_i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    solved = False
    for solver in [cp.SCS, cp.ECOS, cp.CLARABEL]:
        try:
            prob.solve(solver=solver, max_iters=1000000, eps=eps)
            if g.value is not None and not np.any(np.isnan(g.value)):
                solved = True
                break
        except Exception:
            continue

    if not solved or g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
        return None
    if g_val.sum() <= 1.0 + 1e-6:
        return None
    return g_val


def frank_wolfe_step(f, peaks, M, eps=1e-9):
    """
    One Frank-Wolfe step: solve LP, do exact line search, return new f.
    Returns (new_f, improved).
    """
    n = len(f)
    g = solve_lp_direction(f, n, peaks, M, eps=eps)
    if g is None:
        return f, False
    
    h = g / g.sum()
    
    # Compute convolutions for line search
    B_f = fast_convolve(f)
    B_h = fast_convolve(h)
    C = fast_cross_convolve(f, h)
    
    t_opt, t_obj = exact_line_search(f, h, B_f, C, B_h, t_min=0.0, t_max=0.95, n_iter=60)
    
    if t_opt <= 1e-10:
        return f, False
    
    f_new = (1.0 - t_opt) * f + t_opt * h
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-15:
        f_new /= s
    else:
        return f, False
    
    # Verify improvement
    new_M = fast_convolve(f_new).max()
    if new_M < M - 1e-12:
        return f_new, True
    else:
        return f, False

def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length using cubic-like smoothing."""
    n = len(f)
    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    # Smooth with a small kernel
    kernel_size = max(3, new_n // 500)
    if kernel_size % 2 == 0:
        kernel_size += 1
    kernel = np.ones(kernel_size) / kernel_size
    f_new = np.convolve(f_new, kernel, mode='same')
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses LP-based Frank-Wolfe with exact quadratic line search, adaptive
constraint sets, random restarts for local minima escape, multi-length
exploration, and coordinate descent polish.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def exact_line_search(B_f, C, B_h, t_max=0.95, n_iter=50):
    """
    Minimize max_k[(1-t)^2*B_f[k] + 2t(1-t)*C[k] + t^2*B_h[k]] over t in [0, t_max].
    Uses golden section search (objective is convex in t).
    Returns (t_opt, obj_opt).
    """
    def obj(t):
        val = (1.0 - t) * (1.0 - t) * B_f + 2.0 * t * (1.0 - t) * C + t * t * B_h
        return val.max()

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_t = 0.0
    b_t = t_max
    c_t = b_t - gr * (b_t - a_t)
    d_t = a_t + gr * (b_t - a_t)
    f_c = obj(c_t)
    f_d = obj(d_t)

    for _ in range(n_iter):
        if f_c <= f_d:
            b_t = d_t
            d_t = c_t
            f_d = f_c
            c_t = b_t - gr * (b_t - a_t)
            f_c = obj(c_t)
        else:
            a_t = c_t
            c_t = d_t
            f_c = f_d
            d_t = a_t + gr * (b_t - a_t)
            f_d = obj(d_t)

    t_opt = (a_t + b_t) * 0.5
    return t_opt, obj(t_opt)


def solve_lp_direction(f, n, peak_indices, M, eps=1e-9):
    """
    Solve: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for k in peak_indices.
    Returns normalized g (sum=1) or None.
    """
    K = len(peak_indices)
    A = np.zeros((K, n))
    for row_i in range(K):
        k = int(peak_indices[row_i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    solved = False
    for solver in [cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, max_iters=2000000, eps=eps)
            if g.value is not None:
                gv = np.asarray(g.value, dtype=np.float64)
                if not np.any(np.isnan(gv)) and not np.any(np.isinf(gv)):
                    solved = True
                    break
        except Exception:
            continue

    if not solved or g.value is None:
        return None
    g_val = np.maximum(np.asarray(g.value, dtype=np.float64), 0.0)
    S = g_val.sum()
    if S <= 1.0 + 1e-5:
        return None
    g_val /= S
    return g_val


def frank_wolfe_iteration(f, peaks, M, eps=1e-9):
    """
    One FW step: solve LP for direction, exact line search, return new f.
    Returns (new_f, improved_flag).
    """
    n = len(f)
    g = solve_lp_direction(f, n, peaks, M, eps=eps)
    if g is None:
        return f, False

    h = g
    B_f = fast_convolve(f)
    B_h = fast_convolve(h)
    C = fast_cross_convolve(f, h)

    t_opt, _ = exact_line_search(B_f, C, B_h, t_max=0.95, n_iter=50)

    if t_opt <= 1e-12:
        return f, False

    f_new = (1.0 - t_opt) * f + t_opt * h
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-15:
        return f, False
    f_new /= s

    new_M = fast_convolve(f_new).max()
    if new_M < M - 1e-13:
        return f_new, True
    return f, False

def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto simplex {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def interpolate_to_length(f, new_n):
    """Interpolate and smooth a sequence to a new length."""
    n = len(f)
    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    # Light smoothing
    ksize = max(3, new_n // 400)
    if ksize % 2 == 0:
        ksize += 1
    kernel = np.ones(ksize) / ksize
    f_new = np.convolve(f_new, kernel, mode='same')
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def coordinate_descent(f_init, time_limit, max_rounds=40):
    """Coordinate descent with multiplier search per coordinate."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_M = fast_convolve(f).max()
    multipliers = [0.5, 0.7, 0.8, 0.9, 0.95, 1.05, 1.1, 1.2, 1.4, 1.6, 2.0]

    for _ in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-16:
                continue
            best_val = f[j]
            best_obj = best_M
            for mult in multipliers:
                f_trial = f.copy()
                f_trial[j] = f[j] * mult
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_val = f[j] * mult
            if best_val != f[j]:
                f[j] = best_val
                s = f.sum()
                f /= s
                best_M = best_obj
                improved = True
        if not improved:
            break

    return f, best_M


def propose_candidate(seed=42, budget_s=1800.0):

    Main search: FP-based Frank-Wolfe with exact line search, adaptive

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)

        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(1024) / 1024.0
        p_score = 2.0
        print("No parent available, using uniform(1024)")

    best_seq = parent.copy()
    best_score = p_score

    # === Phase 1: Main FW (tight peaks, high precision) ===
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.38, 580.0)
    if t1 > 60:
        n_p = len(parent)
        print(f"\n=== Phase 1: FW main (n={n_p}, budget={t1:.0f}s) ===")

            best_score = vs1
            best_seq = f1.copy()
            print(f"  *** NEW BEST [fw1]: {vs1:.8f}")

    print(f"{'='*55}")
