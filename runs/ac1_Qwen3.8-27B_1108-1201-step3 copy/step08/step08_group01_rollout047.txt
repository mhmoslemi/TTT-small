    """
    Compute the gradient of soft_max(conv(a,a), tau) w.r.t. a.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    conv = fast_convolve(a)
    
    # Soft-max weights
    conv_max = conv.max()
    exp_conv = np.exp(tau * (conv - conv_max))
    sum_exp = exp_conv.sum()
    w = exp_conv / sum_exp  # length 2n-1
    
    # Compute (a_rev * w) via FFT
    a_rev = a[::-1]
    corr_len = n + (2*n - 1) - 1  # = 3n-2
