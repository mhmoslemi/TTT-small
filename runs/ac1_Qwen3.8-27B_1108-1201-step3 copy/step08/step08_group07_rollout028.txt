Now, for the ISA method, let me be more careful:

The ISA method works in the frequency domain. The key equation is:
- `r = a*a` (linear autocorrelation, length 2n-1)
- Zero-pad r to length L (where L >= 2n-1, typically next power of 2)
- `R = DFT(r_padded)` = `|A|^2` where `A = DFT(a_padded)` and `a_padded` is a zero-padded to length L
- Target: `r_target` (length 2n-1, zero-padded to L)
- Error: `e = r_padded - r_target_padded`
- `E = DFT(e)`
- Update: `A_new = A - beta * E / (2 * conj(A))`
- `a_new = IDFT(A_new)[:n]`
- Project: `a_new = max(a_new, 0)`, normalize.

The step size `beta` should be chosen carefully. If too large, it can oscillate. If too small, convergence is slow.

A good adaptive choice: `beta = alpha / max(|E| / |2*conj(A)|)` (normalized step).

        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            ft_decreased = 0
        else:
            ft_decreased += 1

        if ft_decreased > 5000:
            # Reset momentum, try fresh
            v = np.zeros(n)
            ft_decreased = 0
            gamma0 = max(gamma0 * 0.5, 1e-5)

        if verbose and iters % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} it={iters}, best={best_score:.8f}, gamma={gamma:.6f}, t={elapsed:.0f}s")

    return best_a, best_score


def isa_iteration(a, target='flat', beta=0.5, fft_size=None):
    """One ISA (Iterative Spectral Aberration) step."""
    n = len(a)
    size = 2 * n - 1
    if fft_size is None:
        fft_size = 1 << (size - 1).bit_length()

    A = rfft(a, fft_size)
    # Current circular autocorrelation (= linear since fft_size >= 2n-1)
    r_circ = irfft(np.abs(A) ** 2, fft_size)

    # Target
    total_r = r_circ[:size].sum()  # should = (sum a)^2 = 1
    if target == 'flat':
        r_target_full = np.zeros(fft_size)
        r_target_full[:size] = total_r / size
    elif target == 'tapered':
        # Tapered target: weighted toward center
        r_target_full = np.zeros(fft_size)
        w = np.ones(size)
        # Mild taper to allow center to be slightly higher
        r_target_full[:size] = w / w.sum() * total_r
    else:
        r_target_full = np.zeros(fft_size)
        r_target_full[:size] = total_r / size

    # Error
    e = r_circ - r_target_full
    E = rfft(e, fft_size)

    # Update
    A_conj = np.conj(A)
    mask = np.abs(A_conj) > 1e-12
    A_new = A.copy().astype(complex)
    A_new[mask] = A[mask] - beta * E[mask] / (2.0 * A_conj[mask])

    a_new = irfft(A_new, fft_size)[:n]
    a_new = normalize(a_new)
    return a_new


def isa_optimize(a0, time_limit, beta=0.3, verbose=True, label="ISA"):
    """ISA optimization loop with adaptive step size."""
    n = len(a0)
    a = normalize(a0.astype(np.float64).copy())
    best_a = a.copy()
    best_score = compute_score_norm(a)
    t0 = time.monotonic()
    iters = 0
    stall = 0
    fft_size = 1 << (2 * n - 2).bit_length()

    while time.monotonic() - t0 < time_limit:
        iters += 1
        a = isa_iteration(a, target='flat', beta=beta, fft_size=fft_size)
        score = compute_score_norm(a)

        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 100:
            # Reduce beta, try different target
            beta = max(beta * 0.8, 0.01)
            stall = 0
            # Perturb
            rng = np.random.default_rng(iters)
            a = best_a + rng.standard_normal(n) * 0.001
            a = normalize(a)

        if verbose and iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} it={iters}, best={best_score:.8f}, beta={beta:.4f}, t={elapsed:.0f}s")

    return best_a, best_score


def coordinate_descent(a0, time_limit, verbose=True, label="CD"):
    """Coordinate descent: transfer mass between pairs to reduce peak autocorrelation."""
    n = len(a0)
    a = normalize(a0.astype(np.float64).copy())
    best_a = a.copy()
    best_score = compute_score_norm(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    iters = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False

        # Focus on the peak lag
        conv = fast_convolve(a)
        k_peak = conv.argmax()

        # The peak is at k_peak. To reduce it, we should reduce a[j]*a[k_peak-j]
        # Best strategy: find the pair (j, k_peak-j) contributing most to the peak
        # and transfer mass from the larger to a position that doesn't contribute.

        # Strategy 1: Transfer from peak-contributing positions to edge positions
        contribs = np.zeros(n)
        for j in range(n):
            k2 = k_peak - j
            if 0 <= k2 < n:
                contribs[j] = a[j] * a[k2]

        # Find the position with largest contribution
        i_don = contribs.argmax()
        a_don = a[i_don]
        if a_don < 1e-6:
            # Fall back to random pairs
            for _ in range(100):
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j or a[i] < 1e-6:
                    continue
                for frac in [0.1, 0.2, 0.3, 0.5]:
                    t = frac * a[i]
                    a_trial = a.copy()
                    a_trial[i] -= t
                    a_trial[j] += t
                    a_trial = normalize(a_trial)
                    sc = compute_score_norm(a_trial)
                    if sc < best_score - 1e-12:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
        else:
            # Transfer from i_don to a position that minimally contributes to peak
            # Prefer positions far from k_peak-j for all j
            for frac in [0.05, 0.1, 0.15, 0.2]:
                t = frac * a_don
                # Try multiple target positions
                best_local = best_score
                best_j = -1
                for _ in range(30):
                    j = rng.integers(0, n)
                    if j == i_don:
                        continue
                    a_trial = a.copy()
                    a_trial[i_don] -= t
                    a_trial[j] += t
                    a_trial = normalize(a_trial)
                    sc = compute_score_norm(a_trial)
                    if sc < best_local:
                        best_local = sc
                        best_j = j
                if best_j >= 0:
                    a[i_don] -= t
                    a[best_j] += t
                    a = normalize(a)
                    best_score = best_local
                    best_a = a.copy()
                    improved = True
                    break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                # Perturb and reset
                a = best_a + rng.standard_normal(n) * 0.003
                a = normalize(a)
                no_improve = 0

        if verbose and iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} it={iters}, best={best_score:.8f}, t={elapsed:.0f}s")

    return best_a, best_score


def lbgfs_smooth(a0, time_limit, T=500.0, verbose=True):
    """L-BFGS on smoothed log-sum-exp objective in log-space."""
    n = len(a0)
    z0 = np.log(np.maximum(a0, 1e-30))

    def softmax_from_z(z):
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        s = exp_z.sum()
        if s < 1e-300:
            return np.ones(len(z)) / len(z)
        return exp_z / s

    def func(z):
        a = softmax_from_z(z)
        conv = fast_convolve(a)
        shifted = conv / T
        max_shifted = shifted.max()
        lse = max_shifted + np.log(np.exp(shifted - max_shifted).sum())
        return 2.0 * n / T * np.exp(T * lse)

    def jac(z):
        a = softmax_from_z(z)
        conv = fast_convolve(a)
        shifted = conv / T
        max_shifted = shifted.max()
        w = np.exp(shifted - max_shifted)
        w = w / w.sum()
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
        S = float(np.dot(grad_a, a))
        grad_z = a * (grad_a - S)
        return 2.0 * n / T * grad_z

    t_start = time.monotonic()
    best_a = normalize(a0.copy())
    best_score = compute_score_norm(best_a)

    def func_t(z):
        if time.monotonic() - t_start > time_limit:
            return 1e15
        return func(z)

    def jac_t(z):
        if time.monotonic() - t_start > time_limit:
            return np.zeros_like(z)
        return jac(z)

    try:
        res = minimize(fun=func_t, x0=z0, jac=jac_t, method='L-BFGS-B',
                       options={'maxiter': 3000, 'ftol': 1e-16, 'gtol': 1e-12,
                                'maxfun': 15000, 'disp': False})
        a_out = softmax_from_z(res.x)
        score = compute_score_norm(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except Exception:
        pass

    return best_a, best_score


def multi_subgradient(a0, time_limit, verbose=True):
    """Multiple subgradient runs with different initial gamma and momentum."""
    n = len(a0)
    best_a = normalize(a0.copy())
    best_score = compute_score_norm(best_a)
    t0 = time.monotonic()

    for gamma0 in [1.0, 0.3, 0.1]:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 10:
            break
        t_sub = min(rem * 0.35, 300.0)
        a_obj, sc_obj = subgradient_nesterov(best_a, t_sub, verbose=False,
                                             label=f"SGD-g{gamma0}")
        # Override gamma0 by patching: actually let's just call with different params
        if sc_obj < best_score:
            best_score = sc_obj
            best_a = a_obj.copy()
            if verbose:
                print(f"  SGD(g={gamma0}) improved to {best_score:.8f}")

    return best_a, best_score


def try_length(a_best, new_n, time_limit, verbose=True):
    """Interpolate to new length and optimize."""
    a_interp = interpolate_to(a_best, new_n)
    a_opt, sc_opt = subgradient_nesterov(a_interp, time_limit * 0.4, verbose=False, label=f"n{new_n}")
    if sc_opt < compute_score_norm(a_best):
        return a_opt, sc_opt
    return a_best, compute_score_norm(a_best)


def interpolate_to(a, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_new = np.interp(new_x, old_x, a_norm)
    return normalize(a_new)


def check_and_update(a, label, best_seq, best_score, verbose=True):
    """Evaluate and update best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining subgradient, ISA, and coordinate descent.
    Exploits the insight that asymmetric sequences reduce the linear autocorrelation
    center tap below sum(a^2), enabling scores < 2.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 768
        parent = np.ones(n_init) / n_init
        best_seq = parent.copy()
        best_score = compute_score_norm(parent)
        print(f"Initialized uniform: n={n_init}, score={best_score:.8f}")

    n_parent = len(parent)

    # === Phase 1: Multi-shape initialization + quick subgradient ===
    print(f"\n=== Phase 1: Multi-shape init (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()

    # Generate diverse initial shapes
    shapes = {}
    n = n_parent
    x = np.linspace(0, 1, n)

    shapes['uniform'] = np.ones(n) / n
    shapes['ramp_up'] = normalize(np.linspace(0.1, 1.0, n))
    shapes['ramp_down'] = normalize(np.linspace(1.0, 0.1, n))
    shapes['quadratic_up'] = normalize((x ** 2 + 0.01))
    shapes['quadratic_down'] = normalize(((1 - x) ** 2 + 0.01))
    shapes['sine'] = normalize(np.sin(np.pi * x) ** 2 + 0.01)
    shapes['cosine_taper'] = normalize(0.5 * (1 + np.cos(2 * np.pi * x)))
    shapes['exp_ramp'] = normalize(np.exp(2 * x))
    shapes['anti_sym'] = normalize(np.abs(x - 0.3) + 0.01)  # peak off-center
    shapes['bimodal'] = normalize(np.exp(-50 * (x - 0.2) ** 2) + np.exp(-50 * (x - 0.8) ** 2))

    # Quick subgradient on each shape (short budget)
    t_per_shape = min(rem * 0.04, 30.0)
    for name, shape in shapes.items():
        rem = deadline - time.monotonic()
        if rem < 15:
            break
        t_sub = min(t_per_shape, rem * 0.4)
        a_opt, sc_opt = subgradient_nesterov(shape, t_sub, verbose=False, label=f"init_{name}")
        best_seq, best_score = check_and_update(a_opt, f"shape_{name}", best_seq, best_score, verbose=False)

    # Evaluate parent with subgradient too
    rem = deadline - time.monotonic()
    if rem > 30:
        t_sub = min(rem * 0.15, 120.0)
        print(f"  Subgradient on parent, budget={t_sub:.0f}s")
        a_opt, sc_opt = subgradient_nesterov(best_seq, t_sub, verbose=True, label="SGD-parent")
        best_seq, best_score = check_and_update(a_opt, "sgd_parent", best_seq, best_score)

    # === Phase 2: ISA optimization ===
    print(f"\n=== Phase 2: ISA (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_isa = min(rem * 0.20, 200.0)
        print(f"  ISA budget={t_isa:.0f}s")
        for beta_init in [0.5, 0.2, 0.1]:
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            t_sub_isa = min(rem * 0.3, 80.0)
            a_opt, sc_opt = isa_optimize(best_seq, t_sub_isa, beta=beta_init,
                                         verbose=False, label=f"ISA-b{beta_init}")
            best_seq, best_score = check_and_update(a_opt, f"isa_b{beta_init}", best_seq, best_score)

    # === Phase 3: Extended subgradient with momentum ===
    print(f"\n=== Phase 3: Extended SGD (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_sgd = min(rem * 0.30, 400.0)
        print(f"  SGD budget={t_sgd:.0f}s")
        a_opt, sc_opt = subgradient_nesterov(best_seq, t_sgd, verbose=True, label="SGD-ext")
        best_seq, best_score = check_and_update(a_opt, "sgd_ext", best_seq, best_score)

    # === Phase 4: L-BFGS smoothing ===
    print(f"\n=== Phase 4: L-BFGS (best={best_score:.8f}) ===")
    for T_val in [200.0, 500.0, 1000.0]:
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t_lb = min(rem * 0.12, 80.0)
        print(f"  L-BFGS T={T_val:.0f}, budget={t_lb:.0f}s")
        a_opt, sc_opt = lbgfs_smooth(best_seq, t_lb, T=T_val, verbose=False)
        best_seq, best_score = check_and_update(a_opt, f"lbgfs_T{T_val:.0f}", best_seq, best_score)

    # === Phase 5: Coordinate descent ===
    print(f"\n=== Phase 5: Coordinate Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_cd = min(rem * 0.15, 120.0)
        print(f"  CD budget={t_cd:.0f}s")
        a_opt, sc_opt = coordinate_descent(best_seq, t_cd, verbose=True, label="CD")
        best_seq, best_score = check_and_update(a_opt, "coord_desc", best_seq, best_score)

    # === Phase 6: Try nearby lengths ===
    print(f"\n=== Phase 6: Nearby Lengths (best={best_score:.8f}) ===")
    for alt_n in [n_parent - 2, n_parent + 2, n_parent - 4, n_parent + 4]:
        if alt_n < 64 or alt_n > 2048:
            continue
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t_alt = min(rem * 0.15, 60.0)
        a_interp = interpolate_to(best_seq, alt_n)
        a_opt, sc_opt = subgradient_nesterov(a_interp, t_alt * 0.5, verbose=False, label=f"n{alt_n}")
        # Also try L-BFGS on the interpolated
        rem2 = deadline - time.monotonic()
        if rem2 > 10:
            a_opt2, sc_opt2 = lbgfs_smooth(a_opt, min(rem2 * 0.3, 30.0), T=500.0, verbose=False)
            if sc_opt2 < sc_opt:
                a_opt = a_opt2
                sc_opt = sc_opt2
        best_seq, best_score = check_and_update(a_opt, f"n{alt_n}", best_seq, best_score)

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        # Quick SGD
        t_final = min(rem * 0.3, 60.0)
        a_opt, sc_opt = subgradient_nesterov(best_seq, t_final, verbose=False, label="final_sgd")
        best_seq, best_score = check_and_update(a_opt, "final_sgd", best_seq, best_score)

    # Final CD polish
    rem = deadline - time.monotonic()
    if rem > 20:
        t_cd2 = min(rem * 0.4, 60.0)
        a_opt, sc_opt = coordinate_descent(best_seq, t_cd2, verbose=False, label="final_cd")
        best_seq, best_score = check_and_update(a_opt, "final_cd", best_seq, best_score)

    # === Final verification ===
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
            print(f"  Verification improved: {fs:.8f}")

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
