For the projected gradient method:
1. Compute a, compute conv = a*a, find k* = argmax(conv)
2. Compute gradient g = 2*(P_{k*} a) (i.e., g[j] = 2*a[k*-j] if valid)
3. Update: a_new = a - lr * g
4. Project: a_new = a_new - (sum(a_new) - 1)/n (to enforce sum=1)
5. Project: a_new = max(a_new, 0) (to enforce non-negativity)
6. Renormalize: a_new = a_new / sum(a_new) (to fix any drift)
7. Repeat

This is simple and should work! But it might be slow because it only uses one peak at a time.

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute score for normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    # Sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def soft_max_and_grad(a, T):
    """Compute soft max of auto-convolution and subgradient."""
    n = len(a)

    size = 2 * n - 1

2. Projected gradient with soft-max smoothing at multiple temperatures
3. Multiplicative weights (mirror descent) variant

import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max())


def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def soft_max_grad(a, T):

    conv = fast_convolve(a)

    w = np.exp(shifted - mc)
    w = w / w.sum()
    sm_val = T * (mc + np.log(np.sum(np.exp(shifted - mc))))
    
    # grad[j] = 2 * sum_{i=0}^{n-1} a[i] * w[j+i] (for j+i in [0, 2n-2])
    # Use FFT: grad[j] = 2 * (a * w_rev)[2n-2-j]
    w_rev = w[::-1]
    total_len = n + size - 1  # = 3n-3
    fft_size = 1 << (total_len - 1).bit_length()
    A = rfft(a, fft_size)
    WR = rfft(w_rev, fft_size)
    conv_aw = irfft(A * WR, fft_size)[:total_len]
    # grad[j] = 2 * conv_aw[2n-2-j] for j=0,...,n-1
    grad = 2.0 * conv_aw[2*n-2 : 2*n-2-n : -1]
    
    return sm_val, w, grad


def pg_additive(a0, time_limit, T_seq=None, lr_scale=1.0, verbose=False):
    """Projected gradient with soft-max, additive updates."""
    a = project_simplex(a0.copy())
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)
    t_start = time.monotonic()
    
    if T_seq is None:
        T_seq = [500, 200, 100, 50, 20, 10, 5, 2, 1, 0.5, 0.2, 0.1]
    
    total_iters = 0
    for T in T_seq:
        # Adaptive: run until time budget for this T is used
        n_iters_per_T = 200
        for it in range(n_iters_per_T):
            if time.monotonic() - t_start > time_limit:
                return best_a, best_score
            
            lr = lr_scale / (T + 1.0)  # Learning rate inversely proportional to T
            # Actually, let's use a fixed schedule
            lr = lr_scale * 0.1 / (it + 1) ** 0.5
            
            sm_val, w, grad = soft_max_grad(a, T)
            a_new = a - lr * grad
            a_new = project_simplex(a_new)
            
            sc = compute_score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            a = a_new
            total_iters += 1
    
    return best_a, best_score


def pg_multiplicative(a0, time_limit, T_seq=None, verbose=False):
    """Mirror descent / multiplicative weights with soft-max."""
    a = project_simplex(a0.copy())
    a = np.maximum(a, 1e-15)  # Ensure no zeros for log
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)
    t_start = time.monotonic()
    
    if T_seq is None:
        T_seq = [500, 200, 100, 50, 20, 10, 5, 2, 1, 0.5, 0.2, 0.1]
    
    for T in T_seq:
        n_iters_per_T = 200
        for it in range(n_iters_per_T):
            if time.monotonic() - t_start > time_limit:
                return best_a, best_score
            
            lr = 0.5 / (T + 1.0)
            sm_val, w, grad = soft_max_grad(a, T)
            
            a_new = a * np.exp(-lr * grad)
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-30:
                a_new = a.copy()
            else:
                a_new = a_new / s
            
            sc = compute_score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            a = a_new
    
    return best_a, best_score


def lp_fp(a0, time_limit, verbose=False):
    """LP fictitious play with geometric mean updates."""
    f = project_simplex(a0.copy())
    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)
    t0 = time.monotonic()
    
    # Build constraint matrix once (it depends on f, so we rebuild each iteration)
    for k in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        M = fast_convolve(f).max()
        
        # Build C: C[k,j] = f[k-j] if 0 <= k-j < n else 0
        C = np.zeros((2 * n - 1, n))
        for j in range(n):
            C[j:j+n, j] = f
        
        c_vec = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        res = linprog(c_vec, A_ub=C, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
        if res.status != 0:
            break
        g = np.maximum(res.x, 0)
        
        if g.sum() < 1e-15:
            break
        g = g / g.sum()
        
        t_step = 2.0 / (k + 2)
        f_safe = np.maximum(f, 1e-30)
        g_safe = np.maximum(g, 1e-30)
        log_new = (1 - t_step) * np.log(f_safe) + t_step * np.log(g_safe)
        log_new -= log_new.max()
        a_new = np.exp(log_new)
        a_new = a_new / a_new.sum()
        
        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_f = a_new.copy()
        
        f = a_new
        
        if verbose and k % 100 == 0:
            print(f"      LP iter={k}, M={M:.6e}, best={best_score:.8f}, t={time.monotonic()-t0:.0f}s")
    
    return best_f, best_score


def make_ramp(n, power=1.0):
    """Create a power-law ramp: a[i] = (i+1)^power, normalized."""
    i = np.arange(1, n+1, dtype=np.float64)
    a = i ** power
    return a / a.sum()


def make_ramp_zero(n, power=1.0):
    """Create a power-law ramp starting from 0: a[i] = i^power, normalized."""
    i = np.arange(0, n, dtype=np.float64)
    a = i ** power
    s = a.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return a / s
