import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(f, g):
    """Cross-convolution f*g via FFT."""
    size = len(f) + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def get_active_constraints(b, n_constraints, margin=0.0):
    """Get indices of top constraints (peaks) with optional margin."""
    L = len(b)
    if n_constraints >= L:
        return list(range(L))
    # Get indices where b > margin * max(b)
    thresh = margin * b.max()
    candidates = np.where(b > thresh)[0]
    if len(candidates) >= n_constraints:
        # Take top n_constraints from candidates
        vals = b[candidates]
        top_in_cand = np.argpartition(vals, -n_constraints)[-n_constraints:]
        return sorted(candidates[top_in_cand].tolist())
    else:
        # Not enough above threshold, fill from remaining
        remaining = np.array([i for i in range(L) if i not in set(candidates.tolist())])
        need = n_constraints - len(candidates)
        if len(remaining) >= need:
            vals = b[remaining]
            top = np.argpartition(vals, -need)[-need:]
            return sorted(candidates.tolist() + remaining[top].tolist())
        else:
            return sorted(candidates.tolist() + remaining.tolist())


def solve_lp_direction(f, n, active_list, M_bound, time_limit=30.0):
    """
    Solve LP: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M_bound for k in active_list.
    Returns g or None if infeasible.
    """
    t0 = time.monotonic()
    K = len(active_list)
    
    # Build constraint matrix
    A = np.zeros((K, n))
    for row_i in range(K):
        k = active_list[row_i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]
    
    g_var = cp.Variable(n)
    prob = cp.Problem(
        cp.Maximize(cp.sum(g_var)),
        [g_var >= 0, A @ g_var <= M_bound]
    )
    
    for solver in [cp.PDLP, cp.SCS, cp.CLARABEL]:
        try:
            prob.solve(solver=solver, max_iters=1000000)
            if g_var.value is not None:
                g = np.maximum(g_var.value, 0.0)
                if not np.any(np.isnan(g)):
                    return g
        except Exception:
            continue
    
    return None


def golden_section_line_search(f, direction, time_limit=10.0):
    """Golden section search for best step size t in [0, 1]."""
    t0 = time.monotonic()
    n = len(f)
    
    def peak_at(t):
        f_trial = (1.0 - t) * f + t * direction
        f_trial = np.maximum(f_trial, 0.0)
        s = f_trial.sum()
        if s < 1e-10:
            return 1e10
        f_trial = f_trial / s
        return fast_convolve(f_trial).max()
    
    # Coarse search
    t_candidates = [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7]
    best_t = 0.05
    best_obj = peak_at(0.05)
    for t in t_candidates:
        if time.monotonic() - t0 > time_limit:
            break
        obj = peak_at(t)
        if obj < best_obj:
            best_obj = obj
            best_t = t
    
    # Golden section refinement
    idx = t_candidates.index(best_t)
    lo = t_candidates[max(0, idx - 1)] if idx > 0 else 0.0
    hi = t_candidates[min(len(t_candidates) - 1, idx + 1)] if idx < len(t_candidates) - 1 else 1.0
    
    gr = (np.sqrt(5) - 1) / 2
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = peak_at(c)
    fd = peak_at(d)
    
    for _ in range(25):
        if time.monotonic() - t0 > time_limit:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = peak_at(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = peak_at(d)
        if hi - lo < 1e-6:
            break
    
    best_t = (lo + hi) / 2
    return best_t, peak_at(best_t)

def subgradient_polish(f_init, time_limit, n_top=50):
    """Subgradient optimization with momentum."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()

    lr = 0.002
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(123)
    
    while time.monotonic() - t0 < time_limit:
        batch = min(50, max(10, int((time_limit - (time.monotonic() - t0)) * 3)))
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit:
                break
            it += 1
            b = fast_convolve(f)
            
            # Use softmax weighting over top peaks
            if n_top < len(b):
                top_idx = np.argpartition(b, -n_top)[-n_top:]
            else:
                top_idx = np.arange(len(b))
            
            # Equal weight on top peaks
            w = np.zeros(len(b))
            w[top_idx] = 1.0 / len(top_idx)
            
            # Compute gradient
            g = np.zeros(n)
            for k in top_idx:
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                idx = np.arange(lo, hi + 1)
                g[idx] += w[k] * f[k - idx]
            
            v = 0.9 * v - lr * g
            f_new = project_simplex(f + v)
            f = f_new
        
        obj = fast_convolve(f).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
            lr = min(lr * 1.02, 0.01)
        else:
            no_improve += 1
            if no_improve > 3:
                lr *= 0.9
                no_improve = 0
            if no_improve > 15:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                v = np.zeros(n)
                no_improve = 0
        
        if it % 300 == 0:
            print(f"    sg iter={it} obj={best_obj:.10f} "
                  f"score={2.0*n*best_obj:.6f} t={time.monotonic()-t0:.0f}s")
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, maintaining normalization."""
    old_x = np.linspace(0.0, 1.0, len(f))
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return f_new / s
