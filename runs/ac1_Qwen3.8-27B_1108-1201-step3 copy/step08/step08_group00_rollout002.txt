def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def lbgfs_refine(a0, time_limit, T_start=100.0, T_end=0.3, maxiter=8000):
    """L-BFGS with LogSumExp smoothing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    def make_func(T_val):
        def func(z):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(z):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z)
            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    n_stages = 6
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(50, maxiter // n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxfun': iters_per_stage * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=50000, use_active=False, n_active=30):
    """Nesterov dual averaging LP refinement with optional active set restriction."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        if use_active and it > 10:
            b_y = fast_convolve(y)
            threshold = b_y.max() - 0.1 * b_y.max()
            active = np.where(b_y >= threshold)[0]
            if len(active) > n_active:
                active = np.argsort(b_y)[::-1][:n_active]
            b_new, M_b = solve_active_lp(y, active.tolist(), solver)
        else:
            b_new, M_b = solve_lp_min_max(y, solver)

        if b_new is None:
            b_new, M_b = solve_lp_min_max(y, solver)
            if b_new is None:
                continue

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if it % 500 == 0:
            if best_score < prev_best - 1e-7:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            if stall_count >= 8:
                rng_local = np.random.default_rng(it * 7 + 13)
                noise = rng_local.random(n) * 0.02
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0

        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def adaptive_mirror_descent(a0, time_limit, lr_init=0.05, n_steps=100000):
    """Adaptive mirror descent with subgradient oracle."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    z = z_from_a(a)
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    best_lr = lr_init

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        # Use top peaks with a gap-based criterion
        sorted_b = np.sort(b)[::-1]
        gap_threshold = sorted_b[0] - 0.05 * sorted_b[0]
        active = np.where(b >= gap_threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Subgradient: average of contributions from active peaks
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        # Mirror descent update
        z_new = z - lr * g
        a_new = softmax_from_z(z_new)
        score_new = 2.0 * n * fast_convolve(a_new).max()

        if score_new < best_score - 1e-15:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            z = z_new
            no_improve = 0
            best_lr = lr
        else:
            no_improve += 1

        if no_improve >= 50:
            lr *= 0.8
            no_improve = 0
        elif no_improve == 0 and lr < best_lr * 1.5:
            lr = min(lr * 1.05, best_lr * 2.0)

        lr = max(lr, 1e-10)
        lr = min(lr, 0.2)

        if step % 30000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD {elapsed:.0f}s: score={best_score:.6f} lr={lr:.6f}")

    return best_a, best_score


def local_polish(a0, time_limit, n_steps=20000):
    """Coordinate-wise local optimization with exact evaluation."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]

        # Find coordinates that contribute most to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_n = min(8, len(idx))
        top_coords = idx[np.argsort(contribs)[::-1][:top_n]]

        improved = False
        # Try reducing each top coordinate by a small amount
        for i in top_coords:
            if a[i] < 1e-14:
                continue
            # Try several reduction factors
            for factor in [0.995, 0.99, 0.985, 0.98, 0.97, 0.95]:
                new_val = a[i] * factor
                diff = a[i] - new_val
                if diff < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                # Redistribute the difference to other coordinates proportionally
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] += diff * a_trial[mask] / others_sum
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try increasing a small coordinate (to spread mass away from peak contributors)
            small_coords = np.argsort(a)[:5]
            for i in small_coords:
                for factor in [1.1, 1.2, 1.5, 2.0]:
                    a_trial = a.copy()
                    a_trial[i] *= factor
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            # Try pair adjustments
            n_pairs = 0
            for i_idx in range(len(top_coords) // 2):
                i = top_coords[i_idx]
                j = top_coords[-(i_idx + 1)] if i_idx < len(top_coords) // 2 else top_coords[0]
                if i == j:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.97, 0.95]:
                    for f2 in [1.03, 1.05]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-15:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            n_pairs += 1
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            # Check convergence
            if step > 1000:
                break
            # Otherwise add tiny perturbation
            rng_local = np.random.default_rng(step * 31 + 7)
            a = a + 0.001 * rng_local.standard_normal(n)
            a = np.maximum(a, 0)
            a = a / a.sum()

    return best_a, best_score


def structured_init(n, rng, style=0):
    """Create structured initial sequence with different styles."""
    if style == 0:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x / 2) ** 2
    elif style == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif style == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif style == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-6 * x * (1 - x))
    elif style == 4:
        x = np.linspace(0, 1, n)
        a = (0.5 + 0.5 * np.cos(2 * np.pi * x)) ** 2
    elif style == 5:
        # Triangular (Fejer-like)
        x = np.linspace(0, 1, n)
        a = 1 - np.abs(2 * x - 1)
        a = a ** 1.5
    elif style == 6:
        # Bimodal
        x = np.linspace(0, 1, n)
        a = np.exp(-8 * (x - 0.3) ** 2) + np.exp(-8 * (x - 0.7) ** 2)
    else:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. Load parent / generate candidates at multiple lengths
    2. Quick screening to find best length + initialization
    3. Extended Nesterov LP optimization
    4. L-BFGS LogSumExp refinement
    5. Adaptive mirror descent
    6. Local coordinate polish
    7. Final LP resync
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 960
        best_seq = structured_init(n_init, rng, style=2)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    n_best = len(best_seq)

    # === Phase 1: Extended Nesterov LP on initial length (long run) ===
    print(f"\n=== Phase 1: Nesterov LP n={n_best} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 540.0)
    if t_alloc > 40:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=200000, use_active=True, n_active=40)
        try_update(a_lp, "nlp1")

    # === Phase 2: L-BFGS refinement ===
    print(f"\n=== Phase 2: L-BFGS (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 360.0)
    if t_alloc > 30:
        print(f"  L-BFGS T: 150->0.2, budget={t_alloc:.0f}s")
        a_lb = lbgfs_refine(best_seq, t_alloc, T_start=150.0, T_end=0.2, maxiter=15000)
        try_update(a_lb, "lbgfs1")

    # === Phase 3: Second Nesterov LP with active set ===
    print(f"\n=== Phase 3: Nesterov LP r2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 270.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=100000, use_active=True, n_active=25)
        try_update(a_lp2, "nlp2")

    # === Phase 4: Adaptive mirror descent ===
    print(f"\n=== Phase 4: Mirror Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 180.0)
    if t_alloc > 15:
        a_md, _ = adaptive_mirror_descent(best_seq, t_alloc, lr_init=0.04, n_steps=200000)
        try_update(a_md, "md")

    # === Phase 5: Local coordinate polish ===
    print(f"\n=== Phase 5: Local Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 144.0)
    if t_alloc > 10:
        a_pol, _ = local_polish(best_seq, t_alloc, n_steps=50000)
        try_update(a_pol, "polish1")

    # === Phase 6: Try nearby lengths ===
    print(f"\n=== Phase 6: Nearby Lengths (best={best_score:.6f}) ===")
    test_lengths = []
    for dl in [-120, -80, -40, 40, 80, 120, 160, -160]:
        nl = n_best + dl
        if 384 <= nl <= 2048 and nl != n_best:
            test_lengths.append(nl)
    test_lengths = sorted(set(test_lengths))

    for tl in test_lengths[:3]:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_total = min(rem * 0.20, 180.0)
        if t_total < 30:
            break
        print(f"  n={tl}, budget={t_total:.0f}s")

        # Interpolate from best
        init_tl = interpolate_sequence(best_seq, tl)
        a_tl, _ = nesterov_lp_refine(init_tl, t_total * 0.7, solver=solver, n_max_iters=30000, use_active=True, n_active=30)
        try_update(a_tl, f"lp_n{tl}")

        rem = deadline - time.monotonic()
        if rem > 20:
            a_tl2 = lbgfs_refine(best_seq if len(best_seq) == tl else a_tl, min(rem * 0.25, 60.0),
                                 T_start=50.0, T_end=1.0, maxiter=4000)
            try_update(a_tl2, f"lb_n{tl}")

    # === Phase 7: Perturbation escape ===
    print(f"\n=== Phase 7: Perturbation (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 50:
        n_cur = len(best_seq)
        # Try different perturbation scales
        for scale in [0.03, 0.08]:
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            t_pert = min(rem * 0.20, 120.0)
            if t_pert < 20:
                break
            perturbed = best_seq + scale * rng.standard_normal(n_cur)
            perturbed = np.maximum(perturbed, 0)
            perturbed = perturbed / perturbed.sum()
            a_pert, _ = nesterov_lp_refine(perturbed, t_pert, solver=solver, n_max_iters=20000, use_active=True, n_active=30)
            try_update(a_pert, f"pert_{scale}")

    # === Phase 8: Final intensive optimization ===
    print(f"\n=== Phase 8: Final Push (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        # Long LP run
        t_fin_lp = min(rem * 0.45, 200.0)
        if t_fin_lp > 30:
            a_fin, _ = nesterov_lp_refine(best_seq, t_fin_lp, solver=solver, n_max_iters=80000, use_active=True, n_active=20)
            try_update(a_fin, "final_lp")

    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin_lb = min(rem * 0.35, 100.0)
        if t_fin_lb > 15:
            a_fin2 = lbgfs_refine(best_seq, t_fin_lb, T_start=10.0, T_end=0.05, maxiter=6000)
            try_update(a_fin2, "final_lb")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin_pol = min(rem * 0.4, 60.0)
        if t_fin_pol > 10:
            a_fin3, _ = local_polish(best_seq, t_fin_pol, n_steps=30000)
            try_update(a_fin3, "final_pol")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
