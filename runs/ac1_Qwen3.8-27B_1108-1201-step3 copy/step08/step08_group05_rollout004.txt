import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(f, g):
    """Cross-convolution f*g via FFT."""
    size = len(f) + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f, n, indices):
    """Build constraint matrix A where A[k, j] = f[k-j] if valid, else 0."""
    K = len(indices)
    A = np.zeros((K, n))
    for row_i in range(K):
        k = int(indices[row_i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]
    return A


def solve_lp(f, n, active_indices, M, solver_list=None):
    """Solve the LP: max sum(g) s.t. g>=0, A@g <= M."""
    if solver_list is None:
        solver_list = [cp.GLOP, cp.PDLP, cp.SCS]
    
    K = len(active_indices)
    if K == 0:
        return None
    
    A = build_lp_matrix(f, n, active_indices)
    
    g_var = cp.Variable(n)
    prob = cp.Problem(
        cp.Maximize(cp.sum(g_var)),
        [g_var >= 0, A @ g_var <= M]
    )
    
    for solver in solver_list:
        try:
            prob.solve(solver=solver)
            if g_var.value is not None:
                g = np.maximum(g_var.value, 0.0)
                if not (np.any(np.isnan(g)) or np.any(np.isinf(g))):
                    return g
        except Exception:
            continue
    
    return None

def peak_shave(f_init, time_limit, n_peaks=8, label="shave"):
    """Direct peak shaving: identify top peaks and reduce contributing elements."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    rng = np.random.default_rng(999)
    stuck = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(f)
        M = b.max()
        
        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 100:
                break
        
        # Find top peaks
        k_peaks = min(n_peaks, len(b))
        top_idx = np.argpartition(b, -k_peaks)[-k_peaks:]
        
        # Compute gradient (which elements to reduce)
        grad = np.zeros(n)
        for k in top_idx:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            grad[idx] += f[k - idx]
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm
        
        # Try different step sizes
        best_alpha = None
        best_trial = best_obj
        for alpha in [0.001, 0.003, 0.005, 0.01, 0.02, 0.03, 0.05]:
            f_trial = f - alpha * grad
            f_trial = project_simplex(f_trial)
            obj = fast_convolve(f_trial).max()
            if obj < best_trial - 1e-15:
                best_trial = obj
                best_alpha = alpha
        
        if best_alpha is None:
            # Add small noise to escape
            f = project_simplex(f + rng.standard_normal(n) * 0.001)
            continue
        
        f = f - best_alpha * grad
        f = project_simplex(f)
    
    return best_f, best_obj


def spectral_flatness_step(f, n_target_flat=50, label="spectral"):
    """One step of spectral flatness improvement.
    Modify the frequency response to be flatter, then project back to non-negative.
    """
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    F = rfft(f, fft_size)
    
    # Compute current power spectrum
    P = np.abs(F) ** 2
    
    # Target: make P as flat as possible (excluding DC)
    # Use a geometric mean as a soft target
    P_nonzero = P[P > 1e-15]
    target_power = np.exp(np.mean(np.log(P_nonzero + 1e-30)))
    
    # Blend current and target
    P_target = 0.7 * P + 0.3 * target_power
    P_target[0] = P[0]  # Keep DC (which is sum(f)^2 = 1)
    
    # Take square root to get magnitude (keep phase)
    mag_new = np.sqrt(P_target)
    phase = np.angle(F)
    F_new = mag_new * np.exp(1j * phase)
    
    # Inverse FFT
    f_new = irfft(F_new, fft_size)[:n]
    
    # Project to non-negative simplex
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = f.copy()
    
    return f_new


def spectral_optimize(f_init, time_limit, label="spectral"):

def subgradient_polish(f_init, time_limit, lr_init=0.003, n_top=30, label="sg"):

def interpolate_sequence(f_old, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(f_old)
    old_x = np.linspace(0.0, 1.0, n_old)
    new_x = np.linspace(0.0, 1.0, n_new)
    f_new = np.interp(new_x, old_x, f_old)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    return f_new

            n_b = len(best_seq)
            rng = np.random.default_rng(42)
            t_start = time.monotonic()
            
            for trial in range(50):
                if time.monotonic() - t_start > t8:
                    break
                
                # Random small perturbation
                noise_scale = 0.0005 * (1.0 - trial / 50.0)
                f_pert = best_seq + rng.standard_normal(n_b) * noise_scale
                f_pert = project_simplex(f_pert)
                
                score_pert = float(evaluate_sequence(f_pert.tolist()))
                if np.isfinite(score_pert) and score_pert < best_score:
                    best_score = score_pert
                    best_seq = f_pert.copy()
                    print(f"    pert {trial}: score={best_score:.10f} (new best!)")
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
    
    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    print(f"Target: 1.5027000000")
    print(f"Gap: {best_score - 1.5027:.10f}")
    if best_score < 1.5027:
        print("*** TARGET ACHIEVED! ***")
    else:
        print(f"Still {best_score - 1.5027:.10f} above target")
    print(f"{'='*60}")
    
    return [float(x) for x in best_seq]
