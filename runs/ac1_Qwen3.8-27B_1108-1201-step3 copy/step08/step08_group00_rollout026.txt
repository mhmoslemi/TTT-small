    a = a / a.sum()  # normalize
    m = np.zeros(n)  # Adam first moment
    v = np.zeros(n)  # Adam second moment
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    for t in range(n_iters):
        conv = fast_convolve(a)
        k_star = argmax(conv)
        
        # Compute subgradient
        g = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        for p in range(lo, hi + 1):
            g[p] = a[k_star - p] + a[p]
        
        # Normalize gradient (optional)
        g = g / (np.linalg.norm(g) + eps)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        m_hat = m / (1 - beta1**(t+1))
        v_hat = v / (1 - beta2**(t+1))
        
        lr = lr_schedule(t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project to simplex (non-negative, sum=1)
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        a = a_new
    
    return a

    t0 = time.monotonic()
    
    temps = np.geomspace(T_init, T_min, n_phases)
    
    for phase, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        phase_duration = time_limit / n_phases
        phase_deadline = min(t0 + (phase + 1) * phase_duration, t0 + time_limit)
        step = 0
        use_smooth = T > 1.0
        
        while time.monotonic() < phase_deadline:
            step += 1
            conv = fast_convolve(a)
            
            if use_smooth:
                Tb = T * conv
                max_Tb = Tb.max()
                exp_Tb = np.exp(Tb - max_Tb)
                w = exp_Tb / exp_Tb.sum()
                g = compute_logsumexp_gradient(a, w, T, n)
            else:
                # Exact subgradient: find top peaks
                max_val = conv.max()
                threshold = max_val - 0.005 * max_val
                k_stars = np.where(conv >= threshold)[0]
                if len(k_stars) == 0:
                    k_stars = np.array([int(np.argmax(conv))])
                elif len(k_stars) > 20:
                    # Only use top 20 peaks
                    idx_sorted = np.argsort(conv)[::-1][:20]
                    k_stars = idx_sorted
                g = compute_exact_subgradient(a, k_stars.tolist())
            
            # Normalize gradient for stability
            g_norm = np.linalg.norm(g)
            if g_norm > 1e-15:
                g = g / g_norm
            
            # Adam update
            t_adam = step + 1
            m = beta1 * m + (1 - beta1) * g
            v = beta2 * v + (1 - beta2) * g * g
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)
            
            lr = lr_init / (1 + 0.01 * t_adam)
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            # Project to simplex
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new / s_new
            
            # Check improvement
            if step % 50 == 0 or step < 10:
                score = 2.0 * n * fast_convolve(a_new).max()
                if score < best_score:
                    best_score = score
                    best_a = a_new.copy()
            
            a = a_new
        
        # Print progress
        elapsed = time.monotonic() - t0
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        print(f"    Adam phase {phase+1}/{n_phases} (T={T:.1f}, {elapsed:.0f}s): "
              f"best={best_score:.6f}, curr={score:.6f}")
    
    return best_a, best_score


def build_conv_matrix_sparse(a_fixed):
    """Build the convolution operator as a function (sparse-like)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None):
    """LP: given a (sum=1), find b>=0, sum(b)=1 minimizing max conv(a,b)."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix_sparse(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def lp_alternating(a0, time_limit, solver=None, max_iters=5000):
    """LP-based alternating optimization with Nesterov acceleration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    y = a.copy()
    t0 = time.monotonic()
    stall = 0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, _ = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy
        
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        improved = False
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            improved = True
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            improved = True
        
        if not improved:
            stall += 1
        else:
            stall = 0
        
        if stall >= 8:
            # Perturb to escape
            rng = np.random.default_rng(it * 7 + 3)
            noise = rng.random(n) * 0.02
            y = y + noise
            y = np.maximum(y, 0)
            y = y / y.sum()
            stall = 0
        
        if (it + 1) % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score


def peak_shave(a0, time_limit, n_steps=20000):
    """Aggressive peak shaving targeting the maximum convolution value."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(a)
        peak_val = conv.max()
        peak_k = int(np.argmax(conv))
        
        # Find near-peaks
        threshold = peak_val * 0.98
        near_peaks = np.where(conv >= threshold)[0]
        
        # Compute contribution scores
        contrib = np.zeros(n)
        for k in near_peaks[:10]:
            start = max(0, k - n + 1)
            end = min(n - 1, k)
            idx = np.arange(start, end + 1)
            c = a[idx] * a[k - idx]
            contrib[idx] += c
            contrib[k - idx] += c
        
        # Find best coordinate to reduce
        top_coords = np.argsort(contrib)[::-1][:15]
        
        improved = False
        for i in top_coords:
            if a[i] < 1e-12:
                continue
            for factor in [0.99, 0.97, 0.95, 0.92, 0.88]:
                diff = a[i] * (1 - factor)
                if diff < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] *= factor
                # Redistribute to non-contributing positions
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                # Reduce redistribution to positions with low contribution
                low_contrib = contrib < np.percentile(contrib, 50)
                low_contrib[i] = False
                if low_contrib.sum() > 0:
                    redist = diff * low_contrib / low_contrib.sum()
                else:
                    redist = diff * mask / mask.sum()
                    low_contrib = mask
                a_trial += redist
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-10:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Try pair reduction
            start = max(0, peak_k - n + 1)
            end = min(n - 1, peak_k)
            idx = np.arange(start, end + 1)
            c = a[idx] * a[peak_k - idx]
            top_pairs = np.argsort(c)[::-1][:5]
            
            found = False
            for ci in top_pairs:
                i = idx[ci]
                j = peak_k - i
                if i == j or j >= n:
                    continue
                if a[i] < 1e-12 or a[j] < 1e-12:
                    continue
                for f1 in [0.98, 0.95]:
                    for f2 in [0.98, 0.95]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-10:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            found = True
                            break
                    if found:
                        break
                if found:
                    break
            if not found:
                # Small random perturbation
                rng = np.random.default_rng(step)
                a_trial = a + rng.standard_normal(n) * 0.001
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                else:
                    a = a_trial  # accept anyway for exploration
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase hybrid optimization: Adam-LogSumExp + LP alternating +
    peak shaving + multi-length exploration with Tukey window initialization.
    """

    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if parent is None:
        n_init = 960
        best_seq = tukey_window(n_init, alpha=0.25)
        best_score = evaluate_safe(best_seq)
        print(f"Default init (Tukey): n={len(best_seq)}, score={best_score:.6f}")
    
    n_best = len(best_seq)
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")
    
    # === Phase 1: Adam optimization on parent length ===
    print(f"\n=== Phase 1: Adam n={n_best} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 480.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_adam, _ = adam_optimize(best_seq, t_alloc, T_init=80.0, T_min=0.3,
                                   n_phases=10, lr_init=0.06)
        try_update(a_adam, "adam_1")
    
    # === Phase 2: LP Alternating ===
    print(f"\n=== Phase 2: LP Alternating (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = lp_alternating(best_seq, t_alloc, solver=solver, max_iters=20000)
        try_update(a_lp, "lp_1")
    
    # === Phase 3: Adam round 2 (lower temperature) ===
    print(f"\n=== Phase 3: Adam r2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.18, 250.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s, T: 20->0.2")
        a_adam2, _ = adam_optimize(best_seq, t_alloc, T_init=20.0, T_min=0.2,
                                    n_phases=8, lr_init=0.03)
        try_update(a_adam2, "adam_2")
    
    # === Phase 4: Peak shaving ===
    print(f"\n=== Phase 4: Peak Shaving (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 150.0)
    if t_alloc > 15:
        a_ps, _ = peak_shave(best_seq, t_alloc, n_steps=50000)
        try_update(a_ps, "peak_shave_1")
    
    # === Phase 5: LP round 2 ===
    print(f"\n=== Phase 5: LP r2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 150.0)
    if t_alloc > 15:

    # === Phase 6: Alternative lengths ===
    print(f"\n=== Phase 6: Alternative Lengths (best={best_score:.6f}) ===")
    alt_lengths = []
    for mult in [1.25, 1.5, 0.8, 2.0, 0.65]:
        alt_n = int(n_best * mult)
        if 256 <= alt_n <= 3000 and alt_n != n_best:
            alt_lengths.append(alt_n)

        n_cur = len(best_seq)
        
        # Structured perturbation: modulate with low-frequency noise
        pert_scale = 0.03
        freq = rng.integers(2, 8)
        x = np.linspace(0, freq * 2 * np.pi, n_cur)
        mod = 1 + pert_scale * np.sin(x) * rng.uniform(0.5, 1.5)
        perturbed = best_seq * mod
        perturbed = np.maximum(perturbed, 0)
        perturbed = perturbed / perturbed.sum()
        
        t_adam_pert = t_pert * 0.7
        a_pert, _ = adam_optimize(perturbed, t_adam_pert, T_init=30.0, T_min=0.3,
                                   n_phases=6, lr_init=0.04)

            try_update(a_pert_lp, "perturbed_lp")
    
    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.30, 60.0)
        a_fin, _ = adam_optimize(best_seq, t_fin, T_init=5.0, T_min=0.1,
                                   n_phases=5, lr_init=0.01)
        try_update(a_fin, "final_adam")
    
    rem = deadline - time.monotonic()
    if rem > 15:
        t_ps = min(rem * 0.25, 40.0)
        a_ps2, _ = peak_shave(best_seq, t_ps, n_steps=20000)
        try_update(a_ps2, "final_peak")
    
    rem = deadline - time.monotonic()
    if rem > 15:
        t_lp_fin = min(rem * 0.30, 30.0)
        a_lp_fin, _ = lp_alternating(best_seq, t_lp_fin, solver=solver, max_iters=2000)
        try_update(a_lp_fin, "final_lp")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
