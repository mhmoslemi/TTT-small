    """Project v onto the probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)

    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the optimization score for a normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def softmax_fn(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def gradient_smoothed(a, T):
    """Compute gradient of T*log(sum(exp(conv/T))) w.r.t. a using FFT."""
    n = len(a)
    conv = fast_convolve(a)
    # Softmax weights
    scaled = conv / T
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    # g[i] = 2 * sum_m a[m] * w[m+i] = 2 * cross_correlation(a, w)[i]
    size = 2 * n - 1
    # Need N >= n + size - 1 = 3n-2 for no wrap-around
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    grad = 2.0 * irfft(np.conj(A) * W, fft_size)[:n]
    return grad


def log_space_grad(z, T):
    """Gradient of smoothed objective w.r.t. log-space variables."""
    a = softmax_fn(z)
    g = gradient_smoothed(a, T)
    inner = np.dot(g, a)
    return a * (g - inner)


def project_simplex(v):
    """Project v onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def Adam_logspace(a0, time_limit, T, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8, verbose=True):
    """Adam optimization in log-space with soft-max smoothed objective."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-30))
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)

    while time.monotonic() - t0 < time_limit:
        t += 1
        if time.monotonic() - t0 > time_limit:
            break
        grad = log_space_grad(z, T)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = softmax_fn(z)
        score = compute_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam T={T} iter={t}, score={best_score:.8f}, elapsed={elapsed:.1f}s")
    return best_a, best_score


def build_constraint_matrix(f):
    """Build C where C[k,j] = f[k-j] if 0<=k-j<n else 0."""
    n = len(f)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = f
    return C


def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. C*g <= M*1, g >= 0."""
    n = len(f)
    C = build_constraint_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=C, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
        if res.status == 0:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def greedy_lp(f, M):
    """Greedy approximation: build g one coordinate at a time, largest f first."""
    n = len(f)
    g = np.zeros(n)
    conv_vals = np.zeros(2 * n - 1)
    order = np.argsort(f)[::-1]
    for j in order:
        k_start = j
        k_end = j + n - 1
        if k_end >= 2 * n:
            k_end = 2 * n - 1
        slack = M - conv_vals[k_start:k_end + 1]
        f_slice = f[:k_end - k_start + 1]
        valid = f_slice > 1e-15
        if not valid.any():
            continue
        max_gj = np.min(slack[valid] / f_slice[valid])
        max_gj = max(0, max_gj)
        g[j] = max_gj
        conv_vals[k_start:k_end + 1] += f_slice * max_gj
    return g


def lp_fictitious_play(f_init, time_limit, use_greedy=False, verbose=True, multi_M=False):
    """LP-based fictitious play with geometric mean updates."""
    f = f_init.copy()
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-30:
        return f, compute_score(f)
    f = f / s
    n = len(f)

    f_avg = f.copy()
    best_f = f.copy()
    best_score = compute_score(f)
    t0 = time.monotonic()

    for k in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        M = conv.max()

        if multi_M and k % 3 == 2:
            # Try a slightly tighter M to encourage more spread
            M_try = M * 0.95
            g = solve_lp(f, M_try) if not use_greedy else greedy_lp(f, M_try)
        else:
            if use_greedy:
                g = greedy_lp(f, M)
            else:
                g = solve_lp(f, M)

        if g is None or g.sum() < 1e-15:
            g = greedy_lp(f, M)
            if g is None or g.sum() < 1e-15:
                break

        g = g / g.sum()

        # Adaptive step size
        t_step = 2.0 / (k + 2)
        if k > 100:
            t_step = max(0.005, 1.0 / (k + 1))

        # Geometric mean update in log-space
        f_safe = np.maximum(f, 1e-30)
        g_safe = np.maximum(g, 1e-30)
        log_new = (1 - t_step) * np.log(f_safe) + t_step * np.log(g_safe)
        log_new -= log_new.max()
        a_new = np.exp(log_new)
        s_new = a_new.sum()
        if s_new < 1e-30:
            a_new = f.copy()
        else:
            a_new = a_new / s_new

        f_avg = ((k * f_avg) + a_new) / (k + 1)

        score_avg = compute_score(f_avg)
        score_cur = compute_score(a_new)

        if score_avg < best_score - 1e-14:
            best_score = score_avg
            best_f = f_avg.copy()
        if score_cur < best_score - 1e-14:
            best_score = score_cur
            best_f = a_new.copy()

        f = a_new

        if verbose and k % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP[{use_greedy}] iter={k}, M={M:.6e}, avg={score_avg:.8f}, "
                  f"best={best_score:.8f}, el={elapsed:.0f}s")

    return best_f, best_score


def subgradient_method(a0, time_limit, lr0=0.1, verbose=True):
    """Projected subgradient with heavy-ball momentum and averaging."""
    a = a0.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    n = len(a)
    momentum = np.zeros(n)
    beta = 0.9
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = compute_score(a)
    avg_a = a.copy()
    total_iter = 0

    while time.monotonic() - t0 < time_limit:
        total_iter += 1
        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))
        # Subgradient: sg[i] = 2*a[k_star - i] if 0 <= k_star - i < n
        sg = np.zeros(n)
        idx = k_star - np.arange(n)
        valid = (idx >= 0) & (idx < n)
        sg[valid] = 2.0 * a[idx[valid]]

        # Heavy ball
        lr = lr0 / (1 + 0.01 * total_iter)
        momentum = beta * momentum + lr * sg
        a_new = a - momentum
        a_new = project_simplex(a_new)

        avg_a = ((total_iter - 1) * avg_a + a_new) / total_iter

        sc = compute_score(a_new)
        sc_avg = compute_score(avg_a)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = avg_a.copy()

        a = a_new

        if verbose and total_iter % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Subgrad iter={total_iter}, score={best_score:.8f}, el={elapsed:.0f}s")

    return best_a, best_score


def coordinate_descent(a0, time_limit, verbose=True):
    """Coordinate descent with golden-section search."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    gr = (np.sqrt(5) - 1) / 2
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        # Generate candidate pairs
        n_pairs = min(n, 300)
        pairs = set()
        conv = fast_convolve(a)
        top_peak = int(np.argmax(conv))
        js = max(0, top_peak - n + 1)
        je = min(n - 1, top_peak)

        for _ in range(n_pairs):
            if rng.random() < 0.5:
                i = rng.integers(js, je + 1)
                j = rng.integers(js, je + 1)
            else:
                i = rng.integers(0, n)
                j = rng.integers(0, n)
            if i != j:
                pairs.add((min(i, j), max(i, j)))

        for (i, j) in list(pairs):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-15 and a[j] < 1e-15:
                continue

            t_lo = -a[j] + 1e-15
            t_hi = a[i] - 1e-15
            if t_hi <= t_lo:
                continue

            x1 = t_hi - gr * (t_hi - t_lo)
            x2 = t_lo + gr * (t_hi - t_lo)

            def eval_t(t):
                a_trial = a.copy()
                a_trial[i] -= t
                a_trial[j] += t
                if a_trial[i] < -1e-15 or a_trial[j] < -1e-15:
                    return 1e10
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    return 1e10
                a_trial = a_trial / ss
                return compute_score(a_trial)

            f1 = eval_t(x1)
            f2 = eval_t(x2)
            for _ in range(15):
                if time.monotonic() - t0 > time_limit:
                    break
                if f1 < f2:
                    t_hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = t_hi - gr * (t_hi - t_lo)
                    f1 = eval_t(x1)
                else:
                    t_lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = t_lo + gr * (t_hi - t_lo)
                    f2 = eval_t(x2)

            t_opt = (t_lo + t_hi) / 2
            f_opt = min(f1, f2)
            if f_opt < best_score - 1e-13:
                a[i] = max(0, a[i] - t_opt)
                a[j] = a[j] + t_opt
                a = a / a.sum()
                best_score = compute_score(a)
                best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 3:
                noise = rng.standard_normal(n) * 0.003
                a = best_a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
        else:
            no_improve = 0

        if verbose:
            iters = int((time.monotonic() - t0) * 10)
            if iters % 500 == 0:
                print(f"    CD score={best_score:.8f}, el={time.monotonic()-t0:.0f}s")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score, verbose=True):
    """Evaluate a candidate and return updated best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  *** NEW BEST [{label}]: {score:.8f} (n={len(a_norm)}) ***")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-strategy optimization combining LP fictitious play,
    FFT-accelerated smoothed gradient (Adam), subgradient, and coordinate descent.
    Explores multiple lengths and uses the parent as warm start.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"Could not load parent: {e}")

    n_parent = len(parent) if parent is not None else 768

    # Lengths to explore (prioritize near parent)
    candidate_lengths = [n_parent]
    for alt in [n_parent - 64, n_parent - 32, n_parent + 32, n_parent + 64,
                n_parent - 128, n_parent + 128, n_parent + 256,
                512, 768, 1024, 1280, 1536, 2048]:
        if 128 <= alt <= 4096 and alt not in candidate_lengths:
            candidate_lengths.append(alt)
    candidate_lengths = sorted(set(candidate_lengths))

    print(f"\nCandidate lengths: {candidate_lengths}")
    print(f"Initial best: {best_score:.8f} (n={n_parent})")

    # === Main loop: process each length with time allocation ===
    # Give more time to the parent length and nearby lengths
    n_lengths = len(candidate_lengths)
    base_time = (deadline - time.monotonic()) / n_lengths
    # Prioritize parent length
    time_weights = []
    for n in candidate_lengths:
        if n == n_parent:
            time_weights.append(3.0)
        elif abs(n - n_parent) <= 128:
            time_weights.append(2.0)
        else:
            time_weights.append(1.0)
    total_w = sum(time_weights)
    available_time = deadline - time.monotonic() - 120  # reserve 120s for final polish
    time_alloc = [available_time * w / total_w for w in time_weights]

    for idx, n_try in enumerate(candidate_lengths):
        if time.monotonic() > deadline - 150:
            break
        t_budget = min(time_alloc[idx], deadline - time.monotonic() - 150)
        if t_budget < 20:
            break

        print(f"\n{'='*60}")
        print(f"Length n={n_try}, time budget={t_budget:.0f}s, current best={best_score:.8f}")
        print(f"{'='*60}")

        # Initialize from parent (interpolated)
        if parent is not None:
            a_init = interpolate_sequence(parent, n_try)
        else:
            a_init = np.ones(n_try) / n_try
            t_arr = np.linspace(0, 2 * np.pi * 2, n_try)
            a_init = a_init + 0.03 * np.cos(t_arr)
            a_init = np.maximum(a_init, 0)
            a_init = a_init / a_init.sum()

        # --- Phase A: LP Fictitious Play (main workhorse) ---
        t_a = t_budget * 0.35
        if t_a > 10:
            print(f"  Phase A: LP Fictitious Play ({t_a:.0f}s)")
            a_lp, sc_lp = lp_fictitious_play(a_init, t_a, use_greedy=False, verbose=True, multi_M=True)
            a_init, _ = check_and_update(a_lp, f"n{n_try}_lp", best_seq, best_score)
            a_init = a_init / a_init.sum()

        # --- Phase B: Adam in log-space with multiple temperatures ---
        t_b = t_budget * 0.30
        if t_b > 10:
            for T_val in [500, 2000]:
                t_sub = t_b / 2
                if t_sub < 5:
                    break
                if time.monotonic() > deadline - 150:
                    break
                print(f"  Phase B: Adam T={T_val} ({t_sub:.0f}s)")
                a_adam, sc_adam = Adam_logspace(a_init, t_sub, T=T_val, lr=0.05, verbose=True)
                a_init, _ = check_and_update(a_adam, f"n{n_try}_adam{T_val}", best_seq, best_score)
                a_init = a_init / a_init.sum()

        # --- Phase C: LP FP again (polish) ---
        t_c = t_budget * 0.20
        if t_c > 10:
            print(f"  Phase C: LP Polish ({t_c:.0f}s)")
            a_lp2, sc_lp2 = lp_fictitious_play(a_init, t_c, use_greedy=False, verbose=True)
            a_init, _ = check_and_update(a_lp2, f"n{n_try}_lp2", best_seq, best_score)
            a_init = a_init / a_init.sum()

        # --- Phase D: Coordinate descent ---
        t_d = t_budget * 0.15
        if t_d > 10:
            print(f"  Phase D: Coordinate Descent ({t_d:.0f}s)")
            a_cd, sc_cd = coordinate_descent(a_init, t_d, verbose=True)
            a_init, _ = check_and_update(a_cd, f"n{n_try}_cd", best_seq, best_score)

        print(f"  After n={n_try}: best={best_score:.8f}")

    # === Final polish phase ===
    print(f"\n{'='*60}")
    print(f"FINAL POLISH PHASE: best={best_score:.8f} (n={len(best_seq)})")
    t_remaining = deadline - time.monotonic() - 20
    if t_remaining > 30:
        n_best = len(best_seq)
        a_work = best_seq.copy()

        # Quick LP on best
        t_lp = min(t_remaining * 0.3, 60)
        if t_lp > 10:
            print(f"  Final LP ({t_lp:.0f}s)")
            a_fin, _ = lp_fictitious_play(a_work, t_lp, use_greedy=False, verbose=False)
            a_work, best_score = check_and_update(a_fin, "final_lp", best_seq, best_score)

        # Adam polish
        t_ad = min(t_remaining * 0.3, 40)
        if t_ad > 5:
            print(f"  Final Adam ({t_ad:.0f}s)")
            a_fin2, _ = Adam_logspace(a_work, t_ad, T=1000, lr=0.02, verbose=False)
            a_work, best_score = check_and_update(a_fin2, "final_adam", best_seq, best_score)

        # Coordinate descent
        t_cd = min(t_remaining * 0.3, 40)
        if t_cd > 5:
            print(f"  Final CD ({t_cd:.0f}s)")
            a_fin3, _ = coordinate_descent(a_work, t_cd, verbose=False)
            a_work, best_score = check_and_update(a_fin3, "final_cd", best_seq, best_score)

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs) and fs < best_score:
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq) if best_seq is not None else 'N/A'})")

    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    return [float(x) for x in best_seq]
