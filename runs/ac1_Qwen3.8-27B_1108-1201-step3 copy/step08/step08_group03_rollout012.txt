def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adam smoothed gradient descent (log-sum-exp) with adaptive temperature

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, w):
    """
    Compute gradient of max_k b[k] where b = convolve(f, f), weighted by w.
    grad[j] = 2 * sum_i f[i] * w[i+j]
    Uses FFT via reversal trick.
    """
    n = len(f)
    g = f[::-1].copy()
    conv_size = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_size - 1).bit_length()
    G = rfft(g, fft_size)
    W = rfft(w, fft_size)
    conv = irfft(G * W, fft_size)[:conv_size]
    grad = 2.0 * conv[n - 1: 2 * n - 1]
    return grad

def iaa_diversify(f_init, n_iters=200, peak_fraction=0.3, seed=0):
    """

    """

    f = f_init.copy()
    fft_size = 1 << (2 * n - 1).bit_length()
    rng = np.random.default_rng(seed)
    
    for it in range(n_iters):
        F = rfft(f, fft_size)
        R = irfft(F * F, fft_size)[:2 * n - 1]
        
        peak_val = R.max()
        if peak_val < 1e-15:
            break
        
        # Target: flatten peaks above floor
        floor = peak_fraction * peak_val
        R_target = np.clip(R, floor, peak_val)
        # Preserve total energy
        energy = R.sum()
        if R_target.sum() > 1e-10:
            R_target *= energy / R_target.sum()
        
        # Update amplitude spectrum
        R_t_fft = rfft(R_target, fft_size)
        real_part = np.maximum(R_t_fft.real, 0.0)
        new_amp = np.sqrt(real_part)
        
        # Preserve phase (with small random perturbation for diversification)
        abs_F = np.abs(F)
        phase = F / (abs_F + 1e-30)
        # Add small phase noise
        phase_noise = np.exp(1j * rng.standard_normal(len(F)) * 0.05 * (1 + it / n_iters))
        F_new = new_amp * phase * phase_noise
        
        f = irfft(F_new, fft_size)[:n]
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s
        else:
            f = f_init.copy()
            break
    
    return f


def build_constraint_matrix(f, indices, n):
    """Build constraint matrix for (f*g)[k] = A[k,:] @ g."""
    K = len(indices)
    A = np.zeros((K, n))
    for i, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    return A


def solve_lp_direction(f, n, M, active_idx, eps=1e-8):
    """Solve LP: max sum(g) s.t. g >= 0, (f*g)[k_i] <= M for active constraints."""
    K = len(active_idx)
    if K == 0:
        return None
    A = build_constraint_matrix(f, active_idx, n)
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=500000, eps=eps)
    except Exception:
        g_val = None
    if g.value is None:
        try:
            prob.solve(solver=cp.ECOS, max_iters=50000)
        except Exception:
            return None
    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None
    return g_val


def convolve_direction(f, g):
    """Cross-convolution f*g via FFT."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]

                b_cur = fast_convolve(f)
                new_top = np.argpartition(b_cur, -min(max_k, len(b_cur)))[-min(max_k, len(b_cur)):]
                active_idx = np.unique(np.concatenate([active_idx, new_top]))
            continue
        
        # Cutting plane check
        fg = convolve_direction(f, g)
        violated_mask = fg > M * (1 + 1e-7)
        v_count = violated_mask.sum()
        
        if v_count > 0 and len(active_idx) < max_k:
            v_idx = np.where(violated_mask)[0]
            not_active = v_idx[~np.isin(v_idx, active_idx)]
            if len(not_active) > 0:
                n_add = min(20, len(not_active))
                to_add = not_active[np.argsort(fg[not_active])[-n_add:]]
                active_idx = np.unique(np.concatenate([active_idx, to_add]))
                g = solve_lp_direction(f, n, M, active_idx, eps=1e-8)
                if g is None:
                    stuck += 1
                    if stuck > 5:
                        break
                    continue
                S_g = g.sum()
        
        if S_g <= 1.0 + 1e-6:
            stuck += 1
            if stuck > 25:
                break
            continue
        
        stuck = 0
        h = g / S_g
        
        # Line search
        best_t = 0.05
        best_t_obj = M
        scan_steps = [0.02, 0.04, 0.06, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]
        for t in scan_steps:
            f_trial = np.maximum((1.0 - t) * f + t * h, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t
        
        # Golden-section refinement
        if best_t > 0.02:
            lo = max(0.0, best_t - 0.05)
            hi = min(1.0, best_t + 0.1)
            for _ in range(5):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                fa1 = np.maximum((1 - m1) * f + m1 * h, 0.0)
                sa1 = fa1.sum()
                if sa1 > 1e-10:
                    fa1 /= sa1
                fa2 = np.maximum((1 - m2) * f + m2 * h, 0.0)
                sa2 = fa2.sum()
                if sa2 > 1e-10:
                    fa2 /= sa2
                o1 = fast_convolve(fa1).max()
                o2 = fast_convolve(fa2).max()
                if o1 < o2:
                    hi = m2
                else:
                    lo = m1
            t_mid = (lo + hi) / 2.0
            f_mid = np.maximum((1 - t_mid) * f + t_mid * h, 0.0)
            s_mid = f_mid.sum()
            if s_mid > 1e-10:
                f_mid /= s_mid
                o_mid = fast_convolve(f_mid).max()
                if o_mid < best_t_obj:
                    best_t_obj = o_mid
                    best_t = t_mid
        
        f = np.maximum((1.0 - best_t) * f + best_t * h, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s
        else:
            f = np.ones(n) / n
        
        cur_obj = fast_convolve(f).max()
        M = cur_obj
        
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_f = f.copy()
        
        # Refresh active set periodically
        if it % 12 == 0:
            b = fast_convolve(f)
            if len(active_idx) > initial_k * 2:
                active_scores = b[active_idx]
                keep = active_idx[np.argsort(active_scores)[-initial_k * 2:]]
                new_top = np.argpartition(b, -initial_k)[-initial_k:]
                active_idx = np.unique(np.concatenate([keep, new_top]))
                active_idx = active_idx[np.argsort(b[active_idx])[-max_k:]]
        
        if it % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} it={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} "
                  f"active={len(active_idx)} stuck={stuck} t={elapsed:.0f}s")
    
    return best_f, best_obj


def generate_diverse_init(n, seed=0):
    """Generate diverse initial sequences."""
    rng = np.random.default_rng(seed)
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Edge-concentrated (trapezoidal)
    x = np.linspace(0, 1, n)
    trap = 1.0 - np.abs(2 * x - 1) ** 2
    trap = np.maximum(trap, 0.1)
    inits.append(trap / trap.sum())
    
    # Cosine window
    cos_win = 0.5 * (1 + np.cos(2 * np.pi * x))
    cos_win = np.maximum(cos_win, 0.05)
    inits.append(cos_win / cos_win.sum())
    
    # Random
    r = rng.dirichlet(np.ones(n))
    inits.append(r)
    
    # Two-peak
    center = n // 2
    peak1 = np.exp(-((np.arange(n) - 0.15 * n) ** 2) / (2 * (0.1 * n) ** 2))
    peak2 = np.exp(-((np.arange(n) - 0.85 * n) ** 2) / (2 * (0.1 * n) ** 2))
    two = peak1 + peak2 + 0.1
    inits.append(two / two.sum())
    
    # Log-uniform (more mass in middle)
    log_u = np.exp(-((x - 0.5) ** 2) / 0.05)
    log_u = np.maximum(log_u, 0.01)
    inits.append(log_u / log_u.sum())
    
    return inits


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search: multi-phase optimization combining Adam smoothed gradient,
    IAA diversification, cutting-plane LP, and multi-length exploration.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception as e:
        print(f"Parent load failed: {e}")
    
    if parent is None:
        parent = np.ones(1024) / 1024
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: uniform n=1024, score={p_score:.6f}")
    
    best_seq = parent.copy()
    best_score = p_score
    n_parent = len(parent)
    
    # === Phase 1: Adam smoothed descent on parent ===
    print(f"\n=== Phase 1: Adam smoothed (n={n_parent}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.30, 500.0)
    if t1 > 60:
        print(f"  Budget: {t1:.0f}s")
        f1, obj1 = adam_smooth_descent(parent, time_limit=t1, tau_init=5.0,
                                       tau_end=200.0, lr_init=0.005,
                                       n_restarts=5, seed=42)
        score1 = float(evaluate_sequence(f1.tolist()))
        if np.isfinite(score1) and score1 < best_score:
            best_score = score1
            best_seq = f1.copy()
            print(f"  NEW BEST [adam1]: {score1:.6f}")
    
    # === Phase 2: IAA diversification + Adam ===
    print(f"\n=== Phase 2: IAA + Adam (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t2_total = min(rem * 0.20, 250.0)
    if t2_total > 60:
        for variant in range(2):
            rem2 = deadline - time.monotonic()
            if rem2 < 30:
                break
            t2 = min(rem2 * 0.5, 120.0)
            if t2 < 20:
                break
            pf = 0.3 + 0.2 * variant
            f_iaa = iaa_diversify(best_seq, n_iters=150 + 50 * variant,
                                   peak_fraction=pf, seed=100 + variant)
            f2, _ = adam_smooth_descent(f_iaa, time_limit=t2, tau_init=10.0,
                                         tau_end=300.0, lr_init=0.003,
                                         n_restarts=3, seed=200 + variant)
            s2 = float(evaluate_sequence(f2.tolist()))
            if np.isfinite(s2) and s2 < best_score:
                best_score = s2
                best_seq = f2.copy()
                print(f"  NEW BEST [iaa_adam{variant}]: {s2:.6f}")
    
    # === Phase 3: LP cutting-plane polish ===
    print(f"\n=== Phase 3: LP CP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.22, 350.0)
    if t3 > 60:
        n_b = len(best_seq)
        print(f"  Budget: {t3:.0f}s, n={n_b}")
        f3, obj3 = lp_cutting_plane(best_seq, time_limit=t3, initial_k=60,
                                     max_k=450, label="cp1")
        score3 = float(evaluate_sequence(f3.tolist()))
        if np.isfinite(score3) and score3 < best_score:
            best_score = score3
            best_seq = f3.copy()
            print(f"  NEW BEST [cp1]: {score3:.6f}")
    
    # === Phase 4: Try diverse initializations at same length ===
    print(f"\n=== Phase 4: Diverse inits (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    n_b = len(best_seq)
    inits = generate_diverse_init(n_b, seed=7)
    
    # Sort by initial score, try best 2 that aren't too similar to current
    init_scores = []
    for i, init in enumerate(inits):
        b = fast_convolve(init)
        sc = 2.0 * n_b * b.max()
        init_scores.append((sc, i))
    init_scores.sort()
    
    tried = 0
    for sc, idx in init_scores[:4]:
        rem4 = deadline - time.monotonic()
        if rem4 < 40 or tried >= 2:
            break
        # Skip if too similar to best
        similarity = np.dot(inits[idx], best_seq) / (np.linalg.norm(inits[idx]) * np.linalg.norm(best_seq) + 1e-10)
        if similarity > 0.98:
            continue
        t4 = min(rem4 * 0.4, 100.0)
        if t4 < 20:
            break
        tried += 1
        print(f"  Trying init {idx} (init_score={sc:.4f}, sim={similarity:.3f}), budget={t4:.0f}s")
        f4, _ = adam_smooth_descent(inits[idx], time_limit=t4, tau_init=5.0,
                                     tau_end=200.0, lr_init=0.005,
                                     n_restarts=3, seed=300 + idx)
        s4 = float(evaluate_sequence(f4.tolist()))
        if np.isfinite(s4) and s4 < best_score:
            best_score = s4
            best_seq = f4.copy()
            print(f"  NEW BEST [diverse{idx}]: {s4:.6f}")
    
    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length (best={best_score:.6f}) ===")
    n_base = len(best_seq)
    target_lengths = set()
    for factor in [0.9, 1.08, 1.15, 0.85]:
        tl = int(n_base * factor)
        tl = max(512, min(3000, tl))
        if tl != n_base:
            target_lengths.add(tl)
    target_lengths = sorted(target_lengths, key=lambda x: abs(x - n_base))
    
    for target_n in target_lengths[:2]:
        rem5 = deadline - time.monotonic()
        if rem5 < 50:
            break
        t5 = min(rem5 * 0.18, 120.0)
        if t5 < 25:
            break
        old_x = np.linspace(0.0, 1.0, n_base)
        new_x = np.linspace(0.0, 1.0, target_n)
        f_interp = np.interp(new_x, old_x, best_seq)
        f_interp = np.maximum(f_interp, 0.0)
        s = f_interp.sum()
        if s > 1e-10:
            f_interp /= s
        else:
            f_interp = np.ones(target_n) / target_n
        
        print(f"  n={target_n}, budget={t5:.0f}s")
        f5, _ = adam_smooth_descent(f_interp, time_limit=t5 * 0.5, tau_init=5.0,
                                     tau_end=150.0, lr_init=0.004,
                                     n_restarts=2, seed=400 + target_n)
        # Quick LP polish
        rem5b = deadline - time.monotonic()
        if rem5b > 30:
            t5b = min(rem5b * 0.4, 60.0)
            f5b, _ = lp_cutting_plane(f5, time_limit=t5b, initial_k=40,
                                       max_k=300, label=f"n{target_n}")
            f5 = f5b
        
        s5 = float(evaluate_sequence(f5.tolist()))
        if np.isfinite(s5) and s5 < best_score:
            best_score = s5
            best_seq = f5.copy()
            print(f"  NEW BEST [n{target_n}]: {s5:.6f}")
    
    # === Phase 6: Final LP polish on best ===
    print(f"\n=== Phase 6: Final LP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.15, 180.0)
    if t6 > 30:
        n_b = len(best_seq)
        print(f"  Budget: {t6:.0f}s, n={n_b}")
        f6, _ = lp_cutting_plane(best_seq, time_limit=t6, initial_k=80,
                                  max_k=500, label="final")
        score6 = float(evaluate_sequence(f6.tolist()))
        if np.isfinite(score6) and score6 < best_score:
            best_score = score6
            best_seq = f6.copy()
            print(f"  NEW BEST [final_lp]: {score6:.6f}")
    
    # === Phase 7: Short Adam burst with high tau ===
    rem = deadline - time.monotonic()
    if rem > 20:
        print(f"\n=== Phase 7: Fine Adam (best={best_score:.6f}) ===")
        t7 = min(rem * 0.5, 60.0)
        n_b = len(best_seq)
        f7, _ = adam_smooth_descent(best_seq, time_limit=t7, tau_init=100.0,
                                     tau_end=500.0, lr_init=0.001,
                                     n_restarts=2, seed=500)
        score7 = float(evaluate_sequence(f7.tolist()))
        if np.isfinite(score7) and score7 < best_score:
            best_score = score7
            best_seq = f7.copy()
            print(f"  NEW BEST [fine_adam]: {score7:.6f}")
    
    # Final validation
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
    
    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")
    
    return [float(x) for x in best_seq]
