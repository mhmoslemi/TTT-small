from scipy.fft import rfft, irfft
import cvxpy as cp
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_cross_convolve(f, g):
    size = len(f) + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]

def project_simplex(v):
    # Standard simplex projection
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def lp_optimize(f, n, time_limit, active_indices=None, max_constraints=500):
    """
    Solve the LP: max sum(g) s.t. g >= 0, (f*g)_k <= M for active constraints.
    Returns (g, M) where M = max(f*f).
    """
    t0 = time.monotonic()
    b = fast_convolve(f)
    M = b.max()

        hi = min(n - 1, k)
        A[i, lo:hi+1] = f[lo:hi+1]
    
    g_var = cp.Variable(n)
    constraints = [g_var >= 0, A @ g_var <= M]

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_cross_convolve(f, g):
    nf, ng = len(f), len(g)
    size = nf + ng - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)

def score(a):
    """Compute the objective: 2*n*max(conv(a)) / sum(a)^2"""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    b = fast_convolve(a / s)
    return 2.0 * len(a) * b.max()

def lp_direction(f, active_set, M):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)_k <= M for k in active_set"""
    n = len(f)
    K = len(active_set)
    
    A = np.zeros((K, n))
    for i, k in enumerate(active_set):
        lo = max(0, int(k) - n + 1)
        hi = min(n - 1, int(k))
        if lo <= hi:
            A[i, lo:hi+1] = f[lo:hi+1]
    
    g_var = cp.Variable(n)
    prob = cp.Problem(
        cp.Maximize(cp.sum(g_var)),
        [g_var >= 0, A @ g_var <= M]
    )
    
    for solver in [cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            prob.solve(solver=solver, max_iters=100000)
            if g_var.value is not None:
                g = np.maximum(g_var.value, 0)
                if not np.any(np.isnan(g)) and g.sum() > 1.001:
                    return g
        except:
            continue
    return None

def update_active_set(f, g, b, active_set, M, add_top_k=20):
    """Find violated constraints and add to active set."""
    fg = fast_cross_convolve(f, g)
    violations = np.where(fg > M + 1e-10)[0]
    new_set = set(int(x) for x in active_set)
    if len(violations) > 0:
        # Add most violated
        worst = np.argsort(fg[violations])[-add_top_k:]
        for v in worst:
            new_set.add(int(v))
    # Also ensure top peaks are in the set
    top_idx = np.argpartition(b, -len(active_set))[-len(active_set):]
    for t in top_idx:
        new_set.add(int(t))
    return new_set

def optimize_lp(f_init, time_limit, top_k_init=60, max_constraints=400):
    """Main LP-based optimization loop."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    f = f / f.sum()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    b = fast_convolve(f)
    active_set = set(int(x) for x in np.argpartition(b, -top_k_init)[-top_k_init:])
    
    stuck = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()
        
        # Trim active set if too large
        if len(active_set) > max_constraints:
            b_active = np.array([b[k] for k in active_set])
            keep = np.argsort(b_active)[-max_constraints:]
            active_list = sorted(list(active_set))
            active_set = set(int(active_list[i]) for i in keep)
        
        # Solve LP
        g = lp_direction(f, sorted(active_set), M)
        
        if g is None:
            # Fallback: subgradient direction
            top_idx = np.array(sorted(active_set))[:10]
            dir = np.zeros(n)
            for k in top_idx:
                lo = max(0, int(k) - n + 1)
                hi = min(n - 1, int(k))
                dir[lo:hi+1] += f[lo:hi+1]
            if dir.max() < 1e-15:
                break
            g = dir
        else:
            # Update active set
            active_set = update_active_set(f, g, b, active_set, M, add_top_k=10)
        
        # Normalize direction (sum to 1)
        g_sum = g.sum()
        if g_sum <= 1.0 + 1e-5:
            stuck += 1
            if stuck > 30:
                break
            continue
        
        h = g / g_sum
        
        # Line search: find best t in (0, 1)
        best_t = 0.05
        best_score = best_obj
        for t in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7]:
            f_trial = (1 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial = f_trial / s
            obj = fast_convolve(f_trial).max()
            if obj < best_score:
                best_score = obj
                best_t = t
        
        if best_score >= best_obj - 1e-15:
            stuck += 1
            if stuck > 50:
                break
            continue
        
        # Apply update
        f = (1 - best_t) * f + best_t * h
        f = np.maximum(f, 0)
        f = f / f.sum()
        
        best_obj = best_score
        best_f = f.copy()
        stuck = 0
        
        if it % 20 == 0:
            print(f"  LP iter={it} peak={best_obj:.10f} score={2*n*best_obj:.6f} "
                  f"constraints={len(active_set)} t={time.monotonic()-t0:.0f}s")
    
    return best_f, best_obj

    if s < 1e-10:

def subgradient_polish(f_init, time_limit, n_top=50):
    """Subgradient descent with momentum."""
    t0 = time.monotonic()
    n = len(f_init)
    f = np.maximum(f_init, 0.0)
    f = f / f.sum()
    v = np.zeros(n)
    best_f = f.copy()
    b = fast_convolve(f)
    best_peak = b.max()
    lr = 0.0005
    no_improve = 0
    rng = np.random.default_rng(123)
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(f)
        
        # Top-k peaks
        top_idx = np.argpartition(b, -n_top)[-n_top:]
        
        # Subgradient: average of gradients at top peaks
        direction = np.zeros(n)
        for k in top_idx:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            direction[lo:hi+1] += f[lo:hi+1]
        direction /= len(top_idx)
        
        v = 0.9 * v - lr * direction
        f_new = project_simplex(f + v)
        f = f_new
        
        obj = fast_convolve(f).max()
        if obj < best_peak - 1e-14:
            best_peak = obj
            best_f = f.copy()
            no_improve = 0
            lr = min(lr * 1.02, 0.005)
        else:
            no_improve += 1
            if no_improve % 20 == 0:
                lr *= 0.9
            if no_improve > 100:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                v = np.zeros(n)
                no_improve = 0
    
    return best_f, best_peak

def flat_top_window(n):

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return f_new / s

import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(f, g):
    nf, ng = len(f), len(g)
    size = nf + ng - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def compute_score(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    a = a / s
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max()


def build_constraint_sparse(f, active_indices):

    rows = []
    cols = []
    vals = []

                    rows.append(i)
                    cols.append(j)
                    vals.append(f[j])
    A = csr_matrix((vals, (rows, cols)), shape=(K, n))
    return A


def solve_lp(f, active_indices, M):

    K = len(active_indices)
    if K == 0:
        return None

    A = build_constraint_sparse(f, active_indices)

    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), [g_var >= 0, A @ g_var <= M])

    best_g = None
    best_obj = -np.inf

    for solver in [cp.SCS, cp.ECOS, cp.PDLP]:
        try:
            prob.solve(solver=solver, max_iters=200000, verbose=False)
            if g_var.value is not None:
                g = np.maximum(g_var.value, 0.0)
                if np.all(np.isfinite(g)):
                    obj = float(g.sum())
                    if obj > best_obj:
                        best_obj = obj
                        best_g = g
        except Exception:
            continue

    return best_g


def line_search(f, h, best_peak, max_iters=25):
    """Golden section line search on the peak objective."""
    def peak_at(t):
        f_trial = (1.0 - t) * f + t * h
        f_trial = np.maximum(f_trial, 0.0)
        s = f_trial.sum()
        if s < 1e-10:
            return 1e10
        f_trial = f_trial / s
        return fast_convolve(f_trial).max()

    # Coarse search
    candidates = [0.01, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7]
    best_t = 0.05
    best_obj = best_peak
    for t in candidates:
        obj = peak_at(t)
        if obj < best_obj:
            best_obj = obj
            best_t = t

    # Golden section refinement
    if best_t > 0.005:
        idx = candidates.index(best_t)
        lo = candidates[max(0, idx - 1)]
        hi = candidates[min(len(candidates) - 1, idx + 1)]
        gr = (np.sqrt(5) - 1) / 2
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        obj_c1 = peak_at(c1)
        obj_c2 = peak_at(c2)
        for _ in range(10):
            if obj_c1 <= obj_c2:
                hi = c2
                c2 = c1
                obj_c2 = obj_c1
                c1 = hi - gr * (hi - lo)
                obj_c1 = peak_at(c1)
            else:
                lo = c1
                c1 = c2
                obj_c1 = obj_c2
                c2 = lo + gr * (hi - lo)
                obj_c2 = peak_at(c2)
            if hi - lo < 1e-5:
                break
        best_t = (lo + hi) / 2
        best_obj = peak_at(best_t)

    return best_t, best_obj

def subgradient_polish(f_init, time_limit, n_top=40):
    """Subgradient descent with momentum and periodic restarts."""
    t0 = time.monotonic()
    n = len(f_init)
    f = np.maximum(f_init, 0.0)
    f = f / f.sum()
    v = np.zeros(n)
    best_f = f.copy()

    lr = 0.0003
    no_improve = 0
    rng = np.random.default_rng(42)
    batch_no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 2:
            break

        # Batch of small steps
        batch_size = min(50, max(10, int((time_limit - (time.monotonic() - t0)) * 5)))
        for _ in range(batch_size):
            if time.monotonic() - t0 > time_limit - 1:
                break

            b = fast_convolve(f)
            top_idx = np.argpartition(b, -n_top)[-n_top:]

            direction = np.zeros(n)
            for k in top_idx:
                k = int(k)
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                direction[lo:hi + 1] += f[lo:hi + 1]
            direction /= len(top_idx)

            v = 0.85 * v - lr * direction
            f = project_simplex(f + v)

        obj = fast_convolve(f).max()
        if obj < best_peak - 1e-13:
            best_peak = obj
            best_f = f.copy()
            no_improve = 0
            batch_no_improve = 0
            lr = min(lr * 1.03, 0.004)
        else:
            no_improve += 1
            batch_no_improve += 1
            if no_improve % 10 == 0:
                lr *= 0.93
            if batch_no_improve > 15:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                v = np.zeros(n)
                batch_no_improve = 0

    return best_f, best_peak


def flat_top_window(n):
    w = np.ones(n)
    a = [0.21557895, 0.41663158, 0.277263158, 0.083578947, 0.006947368]
    x = np.arange(n)
    for i, ai in enumerate(a):
        w -= ai * np.cos(2 * np.pi * i * x / max(1, n - 1) - i * np.pi)
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def hann_window(n):
    w = 0.5 * (1.0 + np.cos(2.0 * np.pi * np.arange(n) / max(1, n)))
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def blackman_window(n):
    x = np.arange(n)
    w = 0.42 - 0.5 * np.cos(2 * np.pi * x / max(1, n - 1))
    w += 0.08 * np.cos(4 * np.pi * x / max(1, n - 1))
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def interpolate_seq(f, new_n):
    old_n = len(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s < 1e-10:
        return np.ones(new_n) / new_n
    return f_new / s

    multi-length exploration, perturbation restarts, and subgradient polishing.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = flat_top_window(1152)
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default flat-top: n=1152, score={p_score:.8f}")

    best_seq = parent.copy()
    best_score = p_score

    # === Phase 1: Primary LP optimization on parent ===
    print(f"\n{'='*60}")
    print(f"Phase 1: LP optimization (n={len(parent)})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.35, 550.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        f1, peak1 = main_optimize(parent, time_limit=t1, top_k_init=80, max_constraints=400)
        score1 = compute_score(f1)
        if score1 < best_score:
            best_score = score1
            best_seq = f1.copy()
            print(f"  NEW BEST: score={best_score:.8f}")
        else:
            print(f"  Phase 1: score={score1:.8f} (parent was {best_score:.8f})")

    # === Phase 2: Subgradient polish ===
    print(f"\n{'='*60}")
    print(f"Phase 2: Subgradient polish (n={len(best_seq)})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.08, 60.0)
    if t2 > 10:
        f2, peak2 = subgradient_polish(best_seq, time_limit=t2, n_top=30)
        score2 = compute_score(f2)
        if score2 < best_score:
            best_score = score2
            best_seq = f2.copy()
            print(f"  NEW BEST: score={best_score:.8f}")
        else:
            print(f"  Phase 2: score={score2:.8f}")

    # === Phase 3: LP deep pass ===
    print(f"\n{'='*60}")
    print(f"Phase 3: LP deep (n={len(best_seq)})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.22, 300.0)
    if t3 > 30:
        print(f"  Budget: {t3:.0f}s")
        f3, peak3 = main_optimize(best_seq, time_limit=t3, top_k_init=100, max_constraints=500)
        score3 = compute_score(f3)
        if score3 < best_score:
            best_score = score3
            best_seq = f3.copy()
            print(f"  NEW BEST: score={best_score:.8f}")
        else:
            print(f"  Phase 3: score={score3:.8f}")

    # === Phase 4: Multi-length exploration ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Multi-length exploration")
    print(f"{'='*60}")
    n_base = len(best_seq)
    candidate_lengths = []
    for factor in [0.85, 0.9, 0.95, 1.05, 1.1, 1.15, 1.2, 1.3, 1.5, 2.0]:
        target_n = int(round(n_base * factor))
        if 512 <= target_n <= 4096 and target_n != n_base:
            candidate_lengths.append(target_n)
    # Also try some absolute values
    for abs_n in [1024, 1280, 1536, 2048, 2560]:
        if 512 <= abs_n <= 4096 and abs_n != n_base:
            candidate_lengths.append(abs_n)
    candidate_lengths = sorted(set(candidate_lengths), key=lambda x: abs(x - n_base))

    for target_n in candidate_lengths[:5]:
        rem = deadline - time.monotonic()
        if rem < 35:
            break
        t4 = min(rem * 0.1, 90.0)
        if t4 < 15:
            break

        print(f"  Trying n={target_n} (budget={t4:.0f}s)")
        inits = [interpolate_seq(best_seq, target_n), flat_top_window(target_n)]
        best_for_len = None
        best_score_len = float('inf')

        for init in inits:
            rem_init = (deadline - time.monotonic()) * 0.45
            rem_init = min(rem_init, t4 / len(inits) + 5)
            if rem_init < 8:
                break
            f_reg, obj_reg = main_optimize(init, time_limit=rem_init,
                                            top_k_init=60, max_constraints=300, verbose=False)
            score_reg = compute_score(f_reg)
            if score_reg < best_score_len:
                best_score_len = score_reg
                best_for_len = f_reg.copy()

        if best_for_len is not None and best_score_len < best_score:
            best_score = best_score_len
            best_seq = best_for_len.copy()
            print(f"  NEW BEST at n={target_n}: score={best_score:.8f}")
        else:
            print(f"  n={target_n}: best={best_score_len:.8f} (no improvement)")

    # === Phase 5: Perturbation + re-optimization ===
    print(f"\n{'='*60}")
    print(f"Phase 5: Perturbation + re-optimization (n={len(best_seq)})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic()
    if rem > 50:
        n_pert = len(best_seq)
        num_attempts = min(4, int(rem // 30))
        for attempt in range(num_attempts):
            rem = deadline - time.monotonic()
            if rem < 25:
                break
            t5 = min(rem * 0.25, 55.0)
            if t5 < 10:
                break

            # Different perturbation strategies
            if attempt % 3 == 0:
                pert = rng.standard_normal(n_pert) * 0.008
            elif attempt % 3 == 1:
                # Targeted: perturb near peaks
                b = fast_convolve(best_seq)
                peak_idx = b.argmax()
                center = int(peak_idx)
                width = n_pert // 4
                pert = np.zeros(n_pert)
                lo = max(0, center - width)
                hi = min(n_pert, center + width)
                pert[lo:hi] = rng.standard_normal(hi - lo) * 0.015
            else:
                # Smooth perturbation
                pert = np.sin(2 * np.pi * rng.uniform(1, 5) * np.arange(n_pert) / n_pert) * 0.005
                pert += rng.standard_normal(n_pert) * 0.003

            f_pert = best_seq + pert
            f_pert = np.maximum(f_pert, 0)
            f_pert = f_pert / f_pert.sum()

            f5, obj5 = main_optimize(f_pert, time_limit=t5,
                                      top_k_init=80, max_constraints=400, verbose=False)
            score5 = compute_score(f5)
            if score5 < best_score:
                best_score = score5
                best_seq = f5.copy()
                print(f"  Attempt {attempt+1}: NEW BEST score={best_score:.8f}")
            else:
                print(f"  Attempt {attempt+1}: score={score5:.8f}")

    # === Phase 6: Final LP burst ===
    print(f"\n{'='*60}")
    print(f"Phase 6: Final LP (n={len(best_seq)}, score={best_score:.8f})")
    print(f"{'='*60}")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.4, 120.0)
    if t6 > 15:
        print(f"  Budget: {t6:.0f}s")
        f6, peak6 = main_optimize(best_seq, time_limit=t6, top_k_init=100, max_constraints=500)
        score6 = compute_score(f6)
        if score6 < best_score:
            best_score = score6
            best_seq = f6.copy()
            print(f"  NEW BEST: score={best_score:.8f}")
        else:
            print(f"  Final LP: score={score6:.8f}")

    # === Phase 7: Final subgradient polish ===
    rem = deadline - time.monotonic()
    if rem > 8:
        t7 = min(rem * 0.35, 25.0)
        if t7 > 5:
            f7, obj7 = subgradient_polish(best_seq, time_limit=t7, n_top=25)
            score7 = compute_score(f7)
            if score7 < best_score:
                best_score = score7
                best_seq = f7.copy()
                print(f"  Final SG: NEW BEST score={best_score:.8f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        if fs < best_score:
            best_score = fs
    else:
        print("WARNING: final sequence invalid, falling back to parent")
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"Target: 1.50270000")
    print(f"Gap: {best_score - 1.5027:+.6f}")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
