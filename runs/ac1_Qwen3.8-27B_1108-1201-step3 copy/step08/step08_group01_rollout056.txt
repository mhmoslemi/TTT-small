import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    global _lp_solver
    if '_lp_solver' in globals():
        try:
            if _lp_solver is not None:
                return _lp_solver
        except Exception:
            pass
    _lp_solver = None
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.ECOS, cp.SCS, cp.PDLP, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.GLPK, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def exact_coord_descent(a0, time_limit, verbose=False, label=""):
    """
    Exact coordinate descent: for each coordinate, binary search on the
    optimal delta by checking 1D feasibility of the minimax problem.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            ai = a[i]

            # Skip if very small (but allow small increases from zero)
            if ai < 1e-18:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            # Unaffected max
            unaff = -1e30
            if k_start > 0:
                unaff = max(unaff, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff = max(unaff, float(conv[k_end + 1:].max()))
            if unaff < 0:
                unaff = 0.0

            # If unaffected max is already >= M, no improvement possible
            if unaff >= M - 1e-13:
                continue

            affected_conv = conv[k_start:k_end + 1]
            shifted_a = a  # a[k-i] for k=i..i+n-1 gives a[0]..a[n-1]

            # Binary search on T (target max value for affected positions)
            # Feasible range: [unaff, M]
            T_lo = unaff
            T_hi = M
            best_delta = 0.0
            found = False

            for _ in range(40):
                T_mid = 0.5 * (T_lo + T_hi)

                # Check feasibility: exists delta >= -ai such that
                # for all j in [0, n-1], j != i: conv[i+j] + 2*delta*a[j] <= T_mid
                # and: conv[2i] + 2*delta*ai + delta^2 <= T_mid

                # Linear constraints (exclude j=i which is the quadratic term)
                # For j where a[j] > 0: delta <= (T_mid - conv[i+j]) / (2*a[j])
                # For j where a[j] < 0: delta >= (T_mid - conv[i+j]) / (2*a[j])
                # For j where a[j] == 0: need conv[i+j] <= T_mid

                pos_mask = shifted_a > 1e-18
                neg_mask = shifted_a < -1e-18
                # Exclude j=i from linear constraints
                if 0 <= i < n:
                    pos_mask = pos_mask.copy()
                    neg_mask = neg_mask.copy()
                    pos_mask[i] = False
                    neg_mask[i] = False

                feasible = True

                # Check zero-coefficient positions
                zero_mask = (shifted_a <= 1e-18) & (shifted_a >= -1e-18)
                zero_mask[i] = False
                if zero_mask.any():
                    if float(affected_conv[zero_mask].max()) > T_mid:
                        feasible = False

                if feasible:
                    # Upper bounds from positive coefficients
                    if pos_mask.any():
                        ub = (T_mid - affected_conv[pos_mask]) / (2.0 * shifted_a[pos_mask])
                        min_ub = float(ub.min())
                    else:
                        min_ub = 1e30

                    # Lower bounds from negative coefficients
                    if neg_mask.any():
                        lb = (T_mid - affected_conv[neg_mask]) / (2.0 * shifted_a[neg_mask])
                        max_lb = float(lb.max())
                    else:
                        max_lb = -1e30

                    # Quadratic constraint at j=i: conv[2i] + 2*delta*ai + delta^2 <= T_mid
                    # delta^2 + 2*ai*delta + (conv[2i] - T_mid) <= 0
                    conv_2i = float(affected_conv[i])
                    disc = ai * ai - conv_2i + T_mid
                    if disc < 0:
                        feasible = False
                    else:
                        sqrt_disc = np.sqrt(disc)
                        q_lb = -ai - sqrt_disc
                        q_ub = -ai + sqrt_disc

                if feasible:
                    total_lb = max(-ai, max_lb, q_lb)
                    total_ub = min(min_ub, q_ub)
                    if total_lb <= total_ub:
                        # Feasible! Record and tighten upper bound
                        best_delta = 0.5 * (total_lb + total_ub)
                        T_hi = T_mid
                        found = True
                    else:
                        T_lo = T_mid
                else:
                    T_lo = T_mid

            if found and abs(best_delta) > 1e-15:
                new_ai = ai + best_delta
                if new_ai < 0:
                    new_ai = 0.0
                    best_delta = -ai
                a[i] = new_ai
                actual_delta = new_ai - ai

                if abs(actual_delta) > 1e-16:
                    # Update convolution
                    linear_changes = 2.0 * actual_delta * shifted_a
                    # Fix the quadratic term at position i (k=2i)
                    linear_changes = linear_changes.copy()
                    linear_changes[i] = 2.0 * actual_delta * ai + actual_delta * actual_delta
                    conv[k_start:k_end + 1] += linear_changes

                    new_aff_max = float(conv[k_start:k_end + 1].max())
                    M = max(unaff, new_aff_max)
                    M = max(M, 0.0)

                    cur_score = 2.0 * n * M
                    if cur_score < best_score - 1e-14:
                        best_score = cur_score
                        best_a = a.copy()
                    improved = True

        if not improved:
            stall += 1
            if stall > 30:
                break
        else:
            stall = 0

        if verbose and sweeps % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ECD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.7f} M={M:.8f}")

    return best_a, best_score, sweeps

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def init_fourier(n, rng):

    n_harmonics = int(rng.integers(3, 20))
    amplitudes = rng.random(n_harmonics) * 0.4
    phases = rng.uniform(0, 2 * np.pi, n_harmonics)
    freqs = np.arange(1, n_harmonics + 1)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harmonics):
        a += amplitudes[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_cosine(n, rng):
    """Initialize with a cosine shape."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    width = n * rng.uniform(0.3, 0.6)
    a = np.exp(-0.5 * ((t - center) / width) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_u_shape(n, rng):
    """Initialize with U-shape (more mass at edges)."""
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    alpha = rng.uniform(0.5, 2.5)
    a = 1.0 + alpha * np.cos(np.pi * t) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def perturb_sequence(a, rng, magnitude=0.03):
    """Perturb the sequence to escape local optima."""
    n = len(a)
    a_new = a.copy()
    conv = fast_convolve(a)
    top_k = min(10, len(conv))
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    for k in peaks:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1)
        n_pick = max(2, len(idx) // 5)
        picks = rng.choice(idx, size=min(n_pick, len(idx)), replace=False)
        for i in picks:
            a_new[i] *= (1.0 + rng.standard_normal() * magnitude)
    # Also add global noise
    a_new += rng.standard_normal(n) * magnitude * 0.5
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def check_and_update(a, best_score, label=""):
    """Evaluate candidate and return updated best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return None, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        print(f"  NEW BEST [{label}]: {score:.7f} (n={len(a_norm)})")
        return a_norm, best_score
    return None, best_score
