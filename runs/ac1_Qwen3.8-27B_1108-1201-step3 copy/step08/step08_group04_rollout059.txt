    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp(f, M, constraint_indices, n):
    """
    Solve: max sum(g) s.t. (f*g)[k] <= M for k in constraint_indices, g >= 0
    Returns optimal g or None.
    """
    K = len(constraint_indices)
    if K == 0:
        return np.ones(n)
    
    # Build constraint matrix
    A = np.zeros((K, n))
    for i, k in enumerate(constraint_indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi+1] = f[max(0,k-n+1):hi+1]  # This needs care

    A = np.zeros((K, n))
    for i, k in enumerate(constraint_indices):
        # (f*g)[k] = sum_j f[j] * g[k-j] for j in [max(0,k-n+1), min(n-1,k)]
        # = sum_l g[l] * f[k-l] for l in [max(0,k-n+1), min(n-1,k)]  (with l = k-j)

        # l ranges from max(0, k-(n-1)) to min(n-1, k)
        lo = max(0, k - n + 1)

        # A[i, l] = f[k - l] for l in [lo, hi]
        for l in range(lo, hi + 1):
            A[i, l] = f[k - l]
    
    # This loop is slow for large n. Let me vectorize.
    # Actually for the vectorized version:
    # For each constraint k, the row is: A[i, j] = f[k-j] if 0 <= k-j < n, else 0
    # = f[k-j] if j in [k-n+1, k] intersect [0, n-1]
    
    # Let me just use the vectorized approach
    A = np.zeros((K, n))
    for i, k in enumerate(constraint_indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # For g indices l in [lo, hi]: A[i,l] = f[k-l]
        # f[k-l] for l in [lo,hi] = f[k-hi : k-lo+1][::-1]
        A[i, lo:hi+1] = f[k-hi:k-lo+1][::-1]
    
    g = cp.Variable(n)
    constraints = [g >= 0]
    constraints.append(A @ g <= M)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    
    # Try solvers in order
    for solver in [cp.ECOS, cp.SCS, cp.GLOP_CPLEX, cp.PDLP]:

            break
        except Exception:
            continue
    
    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0)
    if np.any(np.isnan(g_val)):
        return None
    return g_val

Hmm, the vectorized constraint matrix building is: for constraint k, A[i, l] = f[k-l] for l in [max(0,k-n+1), min(n-1,k)].

Let me verify: (f*g)[k] = sum_{j=0}^{n-1} f[j] * g[k-j] where g[m] = 0 for m outside [0,n-1].
So the non-zero terms are those where 0 <= k-j < n, i.e., k-n+1 <= j <= k, intersected with 0 <= j < n.
So j ranges from max(0, k-n+1) to min(n-1, k).

Now, (f*g)[k] = sum_{j=max(0,k-n+1)}^{min(n-1,k)} f[j] * g[k-j]

Let l = k-j, so j = k-l. When j = max(0,k-n+1), l = k - max(0,k-n+1) = min(k, n-1).
When j = min(n-1,k), l = k - min(n-1,k) = max(0, k-n+1).

So l ranges from max(0, k-n+1) to min(n-1, k), and A[i,l] = f[k-l].

So A[i, l] = f[k-l] for l in [max(0,k-n+1), min(n-1,k)].

In NumPy: for l in [lo, hi] where lo = max(0,k-n+1), hi = min(n-1,k):
A[i, lo:hi+1] = f[k-lo : k-hi+1][::-1]

Because f[k-l] for l=lo gives f[k-lo], and for l=hi gives f[k-hi].
So the array is f[k-lo], f[k-lo-1], ..., f[k-hi] which is f[k-hi:k-lo+1][::-1].

Let me verify with an example: n=5, k=3.
lo = max(0, 3-4) = 0, hi = min(4, 3) = 3.
A[i, 0:4] should be [f[3], f[2], f[1], f[0]].
f[k-hi:k-lo+1][::-1] = f[3-3:3-0+1][::-1] = f[0:4][::-1] = [f[3], f[2], f[1], f[0]]. ✓

Wait, I realize I should also think about the relationship between the LP approach and the Frank-Wolfe algorithm.

The problem is: minimize f(a) = max_k (a*a)[k] subject to a in Δ (simplex).

Frank-Wolfe would:
1. At current a, compute subgradient g = (e_{k*} * a) where k* = argmax (a*a)
2. Solve linear minimization: min_{s in Δ} <g, s> = min_{s in Δ} (a*s)[k*]
   The optimal s is a point mass at argmin_j a[k*-j] (i.e., s[j] = 1 where j minimizes a[k*-j])
3. Move: a <- (1-t)*a + t*s

This is much simpler than the LP approach but uses only one active constraint. The LP approach uses multiple constraints, which should give better directions.

However, there's a middle ground: use a "composite" subgradient that accounts for multiple peaks. This is essentially what the LP does, but we can approximate it.

For my implementation, I'll stick with the LP approach but make it more efficient and robust.

Let me also think about the "slope trick" or "Dinkelbach's algorithm" approach:
- The objective is 2*n*max_conv/sum^2. With sum=1, it's 2*n*max_conv.
- We want to minimize max_conv.
- Dinkelbach: minimize max_conv - λ for some λ, check if optimal < 0.

This doesn't directly help since the problem is non-convex.

"""
Optimized sequence search using LP-based Frank-Wolfe direction with 
adaptive constraint management, multi-scale exploration, and diverse 
polishing methods to minimize 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f, indices, n):
    """Build constraint matrix for LP: A[i,l] = f[indices[i] - l] if valid."""
    K = len(indices)
    A = np.zeros((K, n))
    for i in range(K):
        k = indices[i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi+1] = f[hi:k-lo+1][::-1] if hi >= lo else 0
    return A


def solve_direction_lp(f, M, indices, n):
    """
    Solve: max sum(g) s.t. (f*g)[k] <= M for k in indices, g >= 0.
    Returns (g_opt, obj_value) or (None, 0).
    """
    K = len(indices)
    if K == 0:
        return np.ones(n), n
    
    A = build_lp_matrix(f, indices, n)
    g = cp.Variable(n)
    obj = cp.Maximize(cp.sum(g))
    cons = [g >= 0, A @ g <= M]
    prob = cp.Problem(obj, cons)
    
    last_err = None
    for solver_name in ['ECOS', 'SCS', 'GLOP_CPLEX', 'PDLP', 'CLARABEL']:
        try:
            solver = getattr(cp, solver_name, None)
            if solver is None:
                continue
            if solver_name == 'ECOS':
                prob.solve(solver=solver, max_iters=50000, abs_acceleration=1e-9)
            elif solver_name == 'SCS':
                prob.solve(solver=solver, max_iters=200000, eps=1e-9)
            elif solver_name == 'GLOP_CPLEX':
                prob.solve(solver=solver)
            elif solver_name == 'PDLP':
                prob.solve(solver=solver, max_iter=100000)
            elif solver_name == 'CLARABEL':
                prob.solve(solver=solver)
            
            if g.value is not None:
                g_val = np.maximum(g.value, 0.0)
                if np.all(np.isfinite(g_val)):
                    return g_val, float(prob.value)
        except Exception as e:
            last_err = e
            continue
    
    return None, 0.0


def line_search(f, direction, current_max_conv):
    """
    Find best step size t: f_new = (1-t)*f + t*direction, normalize, 
    minimize max_conv(f_new).
    Returns (best_t, best_f, best_obj).
    """
    best_t = 0.0
    best_obj = current_max_conv
    best_f = f.copy()
    
    # Try several step sizes
    for t in [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7]:
        f_trial = (1.0 - t) * f + t * direction
        f_trial = np.maximum(f_trial, 0.0)
        s = f_trial.sum()
        if s < 1e-15:
            continue
        f_trial = f_trial / s
        obj = fast_convolve(f_trial).max()
        if obj < best_obj:
            best_obj = obj
            best_t = t
            best_f = f_trial.copy()
    
    # Refine around best_t if it's not at the boundary
    if 0.05 < best_t < 0.6:
        for t in np.linspace(max(0.01, best_t - 0.1), min(0.9, best_t + 0.1), 9):
            f_trial = (1.0 - t) * f + t * direction
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-15:
                continue
            f_trial = f_trial / s
            obj = fast_convolve(f_trial).max()
            if obj < best_obj:
                best_obj = obj
                best_t = t
                best_f = f_trial.copy()
    
    return best_t, best_f, best_obj


def lp_optimize(f_init, time_limit, max_constraints=500, label="lp"):
    """
    LP-based optimization with adaptive constraint set.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    
    # Start with just the top peak
    b = fast_convolve(f)
    constraint_set = {int(np.argmax(b))}
    
    it = 0
    stuck = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        b = fast_convolve(f)
        M = b.max()
        
        # Solve LP with current constraint set
        indices = sorted(constraint_set)
        # Cap the number of constraints for efficiency
        if len(indices) > max_constraints:
            # Keep the most important constraints (those closest to M)
            vals = [b[i] for i in indices]
            order = np.argsort(vals)[::-1][:max_constraints]
            indices = sorted([indices[i] for i in order])
        
        g, lp_obj = solve_direction_lp(f, M, indices, n)
        
        if g is None:
            stuck += 5
            if stuck > 50:
                break
            # Try with fewer constraints
            top_5 = np.argpartition(b, -5)[-5:]
            constraint_set = set(int(x) for x in top_5)
            continue
        
        sum_g = g.sum()
        if sum_g <= 1.0 + 1e-8:
            # LP says we can't improve - add more constraints or stop
            stuck += 1
            if stuck > 30:
                # Try to restart with perturbation
                rng = np.random.default_rng(it * 12345)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.01)
                b = fast_convolve(f)
                constraint_set = set(int(x) for x in np.argpartition(b, -10)[-10:])
                stuck = 0
                continue
            # Add the next most peaked constraint
            # Find peak not in constraint_set
            b_copy = b.copy()
            for idx in constraint_set:
                b_copy[idx] = -np.inf
            if b_copy.max() > 0:
                constraint_set.add(int(np.argmax(b_copy)))
            else:
                # All peaks are constrained, try different M
                M_new = M * 0.999
                g, lp_obj = solve_direction_lp(f, M_new, indices, n)
                if g is None or g.sum() <= 1.0 + 1e-8:
                    break
                sum_g = g.sum()
        
        if sum_g <= 1.0 + 1e-8:
            break
        
        # Normalize direction
        h = g / sum_g
        
        # Line search
        t, f_new, obj_new = line_search(f, h, M)
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_score = 2.0 * n * best_obj
            best_f = f_new.copy()
            f = f_new
            stuck = 0
            
            # Update constraint set: add peaks that are now close to max
            b_new = fast_convolve(f)
            M_new = b_new.max()
            # Add peaks within 5% of max
            threshold = M_new * 0.95
            for k in range(len(b_new)):
                if b_new[k] > threshold:
                    constraint_set.add(k)
            # Cap constraint set size
            if len(constraint_set) > max_constraints:
                # Keep only the most important ones
                peaks = list(constraint_set)
                peak_vals = [(b_new[p], p) for p in peaks]
                peak_vals.sort(reverse=True)
                constraint_set = set(p for _, p in peak_vals[:max_constraints])
        else:
            stuck += 1
            if stuck > 40:
                # Perturb and restart
                rng = np.random.default_rng(it * 67890)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                b = fast_convolve(f)
                constraint_set = set(int(x) for x in np.argpartition(b, -20)[-20:])
                stuck = 0
                if time.monotonic() - t0 > time_limit * 0.9:
                    break
        
        if it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} it={it} obj={best_obj:.10f} score={best_score:.6f} "
                  f"|S|={len(constraint_set)} t={elapsed:.0f}s")
    
    # Final check
    final_obj = fast_convolve(best_f).max()
    final_score = 2.0 * n * final_obj
    return best_f, final_score


def subgradient_polish(f_init, time_limit, n_iter_per_check=100):
    """Subgradient method with adaptive step size."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    v = np.zeros(n)  # momentum
    momentum = 0.9
    lr = 0.01
    it = 0
    no_improve = 0
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()
        k_star = int(np.argmax(b))
        
        # Subgradient: (e_{k*} * f)[j] = f[k*-j] if valid, else 0
        sg = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        for j in range(lo, hi + 1):
            sg[j] = f[k_star - j]
        
        # Update
        v = momentum * v - lr * sg
        f = project_simplex(f + v)
        
        if it % n_iter_per_check == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
                lr = min(lr * 1.02, 0.05)
            else:
                no_improve += 1
                if no_improve > 3:
                    lr *= 0.9
                    no_improve = 0
                if no_improve > 20:
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                    v = np.zeros(n)
                    no_improve = 0
    
    return best_f, 2.0 * n * best_obj


def coordinate_descent(f_init, time_limit):
    """Coordinate descent with multiple multipliers."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    multipliers = [0.3, 0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0]
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-15:
                continue
            
            best_val = f[j]
            best_trial_obj = best_obj
            
            for mult in multipliers:
                f_trial = f.copy()
                f_trial[j] = f[j] * mult
                s = f_trial.sum()
                if s < 1e-12:
                    continue
                f_trial = f_trial / s
                obj = fast_convolve(f_trial).max()
                if obj < best_trial_obj - 1e-13:
                    best_trial_obj = obj
                    best_val = f[j] * mult
            
            if best_val != f[j]:
                f[j] = best_val
                s = f.sum()
                if s > 1e-12:
                    f = f / s
                if best_trial_obj < best_obj - 1e-13:
                    best_obj = best_trial_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_f, 2.0 * n * best_obj


def make_init(n, method="uniform", seed=0):
    """Create various initializations."""
    if method == "uniform":
        return np.ones(n) / n
    elif method == "hann":
        t = np.arange(n)
        w = 0.5 * (1 + np.cos(2 * np.pi * t / (n - 1)))
        return w / w.sum()
    elif method == "edge":
        # Concentrate at edges
        t = np.arange(n)
        w = np.exp(-2.0 * t / n) + np.exp(-2.0 * (n - 1 - t) / n)
        return w / w.sum()
    elif method == "two_clump":
        # Two peaks at 1/3 and 2/3
        t = np.arange(n)
        sigma = n / 8.0
        w = np.exp(-0.5 * ((t - n/3.0) / sigma)**2) + np.exp(-0.5 * ((t - 2*n/3.0) / sigma)**2)
        return w / w.sum()
    elif method == "random":
        rng = np.random.default_rng(seed)
        w = rng.random(n)
        return w / w.sum()
    else:
        return np.ones(n) / n


def interp_sequence(src, target_n):
    """Interpolate a sequence to a new length."""
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result

The LP: max sum(g) s.t. (f*g)[k] <= M for k in S, g >= 0

The optimal value sum(g*) > 1 iff there exists a g with sum > 1 that satisfies all constraints. This means we can normalize g* to h = g*/sum(g*) and have (f*h)[k] <= M/sum(g*) < M for all k in S.

The key insight is: if we move from f to h (or (1-t)f + th), the new self-convolution will be:
((1-t)f + th) * ((1-t)f + th) = (1-t)^2 (f*f) + t(1-t)(f*h + h*f) + t^2 (h*h)

For small t, this is approximately:
(f*f) + t((f*h + h*f) - 2(f*f)) + O(t^2)

We want this to be <= M for all k. The first-order condition is:
(f*h)[k] + (h*f)[k] - 2(f*f)[k] <= 0 for all k

Note that (f*h)[k] = (h*f)[k] by commutativity of convolution. So:
2(f*h)[k] - 2(f*f)[k] <= 0, i.e., (f*h)[k] <= (f*f)[k] for all k.

So the LP constraint (f*g)[k] <= M = (f*f)[k_max] is a sufficient (but not necessary) condition for improvement. The necessary condition is (f*h)[k] <= (f*f)[k] for ALL k, not just the tracked ones.

This means: if the LP says sum(g) > 1 with constraints only on S, the resulting h might still have (f*h)[k] > (f*f)[k] for some k not in S, which could cause (h*h)[k] to increase.

This is why the adaptive constraint set is important: we need to track enough peaks to ensure the improvement is valid.

Uses LP-based Frank-Wolfe directions with adaptive constraint sets,
multi-scale n exploration, diverse restarts, and multi-method polishing.

- Adaptive growing constraint set with violation detection

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f, indices, n):
    """Build K x n matrix where A[i,l] = f[indices[i] - l] if 0<=indices[i]-l<n."""
    K = len(indices)
    A = np.zeros((K, n))
    for i in range(K):
        k = indices[i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if hi >= lo:
            A[i, lo:hi + 1] = f[hi:k - lo + 1][::-1]
    return A


def solve_direction_lp(f, M, indices, n):
    """
    Solve: max sum(g) s.t. (f*g)[k] <= M for k in indices, g >= 0.
    Returns (g_opt, obj_value) or (None, 0).
    """
    K = len(indices)
    if K == 0:
        return np.ones(n), n

    if K > 1500:
        # Too many constraints, subsample
        idx_sorted = sorted(range(K), key=lambda i: -M)  # all same M, just take first
        indices = indices[:1200]
        K = 1200

    A = build_lp_matrix(f, indices, n)
    g = cp.Variable(n)
    obj = cp.Maximize(cp.sum(g))
    cons = [g >= 0, A @ g <= M]
    prob = cp.Problem(obj, cons)

    for solver_name in ['ECOS', 'SCS', 'GLOP_CPLEX', 'PDLP']:
        try:
            solver = getattr(cp, solver_name, None)
            if solver is None:
                continue
            if solver_name == 'ECOS':
                prob.solve(solver=solver, max_iters=80000, abs_acceleration=1e-10)
            elif solver_name == 'SCS':
                prob.solve(solver=solver, max_iters=300000, eps=1e-10)
            elif solver_name == 'GLOP_CPLEX':
                prob.solve(solver=solver)
            elif solver_name == 'PDLP':
                prob.solve(solver=solver, max_iter=200000)

            if g.value is not None and prob.value is not None:
                g_val = np.maximum(g.value, 0.0)
                if np.all(np.isfinite(g_val)):
                    return g_val, float(prob.value)
        except Exception:
            continue

    try:

        if g.value is not None:
            g_val = np.maximum(g.value, 0.0)
            if np.all(np.isfinite(g_val)):
                return g_val, float(cp.sum(g_val).value)
    except Exception:
        pass

    return None, 0.0


def line_search_dense(f, direction, current_obj):
    """

    best_obj = current_obj
    best_f = f.copy()

        f_trial = np.maximum(f_trial, 0.0)
        s = f_trial.sum()
        if s < 1e-15:

        f_trial = f_trial / s

            print(f"    {label} it={it} obj={best_obj:.10f} score={best_score:.6f} "
                  f"|S|={len(constraint_set)} t={elapsed:.0f}/{time_limit:.0f}s")

    final_obj = fast_convolve(best_f).max()
    return best_f, 2.0 * n * final_obj


def subgradient_polish(f_init, time_limit):
    """Subgradient with softmax peak weighting and adaptive LR."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    v = np.zeros(n)
    momentum = 0.88
    lr = 0.005
    temp = 0.0003
    it = 0
    no_improve = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()

        # Softmax weighting on peaks
        scaled = (b - M) / temp
        np.clip(scaled, -50, 50, out=scaled)
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(b))
            w[int(np.argmax(b))] = 1.0
        else:
            w = w / w_sum

        # Restrict to top peaks for efficiency
        n_top = min(n, len(b))
        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
            w_sel = w[top_idx].copy()
            ws = w_sel.sum()
            if ws > 1e-30:
                w_sel = w_sel / ws
            w_new = np.zeros(len(b))
            w_new[top_idx] = w_sel
            w = w_new

        # Compute weighted subgradient
        g = np.zeros(n)
        active = np.where(w > 1e-12)[0]
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                g[j] += w[k] * f[k - j]

        v = momentum * v - lr * g
        f = project_simplex(f + v)

        if it % 50 == 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
                lr = min(lr * 1.03, 0.02)
                temp = max(temp * 0.95, 1e-5)
            else:
                no_improve += 1
                if no_improve > 3:
                    lr *= 0.93
                    no_improve = 0
                if no_improve > 20:
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                    v = np.zeros(n)
                    no_improve = 0

            if it % 500 == 0:
                print(f"    sg it={it} obj={best_obj:.10f} score={2*n*best_obj:.6f} "
                      f"lr={lr:.5f} t={time.monotonic()-t0:.0f}s")

    return best_f, 2.0 * n * best_obj


def coordinate_descent(f_init, time_limit):
    """Coordinate descent with multipliers."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    multipliers = [0.3, 0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0]

    passes = 0
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-15:
                continue
            best_val = f[j]
            best_trial_obj = best_obj
            for mult in multipliers:
                f_trial = f.copy()
                f_trial[j] = f[j] * mult
                s = f_trial.sum()
                if s < 1e-12:
                    continue
                f_trial = f_trial / s
                obj = fast_convolve(f_trial).max()
                if obj < best_trial_obj - 1e-13:
                    best_trial_obj = obj
                    best_val = f[j] * mult
            if best_val != f[j]:
                f[j] = best_val
                s = f.sum()
                if s > 1e-12:
                    f = f / s
                if best_trial_obj < best_obj - 1e-13:
                    best_obj = best_trial_obj
                    best_f = f.copy()
                    improved = True
        if not improved:
            break

    return best_f, 2.0 * n * best_obj


def make_init(n, method="uniform", seed=0):
    """Create various structured initializations."""
    if method == "uniform":
        return np.ones(n) / n
    elif method == "hann":
        t = np.arange(n)
        w = 0.5 * (1.0 + np.cos(2.0 * np.pi * t / max(1, n - 1)))
        return w / w.sum()
    elif method == "edge":
        t = np.arange(n)
        w = np.exp(-3.0 * t / n) + np.exp(-3.0 * (n - 1 - t) / n)
        return w / w.sum()
    elif method == "two_clump":
        t = np.arange(n)
        sigma = n / 6.0
        w = np.exp(-0.5 * ((t - n / 3.0) / sigma) ** 2) + np.exp(-0.5 * ((t - 2.0 * n / 3.0) / sigma) ** 2)
        return w / w.sum()
    elif method == "random":
        rng = np.random.default_rng(seed)
        w = rng.random(n)
        return w / w.sum()
    elif method == "cos_ramp":
        t = np.arange(n)
        w = np.sin(np.pi * (t + 0.5) / n) ** 2
        return w / w.sum()
    else:
        return np.ones(n) / n


def interp_sequence(src, target_n):
    """Interpolate a sequence to a new length."""
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining LP-based directions, multi-scale
    exploration, diverse restarts, and polishing heuristics.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    # Load parent
    parent = None
    parent_score = 2.0
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={parent_score:.6f}")
    except Exception as e:
        print(f"No parent: {e}")

    if parent is None:
        parent = np.ones(512) / 512.0
        parent_score = 2.0

    best_seq = parent.copy()
    best_score = parent_score

    # ============================================================
    # Phase 1: Multi-n exploration
    # ============================================================
    print(f"\n{'=' * 60}")
    print(f"Phase 1: n-exploration (best: {best_score:.6f})")
    print(f"{'=' * 60}")

    rem = deadline - time.monotonic()
    t1 = min(rem * 0.33, 520.0)

    if t1 > 80:
        n_parent = len(parent)
        n_candidates = []
        for factor in [0.5, 0.65, 0.8, 0.9, 1.0, 1.1, 1.25, 1.4, 1.7, 2.0]:
            n_val = int(n_parent * factor)
            n_val = max(250, min(2500, n_val))
            if n_val not in n_candidates:
                n_candidates.append(n_val)
        for n_fix in [300, 450, 600, 800, 1000, 1300, 1700, 2200]:
            if n_fix not in n_candidates:
                n_candidates.append(n_fix)
        n_candidates.sort()

        per_n = t1 / len(n_candidates)
        for n_trial in n_candidates:
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            alloc = min(per_n, rem * 0.8)
            if alloc < 40:
                break
            # Give more time to n values close to parent
            if abs(n_trial - n_parent) < n_parent * 0.2:
                alloc = min(alloc * 1.5, rem * 0.8)

            f_init = interp_sequence(parent, n_trial)
            print(f"  n={n_trial}, alloc={alloc:.0f}s")
            f_res, _ = lp_optimize(f_init, time_limit=alloc,
                                   max_constraints=400, label=f"E{n}")
            real_score = float(evaluate_sequence(f_res.tolist()))
            if np.isfinite(real_score) and real_score < best_score:
                best_score = real_score
                best_seq = f_res.copy()
                print(f"  NEW BEST: n={n_trial}, score={real_score:.6f}")

    # ============================================================
    # Phase 2: Deep LP on best n
    # ============================================================
    print(f"\n{'=' * 60}")
    print(f"Phase 2: Deep LP (best: {best_score:.6f}, n={len(best_seq)})")
    print(f"{'=' * 60}")

    rem = deadline - time.monotonic()
    t2 = min(rem * 0.28, 450.0)

    if t2 > 60:
        n_b = len(best_seq)
        print(f"  n={n_b}, alloc={t2:.0f}s")
        f_res, _ = lp_optimize(best_seq, time_limit=t2,
                               max_constraints=800, label="deep", use_momentum=True)
        real_score = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_score) and real_score < best_score:
            best_score = real_score
            best_seq = f_res.copy()
            print(f"  NEW BEST: score={real_score:.6f}")

    # ============================================================
    # Phase 3: Diverse restarts
    # ============================================================
    print(f"\n{'=' * 60}")
    print(f"Phase 3: Diverse restarts (best: {best_score:.6f})")
    print(f"{'=' * 60}")

    rem = deadline - time.monotonic()
    t3 = min(rem * 0.18, 280.0)

    if t3 > 60:
        n_b = len(best_seq)
        init_methods = ["hann", "edge", "two_clump", "cos_ramp", "random"]
        per_init = t3 / len(init_methods)

        for idx, method in enumerate(init_methods):
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            alloc = min(per_init, rem * 0.8)
            if alloc < 30:
                break

            f_raw = make_init(n_b, method=method, seed=seed + idx)
            # Blend with current best
            blend = 0.6 * best_seq + 0.4 * f_raw
            f_init = blend / blend.sum()

            print(f"  [{method}] alloc={alloc:.0f}s")
            f_res, _ = lp_optimize(f_init, time_limit=alloc,
                                   max_constraints=350, label=f"R{method}")
            real_score = float(evaluate_sequence(f_res.tolist()))
            if np.isfinite(real_score) and real_score < best_score:
                best_score = real_score
                best_seq = f_res.copy()
                print(f"  NEW BEST: score={real_score:.6f}")

    # ============================================================
    # Phase 4: Subgradient polish
    # ============================================================
    print(f"\n{'=' * 60}")
    print(f"Phase 4: SG polish (best: {best_score:.6f})")
    print(f"{'=' * 60}")

    rem = deadline - time.monotonic()
    t4 = min(rem * 0.12, 180.0)

    if t4 > 30:
        print(f"  n={len(best_seq)}, alloc={t4:.0f}s")
        f_res, _ = subgradient_polish(best_seq, time_limit=t4)
        real_score = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_score) and real_score < best_score:
            best_score = real_score
            best_seq = f_res.copy()
            print(f"  NEW BEST: score={real_score:.6f}")

    # ============================================================
    # Phase 5: Coordinate descent
    # ============================================================
    print(f"\n{'=' * 60}")
    print(f"Phase 5: CD (best: {best_score:.6f})")
    print(f"{'=' * 60}")

    rem = deadline - time.monotonic()
    t5 = min(rem * 0.06, 70.0)

    if t5 > 10:
        print(f"  n={len(best_seq)}, alloc={t5:.0f}s")
        f_res, _ = coordinate_descent(best_seq, time_limit=t5)
        real_score = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_score) and real_score < best_score:
            best_score = real_score
            best_seq = f_res.copy()
            print(f"  NEW BEST: score={real_score:.6f}")

    # ============================================================
    # Phase 6: Final LP push
    # ============================================================
    print(f"\n{'=' * 60}")
    print(f"Phase 6: Final LP (best: {best_score:.6f})")
    print(f"{'=' * 60}")

    rem = deadline - time.monotonic()
    t6 = max(rem * 0.85, 0.0)

    if t6 > 30:
        n_b = len(best_seq)
        print(f"  n={n_b}, alloc={t6:.0f}s")
        f_res, _ = lp_optimize(best_seq, time_limit=t6,
                               max_constraints=1200, label="final", use_momentum=True)
        real_score = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_score) and real_score < best_score:
            best_score = real_score
            best_seq = f_res.copy()
            print(f"  NEW BEST: score={real_score:.6f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'=' * 60}")
    print(f"FINAL: score={best_score:.6f}, n={len(best_seq)}")
    print(f"{'=' * 60}")

    return [float(x) for x in best_seq]
