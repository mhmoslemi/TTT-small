`irfft(conj(FFT(a)) * FFT(w), N)[i] = Σ_j conj(a[j]) * w[(j-i) mod N] = Σ_j a[j] * w[j-i]` (for real a).

Hmm, that gives `w[j-i]` not `w[j+i]`.

OK let me just try both and see which one gives the right answer numerically. Or better yet, let me use a different formulation.

Actually, the simplest approach: I want to compute `c[i] = Σ_k w[k] * a[k-i]` for `i = 0, ..., n-1`.

This is the same as: `c = convolve(w, flip(a))` (linear convolution), taking the first `n` elements? Let me check.

Linear convolution: `(w * a_flip)[m] = Σ_k w[k] * a_flip[m-k] = Σ_k w[k] * a[n-1-(m-k)] = Σ_k w[k] * a[n-1-m+k]`

I want `c[i] = Σ_k w[k] * a[k-i]`.

With `a_flip[m] = a[n-1-m]`:
`a[k-i] = a_flip[n-1-k+i]`

So `c[i] = Σ_k w[k] * a_flip[n-1-k+i]`

Let `m = n-1+i`:
`c[i] = Σ_k w[k] * a_flip[m-k] = (w * a_flip)[m] = (w * a_flip)[n-1+i]`

So `c = (w * a_flip)[n-1 : n-1+n]` (the middle n elements of the linear convolution of w and a_flip).

The linear convolution of two length-n arrays has length `2n-1`. The elements are indexed from 0 to 2n-2. I want elements from index `n-1` to `2n-2` (which is `n` elements).

Great, so:
```python
conv_wa = linear_conv(w, a[::-1])
sg = 2.0 * conv_wa[n-1 : 2*n-1]  # indices n-1 through 2n-2
```

Let me verify with a tiny example:
- `a = [a0, a1]`, `w = [w0, w1]` (both length 2)
- `a_flip = [a1, a0]`
- Linear conv of `[w0, w1]` and `[a1, a0]`:
  - index 0: w0*a1
  - index 1: w0*a0 + w1*a1
  - index 2: w1*a0
- I want elements from index n-1=1 to 2n-2=2: `[w0*a0 + w1*a1, w1*a0]`

Now let me compute directly:
- `c[0] = Σ_k w[k] * a[k-0] = w0*a0 + w1*a1` ✓
- `c[1] = Σ_k w[k] * a[k-1] = w0*a[-1] + w1*a[0] = 0 + w1*a0 = w1*a0` ✓

So the formula `sg = 2.0 * conv_wa[n-1 : 2*n-1]` is correct.

"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Combines Nesterov-accelerated subgradient, LP fictitious play, mirror descent,
and coordinate descent with adaptive time allocation.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.linalg import toeplitz
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = next_fast_len(size)
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.max(np.nonzero(cond)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def soft_subgradient(a, T=1.0):
    """Compute soft subgradient of max(conv(a,a)) using temperature T."""
    n = len(a)
    conv = fast_convolve(a)
    max_val = conv.max()
    shifted = conv - max_val
    weights = np.exp(shifted / T)
    wsum = weights.sum()
    if wsum < 1e-300:
        weights = np.zeros_like(conv)
        weights[np.argmax(conv)] = 1.0
    else:
        weights /= wsum
    
    # Compute 2 * sum_k weights[k] * a[k-i] for i = 0..n-1
    # = 2 * (linear_conv(weights, a_flip))[n-1 : 2n-1]
    a_flip = a[::-1]
    fft_size = next_fast_len(2 * n - 1)
    W = rfft(weights, fft_size)
    F = rfft(a_flip, fft_size)
    conv_wa = irfft(W * F, fft_size)[:2*n-1]
    grad = 2.0 * conv_wa[n-1:]
    return grad, max_val


def nesterov_subgradient(a0, time_limit, T_init=5.0, T_min=0.01, verbose=True):
    """Nesterov-accelerated subgradient method on the simplex."""
    n = len(a0)
    a = a0.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score(a0)
    a = a / s
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    # Nesterov parameters
    a_prev = a.copy()
    T = T_init
    
    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Extrapolation
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * (it + 1)**2)) / 2.0
        t_km1 = (1.0 + np.sqrt(1.0 + 4.0 * it**2)) / 2.0 if it > 0 else 1.0
        beta = (t_km1 - 1.0) / t_k if it > 0 else 0.0
        y = a + beta * (a - a_prev)
        a_prev = a.copy()
        
        # Clip y to simplex
        y = np.maximum(y, 0)
        ys = y.sum()
        if ys < 1e-15:
            y = a.copy()
        else:
            y = y / ys
        
        # Compute subgradient
        grad, max_val = soft_subgradient(y, T)
        
        # Step size: eta = c / sqrt(it + 1)
        # For the objective 2n*max(conv), the gradient norm is bounded by ~2n*max(a)
        # Use adaptive step
        eta = 1.0 / (np.sqrt(it + 1) * np.sqrt(n))
        
        a_new = y - eta * grad
        a_new = project_simplex(a_new)
        
        # Adaptive temperature
        T = max(T_min, T_init / (1.0 + 0.01 * (it + 1)))
        
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"  NS iter={it+1}, score={best_score:.8f}, T={T:.4f}, time={elapsed:.0f}s")
    
    return best_a, best_score


def build_conv_matrix(f):
    n = len(f)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)


def solve_best_response(f, M, prev_g=None):
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    if prev_g is not None:
        g.value = prev_g
    C = build_conv_matrix(f)
    constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL, warm_start=(prev_g is not None))
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            try:
                prob.solve(solver=cp.PDLP)
            except Exception:
                return prev_g if prev_g is not None else f.copy()
    if g.value is None:
        return prev_g if prev_g is not None else f.copy()
    return np.maximum(g.value, 0)


def lp_fictitious_play(a0, time_limit, step=0.4, verbose=True, label="LP-FP"):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    prev_g = None
    stall_count = 0
    
    while time.monotonic() - t0 < time_limit and iters < 50000:
        iters += 1
        conv = fast_convolve(a)
        M = conv.max()
        g = solve_best_response(a, M, prev_g=prev_g)
        prev_g = g
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g_norm = g / g_sum
        a_new = (1 - step) * a + step * g_norm
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        score = compute_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1
        a = a_new
        
        if stall_count > 300:
            step = max(0.05, step * 0.7)
            rng = np.random.default_rng(iters)
            a = best_a + rng.standard_normal(n) * 0.001
            a = np.maximum(a, 0)
            a = a / a.sum()
            prev_g = None
            stall_count = 0
        
        if verbose and iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, best={best_score:.8f}, step={step:.4f}, time={elapsed:.0f}s")
    
    return best_a, best_score


def mirror_descent(a0, time_limit, eta=0.1, verbose=True):
    """Mirror descent with entropy regularization (operates in log-space)."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-15)
    s = a.sum()
    a = a / s
    z = np.log(a)  # log-space
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        a_curr = np.exp(z - z.max())
        a_curr = a_curr / a_curr.sum()
        
        # Gradient of the objective (in a-space)
        conv = fast_convolve(a_curr)
        max_val = conv.max()
        # Soft subgradient
        T = max(0.01, 1.0 / (1.0 + 0.005 * it))
        shifted = conv - max_val
        weights = np.exp(shifted / T)
        weights /= weights.sum()
        
        # Grad in a-space
        a_flip = a_curr[::-1]
        fft_size = next_fast_len(2 * n - 1)
        W = rfft(weights, fft_size)
        F = rfft(a_flip, fft_size)
        conv_wa = irfft(W * F, fft_size)[:2*n-1]
        grad_a = 2.0 * n * conv_wa[n-1:]
        
        # Mirror descent step: z_new = z - eta * grad_a / a_curr
        # (since the mirror map is exp, the gradient step in a-space is grad/a)
        step_size = eta / (1.0 + 0.01 * it)
        z = z - step_size * grad_a / np.maximum(a_curr, 1e-30)
        
        # Renormalize (the mirror descent on the simplex needs a shift)
        a_new = np.exp(z - z.max())
        a_new = a_new / a_new.sum()
        
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        if verbose and (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"  MD iter={it+1}, score={best_score:.8f}, eta={step_size:.5f}, time={elapsed:.0f}s")
    
    return best_a, best_score


def coordinate_descent(a0, time_limit, n_pairs=200, verbose=True):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    iters = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            if a[i] < 1e-14:
                continue
            
            best_local = best_score
            best_t = 0.0
            base = a[i] * 0.8
            
            for frac in [0.02, 0.05, 0.1, 0.2, 0.4]:
                t = frac * base
                a_trial = a.copy()
                a_trial[i] -= t
                a_trial[j] += t
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score(a_trial)
                if sc < best_local - 1e-14:
                    best_local = sc
                    best_t = t
            
            if best_t != 0.0:
                a[i] -= best_t
                a[j] += best_t
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = best_local
                best_a = a.copy()
                improved = True
                no_improve = 0
            else:
                no_improve += 1
        
        if not improved:
            no_improve += n_pairs
        
        if no_improve > 500:
            noise = rng.standard_normal(n) * 0.003
            a = best_a.copy() + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            no_improve = 0
        
        if verbose and iters % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"  CD iter={iters}, best={best_score:.8f}, time={elapsed:.0f}s")
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Combines Nesterov-accelerated subgradient, LP fictitious play, mirror descent,
Bussgang spectral flattening, and coordinate descent with adaptive time allocation
and multiple restarts to escape local minima.

import numpy as np
import time
import cvxpy as cp
from scipy.linalg import toeplitz
from scipy.fft import rfft, irfft, next_fast_len
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = next_fast_len(size)
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    n = len(a)
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max())


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.max(np.nonzero(cond)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def soft_subgradient(a, T=1.0):
    n = len(a)
    conv = fast_convolve(a)
    max_val = conv.max()
    shifted = conv - max_val
    weights = np.exp(shifted / T)
    wsum = weights.sum()
    if wsum < 1e-300:
        weights = np.zeros_like(conv)
        weights[np.argmax(conv)] = 1.0
    else:
        weights /= wsum
    a_flip = a[::-1]
    fft_size = next_fast_len(2 * n - 1)
    W = rfft(weights, fft_size)
    F = rfft(a_flip, fft_size)
    conv_wa = irfft(W * F, fft_size)[:2 * n - 1]
    grad = 2.0 * conv_wa[n - 1:]
    return grad


def nesterov_subgradient(a0, time_limit, T_init=3.0, T_min=0.05, verbose=True):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    a_prev = a.copy()
    c_step = 0.3 / (n ** 0.25)

    for it in range(500000):
        if time.monotonic() - t0 > time_limit:
            break
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * (it + 1.0) ** 2)) / 2.0
        t_km1 = (1.0 + np.sqrt(1.0 + 4.0 * it ** 2)) / 2.0 if it > 0 else 1.0
        beta = (t_km1 - 1.0) / t_k if it > 0 else 0.0
        y = a + beta * (a - a_prev)
        a_prev = a.copy()
        y = np.maximum(y, 0)
        ys = y.sum()
        if ys < 1e-15:
            y = a.copy()
        else:
            y = y / ys
        T = max(T_min, T_init / (1.0 + 0.005 * (it + 1)))
        grad = soft_subgradient(y, T)
        eta = c_step / (np.sqrt(it + 1.0) * (1.0 + 0.001 * it))
        a_new = y - eta * grad
        a_new = project_simplex(a_new)
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
        if verbose and (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"  NS iter={it+1}, score={best_score:.8f}, T={T:.4f}, eta={eta:.6f}, time={elapsed:.0f}s")
    return best_a, best_score


def build_conv_matrix(f):
    n = len(f)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)


def solve_best_response(f, M, prev_g=None):
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    if prev_g is not None:
        g.value = prev_g
    C = build_conv_matrix(f)
    constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    solver_order = [cp.CLARABEL, cp.ECOS, cp.PDLP]
    solved = False
    for solver in solver_order:
        try:
            prob.solve(solver=solver, warm_start=(prev_g is not None))
            solved = True
            break
        except Exception:
            continue
    if not solved or g.value is None:
        return prev_g if prev_g is not None else f.copy()
    return np.maximum(g.value, 0)


def lp_fictitious_play(a0, time_limit, step=0.4, verbose=True, label="LP-FP"):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    prev_g = None
    stall_count = 0

    while time.monotonic() - t0 < time_limit and iters < 30000:
        iters += 1
        conv = fast_convolve(a)
        M = conv.max()
        g = solve_best_response(a, M, prev_g=prev_g)
        prev_g = g
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g_norm = g / g_sum
        a_new = (1 - step) * a + step * g_norm
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        score = compute_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1
        a = a_new
        if stall_count > 200:
            step = max(0.05, step * 0.6)
            rng = np.random.default_rng(iters * 7 + 3)
            a = best_a + rng.standard_normal(n) * 0.002
            a = np.maximum(a, 0)
            a = a / a.sum()
            prev_g = None
            stall_count = 0
        if verbose and iters % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, best={best_score:.8f}, step={step:.4f}, time={elapsed:.0f}s")
    return best_a, best_score


def mirror_descent(a0, time_limit, eta_init=0.5, verbose=True):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-15)
    s = a.sum()
    a = a / s
    z = np.log(a)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    for it in range(500000):
        if time.monotonic() - t0 > time_limit:
            break
        z_shift = z - z.max()
        a_curr = np.exp(z_shift)
        a_curr = a_curr / a_curr.sum()
        conv = fast_convolve(a_curr)
        max_val = conv.max()
        T = max(0.05, 1.0 / (1.0 + 0.003 * it))
        shifted = conv - max_val
        weights = np.exp(shifted / T)
        weights /= weights.sum()
        a_flip = a_curr[::-1]
        fft_size = next_fast_len(2 * n - 1)
        W = rfft(weights, fft_size)
        F = rfft(a_flip, fft_size)
        conv_wa = irfft(W * F, fft_size)[:2 * n - 1]
        grad_a = 2.0 * n * conv_wa[n - 1:]
        step_size = eta_init / (1.0 + 0.02 * it)
        z = z - step_size * grad_a / np.maximum(a_curr, 1e-20)
        a_new = np.exp(z - z.max())
        a_new = a_new / a_new.sum()
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        if verbose and (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"  MD iter={it+1}, score={best_score:.8f}, step={step_size:.5f}, time={elapsed:.0f}s")
    return best_a, best_score


def bussgang_iteration(a0, time_limit, tau=0.5, verbose=True):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    fft_size = next_fast_len(2 * n - 1)

    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        A = rfft(a, fft_size)
        power = np.abs(A) ** 2
        total_power = power.sum()
        target = total_power / fft_size
        ratio = np.where(power > 1e-30, target / power, 1.0)
        ratio = np.clip(ratio, 0.05, 20.0)
        A_new = A * (ratio ** tau)
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
        if verbose and (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"  BG iter={it+1}, score={best_score:.8f}, time={elapsed:.0f}s")
    return best_a, best_score


def coordinate_descent(a0, time_limit, n_pairs=200, verbose=True):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    iters = 0
    no_improve_count = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved_this_round = False
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j or a[i] < 1e-14:
                continue
            best_local = best_score
            best_t = 0.0
            base = a[i] * 0.9
            for frac in [0.02, 0.05, 0.1, 0.2, 0.35, 0.6]:
                t = frac * base
                a_trial = a.copy()
                a_trial[i] = a[i] - t
                a_trial[j] = a[j] + t
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score(a_trial)
                if sc < best_local - 1e-14:
                    best_local = sc
                    best_t = t
            if best_t != 0.0:
                a[i] -= best_t
                a[j] += best_t
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = best_local
                best_a = a.copy()
                improved_this_round = True
                no_improve_count = 0
            else:
                no_improve_count += 1
        if not improved_this_round:
            no_improve_count += n_pairs
        if no_improve_count > 400:
            noise = rng.standard_normal(n) * 0.004
            a = best_a.copy() + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            no_improve_count = 0
        if verbose and iters % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"  CD iter={iters}, best={best_score:.8f}, time={elapsed:.0f}s")
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def lbgfs_smoothed(a0, time_limit, T=500, maxiter=3000):
    n = len(a0)
    a0 = np.maximum(a0, 1e-15)
    s = a0.sum()
    a0 = a0 / s
    z0 = np.log(a0)
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)

    def objective_func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        z_s = z - z.max()
        exp_z = np.exp(z_s)
        a = exp_z / exp_z.sum()
        conv = fast_convolve(a)
        shifted = conv / T
        max_sh = shifted.max()
        lse = max_sh + np.log(np.exp(shifted - max_sh).sum())
        return (2.0 * n / T) * np.exp(T * lse)

    def objective_grad(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        z_s = z - z.max()
        exp_z = np.exp(z_s)
        a = exp_z / exp_z.sum()
        conv = fast_convolve(a)
        shifted = conv / T
        max_sh = shifted.max()
        w = np.exp(shifted - max_sh)
        w = w / w.sum()
        fft_size = next_fast_len(2 * n - 1)
        Wf = rfft(w, fft_size)
        Af = rfft(a, fft_size)
        grad_a = 2.0 * irfft(np.conj(Wf) * Af, fft_size)[:n]
        S = float(np.dot(grad_a, a))
        grad_z = a * (grad_a - S)
        return (2.0 * n / T) * grad_z

    try:
        res = minimize(
            fun=objective_func, x0=z0, jac=objective_grad,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-15,
                     'maxfun': maxiter * 30, 'disp': False}
        )
        z_s = res.x - res.x.max()
        a_out = np.exp(z_s)
        a_out = a_out / a_out.sum()
        score = compute_score(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except Exception:
        pass
    return best_a, best_score

        rem = deadline - time.monotonic()
        t5 = min(rem * 0.10, 70.0)
        if rem < 40:
            break
        print(f"  L-BFGS T={T_val}, budget={t5:.0f}s")
        a_lb, sc_lb = lbgfs_smoothed(best_seq, t5, T=T_val, maxiter=2000)
        best_seq, best_score = check_and_update(a_lb, f"lbgfs_T{T_val}", best_seq, best_score)

    # Phase 6: LP Fictitious Play (moderate)
    print(f"\n=== Phase 6: LP-FP Moderate (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.18, 200.0)
    if t6 > 60:
        print(f"  Budget: {t6:.0f}s")
        a_lp2, sc_lp2 = lp_fictitious_play(best_seq, t6, step=0.25, label="LP-mod")
        best_seq, best_score = check_and_update(a_lp2, "lp_mod", best_seq, best_score)

    # Phase 7: Coordinate Descent
    print(f"\n=== Phase 7: Coordinate Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t7 = min(rem * 0.12, 100.0)
    if t7 > 30:
        print(f"  Budget: {t7:.0f}s")
        a_cd, sc_cd = coordinate_descent(best_seq, t7, n_pairs=150, verbose=True)
        best_seq, best_score = check_and_update(a_cd, "coord_desc", best_seq, best_score)

    # Phase 8: Try nearby lengths
    print(f"\n=== Phase 8: Nearby Lengths (best={best_score:.8f}) ===")
    alt_lengths = [n_parent + 2, n_parent + 4, n_parent - 2, n_parent + 6, n_parent - 4]
    for alt_n in alt_lengths:
        if alt_n < 64 or alt_n > 4096:
            continue
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        t8 = min(rem * 0.25, 80.0)
        init_alt = interpolate_sequence(best_seq, alt_n)
        print(f"  Trying n={alt_n}, budget={t8:.0f}s")
        a_alt, sc_alt = lp_fictitious_play(init_alt, t8, step=0.3, label=f"n={alt_n}")
        best_seq, best_score = check_and_update(a_alt, f"n{alt_n}", best_seq, best_score)

    # Phase 9: Final Nesterov on best
    print(f"\n=== Phase 9: Final Nesterov (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t9 = min(rem * 0.30, 100.0)
        print(f"  Budget: {t9:.0f}s")
        a_fin, sc_fin = nesterov_subgradient(best_seq, t9, T_init=1.0, T_min=0.02, verbose=True)
        best_seq, best_score = check_and_update(a_fin, "final_ns", best_seq, best_score)

    # Phase 10: Final LP polish
    print(f"\n=== Phase 10: Final LP Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t10 = min(rem * 0.50, 100.0)
        a_end, sc_end = lp_fictitious_play(best_seq, t10, step=0.1, label="LP-end")
        best_seq, best_score = check_and_update(a_end, "lp_end", best_seq, best_score)

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs) and fs < best_score:
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
