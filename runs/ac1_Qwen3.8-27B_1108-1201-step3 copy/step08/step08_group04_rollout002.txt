def build_constraint_matrix(f, indices, n):

        k = int(indices[i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    return A


def solve_lp_direction(f, indices, M, solver=None):
    """Solve LP: max sum(g) s.t. g >= 0, A@g <= M."""
    n = len(f)
    K = len(indices)
    A = build_constraint_matrix(f, indices, n)
    
    g = cp.Variable(n)
    objective = cp.Maximize(cp.sum(g))
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(objective, constraints)
    
    try:
        if solver is None:
            if K <= 500 and n <= 1200:
                prob.solve(solver=cp.ECOS, max_iters=30000)
            else:
                prob.solve(solver=cp.SCS, max_iters=200000, eps_abs=1e-8, eps_rel=1e-8)
        else:
            prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=100000, eps_abs=1e-7, eps_rel=1e-7)

            return None
    
    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):

            continue
        
        S_g = g.sum()
        if S_g <= 1.0 + 1e-5:
            stuck += 1
            if stuck > max_stuck // 2:
                break
            continue
        
        h = g / S_g
        momentum_dir = momentum_beta * momentum_dir + (1.0 - momentum_beta) * h
        mom_sum = momentum_dir.sum()
        if mom_sum > 1e-10:
            h_mom = momentum_dir / mom_sum
        else:
            h_mom = h
        
        # Line search
        best_t = 0.0
        best_t_obj = M
        for direction in [h, h_mom]:
            for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
                f_trial = (1.0 - t) * f + t * direction
                f_trial = np.maximum(f_trial, 0.0)
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial = f_trial / ss
                obj = fast_convolve(f_trial).max()
                if obj < best_t_obj:
                    best_t_obj = obj
                    best_t = t
        
        if best_t == 0.0:
            stuck += 1
            if stuck > max_stuck:
                break
            continue
        
        f = (1.0 - best_t) * f + best_t * (h if h_mom is h else h_mom)
        f = np.maximum(f, 0.0)
        f = f / f.sum()
        cur_obj = fast_convolve(f).max()
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_score = 2.0 * n * best_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > max_stuck:
                break
        
        if stuck > 30 and it % 10 == 0:
            rng = np.random.default_rng(it * 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            stuck = 0
            momentum_dir = np.zeros(n)
        
        if it % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} it={it} obj={best_obj:.10f} score={best_score:.6f} "
                  f"k={len(top_idx)} t={elapsed:.0f}s")
    
    return best_f, best_score


def coordinate_descent(f_init, time_limit):
    """Coordinate descent with multiple step sizes."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    best_f = f.copy()
    
    multipliers = [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0]
    no_pass = 0
    
    while time.monotonic() - t0 < time_limit:
        any_improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-16:
                continue
            best_j_obj = best_obj
            best_j_val = f[j]
            for mult in multipliers:
                new_val = f[j] * mult
                if new_val < 0:
                    continue
                f[j] = new_val
                s = f.sum()
                if s < 1e-12:
                    f[j] = best_j_val
                    continue
                f = f / s
                obj = fast_convolve(f).max()
                f[j] = f[j] * s  # undo normalization for next trial
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = new_val / s
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-12:
                    f = f / s
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_score = 2.0 * n * best_obj
                    best_f = f.copy()
                    any_improved = True
        
        if not any_improved:
            no_pass += 1
            if no_pass >= 2:
                break
        else:
            no_pass = 0
    
    return best_f, best_score


def interp_sequence(src, target_n):
    """Interpolate a sequence to a new length."""
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    Phase 1: Softmax subgradient on parent + nearby lengths
    Phase 2: LP-based direction finding
    Phase 3: Coordinate descent + final LP polish
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
    except Exception:
        pass
    
    if parent is not None:
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    else:
        parent = np.ones(512) / 512.0
        p_score = 2.0
        print(f"No parent, using uniform n=512, score=2.0")
    
    best_seq = parent.copy()
    best_score = p_score
    
    # ============================================================
    # Phase 1: Softmax subgradient (50% of time)
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 1: Softmax subgradient (best: {best_score:.6f})")
    print(f"{'='*60}")
    
    rem = deadline - time.monotonic()
    t1 = rem * 0.45
    
    # Try parent length first with the most time
    n_parent = len(parent)
    alloc_parent = t1 * 0.5
    
    print(f"  SG n={n_parent}, budget={alloc_parent:.0f}s")
    f_res, _ = run_softmax_subgradient(
        parent, time_limit=alloc_parent,
        beta_init=50.0, beta_max=3000.0,
        lr_init=0.004, momentum_init=0.6,
        label="sg_parent", top_k_frac=0.08, seed=42)
    
    real_score = float(evaluate_sequence(f_res.tolist()))
    if np.isfinite(real_score) and real_score < best_score:
        best_score = real_score
        best_seq = f_res.copy()
        print(f"  NEW BEST [sg_parent]: {real_score:.6f}")
    
    # Try a few other lengths
    rem = deadline - time.monotonic()
    t1_rem = rem * 0.35
    
    n_trials = []
    for factor in [0.7, 0.85, 1.15, 1.3, 1.5]:
        nt = int(n_parent * factor)
        nt = max(200, min(3000, nt))
        if nt not in n_trials and abs(nt - n_parent) > n_parent * 0.1:
            n_trials.append(nt)
    n_trials.sort()
    
    if n_trials and t1_rem > 30:
        per_trial = t1_rem / len(n_trials)
        for nt in n_trials:
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            alloc = min(per_trial, rem * 0.8)
            if alloc < 15:
                break
            f_init = interp_sequence(best_seq, nt)
            print(f"  SG n={nt}, budget={alloc:.0f}s")
            f_res, _ = run_softmax_subgradient(
                f_init, time_limit=alloc,
                beta_init=30.0, beta_max=2000.0,
                lr_init=0.003, momentum_init=0.55,
                label=f"sg{n}", top_k_frac=0.1, seed=nt)
            
            real_score = float(evaluate_sequence(f_res.tolist()))
            if np.isfinite(real_score) and real_score < best_score:
                best_score = real_score
                best_seq = f_res.copy()
                print(f"  NEW BEST [sg n={nt}]: {real_score:.6f}")
    
    # ============================================================
    # Phase 2: LP-based direction finding (30% of time)
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 2: LP optimization (best: {best_score:.6f})")
    print(f"{'='*60}")
    
    rem = deadline - time.monotonic()
    t2 = rem * 0.30
    
    n_b = len(best_seq)

    if np.isfinite(real_score) and real_score < best_score:
        best_score = real_score
        best_seq = f_res.copy()
        print(f"  NEW BEST [lp_main]: {real_score:.6f}")
    
    # ============================================================
    # Phase 3: Coordinate descent + final polish (20% of time)
