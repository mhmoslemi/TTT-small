def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_half(a):
    """
    Compute only the first n values of conv(a,a) using symmetry.
    For symmetric a, conv(a,a) is symmetric around n-1.
    Returns the full convolution.

    return fast_convolve(a)


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def symmetrize(a):

    result[:n//2] = half[:n//2]
    if n % 2 == 1:

        result[n//2+1:] = half[n//2-1::-1]

def nesterov_lp(a0, time_limit, solver=None, symmetrize_flag=True, n_max_iters=50000):
    """Nesterov dual averaging LP refinement with optional symmetry."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    if symmetrize_flag:
        a = symmetrize(a)
        a = normalize(a)
    y = a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        if symmetrize_flag:
            y = symmetrize(y)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        # Nesterov accelerated point
        a_new = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a_new = np.maximum(a_new, 0)
        if symmetrize_flag:
            a_new = symmetrize(a_new)
        sa = a_new.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a_new / sa

        score_a = compute_score(a)
        score_y = compute_score(y)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def gradient_lp(a0, time_limit, solver=None, symmetrize_flag=True, n_steps=20000):
    """
    Gradient-based optimization: at each step, identify the top peaks of conv(a,a),
    compute the subgradient, and take a step in log-space using mirror descent.
    More aggressive peak targeting than simple mirror descent.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    if symmetrize_flag:
        a = symmetrize(a)
        a = normalize(a)
    
    best_a = a.copy()
    best_score = compute_score(a)
    z = np.log(np.maximum(a, 1e-30))
    t0 = time.monotonic()
    lr = 0.1
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        
        # Identify top peaks (top 5% or at least 3 peaks)
        n_peaks = max(3, len(b) // 20)
        peak_threshold = max_b * (1 - 0.02)
        active_peaks = np.where(b >= peak_threshold)[0]
        if len(active_peaks) < 3:
            # Use top k peaks
            top_k = np.argsort(b)[::-1][:n_peaks]
            active_peaks = top_k
        active_peaks = active_peaks[:n_peaks]
        
        # Compute subgradient
        g = np.zeros(n)
        for k in active_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active_peaks)
        
        # Mirror descent step in log-space
        z_new = z - lr * g
        # Symmetry in log-space
        if symmetrize_flag:
            z_new = symmetrize(z_new)
        
        a_new = np.exp(z_new - z_new.max())
        a_new = a_new / a_new.sum()
        
        score_new = compute_score(a_new)
        
        if score_new < best_score - 1e-15:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            z = z_new
        else:
            # Reduce learning rate
            lr *= 0.95
            z = np.log(np.maximum(a, 1e-30))
        
        # Adaptive learning rate
        lr = max(lr * 0.999, 0.001)
        
        if step % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GradLP {elapsed:.0f}s: score={best_score:.6f} lr={lr:.6f}")

    return best_a, best_score


def adam_optimize(a0, time_limit, symmetrize_flag=True, n_steps=50000):
    """
    Adam optimizer in log-space targeting the top peaks of convolution.
    Uses adaptive learning rates per coordinate.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    if symmetrize_flag:
        a = symmetrize(a)
        a = normalize(a)
    
    best_a = a.copy()
    best_score = compute_score(a)
    z = np.log(np.maximum(a, 1e-30))
    
    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    lr = 0.05
    t0 = time.monotonic()
    t = 0
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        t += 1

        b = fast_convolve(a)
        max_b = b.max()
        
        # Identify active peaks
        n_peaks = max(5, len(b) // 15)
        peak_threshold = max_b - 0.01 * max_b
        active_peaks = np.where(b >= peak_threshold)[0]
        if len(active_peaks) > n_peaks:
            # Keep the top n_peaks
            peak_vals = b[active_peaks]
            top_idx = np.argsort(peak_vals)[::-1][:n_peaks]
            active_peaks = active_peaks[top_idx]
        
        # Compute gradient
        g = np.zeros(n)
        for k in active_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active_peaks)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        z_new = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        if symmetrize_flag:
            z_new = symmetrize(z_new)
        
        a_new = np.exp(z_new - z_new.max())
        a_new = a_new / a_new.sum()
        
        score_new = compute_score(a_new)
        
        if score_new < best_score - 1e-15:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            z = z_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                lr *= 0.9
                no_improve = 0
        
        # Decay learning rate
        lr = max(0.05 / (1.0 + 0.001 * step), 0.001)
        
        if step % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam {elapsed:.0f}s: score={best_score:.6f} lr={lr:.6f}")

    return best_a, best_score


def peak_shave(a0, time_limit, symmetrize_flag=True, n_steps=30000):
    """
    Aggressive peak shaving: identify top contributors to peaks and reduce them,
    redistributing mass to non-contributing positions.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    if symmetrize_flag:
        a = symmetrize(a)
        a = normalize(a)
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        
        # Find top peaks (excluding edges)
        b_work = b.copy()
        b_work[:2] = -1
        b_work[-2:] = -1
        peak_k = int(np.argmax(b_work))
        peak_val = b_work[peak_k]
        
        # Find near-peaks
        threshold = peak_val - 0.02 * peak_val
        near_peaks = np.where(b_work >= threshold)[0][:8]

        # Find contributors to each near-peak
        all_contributors = np.zeros(n)
        for k in near_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[k - idx]
            all_contributors[idx] += contribs
        
        # Top contributors
        top_n = max(5, n // 50)
        top_coords = np.argsort(all_contributors)[::-1][:top_n]
        
        improved = False
        best_trial = None
        best_trial_score = best_score
        
        for i in top_coords:
            if a[i] < 1e-14:
                continue
            for factor in [0.98, 0.96, 0.94, 0.92]:
                new_val = a[i] * factor
                diff = a[i] - new_val
                if diff < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                if symmetrize_flag:
                    a_trial[n-1-i] = new_val
                    diff_total = diff * (1 if i == n-1-i else 2)
                else:
                    diff_total = diff
                
                # Redistribute to positions with low contribution
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                if symmetrize_flag:
                    mask[n-1-i] = False
                
                others = a_trial[mask]
                if others.sum() < 1e-15:
                    continue
                a_trial[mask] = others + diff_total * others / others.sum()
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                
                score = compute_score(a_trial)
                if score < best_trial_score:
                    best_trial_score = score
                    best_trial = a_trial.copy()
                
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved and best_trial is not None:
            # Even if not improving global best, accept if it improves peak
            a = best_trial
            if compute_score(a) < best_score + 0.001:
                best_a = a.copy()
        elif not improved:
            # Try pair reduction
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_contrib = np.argsort(contribs)[::-1][:3]
            
            pair_improved = False
            for ci in top_contrib:
                i = idx[ci]
                j = peak_k - i
                if i == j or i >= n or j >= n:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.97, 0.95]:
                    for f2 in [0.97, 0.95]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        if symmetrize_flag:
                            a_trial[n-1-i] = a[i] * f1
                            a_trial[n-1-j] = a[j] * f2
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        score = compute_score(a_trial)
                        if score < best_score - 1e-15:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            pair_improved = True
                            break
                    if pair_improved:
                        break
                if pair_improved:
                    break
            
            if not pair_improved:
                break

    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization and symmetry."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    f_interp = symmetrize(f_interp)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def symmetrize(a):
    """Enforce symmetry: a[j] = a[n-1-j]."""
    return (a + a[::-1]) / 2.0


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def compute_score(a):
    """Compute objective assuming sum=1."""

def nesterov_lp(a0, time_limit, solver=None, use_sym=True, n_max_iters=50000):

    if use_sym:

        a = normalize(a)

            y = symmetrize(y)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

def adam_optimize(a0, time_limit, use_sym=True, n_steps=50000):
    """Adam optimizer in log-space targeting top peaks of convolution."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    if use_sym:
        a = symmetrize(a)
        a = normalize(a)

    best_a = a.copy()
    best_score = compute_score(a)
    z = np.log(np.maximum(a, 1e-30))

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    lr = 0.05
    t0 = time.monotonic()
    t = 0
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        t += 1

        b = fast_convolve(a)
        max_b = b.max()

        n_peaks = max(5, len(b) // 15)
        peak_threshold = max_b - 0.01 * max_b
        active_peaks = np.where(b >= peak_threshold)[0]
        if len(active_peaks) > n_peaks:
            peak_vals = b[active_peaks]
            top_idx = np.argsort(peak_vals)[::-1][:n_peaks]
            active_peaks = active_peaks[top_idx]
        if len(active_peaks) == 0:
            active_peaks = np.array([int(np.argmax(b))])

        g = np.zeros(n)
        for k in active_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active_peaks)

        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        z_new = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        if use_sym:
            z_new = symmetrize(z_new)

        a_new = np.exp(z_new - z_new.max())
        a_new = a_new / a_new.sum()

        score_new = compute_score(a_new)

        if score_new < best_score - 1e-15:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            z = z_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                lr *= 0.9
                no_improve = 0

        lr = max(0.05 / (1.0 + 0.0005 * step), 0.002)

        if step % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam {elapsed:.0f}s: score={best_score:.6f} lr={lr:.6f}")

    return best_a, best_score


def _lbgfs_fun(z, T, n, t0, time_limit):
    """L-BFGS objective (top-level)."""
    if time.monotonic() - t0 > time_limit:
        return 1e15
    z_shifted = z - z.max()
    a_in = np.exp(z_shifted)
    a_in = a_in / a_in.sum()
    b = fast_convolve(a_in)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    return (2.0 * n / T) * log_sum_exp


def _lbgfs_jac(z, T, n, t0, time_limit, use_sym):
    """L-BFGS Jacobian (top-level)."""
    if time.monotonic() - t0 > time_limit:
        return np.zeros_like(z)
    z_shifted = z - z.max()
    a_in = np.exp(z_shifted)
    a_in = a_in / a_in.sum()
    b = fast_convolve(a_in)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a_in[k - idx]

    S = np.dot(grad_a, a_in)
    grad_z = a_in * (grad_a - S)
    if use_sym:
        grad_z = (grad_z + grad_z[::-1]) / 2.0
    return grad_z


def lbgfs_logsumexp(a0, time_limit, T_start=100.0, T_end=0.5, maxiter=8000, use_sym=True):
    """L-BFGS with LogSumExp smoothing and temperature annealing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    if use_sym:
        a = symmetrize(a)
        a = normalize(a)
    z = np.log(np.maximum(a, 1e-30))
    t0 = time.monotonic()

    n_stages = 8
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(30, maxiter // n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break
        try:
            res = minimize(
                fun=_lbgfs_fun,
                x0=z,
                args=(stage_T, n, t0, time_limit),
                jac=_lbgfs_jac,
                x0_pass=True,
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 50, 'disp': False}
            ) if False else minimize(
                fun=lambda zz: _lbgfs_fun(zz, stage_T, n, t0, time_limit),
                x0=z,
                jac=lambda zz: _lbgfs_jac(zz, stage_T, n, t0, time_limit, use_sym),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 50, 'disp': False}
            )
            z = res.x
            if use_sym:
                z = symmetrize(z)
        except Exception:
            pass

    a_result = np.exp(z - z.max())
    a_result = a_result / a_result.sum()
    return a_result


def peak_shave(a0, time_limit, use_sym=True, n_steps=30000):
    """
    Aggressive peak shaving: identify top contributors to peaks and reduce them,
    redistributing mass to non-contributing positions.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    if use_sym:
        a = symmetrize(a)
        a = normalize(a)

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        b_work = b.copy()
        b_work[:2] = -1
        b_work[-2:] = -1
        peak_k = int(np.argmax(b_work))
        peak_val = b_work[peak_k]

        threshold = peak_val - 0.02 * peak_val
        near_peaks = np.where(b_work >= threshold)[0][:8]

        all_contributors = np.zeros(n)
        for k in near_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[k - idx]
            all_contributors[idx] += contribs

        top_n = max(5, n // 50)
        top_coords = np.argsort(all_contributors)[::-1][:top_n]

        improved = False
        for i in top_coords:
            if a[i] < 1e-14:
                continue
            for factor in [0.98, 0.96, 0.94, 0.92]:
                new_val = a[i] * factor
                diff = a[i] - new_val
                if diff < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                if use_sym:
                    a_trial[n - 1 - i] = new_val
                    diff_total = diff * (1 if i == n - 1 - i else 2)
                else:
                    diff_total = diff

                mask = np.ones(n, dtype=bool)
                mask[i] = False
                if use_sym:
                    mask[n - 1 - i] = False

                others = a_trial[mask]
                if others.sum() < 1e-15:
                    continue
                a_trial[mask] = others + diff_total * others / others.sum()
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()

                score = compute_score(a_trial)
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_contrib = np.argsort(contribs)[::-1][:3]

            pair_improved = False
            for ci in top_contrib:
                i = idx[ci]
                j = peak_k - i
                if i == j or i >= n or j >= n:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.97, 0.95]:
                    for f2 in [0.97, 0.95]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        if use_sym:
                            a_trial[n - 1 - i] = a[i] * f1
                            a_trial[n - 1 - j] = a[j] * f2
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        score = compute_score(a_trial)
                        if score < best_score - 1e-15:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            pair_improved = True
                            break
                    if pair_improved:
                        break
                if pair_improved:
                    break

            if not pair_improved:
                # Try triple reduction
                triple_improved = False
                for ci1 in top_contrib[:2]:
                    for ci2 in top_contrib[1:3]:
                        if ci1 == ci2:
                            continue
                        i1 = idx[ci1]
                        j1 = peak_k - i1
                        i2 = idx[ci2]
                        if i1 >= n or j1 >= n or i2 >= n:
                            continue
                        a_trial = a.copy()
                        a_trial[i1] *= 0.97
                        a_trial[j1] *= 0.97
                        a_trial[i2] *= 0.97
                        if use_sym:
                            for pos in [i1, j1, i2]:
                                a_trial[n - 1 - pos] = a[pos] * 0.97
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        score = compute_score(a_trial)
                        if score < best_score - 1e-15:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            triple_improved = True
                            pair_improved = True
                            break
                    if pair_improved:
                        break

                if not triple_improved:
                    break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization and symmetry."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    f_interp = symmetrize(f_interp)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization with symmetry enforcement:
    1. Nesterov LP alternating (long)
    2. Adam gradient descent on active peaks
    3. L-BFGS LogSumExp refinement
    4. Nesterov LP round 2
    5. Peak shaving
    6. Multi-length exploration
    7. Perturbation escape
    8. Final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = normalize(p)
            parent = symmetrize(parent)
            parent = normalize(parent)
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 960
        best_seq = structured_init(n_init, rng, seed_type=0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n_best = len(best_seq)

    # Helper to try updating best (inlined to avoid closure)
    # We'll just do the comparison inline each time

    # === Phase 1: Nesterov LP (long run) ===
    print(f"\n=== Phase 1: Nesterov LP n={n_best} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 500.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp(best_seq, t_alloc, solver=solver, use_sym=True, n_max_iters=100000)
        a_lp = symmetrize(np.maximum(a_lp, 0))
        a_lp = normalize(a_lp)
        sc = evaluate_safe(a_lp)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_lp.copy()
            print(f"  NEW BEST [nlp_1]: {sc:.6f} (n={len(a_lp)})")

    # === Phase 2: Adam optimization ===
    print(f"\n=== Phase 2: Adam (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_adam, _ = adam_optimize(best_seq, t_alloc, use_sym=True, n_steps=80000)
        a_adam = symmetrize(np.maximum(a_adam, 0))
        a_adam = normalize(a_adam)
        sc = evaluate_safe(a_adam)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_adam.copy()
            print(f"  NEW BEST [adam]: {sc:.6f} (n={len(a_adam)})")

    # === Phase 3: L-BFGS ===
    print(f"\n=== Phase 3: L-BFGS (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.18, 280.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lb = lbgfs_logsumexp(best_seq, t_alloc, T_start=150.0, T_end=0.3, maxiter=10000, use_sym=True)
        a_lb = symmetrize(np.maximum(a_lb, 0))
        a_lb = normalize(a_lb)
        sc = evaluate_safe(a_lb)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_lb.copy()
            print(f"  NEW BEST [lbgfs]: {sc:.6f} (n={len(a_lb)})")

    # === Phase 4: Nesterov LP round 2 ===
    print(f"\n=== Phase 4: NLP r2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.14, 200.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, _ = nesterov_lp(best_seq, t_alloc, solver=solver, use_sym=True, n_max_iters=50000)
        a_lp2 = symmetrize(np.maximum(a_lp2, 0))
        a_lp2 = normalize(a_lp2)
        sc = evaluate_safe(a_lp2)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_lp2.copy()
            print(f"  NEW BEST [nlp_2]: {sc:.6f} (n={len(a_lp2)})")

    # === Phase 5: Peak shaving ===
    print(f"\n=== Phase 5: Peak Shaving (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 100.0)
    if t_alloc > 10:
        a_ps, _ = peak_shave(best_seq, t_alloc, use_sym=True, n_steps=30000)
        a_ps = symmetrize(np.maximum(a_ps, 0))
        a_ps = normalize(a_ps)
        sc = evaluate_safe(a_ps)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_ps.copy()
            print(f"  NEW BEST [peak_shave]: {sc:.6f} (n={len(a_ps)})")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-length (best={best_score:.6f}) ===")
    alt_lengths = []
    for mult in [1.25, 1.5, 2.0, 0.75]:
        alt_n = int(n_best * mult)
        if 512 <= alt_n <= 3000 and alt_n != n_best:
            alt_lengths.append(alt_n)
    alt_lengths = sorted(set(alt_lengths))

    for alt_n in alt_lengths[:2]:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_total = min(rem * 0.22, 160.0)
        if t_total < 30:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_seq, alt_n)

        t_lp = t_total * 0.7
        a_alt, _ = nesterov_lp(init_alt, t_lp, solver=solver, use_sym=True, n_max_iters=30000)
        a_alt = symmetrize(np.maximum(a_alt, 0))
        a_alt = normalize(a_alt)
        sc = evaluate_safe(a_alt)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_alt.copy()
            print(f"  NEW BEST [nlp_n{alt_n}]: {sc:.6f} (n={alt_n})")

        rem = deadline - time.monotonic()
        if rem > 20:
            t_lb = min(rem * 0.25, 40.0)
            if t_lb > 10:
                a_alt2 = lbgfs_logsumexp(best_seq, t_lb, T_start=30.0, T_end=1.0, maxiter=3000, use_sym=True)
                a_alt2 = symmetrize(np.maximum(a_alt2, 0))
                a_alt2 = normalize(a_alt2)
                sc2 = evaluate_safe(a_alt2)
                if np.isfinite(sc2) and sc2 < best_score:
                    best_score = sc2
                    best_seq = a_alt2.copy()
                    print(f"  NEW BEST [lbgfs_n{alt_n}]: {sc2:.6f} (n={len(a_alt2)})")

    # === Phase 7: Perturbation escape ===
    print(f"\n=== Phase 7: Perturbation (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        n_cur = len(best_seq)
        for noise_scale in [0.03, 0.08]:
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            perturbed = best_seq + noise_scale * rng.standard_normal(n_cur)
            perturbed = np.maximum(perturbed, 0)
            perturbed = symmetrize(perturbed)
            perturbed = normalize(perturbed)

            t_lp_p = min(rem * 0.35, 45.0)
            if t_lp_p < 10:
                break
            a_pert, _ = nesterov_lp(perturbed, t_lp_p, solver=solver, use_sym=True, n_max_iters=15000)
            a_pert = symmetrize(np.maximum(a_pert, 0))
            a_pert = normalize(a_pert)
            sc = evaluate_safe(a_pert)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_pert.copy()
                print(f"  NEW BEST [pert_{noise_scale}]: {sc:.6f} (n={n_cur})")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.25, 50.0)
        a_fin = lbgfs_logsumexp(best_seq, t_fin, T_start=5.0, T_end=0.1, maxiter=3000, use_sym=True)
        a_fin = symmetrize(np.maximum(a_fin, 0))
        a_fin = normalize(a_fin)
        sc = evaluate_safe(a_fin)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_fin.copy()
            print(f"  NEW BEST [final_lbgfs]: {sc:.6f}")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_ps = min(rem * 0.20, 25.0)
        a_ps2, _ = peak_shave(best_seq, t_ps, use_sym=True, n_steps=10000)
        a_ps2 = symmetrize(np.maximum(a_ps2, 0))
        a_ps2 = normalize(a_ps2)
        sc = evaluate_safe(a_ps2)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_ps2.copy()
            print(f"  NEW BEST [final_ps]: {sc:.6f}")

    rem = deadline - time.monotonic()
    if rem > 10:
        t_lp_fin = min(rem * 0.35, 20.0)
        a_lp_fin, _ = nesterov_lp(best_seq, t_lp_fin, solver=solver, use_sym=True, n_max_iters=5000)
        a_lp_fin = symmetrize(np.maximum(a_lp_fin, 0))
        a_lp_fin = normalize(a_lp_fin)
        sc = evaluate_safe(a_lp_fin)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_lp_fin.copy()
            print(f"  NEW BEST [final_lp]: {sc:.6f}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
