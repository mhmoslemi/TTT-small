    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)

def fast_convolve(a):
    """Full auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def get_weights(b, T, n_top=None):
    """Compute softmax weights on b, optionally restricted to top-n_top peaks."""
    M = b.max()
    shifted = (b - M) / T

    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros_like(b)
        w[np.argmax(b)] = 1.0
        return w
    w /= ws

        top_idx = np.argpartition(b, -n_top)[-n_top:]
        w_sel = w[top_idx].copy()
        w_new = np.zeros_like(b)
        w_new[top_idx] = w_sel
        wn = w_new.sum()
        if wn > 1e-300:
            w_new /= wn
        w = w_new
    return w
