```python
"""
Smooth gradient descent with log-sum-exp approximation, full LP polish,
and multi-length exploration for minimizing auto-convolution peak.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_grad_from_w(w, a):
    """
    Compute gradient of soft-max objective.
    w: soft-max weights (length 2n-1)
    a: sequence (length n)
    Returns: gradient vector (length n)
    grad_i = sum_{j=0}^{n-1} w[i+j] * a[j]
    """
    n = len(a)
    a_flip = a[::-1].copy()
    size = n + len(w) - 1  # = n + (2n-1) - 1 = 3n-2
    fft_size = 1 << (size - 1).bit_length()
    AF = rfft(a_flip, fft_size)
    W = rfft(w, fft_size)
    conv = irfft(AF * W, fft_size)[:size]
    return conv[n-1:2*n-1]

def smooth_descent(a_init, time_limit, label="smooth", 
                   T_ratio=5.0, T_min_ratio=100.0, gamma=0.9995,
                   lr_init=0.05, momentum=0.9, batch=100,
                   restart_freq=50, noise_scale=0.003):
    """
    Smooth gradient descent with log-sum-exp soft-max approximation.
    Minimizes the peak of auto-convolution.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return a, fast_convolve(a).max()
    a /= s
    
    b0 = fast_convolve(a)
    T = b0.max() / T_ratio
    T_min = b0.max() / T_min_ratio
    
    v = np.zeros(n)
    best_obj = b0.max()
    best_a = a.copy()
    no_improve = 0
    it = 0
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        # Batch of gradient steps
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit:
                break
            it += 1
            
            b = fast_convolve(a)
            b_max = b.max()
            
            # Soft-max weights
            b_norm = (b - b_max) / T
            b_norm = np.clip(b_norm, -100.0, 50.0)
            exp_b = np.exp(b_norm)
            w = exp_b / exp_b.sum()
            
            # Gradient
            grad = compute_grad_from_w(w, a)
            
            # Adaptive learning rate
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            lr = lr_init * b_max / (n * grad_norm)
            lr = min(lr, 0.1)
            
            # Momentum update
            v = momentum * v - lr * grad
            a_new = a + v
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:
                a_new = a
            else:
                a_new /= s_new
            a = a_new
            
            # Anneal temperature
            T = max(T * gamma, T_min)
        
        # Evaluate
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Restart if stuck
        if no_improve >= restart_freq:
            a = best_a + rng.standard_normal(n) * noise_scale
            a = np.maximum(a, 0.0)
            a /= a.sum()
            v = np.zeros(n)
            T = fast_convolve(a).max() / T_ratio
            no_improve = 0
            noise_scale = max(noise_scale * 0.7, 1e-5)
        
        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label}: it={it}, obj={best_obj:.12f}, "
                  f"score={2*n*best_obj:.8f}, T={T:.8f}, t={elapsed:.0f}s")
    
    return best_a, best_obj
```

def lp_polish(f, n, max_iters=10, time_limit=60.0):
    """
    LP-based polish: solve LP with all constraints, line search, repeat.
    """
    t0 = time.monotonic()
    f = f.copy()
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(f)
        M = b.max()
        
        # Build full constraint matrix
        # A[k, i] = f[i] if 0 <= k-i < n, else 0
        # i.e., A[k, i] = f[i] if max(0, k-n+1) <= i <= min(k, n-1)
        A = np.zeros((2*n-1, n))
        for k in range(2*n-1):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            A[k, lo:hi+1] = f[lo:hi+1]
        
        # Solve LP: max sum(g) s.t. g >= 0, A@g <= M
        g_var = cp.Variable(n)
        prob = cp.Problem(cp.Maximize(cp.sum(g_var)),
                          [g_var >= 0, A @ g_var <= M])
        prob.solve(solver=cp.PDLP, max_iters=300000)
        
        if g_var.value is None:
            break
        g = np.maximum(g_var.value, 0.0)
        S_g = g.sum()
        
        if S_g <= 1.0 + 1e-5:
            # Try to add a small perturbation
            break
        
        h = g / S_g
        
        # Line search
        best_t = 0.05
        best_trial = best_obj
        for t in [0.01, 0.02, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3]:
            f_trial = (1 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_trial - 1e-14:
                best_trial = obj
                best_t = t
        
        if best_t == 0.0 and best_trial >= best_obj:
            break
        
        f = (1 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        f /= f.sum()
        
        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            print(f"    LP iter {it}: obj={best_obj:.12f}, t={best_t:.4f}")
        else:
            break
    
    return best_f, best_obj

"""
Smooth gradient descent with log-sum-exp soft-max, full LP polish, and multi-length
exploration for minimizing the auto-convolution peak of non-negative sequences.
Uses Nesterov momentum, adaptive temperature annealing, and constraint-complete LP
for final refinement.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(f, g):
    """Cross-convolution f*g via FFT."""
    size = len(f) + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def compute_grad_from_w(w, a):
    """
    Compute 2 * cross-correlation of w (length 2n-1) and a (length n).
    Result_i = 2 * sum_{j=0}^{n-1} w[i+j] * a[j]
    """
    n = len(a)
    a_flip = a[::-1].copy()
    size = n + len(w) - 1
    fft_size = 1 << (size - 1).bit_length()
    AF = rfft(a_flip, fft_size)
    W = rfft(w, fft_size)
    conv = irfft(AF * W, fft_size)[:size]
    return 2.0 * conv[n - 1:2 * n - 1]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_descent(a_init, time_limit, label="smooth", T_ratio=5.0,
                   gamma=0.9997, lr_scale=1.0, momentum=0.9,
                   batch=200, restart_freq=30, noise_init=0.005):
    """
    Smooth gradient descent with log-sum-exp soft-max approximation.
    Minimizes peak of auto-convolution subject to a >= 0, sum(a) = 1.
    """
    n = len(a_init)
    a = a_init.copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return a, fast_convolve(a).max()
    a /= s

    b0 = fast_convolve(a)
    b0_max = b0.max()
    T = b0_max / T_ratio
    T_floor = b0_max * 0.01

    v = np.zeros(n)
    best_obj = b0_max
    best_a = a.copy()
    no_improve = 0
    it = 0
    t0 = time.monotonic()
    rng = np.random.default_rng(99999)
    noise_scale = noise_init

        for _ in range(batch):
            if time.monotonic() - t0 > time_limit:
                break
            it += 1

            b = fast_convolve(a)
            b_max = b.max()

            b_norm = (b - b_max) / T
            b_norm = np.clip(b_norm, -80.0, 30.0)
            exp_b = np.exp(b_norm)
            w_sum = exp_b.sum()
            if w_sum < 1e-300:

                w[b.argmax()] = 1.0
            else:
                w = exp_b / w_sum

            grad = compute_grad_from_w(w, a)

            grad_norm = np.linalg.norm(grad)

            lr = lr_scale * b_max / (n * grad_norm)
            lr = min(lr, 0.5)

            a_new = a + v
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-10:

            else:

            a = a_new

            best_a = a.copy()
            no_improve = 0

            a = np.maximum(a, 0.0)

            else:
                a = best_a.copy()
            v = np.zeros(n)

def lp_full_solve(f, n):
    """Solve LP with all constraints: max sum(g) s.t. g>=0, (f*g)[k] <= M for all k."""
    b = fast_convolve(f)
    M = b.max()

    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[k, lo:hi + 1] = f[lo:hi + 1]

    g_var = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)),
                      [g_var >= 0, A @ g_var <= M])

    for solver in [cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, max_iters=500000)
            if g_var.value is not None:
                g = np.maximum(g_var.value, 0.0)
                if not np.any(np.isnan(g)):
                    return g, M
        except Exception:
            continue
    return None, M


def lp_polish(f_init, time_limit, label="lp", max_iters=30):
    """LP-based polish with full constraints and line search."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        g, M = lp_full_solve(f, n)
        if g is None:
            break

        S_g = g.sum()
        if S_g <= 1.0 + 1e-5:
            break

        h = g / S_g

        best_t = 0.03
        best_trial = best_obj
        for t in [0.01, 0.02, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4]:
            f_trial = (1.0 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_trial - 1e-14:
                best_trial = obj
                best_t = t

        if best_trial >= best_obj - 1e-14:
            break

        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        f /= f.sum()

        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            elapsed = time.monotonic() - t0
            print(f"    {label} iter={it}: obj={best_obj:.12f}, score={2.0*n*best_obj:.8f}, "
                  f"t_step={best_t:.4f}, time={elapsed:.0f}s")

    return best_f, best_obj


def multi_peak_descent(f_init, time_limit, n_peaks=5, label="multi_peak"):
    """Subgradient descent focusing on top-k peaks."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    v = np.zeros(n)
    momentum = 0.9
    lr = 0.01
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    it = 0
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit:
        for _ in range(100):
            if time.monotonic() - t0 > time_limit:
                break
            it += 1
            b = fast_convolve(f)

            if n_peaks < len(b):
                top_idx = np.argpartition(b, -n_peaks)[-n_peaks:]
            else:
                top_idx = np.arange(len(b))

            grad = np.zeros(n)
            for k in top_idx:
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                idx = np.arange(lo, hi + 1)
                grad[idx] += 2.0 * f[k - idx]

            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break

            step_lr = lr * b.max() / (n * grad_norm)
            step_lr = min(step_lr, 0.3)

            v = momentum * v - step_lr * grad
            f_new = f + v
            f_new = np.maximum(f_new, 0.0)
            s = f_new.sum()
            if s < 1e-10:
                f_new = f
            else:
                f_new /= s
            f = f_new

        obj = fast_convolve(f).max()
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f.copy()
            lr = min(lr * 1.02, 0.1)
        else:
            lr *= 0.95
            if it % 2000 == 0 and lr < 0.001:
                f = best_f + rng.standard_normal(n) * 0.002
                f = np.maximum(f, 0.0)
                f /= f.sum()
                v = np.zeros(n)
                lr = 0.01

    return best_f, best_obj


def flat_top_window(n):
    """Generate a normalized flat-top window."""
    w = np.ones(n)
    a_coeffs = [0.21557895, 0.41663158, 0.277263158, 0.083578947, 0.006947368]
    x = np.arange(n)
    for i, ai in enumerate(a_coeffs):
        w -= ai * np.cos(2.0 * np.pi * i * x / max(1, n - 1) - i * np.pi)
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def hann_window(n):
    """Generate a normalized Hann window."""
    w = 0.5 * (1.0 + np.cos(2.0 * np.pi * np.arange(n) / max(1, n)))
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def blackman_window(n):
    """Generate a normalized Blackman window."""
    x = np.arange(n)
    w = 0.42 - 0.5 * np.cos(2.0 * np.pi * x / max(1, n - 1)) \
        + 0.08 * np.cos(4.0 * np.pi * x / max(1, n - 1))
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def interpolate_sequence(f, target_n):
    """Interpolate a sequence to a target length and renormalize."""
    n = len(f)
    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, target_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    s = f_interp.sum()
    if s < 1e-10:
        return np.ones(target_n) / target_n
    return f_interp / s
