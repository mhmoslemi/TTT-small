OK so now I have a clean gradient computation. Let me think about the full algorithm:

1. **FISTA with smoothed max**: 
   - Smoothed objective: `f_eps(a) = eps * log(sum_k exp(b_k/eps))`
   - Gradient via FFT: O(n log n) per iteration
   - Nesterov acceleration for O(1/k^2) convergence
   - Step size: need Lipschitz constant of gradient

2. **Lipschitz constant**: The gradient is `grad = (2/eps) * conv(reversed(a), p)[n-1:2n-1]`. The Hessian involves second derivatives of the smoothed max, which are bounded by `1/eps` times the spectral norm of the "mixing" matrix. A simple bound: `L <= 8*n/eps` (crude but safe). In practice, we can use adaptive step size (backtracking).

3. **Convergence**: FISTA converges in O(1/k^2) for smooth convex functions. With `L` as the Lipschitz constant, after K iterations we get `f(a_K) - f* <= 2*L*n0^2 / K^2` where `n0` is the initial distance.

4. **Practical considerations**:
   - Start with a reasonable eps (e.g., `eps = 0.01 * M_initial`)
   - Gradually decrease eps to sharpen the gradient
   - Use backtracking line search for step size
   - Periodically evaluate the true (unsmoothed) objective

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FISTA (accelerated proximal gradient) on smoothed max autocorrelation
2. Adaptive epsilon annealing for sharpening
3. FFT-based O(n log n) gradient computation
4. Coordinate descent polishing
5. Multi-length exploration with diverse initialization
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def compute_gradient_smoothed(a, eps):
    """
    Compute gradient of f_eps(a) = eps * log(sum_k exp(conv(a,a)_k / eps)).
    Returns (gradient, b_values, p_weights, M).
    """
    n = len(a)
    b = fast_convolve(a)
    # Stable softmax
    b_shifted = b - b.max()
    exp_b = np.exp(b_shifted / eps)
    p = exp_b / exp_b.sum()
    M = float(b.max())
    
    # grad = 2 * conv(reversed(a), p)[n-1 : 2n-1]
    rev_a = a[::-1]
    conv_ap = cross_convolve(rev_a, p)
    grad = 2.0 * conv_ap[n-1 : 2*n-1]
    
    return grad, b, p, M


def fista_smoothed(a0, time_limit, eps_init=None, verbose=False, label=""):
    """
    FISTA on the smoothed max autocorrelation.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    # Initial estimate
    b_init = fast_convolve(a)
    M_init = float(b_init.max())
    if eps_init is None:
        eps = 0.005 * M_init
    else:
        eps = eps_init
    
    # FISTA initialization
    y = a.copy()
    t = 1.0
    step = 1.0  # adaptive step
    
    # Estimate Lipschitz constant
    # L ~ 8*n/eps (crude upper bound)
    L = 8.0 * n / eps
    step = 1.0 / L
    
    best_a = a.copy()
    best_true_score = 2.0 * n * M_init
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    eps_anneal_counter = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute gradient at y
        grad, b_y, p_y, M_y = compute_gradient_smoothed(y, eps)
        
        # FISTA update
        step = 1.0 / L
        y_new = simplex_project(y - step * grad)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y = (1.0 + t / t_new) * y_new - (t - 1.0) / t_new * a
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy  # Hmm, this breaks FISTA. Let me not normalize y.
        t = t_new
        
        # Evaluate true objective at y_new (the "current iterate")
        b_check = fast_convolve(y_new)
        M_check = float(b_check.max())
        score_check = 2.0 * n * M_check
        
        if score_check < best_true_score - 1e-14:
            best_true_score = score_check
            best_a = y_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive epsilon: decrease eps every 500 iterations
        eps_anneal_counter += 1
        if eps_anneal_counter >= 300:
            eps *= 0.7
            eps_anneal_counter = 0
            L = 8.0 * n / eps
            if eps < 1e-10 * M_init:
                break
        
        # If no improvement for a while, try to restart with smaller step
        if no_improve > 500:
            step *= 0.5
            L *= 2.0
            no_improve = 0
            if step < 1e-15:
                break
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FISTA {label} it={it} ({elapsed:.0f}s): best={best_true_score:.6f} eps={eps:.2e} M={M_check:.8f}")
    
    return best_a, best_true_score, it


def fista_backtracking(a0, time_limit, eps_init=None, verbose=False, label=""):

    FISTA with backtracking line search for the step size.
    """

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    b_init = fast_convolve(a)
    M_init = float(b_init.max())
    if eps_init is None:
        eps = 0.005 * M_init
    else:
        eps = eps_init
    
    y = a.copy()
    t = 1.0
    step = 1.0 / (4.0 * n)  # initial step
    c = 1e-4  # backtracking parameter
    
    best_a = a.copy()
    best_true_score = 2.0 * n * M_init
    t0 = time.monotonic()
    it = 0
    eps_anneal_counter = 0
    stagnation = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Backtracking line search
        while True:
            y_trial = simplex_project(y - step * compute_gradient_smoothed(y, eps)[0])
            # Check descent: f(y_trial) <= f(y) + grad^T(y_trial - y) + (1/(2*step)) * ||y_trial - y||^2
            f_y_trial = eps * np.log(np.sum(np.exp((fast_convolve(y_trial) - fast_convolve(y_trial).max()) / eps)))
            # This is getting expensive. Let me use a simpler criterion.
            # Just check that the step doesn't increase the smoothed objective too much.
            b_trial = fast_convolve(y_trial)
            f_trial = eps * np.log(np.sum(np.exp((b_trial - b_trial.max()) / eps)))
            b_y = fast_convolve(y)
            f_y = eps * np.log(np.sum(np.exp((b_y - b_y.max()) / eps)))
            grad_y = compute_gradient_smoothed(y, eps)[0]
            sufficient = f_trial <= f_y + np.dot(grad_y, y_trial - y) + (1.0 / (2.0 * step)) * np.dot(y_trial - y, y_trial - y) + 1e-10
            if sufficient:
                break
            step *= 0.5
            if step < 1e-18:
                break
        
        y_new = simplex_project(y - step * compute_gradient_smoothed(y, eps)[0])
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        y_new_f = (1.0 + t / t_new) * y_new - (t - 1.0) / t_new * a
        y_new_f = np.maximum(y_new_f, 0.0)
        s_yf = y_new_f.sum()
        if s_yf > 1e-15:
            y_new_f = y_new_f / s_yf
        t = t_new
        a_old = a.copy()
        a = y_new
        y = y_new_f
        
        # Evaluate true objective
        b_check = fast_convolve(a)
        M_check = float(b_check.max())
        score_check = 2.0 * n * M_check
        
        if score_check < best_true_score - 1e-14:
            best_true_score = score_check
            best_a = a.copy()
            stagnation = 0
        else:
            stagnation += 1
        
        # Anneal epsilon
        eps_anneal_counter += 1
        if eps_anneal_counter >= 200:
            eps *= 0.6
            eps_anneal_counter = 0
            step = 1.0 / (4.0 * n / eps)  # reset step
            if eps < 1e-12 * M_init:
                break
        
        if stagnation > 300:
            break
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FISTA-BT {label} it={it} ({elapsed:.0f}s): best={best_true_score:.6f} eps={eps:.2e} step={step:.2e}")
    
    return best_a, best_true_score, it
