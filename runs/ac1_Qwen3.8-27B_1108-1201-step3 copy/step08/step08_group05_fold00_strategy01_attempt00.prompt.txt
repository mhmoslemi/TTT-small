import numpy as np

def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(f, g):
    """Cross-convolution f*g via FFT."""
    size = len(f) + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def quadratic_line_search(f, direction, max_evals=20):
    """Find best step size using golden section search on the peak objective."""
    n = len(f)

    def peak_at(t):
        f_trial = (1.0 - t) * f + t * direction
        f_trial = np.maximum(f_trial, 0.0)
        s = f_trial.sum()
        if s < 1e-10:
            return 1e10
        f_trial = f_trial / s
        return fast_convolve(f_trial).max()

    # Initial coarse search
    t_candidates = [0.02, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]
    best_t = 0.05
    best_obj = peak_at(0.05)
    for t in t_candidates:
        obj = peak_at(t)
        if obj < best_obj:
            best_obj = obj
            best_t = t

    # Refine with golden section
    idx = t_candidates.index(best_t)
    lo = t_candidates[max(0, idx - 1)]
    hi = t_candidates[min(len(t_candidates) - 1, idx + 1)]
    if lo == hi:
        return best_t

    gr = (np.sqrt(5) - 1) / 2
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = peak_at(c)
    fd = peak_at(d)

    for _ in range(12):
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = peak_at(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = peak_at(d)
        if hi - lo < 1e-5:
            break

    best_t = (lo + hi) / 2
    return best_t

def focused_descent(f, time_limit, n_peaks=5):
    """Direct steepest descent on the top peaks with adaptive step size."""
    t0 = time.monotonic()
    n = len(f)
    f = f.copy()
    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(f)
        # Find top peaks
        if n_peaks < len(b):
            top_idx = np.argpartition(b, -n_peaks)[-n_peaks:]
        else:
            top_idx = np.arange(len(b))

        # Compute combined gradient (direction to reduce peaks)
        grad = np.zeros(n)
        for k in top_idx:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                grad[j] += f[k - j]

        if grad.max() < 1e-15:
            break

        # Try different step sizes
        best_alpha = None
        best_trial = best_obj
        for alpha in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.15, 0.2]:
            f_trial = f - alpha * grad
            f_trial = project_simplex(f_trial)
            obj = fast_convolve(f_trial).max()
            if obj < best_trial - 1e-14:
                best_trial = obj
                best_alpha = alpha

        if best_alpha is None:
            # Try very small steps
            for alpha in [0.0005, 0.001]:
                f_trial = f - alpha * grad
                f_trial = project_simplex(f_trial)
                obj = fast_convolve(f_trial).max()
                if obj < best_trial - 1e-14:
                    best_trial = obj
                    best_alpha = alpha

        if best_alpha is None:
            break

        f_new = f - best_alpha * grad
        f_new = project_simplex(f_new)
        obj_new = fast_convolve(f_new).max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new

    return best_f, best_obj


def subgradient_polish(f_init, time_limit, lr_init=0.001, n_top=30):
    """Subgradient optimization with momentum and restarts."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    v = np.zeros(n)
    lr = lr_init
    temp = 0.001
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        # Batch of steps
        batch = min(100, max(20, int((time_limit - (time.monotonic() - t0)) * 2)))
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit:
                break
            it += 1
            b = fast_convolve(f)

            # Temperature-based weighting
            scaled = b / max(temp, 1e-10)
            scaled -= scaled.max()
            w = np.exp(scaled)
            w_sum = w.sum()
            if w_sum < 1e-30:
                w = np.zeros(len(b))
                w[b.argmax()] = 1.0
            else:
                w /= w_sum

            # Top-k restriction
            if n_top < len(b):
                top_idx = np.argpartition(b, -n_top)[-n_top:]
                w_sel = w[top_idx].copy()
                ws = w_sel.sum()
                if ws > 1e-30:
                    w_sel /= ws
                else:
                    w_sel = np.ones(len(top_idx)) / len(top_idx)
                w_new = np.zeros(len(b))
                w_new[top_idx] = w_sel
                w = w_new

            # Compute gradient
            g = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-13:
                    continue
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                idx = np.arange(lo, hi + 1)
                g[idx] += w[k] * f[k - idx]

            v = 0.85 * v - lr * g
            f_new = project_simplex(f + v)
            f = f_new

        # Check improvement periodically
        obj = fast_convolve(f).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
            lr = min(lr * 1.03, 0.008)
        else:
            no_improve += 1
            if no_improve > 2:
                lr *= 0.92
                no_improve = 0
            if no_improve > 10:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                v = np.zeros(n)
                no_improve = 0

        if it % 500 == 0:
            temp = min(temp * 1.3, 1.0)
            print(f"    sg iter={it} obj={best_obj:.12f} "
                  f"score={2.0 * n * best_obj:.6f} t={time.monotonic() - t0:.0f}s")

    return best_f, best_obj


def flat_top_window(n):
    """Generate a normalized flat top window."""
    w = np.ones(n)
    a = [0.21557895, 0.41663158, 0.277263158, 0.083578947, 0.006947368]
    for i, ai in enumerate(a):
        w -= ai * np.cos(2 * np.pi * i * np.arange(n) / max(1, n - 1) - i * np.pi)
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def hann_window(n):
    """Generate a normalized Hann window."""
    w = 0.5 * (1.0 + np.cos(2.0 * np.pi * np.arange(n) / max(1, n)))
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def raised_cosine_window(n):
    """Generate a normalized raised cosine (tukey-like) window."""
    x = np.linspace(0, 1, n)
    w = 0.5 * (1 + np.cos(2 * np.pi * (x - 0.5)))
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s
