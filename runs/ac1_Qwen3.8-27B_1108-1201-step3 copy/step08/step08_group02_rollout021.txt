def fast_convolve(a):
    n = len(a)

    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def solve_lp(f, n, active_k, M, tol=1e-8):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)[k] <= M for k in active_k"""
    K = len(active_k)
    if K == 0:
        return np.ones(n) / n

    for row in range(K):

        for i in range(n):

            if 0 <= j < n:
                A_mat[row, i] = f[i]
    
    g = cp.Variable(n)
    constraints = [g >= 0, A_mat @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

            if g.value is not None:
                g_val = np.maximum(g.value, 0.0)
                if not np.any(np.isnan(g_val)):

        except:
            continue
    return None

"""

dynamic active set management, subgradient polishing, and multi-length exploration.
Target: minimize 2*n*max(conv(a,a))/(sum(a))^2 below 1.5027.

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, b, peak_indices, temperature=0.003):
    """Compute soft subgradient of max(conv(a,a)) using softmax weights on peaks."""
    n = len(a)
    top_b = b[peak_indices]
    max_b = top_b.max()
    if max_b <= 0:
        return np.zeros(n)
    scaled = top_b / (temperature * max_b)
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()

    grad = np.zeros(n)
    for idx, wk in zip(peak_indices, w):
        k = int(idx)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        m = np.arange(lo, hi + 1)
        grad[m] += 2.0 * wk * a[k - m]
    return grad


def solve_lp_direction(f, n, active_k, M, eps=1e-8):
    """
    Solve LP: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for k in active_k.
    Returns (g, sum_g) or (None, 0).
    """
    K = len(active_k)
    if K == 0:
        return np.ones(n) / n, 1.0

    # Build constraint matrix efficiently
    A_mat = np.zeros((K, n))
    for row in range(K):
        k = int(active_k[row])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            m = np.arange(lo, hi + 1)
            A_mat[row, lo:hi + 1] = f[k - m]

    g = cp.Variable(n)
    constraints = [g >= 0, A_mat @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    solvers = [cp.SCS, cp.CLARABEL, cp.ECOS]
    for solver in solvers:
        try:
            kwargs = {'verbose': False}
            if solver == cp.SCS:
                kwargs['max_iters'] = 1000000
                kwargs['eps'] = eps
            elif solver == cp.ECOS:
                kwargs['max_iters'] = 100000
            prob.solve(solver=solver, **kwargs)
            if g.value is not None:
                g_val = np.maximum(g.value, 0.0)
                if not np.any(np.isnan(g_val)) and not np.any(np.isinf(g_val)):
                    return g_val, g_val.sum()
        except Exception:
            continue
    return None, 0.0


def line_search(f, h, M_current, max_evals=20):
    """
    Golden-section + grid line search for best t in [0, 1].
    Returns (best_t, best_obj).
    """
    best_t = 0.0
    best_obj = M_current

    # Coarse grid
    t_grid = np.array([0.03, 0.05, 0.08, 0.12, 0.18, 0.25, 0.35, 0.50, 0.65, 0.80])
    for t in t_grid:
        f_trial = (1.0 - t) * f + t * h
        f_trial = np.maximum(f_trial, 0.0)
        s = f_trial.sum()
        if s < 1e-10:
            continue
        f_trial = f_trial / s
        obj = fast_convolve(f_trial).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_t = t

    # Refine with golden section around best_t
    if best_t > 0:
        lo = max(0.0, best_t * 0.5)
        hi = min(1.0, best_t * 1.5 + 0.05)
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c = hi - gr * (hi - lo)
        d = lo + gr * (hi - lo)
        fc = 0.0
        fd = 0.0
        for _ in range(10):
            for t_val in [c, d]:
                f_trial = (1.0 - t_val) * f + t_val * h
                f_trial = np.maximum(f_trial, 0.0)
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial = f_trial / s
                obj = fast_convolve(f_trial).max()
                if t_val == c:
                    fc = obj
                else:
                    fd = obj
            if fc < fd:
                hi = d
                d = c
                fd = fc
                c = hi - gr * (hi - lo)
            else:
                lo = c
                c = d
                fc = fd
                d = lo + gr * (hi - lo)
        t_best = (lo + hi) / 2.0
        f_trial = (1.0 - t_best) * f + t_best * h
        f_trial = np.maximum(f_trial, 0.0)
        s = f_trial.sum()
        if s > 1e-10:
            f_trial = f_trial / s
            obj = fast_convolve(f_trial).max()
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_t = t_best

    return best_t, best_obj


def run_lp_stagewise(f_init, time_limit, label="lp", max_active=None):
    """
    LP-based optimization with dynamic active set management.
    Uses staging: start with few peaks, gradually add more.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-10:
        f = f / s
    else:
        f = np.ones(n) / n

    best_f = f.copy()
    best_b = fast_convolve(f).max()
    it = 0
    stuck = 0
    rng = np.random.default_rng(1234)

    if max_active is None:
        max_active = min(2 * n - 1, 600)

    # Staging: start small, grow
    stage_schedule = [
        (0, 30),       # first 30 iters: 30 peaks
        (30, 60),      # next 30: 60 peaks
        (60, 100),     # next 40: 100 peaks
        (100, 200),    # next 100: 200 peaks
        (200, 400),    # next 200: 400 peaks
        (400, max_active),  # rest: max
    ]

    current_active_max = 30

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Update active set size based on iteration
        for start, end in stage_schedule:
            if it >= start and it < end:
                current_active_max = end
                break
        else:
            current_active_max = max_active

        b = fast_convolve(f)
        M = b.max()

        # Select active peaks: all peaks above 99.5% of max, up to current_active_max
        threshold = 0.995 * M
        all_peaks = np.where(b >= threshold)[0]
        if len(all_peaks) > current_active_max:
            active_k = np.argpartition(b, -current_active_max)[-current_active_max:]
        elif len(all_peaks) < 5:
            active_k = np.argpartition(b, -max(10, n // 50))[-max(10, n // 50):]
        else:
            active_k = all_peaks

        # Ensure the actual max position is included
        max_pos = np.argmax(b)
        if max_pos not in active_k:
            active_k = np.append(active_k, max_pos)

        # Solve LP
        g, S_g = solve_lp_direction(f, n, active_k, M, eps=1e-8)

        if g is None:
            stuck += 1
            if stuck > 30:
                # Perturb and reset
                f = f + rng.standard_normal(n) * 0.001 * f
                f = np.maximum(f, 0.0)
                s = f.sum()
                if s > 1e-10:
                    f = f / s
                stuck = 0
            continue

        if S_g <= 1.0 + 1e-5:
            stuck += 1
            if stuck > 30:
                f = f + rng.standard_normal(n) * 0.002 * f
                f = np.maximum(f, 0.0)
                s = f.sum()
                if s > 1e-10:
                    f = f / s
                stuck = 0
            continue

        # Line search
        h = g / S_g
        best_t, best_t_obj = line_search(f, h, M)

        if best_t <= 0:
            stuck += 1
            if stuck > 30:
                f = f + rng.standard_normal(n) * 0.001 * f
                f = np.maximum(f, 0.0)
                s = f.sum()
                if s > 1e-10:
                    f = f / s
                stuck = 0
            continue

        # Update
        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f = f / s
        else:
            break

        cur_obj = fast_convolve(f).max()
        if cur_obj < best_b - 1e-14:
            best_b = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1

        # Periodic perturbation when stuck
        if stuck > 15 and stuck % 10 == 0:
            f = f + rng.standard_normal(n) * 0.0005 * np.maximum(f, 0.01)
            f = np.maximum(f, 0.0)
            s = f.sum()
            if s > 1e-10:
                f = f / s

        if it % 20 == 0:
            score = 2.0 * n * best_b
            print(f"    {label} it={it} obj={best_b:.10f} score={score:.6f} "
                  f"active={len(active_k)} Sg={S_g:.4f} t={time.monotonic()-t0:.0f}s")

    return best_f, best_b


def subgradient_polish(f_init, time_limit, label="sg", lr_init=0.003):
    """Subgradient descent with momentum and adaptive temperature."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    v = np.zeros(n)
    best_f = f.copy()
    best_b = fast_convolve(f).max()
    lr = lr_init
    temp = 0.003
    no_improve = 0
    it = 0
    rng = np.random.default_rng(777)
    momentum = 0.85

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()

        # Select top peaks
        n_top = min(len(b), max(50, n // 10))
        top_idx = np.argpartition(b, -n_top)[-n_top:]

        # Compute soft subgradient
        grad = compute_subgradient(f, b, top_idx, temperature=temp)

        # Normalized gradient step
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-15:
            grad = grad / grad_norm

        # Momentum update
        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)

        # Check improvement
        obj_new = fast_convolve(f_new).max()
        if obj_new < best_b - 1e-14:
            best_b = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            lr = min(lr * 1.01, 0.02)
        else:
            f = f_new
            no_improve += 1
            if no_improve % 10 == 0:
                lr *= 0.93
                temp = max(temp * 1.05, 0.001)

            if no_improve > 50:
                # Reset from best
                f = best_f + rng.standard_normal(n) * 0.002
                f = np.maximum(f, 0.0)
                s = f.sum()
                if s > 1e-10:
                    f = f / s
                v = np.zeros(n)
                lr = lr_init
                temp = 0.003
                no_improve = 0

        if it % 100 == 0:
            score = 2.0 * n * best_b
            print(f"    {label} it={it} obj={best_b:.10f} score={score:.6f} "
                  f"lr={lr:.5f} temp={temp:.4f} t={time.monotonic()-t0:.0f}s")

    return best_f, best_b


def coordinate_descent(f_init, time_limit, label="cd"):
    """Coordinate descent for final polishing."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_b = fast_convolve(f).max()
    multipliers = [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0]

    passes = 0
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-15:
                continue
            best_j_val = f[j]
            best_j_obj = best_b
            for mult in multipliers:
                new_val = f[j] * mult
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial = f_trial / s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-14:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f = f / s
                if best_j_obj < best_b - 1e-14:
                    best_b = best_j_obj
                    improved = True
        if not improved:
            break

    return f, best_b


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    n = len(f)
    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new = f_new / s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def random_warm_start(n, rng, parent=None):
    """Create a random warm start, optionally biased toward parent."""
    if parent is not None and rng.random() < 0.5:
        # Start from parent with perturbation
        f = parent + rng.standard_normal(n) * 0.1
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f = f / s
        else:
            f = np.ones(n) / n
        return f
    else:
        # Random Dirichlet
        alpha = rng.random(n) * 0.5 + 0.1
        f = rng.dirichlet(alpha)
        return f

        f2, obj2 = run_lp_stagewise(best_seq, time_limit=t2, label="lp2",
                                     max_active=min(2 * n_b - 1, 700))
        vs2 = float(evaluate_sequence(f2.tolist()))
        if np.isfinite(vs2) and vs2 < best_score:
            best_score = vs2
            best_seq = f2.copy()
            print(f"  NEW BEST [lp2]: {vs2:.6f}")

    # === Phase 3: Subgradient polish (15% of time) ===
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.15, 200.0)
    if t3 > 30:
        n_b = len(best_seq)
        print(f"\n=== Phase 3: Subgradient (n={n_b}, budget={t3:.0f}s) ===")
        f3, obj3 = subgradient_polish(best_seq, time_limit=t3, label="sg",
                                       lr_init=0.003)
        vs3 = float(evaluate_sequence(f3.tolist()))
        if np.isfinite(vs3) and vs3 < best_score:
            best_score = vs3
            best_seq = f3.copy()
            print(f"  NEW BEST [sg]: {vs3:.6f}")

    # === Phase 4: Multi-length exploration (15% of time) ===
    rem = deadline - time.monotonic()
    n_base = len(best_seq)
    target_lengths = []
    for factor in [0.85, 1.1, 1.25, 1.5, 0.75]:
        tl = int(n_base * factor)
        tl = max(512, min(3000, tl))
        if tl != n_base and tl not in target_lengths:
            target_lengths.append(tl)
    target_lengths.sort(key=lambda x: abs(x - n_base))

    for target_n in target_lengths[:3]:
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t4 = min(rem * 0.10, 120.0)
        if t4 < 20:
            break
        f_interp = interpolate_to_length(best_seq, target_n)
        print(f"\n=== Phase 4: LP n={target_n} (budget={t4:.0f}s) ===")
        f4, obj4 = run_lp_stagewise(f_interp, time_limit=t4, label=f"lp_n{target_n}",
                                     max_active=min(2 * target_n - 1, 400))
        vs4 = float(evaluate_sequence(f4.tolist()))
        if np.isfinite(vs4) and vs4 < best_score:
            best_score = vs4
            best_seq = f4.copy()
            print(f"  NEW BEST [n{target_n}]: {vs4:.6f}")

    # === Phase 5: Second LP pass on best (10% of time) ===
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.10, 150.0)
    if t5 > 20:
        n_b = len(best_seq)
        print(f"\n=== Phase 5: LP second pass (n={n_b}, budget={t5:.0f}s) ===")
        f5, obj5 = run_lp_stagewise(best_seq, time_limit=t5, label="lp5",
                                     max_active=min(2 * n_b - 1, 600))
        vs5 = float(evaluate_sequence(f5.tolist()))
        if np.isfinite(vs5) and vs5 < best_score:
            best_score = vs5
            best_seq = f5.copy()
            print(f"  NEW BEST [lp5]: {vs5:.6f}")

    # === Phase 6: Final subgradient + coordinate descent ===
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.08, 80.0)
    if t6 > 15:
        n_b = len(best_seq)
        print(f"\n=== Phase 6: SG polish (n={n_b}, budget={t6:.0f}s) ===")
        f6, obj6 = subgradient_polish(best_seq, time_limit=t6, label="sg2",
                                       lr_init=0.002)
        vs6 = float(evaluate_sequence(f6.tolist()))
        if np.isfinite(vs6) and vs6 < best_score:
            best_score = vs6
            best_seq = f6.copy()
            print(f"  NEW BEST [sg2]: {vs6:.6f}")

    rem = deadline - time.monotonic()
    t7 = min(rem * 0.12, 100.0)
    if t7 > 10:
        print(f"\n=== Phase 7: Coordinate descent (budget={t7:.0f}s) ===")
        f7, obj7 = coordinate_descent(best_seq, time_limit=t7, label="cd")
        vs7 = float(evaluate_sequence(f7.tolist()))
        if np.isfinite(vs7) and vs7 < best_score:
            best_score = vs7
            best_seq = f7.copy()
            print(f"  NEW BEST [cd]: {vs7:.6f}")

    # === Final verification ===
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
