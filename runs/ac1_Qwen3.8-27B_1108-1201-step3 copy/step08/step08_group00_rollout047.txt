import cvxpy as cp
import time
from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def active_set_lp(a_fixed, K=20, solver=None):
    """
    Solve LP: minimize M subject to:
    - conv(a_fixed, b)[k] <= M for k in active_set (top K peaks)
    - conv(a_fixed, b)[k] >= 0 for all k (implied by b >= 0)
    - b >= 0
    - sum(b) = 1
    """
    if solver is None:
        solver = get_lp_solver()
    
    n = len(a_fixed)
    b = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    
    # Compute active set
    conv_af = fast_convolve(a_fixed)
    # We want to constrain conv(a_fixed, b) at the peak positions
    # But actually, we should constrain ALL positions where the current
    # conv is high, to prevent the new b from creating new peaks elsewhere.
    # So let's use top K positions of conv(a_fixed, a_fixed).
    
    active_indices = np.argsort(conv_af)[::-1][:K]
    
    # Build constraint: for each active index k, conv(a_fixed, b)[k] <= M
    # conv(a_fixed, b)[k] = sum_j a_fixed[j] * b[k-j] (with proper bounds)
    
    constraints = [cp.sum(b) == 1]
    
    for k in active_indices:
        # conv(a_fixed, b)[k] = sum_{j=max(0,k-n+1)}^{min(n-1,k)} a_fixed[j] * b[k-j]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        # This is a linear expression in b
        linear = sum(a_fixed[j] * b[k - j] for j in range(js, je + 1))
        constraints.append(linear <= M)
    
    # Also add a constraint that the total convolution sum is bounded
    # (this is automatically satisfied since sum(b)=1 and a_fixed >= 0)
    
    prob = cp.Problem(cp.Minimize(M), constraints)
    prob.solve(solver=solver)
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.asarray(b.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Nesterov-accelerated LP squeezing
with full constraints, adaptive two-timescale updates, L-BFGS LogSumExp refinement,
and multi-length exploration. Key improvements: two-timescale LP (multiple best
responds per outer step), tighter solver tolerances, adaptive active-set fallback
for large n, and systematic length exploration.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix_rows(a_fixed, row_indices):
    """Build rows of convolution matrix T_a for given row indices.
    Returns matrix T such that T @ b = [conv(a_fixed,b)[k] for k in row_indices]."""
    n = len(a_fixed)
    T = np.zeros((len(row_indices), n))
    for i, k in enumerate(row_indices):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            T[i, js:je+1] = a_fixed[js:je+1]
    return T


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for name in [cp.SCS, cp.ECOS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            p = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            p.solve(solver=name)
            if p.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = name
                return name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None, topK=None):
    """Solve LP: min M s.t. conv(a_fixed, b) <= M (all or topK positions), sum(b)=1, b>=0."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)

    if topK is not None and topK < 2 * n - 1:
        conv_aa = fast_convolve(a_fixed)
        active = np.argsort(conv_aa)[::-1][:topK]
    else:
        active = np.arange(2 * n - 1)

    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix_rows(a_fixed, active)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        pass

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.SCS, cp.ECOS, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_two_timescale(a0, time_limit, solver=None, inner_steps=3,
                               topK=None, n_max_iters=200000, restart_thresh=0.005):
    """
    Nesterov-accelerated LP with two-timescale inner loop.
    Inner loop does inner_steps LP solves per outer Nesterov update.
    Uses adaptive topK for speed.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_iter = 0
    b_prev = None

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Adaptive topK: start with fewer constraints, grow over time
        if topK is None:
            if it < 200:
                k_use = min(50, 2 * n - 1)
            elif it < 1000:
                k_use = min(150, 2 * n - 1)
            elif it < 5000:
                k_use = min(400, 2 * n - 1)
            else:
                k_use = 2 * n - 1  # full constraints
        else:
            k_use = topK

        # Inner loop: multiple LP steps
        b_best = None
        M_best = None
        for _ in range(inner_steps):
            b_new, M_new = solve_lp_best_response(y, solver=solver, topK=k_use)
            if b_new is None:
                continue
            if M_best is None or M_new < M_best:
                b_best = b_new
                M_best = M_new

        if b_best is None:
            # Fallback: use current y
            b_best = y.copy()

        b_prev = b_best

        # Nesterov update
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_best
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = b_best.copy()
            sy = 1.0
        y = y / sy

        # Evaluate
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()
            stall_iter = 0
        else:
            stall_iter += 1

        # Restart if stalled
        if stall_iter > 500:
            noise = np.random.default_rng(it * 7 + 13).random(n) * 0.03
            y = y + noise
            y = np.maximum(y, 0)
            y = y / y.sum()
            best_a = y.copy()
            best_score = 2.0 * n * fast_convolve(y).max()
            stall_iter = 0

        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NTLP it={it+1} ({elapsed:.0f}s): score={best_score:.7f} k_use={k_use}")

    return best_a, best_score


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def lbgfs_refine(a0, time_limit, T_start=80.0, T_end=0.2, maxiter=10000):
    """L-BFGS with LogSumExp smoothing and geometric annealing."""
    from scipy.optimize import minimize
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()

    def make_func(T_val):
        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    n_stages = 8
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(30, maxiter // n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def mirror_descent_opt(a0, time_limit, lr_init=0.04, n_steps=50000):
    """Mirror descent on the simplex with active-set subgradient."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    z = np.log(np.maximum(a, 1e-300))
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b - 0.02 * max_b
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        active = active[:min(len(active), 20)]

        grad = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * a[k - idx]
        grad /= len(active)

        z_new = z - lr * grad
        a_new = softmax_from_z(z_new)
        score_new = 2.0 * n * fast_convolve(a_new).max()

        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            z = z_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 100:
            lr *= 0.9
            no_improve = 0
        lr = max(lr_init / (1.0 + step * 0.0003), 1e-9)

        if step % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD {elapsed:.0f}s: score={best_score:.7f} lr={lr:.6f}")

    return best_a, best_score


def structured_init(n, rng):
    """Create structured initial sequence."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.sin(np.pi * x) ** 2
    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = a ** 2
    elif choice == 4:
        a = np.exp(-3 * (x - 0.5) ** 2)
    else:
        a = (1 + np.cos(2 * np.pi * x)) * (1 + np.cos(4 * np.pi * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization: two-timescale Nesterov LP (main), L-BFGS LogSumExp,
    mirror descent, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.7f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 960
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.7f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.7f} (n={len(a)})")

    # === Phase 1: Two-timescale Nesterov LP on primary length ===
    print(f"\n=== Phase 1: NTLP n={n_best} (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.50, 900.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_two_timescale(best_seq, t_alloc, solver=solver,
                                             inner_steps=3, n_max_iters=200000)
        try_update(a_lp, "ntlp_main")

        # Second pass with full constraints
        rem = deadline - time.monotonic()
        if rem > 120 and best_seq is not None:
            t2 = min(rem * 0.30, 400.0)
            print(f"  Phase 1b: full-constraint LP, budget={t2:.0f}s")
            a_lp2, _ = nesterov_lp_two_timescale(best_seq, t2, solver=solver,
                                                  inner_steps=2, topK=None,
                                                  n_max_iters=100000)
            try_update(a_lp2, "ntlp_full")

    # === Phase 2: L-BFGS LogSumExp ===
    print(f"\n=== Phase 2: L-BFGS (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.20, 300.0)
    if t_alloc > 30:
        a_lb = lbgfs_refine(best_seq, t_alloc, T_start=100.0, T_end=0.1, maxiter=15000)
        try_update(a_lb, "lbgfs_main")

    # === Phase 3: Mirror descent ===
    print(f"\n=== Phase 3: Mirror Descent (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 120.0)
    if t_alloc > 15:
        a_md, _ = mirror_descent_opt(best_seq, t_alloc, lr_init=0.03, n_steps=80000)
        try_update(a_md, "mirror_desc")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-Length (best={best_score:.7f}) ===")
    candidates_n = [n_best]
    for mult in [1.25, 1.5, 0.8, 2.0]:
        alt_n = int(n_best * mult)
        if 256 <= alt_n <= 3000 and alt_n != n_best:
            candidates_n.append(alt_n)
    candidates_n = sorted(set(candidates_n))[:3]

    for alt_n in candidates_n:
        if alt_n == n_best:
            continue
        rem = deadline - time.monotonic()
        if rem < 100:
            break
        t_total = min(rem * 0.35, 400.0)
        if t_total < 50:
            break
        print(f"  Exploring n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_seq, alt_n)
        # Also try structured init
        init_struct = structured_init(alt_n, rng)

        # LP on interpolated
        t_lp = t_total * 0.7
        a_alt, _ = nesterov_lp_two_timescale(init_alt, t_lp, solver=solver,
                                              inner_steps=2, n_max_iters=50000)
        try_update(a_alt, f"lp_n{alt_n}")

        # LP on structured
        rem = deadline - time.monotonic()
        if rem > 40:
            t_lp2 = min(rem * 0.3, 100.0)
            a_alt2, _ = nesterov_lp_two_timescale(init_struct, t_lp2, solver=solver,
                                                   inner_steps=2, n_max_iters=30000)
            try_update(a_alt2, f"lp_struct_n{alt_n}")

    # === Phase 5: Final polish ===
    print(f"\n=== Phase 5: Final Polish (best={best_score:.7f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t_fin = min(rem * 0.4, 100.0)
        a_fin = lbgfs_refine(best_seq, t_fin, T_start=5.0, T_end=0.05, maxiter=8000)
        try_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 30:
        t_md = min(rem * 0.3, 60.0)
        a_md2, _ = mirror_descent_opt(best_seq, t_md, lr_init=0.01, n_steps=30000)
        try_update(a_md2, "final_md")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.7f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
