```python
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Algorithm: find threshold tau such that sum(max(v - tau, 0)) = 1
    # Use sorting
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.argmax(1 - u * np.arange(1, n + 1) > cssv[n-1] - cssv + np.arange(1, n+1))
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Entropic mirror descent with adaptive soft/hard subgradients (fast global exploration)
2. Precise pair-transfer coordinate descent with golden section (local refinement)
3. LP-based direction finding for escape from local minima
4. Multi-length exploration
5. Nesterov-accelerated mirror descent variant

Key differences from previous approach:
- Mirrors descent uses multiplicative updates (exp(-eta*grad)) instead of additive
- Adaptive switching between soft and hard subgradients
- More aggressive coordinate descent with peak-aware pair selection
- Better length exploration with targeted budgets
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the optimization score for a normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = 1 + np.arange(1, n + 1) * u > cssv
    if cond.all():
        rho = n
    else:
        rho = int(np.argmax(~cond))
    tau = (cssv[rho - 1] - 1.0) / rho
    x = np.maximum(v - tau, 0.0)
    s = x.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return x / s


def cross_correlation(w, a):
    """Compute (w cross-corr a)[i] = sum_m w[m+i]*a[m] for i=0..len(a)-1."""
    n = len(a)
    N = 1 << (3 * n - 2 - 1).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    return irfft(np.conj(W) * A, N)[:n]


def mirror_descent(a0, time_limit, verbose=True, eta0=None, use_nesterov=False):
    """Entropic mirror descent with adaptive step size."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    if eta0 is None:
        eta0 = 0.5 / n

    eta = eta0
    eta_decay = 0.9995
    eta_min = 1e-8
    a_prev = a.copy()
    momentum = 0.0
    T_smooth = max(1.0, n * 0.1)

    for t in range(5000000):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_convolve(a)
        M = c.max()

        # Adaptive softness: use harder subgradient as we converge
        progress = min(1.0, t / 50000.0)
        T_eff = T_smooth * (1.0 - 0.9 * progress) + 0.1 * progress * M * 0.01
        T_eff = max(T_eff, 1e-6)

        # Soft weights
        c_shift = c / T_eff
        c_shift -= c_shift.max()
        w = np.exp(c_shift)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        # Gradient
        g = 4.0 * n * cross_correlation(w, a)

        # Optional Nesterov extrapolation
        if use_nesterov and t > 0:
            m_coef = (t - 1) / (t + 2)
            y = a + m_coef * (a - a_prev)
            y = np.maximum(y, 1e-30)
            y = y / y.sum()
            # Recompute gradient at y
            c_y = fast_convolve(y)
            M_y = c_y.max()
            c_y_shift = c_y / T_eff
            c_y_shift -= c_y_shift.max()
            w_y = np.exp(c_y_shift)
            w_y_sum = w_y.sum()
            if w_y_sum > 1e-30:
                w_y = w_y / w_y_sum
                g = 4.0 * n * cross_correlation(w_y, y)

        # Mirror descent update (multiplicative)
        a_new = a * np.exp(-eta * g)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-30:
            eta *= 0.5
            continue
        a_new = a_new / s

        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        a_prev = a
        a = a_new
        eta = max(eta * eta_decay, eta_min)

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD iter={t}, eta={eta:.2e}, T_eff={T_eff:.4f}, "
                  f"score={score:.8f}, best={best_score:.8f}, elapsed={elapsed:.0f}s")

    return best_a, best_score


def coordinate_descent_pairs(a0, time_limit, verbose=True, max_pairs_per_sweep=200):
    """Coordinate descent with pair transfers and golden section search."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    sweep = 0
    no_improve_sweeps = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        sweep_improved = False

        # Get current convolution for peak-aware pair selection
        conv = fast_convolve(a)
        top_peak = int(np.argmax(conv))

        # Generate candidate pairs
        pairs = []

        # Peak-focused pairs
        js = max(0, top_peak - n + 1)
        je = min(n - 1, top_peak)
        peak_range = je - js + 1
        if peak_range > 1:
            for _ in range(max_pairs_per_sweep // 2):
                i = int(rng.integers(js, je + 1))
                j = int(rng.integers(js, je + 1))
                if i != j:
                    pairs.append((min(i, j), max(i, j)))

        # Random pairs
        for _ in range(max_pairs_per_sweep // 3):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                pairs.append((min(i, j), max(i, j)))

        # Edge pairs (to reduce edge effects)
        edge_width = max(2, n // 16)
        for _ in range(max_pairs_per_sweep // 6):
            i = int(rng.integers(0, edge_width))
            j = int(rng.integers(n - edge_width, n))
            if i != j:
                pairs.append((min(i, j), max(j, i)))

        # Pairs involving the largest and smallest elements
        for _ in range(max_pairs_per_sweep // 6):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j and a[i] > 0 and a[j] > 0:
                pairs.append((min(i, j), max(i, j)))

        # Shuffle
        rng.shuffle(pairs)

        for (i, j) in pairs[:max_pairs_per_sweep]:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-15 and a[j] < 1e-15:
                continue

            t_lo = -a[j] + 1e-15
            t_hi = a[i] - 1e-15
            if t_hi <= t_lo:
                continue

            # Golden section search
            x1 = t_hi - gr * (t_hi - t_lo)
            x2 = t_lo + gr * (t_hi - t_lo)

            def eval_transfer(t_val):
                a[i] -= t_val
                a[j] += t_val
                if a[i] < -1e-12 or a[j] < -1e-12:
                    return 1e10
                nn = max(a[i], 0.0)
                a[i] = nn
                sc = compute_score(a)
                a[j] -= t_val
                a[i] += t_val
                return sc

            f1 = eval_transfer(x1)
            f2 = eval_transfer(x2)

            for _ in range(25):
                if time.monotonic() - t0 > time_limit:
                    break
                if f1 < f2:
                    t_hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = t_hi - gr * (t_hi - t_lo)
                    f1 = eval_transfer(x1)
                else:
                    t_lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = t_lo + gr * (t_hi - t_lo)
                    f2 = eval_transfer(x2)

            t_opt = (t_lo + t_hi) / 2.0
            f_opt = min(f1, f2)

            if f_opt < best_score - 1e-12:
                a[i] -= t_opt
                a[j] += t_opt
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                best_score = compute_score(a)
                best_a = a.copy()
                sweep_improved = True

        if not sweep_improved:
            no_improve_sweeps += 1
            if no_improve_sweeps >= 5:
                # Perturb to escape local minimum
                noise_scale = 0.005 / np.sqrt(n)
                a = best_a + rng.standard_normal(n) * noise_scale
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve_sweeps = 0
        else:
            no_improve_sweeps = 0

        if verbose and sweep % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD sweep={sweep}, score={best_score:.8f}, elapsed={elapsed:.0f}s")

    return best_a, best_score


def coordinate_descent_single(a0, time_limit, verbose=True):
    """Coordinate descent optimizing one coordinate at a time (transfer to all others)."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    sweep = 0
    while time.monotonic() - t0 < time_limit:
        sweep += 1
        improved = False

        # Random order of coordinates
        order = list(range(n))
        rng.shuffle(order)

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-15:
                continue

            # Try transferring a fraction of a[i] to a uniformly chosen target
            # Actually, try several targets
            n_targets = min(5, n - 1)
            targets = rng.choice([j for j in range(n) if j != i], size=n_targets, replace=False)

            for j in targets:
                if a[j] + a[i] < 1e-15:
                    continue

                t_lo = -a[j] + 1e-15
                t_hi = a[i] - 1e-15
                if t_hi <= t_lo:
                    continue

                x1 = t_hi - gr * (t_hi - t_lo)
                x2 = t_lo + gr * (t_hi - t_lo)

                def eval_t_single(t_val):
                    a[i] -= t_val
                    a[j] += t_val
                    nn = max(a[i], 0.0)
                    a[i] = nn
                    sc = compute_score(a)
                    a[j] -= t_val
                    a[i] += t_val
                    return sc

                f1 = eval_t_single(x1)
                f2 = eval_t_single(x2)

                for _ in range(18):
                    if f1 < f2:
                        t_hi = x2
                        x2 = x1
                        f2 = f1
                        x1 = t_hi - gr * (t_hi - t_lo)
                        f1 = eval_t_single(x1)
                    else:
                        t_lo = x1
                        x1 = x2
                        f1 = f2
                        x2 = t_lo + gr * (t_hi - t_lo)
                        f2 = eval_t_single(x2)

                t_opt = (t_lo + t_hi) / 2.0
                f_opt = min(f1, f2)

                if f_opt < best_score - 1e-13:
                    a[i] -= t_opt
                    a[j] += t_opt
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    best_score = compute_score(a)
                    best_a = a.copy()
                    improved = True
                    break  # Move to next coordinate i

        if verbose and sweep % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD1 sweep={sweep}, score={best_score:.8f}, elapsed={elapsed:.0f}s")

    return best_a, best_score


def lp_direction_search(a0, time_limit, verbose=True):
    """LP-based direction finding followed by line search."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(55)

    iters = 0
    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        M = conv.max()

        # Build LP: max sum(g) s.t. conv(a, g) <= M, g >= 0
        # conv(a, g)[k] = sum_j a[j]*g[k-j] <= M for all k
        # This is: C @ g <= M * 1, where C[k,j] = a[k-j] if valid
        # For efficiency, use a sparse/diagonal structure
        C = np.zeros((2 * n - 1, n))
        for j in range(n):
            k_start = j
            k_end = min(j + n - 1, 2 * n - 2)
            C[k_start:k_end + 1, j] = a[:k_end - k_start + 1]

        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        try:
            res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs',
                          options={'time_limit': 5})
            if res.status == 0 and res.x is not None:
                g = np.maximum(res.x, 0)
                g_sum = g.sum()
                if g_sum > 1e-10:
                    g = g / g_sum

                    # Line search: a_new = (1-t)*a + t*g
                    best_t = 0
                    best_f = best_score
                    for t_candidate in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                        a_trial = (1 - t_candidate) * a + t_candidate * g
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        sc = compute_score(a_trial)
                        if sc < best_f:
                            best_f = sc
                            best_t = t_candidate

                    if best_t > 0:
                        a = (1 - best_t) * a + best_t * g
                        a = np.maximum(a, 0)
                        a = a / a.sum()
                        if compute_score(a) < best_score:
                            best_score = compute_score(a)
                            best_a = a.copy()
        except Exception:
            pass

        if verbose and iters % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={iters}, M={M:.6e}, score={best_score:.8f}, elapsed={elapsed:.0f}s")

    return best_a, best_score


def subgradient_projected(a0, time_limit, verbose=True):
    """Projected subgradient descent with Nesterov acceleration."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    a_prev = a.copy()
    y = a.copy()
    eta0 = 1.0 / np.sqrt(n)

    for t in range(5000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov extrapolation
        if t > 0:
            beta = (t - 1) / (t + 2)
            y = a + beta * (a - a_prev)
            y = np.maximum(y, 1e-30)
            y = y / y.sum()
        else:
            y = a.copy()

        # Compute subgradient at y
        c = fast_convolve(y)
        M = c.max()
        # Use soft subgradient
        T_eff = max(0.01 * n, M * 0.05)
        c_shift = c / T_eff
        c_shift -= c_shift.max()
        w = np.exp(c_shift)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break
        w = w / w_sum

        g = 4.0 * n * cross_correlation(w, y)

        # Step
        eta = eta0 / np.sqrt(t + 1)
        a_new = project_simplex(y - eta * g)

        # Check
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        a_prev = a
        a = a_new

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PGD iter={t}, eta={eta:.2e}, score={score:.8f}, best={best_score:.8f}, elapsed={elapsed:.0f}s")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate a candidate and return updated best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-stage optimization:
    1. Entropic mirror descent (fast, good global exploration)
    2. Projected subgradient with Nesterov acceleration
    3. LP direction search
    4. Pair coordinate descent (precise local refinement)
    5. Length exploration
    6. Final polish
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        n_init = 768
        parent = np.ones(n_init) / n_init
        best_seq = parent.copy()
        best_score = compute_score(parent)
        print(f"Initialized uniform: n={n_init}, score={best_score:.8f}")

    n_parent = len(parent)

    # === Phase 1: Mirror Descent (main workhorse) ===
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 500.0)
    print(f"\n=== Phase 1: Mirror Descent (budget={t_alloc:.0f}s) ===")
    a_md, sc_md = mirror_descent(best_seq, t_alloc * 0.6, verbose=True, eta0=0.3 / n_parent)
    best_seq, best_score = check_and_update(a_md, "mirror_desc", best_seq, best_score)

    # Mirror descent with Nesterov on the result
    rem = deadline - time.monotonic()
    if rem > 30:
        t_alloc = min(rem * 0.2, 150.0)
        a_md2, sc_md2 = mirror_descent(best_seq, t_alloc, verbose=True, eta0=0.1 / n_parent, use_nesterov=True)
        best_seq, best_score = check_and_update(a_md2, "mirror_nest", best_seq, best_score)

    # === Phase 2: Projected Subgradient ===
    rem = deadline - time.monotonic()
    if rem > 60:
        t_alloc = min(rem * 0.2, 250.0)
        print(f"\n=== Phase 2: Projected Subgradient (budget={t_alloc:.0f}s) ===")
        a_pg, sc_pg = subgradient_projected(best_seq, t_alloc, verbose=True)
        best_seq, best_score = check_and_update(a_pg, "proj_subgrad", best_seq, best_score)

    # === Phase 3: LP Direction Search ===
    rem = deadline - time.monotonic()
    if rem > 60:
        t_alloc = min(rem * 0.15, 150.0)
        print(f"\n=== Phase 3: LP Direction Search (budget={t_alloc:.0f}s) ===")
        a_lp, sc_lp = lp_direction_search(best_seq, t_alloc, verbose=True)
        best_seq, best_score = check_and_update(a_lp, "lp_dir", best_seq, best_score)

    # === Phase 4: Pair Coordinate Descent ===
    rem = deadline - time.monotonic()
    if rem > 60:
        t_alloc = min(rem * 0.3, 350.0)
        print(f"\n=== Phase 4: Pair Coordinate Descent (budget={t_alloc:.0f}s) ===")
        a_cd, sc_cd = coordinate_descent_pairs(best_seq, t_alloc, verbose=True, max_pairs_per_sweep=300)
        best_seq, best_score = check_and_update(a_cd, "cd_pairs", best_seq, best_score)

    # === Phase 5: Length Exploration ===
    rem = deadline - time.monotonic()
    if rem > 120:
        print(f"\n=== Phase 5: Length Exploration ===")
        # Try nearby and related lengths
        candidate_lengths = []
        for delta in [-2, -1, 1, 2, 4, 8, 16, -16, -32]:
            alt_n = n_parent + delta
            if 128 <= alt_n <= 4096 and alt_n not in candidate_lengths:
                candidate_lengths.append(alt_n)
        # Also try powers of 2 and other "nice" lengths
        for nice_n in [512, 1024, 1536, 2048, 3072, 4096]:
            if nice_n not in candidate_lengths and 128 <= nice_n <= 4096:
                candidate_lengths.append(nice_n)

        for alt_n in candidate_lengths[:8]:  # Limit number of lengths tried
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            t_alloc = min(rem * 0.25, 80.0)
            if t_alloc < 20:
                break

            init_alt = interpolate_sequence(best_seq, alt_n)
            print(f"  Trying n={alt_n}, budget={t_alloc:.0f}s")

            # Quick mirror descent
            a_alt, sc_alt = mirror_descent(init_alt, t_alloc * 0.4, verbose=False)
            best_seq, best_score = check_and_update(a_alt, f"n{alt_n}_md", best_seq, best_score)

            # Quick coordinate descent
            rem2 = deadline - time.monotonic()
            if rem2 > 20:
                a_alt2, sc_alt2 = coordinate_descent_pairs(
                    best_seq if len(best_seq) == alt_n else a_alt,
                    min(rem2 * 0.3, 30.0), verbose=False, max_pairs_per_sweep=100)
                best_seq, best_score = check_and_update(a_alt2, f"n{alt_n}_cd", best_seq, best_score)

    # === Phase 6: Final Polish ===
    rem = deadline - time.monotonic()
    if rem > 30:
        t_alloc = min(rem * 0.5, 100.0)
        print(f"\n=== Phase 6: Final Coordinate Descent Polish (budget={t_alloc:.0f}s) ===")
        a_fin, sc_fin = coordinate_descent_pairs(best_seq, t_alloc, verbose=True, max_pairs_per_sweep=500)
        best_seq, best_score = check_and_update(a_fin, "final_cd", best_seq, best_score)

    # === Phase 7: Very fine polish ===
    rem = deadline - time.monotonic()
    if rem > 20:
        t_alloc = min(rem * 0.7, 50.0)
        print(f"\n=== Phase 7: Fine Polish (budget={t_alloc:.0f}s) ===")
        a_fin2, sc_fin2 = coordinate_descent_single(best_seq, t_alloc, verbose=False)
        best_seq, best_score = check_and_update(a_fin2, "fine_cd", best_seq, best_score)

    # Fallback
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
