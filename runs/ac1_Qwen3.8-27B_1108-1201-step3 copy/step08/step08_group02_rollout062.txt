import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def get_score(f):
    """Compute 2*n*max_conv (assuming sum(f)=1)."""
    n = len(f)
    b = fast_convolve(f)
    return 2.0 * n * b.max()

def run_lp_bisection(f_init, time_limit, label="lp"):

            A[row_i, lo:hi + 1] = f[lo:hi + 1]

import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def normalize(f):
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-10:
        return np.ones(len(f)) / len(f)
    return f / s


def project_simplex(v, target_sum=1.0):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient(b, w, f, top_idx):
    """Compute gradient of sum_k w[k]*conv(f,f)[k] w.r.t. f using top peaks."""
    n = len(f)
    grad = np.zeros(n)
    for k in top_idx:
        wk = w[k]
        if wk < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        j_range = np.arange(lo, hi + 1)
        grad[j_range] += 2.0 * wk * f[k - j_range]
    return grad

    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    return normalize(f_new)


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. Generate diverse initializations (asymmetric variants of parent)
    2. Fast gradient optimization on each
    3. LP refinement on best
    4. Coordinate descent polish
    5. Multi-length exploration
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        n_default = 1024
        parent = np.ones(n_default) / n_default
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default uniform n={n_default}, score={p_score:.6f}")

    best_seq = parent.copy()
    best_score = p_score

    n = len(parent)

    # Generate diverse initializations
    inits = []
    inits.append(("parent", parent.copy()))

    # Asymmetric: multiply by exponential decay
    ramp = np.exp(np.linspace(0, -2.0, n))
    inits.append(("exp_decay", normalize(parent * ramp)))

    # Asymmetric: linear ramp
    lin = np.linspace(0.3, 1.7, n)
    inits.append(("linear", normalize(parent * lin)))

    # Asymmetric: quadratic
    quad = np.linspace(0.1, 2.5, n) ** 2
    inits.append(("quadratic", normalize(parent * quad)))

    # Random non-negative
    for ri in range(3):
        r = np.abs(rng.standard_normal(n))
        inits.append((f"rand{ri}", normalize(r)))

    # Smooth random (random + Gaussian filter)
    from scipy.ndimage import gaussian_filter1d
    for ri in range(2):
        r = np.abs(rng.standard_normal(n))
        r_smooth = gaussian_filter1d(r, sigma=n / 20.0)
        inits.append((f"smrand{ri}", normalize(r_smooth)))

    # Bimodal
    two_bumps = np.exp(-((np.arange(n) - n * 0.2) / (n * 0.1)) ** 2) + \
                np.exp(-((np.arange(n) - n * 0.8) / (n * 0.1)) ** 2)
    inits.append(("bimodal", normalize(parent * two_bumps + 0.1 * two_bumps)))

    # Phase 1: Fast gradient optimization on all initializations
    print(f"\n=== Phase 1: Gradient optimization (n={n}) ===")
    grad_results = []
    n_inits = len(inits)
    # Allocate time: give more time to parent, less to others
    time_per_init = {}
    time_per_init["parent"] = min(300.0, (deadline - time.monotonic()) * 0.35)
    remaining = (deadline - time.monotonic()) - time_per_init["parent"]
    others = [name for name, _ in inits if name != "parent"]
    time_per_other = max(30.0, remaining * 0.4 / max(1, len(others)))

    for name, f_init in inits:
        rem = deadline - time.monotonic()
        if rem < 20:
            break
        if name == "parent":
            t_budget = min(time_per_init[name], rem * 0.5)
        else:
            t_budget = min(time_per_other, rem * 0.3)
        if t_budget < 15:
            t_budget = max(15, t_budget)
            if rem < 15:
                break

        print(f"  Initializing: {name}...")
        f_opt, obj = gradient_optimize(f_init, time_limit=t_budget, label=name,
                                        top_k=min(200, n), lr_init=0.008)
        score = 2.0 * n * obj
        vs = float(evaluate_sequence(f_opt.tolist()))
        print(f"  {name}: M={obj:.10f} score={vs:.6f}")
        grad_results.append((name, f_opt, obj, vs))
        if np.isfinite(vs) and vs < best_score:
            best_score = vs
            best_seq = f_opt.copy()
            print(f"  *** NEW BEST: {vs:.6f} from {name} ***")

    # Phase 2: LP refinement on top gradient results
    print(f"\n=== Phase 2: LP refinement ===")
    grad_results.sort(key=lambda x: x[3] if np.isfinite(x[3]) else 1e10)
    rem = deadline - time.monotonic()
    top_k_results = min(3, len(grad_results))
    for i in range(top_k_results):
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        name, f_g, obj_g, vs_g = grad_results[i]
        t_budget = min(200.0, rem * 0.25)
        if t_budget < 20:
            break
        print(f"  LP refining {name} (budget={t_budget:.0f}s)...")
        f_lp, obj_lp = lp_refine(f_g, time_limit=t_budget, label=f"lp_{name}")
        score_lp = 2.0 * n * obj_lp
        vs_lp = float(evaluate_sequence(f_lp.tolist()))
        print(f"  lp_{name}: M={obj_lp:.10f} score={vs_lp:.6f}")
        if np.isfinite(vs_lp) and vs_lp < best_score:
            best_score = vs_lp
            best_seq = f_lp.copy()
            print(f"  *** NEW BEST: {vs_lp:.6f} from lp_{name} ***")

    # Phase 3: More gradient refinement on best
    rem = deadline - time.monotonic()
    if rem > 60:
        t3 = min(200.0, rem * 0.25)
        print(f"\n=== Phase 3: Gradient on best (n={len(best_seq)}, budget={t3:.0f}s) ===")
        f3, obj3 = gradient_optimize(best_seq, time_limit=t3, label="grad_best",
                                      top_k=min(300, len(best_seq)), lr_init=0.005)
        vs3 = float(evaluate_sequence(f3.tolist()))
        print(f"  grad_best: M={obj3:.10f} score={vs3:.6f}")
        if np.isfinite(vs3) and vs3 < best_score:
            best_score = vs3
            best_seq = f3.copy()
            print(f"  *** NEW BEST: {vs3:.6f} ***")

    # Phase 4: Multi-length exploration
    print(f"\n=== Phase 4: Multi-length exploration ===")
    n_base = len(best_seq)
    target_lengths = set()
    for factor in [0.8, 1.2, 1.5, 0.7, 1.8]:
        tl = int(n_base * factor)
        tl = max(512, min(3000, tl))
        if tl != n_base:
            target_lengths.add(tl)
    target_lengths = sorted(target_lengths)

    for target_n in target_lengths[:4]:
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        f_interp = interpolate_to_length(best_seq, target_n)
        t4 = min(120.0, rem * 0.15)
        if t4 < 20:
            break
        print(f"  Exploring n={target_n} (budget={t4:.0f}s)...")
        f4, obj4 = gradient_optimize(f_interp, time_limit=t4, label=f"n{target_n}",
                                      top_k=min(200, target_n), lr_init=0.006)
        vs4 = float(evaluate_sequence(f4.tolist()))
        print(f"  n={target_n}: score={vs4:.6f}")
        if np.isfinite(vs4) and vs4 < best_score:
            best_score = vs4
            best_seq = f4.copy()
            print(f"  *** NEW BEST: {vs4:.6f} at n={target_n} ***")

    # Phase 5: Coordinate descent polish
    rem = deadline - time.monotonic()
    if rem > 20:
        t5 = min(100.0, rem * 0.25)
        print(f"\n=== Phase 5: Coordinate descent (budget={t5:.0f}s) ===")
        f5, obj5 = coordinate_descent(best_seq, time_limit=t5, label="cd")
        vs5 = float(evaluate_sequence(f5.tolist()))
        print(f"  cd: score={vs5:.6f}")
        if np.isfinite(vs5) and vs5 < best_score:
            best_score = vs5
            best_seq = f5.copy()
            print(f"  *** NEW BEST: {vs5:.6f} ***")

    # Phase 6: Final LP burst
    rem = deadline - time.monotonic()
    if rem > 30:
        t6 = min(100.0, rem * 0.3)
        print(f"\n=== Phase 6: Final LP (budget={t6:.0f}s) ===")
        f6, obj6 = lp_refine(best_seq, time_limit=t6, label="lp_final")
        vs6 = float(evaluate_sequence(f6.tolist()))
        print(f"  lp_final: score={vs6:.6f}")
        if np.isfinite(vs6) and vs6 < best_score:
            best_score = vs6
            best_seq = f6.copy()
            print(f"  *** NEW BEST: {vs6:.6f} ***")

    # Phase 7: Final CD
    rem = deadline - time.monotonic()
    if rem > 15:
        t7 = min(60.0, rem * 0.4)
        print(f"\n=== Phase 7: Final CD (budget={t7:.0f}s) ===")
        f7, obj7 = coordinate_descent(best_seq, time_limit=t7, label="cd_final")
        vs7 = float(evaluate_sequence(f7.tolist()))
        if np.isfinite(vs7) and vs7 < best_score:
            best_score = vs7
            best_seq = f7.copy()
            print(f"  *** NEW BEST: {vs7:.6f} ***")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
