```python
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    """Score for normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def smooth_max_grad(a, T):
    """Compute smooth max of conv(a) and its gradient w.r.t. a."""
    n = len(a)
    conv = fast_convolve(a)
    # Soft max: T * log(sum(exp(conv/T)))
    conv_scaled = conv / T
    max_c = conv_scaled.max()
    exp_c = np.exp(conv_scaled - max_c)
    w = exp_c / exp_c.sum()
    smooth_max = T * (max_c + np.log(exp_c.sum()))
    
    # Gradient: w[k] * ∂conv[k]/∂a
    # ∂conv[k]/∂a[j] = 2*a[k-j] (when k-j is valid)
    # So gradient[j] = 2 * sum_k w[k] * a[k-j] (where k-j valid)
    # This is 2 * (w * a)[j] in the "valid" range... 
    # Actually it's the cross-correlation of w and a, scaled.
    # (w cross-correlate a)[j] = sum_k w[k] * a[k-j]
    # This can be computed via FFT!
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    # Cross-correlation: ifft(conj(W) * A) gives sum_k w[k] * a[k-j] (with appropriate indexing)
    # Actually: ifft(conj(W)*A)[j] = sum_k w[k] * a[(k-j) mod fft_size]
    # For our purposes (linear, not circular), this works because w has support in [0, 2n-2]
    # and a has support in [0, n-1], so the linear cross-correlation fits.
    grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
    
    return smooth_max, grad_a
```

```python
1. Exponentiated gradient (mirror descent) with log-sum-exp smooth max
2. Multi-temperature annealing
3. Coordinate descent polish with peak-focused pair selection
4. Multi-length exploration
5. L-BFGS final polish

import numpy as np

from scipy.fft import rfft, irfft
from scipy.optimize import minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_score(a):
    """Compute score for normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def smooth_max_and_grad(a, T):
    """
    Compute the log-sum-exp smooth maximum of conv(a) and its gradient.
    Returns (smooth_max_value, gradient_vector).
    """
    n = len(a)
    conv = fast_convolve(a)
    conv_scaled = conv / T
    max_c = conv_scaled.max()
    exp_c = np.exp(conv_scaled - max_c)
    exp_sum = exp_c.sum()
    w = exp_c / exp_sum
    smooth_max = T * (max_c + np.log(exp_sum))
    
    # Gradient: grad[j] = 2 * sum_k w[k] * a[k-j] (valid range)
    # Computed via FFT cross-correlation
    fft_size = 1 << (3 * n - 2).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    grad = 2.0 * irfft(W * np.conj(A), fft_size)[:n]
    
    return smooth_max, grad, conv

def exponentiated_gradient_descent(a0, time_limit, T_schedule=None, eta0=0.1, verbose=True):
    """
    Mirror descent with entropy mirror (exponentiated gradient).
    Minimizes smooth_max(conv(a)) subject to sum(a)=1, a>=0.
    """
    a = a0.copy()
    a = np.maximum(a, 1e-30)
    a = a / a.sum()
    n = len(a)
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    if T_schedule is None:
        T_schedule = [2000, 1000, 500, 200, 100, 50]
    
    max_iter = 500000
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adaptive temperature
        frac = it / max_iter
        T_idx = min(int(frac * len(T_schedule)), len(T_schedule) - 1)
        T = T_schedule[T_idx]
        
        # Adaptive step size
        eta = eta0 / (1.0 + 0.01 * it)
        
        smooth_val, grad, conv = smooth_max_and_grad(a, T)
        
        # Project gradient to tangent space (sum of direction = 0)
        grad_mean = grad.dot(a)  # For entropy mirror, project using a-weighted mean
        # Actually for mirror descent on simplex with entropy: 
        # direction = grad - (sum(a*grad)/sum(a)) = grad - sum(a*grad) since sum(a)=1
        grad_proj = grad - grad_mean
        
        # Multiplicative update
        log_update = -eta * grad_proj
        # Clip for stability
        log_update = np.clip(log_update, -50, 50)
        a_new = a * np.exp(log_update)
        s = a_new.sum()
        if s < 1e-30:
            break
        a_new = a_new / s
        
        # Check improvement
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and (it + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    EG iter={it+1}, T={T}, eta={eta:.6f}, score={best_score:.8f}, elapsed={elapsed:.1f}s")
    
    return best_a, best_score
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Exponentiated gradient (mirror descent) with log-sum-exp smooth max at multiple temperatures
2. Efficient coordinate descent with O(1) per-evaluation cost via precomputed linear/quadratic coefficients
3. Multi-temperature annealing with adaptive step sizes
4. Multi-length exploration and random restarts
5. L-BFGS final polish on smooth approximation
Key innovation: coordinate descent evaluates max(conv + lin*t + quad*t^2) in O(n) without FFT,
enabling many more iterations. Combined with fast mirror descent for global progress.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute score for normalized sequence (sum=1): 2*n*max_conv."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def smooth_max_and_grad(a, T):
    """
    Compute log-sum-exp smooth maximum of conv(a) and its gradient.
    Returns (smooth_max_value, gradient_vector, conv).
    """
    n = len(a)
    conv = fast_convolve(a)
    conv_scaled = conv / T
    max_c = conv_scaled.max()
    exp_c = np.exp(conv_scaled - max_c)
    exp_sum = exp_c.sum()
    w = exp_c / exp_sum
    smooth_max = T * (max_c + np.log(exp_sum))

    # Gradient: grad[j] = 2 * sum_k w[k] * a[k-j] (valid range)
    fft_size = 1 << (3 * n - 2).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    grad = 2.0 * irfft(W * np.conj(A), fft_size)[:n]

    return smooth_max, grad, conv


def exponentiated_gradient_descent(a0, time_limit, verbose=True):
    """
    Mirror descent with entropy mirror (exponentiated gradient).
    Minimizes smooth_max(conv(a)) subject to sum(a)=1, a>=0.
    Uses adaptive temperature schedule.
    """
    a = np.maximum(a0, 1e-30)
    a = a / a.sum()
    n = len(a)

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    # Temperature schedule: start high (smooth), end low (sharp)
    T_schedule = [5000, 3000, 1500, 800, 400, 200, 100, 50, 25, 10]
    # Each T gets a fraction of iterations
    T_durations = [0.10, 0.10, 0.12, 0.12, 0.13, 0.13, 0.13, 0.10, 0.10, 0.07]

    max_total_iter = 200000
    it = 0
    idx = 0

    for idx, T in enumerate(T_schedule):
        n_iters_this = int(max_total_iter * T_durations[idx])
        for _ in range(n_iters_this):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_score

            it += 1
            eta = 0.5 / (1.0 + 0.005 * it)

            smooth_val, grad, conv = smooth_max_and_grad(a, T)

            # Project to tangent space of simplex (a-weighted mean for entropy mirror)
            grad_proj = grad - grad.dot(a)

            # Multiplicative update with clipping
            log_update = -eta * grad_proj
            log_update = np.clip(log_update, -30, 30)
            a_new = a * np.exp(log_update)
            s = a_new.sum()
            if s < 1e-30:
                break
            a_new = a_new / s

            score = compute_score(a_new)
            if score < best_score:
                best_score = score
                best_a = a_new.copy()

            a = a_new

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    EG T={T}, iter={it}, score={best_score:.8f}, elapsed={elapsed:.1f}s")

    return best_a, best_score


def efficient_coordinate_descent(a0, time_limit, n_pairs_per_iter=150, verbose=True):
    """
    Coordinate descent with O(n) per-evaluation cost.
    Precomputes linear and quadratic coefficients for each pair transfer,
    then does golden section search on max(conv + lin*t + quad*t^2).
    """
    n = len(a0)
    a = np.maximum(a0, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    iters = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False

        conv = fast_convolve(a)
        n_entries = 2 * n - 1

        # Select pairs
        pairs = set()

        # Peak-focused pairs
        top_k = min(8, n_entries)
        peak_idxs = np.argsort(conv)[::-1][:top_k]
        for pk in peak_idxs:
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            if je < js:
                continue
            cnt = max(5, n_pairs_per_iter // (top_k * 3))
            for _ in range(cnt):
                i = int(rng.integers(js, je + 1))
                j = int(rng.integers(js, je + 1))
                if i != j:
                    pairs.add((min(i, j), max(i, j)))

        # Random pairs
        for _ in range(n_pairs_per_iter // 2):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                pairs.add((min(i, j), max(i, j)))

        # Edge pairs
        edge = max(3, n // 15)
        for _ in range(n_pairs_per_iter // 4):
            i = int(rng.integers(0, edge))
            j = int(rng.integers(n - edge, n))
            if i != j:
                pairs.add((min(i, j), max(i, j)))

        # Center-heavy pairs (to reduce center peak)
        center_start = n // 3
        center_end = 2 * n // 3
        for _ in range(n_pairs_per_iter // 4):
            i = int(rng.integers(center_start, center_end))
            j = int(rng.integers(center_start, center_end))
            if i != j:
                pairs.add((min(i, j), max(i, j)))

        gr = (np.sqrt(5) - 1) / 2

        for (i, j) in list(pairs):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14 and a[j] < 1e-14:
                continue

            # Precompute linear and quadratic coefficients
            # lin[k] = 2*(a[k-j] - a[k-i]) for valid k
            lin = np.zeros(n_entries)
            # a[k-i] is valid for k in [i, i+n-1], value is a[k-i]
            k1_s = i
            k1_e = min(n_entries - 1, i + n - 1)
            lin[k1_s:k1_e + 1] -= 2.0 * a[0:(k1_e - k1_s + 1)]
            # a[k-j] is valid for k in [j, j+n-1], value is a[k-j]
            k2_s = j
            k2_e = min(n_entries - 1, j + n - 1)
            lin[k2_s:k2_e + 1] += 2.0 * a[0:(k2_e - k2_s + 1)]

            # Quadratic
            quad = np.zeros(n_entries)
            ki = 2 * i
            kj = i + j
            kj2 = 2 * j
            if ki < n_entries:
                quad[ki] += 1.0
            if kj < n_entries:
                quad[kj] -= 2.0
            if kj2 < n_entries:
                quad[kj2] += 1.0

            # Golden section search on [t_lo, t_hi]
            t_lo = -a[j] + 1e-15
            t_hi = a[i] - 1e-15
            if t_hi <= t_lo:
                continue

            x1 = t_hi - gr * (t_hi - t_lo)
            x2 = t_lo + gr * (t_hi - t_lo)

            # Evaluate: max of conv + lin*t + quad*t^2
            vals1 = conv + lin * x1 + quad * x1 * x1
            f1 = vals1.max()
            vals2 = conv + lin * x2 + quad * x2 * x2
            f2 = vals2.max()

            for _ in range(20):
                if time.monotonic() - t0 > time_limit:
                    break
                if f1 < f2:
                    t_hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = t_hi - gr * (t_hi - t_lo)
                    vals1 = conv + lin * x1 + quad * x1 * x1
                    f1 = vals1.max()
                else:
                    t_lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = t_lo + gr * (t_hi - t_lo)
                    vals2 = conv + lin * x2 + quad * x2 * x2
                    f2 = vals2.max()

            t_opt = (t_lo + t_hi) / 2
            # Clamp
            t_opt = max(t_lo, min(t_hi, t_opt))

            # Apply
            a[i] -= t_opt
            a[j] += t_opt
            a = np.maximum(a, 0)
            ss = a.sum()
            if ss < 1e-15:
                a = best_a.copy()
                continue
            a = a / ss

            score = compute_score(a)
            if score < best_score - 1e-13:
                best_score = score
                best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 15:
                # Perturb to escape local minimum
                noise_scale = 0.002 * (1.0 + no_improve * 0.1)
                noise = rng.standard_normal(n) * noise_scale
                a = best_a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
        else:
            no_improve = 0

        if verbose and iters % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD iter={iters}, score={best_score:.8f}, elapsed={elapsed:.1f}s")

    return best_a, best_score


def lbgfs_smooth(a0, time_limit, T=300, maxiter=3000):
    """L-BFGS on smooth max in log-space."""
    n = len(a0)

    def softmax_from_z(z):
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        s = exp_z.sum()
        if s < 1e-300:
            return np.ones(n) / n
        return exp_z / s

    a0_safe = np.maximum(a0, 1e-30)
    z0 = np.log(a0_safe)
    z0 -= z0.mean()  # center for numerical stability

    best_a = a0.copy()
    best_score = compute_score(a0)
    t0 = time.monotonic()

    def func_and_grad(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15, np.zeros(n)
        a_tmp = softmax_from_z(z)
        smooth_val, grad_a, _ = smooth_max_and_grad(a_tmp, T)
        # Project to log-space: grad_z = a * (grad_a - dot(a, grad_a))
        S = grad_a.dot(a_tmp)
        grad_z = a_tmp * (grad_a - S)
        return 2.0 * n * smooth_val, 2.0 * n * grad_z

    def func(z):
        val, _ = func_and_grad(z)
        return val

    def jac(z):
        _, g = func_and_grad(z)
        return g

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10,
                     'maxfun': maxiter * 30, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
        score = compute_score(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except Exception:
        pass

    return best_a, best_score

        print(f"    RR: initial score={best_score:.8f}, per_restart_budget={per_restart:.1f}s")

    a_curr, sc_curr = exponentiated_gradient_descent(best_a, per_restart * 0.7, verbose=False)
    if sc_curr < best_score:
        best_score = sc_curr
        best_a = a_curr.copy()

    a_curr, sc_curr = efficient_coordinate_descent(best_a, per_restart * 0.3, verbose=False)
    if sc_curr < best_score:
        best_score = sc_curr
        best_a = a_curr.copy()

    # Then: random restarts
    for r in range(n_restarts):
        if time.monotonic() - t0 > time_limit - 20:
            break

        # Generate perturbation
        scale = 0.05 / (1 + r * 0.5)
        noise = rng.standard_normal(n) * scale
        a_pert = best_a + noise
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()

        per_r = (time_limit - (time.monotonic() - t0) - 20) / (n_restarts - r)
        eg_budget = min(per_r * 0.5, 100.0)
        cd_budget = min(per_r * 0.4, 80.0)

        a_r, sc_r = exponentiated_gradient_descent(a_pert, eg_budget, verbose=False)
        if sc_r < best_score:
            best_score = sc_r
            best_a = a_r.copy()

        a_r, sc_r = efficient_coordinate_descent(best_a, cd_budget, verbose=False)
        if sc_r < best_score:
            best_score = sc_r
            best_a = a_r.copy()

        if verbose:
            print(f"    RR restart {r+1}: score={best_score:.8f}, elapsed={time.monotonic()-t0:.1f}s")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score):
    """Evaluate and update best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. Load parent or initialize
    2. Exponentiated gradient descent (main workhorse)
    3. Efficient coordinate descent (polish)
    4. L-BFGS smooth optimization
    5. Randomized restarts
    6. Length exploration
    7. Final polish
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        n_init = 768
        parent = np.ones(n_init) / n_init
        # Add slight structure
        t_arr = np.linspace(0, 1, n_init)
        parent = parent * (1 + 0.1 * np.sin(4 * np.pi * t_arr))
        parent = np.maximum(parent, 0)
        parent = parent / parent.sum()
        best_seq = parent.copy()
        best_score = compute_score(parent)
        print(f"Initialized: n={n_init}, score={best_score:.8f}")

    n_parent = len(parent)
    a = parent.copy()

    # === Phase 1: Exponentiated Gradient Descent ===
    rem = deadline - time.monotonic() - 600
    if rem > 30:
        t_alloc = min(rem * 0.3, 300.0)
        print(f"\n=== Phase 1: EG Descent (budget={t_alloc:.0f}s, best={best_score:.8f}) ===")
        a_eg, sc_eg = exponentiated_gradient_descent(a, t_alloc, verbose=True)
        best_seq, best_score = check_and_update(a_eg, "EG", best_seq, best_score)
        a = best_seq.copy()

    # === Phase 2: Efficient Coordinate Descent ===
    rem = deadline - time.monotonic() - 400
    if rem > 20:
        t_alloc = min(rem * 0.3, 200.0)
        print(f"\n=== Phase 2: Coord Descent (budget={t_alloc:.0f}s, best={best_score:.8f}) ===")
        a_cd, sc_cd = efficient_coordinate_descent(a, t_alloc, n_pairs_per_iter=200, verbose=True)
        best_seq, best_score = check_and_update(a_cd, "CD", best_seq, best_score)
        a = best_seq.copy()

    # === Phase 3: L-BFGS Smooth ===
    rem = deadline - time.monotonic() - 300
    if rem > 20:
        t_alloc = min(rem * 0.2, 100.0)
        print(f"\n=== Phase 3: L-BFGS (budget={t_alloc:.0f}s, best={best_score:.8f}) ===")
        for T_val in [500, 200]:
            rem2 = deadline - time.monotonic() - 200
            if rem2 < 15:
                break
            t_a = min(rem2 * 0.4, 60.0)
            a_lb, sc_lb = lbgfs_smooth(a, t_a, T=T_val, maxiter=2000)
            best_seq, best_score = check_and_update(a_lb, f"LBF_T{T_val}", best_seq, best_score)
            a = best_seq.copy()

    # === Phase 4: Second round EG + CD ===
    rem = deadline - time.monotonic() - 250
    if rem > 30:
        t_alloc = min(rem * 0.3, 150.0)
        print(f"\n=== Phase 4: EG+CD Round 2 (budget={t_alloc:.0f}s, best={best_score:.8f}) ===")
        a_eg2, sc_eg2 = exponentiated_gradient_descent(a, t_alloc * 0.6, verbose=False)
        best_seq, best_score = check_and_update(a_eg2, "EG2", best_seq, best_score)
        a = best_seq.copy()

        rem2 = deadline - time.monotonic() - 150
        if rem2 > 15:
            t_a2 = min(rem2 * 0.5, 100.0)
            a_cd2, sc_cd2 = efficient_coordinate_descent(a, t_a2, n_pairs_per_iter=250, verbose=True)
            best_seq, best_score = check_and_update(a_cd2, "CD2", best_seq, best_score)
            a = best_seq.copy()

    # === Phase 5: Randomized Restarts ===
    rem = deadline - time.monotonic() - 150
    if rem > 30:
        t_alloc = min(rem * 0.4, 150.0)
        print(f"\n=== Phase 5: Random Restarts (budget={t_alloc:.0f}s, best={best_score:.8f}) ===")
        a_rr, sc_rr = randomized_restart_optimize(a, t_alloc, n_restarts=3, verbose=True)
        best_seq, best_score = check_and_update(a_rr, "RR", best_seq, best_score)
        a = best_seq.copy()

    # === Phase 6: Length Exploration ===
    rem = deadline - time.monotonic() - 100
    if rem > 30:
        print(f"\n=== Phase 6: Length Exploration (best={best_score:.8f}) ===")
        target_lengths = [n_parent, n_parent + 128, n_parent - 128, 1024, 512]
        # Remove duplicates and invalid
        seen = set()
        valid_lengths = []
        for ln in target_lengths:
            if ln < 64 or ln > 4096 or ln in seen:
                continue
            seen.add(ln)
            valid_lengths.append(ln)

        for alt_n in valid_lengths:
            rem2 = deadline - time.monotonic() - 80
            if rem2 < 20:
                break
            if alt_n == n_parent:
                continue  # Already at this length

            t_a = min(rem2 * 0.3, 60.0)
            if t_a < 15:
                continue

            init_alt = interpolate_sequence(best_seq, alt_n)
            print(f"  Trying n={alt_n}, budget={t_a:.0f}s")

            # Quick EG
            a_alt, sc_alt = exponentiated_gradient_descent(init_alt, t_a * 0.5, verbose=False)
            best_seq, best_score = check_and_update(a_alt, f"n{alt_n}_EG", best_seq, best_score)

            rem3 = deadline - time.monotonic() - 60
            if rem3 > 10:
                a_alt2, sc_alt2 = efficient_coordinate_descent(a_alt, min(rem3 * 0.5, 30.0),
                                                              n_pairs_per_iter=100, verbose=False)
                best_seq, best_score = check_and_update(a_alt2, f"n{alt_n}_CD", best_seq, best_score)

    # === Phase 7: Final Polish ===
    rem = deadline - time.monotonic() - 30
    if rem > 10:
        t_alloc = min(rem * 0.6, 60.0)
        print(f"\n=== Phase 7: Final Polish (budget={t_alloc:.0f}s, best={best_score:.8f}) ===")
        a_fin, sc_fin = efficient_coordinate_descent(best_seq, t_alloc, n_pairs_per_iter=300, verbose=False)
        best_seq, best_score = check_and_update(a_fin, "Final_CD", best_seq, best_score)

    # === Phase 8: Very final L-BFGS ===
    rem = deadline - time.monotonic() - 10
    if rem > 5:
        t_alloc = min(rem * 0.8, 20.0)
        a_fin2, sc_fin2 = lbgfs_smooth(best_seq, t_alloc, T=100, maxiter=1000)
        best_seq, best_score = check_and_update(a_fin2, "Final_LBF", best_seq, best_score)

    # Fallback
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        if fs < best_score:
            best_score = fs
        else:
            # Use the actual evaluated score
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
