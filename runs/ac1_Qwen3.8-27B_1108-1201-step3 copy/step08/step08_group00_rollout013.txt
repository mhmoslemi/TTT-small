import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    solved = False
    for s in [solver] + [s2 for s2 in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL] if s2 != solver]:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solved = True
                break
        except Exception:
            continue
    
    if not solved:
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def lbgfs_refine(a0, time_limit, T_start=100.0, T_end=0.5, maxiter=8000):
    """L-BFGS with LogSumExp smoothing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    
    def make_func(T_val):
        def func(z_val):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z_val)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func
    
    def make_jac(T_val):
        def jac(z_val):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_val)
            a = softmax_from_z(z_val)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac
    
    n_stages = 7
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(50, maxiter // n_stages)
    
    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxfun': iters_per_stage * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass
    
    return softmax_from_z(z)

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    tried_solvers = [solver]
    for s2 in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        if s2 not in tried_solvers:
            tried_solvers.append(s2)

    solved = False
    for s in tried_solvers:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                solved = True
                break
        except Exception:
            continue

    if not solved:
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def lbgfs_refine(a0, time_limit, T_start=100.0, T_end=0.5, maxiter=8000):
    """L-BFGS with LogSumExp smoothing and geometric temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    def make_func(T_val):
        def func(z_val):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z_val)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(z_val):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_val)
            a = softmax_from_z(z_val)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    n_stages = 7
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(50, maxiter // n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxfun': iters_per_stage * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)

def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    else:
        f_interp = np.ones(new_n) / new_n
    return f_interp


def structured_init(n, rng):

    elif choice == 2:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 3:
        xx = np.linspace(-1, 1, n)
        a = np.exp(-xx ** 2 * 3)
    elif choice == 4:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.05)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:

        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n_parent = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Rapid length scan ===
    print(f"\n{'='*60}")
    print(f"Phase 1: Rapid Length Scan")
    print(f"{'='*60}")

    # Generate candidate lengths around parent
    candidate_lengths = set()
    for mult in [0.67, 0.75, 0.83, 0.9, 1.0, 1.1, 1.2, 1.33, 1.5, 1.67]:
        cl = int(n_parent * mult)
        if 384 <= cl <= 2560:
            candidate_lengths.add(cl)
    # Add some standard lengths
    for std_n in [512, 640, 768, 832, 896, 960, 1024, 1088, 1152, 1216, 1280, 1408, 1536]:
        if 384 <= std_n <= 2560:
            candidate_lengths.add(std_n)

    candidate_lengths = sorted(candidate_lengths)
    # Limit to at most 10 for the scan
    if len(candidate_lengths) > 10:
        # Prefer lengths near the parent
        scored = sorted(candidate_lengths, key=lambda x: abs(x - n_parent))
        candidate_lengths = scored[:10]

    t_per_scan = 20  # seconds per scan entry
    scan_results = []

    for cand_n in candidate_lengths:
        rem = deadline - time.monotonic()
        if rem < 80:
            break

        t_scan = min(t_per_scan, rem * 0.1)
        if t_scan < 5:
            break

        # Try two initializations: interpolated parent and structured
        init1 = interpolate_sequence(best_seq, cand_n)
        init2 = structured_init(cand_n, rng)

        a1, s1 = nesterov_lp_refine(init1, t_scan * 0.6, solver=solver, n_max_iters=3000)
        a2, s2 = nesterov_lp_refine(init2, t_scan * 0.4, solver=solver, n_max_iters=2000)

        # Pick the better one
        if s1 <= s2:
            best_at_len = a1
            best_sc_len = s1
        else:
            best_at_len = a2
            best_sc_len = s2

        eval_sc = evaluate_safe(best_at_len)
        scan_results.append((cand_n, eval_sc, best_at_len))
        print(f"  n={cand_n}: score={eval_sc:.6f}")

    # Sort by actual evaluated score
    scan_results.sort(key=lambda x: x[1])
    print(f"\n  Scan ranking:")
    for i, (cn, cs, _) in enumerate(scan_results):
        print(f"    {i+1}. n={cn}: {cs:.6f}")

    # Pick top 3 for deep optimization
    top_entries = scan_results[:3]

    # === Phase 2: Deep optimization on top lengths ===
    print(f"\n{'='*60}")
    print(f"Phase 2: Deep Optimization on Top Lengths")
    print(f"{'='*60}")

    deep_budgets = [0.40, 0.30, 0.18]  # fractions of remaining budget

    for rank, (deep_n, deep_init_sc, deep_init) in enumerate(top_entries):
        rem = deadline - time.monotonic()
        if rem < 80:
            break

        # Use the deep_init from scan as starting point, but also try from best_seq interpolation
        init_from_best = interpolate_sequence(best_seq, deep_n)
        # Use whichever has better score
        sc_from_best = evaluate_safe(init_from_best)
        if sc_from_best <= deep_init_sc:
            deep_start = init_from_best
        else:
            deep_start = deep_init

        frac = deep_budgets[rank] if rank < len(deep_budgets) else 0.10
        t_deep = min(rem * frac, 600.0)
        if t_deep < 40:
            t_deep = rem * 0.6
            if t_deep < 40:
                break

        print(f"\n  Deep optimizing n={deep_n} (budget={t_deep:.0f}s, init_score={evaluate_safe(deep_start):.6f})")
        a_deep, sc_deep = deep_optimize(deep_start, t_deep, solver, label=f"n{deep_n}")
        try_update(a_deep, f"deep_n{deep_n}")

    # === Phase 3: Perturbation escape ===
    print(f"\n{'='*60}")
    print(f"Phase 3: Perturbation Escape")
    print(f"{'='*60}")

    rem = deadline - time.monotonic()
    if rem > 80:
        n_cur = len(best_seq)
        t_pert = min(rem * 0.25, 200.0)

        # Try several structured perturbations
        best_pert_a = best_seq.copy()
        best_pert_sc = best_score

        for pert_type in range(4):
            rem2 = deadline - time.monotonic()
            if rem2 < 30:
                break
            t_pert_one = min(t_pert / (4 - pert_type), rem2 * 0.35)
            if t_pert_one < 15:
                break

            if pert_type == 0:
                # Random perturbation
                perturbed = best_seq + 0.04 * rng.standard_normal(n_cur)
            elif pert_type == 1:
                # Sinusoidal perturbation
                x = np.linspace(0, 1, n_cur)
                perturbed = best_seq * (1.0 + 0.08 * np.sin(2 * np.pi * rng.uniform(2, 8) * x))
            elif pert_type == 2:
                # Edge boost
                perturbed = best_seq.copy()
                perturbed[:n_cur // 8] *= 1.1
                perturbed[-n_cur // 8:] *= 1.1
            else:
                # Center boost
                perturbed = best_seq.copy()
                mid = n_cur // 2
                w = n_cur // 6
                perturbed[mid - w:mid + w] *= 1.08

            perturbed = np.maximum(perturbed, 0)
            perturbed = perturbed / perturbed.sum()

            # Quick LP optimization on perturbed
            a_pt, _ = nesterov_lp_refine(perturbed, t_pert_one * 0.6, solver=solver, n_max_iters=10000)
            sc_pt = evaluate_safe(a_pt)

            # L-BFGS refinement
            rem3 = deadline - time.monotonic()
            if rem3 > 20 and sc_pt < best_pert_sc + 0.01:
                t_lb_pt = min(rem3 * 0.3, 60.0)
                a_pt2 = lbgfs_refine(a_pt, t_lb_pt, T_start=50.0, T_end=0.5, maxiter=5000)
                sc_pt2 = evaluate_safe(a_pt2)
                if sc_pt2 < sc_pt:
                    a_pt = a_pt2
                    sc_pt = sc_pt2

            if sc_pt < best_pert_sc:
                best_pert_sc = sc_pt
                try_update(a_pt, f"pert_{pert_type}")

    # === Phase 4: Final polish on overall best ===
    print(f"\n{'='*60}")
    print(f"Phase 4: Final Polish (current best={best_score:.6f}, n={len(best_seq)})")
    print(f"{'='*60}")

    rem = deadline - time.monotonic()
    if rem > 30:
        # Peak shaving
        t_ps = min(rem * 0.3, 60.0)
        if t_ps > 10:
            a_ps, _ = peak_shave_local(best_seq, t_ps, n_steps=20000)
            try_update(a_ps, "final_ps")

        rem = deadline - time.monotonic()
        if rem > 20:
            # L-BFGS low temperature
            t_lb = min(rem * 0.35, 50.0)
            if t_lb > 10:
                a_fin = lbgfs_refine(best_seq, t_lb, T_start=5.0, T_end=0.05, maxiter=8000)
                try_update(a_fin, "final_lbgfs")

        rem = deadline - time.monotonic()
        if rem > 15:
            # Quick LP
            t_lp = min(rem * 0.4, 30.0)
            if t_lp > 5:
                a_lp_fin, _ = nesterov_lp_refine(best_seq, t_lp, solver=solver, n_max_iters=5000)
                try_update(a_lp_fin, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'='*60}")
    return [float(x) for x in best_seq]
