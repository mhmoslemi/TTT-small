import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    return e / e.sum()

def solve_lp_best_response(a_fixed, solver=None):

def logsumexp_objective(x, tau):
    """Compute the log-sum-exp of the auto-convolution."""
    a = softmax(x)
    b = fast_convolve(a)
    # Return tau * log(sum(exp(b/tau)))
    b_shifted = b / tau
    b_shifted -= b_shifted.max()
    val = tau * (b_shifted.max() + np.log(np.exp(b_shifted).sum()))
    return val


def logsumexp_gradient(x, tau):
    """Compute the gradient of the log-sum-exp objective w.r.t. x."""
    n = len(x)
    a = softmax(x)
    b = fast_convolve(a)
    
    # p_k = exp(b[k]/tau) / sum_j exp(b[j]/tau)
    b_scaled = b / tau
    b_scaled -= b_scaled.max()
    p = np.exp(b_scaled)
    p /= p.sum()
    
    # grad_a[i] = (2/tau) * sum_k p[k] * a[k-i] (where 0 <= k-i < n)
    # This is (2/tau) * cross_correlation(p, a)[i]

    # = conv(p, a_flipped)[i + n - 1]
    a_flipped = a[::-1]
    # conv(p, a_flipped) has length (2n-1) + n - 1 = 3n-2

                    changes[idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end+1] += changes
                M = max(unaff_max, float(conv[k_start:k_end+1].max()))

                cur_score = 2.0 * n * M / (s * s)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 10:
                break
        else:
            no_improve = 0

        if verbose and sweeps % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD {label} sweep={sweeps} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def fictitious_play_improved(n, time_limit, initial=None, solver=None, verbose=False):
    """Improved fictitious play with averaging and damped updates."""
    if solver is None:
        solver = get_lp_solver()
    t0 = time.monotonic()
    
    if initial is None:
        a = np.ones(n) / n
    else:
        a = np.maximum(np.asarray(initial, dtype=np.float64), 0)
        s = a.sum()
        a = a / s

    b = a.copy()
    a_running = np.zeros(n)
    b_running = np.zeros(n)
    best_a = a.copy()
    best_score = float('inf')
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        
        # Best response for b given a
        b_new, _ = solve_lp_best_response(a, solver)
        if b_new is None:
            b_new = b
        # Best response for a given b
        a_new, _ = solve_lp_best_response(b_new, solver)
        if a_new is None:
            a_new = a
        
        a = a_new
        b = b_new
        
        # Running average
        a_running = (iters * a_running + a) / (iters + 1)
        b_running = (iters * b_running + b) / (iters + 1)
        
        # Evaluate candidates
        for cand in [a, b, a_running, b_running, 0.5*(a+b), a_running + 0.1*(b_running - a_running)]:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-10:
                continue
            cand = cand / sc
            score = 2.0 * n * fast_convolve(cand).max()
            if score < best_score:
                best_score = score
                best_a = cand.copy()
        
        if verbose and iters % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FP n={n} it={iters} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score, iters


def subgradient_descent(a0, time_limit, lr_init=0.1, verbose=False, label=""):
    """Subgradient descent for the max of auto-conv on the simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(a)
        M = b.max()
        score = 2.0 * n * M
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 50:
            lr *= 0.8
            no_improve = 0
            if lr < 1e-8:
                break
        
        # Compute subgradient
        # Find peaks
        n_peaks = min(20, 2*n-1)
        peaks = np.argpartition(b, -n_peaks)[-n_peaks:]
        # Soft weights
        temp = 0.02 * M
        w = np.exp(b[peaks] / temp)
        w /= w.sum()
        
        grad = np.zeros(n)
        for idx, k in enumerate(peaks):
            if w[idx] < 1e-14:
                continue
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            # grad[i] += 2 * w[idx] * a[k-i] for i in [i_start, i_end]
            # a[k-i] for i in [i_start, i_end] is a[k-i_end : k-i_start+1] reversed
            # = a[i_end-k+i_start... ] hmm let me just do it directly
            grad[i_start:i_end+1] += 2.0 * w[idx] * a[k - np.arange(i_start, i_end+1)]
        
        # Step and project
        a_new = a - lr * grad
        a = simplex_project(a_new)
        
        lr *= 0.9999
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD {label} it={it} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_candidate(a):
    """Evaluate a candidate sequence."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a_norm = a / s
    return float(evaluate_sequence(a_norm.tolist()))
