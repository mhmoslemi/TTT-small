```python
def score_at_d(a, conv, i, d, S):
    """Compute score when a[i] is changed by d."""
    n = len(a)
    k_start = i
    k_end = min(i + n - 1, 2*n - 2)
    
    # Unaffected max
    if k_start > 0:
        unaff_left = conv[:k_start].max()
    else:
        unaff_left = -np.inf
    if k_end < 2*n - 2:
        unaff_right = conv[k_end+1:].max()
    else:
        unaff_right = -np.inf
    unaff_max = max(unaff_left, unaff_right)
    
    # Affected
    a_shift = a[0:n]  # a[k-i] for k = i, ..., i+n-1
    # Actually a[k-i] for k from i to min(i+n-1, 2n-2)
    # k-i ranges from 0 to min(n-1, 2n-2-i)
    length = k_end - k_start + 1
    a_shift = a[0:length]
    
    new_affected = conv[k_start:k_end+1] + 2.0 * d * a_shift
    
    # Handle k = 2i
    if 2*i >= k_start and 2*i <= k_end:
        idx = 2*i - k_start
        new_affected[idx] = conv[2*i] + 2.0*d*a[i] + d*d
    
    new_max = max(unaff_max, new_affected.max())
    new_S = S + d
    if new_S <= 0:
        return np.inf
    return 2.0 * n * new_max / (new_S * new_S)
```

```python
def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    return np.ones(n) / n

def golden_section_1d(f, lo, hi, iters=35):
    """Minimize unimodal f on [lo, hi] using golden section search."""
    gr = (np.sqrt(5) - 1) / 2

            c = d
            fc = fd
```

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using exact 1D coordinate descent
with golden section search, randomized coordinate ordering, and multi-length
exploration. Exploits the convexity of the normalized problem to guarantee
convergence to the global optimum.

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
