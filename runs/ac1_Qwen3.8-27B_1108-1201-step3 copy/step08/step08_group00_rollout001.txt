import numpy as np

from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def build_conv_matrix_sparse(a):
    """Build sparse representation for memory efficiency."""
    n = len(a)
    size = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(size):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[j])
    return np.array(rows), np.array(cols), np.array(vals)

def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def compute_subgradient(a, n_active=5):
    """Compute subgradient of max_k conv(a,a)[k] w.r.t. a."""

        active = top_vals
    
    g = np.zeros(len(a))

        je = min(len(a) - 1, k)
        idx = np.arange(js, je + 1)
        g[idx] += a[k - idx]
    g /= len(active)
    return g, max_b, active


def lbgfs_refine(a0, time_limit, T_start=150.0, T_end=0.2, maxiter=10000):
    """L-BFGS with LogSumExp smoothing and geometric temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    def make_func(T_val):
        def func(z):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(z):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z)
            a = softmax_from_z(z)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    n_stages = 8
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(30, maxiter // n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

def peak_focused_descent(a0, time_limit, n_steps=50000):
    """Novel peak-focused coordinate descent with block updates."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0
    block_size = max(4, n // 64)

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        
        # Find top peaks
        sorted_idx = np.argsort(b)[::-1]
        top_peaks = sorted_idx[:min(10, len(sorted_idx))]
        peak_vals = b[top_peaks]
        
        # Weighted gradient from peaks
        g = np.zeros(n)
        for k in top_peaks:
            weight = peak_vals[numpy.where(top_peaks == k)[0][0]] 
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(top_peaks)
        
        # Adaptive learning rate based on progress
        base_lr = 0.05 / (1 + 0.001 * step)
        if no_improve > 50:
            base_lr *= 0.5
        no_improve_counter = no_improve
        
        # Try different block locations
        improved = False
        n_blocks = n // block_size
        # Focus on blocks that contain peak-contributing indices
        contributing = np.zeros(n)
        for k in top_peaks[:5]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            contributing[js:je+1] += 1
        
        block_priority = np.zeros(n_blocks)
        for bi in range(n_blocks):
            start = bi * block_size
            end = min(start + block_size, n)
            block_priority[bi] = contributing[start:end].sum()
        
        # Try top priority blocks
        block_order = np.argsort(block_priority)[::-1]
        
        for bi in block_order[:8]:
            start = bi * block_size
            end = min(start + block_size, n)
            
            # For this block, try reducing elements that contribute most to peaks
            block_contrib = np.zeros(end - start)
            for k in top_peaks[:5]:
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                for local_i in range(end - start):
                    global_i = start + local_i
                    if js <= global_i <= je:
                        partner = k - global_i
                        if 0 <= partner < n:
                            block_contrib[local_i] += a[partner]
            
            # Try reducing the top contributor
            if block_contrib.max() < 1e-15:
                continue
            top_local = int(np.argmax(block_contrib))
            top_global = start + top_local
            
            if a[top_global] < 1e-14:
                continue
            
            for factor in [0.95, 0.90, 0.85, 0.80]:
                a_trial = a.copy()
                old_val = a_trial[top_global]
                a_trial[top_global] = old_val * factor
                diff = old_val - a_trial[top_global]
                if diff < 1e-16:
                    continue
                # Redistribute to non-contributing positions
                mask = np.ones(n, dtype=bool)
                mask[top_global] = False
                # Prefer positions far from peaks
                dist_to_peak = np.minimum(np.abs(np.arange(n) - top_global), 
                                          n - np.abs(np.arange(n) - top_global))
                dist_to_peak[top_global] = 1e10
                # Also avoid peak-contributing positions
                for k in top_peaks[:3]:
                    js = max(0, k - n + 1)
                    je = min(n - 1, k)
                    dist_to_peak[js:je+1] = np.minimum(dist_to_peak[js:je+1], 0.1)
                
                weights = np.maximum(dist_to_peak, 0.01)
                weights[top_global] = 0
                w_sum = weights[mask].sum()
                if w_sum < 1e-15:
                    continue
                a_trial[mask] += diff * weights[mask] / w_sum
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-12:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    no_improve = 0
                    break
            
            if improved:
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 200:
                # Random restart within neighborhood
                rng = np.random.default_rng(step)
                perturbation = rng.standard_normal(n) * 0.02
                a_trial = a + perturbation
                a_trial = np.maximum(a_trial, 0)
                if a_trial.sum() > 1e-10:
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        no_improve = 0
                    else:
                        break

    return best_a, best_score


def adaptive_mirror_descent(a0, time_limit, lr_init=0.1, n_steps=80000):
    """Mirror descent with adaptive step size and smart active set."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    z = z_from_a(a)
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    momentum = 0.9
    v = np.zeros(n)

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        
        # Adaptive active set: positions within 5% of max
        threshold = max_b * 0.95
        active = np.where(b >= threshold)[0]
        if len(active) > 20:
            active = np.argsort(b)[::-1][:20]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += a[k - idx]
        g /= len(active)

        # Momentum
        v = momentum * v + (1 - momentum) * g
        z_new = z - lr * v
        a_new = softmax_from_z(z_new)
        score_new = 2.0 * n * fast_convolve(a_new).max()

        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            z = z_new
            no_improve = 0
        else:
            no_improve += 1
            # Rollback and reduce lr
            lr *= 0.95

        if no_improve >= 100:
            lr = max(lr_init / (1.0 + 0.0002 * step), 1e-10)
            no_improve = 0

        if step % 30000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      MirrorDesc {elapsed:.0f}s: score={best_score:.8f} lr={lr:.8f}")

    return best_a, best_score


def structured_init(n, rng):
    """Create diverse structured initial sequences."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x) ** 2
    elif choice == 1:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 2:
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        # U-shaped
        a = x ** 2 + (1 - x) ** 2
    elif choice == 4:
        # Bimodal
        a = np.exp(-20 * (x - 0.25) ** 2) + np.exp(-20 * (x - 0.75) ** 2)
    elif choice == 5:
        # Power law
        a = x ** 0.5 + (1 - x) ** 0.5
    elif choice == 6:
        # Cosine power
        a = (0.5 * (1 + np.cos(2 * np.pi * x))) ** 2
    else:
        # Gaussian centered
        a = np.exp(-10 * (x - 0.5) ** 2)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

        if s < 0.01:
