Optimization of step function heights to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses Adam optimizer with log-sum-exp smoothing, adaptive temperature,
simplex projection, and LP-based final polish.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_gradient(f, p):
    """
    Compute gradient of log-sum-exp objective.
    f: sequence (length n), p: softmax weights on convolution (length 2n-1)
    Returns gradient vector of length n.
    """
    n = len(f)
    grad = np.zeros(n)
    for j in range(n):
        grad[j] = 2.0 * np.dot(p[j:j+n], f)
    return grad


def smooth_optimize(f_init, time_limit, lr=0.005, beta1=0.9, beta2=0.999,
                    sigma_init=None, sigma_min=1e-6, sigma_decay=0.995,
                    restart_interval=300, max_no_improve=100, label="smooth"):
    """
    Adam-based optimization with log-sum-exp smoothing and simplex projection.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
        s = 1.0
    else:
        f = f / s

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    if sigma_init is None:
        sigma_init = best_obj / 20.0
        sigma_init = max(sigma_init, 1e-8)
    sigma = sigma_init

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    no_improve = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        t += 1
        
        # Compute convolution and softmax weights
        c = fast_convolve(f)
        c_max = c.max()
        scaled = (c - c_max) / sigma
        p = np.exp(scaled)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.zeros(len(c))
            p[np.argmax(c)] = 1.0
        else:
            p = p / p_sum

        # Compute gradient
        grad = compute_smooth_gradient(f, p)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        f_new = f - step
        f_new = project_simplex(f_new)
        
        # Evaluate
        obj = fast_convolve(f_new).max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
        
        f = f_new
        
        # Anneal temperature
        if t % 50 == 0:
            sigma = max(sigma * sigma_decay, sigma_min)
        
        # Adaptive learning rate
        if no_improve > 20:
            lr *= 0.9
            no_improve = 0
        
        # Restart

            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
        
        if t % 200 == 0:

            print(f"    {label} it={t} obj={best_obj:.12f} score={score:.6f} "
                  f"sigma={sigma:.2e} lr={lr:.5f} t={elapsed:.0f}s")

def lp_polish(f_init, time_limit, top_k=200):
    """LP-based polish using the same approach as before but focused."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    full_len = 2 * n - 1
    stuck = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()

        # Select top peaks
        if top_k < full_len:
            idx = np.argpartition(b, -top_k)[-top_k:]
            idx = np.sort(idx)
        else:
            idx = np.arange(full_len)

        # Build constraint matrix
        K = len(idx)
        A = np.zeros((K, n))
        for i in range(K):

            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            A[i, lo:hi + 1] = f[lo:hi + 1]

        g_var = cp.Variable(n)
        prob = cp.Problem(cp.Maximize(cp.sum(g_var)), [g_var >= 0, A @ g_var <= M])
        
        try:
            prob.solve(solver=cp.ECOS, max_iters=50000)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, max_iters=50000, eps=1e-8)

                stuck += 1
                if stuck > 30:
                    break
                continue

            stuck += 1
            if stuck > 30:
                break
            continue

        g_val = np.maximum(g_var.value, 0.0)
        S_g = g_val.sum()
        
        if S_g <= 1.0 + 1e-5:
            stuck += 1
            if stuck > 30:
                break
            continue

        h = g_val / S_g
        
        # Try different step sizes

        best_trial_obj = M
        for t_val in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7]:

            f_trial = np.maximum(f_trial, 0.0)

            if s_tr < 1e-15:
                continue
            f_trial = f_trial / s_tr
            obj = fast_convolve(f_trial).max()
            if obj < best_trial_obj:
                best_trial_obj = obj
                best_t = t_val

        if best_t == 0.0:
            stuck += 1
            if stuck > 30:
                break
            continue

        f = (1 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        f = f / f.sum()
        cur_obj = fast_convolve(f).max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1

        if it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    lp it={it} obj={best_obj:.12f} score={2*n*best_obj:.6f} "
                  f"k={K} t={elapsed:.0f}s")

    return best_f, 2.0 * n * best_obj


def interp_sequence(src, target_n):
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, np.maximum(src, 0))
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result

"""
Optimization of step function heights to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses Adam optimizer with log-sum-exp smoothing, adaptive temperature annealing,
FFT-based gradient computation, simplex projection, multi-start strategy,
and LP-based final polish.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient_fft(f, p):
    """
    Compute gradient of log-sum-exp objective using FFT cross-correlation.
    f: sequence length n, p: softmax weights length 2n-1.
    Returns gradient of length n.
    """
    n = len(f)
    fft_len = 1 << (3 * n).bit_length()
    P = np.fft.fft(p, fft_len)
    F = np.fft.fft(f, fft_len)
    Q = np.fft.ifft(P * np.conj(F))
    return 2.0 * Q[:n].real


def smooth_optimize(f_init, time_limit, lr=0.005, beta1=0.9, beta2=0.999,
                    sigma_factor_init=25.0, sigma_factor_min=2.0,
                    sigma_decay=0.997, restart_interval=400,
                    max_no_improve=150, label="smooth", seed=42):
    """
    Adam-based optimization with log-sum-exp smoothing and simplex projection.
    Minimizes max(conv(f,f)) subject to f>=0, sum(f)=1.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    c = fast_convolve(f)
    best_obj = c.max()
    best_f = f.copy()

    sigma = best_obj / sigma_factor_init
    rng = np.random.default_rng(seed)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    no_improve = 0
    eval_count = 0

    while time.monotonic() - t0 < time_limit:
        t += 1

        c = fast_convolve(f)
        c_max = c.max()
        scaled = (c - c_max) / sigma
        p = np.exp(scaled)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.zeros(len(c))
            p[np.argmax(c)] = 1.0
        else:
            p = p / p_sum

        grad = compute_gradient_fft(f, p)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        f_new = f - step
        f_new = project_simplex(f_new)

        obj = fast_convolve(f_new).max()
        eval_count += 1

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0

        f = f_new

        # Anneal temperature
        if t % 30 == 0:
            sigma = max(sigma * sigma_decay, best_obj / 1000.0)

        # Adaptive learning rate
        if no_improve > 30 and no_improve % 30 == 0:
            lr = max(lr * 0.85, 1e-6)
            no_improve = 0

        # Restart
        if no_improve > restart_interval:
            noise_scale = 0.001 + 0.002 * (no_improve / restart_interval)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0
            lr = max(lr, 0.002)

        if t % 500 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_obj
            print(f"    {label} it={t} obj={best_obj:.12f} score={score:.6f} "
                  f"sigma={sigma:.2e} lr={lr:.5f} t={elapsed:.0f}s")

    return best_f, 2.0 * n * best_obj


def lp_polish(f_init, time_limit, top_k=200):
    """LP-based polish for final refinement."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    full_len = 2 * n - 1
    stuck = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()

        if top_k < full_len:
            idx = np.argpartition(b, -top_k)[-top_k:]
            idx = np.sort(idx)
        else:
            idx = np.arange(full_len)

        K = len(idx)
        A = np.zeros((K, n))
        for i in range(K):
            k = idx[i]
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            A[i, lo:hi + 1] = f[lo:hi + 1]

        g_var = cp.Variable(n)
        prob = cp.Problem(cp.Maximize(cp.sum(g_var)), [g_var >= 0, A @ g_var <= M])

        try:
            prob.solve(solver=cp.ECOS, max_iters=50000)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, max_iters=50000, eps=1e-8)
            except Exception:
                stuck += 2
                if stuck > 40:
                    break
                continue

        if g_var.value is None:
            stuck += 2
            if stuck > 40:
                break
            continue

        g_val = np.maximum(g_var.value, 0.0)
        if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
            stuck += 2
            if stuck > 40:
                break
            continue

        S_g = g_val.sum()
        if S_g <= 1.0 + 1e-5:
            stuck += 1
            if stuck > 40:
                break
            continue

        h = g_val / S_g

        best_t = 0.0
        best_trial_obj = M
        for t_val in [0.03, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
            f_trial = (1.0 - t_val) * f + t_val * h
            f_trial = np.maximum(f_trial, 0.0)
            s_tr = f_trial.sum()
            if s_tr < 1e-15:
                continue
            f_trial = f_trial / s_tr
            obj = fast_convolve(f_trial).max()
            if obj < best_trial_obj:
                best_trial_obj = obj
                best_t = t_val

        if best_t == 0.0:
            stuck += 1
            if stuck > 40:
                break
            continue

        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        f = f / f.sum()
        cur_obj = fast_convolve(f).max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1

        if it % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    lp it={it} obj={best_obj:.12f} score={2*n*best_obj:.6f} "
                  f"k={K} stuck={stuck} t={elapsed:.0f}s")

    return best_f, 2.0 * n * best_obj


def interp_sequence(src, target_n):
    """Interpolate a sequence to a new length."""
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, np.maximum(src, 0.0))
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result
