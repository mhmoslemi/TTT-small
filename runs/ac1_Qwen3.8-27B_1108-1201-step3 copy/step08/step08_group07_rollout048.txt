```python
def simplex_project(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Binary search for the threshold
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cv - 1))[0][-1]
    theta = (cv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def compute_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = rfft(a, N)
    return irfft(A * A, N)[:size]

def compute_gradient_smoothed(a, T):
    """Gradient of (2n/T)*log(sum exp(T*c[k])) with respect to a."""
    n = len(a)
    c = compute_conv(a)
    # Softmax weights
    c_max = c.max()
    exp_tc = np.exp(T * (c - c_max))
    Z = exp_tc.sum()
    w = exp_tc / Z  # w[k] = exp(T*c[k]) / Z
    
    # h[j] = sum_i w[j+i] * a[i] for j = 0, ..., n-1
    a_rev = a[::-1]

    W = rfft(w, N)
    Ar = rfft(a_rev, N)
    conv_result = irfft(W * Ar, N)
    h = conv_result[n-1 : 2*n-1]
    
    return 4.0 * n * h
```

def exponentiated_gradient(a0, time_limit, T_init=20, T_final=300, lr=0.5, verbose=True):

    a = np.maximum(a0.copy(), 1e-15)

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    total_time = time_limit
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        # Adaptive T
        progress = min(1.0, (time.monotonic() - t0) / total_time)
        T = T_init + (T_final - T_init) * progress
        
        grad = grad_smoothed(a, T)
        
        # Exponentiated gradient update
        scale = lr / (n * T)  # normalize by n*T to keep exponent O(1)

def build_conv_matrix(f):
    n = len(f)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)


def solve_best_response(f, M, prev_g=None):
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    if prev_g is not None:
        g.value = prev_g
    C = build_conv_matrix(f)
    constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL, warm_start=(prev_g is not None))
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return prev_g if prev_g is not None else f.copy()
    if g.value is None:
        return prev_g if prev_g is not None else f.copy()
    return np.maximum(g.value, 0)
