from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def general_convolve(a, b):
    """Linear convolution of two arrays via FFT."""
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_grad(a, temperature):
    """Compute smooth gradient of max(a*a) using log-sum-exp."""
    n = len(a)
    b = fast_convolve(a)
    # Softmax weights
    b_scaled = b / temperature
    b_max = b_scaled.max()
    exp_b = np.exp(b_scaled - b_max)
    w = exp_b / exp_b.sum()
    
    # grad[j] = sum_k w[k] * a[k-j]
    fft_size = 1 << (n + len(w) - 1 - 1).bit_length()
    W = rfft(w, fft_size)
    A_rev = rfft(a[::-1], fft_size)
    conv = irfft(W * A_rev, fft_size)[:n + len(w) - 1]
    grad = conv[n-1:n-1+n]
    return grad, b, w

                                temp_growth=1.02, max_temp=1.0, 
                                adaptive_lr=True, label="sgd"):

    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    
    lr = lr_init
    temp = temp_init
    v = np.zeros(n)  # momentum
    beta = 0.7
    no_improve = 0
    it = 0
    rng = np.random.default_rng(123)
    
    while time.monotonic() - t0 < time_limit:

        f_new = project_simplex(f + v)
        
        obj = fast_convolve(f_new).max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_score = 2.0 * n * obj
            best_f = f_new.copy()
            no_improve = 0
            if adaptive_lr and it % 10 == 0:
                lr = min(lr * 1.01, 0.1)
            f = f_new

            if no_improve > 20:

                no_improve = 0
                temp = min(temp * temp_growth, max_temp)
            f = f  # don't update

            elapsed = time.monotonic() - t0

def solve_lp_pdslp(f, indices, M, solver=None):
    """Solve LP: max sum(g) s.t. g >= 0, A@g <= M."""
    n = len(f)
    K = len(indices)
    
    # Build constraint matrix
    A = np.zeros((K, n))
    for i in range(K):
        k = indices[i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi+1] = f[lo:hi+1]
    
    g = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [g >= 0, A @ g <= M])
    
    try:
        if solver is None:
            # Try PDLP first (much faster for large problems)
            prob.solve(solver=cp.PDLP, max_iters=5000, rho=0.1, scale="auto")
        else:
            prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=100000, eps=1e-8)
        except Exception:
            return None
    
    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None
    return g_val

Optimized search for step function minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses PDLP solver for fast LP, smoothed gradient descent with adaptive temperature,

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def general_convolve(a, b):
    """Linear convolution of two arrays via FFT."""
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_grad(a, temperature):
    """
    Compute smooth gradient of max(a*a) using log-sum-exp softmax.
    Returns (gradient, convolution, softmax_weights).
    """
    n = len(a)
    b = fast_convolve(a)
    b_scaled = b / temperature
    b_max = b_scaled.max()
    exp_b = np.exp(b_scaled - b_max)
    w = exp_b / exp_b.sum()

    # grad[j] = sum_k w[k] * a[k-j]
    # = conv(w, a_rev)[n-1 : n-1+n] where a_rev = a[::-1]
    fft_size = 1 << (n + len(w) - 1 - 1).bit_length()
    W = rfft(w, fft_size)
    A_rev = rfft(a[::-1], fft_size)
    conv = irfft(W * A_rev, fft_size)[:n + len(w) - 1]
    grad = conv[n-1:n-1+n]
    return grad, b, w


def smoothed_gradient_descent(f_init, time_limit, lr_init=0.005, temp_init=0.005,

    it = 0

        it += 1

        v = beta * v - lr * grad

def build_A_matrix(f, indices, n):
    """Build K x n constraint matrix for peak indices."""
    K = len(indices)
    A = np.zeros((K, n))
    for i in range(K):
        k = indices[i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi+1] = f[lo:hi+1]
    return A


def solve_lp(f, indices, M, use_pdlp=True):
    """Solve LP: max sum(g) s.t. g >= 0, A@g <= M. Returns g or None."""
    n = len(f)
    K = len(indices)
    if K == 0 or K > 2 * n - 1:
        return None

    A = build_A_matrix(f, indices, n)

    g = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [g >= 0, A @ g <= M])

    try:
        if use_pdlp and K <= 5000 and n <= 5000:
            prob.solve(solver=cp.PDLP, max_iters=10000, rho=0.1, scale="auto",
                       verbose=False)
        else:
            prob.solve(solver=cp.PDLP, max_iters=5000, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=50000, eps=1e-8, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iters=50000, verbose=False)
            except Exception:
                return None

    if g.value is None:
        return None
    g_val = np.maximum(np.array(g.value).flatten(), 0.0)
    if np.any(~np.isfinite(g_val)):
        return None
    return g_val

def coordinate_descent(f_init, time_limit):
    """Coordinate descent with adaptive multipliers."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    best_f = f.copy()

    multipliers = [0.3, 0.5, 0.7, 0.85, 0.92, 1.05, 1.15, 1.3, 1.5, 1.8]
    no_pass = 0

    while time.monotonic() - t0 < time_limit and no_pass < 2:
        any_imp = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-14:
                continue
            best_j_val = f[j]
            best_j_obj = best_obj
            for mult in multipliers:
                f_trial = f.copy()
                f_trial[j] = f[j] * mult
                st = f_trial.sum()
                if st < 1e-12:
                    continue
                f_trial = f_trial / st
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = f[j] * mult
            if best_j_val != f[j]:
                f[j] = best_j_val
                st = f.sum()
                f = f / st
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_score = 2.0 * n * best_obj
                    best_f = f.copy()
                    any_imp = True
        no_pass = 0 if any_imp else no_pass + 1

    return best_f, best_score


def interp_sequence(src, target_n):
    """Interpolate a sequence to a new length."""
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result


def gaussian_init(n, sigma_frac=0.3):
    """Gaussian-shaped initialization."""
    x = np.arange(n) - n / 2.0
    sigma = sigma_frac * n
    g = np.exp(-x**2 / (2.0 * sigma**2))
    g = np.maximum(g, 0.0)
    s = g.sum()
    return g / s

            print(f"  NEW BEST [final]: {real_score:.6f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
