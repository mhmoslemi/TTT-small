"""
Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-accelerated custom LP solver, Nesterov momentum subgradient,
and multi-scale n exploration.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    n, m = len(a), len(b)
    if n == 0 or m == 0:
        return np.array([0.0])
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def adj_convolve(f, v):
    """Compute A^T @ v where (A g)_k = conv(f,g)_k.
    (A^T v)[j] = sum_i f[i] * v[i+j] for valid i,j.
    Computed as: conv(f, v[::-1])[len(v)-1-j]
    """
    n = len(f)
    vlen = len(v)
    conv_res = fast_cross_convolve(f, v[::-1])
    idx = vlen - 1 - np.arange(n)
    result = np.zeros(n)
    valid = (idx >= 0) & (idx < len(conv_res))
    result[valid] = conv_res[idx[valid]]
    return result


def obj_value(f):
    """Compute max(conv(f,f)) for f with sum=1."""
    s = f.sum()
    if s < 1e-15:
        return np.inf
    fn = f / s
    return fast_convolve(fn).max()


def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    v = v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho_star = np.nonzero(cond)[0][-1]
    theta = (cssv[rho_star] - 1.0) / (rho_star + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, M, n_inner=30, mu_init=0.1, warm_start=None):
    """
    Solve (approximately): max sum(g) s.t. conv(f,g) <= M, g >= 0
    using projected subgradient with method of multipliers.
    Returns g (not normalized).
    """
    n = len(f)
    full_len = 2 * n - 1
    
    if warm_start is not None:
        g = warm_start.copy()
    else:
        g = np.ones(n) / n
    
    lam = np.zeros(full_len)
    mu = mu_init
    
    for outer in range(3):
        for inner in range(n_inner):
            # Compute r = conv(f, g)
            r = fast_cross_convolve(f, g)
            
            # Compute effective dual: w = lam + mu * max(0, r - M)
            excess = np.maximum(r - M, 0.0)
            w = lam + mu * excess
            
            # Compute A^T @ w
            At_w = adj_convolve(f, w)
            
            # Gradient of -sum(g) + penalty
            grad = -np.ones(n) + At_w
            
            # Step size
            lr = 0.5 / (np.abs(grad).max() + 1e-10)
            g_new = np.maximum(g - lr * grad, 0.0)
            
            # Scale to prevent g from growing too large
            g_sum = g_new.sum()
            if g_sum > 10.0 * n:
                g_new = g_new / g_sum * n
            elif g_sum < 0.01:
                g_new = np.ones(n) / n
            
            g = g_new
        
        # Update dual
        r = fast_cross_convolve(f, g)
        lam = np.maximum(0.0, lam + mu * (r - M))
        mu = min(mu * 1.8, 10.0)
    
    return g

def coordinate_descent(f_init, time_limit, label="cd"):
    """Coordinate descent polish."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    best_f = f.copy()
    
    multipliers = [0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0]
    no_pass = 0
    
    while time.monotonic() - t0 < time_limit:
        any_improved = False
        for j in range(0, n, max(1, n // 500)):  # Subsample for speed
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-14:
                continue
            
            best_j_val = f[j]
            best_j_obj = best_obj
            for mult in multipliers:
                f_trial = f.copy()
                f_trial[j] = f[j] * mult
                st = f_trial.sum()
                if st < 1e-12:
                    continue
                f_trial = f_trial / st
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-14:
                    best_j_obj = obj
                    best_j_val = f[j] * mult
            
            if best_j_val != f[j]:
                f[j] = best_j_val
                st = f.sum()
                if st > 1e-12:
                    f = f / st
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_score = 2.0 * n * best_obj
                    best_f = f.copy()
                    any_improved = True
        
        if not any_improved:
            no_pass += 1
            if no_pass >= 2:
                break
        else:
            no_pass = 0
    
    return best_f, best_score


def interp_sequence(src, target_n):
    """Interpolate a sequence to a new length."""
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    return result
