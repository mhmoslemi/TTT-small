"""
Multi-stage optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses: (1) Bussgang spectral flattening for fast initialization, (2) LP fictitious play
with scipy linprog for main optimization, (3) L-BFGS log-sum-exp polishing, (4) coordinate
descent for fine-tuning, and (5) multi-length exploration.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.linalg import toeplitz
from scipy.optimize import minimize, linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_conv_matrix(f):
    """Build (2n-1) x n matrix C such that C @ g = conv(f, g)."""
    n = len(f)
    size = 2 * n - 1
    C = np.zeros((size, n))
    for k in range(size):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for i in range(lo, hi):
            C[k, i] = f[k - i]
    return C


def build_conv_matrix_fast(f):
    """Build convolution matrix using Toeplitz structure."""
    n = len(f)
    size = 2 * n - 1
    col0 = np.zeros(size)
    col0[:n] = f
    C = toeplitz(col0, col0[:1][0] * np.ones(1))
    # Actually toeplitz needs first column and first row
    # For convolution: C[k,i] = f[k-i] when 0<=k-i<n
    # First column (i=0): C[k,0] = f[k] for k<n, 0 otherwise
    # First row (k=0): C[0,i] = f[-i] = 0 for i>0, f[0] for i=0
    # So first_col = [f[0], f[1], ..., f[n-1], 0, ..., 0] (length 2n-1)
    # first_row = [f[0], 0, 0, ..., 0] (length n)
    first_col = np.zeros(size)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)


def solve_best_response_lp(f, M, g_prev=None):
    """Solve: max sum(g) s.t. C(f)@g <= M, g >= 0, using scipy linprog."""
    n = len(f)
    C = build_conv_matrix_fast(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 5.0})
    if res.status == 0 and res.x is not None:
        return np.maximum(res.x, 0)
    elif g_prev is not None:
        return g_prev.copy()
    else:
        return f.copy()


def bussgang_iteration(a0, time_limit, tau=0.5, verbose=True):
    """Bussgang-type spectral flattening to reduce peak autocorrelation."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()
    iters = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        A = rfft(a, fft_size)
        power = np.abs(A) ** 2
        total_power = power.sum()
        target = total_power / fft_size
        ratio = np.where(power > 1e-30, target / power, 1.0)
        # Clip ratio to avoid numerical issues
        ratio = np.clip(ratio, 0.01, 100.0)
        A_new = A * (ratio ** tau)
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        score = compute_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        a = a_new

        if no_improve > 200:
            # Try different tau
            tau = tau * 0.8 if tau > 0.1 else tau + 0.1
            no_improve = 0
            # Add small perturbation
            rng = np.random.default_rng(iters)
            a = best_a + rng.standard_normal(n) * 0.001
            a = np.maximum(a, 0)
            a = a / a.sum()

        if verbose and iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Bussgang iter={iters}, score={best_score:.8f}, tau={tau:.3f}, time={elapsed:.0f}s")

    return best_a, best_score


def lp_fictitious_play(a0, time_limit, step=0.4, verbose=True):
    """LP-based fictitious play with adaptive step size."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    prev_g = None
    stall_count = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        M = conv.max()

        g = solve_best_response_lp(a, M, g_prev=prev_g)
        prev_g = g
        g_sum = g.sum()

        if g_sum < 1e-10:
            break

        g_norm = g / g_sum
        a_new = (1 - step) * a + step * g_norm
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss

        score = compute_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1

        a = a_new

        if stall_count > 300:
            step = max(0.05, step * 0.7)
            rng = np.random.default_rng(iters * 7 + 3)
            a = best_a.copy() + rng.standard_normal(n) * 0.0003
            a = np.maximum(a, 0)
            a = a / a.sum()
            prev_g = None
            stall_count = 0

        if verbose and iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-FP iter={iters}, best={best_score:.8f}, step={step:.4f}, time={elapsed:.0f}s/{time_limit:.0f}s")

    return best_a, best_score


def softmax_from_z(z):
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def lse_objective(z, T, n):
    """Smoothed log-sum-exp objective in log-space."""
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    lse = max_shifted + np.log(np.exp(shifted - max_shifted).sum())
    return (2.0 * n / T) * np.exp(T * lse)


def lse_gradient(z, T, n):
    """Gradient of smoothed objective."""
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    w = np.exp(shifted - max_shifted)
    w = w / w.sum()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
    S = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - S)
    return (2.0 * n / T) * grad_z


def lbgfs_optimize(a0, time_limit, T=500, maxiter=5000):
    """L-BFGS on smoothed log-sum-exp objective in log-space."""
    n = len(a0)
    z0 = np.log(np.maximum(a0, 1e-30))
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        return lse_objective(z, T, n)

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        return lse_gradient(z, T, n)

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-15,
                     'maxfun': maxiter * 100, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
        score = compute_score(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except Exception as e:
        print(f"    L-BFGS error: {e}")
    return best_a, best_score


def coordinate_descent(a0, time_limit, verbose=True):
    """Coordinate descent with pairwise transfers for fine-tuning."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False
        # Try multiple random pairs per iteration
        for _ in range(500):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            base = min(max(a[i], 0), max(a[j], 0))
            if base < 1e-12:
                continue

            best_local = best_score
            best_t = 0.0
            for frac in [0.01, 0.05, 0.1, 0.2, 0.3, 0.5]:
                t = frac * base
                if t >= a[i] - 1e-15:
                    t = a[i] * 0.4
                a_trial = a.copy()
                a_trial[i] -= t
                a_trial[j] += t
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score(a_trial)
                if sc < best_local - 1e-14:
                    best_local = sc
                    best_t = t

            if best_t != 0.0:
                a[i] -= best_t
                a[j] += best_t
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = best_local
                best_a = a.copy()
                improved = True

        if not improved and iters > 10:
            noise_scale = 0.001 * max(1.0, (iters // 20) * 0.5)
            noise = rng.standard_normal(n) * noise_scale
            a = best_a.copy() + noise
            a = np.maximum(a, 0)
            a = a / a.sum()

        if verbose and iters % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD iter={iters}, best={best_score:.8f}, time={elapsed:.0f}s")

    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        best_seq, best_score = check_and_update(a_lp4, best_seq, best_score)
        print(f"  After final LP: score={best_score:.8f}")

    # === Phase 8: Try nearby lengths ===
    rem = deadline - time.monotonic()
    print(f"\n=== Phase 8: Nearby lengths ===")
    alt_lengths = []
    # Try small perturbations of the parent length
    for delta in [2, 4, 8, -2, -4, 16, -8, 32]:
        alt_n = n_parent + delta
        if 64 <= alt_n <= 4096 and alt_n not in alt_lengths:
            alt_lengths.append(alt_n)

    for alt_n in alt_lengths[:6]:
        rem = deadline - time.monotonic()
        if rem < 20:
            break
        t8 = min(rem * 0.15, 60.0)
        init_alt = interpolate_sequence(best_seq, alt_n)
        print(f"  Trying n={alt_n}, budget={t8:.0f}s")
        # Quick LP-FP then L-BFGS
        a_alt1, _ = lp_fictitious_play(init_alt, t8 * 0.4, step=0.3, verbose=False)
        a_alt2, _ = lbgfs_optimize(a_alt1, t8 * 0.5, T=500, maxiter=2000)
        best_seq, best_score = check_and_update(a_alt2, best_seq, best_score)
        print(f"    n={alt_n}: score={best_score:.8f}")

    # === Phase 9: Final coordinate descent on best ===
    rem = deadline - time.monotonic()
    if rem > 20:
        t9 = min(rem * 0.3, 60.0)
        print(f"\n=== Phase 9: Final CD (budget={t9:.0f}s) ===")
        a_final, sc_final = coordinate_descent(best_seq, t9, verbose=True)
        best_seq, best_score = check_and_update(a_final, best_seq, best_score)
        print(f"  Final score: {best_score:.8f}")

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}")

    if best_seq is None:
        best_seq = parent

    return [float(x) for x in best_seq]
