```python
def project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.find_last((u * np.arange(1, n+1) > cssv - 1) & (np.arange(1, n+1) > 0))
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. Multiplicative weights / Hedge algorithm for the min-max game formulation
2. Exponentiated gradient (closed-form) inner minimization
3. Log-domain smoothed L-BFGS polishing
4. Pairwise coordinate descent with golden-section line search
5. Multi-length and multi-start exploration

The key insight is the min-max reformulation:
  min_a max_k (a*a)[k] = max_p min_a sum_k p[k](a*a)[k]
where p ranges over probability distributions on {0,...,2n-2}.
The inner minimization is a convex QP on the simplex, solvable by
projected gradient or exponentiated gradient. FFT accelerates the
key matrix-vector products to O(n log n).
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute score = 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def matvec_hankel(p, a):
    """Compute M_p @ a where M_p[i,j] = p[i+j].
    p has length 2n-1, a has length n.
    Uses FFT for O(n log n) complexity."""
    n = len(a)
    # We want: result[i] = sum_{j=0}^{n-1} p[i+j] * a[j]
    # This equals conv(a, p_rev)[2n-2-i] where p_rev = p[::-1]
    b_rev = p[::-1]
    conv_size = n + (2 * n - 1) - 1  # = 3n - 2
    fft_size = 1 << (conv_size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b_rev, fft_size)
    conv = irfft(A * B, fft_size)[:conv_size]
    # result[i] = conv[2n-2-i] for i=0,...,n-1
    # So result = conv[n-1 : 2n-1][::-1]
    return conv[n-1 : 2*n-1][::-1]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > cssv - 1)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def expgrad_inner(p, a_init, eta, max_iter=200, tol=1e-10):
    """Exponentiated gradient method for min_a a^T M_p a on simplex.
    
    The optimality condition is: a[i] ∝ exp(-eta * (M_p a)[i]).
    We solve by fixed-point iteration (self-consistent equation).
    """
    a = a_init.copy()
    a = np.maximum(a, 1e-30)
    a = a / a.sum()
    
    for it in range(max_iter):
        Ma = matvec_hankel(p, a)
        # a_new[i] = exp(-eta * Ma[i]) / Z
        log_a = -eta * Ma
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        s = a_new.sum()
        if s < 1e-300:
            break
        a_new /= s
        
        diff = np.abs(a_new - a).max()
        a = a_new
        if diff < tol:
            break
    
    return a


def hedge_game(a_init, time_limit, eta_outer=1.0, eta_inner=10.0,
               inner_iter=100, verbose=True):
    """Multiplicative weights algorithm for the min-max game.
    
    min_a max_k (a*a)[k] = max_p min_a sum_k p[k](a*a)[k]
    
    Row player (a): exponentiated gradient
    Column player (p): multiplicative weights
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-15)
    a = a / a.sum()
    
    # Initialize p as uniform over 2n-1 peaks
    p = np.ones(2 * n - 1) / (2 * n - 1)
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    # Running average of a (for final output)
    a_avg = a.copy()
    
    for t in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Inner: solve min_a a^T M_p a using exponentiated gradient
        a_inner = expgrad_inner(p, a, eta_inner, max_iter=inner_iter)
        
        # Compute convolution
        c = fast_convolve(a_inner)
        
        # Update p: multiplicative weights
        eta_p = eta_outer
        log_p = np.log(p + 1e-300) + eta_p * c
        log_p -= log_p.max()
        p = np.exp(log_p)
        p /= p.sum()
        
        # Update a: multiplicative weights on a
        # (This is the "dual" of the column player's update)
        Ma = matvec_hankel(p, a)
        log_a = np.log(a + 1e-300) - eta_outer * 2.0 * Ma
        log_a -= log_a.max()
        a_mul = np.exp(log_a)
        a_mul /= a_mul.sum()
        
        # Blend
        alpha = 2.0 / (t + 3)
        a = (1 - alpha) * a + alpha * a_inner
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        
        # Update average
        a_avg = ((t * a_avg) + a) / (t + 1)
        
        # Track best
        score_inner = compute_score(a_inner)
        score_avg = compute_score(a_avg)
        
        if score_inner < best_score:
            best_score = score_inner
            best_a = a_inner.copy()
        if score_avg < best_score:
            best_score = score_avg
            best_a = a_avg.copy()
        
        if verbose and t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"  Hedge t={t}, score_inner={score_inner:.8f}, "
                  f"score_avg={score_avg:.8f}, best={best_score:.8f}, "
                  f"elapsed={elapsed:.0f}s")
    
    return best_a, best_score

    # = 4n * sum_k w[k] a[k-i]
    # This is 4n times the cross-correlation of w and a.
    
    # sum_k w[k] a[k-i] for i=0,...,n-1
    # = (w cross-correlate a)[i] ... 
    # Let me compute it as: for each i, sum over k of w[k] * a[k-i] where 0<=k-i<n
    # = sum_{j=0}^{n-1} w[i+j] * a[j]  (substituting j = k-i)
    # This is exactly matvec_hankel(w, a) but w has length 2n-1 and a has length n!
    
    grad = 2.0 * n * 2.0 * matvec_hankel(w, a)
    # So grad[i] = 4n * sum_j w[i+j] a[j]
    
    return 2.0 * n * smooth_max, grad


def smoothed_max_obj_and_grad(a, T, n):
    """Compute 2n * smooth_max(conv(a,a)) and gradient w.r.t. a."""
    conv = fast_convolve(a)
    max_c = conv.max()
    exp_shifted = np.exp((conv - max_c) / T)
    sum_exp = exp_shifted.sum()
    smooth_max = max_c + T * np.log(sum_exp)
    obj = 2.0 * n * smooth_max
    
    w = exp_shifted / sum_exp
    # grad[i] = 2n * 2 * sum_j w[i+j] * a[j] = 4n * matvec_hankel(w, a)
    grad = 4.0 * n * matvec_hankel(w, a)
    
    return obj, grad


def log_lbgfs(a_init, time_limit, T_list=[50, 200, 1000], verbose=True):
    """L-BFGS in log-space with smoothed max objective."""
    a = np.maximum(a_init, 1e-15)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    def z_from_a(a_vec):
        return np.log(np.maximum(a_vec, 1e-30))
    
    def a_from_z(z):
        z_shifted = z - z.max()
        a_vec = np.exp(z_shifted)
        return a_vec / a_vec.sum()
    
    for T in T_list:
        if time.monotonic() - t0 > time_limit:
            break
        
        z0 = z_from_a(best_a)
        
        max_iter = 3000
        t_start = time.monotonic()
        
        def func_and_grad(z):
            if time.monotonic() - t0 > time_limit:
                return 1e15, np.zeros_like(z)
            a_tmp = a_from_z(z)
            obj, grad_a = smoothed_max_obj_and_grad(a_tmp, T, n)
            # Project gradient to log-space: grad_z[i] = a[i] * (grad_a[i] - sum grad_a*a)
            S = np.dot(grad_a, a_tmp)
            grad_z = a_tmp * (grad_a - S)
            return obj, grad_z
        
        try:
            res = minimize(
                fun=lambda z: func_and_grad(z)[0],
                x0=z0,
                jac=lambda z: func_and_grad(z)[1],
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12,
                         'maxfun': max_iter * 100, 'disp': False}
            )
            a_out = a_from_z(res.x)
            score = compute_score(a_out)
            if score < best_score:
                best_score = score
                best_a = a_out
                if verbose:
                    print(f"    L-BFGS T={T}: score={score:.8f}")
        except Exception as e:
            if verbose:
                print(f"    L-BFGS T={T} failed: {e}")
    
    return best_a, best_score


def coord_descent(a_init, time_limit, verbose=True):
    """Pairwise coordinate descent with golden-section search."""
    n = len(a_init)
    a = np.maximum(a_init, 0.0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    gr = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio
    iters = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False
        
        # Get current peaks
        conv = fast_convolve(a)
        
        # Generate pairs
        n_pairs = min(n, 300)
        pairs = []
        
        # Random pairs
        for _ in range(n_pairs * 2 // 3):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                pairs.append((min(i, j), max(i, j)))
        
        # Peak-focused pairs
        top_k = min(3, len(conv))
        peak_idx = np.argsort(conv)[::-1][:top_k]
        for pk in peak_idx:
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            for _ in range(n_pairs // 6):
                i = rng.integers(js, je + 1)
                j = rng.integers(js, je + 1)
                if i != j:
                    pairs.append((min(int(i), int(j)), max(int(i), int(j))))
        
        # Deduplicate
        pairs = list(set(pairs))
        
        for (i, j) in pairs:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14 and a[j] < 1e-14:
                continue
            
            t_lo = -a[j] + 1e-15
            t_hi = a[i] - 1e-15
            if t_hi <= t_lo:
                continue
            
            x1 = t_hi - gr * (t_hi - t_lo)
            x2 = t_lo + gr * (t_hi - t_lo)
            
            def eval_t(t):
                a_i = a[i] - t
                a_j = a[j] + t
                if a_i < -1e-15 or a_j < -1e-15:
                    return 1e10
                a_trial = a.copy()
                a_trial[i] = max(0.0, a_i)
                a_trial[j] = max(0.0, a_j)
                ss = a_trial.sum()
                if ss < 1e-15:
                    return 1e10
                a_trial /= ss
                conv_trial = fast_convolve(a_trial)
                return 2.0 * n * conv_trial.max()
            
            f1 = eval_t(x1)
            f2 = eval_t(x2)
            
            for _ in range(15):
                if time.monotonic() - t0 > time_limit:
                    break
                if f1 < f2:
                    t_hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = t_hi - gr * (t_hi - t_lo)
                    f1 = eval_t(x1)
                else:
                    t_lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = t_lo + gr * (t_hi - t_lo)
                    f2 = eval_t(x2)
            
            t_opt = (t_lo + t_hi) / 2.0
            f_opt = min(f1, f2)
            
            if f_opt < best_score - 1e-12:
                a[i] = max(0.0, a[i] - t_opt)
                a[j] = max(0.0, a[j] + t_opt)
                a = a / a.sum()
                best_score = compute_score(a)
                best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 8:
                # Perturb
                a = best_a + rng.standard_normal(n) * 0.003
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
        else:
            no_improve = 0
        
        if verbose and iters % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"  CD iter={iters}, score={best_score:.8f}, elapsed={elapsed:.0f}s")
    
    return best_a, best_score


def hedge_game_v2(a_init, time_limit, verbose=True):
    """Improved Hedge algorithm with adaptive parameters."""
    n = len(a_init)
    a = np.maximum(a_init, 1e-15)
    a = a / a.sum()
    
    p = np.ones(2 * n - 1) / (2 * n - 1)
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    a_avg = a.copy()
    
    # Adaptive parameters
    base_eta = 5.0  # initial learning rate for both players
    
    for t in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        eta = base_eta / (1.0 + 0.01 * t)  # decay learning rate
        
        # Inner: exponentiated gradient for min_a a^T M_p a
        a_curr = a
        for inner_it in range(30):
            Ma = matvec_hankel(p, a_curr)
            log_a_new = np.log(a_curr + 1e-300) - eta * 2.0 * Ma
            log_a_new -= log_a_new.max()
            a_new = np.exp(log_a_new)
            a_new /= a_new.sum()
            if np.abs(a_new - a_curr).max() < 1e-12:
                break
            a_curr = a_new
        a_inner = a_curr
        
        # Compute convolution of inner solution
        c = fast_convolve(a_inner)
        
        # Update p: multiplicative weights (column player)
        log_p = np.log(p + 1e-300) + eta * c
        log_p -= log_p.max()
        p = np.exp(log_p)
        p /= p.sum()
        
        # Blend a with inner solution
        alpha = min(1.0, 5.0 / (t + 5))
        a = (1 - alpha) * a + alpha * a_inner
        a = np.maximum(a, 1e-15)
        a /= a.sum()
        
        # Update running average
        a_avg = ((t * a_avg) + a) / (t + 1)
        
        # Track best
        sc_inner = compute_score(a_inner)
        sc_avg = compute_score(a_avg)
        
        if sc_inner < best_score:
            best_score = sc_inner
            best_a = a_inner.copy()
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = a_avg.copy()
        
        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"  Hedge2 t={t}, inner={sc_inner:.8f}, avg={sc_avg:.8f}, "
                  f"best={best_score:.8f}, eta={eta:.4f}, elapsed={elapsed:.0f}s")
    
    return best_a, best_score


def interp_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n, endpoint=False)
    new_x = np.linspace(0, 1, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def check_and_update(a, label, best_seq, best_score, verbose=True):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score
