Multi-phase optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.

import numpy as np
import time
import cvxpy as cp
from scipy.linalg import toeplitz
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_conv_matrix(f):
    """Build (2n-1) x n dense matrix C such that C @ g = f * g (linear conv)."""
    n = len(f)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)

    n = len(f)

    if prev_g is not None:

    C = build_conv_matrix(f)
    constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL, warm_start=(prev_g is not None))
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return prev_g if prev_g is not None else f.copy()
    if g.value is None:
        return prev_g if prev_g is not None else f.copy()
    return np.maximum(g.value, 0)


def lp_fictitious_play(a0, time_limit, step0=0.5, decay=0.995, verbose=True, label="LP-FP"):
    """LP fictitious play with exponentially decaying step size."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    prev_g = None
    stall_count = 0
    step = step0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        M = conv.max()

        g = solve_best_response(a, M, prev_g=prev_g)
        prev_g = g
        g_sum = g.sum()

        if g_sum < 1e-10:
            break

        g_norm = g / g_sum
        a_new = (1 - step) * a + step * g_norm
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss

        score = compute_score(a_new)
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1

        a = a_new
        step = max(0.02, step * decay)

        if stall_count > 200:
            rng = np.random.default_rng(iters + 7)
            a = best_a + rng.standard_normal(n) * 0.001
            a = np.maximum(a, 0)
            a = a / a.sum()
            prev_g = None
            stall_count = 0
            step = max(step, 0.1)

        if verbose and iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, best={best_score:.8f}, step={step:.4f}, "
                  f"time={elapsed:.0f}s/{time_limit:.0f}s")

    return best_a, best_score


def lp_fictitious_play_decreasing(a0, time_limit, alpha=0.02, verbose=True, label="LP-dec"):
    """LP fictitious play with 1/(1+alpha*k) step schedule."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    prev_g = None

    while time.monotonic() - t0 < time_limit:
        iters += 1
        step = 1.0 / (1.0 + alpha * iters)
        conv = fast_convolve(a)
        M = conv.max()

        g = solve_best_response(a, M, prev_g=prev_g)
        prev_g = g
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g_norm = g / g_sum
        a_new = (1 - step) * a + step * g_norm
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        score = compute_score(a_new)
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        if verbose and iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, best={best_score:.8f}, step={step:.6f}, "
                  f"time={elapsed:.0f}s/{time_limit:.0f}s")

    return best_a, best_score


def softmax_from_z(z):
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-30))


def lse_objective_and_grad_z(z, T, n):
    """Compute LSE-smoothed objective and gradient in z-space."""
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    # LSE of conv
    max_c = conv.max()
    shifted = conv - max_c
    exp_shifted = np.exp(shifted / T)
    lse_val = max_c + T * np.log(exp_shifted.sum())
    # Weights
    w = exp_shifted / exp_shifted.sum()
    # grad of LSE w.r.t. a: C^T @ w
    # C^T @ w can be computed as: for each j, grad_j = sum_k a_k * w_{j+k}
    # This equals conv(w, a_reversed)[n-1 : 2n-1]
    a_rev = a[::-1]
    # Use FFT for the convolution w * a_rev
    size = 2 * n - 1
    fft_size = 1 << (int(np.log2(size + n - 1)) + 1) if (size + n - 1) > 1 else 2
    # More safely:
    need = len(w) + len(a_rev) - 1
    fft_size = 1 << (need - 1).bit_length()
    W = rfft(w, fft_size)
    AR = rfft(a_rev, fft_size)
    conv_wa = irfft(W * AR, fft_size)
    grad_a = conv_wa[n - 1: 2 * n - 1]  # length n

    # Full gradient of score = 2n * lse_val (with sum(a)=1 via softmax)
    # d(score)/dz_j = a_j * (d(score)/da_j - sum_k a_k * d(score)/da_k)
    # d(score)/da_j = 2n * grad_a_j (since lse_val doesn't depend on a directly in this form)
    # Actually score = 2n * lse_val, and lse_val is a function of a.
    # d(lse_val)/da = grad_a (as computed above, which is dLSE/da)
    # So d(score)/da = 2n * grad_a
    # In z-space: d(score)/dz_j = a_j * (2n * grad_a_j - sum_k a_k * 2n * grad_a_k)
    S = float(np.dot(grad_a, a))
    grad_z = a * (2.0 * n * (grad_a - S))
    obj = 2.0 * n * lse_val
    return obj, grad_z


def adam_optimize(a0, time_limit, T0=200.0, T_end=50.0, lr=0.01, betas=(0.9, 0.999),
                  eps=1e-8, verbose=True, label="ADAM"):
    """ADAM optimization in log-space with LSE smoothing and temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    best_a = a0.copy()
    best_score = compute_score(a0)
    t0 = time.monotonic()
    iters = 0
    m = np.zeros(n)
    v = np.zeros(n)

    total_budget = time_limit
    while time.monotonic() - t0 < total_budget:
        iters += 1
        frac = (time.monotonic() - t0) / total_budget
        T = T0 * (T_end / T0) ** frac
        T = max(T_end, T)

        obj, grad_z = lse_objective_and_grad_z(z, T, n)

        m = betas[0] * m + (1 - betas[0]) * grad_z
        v = betas[1] * v + (1 - betas[1]) * grad_z ** 2
        m_hat = m / (1 - betas[0] ** iters)
        v_hat = v / (1 - betas[1] ** iters)

        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        a = softmax_from_z(z)
        score = compute_score(a)

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()

        if verbose and iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, T={T:.1f}, score={score:.8f}, "
                  f"best={best_score:.8f}, time={elapsed:.0f}s/{total_budget:.0f}s")

    return best_a, best_score


def lbgfs_optimize(a0, time_limit, T=500, maxiter=5000):
    """L-BFGS on smoothed log-sum-exp objective in log-space."""
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        obj, _ = lse_objective_and_grad_z(z, T, n)
        return obj

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        _, g = lse_objective_and_grad_z(z, T, n)
        return g

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 50, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
        score = compute_score(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except Exception as e:
        print(f"  L-BFGS error: {e}")
    return best_a, best_score


def coordinate_descent(a0, time_limit, n_pairs=300, verbose=True):
    """Coordinate descent with pairwise transfers."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(54321)
    iters = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue

            best_local = best_score
            best_t = 0.0
            base = min(a[i], a[j]) if a[j] > 1e-15 else a[i]
            if base < 1e-15:
                continue

            for frac in [0.05, 0.1, 0.25, 0.5]:
                t = frac * base
                a_trial = a.copy()
                a_trial[i] -= t
                a_trial[j] += t
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score(a_trial)
                if sc < best_local - 1e-14:
                    best_local = sc
                    best_t = t

            if best_t != 0.0:
                a[i] -= best_t
                a[j] += best_t
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = best_local
                best_a = a.copy()
                improved = True
                no_improve = 0
            else:
                no_improve += 1

        if not improved:
            no_improve += n_pairs

        if no_improve > 5000:
            noise = rng.standard_normal(n) * 0.003
            a = best_a.copy() + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            no_improve = 0

        if verbose and iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"  CD iter={iters}, best={best_score:.8f}, time={elapsed:.0f}s")

    return best_a, best_score


def bussgang_iteration(a0, time_limit, tau=0.3, verbose=True):
    """Bussgang-type spectral flattening."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        A = rfft(a, fft_size)
        power = np.abs(A) ** 2
        total_power = power.sum()
        target = total_power / fft_size
        ratio = np.where(power > 1e-30, target / power, 1.0)
        ratio = np.clip(ratio, 0.05, 20.0)
        A_new = A * (ratio ** tau)
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        score = compute_score(a_new)
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        if verbose and iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"  Bussgang iter={iters}, best={best_score:.8f}, time={elapsed:.0f}s")

    return best_a, best_score


def check_and_update(a, label, best_seq, best_score, verbose=True):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. LP-FP with exponential decay (main driver)
    2. ADAM in log-space with LSE smoothing
    3. LP-FP with decreasing step (convergence)
    4. Bussgang spectral flattening
    5. L-BFGS polish
    6. Coordinate descent
    7. Try n=1024
    8. Final polish
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 35.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"Parent load error: {e}")

    if parent is None:
        n_init = 768
        parent = np.ones(n_init) / n_init
        best_seq = parent.copy()
        best_score = compute_score(parent)
        print(f"Initialized uniform: n={n_init}, score={best_score:.8f}")

    n_parent = len(parent)
    print(f"\nStarting optimization. Best={best_score:.8f}, n={n_parent}")

    # Phase 1: LP-FP with exponential decay (strong step initially)
    print(f"\n=== Phase 1: LP-FP exp decay step0=0.5 (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.28, 450.0)
    if rem > 60:
        print(f"  Budget: {t1:.0f}s, n={n_parent}")
        a_lp1, sc_lp1 = lp_fictitious_play(best_seq, t1, step0=0.5, decay=0.997, label="LP-exp")
        best_seq, best_score = check_and_update(a_lp1, "lp_exp", best_seq, best_score)

    # Phase 2: ADAM in log-space (key new addition)
    print(f"\n=== Phase 2: ADAM log-space (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t2 = min(rem * 0.22, 350.0)
        print(f"  Budget: {t2:.0f}s")
        a_adam, sc_adam = adam_optimize(best_seq, t2, T0=500.0, T_end=20.0, lr=0.005,
                                        verbose=True, label="ADAM")
        best_seq, best_score = check_and_update(a_adam, "adam", best_seq, best_score)

    # Phase 3: LP-FP with 1/(1+alpha*k) schedule (for fine convergence)
    print(f"\n=== Phase 3: LP-FP decreasing step (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t3 = min(rem * 0.18, 300.0)
        print(f"  Budget: {t3:.0f}s")
        a_lp3, sc_lp3 = lp_fictitious_play_decreasing(best_seq, t3, alpha=0.01, label="LP-dec")
        best_seq, best_score = check_and_update(a_lp3, "lp_dec", best_seq, best_score)

    # Phase 4: Bussgang spectral flattening
    print(f"\n=== Phase 4: Bussgang (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t4 = min(rem * 0.06, 60.0)
        print(f"  Budget: {t4:.0f}s")
        a_bg, sc_bg = bussgang_iteration(best_seq, t4, tau=0.3, verbose=True)
        best_seq, best_score = check_and_update(a_bg, "bussgang", best_seq, best_score)

    # Phase 5: L-BFGS polish at multiple temperatures
    print(f"\n=== Phase 5: L-BFGS Polish (best={best_score:.8f}) ===")
    for T_val in [100, 500]:
        rem = deadline - time.monotonic()
        t5 = min(rem * 0.08, 60.0)
        if rem < 30:
            break
        print(f"  L-BFGS T={T_val}, budget={t5:.0f}s")
        a_lb, sc_lb = lbgfs_optimize(best_seq, t5, T=T_val, maxiter=3000)
        best_seq, best_score = check_and_update(a_lb, f"lbgfs_T{T_val}", best_seq, best_score)

    # Phase 6: Coordinate descent (fine-tuning)
    print(f"\n=== Phase 6: Coordinate Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t6 = min(rem * 0.12, 100.0)
        print(f"  Budget: {t6:.0f}s")
        a_cd, sc_cd = coordinate_descent(best_seq, t6, n_pairs=200, verbose=True)
        best_seq, best_score = check_and_update(a_cd, "coord_desc", best_seq, best_score)

    # Phase 7: Try n=1024 (larger sequence)
    print(f"\n=== Phase 7: Try n=1024 (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 120 and n_parent != 1024:
        t7 = min(rem * 0.15, 150.0)
        init_1024 = interpolate_sequence(best_seq, 1024)
        print(f"  Interpolated to n=1024, initial score={compute_score(init_1024):.8f}")
        # Quick LP-FP on n=1024
        a_1024, sc_1024 = lp_fictitious_play(init_1024, t7 * 0.6, step0=0.3, decay=0.998,
                                              label="LP-1024")
        best_seq, best_score = check_and_update(a_1024, "lp_n1024", best_seq, best_score)
        # ADAM on n=1024
        rem2 = deadline - time.monotonic()
        if rem2 > 30:
            t7b = min(rem2 * 0.4, 80.0)
            a_1024b, sc_1024b = adam_optimize(a_1024, t7b, T0=300.0, T_end=30.0, lr=0.003,
                                               label="ADAM-1024")
            best_seq, best_score = check_and_update(a_1024b, "adam_n1024", best_seq, best_score)

    # Phase 8: LP-FP with small step on current best (final polish)
    print(f"\n=== Phase 8: Final LP polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t8 = min(rem * 0.25, 100.0)
        print(f"  Budget: {t8:.0f}s, n={len(best_seq)}")
        a_lp8, sc_lp8 = lp_fictitious_play(best_seq, t8, step0=0.15, decay=0.999, label="LP-fin")
        best_seq, best_score = check_and_update(a_lp8, "lp_final", best_seq, best_score)

    # Phase 9: One more ADAM pass on the best found
    print(f"\n=== Phase 9: Final ADAM (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t9 = min(rem * 0.35, 80.0)
        print(f"  Budget: {t9:.0f}s")
        a_adam2, sc_adam2 = adam_optimize(best_seq, t9, T0=50.0, T_end=10.0, lr=0.002,
                                          label="ADAM-fin")
        best_seq, best_score = check_and_update(a_adam2, "adam_final", best_seq, best_score)

    # Final verification
    if best_seq is not None:
        final_score = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(final_score) and final_score < best_score:
            best_score = final_score
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(x) for x in best_seq]
