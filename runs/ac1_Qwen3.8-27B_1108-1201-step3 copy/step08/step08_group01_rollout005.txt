def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    import cvxpy as cp
    if solver is None:

    M_var = cp.Variable()
    
    # Build constraint: T @ b <= M, where T[k,j] = a[k-j] if valid
    # Instead of explicitly building T (which is (2n-1) x n), use the convolution structure
    # conv(a, b)[k] = sum_j a[k-j] * b[j] for valid j
    
    # For efficiency, build the sparse matrix
    from scipy.sparse import coo_matrix
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for k in range(j, j + n):
            if k < 2 * n - 1:
                rows.append(k)
                cols.append(j)
                vals.append(a_fixed[k - j])
    
    T = coo_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    T = T.tocsr()
    
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]

        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def subgradient_descent(a0, time_limit, lr_init=0.05, verbose=False, label=""):
    """
    Subgradient descent for min max_k conv(a,a)[k] s.t. sum(a)=1, a>=0.
    Uses Polyak step size and averaging.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    conv = fast_convolve(a)
    best_M = float(conv.max())
    best_score = 2.0 * n * best_M
    
    t0 = time.monotonic()
    it = 0
    avg_a = a.copy()
    
    # Lower bound: by pigeonhole, max(conv) >= 1/(2n-1) (since sum(conv) = 1)
    lower_bound_M = 1.0 / (2*n - 1)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = float(conv.max())
        score = 2.0 * n * M
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
        
        # Subgradient: 2 * a[peak - j] for the peak position
        peak_k = int(np.argmax(conv))
        # Multiple peaks for more stable gradient
        top_k = min(10, 2*n-1)
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        temp = 0.005 * max(pv.max(), 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        ws = w.sum()
        w = w / ws
        
        grad = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            j = np.arange(i_start, i_end + 1)
            grad[j] += w[idx] * a[k - j]
        grad *= 2.0
        
        # Polyak step size: (f(x) - f*) / ||g||^2
        gap = M - lower_bound_M
        grad_norm_sq = float(np.dot(grad, grad))
        if grad_norm_sq < 1e-30:
            break
        lr = min(lr_init, gap / grad_norm_sq)
        
        # Update: a_new = a - lr * grad, then project to simplex
        a_raw = a - lr * grad
        a_new = simplex_project(a_raw)
        
        avg_a = (it * avg_a + a_new) / (it + 1)
        a = a_new
        
        # Check averaged iterate
        avg_conv = fast_convolve(avg_a)
        avg_M = float(avg_conv.max())
        avg_score = 2.0 * n * avg_M
        if avg_score < best_score - 1e-14:
            best_score = avg_score
            best_a = avg_a.copy()
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad {label} it={it} ({elapsed:.0f}s): best={best_score:.6f} M={M:.8f} lr={lr:.6f}")
    
    return best_a, best_score, it

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def init_tapered(n, rng, beta=1.0):
    """Initialize with a power-law taper (more mass in the middle or edges)."""
    t = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    # beta > 0: more mass at center, beta < 0: more mass at edges
    dist = np.abs(t - center) / center
    a = (1.0 - dist) ** beta
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def check_and_update(a, best_score, label=""):

        return a_norm.copy(), best_score
