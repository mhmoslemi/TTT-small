    # We want c_i = sum_j a_j * w_{j+i}
    # Set len(w)-1-m+j = j+i => len(w)-1-m = i => m = len(w)-1-i
    return conv_result[len(w) - 1 : len(w) - 1 + n]


def softmax_z(z):
    """Convert log-domain to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def lbgfs_logsumexp(a0, time_limit, T_start=5000.0, T_end=5.0, n_stages=25, maxiter_per_stage=200):
    """
    Aggressive L-BFGS with log-sum-exp smoothing and geometric temperature annealing.
    """
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()

    def make_fun(T_val):
        def fun(z_vec):
            a = softmax_z(z_vec)
            b = fast_convolve(a)
            Tb = T_val * b
            m = Tb.max()
            val = m + np.log(np.exp(Tb - m).sum())
            return (2.0 * n / T_val) * val
        return fun

    def make_jac(T_val):
        def jac(z_vec):
            a = softmax_z(z_vec)
            b = fast_convolve(a)
            Tb = T_val * b
            m = Tb.max()
            exp_Tb = np.exp(Tb - m)
            w = exp_Tb / exp_Tb.sum()
            # c_i = sum_k w_k * a_{k-i}
            c = cross_corr_w_a(w, a)
            g = -2.0 * c
            S = np.dot(g, a)
            return a * (g - S)
        return jac

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        try:
            res = minimize(
                fun=make_fun(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={
                    'maxiter': maxiter_per_stage,
                    'ftol': 1e-20,
                    'gtol': 1e-14,
                    'maxfun': maxiter_per_stage * 200,
                    'disp': False
                }
            )
            z = res.x
        except Exception:
            pass

        a = softmax_z(z)
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    L-BFGS stage {si}/{n_stages} T={stage_T:.1f} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def subgradient_peak_deflation(a0, time_limit, n_steps=200000):
    """
    Fast subgradient method focused on reducing the convolution peak.
    Uses adaptive step size and active set (near-max) indices.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    lr = 0.05
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        # Active set: indices where b is within 1% of max
        threshold = max_b * 0.995
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])

        # Subgradient: g_i = sum_{k in active} 2*a_{k-i} (with boundary handling)
        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * a[k - idx]
        g /= len(active)

        # Step
        a_new = a - lr * g
        # Project to simplex
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score_new = 2.0 * n * fast_convolve(a_new).max()

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive step size
        if no_improve >= 30:
            lr *= 0.8
            no_improve = 0
        if no_improve >= 200:
            # Stuck, try a perturbation
            rng = np.random.default_rng(step)
            a_pert = a + 0.01 * rng.standard_normal(n)
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            a = a_pert
            lr = 0.03
            no_improve = 0

        if step % 50000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad {elapsed:.0f}s: score={best_score:.6f} lr={lr:.6f}")

    return best_a, best_score


def peak_shave_local(a0, time_limit, n_steps=50000):
    """
    Local peak shaving: identify top contributors to the peak and reduce them.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        b_work = b.copy()
        if len(b_work) > 2:
            b_work[0] = -np.inf
            b_work[-1] = -np.inf
        peak_k = int(np.argmax(b_work))
        peak_val = b_work[peak_k]

        # Find contributors to this peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        total_contrib = contribs.sum()
        if total_contrib < 1e-15:
            break

        # Top 3 contributors
        top = np.argsort(contribs)[::-1][:3]

        improved = False
        for ci in top:
            i = idx[ci]
            # Try reducing a[i] by various amounts
            for factor in [0.97, 0.94, 0.90, 0.85]:
                a_trial = a.copy()
                diff = a_trial[i] * (1 - factor)
                if diff < 1e-15:
                    continue
                a_trial[i] -= diff
                # Redistribute to other elements proportionally
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] += diff * a_trial[mask] / others_sum
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try pair modifications
            for ci1 in top[:2]:
                for ci2 in top[:2]:
                    if ci1 == ci2:
                        continue
                    i = idx[ci1]
                    j = idx[ci2]
                    for f1 in [0.95, 0.90]:
                        for f2 in [1.05, 1.02]:
                            a_trial = a.copy()
                            d1 = a_trial[i] * (1 - f1)
                            d2 = a_trial[j] * (f2 - 1)
                            a_trial[i] = a[i] * f1
                            a_trial[j] = a[j] * f2
                            # Compensate sum
                            net = d2 - d1
                            if abs(net) > 1e-15:
                                mask = np.ones(n, dtype=bool)
                                mask[i] = False
                                mask[j] = False
                                others_sum = a_trial[mask].sum()
                                if others_sum > 1e-15:
                                    a_trial[mask] -= net * a_trial[mask] / others_sum
                            a_trial = np.maximum(a_trial, 0)
                            ss = a_trial.sum()
                            if ss > 1e-15:
                                a_trial = a_trial / ss
                                score = 2.0 * n * fast_convolve(a_trial).max()
                                if score < best_score - 1e-14:
                                    a = a_trial
                                    best_score = score
                                    best_a = a.copy()
                                    improved = True
                                    break
                        if improved:
                            break
                if improved:
                    break

        if not improved:
            break

    return best_a, best_score


def nesterov_alt_lp(a0, time_limit, n_max_iters=5000, tolerance=1e-8):
    """
    Fast alternating subgradient method (dual of the LP).
    Instead of solving LPs, uses multiplicative weights on the dual.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    # Dual variables: probability distribution y over [0, 2n-2]
    m = 2 * n - 1
    y = np.ones(m) / m
    eta = 0.1

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Given y, the optimal b minimizes max_k (a*b)_k
        # Actually, let's do: given a, find b to minimize max_k (a*b)_k
        # The dual is: max mu s.t. sum(y)=1, (T^T y)_j <= mu, y>=0
        # (T^T y)_j = sum_k y_k * a_{k-j}
        # Subgradient method on the dual:
        # mu = min_j (T^T y)_j
        # To increase mu, increase y on k's that contribute to the min j's

        TTy = cross_corr_w_a(y, a)  # (T^T y)_j = sum_k y_k a_{k-j}
        # Note: this gives (T^T y)_j for j=0..n-1
        min_val = TTy.min()
        min_idx = np.where(TTy <= min_val + 1e-6)[0]

        # Update y: increase weight on k's that help the min indices
        # (T^T y)_j = sum_k y_k * a_{k-j}
        # To increase (T^T y)_{j*}, we need to increase y_k for k where a_{k-j*} is large
        for jstar in min_idx[:3]:
            # a_{k-j*} is large when k-j* is in the high-value region of a
            for k in range(len(a) + jstar):
                if k < m and k - jstar >= 0 and k - jstar < n:
                    y[k] *= (1 + eta * a[k - jstar])
        y = np.maximum(y, 1e-15)
        y = y / y.sum()

        # Recover b from y (complementary slackness: b_j > 0 only when (T^T y)_j = mu)
        # Simple: b is proportional to the "slack" - use multiplicative update
        # b_j should be large where (T^T y)_j is close to mu (tight constraints)
        slack = min_val - TTy + abs(min_val) + 1e-10
        b_new = slack ** 2  # or exp based
        b_new = np.maximum(b_new, 0)
        sb = b_new.sum()
        if sb < 1e-15:
            b_new = np.ones(n) / n
        else:
            b_new = b_new / sb

        # Combine: a_new = average of a and b_new
        alpha = 2.0 / (it + 2)
        a = (1 - alpha) * a + alpha * b_new
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = b_new.copy()
        else:
            a = a / sa

        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()

        # Decay eta
        if it % 1000 == 0 and it > 0:
            eta *= 0.8
            elapsed = time.monotonic() - t0
            print(f"    AltLP iter={it} ({elapsed:.0f}s): score={best_score:.6f} eta={eta:.4f}")

    return best_a, best_score


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) < 1:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Create structured initial sequence."""
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    elif choice == 2:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 3:
        # Two-peaked
        x = np.linspace(0, 1, n)
        a = np.exp(-20 * (x - 0.3) ** 2) + np.exp(-20 * (x - 0.7) ** 2)
    elif choice == 4:
        # Edge-concentrated
        x = np.linspace(0, 1, n)
        a = x ** 0.5 + (1 - x) ** 0.5
    else:
        # Slightly asymmetric
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2 * (0.7 + 0.3 * x)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization focusing on fast L-BFGS with log-sum-exp,
    subgradient peak deflation, and adaptive multi-length search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 960
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # === Phase 1: Aggressive L-BFGS (main workhorse, 50% of time) ===
    print(f"\n=== Phase 1: L-BFGS LogSumExp n={n_best} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = rem * 0.50
    print(f"  Budget: {t_alloc:.0f}s")
    a1, _ = lbgfs_logsumexp(best_seq, t_alloc, T_start=8000.0, T_end=3.0, n_stages=30, maxiter_per_stage=300)
    try_update(a1, "lbgfs_p1")

    # === Phase 2: Second L-BFGS pass with different params (15% of time) ===
    print(f"\n=== Phase 2: L-BFGS r2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = rem * 0.15
    if t_alloc > 10:
        print(f"  Budget: {t_alloc:.0f}s")
        # Slightly different temperature range
        a2, _ = lbgfs_logsumexp(best_seq, t_alloc, T_start=2000.0, T_end=1.0, n_stages=20, maxiter_per_stage=200)
        try_update(a2, "lbgfs_p2")

    # === Phase 3: Subgradient peak deflation (10% of time) ===
    print(f"\n=== Phase 3: Subgradient Deflation (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = rem * 0.10
    if t_alloc > 10:
        print(f"  Budget: {t_alloc:.0f}s")
        a3, _ = subgradient_peak_deflation(best_seq, t_alloc, n_steps=300000)
        try_update(a3, "subgrad_p3")

    # === Phase 4: Peak shaving local search (8% of time) ===
    print(f"\n=== Phase 4: Peak Shaving (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = rem * 0.08
    if t_alloc > 5:
        print(f"  Budget: {t_alloc:.0f}s")
        a4, _ = peak_shave_local(best_seq, t_alloc, n_steps=100000)
        try_update(a4, "peakshave_p4")

    # === Phase 5: Another L-BFGS refinement (7% of time) ===
    print(f"\n=== Phase 5: L-BFGS refine (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = rem * 0.07
    if t_alloc > 5:
        print(f"  Budget: {talloc if False else t_alloc:.0f}s")
        a5, _ = lbgfs_logsumexp(best_seq, t_alloc, T_start=500.0, T_end=0.5, n_stages=15, maxiter_per_stage=200)
        try_update(a5, "lbgfs_p5")

    # === Phase 6: Alternative lengths (7% of time) ===
    print(f"\n=== Phase 6: Alternative Lengths (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        alt_lengths = []
        for mult in [1.25, 1.5, 2.0, 0.75]:
            alt_n = int(n_best * mult)
            if 256 <= alt_n <= 4096 and alt_n != n_best:
                alt_lengths.append(alt_n)
        alt_lengths = sorted(set(alt_lengths))[:2]

        for alt_n in alt_lengths:
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            t_total = min(rem * 0.45, 200.0)
            if t_total < 20:
                break
            print(f"  Trying n={alt_n}, budget={t_total:.0f}s")

            init_alt = interpolate_sequence(best_seq, alt_n)
            a_alt, _ = lbgfs_logsumexp(init_alt, t_total, T_start=5000.0, T_end=2.0, n_stages=20, maxiter_per_stage=200)
            try_update(a_alt, l"lbgfs_n{alt_n}".replace("l", "f"))

    # === Phase 7: Final polish with very low temperature ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t_fin = min(rem * 0.5, 60.0)
        print(f"  Budget: {t_fin:.0f}s")
        a_fin, _ = lbgfs_logsumexp(best_seq, t_fin, T_start=200.0, T_end=0.3, n_stages=10, maxiter_per_stage=200)
        try_update(a_fin, "final_polish")

    # === Phase 8: Very final peak shaving ===
    rem = deadline - time.monotonic()
    if rem > 10:
        t_ps = min(rem * 0.5, 30.0)
        a_ps, _ = peak_shave_local(best_seq, t_ps, n_steps=50000)
        try_update(a_ps, "final_ps")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
