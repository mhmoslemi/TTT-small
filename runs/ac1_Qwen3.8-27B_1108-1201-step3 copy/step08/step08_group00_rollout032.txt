def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


_lp_solver_cached = None

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def lbgfs_refine(a0, time_limit, T_start=100.0, T_end=0.3, maxiter=10000):
    """L-BFGS with LogSumExp smoothing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    def make_func(T_val):
        def func(z_val):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z_val)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(z_val):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_val)
            a = softmax_from_z(z_val)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    n_stages = 8
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(50, maxiter // n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)

def spectral_shaping(a0, time_limit, n_iter=500):
    """Spectral shaping: try to flatten the power spectrum while maintaining non-negativity."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a, 2.0 * n
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()
    
    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        A = rfft(a, fft_size)
        power = np.abs(A) ** 2
        
        # Target: flatten the power spectrum
        # Use a "soft" target: geometric mean of current power
        log_power = np.log(np.maximum(power, 1e-30))
        target_log_power = np.full_like(log_power, np.median(log_power))
        
        # Interpolate towards flat spectrum
        blend = 0.05
        new_log_power = (1 - blend) * log_power + blend * target_log_power
        new_power = np.exp(new_log_power)
        
        # Need to find a non-negative sequence with this power spectrum
        # This is an inverse problem; use iterative projection
        # Start from current a
        a_curr = a.copy()
        for proj_iter in range(20):
            # FFT
            A_curr = rfft(a_curr, fft_size)
            # Scale to match target power
            magnitude_target = np.sqrt(new_power)
            phase_curr = A_curr / np.maximum(np.abs(A_curr), 1e-30)
            A_new = phase_curr * magnitude_target
            # Inverse FFT
            a_new = irfft(A_new, fft_size)[:n]
            # Project onto non-negative simplex
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                break
            a_new = a_new / s_new
            a_curr = a_new
        
        a = a_curr
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def coordinate_descent_polish(a0, time_limit, n_rounds=200000):
    """Coordinate descent: try tiny changes to individual coordinates."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(123)

    # Pre-compute which coordinates matter most
    b = fast_convolve(a)
    top_peaks = np.argsort(b)[::-1][:8]

    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        if rnd % 5000 == 0 and rnd > 0:
            # Periodically refresh peak positions
            b = fast_convolve(a)
            top_peaks = np.argsort(b)[::-1][:8]

        # Pick a coordinate to try (weighted by contribution to peaks)
        if rnd % 3 == 0:
            # Targeted: pick from top contributors
            peak_k = top_peaks[rnd % len(top_peaks)]
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            candidates = np.arange(js, je + 1)
            contribs = a[candidates] * a[peak_k - candidates]
            i = candidates[np.argmax(contribs)]
        else:
            # Random
            i = rng.integers(0, n)

        if a[i] < 1e-12:
            continue

        # Try both increasing and decreasing
        for direction in [-1, 1]:
            for mag in [0.001, 0.005, 0.01, 0.02]:
                diff = direction * mag * a[i]
                a_trial = a.copy()
                a_trial[i] += diff
                if a_trial[i] < 0:
                    continue
                # Redistribute
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    continue
                deficit = a.sum() - a_trial.sum()
                if abs(deficit) > 1e-10:
                    a_trial[mask] += deficit * a_trial[mask] / others_sum
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                a_trial = a_trial / a_trial.sum()

                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    break

    return best_a, best_score

            if a[i] < 1e-14:
                continue
            for factor in [0.995, 0.99, 0.98, 0.96, 0.93]:
                diff = a[i] * (1 - factor)
                if diff < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[i] -= diff
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                if others.sum() < 1e-15:
                    continue
                a_trial[mask] = others + diff * others / others.sum()
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try pair modification
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_contrib = np.argsort(contribs)[::-1][:4]
            for ci in top_contrib:
                i = idx[ci]
                j = peak_k - i
                if i == j or j < 0 or j >= n:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.99, 0.97]:
                    a_trial = a.copy()
                    a_trial[i] *= f1
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            # Simultaneous tiny reduction
            a_trial = a.copy()
            for i in top_coords[:4]:
                if a[i] > 1e-14:
                    a_trial[i] *= 0.998
            a_trial = a_trial / a_trial.sum()
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-15:
                a = a_trial
                best_score = score
                best_a = a.copy()
            elif step > 10000:
                break

    return best_a, best_score
