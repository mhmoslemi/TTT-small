```python
def smoothed_gradient(a, w, n):
    """Compute gradient of sum(w * b) where b = conv(a, a)."""
    # grad_i = 2 * sum_j a[j] * w[i+j] for j=0..n-1
    # This is 2 * cross_correlation at positive lags
    N = 1 << (3*n-1).bit_length()  # next power of 2 >= 3n-2
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    # Cross-correlation: ifft(conj(A) * W) gives... 
    # Actually let me use: ifft(A * conj(W)) 
    # For real signals, conj(rfft) is just flipping the imaginary parts
    # Let me use full FFT for clarity
    A_full = np.fft.fft(a, N)
    W_full = np.fft.fft(w, N)
    R = np.fft.ifft(A_full * np.conj(W_full)).real
    # X[i] = R[(N-i) % N] for i=0..n-1
    # X[0] = R[0], X[1] = R[N-1], ..., X[n-1] = R[N-n+1]
    X = np.zeros(n)
    X[0] = R[0]
    if n > 1:
        X[1:] = R[N-n+1:N][::-1]
    return 2.0 * X
```

"""
Smoothed gradient optimization with FFT-based gradient computation for minimizing
2*n*max(conv(a,a))/(sum(a))^2. Uses log-sum-exp smoothing, Nesterov momentum,
adaptive step sizes, and hybrid LP polishing. Key innovation: efficient O(n log n)
gradient computation via cross-correlation FFT, enabling millions of iterations.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time
import sys


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient(a, w):
    """
    Compute gradient of sum(w * conv(a,a)) w.r.t. a.
    grad_i = 2 * sum_{j=0}^{n-1} a[j] * w[i+j]
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    # Need N >= (2n-1) + n - 1 = 3n-2 for linear cross-correlation
    N = 1 << (3 * n - 2 + 1).bit_length()
    A = fft(a, N)
    W = fft(w, N)
    R = ifft(A * np.conj(W)).real
    # X[i] = R[(N-i) % N] for i=0..n-1
    # X[0] = R[0], X[1] = R[N-1], X[2] = R[N-2], ..., X[n-1] = R[N-n+1]
    X = np.empty(n)
    X[0] = R[0]
    if n > 1:
        X[1:] = R[N - n + 1 : N][::-1]
    return 2.0 * X


def compute_gradient_direct(a, w):
    """Direct O(n^2) gradient computation for verification."""
    n = len(a)
    grad = np.zeros(n)
    for i in range(n):
        s = 0.0
        for j in range(n):
            k = i + j
            if k < len(w):
                s += a[j] * w[k]
        grad[i] = 2.0 * s
    return grad


def softmax_weights(b, tau):
    """Compute softmax(b/tau) for peak weighting."""
    shifted = b / tau - (b.max() / tau)
    w = np.exp(shifted)
    s = w.sum()
    if s > 0:
        return w / s
    w = np.zeros(len(b))
    w[np.argmax(b)] = 1.0
    return w


def recursive_init(n):
    """
    Initialize using the recursive flat-convolution construction.
    a_0 = 1/sqrt(2n-1), a_k = max(0, [target - sum_{j=1}^{k-1} a_j*a_{k-j}]/(2*a_0))
    This gives an 'edge-heavy' shape that approximates flat convolution.
    """
    target = 1.0 / (2 * n - 1)
    a = np.zeros(n)
    a[0] = np.sqrt(target)
    for k in range(1, n):
        s = 0.0
        for j in range(1, k):
            if j < n and (k - j) < n:
                s += a[j] * a[k - j]
        val = (target - s) / (2.0 * a[0])
        a[k] = max(0.0, val)
    total = a.sum()
    if total > 1e-15:
        a = a / total
    else:
        a = np.ones(n) / n
    return a


def edge_heavy_init(n, alpha=2.0):
    """Edge-heavy initialization: more mass near boundaries."""
    j = np.arange(n, dtype=np.float64)
    a = (j + 1.0) ** alpha * (n - j) ** alpha
    a = np.maximum(a, 1e-10)
    a = a / a.sum()
    return a


def smooth_gradient_optimize(a_init, time_limit, tau_init=0.005, tau_min=0.0005,
                              lr_init=0.01, momentum=0.92, label="sg"):
    """
    Main optimization using smoothed gradient with momentum and adaptive step size.
    Minimize log(sum(exp(b_k/tau))) as a smooth proxy for max(b_k).
    """
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-15:
        a = a / s

    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    best_score = 2.0 * n * best_obj

    v = np.zeros(n)  # momentum
    tau = tau_init
    lr = lr_init
    it = 0
    no_improve = 0
    full_len = 2 * n - 1

    while time.monotonic() - t0 < time_limit:
        it += 1

        b = fast_convolve(a)
        cur_obj = b.max()

        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_score = 2.0 * n * best_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Anneal temperature
        tau = max(tau_min, tau_init * (0.9995 ** it))

        # Compute weights (softmax over peaks)
        w = softmax_weights(b, tau)

        # Compute gradient via FFT
        grad = compute_gradient(a, w)

        # Adaptive learning rate based on gradient magnitude
        gnorm = np.sqrt(np.sum(grad * grad))
        if gnorm < 1e-15:
            break

        # Step size: balance between gradient descent and stability
        step = lr / (gnorm + 1e-10)
        step = min(step, 0.5)

        # Momentum update
        v = momentum * v - step * grad

        # Projected step
        a_trial = project_simplex(a + v)

        # Verify improvement (accept/reject with occasional exploration)
        b_trial = fast_convolve(a_trial)
        obj_trial = b_trial.max()

        if obj_trial < cur_obj + 1e-12:
            a = a_trial
        else:
            # Try smaller step
            v = 0.5 * v
            a_trial2 = project_simplex(a + v)
            b_trial2 = fast_convolve(a_trial2)
            if b_trial2.max() < cur_obj:
                a = a_trial2
            else:
                # Rejected, dampen
                v *= 0.5
                if no_improve > 50:
                    # Reset to best with perturbation
                    rng = np.random.default_rng(it)
                    a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
                    v = np.zeros(n)
                    no_improve = 0
                    lr = max(lr * 0.8, 0.001)

        # Adapt learning rate
        if no_improve == 0:
            lr = min(lr * 1.02, 0.1)
        elif no_improve % 30 == 0:
            lr = max(lr * 0.85, 0.002)

        # Periodic logging
        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} it={it} obj={best_obj:.10f} score={best_score:.6f} "
                  f"tau={tau:.5f} lr={lr:.5f} t={elapsed:.0f}s")

        # Restart if stuck too long
        if no_improve > 200:
            rng = np.random.default_rng(it * 7)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            v = np.zeros(n)
            no_improve = 0
            tau = max(tau * 1.5, tau_init * 0.5)  # Reheat

    return best_a, best_score


def lp_polish(a_init, time_limit, top_k=200):

def interp_sequence(src, target_n):
    """Interpolate a sequence to a new length (preserving shape)."""
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result

    return results
