    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)

"""
Fast FFT-based subgradient optimization with Nesterov momentum, adaptive peak
weighting, and multi-length search to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Key improvement over LP-based approach: O(n log n) per iteration enables 100x+
more iterations, allowing finer convergence to the optimum.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_subgradient_fft(f, W):
    """Compute weighted subgradient g_j = sum_k W_k * f[k-j] via FFT cross-correlation."""
    n = len(f)
    N = 1 << (3 * n - 1).bit_length()
    F = rfft(f, N)
    Wf = rfft(W, N)
    g_full = irfft(F * np.conj(Wf), N)
    return g_full[:n]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interp_sequence(src, target_n):
    """Interpolate a sequence to a new length, normalized to sum 1."""
    src = np.asarray(src, dtype=np.float64)
    src_n = len(src)
    if src_n == target_n:
        result = src.copy()
    else:
        old_x = np.linspace(0.0, 1.0, src_n)
        new_x = np.linspace(0.0, 1.0, target_n)
        result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result

            best_seq = f_res.copy()
            print(f"  NEW BEST [shave]: {real_score:.6f}")

    # Phase 4: Final fine-tuning with very small steps
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.5, 150.0)
    if t4 > 10:
        print(f"\n{'='*60}")
        print(f"Phase 4: Fine tune n={len(best_seq)}, budget={t4:.0f}s")
        print(f"{'='*60}")

        f_res, score_res = subgradient_optimize(
            best_seq, time_limit=t4, label="fine",
            T_init=0.001, T_min=0.00002,
            lr_init=0.001, momentum_init=0.92,
            n_peaks_frac_start=0.2, n_peaks_frac_end=0.8,
            restart_threshold=150
        )
        real_score = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_score) and real_score < best_score:
            best_score = real_score
            best_seq = f_res.copy()
            print(f"  NEW BEST [fine]: {real_score:.6f}")

    # Final verification
    fs = float(evaluate_sequence([float(max(0.0, x)) for x in best_seq]))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*60}")

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
