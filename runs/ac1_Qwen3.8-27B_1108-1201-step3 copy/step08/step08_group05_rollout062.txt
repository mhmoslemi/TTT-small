"""
Optimized step function search combining LP-based descent with constraint generation,
soft-max gradient polishing, perturbation-restart, and multi-length exploration.
Targets score < 1.5027 for the auto-convolution peak problem.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(f, g):
    """Cross-convolution f*g via FFT."""
    size = len(f) + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_solve(f, active_indices, M, n, max_solves=3):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)[k]<=M for k in active_indices."""
    K = len(active_indices)
    if K == 0 or K > n * 3:
        return None
    
    # Build constraint matrix
    A = np.zeros((K, n))
    for row_i, k in enumerate(active_indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]
    
    g_var = cp.Variable(n)
    prob = cp.Problem(
        cp.Maximize(cp.sum(g_var)),
        [g_var >= 0, A @ g_var <= M]
    )
    
    best_obj = -np.inf
    best_g = None
    
    for solver in [cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            start = time.monotonic()
            prob.solve(solver=solver)
            elapsed = time.monotonic() - start
            
            if g_var.value is not None and not np.any(np.isnan(g_var.value)):
                g = np.maximum(g_var.value, 0.0)
                obj = g.sum()
                if obj > best_obj:
                    best_obj = obj
                    best_g = g
            if best_g is not None and elapsed > 1.0:
                break  # Got a solution, don't waste time on others
        except Exception:
            continue
    
    return best_g

def soft_max_descend(f_init, time_limit, t_start=0.005, t_end=0.0001, label="sm"):
    """Soft-max gradient descent with Adam optimizer."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    f = f / f.sum()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    step = 0
    total_steps = 3000
    
    while time.monotonic() - t0 < time_limit:
        step += 1
        progress = min(step / total_steps, 1.0)
        temperature = t_end + (t_start - t_end) * 0.5 * (1 + np.cos(np.pi * progress))
        lr = 0.005 * (1 - 0.8 * progress) + 0.001
        
        b = fast_convolve(f)
        max_b = b.max()
        shifted = (b - max_b) / temperature
        shifted = np.clip(shifted, -80, 80)
        exp_s = np.exp(shifted)
        w_sum = exp_s.sum()
        if w_sum < 1e-30:
            weights = np.zeros(len(b))
            weights[b.argmax()] = 1.0
        else:
            weights = exp_s / w_sum
        
        # Compute gradient
        grad = np.zeros(n)
        grad += 2.0 * weights[0] * f
        for k in range(1, min(n, len(b))):
            w = weights[k]
            if w < 1e-12:
                continue
            grad[:n-k] += w * f[k:]
            grad[k:] += w * f[:n-k]
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** (step + 1))
        v_hat = v / (1 - beta2 ** (step + 1))
        
        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-15:
            f_new = f
        else:
            f_new = f_new / s
        
        f = f_new
        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_f = f.copy()
        
        if step % 300 == 0:
            print(f"    {label} step={step} temp={temperature:.6f} obj={best_obj:.12f} score={2*n*best_obj:.8f}")
    
    return best_f, best_obj


def focused_peak_reduction(f_init, time_limit, n_peaks=5, label="peak"):

        grad = np.zeros(n)

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(f, g):
    size = len(f) + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_solve(f, active_indices, M, n):

        return None
    if K > 5 * n:

        return None
    A = np.zeros((K, n))
    for row_i in range(K):
        k = int(active_indices[row_i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]
    g_var = cp.Variable(n)
    constraints = [g_var >= 0, A @ g_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)

                if not np.any(np.isnan(g)):
                    best_g = g
                    break
        except Exception:
            continue
    return best_g

def soft_max_descend(f_init, time_limit, t_start=0.004, t_end=0.00005, label="sm"):
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    f = f / f.sum()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    step = 0
    total_steps = 4000
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        step += 1
        progress = min(step / total_steps, 1.0)
        temperature = t_end + (t_start - t_end) * 0.5 * (1.0 + np.cos(np.pi * progress))
        lr = 0.003 * (1.0 - 0.7 * progress) + 0.0005

        b = fast_convolve(f)
        max_b = b.max()
        shifted = (b - max_b) / temperature
        shifted = np.clip(shifted, -80, 80)
        exp_s = np.exp(shifted)
        w_sum = exp_s.sum()
        if w_sum < 1e-30:
            weights = np.zeros(len(b))
            weights[int(b.argmax())] = 1.0
        else:
            weights = exp_s / w_sum

        # Gradient computation
        grad = np.zeros(n)
        grad += 2.0 * weights[0] * f
        max_k = min(n, len(b))
        for k in range(1, max_k):
            w = weights[k]
            if w < 1e-12:
                continue
            grad[:n - k] += w * f[k:]
            grad[k:] += w * f[:n - k]

        # Adam
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (step + 1))
        v_hat = v / (1.0 - beta2 ** (step + 1))
        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-15:
            f_new = f.copy()
        else:
            f_new = f_new / s
        f = f_new

        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                # Restart with small perturbation
                f = project_simplex(best_f + np.random.default_rng(step).standard_normal(n) * 0.005)
                m = np.zeros(n)
                v = np.zeros(n)
                no_improve = 0

        if step % 400 == 0:
            print(f"    {label} step={step} temp={temperature:.6f} obj={best_obj:.12f} score={2*n*best_obj:.8f}")

    return best_f, best_obj


def focused_descent(f_init, time_limit, n_peaks=5, label="fd"):
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    f = f / f.sum()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(f)
        if n_peaks < len(b):
            top_idx = np.argpartition(b, -n_peaks)[-n_peaks:]
        else:
            top_idx = np.arange(len(b))

        grad = np.zeros(n)
        for k in top_idx:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                grad[j] += f[k - j]

        gs = grad.sum()
        if gs < 1e-15:
            break

        best_alpha = None
        best_trial = best_obj
        for alpha in [0.0005, 0.001, 0.002, 0.005, 0.01, 0.02, 0.05]:
            f_trial = f - alpha * grad
            f_trial = project_simplex(f_trial)
            obj = fast_convolve(f_trial).max()
            if obj < best_trial - 1e-14:
                best_trial = obj
                best_alpha = alpha
        if best_alpha is None:
            break
        f = f - best_alpha * grad
        f = project_simplex(f)
        best_obj = best_trial
        best_f = f.copy()

    return best_f, best_obj


def parametric_window(n, p, c):
    """a_j = max(0, 1 - c*(2j/(n-1)-1)^p)"""
    j = np.arange(n, dtype=np.float64)
    x = 2.0 * j / max(1, n - 1) - 1.0
    w = 1.0 - c * np.abs(x) ** p
    w = np.maximum(w, 0.0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def cosine_power_window(n, p):
    j = np.arange(n, dtype=np.float64)
    x = np.pi * (j + 0.5) / n
    w = np.cos(x) ** (2.0 * p)
    w = np.maximum(w, 0.0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def flat_top_window(n):
    a = [0.21557895, 0.41663158, 0.277263158, 0.083578947, 0.006947368]
    j = np.arange(n, dtype=np.float64)
    w = np.ones(n)
    for i, ai in enumerate(a):
        w = w - ai * np.cos(2.0 * np.pi * i * j / max(1, n - 1) - i * np.pi)
    w = np.maximum(w, 0.0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def try_parametric_init(n):
    """Try various parametric windows and return the best."""
    best_f = None
    best_score = float('inf')
    rng = np.random.default_rng(123)

    # Grid over cosine power
    for p in [0.5, 1.0, 1.5, 2.0, 2.5, 3.0, 4.0, 5.0, 8.0, 12.0, 20.0]:
        f = cosine_power_window(n, p)
        b = fast_convolve(f)
        score = 2.0 * n * b.max()
        if score < best_score:
            best_score = score
            best_f = f.copy()

    # Grid over parametric windows
    for p in [1.5, 2.0, 2.5, 3.0, 4.0, 6.0]:
        for c in [0.5, 0.8, 1.0, 1.2, 1.5]:
            f = parametric_window(n, p, c)
            b = fast_convolve(f)
            score = 2.0 * n * b.max()
            if score < best_score:
                best_score = score
                best_f = f.copy()

    # Flat top
    f = flat_top_window(n)
    b = fast_convolve(f)
    score = 2.0 * n * b.max()
    if score < best_score:
        best_score = score
        best_f = f.copy()

    if best_f is not None:
        print(f"    Parametric best: score={best_score:.8f}")
    return best_f, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main optimization: LP descent with constraint generation, soft-max polishing,
    multi-scale basin hopping, parametric initialization, and multi-length search.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
    except Exception:
        pass

    if parent is None:
        parent = cosine_power_window(1152, 2.0)

    best_seq = parent.copy()
    best_score = float(evaluate_sequence(parent.tolist()))
    n_parent = len(parent)
    print(f"Parent: n={n_parent}, score={best_score:.8f}")

    # === Phase 0: Parametric exploration for nearby lengths ===
    print(f"\n=== Phase 0: Parametric init ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t0 = min(rem * 0.04, 60.0)
        if t0 > 5:
            pstart = time.monotonic()
            for n_try in [n_parent]:
                pf, ps = try_parametric_init(n_try)
                if pf is not None:
                    real_score = float(evaluate_sequence(pf.tolist()))
                    if np.isfinite(real_score) and real_score < best_score:
                        best_score = real_score
                        best_seq = pf.copy()
                        print(f"  NEW BEST [parametric n={n_try}]: {best_score:.8f}")
            elapsed = time.monotonic() - pstart
            print(f"  Phase 0 done in {elapsed:.1f}s")

    # === Phase 1: LP main ===
    print(f"\n=== Phase 1: LP main (n={len(best_seq)}, score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.28, 480.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        f1, obj1 = run_lp_optimize(best_seq, time_limit=t1, top_k_init=80, top_k_max=500, label="lp1")
        score1 = float(evaluate_sequence(f1.tolist()))
        if np.isfinite(score1) and score1 < best_score:
            best_score = score1
            best_seq = f1.copy()
            print(f"  NEW BEST: {best_score:.8f}")

    # === Phase 2: Soft-max polish ===
    print(f"\n=== Phase 2: Soft-max polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.10, 120.0)
    if t2 > 15:
        f2, _ = soft_max_descend(best_seq, time_limit=t2, t_start=0.003, t_end=0.00003, label="sm1")
        score2 = float(evaluate_sequence(f2.tolist()))
        if np.isfinite(score2) and score2 < best_score:
            best_score = score2
            best_seq = f2.copy()
            print(f"  NEW BEST: {best_score:.8f}")

    # === Phase 3: Focused descent ===
    print(f"\n=== Phase 3: Focused descent (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.06, 60.0)
    if t3 > 10:
        f3, _ = focused_descent(best_seq, time_limit=t3, n_peaks=5, label="fd1")
        score3 = float(evaluate_sequence(f3.tolist()))
        if np.isfinite(score3) and score3 < best_score:
            best_score = score3
            best_seq = f3.copy()
            print(f"  NEW BEST: {best_score:.8f}")

    # === Phase 4: Basin hopping with perturbation ===
    print(f"\n=== Phase 4: Basin hopping (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    n_hops = 6
    for hop in range(n_hops):
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        t_hop = rem / (n_hops - hop) * 0.85
        t_hop = min(t_hop, 300.0)
        if t_hop < 40:
            break

        # Multi-scale perturbation
        scale = 0.005 * (1.0 + 0.3 * hop)
        f_pert = best_seq + rng.standard_normal(len(best_seq)) * scale
        f_pert = np.maximum(f_pert, 0.0)
        f_pert = f_pert / f_pert.sum()

        # Short LP
        t_lp = t_hop * 0.7
        f_hop, _ = run_lp_optimize(f_pert, time_limit=t_lp, top_k_init=60, top_k_max=300, label=f"hop{hop}")
        score_hop = float(evaluate_sequence(f_hop.tolist()))

        # Polish with soft-max if time allows
        rem2 = deadline - time.monotonic()
        if rem2 > 20 and np.isfinite(score_hop) and score_hop < best_score + 0.005:
            t_sm = min(rem2 * 0.3, 40.0)
            if t_sm > 10:
                f_hop2, _ = soft_max_descend(f_hop, time_limit=t_sm, t_start=0.002, t_end=0.00002, label=f"hotsm{hop}")
                score_hop2 = float(evaluate_sequence(f_hop2.tolist()))
                if np.isfinite(score_hop2) and score_hop2 < score_hop:
                    score_hop = score_hop2
                    f_hop = f_hop2

        if np.isfinite(score_hop) and score_hop < best_score:
            best_score = score_hop
            best_seq = f_hop.copy()
            print(f"  NEW BEST [hop{hop}]: {best_score:.8f}")
        else:
            print(f"  hop{hop}: {score_hop:.8f} (no improvement)")

    # === Phase 5: LP deep on best ===
    print(f"\n=== Phase 5: LP deep (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.22, 350.0)
    if t5 > 30:
        f5, _ = run_lp_optimize(best_seq, time_limit=t5, top_k_init=120, top_k_max=600, label="lp_deep")
        score5 = float(evaluate_sequence(f5.tolist()))
        if np.isfinite(score5) and score5 < best_score:
            best_score = score5
            best_seq = f5.copy()
            print(f"  NEW BEST: {best_score:.8f}")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-length (score={best_score:.8f}) ===")
    n_base = len(best_seq)
    length_candidates = [n_base - 24, n_base - 12, n_base + 12, n_base + 24, n_base + 48, n_base + 96]
    length_candidates = [lc for lc in length_candidates if 600 <= lc <= 3500]

    for target_n in length_candidates[:3]:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t6 = min(rem * 0.18, 180.0)
        if t6 < 30:
            break

        # Interpolate best to new length
        old_x = np.linspace(0.0, 1.0, n_base)
        new_x = np.linspace(0.0, 1.0, target_n)
        f_interp = np.interp(new_x, old_x, best_seq)
        f_interp = np.maximum(f_interp, 0.0)
        s = f_interp.sum()
        if s > 1e-10:
            f_interp /= s

        # Also try parametric at this length
        pf, ps = try_parametric_init(target_n)

        # Pick better init
        f_init = f_interp
        init_score = 2.0 * target_n * fast_convolve(f_interp).max()
        if pf is not None:
            ps2 = 2.0 * target_n * fast_convolve(pf).max()
            if ps2 < init_score:
                f_init = pf
                init_score = ps2

        f6, _ = run_lp_optimize(f_init, time_limit=t6, top_k_init=60, top_k_max=300, label=f"n{target_n}")
        score6 = float(evaluate_sequence(f6.tolist()))
        if np.isfinite(score6) and score6 < best_score:
            best_score = score6
            best_seq = f6.copy()
            print(f"  NEW BEST [n={target_n}]: {best_score:.8f}")
        else:
            print(f"  n={target_n}: {score6:.8f}")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final polish (score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        # LP final
        t7a = min(rem * 0.4, 100.0)
        if t7a > 10:
            f7a, _ = run_lp_optimize(best_seq, time_limit=t7a, top_k_init=150, top_k_max=600, label="lp_fin")
            score7a = float(evaluate_sequence(f7a.tolist()))
            if np.isfinite(score7a) and score7a < best_score:
                best_score = score7a
                best_seq = f7a.copy()
                print(f"  NEW BEST: {best_score:.8f}")

        # Soft-max final
        rem2 = deadline - time.monotonic()
        t7b = min(rem2 * 0.4, 60.0)
        if t7b > 10:
            f7b, _ = soft_max_descend(best_seq, time_limit=t7b, t_start=0.001, t_end=0.00001, label="sm_fin")
            score7b = float(evaluate_sequence(f7b.tolist()))
            if np.isfinite(score7b) and score7b < best_score:
                best_score = score7b
                best_seq = f7b.copy()
                print(f"  NEW BEST: {best_score:.8f}")

        # Focused final
        rem3 = deadline - time.monotonic()
        t7c = min(rem3 * 0.5, 30.0)
        if t7c > 5:
            f7c, _ = focused_descent(best_seq, time_limit=t7c, n_peaks=8, label="fd_fin")
            score7c = float(evaluate_sequence(f7c.tolist()))
            if np.isfinite(score7c) and score7c < best_score:
                best_score = score7c
                best_seq = f7c.copy()
                print(f"  NEW BEST: {best_score:.8f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'=' * 60}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"Target: 1.5027")
    print(f"Gap: {best_score - 1.5027:.6f}")
    print(f"{'=' * 60}")

    return [float(x) for x in best_seq]
