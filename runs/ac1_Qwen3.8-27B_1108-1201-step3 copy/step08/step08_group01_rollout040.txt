from scipy.fft import rfft, irfft
import time

# Global state
_best_seq = None
_best_score = float('inf')

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    ub = np.sort(v)[::-1]
    cssv = np.cumsum(ub)
    cond = ub * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1] if cond.any() else n - 1
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_soft_max_grad(a, tau, fft_size):
    """Compute gradient of soft max of conv(a,a)."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = conv / tau
    max_s = scaled.max()
    exp_s = np.exp(scaled - max_s)
    p = exp_s / exp_s.sum()
    
    # Cross-correlation: grad[i] = 2 * sum_k p[k] * a[k-i]
    # Using FFT
    N = fft_size
    P = np.fft.fft(p, N)
    A = np.fft.fft(a, N)
    cross = np.fft.ifft(np.conj(P) * A).real[:n]
    grad = 2.0 * cross
    
    return grad, conv, p

    while time.monotonic() - t0 < time_limit:
        it += 1

        grad, conv_z, p = compute_soft_max_grad(z, tau, fft_size)

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated soft-max gradient with FISTA and annealing (main optimizer)
2. Peak-matching coordinate descent (polishing)
3. Multi-length and multi-start search
4. Simulated annealing for escaping local minima
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    ub = np.sort(v)[::-1]
    cssv = np.cumsum(ub)
    cond = ub * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
    else:
        rho = n - 1
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def get_gradient_soft_max(a, tau, fft_size):
    """Compute gradient of soft-max(conv(a,a)/tau)*tau."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = conv / tau
    max_s = scaled.max()
    exp_s = np.exp(scaled - max_s)
    p = exp_s / exp_s.sum()
    # grad_i = 2 * sum_k p[k] * a[k-i] (cross-correlation)
    P = np.fft.fft(p, fft_size)
    A = np.fft.fft(a, fft_size)
    cross = np.fft.ifft(np.conj(P) * A).real[:n]
    return 2.0 * cross, conv, p

def softmax_fista(a0, time_limit, tau_init=0.3, tau_end=0.003, verbose=False):
    """FISTA on soft-max approximation with geometric annealing."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    fft_size = 1 << (3 * n).bit_length()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    tau = tau_init
    it = 0
    x = a.copy()
    z = a.copy()
    t_mom = 1.0
    alpha = 0.01
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        grad, conv_z, _ = get_gradient_soft_max(z, tau, fft_size)
        
        # Backtracking line search
        x_new = z - alpha * grad
        x_new = simplex_project(x_new)
        
        # Check improvement every 100 iterations
        if it % 100 == 0 or it < 20:
            conv_x = fast_convolve(x_new)
            score = 2.0 * n * conv_x.max()
            if score < best_score:
                best_score = score
                best_a = x_new.copy()
        
        # Update momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        z = x_new + ((t_mom - 1.0) / t_new) * (x_new - x)
        z = np.maximum(z, 0)
        sz = z.sum()
        if sz > 1e-15:
            z = z / sz
        x = x_new
        t_mom = t_new
        
        # Anneal tau
        # Geometric: tau = tau_init * (tau_end/tau_init)^(it/total_iters)
        # But we don't know total_iters, so use exponential decay
        tau = max(tau_end, tau * 0.9997)
        
        # Adjust step size based on tau
        alpha = min(0.1, max(1e-5, tau * 0.05))
        
        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            sc = 2.0 * n * fast_convolve(x).max()
            print(f"  SMFISTA it={it} ({elapsed:.0f}s): tau={tau:.5f} cur={sc:.6f} best={best_score:.6f}")
    
    return best_a, best_score

def peak_matching_cd(a0, time_limit, verbose=False):
    """Coordinate descent with peak-matching step selection."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    M = conv.max()
    best_a = a.copy()
    best_score = 2.0 * n * M
    t0 = time.monotonic()
    sweeps = 0
    
    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        M_before = M
        
        for i in range(n):
            if a[i] < 1e-14:
                continue
            
            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            
            # Unaffected max
            unaff = -1e30
            if k_start > 0:
                unaff = max(unaff, conv[:k_start].max())
            if k_end < 2 * n - 2:
                unaff = max(unaff, conv[k_end + 1:].max())
            
            # Affected entries: conv[k] for k in [k_start, k_end]
            # When we change a[i] by delta, conv[k] changes by 2*delta*a[k-i] (for k != 2i)
            # and by 2*delta*a[i] + delta^2 (for k = 2i)
            
            # Peak matching: find delta that makes the top affected peak equal to second
            # or equal to unaff
            affected_idx = np.arange(k_start, k_end + 1)
            affected_vals = conv[affected_idx].copy()
            slopes = np.zeros(len(affected_idx))
            for m in range(len(affected_idx)):
                k = affected_idx[m]
                j = k - i
                if 0 <= j < n:
                    slopes[m] = 2.0 * a[j]
            
            # Handle k = 2i (quadratic term)
            if 2 * i >= k_start and 2 * i <= k_end:
                m_2i = 2 * i - k_start
                slopes[m_2i] = 2.0 * a[i]  # linear part; quadratic is delta^2
            
            # Find the current max and second max among affected
            max_m = np.argmax(affected_vals)
            max_val = affected_vals[max_m]
            
            # Find delta to reduce max_val to second_max or unaff
            second_max = np.partition(affected_vals, -2)[-2] if len(affected_vals) >= 2 else 0
            target = min(second_max, unaff) if unaff > 0 else second_max
            
            if slopes[max_m] > 1e-14 and max_val > target:
                # Linear: max_val + slope * delta = target
                delta = (target - max_val) / slopes[max_m]
                # Clamp to keep a[i] >= 0
                delta = max(delta, -a[i] * 0.99)
                delta = min(delta, a[i])  # Don't increase too much
                
                # Try this delta and a few nearby
                candidates = [delta, delta * 0.5, delta * 0.8, 0.0, -a[i] * 0.1, a[i] * 0.1]
            else:
                candidates = [-a[i] * 0.3, -a[i] * 0.1, a[i] * 0.1, a[i] * 0.3]
            
            best_delta = 0.0
            best_new_M = M
            
            for delta in candidates:
                new_val = a[i] + delta
                if new_val < -1e-14:
                    continue
                
                # Compute new affected values
                new_affected = affected_vals + slopes * delta
                # Quadratic correction at k = 2i
                if 2 * i >= k_start and 2 * i <= k_end:
                    m_2i = 2 * i - k_start
                    new_affected[m_2i] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta
                
                new_M = max(unaff, new_affected.max())
                if new_M < best_new_M:
                    best_new_M = new_M
                    best_delta = delta
            
            if best_delta != 0.0 and best_new_M < M - 1e-16:
                # Apply
                a[i] += best_delta
                a[i] = max(a[i], 0.0)
                
                # Update conv
                changes = slopes * best_delta
                if 2 * i >= k_start and 2 * i <= k_end:
                    m_2i = 2 * i - k_start
                    changes[m_2i] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff, conv[k_start:k_end + 1].max())
                
                score = 2.0 * n * M
                if score < best_score - 1e-14:
                    best_score = score
                    best_a = a.copy()
                improved = True
        
        if verbose and sweeps % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"  PMCD sweep={sweeps} ({elapsed:.0f}s): M={M:.8f} score={2*n*M:.6f}")
        
        if not improved and sweeps > 5:
            break
    
    return best_a, best_score

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated soft-max gradient with FISTA acceleration and geometric annealing
2. Peak-matching coordinate descent for precise polishing
3. Simulated annealing (perturb + re-optimize) for escaping local minima
4. Multi-length exploration
5. Adaptive step sizing with backtracking

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via real FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def simplex_project(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    ub = np.sort(v)[::-1]
    cssv = np.cumsum(ub)
    cond = ub * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
    else:
        rho = n - 1
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_soft_max_grad(a, tau, fft_size):
    """
    Compute gradient of f_tau(a) = tau * log(sum_k exp(conv(a,a)_k / tau)).
    Returns (gradient, convolution, softmax_weights).
    """
    n = len(a)
    conv = fast_convolve(a)
    scaled = conv / tau
    max_s = scaled.max()
    exp_s = np.exp(scaled - max_s)
    p = exp_s / exp_s.sum()
    # grad_i = 2 * sum_k p[k] * a[k-i] (cross-correlation)
    P = np.fft.fft(p, fft_size)
    A = np.fft.fft(a, fft_size)
    cross = np.fft.ifft(np.conj(P) * A).real[:n]
    grad = 2.0 * cross
    return grad, conv, p


def softmax_fista(a0, time_limit, tau_init=1.0, tau_end=0.01, decay=0.9998,
                  verbose=False, label=""):

    FISTA on soft-max approximation with geometric tau annealing.
    Uses backtracking line search for robustness.
    """

    tau = tau_init
    it = 0

    x_prev = a.copy()
    t_mom = 1.0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # FISTA extrapolation
        if it == 1:
            z = x.copy()
        else:
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
            z = x + ((t_mom - 1.0) / t_new) * (x - x_prev)
            z = np.maximum(z, 0)
            sz = z.sum()
            if sz > 1e-15:
                z = z / sz
            x_prev = x.copy()
            t_mom = t_new

        # Compute gradient at z
        grad, conv_z, p = compute_soft_max_grad(z, tau, fft_size)
        f_z = tau * (np.log(p.sum() * np.exp((conv_z.max() - conv_z.max()))) )  # just for reference
        # Actually f_z = tau * logsumexp(conv_z / tau)

            best_score = score_check

            no_improve = 0

            print(f"    SFISTA[{label}] it={it} ({elapsed:.0f}s): tau={tau:.5f} best={best_score:.6f}")

def peak_matching_cd(a0, time_limit, verbose=False, label="", max_sweeps=5000):
    """
    Coordinate descent with intelligent peak-matching step selection.
    For each coordinate, finds the delta that equalizes the top peaks.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_a = a.copy()
    best_score = 2.0 * n * M
    t0 = time.monotonic()
    sweeps = 0
    stall = 0

    while time.monotonic() - t0 < time_limit and sweeps < max_sweeps:
        sweeps += 1
        improved = False
        M_start = M

        for i in range(n):
            if a[i] < 1e-14:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            # Unaffected max
            unaff = -1e30
            if k_start > 0:
                unaff = max(unaff, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff = max(unaff, float(conv[k_end + 1:].max()))

            # Affected values and slopes
            affected = conv[k_start:k_end + 1].copy()
            affected_len = k_end - k_start + 1
            slopes = np.empty(affected_len)
            for m in range(affected_len):
                k = k_start + m
                j = k - i
                if 0 <= j < n:
                    slopes[m] = 2.0 * a[j]
                else:
                    slopes[m] = 0.0

            has_quad = (2 * i >= k_start) and (2 * i <= k_end)
            m_2i = 2 * i - k_start if has_quad else -1

            # Find current top affected peak
            max_m = int(np.argmax(affected))
            max_val = affected[max_m]

            # Second max among affected
            if affected_len >= 2:
                second_val = float(np.partition(affected, -2)[-2])
            else:
                second_val = 0.0

            # Target: min of second peak and unaffected max
            target = min(second_val, unaff) if unaff > -1e29 else second_val
            # Also consider reducing to a fraction of current max
            target_frac = max_val * 0.95

            # Generate candidate deltas
            candidates = set()
            # Peak matching delta
            if slopes[max_m] > 1e-14 and max_val > target:
                d_pm = (target - max_val) / slopes[max_m]
                candidates.add(d_pm)
                candidates.add(d_pm * 0.5)
                candidates.add(d_pm * 0.8)
                candidates.add(d_pm * 1.2)
            # Fractional reduction
            if slopes[max_m] > 1e-14:
                d_frac = (target_frac - max_val) / slopes[max_m]
                candidates.add(d_frac)
            # Fixed fractions of a[i]
            for f in [0.3, 0.1, -0.1, -0.3, 0.5, -0.5, 0.05, -0.05]:
                candidates.add(f * a[i])
            # Zero
            candidates.add(0.0)

            best_delta = 0.0
            best_new_M = M

            for delta in candidates:
                new_val_i = a[i] + delta
                if new_val_i < -1e-14:
                    continue

                # Compute new affected
                new_affected = affected + slopes * delta
                if has_quad:
                    new_affected[m_2i] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_M = max(unaff, float(new_affected.max()))
                if new_M < best_new_M - 1e-15:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-15:
                # Apply update
                a[i] += best_delta
                a[i] = max(a[i], 0.0)

                # Update convolution
                changes = slopes * best_delta
                if has_quad:
                    changes = changes.copy()
                    changes[m_2i] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff, float(conv[k_start:k_end + 1].max()))

                score = 2.0 * n * M
                if score < best_score - 1e-13:
                    best_score = score
                    best_a = a.copy()
                improved = True

        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PMCD[{label}] sweep={sweeps} ({elapsed:.0f}s): M={M:.9f} score={2*n*M:.6f}")

        if not improved:
            stall += 1
            if stall > 10:
                break
        else:
            stall = 0

    return best_a, best_score

def interpolate_seq(f, new_n):
    """Linearly interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def perturb_sequence(a, rng, magnitude=0.02):
    """Perturb a sequence and renormalize."""
    n = len(a)
    a_new = a.copy() * (1.0 + rng.standard_normal(n) * magnitude)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def block_perturb(a, rng, block_size=32):
    """Perturb a random block of the sequence."""
    n = len(a)
    a_new = a.copy()
    start = rng.integers(0, max(1, n - block_size))
    end = min(start + block_size, n)
    a_new[start:end] *= (1.0 + rng.standard_normal(end - start) * 0.05)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s
