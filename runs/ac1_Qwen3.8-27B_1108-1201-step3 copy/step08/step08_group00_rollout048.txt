        def jac(z_cur):

        try:
            res = minimize(func, z, jac=jac, method='L-BFGS-B',
                          options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                                   'gtol': 1e-14, 'maxfun': iters_per_stage * 200,
                                   'disp': False, 'maxls': 50})
            z = res.x
            if verbose:
                a_cur = softmax_from_z(z)
                score = 2.0 * n * fast_convolve(a_cur).max()
                elapsed = time.monotonic() - t0
                print(f"    L-BFGS T={T:.1f} ({elapsed:.0f}s): score={score:.6f}")
        except Exception as e:
            if verbose:
                print(f"    L-BFGS T={T:.1f} error: {e}")
    
    return softmax_from_z(z)


def nesterov_momentum(a0, time_limit, T=2.0, n_steps=50000, lr_init=0.01, momentum=0.9):
    n = len(a0)
    z = z_from_a(a0).copy()
    v = np.zeros(n)
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        lr = lr_init * (0.999 ** step)
        grad = logsumexp_grad_z(z, T, n)
        v = momentum * v - lr * grad
        z_new = z + v
        a_new = softmax_from_z(z_new)
        score = 2.0 * n * fast_convolve(a_new).max()
        
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        z = z_new
        
        if step % 5000 == 0 and step > 0:
            elapsed = time.monotonic() - t0
            print(f"    Mom {elapsed:.0f}s: score={best_score:.6f} lr={lr:.6f}")
    
    return best_a


def peak_shave(a0, time_limit, n_steps=30000):
    n = len(a0)
    a = np.maximum(a0, 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top = np.argsort(contribs)[::-1][:12]
        
        improved = False
        for ci in top:
            i = idx[ci]
            if a[i] < 1e-15:
                continue
            for factor in [0.995, 0.99, 0.98, 0.97, 0.95, 0.92, 0.88]:
                a_trial = a.copy()
                a_trial[i] *= factor
                diff = a[i] * (1 - factor)
                if diff < 1e-18:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] += diff * a_trial[mask] / others_sum
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Random perturbation
            i = int(rng.integers(n))
            if a[i] > 1e-15:
                a_trial = a.copy()
                a_trial[i] *= 0.95
                diff = a[i] * 0.05
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                a_trial[mask] += diff * a_trial[mask] / a_trial[mask].sum()
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                else:
                    break
        
        if step % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PS {elapsed:.0f}s: score={best_score:.6f}")
    
    return best_a


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def structured_init(n, rng):
    choice = int(rng.integers(0, 6))
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    elif choice == 2:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (0.5 + 0.5 * np.cos(2 * np.pi * x)) ** 2
    elif choice == 4:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 4
    else:
        a = np.ones(n) * (1.0 + 0.3 * np.sin(2 * np.pi * 3 * np.linspace(0, 1, n)))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
