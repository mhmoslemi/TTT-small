def fast_convolve(a):

    return irfft(A * A, fft_size)[:size]

def softmax_weights(conv, T):
    """Compute softmax weights over convolution values."""
    shifted = conv / T
    max_s = shifted.max()
    w = np.exp(shifted - max_s)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(conv)
        w[conv.argmax()] = 1.0
    else:
        w = w / s
    return w


def subgradient_smoothed(a, T):
    """Compute the gradient of the smoothed max (log-sum-exp * T) of conv(a,a).
    
    Returns (gradient, smoothed_value, conv).
    """
    n = len(a)
    conv = fast_convolve(a)
    w = softmax_weights(conv, T)

    # d(conv[k])/da[l] = 2*a[k-l] (if 0<=k-l<n) [as derived above]
    # So g[l] = 2n * T * 2 * sum_k w[k] * a[k-l] (for valid k-l)
    # = 4n*T * cross_corr(w, a)[l]

def project_simplex(v):
    """Project vector v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the optimization score for a normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def softmax_weights(conv, T):
    """Compute softmax weights over convolution values with temperature T."""
    shifted = conv / T
    max_s = shifted.max()
    w = np.exp(shifted - max_s)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(conv)
        w[conv.argmax()] = 1.0
    else:
        w = w / s
    return w


def subgradient_smoothed(a, T):
    """Compute gradient of smoothed max of conv(a,a) w.r.t. a.
    
    Returns (gradient, smoothed_value, conv).
    Uses FFT cross-correlation for O(n log n) computation.
    """
    n = len(a)
    conv = fast_convolve(a)
    w = softmax_weights(conv, T)
    
    # g[l] = 4n*T * cross_corr(w, a)[l]
    # where cross_corr(w, a)[l] = sum_m a[m] * w[m+l]
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    cross = irfft(np.conj(A) * W, fft_size)[:n]
    
    grad = 4.0 * n * T * cross
    
    # Smoothed value: 2n * T * log(sum(exp(conv/T)))
    shifted = conv / T
    max_s = shifted.max()
    smooth_val = 2.0 * n * T * (max_s + np.log(np.exp(shifted - max_s).sum()))
    
    return grad, smooth_val, conv


def project_simplex(v):
    """Project vector v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def nesterov_subgradient(a0, time_limit, T0=100.0, T_min=5.0, T_decay=0.9995,
                          momentum0=0.85, lr_scale=1.0, verbose=True):
    """Nesterov momentum subgradient method on smoothed objective."""
    n = len(a0)
    a = a0.copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    v = np.zeros(n)
    mu = momentum0
    T = T0
    lr = lr_scale / np.sqrt(n)
    t0 = time.monotonic()
    
    iters = 0
    no_improve = 0
    rng_local = np.random.default_rng(99)
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        
        # Nesterov lookahead
        a_look = a + mu * v
        a_look = project_simplex(a_look)
        
        # Compute gradient at lookahead
        grad, _, _ = subgradient_smoothed(a_look, T)
        
        # Center gradient (handle sum constraint)
        grad_centered = grad - grad.mean()
        
        # Update
        v = mu * v - lr * grad_centered
        a_new = project_simplex(a + v)
        
        # Adaptive parameters
        T = max(T_min, T * T_decay)
        lr = max(0.001, lr * 0.99995)
        mu = min(0.999, 0.85 + iters * 0.00005)
        
        # Track best
        score = compute_score(a_new)
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        if verbose and iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NS iter={iters}, T={T:.1f}, mu={mu:.4f}, "
                  f"score={score:.8f}, best={best_score:.8f}, t={elapsed:.0f}s")
        
        if no_improve > 800:
            # Perturb to escape
            a = best_a + rng_local.standard_normal(n) * 0.002
            a = project_simplex(a)
            v = np.zeros(n)
            no_improve = 0
    
    return best_a, best_score


def eval_transfer(a, i, j, t):
    """Evaluate score after transferring t from position i to position j."""
    n = len(a)
    ai = a[i]
    aj = a[j]
    if ai - t < -1e-14 or aj + t < -1e-14:
        return 1e10
    a[i] -= t
    a[j] += t
    # Compute score
    score = compute_score(a)
    # Undo
    a[i] = ai
    a[j] = aj
    return score


def coord_descent_aggressive(a0, time_limit, verbose=True, seed=123):
    """Aggressive coordinate descent with peak-aware pair selection."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    pass_count = 0
    no_improve_passes = 0
    
    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        
        # Determine number of pairs based on remaining time
        n_pairs = min(3000, n * 4)
        
        # Pre-compute convolution for peak-aware selection
        conv = fast_convolve(a)
        max_conv = conv.max()
        
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Mix pair selection strategies
            r = rng.random()
            if r < 0.4:
                # Random pair
                i = rng.integers(0, n)
                j = rng.integers(0, n)
            elif r < 0.7:
                # Peak-contributing pair
                peak_idx = int(np.argmax(conv))
                js = max(0, peak_idx - n + 1)
                je = min(n - 1, peak_idx)
                i = rng.integers(js, je + 1)
                j = rng.integers(js, je + 1)
            elif r < 0.85:
                # Peak vs non-peak
                peak_idx = int(np.argmax(conv))
                js = max(0, peak_idx - n + 1)
                je = min(n - 1, peak_idx)
                peak_positions = list(range(js, je + 1))
                non_peak = np.setdiff1d(np.arange(n), peak_positions)
                if len(non_peak) == 0:
                    continue
                i = peak_positions[rng.integers(0, len(peak_positions))]
                j = non_peak[rng.integers(0, len(non_peak))]
            else:
                # Adjacent pair (local smoothing)
                i = rng.integers(0, n - 1)
                j = i + 1
            
            if i == j:
                continue
            if a[i] < 1e-15 and a[j] < 1e-15:
                continue
            
            # Try transfer using scipy's bounded minimizer
            t_lo = -a[j] + 1e-14
            t_hi = a[i] - 1e-14
            if t_hi - t_lo < 1e-14:
                continue
            
            # Quick check: try a few fixed amounts first
            best_t = 0.0
            best_f = compute_score(a)
            
            candidates = [0.0]
            step = (t_hi - t_lo) / 10.0
            for k in range(1, 10):
                candidates.append(t_lo + k * step)
            
            for t_cand in candidates:
                f_cand = eval_transfer(a, i, j, t_cand)
                if f_cand < best_f - 1e-14:
                    best_f = f_cand
                    best_t = t_cand
            
            if best_t != 0.0:
                # Refine with bounded minimizer around the best candidate
                local_lo = max(t_lo, best_t - step)
                local_hi = min(t_hi, best_t + step)
                
                def obj_1d(t):
                    return eval_transfer(a, i, j, t)
                
                res = minimize_scalar(obj_1d, bounds=(local_lo, local_hi),
                                      method='bounded', options={'xatol': 1e-14})
                if res.fun < best_f - 1e-14:
                    best_t = res.x
                    best_f = res.fun
            
            if best_f < best_score - 1e-14 and best_t != 0.0:
                a[i] -= best_t
                a[j] += best_t
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = compute_score(a)
                best_a = a.copy()
                improved = True
                conv = fast_convolve(a)  # Update conv for next iteration
        
        if verbose and pass_count % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={pass_count}, score={best_score:.8f}, t={elapsed:.0f}s")
        
        if not improved:
            no_improve_passes += 1
            if no_improve_passes > 8:
                # Perturb
                a = best_a + rng.standard_normal(n) * 0.001
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve_passes = 0
        else:
            no_improve_passes = 0
    
    return best_a, best_score


def lbgfs_smoothed(a0, time_limit, T=500.0, maxiter=3000):
    """L-BFGS optimization on the smoothed log-sum-exp objective."""
    n = len(a0)
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)
    
    def obj_func(x):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a_trial = np.maximum(x, 0)
        s = a_trial.sum()
        if s < 1e-15:
            return 1e15
        a_trial = a_trial / s
        _, smooth_val, _ = subgradient_smoothed(a_trial, T)
        return smooth_val
    
    def grad_func(x):
        if time.monotonic() - t0 > time_limit:
            return np.zeros(n)
        a_trial = np.maximum(x, 0)
        s = a_trial.sum()
        if s < 1e-15:
            return np.zeros(n)
        a_trial = a_trial / s
        grad, _, _ = subgradient_smoothed(a_trial, T)
        # Project to account for normalization constraint
        grad_proj = grad - a_trial * np.dot(grad, a_trial)
        return grad_proj
    
    try:
        res = minimize(
            fun=obj_func, x0=a0.copy(), jac=grad_func,
            method='L-BFGS-B',
            bounds=[(0, None)] * n,
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12,
                     'maxfun': maxiter * 30, 'disp': False}
        )
        a_out = np.maximum(res.x, 0)
        s = a_out.sum()
        if s > 1e-15:
            a_out = a_out / s
            score = compute_score(a_out)
            if score < best_score:
                best_a = a_out
                best_score = score
    except Exception:
        pass
    
    return best_a, best_score


def coord_descent_fine(a0, time_limit, verbose=True, seed=456):
    """Finer coordinate descent: uses all pairs systematically in rounds."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    gr = (np.sqrt(5) - 1) / 2  # golden ratio
    
    rounds = 0
    while time.monotonic() - t0 < time_limit:
        rounds += 1
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        
        # Systematic sweep with randomization
        indices = list(range(n))
        rng.shuffle(indices)
        
        for i in indices:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-15:
                continue
            
            # Try transferring to a few random targets
            n_targets = min(20, n)
            targets = rng.choice(n, size=n_targets, replace=False)
            
            for j in targets:
                if j == i:
                    continue
                if a[j] < 1e-15 and a[i] < 1e-15:
                    continue
                
                # Golden section search
                t_lo = -a[j] + 1e-14
                t_hi = a[i] - 1e-14
                if t_hi - t_lo < 1e-14:
                    continue
                
                x1 = t_hi - gr * (t_hi - t_lo)
                x2 = t_lo + gr * (t_hi - t_lo)
                f1 = eval_transfer(a, i, int(j), x1)
                f2 = eval_transfer(a, i, int(j), x2)
                
                for _ in range(18):
                    if f1 < f2:
                        t_hi = x2
                        x2 = x1
                        f2 = f1
                        x1 = t_hi - gr * (t_hi - t_lo)
                        f1 = eval_transfer(a, i, int(j), x1)
                    else:
                        t_lo = x1
                        x1 = x2
                        f1 = f2
                        x2 = t_lo + gr * (t_hi - t_lo)
                        f2 = eval_transfer(a, i, int(j), x2)
                
                t_opt = (t_lo + t_hi) / 2
                f_opt = min(f1, f2)
                
                if f_opt < best_score - 1e-14 and t_opt != 0.0:
                    a[i] -= t_opt
                    a[int(j)] += t_opt
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    best_score = compute_score(a)
                    best_a = a.copy()
                    improved = True
        
        if verbose and rounds % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD-fine round={rounds}, score={best_score:.8f}, t={elapsed:.0f}s")
        
        if not improved and rounds > 3:
            a = best_a + rng.standard_normal(n) * 0.0005
            a = np.maximum(a, 0)
            a = a / a.sum()
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score, verbose=True):
    """Evaluate a candidate and return updated best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):

    3. Aggressive coordinate descent with peak-aware selection (main)
    4. Fine systematic coordinate descent (polish)
    5. Length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"Could not load parent: {e}")
    
    if parent is None:
        n_init = 768
        parent = np.ones(n_init) / n_init
        # Add some structure to avoid trivial starting point
        t_arr = np.linspace(0, 2 * np.pi * 3, n_init)
        parent = parent + 0.03 * np.cos(t_arr) + 0.02 * np.sin(2 * t_arr)
        parent = np.maximum(parent, 0)
        parent = parent / parent.sum()
        best_seq = parent.copy()
        best_score = compute_score(parent)
        print(f"Initialized: n={n_init}, score={best_score:.8f}")
    
    n_parent = len(parent)

    print(f"\n=== Phase 1: Nesterov Subgradient (best={best_score:.8f}) ===")

        a_fin, sc_fin = lbgfs_smoothed(best_seq, t_alloc, T=200, maxiter=2000)

    # === Phase 9: Final CD ===
    rem = deadline - time.monotonic()
    if rem > 20:
        t_alloc = min(60.0, rem * 0.6)
        print(f"\n=== Phase 9: Final CD (best={best_score:.8f}) ===")
        a_fin2, sc_fin2 = coord_descent_fine(best_seq, t_alloc, verbose=False, seed=999)
