def interpolate_to_length(a, new_n):
    """Interpolate sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    s = a.sum()
    a_norm = a / s
    old_x = np.linspace(0, 1, n, endpoint=False)
    new_x = np.linspace(0, 1, new_n, endpoint=False)
    a_new = np.interp(new_x, old_x, a_norm)
    a_new = np.maximum(a_new, 0)
    ss = a_new.sum()
    if ss > 1e-15:
        a_new = a_new / ss
    return a_new
