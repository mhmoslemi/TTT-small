def project_simplex(v):

def score_of(f):
    """Compute the score for a normalized (sum=1, non-negative) sequence."""
    b = fast_convolve(f)
    s = f.sum()
    if s < 1e-10:
        return np.inf
    return 2.0 * len(f) * b.max() / (s * s)

def solve_lp(f, max_constraints=None):
    """
    Solve: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for selected k.
    Returns (g, M, constraint_indices).
    """
    n = len(f)
    b = fast_convolve(f)
    M = b.max()
    
    if max_constraints is None or max_constraints >= len(b):
        indices = np.arange(len(b))
    else:
        # Top peaks + uniform sampling
        n_top = min(max_constraints // 2, len(b))
        top_idx = np.argpartition(b, -n_top)[-n_top:]
        n_rest = max_constraints - len(top_idx)
        if n_rest > 0:
            remaining = np.setdiff1d(np.arange(len(b)), top_idx)
            if len(remaining) > 0 and n_rest < len(remaining):
                step = len(remaining) / n_rest
                extra = np.unique((np.arange(len(remaining)) * step).astype(int))
                indices = np.sort(np.concatenate([top_idx, remaining[extra]]))[:max_constraints]
            else:

        else:
            indices = np.sort(top_idx)

    A = np.zeros((K, n))
    for i, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi+1] = f[lo:hi+1]

    constraints = [A @ g_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
    
    solvers = [cp.PDLP, cp.CBC, cp.SCS, cp.GLPK, cp.ECOS]

        except Exception:
            continue

def nesterov_polish(f, time_limit, moment=0.9):
    """Nesterov-accelerated projected gradient on the peak."""
    t0 = time.monotonic()
    n = len(f)
    x = f.copy()
    y = f.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    v = np.zeros(n)
    step = 0.01
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(y)
        
        # Subgradient of max at the maximizing points
        peak_vals = b.max()
        peak_mask = b >= peak_vals * 0.999
        n_peaks = peak_mask.sum()
        
        grad = np.zeros(n)
        for k in range(len(b)):
            if not peak_mask[k]:
                continue
            w = 1.0 / n_peaks
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            grad[idx] += w * f[k - idx]
        
        # Nesterov update
        v = moment * v + grad
        y = x - step * v
        y = project_simplex(y)
        x_new = (1 + moment) * y - moment * x
        x_new = project_simplex(x_new)
        
        obj = fast_convolve(x_new).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = x_new.copy()
            no_improve = 0
            step = min(step * 1.01, 0.05)
        else:
            no_improve += 1
            if no_improve > 50:
                step *= 0.9
                no_improve = 0
            if no_improve > 500:
                break
        
        x = x_new
    
    return best_f, best_obj

def resize_sequence(f, new_n):

    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s > 1e-10:

"""
Multi-strategy optimization for auto-convolution peak minimization.
Combines full-constraint LP (all 2n-1 constraints), adaptive line search,
Nesterov-accelerated subgradient polish, and multi-length exploration.
Key improvement: uses ALL convolution constraints in the LP for exact

restarts to escape local optima.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(f, g):
    """Cross-convolution f*g via FFT."""
    size = len(f) + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def peak_of(f):
    """Compute max of auto-convolution for normalized f."""
    return fast_convolve(f).max()


def solve_lp_full(f, n, use_all_constraints=True):
    """
    Solve LP: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for all/some k.
    Returns (g, M).
    """
    b = fast_convolve(f)
    M = b.max()
    n_conv = len(b)

    if use_all_constraints:
        indices = np.arange(n_conv)
    else:
        # Top peaks + uniform spacing
        n_top = min(n_conv // 2, 1500)
        top_idx = np.argpartition(b, -n_top)[-n_top:]
        n_rest = 2000 - len(top_idx)
        if n_rest > 0:
            remaining = np.setdiff1d(np.arange(n_conv), top_idx)
            if len(remaining) > 0:
                step = max(1, len(remaining) // n_rest)
                extra = remaining[::step][:n_rest]
                indices = np.sort(np.concatenate([top_idx, extra]))
            else:
                indices = np.sort(top_idx)
        else:
            indices = np.sort(top_idx)

    K = len(indices)
    A = np.zeros((K, n))
    for i, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]

    g_var = cp.Variable(n, nonneg=True)
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), [A @ g_var <= M])

    solvers = [cp.PDLP, cp.CBC, cp.SCS, cp.GLPK, cp.ECOS]
    for solver in solvers:
        try:
            prob.solve(solver=solver, max_iters=2000000)
            gv = g_var.value
            if gv is not None and np.all(np.isfinite(gv)):
                return np.maximum(gv, 0.0), M
        except Exception:
            continue
    return None, M


def line_search(f, h, precision=8):
    """Ternary search for optimal step size t in (0,1]."""
    best_t = 0.0
    best_obj = peak_of(f)

    # Coarse search

        s = f_trial.sum()
        if s < 1e-10:
            continue
        f_trial = f_trial / s
        obj = peak_of(f_trial)
        if obj < best_obj - 1e-16:
            best_obj = obj
            best_t = t

    if best_t == 0:
        return 0.0

    # Fine search around best_t
    lo = max(0.0, best_t * 0.6)
    hi = min(1.0, best_t * 1.4)
    for _ in range(precision):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        fm1 = np.maximum((1 - m1) * f + m1 * h, 0)
        fm1 = fm1 / max(fm1.sum(), 1e-10)
        fm2 = np.maximum((1 - m2) * f + m2 * h, 0)
        fm2 = fm2 / max(fm2.sum(), 1e-10)

        o2 = peak_of(fm2)
        if o1 < o2:
            hi = m2
        else:
            lo = m1
        if hi - lo < 1e-6:
            break

    return best_t


def lp_step(f, n, use_all=True):
    """One LP-based iteration. Returns (f_new, improved, obj)."""
    g, M = solve_lp_full(f, n, use_all_constraints=use_all)

    if g is None:
        return f, False, peak_of(f)

    S = g.sum()
    if S <= 1.0 + 1e-8:
        return f, False, peak_of(f)

    t = line_search(f, h)

        return f, False, peak_of(f)

    f_new = np.maximum((1 - t) * f + t * h, 0)
    s = f_new.sum()
    if s > 1e-10:
        f_new = f_new / s

    obj = peak_of(f_new)
    return f_new, True, obj


def nesterov_polish(f_init, time_limit, moment=0.85):
    """Nesterov-accelerated projected subgradient on the peak."""
    t0 = time.monotonic()
    n = len(f_init)
    x = f_init.copy()
    y = f_init.copy()
    best_f = f_init.copy()
    best_obj = peak_of(f_init)
    v = np.zeros(n)
    step = 0.005
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(y)
        max_b = b.max()

        # Subgradient: average over near-peak lags
        threshold = max_b * 0.999
        peak_mask = b >= threshold
        n_peaks = max(1, int(peak_mask.sum()))

        grad = np.zeros(n)
        for k in np.where(peak_mask)[0]:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            grad[idx] += (1.0 / n_peaks) * f_init[k - idx]

        v = moment * v + grad
        y_new = project_simplex(x - step * v)
        x_new = project_simplex((1 + moment) * y_new - moment * x)

        obj = peak_of(x_new)
        if obj < best_obj - 1e-16:
            best_obj = obj
            best_f = x_new.copy()
            no_improve = 0
            step = min(step * 1.005, 0.03)
        else:
            no_improve += 1
            if no_improve % 100 == 0:
                step *= 0.95
            if no_improve > 2000:
                break

        x = x_new
        y = y_new

    return best_f, best_obj

    print(f"Starting: n={n_parent}, score={best_score:.8f}")

    # === Phase 1: Full-constraint LP on parent ===
    print(f"\n=== Phase 1: Full LP (n={n_parent}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.40, 650.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        f1, obj1 = run_optimization(parent, time_limit=t1,
                                     use_all_constraints=True, label="P1_full")
        score1 = float(evaluate_sequence(f1.tolist()))
        if np.isfinite(score1) and score1 < best_score:
            best_score = score1
            best_seq = f1.copy()
            print(f"  Phase 1 NEW BEST: {score1:.8f}")
        else:
            print(f"  Phase 1: {score1:.8f} (no improvement)")

    # === Phase 2: Nesterov polish ===
    print(f"\n=== Phase 2: Nesterov polish (n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.12, 120.0)
    if t2 > 10:
        f2, obj2 = nesterov_polish(best_seq, time_limit=t2, moment=0.85)
        score2 = float(evaluate_sequence(f2.tolist()))
        if np.isfinite(score2) and score2 < best_score:
            best_score = score2
            best_seq = f2.copy()
            print(f"  Phase 2 NEW BEST: {score2:.8f}")
        else:
            print(f"  Phase 2: {score2:.8f} (no improvement)")

    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-length exploration ===")
    n_base = len(best_seq)
    target_lengths = [1024, 1280, 1536, 1920, 2048, 2560]
    target_lengths = [tl for tl in target_lengths if tl != n_base and 512 <= tl <= 2800]
    # Sort by proximity to current best length
    target_lengths.sort(key=lambda x: abs(x - n_base))

    for tl in target_lengths[:4]:
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t3 = min(rem * 0.14, 110.0)
        if t3 < 15:
            break

        # Try multiple initializations
        inits = [resize_sequence(best_seq, tl),
                 flat_top_window(tl),
                 blackmanharris_window(tl)]

        best_for_tl = None
        best_score_tl = float('inf')

        for init_idx, f_init in enumerate(inits):
            remaining = t3 / len(inits)
            remaining = min(remaining, rem - time.monotonic())
            if remaining < 8:
                break

            init_names = ["interp", "flat_top", "blackman"]
            print(f"  n={tl} init={init_names[init_idx]} t={remaining:.0f}s")

            f_try, obj_try = run_optimization(f_init, time_limit=remaining,
                                               use_all_constraints=(tl <= 1600),
                                               label=f"n{tl}_{init_names[init_idx]}")
            score_try = float(evaluate_sequence(f_try.tolist()))
            if np.isfinite(score_try) and score_try < best_score_tl:
                best_score_tl = score_try
                best_for_tl = f_try.copy()

        if best_for_tl is not None and best_score_tl < best_score:
            best_score = best_score_tl
            best_seq = best_for_tl.copy()
            print(f"  n={tl}: NEW BEST {best_score:.8f}")
        else:
            print(f"  n={tl}: best={best_score_tl:.8f} (no improvement over {best_score:.8f})")

    # === Phase 4: Final LP polish on best ===
    print(f"\n=== Phase 4: Final LP polish (n={len(best_seq)}, score={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.35, 180.0)
    if t4 > 10:
        f4, obj4 = run_optimization(best_seq, time_limit=t4,
                                     use_all_constraints=True, label="P4_final")
        score4 = float(evaluate_sequence(f4.tolist()))
        if np.isfinite(score4) and score4 < best_score:
            best_score = score4
            best_seq = f4.copy()
            print(f"  Phase 4 NEW BEST: {score4:.8f}")
        else:
            print(f"  Phase 4: {score4:.8f}")

    # === Phase 5: Final Nesterov ===
    rem = deadline - time.monotonic()
    if rem > 15:
        t5 = min(rem * 0.5, 60.0)
        if t5 > 8:
            f5, obj5 = nesterov_polish(best_seq, time_limit=t5, moment=0.9)
            score5 = float(evaluate_sequence(f5.tolist()))
            if np.isfinite(score5) and score5 < best_score:
                best_score = score5
                best_seq = f5.copy()
                print(f"  Phase 5 NEW BEST: {score5:.8f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'=' * 60}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"Target: 1.5027")
    print(f"Gap: {best_score - 1.5027:.6f}")
    print(f"{'=' * 60}")

    return [float(x) for x in best_seq]
