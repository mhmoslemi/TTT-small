```python
    """Compute gradient of (1/T)*log(sum(exp(T*b))) where b = auto-conv of f."""
    n = len(f)
    b = fast_convolve(f)

    b_scaled = T * b
    b_scaled -= b_scaled.max()  # for numerical stability
    w = np.exp(b_scaled)

    # Gradient: sum_k w_k * db_k/df_i

        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            grad[i] += w[k] * f[k - i]
    return grad / T  # because we're differentiating (1/T)*log(sum exp(T*b))
```

```python
"""
Spectral gradient + LP hybrid optimizer for auto-convolution peak minimization.
Uses log-sum-exp smoothing with Adam for fast initial optimization,
then LP-based constraint generation for final polish.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(f, g):
    """Cross-convolution f*g via FFT."""
    size = len(f) + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def compute_grad_topk(f, b, top_k=10, T=10.0):
    """
    Compute gradient of (1/T)*log(sum(exp(T*b_k))) using only top-k terms.
    Returns (objective_value, gradient).
    """
    n = len(f)
    if top_k < len(b):
        idx = np.argpartition(b, -top_k)[-top_k:]
    else:
        idx = np.arange(len(b))
    
    b_top = b[idx]
    b_max = b_top.max()
    b_scaled = T * (b_top - b_max)
    w = np.exp(b_scaled)
    w_sum = w.sum()
    obj = (b_max + np.log(w_sum)) / T
    w /= (w_sum * T)
    
    grad = np.zeros(n)
    for i in range(len(idx)):
        wk = w[i]
        if wk < 1e-15:
            continue
        k = int(idx[i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi+1] += wk * f[lo:hi+1]
    
    return obj, grad


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def adam_optimize(f_init, time_limit, T_start=5.0, T_end=50.0, top_k=10,
                  lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8,
                  label="adam", n_restart=3):
    """
    Adam optimizer for minimizing the smoothed max auto-conv peak.
    Works on the simplex (sum=1, f>=0).
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    f = project_simplex(f)
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    no_improve = 0
    it = 0
    
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    
    total_time = max(time_limit, 1.0)
    n_restarts = n_restart
    time_per_segment = total_time / n_restarts
    
    for restart in range(n_restarts):
        seg_start = time.monotonic()
        seg_limit = seg_start + time_per_segment
        
        if restart > 0:
            # Restart from best + noise
            f = project_simplex(best_f + np.random.default_rng(42 + restart).standard_normal(n) * 0.01)
            m = np.zeros(n)
            v = np.zeros(n)
        
        while time.monotonic() < seg_limit:
            it += 1
            elapsed = time.monotonic() - t0
            progress = min(1.0, elapsed / total_time)
            T = T_start + (T_end - T_start) * progress
            
            b = fast_convolve(f)
            cur_peak = b.max()
            
            if cur_peak < best_obj - 1e-14:
                best_obj = cur_peak
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            if no_improve > 200:
                break
            
            obj, grad = compute_grad_topk(f, b, top_k=top_k, T=T)
            
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad * grad
            m_hat = m / (1 - beta1 ** it)
            v_hat = v / (1 - beta2 ** it)
            
            f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
            f = project_simplex(f_new)
            
            if it % 500 == 0:
                print(f"    {label} restart={restart} iter={it} T={T:.1f} "
                      f"peak={best_obj:.10f} score={2.0*n*best_obj:.6f} "
                      f"t={time.monotonic()-t0:.0f}s")
    
    return best_f, best_obj


def lp_refine(f_init, time_limit, top_k_init=50, max_constraints=300):
    """LP-based refinement using constraint generation."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    f = project_simplex(f)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()
        
        # Select active constraints: top peaks
        top_k = min(top_k_init + it * 2, max_constraints)
        if top_k < len(b):
            top_idx = np.argpartition(b, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(b))
        top_idx = np.sort(top_idx)
        
        # Build constraint matrix
        K = len(top_idx)
        A = np.zeros((K, n))
        for row_i in range(K):
            k = int(top_idx[row_i])
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            A[row_i, lo:hi+1] = f[lo:hi+1]
        
        # Solve LP: maximize sum(g) s.t. g >= 0, A @ g <= M
        g_var = cp.Variable(n)
        prob = cp.Problem(
            cp.Maximize(cp.sum(g_var)),
            [g_var >= 0, A @ g_var <= M]
        )
        
        solved = False
        for solver in [cp.PDLP, cp.SCS]:
            try:
                prob.solve(solver=solver, max_iters=100000)
                solved = True
                break
            except Exception:
                continue
        
        if not solved or g_var.value is None:
            break
        
        g = np.maximum(g_var.value, 0.0)
        S_g = g.sum()
        
        if S_g <= 1.0 + 1e-5:
            break
        
        # The direction h = g / S_g has sum 1
        h = g / S_g
        
        # Line search: find best t in (0, 1)
        best_t = 0.1
        best_trial = best_obj
        for t in [0.05, 0.1, 0.2, 0.3, 0.5]:
            f_trial = (1 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_trial:
                best_trial = obj
                best_t = t
        
        # Golden section refinement
        if best_t > 0.05:
            lo, hi = max(0.0, best_t - 0.1), min(1.0, best_t + 0.1)
            gr = (np.sqrt(5) - 1) / 2
            c = hi - gr * (hi - lo)
            d = lo + gr * (hi - lo)
            
            def peak_at(t):
                ft = (1-t) * f + t * h
                ft = np.maximum(ft, 0.0)
                s = ft.sum()
                if s < 1e-10:
                    return 1e10
                ft /= s
                return fast_convolve(ft).max()
            
            fc, fd = peak_at(c), peak_at(d)
            for _ in range(10):
                if fc <= fd:
                    hi, d, fd = d, c, fc
                    c = hi - gr * (hi - lo)
                    fc = peak_at(c)
                else:
                    lo, c, fc = c, d, fc
                    d = lo + gr * (hi - lo)
                    fd = peak_at(d)
                if hi - lo < 1e-5:
                    break
            best_t = (lo + hi) / 2
        
        f_new = (1 - best_t) * f + best_t * h
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s > 1e-10:
            f_new /= s
        
        obj_new = fast_convolve(f_new).max()
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
        else:
            break
        
        if it % 10 == 0:
            print(f"    LP iter={it} obj={best_obj:.10f} "
                  f"score={2.0*n*best_obj:.6f} t={time.monotonic()-t0:.0f}s")
    
    return best_f, best_obj


def hann_window(n):
```

"""
Hybrid spectral-Adam + LP optimizer for auto-convolution peak minimization.
Combines fast gradient-based exploration with LP-based precise refinement.
Multi-start and multi-length exploration for global optimality.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(f, g):
    size = len(f) + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_grad(f, b, top_k, T):
    """Compute (smoothed_obj, gradient) using top-k log-sum-exp."""
    n = len(f)
    if top_k < len(b):
        idx = np.argpartition(b, -top_k)[-top_k:]
    else:
        idx = np.arange(len(b))
    
    b_top = b[idx]
    b_max = b_top.max()
    b_scaled = T * (b_top - b_max)
    w = np.exp(b_scaled)
    w_sum = w.sum()
    obj = (b_max + np.log(w_sum)) / T
    w /= (w_sum * T)
    
    grad = np.zeros(n)
    for i in range(len(idx)):
        wk = w[i]
        if wk < 1e-15:
            continue
        k = int(idx[i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi+1] += wk * f[lo:hi+1]
    
    return obj, grad


def adam_optimize(f_init, time_limit, T_start=5.0, T_end=100.0, top_k=10,

    f = project_simplex(f_init.copy())
    best_f = f.copy()
    best_obj = fast_convolve(f).max()

def lp_refine(f_init, time_limit, top_k_init=50, max_constraints=250,
              label="lp"):
    t0 = time.monotonic()
    n = len(f_init)
    f = project_simplex(f_init.copy())
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()
        
        top_k = min(top_k_init + it * 3, max_constraints)
        if top_k < len(b):
            top_idx = np.argpartition(b, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(b))
        top_idx = np.sort(top_idx)
        
        K = len(top_idx)
        A = np.zeros((K, n))
        for row_i in range(K):
            k = int(top_idx[row_i])
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                A[row_i, lo:hi+1] = f[lo:hi+1]
        
        g_var = cp.Variable(n)
        prob = cp.Problem(
            cp.Maximize(cp.sum(g_var)),
            [g_var >= 0, A @ g_var <= M]
        )
        
        solved = False
        for solver in [cp.PDLP, cp.SCS]:
            try:
                prob.solve(solver=solver, max_iters=200000)
                solved = True
                break
            except Exception:
                continue
        
        if not solved or g_var.value is None:
            break
        
        g = np.maximum(g_var.value, 0.0)
        if np.any(np.isnan(g)):
            break
        S_g = g.sum()
        
        if S_g <= 1.0 + 1e-5:
            break
        
        h = g / S_g
        
        # Line search
        best_t = 0.1
        best_trial = best_obj
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
            f_trial = (1 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_trial:
                best_trial = obj
                best_t = t
        
        # Golden section
        if best_t > 0.02:
            lo = max(0.0, best_t - 0.08)
            hi = min(0.8, best_t + 0.08)
            gr = (np.sqrt(5) - 1) / 2
            c = hi - gr * (hi - lo)
            d = lo + gr * (hi - lo)
            
            fc = _peak_at(f, h, c)
            fd = _peak_at(f, h, d)
            for _ in range(10):
                if fc <= fd:
                    hi, d, fd = d, c, fc
                    c = hi - gr * (hi - lo)
                    fc = _peak_at(f, h, c)
                else:
                    lo, c, fc = c, d, fc
                    d = lo + gr * (hi - lo)
                    fd = _peak_at(f, h, d)
                if hi - lo < 1e-6:
                    break
            best_t = (lo + hi) / 2
        
        f_new = (1 - best_t) * f + best_t * h
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s > 1e-10:
            f_new /= s
        
        obj_new = fast_convolve(f_new).max()
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
        else:
            break
        
        if it % 5 == 0:
            print(f"    {label} iter={it} obj={best_obj:.10f} "
                  f"score={2.0*n*best_obj:.6f} t={time.monotonic()-t0:.0f}s")
    
    return best_f, best_obj


def _peak_at(f, h, t):
    ft = (1-t) * f + t * h
    ft = np.maximum(ft, 0.0)
    s = ft.sum()
    if s < 1e-10:
        return 1e10
    ft /= s
    return fast_convolve(ft).max()


def hann_window(n):
    w = 0.5 * (1.0 + np.cos(2.0 * np.pi * np.arange(n) / max(1, n - 1)))
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def flat_top_window(n):
    w = np.ones(n)
    a = [0.21557895, 0.41663158, 0.277263158, 0.083578947, 0.006947368]
    for i, ai in enumerate(a):
        w -= ai * np.cos(2 * np.pi * i * np.arange(n) / max(1, n - 1) - i * np.pi)
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def gaussian_window(n, sigma=0.3):
    x = np.linspace(-1, 1, n)
    w = np.exp(-x**2 / (2 * sigma**2))
    s = w.sum()
    return w / s
