    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix_subset(a_fixed, active_indices):
    """Build |active| x n matrix T such that T @ b = convolve(a_fixed, b)[active_indices]."""
    n = len(a_fixed)
    T = np.zeros((len(active_indices), n))
    for row, k in enumerate(active_indices):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        T[row, idx] = a_fixed[k - idx]
    return T

def solve_lp_best_response(a_fixed, active_indices, solver):
    """Given fixed a (non-negative, sum=1), find b minimizing max over active conv(a,b)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix_subset(a_fixed, active_indices)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def lbgfs_refine(a0, time_limit, T_start, T_end, maxiter, n_stages=8):
    """L-BFGS with LogSumExp and staged temperature annealing."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()

    # Precompute stage parameters
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(50, maxiter // n_stages)

    for stage_idx, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        # Local function for this stage
        def stage_func(z_local):
            a_local = softmax_from_z(z_local)
            b_local = fast_convolve(a_local)
            Tb = stage_T * b_local
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / stage_T) * log_sum_exp

        def stage_jac(z_local):
            a_local = softmax_from_z(z_local)
            b_local = fast_convolve(a_local)
            Tb = stage_T * b_local
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            # Gradient of log_sum_exp w.r.t. a
            grad_a = np.zeros(n)
            for k in range(len(b_local)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += w[k] * a_local[k - idx]
            S = np.dot(grad_a, a_local)
            return a_local * (grad_a - S)

        try:
            res = minimize(
                fun=stage_func,
                x0=z,
                jac=stage_jac,
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters, active_frac=0.3):
    """Nesterov LP with adaptive active set."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    b = fast_convolve(a)
    best_score = 2.0 * n * b.max()
    t0 = time.monotonic()
    
    # Active set: fraction of peaks to include
    n_active = max(10, int(active_frac * (2 * n - 1)))

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute autocorrelation of y
        b_y = fast_convolve(y)
        
        # Identify active set (top peaks)
        threshold = np.partition(b_y, -n_active)[-n_active]
        active = np.where(b_y >= threshold)[0]
        
        # Solve LP for best response
        b_new, M_val = solve_lp_best_response(y, active, solver)
        if b_new is None:
            continue

        # Nesterov update
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        # Evaluate both a and y
        score_a = 2.0 * n * fast_convolve(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        score_y = 2.0 * n * b_y.max()  # reuse
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def peak_equalize(a0, time_limit, n_iter=5000):
    """Peak equalization: iteratively reduce the highest peak by adjusting its contributors."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]
        
        # Find second peak (excluding edges)
        b_check = b.copy()
        b_check[0] = -1e30
        b_check[-1] = -1e30
        second_k = int(np.argmax(b_check))
        second_val = b_check[second_k]
        
        # Pick the peak to reduce (the one that's more "out of line")
        target_k = peak_k if peak_val > second_val else second_k
        target_val = b[target_k]

        # Find top contributors to this peak
        js = max(0, target_k - n + 1)
        je = min(n - 1, target_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[target_k - idx]
        
        # Try reducing the top contributor
        top_i = idx[np.argmax(contribs)]
        
        # Line search: find best reduction factor
        best_factor = 1.0
        best_local_score = best_score
        for factor in [0.99, 0.98, 0.97, 0.95, 0.93, 0.90, 0.85, 0.80]:
            a_trial = a.copy()
            diff = a_trial[top_i] * (1 - factor)
            a_trial[top_i] *= factor
            # Redistribute the reduction proportionally to others
            mask = np.ones(n, dtype=bool)
            mask[top_i] = False
            others_sum = a_trial[mask].sum()
            if others_sum < 1e-15:
                continue
            a_trial[mask] += diff * a_trial[mask] / others_sum
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_local_score:
                best_local_score = score
                best_factor = factor
                best_trial = a_trial.copy()

        if best_factor < 1.0:
            a = best_trial
            if best_local_score < best_score:
                best_score = best_local_score
                best_a = a.copy()
        else:
            # Try pair reduction
            top2 = np.argsort(contribs)[::-1][:3]
            improved = False
            for ti in top2:
                i = idx[ti]
                j = target_k - i
                if i == j or i >= n or j >= n:
                    continue
                for f in [0.98, 0.95, 0.90]:
                    a_trial = a.copy()
                    a_trial[i] *= f
                    a_trial[j] *= f
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if not improved:
                break

    return best_a, best_score


def coordinate_descent_finetune(a0, time_limit, n_iter=10000):
    """Coordinate descent with line search on the most impactful coordinates."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0

    for it in range(n_iter):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_i = idx[np.argmax(contribs)]

        # Try small adjustments
        improved = False
        for delta_frac in [0.01, 0.005, 0.002, -0.01, -0.005, -0.002]:
            a_trial = a.copy()
            a_trial[top_i] *= (1 + delta_frac)
            if a_trial[top_i] < 0:
                a_trial[top_i] = 0
            a_trial = a_trial / a_trial.sum()
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-12:
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True
                break

        if not improved:
            stall += 1
            if stall > 20:
                break
        else:
            stall = 0

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def structured_init(n, rng):
    choice = rng.integers(0, 4)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 0.5
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * (x - 0.5))
        a = a ** 0.7
    else:
        x = np.linspace(0, 1, n)
        a = np.exp(-2 * (x - 0.3) ** 2) * 0.5 + np.exp(-2 * (x - 0.7) ** 2) * 0.5

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using active-set LP with Nesterov
acceleration, multi-restart L-BFGS with adaptive LogSumExp annealing,

import numpy as np

from scipy.fft import rfft, irfft

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix_subset(a_fixed, active_indices):
    n = len(a_fixed)
    T = np.zeros((len(active_indices), n))
    for row, k in enumerate(active_indices):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        T[row, idx] = a_fixed[k - idx]
    return T


_lp_solver = None

def solve_lp_best_response(a_fixed, active_indices, solver):
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix_subset(a_fixed, active_indices)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.CLARABEL]:
            if alt == solver:
                continue

                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def lbgfs_func(z, n, T):
    a_local = softmax_from_z(z)
    b_local = fast_convolve(a_local)
    Tb = T * b_local
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    return (2.0 * n / T) * log_sum_exp


def lbgfs_jac(z, n, T):
    a_local = softmax_from_z(z)
    b_local = fast_convolve(a_local)
    Tb = T * b_local
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    grad_a = np.zeros(n)
    for k in range(len(b_local)):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += w[k] * a_local[k - idx]
    S = np.dot(grad_a, a_local)
    return a_local * (grad_a - S)


def lbgfs_refine(a0, time_limit, T_start, T_end, maxiter, n_stages):
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(50, maxiter // n_stages)

    for stage_idx in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        stage_T = temps[stage_idx]
        try:
            res = minimize(
                fun=lbgfs_func,
                x0=z,
                jac=lbgfs_jac,
                args=(n, stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)
