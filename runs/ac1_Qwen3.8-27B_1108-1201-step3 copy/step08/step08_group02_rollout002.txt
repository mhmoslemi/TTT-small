Uses full-constraint LP with precomputed matrix structure,

from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
import cvxpy as cp
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def precompute_conv_indices(n):
    row_idx = []
    col_idx = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            row_idx.append(k)
            col_idx.append(j)
    return np.array(row_idx, dtype=np.int32), np.array(col_idx, dtype=np.int32)


def build_A(f, n, row_idx, col_idx):
    vals = f[row_idx - col_idx]
    return csr_matrix((vals, (row_idx, col_idx)), shape=(2*n-1, n))

        sel_rows = peak_indices
        mask = np.isin(row_idx, peak_indices)

        c_sel = col_idx[mask]
        vals = f[r_sel - c_sel]
        A = csr_matrix((vals, (r_sel, c_sel)), shape=(len(peak_indices), n))
    else:

def project_simplex(v, target_sum=1.0):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]

"""
LP-based optimization for minimizing 2*n*max(auto-conv)/(sum)^2.
Uses adaptive full-partial constraint LP with precomputed sparse structure,
exponentiated gradient warm-up, careful line search, coordinate descent,
and multi-length exploration to beat the 1.5027 target.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def precompute_conv_indices(n):
    """Precompute (row, col) indices for the convolution matrix A where A@g = conv(f,g,full)."""
    row_list = []
    col_list = []
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            row_list.append(k)
            col_list.append(j)
    return np.array(row_list, dtype=np.int32), np.array(col_list, dtype=np.int32)


def build_conv_matrix(f, row_idx, col_idx, n):
    """Build sparse convolution matrix A where A[k,j] = f[k-j]."""
    vals = f[row_idx - col_idx]
    return csr_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n))


def solve_lp_full(f, n, M, row_idx, col_idx):
    """Solve LP with ALL constraints: max sum(g) s.t. g>=0, (f*g)[k]<=M for all k."""
    A = build_conv_matrix(f, row_idx, col_idx, n)
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    for solver in [cp.GLOP, cp.ECOS, cp.CBC]:
        try:
            prob.solve(solver=solver, max_iters=200000)
            if g.value is not None and not np.any(np.isnan(g.value)):
                return np.maximum(g.value, 0.0)
        except Exception:
            continue
    return None


def solve_lp_peaks(f, n, M, row_idx, col_idx, peak_set):

    mask = np.isin(row_idx, peak_set_arr)
    r_sel = row_idx[mask]
    c_sel = col_idx[mask]
    vals = f[r_sel - c_sel]
    A = csr_matrix((vals, (r_sel, c_sel)), shape=(len(peak_set_arr), n))

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

def line_search_1d(f, direction, n, M_current):

    1D line search: find best t in (0,1] minimizing max(conv((1-t)*f + t*direction)).
    Uses golden-section refinement after coarse grid.
    """

    best_obj = M_current

        f_trial = np.maximum(f_trial, 0.0)
        s = f_trial.sum()

        f_trial /= s
        obj = fast_convolve(f_trial).max()

    # Refine with finer grid around best

            if obj < best_obj - 1e-15:

def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def exp_gradient_step(f, n, lr, temp, n_top=200):
    """One exponentiated gradient step on the convolution peak."""
    b = fast_convolve(f)
    M = b.max()

    # Softmax weights on peaks
    scaled = b / (temp * M)
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()

    # Focus on top peaks
    if n_top < len(b):
        top_idx = np.argpartition(b, -n_top)[-n_top:]
        w_sel = w[top_idx].copy()
        ws = w_sel.sum()
        if ws > 1e-30:
            w_sel /= ws
        w = np.zeros(len(b))
        w[top_idx] = w_sel

    # Compute gradient: grad[i] = sum_k w[k] * f[k-i]
    grad = np.zeros(n)
    active_idx = np.where(w > 1e-14)[0]
    for k in active_idx:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi + 1] += w[k] * f[k - lo:hi + 1 - (hi - lo) * 0]
        # More carefully:
        # For each j in [lo, hi], grad[j] += w[k] * f[k-j]
        # But we want grad[j] = sum_k w[k] * f[k-j]
        # So for fixed k, j ranges over [lo, hi], and f[k-j] is the value

    # Redo gradient properly
    grad = np.zeros(n)
    for k in active_idx:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            grad[j] += w[k] * f[k - j]

    # Exponentiated gradient update
    log_f = np.log(f + 1e-30)
    log_f_new = log_f - lr * grad
    log_f_new -= log_f_new.max()
    f_new = np.exp(log_f_new)
    s = f_new.sum()
    if s < 1e-30:
        return f
    f_new /= s
    return f_new

def coordinate_descent(f_init, time_limit, label="cd"):
    """Coordinate descent with adaptive multipliers."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_b = fast_convolve(f).max()
    multipliers = [0.5, 0.7, 0.85, 0.95, 1.05, 1.1, 1.2, 1.4, 1.7, 2.2]
    pass_count = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-12:
                continue
            best_j_val = f[j]
            best_j_obj = best_b
            for mult in multipliers:
                f_trial = f.copy()
                f_trial[j] = f[j] * mult
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-14:
                    best_j_obj = obj
                    best_j_val = f[j] * mult
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f /= s
                if best_j_obj < best_b - 1e-14:
                    best_b = best_j_obj
                    improved = True

        pass_count += 1
        if not improved:
            # Try adding small values to zero entries
            zero_mask = f < 1e-12
            if zero_mask.any():
                j = int(np.random.choice(np.where(zero_mask)[0]))
                f_trial = f.copy()
                f_trial[j] = f.max() * 0.1
                f_trial /= f_trial.sum()
                obj = fast_convolve(f_trial).max()
                if obj < best_b - 1e-14:
                    f = f_trial
                    best_b = obj
                    improved = True
        if not improved and pass_count > 3:
            break

    return f, best_b


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    n = len(f)
    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new
