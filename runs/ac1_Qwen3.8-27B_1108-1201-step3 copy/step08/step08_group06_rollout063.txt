def fast_convolve(a):
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

    """Score for normalized p (sum=1): 2n * max(p*p)"""

def build_conv_matrix(f):
    """Build C where (C*g)[k] = (f*g)[k] for linear convolution."""
    n = len(f)

    for j in range(n):

def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. C*f*g <= M, g >= 0"""
    n = len(f)

    b_ub = np.full(2*n-1, M)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based fictitious play with both linear and geometric mean updates
2. Multiplicative weight updates for fast exploration of large n
3. Subgradient method with adaptive step sizes
4. Coordinate descent with exact golden-section line search

import numpy as np
import time
import math
from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(p):
    """Score for normalized p (sum=1): 2n * max(p*p)."""
    n = len(p)
    conv = fast_convolve(p)
    return 2.0 * n * conv.max()


def build_conv_matrix(f):
    """Build C where (C*g)[k] = (f*g)[k] for linear convolution."""
    n = len(f)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = f
    return C


def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    C = build_conv_matrix(f)
    c = -np.ones(n)
    A_ub = C
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status != 0 or res.x is None:
        return None
    g = np.maximum(res.x, 0)
    if g.sum() < 1e-15:
        return None
    return g


def lp_fictitious_play(f_init, time_limit, use_geom=True, verbose=False):
    """LP-based fictitious play with optional geometric mean updates."""
    f = f_init.astype(np.float64).copy()
    f = f / f.sum()
    n = len(f)
    best_f = f.copy()
    best_score = compute_score(f)
    f_avg = f.copy()
    t0 = time.monotonic()
    k = 0
    stagnant = 0
    prev_best = best_score

    while time.monotonic() - t0 < time_limit:
        M = fast_convolve(f).max()
        
        # Slightly inflate M to avoid infeasibility
        g = solve_lp_direction(f, M * (1 + 1e-10))
        if g is None:
            g = solve_lp_direction(f, M * 1.001)
        if g is None:
            break
        g = g / g.sum()
        
        # Adaptive step size
        t_step = 2.0 / (k + 2)
        if k > 200:
            t_step = max(0.01, 2.0 / (k + 2))
        
        if use_geom:
            # Geometric mean in log space
            f_safe = np.maximum(f, 1e-30)
            g_safe = np.maximum(g, 1e-30)
            log_new = (1 - t_step) * np.log(f_safe) + t_step * np.log(g_safe)
            a_new = np.exp(log_new - log_new.max())
            a_new = a_new / a_new.sum()
        else:
            a_new = (1 - t_step) * f + t_step * g
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                break
            a_new = a_new / s
        
        # Update running average
        f_avg = (k * f_avg + a_new) / (k + 1)
        
        sc_avg = compute_score(f_avg)
        sc_cur = compute_score(a_new)
        
        if sc_avg < best_score - 1e-14:
            best_score = sc_avg
            best_f = f_avg.copy()
            stagnant = 0
        elif sc_cur < best_score - 1e-14:
            best_score = sc_cur
            best_f = a_new.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        if stagnant > 500:
            # Perturb to escape
            rng = np.random.default_rng(k)
            a_new = best_f + rng.standard_normal(n) * 0.002
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            stagnant = 0
        
        f = a_new
        k += 1
        
        if verbose and k % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={k}, M={M:.6e}, best={best_score:.8f}, elapsed={elapsed:.0f}s")

    return best_f, best_score


def multiplicative_weight_updates(p_init, time_limit, eta=0.1, verbose=False):
    """Multiplicative weight updates: fast method for large n."""
    p = p_init.astype(np.float64).copy()
    p = p / p.sum()
    n = len(p)
    best_p = p.copy()
    best_score = compute_score(p)
    t0 = time.monotonic()
    k = 0
    stagnant = 0
    p_avg = p.copy()
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(p)
        k_star = int(np.argmax(conv))
        M = conv.max()
        
        # Subgradient: grad[j] = 2*p[k_star-j] for valid j
        grad = np.zeros(n)
        for j in range(min(k_star + 1, n)):
            idx = k_star - j
            if 0 <= idx < n:
                grad[j] = 2.0 * p[idx]
        
        # Multiplicative update
        p_new = p * np.exp(-eta * grad)
        s = p_new.sum()
        if s < 1e-300:
            p_new = np.ones(n) / n
        else:
            p_new = p_new / s
        
        p = p_new
        p_avg = (k * p_avg + p) / (k + 1)
        
        sc = compute_score(p)
        sc_avg = compute_score(p_avg)
        
        if sc_avg < best_score - 1e-14:
            best_score = sc_avg
            best_p = p_avg.copy()
            stagnant = 0
        elif sc < best_score - 1e-14:
            best_score = sc
            best_p = p.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        # Adaptive eta
        if stagnant > 200:
            eta *= 0.95
            stagnant = 0
        elif k > 1000:
            eta *= 0.9995
        
        k += 1
        
        if verbose and k % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MWU iter={k}, eta={eta:.4f}, best={best_score:.8f}, elapsed={elapsed:.0f}s")

    return best_p, best_score


def subgradient_method(p_init, time_limit, eta0=0.05, verbose=False):
    """Projected subgradient method."""
    p = p_init.astype(np.float64).copy()
    p = p / p.sum()
    n = len(p)
    best_p = p.copy()
    best_score = compute_score(p)
    t0 = time.monotonic()
    k = 0
    eta = eta0
    p_avg = p.copy()
    stagnant = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(p)
        k_star = int(np.argmax(conv))
        
        # Subgradient
        grad = np.zeros(n)
        for j in range(min(k_star + 1, n)):
            idx = k_star - j
            if 0 <= idx < n:
                grad[j] = 2.0 * p[idx]
        
        # Gradient step with projection
        p_new = p - eta * grad
        p_new = np.maximum(p_new, 0)
        s = p_new.sum()
        if s < 1e-15:
            p_new = np.ones(n) / n
        else:
            p_new = p_new / s
        
        p = p_new
        p_avg = (k * p_avg + p) / (k + 1)
        
        sc = compute_score(p)
        sc_avg = compute_score(p_avg)
        
        if sc_avg < best_score - 1e-14:
            best_score = sc_avg
            best_p = p_avg.copy()
            stagnant = 0
        elif sc < best_score - 1e-14:
            best_score = sc
            best_p = p.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        # Decay eta
        eta = eta0 / math.sqrt(k + 1)
        if stagnant > 500:
            p = best_p.copy()
            p_avg = p.copy()
            stagnant = 0
        
        k += 1

    return best_p, best_score


def coordinate_descent(p_init, time_limit, n_pairs=100, verbose=False):
    """Coordinate descent with golden-section search."""
    p = p_init.astype(np.float64).copy()
    p = p / p.sum()
    n = len(p)
    best_p = p.copy()
    best_score = compute_score(p)
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    stagnant = 0
    iters = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False
        
        conv = fast_convolve(p)
        top_peak = int(np.argmax(conv))
        
        # Generate pairs
        pairs = set()
        js = max(0, top_peak - n + 1)
        je = min(n - 1, top_peak)
        peak_range = je - js + 1
        
        n_peak_pairs = n_pairs * 2 // 3
        n_rand_pairs = n_pairs - n_peak_pairs
        
        for _ in range(n_peak_pairs):
            i = rng.integers(js, je + 1)
            j = rng.integers(js, je + 1)
            if i != j:
                pairs.add((min(i, j), max(i, j)))
        for _ in range(n_rand_pairs):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                pairs.add((min(i, j), max(i, j)))
        
        for (i, j) in list(pairs):
            if time.monotonic() - t0 > time_limit:
                break
            if p[i] < 1e-14 and p[j] < 1e-14:
                continue
            
            t_lo = -p[j] + 1e-14
            t_hi = p[i] - 1e-14
            if t_hi <= t_lo:
                continue
            
            x1 = t_hi - gr * (t_hi - t_lo)
            x2 = t_lo + gr * (t_hi - t_lo)
            
            # Golden section
            p_trial = p.copy()
            
            def f_x_x1():
                p_trial[i] = p[i] - x1
                p_trial[j] = p[j] + x1
                if p_trial[i] < 0 or p_trial[j] < 0:
                    return 1e10
                return compute_score(p_trial / p_trial.sum())

            def f_x_x2():
                p_trial[i] = p[i] - x2
                p_trial[j] = p[j] + x2
                if p_trial[i] < 0 or p_trial[j] < 0:
                    return 1e10
                return compute_score(p_trial / p_trial.sum())
            
            f1 = f_x_x1()
            f2 = f_x_x2()
            
            for _ in range(15):
                if time.monotonic() - t0 > time_limit:
                    break
                if f1 < f2:
                    t_hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = t_hi - gr * (t_hi - t_lo)
                    f1 = f_x_x1()
                else:
                    t_lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = t_lo + gr * (t_hi - t_lo)
                    f2 = f_x_x2()
            
            t_opt = (t_lo + t_hi) / 2.0
            f_opt = min(f1, f2)
            
            if f_opt < best_score - 1e-12:
                p[i] -= t_opt
                p[j] += t_opt
                p = np.maximum(p, 0)
                p = p / p.sum()
                best_score = compute_score(p)
                best_p = p.copy()
                improved = True
        
        if not improved:
            stagnant += 1
            if stagnant > 8:
                p = best_p + rng.standard_normal(n) * 0.003
                p = np.maximum(p, 0)
                p = p / p.sum()
                stagnant = 0
        else:
            stagnant = 0
        
        if verbose and iters % 20 == 0:
            print(f"    CD iter={iters}, best={best_score:.8f}")

    return best_p, best_score


def gen_cosine(n):
    j = np.arange(n)
    p = np.cos(np.pi * (j + 0.5) / (2.0 * n)) ** 2
    p = np.maximum(p, 1e-10)
    return p / p.sum()


def gen_gaussian(n, sigma_frac=0.3):
    j = np.arange(n)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    p = np.exp(-(j - center) ** 2 / (2.0 * sigma ** 2))
    return p / p.sum()


def gen_uniform(n):
    return np.ones(n) / n


def gen_triangle(n):
    j = np.arange(n)
    center = (n - 1) / 2.0
    p = 1.0 - np.abs(j - center) / center
    p = np.maximum(p, 1e-10)
    return p / p.sum()


def interpolate_to_length(p, new_n):
    old_n = len(p)
    if old_n == new_n:
        return p.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    p_interp = np.interp(new_x, old_x, p)
    p_interp = np.maximum(p_interp, 0)
    s = p_interp.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return p_interp / s


def peak_flatten(p, n_reductions=5, verbose=False):
    """Iteratively reduce the top peaks by redistributing mass."""
    p = p.astype(np.float64).copy()
    p = p / p.sum()
    n = len(p)
    best_p = p.copy()
    best_score = compute_score(p)
    rng = np.random.default_rng(99)
    
    for _ in range(n_reductions):
        conv = fast_convolve(p)
        top_k = np.argsort(conv)[::-1][:5]
        
        for pk in top_k:
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            
            # Find top contributors to this peak
            contribs = p[idx] * p[pk - idx]
            top_contrib = np.argsort(contribs)[::-1][:3]
            
            for ci in top_contrib:
                i_idx = idx[ci]
                if p[i_idx] < 1e-12:
                    continue
                
                # Try to move some mass from i_idx to a position that doesn't
                # contribute to this peak
                noncontrib = np.setdiff1d(np.arange(n), idx)
                if len(noncontrib) == 0:
                    continue
                
                # Pick a position in noncontrib with low current value
                candidates = noncontrib[np.argsort(p[noncontrib])[:10]]
                
                for j_idx in candidates[:3]:
                    for frac in [0.03, 0.07, 0.15]:
                        delta = p[i_idx] * frac
                        p_trial = p.copy()
                        p_trial[i_idx] -= delta
                        p_trial[j_idx] += delta
                        p_trial = np.maximum(p_trial, 0)
                        s = p_trial.sum()
                        if s < 1e-15:
                            continue
                        p_trial = p_trial / s
                        sc = compute_score(p_trial)
                        if sc < best_score - 1e-14:
                            p = p_trial
                            best_score = sc
                            best_p = p.copy()
                            break
    
    return best_p, best_score


def check_and_update(a, label, best_seq, best_score, verbose=False):
    """Evaluate and update best if improved."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:

    if np.isfinite(score) and score < best_score:

        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score

    # === Phase 4: Multiplicative weight updates (fast, good for convergence) ===
    print(f"\n=== Phase 4: MWU Polish (best={best_score:.8f}) ===")
    if time.monotonic() < deadline - 150:
        t_alloc = min(200, max(30, (deadline - time.monotonic()) * 0.3))
        print(f"  Budget: {t_alloc:.0f}s")
        # Try multiple eta values
        for eta_try in [0.03, 0.08, 0.15]:
            if time.monotonic() > deadline - 80:
                break
            t_sub = min(80, (deadline - time.monotonic()) * 0.3)
            if t_sub < 10:
                break
            p_mwu, sc_mwu = multiplicative_weight_updates(best_seq, t_sub, eta=eta_try)
            best_seq, best_score = check_and_update(p_mwu, f"mwu_eta{eta_try}", best_seq, best_score, verbose=True)

    # === Phase 5: Subgradient method ===
    print(f"\n=== Phase 5: Subgradient (best={best_score:.8f}) ===")
    if time.monotonic() < deadline - 100:
        t_alloc = min(150, max(20, (deadline - time.monotonic()) * 0.3))
        print(f"  Budget: {t_alloc:.0f}s")
        p_sg, sc_sg = subgradient_method(best_seq, t_alloc, eta0=0.03)
        best_seq, best_score = check_and_update(p_sg, "subgrad", best_seq, best_score, verbose=True)

    # === Phase 6: Peak flattening ===
    print(f"\n=== Phase 6: Peak Flattening (best={best_score:.8f}) ===")
    if time.monotonic() < deadline - 80:
        t_alloc = 30
        print(f"  Budget: {t_alloc}s")
        p_pf, sc_pf = peak_flatten(best_seq, n_reductions=10)
        best_seq, best_score = check_and_update(p_pf, "peak_flat", best_seq, best_score, verbose=True)

    # === Phase 7: Coordinate descent ===
    print(f"\n=== Phase 7: Coordinate Descent (best={best_score:.8f}) ===")
    if time.monotonic() < deadline - 50:
        t_alloc = min(150, max(20, (deadline - time.monotonic()) * 0.4))
        print(f"  Budget: {t_alloc:.0f}s")
        p_cd, sc_cd = coordinate_descent(best_seq, t_alloc, n_pairs=150)
        best_seq, best_score = check_and_update(p_cd, "coord_desc", best_seq, best_score, verbose=True)

    # === Phase 8: Final LP polish ===
    print(f"\n=== Phase 8: Final LP Polish (best={best_score:.8f}) ===")
    if time.monotonic() < deadline - 20:
        t_alloc = min(80, max(10, (deadline - time.monotonic()) * 0.5))
        if t_alloc > 10:
            print(f"  Budget: {t_alloc:.0f}s")
            p_fin, sc_fin = lp_fictitious_play(best_seq, t_alloc, use_geom=True)
            best_seq, best_score = check_and_update(p_fin, "final_lp", best_seq, best_score, verbose=True)

    # === Phase 9: Final coordinate descent ===
    if time.monotonic() < deadline - 10:
        t_alloc = min(30, max(5, (deadline - time.monotonic()) * 0.6))
        if t_alloc > 5:
            p_fin2, sc_fin2 = coordinate_descent(best_seq, t_alloc, n_pairs=50)
            best_seq, best_score = check_and_update(p_fin2, "final_cd", best_seq, best_score, verbose=True)

    # Fallback
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence([float(x) for x in best_seq]))
    if np.isfinite(fs):
        if fs != best_score:
            print(f"  Note: evaluate_sequence gives {fs:.8f} vs tracked {best_score:.8f}")
        best_score = min(best_score, fs)

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in np.maximum(best_seq, 0.0)]
