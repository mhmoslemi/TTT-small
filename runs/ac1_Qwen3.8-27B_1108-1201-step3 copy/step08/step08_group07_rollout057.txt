Adaptive mirror descent with log-sum-exp smoothing for minimizing
2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Key innovation: Fast FFT-based gradient computation for the soft-max
objective, combined with multiplicative weights (mirror descent) updates
on the simplex, adaptive learning rate, and beta-annealing schedule.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_linear_conv(a):
    """Compute full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score_fast(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_linear_conv(a)
    return 2.0 * n * conv.max() / (s * s)


def lse_objective(a, beta):
    """Compute log-sum-exp of beta*conv, plus normalization penalty.
    Returns (value, gradient) with respect to a (unconstrained)."""
    n = len(a)
    s = a.sum()
    conv = fast_linear_conv(a)
    bconv = beta * conv
    max_bc = bconv.max()
    exp_bc = np.exp(bconv - max_bc)
    lse = max_bc + np.log(exp_bc.sum())
    value = (2.0 * n / beta) * lse - 2.0 * np.log(s) if s > 0 else np.inf
    # Actually, let's keep it simple: normalize to sum=1 and just minimize 2n*max
    return value, exp_bc / exp_bc.sum()


def mirror_descent_step(a, w, conv, beta, eta):
    """One mirror descent step on the simplex.
    a: current point (sum=1, a>=0)
    w: softmax weights over conv lags
    conv: current convolution
    beta: smoothing parameter
    eta: step size
    Returns: new a (sum=1, a>=0)"""
    n = len(a)
    # Gradient: grad[j] = 2 * sum_k w[k] * a[k-j] (with a[m]=0 for m not in [0,n-1])
    # This is 2 * cross-correlation of w and a

    w_pad[:2*n-1] = w

    # = ifft(conj(fft(w_pad)) * fft(a_pad)).real
    A = rfft(a_pad, fft_size)  # since a_pad is real, rfft is fine
    W = rfft(w_pad, fft_size)
    # For real sequences: ifft(conj(W_full) * A_full) = ifft(W_full * A_full) since both are real... 
    # Actually no. Let me use full complex FFT to be safe.
    # Actually for real inputs, the correlation is:
    # corr[j] = sum_k w[k] * a[k-j]
    # This equals ifft(FFT(w) * conj(FFT(a)))... let me verify.
    # 
    # Standard: circular cross-correlation R_xy[m] = sum_n x[n+m] y*n]
    # FFT: R = ifft(conj(X) * Y) where X=FFT(x), Y=FFT(y)
    #
    # We want: corr[j] = sum_k w[k] a[k-j]
    # Let x = w, y = a. Then R[m] = sum_n w[n+m] a[n]
    # corr[j] = sum_k w[k] a[k-j]. Let n = k-j: sum_n w[n+j] a[n] = R[j].
    # So corr[j] = R[j] = ifft(conj(W) * A)[j]
    
    W_full = np.fft.rfft(w_pad, fft_size)
    A_full = np.fft.rfft(a_pad, fft_size)
    corr = np.fft.irfft(np.conj(W_full) * A_full, fft_size)[:n]
    
    grad = 2.0 * corr

    a_new = a * update
    Z = a_new.sum()
    if Z < 1e-300:

    a_new = a_new / Z
    return a_new

                eta = eta_init * 0.1
