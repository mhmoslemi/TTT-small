The optimal solution likely has a structure where the auto-convolution is as flat as possible (like a "perfect" autocorrelation sequence but with non-negative constraint).

For non-negative sequences, the best we can do is related to the "Sompiers" or "Zadoff-Chu" type sequences adapted to be non-negative. But these are typically complex-valued.

For real non-negative sequences, the constraint is much more restrictive. The auto-convolution is always non-negative, and by Parseval, `sum_k conv(a,a)[k] = (sum a)^2`. So the average value is `(sum a)^2 / (2n-1)`. The minimum possible max is at least this average, giving a lower bound of `2n * (sum a)^2/(2n-1) / (sum a)^2 = 2n/(2n-1)` which approaches 1 for large n.

But for non-negative a, achieving a flat auto-convolution is harder because you can't have negative lobes to cancel out.

"""
Algorithm: Adaptive Smoothed Subgradient Descent with Momentum for minimizing
2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Key innovations over previous approach:
1. Smoothed gradient (logsumexp) instead of LP for faster iterations
2. Barzilai-Borwein adaptive step sizes
3. Nesterov momentum for faster convergence
4. Adaptive epsilon cooling schedule
5. Multi-restart with diverse perturbations
6. Final LP polish for fine-tuning
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def gradient_smoothed(f, conv_vals, eps):
    """
    Compute gradient of f_eps(a) = eps * logsumexp(conv(a,a)/eps)
    grad[j] = 2 * sum_l w[j+l] * a[l] where w = softmax(conv/eps)
    """
    n = len(f)
    # Compute weights
    scaled = conv_vals / eps
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(len(conv_vals)) / len(conv_vals)
    else:
        w /= w_sum
    
    # Compute gradient via cross-correlation
    # grad[j] = 2 * sum_{l=0}^{n-1} w[j+l] * a[l]
    # Use FFT: conv(w, a_flip)[j+n-1]
    a_flip = f[::-1]
    size = len(w) + n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    AF = rfft(a_flip, fft_size)
    conv_result = irfft(W * AF, fft_size)
    g = conv_result[n-1 : 2*n-2]
    
    return 2.0 * g


def gradient_hard(f, conv_vals, top_k=20):
    """
    Compute subgradient of max(conv(a,a)) using top-k peaks.
    grad[j] = 2 * sum_{k in top} a[k-j] (for valid k-j)
    This is an averaged subgradient over the top-k peaks.
    """
    n = len(f)
    # Select top-k indices
    if top_k < len(conv_vals):
        top_idx = np.argpartition(conv_vals, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(conv_vals))
    
    # Equal weights on top peaks
    w = np.zeros(len(conv_vals))
    w[top_idx] = 1.0 / len(top_idx)
    
    # Same computation as smoothed gradient
    a_flip = f[::-1]
    size = len(conv_vals) + n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    AF = rfft(a_flip, fft_size)
    conv_result = irfft(W * AF, fft_size)
    g = conv_result[n-1 : 2*n-2]
    
    return 2.0 * g


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def bb_step(f_prev, f_curr, g_prev, g_curr):
    """Barzilai-Borwein step size."""
    s = f_curr - f_prev
    y = g_curr - g_prev
    ys = np.dot(y, s)
    sn = np.dot(s, s)
    yn = np.dot(y, y)
    if abs(ys) < 1e-30:
        return 0.01
    gamma = ys / yn if yn > 1e-30 else 0.01
    rho_bb = sn / ys if abs(ys) > 1e-30 else gamma
    # Alternate or use geometric mean
    if sn > 1e-30 and abs(ys) > 1e-30:
        return np.sqrt(gamma * rho_bb)
    return max(gamma, 1e-10)


def smoothed_gradient_descent(f_init, time_limit, eps_init=0.01, eps_min=0.001, 
                               momentum_init=0.9, max_momentum=0.98,
                               label="sgd"):
    """
    Smoothed subgradient descent with Nesterov momentum and BB steps.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f /= s
    
    v = np.zeros(n)  # momentum
    best_f = f.copy()
    conv_best = fast_convolve(best_f)
    best_obj = conv_best.max()
    
    # BB state
    f_prev = f.copy()
    g_prev = None
    lr = 1.0 / (2.0 * np.abs(f).max() + 1.0)  # initial learning rate
    
    it = 0
    no_improve = 0
    eps = eps_init
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute current convolution
        conv_vals = fast_convolve(f)
        cur_obj = conv_vals.max()
        
        # Update best
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Compute gradient
        if no_improve > 50:
            # Switch to hard gradient for getting out of plateaus
            g = gradient_hard(f, conv_vals, top_k=min(30, n//4))
        else:
            g = gradient_smoothed(f, conv_vals, eps)
        
        # BB step size
        if g_prev is not None:
            lr_bb = bb_step(f_prev, f, g_prev, g)
            lr = 0.5 * lr + 0.5 * lr_bb
            lr = np.clip(lr, 1e-10, 0.1)
        
        f_prev = f.copy()
        g_prev = g.copy()
        
        # Nesterov momentum
        momentum = momentum_init + (max_momentum - momentum_init) * min(1.0, it / 1000.0)
        v = momentum * v - lr * g
        f_new = project_simplex(f + v)
        
        # Line search (backtracking)
        obj_new = fast_convolve(f_new).max()
        step_factor = 1.0
        trial = 0
        while obj_new > cur_obj * (1 + 1e-10) and trial < 10:
            step_factor *= 0.5
            v = momentum * v - lr * step_factor * g
            f_new = project_simplex(f + v)
            obj_new = fast_convolve(f_new).max()
            trial += 1
            if step_factor < 1e-10:
                break
        
        f = f_new
        
        # Cool epsilon
        if it % 50 == 0:
            eps = max(eps_min, eps * 0.95)
        
        # Restart if stuck
        if no_improve > 200:
            rng_restart = np.random.default_rng(it)
            noise = rng_restart.standard_normal(n) * 0.01
            f = project_simplex(best_f + noise)
            v = np.zeros(n)
            no_improve = 0
            g_prev = None
        
        if it % 2000 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_obj
            print(f"    {label} it={it} obj={best_obj:.10f} score={score:.6f} "
                  f"eps={eps:.6f} lr={lr:.6f} t={elapsed:.0f}s")
    
    return best_f, best_obj


def lp_polish(f_init, time_limit, initial_k=50, max_k=400, label="lp"):
    """LP-based direction finding with cutting planes for final polish."""
    from scipy.fft import rfft, irfft
    
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / s
    
    b = fast_convolve(f)
    M = b.max()
    best_f = f.copy()
    best_obj = M
    
    # Active set: top K peaks
    initial_k_eff = min(initial_k, len(b))
    active_idx = np.sort(np.argpartition(b, -initial_k_eff)[-initial_k_eff:])
    
    def build_A(f, indices):
        K = len(indices)
        A = np.zeros((K, n))
        for i, k in enumerate(indices):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            A[i, lo:hi+1] = f[lo:hi+1]
        return A
    
    def solve_lp(f, M, idx):
        A = build_A(f, idx)
        g = cp.Variable(n)
        constraints = [g >= 0, A @ g <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=300000, eps=1e-8)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iters=50000)
            except Exception:
                return None
        if g.value is None:
            return None
        gv = np.maximum(g.value, 0.0)
        if np.any(np.isnan(gv)):
            return None
        return gv
    
    it = 0
    stuck = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break
        
        g = solve_lp(f, M, active_idx)
        if g is None:
            stuck += 1
            if stuck > 10:
                break
            continue
        
        S_g = g.sum()
        if S_g <= 1.0 + 1e-6:
            stuck += 1
            if stuck > 30:

            continue
        stuck = 0
        
        # Check for violated constraints (cutting plane)

        violated = fg > M * (1 + 1e-7)

            v_idx = np.where(violated)[0]
            not_active = v_idx[~np.isin(v_idx, active_idx)]
            if len(not_active) > 0:
                n_add = min(20, len(not_active))
                to_add = not_active[np.argsort(fg[not_active])[-n_add:]]
                active_idx = np.unique(np.concatenate([active_idx, to_add]))
                g = solve_lp(f, M, active_idx)
                if g is None:
                    stuck += 1
                    if stuck > 5:
                        break
                    continue
                S_g = g.sum()
                if S_g <= 1.0 + 1e-6:
                    stuck += 1
                    if stuck > 20:
                        break
                    continue
        
        h = g / S_g
        
        # Line search
        best_t = 0.1
        best_t_obj = M
        for t in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7]:
            f_trial = (1-t)*f + t*h
            f_trial = np.maximum(f_trial, 0)
            st = f_trial.sum()
            if st < 1e-10:
                continue
            f_trial /= st
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t
        
        f = (1 - best_t)*f + best_t*h
        f = np.maximum(f, 0)
        s = f.sum()
        if s > 1e-10:
            f /= s
        
        cur = fast_convolve(f).max()
        M = cur
        if cur < best_obj - 1e-13:
            best_obj = cur

                active_idx = np.unique(np.concatenate([keep, new_top]))
                active_idx = np.sort(active_idx[-max_k:])
    
    return best_f, best_obj


def convolve_cross_func(a, b):
    """Cross-convolution via FFT."""
    n1, n2 = len(a), len(b)
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

"""
Algorithm: Adaptive Smoothed Gradient Descent with Nesterov Momentum,
QCQP polish, and multi-length exploration for minimizing
2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Key improvements over previous LP-based approach:
1. O(n log n) per-iteration cost (FFT-based gradient) vs O(n^2) LP
2. Nesterov-accelerated momentum for faster convergence
3. Adaptive epsilon cooling for smooth->sharp gradient transition
4. Barzilai-Borwein step with backtracking safety
5. QCQP polish phase using cvxpy for fine-tuning
6. Multi-restart with diverse perturbations for escaping local minima
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_cross_func(a, b):
    """Linear cross-convolution a*b via FFT."""
    n1, n2 = len(a), len(b)
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient(f, conv_vals, eps):
    """
    Gradient of f_eps(a) = eps * logsumexp(conv(a,a)/eps).
    grad[j] = 2 * sum_{m=0}^{n-1} w[j+m] * a[m]
    where w = softmax(conv/eps).
    Computed via FFT cross-correlation.
    """
    n = len(f)
    scaled = conv_vals / eps
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w /= w_sum

    a_flip = f[::-1]
    size = (2 * n - 1) + n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    AF = rfft(a_flip, fft_size)
    conv_result = irfft(W * AF, fft_size)
    g = conv_result[n - 1: 2 * n - 2]
    return 2.0 * g


def compute_gradient_hard(f, conv_vals, top_k):
    """
    Averaged subgradient over top-k peaks of conv(a,a).
    Same structure as compute_gradient but with hard (one-hot averaged) weights.
    """
    n = len(f)
    k_eff = min(top_k, 2 * n - 1)
    top_idx = np.argpartition(conv_vals, -k_eff)[-k_eff:]
    w = np.zeros(2 * n - 1)
    w[top_idx] = 1.0 / k_eff

    a_flip = f[::-1]
    size = (2 * n - 1) + n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    AF = rfft(a_flip, fft_size)
    conv_result = irfft(W * AF, fft_size)
    g = conv_result[n - 1: 2 * n - 2]
    return 2.0 * g


def bb_step_size(s, y):
    """Barzilai-Borwein step size from step vector s and gradient diff y."""
    ys = np.dot(y, s)
    sn = np.dot(s, s)
    yn = np.dot(y, y)
    if abs(ys) < 1e-30 or sn < 1e-30:
        return 0.01
    gamma = ys / max(yn, 1e-30)
    rho_bb = sn / ys
    gamma = max(gamma, 1e-12)
    rho_bb = max(rho_bb, 1e-12)
    return np.sqrt(gamma * rho_bb)


def smoothed_gradient_descent(f_init, time_limit, eps_init=0.02, eps_min=0.002,
                               momentum_init=0.85, momentum_end=0.97,
                               label="sgd"):
    """
    Nesterov-accelerated smoothed subgradient descent on the simplex.
    Minimizes max(conv(a,a)) with sum(a)=1, a>=0.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f /= s

    v = np.zeros(n)
    best_f = f.copy()
    conv_best = fast_convolve(best_f)
    best_obj = conv_best.max()

    f_prev = f.copy()
    g_prev = None
    lr = 0.05
    eps = eps_init
    it = 0
    no_improve = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2:
            break

        conv_vals = fast_convolve(f)
        cur_obj = conv_vals.max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Choose gradient type based on stagnation
        if no_improve > 80:
            top_k = max(5, n // 8)
            g = compute_gradient_hard(f, conv_vals, top_k)
        else:
            g = compute_gradient(f, conv_vals, eps)

        # BB stepsize
        if g_prev is not None and it > 1:
            s_vec = f - f_prev
            y_vec = g - g_prev
            lr_bb = bb_step_size(s_vec, y_vec)
            lr = 0.3 * lr + 0.7 * lr_bb
            lr = np.clip(lr, 1e-8, 0.2)

        f_prev = f.copy()
        g_prev = g.copy()

        # Nesterov momentum
        momentum = momentum_init + (momentum_end - momentum_init) * min(1.0, it / 2000.0)
        v = momentum * v - lr * g
        f_new = project_simplex(f + v)

        # Backtracking line search
        obj_new = fast_convolve(f_new).max()
        trial = 0
        scale = 1.0
        while obj_new > cur_obj + 1e-12 and trial < 8:
            scale *= 0.5
            v = momentum * v - lr * scale * g
            f_new = project_simplex(f + v)
            obj_new = fast_convolve(f_new).max()
            trial += 1
            if scale < 1e-12:
                break
        f = f_new

        # Cool epsilon
        if it % 40 == 0:
            eps = max(eps_min, eps * 0.96)

        # Periodic restart from best with noise
        if no_improve > 300:
            noise_scale = 0.005 + 0.005 * (no_improve / 500.0)
            noise = rng.standard_normal(n) * noise_scale
            f = project_simplex(best_f + noise)
            v = np.zeros(n)
            no_improve = 0
            g_prev = None
            lr = 0.05

        # Track objective improvement for adaptive restart
        if no_improve > 150:
            noise_scale = 0.003
            noise = rng.standard_normal(n) * noise_scale
            f_candidate = project_simplex(f + noise)
            obj_candidate = fast_convolve(f_candidate).max()
            if obj_candidate < cur_obj:
                f = f_candidate
                v = np.zeros(n)
                no_improve = 0
                g_prev = None

        if it % 5000 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_obj
            print(f"    {label} it={it} obj={best_obj:.10f} score={score:.6f} "
                  f"eps={eps:.6f} lr={lr:.6f} noimp={no_improve} t={elapsed:.0f}s")

    return best_f, best_obj


def qcqp_polish(f_init, time_limit, n_constraints=50, label="qcqp"):
    """
    QCQP polish: minimize t s.t. a^T B_k a <= t for top-k peaks,
    a >= 0, sum(a) = 1.
    Solved via cvxpy (SOCP-representable).
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f /= s

    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 10:
            break

        conv_vals = fast_convolve(f)
        M = conv_vals.max()

        # Select active constraints (top peaks)
        k_eff = min(n_constraints,
                    int(n_constraints * (1.0 + 0.5 * np.log1p(it))))
        k_eff = min(k_eff, 2 * n - 1)
        top_idx = np.sort(np.argpartition(conv_vals, -k_eff)[-k_eff:])

        # Build QCQP
        a_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [a_var >= 0, cp.sum(a_var) == 1, t_var >= 0]

        for k in top_idx:
            # conv(a,a)[k] = a^T B_k a where B_k[j,l] = 1 if j-l = k-(n-1) adjusted
            # Actually B_k is the symmetric matrix such that a^T B_k a = conv[k]
            # conv[k] = sum_{j,l: 0<=j<n, 0<=l<n, j-l = k-(n-1)} a[j]*a[l]
            # Wait no: conv[k] = sum_l a[l]*a[k-l] for valid l
            # = sum_{l: max(0,k-n+1) <= l <= min(n-1,k)} a[l]*a[k-l]
            # This is a^T Q_k a where Q_k is the matrix with Q_k[i,j] = 1/2 if 
            # i+j = 2*(n-1)-k or... 
            # Actually: conv[k] = sum_i a[i] * sum_j (i+j = k mod something) ...
            # Let me just use: conv[k] = sum_l a[l]*a[k-l]
            # = sum_{l,j: j = k-l} a[l]*a[j] = sum_{l,j: l+j = k} a[l]*a[j]
            # Wait that's not right for linear convolution.
            # Linear conv: (a*a)[k] = sum_{l=0}^{n-1} a[l]*a[k-l] where 0<=k-l<n
            # So (a*a)[k] = sum_{l: 0<=l<n, 0<=k-l<n} a[l]*a[k-l]
            # Let j = k-l, so l = k-j: sum_{j: 0<=j<n, 0<=k-j<n} a[k-j]*a[j]
            # = sum_{j: max(0,k-n+1) <= j <= min(n-1,k)} a[j]*a[k-j]
            
            # As a quadratic form: a^T B_k a where B_k is symmetric with
            # B_k[j, k-j] = 1/2 and B_k[k-j, j] = 1/2 (for j != k/2)
            # and B_k[j,j] = 1 if j = k-j (i.e., j = k/2)
            
            # For cvxpy, we can express this as:
            # sum_{j} a[j] * a[k-j] (with valid indices)
            # = 2*sum_{j < k-j} a[j]*a[k-j] + a[k/2]^2 (if k even)
            
            # Simpler: just express as a @ diag stuff... 
            # Actually the cleanest way: 
            # conv[k] = a^T (shift_k) a where shift_k is the anti-diagonal matrix
            # For cvxpy, we can use: (a[j] * a[k-j]) terms explicitly
            
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            # Terms: a[l] * a[k-l] for l in [lo, hi]
            # This is a quadratic: sum_{l=lo}^{hi} a[l] * a[k-l]
            # In cvxpy: cp.quad_form with appropriate matrix is expensive
            # Instead, express as sum of products
            
            # For small k_eff, just add the constraint directly
            # conv[k] <= t means sum_{l} a[l]*a[k-l] <= t
            # This is a quadratic constraint. cvxpy can handle it if we write
            # it as cp.sum(cp.multiply(a_var[l], a_var[k-l]))
            
            # But this might be slow. Alternative: use the fact that
            # sum_l a[l]*a[k-l] <= t is equivalent to
            # [a, t^{1/2}] being in a rotated cone... but that's for a^T a forms.
            
            # For now, let's use a simpler approach: iterative LP within QCQP
            # Just skip QCQP if it's too complex and use LP instead.
            pass

        # Alternative: use successive LP with updated M
        # (This is essentially what the previous code did, just with tighter M updates)
        # Let me use a direct approach: solve the LP for direction, update, iterate
        
        # Build LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M for k in top_idx
        A_mat = np.zeros((k_eff, n))
        for i, k in enumerate(top_idx):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            A_mat[i, lo:hi + 1] = f[lo:hi + 1]
        
        g_var = cp.Variable(n)
        lp_constraints = [g_var >= 0, A_mat @ g_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g_var)), lp_constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=200000, eps=1e-8)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, max_iters=30000)
            except Exception:
                continue
        
        if g_var.value is None:
            continue
        g_val = np.maximum(g_var.value, 0.0)
        S_g = g_val.sum()
        if S_g <= 1.0 + 1e-6:
            continue
        
        h = g_val / S_g
        
        # Line search
        best_t = 0.1
        best_t_obj = M
        for t in [0.03, 0.07, 0.12, 0.2, 0.3, 0.45, 0.6]:
            f_trial = (1 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0)
            st = f_trial.sum()
            if st < 1e-10:
                continue
            f_trial /= st
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t
        
        f = (1 - best_t) * f + best_t * h
        f = np.maximum(f, 0)
        s = f.sum()
        if s > 1e-10:
            f /= s
        
        cur = fast_convolve(f).max()
        if cur < best_obj - 1e-13:
            best_obj = cur
            best_f = f.copy()
        
        if it % 10 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_obj
            print(f"    {label} it={it} obj={best_obj:.10f} score={score:.6f} t={elapsed:.0f}s")

    return best_f, best_obj


def diverse_restart_search(f_init, time_limit, n_restarts=5, label="restart"):
    """Try multiple diverse perturbations and run short SGD from each."""
    t0 = time.monotonic()
    n = len(f_init)
    best_f = f_init.copy()
    best_obj = fast_convolve(f_init).max()
    rng = np.random.default_rng(777)
    
    time_per_restart = (time_limit - 10.0) / n_restarts
    
    for r in range(n_restarts):
        if time.monotonic() - t0 > time_limit - 5:
            break
        
        # Generate diverse perturbation
        noise = rng.standard_normal(n)
        # Apply structured perturbation (low-frequency)
        freq = rfft(noise)
        decay = np.exp(-np.linspace(0, 3, len(freq)))
        freq *= decay
        noise = irfft(freq) / (np.linalg.norm(noise) + 1e-10)
        
        scale = 0.01 * (1 + 0.5 * r)
        f_trial = project_simplex(f_init + scale * noise)
        
        # Run short SGD
        t_sub = time_per_restart
        f_sub, obj_sub = smoothed_gradient_descent(
            f_trial, time_limit=t_sub,
            eps_init=0.01, eps_min=0.003,
            momentum_init=0.8, momentum_end=0.93,
            label=f"{label}_r{r}")
        
        score = evaluate_sequence(f_sub.tolist())
        if np.isfinite(score) and score < 2 * n * best_obj:
            best_obj = obj_sub
            best_f = f_sub.copy()
            print(f"    {label} restart {r}: improved to score={2.0*n*best_obj:.6f}")
    
    return best_f, best_obj


def multi_length_search(f_init, time_limit, label="multilength"):
    """Try different sequence lengths by interpolation."""
    t0 = time.monotonic()
    n_base = len(f_init)
    best_f = f_init.copy()
    best_obj = fast_convolve(f_init).max()
    
    # Candidate lengths
    factors = [0.85, 0.92, 1.08, 1.15, 1.3]
    target_lengths = []
    for fac in factors:
        tl = int(n_base * fac)
        tl = max(256, min(4000, tl))
        if tl != n_base and tl not in target_lengths:
            target_lengths.append(tl)
    
    if not target_lengths:
        return best_f, best_obj
    
    time_per = (time_limit - 10.0) / len(target_lengths)
    
    for tgt_n in target_lengths:
        if time.monotonic() - t0 > time_limit - 5:
            break
        if time_per < 30:
            break
        
        # Interpolate
        old_x = np.linspace(0.0, 1.0, n_base)
        new_x = np.linspace(0.0, 1.0, tgt_n)
        f_interp = np.interp(new_x, old_x, f_init)
        f_interp = np.maximum(f_interp, 0.0)
        s = f_interp.sum()
        if s > 1e-10:
            f_interp /= s
        else:
            f_interp = np.ones(tgt_n) / tgt_n
        
        # Run SGD
        f_out, obj_out = smoothed_gradient_descent(
            f_interp, time_limit=time_per,
            eps_init=0.015, eps_min=0.003,
            label=f"{label}_n{tgt_n}")
        
        score_out = evaluate_sequence(f_out.tolist())
        score_best = evaluate_sequence(best_f.tolist())
        if np.isfinite(score_out) and score_out < score_best:
            best_f = f_out.copy()
            best_obj = obj_out
            print(f"    {label} n={tgt_n}: improved to score={score_out:.6f}")
    
    return best_f, fast_convolve(best_f).max()
