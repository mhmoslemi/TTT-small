import numpy as np
import time
import math
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute score for normalized sequence (sum=1): 2*n*max(conv)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def compute_score_full(a):
    """Full score: 2*n*max_conv/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def build_C(f):
    """Build constraint matrix C where C[k,j] = f[k-j] if 0<=k-j<n else 0."""
    n = len(f)
    # C has shape (2n-1, n)
    # C[k,j] = f[k-j] if 0 <= k-j <= n-1, else 0
    # For row k: C[k,j] = f[k-j] for j in [max(0,k-n+1), min(k, n-1)]
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        # f[k-j] for k from j to j+n-1
        C[j:j + n, j] = f
    return C


def solve_lp(f, M, solver='CLARABEL', timeout=30):
    """Solve LP: max sum(g) s.t. C*g <= M*1, g >= 0."""
    n = len(f)
    C = build_C(f)
    g_var = cp.Variable(n, nonneg=True)
    constraints = [C @ g_var <= M]
    obj = cp.Maximize(cp.sum(g_var))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if g_var.value is None:
        return None
    return np.maximum(g_var.value, 0)


def solve_lp_scipy(f, M):
    """Fallback LP solver using scipy."""
    n = len(f)
    C = build_C(f)
    c = -np.ones(n)
    A_ub = C
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    if res.status != 0:
        return None
    return np.maximum(res.x, 0)


def geometric_mean_update(f, g, t):
    """Geometric mean interpolation in log-space."""
    f_safe = np.maximum(f, 1e-30)
    g_safe = np.maximum(g, 1e-30)
    log_new = (1 - t) * np.log(f_safe) + t * np.log(g_safe)
    log_new -= log_new.max()
    a = np.exp(log_new)
    s = a.sum()
    if s < 1e-30:
        return f.copy()
    return a / s


def lp_fictitious_play(f_init, time_limit, use_adaptive_M=True, verbose=True):
    """LP-based fictitious play with geometric mean updates and adaptive M."""
    f = np.maximum(f_init.astype(np.float64), 0)
    s = f.sum()
    f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_score = compute_score(f)
    t0 = time.monotonic()
    
    # Try different solvers
    solvers = ['CLARABEL', 'ECOS', 'SCS']
    solver_idx = 0
    
    for k in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(f)
        M = conv.max()
        
        # Adaptive M: try slightly below max to push harder
        if use_adaptive_M and k > 10:
            M_eff = M * (1.0 - 0.01 * min(1.0, k / 100.0))
        else:
            M_eff = M
        
        # Solve LP
        g = solve_lp(f, M_eff, solver=solvers[solver_idx % len(solvers)])
        if g is None:
            # Try next solver
            solver_idx += 1
            g = solve_lp(f, M_eff, solver=solvers[solver_idx % len(solvers)])
        if g is None:
            g = solve_lp_scipy(f, M_eff)
        
        if g is None or g.sum() < 1e-15:
            # If adaptive M is too aggressive, fall back to M
            if M_eff < M:
                g = solve_lp(f, M, solver=solvers[0])
                if g is None:
                    g = solve_lp_scipy(f, M)
            if g is None or g.sum() < 1e-15:
                break
        
        g = g / g.sum()
        
        # Adaptive step size
        t_step = 2.0 / (k + 2)
        if k > 100:
            t_step = max(0.005, 1.5 / (k + 2))
        
        a_new = geometric_mean_update(f, g, t_step)
        
        score_cur = compute_score(a_new)
        if score_cur < best_score - 1e-14:
            best_score = score_cur
            best_f = a_new.copy()
        
        # Also check: what if we use the LP solution directly (normalized)?
        g_norm = g / g.sum()
        score_g = compute_score(g_norm)
        if score_g < best_score - 1e-14:
            best_score = score_g
            best_f = g_norm.copy()
        
        f = a_new
        
        if verbose and k % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter={k}, M={M:.6f}, M_eff={M_eff:.6f}, "
                  f"score={score_cur:.8f}, best={best_score:.8f}, t={elapsed:.0f}s")
    
    return best_f, best_score


def dual_guided_cd(a0, time_limit, verbose=True):
    """Coordinate descent guided by LP dual variables."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    iters = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False
        
        conv = fast_convolve(a)
        n_conv = len(conv)
        
        # Find top peaks
        top_k = min(10, n_conv)
        peak_indices = np.argsort(conv)[::-1][:top_k]
        
        # For each peak, find contributing coordinates and try to reduce them
        for pk in peak_indices:
            if time.monotonic() - t0 > time_limit:
                break
            
            js = max(0, int(pk) - n + 1)
            je = min(n - 1, int(pk))
            idx = np.arange(js, je + 1)
            
            if len(idx) == 0:
                continue
            
            # Contributions: a[idx] * a[pk - idx]
            contribs = a[idx] * a[int(pk) - idx]
            
            # Try reducing top contributors by transferring to non-contributing positions
            top_contrib = np.argsort(contribs)[::-1][:5]
            
            # Non-contributing positions
            noncontrib = np.setdiff1d(np.arange(n), idx)
            
            for ci in top_contrib:
                if time.monotonic() - t0 > time_limit:
                    break
                i_idx = idx[ci]
                if a[i_idx] < 1e-15:
                    continue
                
                # Try different fractions to transfer
                for frac in [0.01, 0.03, 0.05, 0.1, 0.2]:
                    delta = a[i_idx] * frac
                    
                    # Choose target: prefer positions that contribute to fewer peaks
                    if len(noncontrib) > 0:
                        # Score each noncontrib position by how many top peaks it contributes to
                        target_scores = np.zeros(len(noncontrib))
                        for tpi in peak_indices:
                            tj = noncontrib
                            # Position tj contributes to peak tpi if 0 <= tpi - tj < n
                            # and 0 <= tpi - (n-1-tj) < n (symmetric)
                            # Actually, position tj contributes to convolution at positions tj and 2n-2-tj
                            # More precisely, a[tj] contributes to conv[k] for k in [tj, tj+n-1]
                            for pos in tj:
                                if pos in peak_indices or (2*n-2-pos) in peak_indices:
                                    target_scores[list(tj).index(pos)] += 1
                        j_local = np.argmin(target_scores)
                        j_idx = noncontrib[j_local]
                    else:
                        j_idx = i_idx + 1 if i_idx + 1 < n else i_idx - 1
                        if j_idx == i_idx:
                            continue
                    
                    if j_idx == i_idx:
                        continue
                    
                    a_trial = a.copy()
                    a_trial[i_idx] -= delta
                    a_trial[j_idx] += delta
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-13:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
        
        # Also do some random pair moves
        n_random = min(100, n)
        for _ in range(n_random):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            for frac in [0.01, 0.05, 0.1]:
                delta = min(a[i], a[j]) * frac
                for sign in [1, -1]:
                    a_trial = a.copy()
                    a_trial[i] -= sign * delta
                    a_trial[j] += sign * delta
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-13:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
        
        if not improved:
            no_improve += 1
            if no_improve > 20:
                # Perturb
                noise = rng.standard_normal(n) * 0.003
                a = best_a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
        else:
            no_improve = 0
        
        if verbose and iters % 50 == 0:
            print(f"    DCD iter={iters}, score={best_score:.8f}, t={time.monotonic()-t0:.0f}s")
    
    return best_a, best_score


def flatten_convolution(a0, time_limit, verbose=True):
    """Directly try to flatten the convolution by adjusting peaks."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(456)
    
    iters = 0
    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        mean_conv = conv.mean()
        max_conv = conv.max()
        
        if max_conv < mean_conv * 1.01:
            # Already quite flat
            break
        
        # Find the peak
        pk = np.argmax(conv)
        js = max(0, int(pk) - n + 1)
        je = min(n - 1, int(pk))
        idx = np.arange(js, je + 1)
        
        # The peak value is sum of a[i]*a[pk-i] for i in idx
        # To reduce it, we want to reduce the product terms
        # Strategy: for the largest product term, reduce one factor and increase another
        contributors = a[idx] * a[int(pk) - idx]
        
        # Find the maximum contributor
        ci = np.argmax(contributors)
        i_main = idx[ci]
        j_main = int(pk) - i_main
        
        # Try reducing a[i_main] and a[j_main] slightly, and redistributing
        for frac in [0.005, 0.01, 0.02, 0.05]:
            delta = min(a[i_main], a[j_main]) * frac / 2
            a_trial = a.copy()
            a_trial[i_main] -= delta
            a_trial[j_main] -= delta
            # Redistribute to random positions far from peak
            redist = 2 * delta
            # Add to positions that don't contribute to this peak
            noncontrib = np.setdiff1d(np.arange(n), idx)
            if len(noncontrib) > 0:
                targets = noncontrib[rng.integers(0, len(noncontrib), size=min(5, len(noncontrib)))]
                for t in targets:
                    a_trial[t] += redist / len(targets)
            
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-15:
                continue
            a_trial = a_trial / ss
            sc = compute_score(a_trial)
            if sc < best_score - 1e-13:
                a = a_trial
                best_score = sc
                best_a = a.copy()
                break
        
        if verbose and iters % 100 == 0:
            print(f"    Flatten iter={iters}, max_conv={max_conv:.6f}, mean={mean_conv:.6f}, "
                  f"score={best_score:.8f}, t={time.monotonic()-t0:.0f}s")
    
    return best_a, best_score


def subgradient_method(a0, time_limit, num_restarts=3, verbose=True):
    """Subgradient method for the smoothed objective."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(789)
    
    for restart in range(num_restarts):
        if time.monotonic() - t0 > time_limit:
            break
        
        if restart > 0:
            # Perturb
            noise = rng.standard_normal(n) * 0.01
            a = best_a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
        
        step_size = 0.1
        patient = 0
        
        for k in range(5000):
            if time.monotonic() - t0 > time_limit:
                break
            
            conv = fast_convolve(a)
            max_val = conv.max()
            max_idx = np.argmax(conv)
            
            # Subgradient of max(conv) w.r.t. a:
            # If conv[k] = a * a (cross correlation), then dcov[k]/da[j] = a[k-j] (if 0<=k-j<n)
            # Subgradient of max is the gradient at the argmax
            # dmax/da[j] = a[max_idx - j] if 0 <= max_idx - j <= n-1, else 0
            
            grad = np.zeros(n)
            j_start = max(0, int(max_idx) - n + 1)
            j_end = min(n - 1, int(max_idx))
            for j in range(j_start, j_end + 1):
                grad[j] = a[int(max_idx) - j]
            
            # Normalize gradient
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            
            # Update: move in negative gradient direction, projected to simplex
            a_new = a - step_size * grad / grad_norm
            # Project to simplex (sum=1, nonnegative)
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                break
            a_new = a_new / s
            
            sc = compute_score(a_new)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a_new.copy()
                patient = 0
            else:
                patient += 1
            
            a = a_new
            
            # Adaptive step size
            if patient > 50:
                step_size *= 0.9
                patient = 0
        
        if verbose:
            print(f"    Subgrad restart={restart}, best={best_score:.8f}")
    
    return best_a, best_score


def try_length(a_best, target_n, time_limit, verbose=True):
    """Interpolate to new length and optimize."""
    old_n = len(a_best)
    if old_n == target_n:
        a = a_best.copy()
    else:
        s = a_best.sum()
        a_norm = a_best / s
        old_x = np.linspace(0, 1, old_n)
        new_x = np.linspace(0, 1, target_n)
        a = np.interp(new_x, old_x, a_norm)
        a = np.maximum(a, 0)
        ss = a.sum()
        if ss > 1e-15:
            a = a / ss
    
    # Run LP fictitious play on new length
    a_opt, sc = lp_fictitious_play(a, time_limit * 0.6, verbose=False)
    
    # Run flatten on the result
    a_opt2, sc2 = flatten_convolution(a_opt, time_limit * 0.3, verbose=False)
    
    return a_opt2, sc2

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Score for normalized sequence (sum=1): 2*n*max(conv)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_C(f):
    """Build constraint matrix C: C[k,j] = f[k-j] if 0<=k-j<n else 0."""
    n = len(f)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = f
    return C


def solve_max_width_lp(f, M, solver='CLARABEL'):
    """LP: max sum(g) s.t. C*g <= M, g >= 0."""
    n = len(f)
    C = build_C(f)
    g_var = cp.Variable(n, nonneg=True)
    constraints = [C @ g_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        try:
            prob.solve(solver='ECOS', verbose=False)
        except Exception:
            return None
    if g_var.value is None:
        return None
    return np.maximum(g_var.value, 0)


def solve_min_peak_lp(f, solver='CLARABEL'):
    """LP: min t s.t. (f*g)(k) <= t for all k, sum(g)=1, g>=0.
    Returns (g_opt, t_opt)."""
    n = len(f)
    C = build_C(f)
    g_var = cp.Variable(n, nonneg=True)
    t_var = cp.Variable()
    constraints = [
        C @ g_var <= t_var,
        cp.sum(g_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        try:
            prob.solve(solver='ECOS', verbose=False)
        except Exception:
            try:
                prob.solve(solver='SCS', verbose=False)
            except Exception:
                return None, None
    if g_var.value is None:
        return None, None
    return np.maximum(g_var.value, 0), float(t_var.value)


def geometric_mean_update(f, g, t):
    """Geometric mean interpolation in log-space, normalized."""
    f_safe = np.maximum(f, 1e-30)
    g_safe = np.maximum(g, 1e-30)
    log_new = (1.0 - t) * np.log(f_safe) + t * np.log(g_safe)
    log_new -= log_new.max()
    a = np.exp(log_new)
    s = a.sum()
    if s < 1e-30:
        return f.copy()
    return a / s

            je = min(n - 1, int(pk))
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue

            contribs = a[idx] * a[int(pk) - idx]
            top_ci = np.argsort(contribs)[::-1][:3]

            for ci in top_ci:
                i_idx = idx[ci]
                j_idx = int(pk) - i_idx
                if a[i_idx] < 1e-15 and a[j_idx] < 1e-15:
                    continue

                for frac in [0.005, 0.01, 0.02, 0.05]:
                    delta = min(a[i_idx], a[j_idx]) * frac * 0.5
                    if delta < 1e-18:
                        continue
                    a_trial = a.copy()
                    a_trial[i_idx] -= delta
                    if j_idx != i_idx:
                        a_trial[j_idx] -= delta
                        redist = 2.0 * delta
                    else:
                        redist = delta

                    # Redistribute to positions far from this peak
                    noncontrib = np.setdiff1d(np.arange(n), idx)
                    if len(noncontrib) > 0:
                        ntargets = min(10, len(noncontrib))
                        targets = noncontrib[rng.integers(0, len(noncontrib), size=ntargets)]
                        for t_pos in targets:
                            a_trial[int(t_pos)] += redist / len(targets)

                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-13:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break

        if verbose and iters % 100 == 0:
            print(f"    Flatten iter={iters}, max={max_val:.6e}, "
                  f"score={best_score:.8f}, t={time.monotonic()-t0:.0f}s")

    return best_a, best_score


def coord_descent_refine(a0, time_limit, verbose=True):
    """Coordinate descent with golden section on pairs."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(99)

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    iters = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False

        conv = fast_convolve(a)
        n_conv = len(conv)

        # Generate candidate pairs
        pairs = []
        # Peak-focused pairs
        top_pk = np.argmax(conv)
        js = max(0, int(top_pk) - n + 1)
        je = min(n - 1, int(top_pk))
        for _ in range(50):
            i = rng.integers(js, je + 1)
            j = rng.integers(0, n)
            if i != j:
                pairs.append((min(i, j), max(i, j)))
        # Random pairs
        for _ in range(100):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                pairs.append((min(i, j), max(i, j)))

        for (i, j) in pairs:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14 and a[j] < 1e-14:
                continue

            t_lo = -a[j] + 1e-15
            t_hi = a[i] - 1e-15
            if t_hi <= t_lo:
                continue

            x1 = t_hi - gr * (t_hi - t_lo)
            x2 = t_lo + gr * (t_hi - t_lo)

            a_trial = a.copy()

            def score_at(t_val):
                trial_i = a[i] - t_val
                trial_j = a[j] + t_val
                if trial_i < -1e-15 or trial_j < -1e-15:
                    return 1e10
                a[i] = max(0, trial_i)
                a[j] = max(0, trial_j)
                return compute_score(a)

            f1 = score_at(x1)
            f2 = score_at(x2)

            for _ in range(15):
                if f1 < f2:
                    t_hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = t_hi - gr * (t_hi - t_lo)
                    f1 = score_at(x1)
                else:
                    t_lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = t_lo + gr * (t_hi - t_lo)
                    f2 = score_at(x2)

            t_opt = (t_lo + t_hi) / 2.0
            f_opt = min(f1, f2)

            if f_opt < best_score - 1e-13:
                a[i] = max(0, a[i] - t_opt)
                a[j] = max(0, a[j] + t_opt)
                ss = a.sum()
                if ss > 1e-15:
                    a = a / ss
                best_score = compute_score(a)
                best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 10:
                noise = rng.standard_normal(n) * 0.003
                a = best_a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
        else:
            no_improve = 0

        if verbose and iters % 20 == 0:
            print(f"    CD iter={iters}, score={best_score:.8f}, t={time.monotonic()-t0:.0f}s")

    return best_a, best_score


def subgradient_refine(a0, time_limit, verbose=True):
    """Subgradient descent on the max-convolution objective."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(555)

    step_size = 0.05
    patience_count = 0

    for k in range(20000):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(a)
        pk = int(np.argmax(conv))

        # Subgradient at the peak
        grad = np.zeros(n)
        j_start = max(0, pk - n + 1)
        j_end = min(n - 1, pk)
        for j in range(j_start, j_end + 1):
            grad[j] = a[pk - j]

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        a_new = a - step_size * grad / gnorm
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss

        sc = compute_score(a_new)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()
            patience_count = 0
        else:
            patience_count += 1

        a = a_new

        if patience_count > 30:
            step_size *= 0.95
            patience_count = 0

        if verbose and k % 500 == 0:
            print(f"    Subgrad k={k}, step={step_size:.4f}, "
                  f"score={best_score:.8f}, t={time.monotonic()-t0:.0f}s")

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_update(a, label, best_seq, best_score, verbose=True):
    """Evaluate via evaluate_sequence and update best."""
    if a is None:
        return best_seq, best_score
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
        best_score = score
        best_seq = a_norm.copy()
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining min-peak LP, max-width LP,
    alternating LP, peak flattening, coordinate descent, and subgradient.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    n_parent = 0
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            n_parent = len(parent)
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={n_parent}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        n_parent = 768
        parent = np.ones(n_parent) / n_parent
        best_seq = parent.copy()
        best_score = compute_score(parent)
        print(f"Initialized uniform: n={n_parent}, score={best_score:.8f}")

    # === Phase 1: Min-Peak LP Fictitious Play (main novelty) ===
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.30, 540.0)
    print(f"\n=== Phase 1: Min-Peak LP Play (budget={t1:.0f}s) ===")
    if t1 > 30:
        a1, sc1 = lp_min_peak_play(best_seq, t1, verbose=True)
        best_seq, best_score = check_update(a1, "minpeak_lp", best_seq, best_score)

    # === Phase 2: Max-Width LP Fictitious Play ===
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.20, 300.0)
    print(f"\n=== Phase 2: Max-Width LP Play (budget={t2:.0f}s) ===")
    if t2 > 30:
        a2, sc2 = lp_max_width_play(best_seq, t2, verbose=True)
        best_seq, best_score = check_update(a2, "maxwidth_lp", best_seq, best_score)

    # === Phase 3: Alternating LP ===
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.20, 300.0)
    print(f"\n=== Phase 3: Alternating LP (budget={t3:.0f}s) ===")
    if t3 > 30:
        a3, sc3 = lp_alternating_play(best_seq, t3, verbose=True)
        best_seq, best_score = check_update(a3, "alt_lp", best_seq, best_score)

    # === Phase 4: Peak Flattening ===
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.15, 180.0)
    print(f"\n=== Phase 4: Peak Flattening (budget={t4:.0f}s) ===")
    if t4 > 20:
        a4, sc4 = flatten_peaks(best_seq, t4, verbose=True)
        best_seq, best_score = check_update(a4, "flatten", best_seq, best_score)

    # === Phase 5: Min-Peak LP again on flattened result ===
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.15, 200.0)
    print(f"\n=== Phase 5: Min-Peak LP Polish (budget={t5:.0f}s) ===")
    if t5 > 20:
        a5, sc5 = lp_min_peak_play(best_seq, t5, verbose=False)
        best_seq, best_score = check_update(a5, "minpeak_polish", best_seq, best_score)

    # === Phase 6: Coordinate Descent ===
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.10, 120.0)
    print(f"\n=== Phase 6: Coordinate Descent (budget={t6:.0f}s) ===")
    if t6 > 15:
        a6, sc6 = coord_descent_refine(best_seq, t6, verbose=False)
        best_seq, best_score = check_update(a6, "cd", best_seq, best_score)

    # === Phase 7: Subgradient ===
    rem = deadline - time.monotonic()
    t7 = min(rem * 0.08, 100.0)
    print(f"\n=== Phase 7: Subgradient (budget={t7:.0f}s) ===")
    if t7 > 15:
        a7, sc7 = subgradient_refine(best_seq, t7, verbose=False)
        best_seq, best_score = check_update(a7, "subgrad", best_seq, best_score)

    # === Phase 8: Length Exploration ===
    print(f"\n=== Phase 8: Length Exploration ===")
    target_lengths = [1024, 1536, 2048, 512, 640, 896]
    target_lengths = [t for t in target_lengths if t != n_parent and 128 <= t <= 4096]

    for tl in target_lengths:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_alloc = min(rem * 0.25, 120.0)
        if t_alloc < 30:
            break
        print(f"  Length n={tl} (budget={t_alloc:.0f}s)")
        a_interp = interpolate_seq(best_seq, tl)
        a_opt, sc_opt = lp_min_peak_play(a_interp, t_alloc * 0.5, verbose=False)
        best_seq, best_score = check_update(a_opt, f"n{tl}_minpeak", best_seq, best_score)

        rem2 = deadline - time.monotonic()
        if rem2 > 20:
            a_fl, sc_fl = flatten_peaks(a_opt, min(rem2 * 0.3, 40.0), verbose=False)
            best_seq, best_score = check_update(a_fl, f"n{tl}_flat", best_seq, best_score)

    # === Phase 9: Final polish ===
    rem = deadline - time.monotonic()
    t9 = min(rem * 0.4, 90.0)
    print(f"\n=== Phase 9: Final Min-Peak LP Polish (budget={t9:.0f}s) ===")
    if t9 > 15:
        a9, sc9 = lp_min_peak_play(best_seq, t9, verbose=False)
        best_seq, best_score = check_update(a9, "final_minpeak", best_seq, best_score)

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs) and fs < best_score:
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    if best_seq is None:
        best_seq = parent if parent is not None else np.ones(100) / 100.0
        best_score = float(evaluate_sequence(best_seq.tolist()))

    return [float(x) for x in best_seq]
