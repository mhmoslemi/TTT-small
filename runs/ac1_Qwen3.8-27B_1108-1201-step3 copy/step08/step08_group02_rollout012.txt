```python
def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    n = len(f)
    if n == new_n:
        return f.copy()
    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

def gaussian_init(n, sigma_frac=0.3, seed=0):
    """Gaussian-shaped initialization."""
    x = np.arange(n)
    center = n / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a /= a.sum()
    return a

def two_bump_init(n, sep_frac=0.3, seed=0):
    """Two-bump initialization."""
    x = np.arange(n)
    c1 = n * (0.5 - sep_frac / 2)
    c2 = n * (0.5 + sep_frac / 2)
    sigma = n * 0.1
    a = np.exp(-0.5 * ((x - c1) / sigma) ** 2) + np.exp(-0.5 * ((x - c2) / sigma) ** 2)
    a /= a.sum()
    return a
```

```python
"""
Smoothed gradient optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses log-sum-exp smoothing with FFT-based O(n log n) gradient computation,
momentum-accelerated projected gradient descent on the simplex,
adaptive temperature annealing, multi-length exploration, and
coordinate descent polishing.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    v = v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_grad(a, w, tau):
    """
    Compute gradient of log-sum-exp(conv(a,a)/tau) w.r.t. a.
    Uses FFT for O(n log n) computation.
    grad[j] = (1/tau) * sum_k w[k] * (a[k-j] + a[j-k])
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    # Cross-correlation: sum_k w[k] * a[k-j]
    c1 = irfft(np.conj(W) * A, fft_size)[:n]
    # Convolution: sum_k w[k] * a[j-k]
    c2 = irfft(W * A, fft_size)[:n]
    return (c1 + c2) / tau


def smoothed_pgd(a_init, time_limit, tau_init=None, tau_min=None,
                 lr_init=0.05, momentum=0.85, label="pgd",
                 perturbation_rate=0.005, verbose=True):
    """
    Smoothed projected gradient descent with momentum for minimizing
    max(conv(a,a)) subject to a >= 0, sum(a) = 1.
    """
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n

    b = fast_convolve(a)
    M = b.max()
    best_a = a.copy()
    best_M = M

    if tau_init is None:
        tau_init = 0.05 * M
    if tau_min is None:
        tau_min = 0.003 * M
    if tau_min > tau_init:
        tau_min = tau_init * 0.05

    lr = lr_init
    v = np.zeros(n)
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Adaptive tau: geometric decay
        tau = max(tau_min, tau_init * (0.985 ** it))

        # Compute weights from current convolution
        b = fast_convolve(a)
        b_scaled = b / tau
        b_max_val = b_scaled.max()
        exp_b = np.exp(b_scaled - b_max_val)
        w = exp_b / exp_b.sum()

        # Compute gradient
        grad = compute_smoothed_grad(a, w, tau)

        # Project gradient to tangent space of simplex (sum=0)
        grad_tan = grad - np.mean(grad)

        # Momentum update
        v = momentum * v - lr * grad_tan
        a_new = project_simplex(a + v)

        # Evaluate
        b_new = fast_convolve(a_new)
        M_new = b_new.max()

        if M_new < M - 1e-13:
            a = a_new
            M = M_new
            if M < best_M - 1e-13:
                best_M = M
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            # Slowly increase learning rate
            lr = min(lr * 1.005, lr_init * 4.0)
        else:
            no_improve += 1
            lr *= 0.85
            v *= 0.5

            if lr < lr_init * 0.01:
                # Perturb from best and restart
                rng = np.random.default_rng(it)
                a = best_a + rng.standard_normal(n) * perturbation_rate * best_a
                a = np.maximum(a, 0)
                s = a.sum()
                if s > 1e-10:
                    a /= s
                else:
                    a = best_a.copy()
                v = np.zeros(n)
                lr = lr_init
                no_improve = 0
                # Reset tau to be a bit larger

            score = 2.0 * n * best_M

            break

    b_final = fast_convolve(a).max()
    if b_final < best_M:
        best_M = b_final
        best_a = a.copy()

    return best_a, best_M


def coordinate_descent(a_init, time_limit, label="cd", verbose=True):
    """
    Coordinate descent: try scaling each element by various multipliers.
    """
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy()
    s = a.sum()
    if s > 1e-10:
        a /= s

    best_M = fast_convolve(a).max()
    best_a = a.copy()

    multipliers = [0.5, 0.7, 0.85, 0.92, 1.08, 1.15, 1.3, 1.5, 2.0]

    passes = 0
    consecutive_no_improve = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if a[j] < 1e-15:
                continue

            cur_val = a[j]
            best_val = cur_val
            best_obj = best_M

            for mult in multipliers:
                trial = a.copy()
                trial[j] = cur_val * mult
                s_t = trial.sum()
                if s_t < 1e-10:
                    continue
                trial /= s_t
                obj = fast_convolve(trial).max()
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_val = cur_val * mult

            if best_val != cur_val:
                a[j] = best_val
                s = a.sum()
                if s > 1e-10:
                    a /= s
                if best_obj < best_M - 1e-14:
                    best_M = best_obj
                    best_a = a.copy()
                    improved = True

        if verbose and passes % 3 == 0:
            score = 2.0 * n * best_M
            print(f"    {label} pass={passes} M={best_M:.10f} "
                  f"score={score:.6f} t={time.monotonic()-t0:.0f}s")

        if not improved:
            consecutive_no_improve += 1
            if consecutive_no_improve >= 3:
                break
        else:
            consecutive_no_improve = 0

    return best_a, best_M


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    n = len(f)
    if n == new_n:
        return f.copy()
    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def gaussian_init(n, sigma_frac=0.25):
    """Gaussian-shaped initialization."""
    x = np.arange(n, dtype=float)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a /= a.sum()
    return a


def raised_cosine_init(n):
    """Raised cosine window initialization."""
    x = np.arange(n, dtype=float)
    a = 0.5 * (1 + np.cos(2 * np.pi * (x - (n - 1) / 2.0) / (n - 1)))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    return a


def two_bump_init(n, sep_frac=0.3):
    """Two-bump initialization."""
    x = np.arange(n, dtype=float)
    c1 = n * (0.5 - sep_frac / 2.0)
    c2 = n * (0.5 + sep_frac / 2.0)
    sigma = n * 0.12
    a = np.exp(-0.5 * ((x - c1) / sigma) ** 2) + np.exp(-0.5 * ((x - c2) / sigma) ** 2)
    a /= a.sum()
    return a


def random_init(n, seed=0):
    """Random non-negative initialization."""
    rng = np.random.default_rng(seed)
    a = rng.random(n)
    a /= a.sum()
    return a

        rem = deadline - time.monotonic()
        if rem < 70:
            break
        t_alloc = min(rem * 0.12, 120.0)
        if t_alloc < 30:
            break

        print(f"\n=== Phase 3: PGD {name} (n={n_best}, budget={t_alloc:.0f}s) ===")
        a_opt, M_opt = smoothed_pgd(a_init, time_limit=t_alloc, label=f"pgd_{name}",
                                    lr_init=0.04, momentum=0.85)
        s_opt = float(evaluate_sequence(a_opt.tolist()))
        print(f"  {name}: M={M_opt:.10f} score={s_opt:.6f}")
        if np.isfinite(s_opt) and s_opt < best_score:
            best_score = s_opt
            best_seq = a_opt.copy()
            print(f"  *** NEW BEST: {best_score:.6f} ({name}) ***")

    # === Phase 4: Coordinate descent polish ===
    rem = deadline - time.monotonic()
    if rem > 50:
        t_alloc = min(rem * 0.25, 250.0)
        print(f"\n=== Phase 4: CD polish (budget={t_alloc:.0f}s) ===")
        a_cd, M_cd = coordinate_descent(best_seq, time_limit=t_alloc, label="cd1")
        s_cd = float(evaluate_sequence(a_cd.tolist()))
        print(f"  CD: M={M_cd:.10f} score={s_cd:.6f}")
        if np.isfinite(s_cd) and s_cd < best_score:
            best_score = s_cd
            best_seq = a_cd.copy()
            print(f"  *** NEW BEST: {best_score:.6f} ***")

    # === Phase 5: Deep PGD with lower temperature ===
    rem = deadline - time.monotonic()
    if rem > 50:
        t_alloc = min(rem * 0.25, 250.0)
        M_cur = fast_convolve(best_seq).max()
        print(f"\n=== Phase 5: Deep PGD (budget={t_alloc:.0f}s) ===")
        a_deep, M_deep = smoothed_pgd(best_seq, time_limit=t_alloc,
                                      tau_init=0.02 * M_cur, tau_min=0.002 * M_cur,
                                      lr_init=0.03, momentum=0.87, label="pgd_deep")
        s_deep = float(evaluate_sequence(a_deep.tolist()))
        print(f"  Deep: M={M_deep:.10f} score={s_deep:.6f}")
        if np.isfinite(s_deep) and s_deep < best_score:
            best_score = s_deep
            best_seq = a_deep.copy()
            print(f"  *** NEW BEST: {best_score:.6f} ***")

    # === Phase 6: Final coordinate descent ===
    rem = deadline - time.monotonic()
    if rem > 30:
        t_alloc = min(rem * 0.5, 100.0)
        print(f"\n=== Phase 6: Final CD (budget={t_alloc:.0f}s) ===")
        a_cd2, M_cd2 = coordinate_descent(best_seq, time_limit=t_alloc, label="cd2")
        s_cd2 = float(evaluate_sequence(a_cd2.tolist()))
        if np.isfinite(s_cd2) and s_cd2 < best_score:
            best_score = s_cd2
            best_seq = a_cd2.copy()
            print(f"  *** NEW BEST: {best_score:.6f} ***")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if not np.isfinite(fs):
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
    else:
        best_score = fs

    print(f"\n{'='*55}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*55}")

    return [float(x) for x in best_seq]
```
