**Key insight for local optimization**: The objective `f(a) = 2n * max_k conv(a,a)[k] / (sum a)^2` is piecewise differentiable. The active set (which k achieves the max) changes only when the sequence crosses a "boundary" in the parameter space. Within a region where the active set is fixed, the objective is a smooth rational function, and we can use Newton's method or damped gradient descent.

**Active set method:**
1. Identify the active set: S = {k : conv(a,a)[k] = max}
2. The objective (on this active set) is: f = 2n * max_{k in S} conv(a,a)[k] / (sum a)^2
3. If the active set is a single point k*, then f = 2n * conv(a,a)[k*] / (sum a)^2, which is a smooth function
4. We can use gradient descent on this smooth function
5. Periodically check if the active set has changed (if another k has caught up)
6. If the active set changes, recompute the active set and continue

This is essentially what the previous code does (it tracks the max and its position), but let me make it more explicit and use second-order information.

**Damped Newton method on the active constraint:**

If the active set is {k*}, then we minimize g(a) = conv(a,a)[k*] / (sum a)^2.

The gradient of g is:
∂g/∂a_j = [2*a_{k*-j} * (sum a) - 2 * conv(a,a)[k*] * 1] / (sum a)^2

(simplified for the case where the constraint is just conv(a,a)[k*])

The Hessian is more complex but can be computed.

However, in practice, the active set often has multiple elements, and switching between them makes Newton's method tricky. So I'll stick with subgradient methods but make them more efficient.

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def get_lp_solver():
    import cvxpy as cp
    global _lp_solver
    if '_lp_solver' in globals() and _lp_solver is not None:
        return _lp_solver
    _lp_solver = None
    for solver_name in [cp.GUROBI, cp.MOSEK, cp.GLPK, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:

    return 'ECOS'

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def block_coord_descent(a0, time_limit, block_size=3, n_trials=8, verbose=False, label=""):
    """Block coordinate descent: update blocks of coordinates jointly."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    M = float(conv.max())
    sum_a = 1.0
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    sweeps = 0
    no_improve = 0
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False
        
        # Randomize block order
        block_starts = list(range(0, n, block_size))
        rng.shuffle(block_starts)
        
        for start in block_starts:
            end = min(start + block_size, n)
            block_idx = np.arange(start, end)
            
            # Collect current values
            a_block = a[block_idx].copy()
            block_sum = a_block.sum()
            
            if block_sum < 1e-15:
                continue
            
            # Compute affected range
            k_start = start
            k_end = min(end - 1 + n - 1, 2 * n - 2)
            
            # Unaffected max
            if k_start > 0:
                unaff_left = conv[:k_start].max() if k_start > 0 else -1e30
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max() if k_end < 2 * n - 2 else -1e30
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)
            
            # Try perturbations on the block
            best_block = a_block.copy()
            best_new_max = M
            
            for trial in range(n_trials):
                if trial == 0:
                    # Small random perturbation
                    candidate_block = a_block * (1 + rng.standard_normal(len(block_idx)) * 0.05)
                elif trial == 1:
                    # Scale entire block
                    candidate_block = a_block * 0.9
                elif trial == 2:
                    candidate_block = a_block * 1.1
                elif trial == 3:
                    # Move mass within block
                    candidate_block = a_block.copy()
                    donor = rng.integers(0, len(block_idx))
                    recipient = rng.integers(0, len(block_idx))
                    if donor != recipient:
                        amt = candidate_block[donor] * 0.2
                        candidate_block[donor] -= amt
                        candidate_block[recipient] += amt
                elif trial == 4:
                    # Flatten block
                    candidate_block = np.full(len(block_idx), block_sum / len(block_idx))
                else:
                    # Larger random perturbation
                    candidate_block = a_block * (1 + rng.standard_normal(len(block_idx)) * 0.15)
                
                candidate_block = np.maximum(candidate_block, 0)
                new_sum = sum_a - block_sum + candidate_block.sum()
                if new_sum < 0.01:
                    continue
                
                # Compute new convolution in affected region
                # conv_new[k] = conv_old[k] - contributions from old block + contributions from new block
                # This is complex; use FFT for small blocks
                if k_end - k_start + 1 > n:
                    # Just recompute full convolution
                    a_test = a.copy()
                    a_test[block_idx] = candidate_block
                    conv_test = fast_convolve(a_test)
                    new_max = float(conv_test.max())
                else:
                    # Incremental update
                    new_affected = conv[k_start:k_end+1].copy()
                    for di, old_val in enumerate(a_block):
                        if old_val < 1e-15:
                            continue
                        j = start + di
                        # Remove old contribution
                        for k in range(k_start, k_end + 1):
                            k_mod = k - j
                            if 0 <= k_mod < n:
                                new_affected[k - k_start] -= 2.0 * old_val * a[k_mod] if k_mod != j else old_val * old_val
                        # This is getting too complicated; just use FFT
                    new_max = None
                
                if new_max is None:
                    a_test = a.copy()
                    a_test[block_idx] = candidate_block
                    new_max = float(fast_convolve(a_test).max())
                
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_block = candidate_block.copy()
            
            if best_new_max < M - 1e-16:
                a[block_idx] = best_block
                sum_a = a.sum()
                conv = fast_convolve(a)
                M = float(conv.max())
                cur_score = 2.0 * n * M / (sum_a * sum_a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve > 20:
                break
        else:
            no_improve = 0
        
        if verbose and sweeps % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    BCD {label} sweep={sweeps} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score

def simulated_annealing(a0, time_limit, T_init=0.005, T_final=0.0001, verbose=False, label=""):
    """Simulated annealing with diverse moves."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    cur_score = best_score
    t0 = time.monotonic()
    total_time = time_limit
    rng = np.random.default_rng(123)
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Temperature schedule
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_init * (T_final / T_init) ** progress
        T = max(T, T_final)
        
        # Choose a move type
        move_type = rng.integers(0, 4)
        a_new = a.copy()
        
        if move_type == 0:
            # Coordinate perturbation
            n_perturb = rng.integers(1, 4)
            idx = rng.integers(0, n, size=n_perturb)
            for i in idx:
                a_new[i] = max(0, a_new[i] + rng.standard_normal() * 0.05 * a.mean())
        elif move_type == 1:
            # Block swap
            block_len = rng.integers(5, 20)
            i1 = rng.integers(0, n - block_len)
            i2 = rng.integers(0, n - block_len)
            if i1 != i2:
                a_new[i1:i1+block_len], a_new[i2:i2+block_len] = \
                    a_new[i2:i2+block_len].copy(), a_new[i1:i1+block_len].copy()
        elif move_type == 2:
            # Spectral rotation (modify one Fourier mode)
            N = 1 << (2*n-1).bit_length()
            F = rfft(a, N)
            k = rng.integers(1, N//2)
            phase = 2 * np.pi * rng.random()
            F[k] *= np.exp(1j * phase)
            a_new = irfft(F, N)[:n]
        else:
            # Multiplicative noise
            a_new = a * np.exp(rng.standard_normal(n) * 0.1)
        
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 0.01:
            continue
        a_new = a_new / s_new
        
        new_score = 2.0 * n * fast_convolve(a_new).max()
        delta = new_score - cur_score
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_new
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
        
        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SA {label} it={it} ({elapsed:.0f}s): T={T:.6f} best={best_score:.8f} cur={cur_score:.8f}")
    
    return best_a, best_score
