    n = len(f)

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_constraint_matrix(f, indices, n, fft_size):
    """Build constraint matrix A where row i encodes (f*g)[k_i] = A[i,:] @ g."""
    K = len(indices)
    A = np.zeros((K, n))
    for i, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    return A


def solve_lp_direction(f, n, M, active_idx, fft_size, solver_priority=None):

    n = len(a_init)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    a = a_init.copy()
    best_a = a.copy()
    R = fast_convolve(a)
    best_obj = R.max()
    
    step = step_init
    prev_subgrad = np.zeros(fft_size // 2 + 1)
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if (it - 1) % 200 == 0:
            rem = time_limit - (time.monotonic() - t0)
            if rem < 1.0:
                break
        
        F = rfft(a, fft_size)
        P = F.real ** 2 + F.imag ** 2
        
        # Softmax weights on autocorrelation (higher weight on peaks)
        R_max = R.max()
        if R_max < 1e-15:
            break
        R_scaled = R / (temp * R_max + 1e-15)
        w = np.exp(-R_scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(size)
            w[0] = 1.0
        else:
            w /= w_sum
        
        # Subgradient in frequency domain
        w_padded = np.zeros(fft_size)
        w_padded[:size] = w
        subgrad_full = np.fft.ifft(w_padded)
        subgrad = np.real(subgrad_full[:fft_size // 2 + 1])
        
        # Momentum
        subgrad_mom = subgrad + momentum_coeff * prev_subgrad
        prev_subgrad = subgrad
        
        # Adaptive step: scale by P magnitude
        P_med = np.median(P[P > 1e-10]) if np.any(P > 1e-10) else 1.0
        effective_step = step * P_med / (np.max(np.abs(subgrad_mom)) + 1e-15)
        effective_step = min(effective_step, 2.0)  # Cap relative change
        
        # Update power spectrum
        P_new = P - effective_step * subgrad_mom
        P_new = np.maximum(P_new, 0)
        
        # Preserve DC component approximately
        if P_new[0] < 0.5 * P[0]:
            P_new[0] = 0.5 * P[0]
        
        # Reconstruct
        new_amp = np.sqrt(P_new)
        abs_F = np.abs(F)
        phase = F / (abs_F + 1e-30)
        F_new = new_amp * phase
        a_new = irfft(F_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            step *= 0.5
            continue
        a_new /= s
        
        # Evaluate
        R_new = fast_convolve(a_new)
        obj_new = R_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                step *= 0.7
                temp *= 1.05
                no_improve = 0
                if step < 0.01:
                    # Restart with perturbation
                    a = best_a + np.random.default_rng(it).standard_normal(n) * 0.01
                    a = np.maximum(a, 0)
                    a /= a.sum()
                    step = step_init * 0.5
        
        a = a_new
        step *= decay
        
        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} it={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} "
                  f"step={step:.4f} temp={temp:.4f} t={elapsed:.0f}s")
    
    return best_a, best_obj


def lp_optimize(f_init, time_limit, initial_k=50, max_k=400, label="lp",
                solver_priority=None):
    """
    Cutting-plane LP optimization with adaptive active set management.
    """
    t0 = time.monotonic()
    n = len(f_init)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    f = f_init.copy()
    best_f = f.copy()
    R = fast_convolve(f)
    M = R.max()
    best_obj = M
    
    # Initialize active set with top peaks
    if initial_k < size:
        active_idx = np.argpartition(R, -initial_k)[-initial_k:]
    else:
        active_idx = np.arange(size)
    active_idx = np.sort(active_idx)
    
    it = 0
    stuck = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break
        
        # Solve LP
        g = solve_lp_direction(f, n, M, active_idx, fft_size, solver_priority)
        
        if g is None:
            stuck += 1
            if stuck > 10:
                break
            # Reduce active set
            if len(active_idx) > initial_k:
                active_idx = active_idx[np.argsort(R[active_idx])[-initial_k:]]
            continue
        
        S_g = g.sum()
        if S_g <= 1.0 + 1e-5:
            stuck += 1
            if stuck > 25:
                break
            # Try expanding
            if len(active_idx) < max_k:
                R_cur = fast_convolve(f)
                new_top = np.argpartition(R_cur, -min(max_k, size))[-min(max_k, size):]
                active_idx = np.unique(np.concatenate([active_idx, new_top]))
            continue
        
        # Cutting plane: find violated constraints
        fg = convolve_cross(f, g, fft_size)
        violated = fg > M * (1 + 1e-6)
        v_idx = np.where(violated)[0]
        not_active = v_idx[~np.isin(v_idx, active_idx)]
        
        if len(not_active) > 0 and len(active_idx) < max_k:
            n_add = min(20, len(not_active))
            to_add = not_active[np.argsort(fg[not_active])[-n_add:]]
            active_idx = np.unique(np.concatenate([active_idx, to_add]))
            # Re-solve
            g = solve_lp_direction(f, n, M, active_idx, fft_size, solver_priority)
            if g is None:
                stuck += 1
                if stuck > 5:
                    break
                continue
            S_g = g.sum()
            if S_g <= 1.0 + 1e-5:
                stuck += 1
                if stuck > 20:
                    break
                continue
        
        stuck = 0
        h = g / S_g
        
        # Line search
        best_t = 0.05
        best_t_obj = M
        scan = [0.02, 0.05, 0.1, 0.15, 0.25, 0.4, 0.6]
        for t in scan:
            f_trial = np.maximum((1.0 - t) * f + t * h, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t
        
        # Golden section refinement
        if best_t > 0.03:
            lo = max(0.0, best_t - 0.08)
            hi = min(1.0, best_t + 0.15)
            for _ in range(7):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                fa1 = np.maximum((1 - m1) * f + m1 * h, 0.0)
                sa1 = fa1.sum()
                if sa1 > 1e-10:
                    fa1 /= sa1
                fa2 = np.maximum((1 - m2) * f + m2 * h, 0.0)
                sa2 = fa2.sum()
                if sa2 > 1e-10:
                    fa2 /= sa2
                o1 = fast_convolve(fa1).max()
                o2 = fast_convolve(fa2).max()
                if o1 < o2:
                    hi = m2
                else:
                    lo = m1
            t_mid = (lo + hi) / 2.0
            f_mid = np.maximum((1 - t_mid) * f + t_mid * h, 0.0)
            s_mid = f_mid.sum()
            if s_mid > 1e-10:
                f_mid /= s_mid
                o_mid = fast_convolve(f_mid).max()
                if o_mid < best_t_obj:
                    best_t_obj = o_mid
                    best_t = t_mid
        
        # Update
        f = np.maximum((1.0 - best_t) * f + best_t * h, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s
        else:
            f = f_init.copy()
            break
        
        cur_obj = fast_convolve(f).max()
        M = cur_obj
        
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_f = f.copy()
        
        # Periodic active set refresh
        if it % 10 == 0:
            R = fast_convolve(f)
            if len(active_idx) > initial_k * 3:
                scores = R[active_idx]
                keep = active_idx[np.argsort(scores)[-initial_k * 2:]]
                new_top = np.argpartition(R, -initial_k)[-initial_k:]
                active_idx = np.unique(np.concatenate([keep, new_top]))
                active_idx = active_idx[np.argsort(R[active_idx])[-max_k:]]
        
        if it % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} it={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} "
                  f"active={len(active_idx)} t={elapsed:.0f}s")
    
    return best_f, best_obj


def subgradient_polish(a_init, time_limit, lr_init=0.003, n_top=50, momentum=0.8):
    """Subgradient method with momentum for final polish."""
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    v = np.zeros(n)
    lr = lr_init
    no_improve = 0
    it = 0
    rng = np.random.default_rng(123)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        R = fast_convolve(a)
        R_max = R.max()
        
        # Softmax weights
        w = np.exp(-R / (0.01 * R_max + 1e-10))
        w /= w.sum()
        
        if n_top < len(R):
            top_idx = np.argpartition(R, -n_top)[-n_top:]
            w_sel = w[top_idx].copy()
            ws = w_sel.sum()
            if ws > 1e-30:
                w_sel /= ws
            else:
                w_sel = np.ones(len(top_idx)) / len(top_idx)
            w = np.zeros(len(R))
            w[top_idx] = w_sel
        
        # Subgradient
        g = np.zeros(n)
        for k in range(len(R)):
            if w[k] < 1e-12:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += w[k] * 2.0 * a[k - idx]
        
        v = momentum * v - lr * g
        a_new = project_simplex(a + v)
        
        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                lr *= 0.85
                no_improve = 0
                if lr < 1e-5:
                    a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
                    v = np.zeros(n)
                    lr = lr_init * 0.5
        
        a = a_new
        
        if it % 200 == 0:
            print(f"    sg it={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} "
                  f"lr={lr:.6f} t={time.monotonic()-t0:.0f}s")
    
    return best_a, best_obj


def make_cosine_taper(n, power=1.0):
    """Create a cosine-tapered sequence."""
    x = np.linspace(0, np.pi, n)
    a = np.cos(x / 2) ** (2 * power)
    a = np.maximum(a, 0)
    a /= a.sum()
    return a


def make_edge_concentrated(n, alpha=2.0):
    """Create a sequence concentrated at edges."""
    i = np.arange(n)
    mid = (n - 1) / 2.0
    dist_from_edge = np.minimum(i, n - 1 - i).astype(float)
    a = (dist_from_edge / (n / 2.0)) ** alpha
    a = np.maximum(a, 1e-10)
    a /= a.sum()
    return a


def make_sinusoid_perturbed(base, freq=3.0, eps=0.1):
    """Perturb a base sequence with a sinusoid."""
    n = len(base)
    t = np.linspace(0, 2 * np.pi * freq, n)
    a = base * (1.0 + eps * np.sin(t))
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = base.copy()
    return a

            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")

        parent = np.ones(1024) / 1024
