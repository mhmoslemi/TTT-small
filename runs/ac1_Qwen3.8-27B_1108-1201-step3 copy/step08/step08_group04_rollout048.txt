def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():

            b_now = fast_convolve(f)
            active_list = sorted(int(x) for x in np.argpartition(b_now, -max_active)[-max_active:])
            active = set(active_list)

        g = solve_lp_dir(f, active_list, M)

        if g is None:
            lp_fail_count += 1
            if lp_fail_count > 5:
                if len(active_list) > 10:
                    top10 = sorted(int(x) for x in np.argpartition(fast_convolve(f), -10)[-10:])
                    active = set(top10)
                    lp_fail_count = 0
                    continue
                break
            continue

        lp_fail_count = 0
        S_g = g.sum()

        if S_g <= 1.0 + 1e-7:
            # No improvement with current active set
            fg = cross_convolve(f, g)
            violated = np.where(fg > M + 1e-10)[0]
            if len(violated) > 0:
                worst = int(violated[np.argmax(fg[violated])])
                active.add(worst)
                stuck += 1
            else:
                stuck += 2
            if stuck > max_stuck:
                break
            continue

        h = g / S_g
        t_star, new_peak = line_search_convex(f, h, n_coarse=n_coarse, n_fine=n_fine)

        if new_peak < M - 1e-14:
            f_new = (1.0 - t_star) * f + t_star * h
            f_new = np.maximum(f_new, 0.0)
            s_new = f_new.sum()
            if s_new > 1e-15:
                f_new = f_new / s_new
            else:
                f_new = f.copy()
                new_peak = M
            f = f_new
            stuck = 0
            if new_peak < best_peak - 1e-15:
                best_peak = new_peak
                best_score = 2.0 * n * best_peak
                best_f = f.copy()
        else:
            stuck += 1
            if stuck > max_stuck:
                break

        if it % 8 == 0:
            b_now = fast_convolve(f)
            M_cur = b_now.max()
            thresh = prune_frac * M_cur
            new_active = set(int(x) for x in np.where(b_now > thresh)[0])
            if len(new_active) < init_k:
                top_idx = np.argpartition(b_now, -init_k)[-init_k:]
                new_active.update(int(x) for x in top_idx)
            if len(new_active) > max_active:
                items = sorted(new_active)
                vals = [b_now[i] for i in items]
                keep_idx = np.argpartition(vals, -max_active)[-max_active:]
                new_active = set(items[i] for i in keep_idx)
            active = new_active

        if it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} it={it} score={best_score:.7f} |A|={len(active)} t={elapsed:.0f}s")

    return best_f, best_score


def coordinate_descent(f_init, time_limit, n_multipliers=7):
    """Coordinate descent polished with multipliers."""
    t0 = time.monotonic()
    n = len(f_init)
    f = np.asarray(f_init, dtype=np.float64).copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s
    best_peak = fast_convolve(f).max()
    best_score = 2.0 * n * best_peak
    best_f = f.copy()
    multipliers = np.array([0.5, 0.7, 0.8, 0.9, 1.0, 1.1, 1.2, 1.4, 1.7, 2.0])
    no_pass = 0

    while time.monotonic() - t0 < time_limit:
        any_imp = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit - 1:
                break
            if f[j] < 1e-16:
                continue
            best_j_val = f[j]
            best_j_peak = best_peak
            for mult in multipliers:
                nv = f[j] * mult
                trial = f.copy()
                trial[j] = nv
                st = trial.sum()
                if st < 1e-12:
                    continue
                trial = trial / st
                pk = fast_convolve(trial).max()
                if pk < best_j_peak - 1e-14:
                    best_j_peak = pk
                    best_j_val = nv
            if best_j_val != f[j]:
                f[j] = best_j_val
                st = f.sum()
                if st > 1e-12:
                    f = f / st
                if best_j_peak < best_peak - 1e-14:
                    best_peak = best_j_peak
                    best_score = 2.0 * n * best_peak
                    best_f = f.copy()
                    any_imp = True
        if not any_imp:
            no_pass += 1
            if no_pass >= 3:
                break
        else:
            no_pass = 0

    return best_f, best_score


def interp_seq(src, target_n):
    """Interpolate a normalized sequence to a new length."""
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result


def trimmed_active(b, max_active, min_active=15):
    """Select active set from convolution values."""
    n_b = len(b)
    if n_b <= max_active:
        return set(range(n_b))
    threshold_idx = np.argpartition(b, -max_active)[-max_active:]
    active = set(int(x) for x in threshold_idx)
    if len(active) < min_active:
        top = np.argpartition(b, -min_active)[-min_active:]
        active.update(int(x) for x in top)
    return active


def subgradient_polish(f_init, time_limit, lr_init=0.002, n_top=60, momentum=0.8):
    """Subgradient method with softmax peak weighting."""
    t0 = time.monotonic()
    n = len(f_init)
    f = np.asarray(f_init, dtype=np.float64).copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s
    v = np.zeros(n)
    lr = lr_init
    temp = 0.0003
    best_f = f.copy()
    best_peak = fast_convolve(f).max()
    no_imp = 0
    it = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()

        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
        else:
            top_idx = np.arange(len(b))

        scaled = (b[top_idx] - M) / temp
        exp_vals = np.exp(scaled - scaled.max())
        w = exp_vals / exp_vals.sum()

        g_sub = np.zeros(n)
        for i, k in enumerate(top_idx):
            if w[i] < 1e-14:
                continue
            lo = max(0, int(k) - n + 1)
            hi = min(n - 1, int(k))
            idx = np.arange(lo, hi + 1)
            g_sub[idx] += w[i] * f[int(k) - idx]

        v = momentum * v - lr * g_sub
        f = project_simplex(f + v)

        if it % 40 == 0:
            obj = fast_convolve(f).max()
            if obj < best_peak - 1e-14:
                best_peak = obj
                best_f = f.copy()
                no_imp = 0
                lr = min(lr * 1.03, 0.015)
            else:
                no_imp += 1
                if no_imp > 5:
                    lr *= 0.9
                    no_imp = 0
            if it % 200 == 0:
                temp = min(temp * 1.15, 0.4)
            if no_imp > 30:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                v = np.zeros(n)
                no_imp = 0

    return best_f, 2.0 * n * best_peak


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    Phase 1: Quick multi-n exploration with cut-LP
    Phase 2: Deep cut-LP on best candidate
    Phase 3: Subgradient + coordinate descent polishing
    Phase 4: Final LP burst
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
    except Exception:
        pass

    if parent is not None:
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Parent: n={len(parent)}, score={p_score:.7f}")
    else:
        parent = np.ones(512) / 512.0
        p_score = 2.0
        print(f"No parent found, using uniform n=512, score=2.0")

    best_seq = parent.copy()
    best_score = p_score

    # ============================================================
    # Phase 1: Multi-n quick exploration (25% of budget)
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 1: Multi-n exploration (target: {best_score:.7f})")
    print(f"{'='*60}")

    rem = deadline - time.monotonic()
    t1 = min(rem * 0.25, 400.0)

    if t1 > 60:
        n_parent = len(parent)
        n_candidates = []
        for factor in [0.5, 0.65, 0.8, 0.9, 1.0, 1.1, 1.25]:
            nc = int(n_parent * factor)
            nc = max(192, min(2000, nc))
            if nc not in n_candidates:
                n_candidates.append(nc)
        n_candidates.sort()
        per_n = t1 / len(n_candidates)

        for nc in n_candidates:
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            alloc = min(per_n, rem * 0.9)
            if alloc < 15:
                break
            f_init = interp_seq(best_seq, nc)
            print(f"  CutLP n={nc}, budget={alloc:.0f}s")
            f_res, sc = cutlp_optimize(f_init, time_limit=alloc, label=f"p1_n{nc}",
                                       max_active=300, init_k=20, max_stuck=60)
            real_sc = float(evaluate_sequence(f_res.tolist()))
            if np.isfinite(real_sc) and real_sc < best_score:
                best_score = real_sc
                best_seq = f_res.copy()
                print(f"    NEW BEST [n={nc}]: {real_sc:.7f}")

    # ============================================================
    # Phase 2: Deep cut-LP on best (40% of remaining)
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 2: Deep CutLP (best: {best_score:.7f}, n={len(best_seq)})")
    print(f"{'='*60}")

    rem = deadline - time.monotonic()
    t2 = min(rem * 0.40, 600.0)

    if t2 > 30:
        f_res, sc = cutlp_optimize(best_seq, time_limit=t2, label="deep",
                                   max_active=500, init_k=40, max_stuck=120,
                                   n_coarse=18, n_fine=10, prune_frac=0.90)
        real_sc = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_sc) and real_sc < best_score:
            best_score = real_sc
            best_seq = f_res.copy()
            print(f"  NEW BEST [deep]: {real_sc:.7f}")

    # ============================================================
    # Phase 3: Subgradient + CD polishing (15%)
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 3: SG + CD polish (best: {best_score:.7f})")
    print(f"{'='*60}")

    rem = deadline - time.monotonic()
    t3 = min(rem * 0.15, 200.0)

    if t3 > 20:
        n_sg = t3 * 0.5
        n_cd = t3 * 0.5
        f_res, sc = subgradient_polish(best_seq, time_limit=n_sg, lr_init=0.002,
                                       n_top=max(40, len(best_seq) // 15))
        real_sc = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_sc) and real_sc < best_score:
            best_score = real_sc
            best_seq = f_res.copy()
            print(f"  NEW BEST [sg]: {real_sc:.7f}")

        rem = deadline - time.monotonic()
        if rem > 10:
            f_res, sc = coordinate_descent(best_seq, time_limit=min(n_cd, rem))
            real_sc = float(evaluate_sequence(f_res.tolist()))
            if np.isfinite(real_sc) and real_sc < best_score:
                best_score = real_sc
                best_seq = f_res.copy()
                print(f"  NEW BEST [cd]: {real_sc:.7f}")

    # ============================================================
    # Phase 4: Final LP burst with high constraint count (20%)
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Final LP burst (best: {best_score:.7f})")
    print(f"{'='*60}")

    rem = deadline - time.monotonic()
    t4 = min(rem * 0.20, 250.0)

    if t4 > 20:
        f_res, sc = cutlp_optimize(best_seq, time_limit=t4, label="final",
                                   max_active=600, init_k=50, max_stuck=150,
                                   n_coarse=20, n_fine=12, prune_frac=0.88)
        real_sc = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_sc) and real_sc < best_score:
            best_score = real_sc
            best_seq = f_res.copy()
            print(f"  NEW BEST [final]: {real_sc:.7f}")

    # ============================================================
    # Phase 5: Tiny coordinate descent + verify
    # ============================================================
    rem = deadline - time.monotonic()
    if rem > 10:
        f_res, sc = coordinate_descent(best_seq, time_limit=min(rem * 0.5, 60))
        real_sc = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_sc) and real_sc < best_score:
            best_score = real_sc
            best_seq = f_res.copy()
            print(f"  NEW BEST [final cd]: {real_sc:.7f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.7f} (n={len(best_seq)})")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
