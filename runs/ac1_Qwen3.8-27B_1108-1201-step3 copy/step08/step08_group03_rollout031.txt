    Optimize using smoothed max (log-sum-exp) with Adam.
    """

    f = f_init.copy()
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8
    lr = 0.01
    t = 0
    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    T = T_init

        if obj < best_obj:
            best_obj = obj
            best_f = f.copy()

"""
Optimized search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses smoothed max (log-sum-exp) with Adam optimizer, FFT-based gradient,
periodic LP escapes, and multi-start with diverse initializations.

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad(f, b, T):
    """

    Uses FFT for O(n log n) computation.

    size_b = 2 * n - 1

"""
Optimized search for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Novel approach: Smoothed max (log-sum-exp) with Adam optimizer and FFT-based
gradient computation, combined with periodic LP-based escape, multi-start
diversification with structured initializations, and coordinate descent polish.
Key innovation: O(n log n) gradient via FFT cross-correlation, enabling
thousands of optimization steps within the time budget.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_grad(f, b, T):
    """
    Compute gradient of T*log(sum(exp(b/T))) w.r.t. f using FFT.
    Returns (gradient, smoothed_objective_value).
    """
    n = len(f)
    size_b = 2 * n - 1

    # Softmax weights (numerically stable)
    scaled = b / T
    scaled -= scaled.max()
    w_unnorm = np.exp(scaled)
    w_sum = w_unnorm.sum()
    w = w_unnorm / w_sum

    # Smoothed objective: T * log(sum(exp(b/T))) = max(b) + T * log(w_sum)
    obj_smooth = b.max() + T * np.log(w_sum)

    # FFT size for convolving w (len 2n-1) with f (len n): need 3n-2
    fft_size = 1 << (3 * n - 2).bit_length()

    W = rfft(w, fft_size)
    F = rfft(f, fft_size)

    # term2[i] = sum_j w[i-j] * f[j] = (w * f)[i]
    conv_wf = irfft(W * F, fft_size)
    term2 = conv_wf[:n]

    # term1[i] = sum_j w[i+j] * f[j] = (w * f_rev)[n-1+i]
    f_rev = f[::-1]
    F_rev = rfft(f_rev, fft_size)
    conv_wfrev = irfft(W * F_rev, fft_size)
    term1 = conv_wfrev[n - 1:2 * n - 1]

    grad = term1 + term2
    return grad, obj_smooth


def adam_smoothed_max(f_init, time_limit, T_init=0.05, T_end=0.002,
                      lr=0.005, beta1=0.85, beta2=0.999, eps=1e-8,
                      label="adam", verbose_every=200):
    """
    Adam optimizer on the log-sum-exp smoothed max of conv(f,f).
    Projects onto simplex at each step.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        step += 1
        # Adaptive temperature: exponential decay
        progress = min(1.0, step / (time_limit * 80))
        T = T_init * (T_end / T_init) ** progress

        b = fast_convolve(f)
        grad, _ = compute_smoothed_grad(f, b, T)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Project to simplex
        f = project_simplex(f_new)

        # Evaluate every few steps (expensive)
        if step % 10 == 0 or step < 5:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1

            # Adaptive learning rate
            if no_improve > 30:
                lr *= 0.9
                no_improve = 0

            if verbose_every > 0 and step % verbose_every == 0:
                elapsed = time.monotonic() - t0
                print(f"    {label} step={step} obj={best_obj:.10f} "
                      f"score={2.0*n*best_obj:.6f} T={T:.5f} "
                      f"lr={lr:.5f} t={elapsed:.0f}s")

    return best_f, best_obj


def lp_escape(f_init, n_constraints=60, label="lp"):
    """
    Solve LP to find a direction that reduces the peak.
    max sum(g) s.t. g>=0, A@g <= M
    Returns a normalized direction.
    """
    n = len(f_init)
    b = fast_convolve(f_init)
    M = b.max()

    # Get top constraint indices
    k = min(n_constraints, 2 * n - 1)
    active_idx = np.argpartition(b, -k)[-k:]
    active_idx = np.sort(active_idx)

    # Build constraint matrix
    A = np.zeros((k, n))
    for i, kk in enumerate(active_idx):
        lo = max(0, kk - n + 1)
        hi = min(n - 1, kk)
        A[i, lo:hi + 1] = f_init[lo:hi + 1]

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=200000, eps=1e-8)
    except Exception:
        return None

    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None
    s = g_val.sum()
    if s < 1.0 + 1e-6:
        return None
    return g_val / s


def mixed_optimize(f_init, time_limit, label="mixed"):
    """
    Combined Adam + LP escape optimization.
    Runs Adam for a while, then if stuck, uses LP to find a direction,
    blends, and continues.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    lr = 0.005
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8
    T_init = 0.05
    T_end = 0.001
    step = 0

    # LP escape parameters
    lp_cooldown = 0
    stuck_count = 0
    moments = 0

    while time.monotonic() - t0 < time_limit:
        step += 1

        # Temperature schedule
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / max(1.0, time_limit))
        T = T_init * (T_end / T_init) ** progress

        # Adam step
        b = fast_convolve(f)
        grad, _ = compute_smoothed_grad(f, b, T)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f = project_simplex(f_new)

        # Evaluate and track
        if step % 8 == 0 or lp_cooldown > 0:
            obj = fast_convolve(f).max()
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_f = f.copy()
                stuck_count = 0
            else:
                stuck_count += 1

            # LP escape if stuck
            if stuck_count > 40 and lp_cooldown <= 0:
                rem = time_limit - (time.monotonic() - t0)
                if rem > 5:
                    direction = lp_escape(f, n_constraints=80)
                    if direction is not None:
                        # Blend with LP direction

                            f_new_lp /= s
                            obj_lp = fast_convolve(f_new_lp).max()
                            if obj_lp < best_obj:
                                f = f_new_lp
                                best_obj = obj_lp
                                best_f = f.copy()
                        stuck_count = 0
                lp_cooldown = 100

            if lp_cooldown > 0:
                lp_cooldown -= 1

            # Adaptive LR
            if stuck_count > 20:
                lr *= 0.85
                stuck_count = 0

            if step % 300 == 0:
                print(f"    {label} step={step} obj={best_obj:.10f} "
                      f"score={2.0*n*best_obj:.6f} T={T:.5f} "
                      f"lr={lr:.5f} stuck={stuck_count} t={elapsed:.0f}s")

    return best_f, best_obj


def coordinate_polish(f_init, time_limit, n_sweeps=None):
    """Coordinate descent with multiplicative updates on individual elements."""
    t0 = time.monotonic()

    f = f.init.copy() if False else f_init.copy()

    sweep = 0
    while time.monotonic() - t0 < time_limit:
        sweep += 1
        improved = False
        for j in range(0, n, step_size):
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-14:
                continue
            orig = f[j]
            cur_best = orig
            cur_obj = best_obj
            for mult in [0.7, 0.85, 1.15, 1.3]:
                f[j] = orig * mult
                s = f.sum()
                if s < 1e-10:
                    f[j] = orig
                    continue
                f /= s
                obj = fast_convolve(f).max()
                if obj < cur_obj - 1e-13:
                    cur_obj = obj
                    cur_best = orig * mult
            f[j] = cur_best
            s = f.sum()
            if s > 1e-10:
                f /= s
            else:
                f[j] = orig
            if cur_obj < best_obj - 1e-13:
                best_obj = cur_obj
                improved = True
        if not improved and sweep > 3:
            break
    return f, best_obj


def make_front_loaded(n, alpha=2.0):
    """Create a front-loaded initialization."""
    x = np.arange(n, 0, -1, dtype=np.float64)
    f = x ** alpha
    f /= f.sum()
    return f


def make_smooth_bell(n, center_frac=0.3, width_frac=0.3):
    """Create a smooth bell-shaped initialization centered at center_frac*n."""
    x = np.arange(n, dtype=np.float64)
    center = center_frac * n
    width = width_frac * n
    f = np.exp(-0.5 * ((x - center) / width) ** 2)
    f = np.maximum(f, 0.0)
    f /= f.sum()
    return f


def make_two_cluster(n, frac1=0.25, frac2=0.75, width=0.1):
    """Two-cluster initialization."""
    x = np.arange(n, dtype=np.float64)
    c1 = frac1 * n
    c2 = frac2 * n
    w = width * n
    f1 = np.exp(-0.5 * ((x - c1) / w) ** 2)
    f2 = np.exp(-0.5 * ((x - c2) / w) ** 2)
    f = f1 + f2
    f = np.maximum(f, 0.0)
    f /= f.sum()
    return f

        f6, _ = coordinate_polish(best_seq, time_limit=t6)
        s6 = float(evaluate_sequence(f6.tolist()))
        if np.isfinite(s6) and s6 < best_score:
            best_score = s6
            best_seq = f6.copy()
            print(f"  NEW BEST [cd]: {s6:.6f}")

    # Final validation
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'=' * 50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'=' * 50}")

    return [float(x) for x in best_seq]
