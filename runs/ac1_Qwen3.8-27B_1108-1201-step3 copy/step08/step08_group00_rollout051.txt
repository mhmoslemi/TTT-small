            a = a_new

        if (it + 1) % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FTRL iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def structured_init_linear(n):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x / x.sum()
    return a


def structured_init_quad(n):
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** 2
    a = a / a.sum()
    return a


def structured_init_exp(n, lam=1.0):
    x = np.arange(n, dtype=np.float64)
    a = lam ** x
    a = a / a.sum()
    return a


def structured_init_cos(n, p=1.0):
    x = np.linspace(0.0, 1.0, n)
    a = np.sin(np.pi * x * 0.5) ** p
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def structured_init_bimodal(n, alpha=0.3):
    x = np.linspace(0.0, 1.0, n)
    a = np.exp(-((x - alpha) ** 2) / 0.01) + np.exp(-((x - (1 - alpha)) ** 2) / 0.01)
    a = a / a.sum()
    return a


def interpolated_init(parent, new_n):
    old_n = len(parent)
    if old_n == new_n:
        return parent.copy()
    s = parent.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = parent / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def peak_shave(a0, time_limit, n_steps=10000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autoconv(a).max()
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autoconv(a)
        b_work = b.copy()
        if len(b_work) > 2:
            b_work[0] = -np.inf
            b_work[-1] = -np.inf
        peak_k = int(np.argmax(b_work))
        peak_val = b_work[peak_k]
        threshold = peak_val - 0.02 * peak_val
        near_peaks = np.where(b_work >= threshold)[0]

        contrib_count = np.zeros(n)
        for k in near_peaks[:4]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[k - idx]
            top_idx = np.argsort(contribs)[::-1][:5]
            for ci in top_idx:
                contrib_count[idx[ci]] += 1

        top_coords = np.argsort(contrib_count)[::-1][:8]
        improved = False

        for i in top_coords:
            if a[i] < 1e-14:
                continue
            for factor in [0.97, 0.94, 0.90, 0.85]:
                new_val = a[i] * factor
                diff = a[i] - new_val
                if diff < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                if others.sum() < 1e-15:
                    continue
                a_trial[mask] = others + diff * others / others.sum()
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_autoconv(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_contrib = np.argsort(contribs)[::-1][:4]
            for ci in top_contrib:
                i = idx[ci]
                j = peak_k - i
                if i == j or i >= n or j >= n:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.95, 0.90, 0.85]:
                    a_trial = a.copy()
                    a_trial[i] *= f1
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autoconv(a_trial).max()
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            a_trial = a.copy()
            for i in top_coords[:3]:
                if a[i] > 1e-14:
                    a_trial[i] *= 0.97
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            score = 2.0 * n * fast_autoconv(a_trial).max()
            if score < best_score - 1e-15:
                a = a_trial
                best_score = score
                best_a = a.copy()
            else:
                break

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 45.0)

    best_seq = None
    best_score = float('inf')

    solver = get_solver()
    print(f"Using LP solver: {solver}")

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(arr, label):
        nonlocal best_seq, best_score
        arr = np.maximum(arr, 0).astype(np.float64)
        s = arr.sum()
        if s < 0.01:
            return
        arr = arr / s
        sc = evaluate_safe(arr)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = arr.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(arr)})")

    if parent is None:
        n_init = 960
        best_seq = structured_init_linear(n_init)
        best_score = evaluate_safe(best_seq)
        print(f"Linear init: n={n_init}, score={best_score:.6f}")

    n_base = len(best_seq)
    print(f"Starting: n={n_base}, score={best_score:.6f}")

    # Phase 1: LP alternating on base length with best initial
    print(f"\n=== Phase 1: Nesterov LP n={n_base} ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.30, 500.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=80000)
        try_update(a_lp, "lp_phase1")

    # Phase 2: L-BFGS refinement
    print(f"\n=== Phase 2: L-BFGS (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.18, 300.0)
    if t_alloc > 20:
        print(f"  L-BFGS T: 200->0.2, budget={t_alloc:.0f}s")
        a_lb = lse_lbgfs(best_seq, t_alloc, T_start=200.0, T_end=0.2, maxiter=15000)
        try_update(a_lb, "lbgfs_phase2")

    # Phase 3: Second LP round
    print(f"\n=== Phase 3: Nesterov LP r2 (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 250.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=50000)
        try_update(a_lp2, "lp_phase3")

    # Phase 4: Ultra-fine L-BFGS
    print(f"\n=== Phase 4: Ultra-fine L-BFGS (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 200.0)
    if t_alloc > 15:
        print(f"  L-BFGS T: 20->0.1, budget={t_alloc:.0f}s")
        a_lb2 = lse_lbgfs(best_seq, t_alloc, T_start=20.0, T_end=0.1, maxiter=10000)
        try_update(a_lb2, "lbgfs_phase4")

    # Phase 5: Peak shaving
    print(f"\n=== Phase 5: Peak Shaving (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.05, 60.0)
    if t_alloc > 10:
        a_ps, _ = peak_shave(best_seq, t_alloc, n_steps=10000)
        try_update(a_ps, "peak_shave")

    # Phase 6: Try alternative lengths
    print(f"\n=== Phase 6: Alternative Lengths (score={best_score:.6f}) ===")
    alt_lengths = []
    for mult in [1.25, 1.5, 0.8, 2.0]:
        alt_n = int(n_base * mult)
        if 256 <= alt_n <= 3000 and alt_n != n_base:
            alt_lengths.append(alt_n)
    alt_lengths = sorted(set(alt_lengths))

    for alt_n in alt_lengths[:2]:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_total = min(rem * 0.35, 250.0)
        if t_total < 30:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolated_init(best_seq, alt_n)
        init_lin = structured_init_linear(alt_n)

        # Try both initializations
        for init_a, init_name in [(init_alt, "interp"), (init_lin, "linear")]:
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            t_lp = min(rem * 0.6, 120.0)
            if t_lp < 15:
                continue
            a_alt, _ = nesterov_lp_refine(init_a, t_lp, solver=solver, n_max_iters=30000)
            try_update(a_alt, f"lp_n{alt_n}_{init_name}")

            rem = deadline - time.monotonic()
            if rem > 15 and best_seq is not None and len(best_seq) == alt_n:
                t_lb = min(rem * 0.3, 50.0)
                if t_lb > 10:
                    a_alt2 = lse_lbgfs(best_seq, t_lb, T_start=30.0, T_end=0.5, maxiter=5000)
                    try_update(a_alt2, f"lbgfs_n{alt_n}")

    # Phase 7: Perturbation escape
    print(f"\n=== Phase 7: Perturbation (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        n_cur = len(best_seq)
        for noise_scale in [0.03, 0.08]:
            perturbed = best_seq + noise_scale * rng.standard_normal(n_cur)
            perturbed = np.maximum(perturbed, 0)
            perturbed = perturbed / perturbed.sum()
            t_pert = min(deadline - time.monotonic(), 80.0)
            if t_pert < 15:
                break
            a_pert, _ = nesterov_lp_refine(perturbed, t_pert * 0.7, solver=solver, n_max_iters=20000)
            try_update(a_pert, f"pert_{noise_scale}")
            rem = deadline - time.monotonic()
            if rem > 10:
                a_pert2 = lse_lbgfs(best_seq, min(rem * 0.3, 30.0),
                                     T_start=5.0, T_end=0.3, maxiter=3000)
                try_update(a_pert2, f"pert_lbgfs_{noise_scale}")

    # Phase 8: Final polish
    print(f"\n=== Phase 8: Final Polish (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.4, 60.0)
        a_fin = lse_lbgfs(best_seq, t_fin, T_start=5.0, T_end=0.05, maxiter=5000)
        try_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 10:
        a_ps2, _ = peak_shave(best_seq, min(rem * 0.3, 20.0), n_steps=5000)
        try_update(a_ps2, "final_ps")

    rem = deadline - time.monotonic()
    if rem > 8:
        a_lp_fin, _ = nesterov_lp_refine(best_seq, min(rem * 0.4, 15.0),
                                         solver=solver, n_max_iters=3000)
        try_update(a_lp_fin, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq) if best_seq is not None else 0})")
    if best_seq is None:
        return [1.0]
    return [float(x) for x in np.maximum(best_seq, 0)]
