def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(f):
    """Compute the objective score for a normalized sequence (sum=1)."""
    n = len(f)
    b = fast_convolve(f)
    return 2.0 * n * b.max()


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_ecos(f, n, peak_indices, M, eps=1e-9):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)[k] <= M for k in peaks."""
    K = len(peak_indices)
    if K == 0:
        return None
    # Build constraint matrix efficiently
    A = np.zeros((K, n))
    for idx in range(K):
        k = int(peak_indices[idx])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[idx, lo:hi + 1] = f[lo:hi + 1]

    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    for solver in [cp.ECOS, cp.SCS, cp.CBC]:
        try:
            prob.solve(solver=solver, eps=eps, max_iters=200000)
            break
        except Exception:
            continue

    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None
    return g_val

            continue

        # Update
        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s < 1e-10:
            break
        f = f / s

        cur_obj = fast_convolve(f).max()
        if cur_obj < best_b - 1e-14:
            best_b = cur_obj
            best_f = f.copy()
            stuck = 0
            # Adapt: increase step slightly if improving well
            step_size = min(step_size * 1.02, 0.5)
        else:
            stuck += 1
            step_size = max(step_size * 0.95, 0.02)

        # Gradually lower fraction
        if it % 20 == 0:
            frac = max(final_frac, frac - 0.008)

        # Adjust peak count
        if it % 50 == 0 and len(peaks) < max_peaks:
            pass  # max_peaks is fixed

        if it % 20 == 0:
            score = 2.0 * n * best_b
            print(f"    {label} it={it} obj={best_b:.12f} score={score:.6f} "
                  f"peaks={len(peaks)} frac={frac:.3f} t={time.monotonic()-t0:.0f}s")

    return best_f, best_b


def coordinate_descent(f_init, time_limit, label="cd"):
    """Fast coordinate descent with 1D optimization per coordinate."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_b = fast_convolve(f).max()
    rng = np.random.default_rng(777)
    no_improve = 0
    multipliers = [0.5, 0.65, 0.78, 0.87, 0.93, 1.0, 1.07, 1.15, 1.25, 1.4, 1.65]
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        improved = False
        # Randomize order
        order = np.random.default_rng(it).permutation(n)

        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-15:
                continue

            cur_val = f[j]
            best_val = cur_val
            best_obj = best_b

            for mult in multipliers:
                new_val = cur_val * mult
                if new_val < 0:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial = f_trial / s
                obj = fast_convolve(f_trial).max()
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_val = new_val

            if best_val != cur_val:
                f[j] = best_val
                s = f.sum()
                f = f / s
                if best_obj < best_b - 1e-14:
                    best_b = best_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            no_improve += 1
            if no_improve > 3:
                # Try perturbation
                f = f * (1.0 + rng.standard_normal(n) * 0.001)
                f = np.maximum(f, 0.0)
                f = f / f.sum()
                no_improve = 0
            elif no_improve > 10:
                break
        else:
            no_improve = 0

        if it % 10 == 0:
            score = 2.0 * n * best_b
            print(f"    {label} pass={it} score={score:.6f} t={time.monotonic()-t0:.0f}s")

    return best_f, best_b


def pair_optimization(f_init, time_limit, label="pair"):
    """Optimize pairs of coordinates that interact at peak locations."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_b = fast_convolve(f).max()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()
        # Find top peak
        k_star = int(np.argmax(b))
        # Find coordinates that contribute to this peak
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        active = list(range(lo, hi + 1))
        if len(active) < 2:
            break

        improved = False
        # Try optimizing pairs within the active set
        for ii in range(len(active)):
            if time.monotonic() - t0 > time_limit:
                break
            for jj in range(ii + 1, min(ii + 5, len(active))):
                i = active[ii]
                j = active[jj]
                if f[i] < 1e-14 or f[j] < 1e-14:
                    continue
                # Try scaling i up/down while scaling j opposite
                cur_obj = best_b
                best_scale = 1.0
                for delta in [-0.3, -0.2, -0.1, 0.1, 0.2, 0.3]:
                    f_trial = f.copy()
                    add = f[j] * delta
                    f_trial[i] = f[i] + add
                    f_trial[j] = f[j] - add
                    if f_trial[i] < 0 or f_trial[j] < 0:
                        continue
                    s = f_trial.sum()
                    f_trial = f_trial / s
                    obj = fast_convolve(f_trial).max()
                    if obj < cur_obj - 1e-14:
                        cur_obj = obj
                        best_scale = delta

                if best_scale != 1.0:
                    add = f[j] * best_scale
                    f[i] += add
                    f[j] -= add
                    f = np.maximum(f, 0.0)
                    f = f / f.sum()
                    if cur_obj < best_b - 1e-14:
                        best_b = cur_obj
                        best_f = f.copy()
                        improved = True

        if not improved:
            break

        if it % 5 == 0:
            score = 2.0 * n * best_b
            print(f"    {label} it={it} score={score:.6f} t={time.monotonic()-t0:.0f}s")

    return best_f, best_b


def interpolate_to_length(f, new_n):
    """Interpolate sequence to new length."""
    n = len(f)
    if n == new_n:
        return f.copy()
    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new = f_new / s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def peak_shaving(f_init, time_limit, label="ps"):
    """Directly reduce the highest peaks by redistributing mass."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_b = fast_convolve(f).max()
    rng = np.random.default_rng(999)
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()
        k_star = int(np.argmax(b))

        # Find contributions to the peak
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        contribs = np.array([f[i] * f[k_star - i] for i in range(lo, hi + 1)])
        total = contribs.sum()
        if total < 1e-15:
            break

        # Find the most "concentrated" contribution
        # We want to reduce the dominant pair
        # Strategy: reduce one of the pair members and add to a "safe" coordinate
        # Safe coordinate: one that contributes little to any peak
        peak_list = np.argpartition(b, -min(20, len(b)))[-min(20, len(b)):]

        # Find safe coordinates (low contribution to all top peaks)
        contrib_to_peaks = np.zeros(n)
        for pk in peak_list:
            plo = max(0, pk - n + 1)
            phi = min(n - 1, pk)
            for i in range(plo, phi + 1):
                contrib_to_peaks[i] += f[i] * f[pk - i]

        # Pick safe coordinate (low total contribution, existing mass)
        safe_mask = (contrib_to_peaks < np.median(contrib_to_peaks)) & (f > 1e-6)
        if not safe_mask.any():
            safe_mask = f > np.percentile(f, 25)

        safe_coords = np.where(safe_mask)[0]
        if len(safe_coords) == 0:
            break

        # Try reducing peak contributors and adding to safe coords
        best_improvement = 0
        best_action = None

        for i in range(lo, hi + 1):
            for j in range(lo, hi + 1):
                if i == j:
                    continue
                pair_contrib = f[i] * f[j]
                if pair_contrib < 0.05 * M:
                    continue
                # Try reducing f[i] by some fraction and adding to a safe coord
                for frac in [0.05, 0.1, 0.15, 0.2]:
                    reduce_amount = f[i] * frac
                    if reduce_amount < 1e-10:
                        continue
                    for sc in safe_coords[:3]:
                        f_trial = f.copy()
                        f_trial[i] -= reduce_amount
                        f_trial[sc] += reduce_amount
                        if f_trial[i] < 0:
                            continue
                        s = f_trial.sum()
                        f_trial = f_trial / s
                        obj = fast_convolve(f_trial).max()
                        if obj < M - 1e-12 - best_improvement:
                            best_improvement = M - obj
                            best_action = (i, sc, reduce_amount)

        if best_action is not None:
            i, sc, amt = best_action
            f[i] -= amt
            f[sc] += amt
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            cur_obj = fast_convolve(f).max()
            if cur_obj < best_b - 1e-14:
                best_b = cur_obj
                best_f = f.copy()
        else:
            # Perturb
            f = f * (1.0 + rng.standard_normal(n) * 0.002)
            f = np.maximum(f, 0.0)
            f = f / f.sum()

        if it % 30 == 0:
            score = 2.0 * n * best_b
            print(f"    {label} it={it} score={score:.6f} t={time.monotonic()-t0:.0f}s")

    return best_f, best_b


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main optimization: combines LP-based direction finding, coordinate descent,
    pair optimization, and peak shaving to minimize 2*n*max(conv(a,a))/sum(a)^2.
    """
    deadline = time.monotonic() + max(10.0, float(budget_s) - 15.0)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(1024) / 1024
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"No parent, using uniform n=1024, score={p_score:.6f}")

    best_seq = parent.copy()
    best_score = p_score

    # === Phase 1: LP optimization on parent ===
    n_p = len(parent)
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.35, 500.0)
    if t1 > 60:
        print(f"\n=== Phase 1: LP main (n={n_p}, budget={t1:.0f}s) ===")
        f1, obj1 = run_lp_optimize(parent, time_limit=t1, label="lp1",
                                    initial_frac=0.97, final_frac=0.88,
                                    max_peaks=350, patience=12, start_step=0.12)
        score1 = 2.0 * n_p * obj1
        vs1 = float(evaluate_sequence(f1.tolist()))
        if np.isfinite(vs1) and vs1 < best_score:
            best_score = vs1
            best_seq = f1.copy()
            print(f"  NEW BEST [lp1]: {vs1:.6f}")

    # === Phase 2: Coordinate descent polish ===
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.15, 200.0)
    if t2 > 30:
        n_b = len(best_seq)
        print(f"\n=== Phase 2: CD polish (n={n_b}, budget={t2:.0f}s) ===")
        f2, obj2 = coordinate_descent(best_seq, time_limit=t2, label="cd1")
        vs2 = float(evaluate_sequence(f2.tolist()))
        if np.isfinite(vs2) and vs2 < best_score:
            best_score = vs2
            best_seq = f2.copy()
            print(f"  NEW BEST [cd1]: {vs2:.6f}")

    # === Phase 3: Deeper LP ===
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.25, 400.0)
    if t3 > 40:
        n_b = len(best_seq)
        print(f"\n=== Phase 3: LP deep (n={n_b}, budget={t3:.0f}s) ===")
        f3, obj3 = run_lp_optimize(best_seq, time_limit=t3, label="lp3",
                                    initial_frac=0.95, final_frac=0.80,
                                    max_peaks=500, patience=20, start_step=0.10)
        score3 = 2.0 * n_b * obj3
        vs3 = float(evaluate_sequence(f3.tolist()))
        if np.isfinite(vs3) and vs3 < best_score:
            best_score = vs3
            best_seq = f3.copy()
            print(f"  NEW BEST [lp3]: {vs3:.6f}")

    # === Phase 4: Peak shaving ===
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.10, 100.0)
    if t4 > 20:
        n_b = len(best_seq)
        print(f"\n=== Phase 4: Peak shaving (n={n_b}, budget={t4:.0f}s) ===")
        f4, obj4 = peak_shaving(best_seq, time_limit=t4, label="ps1")
        vs4 = float(evaluate_sequence(f4.tolist()))
        if np.isfinite(vs4) and vs4 < best_score:
            best_score = vs4
            best_seq = f4.copy()
            print(f"  NEW BEST [ps1]: {vs4:.6f}")

    # === Phase 5: Multi-length exploration ===
    rem = deadline - time.monotonic()
    n_base = len(best_seq)
    target_lengths = []
    for factor in [0.9, 1.05, 1.15, 1.3, 1.5, 2.0]:
        tl = int(n_base * factor)
        tl = max(512, min(8000, tl))
        if tl != n_base:
            # Round to nice number
            tl = int(round(tl / 64.0)) * 64
            if tl not in target_lengths:
                target_lengths.append(tl)
    target_lengths.sort(key=lambda x: abs(x - n_base))

    for target_n in target_lengths[:4]:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t5 = min(rem * 0.15, 150.0)
        if t5 < 25:
            break
        f_interp = interpolate_to_length(best_seq, target_n)
        print(f"\n=== Phase 5: LP n={target_n} (budget={t5:.0f}s) ===")
        f5, obj5 = run_lp_optimize(f_interp, time_limit=t5, label=f"lp_n{target_n}",
                                    initial_frac=0.97, final_frac=0.85,
                                    max_peaks=min(400, target_n // 2),
                                    patience=15, start_step=0.10)
        score5 = 2.0 * target_n * obj5

    # === Phase 6: Pair optimization ===
    rem = deadline - time.monotonic()

    if t6 > 15:
        n_b = len(best_seq)
        print(f"\n=== Phase 6: Pair opt (n={n_b}, budget={t6:.0f}s) ===")
        f6, obj6 = pair_optimization(best_seq, time_limit=t6, label="pair1")
        vs6 = float(evaluate_sequence(f6.tolist()))

            print(f"  NEW BEST [pair1]: {vs6:.6f}")

    # === Phase 7: Final LP burst ===
    rem = deadline - time.monotonic()
    t7 = min(rem * 0.25, 200.0)
    if t7 > 20:
        n_b = len(best_seq)
        print(f"\n=== Phase 7: Final LP (n={n_b}, budget={t7:.0f}s) ===")
        f7, obj7 = run_lp_optimize(best_seq, time_limit=t7, label="lp_final",
                                    initial_frac=0.93, final_frac=0.75,
                                    max_peaks=600, patience=25, start_step=0.08)
        score7 = 2.0 * n_b * obj7
        vs7 = float(evaluate_sequence(f7.tolist()))
        if np.isfinite(vs7) and vs7 < best_score:
            best_score = vs7
            best_seq = f7.copy()
            print(f"  NEW BEST [lp_final]: {vs7:.6f}")

    # === Phase 8: Final coordinate descent ===
    rem = deadline - time.monotonic()
    t8 = min(rem * 0.20, 100.0)
    if t8 > 15:
        n_b = len(best_seq)
        print(f"\n=== Phase 8: Final CD (n={n_b}, budget={t8:.0f}s) ===")
        f8, obj8 = coordinate_descent(best_seq, time_limit=t8, label="cd_final")
