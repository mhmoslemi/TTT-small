    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = next_fast_len(size)
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

    """Score = 2n * max(conv(a,a)) assuming sum(a) = 1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def hankel_matvec(w, a, eps=0.0):
    """Compute (W + eps*I) @ a where W[i,k] = w[i+k]. w has length 2n-1."""
    n = len(a)

    conv = np.convolve(w_rev, a, mode='full')
    result = conv[2*n-2::-1][:n]
    if eps > 0:

    return result

def proj_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    """Conjugate gradient for Ax = b, A symmetric PD."""

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Annealing QP (iteratively solve min a^T W_w a with increasing weight sharpness)
2. Conjugate gradient for QP matrix solves with Hankel-structured matvec via FFT
3. Coordinate descent with exact golden-section search (pairwise transfers)
4. Length exploration and multiple random restarts

import numpy as np
import time
from scipy.fft import rfft, irfft, next_fast_len
import cvxpy as cp


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = next_fast_len(size)
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Score = 2n * max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def hankel_matvec(w, a, eps=0.0):
    """Compute (W + eps*I) @ a where W[i,k] = w[i+k]. w has length 2n-1."""
    n = len(a)
    w_rev = w[::-1]
    conv = np.convolve(w_rev, a, mode='full')
    result = conv[2 * n - 2::-1][:n]
    if eps > 0:
        result = result + eps * a
    return result


def proj_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_idx = np.nonzero(cond)[0]
    if len(rho_idx) == 0:
        return np.ones(n) / n
    rho = rho_idx[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def cg_solve(matvec_fn, b, x0=None, max_iter=500, tol=1e-12):
    """Conjugate gradient for symmetric PD system Ax = b."""
    n = len(b)
    if x0 is not None:
        x = x0.astype(np.float64).copy()
    else:
        x = np.ones(n) / n
    r = b - matvec_fn(x)
    p = r.copy()
    rs = r @ r
    for _ in range(max_iter):
        if np.sqrt(rs) < tol:
            break
        Ap = matvec_fn(p)
        pAp = p @ Ap
        if pAp < 1e-30:
            break
        alpha = rs / pAp
        x = x + alpha * p
        r = r - alpha * Ap
        rs_new = r @ r
        if rs_new > rs * 1.01 or rs_new < 1e-30:
            p = r.copy()
        else:
            p = r + (rs_new / max(rs, 1e-30)) * p
        rs = rs_new
    return x


def solve_qp_weighted(w, a_init, eps=1e-8):
    """Solve min a^T (W+eps*I) a s.t. sum(a)=1, a>=0, W[i,k]=w[i+k]."""
    n = len(a_init)

    def matvec(x):
        return hankel_matvec(w, x, eps)

    # Try CG (unconstrained solution)
    x = cg_solve(matvec, np.ones(n), x0=a_init, max_iter=400, tol=1e-12)

    if x.min() > 1e-10 and x.sum() > 1e-15:
        return x / x.sum()

    # CG solution infeasible; use cvxpy
    W = np.zeros((n, n))
    for i in range(n):
        W[i, :] = w[i:i + n]
    W = (W + W.T) / 2.0
    W += eps * np.eye(n)

    a_var = cp.Variable(n)
    obj = cp.quad_form(a_var, W, assume_PSD=True)
    constraints = [cp.sum(a_var) == 1, a_var >= 0]
    prob = cp.Problem(cp.Minimize(obj), constraints)
    try:
        prob.solve(solver=cp.CLARABEL, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            return None
    if a_var.value is not None:
        result = np.maximum(a_var.value, 0)
        s = result.sum()
        if s > 1e-15:
            return result / s
    return None

def interpolate_sequence(f, new_n):
    """Interpolate to new length using linear interpolation on [0,1]."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    result = np.interp(new_x, old_x, f_norm)
    result = np.maximum(result, 0)
    ss = result.sum()
    if ss > 1e-15:
        result = result / ss
    else:
        result = np.ones(new_n) / new_n
    return result

        a_qp2, _ = annealing_qp(best_seq, t_alloc, verbose=False)
        best_seq, best_score = check_and_update(a_qp2, "final_qp", best_seq, best_score)

    # === Phase 7: Final CD squeeze ===
    print(f"\n=== Phase 7: Final CD Squeeze ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_alloc = min(120.0, rem * 0.8)
        a_cd3, _ = coord_descent(best_seq, t_alloc, verbose=False, seed=555)
        best_seq, best_score = check_and_update(a_cd3, "final_cd", best_seq, best_score)

    # Fallback
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
