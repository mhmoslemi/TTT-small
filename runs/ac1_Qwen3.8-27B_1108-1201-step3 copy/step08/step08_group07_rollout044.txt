def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def build_conv_matrix(f):
    """Build (2n-1) x n matrix C such that C @ g = f * g (linear conv)."""
    n = len(f)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)


def solve_best_response(f, M, n_constraints=None):
    """Solve: max sum(g) s.t. C(f) @ g <= M, g >= 0."""
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    C = build_conv_matrix(f)
    if n_constraints is not None:
        # Use only the most important constraints (top n_constraints)
        # Heuristic: use constraints around the peak
        conv = fast_convolve(f)
        peak_idx = np.argmax(conv)
        idxs = np.arange(peak_idx - n_constraints // 2, peak_idx + n_constraints // 2 + 1)
        idxs = idxs[(idxs >= 0) & (idxs < 2 * n - 1)]
        C_sub = C[idxs]
        M_sub = np.full(len(idxs), M)
        constraints = [C_sub @ g <= M_sub]
    else:
        constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            try:
                prob.solve(solver=cp.SCS)
            except Exception:
                return f.copy()
    if g.value is None:
        return f.copy()
    return np.maximum(g.value, 0)

        if not improved and iters > 10:

        if verbose and iters % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"  Mirror iter={iters}, best={best_score:.8f}, time={elapsed:.0f}s")

    return best_a, best_score


def truncate_padding(a0, time_limit, verbose=True):
    """
    Try finding optimal padding lengths.
    For a core sequence, try different total lengths by zero-padding.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    # Try different total lengths
    for new_n in range(n // 2, n * 2 + 1, max(1, n // 16)):
        if time.monotonic() - t0 > time_limit:
            break
        if new_n < 4:
            continue
        a_padded = np.zeros(new_n)
        # Center the core
        offset = (new_n - n) // 2
        offset = max(0, min(offset, new_n - n))
        a_padded[offset:offset + n] = a
        ss = a_padded.sum()
        if ss < 1e-15:
            continue
        a_padded = a_padded / ss
        sc = compute_score(a_padded)
        if sc < best_score:
            best_score = sc
            best_a = a_padded.copy()
            if verbose:
                print(f"  Truncate: n={new_n}, score={sc:.8f}")

    return best_a, best_score


def subsequence_search(a0, time_limit, verbose=True):
    """Try to find better subsequences by removing elements."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    # Try removing random subsets
    for trial in range(100):
        if time.monotonic() - t0 > time_limit:
            break
        # Try different subset sizes
        for frac in [0.5, 0.7, 0.8, 0.9]:
            new_n = max(4, int(n * frac))
            if new_n >= n:
                continue
            idx = rng.choice(n, size=new_n, replace=False)
            idx.sort()
            a_sub = a[idx]
            ss = a_sub.sum()
            if ss < 1e-15:
                continue
            a_sub = a_sub / ss
            sc = compute_score(a_sub)
            if sc < best_score:
                best_score = sc
                best_a = a_sub.copy()
                if verbose:
                    print(f"  Subseq: n={new_n}, score={sc:.8f}")

    return best_a, best_score


def check_and_update(a, label, best_seq, best_score, verbose=True):
    """Check a candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def uniform_sequence(n):
    return np.ones(n) / n
