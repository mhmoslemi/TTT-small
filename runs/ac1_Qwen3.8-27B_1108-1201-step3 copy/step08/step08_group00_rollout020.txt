def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)

    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)

    for j in range(n):
        T[j:j + n, j] = a
    return T

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    def make_func(T_val):
        def func(z_in):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(z_in):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_in)
            a = softmax_from_z(z_in)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(50, maxiter // n_stages)

        if time.monotonic() - t0 > time_limit:
            break

        if remaining < 5:
            break

                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),

            )
            z = res.x
        except Exception:
            pass

        # Linear minimization oracle (toward step)
        # min_x g^T x over simplex = min_i g[i]
        i_toward = int(np.argmin(g))

        if len(support) > 1:
            sup_list = list(support)
            g_sup = g[sup_list]
            i_away = sup_list[int(np.argmax(g_sup))]

                i_away = -1
        else:
            i_away = -1

        a_toward = a.copy()
        a_toward[i_toward] = 1.0 - a_toward[i_toward]
        # Actually for FW, the step is: s = (1-t)*a + t*e_i_toward
        # Choose t by line search or fixed schedule
        t_toward = 2.0 / (step + 2)
        
        best_trial = None
        best_trial_score = best_score

        a_trial = (1 - t_toward) * a + t_toward * np.zeros(n)
        a_trial[i_toward] = 1.0
        a_trial = a_trial / a_trial.sum()
        score_trial = 2.0 * n * fast_convolve(a_trial).max()
        if score_trial < best_trial_score:
            best_trial = a_trial.copy()
            best_trial_score = score_trial
        
        # Try away step
        if i_away >= 0 and a[i_away] > 1e-15:
            gamma = min(1.0, a[i_away] * 0.5)
            a_trial_away = a.copy()
            a_trial_away[i_away] *= (1 - gamma)
            a_trial_away = a_trial_away / a_trial_away.sum()
            score_away = 2.0 * n * fast_convolve(a_trial_away).max()
            if score_away < best_trial_score:
                best_trial = a_trial_away.copy()
                best_trial_score = score_away

            a = best_trial
            best_score = best_trial_score
            best_a = a.copy()
            # Update support
            if i_toward >= 0:
                support.add(i_toward)
            if i_away >= 0:
                if a[i_away] < 1e-10:
                    support.discard(i_away)
            # Prune support
            if len(support) > 2 * n // 3:
                support = set(np.nonzero(a > 1e-8)[0])

            break

1. Find s = argmin_{x in C} f(a) + <∇f(a), x - a> (linear oracle)
2. Update: a_new = (1-t) * a + t * s

For the simplex C = {a >= 0, sum(a) = 1}, the linear oracle finds s = e_i where i = argmin <∇f(a), e_i> = argmin ∇f(a)[i].

The gradient of max_k a^T Q_k a is 2 * Q_{k*} * a (where k* is active). But since we're using the subgradient, it's Q_{k*} * a (the factor of 2 doesn't matter for argmin).

                score_trial = 2.0 * n * fast_convolve(a_trial).max()
                if score_trial < best_trial_score:
                    best_trial_score = score_trial
                    best_trial = a_trial.copy()

        if best_trial is not None:
            a = best_trial
            best_score = best_trial_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                break

    return best_a, best_score


def structured_init(n, rng, mode=0):
    """Create structured initial sequence."""
    if mode == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif mode == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif mode == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 1.5
    elif mode == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-6 * x * (1 - x))
    elif mode == 4:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n))
        a = a ** 2
    elif mode == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-x * (1 - x) * 8) * (1 + 0.3 * np.cos(4 * np.pi * x))
    else:
        a = np.ones(n) * (1 + 0.1 * np.sin(2 * np.pi * rng.random() * 3 * np.linspace(0, 1, n)))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = f / f.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def peak_shave_local(a0, time_limit, n_steps=15000):
    """Multi-peak shaving with adaptive coordinate adjustments."""
    n = len(a0)
    a = norm_sequence(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        b_work = b.copy()
        if len(b_work) > 2:
            b_work[0] = -np.inf
            b_work[-1] = -np.inf
        peak_k = int(np.argmax(b_work))

        peak_val = b_work[peak_k]
        near_peak_threshold = peak_val - 0.02 * peak_val
        near_peaks = np.where(b_work >= near_peak_threshold)[0]

        # Find top contributors
        contrib_count = np.zeros(n)
        for k in near_peaks[:8]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[k - idx]
            top_idx = np.argsort(contribs)[::-1][:8]
            for ci in top_idx:
                if contribs[ci] > 0:
                    contrib_count[idx[ci]] += contribs[ci]

        top_coords = np.argsort(contrib_count)[::-1][:15]

        improved = False
        for i in top_coords:
            if a[i] < 1e-14:
                continue
            for factor in [0.97, 0.94, 0.90, 0.85, 0.80]:
                new_val = a[i] * factor
                diff = a[i] - new_val
                if diff < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                if others.sum() < 1e-15:
                    continue
                # Redistribute to low-contrib positions
                low_coords = np.argsort(contrib_count)[:n // 4]
                low_mask = np.zeros(n, dtype=bool)
                low_mask[low_coords] = True
                low_mask[i] = False
                if low_mask.sum() > 0:
                    dist = a_trial[low_mask] + 1e-10
                    a_trial[low_mask] = dist + diff * dist / dist.sum()
                else:
                    a_trial[mask] = others + diff * others / others.sum()
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try pair adjustment
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_contrib = np.argsort(contribs)[::-1][:6]
            found = False
            for ci in top_contrib:
                i = idx[ci]
                j = peak_k - i
                if i == j or i >= n or j >= n:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.93, 0.88, 0.82]:
                    for f2 in [0.93, 0.88, 0.82]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = norm_sequence(a_trial)
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-15:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            found = True
                            improved = True
                            break
                    if found:
                        break
                if found:
                    break
            if not improved:
                break

    return best_a, best_score


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining LP-based Nesterov, extended L-BFGS,
    Frank-Wolfe, peak shaving, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 768
        best_seq = structured_init(n_init, rng, mode=0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = norm_sequence(a)
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # === Phase 1: Nesterov LP (main global optimization) ===
    print(f"\n=== Phase 1: Nesterov LP n={n_best} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.28, 500.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=100000)
        try_update(a_lp, "nesterov_lp_1")

    # === Phase 2: Extended L-BFGS with multi-stage annealing ===
    print(f"\n=== Phase 2: L-BFGS Extended (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 450.0)
    if t_alloc > 30:
        print(f"  L-BFGS T: 150->0.1, budget={t_alloc:.0f}s")
        a_lb = lbgfs_refine(best_seq, t_alloc, T_start=150.0, T_end=0.1, maxiter=15000, n_stages=12)
        try_update(a_lb, "lbgfs_main")

    # === Phase 3: Frank-Wolfe local refinement ===
    print(f"\n=== Phase 3: Frank-Wolfe (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 120.0)
    if t_alloc > 10:
        a_fw, _ = frank_wolfe_refine(best_seq, t_alloc, n_steps=50000)
        try_update(a_fw, "frank_wolfe")

    # === Phase 4: Second Nesterov LP round ===
    print(f"\n=== Phase 4: Nesterov LP r2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 200.0)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp2, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=50000)
        try_update(a_lp2, "nesterov_lp_2")

    # === Phase 5: L-BFGS low-temperature polish ===
    print(f"\n=== Phase 5: L-BFGS Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 200.0)
    if t_alloc > 20:
        print(f"  L-BFGS T: 20->0.05, budget={t_alloc:.0f}s")
        a_lb2 = lbgfs_refine(best_seq, t_alloc, T_start=20.0, T_end=0.05, maxiter=10000, n_stages=10)
        try_update(a_lb2, "lbgfs_polish")

    # === Phase 6: Peak shaving ===
    print(f"\n=== Phase 6: Peak Shaving (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.06, 80.0)
    if t_alloc > 10:
        a_ps, _ = peak_shave_local(best_seq, t_alloc, n_steps=20000)
        try_update(a_ps, "peak_shave")

    # === Phase 7: Multi-length exploration ===
    print(f"\n=== Phase 7: Multi-Length (best={best_score:.6f}) ===")
    alt_lengths = []
    for mult in [1.5, 1.25, 2.0, 0.75, 1.33]:
        alt_n = int(n_best * mult)
        if 256 <= alt_n <= 5120 and alt_n != n_best:
            alt_lengths.append(alt_n)
    # Also try some fixed lengths
    for fixed_n in [512, 768, 1024, 1280, 1536, 2048]:
        if 256 <= fixed_n <= 5120 and fixed_n != n_best:
            alt_lengths.append(fixed_n)
    alt_lengths = sorted(set(alt_lengths))

    for alt_n in alt_lengths[:3]:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_total = min(rem * 0.25, 200.0)
        if t_total < 30:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_seq, alt_n)

        # Quick LP
        t_lp = t_total * 0.5
        a_alt, _ = nesterov_lp_refine(init_alt, t_lp, solver=solver, n_max_iters=20000)
        try_update(a_alt, f"lp_n{alt_n}")

        # L-BFGS on the result
        rem = deadline - time.monotonic()
        if rem > 30 and best_seq is not None and len(best_seq) == alt_n:
            t_lb = min(rem * 0.4, 80.0)
            if t_lb > 15:
                a_alt2 = lbgfs_refine(best_seq, t_lb, T_start=30.0, T_end=0.1, maxiter=5000, n_stages=8)
                try_update(a_alt2, f"lbgfs_n{alt_n}")

    # === Phase 8: Perturbation + re-optimization ===
    print(f"\n=== Phase 8: Perturbation Escape (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 50:
        t_pert = min(rem * 0.2, 120.0)
        n_cur = len(best_seq)

        # Try multiple perturbation scales
        for scale in [0.03, 0.08, 0.15]:
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            perturbed = best_seq + scale * rng.standard_normal(n_cur)
            perturbed = norm_sequence(perturbed)

            t_sub = min(t_pert / 3, rem * 0.8)
            t_lp_pert = t_sub * 0.5
            a_pert, _ = nesterov_lp_refine(perturbed, t_lp_pert, solver=solver, n_max_iters=10000)
            try_update(a_pert, f"pert_{scale}")

            rem = deadline - time.monotonic()
            if rem > 15:
                t_lb_pert = min(rem * 0.5, t_sub * 0.4)
                if t_lb_pert > 10:
                    a_pert2 = lbgfs_refine(best_seq, t_lb_pert, T_start=10.0, T_end=0.1, maxiter=3000, n_stages=6)
                    try_update(a_pert2, f"pert_lbgfs_{scale}")

    # === Phase 9: Final multi-algorithm polish ===
    print(f"\n=== Phase 9: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_fin = min(rem * 0.3, 60.0)
        a_fin = lbgfs_refine(best_seq, t_fin, T_start=5.0, T_end=0.03, maxiter=6000, n_stages=8)
        try_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_fw = min(rem * 0.2, 30.0)
        a_fw2, _ = frank_wolfe_refine(best_seq, t_fw, n_steps=20000)
        try_update(a_fw2, "final_fw")

    rem = deadline - time.monotonic()
    if rem > 10:
        t_ps = min(rem * 0.2, 25.0)
        a_ps2, _ = peak_shave_local(best_seq, t_ps, n_steps=8000)
        try_update(a_ps2, "final_ps")

    rem = deadline - time.monotonic()
    if rem > 10:
        t_lp_fin = min(rem * 0.3, 20.0)
        a_lp_fin, _ = nesterov_lp_refine(best_seq, t_lp_fin, solver=solver, n_max_iters=5000)
        try_update(a_lp_fin, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
