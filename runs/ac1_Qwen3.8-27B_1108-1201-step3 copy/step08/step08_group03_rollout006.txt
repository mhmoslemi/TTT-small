import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_direction(f, g):
    """Compute cross-convolution f*g via FFT."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_constraint_matrix(f, indices, n):
    """Build constraint matrix where row i encodes (f*g)[k_i] = A[i,:] @ g."""
    K = len(indices)
    A = np.zeros((K, n))
    for i, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    return A


def solve_lp_direction(f, n, M, active_idx):
    """Solve: max sum(g) s.t. g>=0, A@g <= M. Returns g and dual info."""
    K = len(active_idx)
    if K == 0:
        return None, None
    A = build_constraint_matrix(f, active_idx, n)
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    g_val = None
    dual_vals = None
    try:
        prob.solve(solver=cp.SCS, max_iters=500000, eps=1e-9)
        if g.value is not None:
            g_val = g.value
            # Try to extract dual variables
            for con in constraints:
                if con.equals(A @ g <= M):
                    dual_vals = con.dual_value
    except Exception:
        pass
    if g_val is None:
        if n <= 3000:
            try:
                prob.solve(solver=cp.ECOS, max_iters=30000)
                if g.value is not None:
                    g_val = g.value
            except Exception:
                pass
    if g_val is None:
        try:
            prob.solve(solver=cp.PDLP)
            if g.value is not None:
                g_val = g.value
        except Exception:
            pass
    if g_val is None:
        return None, None
    g_val = np.maximum(g_val, 0.0)
    if np.any(np.isnan(g_val)):
        return None, None
    return g_val, dual_vals


def parabolic_linesearch(f, h, M, f0, max_iter=20):
    """
    Parabolic/quadratic interpolation line search.
    Find t in [0,1] minimizing max_conv of (1-t)*f + t*h.
    """
    def obj_at_t(t):
        f_trial = np.maximum((1.0 - t) * f + t * h, 0.0)
        s = f_trial.sum()
        if s < 1e-12:
            return 1e10
        f_trial = f_trial / s
        return fast_convolve(f_trial).max()

    # Initial scan
    candidates = [0.02, 0.05, 0.1, 0.2, 0.35, 0.5, 0.7, 0.9]
    best_t = 0.05
    best_val = obj_at_t(0.05)
    f0_val = f0

    for t in candidates:
        val = obj_at_t(t)
        if val < best_val:
            best_val = val
            best_t = t

    if best_t <= 0.02:
        return best_t, best_val

    # Golden section + parabolic refinement
    lo = max(0.0, best_t - 0.15)
    hi = min(1.0, best_t + 0.15)
    vals缓存 = {}

    def cached_obj(t):
        key = round(t, 8)
        if key not in vals缓存:
            vals缓存[key] = obj_at_t(t)
        return vals缓存[key]

    for _ in range(max_iter):
        if hi - lo < 1e-6:
            break
        m1 = lo + (hi - lo) * 0.382
        m2 = hi - (hi - lo) * 0.382
        v1 = cached_obj(m1)
        v2 = cached_obj(m2)
        if v1 < v2:
            hi = m2
        else:
            lo = m1

    t_best = (lo + hi) / 2.0
    v_best = cached_obj(t_best)

    if v_best < best_val:
        best_t = t_best
        best_val = v_best

    return best_t, best_val

def subgradient_polish(f_init, time_limit, lr_init=0.003, n_top=60, temp_init=0.002):
    """
    Softmax-smoothed subgradient method with momentum and adaptive parameters.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    v = np.zeros(n)
    lr = lr_init
    temp = temp_init
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(123)
    momentum = 0.85

    while time.monotonic() - t0 < time_limit:
        # Batch of iterations
        for _ in range(5):
            if time.monotonic() - t0 > time_limit:
                break
            it += 1
            b = fast_convolve(f)

            # Softmax weights on top-n peaks
            if n_top < len(b):
                top_idx = np.argpartition(b, -n_top)[-n_top:]
                top_vals = b[top_idx]
                scaled = top_vals / temp
                scaled -= scaled.max()
                w_top = np.exp(scaled)
                w_top /= w_top.sum()
            else:
                top_idx = np.arange(len(b))
                scaled = b / temp
                scaled -= scaled.max()
                w_top = np.exp(scaled)
                w_top /= w_top.sum()

            # Compute subgradient
            g = np.zeros(n)
            for i, k in enumerate(top_idx):
                if w_top[i] < 1e-14:
                    continue
                lo = max(0, k - n + 1)
                hi = min(n - 1, k)
                idx = np.arange(lo, hi + 1)
                g[idx] += w_top[i] * 2.0 * f[k - idx]

            # Projected gradient step with momentum
            v = momentum * v - lr * g
            f_new = f + v
            f = project_simplex(f_new)

        # Evaluate and update
        obj = fast_convolve(f).max()
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 5:
                lr *= 0.85
                temp *= 1.15
                no_improve = 0
            if no_improve > 30:
                # Restart from best with perturbation
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                v = np.zeros(n)
                no_improve = 0

    return best_f, best_obj


def multi_length_search(best_seq, best_score, time_limit, n_candidates):

        new_x = np.linspace(0.0, 1.0, target_n)
        f_interp = np.interp(new_x, old_x, best_seq)
        f_interp = np.maximum(f_interp, 0.0)
        s = f_interp.sum()
        if s > 1e-10:
            f_interp = f_interp / s
        else:
            f_interp = np.ones(target_n) / target_n

        # Run LP
        f_opt, _ = lp_cutting_planes(f_interp, time_limit=t_alloc,
                                      initial_k=40, max_k=300,
                                      label=f"n{target_n}")
        score = float(evaluate_sequence(f_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = f_opt.copy()
            print(f"  NEW BEST [n={target_n}]: {score:.8f}")

    return best_seq, best_score


def coordinate_descent(f_init, time_limit):
    """Coordinate descent with multiplicative updates on subsets."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    step = max(1, n // 200)

    improved = True
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(0, n, step):
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            if f[j] < 1e-15:
                continue
            orig = f[j]
            best_val = orig
            best_obj_j = best_obj
            for mult in [0.8, 0.9, 1.1, 1.2]:
                f[j] = orig * mult
                s = f.sum()
                if s < 1e-10:
                    f[j] = orig
                    continue
                f_norm = f / s
                obj = fast_convolve(f_norm).max()
                if obj < best_obj_j - 1e-14:
                    best_obj_j = obj
                    best_val = orig * mult
            f[j] = best_val
            s = f.sum()
            if s > 1e-10:
                f = f / s
            if best_obj_j < best_obj - 1e-14:
                best_obj = best_obj_j
                improved = True

    return f, best_obj
