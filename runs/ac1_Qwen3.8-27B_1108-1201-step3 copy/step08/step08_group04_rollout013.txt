            top_idx = np.argpartition(b, -top_k)[-top_k:]
            structural = [0, full_len - 1, (full_len - 1) // 2, n - 1, 2 * (n - 1)]
            structural = [i for i in structural if 0 <= i < full_len]
            all_idx = list(set(list(top_idx.tolist()) + structural))
            all_idx.sort(key=lambda i: -b[i])
            indices = np.array(all_idx[:top_k])
            indices = np.sort(indices)

        # Solve LP
        g = solve_lp_dir(f, indices, M)
        if g is None:
            if len(indices) > 10:
                small = np.sort(np.argpartition(b, -10)[-10:])
                g = solve_lp_dir(f, small, M)
            if g is None:
                stuck += 3
                if stuck > 80:
                    break
                continue

        S_g = g.sum()
        if S_g <= 1.0 + 1e-5:
            stuck += 1
            if stuck > 50:
                break
            continue

        h = g / S_g

        # Momentum smoothing
        mom_dir = mom_beta * mom_dir + (1.0 - mom_beta) * h
        mom_sum = mom_dir.sum()
        if mom_sum > 1e-10:
            h_mom = mom_dir / mom_sum
        else:
            h_mom = h

        # Line search: try both raw and momentum directions
        best_trial_obj = M
        best_trial_f = f.copy()
        best_t = 0.0

        for direction in [h, h_mom]:
            for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7]:
                f_trial = (1.0 - t) * f + t * direction
                f_trial = np.maximum(f_trial, 0.0)
                s = f_trial.sum()
                if s < 1e-12:
                    continue
                f_trial = f_trial / s
                obj = fast_convolve(f_trial).max()
                if obj < best_trial_obj - 1e-15:
                    best_trial_obj = obj
                    best_t = t
                    best_trial_f = f_trial.copy()

        if best_t == 0.0:
            stuck += 1
            if stuck > 80:
                break
            if stuck % 25 == 0:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
                stuck = 0
                mom_dir = np.zeros(n)
            continue

        f = best_trial_f
        cur_obj = fast_convolve(f).max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1

        if it % 40 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} it={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} k={top_k} t={elapsed:.0f}s")

    return best_f, best_obj


def run_mirror_descent(f_init, time_limit, eta_init=0.4, tau_init=0.005, label="md"):
    """KL mirror descent with adaptive step size and temperature."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-15:
        f = f / s
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    eta = eta_init
    tau = tau_init
    it = 0
    no_improve = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()

        # Softmax peak weights
        scaled = (b - M) / max(tau, 1e-10)
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(b))
            w[int(np.argmax(b))] = 1.0
        else:
            w = w / w_sum

        # Subgradient: g(j) = 2 * sum_m w(j+m) * f(m)
        g = 2.0 * np.correlate(w, f, mode='valid')

        # Mirror descent update (KL-based)
        f_new = f * np.exp(-eta * g)
        s_new = f_new.sum()
        if s_new < 1e-15:
            f_new = np.ones(n) / n
        else:
            f_new = f_new / s_new
        f_new = np.maximum(f_new, 1e-16)
        f_new = f_new / f_new.sum()

        cur_obj = fast_convolve(f_new).max()

        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_f = f_new.copy()
            no_improve = 0
            eta = min(eta * 1.03, 3.0)
        else:
            no_improve += 1
            if no_improve > 15:
                eta *= 0.85
                no_improve = 0
            if no_improve > 80:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                eta = eta_init
                tau = tau_init
                no_improve = 0

        f = f_new

        # Gradually warm up temperature
        if it % 200 == 0:
            tau = min(tau * 1.08, 0.3 * best_obj)

        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} it={it} obj={best_obj:.10f} score={2.0*n*best_obj:.6f} eta={eta:.4f} t={elapsed:.0f}s")

    return best_f, best_obj


def run_coord_descent(f_init, time_limit, label="cd"):
    """Coordinate descent with multiple step sizes."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    multipliers = np.array([0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 1.8, 2.5])
    no_pass = 0

    while time.monotonic() - t0 < time_limit:
        any_improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_f, best_obj
            if f[j] < 1e-14:
                continue
            best_j_val = f[j]
            best_j_obj = best_obj
            for mult in multipliers:
                new_val = f[j] * mult
                if new_val < 0:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-12:
                    continue
                f_trial = f_trial / s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-14:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-12:
                    f = f / s
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    any_improved = True
        if not any_improved:
            no_pass += 1
            if no_pass >= 2:
                break
        else:
            no_pass = 0

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization: mirror descent for rapid convergence, LP for
    careful direction finding, coordinate descent for polishing.
    Tries multiple sequence lengths to find the global best.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
    except Exception:
        pass

    if parent is not None:
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    else:
        parent = np.ones(512) / 512.0
        p_score = 2.0
        print(f"No parent, using uniform n=512, score=2.0")

    best_seq = parent.copy()
    best_score = p_score
    n_parent = len(parent)

    # ============================================================
    # Phase 1: Mirror descent on parent length (fast progress)
    # ============================================================
    print(f"\n{'=' * 60}")
    print(f"Phase 1: Mirror descent n={n_parent}")
    print(f"{'=' * 60}")

    rem = deadline - time.monotonic()
    t1 = min(rem * 0.18, 280.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        f_res, _ = run_mirror_descent(parent, time_limit=t1, eta_init=0.5, tau_init=0.003, label="md1")
        real_score = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_score) and real_score < best_score:
            best_score = real_score
            best_seq = f_res.copy()
            print(f"  NEW BEST [md parent]: {real_score:.6f}")
        else:
            print(f"  No improvement, score={real_score:.6f}")

    # ============================================================
    # Phase 2: LP optimization on current best
    # ============================================================
    print(f"\n{'=' * 60}")
    print(f"Phase 2: LP optimization n={len(best_seq)} (best={best_score:.6f})")
    print(f"{'=' * 60}")

    rem = deadline - time.monotonic()
    t2 = min(rem * 0.30, 520.0)
    if t2 > 30:
        n_b = len(best_seq)
        print(f"  Budget: {t2:.0f}s, n={n_b}")
        f_res, _ = run_lp_opt(
            best_seq, time_limit=t2,
            top_k_init=30, top_k_max=min(500, 2 * n_b - 1),
            label="lp2")
        real_score = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_score) and real_score < best_score:
            best_score = real_score
            best_seq = f_res.copy()
            print(f"  NEW BEST [lp main]: {real_score:.6f}")
        else:
            print(f"  No improvement, score={real_score:.6f}")

    # ============================================================
    # Phase 3: Try nearby sequence lengths
    # ============================================================
    print(f"\n{'=' * 60}")
    print(f"Phase 3: Length exploration (best={best_score:.6f}, n={len(best_seq)})")
    print(f"{'=' * 60}")

    rem = deadline - time.monotonic()
    t3 = min(rem * 0.25, 420.0)
    if t3 > 40:
        n_base = len(best_seq)
        # Try a few nearby lengths
        length_factors = [0.88, 0.93, 1.07, 1.12]
        candidates = []
        for fac in length_factors:
            n_try = int(n_base * fac)
            n_try = max(256, min(2500, n_try))
            if n_try not in candidates and abs(n_try - n_base) > n_base * 0.05:
                candidates.append(n_try)
        candidates.sort()

        if candidates:
            per_alloc = t3 / len(candidates)
            for n_try in candidates:
                rem = deadline - time.monotonic()
                if rem < 20:
                    break
                alloc = min(per_alloc, rem * 0.9)
                if alloc < 15:
                    break
                print(f"  Trying n={n_try}, budget={alloc:.0f}s")
                f_init = interp_sequence(best_seq, n_try)
                # Quick mirror descent first
                t_md = alloc * 0.3
                t_lp = alloc * 0.7
                f_md, _ = run_mirror_descent(f_init, time_limit=t_md, eta_init=0.3, tau_init=0.005, label=f"md_n{n_try}")
                real_md = float(evaluate_sequence(f_md.tolist()))
                f_lp, _ = run_lp_opt(f_md, time_limit=t_lp, top_k_init=25,
                                      top_k_max=min(400, 2 * n_try - 1), label=f"lp_n{n_try}")
                real_lp = float(evaluate_sequence(f_lp.tolist()))
                best_local = f_lp if real_lp < real_md else f_md
                best_local_score = min(real_md, real_lp)
                if np.isfinite(best_local_score) and best_local_score < best_score:
                    best_score = best_local_score
                    best_seq = best_local.copy()
                    print(f"  NEW BEST [n={n_try}]: {best_local_score:.6f}")
                else:
                    print(f"  n={n_try}: best={best_local_score:.6f} (no improvement)")

    # ============================================================
    # Phase 4: Deep LP on current best (more iterations, more peaks)
    # ============================================================
    print(f"\n{'=' * 60}")
    print(f"Phase 4: Deep LP n={len(best_seq)} (best={best_score:.6f})")
    print(f"{'=' * 60}")

    rem = deadline - time.monotonic()
    t4 = min(rem * 0.15, 250.0)
    if t4 > 20:
        n_b = len(best_seq)
        print(f"  Budget: {t4:.0f}s")
        f_res, _ = run_lp_opt(
            best_seq, time_limit=t4,
            top_k_init=50, top_k_max=min(800, 2 * n_b - 1),
            label="lp_deep")
        real_score = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_score) and real_score < best_score:
            best_score = real_score
            best_seq = f_res.copy()
            print(f"  NEW BEST [deep]: {real_score:.6f}")

    # ============================================================
    # Phase 5: Coordinate descent polish
    # ============================================================
    print(f"\n{'=' * 60}")
    print(f"Phase 5: Coordinate descent (best={best_score:.6f})")
    print(f"{'=' * 60}")

    rem = deadline - time.monotonic()
    t5 = min(rem * 0.08, 100.0)
    if t5 > 5:
        print(f"  Budget: {t5:.0f}s, n={len(best_seq)}")
        f_res, _ = run_coord_descent(best_seq, time_limit=t5, label="cd")
        real_score = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_score) and real_score < best_score:
            best_score = real_score
            best_seq = f_res.copy()
            print(f"  NEW BEST [cd]: {real_score:.6f}")

    # ============================================================
    # Phase 6: Final LP burst
    # ============================================================
    print(f"\n{'=' * 60}")
    print(f"Phase 6: Final LP (best={best_score:.6f})")
    print(f"{'=' * 60}")

    rem = deadline - time.monotonic()
    t6 = max(0.0, rem * 0.85)
    if t6 > 10:
        n_b = len(best_seq)
        print(f"  Budget: {t6:.0f}s, n={n_b}")
        f_res, _ = run_lp_opt(
            best_seq, time_limit=t6,
            top_k_init=80, top_k_max=min(1000, 2 * n_b - 1),
            label="final")
        real_score = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_score) and real_score < best_score:
            best_score = real_score
            best_seq = f_res.copy()
            print(f"  NEW BEST [final]: {real_score:.6f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'=' * 60}")
    print(f"FINAL: score={best_score:.6f}, n={len(best_seq)}")
    print(f"{'=' * 60}")

    return [float(x) for x in best_seq]
