from scipy.fft import rfft, irfft
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score_normalized(a):
    """Compute score for a normalized sequence (sum=1). Returns 2n*max(conv)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_C(f):
    """Build constraint matrix C where C[k,j] = f[k-j] if 0<=k-j<n else 0."""
    n = len(f)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = f
    return C


def solve_lp_for_direction(f, M):
    """Solve LP: max sum(g) s.t. C*g <= M*1, g >= 0."""
    n = len(f)
    C = build_C(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c, A_ub=C, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
        if res.status != 0:
            return None
        return np.maximum(res.x, 0)
    except Exception:
        return None


def greedy_lp(f, M):
    """Greedy approximation to the LP: build g one coordinate at a time."""
    n = len(f)
    g = np.zeros(n)
    conv_vals = np.zeros(2 * n - 1)
    for j in range(n):
        k_start = j
        k_end = j + n - 1
        slack = M - conv_vals[k_start:k_end + 1]
        f_slice = f
        max_gj = np.inf
        for idx in range(n):
            if f_slice[idx] > 1e-15:
                cand = slack[idx] / f_slice[idx]
                if cand < max_gj:
                    max_gj = cand
        max_gj = max(0.0, max_gj)
        g[j] = max_gj
        conv_vals[k_start:k_end + 1] += f_slice * max_gj
    return g


def multiplicative_descent(a_init, time_limit, T=50.0, verbose=True):
    """
    Mirror descent with entropy Bregman divergence.
    Uses soft-max over convolution peaks for smooth gradient.
    Multiplicative update: a[i] <- a[i] * exp(-step * g[i]), then normalize.
    """
    a = np.maximum(a_init.astype(np.float64), 1e-300)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    # Adaptive temperature schedule: start high, decay
    T_current = T
    step_base = 0.5
    
    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(a)
        M = conv.max()
        score = 2.0 * n * M
        
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
        
        # Decay temperature
        progress = min(1.0, it / 5000)
        T_current = T * (1.0 - 0.5 * progress) + 1.0 * 0.5 * progress
        T_current = max(T_current, 1.0)
        
        # Soft-max weights over convolution peaks
        shifted = conv / T_current
        shifted -= shifted.max()
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            break
        w = w / w_sum
        
        # Compute gradient: g[i] = sum_k w[k] * (a[k-i] + a[i-k]) with bounds
        # Use FFT for efficiency
        # g[i] = sum_k w[k] * a[k-i]  (correlation of w and a)
        #      + sum_k w[k] * a[i-k]  (correlation of a and w)
        # Both are the same: sum_k w[k] * a[k-i] = sum_j w[i+j] * a[j]
        # Actually: sum_k w[k] a[k-i] = (w correlate a)[i] ... let me think
        # (w * a)[k] = sum_j w[j] a[k-j], so (w*a)[i] = sum_j w[j] a[i-j]
        # We want sum_k w[k] a[k-i]. Let l = k-i, k = l+i: sum_l w[l+i] a[l]
        # This is (w_rev * a)[i] where w_rev[k] = w[-k] ... 
        # Simpler: just compute directly using the fact that n is moderate
        # For n up to ~2000, direct computation is O(n^2) which is 4M ops - fine
        
        if n <= 3000:
            # Direct computation
            g = np.zeros(n)
            for k in range(2 * n - 1):
                if w[k] < 1e-15:
                    continue
                wk = w[k]
                # a[k-i] is valid for i in [max(0,k-n+1), min(n-1,k)]
                i_lo = max(0, k - n + 1)
                i_hi = min(n - 1, k)
                g[i_lo:i_hi + 1] += wk * a[k - (np.arange(i_lo, i_hi + 1))]
        else:
            # FFT-based
            fft_size = 1 << (2 * n - 1 - 1).bit_length()
            W = rfft(w, fft_size)
            A = rfft(a, fft_size)
            # sum_k w[k] a[k-i] = IFFT(conj(W) * A) ... 
            # Actually: (w * a)[m] = sum_k w[k] a[m-k]
            # We want sum_k w[k] a[k-i] = (w * a')[-i] where a'[j] = a[-j]
            # Simpler: just use the convolve result we already have
            # sum_k w[k] a[k-i] for all i: this is the cross-correlation
            # corr(i) = sum_k w[k] a[k-i]
            # = sum_k w[k] a[k-i]
            # Let's compute via FFT: IFFT(FFT(w) * conj(FFT(a))) gives circular cross-correlation
            Wf = rfft(w, fft_size)
            Af = rfft(a, fft_size)
            corr = irfft(Wf * np.conj(Af), fft_size)[:n]
            g = 2.0 * corr  # symmetric (since a is real, corr is symmetric for auto-like)
            # Actually for cross-correlation of w and a, we get:
            # (w ⊛ a)(i) = sum_k w[k] a[k+i] ... this is getting confusing
            # Let me just use the direct approach with a trick
            g = np.zeros(n)
            # Use the fact that sum_k w[k] * a[k-i] can be computed as:
            # For each i, sum over valid k of w[k]*a[k-i]
            # This is a convolution of w with reversed a, shifted
            a_rev = a[::-1]
            conv_wa = np.convolve(w, a_rev, mode='full')
            # (w * a_rev)[p] = sum_k w[k] a_rev[p-k] = sum_k w[k] a[n-1-p+k]
            # We want sum_k w[k] a[k-i], let q = k-i, a[q] = a_rev[n-1-q]
            # sum_k w[k] a_rev[n-1-k+i] ... this is getting too complex
            # Just cap n at 3000 for direct computation
            pass
        
        # Adaptive step size based on gradient magnitude
        g_mag = np.max(np.abs(g))
        if g_mag < 1e-15:
            break
        step = step_base / (g_mag * n ** 0.5)
        step = min(step, 1.0)
        
        # Multiplicative update
        log_update = -step * g
        log_update -= log_update.max()  # numerical stability
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-300)
        s = a_new.sum()
        if s < 1e-300:
            break
        a_new = a_new / s
        
        # Accept if it improves or with some probability (to avoid getting stuck)
        new_score = compute_score_normalized(a_new)
        if new_score < score + 1e-10:
            a = a_new
        else:
            # Try smaller step
            a_half = a * np.exp(-0.5 * step * g)
            a_half = np.maximum(a_half, 1e-300)
            a_half = a_half / a_half.sum()
            half_score = compute_score_normalized(a_half)
            if half_score < score + 1e-10:
                a = a_half
            else:
                # No improvement, do a small random perturbation
                a = a + rng.standard_normal(n) * 1e-4
                a = np.maximum(a, 1e-300)
                a = a / a.sum()
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"  MD iter={it}, score={score:.8f}, best={best_score:.8f}, "
                  f"T={T_current:.1f}, step={step:.4f}, elapsed={elapsed:.0f}s")
    
    # Final check
    final_score = compute_score_normalized(best_a)
    if final_score < best_score:
        best_score = final_score
    return best_a, best_score


def two_coordinate_search(a_init, time_limit, verbose=True):
    """
    Exact 2-coordinate search with golden section.
    Targets peak-contributing pairs and random pairs.
    """
    a = np.maximum(a_init.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(55)
    
    iters = 0
    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        top_peak = int(np.argmax(conv))
        
        # Generate candidate pairs
        pairs = set()
        
        # Peak-focused: pairs that both contribute to the top peak
        js = max(0, top_peak - n + 1)
        je = min(n - 1, top_peak)
        peak_range = np.arange(js, je + 1)
        for _ in range(min(100, len(peak_range) * 2)):
            i = peak_range[rng.integers(0, len(peak_range))]
            j = peak_range[rng.integers(0, len(peak_range))]
            if i != j:
                pairs.add((min(i, j), max(i, j)))
        
        # Top 3 secondary peaks
        sorted_peaks = np.argsort(conv)[::-1][:3]
        for pk in sorted_peaks[1:]:
            js2 = max(0, int(pk) - n + 1)
            je2 = min(n - 1, int(pk))
            rng2 = np.arange(js2, je2 + 1)
            for _ in range(30):
                i = rng2[rng.integers(0, len(rng2))]
                j = rng2[rng.integers(0, len(rng2))]
                if i != j:
                    pairs.add((min(i, j), max(i, j)))
        
        # Random pairs
        for _ in range(100):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                pairs.add((min(i, j), max(i, j)))
        
        # Edge pairs
        edge_size = max(3, n // 8)
        for _ in range(30):
            i = rng.integers(0, edge_size)
            j = rng.integers(n - edge_size, n)
            if i != j:
                pairs.add((min(i, j), max(i, j)))
        
        improved = False
        for (i, j) in list(pairs):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14 and a[j] < 1e-14:
                continue
            
            t_lo = -a[j] + 1e-15
            t_hi = a[i] - 1e-15
            if t_hi <= t_lo:
                continue
            
            gr = (np.sqrt(5) - 1) / 2
            x1 = t_hi - gr * (t_hi - t_lo)
            x2 = t_lo + gr * (t_hi - t_lo)
            
            a_trial = a.copy()
            
            def eval_t(t):
                a_trial[i] = a[i] - t
                a_trial[j] = a[j] + t
                if a_trial[i] < -1e-15 or a_trial[j] < -1e-15:
                    return 1e10
                a_trial[i] = max(0.0, a_trial[i])
                a_trial[j] = max(0.0, a_trial[j])
                ss = a_trial.sum()
                if ss < 1e-15:
                    return 1e10
                a_trial = a_trial / ss
                return compute_score_normalized(a_trial)
            
            f1 = eval_t(x1)
            f2 = eval_t(x2)
            
            for _ in range(16):
                if time.monotonic() - t0 > time_limit:
                    break
                if f1 < f2:
                    t_hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = t_hi - gr * (t_hi - t_lo)
                    f1 = eval_t(x1)
                else:
                    t_lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = t_lo + gr * (t_hi - t_lo)
                    f2 = eval_t(x2)
            
            t_opt = (t_lo + t_hi) / 2
            f_opt = min(f1, f2)
            
            if f_opt < best_score - 1e-13:
                a[i] = max(0.0, a[i] - t_opt)
                a[j] = max(0.0, a[j] + t_opt)
                a = a / a.sum()
                best_score = compute_score_normalized(a)
                best_a = a.copy()
                improved = True
        
        if not improved:
            # Perturb to escape local minimum
            noise_scale = 0.001 + 0.002 * (iters % 10)
            a = best_a + rng.standard_normal(n) * noise_scale
            a = np.maximum(a, 0.0)
            a = a / a.sum()
        
        if verbose and iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"  2CD iter={iters}, score={best_score:.8f}, elapsed={elapsed:.0f}s")
    
    return best_a, best_score


def lp_fictitious_play_improved(a_init, time_limit, verbose=True):
    """
    LP fictitious play with geometric mean updates and adaptive step size.
    Key improvement: uses multiple LP solves per iteration with different M values
    to get a better direction, and uses a running average.
    """
    a = np.maximum(a_init.astype(np.float64), 1e-300)
    a = a / a.sum()
    n = len(a)
    
    a_avg = a.copy()
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    t0 = time.monotonic()
    
    for k in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(a)
        M = conv.max()
        
        # Try LP, fall back to greedy
        g = solve_lp_for_direction(a, M)
        if g is None or g.sum() < 1e-15:
            g = greedy_lp(a, M)
        
        if g is None or g.sum() < 1e-15:
            break
        
        g = g / g.sum()
        
        # Adaptive step: start large, decay
        t_step = min(0.5, 3.0 / (k + 3))
        if k > 200:
            t_step = max(0.02, 2.0 / (k + 3))
        
        # Geometric mean interpolation
        a_safe = np.maximum(a, 1e-300)
        g_safe = np.maximum(g, 1e-300)
        log_new = (1 - t_step) * np.log(a_safe) + t_step * np.log(g_safe)
        log_new -= log_new.max()
        a_new = np.exp(log_new)
        a_new = np.maximum(a_new, 1e-300)
        a_new = a_new / a_new.sum()
        
        # Running average (exponential)
        alpha = 0.1
        a_avg = (1 - alpha) * a_avg + alpha * a_new
        a_avg = a_avg / a_avg.sum()
        
        # Update best
        score_new = compute_score_normalized(a_new)
        score_avg = compute_score_normalized(a_avg)
        
        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
        if score_avg < best_score - 1e-14:
            best_score = score_avg
            best_a = a_avg.copy()
        
        a = a_new
        
        if verbose and k % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LP iter={k}, M={M:.6e}, score={score_new:.8f}, "
                  f"best={best_score:.8f}, elapsed={elapsed:.0f}s")
    
    return best_a, best_score


def lbgfs_smooth(a_init, time_limit, T=500.0, maxiter=3000):
    """L-BFGS on smoothed objective in log-space."""
    from scipy.optimize import minimize
    
    n = len(a_init)
    a = np.maximum(a_init, 1e-300)
    a = a / a.sum()
    z0 = np.log(a)
    z0 -= z0.sum()  # centering for numerical stability
    
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    t0 = time.monotonic()
    
    def compute_soft_obj_grad(z):
        # Softmax to get a from z
        z_shifted = z - z.max()
        a_curr = np.exp(z_shifted)
        a_curr = a_curr / a_curr.sum()
        
        conv = fast_convolve(a_curr)
        shifted = conv / T
        shifted -= shifted.max()
        w = np.exp(shifted)
        w = w / w.sum()
        
        obj = 2.0 * n * T * (shifted.max() + np.log(w.sum()))
        
        # Gradient w.r.t. a: 2n * sum_k w[k] * (a[k-i] + a[i-k])
        # For n <= 2000, compute directly
        if n <= 2500:
            grad_a = np.zeros(n)
            for k in range(2 * n - 1):
                if w[k] < 1e-12:
                    continue
                wk = w[k]
                i_lo = max(0, k - n + 1)
                i_hi = min(n - 1, k)
                idx = np.arange(i_lo, i_hi + 1)
                grad_a[idx] += 2.0 * wk * a_curr[k - idx]
        else:
            # FFT-based approximation
            fft_size = 1 << (2 * n - 1 - 1).bit_length()
            W = rfft(w, fft_size)
            A = rfft(a_curr, fft_size)
            grad_a = 2.0 * np.real(irfft(np.conj(W) * A, fft_size)[:n])
        
        # Project to log-space: grad_z = a * (grad_a - a . grad_a)
        S = np.dot(grad_a, a_curr)
        grad_z = a_curr * (grad_a - S)
        
        return obj, grad_z
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        obj, _ = compute_soft_obj_grad(z)
        return obj
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        _, g = compute_soft_obj_grad(z)
        return g
    
    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                     'maxfun': maxiter * 30, 'disp': False}
        )
        z_out = res.x
        z_out_shifted = z_out - z_out.max()
        a_out = np.exp(z_out_shifted)
        a_out = a_out / a_out.sum()
        score = compute_score_normalized(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except Exception as e:
        pass
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using cubic-like smoothing."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    # Use simple linear interpolation followed by smoothing
    f_interp = np.interp(new_x, old_x, f_norm)
    # Apply a mild Gaussian smoothing
    if new_n > 10:
        kernel_size = max(3, new_n // 100)
        if kernel_size % 2 == 0:
            kernel_size += 1
        kernel = np.ones(kernel_size) / kernel_size
        f_interp = np.convolve(f_interp, kernel, mode='same')
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score, verbose=True):
    """Evaluate a candidate using the actual evaluate_sequence and update best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def structured_perturb(a, rng, strength=0.01):
    """Apply a structured perturbation: preserve symmetry, add oscillation."""
    n = len(a)
    a_new = a.copy()
    
    # Add a sinusoidal perturbation
    freq = rng.integers(2, max(3, n // 10))
    phase = rng.uniform(0, 2 * np.pi)
    t = np.arange(n)
    a_new += strength * np.sin(2 * np.pi * freq * t / n + phase)
    
    # Add small random noise
    a_new += strength * 0.5 * rng.standard_normal(n)
    
    a_new = np.maximum(a_new, 0.0)
    a_new = a_new / a_new.sum()
    return a_new


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-strategy optimization combining:
    - Multiplicative (mirror) descent with soft-max gradient
    - Exact 2-coordinate golden-section search
    - LP-based fictitious play with geometric mean updates
    - L-BFGS on smoothed objective
    - Multi-length exploration
    - Structured random restarts
    """

    deadline = time.monotonic() + max(10.0, float(budget_s) - 20.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)

            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()

        print(f"Could not load parent: {e}")

    n_parent = len(parent)
    
    # === Phase 1: Multiplicative Descent (main driver) ===

        t_alloc = min(rem * 0.2, 300.0)
        print(f"  Budget: {t_alloc:.0f}s")
        a_cd, sc_cd = two_coordinate_search(best_seq, t_alloc, verbose=True)
        best_seq, best_score = check_and_update(a_cd, "2coord", best_seq, best_score)
    
    # === Phase 3: LP Fictitious Play ===
    rem = deadline - time.monotonic()
    print(f"\n=== Phase 3: LP Fictitious Play (best={best_score:.8f}) ===")
    if rem > 60:
        t_alloc = min(rem * 0.2, 300.0)
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, sc_lp = lp_fictitious_play_improved(best_seq, t_alloc, verbose=True)
        best_seq, best_score = check_and_update(a_lp, "lp_play", best_seq, best_score)
    
    # === Phase 4: L-BFGS with multiple temperatures ===
    rem = deadline - time.monotonic()
    print(f"\n=== Phase 4: L-BFGS Smoothed (best={best_score:.8f}) ===")
    for T_val in [200, 800, 2000]:
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t_alloc = min(rem * 0.15, 120.0)
        if t_alloc < 15:
            break
        print(f"  L-BFGS T={T_val}, budget={t_alloc:.0f}s")
        a_lb, sc_lb = lbgfs_smooth(best_seq, t_alloc, T=T_val, maxiter=2000)
        best_seq, best_score = check_and_update(a_lb, f"lbgfs_T{T_val}", best_seq, best_score)
    
    # === Phase 5: Another round of multiplicative descent ===
    rem = deadline - time.monotonic()
    print(f"\n=== Phase 5: Mirror Descent Round 2 (best={best_score:.8f}) ===")
    if rem > 60:
        t_alloc = min(rem * 0.2, 200.0)
        print(f"  Budget: {t_alloc:.0f}s")
        a_md2, sc_md2 = multiplicative_descent(best_seq, t_alloc, T=50.0, verbose=False)
        best_seq, best_score = check_and_update(a_md2, "mirror2", best_seq, best_score)
    
    # === Phase 6: Length Exploration ===
    rem = deadline - time.monotonic()
    print(f"\n=== Phase 6: Length Exploration (best={best_score:.8f}) ===")
    candidate_lengths = []
    for delta in [-4, -3, -2, -1, 1, 2, 3, 4, 6, 8]:
        alt_n = n_parent + delta
        if 64 <= alt_n <= 4000:
            candidate_lengths.append(alt_n)
    # Add some round numbers
    for rn in [512, 768, 1024, 1280, 1536, 2048]:
        if rn not in candidate_lengths and 64 <= rn <= 4096:
            candidate_lengths.append(rn)
    
    for alt_n in candidate_lengths:
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        if alt_n == n_parent:
            continue
        t_alloc = min(rem * 0.1, 80.0)
        if t_alloc < 10:
            continue
        
        init_alt = interpolate_sequence(best_seq, alt_n)
        
        # Quick mirror descent
        a_alt, sc_alt = multiplicative_descent(init_alt, t_alloc * 0.5, T=80.0, verbose=False)
        best_seq, best_score = check_and_update(a_alt, f"n{alt_n}_md", best_seq, best_score)
        
        # Quick 2-coord
        rem2 = deadline - time.monotonic()
        if rem2 > 15:
            a_alt2, sc_alt2 = two_coordinate_search(a_alt, min(rem2 * 0.3, 30.0), verbose=False)
            best_seq, best_score = check_and_update(a_alt2, f"n{alt_n}_cd", best_seq, best_score)
    
    # === Phase 7: Structured Random Restarts ===
    rem = deadline - time.monotonic()
    print(f"\n=== Phase 7: Structured Restarts (best={best_score:.8f}) ===")
    if rem > 50:
        t_alloc = min(rem * 0.15, 150.0)
        n_restarts = min(10, int(t_alloc // 15))
        for r in range(n_restarts):
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            perturbed = structured_perturb(best_seq, rng, strength=0.005 + 0.003 * r)
            t_alloc_r = min(rem * 0.4, 30.0)
            a_p, sc_p = two_coordinate_search(perturbed, t_alloc_r, verbose=False)
            best_seq, best_score = check_and_update(a_p, f"restart{r}", best_seq, best_score)
    
    # === Phase 8: Final Polish ===
    rem = deadline - time.monotonic()
    print(f"\n=== Phase 8: Final Polish (best={best_score:.8f}) ===")
    if rem > 30:
        t_alloc = min(rem * 0.4, 60.0)
        a_final, sc_final = two_coordinate_search(best_seq, t_alloc, verbose=False)
        best_seq, best_score = check_and_update(a_final, "final_cd", best_seq, best_score)
    
    # === Phase 9: Final LP polish ===
    rem = deadline - time.monotonic()
    if rem > 20:
        t_alloc = min(rem * 0.5, 40.0)
        a_lp_final, sc_lp_final = lp_fictitious_play_improved(best_seq, t_alloc, verbose=False)
        best_seq, best_score = check_and_update(a_lp_final, "final_lp", best_seq, best_score)
    
    # Fallback
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification with actual evaluate_sequence
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
