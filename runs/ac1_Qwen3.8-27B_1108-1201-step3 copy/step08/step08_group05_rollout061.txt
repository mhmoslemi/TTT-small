def project_simplex(v):
    """Euclidean projection onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]

        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_cross_convolve(f, g):
    size = len(f) + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def subgradient_step(f, n_peaks=1):
    """Compute the subgradient of max_k (f*f)[k] at f."""
    n = len(f)
    conv = fast_convolve(f)
    if n_peaks < len(conv):
        top = np.argpartition(conv, -n_peaks)[-n_peaks:]
    else:
        top = np.arange(len(conv))
    
    # Subgradient: average of 2*R_k*f for active k
    grad = np.zeros(n)
    for k in top:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            grad[j] += 2.0 * f[k - j]
    grad /= len(top)
    return grad, conv, top

def nesterov_subgradient(f_init, time_limit, lr=0.01, momentum=0.9, n_peaks=3):

    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    y = f.copy()  # lookahead point
    v = np.zeros(n)  # velocity
    best_f = f.copy()
    best_conv = fast_convolve(f).max()
    
    best_score_obj = best_conv  # we'll track the normalized peak
    
    it = 0

        # Lookahead
        grad_y, conv_y, top_y = subgradient_step(y, n_peaks)

            # Decay learning rate

"""
Step function optimization using linearized LP descent with
constraint-based peak minimization, multi-length exploration,
and Nesterov-accelerated subgradient polish.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(f, g):
    size = len(f) + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_min_M(f_0, max_iters=100000):
    """
    Solve: min M s.t. 2*(f_0*f)[k] - M <= conv_0[k] for all k,
           f >= 0, sum(f) = 1.
    Returns (f_new, M_val) or (None, inf) on failure.
    """
    n = len(f_0)
    size = 2 * n - 1
    conv_0 = fast_convolve(f_0)
    
    f_var = cp.Variable(n)
    M_var = cp.Variable()
    
    constraints = [f_var >= 0, cp.sum(f_var) == 1]
    
    # Build constraint: for each k, 2 * sum_m f_0[k-m] * f[m] - M <= conv_0[k]
    # Using the structure: (f_0 * f)[k] = sum_{m} f_0[k-m] f[m]
    # We can compute this as a matrix-vector product.
    
    # Build the constraint matrix efficiently
    # A[k, m] = 2 * f_0[k-m] if 0 <= k-m < n, else 0
    # A[k, n] = -1 (coefficient of M)
    
    # For efficiency, we'll build it row by row
    A_rows = []
    for k in range(size):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # Coefficients for f[lo..hi]: 2*f_0[k-hi], 2*f_0[k-hi+1], ..., 2*f_0[k-lo]
        # Wait: for m in [lo, hi], coeff = 2*f_0[k-m]
        # m=lo: 2*f_0[k-lo]
        # m=hi: 2*f_0[k-hi]
        # So the vector is [2*f_0[k-lo], 2*f_0[k-lo-1], ..., 2*f_0[k-hi]]
        # In numpy: 2 * f_0[k-hi : k-lo+1] ... no
        # m goes from lo to hi. k-m goes from k-lo down to k-hi.
        # So coeffs = [f_0[k-lo], f_0[k-lo-1], ..., f_0[k-hi]]
        # = f_0[k-hi : k-lo+1] reversed? No.
        # Let's just use: idx = np.arange(lo, hi+1); coeffs = 2.0 * f_0[k - idx]
        
        idx = np.arange(lo, hi + 1)
        coeffs = 2.0 * f_0[k - idx]
        
        # Constraint: coeffs @ f[lo:hi+1] - M <= conv_0[k]
        # In cvxpy: coeffs[0]*f[lo] + ... + coeffs[-1]*f[hi] - M <= conv_0[k]
        
        expr = 2.0 * sum(f_0[k - m] * f_var[m] for m in range(lo, hi + 1)) - M_var
        constraints.append(expr <= conv_0[k])
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(cp.PDLP, max_iters=max_iters, verbose=False)
    except Exception:
        try:
            prob.solve(cp.SCS, max_iters=max_iters, verbose=False)
        except Exception:
            return None, float('inf')
    
    if f_var.value is None:
        return None, float('inf')
    
    f_new = np.maximum(f_var.value, 0.0)
    s = f_new.sum()
    if s < 1e-10:
        return None, float('inf')
    f_new = f_new / s
    
    return f_new, M_var.value if M_var.value is not None else conv_0.max()

    """Efficient version using explicit matrix."""
    n = len(f_0)
    size = 2 * n - 1
    conv_0 = fast_convolve(f_0)
