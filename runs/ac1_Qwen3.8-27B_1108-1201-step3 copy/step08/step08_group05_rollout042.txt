    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]

def spectral_jump(f, flatten_alpha=0.5, percentile_level=0.90):
    n = len(f)
    N = 1 << (2 * n - 1).bit_length()
    A = np.fft.fft(f, N)
    # Current auto-conv
    b = np.fft.ifft(A * A).real[:2*n-1]
    # Flatten peaks
    threshold = np.percentile(b, percentile_level * 100)
    b_flat = np.minimum(b, threshold)
    # Extend b_flat to length N (zero-pad)
    b_flat_full = np.zeros(N)
    b_flat_full[:2*n-1] = b_flat
    # Get new spectrum
    B_flat = np.fft.fft(b_flat_full)

    # But B_flat might have negative or complex values, so we need to handle this

    # But it might not be, so let's use: mag = sqrt(re(B_flat)) if re(B_flat) > 0
    mag_new = np.sqrt(np.clip(B_flat.real, 0, None))

"""
Multi-strategy optimization for minimum peak auto-convolution.
Uses fast subgradient descent with momentum, spectral error reduction,
LP refinement, and diverse multi-start.

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def make_gaussian(n, sigma_frac):
    center = (n - 1) / 2.0
    sigma = sigma_frac * n / 2.0
    x = np.arange(n)
    a = np.exp(-(x - center)**2 / (2 * sigma**2))
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s

def make_flat_top(n):
    w = np.ones(n)
    a1, a2, a3, a4 = 0.21557895, 0.41663158, 0.277263158, 0.083578947
    x = np.arange(n)
    for i, ai in enumerate([a1, a2, a3, a4]):
        w -= ai * np.cos(2 * np.pi * i * x / max(1, n-1) - i * np.pi)
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s

def make_hann(n):
    w = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / max(1, n)))
    s = w.sum()
    return w / s

def make_blackman_harris(n):
    a = [0.35875, 0.48829, 0.14128, 0.01168]
    x = np.arange(n)
    w = np.zeros(n)
    for i, ai in enumerate(a):
        w += ai * np.cos(2 * np.pi * i * x / max(1, n-1) - i * np.pi)
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s

def compute_subgradient(f, n_peaks=20):
    n = len(f)
    b = fast_convolve(f)
    if n_peaks < len(b):
        top_idx = np.argpartition(b, -n_peaks)[-n_peaks:]
    else:
        top_idx = np.arange(len(b))
    b_top = b[top_idx]
    b_min = b_top.min()
    weights = np.maximum(b_top - b_min, 0.0)
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.ones(len(top_idx)) / len(top_idx)
    else:
        weights /= w_sum
    grad = np.zeros(n)
    for idx in range(len(top_idx)):
        k = int(top_idx[idx])
        w = 2.0 * weights[idx]
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        # f[k-i] for i in [lo, hi]
        # When i=lo: f[k-lo], when i=hi: f[k-hi]
        # So it's f[k-hi : k-lo+1] reversed
        grad[lo:hi+1] += w * f[k-hi:k-lo+1][::-1]
    obj = b.max()
    return grad, obj, b

    best_obj = fast_convolve(f).max()
    it = 0
    no_improve = 0

        for _ in range(batch):
            if time.monotonic() - t0 > time_limit:
                break
            it += 1
            grad, obj, b = compute_subgradient(f, n_peaks)
            v = momentum * v - lr * grad
            f_new = project_simplex(f + v)
            f = f_new
            
            if it % 50 == 0:
                cur_obj = fast_convolve(f).max()
                if cur_obj < best_obj - 1e-15:

def spectral_jump(f, percentile_level=0.85):
    n = len(f)
    N = 1 << (2 * n - 1).bit_length()
    A = np.fft.fft(f, N)
    b = np.fft.ifft(np.fft.fft(f, N) * np.fft.fft(f, N)).real[:2*n-1]
    threshold = np.percentile(b, percentile_level * 100)
    b_flat = np.minimum(b, threshold)
    b_flat_full = np.zeros(N)
    b_flat_full[:2*n-1] = b_flat
    B_flat = np.fft.fft(b_flat_full)
    mag_new = np.sqrt(np.clip(B_flat.real, 0, None))
    abs_A = np.abs(A)
    phase_A = np.where(abs_A > 1e-30, A / abs_A, 0)
    A_new = mag_new * phase_A
    # Enforce Hermitian
    A_new[N//2] = np.abs(A_new[N//2])
    half = N // 2
    for i in range(1, half):
        A_new[N - i] = np.conj(A_new[i])
    a_new = np.fft.ifft(A_new).real[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return f.copy()
    return a_new / s

def lp_refine(f_init, time_limit, top_k=50):
    """Simple LP-based refinement using scipy."""
    try:
        import cvxpy as cp
    except ImportError:
        return f_init, fast_convolve(f_init).max()
    
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(f)
        M = b.max()
        if M < 1e-15:
            break
        
        # Get top constraints
        if top_k < len(b):
            top_idx = np.argpartition(b, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(b))
        top_idx = np.sort(top_idx)
        
        # Build LP
        A_mat = np.zeros((len(top_idx), n))
        for row_i, k in enumerate(top_idx):
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            A_mat[row_i, lo:hi+1] = f[lo:hi+1]
        
        g_var = cp.Variable(n)
        prob = cp.Problem(cp.Maximize(cp.sum(g_var)), [g_var >= 0, A_mat @ g_var <= M])
        
        solved = False
        for solver in [cp.PDLP, cp.SCS]:
            try:
                prob.solve(solver=solver, max_iters=100000)
                solved = True
                break
            except Exception:
                continue
        
        if not solved or g_var.value is None:
            break
        g = np.maximum(g_var.value, 0.0)
        S_g = g.sum()
        if S_g <= 1.0 + 1e-5:
            break
        
        h = g / S_g
        # Line search
        best_t = 0.05
        best_trial = fast_convolve(h).max()
        for t in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            trial = (1-t)*f + t*h
            trial = np.maximum(trial, 0)
            s = trial.sum()
            if s > 1e-10:
                trial /= s
                obj = fast_convolve(trial).max()
                if obj < best_trial:
                    best_trial = obj
                    best_t = t
        
        f = (1 - best_t) * f + best_t * h
        f = np.maximum(f, 0)
        s = f.sum()
        if s > 1e-10:
            f /= s
        
        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
    
    return best_f, best_obj

"""
Multi-strategy optimization for minimum peak auto-convolution of non-negative sequences.
Combines fast subgradient descent with adaptive temperature, spectral error reduction,
LP-based refinement, diverse multi-start initializations, and parameter sweeps.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def make_gaussian(n, sigma_frac):
    center = (n - 1) / 2.0
    sigma = sigma_frac * n / 2.0
    if sigma < 1:
        sigma = 1.0
    x = np.arange(n, dtype=np.float64)
    a = np.exp(-(x - center) ** 2 / (2.0 * sigma ** 2))
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def make_flat_top(n):
    x = np.arange(n, dtype=np.float64)
    w = np.ones(n)
    coeffs = [0.21557895, 0.41663158, 0.277263158, 0.083578947]
    for i, ai in enumerate(coeffs):
        w -= ai * np.cos(2.0 * np.pi * i * x / max(1, n - 1) - i * np.pi)
    w = np.maximum(w, 0.0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def make_hann(n):
    w = 0.5 * (1.0 + np.cos(2.0 * np.pi * np.arange(n, dtype=np.float64) / max(1, n)))
    s = w.sum()
    return w / s


def make_blackman_harris(n):
    coeffs = [0.35875, 0.48829, 0.14128, 0.01168]
    x = np.arange(n, dtype=np.float64)
    w = np.zeros(n)
    for i, ai in enumerate(coeffs):
        w += ai * np.cos(2.0 * np.pi * i * x / max(1, n - 1) - i * np.pi)
    w = np.maximum(w, 0.0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def make_kaiser(n, beta):
    from scipy.signal import kaiser
    w = kaiser(n, beta)
    w = np.maximum(w, 0.0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def compute_soft_gradient(f, n_peaks, temperature):
    n = len(f)
    b = fast_convolve(f)
    if n_peaks < len(b):
        top_idx = np.argpartition(b, -n_peaks)[-n_peaks:]
    else:
        top_idx = np.arange(len(b))
    b_top = b[top_idx]
    scaled = b_top / max(temperature, 1e-10)
    scaled -= scaled.max()
    weights = np.exp(scaled)
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.ones(len(top_idx)) / len(top_idx)
    else:
        weights /= w_sum
    grad = np.zeros(n)
    for idx in range(len(top_idx)):
        k = int(top_idx[idx])
        w = 2.0 * weights[idx]
        if w < 1e-15:
            continue
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        grad[lo:hi + 1] += w * f[k - hi:k - lo + 1][::-1]
    return grad, b.max()


def spectral_jump(f, percentile_level=0.85):
    n = len(f)
    N = 1 << (2 * n - 1).bit_length()
    A = np.fft.fft(f, N)
    b = np.fft.ifft(A * A).real[:2 * n - 1]
    threshold = np.percentile(b, percentile_level * 100)
    b_flat = np.minimum(b, threshold)
    b_flat_full = np.zeros(N)
    b_flat_full[:2 * n - 1] = b_flat
    B_flat = np.fft.fft(b_flat_full)
    mag_new = np.sqrt(np.clip(B_flat.real, 0.0, None))
    abs_A = np.abs(A)
    phase_A = np.where(abs_A > 1e-30, A / abs_A, 0.0)
    A_new = mag_new * phase_A
    A_new[N // 2] = np.abs(A_new[N // 2])
    half = N // 2
    for i in range(1, half):
        A_new[N - i] = np.conj(A_new[i])
    a_new = np.fft.ifft(A_new).real[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return f.copy()
    return a_new / s


def subgradient_optimize(f_init, time_limit, lr_init=0.01, momentum=0.9,
                         n_peaks_init=20, n_peaks_max=80,
                         temp_init=0.01, temp_max=0.5, label="sg"):
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    v = np.zeros(n)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    lr = lr_init
    temp = temp_init
    n_peaks = n_peaks_init
    it = 0
    no_improve = 0
    stuck_count = 0

    while time.monotonic() - t0 < time_limit:
        if it % 200 == 0 and it > 0:
            temp = min(temp * 1.5, temp_max)
            n_peaks = min(n_peaks + 5, n_peaks_max)

        batch = min(50, max(10, int((time_limit - (time.monotonic() - t0)) * 0.4)))
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit:
                break
            it += 1
            grad, obj = compute_soft_gradient(f, n_peaks, temp)
            v = momentum * v - lr * grad
            f_new = project_simplex(f + v)
            f = f_new

        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
            stuck_count = 0
            lr = min(lr * 1.02, 0.05)
        else:
            no_improve += 1
            if no_improve >= 3:
                lr *= 0.93
                no_improve = 0
            if no_improve >= 8:
                stuck_count += 1
                no_improve = 0
                if stuck_count > 3:
                    f = spectral_jump(f, 0.80 + 0.1 * (stuck_count % 4))
                    v = np.zeros(n)
                    lr = lr_init * 0.7
                    stuck_count = 0

    return best_f, best_obj


def lp_refine(f_init, time_limit, top_k_init=50, top_k_max=200):
    try:
        import cvxpy as cp
    except ImportError:
        return f_init, fast_convolve(f_init).max()

    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    top_k = top_k_init
    it = 0
    stuck = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 30 == 0 and top_k < top_k_max:
            top_k = min(top_k + 10, top_k_max)

        b = fast_convolve(f)
        M = b.max()
        if M < 1e-15:
            break

        if top_k < len(b):
            top_idx = np.argpartition(b, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(b))
        top_idx = np.sort(top_idx)

        A_mat = np.zeros((len(top_idx), n))
        for row_i in range(len(top_idx)):
            k = int(top_idx[row_i])
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            A_mat[row_i, lo:hi + 1] = f[lo:hi + 1]

        g_var = cp.Variable(n)
        prob = cp.Problem(cp.Maximize(cp.sum(g_var)), [g_var >= 0, A_mat @ g_var <= M])

        solved = False
        for solver in [cp.PDLP, cp.SCS, cp.ECOS]:
            try:
                prob.solve(solver=solver, max_iters=200000)
                solved = True
                break
            except Exception:
                continue

        if not solved or g_var.value is None:
            break
        g = np.maximum(g_var.value, 0.0)
        if np.any(np.isnan(g)):
            break
        S_g = g.sum()
        if S_g <= 1.0 + 1e-5:
            stuck += 1
            if stuck > 10:
                break
            top_k = min(top_k * 2, top_k_max)
            continue

        h = g / S_g
        best_t = 0.05
        best_trial = best_obj
        for t in [0.01, 0.03, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            trial = (1.0 - t) * f + t * h
            trial = np.maximum(trial, 0.0)
            s = trial.sum()
            if s > 1e-10:
                trial /= s
                obj = fast_convolve(trial).max()
                if obj < best_trial:
                    best_trial = obj
                    best_t = t

        f_new = (1.0 - best_t) * f + best_t * h
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s > 1e-10:
            f_new /= s

        if best_trial < best_obj - 1e-14:
            best_obj = best_trial
            best_f = f_new.copy()
            f = f_new
            stuck = 0
        else:
            stuck += 1
            if stuck > 15:
                break

    return best_f, best_obj


def focused_descent(f_init, time_limit, n_peaks=5):
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(f)
        if n_peaks < len(b):
            top_idx = np.argpartition(b, -n_peaks)[-n_peaks:]
        else:
            top_idx = np.arange(len(b))

        grad = np.zeros(n)
        for k in top_idx:
            k = int(k)
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            grad[lo:hi + 1] += 2.0 * f[k - hi:k - lo + 1][::-1]

        if grad.max() < 1e-15:
            break

        best_alpha = None
        best_trial = best_obj
        for alpha in [0.001, 0.003, 0.005, 0.01, 0.02, 0.05, 0.1]:
            f_trial = project_simplex(f - alpha * grad)
            obj = fast_convolve(f_trial).max()
            if obj < best_trial - 1e-14:
                best_trial = obj
                best_alpha = alpha

        if best_alpha is None:

        f_new = project_simplex(f - best_alpha * grad)
        if best_trial < best_obj - 1e-14:
            best_obj = best_trial
            best_f = f_new.copy()
            f = f_new

    return best_f, best_obj

    if rem > 30:
        t_fd = min(rem * 0.1, 100.0)
        if t_fd > 10:
            print(f"\n[FD] t={t_fd:.0f}s")
            f_fd, _ = focused_descent(best_seq, time_limit=t_fd, n_peaks=8)
            s_fd = float(evaluate_sequence(f_fd.tolist()))
            if np.isfinite(s_fd) and s_fd < best_score:
                best_score = s_fd
                best_seq = f_fd.copy()
                print(f"  NEW BEST [FD]: {s_fd:.8f}")

    # Phase 4: Second SG with different params
    rem = deadline - time.monotonic()
    if rem > 40:
        t_sg2 = min(rem * 0.15, 150.0)
        if t_sg2 > 15:
            print(f"\n[SG2] t={t_sg2:.0f}s")
            f_sg2, _ = subgradient_optimize(best_seq, time_limit=t_sg2, lr_init=0.006,
                                             momentum=0.85, n_peaks_init=30,
                                             n_peaks_max=100, temp_init=0.05,
                                             temp_max=0.8, label="sg2")
            s_sg2 = float(evaluate_sequence(f_sg2.tolist()))
            if np.isfinite(s_sg2) and s_sg2 < best_score:
                best_score = s_sg2
                best_seq = f_sg2.copy()
                print(f"  NEW BEST [SG2]: {s_sg2:.8f}")

    # Phase 5: LP again with more constraints
    rem = deadline - time.monotonic()
    if rem > 50:
        t_lp2 = min(rem * 0.2, 200.0)
        if t_lp2 > 20:
            print(f"\n[LP2] t={t_lp2:.0f}s")
            f_lp2, _ = lp_refine(best_seq, time_limit=t_lp2, top_k_init=80, top_k_max=300)
            s_lp2 = float(evaluate_sequence(f_lp2.tolist()))
            if np.isfinite(s_lp2) and s_lp2 < best_score:
                best_score = s_lp2
                best_seq = f_lp2.copy()
                print(f"  NEW BEST [LP2]: {s_lp2:.8f}")

    # Phase 6: Try nearby lengths
    rem = deadline - time.monotonic()
    if rem > 60:
        n_cur = len(best_seq)
        for delta in [-96, 96, -192, 192, -256, 256]:
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            target_n = n_cur + delta
            if target_n < 384 or target_n > 3000:
                continue
            t_len = min(rem * 0.12, 80.0)
            if t_len < 20:
                break
            old_x = np.linspace(0.0, 1.0, n_cur)
            new_x = np.linspace(0.0, 1.0, target_n)
            f_interp = np.interp(new_x, old_x, best_seq)
            f_interp = np.maximum(f_interp, 0.0)
            s_int = f_interp.sum()
            if s_int < 1e-10:
                continue
            f_interp /= s_int
            print(f"\n[Len] n={target_n}, t={t_len:.0f}s")
            f_new, _ = subgradient_optimize(f_interp, time_limit=t_len, lr_init=0.01,
                                             momentum=0.9, n_peaks_init=20,
                                             n_peaks_max=60, temp_init=0.02,
                                             temp_max=0.4, label=f"n{target_n}")
            s_new = float(evaluate_sequence(f_new.tolist()))
            if np.isfinite(s_new) and s_new < best_score:
                best_score = s_new
                best_seq = f_new.copy()
                print(f"  NEW BEST [n={target_n}]: {s_new:.8f}")

    # Phase 7: Final polish
    rem = deadline - time.monotonic()
    if rem > 20:
        t_final = min(rem * 0.4, 60.0)
        if t_final > 10:
            f_fin, _ = focused_descent(best_seq, time_limit=t_final, n_peaks=10)
            s_fin = float(evaluate_sequence(f_fin.tolist()))
            if np.isfinite(s_fin) and s_fin < best_score:
                best_score = s_fin
                best_seq = f_fin.copy()
                print(f"  NEW BEST [final]: {s_fin:.8f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'=' * 60}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"Target: 1.5027 | Previous: 1.504858")
    print(f"{'=' * 60}")

    return [float(x) for x in best_seq]
