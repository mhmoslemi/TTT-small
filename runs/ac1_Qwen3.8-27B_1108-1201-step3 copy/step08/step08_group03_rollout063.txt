"""
Optimization of non-negative step function to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses smooth (log-sum-exp) projected gradient descent, LP-based active set refinement,
and multi-length exploration. Key innovation: smooth optimization finds a better basin
than pure LP, then LP with adaptive active sets refines the solution.

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_convolve(w, a):
    """Compute result[j] = sum_k w[k]*a[k-j] for j=0..n-1, using FFT.
    w has length 2n-1, a has length n."""
    n = len(a)
    size = 3 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    Ar = rfft(a[::-1], fft_size)
    conv = irfft(W * Ar, fft_size)
    return conv[n - 1: 2 * n - 1]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_optimize(f_init, time_limit, T_start=5000, T_end=500, n_iters_max=50000):
    """
    Projected gradient descent on logsumexp(T * conv(f,f))/T.
    Returns (best_f, best_max_conv).
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s > 1e-10:
        f /= s
    else:
        f = np.ones(n) / n

    best_f = f.copy()
    b = fast_convolve(f)
    best_obj = b.max()
    T = T_start
    total_iters = n_iters_max
    no_improve = 0

    for it in range(1, total_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        # Anneal temperature
        progress = min(1.0, (it - 1) / max(1, total_iters))
        T_cur = T_start * (T_end / T_start) ** progress

        # Compute convolution and softmax weights
        b = fast_convolve(f)
        Tb = T_cur * b
        Tb -= Tb.max()  # numerical stability
        exp_Tb = np.exp(Tb)
        w = exp_Tb / exp_Tb.sum()

        # Compute gradient via FFT: grad[j] = 2 * sum_k w[k] * a[k-j]
        # = 2 * cross_convolve(w, f)
        grad = 2.0 * cross_convolve(w, f)

        # Line search: backtracking
        f_best = f.copy()
        obj_best = best_obj
        step = 1.0
        for _ in range(20):
            f_trial = project_simplex(f - step * grad)
            b_trial = fast_convolve(f_trial)
            obj_trial = b_trial.max()
            if obj_trial < best_obj - 1e-14:
                f_best = f_trial
                obj_best = obj_trial
                break
            step *= 0.5
            if step < 1e-10:
                break

        f = f_best
        if obj_best < best_obj - 1e-14:
            best_obj = obj_best
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                # Boost: add small random perturbation
                rng = np.random.default_rng(it)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                no_improve = 0

            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_obj
            print(f"    smooth it={it} obj={best_obj:.10f} score={score:.6f} T={T_cur:.0f} t={elapsed:.0f}s")

def build_constraint_matrix(f, indices, n):
    """Build A where A[i, j] = f[indices[i] - j] if valid."""
    K = len(indices)
    A = np.zeros((K, n))
    for i, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[k - hi: k - lo + 1][::-1]
    return A


def solve_lp_min_max(f, n, active_idx):
    """
    Solve: min M s.t. g >= 0, sum(g) = 1, (f*g)[k] <= M for k in active_idx.
    Returns (g, M_opt).
    """
    K = len(active_idx)
    if K == 0:
        return None, None
    A = build_constraint_matrix(f, active_idx, n)
    g = cp.Variable(n)
    M_var = cp.Variable()
    constraints = [g >= 0, cp.sum(g) == 1, A @ g <= M_var]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=300000, eps=1e-9)
    except Exception:
        g.value = None
    if g.value is None:
        try:
            prob.solve(solver=cp.ECOS, max_iters=50000)
        except Exception:
            return None, None
    if g.value is None:
        return None, None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None, None
    S = g_val.sum()
    if S < 1e-10:
        return None, None
    g_val /= S
    return g_val, M_var.value


def lp_refine(f_init, time_limit, initial_k=50, max_k=400, label="lp"):
    """
    LP-based refinement with active set management.
    Alternates between solving the LP and updating the active set.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    b = fast_convolve(f)
    best_obj = b.max()

    # Initialize active set
    if initial_k < len(b):
        active_idx = np.argpartition(b, -initial_k)[-initial_k:]
    else:
        active_idx = np.arange(len(b))
    active_idx = np.sort(active_idx)

    it = 0
    stuck = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 3:
            break

        # Solve LP
        g, M_opt = solve_lp_min_max(f, n, active_idx)

        if g is None:
            stuck += 1
            if stuck > 10:
                break
            # Shrink active set
            if len(active_idx) > initial_k:
                b_cur = fast_convolve(f)
                active_idx = np.argpartition(b_cur, -initial_k)[-initial_k:]
            continue

        # Check improvement: what's the max of conv(g, g)?
        b_g = fast_convolve(g)
        obj_g = b_g.max()

        if obj_g < best_obj - 1e-13:
            best_obj = obj_g
            best_f = g.copy()
            stuck = 0
            # Update active set from g's convolution
            if max_k < len(b_g):
                new_top = np.argpartition(b_g, -min(max_k, len(b_g)))[-min(max_k, len(b_g)):]
            else:
                new_top = np.arange(len(b_g))
            active_idx = np.unique(np.concatenate([active_idx, new_top]))
            active_idx = np.sort(active_idx)
            if len(active_idx) > max_k:
                b_g_for_sort = fast_convolve(best_f)
                scores = b_g_for_sort[active_idx]
                keep = active_idx[np.argsort(scores)[-max_k:]]
                active_idx = np.sort(keep)
            f = g.copy()
        else:
            # Try interpolation
            for t in [0.5, 0.3, 0.2, 0.1]:
                f_trial = (1 - t) * f + t * g
                f_trial = np.maximum(f_trial, 0.0)
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj_trial = fast_convolve(f_trial).max()
                if obj_trial < best_obj - 1e-14:
                    best_obj = obj_trial
                    best_f = f_trial.copy()
                    f = f_trial.copy()
                    break
            else:
                stuck += 1
                if stuck > 30:
                    break
                # Refresh active set
                b_cur = fast_convolve(f)
                active_idx = np.argpartition(b_cur, -min(max_k, len(b_cur)))[-min(max_k, len(b_cur)):]
                active_idx = np.sort(active_idx)

        if it % 20 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_obj
            print(f"    {label} it={it} obj={best_obj:.10f} score={score:.6f} active={len(active_idx)} t={elapsed:.0f}s")

def cross_convolve(w, a):
    """Compute result[j] = sum_k w[k]*a[k-j] for j=0..n-1.
    w has length 2n-1, a has length n."""

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)

    return A

            # Try blending
            best_blend = None
            for t in [0.3, 0.5, 0.7]:
                f_blend = (1 - t) * f + t * g
                f_blend = np.maximum(f_blend, 0.0)
                sb = f_blend.sum()
                if sb < 1e-10:
                    continue
                f_blend /= sb
                ob = fast_convolve(f_blend).max()
                if ob < best_obj - 1e-14:
                    if best_blend is None or ob < best_blend[1]:
                        best_blend = (f_blend, ob)
            if best_blend is not None:
                f, best_obj = best_blend[0], best_blend[1]
                best_f = f.copy()
                b_cur = fast_convolve(f)
                tk = min(max_k, len(b_cur))
                active_idx = np.argpartition(b_cur, -tk)[-tk:]
                active_idx = np.sort(active_idx)
            else:
                stuck += 1
                if stuck > 40:
                    break
                # Refresh active set
                b_cur = fast_convolve(f)
                tk = min(max_k, len(b_cur))
                active_idx = np.argpartition(b_cur, -tk)[-tk:]
                active_idx = np.sort(active_idx)

        if it % 15 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_obj
            print(f"    {label} it={it} score={score:.6f} active={len(active_idx)} t={elapsed:.0f}s")

    return best_f, best_obj


def lp_refine_maxsum(f_init, time_limit, initial_k=80, max_k=500, label="lps"):
    """
    LP: max sum(g) s.t. g>=0, A@g <= M. Then rescale and blend.
    With cutting planes.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    b = fast_convolve(f)
    M = b.max()
    best_obj = M

    top_k = min(initial_k, 2 * n - 1)
    active_idx = np.argpartition(b, -top_k)[-top_k:]
    active_idx = np.sort(active_idx)

    it = 0
    stuck = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 3:
            break

        A = build_constraint_matrix_fast(f, active_idx, n)
        g_var = cp.Variable(n)
        constraints = [g_var >= 0, A @ g_var <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iters=500000, eps=1e-9)
        except Exception:
            g_var.value = None
        if g_var.value is None and n <= 3000:
            try:
                prob.solve(solver=cp.ECOS, max_iters=100000)
            except Exception:
                pass
        if g_var.value is None:
            stuck += 1
            if stuck > 15:
                break
            continue

        g = np.maximum(g_var.value, 0.0)
        if np.any(np.isnan(g)):
            stuck += 1
            if stuck > 15:
                break
            continue

        S_g = g.sum()
        if S_g <= 1.0 + 1e-6:
            stuck += 1
            if stuck > 25:
                break
            # Update active set
            b_cur = fast_convolve(f)
            tk = min(max_k, len(b_cur))
            active_idx = np.argpartition(b_cur, -tk)[-tk:]
            active_idx = np.sort(active_idx)
            continue

        stuck = 0
        h = g / S_g

        # Find violated constraints
        fft_size = 1 << (2 * n).bit_length()
        F = rfft(f, fft_size)
        H = rfft(h, fft_size)
        fg = irfft(F * H, fft_size)[:2 * n - 1]

        violated = fg > M * (1 + 1e-7)
        v_idx = np.where(violated)[0]
        not_active = v_idx[~np.isin(v_idx, active_idx)]
        if len(not_active) > 0 and len(active_idx) < max_k:
            n_add = min(25, len(not_active))
            to_add = not_active[np.argsort(fg[not_active])[-n_add:]]
            active_idx = np.unique(np.concatenate([active_idx, to_add]))
            active_idx = np.sort(active_idx)

        # Try blending
        best_t = 0.1
        best_t_obj = M
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            f_trial = (1.0 - t) * f + t * h
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t

        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s
        cur_obj = fast_convolve(f).max()
        M = cur_obj

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()

        # Periodically refresh
        if it % 12 == 0:
            b_cur = fast_convolve(best_f)
            tk = min(max_k, len(b_cur))
            new_top = np.argpartition(b_cur, -tk)[-tk:]
            old_keep = active_idx[np.argsort(b_cur[active_idx])[-initial_k:]] if len(active_idx) >= initial_k else active_idx
            active_idx = np.unique(np.concatenate([old_keep, new_top]))
            if len(active_idx) > max_k:
                active_idx = active_idx[np.argsort(b_cur[active_idx])[-max_k:]]
            active_idx = np.sort(active_idx)

        if it % 20 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_obj
            print(f"    {label} it={it} score={score:.6f} active={len(active_idx)} t={elapsed:.0f}s")

    return best_f, best_obj


def coordinate_descent(f_init, time_limit):
    """Coordinate descent with multiplicative updates."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    rng = np.random.default_rng(456)

    for sweep in range(20):
        if time.monotonic() - t0 > time_limit:
            break
        idx_order = rng.permutation(n)
        improved = False
        for j in idx_order:
            if time.monotonic() - t0 > time_limit:
                return f, best_obj
            best_val = f[j]
            best_obj_j = best_obj
            for mult in [0.75, 0.9, 1.1, 1.25]:
                nv = f[j] * mult
                if nv < 0:
                    nv = 0.0
                f[j] = nv
                s = f.sum()
                if s < 1e-10:
                    f[j] = best_val
                    s = f.sum()
                    if s > 1e-10:
                        f /= s
                    continue
                f /= s
                obj = fast_convolve(f).max()
                if obj < best_obj_j - 1e-14:
                    best_obj_j = obj
                    best_val = nv
            f[j] = best_val
            s = f.sum()
            if s > 1e-10:
                f /= s
            if best_obj_j < best_obj - 1e-14:
                best_obj = best_obj_j
                improved = True
        if not improved:
            break

    return f, best_obj


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search combining smooth optimization, LP refinement (two formulations),
    multi-start, multi-length exploration, and coordinate descent.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        parent = np.ones(1024) / 1024
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: uniform n=1024, score={p_score:.6f}")

    best_seq = parent.copy()
    best_score = p_score

    # === Phase 1: Smooth optimization ===
    print(f"\n=== Phase 1: Smooth opt (n={len(parent)}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.28, 450.0)
    if t1 > 60:
        print(f"  Budget: {t1:.0f}s")
        f1, _ = smooth_optimize(parent, time_limit=t1, T_start=10000, T_end=1000)
        score1 = float(evaluate_sequence(f1.tolist()))
        if np.isfinite(score1) and score1 < best_score:
            best_score = score1
            best_seq = f1.copy()
            print(f"  NEW BEST: {score1:.6f}")

    # === Phase 2: LP min-M formulation ===
    print(f"\n=== Phase 2: LP min-M (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.22, 350.0)
    if t2 > 60:
        n_b = len(best_seq)
        print(f"  Budget: {t2:.0f}s, n={n_b}")
        f2, _ = lp_refine_minmax(best_seq, time_limit=t2, initial_k=100, max_k=500, label="mm")

        if np.isfinite(score2) and score2 < best_score:
            best_score = score2
            best_seq = f2.copy()
            print(f"  NEW BEST: {score2:.6f}")

    # === Phase 3: LP max-sum with cutting planes ===
    print(f"\n=== Phase 3: LP max-sum (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.20, 300.0)
    if t3 > 60:
        n_b = len(best_seq)
        print(f"  Budget: {t3:.0f}s, n={n_b}")
        f3, _ = lp_refine_maxsum(best_seq, time_limit=t3, initial_k=100, max_k=600, label="ms")
        score3 = float(evaluate_sequence(f3.tolist()))
        if np.isfinite(score3) and score3 < best_score:
            best_score = score3
            best_seq = f3.copy()
            print(f"  NEW BEST: {score3:.6f}")

    # === Phase 4: Multi-start smooth ===
    print(f"\n=== Phase 4: Multi-start (score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    n_cur = len(best_seq)
    starts = []
    # Asymmetric
    x = np.linspace(0, 1, n_cur)
    fa = x**3 + 0.05*(1-x)**3
    fa = np.maximum(fa, 0)
    fa /= fa.sum()
    starts.append(fa)
    # Other asymmetric
    fb = (1-x)**3 + 0.05*x**3
    fb = np.maximum(fb, 0)
    fb /= fb.sum()
    starts.append(fb)
    # Random
    rng = np.random.default_rng(99)
    fr = rng.random(n_cur) + 0.01
    fr /= fr.sum()
    starts.append(fr)

    for i, fs in enumerate(starts):

        t4 = min(rem * 0.11, 100.0)

            best_seq = f4.copy()
            print(f"  NEW BEST: {score4:.6f}")
        # Quick LP polish

        f_int = np.interp(new_x, old_x, best_seq)
