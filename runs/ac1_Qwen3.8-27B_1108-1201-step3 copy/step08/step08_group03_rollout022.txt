"""
Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses spectral flattening (Gerchberg-Saxton style), L-BFGS with log-sum-exp
smoothing and annealing, swap-based local search for arrangement, and
multi-start diversification.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_objective(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def spectral_flatten(a_init, n_iters=2000, t_init=0.1, t_end=0.01, verbose=False):
    """
    GS-based spectral flattening.
    Iteratively:
    1. Compute FFT of a
    2. Adjust magnitude toward flat
    3. IFFT, clip to non-negative, normalize
    """
    n = len(a_init)
    fft_size = 1 << (2 * n - 2).bit_length()
    a = np.maximum(a_init, 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a /= s
    
    best_a = a.copy()
    best_obj = compute_objective(a)
    no_improve = 0
    
    for it in range(n_iters):
        # Anneal t
        frac = it / max(n_iters - 1, 1)
        t = t_init + (t_end - t_init) * frac
        # As t decreases, we flatten more aggressively
        # t controls how much we pull toward flat
        alpha = 1.0 - t  # alpha goes from (1-t_init) to (1-t_end), i.e., more aggressive
        
        A = rfft(a, fft_size)
        power = np.abs(A) ** 2
        
        # Flatten: new_power = power^alpha normalized
        # alpha < 1 compresses (flattens), alpha > 1 expands
        p_mean = power.mean()
        if p_mean < 1e-30:
            break
        power_norm = power / p_mean
        new_power = power_norm ** alpha
        # Preserve total power
        new_power = new_power * (power.sum() / (new_power.sum() + 1e-30))
        
        new_amp = np.sqrt(np.maximum(new_power, 0.0))
        phase = A / (np.abs(A) + 1e-30)
        A_new = new_amp * phase
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new /= s_new
        
        obj = compute_objective(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        if verbose and it % 200 == 0:
            print(f"      spectral it={it} obj={obj:.10f} best={best_obj:.10f} t={t:.4f}")
        
        if no_improve > 200:
            # Restart with different t
            t_end = max(0.005, t_end * 0.7)
            no_improve = 0
            a = best_a.copy()
    
    return best_a


def smoothed_objective_and_grad(w, t, n, fft_size):
    """
    Compute smoothed objective and gradient with respect to w (softmax parameterization).
    a = softmax(w)
    F_t = t * log(sum_k exp(b_k / t))
    """
    # Compute a from w
    w_shifted = w - w.max()
    a = np.exp(w_shifted)
    a_sum = a.sum()
    a = a / a_sum
    
    # Compute b = conv(a, a)
    A = rfft(a, fft_size)
    b = irfft(A * A, fft_size)[:2*n-1]
    
    # Softmax weights
    b_scaled = b / t
    b_scaled -= b_scaled.max()
    p = np.exp(b_scaled)
    p_sum = p.sum()
    p = p / p_sum
    
    # Objective
    F = t * (b.max() / t + np.log(p_sum))  # t * logsumexp(b/t)
    
    # Gradient of F w.r.t. a: g_a = 2 * cross_corr(p, a)
    # cross_corr(p, a)_i = sum_k p_k * a_{k-i} for valid indices
    # This is the "reversed convolution"
    # Compute via FFT: IFFT(conj(FFT(p, N)) * FFT(a, N))
    N2 = 1 << (3 * n - 3).bit_length()
    P = rfft(p, N2)
    Af = rfft(a, N2)
    # cross-correlation: sum_k p[k] * a[k-i]
    # = IFFT(conj(P) * Af) ... let me verify
    # Actually: circular cross-correlation R[i] = sum_k p[k] * conj(a[k-i])
    # For real signals: R[i] = sum_k p[k] * a[k-i]
    # FFT(R) = conj(FFT(p)) * FFT(a)
    # So R = IFFT(conj(P) * Af)
    cross = irfft(np.conj(P) * Af, N2)[:n]
    g_a = 2.0 * cross
    
    # Gradient w.r.t. w: df/dw = a * (g_a - a @ g_a)
    a_dot_g = np.dot(a, g_a)
    grad_w = a * (g_a - a_dot_g)
    
    return F, grad_w


def lbfgs_smooth_anneal(a_init, t_schedule, n_lbfgs_iters=100, verbose=False):
    """
    L-BFGS with log-sum-exp smoothing and annealing.
    """
    n = len(a_init)
    fft_size = 1 << (2 * n - 2).bit_length()
    N2 = 1 << (3 * n - 3).bit_length()
    
    a = np.maximum(a_init, 1e-15).copy()
    a /= a.sum()
    w = np.log(a)
    
    best_a = a.copy()
    best_obj = compute_objective(a)
    
    for t_idx, t in enumerate(t_schedule):
        # L-BFGS with current t
        def func(w):
            F, _ = smoothed_objective_and_grad(w, t, n, fft_size)
            return F
        
        def func_grad(w):
            F, g = smoothed_objective_and_grad(w, t, n, fft_size)
            return F, g
        
        try:
            res = minimize(func_grad, w, method='L-BFGS-B', 
                          options={'maxiter': n_lbfgs_iters, 'gtol': 1e-8})
        except Exception:
            res = minimize(func_grad, w, method='L-BFGS-B',
                          options={'maxiter': n_lbfgs_iters // 2})
        
        if res.x is not None:
            w = res.x
            a_new = np.exp(w - w.max())
            a_new /= a_new.sum()
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s > 1e-15:
                a_new /= s
            obj = compute_objective(a_new)
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a_new.copy()
            a = a_new
        
        if verbose:
            print(f"    L-BFGS t={t:.4f} obj={best_obj:.10f}")
    
    return best_a


def swap_local_search(a_init, n_sweeps=10, verbose=False):
    """
    Local search by swapping pairs of elements.
    Try to reduce the max of the auto-convolution by swapping elements.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0.0).copy()
    a /= a.sum()
    
    best_obj = compute_objective(a)
    b = fast_convolve(a)
    
    # Strategy: for the peak location k*, try swapping a_i with a_j 
    # where the swap reduces b[k*]
    # b[k*] changes by (a_j - a_i) * (a_{k*-i} - a_{k*-j})
    # To reduce b[k*], we want (a_j - a_i) * (a_{k*-i} - a_{k*-j}) < 0
    
    for sweep in range(n_sweeps):
        improved = False
        b = fast_convolve(a)
        peak_k = b.argmax()
        
        # Try random swaps
        rng = np.random.default_rng(sweep)
        n_tries = min(500, n * 2)
        
        for _ in range(n_tries):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            
            # Quick check: will this swap reduce the peak?
            # Delta at peak_k: (a[j]-a[i])*(a[peak_k-i] - a[peak_k-j]) (with boundary checks)
            ki = peak_k - i
            kj = peak_k - j
            val_i = a[ki] if 0 <= ki < n else 0.0
            val_j = a[kj] if 0 <= kj < n else 0.0
            delta = (a[j] - a[i]) * (val_i - val_j)
            
            if delta >= 0:  # Would increase the peak
                continue
            
            # Do the swap and check full objective
            a[i], a[j] = a[j], a[i]
            obj = compute_objective(a)
            if obj < best_obj - 1e-13:
                best_obj = obj
                b = fast_convolve(a)
                improved = True
            else:
                # Undo
                a[i], a[j] = a[j], a[i]
        
        if verbose:
            print(f"    swap sweep={sweep} obj={best_obj:.10f} improved={improved}")
        
        if not improved:
            break
    
    return a, best_obj

"""
Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses: (1) Gerchberg-Saxton spectral flattening with adaptive relaxation,
(2) L-BFGS with log-sum-exp smoothing and geometric annealing,
(3) swap-based local search for arrangement optimization,
(4) multi-start with diverse initializations including anti-symmetric arrangements.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_objective(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def spectral_flatten(a_init, n_iters=5000, t_init=0.3, t_end=0.02,
                     momentum=0.0, verbose=False, verbose_every=500):
    """
    GS-based spectral flattening with adaptive relaxation and momentum.
    Alternates between:
      - Frequency domain: flatten |FFT(a)|^2 toward uniform
      - Time domain: enforce non-negativity and normalization
    """
    n = len(a_init)
    fft_size = 1 << (2 * n - 2).bit_length()
    a = np.maximum(a_init, 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a /= s

    best_a = a.copy()
    best_obj = compute_objective(a)
    no_improve = 0
    a_mom = a.copy()

    for it in range(n_iters):
        frac = it / max(n_iters - 1, 1)
        t = t_init + (t_end - t_init) * frac
        alpha = 1.0 - t

        A = rfft(a, fft_size)
        power = np.abs(A) ** 2

        p_mean = power.mean()
        if p_mean < 1e-30:
            break

        power_norm = power / p_mean
        new_power = power_norm ** alpha
        new_power = new_power * (power.sum() / (new_power.sum() + 1e-30))

        new_amp = np.sqrt(np.maximum(new_power, 0.0))
        abs_A = np.abs(A)
        phase = A / (abs_A + 1e-30)
        A_new = new_amp * phase
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new /= s_new

        if momentum > 0:
            a_candidate = (1.0 - momentum) * a_new + momentum * a_mom
            a_candidate = np.maximum(a_candidate, 0.0)
            sc = a_candidate.sum()
            if sc > 1e-15:
                a_candidate /= sc
            a_mom = a_candidate
            a = a_candidate
        else:
            a = a_new

        obj = compute_objective(a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and it % verbose_every == 0:
            print(f"      spectral it={it} obj={obj:.10f} best={best_obj:.10f} t={t:.4f} no_imp={no_improve}")

        if no_improve > 500:
            no_improve = 0
            a = best_a.copy()
            t_end = max(0.005, t_end * 0.8)

    return best_a


def smoothed_grad(w, t, n, fft_size, N2):
    """Compute gradient of smoothed objective w.r.t. w (softmax param)."""
    w_shifted = w - w.max()
    a = np.exp(w_shifted)
    a_sum = a.sum()
    a = a / a_sum

    A = rfft(a, fft_size)
    b = irfft(A * A, fft_size)[:2 * n - 1]

    b_scaled = b / t
    b_scaled -= b_scaled.max()
    p = np.exp(b_scaled)
    p_sum = p.sum()
    p = p / p_sum

    P = rfft(p, N2)
    Af = rfft(a, N2)
    cross = irfft(np.conj(P) * Af, N2)[:n]
    g_a = 2.0 * cross

    a_dot_g = np.dot(a, g_a)
    grad_w = a * (g_a - a_dot_g)
    return grad_w


def lbfgs_stage(w_init, t, n, fft_size, N2, maxiter=80):
    """Run one L-BFGS stage with given smoothing parameter t."""
    def func_grad(w):
        g = smoothed_grad(w, t, n, fft_size, N2)
        w_shifted = w - w.max()
        a = np.exp(w_shifted)
        a_sum = a.sum()
        a = a / a_sum
        A = rfft(a, fft_size)
        b = irfft(A * A, fft_size)[:2 * n - 1]
        b_scaled = b / t
        b_scaled -= b_scaled.max()
        F = t * (b_scaled.max() + np.log(np.exp(b_scaled).sum()))
        return F, g

    try:
        res = minimize(func_grad, w_init, method='L-BFGS-B',
                       options={'maxiter': maxiter, 'gtol': 1e-7, 'maxfun': maxiter * 5})
        return res.x if res.x is not None else w_init
    except Exception:
        return w_init


def lbfgs_anneal(a_init, t_values, n, verbose=False):
    """L-BFGS with geometric annealing of smoothing parameter."""
    fft_size = 1 << (2 * n - 2).bit_length()
    N2 = 1 << (3 * n - 3).bit_length()

    a = np.maximum(a_init, 1e-15).copy()
    a /= a.sum()
    w = np.log(a)

    best_a = a.copy()
    best_obj = compute_objective(a)

    for i, t in enumerate(t_values):
        maxiter = max(30, 200 - 20 * i)
        w = lbfgs_stage(w, t, n, fft_size, N2, maxiter=maxiter)

        a_new = np.exp(w - w.max())
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-15:
            a_new /= s
        obj = compute_objective(a_new)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
        a = a_new

        if verbose:
            print(f"    L-BFGS t={t:.5f} obj={obj:.10f} best={best_obj:.10f}")

    return best_a


def swap_local_search(a_init, max_time=60.0, verbose=False):
    """Local search by element swaps to reduce peak auto-convolution."""
    t0 = time.monotonic()
    n = len(a_init)
    a = np.maximum(a_init, 0.0).copy()
    a /= a.sum()
    best_obj = compute_objective(a)
    rng = np.random.default_rng(12345)

    sweep = 0
    while time.monotonic() - t0 < max_time - 5:
        sweep += 1
        b = fast_convolve(a)
        peak_k = b.argmax()
        improved = False

        n_tries = min(2000, n * 4)
        for _ in range(n_tries):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            if a[i] == a[j]:
                continue

            ki = peak_k - i
            kj = peak_k - j
            val_i = a[ki] if 0 <= ki < n else 0.0
            val_j = a[kj] if 0 <= kj < n else 0.0
            delta = (a[j] - a[i]) * (val_i - val_j)

            if delta >= -1e-15:
                continue

            a[i], a[j] = a[j], a[i]
            obj = compute_objective(a)
            if obj < best_obj - 1e-13:
                best_obj = obj
                improved = True
                b = fast_convolve(a)
                peak_k = b.argmax()
            else:
                a[i], a[j] = a[j], a[i]

        if verbose and sweep % 5 == 0:
            print(f"    swap sweep={sweep} obj={best_obj:.10f}")

        if not improved:
            # Try different peak
            b = fast_convolve(a)
            top_peaks = np.argsort(b)[-5:]
            peak_k = top_peaks[rng.integers(0, len(top_peaks))]
            if sweep > 20:
                break

    return a, best_obj

                a_new = np.maximum(a_new, 0.0)
                s = a_new.sum()
                if s > 1e-15:
                    a_new /= s
                score = float(evaluate_sequence(a_new.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_seq = a_new.copy()
                    print(f"  NEW BEST [LBFGS t={t_val}]: {score:.8f}")
                w_init = w_res

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'='*55}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'='*55}")

    return [float(x) for x in best_seq]
