5. Multiple initialization strategies and length exploration
The key innovation is using the exact analytical gradient of the soft-max
smoothed auto-convolution, computed via FFT cross-correlation, combined
with natural gradient (mirror descent) updates that respect the simplex
geometry, plus a "reheating" schedule to escape local minima.
"""

import numpy as np
import time
from scipy.fft import fft, ifft, rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score_normalized(a):
    """Score for a sequence (assumed sum=1): 2*n*max(conv)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def smooth_max_and_gradient(a, T):
    """
    Compute soft_max(conv(a,a), T) and its gradient w.r.t. a.
    soft_max = T * log(sum_k exp(conv(k)/T))
    gradient(j) = 2 * sum_l w(l+j) * a(l) where w = softmax(conv/T)
    """
    n = len(a)
    conv = fast_convolve(a)
    shifted = conv / T
    m = shifted.max()
    w = np.exp(shifted - m)
    w = w / w.sum()
    soft_max_val = T * (m + np.log(w.sum()))

    # Cross-correlation via FFT: g(j) = 2 * sum_l w(j+l) * a(l)
    N = 1 << (3 * n).bit_length()
    W_pad = np.zeros(N)
    W_pad[:2 * n - 1] = w
    A_pad = np.zeros(N)
    A_pad[:n] = a

    W_fft = fft(W_pad)
    A_fft = fft(A_pad)
    # (W *conj A)^ = sum_k W(k) A(k-j) ... we want sum_l W(l+j) A(l)
    # This is cross-correlation: ifft(FFT(W) * conj(FFT(A)))
    corr = np.real(ifft(W_fft * np.conj(A_fft)))
    g = 2.0 * corr[:n]

    return soft_max_val, g, conv, w


def hard_max_gradient(a, top_k=3):
    """
    Gradient using top-k bins: w(k) = 1/top_k for top-k bins, 0 elsewhere.
    Returns (max_conv, gradient, conv, w).
    """
    n = len(a)
    conv = fast_convolve(a)
    top_idx = np.argsort(conv)[::-1][:top_k]
    w = np.zeros(len(conv))
    w[top_idx] = 1.0 / top_k
    max_val = conv[top_idx[0]]

    N = 1 << (3 * n).bit_length()
    W_pad = np.zeros(N)
    W_pad[:2 * n - 1] = w
    A_pad = np.zeros(N)
    A_pad[:n] = a

    W_fft = fft(W_pad)
    A_fft = fft(A_pad)
    corr = np.real(ifft(W_fft * np.conj(A_fft)))
    g = 2.0 * corr[:n]

    return max_val, g, conv, w


def natural_gradient_update(a, g, lr):
    """
    Multiplicative (natural gradient) update: a_new ∝ a * exp(-lr * g).
    Stable version with gradient normalization.
    """
    # Normalize gradient to have unit L2 norm for stable steps
    g_norm = np.linalg.norm(g)
    if g_norm < 1e-30:
        return a.copy()
    g_normalized = g / g_norm

    # Shift to prevent underflow
    g_shifted = g_normalized - g_normalized.max()
    update = np.exp(-lr * g_shifted)
    a_new = a * update
    s = a_new.sum()
    if s < 1e-300:
        return a.copy()
    a_new = a_new / s
    return a_new


def subgradient_annealing(a0, time_limit, verbose=True, seed=42):
    """
    Subgradient descent with simulated annealing (temperature schedule
    with reheating) and natural gradient updates.
    """
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score_normalized(a0)
    a = a / s

    best_a = a.copy()
    best_score = compute_score_normalized(a)
    rng = np.random.default_rng(seed)

    t0 = time.monotonic()
    lr = 0.5
    T = 1.0  # temperature for soft-max
    no_improve = 0
    max_no_improve = 200

    for it in range(500000):
        if time.monotonic() - t0 > time_limit:
            break

        # Temperature schedule: cool down, with reheating
        if it % 500 == 0:
            T = max(0.05, 1.0 * np.exp(-it / 3000.0))
            lr = max(0.01, 0.5 * np.exp(-it / 5000.0))

        if it % 100 == 0 and verbose:
            pass  # reduce print frequency

        # Compute gradient
        if T > 0.1:
            _, g, conv, _ = smooth_max_and_gradient(a, T)
        else:
            _, g, conv, _ = hard_max_gradient(a, top_k=min(5, n // 10 + 1))

        # Natural gradient update
        a_new = natural_gradient_update(a, g, lr)

        # Check score
        sc = compute_score_normalized(a_new)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Reheating: if stuck, add noise and increase temperature
        if no_improve >= max_no_improve:
            noise_scale = 0.01 * np.exp(-it / 10000.0)
            a = best_a + rng.standard_normal(n) * noise_scale
            a = np.maximum(a, 0)
            a = a / a.sum()
            T = min(1.0, T * 3.0)
            lr = min(1.0, lr * 2.0)
            no_improve = 0
            max_no_improve = max(50, max_no_improve // 2)

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SA iter={it}, T={T:.4f}, lr={lr:.4f}, "
                  f"score={best_score:.8f}, elapsed={elapsed:.0f}s")

    return best_a, best_score


def coordinate_descent_peak(a0, time_limit, verbose=True, seed=123):
    """
    Coordinate descent focusing on peak-contributing pairs.
    Uses exact golden-section search for optimal transfer.
    """
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()

    iters = 0
    while time.monotonic() - t0 < time_limit:
        iters += 1

        conv = fast_convolve(a)
        n_bins = len(conv)

        # Find top peaks
        top_k = min(5, n_bins)
        peak_indices = np.argsort(conv)[::-1][:top_k]

        improved = False
        pairs = set()

        # Pairs affecting top peaks
        for pk in peak_indices:
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            if len(idx) >= 2:
                # Sample pairs from contributing indices
                for _ in range(min(20, len(idx) * (len(idx) - 1) // 2)):
                    i = idx[rng.integers(0, len(idx))]
                    j = idx[rng.integers(0, len(idx))]
                    if i != j:
                        pairs.add((min(int(i), int(j)), max(int(i), int(j))))

        # Also add some random pairs for exploration
        for _ in range(30):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                pairs.add((min(int(i), int(j)), max(int(i), int(j))))

        gr = (np.sqrt(5) - 1) / 2

        for (i, j) in list(pairs):
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14 and a[j] < 1e-14:
                continue

            t_lo = -a[j] + 1e-15
            t_hi = a[i] - 1e-15
            if t_hi <= t_lo:
                continue

            x1 = t_hi - gr * (t_hi - t_lo)
            x2 = t_lo + gr * (t_hi - t_lo)

            a_trial = a.copy()

            def eval_score(t):
                a_trial[i] = a[i] - t
                a_trial[j] = a[j] + t
                if a_trial[i] < -1e-12 or a_trial[j] < -1e-12:
                    return 1e10
                a_trial[i] = max(0.0, a_trial[i])
                a_trial[j] = max(0.0, a_trial[j])
                ss = a_trial.sum()
                if ss < 1e-15:
                    return 1e10
                a_trial = a_trial / ss
                return compute_score_normalized(a_trial)

            f1 = eval_score(x1)
            f2 = eval_score(x2)

            for _ in range(15):
                if f1 < f2:
                    t_hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = t_hi - gr * (t_hi - t_lo)
                    f1 = eval_score(x1)
                else:
                    t_lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = t_lo + gr * (t_hi - t_lo)
                    f2 = eval_score(x2)

            t_opt = (t_lo + t_hi) / 2
            f_opt = min(f1, f2)

            if f_opt < best_score - 1e-13:
                a[i] = max(0.0, a[i] - t_opt)
                a[j] = max(0.0, a[j] + t_opt)
                ss = a.sum()
                a = a / ss
                best_score = compute_score_normalized(a)
                best_a = a.copy()
                improved = True

        if not improved and iters % 20 == 0:
            # Perturb to escape
            scale = 0.003 * np.exp(-iters / 50.0)
            a = best_a + rng.standard_normal(n) * scale
            a = np.maximum(a, 0)
            a = a / a.sum()

        if verbose and iters % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD iter={iters}, score={best_score:.8f}, elapsed={elapsed:.0f}s")

    return best_a, best_score


def lbgfs_smooth(a0, time_limit, T, verbose=False, seed=99):
    """L-BFGS on smooth soft-max objective in log-space."""
    n = len(a0)
    a_init = np.maximum(a0, 1e-30).astype(np.float64)
    a_init = a_init / a_init.sum()
    z0 = np.log(a_init)

    best_a = a_init.copy()
    best_score = compute_score_normalized(a_init)
    t0 = time.monotonic()

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a_tmp = np.exp(z - z.max())
        a_tmp = a_tmp / a_tmp.sum()
        obj, g, _, _ = smooth_max_and_gradient(a_tmp, T)
        return obj

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        a_tmp = np.exp(z - z.max())
        a_tmp = a_tmp / a_tmp.sum()
        _, g, _, _ = smooth_max_and_gradient(a_tmp, T)
        # Project to log-space tangent space
        S = np.dot(g, a_tmp)
        grad_z = a_tmp * (g - S)
        return grad_z

    try:
        res = minimize(fun=func, x0=z0, jac=jac, method='L-BFGS-B',
                       options={'maxiter': 3000, 'ftol': 1e-16, 'gtol': 1e-12,
                                'maxfun': 10000, 'disp': False})
        a_out = np.exp(res.x - res.x.max())
        a_out = a_out / a_out.sum()
        score = compute_score_normalized(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except Exception:
        pass

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score, verbose=True):
    """Evaluate candidate and return updated best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    try:
        score = float(evaluate_sequence(a_norm.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score

    Phase 5: Alternative lengths
    Phase 6: Final polish

    # === Phase 1: Subgradient Annealing ===
    rem = deadline - time.monotonic()

            best_score = fs
    except Exception:
        pass

    return [float(x) for x in best_seq]
