So `c2[j] = (w * a_rev)[n-1-j]`, meaning `c2 = reverse(w * a_rev)`.

Great! So:
`grad = (w * a) + reverse(w * a_rev)`

`c2_full[m] = sum_k w[k] * a_rev[m-k]` for `m = 0, ..., 3n-3`.

We want `c2[j] = c2_full[n-1-j]` for `j = 0, ..., n-1`.

So `c2[0] = c2_full[n-1]`, `c2[1] = c2_full[n-2]`, ..., `c2[n-1] = c2_full[0]`.

Now, for the overall algorithm, let me also think about the **log-sum-exp temperature scheduling**:

- Start with `τ = 10` (smooth approximation)
- Gradually increase to `τ = 100` or `τ = 1000` (sharper)
- At each τ, run several Adam steps
- This is like simulated annealing in the objective landscape

Actually, I realize there might be an issue. The log-sum-exp is always ≥ max, and as τ → ∞, it approaches max. So minimizing the log-sum-exp at high τ is close to minimizing the max. But at low τ, it's more like minimizing the "average" (geometric mean) of the convolution values, which is a different problem.

A better schedule might be:
- Start at moderate τ (e.g., 20-50) to get a good overall shape

"""
Optimization of step function coefficients to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses log-sum-exp smoothed objective with Adam optimizer, LP-based refinement,
frequencydomain insights, and coordinate descent for final polishing.

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time
import sys

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_smoothed_grad(a, tau):
    """
    Compute the smoothed objective value and its gradient.
    Objective: (1/tau) * log(sum(exp(tau * b[k]))) where b = conv(a,a)
    """
    n = len(a)
    b = fast_convolve(a)
    
    # Log-sum-exp for numerical stability
    max_b = b.max()
    scaled = tau * (b - max_b)
    log_sum_exp = max_b + np.log(np.sum(np.exp(scaled))) / tau
    obj = log_sum_exp
    
    # Softmax weights
    weights = np.exp(scaled)
    weights /= weights.sum()
    
    # Compute gradient: grad[j] = sum_m weights[m] * (a[j-m] + a[j+m])
    # = (w * a)[j] + (w * a_rev)[n-1-j]
    a_rev = a[::-1]
    conv_size = (2*n - 1) + n - 1  # 3n - 2
    N = 1 << (conv_size - 1).bit_length()
    W = rfft(weights, N)
    A = rfft(a, N)
    Arev = rfft(a_rev, N)
    c1 = irfft(W * A, N)[:n]
    c2_full = irfft(W * Arev, N)
    c2 = c2_full[:n][::-1]
    grad = c1 + c2
    
    return obj, grad

def adam_optimize(a_init, time_limit, tau_start=20.0, tau_end=200.0, 
                  lr=0.01, n_steps=None, label="adam", seed=42):
    """
    Adam optimizer with increasing tau (sharpening the max approximation).
    """
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy().astype(np.float64)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    best_score = 2.0 * n * best_obj
    
    no_improve = 0
    tau = tau_start
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        obj, grad = compute_smoothed_grad(a, tau)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project to non-negative and normalize
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = best_a.copy()
        else:
            a_new /= s
        
        a = a_new
        
        # Check improvement (every 10 steps to save computation)
        if t % 10 == 0:
            cur_b = fast_convolve(a).max()
            cur_score = 2.0 * n * cur_b
            if cur_score < best_score - 1e-10:
                best_score = cur_score
                best_obj = cur_b
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            # Increase tau periodically
            if no_improve > 50:
                tau = min(tau * 1.3, tau_end)
                no_improve = 0
                lr = max(lr * 0.9, 1e-5)
            
            if t % 200 == 0:
                elapsed = time.monotonic() - t0
                print(f"    {label} iter={t} tau={tau:.1f} score={best_score:.6f} "
                      f"lr={lr:.6f} t={elapsed:.0f}s")
        
        if no_improve > 300:
            break
    
    return best_a, best_score

def solve_lp_direction(f, n, peak_indices, M, eps=1e-8):
    """Solve LP for direction finding."""
    K = len(peak_indices)
    A = np.zeros((K, n))
    for row_i in range(K):
        k = int(peak_indices[row_i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[row_i, lo:hi + 1] = f[lo:hi + 1]
    
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    
    solved = False
    for solver in [cp.SCS, cp.ECOS, cp.CLARABEL]:
        try:
            prob.solve(solver=solver, max_iters=500000, eps=eps)
            solved = True
            break
        except Exception:
            continue
    
    if not solved or g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
        return None
    return g_val

def run_lp_optimize(f_init, time_limit, label="lp", initial_peak_frac=0.95,
                    final_peak_frac=0.85, max_peaks=300, restart_patience=20,
                    lp_eps=1e-8):
    """LP-based optimization with adaptive peak selection."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-10:
        f /= s
    best_f = f.copy()
    best_b = fast_convolve(f).max()
    it = 0
    stuck = 0
    peak_frac = initial_peak_frac
    rng = np.random.default_rng(2024)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()
        
        threshold = peak_frac * M
        peaks = np.where(b >= threshold)[0]
        if len(peaks) > max_peaks:
            peaks = np.argpartition(b, -max_peaks)[-max_peaks:]
        if len(peaks) < 5:
            peaks = np.argpartition(b, -max(10, len(b) // 10))[-max(10, len(b) // 10):]
        
        g = solve_lp_direction(f, n, peaks, M, eps=lp_eps)
        
        if g is None:
            stuck += 1
            if stuck > restart_patience:
                break
            continue
        
        S_g = g.sum()
        if S_g <= 1.0 + 1e-5:
            stuck += 1
            if stuck > restart_patience:
                break
            continue
        
        # Line search
        h = g / S_g
        best_t = 0.0
        best_t_obj = M
        
        for t_val in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            f_trial = (1.0 - t_val) * f + t_val * h
            f_trial = np.maximum(f_trial, 0.0)
            ss = f_trial.sum()
            if ss < 1e-10:
                continue
            f_trial /= ss
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj - 1e-15:
                best_t_obj = obj
                best_t = t_val
        
        if best_t <= 0:
            stuck += 1
            if stuck > restart_patience:
                break
            continue
        
        f = (1.0 - best_t) * f + best_t * h
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s
        
        cur_obj = fast_convolve(f).max()
        if cur_obj < best_b - 1e-14:
            best_b = cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
        
        if it % 20 == 0:
            score = 2.0 * n * best_b
            print(f"    {label} iter={it} obj={best_b:.10f} score={score:.6f} "
                  f"t={time.monotonic()-t0:.0f}s")
        
        if it % 30 == 0:
            peak_frac = max(final_peak_frac, peak_frac - 0.005)
        
        if stuck > restart_patience // 2 and stuck % 5 == 0:
            f = f + rng.standard_normal(n) * 0.001 * f
            f = np.maximum(f, 0.0)
            f /= f.sum()
    
    return best_f, best_b

def coordinate_descent(f_init, time_limit, label="cd"):
    """Coordinate descent for final polishing."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_b = fast_convolve(f).max()
    multipliers = np.array([0.7, 0.8, 0.9, 1.0, 1.1, 1.2, 1.3, 1.5])
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-15:
                continue
            best_j_val = f[j]
            best_j_obj = best_b
            for mult in multipliers:
                f_trial = f.copy()
                f_trial[j] = f[j] * mult
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-14:
                    best_j_obj = obj
                    best_j_val = f[j] * mult
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f /= s
                if best_j_obj < best_b - 1e-14:
                    best_b = best_j_obj
                    improved = True
        if not improved:
            break
    
    return f, best_b

def simulated_annealing(f_init, time_limit, label="sa", seed=123):
    """Simulated annealing for escaping local minima."""
    t0 = time.monotonic()
    n = len(f_init)
    a = f_init.copy()
    best_a = a.copy()
    best_b = fast_convolve(a).max()
    T = 0.01  # initial temperature
    rng = np.random.default_rng(seed)
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Random perturbation
        n_perturb = max(1, n // 20)
        idx = rng.choice(n, size=n_perturb, replace=False)
        a_new = a.copy()
        a_new[idx] *= np.exp(rng.standard_normal(n_perturb) * 0.3)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            continue
        a_new /= s
        
        b_new = fast_convolve(a_new).max()
        b_cur = fast_convolve(a).max()
        delta = b_new - b_cur
        
        if delta < 0 or rng.random() < np.exp(-delta / T):
            a = a_new
            if b_new < best_b - 1e-14:
                best_b = b_new
                best_a = a.copy()
        
        # Cool down
        if it % 50 == 0:
            T *= 0.95
            if it % 200 == 0:
                score = 2.0 * n * best_b
                print(f"    {label} iter={it} T={T:.6f} score={score:.6f} "
                      f"t={time.monotonic()-t0:.0f}s")
        
        if T < 1e-8:
            break
    
    return best_a, best_b

def frequency_init(n, seed=0):
    """Initialize using frequency domain: try to get flat power spectrum."""
    rng = np.random.default_rng(seed)
    # Start with random phases and flat magnitude
    N = 1 << (2 * n - 1).bit_length()
    # Random magnitude with some structure
    mags = np.ones(N // 2 + 1)
    # Add some randomness
    mags[1:] = 1.0 + 0.1 * rng.standard_normal(N // 2)
    mags = np.maximum(mags, 0.1)
    phases = rng.standard_normal(N // 2 + 1) * np.pi
    # Construct spectrum
    spec = mags * np.exp(1j * phases)
    # Handle Hermitian symmetry
    X = np.zeros(N, dtype=complex)
    X[:N // 2 + 1] = spec
    if N % 2 == 0:
        X[N // 2 + 1:] = np.conj(X[N // 2 - 1:0:-1])
    else:
        X[N // 2 + 1:] = np.conj(X[N // 2:0:-1])
    # Inverse FFT
    a_full = np.real(np.fft.ifft(X))
    a = a_full[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a /= s
    return a

def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def nonuniform_weighted_init(n, seed=0):
    """Create initial sequence with non-uniform weights based on decay patterns."""
    rng = np.random.default_rng(seed)
    # Try several patterns
    patterns = [
        np.ones(n),
        np.exp(-np.linspace(0, 3, n)),
        np.exp(-np.linspace(0, 0, n)) * np.cos(np.linspace(0, 4*np.pi, n))**2 + 0.1,
        np.ones(n) * (1.0 + 0.3 * np.sin(np.linspace(0, 2*np.pi * rng.integers(1,5), n))),
    ]
    a = patterns[rng.integers(len(patterns))].copy()
    a += rng.standard_normal(n) * 0.1
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    return a
