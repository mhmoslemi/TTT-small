def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_A_ub(a):
    """Build the (2n-1) x (n+1) constraint matrix for the LP."""
    n = len(a)
    A = np.zeros((2 * n - 1, n + 1))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            cols = np.arange(j_start, j_end + 1)
            A[k, cols] = a[k - cols]
        A[k, n] = -1.0
    return A

def solve_lp(a):
    """Solve min t s.t. conv(a,b)[k] <= t, sum(b)=1, b>=0."""
    n = len(a)
    A_ub = build_A_ub(a)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    b_ub = np.zeros(2 * n - 1)
    bounds = [(0, None)] * n + [(-np.inf, None)]  # t can be negative? No, it should be positive
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        b = np.maximum(res.x[:n], 0)
        s = b.sum()
        if s > 1e-15:
            b = b / s
        return b
    return None

def fictitious_play(a_init, time_budget, verbose=True):

            break

        a = a_new

        alpha = 1.0 / (it + 1)  # Actually for Polyak, it's cumulative
        a_avg = a_avg + (a - a_avg) / it  # This is just running average
        b_avg = b_avg + (b - b_avg) / it

            print(f"  FP it={it} ({elapsed:.0f}s): best={best_score:.6f}, LP_time={conv_time/it:.3f}s")
    
    return best_a, best_score, it

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast LP-based fictitious play (scipy HiGHS) with Polyak averaging
2. Subgradient-based optimization for rapid exploration
3. Fast coordinate descent with sensitivity analysis for local refinement

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_A_ub(a):
    """Build the (2n-1) x (n+1) constraint matrix for the LP."""
    n = len(a)
    A = np.zeros((2 * n - 1, n + 1))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            cols = np.arange(j_start, j_end + 1)
            A[k, cols] = a[k - cols]
        A[k, n] = -1.0
    return A


def solve_lp_fast(a):
    """Solve min t s.t. conv(a,b)[k] <= t for all k, sum(b)=1, b>=0."""
    n = len(a)
    A_ub = build_A_ub(a)
    c = np.zeros(n + 1)
    c[n] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    b_ub = np.zeros(2 * n - 1)
    bounds = [(0, None)] * n + [(-1e10, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            b = np.maximum(res.x[:n], 0)
            s = b.sum()
            if s > 1e-15:
                b = b / s
            return b
    except Exception:
        pass
    return None


def fictitious_play_lp(a_init, time_budget, verbose=True):
    """Fictitious play with fast LP solves and Polyak averaging."""
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    b = a.copy()

    a_avg = a.copy().astype(np.float64)
    b_avg = b.copy().astype(np.float64)

    conv_init = fast_convolve(a)
    best_score = 2.0 * n * float(conv_init.max())
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    total_lp_time = 0.0

    while time.monotonic() - t0 < time_budget:
        it += 1
        t_lp_start = time.monotonic()

        b_new = solve_lp_fast(a)
        a_new = solve_lp_fast(b)
        total_lp_time += time.monotonic() - t_lp_start

        if b_new is None:
            b_new = b
        if a_new is None:
            a_new = a

        b = b_new
        a = a_new

        # Polyak averaging (running average)
        a_avg += (a - a_avg) / it
        b_avg += (b - b_avg) / it

        # Check multiple candidates
        candidates = [a, b]
        if it >= 10:
            candidates.append(a_avg)
            candidates.append(b_avg)
        if it >= 50:
            candidates.append(0.5 * (a_avg + b_avg))

        for cand in candidates:
            cand = np.maximum(cand, 0)
            sc = cand.sum()
            if sc < 1e-10:
                continue
            cand_norm = cand / sc
            conv_c = fast_convolve(cand_norm)
            score_c = 2.0 * n * float(conv_c.max())
            if score_c < best_score - 1e-12:
                best_score = score_c
                best_a = cand_norm.copy()

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            avg_lp = total_lp_time / it * 1000
            print(f"    FP it={it} ({elapsed:.0f}s): best={best_score:.6f}, avg_LP={avg_lp:.0f}ms")

    return best_a, best_score, it


def subgradient_optimize(a_init, time_budget, verbose=True, label=""):
    """Subgradient method for the non-convex peak minimization."""
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    conv = fast_convolve(a)
    best_score = 2.0 * n * float(conv.max())

    t0 = time.monotonic()
    it = 0
    lr_init = 0.5
    lr = lr_init
    no_improve = 0
    top_k = min(10, 2 * n - 1)

    while time.monotonic() - t0 < time_budget:
        it += 1
        conv = fast_convolve(a)
        M = float(conv.max())
        score = 2.0 * n * M

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 500:
            lr *= 0.8
            no_improve = 0
            if lr < 1e-8:
                break

        # Find active (near-max) positions
        threshold = M - 0.05 * max(M, 1e-10)
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(conv))])
        # Weight by proximity to max
        weights = np.exp(-(M - conv[active]) / (0.02 * max(M, 1e-10)))
        weights /= weights.sum()

        # Compute subgradient
        grad = np.zeros(n)
        for idx in range(len(active)):
            k = int(active[idx])
            w = weights[idx]
            # grad[i] += 2*w*a[k-i] for valid i
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            grad[i_start:i_end+1] += 2.0 * w * a[k - np.arange(i_start, i_end + 1)]

        # Update with projection onto simplex
        lr_step = lr / (1.0 + 0.01 * it)
        a_new = a - lr_step * grad
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss > 1e-15:
            a = a_new / ss
        else:
            a = np.ones(n) / n

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG {label} it={it} ({elapsed:.0f}s): best={best_score:.6f} lr={lr:.6f}")

    return best_a, best_score


def coord_descent_fast(a_init, time_budget, n_trials=8, verbose=False, label=""):
    """Fast coordinate descent with sensitivity analysis."""
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    M = float(conv.max())
    best_score = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve_sweeps = 0

    while time.monotonic() - t0 < time_budget:
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)

            unaff_max = -1e30
            if k_start > 0:
                unaff_max = max(unaff_max, float(conv[:k_start].max()))
            if k_end < 2 * n - 2:
                unaff_max = max(unaff_max, float(conv[k_end + 1:].max()))

            affected = conv[k_start:k_end + 1]
            a_coupled = a  # a[k-i] for k in [i, i+n-1] is a[0] to a[n-1]

            best_delta = 0.0
            best_new_M = M

            for trial in range(n_trials):
                if trial == 0:
                    delta = -0.4 * a[i]
                elif trial == 1:
                    delta = -0.2 * a[i]
                elif trial == 2:
                    delta = -0.1 * a[i]
                elif trial == 3:
                    delta = -0.05 * a[i]
                elif trial == 4:
                    delta = 0.05 * a[i]
                elif trial == 5:
                    delta = 0.1 * a[i]
                elif trial == 6:
                    delta = 0.2 * a[i]
                else:
                    delta = 0.4 * a[i]

                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = affected + 2.0 * delta * a_coupled
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    new_affected = new_affected.copy()
                    new_affected[idx_2i] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta
                else:
                    new_affected = new_affected.copy()

                new_M = max(unaff_max, float(new_affected.max()))
                if new_M < best_new_M:
                    best_new_M = new_M
                    best_delta = delta

            if best_delta != 0.0 and best_new_M < M - 1e-16:
                a[i] += best_delta
                a[i] = max(a[i], 0.0)

                changes = 2.0 * best_delta * a_coupled
                if 2 * i >= k_start and 2 * i <= k_end:
                    idx_2i = 2 * i - k_start
                    changes = changes.copy()
                    changes[idx_2i] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps > 30:
                break
        else:
            no_improve_sweeps = 0

        if verbose and (it := 0) == 0 and no_improve_sweeps > 0:
            pass

    return best_a, best_score

def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def perturb_escape(a, rng, magnitude=0.05):
    """Perturb to escape local optimum."""
    n = len(a)
    a_new = a.copy()
    conv = fast_convolve(a)
    top_k = min(5, len(conv))
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    for k in peaks:
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1)
        n_pick = max(1, len(idx) // 5)
        picks = rng.choice(idx, size=min(n_pick, len(idx)), replace=False)
        for i in picks:
            a_new[i] *= (1.0 + rng.standard_normal() * magnitude)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    return a_new / s


def init_fourier(n, rng):
    """Initialize using random Fourier superposition."""
    n_harmonics = int(rng.integers(4, 20))
    amplitudes = rng.random(n_harmonics) * 0.3
    phases = rng.uniform(0, 2 * np.pi, n_harmonics)
    freqs = np.arange(1, n_harmonics + 1)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harmonics):
        a += amplitudes[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def init_skewed(n, rng):
    """Initialize with a skewed (asymmetric) shape."""
    t = np.linspace(0, 1, n)
    alpha = rng.uniform(1.5, 4.0)
    beta = rng.uniform(1.5, 4.0)
    a = t ** alpha * (1 - t) ** beta
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def damped_lp_refine(a_init, time_budget, verbose=False):
    """Damped LP refinement: alternate LP solve with blending."""
    n = len(a_init)
    a = np.maximum(a_init, 0).astype(np.float64).copy()

        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t_fine = min(rem * 0.5, 120.0)
            print(f"  FineCD budget={t_fine:.0f}s")
            a_fine, score_fine = fine_coord_descent(best_seq, t_fine, label="fine")
            if score_fine < best_score:
                best_score = score_fine
                best_seq = a_fine.copy()
                print(f"  NEW BEST (FineCD): {best_score:.6f}")

        rem = deadline - time.monotonic() - 30
        if rem > 10:
            t_cd2 = min(rem * 0.7, 60.0)
            a_cd2, score_cd2 = coord_descent_fast(best_seq, t_cd2, n_trials=16, label="polish")
            if score_cd2 < best_score:
                best_score = score_cd2
                best_seq = a_cd2.copy()
                print(f"  NEW BEST (polish): {best_score:.6f}")

    # === Phase 6: Try alternative length ===
    print(f"\n=== Phase 6: Alt Length (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 40:
            n_best = len(best_seq)
            # Try nearby "nice" lengths
            alt_options = []
            for base in [64, 128, 192, 256]:
                for mult in [1, 2, 3, 4, 5, 6, 7, 8]:
                    candidate_n = base * mult
                    if 512 <= candidate_n <= 2560 and candidate_n != n_best:
                        alt_options.append(candidate_n)
            alt_options = sorted(set(alt_options))[:3]

            for alt_n in alt_options:
                rem = deadline - time.monotonic() - 20
                if rem < 30:
                    break
                t_alt = min(rem * 0.5, 80.0)
                init_alt = interpolate_seq(best_seq, alt_n)
                a_alt, score_alt = coord_descent_fast(init_alt, t_alt, n_trials=10, label=f"alt{alt_n}")
                if score_alt < best_score:
                    best_score = score_alt
                    best_seq = a_alt.copy()
                    print(f"  NEW BEST (alt_n={alt_n}): {best_score:.6f}")

    # === Phase 7: Final intensive LP + CD ===
    print(f"\n=== Phase 7: Final (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 20:
            t_final = min(rem * 0.6, 60.0)
            a_fin, score_fin, _ = fictitious_play_lp(best_seq, t_final, verbose=False)
            if score_fin < best_score:
                best_score = score_fin
                best_seq = a_fin.copy()
                print(f"  NEW BEST (final FP): {best_score:.6f}")

        rem = deadline - time.monotonic() - 5
        if rem > 10:
            a_fin2, score_fin2 = fine_coord_descent(best_seq, rem * 0.8, label="last")
            if score_fin2 < best_score:
                best_score = score_fin2
                best_seq = a_fin2.copy()
                print(f"  NEW BEST (final fine): {best_score:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(1024) / 1024
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    best_seq = np.maximum(best_seq, 0.0)
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
