import numpy as np
import cvxpy as cp

from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

    best_score = 2.0 * n * fast_convolve(softmax_from_z(z)).max()

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        current_iter_iter = iters_per_stage
        if stage_T > 50.0:
            current_iter_iter = max(500, iters_per_stage // 2)
        elif stage_T < 1.0:
            current_iter_iter = iters_per_stage * 2

        def func_T(z_inner):
            a_inner = softmax_from_z(z_inner)
            b_inner = fast_convolve(a_inner)
            Tb = stage_T * b_inner
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / stage_T) * log_sum_exp

        def jac_T(z_inner):
            a_inner = softmax_from_z(z_inner)
            b_inner = fast_convolve(a_inner)
            Tb = stage_T * b_inner
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b_inner)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a_inner[k - idx]
            S = np.dot(grad_a, a_inner)
            return a_inner * (grad_a - S)

        try:
            res = minimize(
                fun=func_T,
                x0=z,
                jac=jac_T,
                method='L-BFGS-B',
                options={'maxiter': current_iter_iter, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': current_iter_iter * 150, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

        a_check = softmax_from_z(z)
        score_check = 2.0 * n * fast_convolve(a_check).max()
        if score_check < best_score:
            best_score = score_check
            best_z = z.copy()

    return softmax_from_z(best_z)


def nesterov_lp_refine(a0, time_limit, solver, n_max_iters=100000):
    """Nesterov dual averaging LP refinement with adaptive step and convergence detection."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    prev_best = best_score
    stall_count = 0
    consecutive_no_improve = 0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        improved = False
        if score_a < best_score - 1e-12:
            best_score = score_a
            best_a = a.copy()
            improved = True
        if score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()
            improved = True

        if improved:
            consecutive_no_improve = 0
        else:
            consecutive_no_improve += 1

        if it % 500 == 0:
            if best_score < prev_best - 1e-8:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score

            if stall_count >= 8:
                noise = np.random.default_rng(it * 7 + 3).random(n) * 0.02
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0
                consecutive_no_improve = 0

            elapsed = time.monotonic() - t0
            if it % 2000 == 0:
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")

        if consecutive_no_improve > 5000:
            break

    return best_a, best_score


def subgradient_flow(a0, time_limit, n_steps=200000, lr_init=0.01):
    """Subgradient flow with Nesterov momentum on the probability simplex."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    v = np.zeros(n)
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = lr_init
    beta = 0.9
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b - 0.02 * max_b
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        active = active[:20]

        g = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                g[j] += a[k - j]
        g /= len(active)

        y_new = a - lr * g
        beta = min(beta * 0.9999, 0.95)
        v = beta * v + (1 - beta) * (a - y_new)
        a_new = y_new + beta * v
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        score_new = 2.0 * n * fast_convolve(a_new).max()
        if score_new < best_score - 1e-14:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            a = a_new
            no_improve += 1

        if no_improve >= 500:
            lr *= 0.9
            no_improve = 0
        lr = max(lr_init / (1.0 + step * 0.001), 1e-10)

    return best_a, best_score


def peak_shave_local(a0, time_limit, n_steps=30000):
    """Multi-peak shaving with adaptive coordinate adjustments and pair operations."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        b_work = b.copy()
        if len(b_work) > 2:
            b_work[0] = -np.inf
            b_work[-1] = -np.inf
        peak_k = int(np.argmax(b_work))
        peak_val = b_work[peak_k]
        near_peak_threshold = peak_val - 0.05 * peak_val
        near_peaks = np.where(b_work >= near_peak_threshold)[0]

        contrib_count = np.zeros(n)
        for k in near_peaks[:8]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[k - idx]
            top_idx = np.argsort(contribs)[::-1][:6]
            for ci in top_idx:
                contrib_count[idx[ci]] += 1

        top_coords = np.argsort(contrib_count)[::-1][:12]

        improved = False
        for i in top_coords:
            if a[i] < 1e-14:
                continue
            for factor in [0.99, 0.97, 0.95, 0.92, 0.88]:
                new_val = a[i] * factor
                diff = a[i] - new_val
                if diff < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] = others + diff * others / others_sum
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_contrib = np.argsort(contribs)[::-1][:6]
            for ci in top_contrib:
                i = idx[ci]
                j = peak_k - i
                if i == j or i >= n or j >= n:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.97, 0.94, 0.90]:
                    for f2 in [0.97, 0.94, 0.90]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-15:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            a_trial = a.copy()
            for i in top_coords[:4]:
                if a[i] > 1e-14:
                    a_trial[i] *= 0.98
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-15:
                a = a_trial
                best_score = score
                best_a = a.copy()
            else:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Create structured initial sequence."""
    choice = rng.integers(0, 5)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * x * (1 - x))
    else:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * np.linspace(0, 1, n))
        a = a ** 1.5
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization: LP-based Nesterov alternating, L-BFGS LogSumExp
    with fine temperature schedule, subgradient flow, peak shaving, and
    multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 960
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.8f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(a)})")

    # === Phase 1: Nesterov LP on initial length ===
    print(f"\n=== Phase 1: Nesterov LP n={n_best} (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.28, 450.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=150000)
        try_update(a_lp, "nesterov_lp_1")

    # === Phase 2: L-BFGS with fine temperature schedule ===
    print(f"\n=== Phase 2: L-BFGS fine schedule (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.35, 600.0)
    if t_alloc > 60:
        print(f"  L-BFGS temperatures: 200->0.05, budget={t_alloc:.0f}s")
        temps_fine = [200.0, 100.0, 50.0, 25.0, 12.0, 6.0, 3.0, 1.5, 0.75,
                      0.4, 0.2, 0.1, 0.05]
        a_lb = lbgfs_multistage(best_seq, t_alloc, temps=temps_fine, iters_per_stage=3000)
        try_update(a_lb, "lbgfs_fine")

    # === Phase 3: Subgradient flow ===
    print(f"\n=== Phase 3: Subgradient Flow (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 20:
        a_sg, _ = subgradient_flow(best_seq, t_alloc, n_steps=300000, lr_init=0.005)
        try_update(a_sg, "subgrad_flow")

    # === Phase 4: Peak shaving ===
    print(f"\n=== Phase 4: Peak Shaving (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 100.0)
    if t_alloc > 15:
        a_ps, _ = peak_shave_local(best_seq, t_alloc, n_steps=50000)
        try_update(a_ps, "peak_shave")

    # === Phase 5: Explore alternative lengths ===
    print(f"\n=== Phase 5: Alternative Lengths (best={best_score:.8f}) ===")
    alt_lengths = []
    for target_n in [1280, 1024, 1152]:
        if target_n != n_best and 512 <= target_n <= 2048:
            alt_lengths.append(target_n)
    alt_lengths = sorted(set(alt_lengths))

    for alt_n in alt_lengths[:2]:
        rem = deadline - time.monotonic()
        if rem < 120:
            break
        t_total = min(rem * 0.30, 350.0)
        if t_total < 60:
            break
        print(f"  n={alt_n}, total budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_seq, alt_n)

        # LP phase
        t_lp = t_total * 0.50
        a_alt, _ = nesterov_lp_refine(init_alt, t_lp, solver=solver, n_max_iters=40000)
        try_update(a_alt, f"lp_n{alt_n}")

        # L-BFGS phase
        rem = deadline - time.monotonic()
        if rem > 40 and best_seq is not None and len(best_seq) == alt_n:
            t_lb = min(rem * 0.40, 150.0)
            if t_lb > 20:
                temps_alt = [50.0, 25.0, 12.0, 6.0, 3.0, 1.5, 0.75, 0.4, 0.2, 0.1]
                a_alt2 = lbgfs_multistage(best_seq, t_lb, temps=temps_alt, iters_per_stage=2000)
                try_update(a_alt2, f"lbgfs_n{alt_n}")

    # === Phase 6: Perturbation + re-optimization ===
    print(f"\n=== Phase 6: Perturbation Escape (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        t_pert = min(rem * 0.30, 200.0)
        n_cur = len(best_seq)

        # Try two different perturbations
        for pert_idx, noise_scale in enumerate([0.03, 0.08]):
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            t_this = min(rem * 0.45, 100.0)
            if t_this < 20:
                break

            perturbed = best_seq + noise_scale * rng.standard_normal(n_cur)
            perturbed = np.maximum(perturbed, 0)

            t_lp_pert = t_this * 0.55
            a_p, _ = nesterov_lp_refine(perturbed, t_lp_pert, solver=solver, n_max_iters=15000)
            try_update(a_p, f"pert_lp_{pert_idx}")

            rem = deadline - time.monotonic()
            if rem > 20 and best_seq is not None and len(best_seq) == n_cur:
                t_lb_pert = min(rem * 0.35, 60.0)
                if t_lb_pert > 10:
                    temps_pert = [10.0, 5.0, 2.5, 1.0, 0.5, 0.2, 0.1]
                    a_p2 = lbgfs_multistage(best_seq, t_lb_pert, temps=temps_pert, iters_per_stage=1500)
                    try_update(a_p2, f"pert_lbgfs_{pert_idx}")

    # === Phase 7: Final polishing ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 40:
        t_fin = min(rem * 0.35, 100.0)
        temps_final = [5.0, 2.5, 1.0, 0.5, 0.25, 0.1, 0.05, 0.02]
        a_fin = lbgfs_multistage(best_seq, t_fin, temps=temps_final, iters_per_stage=2000)
        try_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 25:
        t_ps = min(rem * 0.30, 50.0)
        a_ps2, _ = peak_shave_local(best_seq, t_ps, n_steps=30000)
        try_update(a_ps2, "final_ps")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_lp_fin = min(rem * 0.40, 40.0)
        a_lp_fin, _ = nesterov_lp_refine(best_seq, t_lp_fin, solver=solver, n_max_iters=8000)
        try_update(a_lp_fin, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    elapsed_total = 1800.0 - (deadline - time.monotonic())
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}) after {elapsed_total:.0f}s")
    return [float(x) for x in best_seq]
