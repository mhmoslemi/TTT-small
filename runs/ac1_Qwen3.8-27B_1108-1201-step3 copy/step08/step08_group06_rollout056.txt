"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Frank-Wolfe with smooth max approximation (LoG smoother)
2. Gradient descent with backtracking line search in log-space
3. Coordinate descent with golden-section search

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score_normalized(a):
    """Compute 2*n*max_conv for normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def simplex_project(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    # Standard simplex projection
    mu = np.sort(v)[::-1]
    cv = np.cumsum(mu)
    rho = np.nonzero(mu * np.arange(1, n + 1) > (cv - 1))[0][-1]
    theta = (cv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def smooth_max_and_grad(c, T):
    """Compute T*log(sum(exp(c/T))) and its gradient w.r.t. c."""
    c_shifted = c / T
    c_max = c_shifted.max()
    exp_c = np.exp(c_shifted - c_max)
    S = exp_c.sum()
    w = exp_c / S
    val = T * (c_max + np.log(S))
    return val, w


def compute_gradient(a, w):
    """Compute gradient of smooth max w.r.t. a.
    grad_j = 2 * sum_m w[j+m] * a[m] for m=0..n-1
    """
    n = len(a)
    return 2.0 * np.correlate(w, a, mode='valid')


def frank_wolfe(a_init, T, time_limit, verbose=False):
    """Frank-Wolfe method with smooth max approximation."""
    a = a_init.copy()
    n = len(a)
    best_a = a.copy()
    best_score = compute_score_normalized(a)
    t0 = time.monotonic()
    
    for k in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(a)
        val, w = smooth_max_and_grad(conv, T)
        grad = compute_gradient(a, w)
        # Full objective gradient: 2n * grad (since f_T = 2n * T*log(...))
        grad_full = 2.0 * n * grad
        
        # FW step: find vertex with min gradient
        i_min = np.argmin(grad_full)
        e_i = np.zeros(n)
        e_i[i_min] = 1.0
        
        # Step size: standard FW is t = 2/(k+2), but we can also use the
        # "away-step" variant or exact line search
        t_step = 2.0 / (k + 2)
        
        a_new = (1 - t_step) * a + t_step * e_i
        
        # Check if improvement
        score_new = compute_score_normalized(a_new)
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
        
        a = a_new
        
        # Also try a cheaper vertex: distribute mass to the k smallest gradient coords
        # This gives a better direction than a single vertex
        if k % 10 == 0:
            k_top = min(10, n)
            top_indices = np.argsort(grad_full)[:k_top]
            v_dir = np.zeros(n)
            v_dir[top_indices] = 1.0 / k_top
            t_step2 = 2.0 / (k + 2)
            a_new2 = (1 - t_step2) * a + t_step2 * v_dir
            score_new2 = compute_score_normalized(a_new2)
            if score_new2 < best_score:
                best_score = score_new2
                best_a = a_new2.copy()
                a = a_new2
        
        if verbose and k % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW iter={k}, val={val:.8f}, best={best_score:.8f}, t={elapsed:.0f}s")
    
    return best_a, best_score


def grad_descent_logspace(a_init, T, time_limit, verbose=False):

    a = a_init.copy()
    n = len(a)

            break

        z_shifted = z - z.max()
        a = np.exp(z_shifted)
        a = a / a.sum()
        
        conv = fast_convolve(a)
        val, w = smooth_max_and_grad(conv, T)
        grad_a = 2.0 * n * compute_gradient(a, w)

        # d/ dz [f(a(z))] where a = softmax(z)

                if score_new < best_score:

                break

def structured_init(n, rng):
    """Generate a structured initial sequence."""
    choice = rng.integers(0, 5)

        a = 0.5 + 0.3 * np.cos(t + phase) + 0.2 * np.cos(2 * t + 2 * phase)
    elif choice == 2:

        t = np.linspace(-2, 2, n)
        a = np.exp(-t**2)

        a = np.exp(-((t-0.25)/0.15)**2) + np.exp(-((t-0.75)/0.15)**2)

def interpolate_sequence(a, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0)
    ss = a_interp.sum()
    if ss > 1e-15:
        a_interp = a_interp / ss
    return a_interp

df_T/d a_j = 2n * T * (1/S) * sum_k exp(c_k/T) * (1/T) * dc_k/d a_j

Now, for the Frank-Wolfe method, the FW step is:
1. Compute gradient g at current point a
2. Solve min_v g^T v s.t. v in simplex → v = e_{argmin(g)}

def fast_convolve(a):

    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(a, b):
    """Linear cross-convolution a * b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)

    return irfft(A * B, fft_size)[:size]

def smooth_max_and_weights(c, T):
    c_shifted = c / T
    c_max = c_shifted.max()
    exp_c = np.exp(c_shifted - c_max)
    S = exp_c.sum()
    w = exp_c / S
    val = T * (c_max + np.log(S))
    return val, w


def compute_smooth_gradient(a, w):
    """Gradient of T*log(sum(exp(c/T))) w.r.t. a, where c = conv(a,a)."""
    return 2.0 * np.correlate(w, a, mode='valid')


def build_constraint_matrix(f):
    """Build C where C[k,j] = f[k-j] if 0<=k-j<len(f) else 0."""
    n = len(f)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = f
    return C


def greedy_lp(f, M):
    """Greedy LP: max sum(g) s.t. C*f_convolve g <= M, g >= 0."""
    n = len(f)
    g = np.zeros(n)
    conv_vals = np.zeros(2 * n - 1)
    
    for j in range(n):
        k_start = j
        k_end = j + n - 1
        slack = M - conv_vals[k_start:k_end + 1]
        f_slice = f[:n]  # f[k-j] for k in [j, j+n-1]
        valid = f_slice > 1e-15
        if not valid.any():
            continue
        max_gj = np.inf
        for idx in range(n):
            if f_slice[idx] > 1e-15 and slack[idx] > 0:
                cand = slack[idx] / f_slice[idx]
                if cand < max_gj:
                    max_gj = cand
            elif f_slice[idx] > 1e-15:
                max_gj = 0
                break
        max_gj = max(0, max_gj)
        g[j] = max_gj
        conv_vals[k_start:k_end + 1] += f_slice * max_gj
    
    return g


def solve_lp_scipy(f, M):
    """Solve LP: max sum(g) s.t. C@g <= M*1, g >= 0."""
    from scipy.optimize import linprog
    n = len(f)
    C = build_constraint_matrix(f)
    c_obj = -np.ones(n)

    if res.status != 0:

    return np.maximum(res.x, 0)
