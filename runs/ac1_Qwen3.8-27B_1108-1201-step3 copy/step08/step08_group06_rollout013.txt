        pressure[j_min:j_max + 1] += wk * a[m]
    return pressure


def mirror_descent(a0, time_budget, verbose=True):
    """Fast mirror descent with multi-peak pressure."""
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-30:
        return a0.copy(), compute_score(a0)
    a = a / s
    n = len(a)

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()

    eta = 0.15
    eta_min = 1e-6
    eta_max = 0.5
    K = 8
    no_improve = 0
    iteration = 0

    while time.monotonic() - t0 < time_budget:
        iteration += 1
        b = fast_convolve(a)
        b_mean = b.mean()

        # Select top K peaks
        if K <= len(b):
            top_idx = np.argsort(b)[::-1][:K]
        else:
            top_idx = np.arange(len(b))

        # Compute weights: emphasize peaks above mean
        peak_vals = b[top_idx]
        weights = np.maximum(peak_vals - b_mean, 0)
        if weights.max() < 1e-15:
            weights = peak_vals.copy()
        # Sharpen weights (emphasize the very top)
        weights = weights ** 2

        # Compute pressure
        pressure = compute_pressure_direct(a, top_idx, weights)

        if pressure.max() < 1e-15:
            break

        # Normalize pressure for step size control
        p_norm = pressure / pressure.max()

        # Mirror descent update (exponential)
        a_new = a * np.exp(-eta * p_norm)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-30:
            eta *= 0.5
            continue
        a_new = a_new / s_new

        sc = compute_score(a_new)

        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
            no_improve = 0
            # Slightly increase eta if improving rapidly
            if sc < best_score * 0.999:
                eta = min(eta * 1.05, eta_max)
        else:
            a_new2 = a * np.exp(-eta * 0.5 * p_norm)
            a_new2 = np.maximum(a_new2, 0)
            s2 = a_new2.sum()
            if s2 > 1e-30:
                a_new2 = a_new2 / s2
                sc2 = compute_score(a_new2)
                if sc2 < best_score - 1e-14:
                    best_score = sc2
                    best_a = a_new2.copy()
                    a = a_new2
                    no_improve = 0
                    continue
            # Reduce step size
            eta = max(eta * 0.7, eta_min)
            no_improve += 1

        a = a_new

        # If stuck, try a perturbation
        if no_improve > 50 and iteration % 50 == 0:
            rng_local = np.random.default_rng(iteration)
            noise = rng_local.standard_normal(n) * 0.003
            a_perturbed = a + noise
            a_perturbed = project_simplex(a_perturbed)
            sc_p = compute_score(a_perturbed)
            if sc_p < best_score:
                a = a_perturbed
            else:
                a = best_a.copy()
            no_improve = 0
            eta = min(eta * 1.5, eta_max)

        if verbose and iteration % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD iter={iteration}, score={best_score:.8f}, eta={eta:.4f}, elapsed={elapsed:.0f}s")

    return best_a, best_score


def coordinate_descent(a0, time_budget, verbose=True, seed=42):
    """Coordinate descent with golden-section search."""
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-30:
        return a0.copy(), compute_score(a0)
    a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    gr = (np.sqrt(5) - 1) / 2
    iters = 0
    no_improve_count = 0

    while time.monotonic() - t0 < time_budget:
        iters += 1
        improved_this_pass = False

        # Generate candidate pairs
        n_pairs = min(n, 800)
        pairs = set()

        # Get current peaks for focus
        conv = fast_convolve(a)
        top_peak_idx = np.argmax(conv)
        js = max(0, top_peak_idx - n + 1)
        je = min(n - 1, top_peak_idx)

        # Random pairs
        for _ in range(n_pairs // 3):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                pairs.add((min(i, j), max(i, j)))

        # Peak-focused pairs
        for _ in range(n_pairs // 3):
            i = rng.integers(js, je + 1)
            j = rng.integers(js, je + 1)
            if i != j:
                pairs.add((min(i, j), max(i, j)))

        # Edge pairs
        edge_w = max(2, n // 15)
        for _ in range(n_pairs // 6):
            i = rng.integers(0, edge_w)
            j = rng.integers(n - edge_w, n)
            if i != j:
                pairs.add((min(i, j), max(i, j)))

        # Mid-range pairs (valley filling)
        mid_start = n // 3
        mid_end = 2 * n // 3
        for _ in range(n_pairs // 6):
            i = rng.integers(mid_start, mid_end)
            j = rng.integers(mid_start, mid_end)
            if i != j:
                pairs.add((min(i, j), max(i, j)))

        for (i, j) in list(pairs):
            if time.monotonic() - t0 > time_budget:
                break
            if a[i] < 1e-14 and a[j] < 1e-14:
                continue

            t_lo = -a[j] + 1e-15
            t_hi = a[i] - 1e-15
            if t_hi <= t_lo:
                continue

            x1 = t_hi - gr * (t_hi - t_lo)
            x2 = t_lo + gr * (t_hi - t_lo)

            def eval_t(t):
                ai = a[i] - t
                aj = a[j] + t
                if ai < 0 or aj < 0:
                    return 1e10
                # Quick score estimate: only recompute affected convolution entries
                # For speed, use full FFT
                a_trial = a.copy()
                a_trial[i] = ai
                a_trial[j] = aj
                a_trial = a_trial / a_trial.sum()
                return compute_score(a_trial)

            f1 = eval_t(x1)
            f2 = eval_t(x2)

            for _ in range(18):
                if time.monotonic() - t0 > time_budget:
                    break
                if f1 < f2:
                    t_hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = t_hi - gr * (t_hi - t_lo)
                    f1 = eval_t(x1)
                else:
                    t_lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = t_lo + gr * (t_hi - t_lo)
                    f2 = eval_t(x2)

            t_opt = (t_lo + t_hi) / 2
            f_opt = min(f1, f2)

            if f_opt < best_score - 5e-14:
                a[i] -= t_opt
                a[j] += t_opt
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = compute_score(a)
                best_a = a.copy()
                improved_this_pass = True

        if not improved_this_pass:
            no_improve_count += 1
            if no_improve_count > 8:
                noise = rng.standard_normal(n) * 0.001
                a = best_a + noise
                a = project_simplex(a)
                no_improve_count = 0
        else:
            no_improve_count = 0

        if verbose and iters % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD iter={iters}, score={best_score:.8f}, elapsed={elapsed:.0f}s")

    return best_a, best_score


def coordinate_descent_fine(a0, time_budget, verbose=True, seed=123):
    """Very fine coordinate descent - smaller perturbations, more pairs."""
    n = len(a0)
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-30:
        return a0.copy(), compute_score(a0)
    a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    gr = (np.sqrt(5) - 1) / 2
    iters = 0

    while time.monotonic() - t0 < time_budget:
        iters += 1
        improved = False

        # Focused on the most problematic pairs
        conv = fast_convolve(a)
        # Top 3 peaks
        top_peaks = np.argsort(conv)[::-1][:3]

        for pk in top_peaks:
            if time.monotonic() - t0 > time_budget:
                break
            js = max(0, int(pk) - n + 1)
            je = min(n - 1, int(pk))
            # Contributing indices
            idx = np.arange(js, je + 1)
            if len(idx) < 2:
                continue
            # Sort by contribution to this peak
            contribs = a[idx] * a[int(pk) - idx]
            top_contrib = idx[np.argsort(contribs)[::-1][:5]]

            for i_idx in top_contrib:
                if time.monotonic() - t0 > time_budget:
                    break
                # Try transferring to random non-contributing positions
                for _ in range(5):
                    if time.monotonic() - t0 > time_budget:
                        break
                    j_candidate = rng.integers(0, n)
                    if j_candidate == i_idx:
                        continue
                    i = min(int(i_idx), int(j_candidate))
                    j = max(int(i_idx), int(j_candidate))
                    if a[i] < 1e-14 and a[j] < 1e-14:
                        continue

                    t_lo = -a[j] + 1e-15
                    t_hi = a[i] - 1e-15
                    if t_hi <= t_lo:
                        continue

                    x1 = t_hi - gr * (t_hi - t_lo)
                    x2 = t_lo + gr * (t_hi - t_lo)

                    def eval_t2(t):
                        a_trial = a.copy()
                        a_trial[i] -= t
                        a_trial[j] += t
                        if a_trial[i] < 0 or a_trial[j] < 0:
                            return 1e10
                        a_trial = a_trial / a_trial.sum()
                        return compute_score(a_trial)

                    f1 = eval_t2(x1)
                    f2 = eval_t2(x2)

                    for _ in range(15):
                        if time.monotonic() - t0 > time_budget:
                            break
                        if f1 < f2:
                            t_hi = x2
                            x2 = x1
                            f2 = f1
                            x1 = t_hi - gr * (t_hi - t_lo)
                            f1 = eval_t2(x1)
                        else:
                            t_lo = x1
                            x1 = x2
                            f1 = f2
                            x2 = t_lo + gr * (t_hi - t_lo)
                            f2 = eval_t2(x2)

                    t_opt = (t_lo + t_hi) / 2
                    f_opt = min(f1, f2)

                    if f_opt < best_score - 5e-14:
                        a[i] -= t_opt
                        a[j] += t_opt
                        a = np.maximum(a, 0)
                        a = a / a.sum()
                        best_score = compute_score(a)
                        best_a = a.copy()
                        improved = True

        if verbose and iters % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD-fine iter={iters}, score={best_score:.8f}, elapsed={elapsed:.0f}s")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_and_update(a, label, best_seq, best_score, verbose=True):
    """Evaluate a candidate and return updated best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-strategy optimization combining fast mirror descent,
    aggressive coordinate descent, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        n_init = 768
        t_arr = np.linspace(0, 2 * np.pi * 3, n_init)
        parent = np.ones(n_init) + 0.1 * np.cos(t_arr)
        parent = np.maximum(parent, 0)
        parent = parent / parent.sum()
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Initialized: n={n_init}, score={best_score:.8f}")

    n_parent = len(parent)
    print(f"Starting optimization. Initial best: {best_score:.8f}")

    # === Phase 1: Mirror Descent ===
    print(f"\n=== Phase 1: Mirror Descent (best={best_score:.8f}) ===")
    t1_budget = min(350.0, (deadline - time.monotonic()) * 0.22)
    rem = deadline - time.monotonic()
    if rem > t1_budget + 500:
        t_alloc = t1_budget
    else:
        t_alloc = max(60, rem * 0.15)
    print(f"  Budget: {t_alloc:.0f}s")
    a_md, sc_md = mirror_descent(best_seq, t_alloc, verbose=True)
    best_seq, best_score = check_and_update(a_md, "mirror_descent", best_seq, best_score)

    # === Phase 2: Coordinate Descent (aggressive) ===
    print(f"\n=== Phase 2: Coordinate Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 600
    if rem > 60:
        t_alloc = min(rem * 0.25, 400.0)
        print(f"  Budget: {t_alloc:.0f}s")
        a_cd, sc_cd = coordinate_descent(best_seq, t_alloc, verbose=True, seed=42)
        best_seq, best_score = check_and_update(a_cd, "coord_desc", best_seq, best_score)

    # === Phase 3: Mirror Descent again (on refined solution) ===
    print(f"\n=== Phase 3: Mirror Descent Round 2 (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 500
    if rem > 40:
        t_alloc = min(rem * 0.15, 200.0)
        print(f"  Budget: {t_alloc:.0f}s")
        a_md2, sc_md2 = mirror_descent(best_seq, t_alloc, verbose=True)
        best_seq, best_score = check_and_update(a_md2, "md_round2", best_seq, best_score)

    # === Phase 4: Fine Coordinate Descent ===
    print(f"\n=== Phase 4: Fine Coordinate Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 400
    if rem > 40:
        t_alloc = min(rem * 0.25, 300.0)
        print(f"  Budget: {t_alloc:.0f}s")
        a_cd2, sc_cd2 = coordinate_descent_fine(best_seq, t_alloc, verbose=True, seed=123)
        best_seq, best_score = check_and_update(a_cd2, "coord_fine", best_seq, best_score)

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length Exploration (best={best_score:.8f}) ===")
    offsets = [1, -1, 2, -2, 3, -3, 4, -4, 6, 8, -8, 12, 16, -16, 24, 32, -32, 48, 64]
    tried_lengths = set()
    for offset in offsets:
        rem = deadline - time.monotonic() - 100
        if rem < 50:
            break
        alt_n = n_parent + offset
        if alt_n < 64 or alt_n > 3000 or alt_n in tried_lengths:
            continue
        tried_lengths.add(alt_n)

        t_alloc = min(rem * 0.1, 80.0)
        if t_alloc < 20:
            continue

        init_alt = interpolate_sequence(best_seq, alt_n)
        print(f"  Trying n={alt_n}, budget={t_alloc:.0f}s")

        # Quick mirror descent
        t_md = t_alloc * 0.5
        a_alt_md, _ = mirror_descent(init_alt, t_md, verbose=False)
        best_seq, best_score = check_and_update(a_alt_md, f"n{alt_n}_md", best_seq, best_score)

        # Quick coordinate descent
        rem2 = deadline - time.monotonic() - 80
        if rem2 > 20:
            t_cd = min(rem2 * 0.1, 60.0)
            a_alt_cd, _ = coordinate_descent(a_alt_md, t_cd, verbose=False, seed=alt_n)
            best_seq, best_score = check_and_update(a_alt_cd, f"n{alt_n}_cd", best_seq, best_score)

    # === Phase 6: Final polish - coordinate descent on best ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 60
    if rem > 30:
        t_alloc = min(rem * 0.4, 200.0)
        print(f"  CD fine budget: {t_alloc:.0f}s")
        a_fin, _ = coordinate_descent_fine(best_seq, t_alloc, verbose=True, seed=999)
        best_seq, best_score = check_and_update(a_fin, "final_cd", best_seq, best_score)

    # === Phase 7: Even tighter polish ===
    rem = deadline - time.monotonic() - 30
    if rem > 20:
        t_alloc = min(rem * 0.5, 100.0)
        print(f"\n=== Phase 7: Tight Polish (best={best_score:.8f}) ===")
        print(f"  Budget: {t_alloc:.0f}s")
        a_fin2, _ = coordinate_descent(best_seq, t_alloc, verbose=False, seed=7777)
        best_seq, best_score = check_and_update(a_fin2, "tight_cd", best_seq, best_score)

    # === Phase 8: Final mirror descent pass ===
    rem = deadline - time.monotonic() - 10
    if rem > 15:
        t_alloc = min(rem * 0.7, 50.0)
        print(f"\n=== Phase 8: Final MD (best={best_score:.8f}) ===")
        a_fin3, _ = mirror_descent(best_seq, t_alloc, verbose=False)
        best_seq, best_score = check_and_update(a_fin3, "final_md", best_seq, best_score)

    # Fallback
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    final_score = float(evaluate_sequence([float(x) for x in best_seq]))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
