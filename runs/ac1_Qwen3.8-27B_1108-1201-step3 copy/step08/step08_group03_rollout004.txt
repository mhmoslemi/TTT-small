import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def convolve_direction(f, g):
    n = len(f)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]

def build_constraint_matrix(f, indices, n):
    K = len(indices)
    A = np.zeros((K, n))
    for i, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi+1] = f[lo:hi+1]
    return A

def solve_lp_hiGHS(f, n, M, active_idx):
    """Solve: max sum(g) s.t. g >= 0, A@g <= M using HiGHS."""
    K = len(active_idx)
    if K == 0:
        return None
    A = build_constraint_matrix(f, active_idx, n)
    c = -np.ones(n)  # minimize -sum(g)
    constraints = [A @ g <= M, g >= 0]
    # In scipy linprog form: A_ub @ x <= b_ub, bounds

def fast_convolve(a):
    n = len(a)

def cross_convolve(f, g):
