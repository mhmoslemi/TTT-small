import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def cross_correlate_p_a(p, a):
    """Compute c_j = sum_k p_k * a_{k-j} for j=0..n-1 using FFT."""
    n = len(a)
    fft_size = 1 << (len(p) + n - 1 - 1).bit_length()
    a_flip = a[::-1]
    P = rfft(p, fft_size)
    A = rfft(a_flip, fft_size)
    return irfft(P * A, fft_size)[n - 1:2 * n - 1]


def power_sum_obj_grad(a, p):
    """Compute sum(b^p) and its gradient, where b = conv(a,a)."""
    n = len(a)
    b = fast_convolve(a)
    obj = np.sum(b ** p)
    # grad_j = 2p * sum_k b_k^(p-1) * a_{k-j}
    weights = b ** (p - 1)
    c = cross_correlate_p_a(weights, a)
    grad = 2.0 * p * c
    return obj, grad, b


def softmax_obj_grad(a, tau):
    """Compute soft-max of conv(a,a) and its gradient."""
    n = len(a)
    b = fast_convolve(a)
    b_scaled = b / tau
    b_max = b_scaled.max()
    exp_b = np.exp(b_scaled - b_max)
    p = exp_b / exp_b.sum()
    obj = tau * (b_max + np.log(exp_b.sum()))
    c = cross_correlate_p_a(p, a)
    grad = 2.0 * c
    return obj, grad, p, b

def build_constraint_matrix(f, indices, n):
    """Build K x n constraint matrix for given peak indices."""
    K = len(indices)
    A = np.zeros((K, n))
    for i in range(K):
        k = indices[i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    return A


def solve_lp_direction(f, indices, M, solver='ECOS'):
    """Solve: max sum(g) s.t. g >= 0, A@g <= M. Returns (g, dual) or (None, None)."""
    n = len(f)
    K = len(indices)
    A = build_constraint_matrix(f, indices, n)

    if A.shape[0] == 0:
        return None, None

    g = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [g >= 0, A @ g <= M])

    try:
        if K <= 600 and n <= 1200:
            prob.solve(solver=cp.ECOS, max_iters=30000, abs_acceleration=1e-10)
        else:
            prob.solve(solver=cp.SCS, max_iters=200000, eps_abs=1e-9, eps_rel=1e-9)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=100000, eps_abs=1e-8, eps_rel=1e-8)
        except Exception:
            return None, None

    if g.value is None:
        return None, None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
        return None, None

    dual_vals = None
    try:
        if prob.status in ['optimal', 'optimal_inaccurate']:
            # Try to get dual values from the constraint
            for c in prob.constraints:
                if hasattr(c, 'dual_value') and c.dual_value is not None:
                    dv = c.dual_value
                    if dv is not None and len(dv) == K:
                        dual_vals = np.abs(dv)
                        break
    except Exception:
        pass

    return g_val, dual_vals


def select_peaks_adaptive(b, f, n, top_k, dual_weights=None):
    """Adaptive peak selection combining top-K, structural, and dual-informed peaks."""
    full_len = 2 * n - 1
    candidates = set()

    # Top peaks
    if top_k >= full_len:
        return np.arange(full_len)

    n_top = min(top_k, full_len)
    top_idx = np.argpartition(b, -n_top)[-n_top:]
    candidates.update(top_idx.tolist())

    # Boundary peaks (important for the convolution structure)
    n_bnd = min(n, 20)
    for i in range(n_bnd):
        if i < full_len:
            candidates.add(i)
        if full_len - 1 - i >= 0:
            candidates.add(full_len - 1 - i)

    # Center region
    center = (full_len - 1) // 2
    for di in range(-5, 6):
        idx = center + di
        if 0 <= idx < full_len:
            candidates.add(idx)

    # Dual-informed: peaks with high dual weight
    if dual_weights is not None and len(dual_weights) == full_len:
        dual_top = np.argpartition(dual_weights, -max(10, top_k // 4))[-max(10, top_k // 4):]
        candidates.update(dual_top.tolist())

    all_candidates = list(candidates)
    all_candidates.sort(key=lambda i: b[i], reverse=True)
    indices = np.array(all_candidates[:top_k])
    return np.sort(indices)

def coordinate_descent(f_init, time_limit, label="cd"):
    """Coordinate descent with multiple multiplier candidates."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    best_f = f.copy()

    multipliers = [0.3, 0.5, 0.7, 0.85, 0.92, 1.08, 1.15, 1.3, 1.5, 1.8, 2.5]
    no_pass = 0

    while time.monotonic() - t0 < time_limit and no_pass < 3:
        any_improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-14:
                continue

            best_j_val = f[j]
            best_j_obj = best_obj
            for mult in multipliers:
                new_val = f[j] * mult
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-12:
                    continue
                f_trial = f_trial / s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = new_val

            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-12:
                    f = f / s
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_score = 2.0 * n * best_obj
                    best_f = f.copy()
                    any_improved = True

        if any_improved:
            no_pass = 0
        else:
            no_pass += 1

        if no_pass == 0 and (j + 1) % 200 == 0:
            print(f"    {label} pass no={no_pass} obj={best_obj:.10f} score={best_score:.6f} "
                  f"t={time.monotonic()-t0:.0f}s")

    return best_f, best_score


def interp_sequence(src, target_n):
    """Interpolate a sequence to a new length."""
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result
