    size = 2 * n - 1

    return irfft(F * G, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

The Frank-Wolfe method for minimizing a convex function over a convex set works as follows:

2. Take a step towards the minimizer

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_cross(f, g):
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp(f, n, M, active_idx, time_budget=5.0):
    """Solve LP direction with multiple solver fallbacks."""
    K = len(active_idx)
    if K == 0:
        return None, None
    # Build constraint matrix
    A = np.zeros((K, n))
    for i, k in enumerate(active_idx):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    g = cp.Variable(n)
    y = cp.Variable(K)
    constraints = [g >= 0, A @ g <= M, y >= 0]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    g_val = None
    y_val = None
    # Try solvers in order of preference
    solvers_to_try = []
    if n <= 5000:
        solvers_to_try.append(('PDDP', {'max_iter': 500000, 'eps_abs': 1e-9, 'eps_rel': 1e-9}))
    solvers_to_try.append(('SCS', {'max_iters': 500000, 'eps': 1e-9}))
    if n <= 3000:
        solvers_to_try.append(('ECOS', {'max_iters': 50000}))
    solvers_to_try.append(('SCS', {'max_iters': 300000, 'eps': 1e-8}))
    t_start = time.monotonic()
    for solver_name, solver_opts in solvers_to_try:
        if time.monotonic() - t_start > time_budget:
            break
        try:
            prob.solve(solver=getattr(cp, solver_name), **solver_opts)
            if g.value is not None:
                g_val = np.maximum(g.value, 0.0)
                if np.any(np.isnan(g_val)):
                    g_val = None
                else:
                    # Also get dual variables
                    try:
                        if constraints[1].dual.value is not None:
                            y_val = np.abs(constraints[1].dual.value)
                    except Exception:
                        y_val = None
                    break
        except Exception:
            continue
    return g_val, y_val

def lp_optimize(f_init, time_limit, initial_k=60, max_k=400, label="lp"):

        f = np.ones(n) / n

    b = fast_convolve(f)
    M = b.max()
    best_obj = M

    active_idx = np.sort(active_idx)

def dual_informed_perturb(f, y_dual, active_idx, strength=0.15, rng=None):

        f_new = np.maximum(f + perturb, 0.0)
        s = f_new.sum()

            f_new /= s
        else:
            f_new = f.copy()
        return f_new

    # Use dual variables to create a directed perturbation
    # High dual => constraint is tight => try to move away from satisfying it
    # This creates diversity
    perturb = np.zeros(n)
    for i, k in enumerate(active_idx):
        w = y_dual[i]
        if w < 1e-10:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # The gradient of b_k w.r.t. f is: (f[k-1], f[k-2], ..., f[0], 0, ..., f[1]) ... 
        # Actually db_k/df_j = f[k-j] for valid k-j
        # So the "peak direction" at k is: grad[j] = f[k-j]
        # We want to perturb AWAY from reducing this peak (to explore)
        # Add noise weighted by dual
        lo2 = max(0, k - n + 1)
        hi2 = min(n - 1, k)
        indices = np.arange(lo2, hi2 + 1)
        perturb[indices] += w * f[k - indices] * rng.normal(0, strength)

    # Normalize perturbation
    pnorm = np.linalg.norm(perturb)
    if pnorm > 1e-10:
        perturb = perturb / pnorm * 0.1
    f_new = np.maximum(f + perturb, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = f.copy()
    return f_new


def iaa_flatten(f_init, n_iters=100, floor_frac=0.3):
    """IAA-based spectrum flattening with water-filling."""
    n = len(f_init)
    f = f_init.copy()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    for it in range(n_iters):
        b = fast_convolve(f)
        peak = b.max()
        if peak < 1e-12:
            break
        # Water-filling: reduce peaks above threshold, raise others
        threshold = floor_frac * peak
        b_new = b.copy()
        excess = np.maximum(b_new - threshold, 0)
        total_excess = excess.sum()
        deficit_mask = b_new < threshold
        n_deficit = deficit_mask.sum()
        if n_deficit > 0 and total_excess > 0:
            b_new[deficit_mask] += total_excess / n_deficit
            b_new[~deficit_mask] = threshold
        # Renormalize
        if b_new.sum() > 1e-10:
            b_new *= b.sum() / b_new.sum()
        b_new = np.maximum(b_new, 0)
        # Recover f from b_new (phase retrieval step)
        B = rfft(b_new, fft_size)
        F_cur = rfft(f, fft_size)
        abs_F = np.abs(F_cur)
        abs_B = np.abs(B)
        # We want |F|^2 ~ B, so |F| ~ sqrt(B)
        target_abs = np.sqrt(np.maximum(abs_B, 0))
        # Blend with current
        if abs_F.max() > 1e-10:
            target_abs = 0.7 * target_abs + 0.3 * abs_F
        phase = F_cur / (abs_F + 1e-30)
        F_new = target_abs * phase
        f = irfft(F_new, fft_size)[:n]
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s
        else:
            break
    return f

def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase search combining enhanced cutting-plane LP, smooth-max gradient,
    dual-informed diversification, IAA flattening, and multi-length exploration.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception as e:
        print(f"Parent load failed: {e}")

    if parent is None:
        parent = np.ones(1024) / 1024
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: uniform n=1024, score={p_score:.6f}")

    best_seq = parent.copy()
    best_score = p_score

    # === Phase 1: Main LP optimization ===
    print(f"\n=== Phase 1: LP (n={len(best_seq)}, score={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.35, 550.0)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        f1, obj1 = lp_optimize(parent, time_limit=t1, initial_k=60, max_k=400, label="p1")
        score1 = float(evaluate_sequence(f1.tolist()))
        if np.isfinite(score1) and score1 < best_score:
            best_score = score1
            best_seq = f1.copy()
            print(f"  NEW BEST [p1]: {score1:.6f}")

    # === Phase 2: Smooth-max gradient polish ===
    print(f"\n=== Phase 2: Smooth-max (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.12, 150.0)
    if t2 > 15:
        print(f"  Budget: {t2:.0f}s")
        f2, obj2 = smooth_max_descent(best_seq, time_limit=t2, tau_init=0.005, lr_init=0.003)
        score2 = float(evaluate_sequence(f2.tolist()))
        if np.isfinite(score2) and score2 < best_score:
            best_score = score2
            best_seq = f2.copy()
            print(f"  NEW BEST [p2]: {score2:.6f}")

    # === Phase 3: LP deep dive ===
    print(f"\n=== Phase 3: LP deep (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.20, 300.0)
    if t3 > 30:
        print(f"  Budget: {t3:.0f}s")
        f3, obj3 = lp_optimize(best_seq, time_limit=t3, initial_k=80, max_k=500, label="p3")
        score3 = float(evaluate_sequence(f3.tolist()))
        if np.isfinite(score3) and score3 < best_score:
            best_score = score3
            best_seq = f3.copy()
            print(f"  NEW BEST [p3]: {score3:.6f}")

    # === Phase 4: Diversification rounds ===
    print(f"\n=== Phase 4: Diversify (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    n_diverse = 3
    for d in range(n_diverse):
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_div = min(rem * 0.15, 100.0)
        if t_div < 20:
            break

        # Create diverse starting point
        if d == 0:
            # IAA flatten
            f_div = iaa_flatten(best_seq, n_iters=80, floor_frac=0.25)
        elif d == 1:
            # IAA with different params
            f_div = iaa_flatten(best_seq, n_iters=120, floor_frac=0.5)
        else:
            # Random perturbation + smooth-max
            perturb = rng.standard_normal(len(best_seq)) * 0.08
            f_div = np.maximum(best_seq + perturb, 0.0)
            s = f_div.sum()
            if s > 1e-10:
                f_div /= s
            else:
                f_div = best_seq.copy()
            # Quick smooth-max
            t_sm = min(t_div * 0.3, 30.0)
            f_div, _ = smooth_max_descent(f_div, time_limit=t_sm, tau_init=0.01, lr_init=0.005)

        # LP on diverse point
        t_lp_div = t_div - (t_div * 0.3 if d == 2 else 0)
        f_div_opt, _ = lp_optimize(f_div, time_limit=max(10.0, t_lp_div),
                                    initial_k=40, max_k=300, label=f"div{d}")
        score_div = float(evaluate_sequence(f_div_opt.tolist()))
        if np.isfinite(score_div) and score_div < best_score:
            best_score = score_div
            best_seq = f_div_opt.copy()
            print(f"  NEW BEST [div{d}]: {score_div:.6f}")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length (best={best_score:.6f}) ===")
    n_base = len(best_seq)
    # Try a few nearby lengths
    length_candidates = []
    for delta in [-100, -50, 50, 100, 150, -150, 200]:
        tl = n_base + delta
        if 512 <= tl <= 3000:
            length_candidates.append(tl)
    # Deduplicate and sort by proximity
    length_candidates = sorted(set(length_candidates), key=lambda x: abs(x - n_base))

    for target_n in length_candidates[:3]:
        rem = deadline - time.monotonic()
        if rem < 50:
            break
        t_m = min(rem * 0.12, 100.0)
        if t_m < 20:
            break

        # Interpolate
        old_x = np.linspace(0.0, 1.0, n_base)
        new_x = np.linspace(0.0, 1.0, target_n)
        f_interp = np.interp(new_x, old_x, best_seq)
        f_interp = np.maximum(f_interp, 0.0)
        s = f_interp.sum()
        if s > 1e-10:
            f_interp /= s
        else:
            f_interp = np.ones(target_n) / target_n

        print(f"  Multi-length n={target_n}, budget={t_m:.0f}s")
        f_ml, _ = lp_optimize(f_interp, time_limit=t_m, initial_k=40, max_k=300, label=f"n{target_n}")
        score_ml = float(evaluate_sequence(f_ml.tolist()))
        if np.isfinite(score_ml) and score_ml < best_score:
            best_score = score_ml
            best_seq = f_ml.copy()
            print(f"  NEW BEST [n{target_n}]: {score_ml:.6f}")

    # === Phase 6: Final LP burst ===
    print(f"\n=== Phase 6: Final LP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.25, 200.0)
    if t6 > 20:
        print(f"  Budget: {t6:.0f}s, n={len(best_seq)}")
        f6, _ = lp_optimize(best_seq, time_limit=t6, initial_k=100, max_k=600, label="final")
        score6 = float(evaluate_sequence(f6.tolist()))
        if np.isfinite(score6) and score6 < best_score:
            best_score = score6
            best_seq = f6.copy()
            print(f"  NEW BEST [final]: {score6:.6f}")

    # === Phase 7: Final smooth-max polish ===
    rem = deadline - time.monotonic()
    if rem > 15:
        t7 = min(rem * 0.5, 60.0)
        print(f"\n=== Phase 7: Final smooth-max (best={best_score:.6f}) ===")
        f7, _ = smooth_max_descent(best_seq, time_limit=t7, tau_init=0.002, lr_init=0.001)
        score7 = float(evaluate_sequence(f7.tolist()))
        if np.isfinite(score7) and score7 < best_score:
            best_score = score7
            best_seq = f7.copy()
            print(f"  NEW BEST [sm7]: {score7:.6f}")

    # Final validation
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
