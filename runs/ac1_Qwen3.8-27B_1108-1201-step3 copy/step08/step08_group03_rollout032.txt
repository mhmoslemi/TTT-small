def subgradient_polish(f_init, time_limit, lr=0.001, n_top=40, temp=0.002, label=""):
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_M = fast_convolve(f).max()
    v = np.zeros(n)
    momentum = 0.8
    no_improve = 0
    rng = np.random.default_rng(123)
    
    it = 0
    while time.monotonic() - t0 < time_limit - 2:
        it += 1
        b = fast_convolve(f)
        
        # Soft-max weights on top peaks
        scaled = b / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        
        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
            w_sel = w[top_idx].copy()
            ws = w_sel.sum()
            if ws > 1e-30:
                w_sel /= ws
            w = np.zeros(len(b))
            w[top_idx] = w_sel
        
        # Subgradient
        g = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-12:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += w[k] * 2.0 * f[k - idx]
        
        v = momentum * v - lr * g
        f = project_simplex(f + v)
        
        obj = fast_convolve(f).max()
        if obj < best_M - 1e-12:
            best_M = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 20:
                lr *= 0.9
                temp *= 1.1
                no_improve = 0
            if no_improve > 80:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                v = np.zeros(n)
                no_improve = 0
        
        if it % 500 == 0:
            print(f"  {label} it={it} M={best_M:.10f} score={2*n*best_M:.6f} lr={lr:.5f}")
    
    return best_f, best_M

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP with cutting planes and thermal restarts for diversification,
spectral peak shaving, multi-length exploration, and subgradient polishing.
Key improvements: thermal restarts to escape local optima, adaptive active set
management, diverse initializations including different sequence lengths,
and more aggressive line search.

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_convolve(f, g):
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_A(f, indices, n):
    K = len(indices)
    A = np.zeros((K, n))
    for i, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    return A


def solve_lp(f, n, M, active_idx):
    A = build_A(f, active_idx, n)
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=200000, eps=1e-8)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_iters=50000)
        except Exception:
            return None
    if g.value is None:
        return None
    gv = np.maximum(g.value, 0)
    if np.any(np.isnan(gv)):
        return None
    return gv

def spectral_shave(f_init, time_limit, label=""):
    """Iterative spectral peak shaving to flatten autocorrelation."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_M = fast_convolve(f).max()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    it = 0
    no_imp = 0
    percentile = 88

    while time.monotonic() - t0 < time_limit - 2:
        it += 1
        F = rfft(f, fft_size)
        R = irfft(F * F, fft_size)[:size]

        target = np.percentile(R, percentile) * 1.03
        R_new = np.minimum(R, target)
        e = R.sum()
        if R_new.sum() > 1e-10:
            R_new *= e / R_new.sum()
        else:
            break

        R_fft = rfft(R_new, fft_size)
        new_mag = np.sqrt(np.maximum(R_fft.real, 0))
        phase = F / (np.abs(F) + 1e-30)
        F_new = new_mag * phase
        f_new = irfft(F_new, fft_size)[:n]
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s > 1e-10:
            f_new /= s
        else:
            break

        obj = fast_convolve(f_new).max()
        if obj < best_M - 1e-12:
            best_M = obj
            best_f = f_new.copy()
            no_imp = 0
            f = f_new
        else:
            no_imp += 1
            if no_imp > 50:
                percentile += 3
                if percentile > 97:
                    break
            if no_imp > 200:
                break

        if it % 300 == 0:
            print(f"  {label} it={it} M={best_M:.10f} sc={2*n*best_M:.6f} pct={percentile}")

    return best_f, best_M


def subgradient_polish(f_init, time_limit, label=""):
    """Momentum subgradient with adaptive parameters."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_M = fast_convolve(f).max()
    v = np.zeros(n)
    lr = 0.0015
    temp = 0.003
    momentum = 0.82
    n_top = max(25, n // 12)
    no_imp = 0
    rng = np.random.default_rng(555)

    it = 0
    while time.monotonic() - t0 < time_limit - 2:
        it += 1
        b = fast_convolve(f)

        scaled = b / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        if n_top < len(b):
            top_idx = np.argpartition(b, -n_top)[-n_top:]
            w_sel = w[top_idx].copy()
            ws = w_sel.sum()
            if ws > 1e-30:
                w_sel /= ws
            w = np.zeros(len(b))
            w[top_idx] = w_sel

        g = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-12:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += w[k] * 2.0 * f[k - idx]

        v = momentum * v - lr * g
        f = project_simplex(f + v)

        obj = fast_convolve(f).max()
        if obj < best_M - 1e-12:
            best_M = obj
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 25:
                lr *= 0.88
                temp *= 1.1
                no_imp = 0
            if no_imp > 100:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.008)
                v = np.zeros(n)
                no_imp = 0

        if it % 800 == 0:
            print(f"  {label} it={it} M={best_M:.10f} sc={2*n*best_M:.6f} lr={lr:.5f}")

    return best_f, best_M


def resample_sequence(f, new_n):
    """Resample a normalized sequence to a new length via interpolation."""
    old_n = len(f)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

        if np.isfinite(score6) and score6 < best_score:
            best_score = score6
            best_seq = f6.copy()
            print(f"  *** NEW BEST [final]: {score6:.6f}")

    # Final validation
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'='*55}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*55}")

    return [float(x) for x in best_seq]
