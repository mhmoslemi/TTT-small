import numpy as np
from scipy.fft import rfft, irfft

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_soft_subgradient(f, b, temp, n_top=60):
    """Compute soft subgradient of max(b) using softmax weights over top peaks."""
    n = len(f)
    full_len = 2 * n - 1
    n_use = min(n_top, full_len)
    peak_idx = np.argpartition(b, -n_use)[-n_use:]
    b_peaks = b[peak_idx]
    w = np.exp((b_peaks - b_peaks.max()) / temp)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(n_use)
        w[0] = 1.0
    else:
        w = w / w_sum

    # Build full weight vector
    w_full = np.zeros(full_len)
    w_full[peak_idx] = w

    # Cross-correlation via FFT: g_j^(1) = sum_m w_full[m] * f[m-j]
    corr_size = 1 << (full_len + n - 1).bit_length()
    W = np.fft.rfft(w_full, corr_size)
    F = np.fft.rfft(f, corr_size)
    g_corr = np.fft.irfft(np.conj(W) * F, corr_size)[:n]

    # Add diagonal terms: for each peak k where 2j == k, add w_k * f[j]
    g = g_corr.copy()
    for i in range(n_use):
        k = int(peak_idx[i])
        j = k // 2
        if 2 * j == k and j < n:
            g[j] += w[i] * f[j]

    return g

def frank_wolfe_opt(f_init, time_limit, n_top=40):
    """Frank-Wolfe (conditional gradient) method on the simplex."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj

    it = 0
    no_improve = 0
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()

        # Compute subgradient (use top peaks with equal weight for stability)
        g = compute_soft_subgradient(f, b, temp=0.001, n_top=n_top)

        # Frank-Wolfe: find vertex with minimum inner product
        j_star = int(np.argmin(g))

        # Backtracking line search
        best_t = 0.0
        best_t_obj = M
        t_candidates = [2.0 / (it + 2.0)]
        # Add some alternatives
        t_candidates.append(0.1)
        t_candidates.append(0.05)
        t_candidates.append(0.3)

        for t in t_candidates:
            t = min(t, 1.0)
            v = np.zeros(n)
            v[j_star] = 1.0
            f_trial = (1.0 - t) * f + t * v
            f_trial = np.maximum(f_trial, 0.0)
            s_trial = f_trial.sum()
            if s_trial < 1e-15:
                continue
            f_trial = f_trial / s_trial
            obj_trial = fast_convolve(f_trial).max()
            if obj_trial < best_t_obj:
                best_t_obj = obj_trial
                best_t = t

        if best_t == 0.0:
            no_improve += 1
            if no_improve > 100:
                f = best_f + rng.standard_normal(n) * 0.01
                f = np.maximum(f, 0.0)
                f = f / f.sum()
                no_improve = 0
            continue

        v = np.zeros(n)
        v[j_star] = 1.0
        f = (1.0 - best_t) * f + best_t * v
        f = np.maximum(f, 0.0)
        f = f / f.sum()

        cur_obj = fast_convolve(f).max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_score = 2.0 * n * cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW it={it} obj={best_obj:.12f} score={best_score:.6f} "
                  f"t={elapsed:.0f}s")

    return best_f, best_score


def build_constraint_matrix(f, indices, n):
    """Build K x n constraint matrix for given peak indices."""
    K = len(indices)
    A = np.zeros((K, n))
    for i in range(K):
        k = int(indices[i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    return A


def solve_lp_direction(f, indices, M):
    """Solve: max sum(g) s.t. g >= 0, A@g <= M."""
    n = len(f)
    K = len(indices)
    A = build_constraint_matrix(f, indices, n)

    g_var = cp.Variable(n)
    objective = cp.Maximize(cp.sum(g_var))
    constraints = [g_var >= 0, A @ g_var <= M]
    prob = cp.Problem(objective, constraints)

    try:
        if K <= 500 and n <= 2000:
            prob.solve(solver=cp.ECOS, max_iters=30000, abs_acceleration=1e-10)
        else:
            prob.solve(solver=cp.SCS, max_iters=200000, eps_abs=1e-9, eps_rel=1e-9)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=100000, eps_abs=1e-8, eps_rel=1e-8)
        except Exception:
            return None

    if g_var.value is None:
        return None
    g_val = np.maximum(g_var.value, 0.0)
    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
        return None
    return g_val


def lp_optimize(f_init, time_limit, top_k_init=50, top_k_max=400, label="lp",

    """LP-based optimization with momentum and backtracking."""
    t0 = time.monotonic()

        f = np.ones(n) / n
    else:
        f = f / s

    it = 0
    stuck = 0
    top_k = top_k_init
    momentum_dir = np.zeros(n)
    full_len = 2 * n - 1

    while time.monotonic() - t0 < time_limit:

        h = g_val / S_g

        # Momentum
        momentum_dir = momentum_beta * momentum_dir + (1.0 - momentum_beta) * h
        mom_sum = momentum_dir.sum()
        if mom_sum > 1e-10:
            h_mom = momentum_dir / mom_sum
        else:
            h_mom = h

        # Try both directions
        best_t_obj = M
        best_f_trial = f.copy()
        best_t = 0.0

        for direction in [h, h_mom]:
            for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
                f_trial = (1.0 - t) * f + t * direction
                f_trial = np.maximum(f_trial, 0.0)
                s_trial = f_trial.sum()
                if s_trial < 1e-15:
                    continue
                f_trial = f_trial / s_trial
                obj_trial = fast_convolve(f_trial).max()
                if obj_trial < best_t_obj:
                    best_t_obj = obj_trial
                    best_t = t
                    best_f_trial = f_trial.copy()

        if best_t == 0.0:
            stuck += 1
            if stuck > max_stuck:
                break
            if stuck > max_stuck // 2:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                stuck = 0
                momentum_dir = np.zeros(n)
            continue

        f = best_f_trial
        cur_obj = fast_convolve(f).max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_score = 2.0 * n * cur_obj
            best_f = f.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > max_stuck:
                break

        if stuck > 30 and it % 10 == 0:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            stuck = 0
            momentum_dir = np.zeros(n)

        if it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    {label} it={it} obj={best_obj:.12f} score={best_score:.6f} "
                  f"k={len(indices)} t={elapsed:.0f}s")

    return best_f, best_score


def pairwise_polish(f_init, time_limit, n_sweeps_per_round=3):
    """Pairwise coordinate optimization: for each pair (i,j), optimize their values."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    best_f = f.copy()
    rng = np.random.default_rng(7777)

    no_improve_rounds = 0

    while time.monotonic() - t0 < time_limit:
        any_improvement = False

        for sweep in range(n_sweeps_per_round):
            if time.monotonic() - t0 > time_limit:
                break

            # Random permutation of pairs
            indices = rng.permutation(n)
            for pass_i in range(n // 2):
                i = int(indices[2 * pass_i])
                j = int(indices[2 * pass_i + 1])
                if i == j:
                    continue

                c = f[i] + f[j]
                if c < 1e-15:
                    continue

                # Optimize over fraction alpha: f[i] = alpha*c, f[j] = (1-alpha)*c
                best_alpha = f[i] / c
                best_alpha_obj = best_obj

                # Try a grid of alphas
                alphas = np.linspace(0.0, 1.0, 11)
                for alpha in alphas:
                    f_trial = f.copy()
                    f_trial[i] = alpha * c
                    f_trial[j] = (1.0 - alpha) * c
                    s_trial = f_trial.sum()
                    if s_trial < 1e-15:
                        continue
                    f_trial = f_trial / s_trial
                    obj_trial = fast_convolve(f_trial).max()
                    if obj_trial < best_alpha_obj - 1e-14:
                        best_alpha_obj = obj_trial
                        best_alpha = alpha

                if abs(best_alpha - f[i] / c) > 1e-6:
                    f[i] = best_alpha * c
                    f[j] = (1.0 - best_alpha) * c
                    s = f.sum()
                    f = f / s
                    if best_alpha_obj < best_obj - 1e-14:
                        best_obj = best_alpha_obj
                        best_score = 2.0 * n * best_obj
                        best_f = f.copy()
                        any_improvement = True

        if not any_improvement:
            no_improve_rounds += 1
            if no_improve_rounds >= 2:
                break
        else:
            no_improve_rounds = 0

        # Periodic perturbation
        if no_improve_rounds == 1:
            f = best_f + rng.standard_normal(n) * 0.002
            f = np.maximum(f, 0.0)
            f = f / f.sum()

    return best_f, best_score


def coord_descent_polish(f_init, time_limit):
    """Coordinate descent with multiplicative updates."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_obj = fast_convolve(f).max()
    best_score = 2.0 * n * best_obj
    best_f = f.copy()

    multipliers = [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5]
    no_pass = 0

    while time.monotonic() - t0 < time_limit:
        any_improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-15:
                continue

            best_j_val = f[j]
            best_j_obj = best_obj
            for mult in multipliers:
                f_trial = f.copy()
                f_trial[j] = f[j] * mult
                s = f_trial.sum()
                if s < 1e-12:
                    continue
                f_trial = f_trial / s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-13:
                    best_j_obj = obj
                    best_j_val = f[j] * mult

            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                f = f / s
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_score = 2.0 * n * best_obj
                    best_f = f.copy()
                    any_improved = True

        if not any_improved:
            no_pass += 1
            if no_pass >= 2:
                break
        else:
            no_pass = 0

    return best_f, best_score


def interp_sequence(src, target_n):
    """Interpolate a sequence to a new length."""
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result


def make_raised_cosine(n):
    """Raised cosine window."""
    x = np.arange(n) / n
    a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    a = np.maximum(a, 0.0)
    return a / a.sum()


def make_hann(n):
    """Hann window."""
    x = np.arange(n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    a = np.maximum(a, 1e-6)
    return a / a.sum()


def make_hamming(n):
    """Hamming window."""
    x = np.arange(n)
    a = 0.54 + 0.46 * np.cos(2 * np.pi * x / (n - 1))
    a = np.maximum(a, 1e-6)
    return a / a.sum()


def make_bell(n, sigma_frac=0.3):
    """Bell-shaped window."""
    x = np.linspace(-1, 1, n)
    a = np.exp(-x**2 / (2 * sigma_frac**2))
    a = np.maximum(a, 1e-6)
    return a / a.sum()


def make_dual_bell(n, sep_frac=0.3):
    """Two-bump distribution."""
    x = np.linspace(-1, 1, n)
    a = np.exp(-((x - sep_frac)**2) / 0.05) + np.exp(-((x + sep_frac)**2) / 0.05)
    a = np.maximum(a, 1e-6)
    return a / a.sum()


def make_flat_top(n, frac=0.6):
    """Flat-top window."""
    x = np.linspace(-1, 1, n)
    a = np.ones(n)
    mask = np.abs(x) > frac
    a[mask] = 0.5 * (1 + np.cos(np.pi * (np.abs(x[mask]) - frac) / (1 - frac)))
    a = np.maximum(a, 1e-6)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    Phase 1: Diverse initializations with quick mirror descent
    Phase 2: LP optimization on best candidates
    Phase 3: Frank-Wolfe + Mirror Descent hybrid
    Phase 4: Pairwise polish
    Phase 5: Final LP burst
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
    except Exception:
        pass

    if parent is not None:
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    else:
        parent = np.ones(512) / 512.0
        p_score = 2.0
        print(f"No parent, using uniform n=512")

    best_seq = parent.copy()
    best_score = p_score

    # ============================================================
    # Phase 1: Diverse initializations with quick mirror descent
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 1: Diverse inits + MD (best: {best_score:.6f})")
    print(f"{'='*60}")

    rem = deadline - time.monotonic()
    t1 = min(rem * 0.28, 500.0)

    if t1 > 30:
        n_parent = len(parent)
        # Generate diverse initializations at various n
        candidates = []

        # Parent interpolated to different lengths
        for n_try in [n_parent, max(512, n_parent - 100), min(2000, n_parent + 150)]:
            n_try = max(256, min(2500, n_try))
            candidates.append((interp_sequence(parent, n_try), "parent"))

        # Structured windows
        for n_try in [n_parent, max(512, n_parent // 2 + 256)]:
            n_try = max(256, min(2500, n_try))
            candidates.append((make_raised_cosine(n_try), "rcos"))
            candidates.append((make_hann(n_try), "hann"))
            candidates.append((make_hamming(n_try), "hamm"))
            candidates.append((make_bell(n_try, 0.25), "bell"))
            candidates.append((make_bell(n_try, 0.35), "bell2"))
            candidates.append((make_dual_bell(n_try, 0.25), "dbl"))
            candidates.append((make_flat_top(n_try, 0.5), "flat"))

        # Random structured
        for n_try in [n_parent, n_parent + 64]:
            n_try = max(256, min(2500, n_try))
            # Smoothed random
            a_raw = rng.standard_normal(n_try)
            # Smooth with a running mean
            kernel = np.ones(10) / 10
            from scipy.signal import convolve as sig_conv
            a_smooth = sig_conv(a_raw, kernel, mode='same')
            a_smooth = np.maximum(a_smooth, 0.0)
            if a_smooth.sum() > 1e-10:
                candidates.append((a_smooth / a_smooth.sum(), "rand"))

        # Deduplicate by length
        seen = set()
        unique_candidates = []
        for cand, name in candidates:
            key = (len(cand), name)
            if key not in seen:
                seen.add(key)
                unique_candidates.append((cand, name))

        per_cand_time = t1 / len(unique_candidates)
        per_cand_time = max(5.0, min(per_cand_time, 60.0))

        for cand, name in unique_candidates:
            rem = deadline - time.monotonic()
            if rem < 10:
                break
            alloc = min(per_cand_time, rem * 0.8)
            if alloc < 5:
                break

            print(f"  MD {name} n={len(cand)} budget={alloc:.0f}s", end=" ")
            f_res, score_est = mirror_descent_opt(
                cand, time_limit=alloc, lr_init=0.03, temp_init=0.003, n_top=50)

            real_score = float(evaluate_sequence(f_res.tolist()))
            if np.isfinite(real_score):
                print(f"score={real_score:.6f}", end="")
                if real_score < best_score:
                    best_score = real_score
                    best_seq = f_res.copy()
                    print(f" *** NEW BEST ***")
                else:
                    print()
            else:
                print("invalid")

    # ============================================================
    # Phase 2: LP optimization on best
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 2: LP deep (best: {best_score:.6f}, n={len(best_seq)})")
    print(f"{'='*60}")

    rem = deadline - time.monotonic()
    t2 = min(rem * 0.22, 400.0)

    if t2 > 20:
        n_b = len(best_seq)
        print(f"  LP n={n_b}, budget={t2:.0f}s")
        f_res, _ = lp_optimize(
            best_seq, time_limit=t2, top_k_init=60, top_k_max=min(500, 2*n_b-1),
            label="P2", momentum_beta=0.3, max_stuck=70)

        real_score = float(evaluate_sequence(f_res.tolist()))
        if np.isfinite(real_score) and real_score < best_score:

    # ============================================================
    # Phase 3: Try different n with LP

    print(f"{'='*60}")

    rem = deadline - time.monotonic()

    if t3 > 20:

        other_ns = [max(300, n_b - 128), max(300, n_b + 128), max(300, int(n_b * 0.8)),
                    max(300, int(n_b * 1.2))]
        # Remove duplicates and sort
        other_ns = sorted(set(n for n in other_ns if abs(n - n_b) > 50))
        per_n = t3 / max(1, len(other_ns))

        for n_try in other_ns:

            f_res, _ = lp_optimize(
                f_init, time_limit=alloc, top_k_init=50, top_k_max=min(400, 2*n_try-1),
                label=f"n{n_try}", momentum_beta=0.3, max_stuck=50)

            real_score = float(evaluate_sequence(f_res.tolist()))
            if np.isfinite(real_score) and real_score < best_score:
                best_score = real_score
                best_seq = f_res.copy()
                print(f"  NEW BEST [P3 n={n_try}]: {real_score:.6f}")

    # ============================================================
    # Phase 4: Frank-Wolfe + Mirror Descent hybrid

        t_md_actual = min(t_md, rem * 0.8)

            f_res, _ = mirror_descent_opt(

    # Phase 6: Coordinate descent polish
