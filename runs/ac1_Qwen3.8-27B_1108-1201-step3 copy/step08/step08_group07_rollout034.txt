Ramp-initialized LP fictitious play for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Key improvements: (1) Linear ramp initialization gives immediate score ~4/3 vs 2 for uniform,
(2) Top-K cutting-plane LP for fast best responses, (3) Line-search step sizes,
(4) Multi-length exploration, (5) Nesterov momentum, (6) L-BFGS + coordinate polish.

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_convolve(f, g):
    """Linear cross-convolution f*g via FFT."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) for unit-sum a."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def ramp_init(n, p=1.0, seed=42):
    """Create a power-law ramp initialization."""
    x = np.arange(1, n + 1, dtype=np.float64)
    a = x ** p
    return a / a.sum()


def build_constraint_rows(f, lags):
    """Build K x n matrix where row k has (f*g)[lag_k] = row @ g."""
    n = len(f)
    K = len(lags)
    A = np.zeros((K, n))
    for idx, k in enumerate(lags):
        # (f*g)[k] = sum_i f[k-i] * g[i] for valid k-i
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            A[idx, lo:hi + 1] = f[k - np.arange(lo, hi + 1)]
    return A


def solve_topK_lp(f, M, lags, prev_g=None, max_iters=None):
    """Solve max sum(g) s.t. (f*g)[lag_k] <= M for each lag in lags, g >= 0."""
    n = len(f)
    K = len(lags)
    A = build_constraint_rows(f, lags)
    g = cp.Variable(n, nonneg=True)
    if prev_g is not None:
        g.value = np.maximum(prev_g, 0)
    constraints = [A @ g <= M * np.ones(K)]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.CLARABEL, warm_start=(prev_g is not None), max_iters=max_iters)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return prev_g if prev_g is not None else f.copy()
    if g.value is None:
        return prev_g if prev_g is not None else f.copy()
    return np.maximum(g.value, 0)


def line_search(a, g, n, prev_score):

    best_t = 0.5

def lp_fictitious_play(a0, time_limit, K=80, verbose=True, momentum=0.0):
    """LP-based fictitious play with top-K cutting plane and line search."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    prev_g = None
    active_lags = np.arange(n)  # Start with all lags (first iteration)
    stall_count = 0
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        M = conv.max()

        # Update active lags: top-K by current conv value
        if iters % 5 == 1 or len(active_lags) < K:
            active_lags = np.argsort(conv)[-min(K, 2 * n - 1):][::-1]

        # Solve LP
        g = solve_topK_lp(a, M, active_lags, prev_g=prev_g)
        prev_g = g
        g_sum = g.sum()

        if g_sum < 1e-10:
            break

        g_norm = g / g_sum

        # Check if other constraints are violated (cutting plane)
        fg = cross_convolve(a, g_norm)
        if fg.max() > M * 1.05:
            violated = np.where(fg > M * 0.95)[0]
            active_lags = np.unique(np.concatenate([active_lags, violated]))
            if len(active_lags) > 2 * K:
                active_lags = np.argsort(fg)[-2 * K:]

        # Line search for step size
        t_opt, sc_trial = line_search(a, g_norm, n, best_score)

        if t_opt > 0:
            a_new = (1 - t_opt) * a + t_opt * g_norm
            a_new = np.maximum(a_new, 0)
            ss = a_new.sum()
            if ss < 1e-15:
                break
            a_new = a_new / ss
            score = compute_score(a_new)

            if score < best_score - 1e-12:
                best_score = score
                best_a = a_new.copy()
                stall_count = 0
            else:
                stall_count += 1

            a = a_new
        else:
            stall_count += 1

        # Handle stalling
        if stall_count > 30:
            # Perturb and reset
            noise = rng.standard_normal(n) * 0.001
            a = best_a + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            prev_g = None
            active_lags = np.argsort(fast_convolve(a))[-K:]
            stall_count = 0

        if verbose and iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"  LP-FP iter={iters}, best={best_score:.8f}, K={len(active_lags)}, time={elapsed:.0f}s")

    return best_a, best_score


def softmax_from_z(z):
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def smoothed_objective(z, T, n):
    """Log-sum-exp smoothed version of max(conv)."""
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    lse = max_shifted + np.log(np.exp(shifted - max_shifted).sum())
    return (2.0 * n / T) * np.exp(T * lse)


def smoothed_gradient(z, T, n):
    """Gradient of the smoothed objective w.r.t. z."""
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    w = np.exp(shifted - max_shifted)
    w = w / w.sum()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    # grad of sum(w_k * conv_k) w.r.t. a: conv_k = sum_i a[i]*a[k-i]
    # d(conv_k)/da[j] = 2*a[k-j] (with bounds)
    # So grad_a[j] = sum_k w_k * 2 * a[k-j] (with bounds)
    # This is 2 * cross_correlation(w, a)
    grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
    S = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - S)
    return (2.0 * n / T) * grad_z


def lbgfs_optimize(a0, time_limit, T=500, maxiter=5000):
    """L-BFGS on smoothed log-sum-exp objective in log-space."""
    n = len(a0)
    z0 = np.log(np.maximum(a0, 1e-30))
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        return smoothed_objective(z, T, n)

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        return smoothed_gradient(z, T, n)

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 30, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
        score = compute_score(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except Exception:
        pass
    return best_a, best_score


def coordinate_descent(a0, time_limit, verbose=True):
    """Coordinate descent with pairwise transfers."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False
        n_pairs = min(200, max(20, n // 4))
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or a[i] < 1e-14:
                continue

            best_local = best_score
            best_t = 0.0
            amount = a[i] * 0.3

            for frac in [0.1, 0.2, 0.3, 0.5, 0.7]:
                t = frac * amount
                a_trial = a.copy()
                a_trial[i] -= t
                a_trial[j] += t
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score(a_trial)
                if sc < best_local - 1e-14:
                    best_local = sc
                    best_t = t

            if best_t != 0.0:
                a[i] -= best_t
                a[j] += best_t
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = best_local
                best_a = a.copy()
                improved = True

        if not improved and iters > 10:
            noise = rng.standard_normal(n) * 0.001
            a = best_a.copy() + noise
            a = np.maximum(a, 0)
            a = a / a.sum()

        if verbose and iters % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"  CD iter={iters}, best={best_score:.8f}, time={elapsed:.0f}s")

    return best_a, best_score


def check_and_update(a, label, best_seq, best_score, verbose=True):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization starting from ramp initialization.
    Phases: LP-FP (main) -> L-BFGS -> CD -> different n -> final LP.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Try parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    # Generate ramp initializations for multiple n values
    n_values = [512, 768, 1024]
    ramp_solutions = {}

    for n in n_values:
        a_ramp = ramp_init(n, p=1.0)
        sc_ramp = compute_score(a_ramp)
        ramp_solutions[n] = (a_ramp, sc_ramp)
        print(f"Ramp n={n}: score={sc_ramp:.8f}")
        if sc_ramp < best_score:
            best_score = sc_ramp
            best_seq = a_ramp.copy()

    # Phase 1: LP-FP on the best ramp (typically n=768)
    n_main = 768
    print(f"\n=== Phase 1: LP-FP main (n={n_main}, best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.40, 600.0)
    print(f"  Budget: {t1:.0f}s")
    a_lp1, sc_lp1 = lp_fictitious_play(ramp_solutions[n_main][0], t1, K=100, verbose=True)
    best_seq, best_score = check_and_update(a_lp1, "lp_main", best_seq, best_score)

    # Phase 2: L-BFGS polish on the LP result
    print(f"\n=== Phase 2: L-BFGS Polish (best={best_score:.8f}) ===")
    for T_val in [200, 500, 1000]:
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t2 = min(rem * 0.15, 100.0)
        print(f"  L-BFGS T={T_val}, budget={t2:.0f}s")
        a_lb, sc_lb = lbgfs_optimize(best_seq, t2, T=T_val, maxiter=3000)
        best_seq, best_score = check_and_update(a_lb, f"lbgfs_T{T_val}", best_seq, best_score)

    # Phase 3: Coordinate descent
    print(f"\n=== Phase 3: Coordinate Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.15, 120.0)
    if rem > 30:
        print(f"  Budget: {t3:.0f}s")
        a_cd, sc_cd = coordinate_descent(best_seq, t3, verbose=True)
        best_seq, best_score = check_and_update(a_cd, "coord_desc", best_seq, best_score)

    # Phase 4: Continue LP-FP from current best
    print(f"\n=== Phase 4: LP-FP continued (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.25, 300.0)
    if rem > 60:
        print(f"  Budget: {t4:.0f}s")
        a_lp2, sc_lp2 = lp_fictitious_play(best_seq, t4, K=80, verbose=True)
        best_seq, best_score = check_and_update(a_lp2, "lp_cont", best_seq, best_score)

    # Phase 5: Try other n values with ramp init + quick LP
    print(f"\n=== Phase 5: Other lengths (best={best_score:.8f}) ===")
    for n_alt in [512, 1024]:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        a_alt = ramp_init(n_alt, p=1.0)
        # Quick LP-FP
        t5 = min(rem * 0.20, 120.0)
        print(f"  LP-FP n={n_alt}, budget={t5:.0f}s")
        a_lp_alt, sc_lp_alt = lp_fictitious_play(a_alt, t5, K=60, verbose=False)
        best_seq, best_score = check_and_update(a_lp_alt, f"lp_n{n_alt}", best_seq, best_score)

        # L-BFGS on this
        rem = deadline - time.monotonic()
        if rem > 30:
            t5b = min(rem * 0.10, 50.0)
            a_lb_alt, sc_lb_alt = lbgfs_optimize(a_lp_alt, t5b, T=500, maxiter=2000)
            best_seq, best_score = check_and_update(a_lb_alt, f"lbgfs_n{n_alt}", best_seq, best_score)

    # Phase 6: Try different ramp powers on the main n
    print(f"\n=== Phase 6: Different ramp powers (best={best_score:.8f}) ===")
    for p_val in [0.7, 1.5, 2.0]:
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        a_p = ramp_init(n_main, p=p_val)
        t6 = min(rem * 0.15, 80.0)
        sc_p = compute_score(a_p)
        print(f"  Ramp p={p_val}: initial score={sc_p:.6f}, budget={t6:.0f}s")
        a_lp_p, sc_lp_p = lp_fictitious_play(a_p, t6, K=80, verbose=False)
        best_seq, best_score = check_and_update(a_lp_p, f"ramp_p{p_val}", best_seq, best_score)

    # Phase 7: Final L-BFGS + CD polish on the best
    print(f"\n=== Phase 7: Final polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t7 = min(rem * 0.30, 80.0)
        a_final_lb, sc_final_lb = lbgfs_optimize(best_seq, t7, T=300, maxiter=3000)
        best_seq, best_score = check_and_update(a_final_lb, "final_lbgfs", best_seq, best_score)

    rem = deadline - time.monotonic()
    if rem > 20:
        t7b = min(rem * 0.30, 60.0)
        a_final_cd, sc_final_cd = coordinate_descent(best_seq, t7b, verbose=False)
        best_seq, best_score = check_and_update(a_final_cd, "final_cd", best_seq, best_score)

    # Phase 8: Final short LP-FP
    print(f"\n=== Phase 8: Final LP (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t8 = min(rem * 0.5, 60.0)
        a_final_lp, sc_final_lp = lp_fictitious_play(best_seq, t8, K=60, verbose=False)
        best_seq, best_score = check_and_update(a_final_lp, "final_lp", best_seq, best_score)

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs) and fs < best_score:
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
