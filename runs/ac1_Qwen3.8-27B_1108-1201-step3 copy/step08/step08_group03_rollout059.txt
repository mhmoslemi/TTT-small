Now, there's a subtlety: the smooth objective `L` is NOT the same as the true objective `max(c)/T`. The true objective (after normalizing sum=1) is `2n * max(c)`, and we want to minimize this. def spectrum_flatten(a_init, n_iters=100, alpha=0.1):
    """Iteratively flatten the power spectrum while maintaining non-negativity."""
    n = len(a_init)
    a = a_init.copy()
    a /= a.sum()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    for it in range(n_iters):
        A = rfft(a, fft_size)
        power = np.abs(A)**2
        # Target: flatten power spectrum
        target = np.ones_like(power)
        # Blend
        new_power = (1 - alpha) * power + alpha * target * (power.sum() / target.sum())
        # New magnitude
        new_mag = np.sqrt(np.maximum(new_power, 0))
        # Preserve phase
        phase = A / (np.abs(A) + 1e-30)
        A_new = new_mag * phase
        a = irfft(A_new, fft_size)[:n]
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            a /= s
        else:
            break
    
    return a

Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses a hybrid approach: smooth gradient descent (log-sum-exp), LP cutting-plane refinement,

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def convolve_cross(f, g):
    """Cross-convolution f*g via FFT."""
    n = len(f)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_gradient(a, T):
    """
    Compute gradient of log-sum-exp(conv(a,a)/T) w.r.t. a.
    Returns (gradient, softmax_weights, current_max).
    """
    n = len(a)
    c = fast_convolve(a)
    c_scaled = c / T
    c_scaled -= c_scaled.max()
    exp_c = np.exp(c_scaled)
    w = exp_c / exp_c.sum()
    # Cross-correlation: g[i] = sum_k w[k] * a[k-i]
    a_rev = a[::-1]
    fft_size = 1 << (3 * n - 1).bit_length()
    W = rfft(w, fft_size)
    A_rev = rfft(a_rev, fft_size)
    g_full = irfft(W * A_rev, fft_size)
    g = g_full[n - 1:2 * n - 1]
    return (2.0 / T) * g, w, c.max()

    best_obj = fast_convolve(a).max()
    v = np.zeros_like(a)
    T = T_init

    it = 0
    no_improve = 0
    rng = np.random.default_rng(42 + n % 1000)

        a_new = project_simplex(a + v)

def build_constraint_matrix(f, indices, n):
    """Build constraint matrix where row i encodes (f*g)[k_i]."""
    K = len(indices)
    A = np.zeros((K, n))
    for i, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    return A


def solve_lp_direction(f, n, M, active_idx):
    """Solve: max sum(g) s.t. g>=0, (f*g)[k] <= M for k in active_idx."""
    K = len(active_idx)
    if K == 0:
        return None
    A = build_constraint_matrix(f, active_idx, n)
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=500000, eps=1e-8)
    except Exception:
        g.value = None
    if g.value is None and n <= 3000:
        try:
            prob.solve(solver=cp.ECOS, max_iters=50000)
        except Exception:
            return None
    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None
    return g_val


def lp_cutting_plane(f_init, time_limit, initial_k=50, max_k=500, label="lp"):
    """LP direction finding with cutting planes."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    b = fast_convolve(f)
    M = b.max()
    best_obj = M

        active_idx = np.argpartition(b, -initial_k)[-initial_k:]
    else:
        active_idx = np.arange(len(b))
    active_idx = np.sort(active_idx)

    it = 0
    stuck = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 3:
            break

        g = solve_lp_direction(f, n, M, active_idx)

        if g is None:
            stuck += 1
            if stuck > 10:
                break
            if len(active_idx) > initial_k:
                active_idx = active_idx[np.argsort(b[active_idx])[-initial_k:]]
            continue

        S_g = g.sum()
        if S_g <= 1.0 + 1e-6:
            stuck += 1
            if stuck > 25:
                break
            if len(active_idx) < max_k:
                b_cur = fast_convolve(f)
                new_top = np.argpartition(b_cur, -min(max_k, len(b_cur)))[-min(max_k, len(b_cur)):]
                active_idx = np.unique(np.concatenate([active_idx, new_top]))
            continue

        # Cutting plane
        fg = convolve_cross(f, g)
        violated = fg > M * (1 + 1e-7)
        if violated.sum() > 0 and len(active_idx) < max_k:
            v_idx = np.where(violated)[0]
            not_active = v_idx[~np.isin(v_idx, active_idx)]
            if len(not_active) > 0:
                n_add = min(40, len(not_active))
                to_add = not_active[np.argsort(fg[not_active])[-n_add:]]
                active_idx = np.unique(np.concatenate([active_idx, to_add]))
                g = solve_lp_direction(f, n, M, active_idx)
                if g is None:
                    stuck += 1
                    if stuck > 5:
                        break
                    continue
                S_g = g.sum()

        if S_g <= 1.0 + 1e-6:
            stuck += 1
            if stuck > 25:
                break
            continue

        stuck = 0
        h = g / S_g

        # Line search
        best_t = 0.05
        best_t_obj = M
        for t in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            f_trial = np.maximum((1.0 - t) * f + t * h, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_t_obj:
                best_t_obj = obj
                best_t = t

        # Golden section refinement
        if best_t > 0.02:
            lo = max(0.0, best_t - 0.05)
            hi = min(1.0, best_t + 0.1)
            for _ in range(5):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                fa1 = np.maximum((1 - m1) * f + m1 * h, 0.0)
                sa1 = fa1.sum()
                if sa1 > 1e-10:
                    fa1 /= sa1
                fa2 = np.maximum((1 - m2) * f + m2 * h, 0.0)
                sa2 = fa2.sum()
                if sa2 > 1e-10:
                    fa2 /= sa2
                o1 = fast_convolve(fa1).max()
                o2 = fast_convolve(fa2).max()
                if o1 < o2:
                    hi = m2
                else:
                    lo = m1
            t_mid = (lo + hi) / 2.0
            f_mid = np.maximum((1 - t_mid) * f + t_mid * h, 0.0)
            s_mid = f_mid.sum()
            if s_mid > 1e-10:
                f_mid /= s_mid
                o_mid = fast_convolve(f_mid).max()
                if o_mid < best_t_obj:
                    best_t_obj = o_mid
                    best_t = t_mid

        f = np.maximum((1.0 - best_t) * f + best_t * h, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s
        else:
            f = np.ones(n) / n

        cur_obj = fast_convolve(f).max()
        M = cur_obj

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_f = f.copy()

        # Refresh active set periodically
        if it % 12 == 0:
            b = fast_convolve(f)
            if len(active_idx) > initial_k * 2:
                active_scores = b[active_idx]
                keep = active_idx[np.argsort(active_scores)[-initial_k * 2:]]
                new_top = np.argpartition(b, -initial_k)[-initial_k:]
                active_idx = np.unique(np.concatenate([keep, new_top]))
                active_idx = active_idx[np.argsort(b[active_idx])[-max_k:]]

        if it % 20 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_obj
            print(f"    {label} it={it} obj={best_obj:.10f} score={score:.6f} "
                  f"active={len(active_idx)} t={elapsed:.0f}s")

    return best_f, best_obj


def spectrum_flatten(a_init, n_iters=200, alpha=0.05, label="spec"):
    """Iteratively flatten the power spectrum while maintaining non-negativity."""
    n = len(a_init)
    a = a_init.copy()
    a /= a.sum()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    for it in range(n_iters):
        A = rfft(a, fft_size)
        power = np.abs(A) ** 2
        # Target: geometric mean of power (flat spectrum)
        log_power = np.log(power + 1e-30)
        target_power = np.exp(np.mean(log_power))
        target_power = np.ones_like(power) * target_power
        # Blend current and target
        blend = alpha
        new_power = (1 - blend) * power + blend * target_power * (power.sum() / target_power.sum())
        new_mag = np.sqrt(np.maximum(new_power, 0))
        # Preserve phase
        abs_A = np.abs(A)
        phase = A / (abs_A + 1e-30)
        A_new = new_mag * phase
        a = irfft(A_new, fft_size)[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            a /= s
        else:
            a = best_a.copy()
            break

        obj = fast_convolve(a).max()
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    return best_a


def hann_init(n):
    """Create a Hann window initialization."""
    x = np.linspace(0, 2 * np.pi, n, endpoint=False)
    return 0.5 * (1 + np.cos(x))


def raised_cosine_init(n, beta=2.0):
    """Raised cosine with adjustable roll-off."""
    x = np.linspace(-1, 1, n)
    return 0.5 * (1 + np.cos(np.pi * x * beta / 2.0))


def triangular_init(n):
    """Triangular (Bartlett) window."""
    x = np.linspace(-1, 1, n)
    return np.maximum(0, 1 - np.abs(x))


def super_gaussian_init(n, p=3.0):
    """Super-Gaussian peak-shaped initialization."""
    x = np.linspace(-1, 1, n)
    return np.exp(-p * x ** 2)


def mixed_init(n, seed=0):
    """Random mixture of different window shapes."""
    rng = np.random.default_rng(seed)
    shapes = [hann_init(n), raised_cosine_init(n, 1.5),
              triangular_init(n), super_gaussian_init(n, 2.0),
              np.ones(n) / n]
    weights = rng.dirichlet(np.ones(len(shapes)))
    a = sum(w * s for w, s in zip(weights, shapes))
    a = np.maximum(a, 0)
    a /= a.sum()
    return a


def refine_sequence(a, time_limit, n_sgd=0.35, n_lp=0.45, n_mwu=0.10, n_spec=0.10):
    """Multi-method refinement of a single sequence."""
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()

    # Phase 1: Smooth gradient descent
    t1 = time_limit * n_sgd
    print(f"  [refine] SGD: {t1:.0f}s")
    a_sgd, obj_sgd = smooth_gradient_descent(a, time_limit=t1, label="sgd")
    score_sgd = 2.0 * n * obj_sgd
    if obj_sgd < best_obj:
        best_obj = obj_sgd
        best_a = a_sgd.copy()
    print(f"  [refine] SGD score: {score_sgd:.6f}")

    # Phase 2: LP cutting plane
    rem = time_limit - (time.monotonic() - t0)
    t2 = rem * (n_lp / (n_lp + n_mwu + n_spec))
    if t2 > 10:
        print(f"  [refine] LP: {t2:.0f}s")
        a_lp, obj_lp = lp_cutting_plane(best_a, time_limit=t2,
                                         initial_k=40, max_k=400, label="lp")
        if obj_lp < best_obj:
            best_obj = obj_lp
            best_a = a_lp.copy()
        print(f"  [refine] LP score: {2.0*n*best_obj:.6f}")

    # Phase 3: MWU
    rem = time_limit - (time.monotonic() - t0)
    t3 = rem * (n_mwu / (n_mwu + n_spec))
    if t3 > 10:
        print(f"  [refine] MWU: {t3:.0f}s")
        a_mwu, obj_mwu = multiplicative_weight_update(best_a, time_limit=t3, label="mwu")
        if obj_mwu < best_obj:
            best_obj = obj_mwu
            best_a = a_mwu.copy()
        print(f"  [refine] MWU score: {2.0*n*best_obj:.6f}")

    # Phase 4: Spectrum flattening + quick LP
    rem = time_limit - (time.monotonic() - t0)
    if rem > 15:
        t4a = rem * 0.3
        t4b = rem * 0.7
        a_spec = spectrum_flatten(best_a, n_iters=100, alpha=0.03)
        obj_spec = fast_convolve(a_spec).max()
        if obj_spec < best_obj:
            best_obj = obj_spec
            best_a = a_spec.copy()
        rem4 = time_limit - (time.monotonic() - t0)
        if rem4 > 10:
            a_lp2, obj_lp2 = lp_cutting_plane(best_a, time_limit=rem4 * 0.8,
                                               initial_k=30, max_k=300, label="lp2")
            if obj_lp2 < best_obj:
                best_obj = obj_lp2
                best_a = a_lp2.copy()
        print(f"  [refine] Spec+LP score: {2.0*n*best_obj:.6f}")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search: multi-initialization, multi-length optimization combining
    smooth gradient descent, LP cutting-plane, MWU, and spectrum flattening.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception as e:
        print(f"Parent load failed: {e}")

    if parent is None:
        parent = hann_init(1024)
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Default: Hann n=1024, score={p_score:.6f}")

    best_seq = parent.copy()
    best_score = p_score

    def try_sequence(a, label=""):
        nonlocal best_seq, best_score
        if a is None or len(a) == 0:
            return
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return
        a = a / s
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {score:.6f} (n={len(a)})")

    # === Phase 1: Refine parent with multi-method ===
    n_parent = len(parent)
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.30, 500.0)
    if t1 > 30:
        print(f"\n=== Phase 1: Refine parent n={n_parent}, budget={t1:.0f}s ===")
        f1, obj1 = refine_sequence(parent, time_limit=t1)
        try_sequence(f1, f"parent_refine n={n_parent}")

    # === Phase 2: Different initializations at parent length ===
    rem = deadline - time.monotonic()
    inits = {
        "hann": hann_init(n_parent),
        "rc": raised_cosine_init(n_parent, 1.8),
        "tri": triangular_init(n_parent),
        "sg": super_gaussian_init(n_parent, 2.5),
        "mixed": mixed_init(n_parent, seed=7),
    }
    # Remove inits too similar to current best
    inits_to_try = []
    for name, init in inits.items():
        # Quick check: is it different from best?
        diff = np.abs(init - best_seq / best_seq.sum()).max()
        if diff > 0.05:
            inits_to_try.append((name, init))

    # Sort by initial objective (better = first)
    inits_to_try.sort(key=lambda x: fast_convolve(x[1]).max())
    inits_to_try = inits_to_try[:3]  # Try at most 3

    for name, init in inits_to_try:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t = min(rem * 0.20, 200.0)
        if t < 20:
            break
        print(f"\n=== Phase 2: Init '{name}' n={n_parent}, budget={t:.0f}s ===")
        f2, obj2 = refine_sequence(init, time_limit=t,
                                     n_sgd=0.40, n_lp=0.35, n_mwu=0.15, n_spec=0.10)
        try_sequence(f2, f"init_{name} n={n_parent}")

    # === Phase 3: Different sequence lengths ===
    rem = deadline - time.monotonic()
    print(f"\n=== Phase 3: Multi-length (remaining={rem:.0f}s) ===")
    base_n = len(best_seq)
    target_ns = []
    for factor in [0.8, 0.9, 1.1, 1.25, 1.5]:
        tn = int(base_n * factor)
        tn = max(512, min(2500, tn))
        if tn != base_n and tn not in target_ns:
            target_ns.append(tn)
    # Always include a couple of "round" numbers
    for rn in [1024, 1536, 2048]:
        if rn not in target_ns and abs(rn - base_n) / base_n < 0.5:
            target_ns.append(rn)
    target_ns.sort(key=lambda x: abs(x - base_n))
    target_ns = target_ns[:3]

    for tn in target_ns:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        t = min(rem * 0.22, 250.0)
        if t < 30:
            break
        # Interpolate best_seq to new length
        old_x = np.linspace(0, 1, base_n)
        new_x = np.linspace(0, 1, tn)
        f_interp = np.interp(new_x, old_x, best_seq)
        f_interp = np.maximum(f_interp, 0.0)
        s = f_interp.sum()
        if s > 1e-10:
            f_interp /= s
        else:
            f_interp = hann_init(tn)

        print(f"  n={tn}, budget={t:.0f}s")
        f3, obj3 = refine_sequence(f_interp, time_limit=t,
                                    n_sgd=0.40, n_lp=0.35, n_mwu=0.15, n_spec=0.10)
        try_sequence(f3, f"n={tn}")

    # === Phase 4: Deep LP on best ===
    rem = deadline - time.monotonic()
    if rem > 30:
        n_b = len(best_seq)
        t4 = min(rem * 0.40, 300.0)
        print(f"\n=== Phase 4: Deep LP n={n_b}, budget={t4:.0f}s ===")
        f4, obj4 = lp_cutting_plane(best_seq, time_limit=t4,
                                     initial_k=80, max_k=600, label="deep_lp")
        try_sequence(f4, f"deep_lp n={n_b}")

    # === Phase 5: Final smooth gradient polish ===
    rem = deadline - time.monotonic()
    if rem > 20:
        n_b = len(best_seq)
        t5 = min(rem * 0.50, 150.0)
        print(f"\n=== Phase 5: Final SGD n={n_b}, budget={t5:.0f}s ===")
        f5, obj5 = smooth_gradient_descent(best_seq, time_limit=t5,
                                            T_init=0.002, T_min=0.0003,
                                            lr_init=0.01, label="final_sgd")
        try_sequence(f5, f"final_sgd n={n_b}")

    # === Phase 6: Final LP ===
    rem = deadline - time.monotonic()
    if rem > 15:
        n_b = len(best_seq)
        t6 = min(rem, 100.0)
        print(f"\n=== Phase 6: Final LP n={n_b}, budget={t6:.0f}s ===")
        f6, obj6 = lp_cutting_plane(best_seq, time_limit=t6,
                                     initial_k=100, max_k=600, label="ult_lp")
        try_sequence(f6, f"ult_lp n={n_b}")

    # Final validation
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'='*55}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*55}")

    return [float(x) for x in best_seq]
