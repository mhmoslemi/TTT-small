"""
Convex optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses LP-based direction finding (min-t formulation) with all convolution
constraints, golden section line search, and multi-length exploration.
Exploits the convexity of the objective for guaranteed global convergence.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction_min_t(f, n):
    """
    Solve: min t s.t. (f*g)[k] <= t for all k, sum(g)=1, g>=0.
    Returns (g, t_value) or (None, None) if failed.
    """
    g = cp.Variable(n)
    t = cp.Variable()
    constraints = [cp.sum(g) == 1, g >= 0]
    
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A_k = np.zeros(n)
        idx = np.arange(lo, hi + 1)
        A_k[idx] = f[k - idx]
        constraints.append(A_k @ g <= t)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    for solver in [cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, eps=1e-9, max_iters=200000)
            if g.value is not None:
                g_val = np.maximum(g.value, 0)
                s = g_val.sum()
                if s > 1e-10:
                    g_val /= s
                return g_val, float(t.value)
        except Exception:
            continue
    
    return None, None


def solve_lp_direction_max_sum(f, n, peaks, M):
    """
    Solve: max sum(g) s.t. g>=0, (f*g)[k] <= M for k in peaks.
    Returns g or None.
    """
    K = len(peaks)
    g = cp.Variable(n)
    constraints = [g >= 0]
    for row_i in range(K):
        k = int(peaks[row_i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A_k = np.zeros(n)
        idx = np.arange(lo, hi + 1)
        A_k[idx] = f[k - idx]
        constraints.append(A_k @ g <= M)
    
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    
    for solver in [cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, eps=1e-9, max_iters=200000)
            if g.value is not None:
                g_val = np.maximum(g.value, 0)
                if g_val.sum() > 1.0 + 1e-5:
                    return g_val
        except Exception:
            continue
    
    return None


def line_search_golden(f, g, M_current, max_iters=30):
    """
    Golden section search on alpha in (0, 1] to minimize
    max(((1-alpha)*f + alpha*g)^2).
    Returns (best_alpha, best_obj).
    """
    phi = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ≈ 0.618
    
    a_lo = 0.0
    a_hi = 1.0
    
    # First, check if alpha=1 gives improvement
    f_hi = g.copy()
    s = f_hi.sum()
    if s > 1e-10:
        f_hi /= s
    obj_hi = fast_convolve(f_hi).max()
    
    if obj_hi >= M_current:
        # No improvement at alpha=1, search in (0, some upper bound)
        a_hi = 0.5
    
    # Golden section
    c = a_hi - phi * (a_hi - a_lo)
    d = a_lo + phi * (a_hi - a_lo)
    
    def obj_at(alpha):
        f_trial = (1 - alpha) * f + alpha * g
        f_trial = np.maximum(f_trial, 0)
        s = f_trial.sum()
        if s < 1e-10:
            return M_current
        f_trial /= s
        return fast_convolve(f_trial).max()
    
    f_c = obj_at(c)
    f_d = obj_at(d)
    
    best_alpha = c if f_c < f_d else d
    best_obj = min(f_c, f_d)
    
    for _ in range(max_iters):
        if f_c < f_d:
            a_hi = d
            d = c
            f_d = f_c
            c = a_hi - phi * (a_hi - a_lo)
            f_c = obj_at(c)
        else:
            a_lo = c
            c = d
            f_c = f_d
            d = a_lo + phi * (a_hi - a_lo)
            f_d = obj_at(d)
        
        if f_c < best_obj:
            best_obj = f_c
            best_alpha = c
        if f_d < best_obj:
            best_obj = f_d
            best_alpha = d
        
        if a_hi - a_lo < 1e-6:
            break
    
    # Also try a few grid points
    for alpha in [0.01, 0.02, 0.05, 0.1, 0.15, 0.2]:
        if alpha > a_hi:
            break
        obj = obj_at(alpha)
        if obj < best_obj:
            best_obj = obj
            best_alpha = alpha
    
    return best_alpha, best_obj

def subgradient_optimize(f_init, time_limit, lr_init=0.003, temp_init=0.005, label="sg"):
    """Subgradient method with softmax weighting on peaks."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    v = np.zeros(n)  # momentum
    best_f = f.copy()
    best_b = fast_convolve(f).max()
    lr = lr_init
    temp = temp_init
    no_improve = 0
    it = 0
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()
        
        # Softmax weights
        scaled = b / (temp * M)
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        
        # Focus on top peaks
        n_top = min(200, len(b))
        top_idx = np.argpartition(b, -n_top)[-n_top:]
        w_sel = w[top_idx].copy()
        ws = w_sel.sum()
        if ws > 1e-30:
            w_sel /= ws
        w = np.zeros(len(b))
        w[top_idx] = w_sel
        
        # Subgradient of max_k (f*f)[k]
        # grad[i] = sum_k w[k] * 2 * f[k-i] (with bounds)
        grad = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-12:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            grad[idx] += 2.0 * w[k] * f[k - idx]
        
        # Momentum update
        v = 0.9 * v - lr * grad
        f_new = f + v
        f = project_simplex(f_new, 1.0)
        
        if it % 50 == 0:
            obj = fast_convolve(f).max()
            if obj < best_b - 1e-12:
                best_b = obj
                best_f = f.copy()
                no_improve = 0
                lr = min(lr * 1.02, 0.02)
            else:
                no_improve += 1
                if no_improve > 8:
                    lr *= 0.93
                    no_improve = 0
            
            if it % 200 == 0:
                temp = min(temp * 1.1, 0.05)
                print(f"    {label} iter={it} M={obj:.12f} lr={lr:.5f} "
                      f"score={2*n*best_b:.6f} elapsed={time.monotonic()-t0:.0f}s")
            
            if no_improve > 40:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002, 1.0)
                v = np.zeros(n)
                lr = lr_init * 0.8
                no_improve = 0
    
    return best_f, best_b


def coordinate_descent(f_init, time_limit, label="cd"):
    """Coordinate descent on individual elements."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_b = fast_convolve(f).max()
    multipliers = np.array([0.5, 0.7, 0.85, 0.92, 0.96, 1.04, 1.08, 1.15, 1.3, 1.5, 2.0])
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-15:
                continue
            best_j_val = f[j]
            best_j_obj = best_b
            for mult in multipliers:
                new_val = f[j] * mult
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-14:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f /= s

def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    n = len(f)

    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new

"""
Convex LP-based optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses min-t LP formulation (minimize max cross-convolution over simplex)
with golden section line search, subgradient polishing, coordinate descent,
and multi-length exploration. Exploits convexity for global optimality.
"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_min_t(f, n, eps=1e-8):
    """
    Solve: min t s.t. (f*g)[k] <= t for all k=0..2n-2, sum(g)=1, g>=0.
    Returns (g_normalized, t_value) or (None, None).
    """
    g = cp.Variable(n)
    t = cp.Variable()
    constraints = [cp.sum(g) == 1, g >= 0]

    # Build all cross-convolution constraints
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        A_k = f[k - idx]
        constraints.append(A_k @ g[idx] <= t)

    prob = cp.Problem(cp.Minimize(t), constraints)

    for solver in [cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, eps=eps, max_iters=200000)
            if g.value is not None:
                g_val = np.maximum(g.value, 0.0)
                s = g_val.sum()
                if s > 1e-10 and np.all(np.isfinite(g_val)):
                    g_val /= s
                    return g_val, float(t.value)
        except Exception:
            continue

    return None, None


def solve_lp_max_sum(f, n, peaks, M, eps=1e-8):
    """
    Fallback LP: max sum(g) s.t. g>=0, (f*g)[k] <= M for k in peaks.
    Returns g or None.
    """
    g = cp.Variable(n)
    constraints = [g >= 0]
    for k_int in peaks:
        k_int = int(k_int)
        lo = max(0, k_int - n + 1)
        hi = min(n - 1, k_int)
        idx = np.arange(lo, hi + 1)
        A_k = f[k_int - idx]
        constraints.append(A_k @ g[idx] <= M)

    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    for solver in [cp.SCS, cp.ECOS]:
        try:
            prob.solve(solver=solver, eps=eps, max_iters=200000)
            if g.value is not None:
                g_val = np.maximum(g.value, 0.0)
                if g_val.sum() > 1.0 + 1e-5 and np.all(np.isfinite(g_val)):
                    return g_val
        except Exception:
            continue
    return None


def line_search(f, g, M_current):
    """
    Find best alpha in (0,1] minimizing max(((1-alpha)*f + alpha*g)^2).
    Returns (best_alpha, best_obj).
    """
    best_alpha = 0.0
    best_obj = M_current

    # Grid search first
    grid = [0.01, 0.02, 0.03, 0.05, 0.07, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]
    for alpha in grid:
        f_trial = (1.0 - alpha) * f + alpha * g
        f_trial = np.maximum(f_trial, 0.0)
        s = f_trial.sum()
        if s < 1e-10:
            continue
        f_trial /= s
        obj = fast_convolve(f_trial).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_alpha = alpha

    # Golden section refinement around best
    if best_alpha > 0:
        lo = max(0.0, best_alpha * 0.5)
        hi = min(1.0, best_alpha * 2.0)
        if hi - lo < 1e-4:
            lo = max(0.0, best_alpha - 0.02)
            hi = min(1.0, best_alpha + 0.02)

        phi = (np.sqrt(5.0) - 1.0) / 2.0
        c = hi - phi * (hi - lo)
        d = lo + phi * (hi - lo)

        def obj_at(a):
            ft = (1.0 - a) * f + a * g
            ft = np.maximum(ft, 0.0)
            s = ft.sum()
            if s < 1e-10:
                return M_current
            ft /= s
            return fast_convolve(ft).max()

        fc = obj_at(c)
        fd = obj_at(d)
        if fc < best_obj:
            best_obj = fc
            best_alpha = c
        if fd < best_obj:
            best_obj = fd
            best_alpha = d

        for _ in range(20):
            if hi - lo < 1e-6:
                break
            if fc < fd:
                hi = d
                d = c
                fd = fc
                c = hi - phi * (hi - lo)
                fc = obj_at(c)
            else:
                lo = c
                c = d
                fc = fd
                d = lo + phi * (hi - lo)
                fd = obj_at(d)
            if fc < best_obj:
                best_obj = fc
                best_alpha = c
            if fd < best_obj:
                best_obj = fd
                best_alpha = d

    return best_alpha, best_obj


def run_lp_optimize(f_init, time_limit, label="lp"):
    """
    Main LP-based optimization loop using min-t formulation.
    Falls back to max-sum LP when min-t shows no progress.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s > 1e-10:
        f = f / s
    best_f = f.copy()
    best_b = fast_convolve(f).max()
    it = 0
    no_improve = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()

        # Try min-t LP
        g, t_val = solve_lp_min_t(f, n, eps=1e-8)

        used_fallback = False
        if g is None or t_val >= M - 5e-9:
            # Fallback: max-sum LP with active peaks
            threshold = 0.95 * M
            peaks = np.where(b >= threshold)[0]
            if len(peaks) > 400:
                peaks = np.argpartition(b, -400)[-400:]
            if len(peaks) < 5:
                peaks = np.argpartition(b, -50)[-50:]

            g2 = solve_lp_max_sum(f, n, peaks, M, eps=1e-8)
            if g2 is not None and g2.sum() > 1.0 + 1e-5:
                g = g2 / g2.sum()
                used_fallback = True
            else:
                no_improve += 1
                if no_improve > 30:
                    # Perturb and restart
                    f = f + rng.standard_normal(n) * 0.001 * f
                    f = np.maximum(f, 0.0)
                    f /= f.sum()
                    no_improve = 0
                    continue
                if it % 10 == 0:
                    print(f"    {label} iter={it} M={M:.10f} stuck={no_improve} "
                          f"score={2*n*best_b:.8f} t={time.monotonic()-t0:.0f}s")
                continue

        # Line search
        best_alpha, best_obj = line_search(f, g, M)

        if best_alpha <= 0:
            no_improve += 1
            if no_improve > 20:
                # Small perturbation
                f = f + rng.standard_normal(n) * 0.0005 * f
                f = np.maximum(f, 0.0)
                f /= f.sum()
                no_improve = 0
            continue

        # Update
        f = (1.0 - best_alpha) * f + best_alpha * g
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s < 1e-10:
            break
        f /= s

        cur_b = fast_convolve(f).max()
        if cur_b < best_b - 1e-12:
            best_b = cur_b
            best_f = f.copy()
            no_improve = 0
            if it % 5 == 0 or best_alpha > 0.3:
                print(f"    {label} iter={it} M={cur_b:.10f} alpha={best_alpha:.4f} "
                      f"fb={used_fallback} score={2*n*best_b:.8f} "
                      f"t={time.monotonic()-t0:.0f}s")
        else:
            no_improve += 1
            if it % 10 == 0:
                print(f"    {label} iter={it} M={cur_b:.10f} alpha={best_alpha:.4f} "
                      f"no_imp={no_improve} score={2*n*best_b:.8f} "
                      f"t={time.monotonic()-t0:.0f}s")

        # Check time and convergence

            # Try a larger perturbation
            f = f + rng.standard_normal(n) * 0.003 * f
            f = np.maximum(f, 0.0)
            f /= f.sum()
            no_improve = 0

    return best_f, best_b


def subgradient_optimize(f_init, time_limit, lr_init=0.003, temp_init=0.004, label="sg"):
    """Subgradient method with adaptive softmax weighting."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    v = np.zeros(n)
    best_f = f.copy()
    best_b = fast_convolve(f).max()
    lr = lr_init
    temp = temp_init
    no_improve = 0
    it = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(f)
        M = b.max()

        # Softmax weights (cold: focus on peaks)
        scaled = b / (temp * M)
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        # Restrict to top peaks
        n_top = min(150, len(b))
        top_idx = np.argpartition(b, -n_top)[-n_top:]
        w_sel = w[top_idx].copy()
        ws = w_sel.sum()
        if ws > 1e-30:
            w_sel /= ws
        w = np.zeros(len(b))
        w[top_idx] = w_sel

        # Subgradient: grad[i] = sum_k w[k] * 2 * f[k-i]
        grad = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-12:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            grad[idx] += 2.0 * w[k] * f[k - idx]

        # Nesterov momentum
        v = 0.9 * v - lr * grad
        f_new = f + v
        f = project_simplex(f_new, 1.0)

        if it % 30 == 0:
            obj = fast_convolve(f).max()
            if obj < best_b - 1e-12:
                best_b = obj
                best_f = f.copy()
                no_improve = 0
                lr = min(lr * 1.015, 0.015)
            else:
                no_improve += 1
                if no_improve > 5:
                    lr *= 0.94
                    no_improve = 0

            if it % 150 == 0:
                temp = min(temp * 1.08, 0.04)
                print(f"    {label} iter={it} obj={obj:.10f} lr={lr:.5f} "
                      f"score={2*n*best_b:.8f} t={time.monotonic()-t0:.0f}s")

            if no_improve > 30:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002, 1.0)
                v = np.zeros(n)
                lr = max(lr_init * 0.7, 0.0005)
                no_improve = 0

    return best_f, best_b


def coordinate_descent(f_init, time_limit, label="cd"):
    """Coordinate descent with adaptive multipliers."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_b = fast_convolve(f).max()
    best_f = f.copy()
    multipliers = np.array([0.5, 0.7, 0.85, 0.93, 0.97, 1.03, 1.07, 1.15, 1.3, 1.5, 2.0])

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-15:
                continue
            best_j_val = f[j]
            best_j_obj = best_b
            for mult in multipliers:
                new_val = f[j] * mult
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-14:
                    best_j_obj = obj
                    best_j_val = new_val
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                if s > 1e-10:
                    f /= s
                if best_j_obj < best_b - 1e-14:
                    best_b = best_j_obj
                    best_f = f.copy()
                    improved = True
        if not improved:
            break

    return best_f, best_b


def interpolate_to_length(f, new_n):
    """Interpolate sequence to new length."""
    n = len(f)
    if n == new_n:
        return f.copy()
    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search: convex LP min-t optimization with multi-phase strategy.
    Phases: LP main -> SG polish -> LP deep -> multi-length -> CD -> final LP.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={p_score:.8f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(1024) / 1024.0
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"No parent, using uniform n=1024, score={p_score:.8f}")

    best_seq = parent.copy()
    best_score = p_score

    # === Phase 1: Main LP optimization on parent ===
    n_parent = len(parent)
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.38, 620.0)
    if t1 > 100:
        print(f"\n{'='*60}")
        print(f"Phase 1: LP min-t (n={n_parent}, budget={t1:.0f}s)")
        print(f"{'='*60}")
        f1, obj1 = run_lp_optimize(parent, time_limit=t1, label="lp1")
        vs1 = float(evaluate_sequence(f1.tolist()))
        if np.isfinite(vs1) and vs1 < best_score:
            best_score = vs1
            best_seq = f1.copy()
            print(f"  *** NEW BEST [phase1]: {vs1:.8f} (n={len(best_seq)})")
        else:
            print(f"  [phase1] score={vs1:.8f} (no improvement)")

    # === Phase 2: Subgradient polish ===
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.12, 150.0)
    if t2 > 30:
        n_b = len(best_seq)
        print(f"\nPhase 2: Subgradient (n={n_b}, budget={t2:.0f}s)")
        f2, obj2 = subgradient_optimize(best_seq, time_limit=t2, lr_init=0.003,

            best_score = vs2
            best_seq = f2.copy()
            print(f"  *** NEW BEST [phase2]: {vs2:.8f}")

    # === Phase 3: Deeper LP ===
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.25, 350.0)
    if t3 > 60:
        n_b = len(best_seq)
        print(f"\nPhase 3: LP deep (n={n_b}, budget={t3:.0f}s)")
        f3, obj3 = run_lp_optimize(best_seq, time_limit=t3, label="lp3")
        vs3 = float(evaluate_sequence(f3.tolist()))
        if np.isfinite(vs3) and vs3 < best_score:
            best_score = vs3
            best_seq = f3.copy()
            print(f"  *** NEW BEST [phase3]: {vs3:.8f}")

    # === Phase 4: Multi-length exploration ===
    rem = deadline - time.monotonic()
    n_base = len(best_seq)
    candidate_lengths = []
    for factor in [0.80, 0.90, 1.10, 1.25, 1.40, 1.60]:
        tl = int(round(n_base * factor))
        tl = max(512, min(2800, tl))
        if tl != n_base and tl not in candidate_lengths:
            candidate_lengths.append(tl)
    candidate_lengths.sort(key=lambda x: abs(x - n_base))

    for target_n in candidate_lengths[:4]:
        rem = deadline - time.monotonic()
        if rem < 100:
            break
        t4 = min(rem * 0.15, 180.0)
        if t4 < 40:
            break
        f_interp = interpolate_to_length(best_seq, target_n)
        print(f"\nPhase 4: LP n={target_n} (budget={t4:.0f}s)")
        f4, obj4 = run_lp_optimize(f_interp, time_limit=t4, label=f"lp_n{target_n}")
        vs4 = float(evaluate_sequence(f4.tolist()))

            best_score = vs4
            best_seq = f4.copy()
            print(f"  *** NEW BEST [n={target_n}]: {vs4:.8f}")

    # === Phase 5: Coordinate descent ===
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.08, 60.0)

        print(f"\nPhase 5: CD (n={n_b}, budget={t5:.0f}s)")
        f5, obj5 = coordinate_descent(best_seq, time_limit=t5, label="cd")
        vs5 = float(evaluate_sequence(f5.tolist()))
