    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_conv_matrix(f):
    """Build (2n-1) x n dense matrix C such that C @ g = conv(f, g)."""
    n = len(f)
    size = 2 * n - 1
    first_col = np.zeros(size)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)


def solve_best_response_lp(f, M, prev_g=None):
    """Solve: max sum(g) s.t. C(f) @ g <= M, g >= 0 using cvxpy."""

        Ct_v = np.zeros(n)
        for j in range(n):
            # C[k,j] = f[k-j] if 0<=k-j<n
            # (C^T v)[j] = sum_{k=j}^{j+n-1} f[k-j] v[k] (truncated)
            # = sum_{m=0}^{n-1} f[m] v[j+m] (truncated to valid v indices)
            idx = j + np.arange(min(n, size - j))
            Ct_v[j] = np.dot(f[:len(idx)], viol[idx])
        # Projected gradient step
        eta = 1.0 / (1.0 + np.linalg.norm(grad))
        g = np.maximum(g - eta * grad, 0)
        gs = g.sum()
        if gs > 1e-15:
            g = g / gs
    # Final check
    Cg = np.convolve(f, g)
    if Cg.max() > M * 1.01:
        # Scale down
        g = g * (M / Cg.max())
        g = g / g.sum()
    return g


def lp_fictitious_play(a0, time_limit, step_mode='adaptive', verbose=True, label="LP-FP"):
    """LP fictitious play with adaptive or fixed step."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    prev_g = None
    stall_count = 0
    use_fast = n > 1024  # Use fast method for large n

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        M = conv.max()

        if use_fast:
            g = solve_best_response_fast(a, M, prev_g)
        else:
            g = solve_best_response_lp(a, M, prev_g)
        prev_g = g
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g_norm = g / g_sum

        if step_mode == 'adaptive':
            t = min(0.7, 1.5 / (1.0 + 0.005 * iters))
        elif step_mode == 'fast':
            t = min(0.8, 2.0 / (1.0 + 0.02 * iters))
        elif step_mode == 'slow':
            t = min(0.3, 0.5 / (1.0 + 0.01 * iters))
        elif isinstance(step_mode, float):
            t = step_mode
        else:
            t = 0.4

        a_new = (1 - t) * a + t * g_norm
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss

        score = compute_score(a_new)
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1

        a = a_new

        if stall_count > 300:
            rng = np.random.default_rng((iters * 7919 + n) % (2**31))
            a = best_a + rng.standard_normal(n) * 0.001
            a = np.maximum(a, 0)
            a = a / a.sum()
            prev_g = None
            stall_count = 0

        if verbose and iters % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, best={best_score:.8f}, t={t:.4f}, time={elapsed:.0f}s/{time_limit:.0f}s")

    return best_a, best_score


def softmax_from_z(z):
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def smoothed_objective(z, T, n):
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    lse = max_shifted + np.log(np.exp(shifted - max_shifted).sum())
    return (2.0 * n / T) * np.exp(T * lse)


def smoothed_gradient(z, T, n):
    a = softmax_from_z(z)
    conv = fast_convolve(a)
    shifted = conv / T
    max_shifted = shifted.max()
    w = np.exp(shifted - max_shifted)
    w = w / w.sum()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
    S = float(np.dot(grad_a, a))
    grad_z = a * (grad_a - S)
    return (2.0 * n / T) * grad_z


def lbgfs_optimize(a0, time_limit, T=500, maxiter=5000, verbose=False):
    """L-BFGS on smoothed log-sum-exp objective in log-space."""
    n = len(a0)
    z0 = np.log(np.maximum(a0, 1e-30))
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        return smoothed_objective(z, T, n)

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        return smoothed_gradient(z, T, n)

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 30, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
        score = compute_score(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except Exception:
        pass
    return best_a, best_score


def subgradient_descent(a0, time_limit, verbose=True, label="SGD"):
    """Subgradient descent on the max of conv(a,a)."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    accumulator = a.copy()
    count = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))
        # Gradient of b[k_star] w.r.t. a: 2 * a[k_star - j] (shifted/reversed)
        grad = np.zeros(n)
        # grad[j] = 2 * a[k_star - j] if 0 <= k_star - j < n
        for j in range(n):
            m = k_star - j
            if 0 <= m < n:
                grad[j] = 2.0 * a[m]
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-30:
            break
        grad = grad / gnorm
        # Step size: eta = c / sqrt(iter)
        eta = 0.1 / np.sqrt(iters + 1)
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        # Averaging
        count += 1
        accumulator += a_new
        avg_a = accumulator / count
        score = compute_score(avg_a)
        if score < best_score - 1e-13:
            best_score = score
            best_a = avg_a.copy()
        a = a_new

        if verbose and iters % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  {label} iter={iters}, best={best_score:.8f}, time={elapsed:.0f}s")

    return best_a, best_score


def greedy_peak_reduction(a0, time_limit, verbose=True):
    """Directly reduce the peak of conv(a,a) using greedy mass transfer."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))
        # Find which positions contribute most to b[k_star]
        contrib = np.zeros(n)
        for j in range(n):
            m = k_star - j
            if 0 <= m < n:
                contrib[j] = a[j] * a[m]
        # Move mass from high-contrib positions to low-contrib ones
        order = np.argsort(contrib)
        # Pick a high-contrib source and low-contrib target
        src = order[-1]
        dst = order[0]
        if src == dst or a[src] < 1e-10:
            continue
        # Try to transfer a small amount
        delta = min(a[src] * 0.1, 0.01)
        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        a_new = a_new / ss
        score = compute_score(a_new)
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            delta = delta * 0.1
            a_new[src] = a[src] - delta
            a_new[dst] = a[dst] + delta
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            score = compute_score(a_new)
            if score < best_score - 1e-14:
                best_score = score
                best_a = a_new.copy()
                a = a_new

        if verbose and iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"  Greedy iter={iters}, best={best_score:.8f}, time={elapsed:.0f}s")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Resample sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def coordinate_descent(a0, time_limit, n_pairs=300, verbose=True):
    """Coordinate descent with pairwise transfers."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or a[i] < 1e-14:
                continue
            base = min(a[i], a[j] * 10) if a[j] > 1e-15 else a[i]
            for frac in [0.03, 0.1, 0.3, 0.6]:
                t = frac * base
                if t >= a[i] - 1e-15:
                    t = a[i] * 0.4
                a_trial = a.copy()
                a_trial[i] -= t
                a_trial[j] += t
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score(a_trial)
                if sc < best_score - 1e-14:
                    best_score = sc
                    a[i] -= t
                    a[j] += t
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    best_a = a.copy()
                    improved = True
                    break

        if not improved and iters > 30:
            noise = rng.standard_normal(n) * 0.003
            a = best_a.copy() + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            best_score = compute_score(a)
            if best_score > compute_score(best_a):
                best_score = compute_score(best_a)

    return best_a, best_score


def check_and_update(a, label, best_seq, best_score, verbose=True):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining LP fictitious play, smoothed L-BFGS,
    subgradient descent, greedy peak reduction, and multi-start exploration.
    """
    deadline = time.monotonic() + max(10.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 768
        parent = np.ones(n_init) / n_init
        best_seq = parent.copy()
        best_score = compute_score(parent)
        print(f"Initialized: n={n_init}, score={best_score:.8f}")

    n_parent = len(parent)

    # Phase 1: LP-FP with fast adaptive step (main workhorse)
    print(f"\n=== Phase 1: LP-FP Fast (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.35, 550.0)
    print(f"  Budget: {t1:.0f}s, n={n_parent}")
    if rem > 30:
        a1, s1 = lp_fictitious_play(best_seq, t1, step_mode='fast', label="LP-fast")
        best_seq, best_score = check_and_update(a1, "lp_fast", best_seq, best_score)

    # Phase 2: LP-FP with standard adaptive step
    print(f"\n=== Phase 2: LP-FP Adaptive (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.25, 400.0)
    if rem > 30:
        print(f"  Budget: {t2:.0f}s")
        a2, s2 = lp_fictitious_play(best_seq, t2, step_mode='adaptive', label="LP-adapt")
        best_seq, best_score = check_and_update(a2, "lp_adapt", best_seq, best_score)

    # Phase 3: L-BFGS polishing with multiple temperatures
    print(f"\n=== Phase 3: L-BFGS Polish (best={best_score:.8f}) ===")
    for T_val in [200, 800, 2000]:
        rem = deadline - time.monotonic()
        t3 = min(rem * 0.08, 60.0)
        if rem < 20:
            break
        print(f"  L-BFGS T={T_val}, budget={t3:.0f}s")
        a3, s3 = lbgfs_optimize(best_seq, t3, T=T_val, maxiter=3000)
        best_seq, best_score = check_and_update(a3, f"lbgfs_T{T_val}", best_seq, best_score)

    # Phase 4: Subgradient descent
    print(f"\n=== Phase 4: Subgradient (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.08, 50.0)
    if rem > 20:
        a4, s4 = subgradient_descent(best_seq, t4, verbose=False)
        best_seq, best_score = check_and_update(a4, "subgrad", best_seq, best_score)

    # Phase 5: LP-FP with slow step for fine-tuning
    print(f"\n=== Phase 5: LP-FP Slow (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.15, 150.0)
    if rem > 20:
        print(f"  Budget: {t5:.0f}s")
        a5, s5 = lp_fictitious_play(best_seq, t5, step_mode='slow', label="LP-slow")
        best_seq, best_score = check_and_update(a5, "lp_slow", best_seq, best_score)

    # Phase 6: Coordinate descent
    print(f"\n=== Phase 6: Coordinate Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.08, 50.0)
    if rem > 15:
        a6, s6 = coordinate_descent(best_seq, t6, n_pairs=200, verbose=False)
        best_seq, best_score = check_and_update(a6, "coord_desc", best_seq, best_score)

    # Phase 7: Multi-start with structured perturbations
    print(f"\n=== Phase 7: Multi-Start (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        n_starts = min(5, int(rem / 30))
        rng = np.random.default_rng(seed)
        for idx in range(n_starts):
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            # Structured perturbation
            a_start = best_seq.copy()
            noise_type = idx % 3
            if noise_type == 0:
                # Gaussian noise
                a_start = a_start + rng.standard_normal(len(a_start)) * 0.005
            elif noise_type == 1:
                # Move mass from edges to center
                a_start = a_start.copy()
                edge = max(1, len(a_start) // 10)
                mass_edge = a_start[:edge].sum() + a_start[-edge:].sum()
                a_start[:edge] *= 0.5
                a_start[-edge:] *= 0.5
                center = len(a_start) // 2
                a_start[center - edge:center + edge] += mass_edge / (2 * edge)
            else:
                # Cosine modulation
                x = np.linspace(0, 2 * np.pi, len(a_start))
                mod = 1.0 + 0.1 * np.cos(3 * x)
                a_start = a_start * mod
            a_start = np.maximum(a_start, 0)
            a_start = a_start / a_start.sum()
            t_start = min(rem * 0.4, 40.0)
            a_m, s_m = lp_fictitious_play(a_start, t_start, step_mode='fast',
                                           label=f"MS{idx}", verbose=False)
            best_seq, best_score = check_and_update(a_m, f"multistart{idx}",
                                                     best_seq, best_score)

    # Phase 8: Try different n values
    print(f"\n=== Phase 8: Different n (best={best_score:.8f}) ===")
    for alt_n in [n_parent + 2, n_parent + 4, n_parent - 2, 512, 1024]:
        if alt_n < 64 or alt_n > 2048 or alt_n == n_parent:
            continue
        rem = deadline - time.monotonic()
        if rem < 25:
            break
        t8 = min(rem * 0.2, 40.0)
        init_alt = interpolate_sequence(best_seq, alt_n)
        print(f"  Trying n={alt_n}, budget={t8:.0f}s")
        a_alt, sc_alt = lp_fictitious_play(init_alt, t8, step_mode='fast',
                                            label=f"n{alt_n}", verbose=False)
        best_seq, best_score = check_and_update(a_alt, f"n{alt_n}", best_seq, best_score)

    # Phase 9: Final LP polish
    print(f"\n=== Phase 9: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t9 = min(rem * 0.5, 60.0)
        a_final, sc_final = lp_fictitious_play(best_seq, t9, step_mode='slow',
                                               label="LP-final", verbose=False)
        best_seq, best_score = check_and_update(a_final, "lp_final", best_seq, best_score)
        # One more L-BFGS
        rem = deadline - time.monotonic()
        if rem > 15:
            a_lb, sc_lb = lbgfs_optimize(best_seq, min(rem * 0.7, 30.0), T=500, maxiter=1000)
            best_seq, best_score = check_and_update(a_lb, "lbgfs_final", best_seq, best_score)

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")

    return [float(x) for x in best_seq]
