"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP direction finding, Nesterov accelerated gradient, spectral polishing,
and simulated annealing to escape local optima.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time
import sys

try:
    import cvxpy as cp
    HAS_CVXPY = True
except:
    HAS_CVXPY = False


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f, indices, n):
    """Build K x n matrix where row k represents (f*g)_k as a linear form in g."""
    K = len(indices)
    A = np.zeros((K, n))
    for i in range(K):
        k = indices[i]
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    return A


def solve_lp(f, indices, M, timeout=5.0):
    """
    Solve LP: max sum(g) s.t. A@g <= M, g >= 0.
    Returns g or None.
    """
    if not HAS_CVXPY:
        return None
    n = len(f)
    K = len(indices)
    if K == 0 or K > 2000:
        return None
    
    A = build_lp_matrix(f, indices, n)
    g = cp.Variable(n)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), [cp.sum(A @ g, axis=0) <= M, g >= 0])
    
    t_start = time.monotonic()
    try:
        if K <= 300 and n <= 2000:
            prob.solve(solver=cp.ECOS, max_iters=10000, abs_acceleration=1e-8)
        else:
            prob.solve(solver=cp.SCS, max_iters=50000, eps_abs=1e-7, eps_rel=1e-7)
    except:
        try:
            prob.solve(solver=cp.SCS, max_iters=30000, eps_abs=1e-6, eps_rel=1e-6)
        except:
            return None
    
    if time.monotonic() - t_start > timeout:
        return None
    
    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
        return None
    return g_val


def compute_smoothed_gradient(a, b, tau):
    """
    Compute gradient of smoothed max: tau * log(sum exp(b_k / tau)).
    Returns (gradient, weights).
    """
    # Softmax weights
    scaled = (b - b.max()) / tau
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        idx = np.argmax(b)
        w = np.zeros(len(b))
        w[idx] = 1.0
    else:
        w = w / w_sum
    
    n = len(a)
    # gradient_j = 2 * sum_m a_m * w_{j+m}
    # This is cross-correlation: c_j = sum_m a_m * w_{j+m}
    # Can compute via FFT: c = ifft(fft(a, N) * conj(fft(w, N)))[:n]... 
    # Actually: c_j = sum_m a_m * w_{j+m} = (reverse(a) * w)_{n-1+j}
    # Let a_rev[i] = a[n-1-i]
    # Then (a_rev * w)[n-1+j] = sum_m a_rev[m] * w[n-1+j-m] = sum_m a[n-1-m] * w[n-1+j-m]
    # Let l = n-1-m: = sum_l a[l] * w[j+l]
    # Yes! So c_j = (a_rev * w)[n-1+j]
    
    N = 1 << (n + len(w) - 1 - 1).bit_length()
    a_rev = a[::-1]
    F_a = fft(a_rev, N)
    F_w = fft(w, N)
    c = ifft(F_a * F_w, N)[:n + len(w) - 1]
    # c_j for j = 0..n-1 is at index n-1+j
    grad = 2.0 * c[n-1:n-1+n]
    return grad, w


def nesterov_optimize(a_init, time_limit, tau_init=0.5, tau_decay=0.95, tau_min=0.01):
    """
    Nesterov accelerated proximal gradient with smoothed max.
    """
    t0 = time.monotonic()
    n = len(a_init)
    x = a_init.copy()
    s = x.sum()
    if s > 1e-15:
        x = x / s
    else:
        x = np.ones(n) / n
    
    y = x.copy()
    t_param = 1.0
    tau = tau_init
    best_x = x.copy()
    best_obj = fast_convolve(x).max()
    no_improve = 0
    rng = np.random.default_rng(12345)
    
    # Estimate Lipschitz constant
    L = 2.0 + 4.0 / tau
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(y)
        step = 1.0 / L
        grad, w = compute_smoothed_gradient(y, b, tau)
        
        # Proximal step
        y_new = y - step * grad
        x_new = project_simplex(y_new)
        
        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param)) / 2.0
        beta = (1.0 - t_param) / t_new
        y = x_new + beta * (x_new - x)
        x = x_new
        t_param = t_new
        
        # Update tau
        if t_param > 50:
            tau = max(tau * tau_decay, tau_min)
            L = 2.0 + 4.0 / tau
            if t_param > 500:
                t_param = 1.0  # Reset momentum to escape
        
        # Check improvement
        if t_param % 10 == 0 or t_param < 20:
            obj = fast_convolve(x).max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_x = x.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 20:
                    # Restart with perturbation
                    x = project_simplex(best_x + rng.standard_normal(n) * 0.01)
                    y = x.copy()
                    t_param = 1.0
                    no_improve = 0
    
    return best_x, 2.0 * n * best_obj


def spectral_polish(a_init, time_limit, n_iters=5000):
    """
    Gerchberg-Saxton style spectral polishing.
    """
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    rng = np.random.default_rng(999)
    
    fft_size = 1 << (2*n - 2).bit_length()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # FFT
        A = fft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Flatten magnitude: use a soft power compression
        # Target: make mag closer to its geometric mean
        log_mag = np.log(mag + 1e-15)
        log_mean = np.mean(log_mag)
        # Compress toward mean
        alpha = 0.05  # Compression strength
        new_log_mag = log_mean + (1 - alpha) * (log_mag - log_mean)
        new_mag = np.exp(new_log_mag)
        
        # Construct new FFT
        A_new = new_mag * np.exp(1j * phase)
        
        # Inverse FFT

"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines enhanced LP direction finding, Nesterov accelerated gradient,
spectral polishing, and simulated annealing to find sequences with
score below 1.5027.
"""

import numpy as np
from scipy.fft import rfft, irfft, fft, ifft
import time
import sys

try:
    import cvxpy as cp
    HAS_CVXPY = True
except ImportError:
    HAS_CVXPY = False


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f, indices, n):
    """Build K x n matrix where row k gives coefficients of (f*g)_k as linear form in g."""
    K = len(indices)
    A = np.zeros((K, n))
    for i in range(K):
        k = int(indices[i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    return A


def solve_lp_direction(f, indices, M):
    """
    Solve LP: max sum(g) s.t. (f*g)_k <= M for k in indices, g >= 0.
    Returns g (unnormalized) or None.
    """
    if not HAS_CVXPY:
        return None
    n = len(f)
    K = len(indices)
    if K == 0 or K > 3000 or n > 5000:
        return None

    A = build_lp_matrix(f, indices, n)
    g = cp.Variable(n)
    constraints = [g >= 0]
    # A @ g <= M (element-wise)
    constraints.append(A @ g <= M)

    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        if K <= 500 and n <= 2500:
            prob.solve(solver=cp.ECOS, max_iters=15000, abs_acceleration=1e-9)
        else:
            prob.solve(solver=cp.SCS, max_iters=80000, eps_abs=1e-8, eps_rel=1e-8)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=50000, eps_abs=1e-7, eps_rel=1e-7)
        except Exception:
            return None

    if g.value is None:
        return None
    g_val = np.asarray(g.value, dtype=np.float64).ravel()
    g_val = np.maximum(g_val, 0.0)
    if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
        return None
    if g_val.sum() < 1e-30:
        return None
    return g_val


def solve_lp_dual_direction(f, indices, M):
    """
    Alternative: solve dual LP for direction.
    Dual: min M * sum(lam) s.t. A^T lam >= 1, lam >= 0.
    The primal optimal value = dual optimal value.
    If dual optimal > 1, there exists g with sum(g) > 1.
    We extract a direction from the complementary slackness.
    """
    if not HAS_CVXPY:
        return None
    n = len(f)
    K = len(indices)
    if K == 0 or K > 2000:
        return None

    A = build_lp_matrix(f, indices, n)
    # Dual: min M * sum(lam) s.t. A.T @ lam >= 1, lam >= 0
    lam = cp.Variable(K)
    prob = cp.Problem(cp.Minimize(M * cp.sum(lam)), [A.T @ lam >= 1, lam >= 0])

    try:
        if K <= 500:
            prob.solve(solver=cp.ECOS, max_iters=15000, abs_acceleration=1e-9)
        else:
            prob.solve(solver=cp.SCS, max_iters=80000, eps_abs=1e-8, eps_rel=1e-8)
    except Exception:
        return None

    if lam.value is None:
        return None
    lam_val = np.maximum(np.asarray(lam.value, dtype=np.float64).ravel(), 0.0)

    # Complementary slackness: g_j > 0 only if (A^T lam)_j = 1
    # So g is supported on {j : (A^T lam)_j == 1}
    at_lam = A.T @ lam_val
    support = np.where(at_lam >= 1.0 - 1e-6)[0]
    if len(support) == 0:
        support = np.where(at_lam >= 1.0 - 1e-4)[0]
    if len(support) == 0:
        # Fallback: use all
        support = np.arange(n)

    # On the support, we need to find g such that A@g <= M and g_j = 0 outside support
    # Simple approach: uniform on support
    g = np.zeros(n)
    g[support] = 1.0 / len(support)
    return g


def line_search_direction(f, direction, current_max, n_candidates=15):
    """
    Find best interpolation t that minimizes max(conv((1-t)f+td, (1-t)f+td)).
    direction should be on the simplex.
    """
    best_t = 0.0
    best_obj = current_max
    ts = np.linspace(0.05, 0.95, n_candidates)

    for t in ts:
        f_trial = (1.0 - t) * f + t * direction
        f_trial = np.maximum(f_trial, 0.0)
        s = f_trial.sum()
        if s < 1e-15:
            continue
        f_trial = f_trial / s
        obj = fast_convolve(f_trial).max()
        if obj < best_obj:
            best_obj = obj
            best_t = t

    # Refine around best
    if best_t > 0:
        ts_fine = np.linspace(max(0.01, best_t - 0.08), min(0.99, best_t + 0.08), 8)
        for t in ts_fine:
            f_trial = (1.0 - t) * f + t * direction
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-15:
                continue
            f_trial = f_trial / s
            obj = fast_convolve(f_trial).max()
            if obj < best_obj:
                best_obj = obj
                best_t = t

    return best_t, best_obj

def nesterov_optimize(a_init, time_limit, label="NEG"):
    """
    Nesterov accelerated proximal gradient with smoothed max.
    Uses backtracking line search for robustness.
    """
    t0 = time.monotonic()
    n = len(a_init)
    x = a_init.copy()
    s = x.sum()
    if s > 1e-15:
        x = x / s
    else:
        x = np.ones(n) / n

    y = x.copy()
    t_param = 1.0
    tau = 1.0  # Smoothing parameter
    best_x = x.copy()
    best_obj = fast_convolve(x).max()
    no_improve = 0
    rng = np.random.default_rng(2024)
    full_len = 2 * n - 1

    # Precompute FFT size for cross-correlation
    fft_N = 1 << (3 * n).bit_length()

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 3:
            break

        b = fast_convolve(y)

        # Compute smoothed gradient
        scaled = (b - b.max()) / tau
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(full_len)
            w[np.argmax(b)] = 1.0
        else:
            w = w / w_sum

        # Gradient via cross-correlation
        a_rev = y[::-1]
        F_a = fft(a_rev, fft_N)
        F_w = fft(w, fft_N)
        c = np.real(ifft(F_a * F_w, fft_N))
        grad = 2.0 * c[n - 1:n - 1 + n]

        # Backtracking line search
        step = 0.5
        accepted = False
        cur_obj_y = b.max()
        for _ in range(20):
            y_new = y - step * grad
            x_new = project_simplex(y_new)
            # Check sufficient decrease (Armijo)
            obj_new = fast_convolve(x_new).max()
            if obj_new < cur_obj_y - 1e-6 or step < 0.001:
                accepted = True
                break
            step *= 0.5

        if not accepted:
            x_new = project_simplex(y - 0.001 * grad)

        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param))
        beta = (1.0 - t_param) / t_new
        y = x_new + beta * (x_new - x)
        x = x_new
        t_param = t_new

        # Decrease tau periodically
        if t_param % 20 == 0:
            tau = max(tau * 0.9, 0.02)

        # Track best
        obj_x = fast_convolve(x).max()
        if obj_x < best_obj - 1e-12:
            best_obj = obj_x
            best_x = x.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                # Restart
                x = project_simplex(best_x + rng.standard_normal(n) * 0.01)
                y = x.copy()
                t_param = 1.0
                tau = max(0.5, tau)
                no_improve = 0

        if t_param % 50 == 0:
            print(f"    {label} t={t_param:.0f} obj={best_obj:.12f} "
                  f"score={2.0*n*best_obj:.6f} tau={tau:.4f} "
                  f"elapsed={time.monotonic()-t0:.0f}s")

    return best_x, 2.0 * n * best_obj


def spectral_polish(a_init, time_limit, label="SPEC"):
    """
    Spectral polishing via iterative magnitude flattening in frequency domain.
    """
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    fft_size = 1 << (2 * n - 1).bit_length()
    rng = np.random.default_rng(42)
    alpha = 0.03  # Compression strength
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 2:
            break

        A = fft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)

        # Flattening: compress log-magnitude toward mean
        log_mag = np.log(mag + 1e-15)
        # Use a weighted mean (exclude DC which is special)
        log_mean = np.mean(log_mag)
        new_log_mag = log_mean + (1.0 - alpha) * (log_mag - log_mean)
        new_mag = np.exp(new_log_mag)

        # Ensure DC component preserves sum
        # A[0] = sum(a) = 1 (on simplex)
        # Don't modify DC
        new_mag[0] = mag[0]

        A_new = new_mag * np.exp(1j * phase)
        a_new = np.real(ifft(A_new, fft_size))[:n]
        a_new = project_simplex(a_new)

        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            no_improve += 1
            a = a_new  # Still update to explore
            if no_improve > 100:
                # Perturb
                a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
                no_improve = 0
                alpha *= 0.9  # Reduce compression

        if (time.monotonic() - t0) % 5 < 0.1:
            pass  # Print less frequently

    return best_a, 2.0 * n * best_obj

def interp_sequence(src, target_n):
    """Interpolate a sequence to a new length."""
    src_n = len(src)
    if src_n == target_n:
        return src.copy()
    old_x = np.linspace(0.0, 1.0, src_n)
    new_x = np.linspace(0.0, 1.0, target_n)
    result = np.interp(new_x, old_x, src)
    result = np.maximum(result, 0.0)
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = np.ones(target_n) / target_n
    return result

        print(f"No parent, using uniform n=512, score={p_score:.6f}")

    best_seq = parent.copy()
    best_score = p_score

    def try_update(seq, context=""):
        nonlocal best_seq, best_score
        if seq is None or len(seq) == 0:
            return
        seq = np.asarray(seq, dtype=np.float64)
        seq = np.clip(seq, 0.0, 1000.0)
        s = seq.sum()
        if s < 1e-10:
            return
        try:
            sc = float(evaluate_sequence(seq.tolist()))
        except Exception:
            sc = np.inf
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print(f"  NEW BEST [{context}]: score={sc:.6f} n={len(seq)}")

    # ============================================================
    # Phase A: Enhanced LP (40% of budget)
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase A: Enhanced LP (budget target: {best_score:.6f})")
    print(f"{'='*60}")

    tA = total_budget * 0.40
    rem = deadline - time.monotonic()
    tA = min(tA, rem)

    if tA > 30:
        n_par = len(parent)
        # Run LP on parent length
        print(f"  LP on n={n_par}, time={tA:.0f}s")
        f_res, _ = run_lp_optimization(parent, time_limit=tA, label="LPA")
        try_update(f_res, f"LP n={n_par}")

        # If we have time, run on a slightly different n
        rem = deadline - time.monotonic()
        if rem > 30:
            n_alt = int(n_par * 1.1)
            f_init = interp_sequence(best_seq, n_alt)
            tA2 = min(rem * 0.4, tA * 0.4)
            if tA2 > 15:
                print(f"  LP on n={n_alt}, time={tA2:.0f}s")
                f_res2, _ = run_lp_optimization(f_init, time_limit=tA2, label="LPB")
                try_update(f_res2, f"LP n={n_alt}")

    # ============================================================
    # Phase B: Nesterov accelerated gradient (20% of budget)
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase B: Nesterov gradient (best: {best_score:.6f})")
    print(f"{'='*60}")

    rem = deadline - time.monotonic()
    tB = min(total_budget * 0.20, rem)

    if tB > 30:
        n_b = len(best_seq)
        print(f"  NEG on n={n_b}, time={tB:.0f}s")
        f_res, _ = nesterov_optimize(best_seq, time_limit=tB, label="NEG")
        try_update(f_res, f"NEG n={n_b}")

    # ============================================================
    # Phase C: Spectral polish (10% of budget)
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase C: Spectral polish (best: {best_score:.6f})")
    print(f"{'='*60}")

    rem = deadline - time.monotonic()
    tC = min(total_budget * 0.10, rem)

    if tC > 15:
        n_c = len(best_seq)
        print(f"  Spectral on n={n_c}, time={tC:.0f}s")
        f_res, _ = spectral_polish(best_seq, time_limit=tC, label="SPEC")
        try_update(f_res, f"SPEC n={n_c}")

    # ============================================================
    # Phase D: Coordinate descent (10% of budget)
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase D: Coordinate descent (best: {best_score:.6f})")
    print(f"{'='*60}")

    rem = deadline - time.monotonic()
    tD = min(total_budget * 0.10, rem)

    if tD > 10:
        n_d = len(best_seq)
        print(f"  CD on n={n_d}, time={tD:.0f}s")
        f_res, _ = coordinate_descent_polish(best_seq, time_limit=tD, label="CD")
        try_update(f_res, f"CD n={n_d}")

    # ============================================================
    # Phase E: Multi-n exploration + final LP burst (20% of budget)
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase E: Multi-n + final (best: {best_score:.6f})")
    print(f"{'='*60}")

    rem = deadline - time.monotonic()
    tE = min(total_budget * 0.20, rem)

    if tE > 20:
        n_base = len(best_seq)

        # Try a few different n values
        n_trials = set()
        for factor in [0.8, 0.9, 1.0, 1.1, 1.2, 1.35]:
            n_t = int(n_base * factor)
            n_t = max(256, min(3000, n_t))
            n_trials.add(n_t)
        n_trials = sorted(n_trials)

        per_n = tE * 0.5 / len(n_trials)

        for n_t in n_trials:
            rem = deadline - time.monotonic()
            if rem < 10:
                break
            alloc = min(per_n, rem * 0.8)
            if alloc < 8:
                break

            if n_t == len(best_seq):
                continue  # Already optimized this

            f_init = interp_sequence(best_seq, n_t)
            print(f"  Multi-n: n={n_t}, time={alloc:.0f}s")
            f_res, _ = run_lp_optimization(f_init, time_limit=alloc, label=f"MN{n_t}")
            try_update(f_res, f"Multi-n {n_t}")

        # Final LP burst on best
        rem = deadline - time.monotonic()
        if rem > 15:
            n_final = len(best_seq)
            t_final = min(rem * 0.9, 120.0)
            print(f"  Final LP: n={n_final}, time={t_final:.0f}s")
            f_res, _ = run_lp_optimization(best_seq, time_limit=t_final, label="FIN")
            try_update(f_res, "Final")

            # Quick CD after
            rem = deadline - time.monotonic()
            if rem > 5:
                t_cd = min(rem * 0.5, 30.0)
                if t_cd > 3:
                    f_res2, _ = coordinate_descent_polish(best_seq, time_limit=t_cd, label="CD2")
                    try_update(f_res2, "CD2")

    # Final verification
    rem = deadline - time.monotonic()
    if rem < 1:
        # Emergency: return what we have
        fs = float(evaluate_sequence(best_seq.tolist()))
        if not np.isfinite(fs):
            best_seq = parent.copy()
        return [float(x) for x in best_seq]

    # Do a final LP burst if we have significant time
    rem = deadline - time.monotonic()
    if rem > 30:
        print(f"\n  Extra time: {rem:.0f}s, running final LP burst")
        n_final = len(best_seq)
        t_extra = rem * 0.9
        f_res, _ = run_lp_optimization(best_seq, time_limit=t_extra, label="EXTRA")
        try_update(f_res, "EXTRA")

    # Final check
    fs = float(evaluate_sequence(best_seq.tolist()))
    if not np.isfinite(fs):
        best_seq = parent.copy()
        fs = float(evaluate_sequence(best_seq.tolist()))
    best_score = fs

    print(f"\n{'='*60}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*60}")

    return [float(x) for x in best_seq]
