import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(f, g):
    """Cross-convolution f*g via FFT."""
    size = len(f) + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_full_constraint_matrix(n, f):

    A = np.zeros((2 * n - 1, n))

        return None

    A = np.zeros((K, n))
    for row_i in range(K):
        k = int(k_values[row_i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            A[row_i, lo:hi + 1] = f[k - hi:k - lo + 1]

    g_var = cp.Variable(n)
    constraints = [g_var >= 0, A @ g_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)

    g_val = None
    for solver in [cp.PDLP, cp.SCS]:
        try:
            prob.solve(solver=solver, verbose=False)
            if g_var.value is not None and not np.any(np.isnan(g_var.value)):
                g_val = np.maximum(g_var.value, 0.0)
                break
        except Exception:
            continue

    return g_val


def coordinate_descent_polish(f, max_rounds=3, time_limit=30.0):
    """Polish using coordinate descent on the normalized peak objective."""
    t0 = time.monotonic()
    n = len(f)
    f = f.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max() / (f.sum() ** 2)

    rng = np.random.default_rng(12345)

    for rnd in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        order = rng.permutation(n)
        improved = False

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break

            # Try several values for f[i]
            current = f[i]
            s_other = f.sum() - current

            candidates = [current * 0.5, current * 0.8, current, current * 1.2, current * 1.5,
                          current + 0.01, current - 0.01, 0.0]
            # Add neighborhood
            if current > 0:
                candidates.extend([current * 0.9, current * 1.1])

            best_val = current
            best_obj_i = best_obj * f.sum() ** 2  # track raw peak relative to current sum

            for trial in candidates:
                if trial < 0:
                    trial = 0.0
                f_trial = f.copy()
                f_trial[i] = trial
                total_sum = f_trial.sum()
                if total_sum < 1e-10:
                    continue
                # Normalize
                f_norm = f_trial / total_sum
                peak = fast_convolve(f_norm).max()
                if peak < best_obj:
                    best_obj = peak
                    best_val = trial
                    improved = True

            f[i] = best_val
            f = f / f.sum()

        if not improved:
            break

    f = f / f.sum()
    return f


def global_descent_search(f, n_peaks=5, time_limit=60.0):
    """Steepest descent on top peaks with adaptive step size and momentum."""
    t0 = time.monotonic()
    n = len(f)
    f = f.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    momentum = np.zeros(n)
    beta = 0.7

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(f)
        if n_peaks < len(b):
            top_idx = np.argpartition(b, -n_peaks)[-n_peaks:]
        else:
            top_idx = np.arange(len(b))

        # Compute gradient direction (direction increasing peak)
        grad = np.zeros(n)
        for k in top_idx:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                grad[j] += f[k - j]

        if grad.max() < 1e-15:
            break

        # Nesterov-like momentum
        momentum = beta * momentum - grad
        direction = -momentum

        # Try step sizes
        best_alpha = None
        best_trial = best_obj
        for alpha in [0.001, 0.003, 0.005, 0.01, 0.02, 0.05, 0.1]:
            f_trial = f + alpha * direction
            f_trial = project_simplex(f_trial)
            obj = fast_convolve(f_trial).max()
            if obj < best_trial:
                best_trial = obj
                best_alpha = alpha

        if best_alpha is None:
            # Try smaller
            for alpha in [0.0003, 0.0005, 0.001]:
                f_trial = f + alpha * direction
                f_trial = project_simplex(f_trial)
                obj = fast_convolve(f_trial).max()
                if obj < best_trial:
                    best_trial = obj
                    best_alpha = alpha

        if best_alpha is None:
            break

        f = f + best_alpha * direction
        f = project_simplex(f)
        obj = fast_convolve(f).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
        else:
            # Revert
            f = best_f.copy()

    return best_f, best_obj


def multi_peak_lp_step(f, n, time_budget=10.0):
    """Single LP step using all constraints, returns new f."""
    b = fast_convolve(f)
    M = b.max()
    if M < 1e-15:
        return f

    g = solve_lp_all_constraints(f, n, M)
    if g is None or g.sum() <= 1.0 + 1e-8:
        return None

    g = g / g.sum()
    return g


def line_search_peak(f, g, max_evals=25):
    """Golden section line search to find best mixing parameter t."""
    n = len(f)

    def obj_at_t(t):
        f_trial = (1.0 - t) * f + t * g
        f_trial = np.maximum(f_trial, 0.0)
        s = f_trial.sum()
        if s < 1e-10:
            return 1e10
        f_trial = f_trial / s
        return fast_convolve(f_trial).max()

    # Coarse search
    ts = [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7]
    best_t = 0.05
    best_obj = obj_at_t(0.05)
    for t in ts:
        o = obj_at_t(t)
        if o < best_obj:
            best_obj = o
            best_t = t

    # Golden section refine
    idx = ts.index(best_t)
    lo = ts[max(0, idx - 1)]
    hi = ts[min(len(ts) - 1, idx + 1)]
    if hi - lo < 1e-4:
        return best_t

    gr = (np.sqrt(5) - 1) / 2
    c = hi - gr * (hi - lo)
    d = lo + gr * (hi - lo)
    fc = obj_at_t(c)
    fd = obj_at_t(d)

    for _ in range(15):
        if hi - lo < 1e-5:
            break
        if fc <= fd:
            hi = d
            d = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = obj_at_t(c)
        else:
            lo = c
            c = d
            fc = fd
            d = lo + gr * (hi - lo)
            fd = obj_at_t(d)

    return (lo + hi) / 2


def run_lp_optimization(f_init, time_limit, label="lp"):
    """Main LP optimization loop with all constraints."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    f = f / f.sum()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    it = 0
    stuck = 0
    step_scale = 1.0

    print(f"  [{label}] Start: n={n}, obj={best_obj:.12f}, score={2*n*best_obj:.6f}")

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break

        # LP step
        g = multi_peak_lp_step(f, n, time_budget=min(10, rem * 0.3))
        if g is None:
            stuck += 5
            if stuck > 30:
                break
            # Fallback: use peak direction
            b = fast_convolve(f)
            pk = b.argmax()
            g = np.zeros(n)
            lo = max(0, pk - n + 1)
            hi = min(n - 1, pk)
            g[lo:hi + 1] = f[pk - hi:pk - lo + 1]
            g = g / (g.sum() + 1e-30)

        # Line search
        if rem > 15:
            t_best = line_search_peak(f, g)
        else:
            t_best = 0.05 * step_scale

        t_best = min(t_best, 0.5)

        # Update
        f_new = (1.0 - t_best) * f + t_best * g
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-10:
            f_new = f
        else:
            f_new = f_new / s

        obj_new = fast_convolve(f_new).max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            stuck = 0
            step_scale = min(step_scale * 1.1, 2.0)
        else:
            f = f_new  # Still accept for exploration
            stuck += 1
            step_scale = max(step_scale * 0.8, 0.3)
            if stuck > 40:
                # Restart from best with noise
                f = best_f + np.random.default_rng(it).standard_normal(n) * 0.001
                f = np.maximum(f, 0)
                f = f / f.sum()
                stuck = 0

        if it % 30 == 0:
            print(f"  [{label}] it={it} obj={best_obj:.12f} score={2*n*best_obj:.6f} "
                  f"t={time.monotonic()-t0:.0f}s stuck={stuck}")

    return best_f, best_obj


def run_cd_polish(f_init, time_limit, label="cd"):
    """Coordinate descent polish loop."""
    t0 = time.monotonic()
    f = f_init.copy()
    f = f / f.sum()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    n = len(f)

    print(f"  [{label}] Start: obj={best_obj:.12f}, score={2*n*best_obj:.6f}")

    while time.monotonic() - t0 < time_limit:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break
        f_new = coordinate_descent_polish(f, max_rounds=2, time_limit=min(rem, 60.0))
        obj_new = fast_convolve(f_new).max()
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
        else:
            break

    print(f"  [{label}] End: obj={best_obj:.12f}, score={2*n*best_obj:.6f}")
    return best_f, best_obj


def run_gradient_polish(f_init, time_limit, label="grad"):
    """Gradient-based polish with momentum."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f, best_obj = global_descent_search(f, n_peaks=8, time_limit=time_limit)
    print(f"  [{label}] Score: {2*n*best_obj:.6f}")
    return best_f, best_obj


def interpolate_sequence(f_old, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(f_old)
    if n_old == n_new:
        return f_old.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    f_new = np.interp(x_new, x_old, f_old)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = np.ones(n_new) / n_new
    return f_new
