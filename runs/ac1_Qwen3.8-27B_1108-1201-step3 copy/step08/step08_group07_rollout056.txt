"""
Multi-phase optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Combines improved LP fictitious play with multi-start initialization,
variance-minimizing spectral gradient descent, targeted peak-reduction
local search, and length exploration.

import numpy as np
import time
import cvxpy as cp
from scipy.linalg import toeplitz
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def build_conv_matrix(f):
    """Build (2n-1) x n matrix C such that C @ g = f * g (linear conv)."""
    n = len(f)
    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    return toeplitz(first_col, first_row)


def solve_best_response(f, M, prev_g=None):
    """Solve: max sum(g) s.t. C(f) @ g <= M, g >= 0."""
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    if prev_g is not None:
        g.value = prev_g
    C = build_conv_matrix(f)
    constraints = [C @ g <= M]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL, warm_start=(prev_g is not None))
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            return prev_g if prev_g is not None else f.copy()
    if g.value is None:
        return prev_g if prev_g is not None else f.copy()
    return np.maximum(g.value, 0)


def lp_fictitious_play(a0, time_limit, step=0.4, verbose=True, label="LP-FP"):
    """LP-based fictitious play with adaptive step and periodic reset."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a0.copy(), compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iters = 0
    prev_g = None
    stall_count = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        M = conv.max()

        g = solve_best_response(a, M, prev_g=prev_g)
        prev_g = g
        g_sum = g.sum()

        if g_sum < 1e-10:
            break

        g_norm = g / g_sum

        # Adaptive step: decay over time, reduce when stalled
        base_step = step * (1.0 + 0.5 * np.cos(np.pi * min(iters / max(1, int(time_limit / 5.0)), 1.0)))
        if stall_count > 100:
            base_step = max(0.05, base_step * 0.5)

        a_new = (1 - base_step) * a + base_step * g_norm
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss

        score = compute_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            stall_count = 0
        else:
            stall_count += 1

        a = a_new

        if stall_count > 200:
            rng = np.random.default_rng(iters * 7919)
            a = best_a + rng.standard_normal(n) * (0.001 + 0.002 * stall_count / 500.0)
            a = np.maximum(a, 0)
            ss = a.sum()
            if ss > 1e-15:

            prev_g = None
            stall_count = 0

def spectral_init(n, seed=42, n_bussgang=50):
    """Generate a sequence with approximately flat power spectrum."""
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 2).bit_length()

    # Start with random phases
    A = (rng.standard_normal(fft_size // 2 + 1) + 1j * rng.standard_normal(fft_size // 2 + 1))
    # Fix DC to be positive (for non-negativity)
    A[0] = np.abs(A[0])

    a = irfft(A, fft_size)[:n]
    a = np.maximum(a, 0)
    ss = a.sum()
    if ss < 1e-15:
        return np.ones(n) / n
    a = a / ss

    # Bussgang iterations to flatten spectrum
    for _ in range(n_bussgang):
        A = rfft(a, fft_size)
        power = np.abs(A) ** 2
        total_power = power.sum()
        target = total_power / (fft_size // 2 + 1)
        ratio = np.where(power > 1e-30, target / power, 1.0)
        ratio = np.clip(ratio, 0.05, 20.0)
        tau = 0.3
        A_new = A * (ratio ** tau)
        a = irfft(A_new, fft_size)[:n]
        a = np.maximum(a, 0)
        ss = a.sum()
        if ss < 1e-15:
            return np.ones(n) / n
        a = a / ss

    return a

        iters += 1
        conv = fast_convolve(a)
        mean_conv = 1.0 / (2 * n - 1)  # since sum(a)=1, sum(conv)=1
        residual = conv - mean_conv

        # Gradient of sum(residual^2) w.r.t. a
        # d/da[i] sum_k (conv[k] - c)^2 = 2 * sum_k residual[k] * d(conv[k])/da[i]
        # d(conv[k])/da[i] = a[k-i]*1[valid] + a[i-k]*1[valid] for auto-conv
        # This is: 2 * (residual * a) circular-conv-like, but for linear conv it's trickier
        # Use FFT-based approach: gradient = 2 * (linear cross-conv of residual and a, truncated to n)
        # Actually: d/da[i] (a*a)[k] = a[k-i] + a[i-k] (with valid index checks)
        # This equals: the i-th element of (a correlated with the "extended" a)
        # Simplest: use the fact that for linear conv, grad[i] = 2 * sum_k residual[k] * (a[k-i] + a[i-k])_valid
        # Which is 2 * (residual_extended * a) reversed and truncated...
        # Let's just use: conv_full = fast_convolve(a), grad_a = 2 * (cross-conv of residual_padded and a)
        # Actually the cleanest way: pad residual to size n, then convolve
        # d/da[i] sum_k (a*a)[k]^2 = 2 * sum_k (a*a)[k] * (a[k-i] + (a*[i-k]) where valid)
        # = 2 * [ ((a*a) * a) + ((a*a) * reverse(a)) ][n-1-i] ... this is getting complicated

        # Simpler approach: use the FFT-based gradient
        # For circular conv: grad = 2 * IFFT(conj(FFT(a)) * FFT(2*(conv_circ - c)))
        # For linear conv, approximate with circular conv (works well for large n)
        A = rfft(a, fft_size)
        conv_circ = irfft(A * np.conj(A), fft_size)[:n]  # circular auto-conv
        # Actually conv_circ[k] = sum_j a[j]*a[(k-j) mod n]
        # The gradient of sum (conv_circ - c)^2 w.r.t. a[i]:
        # = 2 * sum_k (conv_circ[k]-c) * d(conv_circ[k])/da[i]
        # d(conv_circ[k])/da[i] = a[(k-i) mod n] + a[(i-k) mod n] = 2*a[(k-i) mod n] (since symmetric)
        # Wait no, conv_circ[k] = sum_j a[j]*a[(k-j)%n]
        # d/da[i] conv_circ[k] = a[(k-i)%n] + (a[i] contributes when j=i: a[(k-i)%n]) + (a[(k-j)%n]=a[i] when j=(k-i)%n: a[j]=a[(k-i)%n])
        # = a[(k-i)%n] + a[(k-(k-i))%n] = a[(k-i)%n] + a[i%n]
        # Hmm this is getting confusing. Let me just use the direct formula.
        #
        # For circular: conv_circ = IFFT(|FFT(a)|^2)
        # d/da conv_circ = IFFT(FFT(a) * conj(FFT(a)) gradient)... 
        # Actually: conv_circ[k] = sum_j a[j] a[(k-j) mod n]
        # Let F = FFT(a). Then conv_circ = IFFT(|F|^2).
        # d(conv_circ)/da = IFFT(conj(F) * d(|F|^2)/da) ... no this is circular.
        # 
        # Simpler: d/da[i] (conv_circ)[k] = a[(k-i) mod n] + a[(i-k) mod n]
        # = 2 * a[(k-i) mod n] (since a is real and the sum is symmetric)
        # Wait no: d/da[i] sum_j a[j]*a[(k-j)%n]
        # = sum_j [delta(j,i)*a[(k-j)%n] + a[j]*delta((k-j)%n, i)]
        # = a[(k-i)%n] + a[(k-i)%n]  [since (k-j)%n = i means j = (k-i)%n]
        # = 2*a[(k-i)%n]
        #
        # So grad[i] = 2 * sum_k residual[k] * 2*a[(k-i) mod n] = 4 * sum_k residual[k] * a[(k-i) mod n]
        # = 4 * (residual circularly correlated with a)[i]
        # = 4 * IFFT(FFT(residual) * conj(FFT(a)))[i]  ... let me verify
        # (residual circ-conv a)[i] = sum_k residual[k] * a[(i-k) mod n]
        # IFFT(FFT(residual) * conj(FFT(a)))[i] = sum_k residual[k] * a[(i-k) mod n] (circular)
        # Yes! So grad = 4 * IFFT(FFT(residual) * conj(FFT(a)))

        R = rfft(residual[:n], fft_size)  # pad residual to circular
        grad = 4.0 * irfft(R * np.conj(A), fft_size)[:n]

        # Gradient step with projection onto simplex {a >= 0, sum(a) = 1}
        a_new = a - lr_current * grad
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss

        score = compute_score(a_new)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a_new.copy()
            lr_current = min(lr * 2.0, 0.1)  # increase step if improving
        else:
            lr_current = max(lr * 0.5, 0.0001)  # decrease step if not improving

        a = a_new

        if verbose and iters % 200 == 0:
            elapsed = time.monotonic() - t0
            var = np.var(conv)
            print(f"  VarGD iter={iters}, best={best_score:.8f}, var={var:.2e}, "
                  f"lr={lr_current:.6f}, time={elapsed:.0f}s")

    return best_a, best_score


def targeted_local_search(a0, time_limit, verbose=True):
    """Targeted local search that focuses on reducing the auto-convolution peak."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(99999)
    iters = 0
    no_improve = 0
    max_no_improve = 500

    while time.monotonic() - t0 < time_limit:
        iters += 1
        if no_improve > max_no_improve:
            # Restart with perturbation
            a = best_a + rng.standard_normal(n) * 0.001
            a = np.maximum(a, 0)
            a = a / a.sum()
            no_improve = 0
            continue

        conv = fast_convolve(a)
        k_peak = int(np.argmax(conv))
        peak_val = conv[k_peak]

        # Try to reduce the peak by moving mass
        improved = False
        n_trials = 50

        for _ in range(n_trials):
            if time.monotonic() - t0 > time_limit:
                break

            # Choose a source position (contributor to the peak)
            # Positions i where (k_peak - i) is valid contribute to conv[k_peak]
            i_lo = max(0, k_peak - (n - 1))
            i_hi = min(n - 1, k_peak)
            if i_hi <= i_lo:
                i_src = rng.integers(0, n)
            else:
                # Weight by contribution
                candidates = list(range(i_lo, i_hi + 1))
                weights = np.array([a[i] * a[k_peak - i] for i in candidates])
                weights = weights / weights.sum()
                i_src = candidates[rng.choice(len(candidates), p=weights)]

            if a[i_src] < 1e-10:
                continue

            # Choose a target position (non-contributor to the peak, or low contributor)
            # Try to find a position j where a[j]*a[k_peak-j] (if valid) is small
            n_targets = 10
            best_target = None
            best_gain = 0

            for _ in range(n_targets):
                j = int(rng.integers(0, n))
                if j == i_src:
                    continue
                # Estimate the change in peak
                j_k = k_peak - j
                new_contrib_j = a[i_src] * a[j_k] if 0 <= j_k < n else 0
                old_contrib_i = a[i_src] * a[k_peak - i_src] if 0 <= k_peak - i_src < n else 0
                # Moving delta from i to j changes peak by:
                # -delta * a[k_peak-i] + delta * a[k_peak-j] (approximately)
                delta_est = a[k_peak - i_src] - (a[k_peak - j] if 0 <= k_peak - j < n else 0)
                if delta_est > best_gain:
                    best_gain = delta_est
                    best_target = j

            if best_target is None or best_gain <= 0:
                # Just try a random target
                best_target = int(rng.integers(0, n))
                if best_target == i_src:
                    best_target = (best_target + 1) % n

            # Try several amounts of transfer
            base = min(a[i_src], 0.05)
            for frac in [0.1, 0.3, 0.5, 0.7, 1.0]:
                delta = frac * base
                if delta >= a[i_src] * 0.9:
                    delta = a[i_src] * 0.5

                a_trial = a.copy()
                a_trial[i_src] -= delta
                a_trial[best_target] += delta
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score(a_trial)
                if sc < best_score - 1e-14:
                    best_score = sc
                    best_a = a_trial.copy()
                    a = a_trial
                    improved = True
                    break

            if improved:
                break

        if not improved:
            no_improve += 1

        if verbose and iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"  TLS iter={iters}, best={best_score:.8f}, peak_lag={k_peak}, "
                  f"no_improve={no_improve}, time={elapsed:.0f}s")

    return best_a, best_score


def lbgfs_smoothed(a0, time_limit, T=500, maxiter=3000):
    """L-BFGS on log-sum-exp smoothed objective in log-space."""
    n = len(a0)

    def softmax_from_z(z):
        z_max = z.max()
        exp_z = np.exp(z - z_max)
        s = exp_z.sum()
        if s < 1e-300:
            return np.ones(len(z)) / len(z)
        return exp_z / s

    z0 = np.log(np.maximum(a0, 1e-30))
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)

    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        a = softmax_from_z(z)
        conv = fast_convolve(a)
        shifted = conv / T
        max_shifted = shifted.max()
        lse = max_shifted + np.log(np.exp(shifted - max_shifted).sum())
        return (2.0 * n / T) * np.exp(T * lse)

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        a = softmax_from_z(z)
        conv = fast_convolve(a)
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        shifted = conv / T
        max_shifted = shifted.max()
        w = np.exp(shifted - max_shifted)
        w = w / w.sum()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        grad_a = 2.0 * irfft(np.conj(W) * A, fft_size)[:n]
        S = float(np.dot(grad_a, a))
        grad_z = a * (grad_a - S)
        return (2.0 * n / T) * grad_z

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 50, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
        score = compute_score(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except Exception:
        pass
    return best_a, best_score


def check_and_update(a, label, best_seq, best_score, verbose=True):
    """Evaluate a candidate and update best if improved."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        if verbose:
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def coordinate_descent(a0, time_limit, n_pairs=200, verbose=True):
    """Coordinate descent with pairwise mass transfers."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(54321)
    iters = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        improved = False
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j or a[i] < 1e-14:
                continue

            best_local = best_score
            best_t = 0.0
            base = min(a[i], max(a[j], 1e-10))

            for frac in [0.05, 0.1, 0.2, 0.4, 0.6, 0.8]:
                t = frac * base
                if t >= a[i] - 1e-15:
                    t = a[i] * 0.5
                a_trial = a.copy()
                a_trial[i] -= t
                a_trial[j] += t
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score(a_trial)
                if sc < best_local - 1e-14:
                    best_local = sc
                    best_t = t

            if best_t != 0.0:
                a[i] -= best_t
                a[j] += best_t
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = best_local
                best_a = a.copy()
                improved = True

        if not improved and iters > 30:
            noise = rng.standard_normal(n) * 0.001
            a = best_a.copy() + noise
            a = np.maximum(a, 0)
            a = a / a.sum()

        if verbose and iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"  CD iter={iters}, best={best_score:.8f}, time={elapsed:.0f}s")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining LP fictitious play, spectral methods,
    variance gradient descent, targeted local search, and length exploration.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 768
        parent = np.ones(n_init) / n_init
        best_seq = parent.copy()
        best_score = compute_score(parent)
        print(f"Initialized: n={n_init}, score={best_score:.8f}")

    n_parent = len(parent)

    # Phase 1: Multi-start initialization + short LP-FP (300s)
    print(f"\n=== Phase 1: Multi-start (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t1 = min(rem * 0.18, 300.0)
    print(f"  Budget: {t1:.0f}s")

    if rem > 60:
        # Try different initializations
        inits = {'parent': parent.copy()}

        # Spectral initialization
        try:
            spec_init = spectral_init(n_parent, seed=seed, n_bussgang=30)
            inits['spectral'] = spec_init
        except Exception:
            pass

        # Random initialization with Bussgang
        try:
            rng = np.random.default_rng(seed + 1)
            rand_init = rng.random(n_parent)
            rand_init = rand_init / rand_init.sum()
            inits['random_bussgang'] = rand_init
        except Exception:
            pass

        # Triangular initialization
        try:
            x = np.linspace(-1, 1, n_parent)
            tri_init = np.maximum(1 - np.abs(x), 0) ** 2
            tri_init = tri_init / tri_init.sum()
            inits['triangular'] = tri_init
        except Exception:
            pass

        # Run short LP-FP on each
        for name, init in inits.items():
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            t_sub = min(rem / max(1, len(inits)), t1 / len(inits))
            print(f"  LP-FP [{name}], t={t_sub:.0f}s")
            a_sub, sc_sub = lp_fictitious_play(init, t_sub, step=0.5, verbose=False,
                                               label=f"LP-{name}")
            best_seq, best_score = check_and_update(a_sub, f"lp_{name}",
                                                    best_seq, best_score)
            # Also update working best
            if sc_sub < best_score:
                best_seq = a_sub.copy()

    # Phase 2: Extended LP-FP on best (500s)
    print(f"\n=== Phase 2: Extended LP-FP (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem * 0.30, 500.0)
    if rem > 60:
        print(f"  Budget: {t2:.0f}s")
        a_lp2, sc_lp2 = lp_fictitious_play(best_seq, t2, step=0.4, verbose=True,
                                           label="LP-extended")
        best_seq, best_score = check_and_update(a_lp2, "lp_extended",
                                                best_seq, best_score)

    # Phase 3: Variance gradient descent (150s)
    print(f"\n=== Phase 3: Variance GD (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t3 = min(rem * 0.10, 150.0)
    if rem > 40:
        print(f"  Budget: {t3:.0f}s")
        a_vg, sc_vg = variance_gradient_descent(best_seq, t3, lr=0.005, verbose=True)
        best_seq, best_score = check_and_update(a_vg, "var_gd",
                                                best_seq, best_score)

    # Phase 4: Bussgang refinement (100s)
    print(f"\n=== Phase 4: Bussgang (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t4 = min(rem * 0.06, 100.0)
    if rem > 30:
        print(f"  Budget: {t4:.0f}s")
        a_bg, sc_bg = bussgang_refine(best_seq, t4, tau=0.3, verbose=True)
        best_seq, best_score = check_and_update(a_bg, "bussgang",
                                                best_seq, best_score)

    # Phase 5: Targeted local search (200s)
    print(f"\n=== Phase 5: Targeted Local Search (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.12, 200.0)
    if rem > 40:
        print(f"  Budget: {t5:.0f}s")
        a_tls, sc_tls = targeted_local_search(best_seq, t5, verbose=True)
        best_seq, best_score = check_and_update(a_tls, "tls",
                                                best_seq, best_score)

    # Phase 6: L-BFGS polish (150s)
    print(f"\n=== Phase 6: L-BFGS Polish (best={best_score:.8f}) ===")
    for T_val in [500, 2000]:
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t6 = min(rem * 0.08, 75.0)
        print(f"  L-BFGS T={T_val}, budget={t6:.0f}s")
        a_lb, sc_lb = lbgfs_smoothed(best_seq, t6, T=T_val, maxiter=2000)
        best_seq, best_score = check_and_update(a_lb, f"lbgfs_{T_val}",
                                                best_seq, best_score)

    # Phase 7: Coordinate descent (150s)
    print(f"\n=== Phase 7: Coordinate Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t7 = min(rem * 0.10, 150.0)
    if rem > 30:
        print(f"  Budget: {t7:.0f}s")
        a_cd, sc_cd = coordinate_descent(best_seq, t7, n_pairs=300, verbose=True)
        best_seq, best_score = check_and_update(a_cd, "coord_desc",
                                                best_seq, best_score)

    # Phase 8: Try nearby lengths (150s)
    print(f"\n=== Phase 8: Nearby Lengths (best={best_score:.8f}) ===")
    alt_lengths = []
    for delta in [2, 4, 8, 16, -2, -4, -8]:
        alt_n = n_parent + delta
        if 128 <= alt_n <= 4096:
            alt_lengths.append(alt_n)

    for alt_n in alt_lengths:
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t8 = min(rem * 0.15, 50.0)
        init_alt = interpolate_sequence(best_seq, alt_n)
        print(f"  LP-FP n={alt_n}, budget={t8:.0f}s")
        a_alt, sc_alt = lp_fictitious_play(init_alt, t8, step=0.3, verbose=False,
                                           label=f"LP-n{alt_n}")
        best_seq, best_score = check_and_update(a_alt, f"n{alt_n}",
                                                best_seq, best_score)

    # Phase 9: Final LP-FP polish (remaining time)
    print(f"\n=== Phase 9: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 20:
        t9 = min(rem * 0.5, 100.0)
        print(f"  Final LP-FP, budget={t9:.0f}s")
        a_final, sc_final = lp_fictitious_play(best_seq, t9, step=0.2,
                                               verbose=True, label="LP-final")
        best_seq, best_score = check_and_update(a_final, "lp_final",
                                                best_seq, best_score)

    # Final targeted local search with remaining time
    rem = deadline - time.monotonic()
    if rem > 30:
        t10 = min(rem * 0.5, 60.0)
        print(f"  Final TLS, budget={t10:.0f}s")
        a_final2, sc_final2 = targeted_local_search(best_seq, t10, verbose=False)
        best_seq, best_score = check_and_update(a_final2, "tls_final",
                                                best_seq, best_score)

    # Final coordinate descent
    rem = deadline - time.monotonic()
    if rem > 20:
        t11 = min(rem * 0.5, 40.0)
        a_final3, sc_final3 = coordinate_descent(best_seq, t11, n_pairs=200, verbose=False)
        best_seq, best_score = check_and_update(a_final3, "cd_final",
                                                best_seq, best_score)

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs) and fs < best_score:
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
