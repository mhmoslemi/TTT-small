    n = len(a)

    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def softmax_from_z(z):
    """Convert log-domain z to probability vector (sum=1, positive)."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def compute_grad_z(a):
    """Compute gradient of 2*n*max(conv(a,a)) with respect to z (log-space)."""
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    # Use top few peaks for a steeper gradient
    threshold = max_b * 0.95
    active = np.where(b >= threshold)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(b))])
    w = b[active] - (b[active] - b[active].max())
    w = w / w.sum()
    grad_a = np.zeros(n)
    for k in active:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k - js if k < len(w) else 0] * a[k - idx]
    # Proper weighted gradient
    grad_a = np.zeros(n)
    for ii, k in enumerate(active):

        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[ii] * a[k - idx]
    S = np.dot(grad_a, a)
    return a * (grad_a - S)


def lbgfs_refine(a0, time_limit, T_start=150.0, T_end=0.1, maxiter=15000):
    """L-BFGS with LogSumExp smoothing and multi-stage geometric annealing."""
    n = len(a0)
    z = z_from_a(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()

    def make_func(T_val):
        def func(zv):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(zv)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            lse = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * lse
        return func

    def make_jac(T_val):
        def jac(zv):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(zv)
            a = softmax_from_z(zv)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    n_stages = 8
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(30, maxiter // n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)

def subgradient_descent(a0, time_limit, n_steps=50000, peak_fraction=0.90):
    """Subgradient descent in log-space with adaptive step size and peak averaging."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    z = z_from_a(a).copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    lr = 0.05
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * peak_fraction
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        # Weight by how close to max
        w = (b[active] - threshold) / (b[active].sum() - threshold * len(active) + 1e-30)
        w = w / (w.sum() + 1e-30)

        grad_a = np.zeros(n)
        for ii, k in enumerate(active):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * w[ii] * a[k - idx]
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)

        z_new = z - lr * grad_z
        a_new = softmax_from_z(z_new)
        score_new = 2.0 * n * fast_convolve(a_new).max()

        if score_new < best_score - 1e-15:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            z = z_new
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 50:
            lr *= 0.9
            no_improve = 0
        lr = max(0.002, lr)
        if step % 5000 == 0:
            lr = max(0.002, 0.05 / (1.0 + step * 0.0003))

        if step % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SubGrad {elapsed:.0f}s: score={best_score:.6f} lr={lr:.5f}")

    return best_a, best_score


def coordinate_polish(a0, time_limit, n_steps=30000):

    n = len(a0)

    a = a / s
    best_a = a.copy()

        # Find the peak (excluding very edges which are hard to fix)
        b_inner = b.copy()
        b_inner[:2] = -np.inf
        b_inner[-2:] = -np.inf
        peak_k = int(np.argmax(b_inner))
        peak_val = b[peak_k]

        # Find top contributors to this peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        top_m = min(6, len(idx))
        top_positions = idx[np.argsort(contribs)[::-1][:top_m]]

        improved = False

            if a[i] < 1e-12:
                continue

                # Put mass where it causes least convolution increase

                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break

                break

        if not improved:
            # Try pair adjustment
            if len(top_positions) >= 2:
                i, j = top_positions[0], top_positions[1]
                for f1 in [0.93, 0.88]:
                    for f2 in [0.93, 0.88]:
                        a_trial = a.copy()
                        d1 = a[i] * (1 - f1)
                        d2 = a[j] * (1 - f2)
                        a_trial[i] -= d1
                        a_trial[j] -= d2
                        a_trial = np.maximum(a_trial, 0)
                        mask = np.ones(n, dtype=bool)
                        mask[max(0, peak_k - n + 1):min(n, peak_k + 1)] = False
                        candidates = np.where(mask)[0]
                        if len(candidates) > 0:
                            n_add = min(5, len(candidates))
                            chosen = candidates[rng.permutation(len(candidates))[:n_add]]
                            a_trial[chosen] += (d1 + d2) / n_add
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break

        if not improved:
            # Stagnation: break after many failed steps
            if step > 100 and step % 500 == 0:
                break

    return best_a, best_score


def asymmetric_init(n, rng):
    """Create an asymmetric initialization that avoids central peak."""
    choice = rng.integers(0, 4)
    if choice == 0:
        # Two separated bumps
        x = np.linspace(0, 1, n)
        center1 = 0.25
        center2 = 0.75
        sigma = 0.15
        a = np.exp(-(x - center1) ** 2 / (2 * sigma ** 2)) + np.exp(-(x - center2) ** 2 / (2 * sigma ** 2))
    elif choice == 1:
        # Skewed Gaussian
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.3) ** 2) / 0.08) + 0.5 * np.exp(-((x - 0.7) ** 2) / 0.04)
    elif choice == 2:
        # Sine-based asymmetric
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2 * (1 + 0.5 * np.cos(2 * np.pi * x))
        a = np.maximum(a, 0)
    else:
        # Power asymmetry
        x = np.linspace(0, 1, n)
        a = x ** 0.3 * (1 - x) ** 0.7
        a = a + 0.3 * np.exp(-10 * (x - 0.8) ** 2)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def smooth_init(n, rng):
    """Create a smooth symmetric initialization."""
    choice = rng.integers(0, 5)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 2:
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = a ** 1.5
    else:
        a = np.exp(-x ** 2 * 3) + np.exp(-(1 - x) ** 2 * 3)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def freq_informed_init(n, rng):
    """Initialize using frequency-domain considerations for flat autocorrelation."""
    # Start with a chirp-like structure (good for flat spectrum)
    x = np.arange(n) / n
    # Use a modulated cosine that has good autocorrelation properties
    freq = rng.uniform(1.0, 3.0)
    phase = rng.uniform(0, 2 * np.pi)
    a_base = 0.5 + 0.5 * np.cos(2 * np.pi * freq * x + phase)
    # Mix with a smooth envelope
    envelope = np.sin(np.pi * x) ** 0.5
    a = a_base * envelope
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def stochastic_search(a0, time_limit, n_restarts=5):
    """Stochastic perturbation + re-optimization to escape local minima."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    per_restart = max(2, int((time_limit - 2) / n_restarts))

    for r in range(n_restarts):
        if time.monotonic() - t0 > time_limit:
            break
        # Perturb
        scale = 0.03 + 0.02 * r
        perturbed = a + scale * rng.standard_normal(n)
        perturbed = np.maximum(perturbed, 0)
        perturbed = perturbed / perturbed.sum()

        # Quick subgradient refinement
        t_sub = per_restart * 0.4
        a_sub, sc_sub = subgradient_descent(perturbed, t_sub, n_steps=5000, peak_fraction=0.92)
        if sc_sub < best_score:
            best_score = sc_sub
            best_a = a_sub.copy()
            a = a_sub.copy()

        # Quick L-BFGS
        t_lb = per_restart * 0.5
        if time.monotonic() - t0 < time_limit:
            a_lb = lbgfs_refine(best_a, t_lb, T_start=20.0, T_end=0.5, maxiter=2000)
            sc_lb = 2.0 * n * fast_convolve(a_lb).max()
            if sc_lb < best_score:
                best_score = sc_lb
                best_a = a_lb.copy()

        elapsed = time.monotonic() - t0
        print(f"    Stoch restart {r+1}/{n_restarts} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining LP, L-BFGS, subgradient, coordinate polish,
    and stochastic search across multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    solver = _get_solver()
    print(f"Using LP solver: {solver}, all solvers: {_SOLVERS}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 768
        best_seq = smooth_init(n_init, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}) ***")

    # === Phase 1: LP on current best length ===
    print(f"\n=== Phase 1: Nesterov LP n={n_best} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp(best_seq, t_alloc, solver=solver, n_max_iters=60000)
        try_update(a_lp, "lp_p1")

    # === Phase 2: L-BFGS aggressive annealing ===
    print(f"\n=== Phase 2: L-BFGS (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.22, 350.0)
    if t_alloc > 20:
        print(f"  T: 300->0.1, budget={t_alloc:.0f}s")
        a_lb = lbgfs_refine(best_seq, t_alloc, T_start=300.0, T_end=0.1, maxiter=20000)
        try_update(a_lb, "lbgfs_p2")

    # === Phase 3: Subgradient refinement ===
    print(f"\n=== Phase 3: Subgradient (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.10, 150.0)
    if t_alloc > 10:
        a_sg, _ = subgradient_descent(best_seq, t_alloc, n_steps=40000, peak_fraction=0.88)
        try_update(a_sg, "subgrad_p3")

    # === Phase 4: Coordinate polish ===
    print(f"\n=== Phase 4: Coordinate Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 120.0)
    if t_alloc > 10:
        a_cp, _ = coordinate_polish(best_seq, t_alloc, n_steps=40000)
        try_update(a_cp, "coord_p4")

    # === Phase 5: Try longer sequences ===
    print(f"\n=== Phase 5: Longer Sequences (best={best_score:.6f}) ===")
    longer_lengths = []
    for mult in [1.25, 1.5, 2.0]:
        ln = int(n_best * mult)
        ln = int(ln / 128) * 128  # Round to multiple of 128
        if ln > n_best and ln <= 4096:
            longer_lengths.append(ln)
    longer_lengths = sorted(set(longer_lengths))

    for alt_n in longer_lengths:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_total = min(rem * 0.25, 250.0)
        if t_total < 40:
            break
        print(f"  Trying n={alt_n}, budget={t_total:.0f}s")

        # Multiple initializations for this length
        inits = [
            interpolate_sequence(best_seq, alt_n),
            freq_informed_init(alt_n, rng),
            asymmetric_init(alt_n, rng),
        ]
        best_for_len = None
        best_sc_for_len = float('inf')

        for ii, init in enumerate(inits):
            rem2 = deadline - time.monotonic()
            if rem2 < 30:
                break
            t_sub = t_total / (len(inits) - ii)
            # Quick LP
            if t_sub > 20 and alt_n <= 2048:
                t_lp = t_sub * 0.5
                a_t, _ = nesterov_lp(init, t_lp, solver=solver, n_max_iters=15000)
                sc_t = 2.0 * alt_n * fast_convolve(a_t).max()
            else:
                # Use subgradient for longer sequences
                t_sg = t_sub * 0.4
                a_t, sc_t = subgradient_descent(init, t_sg, n_steps=8000, peak_fraction=0.90)

            # L-BFGS polish
            rem3 = deadline - time.monotonic()
            if rem3 > 15:
                t_lb2 = min(t_sub * 0.4, rem3 * 0.4)
                if t_lb2 > 5:
                    a_t2 = lbgfs_refine(a_t, t_lb2, T_start=50.0, T_end=0.5, maxiter=3000)
                    sc_t2 = 2.0 * alt_n * fast_convolve(a_t2).max()
                    if sc_t2 < sc_t:
                        a_t, sc_t = a_t2, sc_t2

            if sc_t < best_sc_for_len:
                best_sc_for_len = sc_t
                best_for_len = a_t.copy()

        if best_for_len is not None:
            try_update(best_for_len, f"long_n{alt_n}")

    # === Phase 6: Stochastic escape ===
    print(f"\n=== Phase 6: Stochastic Escape (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t_stoch = min(rem * 0.20, 200.0)
        a_st, _ = stochastic_search(best_seq, t_stoch, n_restarts=4)
        try_update(a_st, "stoch_p6")

    # === Phase 7: Second LP pass on best ===
    print(f"\n=== Phase 7: LP Pass 2 (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_lp2 = min(rem * 0.20, 150.0)
        if len(best_seq) <= 2048:
            a_lp2, _ = nesterov_lp(best_seq, t_lp2, solver=solver, n_max_iters=30000)
            try_update(a_lp2, "lp_p7")
        else:
            a_sg2, _ = subgradient_descent(best_seq, t_lp2, n_steps=20000, peak_fraction=0.90)
            try_update(a_sg2, "sg_p7")

    # === Phase 8: Final L-BFGS polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 15:
        t_fin = min(rem * 0.30, 80.0)
        a_fin = lbgfs_refine(best_seq, t_fin, T_start=10.0, T_end=0.05, maxiter=5000)
        try_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 10:
        t_cp = min(rem * 0.4, 40.0)
        a_cp2, _ = coordinate_polish(best_seq, t_cp, n_steps=15000)
        try_update(a_cp2, "final_coord")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
