def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_direction_primal(f, n, peaks, M, method='highs'):
    """
    Solve: max sum(g) s.t. A@g <= M, g >= 0
    where A[k,j] = f[j] if max(0,k-n+1) <= j <= min(n-1,k)
    """
    K = len(peaks)
    # Build constraint matrix A (K x n)
    # A[k,j] = f[j] if j in [max(0, peaks[k]-n+1), min(n-1, peaks[k])]
    
    # Use sparse matrix for efficiency
    from scipy.sparse import csr_matrix
    
    rows = []
    cols = []
    vals = []
    
    for i, k in enumerate(peaks):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            rows.append(i)
            cols.append(j)
            vals.append(f[j])
    
    A_mat = csr_matrix((vals, (rows, cols)), shape=(K, n))
    
    c = -np.ones(n)  # minimize -sum(g) = max sum(g)
    b_ub = np.full(K, M)
    
    res = linprog(c, A_ub=A_mat, b_ub=b_ub, bounds=[(0, None)] * n, method=method)
    
    if res.status != 0:
        return None
    
    return np.maximum(res.x, 0)

def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
