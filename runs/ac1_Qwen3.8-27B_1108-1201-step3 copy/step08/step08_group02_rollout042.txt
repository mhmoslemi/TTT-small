```python
"""
Optimized minimization of 2*n*max(conv(a,a))/(sum(a))^2 using
convex cutting-plane LP with structured initializations and
multi-scale search.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    result = irfft(A * A, fft_size)[:size]
    return np.maximum(result, 0.0)  # Clip tiny negatives from FFT

def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_convex(f, n, peak_indices, M):
    """
    Solve: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for k in peak_indices.
    Uses scipy linprog.
    """
    from scipy.optimize import linprog
    
    K = len(peak_indices)
    if K == 0:
        return None
    
    # Build constraint matrix A: A[i, j] = (f*g)[i] coefficient for g[j]
    # (f*g)[k] = sum_j f[j] * g[k-j] (with valid indices)
    # For a given k: g appears at positions j where 0 <= k-j < n
    # i.e., max(0, k-n+1) <= j <= min(k, n-1)
    # And the coefficient of g[j] in (f*g)[k] is f[k-j]
    
    A_ub = np.zeros((K, n))
    for row_i in range(K):
        k = int(peak_indices[row_i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A_ub[row_i, j] = f[k - j]
    
    c = -np.ones(n)  # Minimize -sum(g) = maximize sum(g)
    
    res = linprog(c, A_ub=A_ub, b_ub=np.full(K, M), bounds=[(0, None)] * n, method='highs')
    
    if res.status != 0 or res.x is None:
        return None
    
    g = np.maximum(res.x, 0.0)
    if np.any(np.isnan(g)) or np.any(np.isinf(g)):
        return None
    return g
```

```python
"""
Optimized minimization of 2*n*max(conv(a,a))/(sum(a))^2 using
convex cutting-plane LP, subgradient descent, structured initializations,
and multi-scale search. Key improvements: better peak management,
vectorized constraint building, adaptive length exploration.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    result = irfft(A * A, fft_size)[:size]
    return np.maximum(result, 0.0)


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    v = v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_constraint_matrix(f, peak_indices, n):
    """Build the constraint matrix for the LP. Vectorized."""
    K = len(peak_indices)
    A = np.zeros((K, n))
    for row_i in range(K):
        k = int(peak_indices[row_i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        # A[row_i, j] = f[k-j] for j in [lo, hi]
        # k-j ranges from k-hi down to k-lo
        A[row_i, lo:hi + 1] = f[k - hi:k - lo + 1][::-1]
    return A


def solve_lp_direction(f, n, peak_indices, M):
    """Solve: maximize sum(g) s.t. g >= 0, (f*g)[k] <= M for k in peak_indices."""
    K = len(peak_indices)
    if K == 0:
        return None
    A_ub = build_constraint_matrix(f, peak_indices, n)
    c = -np.ones(n)
    b_ub = np.full(K, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 120})
        if res.status != 0 or res.x is None:
            return None
        g = np.maximum(res.x, 0.0)
        if np.any(np.isnan(g)) or np.any(np.isinf(g)):
            return None
        return g
    except Exception:
        return None


def line_search_1d(f, h, current_max, n):
    """Find best step size t in (0,1] to minimize max(conv((1-t)f+th, (1-t)f+th))."""
    best_t = 0.0
    best_obj = current_max
    # Coarse grid
    t_candidates = [0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5, 0.6, 0.7, 0.8]
    for t in t_candidates:
        f_trial = (1.0 - t) * f + t * h
        s = f_trial.sum()
        if s < 1e-10:
            continue
        f_trial = f_trial / s
        obj = fast_convolve(f_trial).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_t = t
    # Refine
    if best_t > 0:
        lo = max(0.0, best_t - 0.1)
        hi = min(0.95, best_t + 0.1)
        for _ in range(10):
            m1 = lo + (hi - lo) / 3
            m2 = hi - (hi - lo) / 3
            f1 = (1.0 - m1) * f + m1 * h
            f1 = np.maximum(f1, 0.0)
            s1 = f1.sum()
            if s1 > 1e-10:
                f1 /= s1
                obj1 = fast_convolve(f1).max()
            else:
                obj1 = np.inf
            f2 = (1.0 - m2) * f + m2 * h
            f2 = np.maximum(f2, 0.0)
            s2 = f2.sum()
            if s2 > 1e-10:
                f2 /= s2
                obj2 = fast_convolve(f2).max()
            else:
                obj2 = np.inf
            if obj1 < obj2:
                hi = m2
            else:
                lo = m1
            if hi - lo < 0.005:
                break
        best_t = (lo + hi) / 2
        f_best = (1.0 - best_t) * f + best_t * h
        f_best = np.maximum(f_best, 0.0)
        sb = f_best.sum()
        if sb > 1e-10:
            f_best /= sb
            best_obj = fast_convolve(f_best).max()
    return best_t, best_obj

        threshold = 0.97 * M
        new_peaks = np.where(b >= threshold)[0]
        # Also ensure we have at least some peaks

            n_top = max(20, len(b) // 20)
            new_peaks = np.argpartition(b, -n_top)[-n_top:]

        max_peaks = min(len(b), max(200, n))
        if len(new_peaks) > max_peaks:

        # Solve LP

        if g is None or g.sum() <= 1.0 + 1e-7:
            no_improve += 1
            if no_improve > 30:

                if fs > 1e-10:
                    f /= fs
                no_improve = 0
                if time.monotonic() - t0 > time_limit:
                    break
                continue
            # Try with all peaks (more constraints)

                    continue

def coordinate_descent(f_init, time_limit, label="cd"):
    """Coordinate descent: try scaling each element."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_max = fast_convolve(f).max()
    multipliers = [0.7, 0.8, 0.9, 1.0, 1.1, 1.2, 1.5, 2.0]
    improved_total = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-12:
                continue
            best_val = f[j]
            best_obj = best_max
            for mult in multipliers:
                new_val = f[j] * mult
                if new_val < 0:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_val = new_val
            if best_val != f[j]:
                f[j] = best_val
                s = f.sum()
                if s > 1e-10:
                    f /= s
                if best_obj < best_max - 1e-14:
                    best_max = best_obj
                    improved = True
                    improved_total += 1
        if not improved:
            break

    return f, best_max


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    n = len(f)
    if n == new_n:
        return f.copy()
    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new


def structured_init(n, kind="uniform"):
    """Create a structured initial sequence of length n."""
    if kind == "uniform":
        return np.ones(n) / n
    elif kind == "raised_cosine":
        x = np.linspace(0, np.pi, n)
        a = np.sin(x / 2) ** 2  # = (1-cos(x))/2
        s = a.sum()
        return a / s
    elif kind == "gaussian":
        x = np.arange(n) - (n - 1) / 2.0
        sigma = n / 4.0
        a = np.exp(-x ** 2 / (2 * sigma ** 2))
        s = a.sum()
        return a / s
    elif kind == "triangular":
        x = np.arange(n)
        mid = (n - 1) / 2.0
        a = np.maximum(0, 1 - np.abs(x - mid) / mid)
        s = a.sum()
        return a / s
    elif kind == "bimodal":
        center1 = n // 4
        center2 = 3 * n // 4
        x = np.arange(n)
        a = np.exp(-((x - center1) ** 2 + (x - center2) ** 2) / (n / 4.0) ** 2)
        s = a.sum()
        return a / s
    else:
        return np.ones(n) / n

    t2 = min(rem * 0.20, 300.0)
    if t2 > 30:
        print(f"\n=== Phase 2: Subgradient (n={n_best}, budget={t2:.0f}s) ===")
        f2, obj2 = subgradient_optimize(best_seq, time_limit=t2, label="sg1")
        sc2 = float(evaluate_sequence(f2.tolist()))
        if np.isfinite(sc2) and sc2 < best_score:
            best_score = sc2
            best_seq = f2.copy()
            print(f"  NEW BEST [sg1]: {sc2:.6f}")
        else:
            print(f"  [sg1] no improvement: {sc2:.6f}")

    # === Phase 3: Multi-length exploration ===
    rem = deadline - time.monotonic()
    n_base = len(best_seq)
    target_lengths = []
    for factor in [0.7, 0.85, 1.15, 1.3, 1.5, 2.0]:
        tl = int(n_base * factor)
        tl = max(256, min(4096, tl))
        if tl != n_base and tl not in target_lengths:
            target_lengths.append(tl)
    target_lengths.sort(key=lambda x: abs(x - n_base))

    for target_n in target_lengths[:4]:
        rem = deadline - time.monotonic()
        if rem < 80:
            break
        t3 = min(rem * 0.15, 200.0)
        if t3 < 30:
            break
        f_interp = interpolate_to_length(best_seq, target_n)
        print(f"\n=== Phase 3: Optimize n={target_n} (budget={t3:.0f}s) ===")
        if target_n <= 2000:
            f3, obj3 = run_cutting_plane_optimize(f_interp, time_limit=t3 * 0.6, label=f"cp_n{target_n}")
            rem3 = deadline - time.monotonic()
            if rem3 > 20:
                f3b, obj3b = subgradient_optimize(f3, time_limit=min(rem3 * 0.5, 60), label=f"sg_n{target_n}")
                sc3b = float(evaluate_sequence(f3b.tolist()))
                if np.isfinite(sc3b) and sc3b < float(evaluate_sequence(f3.tolist())):
                    f3 = f3b
            sc3 = float(evaluate_sequence(f3.tolist()))
        else:
            f3, obj3 = subgradient_optimize(f_interp, time_limit=t3, label=f"sg_n{target_n}")
            sc3 = float(evaluate_sequence(f3.tolist()))

        if np.isfinite(sc3) and sc3 < best_score:
            best_score = sc3
            best_seq = f3.copy()
            print(f"  NEW BEST [n{target_n}]: {sc3:.6f}")
        else:
            print(f"  [n{target_n}] score={sc3:.6f} (no improvement)")

    # === Phase 4: Coordinate descent polish ===
    rem = deadline - time.monotonic()
    n_best = len(best_seq)
    t4 = min(rem * 0.15, 120.0)
    if t4 > 20:
        print(f"\n=== Phase 4: Coordinate descent (n={n_best}, budget={t4:.0f}s) ===")
        f4, obj4 = coordinate_descent(best_seq, time_limit=t4, label="cd1")
        sc4 = float(evaluate_sequence(f4.tolist()))
        if np.isfinite(sc4) and sc4 < best_score:
            best_score = sc4
            best_seq = f4.copy()
            print(f"  NEW BEST [cd1]: {sc4:.6f}")
        else:
            print(f"  [cd1] no improvement: {sc4:.6f}")

    # === Phase 5: Final cutting-plane burst ===
    rem = deadline - time.monotonic()
    n_best = len(best_seq)
    t5 = min(rem * 0.30, 200.0)
    if t5 > 30:
        print(f"\n=== Phase 5: Final LP (n={n_best}, budget={t5:.0f}s) ===")
        f5, obj5 = run_cutting_plane_optimize(best_seq, time_limit=t5, label="cp_final")
        sc5 = float(evaluate_sequence(f5.tolist()))
        if np.isfinite(sc5) and sc5 < best_score:
            best_score = sc5
            best_seq = f5.copy()
            print(f"  NEW BEST [cp_final]: {sc5:.6f}")
        else:
            print(f"  [cp_final] no improvement: {sc5:.6f}")

    # === Phase 6: Final coordinate descent ===
    rem = deadline - time.monotonic()
    t6 = min(rem * 0.20, 80.0)
    if t6 > 10:
        print(f"\n=== Phase 6: Final CD (budget={t6:.0f}s) ===")
        f6, obj6 = coordinate_descent(best_seq, time_limit=t6, label="cd_final")
        sc6 = float(evaluate_sequence(f6.tolist()))
        if np.isfinite(sc6) and sc6 < best_score:
            best_score = sc6
            best_seq = f6.copy()
            print(f"  NEW BEST [cd_final]: {sc6:.6f}")

    # === Final verification ===
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if not np.isfinite(final_score):
        best_seq = parent.copy()
        final_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {final_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
```
