def fast_convolve(a):

    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Adam optimization with
temperature-annealed log-sum-exp, FFT-based gradients, LP refinement,
and multi-length exploration.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax_fn(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_prob(a):
    """Convert probability vector to log-space."""
    return np.log(np.maximum(a, 1e-300))


def compute_grad_fft(a, w, n):
    """Compute g_a = 2 * cross_correlation(w, a) using FFT."""
    if n <= 300:
        g_a = np.zeros(n)
        for k in range(len(w)):
            if w[k] < 1e-18:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g_a[idx] += 2.0 * w[k] * a[k - idx]
        return g_a
    else:
        fft_len = 1 << (3 * n).bit_length()
        W = np.fft.rfft(w, fft_len)
        a_rev = a[::-1]
        Ar = np.fft.rfft(a_rev, fft_len)
        corr = np.fft.irfft(W * Ar, fft_len)
        g_a = 2.0 * corr[n - 1: 2 * n - 1]
        return g_a


def adam_with_annealing(a0, time_limit, T_start=200.0, T_end=0.5, lr_init=0.05, max_iters=50000):
    """Adam optimizer with geometric temperature annealing."""
    n = len(a0)

        v_hat = v / (1 - beta2 ** t_count)
        
        # Adaptive learning rate
        lr = lr_init / (1.0 + 0.01 * t_count ** 0.5)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            a_check = softmax_fn(z)
            score = 2.0 * n * fast_convolve(a_check).max()

                best_score = score
                best_a = a_check.copy()
                no_improve = 0

                no_improve += 1

        best_a = a_final.copy()
    
    return best_a, best_score


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
