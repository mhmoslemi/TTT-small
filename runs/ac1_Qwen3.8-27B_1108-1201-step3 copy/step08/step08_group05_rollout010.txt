import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(f, g):
    size = len(f) + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient(f, peak_indices):
    """Compute subgradient of max convolution at given peaks."""
    n = len(f)
    grad = np.zeros(n)
    for k in peak_indices:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            grad[j] += f[k - j]
    return grad


def peaktored_reduction(f, peak_indices, eta, n_iter=100):
    """
    Multiplicative weight update to reduce peaks.
    f_new[j] = f[j] * exp(-eta * grad[j]) / normalizer
    """
    n = len(f)
    f_work = f.copy()
    best_f = f_work.copy()
    best_obj = fast_convolve(f_work).max()
    
    for it in range(n_iter):
        grad = compute_gradient(f_work, peak_indices)
        if grad.max() < 1e-15:
            break
        # Normalize gradient
        grad_norm = grad.max()
        if grad_norm > 0:
            grad = grad / grad_norm
        # Multiplicative update
        f_work = f_work * np.exp(-eta * grad)
        # Normalize
        s = f_work.sum()
        if s < 1e-15:
            break
        f_work = f_work / s
        
        obj = fast_convolve(f_work).max()
        if obj < best_obj:
            best_obj = obj
            best_f = f_work.copy()
    
    return best_f, best_obj

def hann_window(n):
    w = 0.5 * (1.0 + np.cos(2.0 * np.pi * np.arange(n) / max(1, n)))
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def flat_top_window(n):
    w = np.ones(n)
    a = [0.21557895, 0.41663158, 0.277263158, 0.083578947, 0.006947368]
    for i, ai in enumerate(a):
        w -= ai * np.cos(2 * np.pi * i * np.arange(n) / max(1, n - 1) - i * np.pi)
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def smooth_bump(n):
    """Gaussian-like bump centered in the sequence."""
    x = np.arange(n) / n
    w = np.exp(-50 * (x - 0.5) ** 2)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s

"""
SCA (Successive Convex Approximation) based optimization for minimizing
the auto-convolution peak. Uses tangent-plane LP relaxation with all lags,
combined with multiplicative descent and adaptive constraint LP as fallbacks.
Targets score < 1.5027 (lower is better).

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(f, g):
    """Cross-convolution f*g via FFT."""
    size = len(f) + len(g) - 1
    fft_size = 1 << (size - 1).bit_length()
    F = rfft(f, fft_size)
    G = rfft(g, fft_size)
    return irfft(F * G, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient(f, peak_indices):
    """Compute subgradient of max convolution at given peaks."""
    n = len(f)
    grad = np.zeros(n)
    for k in peak_indices:
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            grad[j] += f[k - j]
    return grad


def solve_sca_lp(f0, time_limit=60.0):
    """
    Solve the SCA (Successive Convex Approximation) LP:
    min t
    s.t. 2*(f0*f)[k] - (f0*f0)[k] <= t, for all k
         f >= 0, sum(f) = 1
    
    Returns the optimal f (normalized) and t value, or (None, None) on failure.
    """
    n = len(f0)
    b0 = fast_convolve(f0)
    
    # Build constraint matrix: for each lag k, the row is 2*f0[k-j] for valid j
    K = 2 * n - 1
    A = np.zeros((K, n))
    for k in range(K):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        # A[k, j] = 2 * f0[k - j] for j in [lo, hi]
        for j in range(lo, hi + 1):
            A[k, j] = 2.0 * f0[k - j]
    
    # LP: min t s.t. A @ f - t <= b0, f >= 0, sum(f) = 1
    f_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        A @ f_var - t_var <= b0,
        f_var >= 0,
        cp.sum(f_var) == 1
    ]
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    
    f_opt = None
    for solver in [cp.PDLP, cp.SCS, cp.ECOS, cp.CLARABEL]:
        try:
            prob.solve(solver=solver)
            if f_var.value is not None:
                f_candidate = np.maximum(f_var.value, 0.0)
                if not (np.any(np.isnan(f_candidate)) or np.any(np.isinf(f_candidate))):
                    s = f_candidate.sum()
                    if s > 1e-10:
                        f_candidate = f_candidate / s
                        f_opt = f_candidate
                        break
        except Exception:
            continue
    
    return f_opt, (t_var.value if t_var.value is not None else None)

def multiplicative_descent(f_init, time_limit, eta_init=0.01, n_top=50):
    """

    Naturally maintains non-negativity.
    """

    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()

    eta = eta_init
    no_improve = 0
    rng = np.random.default_rng(123)

        if time.monotonic() - t0 > time_limit:
            break
        it += 1
        
        b = fast_convolve(f)
        M = b.max()
        
        # Select active peaks
        if n_top < len(b):
            active = np.argpartition(b, -n_top)[-n_top:]
        else:
            active = np.arange(len(b))

        grad = compute_gradient(f, active)
        grad_max = grad.max()
        if grad_max < 1e-15:

        grad = grad / grad_max

        f = f * np.exp(-eta * grad)
        s = f.sum()
        if s < 1e-15:

        f = f / s
        
        obj = fast_convolve(f).max()

    best_obj = fast_convolve(f).max()
    
    it = 0
    stuck = 0
    top_k = top_k_init
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 30 == 0 and top_k < top_k_max:
            top_k = min(top_k + 15, top_k_max)
        
        b = fast_convolve(f)
        M = b.max()
        
        # Select constraints
        if top_k < len(b):
            top_idx = np.argpartition(b, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(b))
        
        # Also include near-active constraints
        threshold = M * 0.98
        near_active = np.where(b >= threshold)[0]
        all_idx = np.unique(np.concatenate([top_idx, near_active]))
        
        # Cap at 400 constraints
        if len(all_idx) > 400:
            all_idx = all_idx[np.argsort(-b[all_idx])][:400]
        all_idx = np.sort(all_idx)
        
        K = len(all_idx)
        A = np.zeros((K, n))
        for row_i in range(K):
            k = int(all_idx[row_i])
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            A[row_i, lo:hi + 1] = f[lo:hi + 1]
        
        # Solve LP: max sum(g) s.t. g >= 0, A@G <= M
        g_var = cp.Variable(n)
        prob = cp.Problem(
            cp.Maximize(cp.sum(g_var)),
            [g_var >= 0, A @ g_var <= M]
        )
        
        g = None
        for solver in [cp.PDLP, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if g_var.value is not None:
                    g_candidate = np.maximum(g_var.value, 0.0)
                    if not (np.any(np.isnan(g_candidate)) or np.any(np.isinf(g_candidate))):
                        g = g_candidate
                        break
            except Exception:
                continue
        
        if g is None:
            # Fallback: negative gradient direction
            top_peaks = np.argsort(b)[-min(5, len(b)):]
            grad = compute_gradient(f, top_peaks)
            g = np.maximum(M - grad, 0.0)
            s_g = g.sum()
            if s_g < 1e-10:
                g = np.ones(n) / n
            else:
                g = g / s_g
        else:
            s_g = g.sum()
            if s_g <= 1.0 + 1e-5:
                stuck += 1
                if stuck > 20:
                    break
                top_k = min(top_k * 2, top_k_max)
                continue
            g = g / s_g
        
        # Line search
        best_t = 0.0
        best_trial = best_obj
        for t in [0.02, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            f_trial = (1.0 - t) * f + t * g
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial = f_trial / s
            obj = fast_convolve(f_trial).max()
            if obj < best_trial:
                best_trial = obj
                best_t = t
        
        if best_t == 0.0:
            stuck += 1
            if stuck > 30:
                break
            continue
        
        f_new = (1.0 - best_t) * f + best_t * g
        f_new = np.maximum(f_new, 0.0)
        f_new = f_new / f_new.sum()
        obj_new = fast_convolve(f_new).max()
        
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 30:
                break
        
        if it % 20 == 0:
            elapsed = time.monotonic() - t0
            score = 2.0 * n * best_obj
            print(f"      LP-Adapt iter={it} obj={best_obj:.10f} score={score:.6f} "
                  f"K={K} t={elapsed:.0f}s")
    
    return best_f, best_obj


def peak_shaving(f_init, time_limit, n_peaks=10):
    """
    Direct peak shaving: iteratively identify the peak and reduce it
    by adjusting the contributing elements proportionally.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    no_improve = 0
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        it += 1
        
        b = fast_convolve(f)
        M = b.max()
        
        # Get top peaks
        if n_peaks < len(b):
            peak_idx = np.argpartition(b, -n_peaks)[-n_peaks:]
        else:
            peak_idx = np.arange(len(b))
        
        # For each peak, compute the "pressure" on each element
        pressure = np.zeros(n)
        for k in peak_idx:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for j in range(lo, hi + 1):
                pressure[j] += f[k - j]
        
        # Normalize pressure
        p_max = pressure.max()
        if p_max < 1e-15:
            break
        pressure = pressure / p_max
        
        # Reduce elements with highest pressure
        for alpha in [0.01, 0.02, 0.05, 0.1, 0.15, 0.2]:
            f_trial = f * (1.0 - alpha * pressure)
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial = f_trial / s
            obj = fast_convolve(f_trial).max()
            if obj < best_obj - 1e-13:
                f = f_trial
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
                break
        
        if no_improve > 100:
            break
        no_improve += 1
        
        if it % 50 == 0:
            score = 2.0 * n * best_obj
            print(f"      PeakShave iter={it} obj={best_obj:.10f} score={score:.6f} "
                  f"t={time.monotonic()-t0:.0f}s")
    
    return best_f, best_obj


def hann_window(n):
    w = 0.5 * (1.0 + np.cos(2.0 * np.pi * np.arange(n) / max(1, n)))
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def flat_top_window(n):
    w = np.ones(n)
    a = [0.21557895, 0.41663158, 0.277263158, 0.083578947, 0.006947368]
    for i, ai in enumerate(a):
        w -= ai * np.cos(2 * np.pi * i * np.arange(n) / max(1, n - 1) - i * np.pi)
    w = np.maximum(w, 0)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def smooth_bump(n):
    x = np.arange(n) / n
    w = np.exp(-40.0 * (x - 0.5) ** 2)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s


def symmetric_gaussian(n, sigma_frac=0.3):
    x = (np.arange(n) - (n - 1) / 2.0) / (n * sigma_frac)
    w = np.exp(-0.5 * x ** 2)
    s = w.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return w / s
