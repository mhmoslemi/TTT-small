def fast_convolve(a):
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def subgradient_peak(a):
    """Compute subgradient of max(convolve(a,a)) at a."""
    n = len(a)
    b = fast_convolve(a)
    P = b.max()
    # Find active peaks (within tolerance)
    tol = 1e-10 * P
    active = np.where(b >= P - tol)[0]
    
    # Subgradient: g[j] = (1/|S|) * sum_{k in S} 2*a[k-j] (with a[i]=0 outside [0,n-1])
    g = np.zeros(n)
    for k in active:
        # a[k-j] is valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        d = k - np.arange(lo, hi + 1)  # d = k - j for j in [lo, hi]
        g[lo:hi+1] += 2 * a[d]
    
    g /= len(active)
    return g, b

def nesterov_gradient_descent(a_init, time_limit):

    n = len(a_init)

    y = a.copy()
    v = np.zeros(n)

    best_a = a.copy()

    lr = 1.0 / n  # initial learning rate

    it = 0

        beta = (1 - t) / t_new
