        for j in range(j_lo, j_hi + 1):
            rows.append(k)
            cols.append(j)
            data.append(f[k - j])
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def solve_best_response(f, M, prev_g=None, frac_constraints=1.0):
    """Solve LP: max sum(g) s.t. C(f)@g <= M, g >= 0. Uses top constraints."""
    n = len(f)
    n_total = 2 * n - 1
    n_cons = max(n, int(n_total * frac_constraints))

    conv_f = fast_convolve(f)
    if n_cons < n_total:
        idx = np.argsort(conv_f)[::-1][:n_cons]
        C = build_conv_matrix(f)
        C_sub = C[idx, :].toarray()
        rhs = M * np.ones(n_cons)
    else:
        C_sub = build_conv_matrix(f).toarray()
        rhs = M * np.ones(n_total)

    g = cp.Variable(n, nonneg=True)
    if prev_g is not None:
        g.value = prev_g
    constraints = [C_sub @ g <= rhs]
    obj = cp.Maximize(cp.sum(g))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.CLARABEL, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            return prev_g if prev_g is not None else f.copy()
    if g.value is None:
        return prev_g if prev_g is not None else f.copy()
    return np.maximum(g.value, 0)


def bb_step(s, y):
    """Barzilai-Borwein step size."""
    sy = np.dot(s, y)
    yy = np.dot(y, y)
    ss = np.dot(s, s)
    if yy < 1e-30:
        return 0.3
    if sy > 1e-15:
        return min(max(sy / yy, 0.05), 2.0)
    return min(max(ss / max(sy, 1e-15), 0.05), 2.0) if sy > 0 else 0.3

        else:

        a_new = np.maximum(a_new, 0)

def peak_targeted_cd(a0, time_limit, n_pairs=500, verbose=True):
    """Coordinate descent targeting the peak convolution position."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score_norm(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    iters = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        peak_idx = conv.argmax()
        improved = False

        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or a[i] < 1e-14:
                continue

            # Check if moving mass from i to j reduces the peak
            # Positions affected: peak_idx could change
            best_local = best_score
            best_t = 0.0
            base = a[i] * 0.4

            for frac in [0.1, 0.3, 0.5]:
                t = frac * base
                if t >= a[i] * 0.9:
                    t = a[i] * 0.4
                a_trial = a.copy()
                a_trial[i] -= t
                a_trial[j] += t
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                sc = compute_score_norm(a_trial)
                if sc < best_local - 1e-14:
                    best_local = sc
                    best_t = t

            if best_t != 0.0:
                a[i] -= best_t
                a[j] += best_t
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = best_local
                best_a = a.copy()
                improved = True
                stall = 0
            else:
                stall += 1

        if not improved and iters > 10:
            noise = rng.standard_normal(n) * 0.003
            a = best_a.copy() + noise
            a = np.maximum(a, 0)
            a = a / a.sum()
            stall = 0

        if verbose and iters % 30 == 0:
            print(f"    CD iter={iters}, best={best_score:.8f}, t={time.monotonic()-t0:.0f}s")

def lbgfs_smooth(a0, time_limit, T=500, maxiter=2000):
    """L-BFGS on smoothed objective in log-space."""
    n = len(a0)
    z0 = np.log(np.maximum(a0, 1e-15))
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score_norm(a0)

    def obj_func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e10
        a = np.exp(z - z.max())
        a = a / a.sum()
        conv = fast_convolve(a)
        shifted = conv / T
        mx = shifted.max()
        lse = mx + np.log(np.exp(shifted - mx).sum())
        return (2.0 * n / T) * np.exp(T * lse)

    def grad_func(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        a = np.exp(z - z.max())
        a = a / a.sum()
        conv = fast_convolve(a)
        shifted = conv / T
        mx = shifted.max()
        w = np.exp(shifted - mx)
        w = w / w.sum()
        size = 2 * n - 1
        fft_sz = 1 << (size - 1).bit_length()
        W = rfft(w, fft_sz)
        A = rfft(a, fft_sz)
        grad_a = 2.0 * irfft(np.conj(W) * A, fft_sz)[:n]
        S = float(np.dot(grad_a, a))
        grad_z = a * (grad_a - S)
        return (2.0 * n / T) * grad_z

    try:
        res = minimize(
            fun=obj_func, x0=z0, jac=grad_func,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12, 'disp': False}
        )
        a_out = np.exp(res.x - res.x.max())
        a_out = a_out / a_out.sum()
        score = compute_score_norm(a_out)
        if score < best_score:
            best_a = a_out
            best_score = score
    except Exception:
        pass
    return best_a, best_score


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

                  f"t={time.monotonic()-t0:.0f}s")

    return best_a, best_score


def bussgang_flatten(a0, time_limit, tau=0.3, verbose=True):
    """Bussgang spectral flattening."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score_norm(a)
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()
    iters = 0
    tau_current = tau

    while time.monotonic() - t0 < time_limit:
        iters += 1
        A = rfft(a, fft_size)
        power = np.abs(A) ** 2
        total_power = power.sum()
        target = total_power / fft_size
        ratio = np.where(power > 1e-30, target / power, 1.0)
        ratio = np.clip(ratio, 0.05, 20.0)
        A_new = A * (ratio ** tau_current)
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss
        score = compute_score_norm(a_new)
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
        a = a_new

        # Gradually reduce tau for finer control
        if iters % 200 == 0:
            tau_current = max(0.1, tau_current * 0.8)

        if verbose and iters % 500 == 0:
            print(f"    Bussgang iter={iters}, best={best_score:.8f}, tau={tau_current:.3f}")

    return best_a, best_score


def geometric_mean_fp(a0, time_limit, t_param=0.25, verbose=True):
    """Geometric mean variant of fictitious play."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score_norm(a)
    t0 = time.monotonic()
    iters = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        conv = fast_convolve(a)
        M = conv.max()
        g = solve_best_response(a, M)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g_norm = g / g_sum
        g_norm = np.maximum(g_norm, 1e-15)

        log_a = np.log(a)
        log_g = np.log(g_norm)
        log_new = (1.0 - t_param) * log_a + t_param * log_g
        a_new = np.exp(log_new - log_new.max())
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss

        score = compute_score_norm(a_new)
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
            stall = 0
        else:
            stall += 1
        a = a_new

        if stall > 200:
            rng = np.random.default_rng(iters)
            a = best_a + rng.standard_normal(n) * 0.002
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            stall = 0

        if verbose and iters % 500 == 0:
            print(f"    GeoFP iter={iters}, best={best_score:.8f}")

    return best_a, best_score
