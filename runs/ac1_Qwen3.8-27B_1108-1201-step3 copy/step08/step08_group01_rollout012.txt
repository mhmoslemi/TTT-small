import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def softmax_weights(values, temperature):
    """Compute softmax weights over values."""
    v = np.asarray(values, dtype=np.float64)
    v = v - v.max()
    w = np.exp(v / temperature)
    s = w.sum()
    if s < 1e-300:
        return np.ones(len(v)) / len(v)
    return w / s

def simplex_project(v):
    """Project onto probability simplex using bisection."""
    n = len(v)
    mu = np.zeros(n)
    sv = np.sort(v)[::-1]
    cumsum = np.cumsum(sv)
    rho = np.where(sv * np.arange(1, n+1) > (cumsum - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cumsum[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def subgradient_at_peak(a, conv, peak_indices, weights):
    """Compute subgradient of max(conv) at a, weighted by peak positions."""
    n = len(a)
    grad = np.zeros(n)
    for k, w in zip(peak_indices, weights):
        k = int(k)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        # grad[i] += 2 * w * a[k-i] for i in [i_start, i_end]
        # a[k-i] for i in [i_start, i_end] is a[k-i_start], ..., a[k-i_end]
        # = a[min(n-1,k)], ..., a[max(0,k-n+1)]
        # which is a reversed slice
        indices = np.arange(i_start, i_end + 1)
        grad[indices] += 2.0 * w * a[k - indices]
    return grad

    n = len(a0)

def softmax_weights(values, temperature):
    """Compute softmax weights over values."""
    v = np.asarray(values, dtype=np.float64)
    v = v - v.max()
    w = np.exp(v / temperature)
    s = w.sum()

def simplex_project(v):
    """Project onto probability simplex."""
    n = len(v)
    sv = np.sort(v)[::-1]
    cumsum = np.cumsum(sv)
    rho = np.where(sv * np.arange(1, n + 1) > (cumsum - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cumsum[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, conv, peak_indices, weights):
    """Compute weighted subgradient of max(conv) at a."""

        i_end = min(n - 1, k)
        indices = np.arange(i_start, i_end + 1)
        grad[indices] += 2.0 * w * a[k - indices]

    factors = np.array([-0.4, -0.3, -0.2, -0.15, -0.1, -0.05,
                        0.05, 0.1, 0.15, 0.2, 0.3, 0.4])

    while time.monotonic() - t0 < time_limit:
        sweeps += 1
        improved = False

        for i in range(n):
            if a[i] < 1e-15:
                continue

            k_start = i
            k_end = min(i + n - 1, 2 * n - 2)
            length = k_end - k_start + 1

            if k_start > 0:
                unaff_left = conv[:k_start].max()
            else:
                unaff_left = -1e30
            if k_end < 2 * n - 2:
                unaff_right = conv[k_end + 1:].max()
            else:
                unaff_right = -1e30
            unaff_max = max(unaff_left, unaff_right)

            a_shifted = a[0:length]

            affected = conv[k_start:k_end + 1]
            best_delta = 0.0
            best_new_max = M
            has_quad = (2 * i >= k_start and 2 * i <= k_end)
            quad_idx = (2 * i - k_start) if has_quad else -1

            for f in factors:
                delta = f * a[i]
                new_val = a[i] + delta
                if new_val < -1e-15:
                    continue

                new_affected = affected + 2.0 * delta * a_shifted
                if has_quad:
                    new_affected = new_affected.copy()
                    new_affected[quad_idx] = conv[2 * i] + 2.0 * delta * a[i] + delta * delta

                new_max = max(unaff_max, float(new_affected.max()))
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_delta = delta

            if best_delta != 0.0 and best_new_max < M - 1e-16:
                a[i] = max(a[i] + best_delta, 0.0)
                sum_a += best_delta

                changes = 2.0 * best_delta * a_shifted
                if has_quad:
                    changes = changes.copy()
                    changes[quad_idx] = 2.0 * best_delta * (a[i] - best_delta) + best_delta * best_delta
                conv[k_start:k_end + 1] += changes
                M = max(unaff_max, float(conv[k_start:k_end + 1].max()))

                cur_score = 2.0 * n * M / (sum_a * sum_a)
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                improved = True

        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps > 30:
                break
        else:
            no_improve_sweeps = 0

        if verbose and sweeps % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD {label} sweep={sweeps} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score

def init_from_parent(parent, n, rng, noise_level=0.01):
    """Interpolate parent to new length with small noise."""
    old_n = len(parent)
    parent_norm = parent / parent.sum()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, n)
    f_interp = np.interp(new_x, old_x, parent_norm)
    f_interp += rng.standard_normal(n) * noise_level
    f_interp = np.maximum(f_interp, 0)
    s = f_interp.sum()
    return f_interp / s


def init_ushape(n, rng, alpha=None):
    """U-shaped initialization: a[j] proportional to (j+1)^(-alpha) * (n-j)^(-alpha)."""
    if alpha is None:
        alpha = rng.uniform(0.3, 1.5)
    t = np.arange(n, dtype=np.float64)
    a = 1.0 / ((t + 1.0) ** alpha * (n - t) ** alpha)
    a += rng.standard_normal(n) * 0.005
    a = np.maximum(a, 0)
    s = a.sum()
    return a / s


def init_fourier(n, rng, n_harm=None):
    """Initialize with random Fourier series."""
    if n_harm is None:
        n_harm = int(rng.integers(5, 30))
    amplitudes = rng.random(n_harm) * 0.25
    phases = rng.uniform(0, 2 * np.pi, n_harm)
    freqs = np.arange(1, n_harm + 1)
    t = np.arange(n, dtype=np.float64)
    a = np.full(n, 0.5)
    for j in range(n_harm):
        a += amplitudes[j] * np.cos(2 * np.pi * freqs[j] * t / n + phases[j])
    a = np.maximum(a, 0)
    s = a.sum()
    return a / s


def init_parametric(n, rng):
    """Try various parametric shapes."""
    t = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    shape = int(rng.integers(0, 5))
    if shape == 0:
        # Power law U-shape
        alpha = rng.uniform(0.2, 1.0)
        a = t ** (-alpha) * (1 - t + 1e-6) ** (-alpha) + 0.1
    elif shape == 1:
        # Cosine-based
        a = 1.0 + rng.uniform(0.3, 1.0) * np.cos(np.pi * t) ** 2
    elif shape == 2:
        # Two-hump
        centers = np.array([0.2, 0.8])
        a = np.zeros(n)
        for c in centers:
            a += np.exp(-0.5 * ((t - c) / 0.15) ** 2)
        a += 0.1
    elif shape == 3:
        # Inverted parabola with noise
        a = 1.0 - 0.5 * (2 * t - 1) ** 2 + rng.uniform(0.1, 0.3)
    else:
        # Exponential tails
        beta = rng.uniform(0.5, 2.0)
        a = np.exp(-beta * np.abs(2 * t - 1))
    a += rng.standard_normal(n) * 0.005
    a = np.maximum(a, 0)
    s = a.sum()
    return a / s


def evaluate_and_track(a, best_score, best_seq, label=""):
    """Evaluate a candidate and return updated best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score
