def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgrad(a, w):
    """
    Compute subgradient of sum_k w[k]*c[k] w.r.t. a, where c=convolve(a,a).
    w has length 2n-1. Returns array of length n.
    Uses np.correlate for efficiency.
    """
    n = len(a)
    # r[j] = sum_m w[j+m] * a[m] which equals sum_k w[k]*a[k-j] for valid k
    r = np.correlate(w, a, mode='valid')
    return 2.0 * r


def softmax_weights(c, temp, n_top=None):
    """Compute softmax weights on convolution peaks."""

        idx = np.argpartition(c, -n_top)[-n_top:]
        w_full = np.zeros(len(c))
        scaled = c[idx] / (temp * M)
        scaled -= scaled.max()
        w_sel = np.exp(scaled)
        w_sel /= w_sel.sum()
        w_full[idx] = w_sel
        return w_full
    else:

        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
