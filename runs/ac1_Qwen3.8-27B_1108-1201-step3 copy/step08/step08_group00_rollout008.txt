"""
Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize

# All helper functions at top level (no closures)

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def get_lp_solver():
    # Try PDLP first (fastest for this problem type)
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS

def solve_lp(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def z_from_a(a):

def nesterov_lp(a0, time_limit, solver=None, max_iters=100000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()

    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new = solve_lp(y, solver)
        if b_new is None:
            continue
        
        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy
        
        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa
        
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
    
    return best_a, best_score

def logsumexp_refine(a0, time_limit, T_start=50.0, T_end=0.1, maxiter=10000):

    z = z_from_a(a0).copy()

            b_v = fast_convolve(a_v)

            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func
    
    def make_jac(T_val):
        def jac(z_v):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_v)
            a_v = softmax_from_z(z_v)
            b_v = fast_convolve(a_v)
            Tb = T_val * b_v
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b_v)):
                if w[k] < 1e-15:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)

                grad_a[idx] += 2.0 * w[k] * a_v[k - idx]
            S = np.dot(grad_a, a_v)
            return a_v * (grad_a - S)
        return jac
    
    n_stages = 8
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(30, maxiter // n_stages)
    
    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        try:

                fun=make_func(stage_T),
                x0=z,

            z = res.x
        except Exception:

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find the fastest working LP solver."""
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp(a_fixed, solver=None):

    if solver is None:

    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):

    b_val = np.asarray(b_var.value, dtype=np.float64)

def softmax_from_z(z):

    if s < 1e-300:
        return np.ones(len(z)) / len(z)

        je = min(n - 1, k)
        idx = np.arange(js, je + 1)

    S = np.dot(grad_a, a)

    t4 = min(rem * 0.15, 200.0)
    if t4 > 20:
        a_lp2, _ = nesterov_lp(best_seq, t4, solver=solver, max_iters=100000)
        sc = evaluate_safe(a_lp2)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_lp2.copy()
            print(f"  NEW BEST [lp_r2]: {sc:.6f}")

    # Phase 5: Mirror descent
    print(f"\n=== Phase 5: Mirror Descent (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.08, 100.0)
    if t5 > 10:
        a_md, _ = mirror_descent(best_seq, t5)
        sc = evaluate_safe(a_md)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_md.copy()
            print(f"  NEW BEST [mirror]: {sc:.6f}")

    # Phase 6: Multi-length exploration
    print(f"\n=== Phase 6: Multi-Length (best={best_score:.6f}) ===")
    alt_lengths = set()
    base = n_best
    for mult in [0.92, 1.08, 0.85, 1.15, 1.25]:
        alt_n = int(round(base * mult))
        if 512 <= alt_n <= 2048 and alt_n != base:
            alt_lengths.add(alt_n)
    for alt_n in sorted(alt_lengths):
        rem = deadline - time.monotonic()
        if rem < 120:
            break
        t_alloc = min(rem * 0.25, 250.0)
        if t_alloc < 40:
            break
        print(f"  n={alt_n}, budget={t_alloc:.0f}s")
        init_alt = interpolate_seq(best_seq, alt_n)
        a_alt, _ = nesterov_lp(init_alt, t_alloc * 0.65, solver=solver, max_iters=50000)
        sc = evaluate_safe(a_alt)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_alt.copy()
            print(f"  NEW BEST [lp_n{alt_n}]: {sc:.6f}")

        rem = deadline - time.monotonic()
        if rem > 40 and best_seq is not None and len(best_seq) == alt_n:
            a_alt2 = logsumexp_refine(best_seq, min(rem * 0.25, 80.0),
                                       T_start=10.0, T_end=0.3, maxiter=5000)
            sc2 = evaluate_safe(a_alt2)
            if np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_seq = a_alt2.copy()
                print(f"  NEW BEST [lse_n{alt_n}]: {sc2:.6f}")

    # Phase 7: Perturbation escape with re-optimization
    print(f"\n=== Phase 7: Perturbation Escape (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 100:
        n_cur = len(best_seq)
        for trial in range(4):
            rem = deadline - time.monotonic()
            if rem < 50:
                break
            noise_scale = 0.02 + 0.015 * trial
            perturbed = best_seq + noise_scale * rng.standard_normal(n_cur)
            perturbed = np.maximum(perturbed, 0)
            sp = perturbed.sum()
            if sp < 1e-15:
                continue
            perturbed = perturbed / sp
            t_pert = min(rem * 0.35, 80.0)
            a_p, _ = nesterov_lp(perturbed, t_pert, solver=solver, max_iters=30000)
            sc = evaluate_safe(a_p)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_p.copy()
                print(f"  NEW BEST [pert_{trial}]: {sc:.6f}")

    # Phase 8: Final ultra-low-T LogSumExp polish
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.50, 100.0)
        a_fin = logsumexp_refine(best_seq, t_fin, T_start=3.0, T_end=0.01, maxiter=10000)
        sc = evaluate_safe(a_fin)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_fin.copy()
            print(f"  NEW BEST [final_lse]: {sc:.6f}")

    rem = deadline - time.monotonic()
    if rem > 20:
        a_fin2, _ = nesterov_lp(best_seq, min(rem * 0.5, 40.0), solver=solver, max_iters=20000)
        sc = evaluate_safe(a_fin2)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_fin2.copy()
            print(f"  NEW BEST [final_lp]: {sc:.6f}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
