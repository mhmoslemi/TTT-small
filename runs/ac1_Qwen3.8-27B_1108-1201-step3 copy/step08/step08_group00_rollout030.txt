def softmax_from_z(z):
    """Convert log-domain z to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))

def adam_smooth_optimize(a0, time_limit, T_start=80.0, T_end=0.3, 
                          lr_init=0.15, max_iters=500000, n=0):
    """
    Adam optimizer in log-domain with adaptive temperature for smooth objective.
    Minimizes F_T(a) = (2n/T) * logsumexp(T * conv(a,a)) as approximation to 2n*max(conv(a,a)).
    """
    if n == 0:
        n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    z = z_from_a(a)
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8
    lr = lr_init
    
    # Nesterov momentum (in z-space)
    z_prev = z.copy()
    momentum = 0.0
    momentum_target = 0.6
    
    stall_count = 0
    prev_best = best_score
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        a = softmax_from_z(z)
        c = fast_convolve(a)
        
        # Adaptive temperature: decrease when losing
        progress = (it + 1) / max_iters
        T = T_start * (T_end / T_start) ** (progress ** 0.7)
        T = max(T, T_end)
        
        # Compute smooth gradient
        Tc = T * c
        max_Tc = Tc.max()
        exp_Tc = np.exp(Tc - max_Tc)
        w = exp_Tc / exp_Tc.sum()
        
        grad_a = smooth_gradient(a, w, n)
        
        # Gradient in z-space
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)
        
        # Nesterov momentum
        if it > 0:
            momentum = momentum + (momentum_target - momentum) * 0.01
            grad_z_momentum = grad_z + momentum * (grad_z - (grad_z - grad_z))  # just use grad_z
            # Simple momentum
            grad_z = grad_z * (1 + momentum * 0.5)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * grad_z ** 2
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        
        # Learning rate schedule
        lr_t = lr_init / (1.0 + it * 0.0002)
        lr_t = max(lr_t, 0.01 * lr_init)
        
        z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        
        # Evaluate true objective (periodically)
        if it % 50 == 0 or it < 10:
            a_new = softmax_from_z(z)
            score = 2.0 * n * fast_convolve(a_new).max()
            if score < best_score - 1e-10:
                best_score = score
                best_a = a_new.copy()
                stall_count = 0
            else:
                stall_count += 1
        
        # Adaptive: if stalled, add perturbation
        if stall_count > 200:
            noise_scale = 0.02 * (1.0 + stall_count / 100.0)
            z = z + noise_scale * np.random.standard_normal(n)
            stall_count = 0
        
        # Progress print
        if it % 10000 == 0 and it > 0:
            elapsed = time.monotonic() - t0
            a_check = softmax_from_z(z)
            sc = 2.0 * n * fast_convolve(a_check).max()
            print(f"    AdamSmooth it={it} ({elapsed:.0f}s): T={T:.2f} score={sc:.6f} best={best_score:.6f}")
    
    # Final evaluation
    a_final = softmax_from_z(z)
    score_final = 2.0 * n * fast_convolve(a_final).max()
    if score_final < best_score:
        best_score = score_final
        best_a = a_final.copy()
    
    return best_a, best_score


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=50000):
    """Nesterov dual averaging LP refinement."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall_count = 0
    prev_best = best_score

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            break
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if it % 200 == 0:
            if best_score < prev_best - 1e-7:
                stall_count = 0
            else:
                stall_count += 1
            prev_best = best_score
            if stall_count >= 8:
                noise = np.random.default_rng(it).random(n) * 0.02
                y = y + noise
                y = np.maximum(y, 0)
                y = y / y.sum()
                stall_count = 0

        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def lbgfs_refine(a0, time_limit, T_start=50.0, T_end=0.1, maxiter=8000):
    """L-BFGS with LogSumExp smoothing and geometric temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    def make_func(T_val):
        def func(zv):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            av = softmax_from_z(zv)
            b = fast_convolve(av)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(zv):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(zv)
            av = softmax_from_z(zv)
            b = fast_convolve(av)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_av = smooth_gradient(av, w, n)
            S = np.dot(grad_av, av)
            return av * (grad_av - S)
        return jac

    n_stages = 8
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(30, maxiter // n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-14,
                         'maxfun': iters_per_stage * 100, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def peak_shave_local(a0, time_limit, n_steps=10000):
    """Multi-peak shaving with adaptive coordinate adjustments."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        b_work = b.copy()
        if len(b_work) > 2:
            b_work[0] = -np.inf
            b_work[-1] = -np.inf
        peak_k = int(np.argmax(b_work))
        peak_val = b_work[peak_k]
        near_peak_threshold = peak_val - 0.02 * peak_val
        near_peaks = np.where(b_work >= near_peak_threshold)[0]

        # Find top contributors
        contrib_count = np.zeros(n)
        for k in near_peaks[:8]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[k - idx]
            top_idx = np.argsort(contribs)[::-1][:8]
            for ci in top_idx:
                contrib_count[idx[ci]] += 1

        top_coords = np.argsort(contrib_count)[::-1][:12]

        improved = False
        for i in top_coords:
            if a[i] < 1e-14:
                continue
            for factor in [0.97, 0.94, 0.90, 0.85]:
                new_val = a[i] * factor
                diff = a[i] - new_val
                if diff < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                if others.sum() < 1e-15:
                    continue
                a_trial[mask] = others + diff * others / others.sum()
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-15:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Pair reduction
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_contrib = np.argsort(contribs)[::-1][:6]
            for ci in top_contrib:
                i = idx[ci]
                j = peak_k - i
                if i == j or i >= n or j >= n:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.94, 0.90, 0.85]:
                    for f2 in [0.94, 0.90, 0.85]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-15:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            # Simultaneous reduction
            a_trial = a.copy()
            for i in top_coords[:5]:
                if a[i] > 1e-14:
                    a_trial[i] *= 0.96
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-15:
                a = a_trial
                best_score = score
                best_a = a.copy()
            elif step > 500:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Create structured initial sequence."""
    choice = rng.integers(0, 8)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.cos(np.pi * x / 2) ** 2
    elif choice == 1:
        a = np.exp(-4 * x * (1 - x))
    elif choice == 2:
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
        a = a ** 2
    elif choice == 4:
        a = np.exp(-2 * (x - 0.5) ** 2)
    elif choice == 5:
        # Triangular
        a = 1 - np.abs(2 * x - 1)
    elif choice == 6:
        # Flat with cosine taper
        taper = 0.3 * np.cos(np.pi * x) ** 2
        a = 1.0 + taper
    else:
        # Fejer-like kernel
        a = 1 - x
        a = a ** 1.5

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def evaluate_safe(a):
    """Evaluate sequence safely."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (non-negative, sum=1), find b minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def multi_start_adam(a0, time_limit, n_starts=3, rng=None):
    """Run Adam smooth optimization from multiple perturbed starting points."""
    if rng is None:
        rng = np.random.default_rng(0)
    n = len(a0)
    best_a = a0.copy()
    best_score = 2.0 * n * fast_convolve(a0).max()
    
    t_per_start = time_limit / n_starts
    for i in range(n_starts):
        rem = time_limit - (time.monotonic() - (time.monotonic() - time_limit))
        if rem < 10:
            break
        
        if i == 0:
            start = a0.copy()
        else:
            # Perturbed start
            noise = rng.standard_normal(n) * 0.08
            start = a0 + noise
            start = np.maximum(start, 0)
            start = start / start.sum()
        
        a_res, score_res = adam_smooth_optimize(start, t_per_start * (1 - i * 0.15),
                                                  T_start=60.0, T_end=0.3)
        if score_res < best_score:
            best_score = score_res
            best_a = a_res.copy()
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization combining Adam smooth optimization, Nesterov LP,
    L-BFGS, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_init = 960
        best_seq = structured_init(n_init, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_init}, score={best_score:.6f}")

    n_best = len(best_seq)

    def try_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {sc:.6f} (n={len(a)})")

    # === Phase 1: Adam smooth optimization (main phase, most time) ===
    print(f"\n=== Phase 1: Adam Smooth n={n_best} (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.40, 700.0)
    if t_alloc > 60:
        print(f"  Budget: {t_alloc:.0f}s, T: 100->0.2")
        a_adam, _ = adam_smooth_optimize(best_seq, t_alloc, 
                                          T_start=100.0, T_end=0.2,
                                          lr_init=0.12, max_iters=1000000)
        try_update(a_adam, "adam_1")

    # === Phase 2: Nesterov LP refinement ===
    print(f"\n=== Phase 2: Nesterov LP (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 400.0)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a_lp, _ = nesterov_lp_refine(best_seq, t_alloc, solver=solver, n_max_iters=80000)
        try_update(a_lp, "nesterov_lp_1")

    # === Phase 3: L-BFGS with very low temperature ===
    print(f"\n=== Phase 3: L-BFGS low-T (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.15, 200.0)
    if t_alloc > 20:
        print(f"  L-BFGS T: 20->0.1, budget={t_alloc:.0f}s")
        a_lb = lbgfs_refine(best_seq, t_alloc, T_start=20.0, T_end=0.1, maxiter=10000)
        try_update(a_lb, "lbgfs_low")

    # === Phase 4: Adam again from LP-refined point ===
    print(f"\n=== Phase 4: Adam from LP point (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.12, 150.0)
    if t_alloc > 20:
        a_adam2, _ = adam_smooth_optimize(best_seq, t_alloc,
                                           T_start=30.0, T_end=0.1,
                                           lr_init=0.08, max_iters=500000)
        try_update(a_adam2, "adam_2")

    # === Phase 5: Peak shaving ===
    print(f"\n=== Phase 5: Peak Shaving (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.08, 80.0)
    if t_alloc > 10:
        a_ps, _ = peak_shave_local(best_seq, t_alloc, n_steps=20000)
        try_update(a_ps, "peak_shave")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-Length (best={best_score:.6f}) ===")
    alt_lengths = []
    n_cur = len(best_seq)
    for mult in [1.25, 1.5, 2.0, 0.75, 0.67]:
        alt_n = int(n_cur * mult)
        if 256 <= alt_n <= 4096 and alt_n != n_cur:
            alt_lengths.append(alt_n)
    alt_lengths = sorted(set(alt_lengths), key=lambda x: abs(x - n_cur))

    for alt_n in alt_lengths[:3]:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_total = min(rem * 0.30, 180.0)
        if t_total < 30:
            break
        print(f"  n={alt_n}, budget={t_total:.0f}s")

        init_alt = interpolate_sequence(best_seq, alt_n)
        
        # Adam optimization for this length
        t_adam = t_total * 0.6
        a_alt, _ = adam_smooth_optimize(init_alt, t_adam,
                                         T_start=50.0, T_end=0.3,
                                         lr_init=0.10, max_iters=300000)
        try_update(a_alt, f"adam_n{alt_n}")

        rem = deadline - time.monotonic()
        if rem > 20 and best_seq is not None and len(best_seq) == alt_n:
            t_lp_alt = min(rem * 0.3, 60.0)
            if t_lp_alt > 15:
                a_alt2, _ = nesterov_lp_refine(best_seq, t_lp_alt, solver=solver, n_max_iters=10000)
                try_update(a_alt2, f"lp_n{alt_n}")

    # === Phase 7: Perturbation escape + re-optimization ===
    print(f"\n=== Phase 7: Perturbation (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 60:
        t_pert = min(rem * 0.25, 120.0)
        n_cur = len(best_seq)
        
        # Multiple perturbations
        for pert_scale in [0.03, 0.06, 0.10]:
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            perturbed = best_seq + pert_scale * rng.standard_normal(n_cur)
            perturbed = np.maximum(perturbed, 0)
            perturbed = perturbed / perturbed.sum()
            
            t_sub = min(rem * 0.35, 40.0)
            a_pert, _ = adam_smooth_optimize(perturbed, t_sub,
                                              T_start=20.0, T_end=0.5,
                                              lr_init=0.08, max_iters=200000)
            try_update(a_pert, f"pert_{pert_scale}")

    # === Phase 8: Final polish ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_fin = min(rem * 0.30, 60.0)
        a_fin = lbgfs_refine(best_seq, t_fin, T_start=5.0, T_end=0.05, maxiter=5000)
        try_update(a_fin, "final_lbgfs")

    rem = deadline - time.monotonic()
    if rem > 20:
        t_ps = min(rem * 0.25, 40.0)
        a_ps2, _ = peak_shave_local(best_seq, t_ps, n_steps=15000)
        try_update(a_ps2, "final_ps")

    rem = deadline - time.monotonic()
    if rem > 15:
        t_lp_fin = min(rem * 0.4, 30.0)
        a_lp_fin, _ = nesterov_lp_refine(best_seq, t_lp_fin, solver=solver, n_max_iters=5000)
        try_update(a_lp_fin, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq) if best_seq is not None else 0})")
    return [float(x) for x in best_seq]
