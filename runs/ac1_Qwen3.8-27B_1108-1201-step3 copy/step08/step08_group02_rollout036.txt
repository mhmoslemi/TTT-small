This is straightforward to compute.

"""
Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses smooth optimization (L-BFGS with soft-max) combined with LP refinement
and coordinate descent for final polish.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_gradient_soft_max(a, t):
    """Compute gradient of soft_max(conv(a,a), t) w.r.t. a."""
    c = fast_convolve(a)
    c_shifted = c - c.max()
    w = np.exp(c_shifted / t)
    w_sum = w.sum()
    w /= w_sum
    # grad[j] = 2 * sum_m w[m] * a[m-j] (with bounds)
    # This is 2 * conv(w, a)[j] truncated to length n
    n = len(a)
    # conv(w, a) has length len(w) + n - 1
    wa = np.convolve(w, a)  # This might be slow for large arrays
    # Actually, let's use FFT
    fft_size = 1 << (len(w) + n - 1 - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    wa_fft = irfft(W * A, fft_size)[:len(w) + n - 1]
    # We need wa_fft[j] for j = 0, ..., n-1
    # But wait, conv(w, a)[j] = sum_i w[i]*a[j-i] which is what we want
    # Actually we need: grad[j] = 2 * sum_{m: 0<=m-j<2n-1} w[m] * a[m-j]
    # = 2 * (w * a)[j] where (w*a) is the linear convolution
    # But w has length 2n-1, a has length n
    # conv(w, a)[j] = sum_i w[i]*a[j-i] for valid i
    # For j in [0, n-1], this gives us what we want
    grad = 2.0 * wa_fft[:n]
    return grad

def soft_max_conv(a, t):
    """Compute soft_max(conv(a,a), t)."""
    c = fast_convolve(a)
    c_max = c.max()
    return c_max + t * np.log(np.sum(np.exp((c - c_max) / t)))

def softmax_param_grad(theta, t):
    """Gradient of soft_max_conv(softmax(theta), t) w.r.t. theta."""
    a = np.exp(theta - theta.max())
    a_sum = a.sum()
    a = a / a_sum
    grad_a = compute_gradient_soft_max(a, t)
    # dF/dtheta[j] = grad_a[j]*a[j] - a[j]*sum_i grad_a[i]*a[i]
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    return grad_theta, a

def soft_max_conv_params(theta, t):
    a = np.exp(theta - theta.max())
    a = a / a.sum()
    return soft_max_conv(a, t)

def run_smooth_optimize(a_init, time_limit, label="smooth"):
    """L-BFGS smooth optimization with decreasing temperature."""
    t0 = time.monotonic()
    n = len(a_init)
    a = a_init.copy()
    S = a.sum()
    a = a / S
    theta = np.log(a + 1e-30)
    best_a = a.copy()

    # Start with high temperature and decrease
    t_init = 0.05 * best_val
    t_final = 0.001 * best_val
    
    current_t = t_init
    total_iters = 0
    
    while time.monotonic() - t0 < time_limit:
        # Phase of optimization at current temperature
        phase_time = min(30.0, time_limit - (time.monotonic() - t0))
        phase_end = time.monotonic() + phase_time
        
        def obj(grad=True):
            val = soft_max_conv_params(theta, current_t)
            if not grad:
                return val
            g, a_curr = softmax_param_grad(theta, current_t)
            return val, g
        
        # Use L-BFGS
        result = minimize(
            lambda th: obj(True)[0] if isinstance(obj(True), tuple) else obj(False),
            theta,
            jac=lambda th: obj(True)[1],
            method='L-BFGS-B',
            options={'maxiter': 1000, 'maxfun': 5000, 'gtol': 1e-8}
        )
        
        theta = result.x
        a = np.exp(theta - theta.max())
        a = a / a.sum()
        
        cur_val = fast_convolve(a).max()
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            
        # Decrease temperature
        current_t *= 0.7
        
        if current_t < t_final:
            current_t = t_final
            # Check if we should stop
            if time.monotonic() - t0 > time_limit * 0.8:
                break
        
        total_iters += 1
        if total_iters % 3 == 0:
            score = 2.0 * n * best_val
            print(f"    {label} t_sched={current_t:.2e} best_val={best_val:.10f} score={score:.6f} time={time.monotonic()-t0:.0f}s")
    
    return best_a, best_val

    Returns (g, M).

Optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

# Helper functions (top-level, no closures, no lambdas)

    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(f, n):
    conv_len = 2 * n - 1
    A = np.zeros((conv_len, n))
    for j in range(n):
        A[j:j+n, j] = f
    return A

def project_simplex(v, target_sum=1.0):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_direct_lp(f, n, solver_list=None):

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Direct LP optimization (min cross-convolution peak)
2. Smooth L-BFGS with soft-max approximation
3. Coordinate descent polish
4. Multi-length exploration

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(f, n):
    """Build matrix A where A @ g = conv(f, g) (full linear convolution)."""
    conv_len = 2 * n - 1
    A = np.zeros((conv_len, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(np.asarray(v, dtype=np.float64))[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(np.asarray(v, dtype=np.float64) - theta, 0.0)


def solve_direct_lp(f, n, solver_list=None, max_iters=100000):
    """
    Solve: minimize M s.t. A@G <= M (element-wise), G >= 0, sum(G) = 1.
    Returns (g, M_value) or (None, None).
    """
    if solver_list is None:
        solver_list = [cp.PDLp, cp.SCS, cp.ECOS]
    
    conv_len = 2 * n - 1
    A_mat = build_conv_matrix(f, n)
    
    g_var = cp.Variable(n)
    M_var = cp.Variable()
    
    constraints = [
        g_var >= 0,
        cp.sum(g_var) == 1,
        A_mat @ g_var <= M_var
    ]
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    for solver in solver_list:
        try:
            prob.solve(solver=solver, max_iters=max_iters)
            if g_var.value is not None:
                g_val = np.maximum(g_var.value, 0.0)
                if np.any(np.isnan(g_val)) or np.any(np.isinf(g_val)):
                    continue
                s = g_val.sum()
                if s < 1e-10:
                    continue
                g_val /= s
                M_val = float(M_var.value)
                if np.isfinite(M_val):
                    return g_val, M_val
        except Exception:
            continue
    
    return None, None


def line_search(f, g, M_current, n_evals=15):
    """Find best t in (0,1] minimizing max(conv((1-t)f+tg, (1-t)f+tg))."""
    best_t = 0.0
    best_obj = M_current
    
    # Log-spaced grid
    t_vals = np.logspace(-2, 0, n_evals)  # 0.01 to 1.0
    t_vals = np.minimum(t_vals, 0.99)
    
    for t in t_vals:
        f_trial = (1.0 - t) * f + t * g
        f_trial = np.maximum(f_trial, 0.0)
        s = f_trial.sum()
        if s < 1e-10:
            continue
        f_trial /= s
        obj = fast_convolve(f_trial).max()
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_t = t
    
    # Refine
    if best_t > 0.01:
        t_range = np.linspace(max(0.005, best_t * 0.7), min(0.98, best_t * 1.3), 8)
        for t in t_range:
            f_trial = (1.0 - t) * f + t * g
            f_trial = np.maximum(f_trial, 0.0)
            s = f_trial.sum()
            if s < 1e-10:
                continue
            f_trial /= s
            obj = fast_convolve(f_trial).max()
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_t = t
    
    return best_t, best_obj

def smooth_obj_and_grad(theta, temp):
    """Objective and gradient for smooth optimization."""
    theta_max = theta.max()
    a = np.exp(theta - theta_max)
    a_sum = a.sum()
    a = a / a_sum
    n = len(a)
    
    # Convolution
    c = fast_convolve(a)
    
    # Soft-max
    c_max = c.max()
    scaled = (c - c_max) / temp
    log_sum = c_max / temp + np.log(np.sum(np.exp(scaled)))
    obj = log_sum * temp  # = c_max + temp * log(sum(exp((c-c_max)/temp)))
    
    # Soft-max weights
    w = np.exp(scaled)
    w_sum = w.sum()
    w = w / w_sum
    
    # Gradient w.r.t. a: grad_a[j] = 2 * sum_m w[m] * a[m-j]
    # = 2 * (w * a)[j] where * is linear convolution, take first n elements
    conv_len = len(w) + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    wa = irfft(W * A, fft_size)[:conv_len]
    grad_a = 2.0 * wa[:n]
    
    # Gradient w.r.t. theta (softmax parametrization)
    inner = np.dot(grad_a, a)
    grad_theta = a * (grad_a - inner)
    
    return obj, grad_theta


def run_smooth_optimize(f_init, time_limit, label="smooth"):
    """L-BFGS smooth optimization with decreasing temperature."""
    t0 = time.monotonic()
    n = len(f_init)
    a = f_init.copy()
    s = a.sum()
    if s > 1e-10:
        a /= s

    best_a = a.copy()
    M_init = fast_convolve(a).max()
    best_b = M_init
    
    temp = 0.05 * M_init
    temp_min = 0.001 * M_init
    
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Run L-BFGS for a batch of iterations
        batch_iter = 200
        
        def obj_func(th):
            return smooth_obj_and_grad(th, temp)[0]
        
        def grad_func(th):
            return smooth_obj_and_grad(th, temp)[1]
        
        result = minimize(
            obj_func, theta, jac=grad_func,
            method='L-BFGS-B',
            options={'maxiter': batch_iter, 'gtol': 1e-10, 'maxls': 20}
        )
        
        theta = result.x
        a = np.exp(theta - theta.max())
        a = a / a.sum()
        
        cur_b = fast_convolve(a).max()
        if cur_b < best_b - 1e-12:

        # Decrease temperature
        temp *= 0.8
        if temp < temp_min:
            temp = temp_min
        
        if no_improve > 5:
            break
    
    return best_a, best_b


def coordinate_descent(f_init, time_limit, label="cd"):
    """Coordinate descent: optimize one element at a time."""
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    best_b = fast_convolve(f).max()
    
    multipliers = np.array([0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0])
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-15:
                # Try adding a small value
                for val in [0.001, 0.005, 0.01]:
                    f_trial = f.copy()
                    f_trial[j] = val
                    s = f_trial.sum()
                    f_trial /= s
                    obj = fast_convolve(f_trial).max()
                    if obj < best_b - 1e-12:
                        f = f_trial
                        best_b = obj
                        improved = True
                        break
                continue
            
            best_j_val = f[j]
            best_j_obj = fast_convolve(f).max()
            
            for mult in multipliers:
                new_val = f[j] * mult
                f_trial = f.copy()
                f_trial[j] = new_val
                s = f_trial.sum()
                if s < 1e-10:
                    continue
                f_trial /= s
                obj = fast_convolve(f_trial).max()
                if obj < best_j_obj - 1e-12:
                    best_j_obj = obj
                    best_j_val = new_val
            
            if best_j_val != f[j]:
                f[j] = best_j_val
                s = f.sum()
                f /= s
                if best_j_obj < best_b - 1e-12:
                    best_b = best_j_obj
                    improved = True
        
        if not improved:
            break
    
    return f, best_b


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    n = len(f)
    old_x = np.linspace(0.0, 1.0, n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_new = np.interp(new_x, old_x, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s > 1e-10:
        f_new /= s
    else:
        f_new = np.ones(new_n) / new_n
    return f_new
