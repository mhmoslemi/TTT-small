def fast_convolve(a, b=None):
    """Linear convolution of a with b (or with itself)."""
    if b is None:
        b = a
    n_a, n_b = len(a), len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def build_sparse_conv_matrix(a):

            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using sparse LP-based best response
with Nesterov acceleration, multi-length exploration, and L-BFGS LogSumExp
refinement. Key improvements: sparse matrix LP enabling larger n, systematic
length sweep, adaptive time allocation, and dual-guided perturbation.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
from scipy.optimize import minimize


def fast_convolve(a, b=None):
    """Linear convolution via FFT. If b is None, computes auto-convolution."""
    if b is None:
        b = a
    n_a = len(a)
    n_b = len(b)
    size = n_a + n_b - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def build_sparse_conv_matrix(a):
    """Build sparse matrix T (2n-1 x n) such that T @ b = convolve(a, b)."""
    n = len(a)
    size = 2 * n - 1
    total_nonzeros = sum(min(k + 1, n, size - k) for k in range(size))
    rows = np.empty(total_nonzeros, dtype=np.int32)
    cols = np.empty(total_nonzeros, dtype=np.int32)
    vals = np.empty(total_nonzeros, dtype=np.float64)
    a = np.asarray(a, dtype=np.float64)
    pos = 0
    for k in range(size):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        cnt = hi - lo + 1
        rows[pos:pos + cnt] = k
        cols[pos:pos + cnt] = np.arange(lo, hi + 1, dtype=np.int32)
        idx = np.arange(lo, hi + 1)
        vals[pos:pos + cnt] = a[k - idx]
        pos += cnt
    return csr_matrix((vals, (rows, cols)), shape=(size, n))


_solver_cache = None


def get_lp_solver():
    """Find a working LP solver, preferring CLARABEL or ECOS."""
    global _solver_cache
    if _solver_cache is not None:
        return _solver_cache
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            ones = np.ones(5)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [x @ ones == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _solver_cache = solver_name
                return solver_name
        except Exception:
            continue
    _solver_cache = cp.ECOS
    return _solver_cache


def solve_lp_best_response(a, solver=None, max_constr=None):
    """Solve LP: min M s.t. conv(a,b)[k] <= M for all k, sum(b)=1, b>=0."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a)
    T = build_sparse_conv_matrix(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    ones = cp.Constant(np.ones(2 * n - 1))
    constraints = [T @ b_var <= M_var * ones, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        for alt in [cp.ECOS, cp.CLARABEL, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None, None
    b_val = b_val / s
    # Get dual variables for active constraints
    try:
        duals = np.asarray(T @ b_var.value - M_var.value, dtype=np.float64)
    except Exception:
        duals = None
    return b_val, float(M_var.value), duals


def softmax_from_z(z):
    """Convert log-domain to probability vector."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def nesterov_lp_refine(a0, time_limit, solver=None, n_max_iters=100000, verbose=False):
    """Nesterov dual averaging LP refinement with stall detection."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    stall = 0
    prev_best = best_score
    consecutive_fail = 0

    for it in range(n_max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_val, _ = solve_lp_best_response(y, solver)
        if b_new is None:
            consecutive_fail += 1
            if consecutive_fail > 20:
                break
            continue
        consecutive_fail = 0

        alpha = 2.0 / (it + 2)
        y = (1 - alpha) * y + alpha * b_new
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
            continue
        y = y / sy

        a = ((it + 1) * a + (alpha * (it + 2)) * b_new) / (it + 1 + alpha * (it + 2))
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa < 1e-15:
            a = y.copy()
        else:
            a = a / sa

        score_a = 2.0 * n * fast_convolve(a).max()
        score_y = 2.0 * n * fast_convolve(y).max()

        if score_a < best_score - 1e-10:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        if score_y < best_score - 1e-10:
            best_score = score_y
            best_a = y.copy()
            stall = 0

        if it % 100 == 0:
            if best_score < prev_best - 1e-8:
                stall = 0
            else:
                stall += 1
            prev_best = best_score
            if verbose:
                elapsed = time.monotonic() - t0
                print(f"      NLP iter={it+1} ({elapsed:.0f}s): score={best_score:.6f}")
            if stall >= 10:
                rng = np.random.default_rng(hash((n, it)) & 0xFFFFFFFF)
                noise_scale = 0.03 * (1 + stall * 0.1)
                y = y + noise_scale * rng.standard_normal(n)
                y = np.maximum(y, 0)
                y = y / y.sum()
                a = (0.7 * a + 0.3 * y)
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0

    return best_a, best_score


def lbgfs_refine(a0, time_limit, T_start=150.0, T_end=0.2, n_stages=8):
    """L-BFGS with LogSumExp smoothing and geometric temperature annealing."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()

    def make_func(T_val):
        def func(z_val):
            if time.monotonic() - t0 > time_limit:
                return 1e15
            a = softmax_from_z(z_val)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            log_sum_exp = max_Tb + np.log(exp_Tb.sum())
            return (2.0 * n / T_val) * log_sum_exp
        return func

    def make_jac(T_val):
        def jac(z_val):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(z_val)
            a = softmax_from_z(z_val)
            b = fast_convolve(a)
            Tb = T_val * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            grad_a = np.zeros(n)
            for k in range(len(b)):
                if w[k] < 1e-14:
                    continue
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                grad_a[idx] += 2.0 * w[k] * a[k - idx]
            S = np.dot(grad_a, a)
            return a * (grad_a - S)
        return jac

    temps = np.geomspace(T_start, T_end, n_stages)
    maxiter_total = 10000
    iters_per_stage = max(80, maxiter_total // n_stages)

    for stage_T in temps:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        try:
            res = minimize(
                fun=make_func(stage_T),
                x0=z,
                jac=make_jac(stage_T),
                method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18, 'gtol': 1e-13,
                         'maxfun': iters_per_stage * 200, 'disp': False}
            )
            z = res.x
        except Exception:
            pass

    return softmax_from_z(z)


def peak_shave_local(a0, time_limit, n_steps=15000):
    """Multi-peak shaving with coordinate adjustments targeting top peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        # Find top peaks (excluding boundary)
        b_work = b.copy()
        if len(b_work) > 2:
            b_work[0] = -np.inf
            b_work[-1] = -np.inf
        peak_k = int(np.argmax(b_work))
        peak_val = b_work[peak_k]
        threshold = peak_val - 0.02 * peak_val
        near_peaks = np.where(b_work >= threshold)[0]

        # Find top contributors
        contrib_count = np.zeros(n)
        for k in near_peaks[:8]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[k - idx]
            total_c = contribs.sum()
            if total_c < 1e-20:
                continue
            top_idx = np.argsort(contribs)[::-1][:6]
            for ci in top_idx:
                contrib_count[idx[ci]] += contribs[ci] / total_c

        top_coords = np.argsort(contrib_count)[::-1][:12]
        improved = False

        for i in top_coords:
            if a[i] < 1e-14:
                continue
            for factor in [0.97, 0.94, 0.90, 0.85, 0.80]:
                new_val = a[i] * factor
                diff = a[i] - new_val
                if diff < 1e-16:
                    continue
                a_trial = a.copy()
                a_trial[i] = new_val
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                if others.sum() < 1e-15:
                    continue
                a_trial[mask] = others + diff * others / others.sum()
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Pair reduction
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_contrib = np.argsort(contribs)[::-1][:5]
            for ci in top_contrib:
                i = idx[ci]
                j = peak_k - i
                if i == j or i >= n or j >= n:
                    continue
                if a[i] < 1e-14 or a[j] < 1e-14:
                    continue
                for f1 in [0.93, 0.88]:
                    for f2 in [0.93, 0.88]:
                        a_trial = a.copy()
                        a_trial[i] *= f1
                        a_trial[j] *= f2
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            # Simultaneous reduction
            a_trial = a.copy()
            for i in top_coords[:4]:
                if a[i] > 1e-14:
                    a_trial[i] *= 0.96
            a_trial = a_trial / a_trial.sum()
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score - 1e-14:
                a = a_trial
                best_score = score
                best_a = a.copy()
            else:

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length, preserving normalization."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Evaluate sequence safely, handling edge cases."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')
