"""

import numpy as np
from scipy.fft import rfft, irfft
import cvxpy as cp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0]**2])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_subgradient_efficient(f, w):
    """
    Compute subgradient of sum_k w_k * (conv(f,f))_k using FFT.
    grad_j = 2 * conv(f, w_rev)[2n-2-j]
    where w has length 2n-1.
    """
    n = len(f)
    w_rev = w[::-1]
    # Need FFT size >= n + (2n-1) - 1 = 3n-2
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    F = rfft(f, fft_size)
    W_rev = rfft(w_rev, fft_size)
    conv_result = irfft(F * W_rev, fft_size)
    # A[j] = conv_result[2n-2-j], so grad = 2 * A
    # A indexes from 2n-2 down to 2n-2-(n-1) = n-1
    A = conv_result[2 * n - 2 - np.arange(n)]
    return 2.0 * A


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fista_optimize(f_init, time_limit, max_tau=300.0, min_tau=10.0,
                   initial_topk=20, max_topk=200, momentum=0.7, label="fista"):
    """
    FISTA with adaptive subgradient for minimizing max(conv(f,f)).
    Uses FFT-based subgradient computation for O(n log n) per iteration.
    """
    t0 = time.monotonic()
    n = len(f_init)
    f = f_init.copy()
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / s

    z = f.copy()
    t_mom = 1.0
    best_f = f.copy()
    b = fast_convolve(f)
    best_obj = b.max()
    it = 0
    no_improve = 0
    lr = 1.0 / (4.0 * best_obj)  # initial step size estimate
    rng = np.random.default_rng(2024)

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 0.5:
            break

        b = fast_convolve(z)
        cur_obj = b.max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = z.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive top-K
        topk = min(max_topk, initial_topk + it // 50)
        if topk > len(b):
            topk = len(b)
        top_idx = np.argpartition(b, -topk)[-topk:]
        w = np.zeros(len(b))
        w[top_idx] = 1.0 / topk

        # Compute subgradient via FFT
        grad = compute_subgradient_efficient(z, w)

        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        beta = (t_mom - 1.0) / t_new

        # Backtracking line search
        alpha = lr
        y_trial = z
        g_trial = grad
        for backtrack in range(10):
            x_trial = y_trial - alpha * g_trial
            x_trial_s = project_simplex(x_trial)
            y_next = x_trial_s + beta * (x_trial_s - y_trial)
            y_next = np.maximum(y_next, 0.0)
            sn = y_next.sum()
            if sn < 1e-10:

            else:
                y_next = y_next / sn
            b_next = fast_convolve(y_next)
            if b_next.max() < cur_obj + np.dot(grad, y_next - y_trial) + (1.0 / (2.0 * alpha)) * np.sum((y_next - y_trial) ** 2) + 1e-12:
                break
            alpha *= 0.5

        x_new = z - alpha * grad
        x_new = project_simplex(x_new)
        y_next = x_new + beta * (x_new - z)
        y_next = np.maximum(y_next, 0.0)
        sn = y_next.sum()
        if sn < 1e-10:
            y_next = best_f.copy()
        else:
            y_next = y_next / sn

        z = y_next
        t_mom = t_new
        lr = alpha

        # Periodic restart / diversification
        if no_improve > 60:
            if no_improve > 200:
                # Full restart from best with perturbation
                z = best_f.copy()
                noise = rng.standard_normal(n) * 0.01
                z = project_simplex(z + noise)
                t_mom = 1.0
                lr = 1.0 / (4.0 * best_obj)
                no_improve = 0
            elif no_improve % 40 == 0:

                  f"topk={topk} lr={lr:.4f} t={elapsed:.0f}s")

    return best_f, best_obj


def build_constraint_matrix(f, indices, n):
    """Build constraint matrix where row i encodes (f*g)[k_i] = A[i,:] @ g."""
    K = len(indices)
    A = np.zeros((K, n))
    for i, k in enumerate(indices):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        A[i, lo:hi + 1] = f[lo:hi + 1]
    return A


def solve_lp_direction(f, n, M, active_idx, solver="PDLP"):
    """Solve: max sum(g) s.t. g>=0, A@g <= M."""
    K = len(active_idx)
    if K == 0:
        return None
    A = build_constraint_matrix(f, active_idx, n)
    g = cp.Variable(n)
    constraints = [g >= 0, A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        if solver == "PDLP" and n <= 5000:
            prob.solve(solver=cp.PDLP, max_iter=500000, verbose=False)
        else:
            prob.solve(solver=cp.SCS, max_iters=500000, eps=1e-9, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, max_iters=300000, eps=1e-8)
        except Exception:
            return None
    if g.value is None:
        return None
    g_val = np.maximum(g.value, 0.0)
    if np.any(np.isnan(g_val)):
        return None
    return g_val

def spectral_factor_init(n, n_peaks=3, rng=None):
    """Generate an initial sequence with multiple peaks for diversity."""
    if rng is None:
        rng = np.random.default_rng(42)
    f = np.zeros(n)
    peak_positions = rng.choice(n, size=n_peaks, replace=False)
    widths = rng.integers(n // 20, n // 5, size=n_peaks)
    heights = rng.uniform(0.5, 1.0, size=n_peaks)
    for pos, width, height in zip(peak_positions, widths, heights):
        for j in range(max(0, pos - width), min(n, pos + width + 1)):
            f[j] += height * np.exp(-((j - pos) ** 2) / (2 * width ** 2))
    f = np.maximum(f, 0)
    s = f.sum()
    if s > 1e-10:
        f /= s
    else:
        f = np.ones(n) / n
    return f


def iaa_iterate(f_init, n_iters=100, floor_frac=0.3):
    """IAA-inspired iteration to flatten the auto-convolution."""
    n = len(f_init)
    f = f_init.copy()
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    for _ in range(n_iters):
        F = rfft(f, fft_size)
        R = irfft(F * F, fft_size)[:size]
        peak_val = R.max()
        floor_val = floor_frac * peak_val
        R_target = np.clip(R, floor_val, peak_val)
        energy = R.sum()
        if R_target.sum() > 1e-10:
            R_target *= energy / R_target.sum()
        R_fft = rfft(R_target, fft_size)
        real_part = np.maximum(R_fft.real, 0.0)
        new_amp = np.sqrt(real_part)
        abs_F = np.abs(F)
        phase = F / (abs_F + 1e-30)
        F_new = new_amp * phase
        f = irfft(F_new, fft_size)[:n]
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s > 1e-10:
            f /= s
        else:
            return f_init
    return f

        t4 = min(rem * 0.12, 100.0)
        if t4 < 30:
            break
        old_x = np.linspace(0.0, 1.0, n_base)
        new_x = np.linspace(0.0, 1.0, target_n)
        f_interp = np.interp(new_x, old_x, best_seq)
        f_interp = np.maximum(f_interp, 0.0)
        s = f_interp.sum()
        if s > 1e-10:
            f_interp /= s
        else:
            f_interp = np.ones(target_n) / target_n

        # Quick FISTA then LP
        t4a = t4 * 0.4
        t4b = t4 * 0.6
        f4a, _ = fista_optimize(f_interp, time_limit=t4a,
                                initial_topk=15, max_topk=80, label=f"fista_n{target_n}")
        f4b, _ = lp_cutting_planes(f4a, time_limit=t4b,
                                   initial_k=40, max_k=300, label=f"cp_n{target_n}")
        s4 = float(evaluate_sequence(f4b.tolist()))
        if np.isfinite(s4) and s4 < best_score:
            best_score = s4
            best_seq = f4b.copy()
            print(f"  NEW BEST [n{target_n}]: {s4:.6f}")
        else:
            print(f"  n={target_n}: score={s4:.6f} (no improvement)")

    # === Phase 5: Continued LP on best ===
    print(f"\n=== Phase 5: LP deep (best={best_score:.6f}, n={len(best_seq)}) ===")
    rem = deadline - time.monotonic()
    t5 = min(rem * 0.25, 300.0)
    if t5 > 40:
        n_b = len(best_seq)
        print(f"  Budget: {t5:.0f}s, n={n_b}")
        f5, obj5 = lp_cutting_planes(best_seq, time_limit=t5,
                                     initial_k=100, max_k=800, label="cp_deep")
        score5 = float(evaluate_sequence(f5.tolist()))
        if np.isfinite(score5) and score5 < best_score:
            best_score = score5
            best_seq = f5.copy()
            print(f"  NEW BEST [cp_deep]: {score5:.6f}")

    # === Phase 6: FISTA polish ===
    rem = deadline - time.monotonic()
    if rem > 30:
        print(f"\n=== Phase 6: FISTA polish (best={best_score:.6f}) ===")
        t6 = min(rem * 0.5, 120.0)
        n_b = len(best_seq)
        f6, _ = fista_optimize(best_seq, time_limit=t6,
                               initial_topk=30, max_topk=200, label="fista_final")
        s6 = float(evaluate_sequence(f6.tolist()))
        if np.isfinite(s6) and s6 < best_score:
            best_score = s6
            best_seq = f6.copy()
            print(f"  NEW BEST [fista_final]: {s6:.6f}")

    # === Phase 7: Final LP burst ===
    rem = deadline - time.monotonic()
    if rem > 20:
        print(f"\n=== Phase 7: Final LP (best={best_score:.6f}) ===")
        t7 = min(rem * 0.8, 80.0)
        n_b = len(best_seq)
        f7, _ = lp_cutting_planes(best_seq, time_limit=t7,
                                  initial_k=120, max_k=1000, label="final")
        s7 = float(evaluate_sequence(f7.tolist()))
        if np.isfinite(s7) and s7 < best_score:
            best_score = s7
            best_seq = f7.copy()
            print(f"  NEW BEST [final]: {s7:.6f}")

    # Final validation
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
