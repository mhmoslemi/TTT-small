"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast exponential barrier (log-sum-exp) with FFT-based gradients + L-BFGS in log-space
2. LP fictitious play with optimal line search and dual-guided weighting
3. Targeted coordinate descent for peak equalization
4. Multi-length exploration
Key improvements: O(n log n) gradient via FFT correlation, adaptive lambda,
optimal step-size line search in LP phase, and smart peak-focused moves.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute 2*n*max(conv) for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def softmax(z):
    z_max = z.max()
    e = np.exp(z - z_max)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def grad_exp_barrier(z, n, lam):
    """
    Compute f_lambda(softmax(z)) and gradient w.r.t. z.
    f_lambda = (1/lam)*log(sum(exp(lam*(a*a))))
    grad_z[i] = a[i]*(grad_a[i] - sum(a*grad_a))
    where grad_a[i] = 2*sum_m p[i+m]*a[m], p=softmax(lam*(a*a))
    """
    a = softmax(z)
    c = fast_convolve(a)
    max_c = c.max()
    lc = lam * (c - max_c)
    exp_lc = np.exp(lc)
    sum_exp = exp_lc.sum()
    f_val = max_c + np.log(sum_exp) / lam
    p = exp_lc / sum_exp
    # grad_a[i] = 2 * correlation(p, a)[i] = 2 * sum_m p[i+m]*a[m]
    total_size = 3 * n - 2
    fft_size = 1 << (total_size - 1).bit_length()
    P = rfft(p, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(P * np.conj(A), fft_size)
    grad_a = 2.0 * corr[:n]
    S = np.dot(a, grad_a)
    grad_z = a * (grad_a - S)
    return f_val, grad_z


def lbfgs_exp_barrier(a0, time_limit, lam_start=100.0, lam_end=8000.0,
                      n_steps=8, maxiter_per=800):
    """
    L-BFGS with exponential barrier, annealing lambda.
    a0: initial normalized non-negative sequence.
    Returns (best_a, best_score).
    """
    n = len(a0)
    a0 = np.maximum(a0, 1e-15)
    a0 = a0 / a0.sum()
    z0 = np.log(a0)
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = compute_score(a0)
    z = z0.copy()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        frac = step / max(1, n_steps - 1)
        lam = lam_start * (lam_end / lam_start) ** frac

        def func(zv):
            if time.monotonic() - t0 > time_limit:
                return 1e10
            fv, _ = grad_exp_barrier(zv, n, lam)
            return fv

        def jac(zv):
            if time.monotonic() - t0 > time_limit:
                return np.zeros_like(zv)
            _, gv = grad_exp_barrier(zv, n, lam)
            return gv

        try:
            res = minimize(func, z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': maxiter_per, 'ftol': 1e-16,
                                    'gtol': 1e-12, 'maxfun': maxiter_per * 20,
                                    'disp': False})
            z = res.x
        except Exception:
            pass

        a_tmp = softmax(z)
        a_tmp = np.maximum(a_tmp, 0)
        s = a_tmp.sum()
        if s > 1e-15:
            a_tmp = a_tmp / s
        sc = compute_score(a_tmp)
        if sc < best_score:
            best_score = sc
            best_a = a_tmp.copy()

    return best_a, best_score


def build_C_matrix(f):
    """Build Toeplitz constraint matrix C[k,j] = f[k-j] if 0<=k-j<n else 0."""
    n = len(f)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = f
    return C


def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. C*g <= M, g >= 0."""
    n = len(f)
    C = build_C_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status != 0:
        return None, None
    g = np.maximum(res.x, 0)
    # Get dual variables (slack multipliers)
    dual = None
    if hasattr(res, 'ineqlin') and res.ineqlin is not None:
        dual = res.ineqlin.get('mult', None)
    return g, dual


def lp_play_with_linesearch(a0, time_limit, max_iters=2000):
    """LP fictitious play with optimal line search."""
    n = len(a0)
    a = np.maximum(a0.copy(), 1e-15)
    a = a / a.sum()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = compute_score(a)

    # Momentum
    v = np.zeros(n)
    beta = 0.9

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(a)
        M = conv.max()

        g, dual = solve_lp_direction(a, M)
        if g is None or g.sum() < 1e-15:
            break
        g = g / g.sum()

        # Line search: find best t in (0, 1]
        best_t = 0.05
        best_f = 1e10
        # Try several t values
        t_candidates = [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]
        for t_try in t_candidates:
            a_try = (1 - t_try) * a + t_try * g
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s < 1e-15:
                continue
            a_try = a_try / s
            f_try = compute_score(a_try)
            if f_try < best_f:
                best_f = f_try
                best_t = t_try

        # Use momentum
        v = beta * v + (1 - beta) * (g - a)
        a_new = a + best_t * v
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a
        else:
            a_new = a_new / s

        sc = compute_score(a_new)
        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
        a = a_new

        # Also try without momentum
        a_plain = (1 - best_t) * a + best_t * g
        a_plain = np.maximum(a_plain, 0)
        sp = a_plain.sum()
        if sp > 1e-15:
            a_plain = a_plain / sp
            sc_p = compute_score(a_plain)
            if sc_p < best_score - 1e-14:
                best_score = sc_p
                best_a = a_plain.copy()
                a = a_plain.copy()
                v = np.zeros(n)

    return best_a, best_score


def coord_descent_peak(a0, time_limit, n_pairs_per_round=300):
    """Coordinate descent focusing on top peaks."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0)
    a = a / a.sum()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = compute_score(a)
    rng = np.random.default_rng(123)

    no_improve = 0
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        # Top peak and its contributing indices
        pk = int(np.argmax(conv))
        js = max(0, pk - n + 1)
        je = min(n - 1, pk)

        improved = False
        # Generate candidate pairs
        pairs = []
        # Pairs within the peak's contributing range
        for _ in range(n_pairs_per_round // 2):
            i = int(rng.integers(js, je + 1))
            j = int(rng.integers(js, je + 1))
            if i != j:
                pairs.append((min(i, j), max(i, j)))
        # Pairs involving peak contributors and non-contributors
        noncontrib = np.setdiff1d(np.arange(n), np.arange(js, je + 1))
        if len(noncontrib) > 0:
            for _ in range(n_pairs_per_round // 4):
                i = int(rng.integers(js, je + 1))
                j = int(noncontrib[rng.integers(0, len(noncontrib))])
                pairs.append((min(i, j), max(i, j)))
        # Random pairs
        for _ in range(n_pairs_per_round // 4):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                pairs.append((min(i, j), max(i, j)))

        for (i, j) in pairs:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14 and a[j] < 1e-14:
                continue
            # Golden section search for transfer t
            t_lo = -a[j] + 1e-14
            t_hi = a[i] - 1e-14
            if t_hi - t_lo < 1e-14:
                continue
            gr = (np.sqrt(5) - 1) / 2
            x1 = t_hi - gr * (t_hi - t_lo)
            x2 = t_lo + gr * (t_hi - t_lo)

            def eval_transfer(t_val):
                a[i] -= t_val
                a[j] += t_val
                if a[i] < -1e-14 or a[j] < -1e-14:
                    a[i] += t_val
                    a[j] -= t_val
                    return 1e10
                tmp_i = a[i]
                tmp_j = a[j]
                a[i] = max(0, tmp_i)
                a[j] = max(0, tmp_j)
                ss = a.sum()
                if ss < 1e-14:
                    a[i] = tmp_i
                    a[j] = tmp_j
                    return 1e10
                a_save = a.copy()
                a_norm = a / ss
                sc = compute_score(a_norm)
                a[:] = a_save
                return sc

            f1 = eval_transfer(x1)
            f2 = eval_transfer(x2)
            for _ in range(15):
                if time.monotonic() - t0 > time_limit:
                    break
                if f1 < f2:
                    t_hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = t_hi - gr * (t_hi - t_lo)
                    f1 = eval_transfer(x1)
                else:
                    t_lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = t_lo + gr * (t_hi - t_lo)
                    f2 = eval_transfer(x2)

            t_opt = (t_lo + t_hi) / 2
            f_opt = min(f1, f2)
            if f_opt < best_score - 1e-13:
                a[i] -= t_opt
                a[j] += t_opt
                a = np.maximum(a, 0)
                a = a / a.sum()
                best_score = compute_score(a)
                best_a = a.copy()
                improved = True

        if not improved:
            no_improve += 1
            if no_improve > 8:
                # Perturb
                noise = rng.standard_normal(n) * 0.003
                a = best_a + noise
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
        else:
            no_improve = 0

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_n = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_i = np.interp(new_x, old_x, f_n)
    f_i = np.maximum(f_i, 0)
    ss = f_i.sum()
    if ss > 1e-15:
        f_i = f_i / ss
    return f_i

                                                 f"lbfgs_{ls}_{le}")
            print(f"  After lbfgs({ls}->{le}): best={best_score:.8f}")

    # === Phase 2: LP Fictitious Play ===
    print(f"\n=== Phase 2: LP Play (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(300.0, rem * 0.18)
    if t_alloc > 30:
        print(f"  Budget: {t_alloc:.0f}s")
        a2, sc2 = lp_play_with_linesearch(best_seq, t_alloc, max_iters=500)
        best_seq, best_score = check_update(a2, best_seq, best_score, "lp_phase2")
        print(f"  After LP: best={best_score:.8f}")

    # === Phase 3: Alternate L-BFGS + LP ===
    print(f"\n=== Phase 3: Alternating (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    n_rounds = 4
    for rnd in range(n_rounds):
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        # L-BFGS sub-step
        t_sub = min(60.0, rem * 0.25)
        if t_sub > 15:
            lam = 2000.0 + 2000.0 * rnd
            a3, sc3 = lbfgs_exp_barrier(best_seq, t_sub,
                                        lam_start=lam, lam_end=lam * 3,
                                        n_steps=3, maxiter_per=400)
            best_seq, best_score = check_update(a3, best_seq, best_score,
                                                 f"alt_lbfgs_{rnd}")
        # LP sub-step
        rem = deadline - time.monotonic()
        t_sub = min(50.0, rem * 0.2)
        if t_sub > 15:
            a4, sc4 = lp_play_with_linesearch(best_seq, t_sub, max_iters=200)
            best_seq, best_score = check_update(a4, best_seq, best_score,
                                                 f"alt_lp_{rnd}")
        print(f"  Round {rnd}: best={best_score:.8f}")

    # === Phase 4: Coordinate Descent ===
    print(f"\n=== Phase 4: Coord Descent (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(200.0, rem * 0.2)
    if t_alloc > 20:
        print(f"  Budget: {t_alloc:.0f}s")
        a5, sc5 = coord_descent_peak(best_seq, t_alloc)
        best_seq, best_score = check_update(a5, best_seq, best_score, "cd")
        print(f"  After CD: best={best_score:.8f}")

    # === Phase 5: Length Exploration ===
    print(f"\n=== Phase 5: Length Exploration (best={best_score:.8f}) ===")
    nearby_ns = [n_parent - 3, n_parent - 2, n_parent - 1, n_parent + 1,
                 n_parent + 2, n_parent + 3]
    for alt_n in nearby_ns:
        if alt_n < 128 or alt_n > 3000:
            continue
        rem = deadline - time.monotonic()
        if rem < 40:
            break
        t_sub = min(50.0, rem * 0.2)
        if t_sub < 15:
            break
        init_alt = interpolate_seq(best_seq, alt_n)
        a_alt, sc_alt = lbfgs_exp_barrier(init_alt, t_sub * 0.6,
                                          lam_start=500, lam_end=5000,
                                          n_steps=3, maxiter_per=400)
        best_seq, best_score = check_update(a_alt, best_seq, best_score,
                                             f"n{alt_n}")
        # Quick CD on this length
        rem2 = deadline - time.monotonic()
        if rem2 > 20:
            a_alt2, sc_alt2 = coord_descent_peak(a_alt, min(20.0, rem2 * 0.3))
            best_seq, best_score = check_update(a_alt2, best_seq, best_score,
                                                 f"n{alt_n}_cd")
        print(f"  n={alt_n}: best={best_score:.8f}")

    # === Phase 6: Final Polish ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic()
    if rem > 30:
        # Coordinate descent
        t_sub = min(60.0, rem * 0.4)
        if t_sub > 10:
            a_fin, sc_fin = coord_descent_peak(best_seq, t_sub)
            best_seq, best_score = check_update(a_fin, best_seq, best_score,
                                                 "final_cd")
        # L-BFGS
        rem2 = deadline - time.monotonic()
        if rem2 > 10:
            t_sub2 = min(40.0, rem2 * 0.5)
            if t_sub2 > 10:
                a_fin2, sc_fin2 = lbfgs_exp_barrier(best_seq, t_sub2,
                                                     lam_start=5000,
                                                     lam_end=20000,
                                                     n_steps=3, maxiter_per=500)
                best_seq, best_score = check_update(a_fin2, best_seq, best_score,
                                                     "final_lbfgs")

    # Final verification
    if best_seq is None:
        best_seq = parent
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = min(best_score, fs)

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
