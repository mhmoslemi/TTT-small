def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def smooth_max_and_weights(x, beta):
    """Compute smooth max and softmax weights with numerical stability."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -200, 200)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-30:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_shifted / total
    sm = x_max + np.log(total) / beta
    return sm, w


def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_conv_at_lag(a, k):
    """Compute gradient of (a*a)[k] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    for i in range(n):
        val = 0.0
        if 0 <= k - i < n:
            val += a[k - i]
        if 0 <= i - k < n:
            val += a[i - k]
        grad[i] = val
    return grad


def compute_grad_conv_fft(a, w):

    W = fft(w, N)
    A = fft(a, N)
    grad = np.real(ifft(np.conj(W) * A, N))[:n] * 2.0

def alternating_projections(n, n_iter=200, rng=None):
    """Gerchberg-Saxton style alternating projections for flat spectrum + simplex."""
    if rng is None:

    fft_size = 1 << (2 * n - 1 - 1).bit_length()

    A = np.ones(fft_size) * np.exp(1j * phases)
    A[0] = np.ones(fft_size)  # DC is real and positive
    a = np.real(ifft(A, fft_size))[:n]
    a = project_simplex(a)
    
    for it in range(n_iter):
        # Frequency domain: flatten magnitude
        A = fft(a, fft_size)
        mags = np.abs(A)

            break
        
        new_mags = np.copy(mags)
        new_mags[1:] = target
        phases = A / np.maximum(mags, 1e-30)

        a_new = np.real(ifft(new_A, fft_size))[:n]
        
        # Time domain: project to simplex
        a = project_simplex(a_new)
    
    return a


def lp_refine_step(a):
    """
    One step of LP-based refinement.
    Given current a (sum=1), find g maximizing sum(g) subject to (a*g)[k] <= M, g >= 0.
    Returns the updated sequence.
    """
    n = len(a)
    conv_a = fast_convolve(a)
    M = conv_a.max()
    
    # Build constraint matrix: (a*g)[k] = sum_j a[k-j] * g[j] for valid indices
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        # (a*g)[k] = sum_j a[k-j] * g[j] where 0 <= k-j < n and 0 <= j < n
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            A_ub[k, j_start:j_end+1] = a[k - j_end:k - j_start + 1][::-1]
    
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            g = np.maximum(result.x, 0)
            s = g.sum()
            if s < 1e-15:
                return a
            g = g / s
            return g
        else:
            return a
    except Exception:
        return a


def lp_refine(a, n_iters=100, t=0.1):
    """Multiple LP refinement steps."""
    a = a.copy()
    best = a.copy()
    best_score = 2.0 * len(a) * fast_convolve(a).max()
    
    for it in range(n_iters):
        g = lp_refine_step(a)
        a_new = (1 - t) * a + t * g

    return best, best_score
