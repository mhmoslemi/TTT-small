    s = a_new.sum()
    if s > 1e-15:

def subgrad_onemax(a):
    """Compute subgradient of max(conv(a,a)) at a. Returns (grad, k_star, max_val)."""
    n = len(a)
    conv = fast_convolve(a)
    k_star = int(np.argmax(conv))

    for j in range(n):
        k_j = k_star - j
        if 0 <= k_j < n:
            grad[j] = 2.0 * a[k_j]
    return grad, k_star, conv.max()


def subgrad_onemax_fft(a):

    conv = fast_convolve(a)
    k_star = int(np.argmax(conv))

    # a[k_star - j]: when j=0, a[k_star]; j=1, a[k_star-1]; ...

def fast_convolve(a):

    n = len(a)
    size = 2 * n - 1

    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def smooth_max_and_weights(x, beta):
    """Compute smooth max (log-sum-exp) and softmax weights."""
    x_max = x.max()
    shifted = beta * (x - x_max)

    if total < 1e-300:

        w[np.argmax(x)] = 1.0
        return x_max, w

    n = len(a)
    fft_size = 1 << (len(w) + n - 1 + 63).bit_length()
    W = fft(w, fft_size)
    A = fft(a, fft_size)
    grad = np.real(ifft(np.conj(W) * A, fft_size))[:n] * 2.0
    return grad


def project_simplex(v):
    """Euclidean projection onto simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgrad_max_conv(a):
    """Subgradient of max_k conv(a,a)[k] at a. Returns (grad, k_star, max_val)."""
    n = len(a)
    conv = fast_convolve(a)
    k_star = int(np.argmax(conv))
    max_val = conv[k_star]
    grad = np.zeros(n)
    # d(conv[k])/d(a[j]) = 2*a[k-j] if 0 <= k-j < n, else 0
    j_lo = max(0, k_star - (n - 1))
    j_hi = min(n - 1, k_star)
    for j in range(j_lo, j_hi + 1):
        grad[j] = 2.0 * a[k_star - j]
    return grad, k_star, max_val


def ert_project(a):
    """One IFT step: flatten spectrum while preserving DC, then project to a>=0."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = fft(a, fft_size)
    
    a_energy = np.sum(a ** 2)
    a_sum = np.sum(a)
    c2 = (fft_size * a_energy - a_sum ** 2) / max(fft_size - 1, 1)
    c = np.sqrt(max(c2, 0.0))
    
    new_mag = np.full(fft_size, c, dtype=np.float64)
    new_mag[0] = abs(A[0])
    new_phase = np.angle(A)
    new_A = new_mag * np.exp(1j * new_phase)
    
    a_new = np.real(ifft(new_A, fft_size))[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    else:
        a_new = np.ones(n) / n
    return a_new


def ert_init(n, n_steps=150, rng=None):
    """Initialize via IFT iterations from random start."""
    if rng is None:
        rng = np.random.default_rng()
    a = rng.dirichlet(np.ones(n) * 0.5)
    for _ in range(n_steps):
        a = ert_project(a)
    return a


def ert_multi_init(n, n_trials=5, n_steps=100, rng=None):
    """Multiple IFT initializations with different random starts."""
    if rng is None:
        rng = np.random.default_rng()
    best_a = None
    best_obj = float('inf')
    for _ in range(n_trials):
        a0 = rng.dirichlet(np.ones(n) * rng.uniform(0.3, 2.0))
        a = a0.copy()
        for _ in range(n_steps):
            a = ert_project(a)
        obj = 2.0 * n * fast_convolve(a).max()
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
    return best_a


def nesterov_optimize(a0, beta, lr, n_iter, momentum=0.9):
    """Momentum (Nesterov-like) projected gradient on simplex."""
    n = len(a0)
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    y = a.copy()
    v = np.zeros(n)
    
    best_obj = float('inf')
    best_a = a.copy()
    
    for it in range(n_iter):
        conv = fast_convolve(y)
        sm, w = smooth_max_and_weights(conv, beta)
        obj = 2.0 * n * sm
        
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        grad = compute_grad_smooth_conv(y, w)
        v = momentum * v + grad
        a_new = project_simplex(a - lr * v)
        
        if np.linalg.norm(a_new - a) < 1e-16:
            break
        
        a = a_new
        y = a
        lr *= 0.9995
    
    return best_a, best_obj


def adam_optimize(a0, beta, lr, n_iter, beta1=0.85, beta2=0.999):
    """Adam optimizer projected onto simplex."""
    n = len(a0)
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    
    m = np.zeros(n)
    v = np.zeros(n)
    eps = 1e-8
    
    best_obj = float('inf')
    best_a = a.copy()
    
    for t in range(n_iter):
        conv = fast_convolve(a)
        sm, w = smooth_max_and_weights(conv, beta)
        obj = 2.0 * n * sm
        
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        grad = compute_grad_smooth_conv(a, w)
        
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (t + 1))
        v_hat = v / (1.0 - beta2 ** (t + 1))
        
        a_new = project_simplex(a - lr * m_hat / (np.sqrt(v_hat) + eps))
        a = a_new
    
    return best_a, best_obj


def subgradient_optimize(a0, lr, n_iter, lr_decay=0.9995, lower_bound=1.0):
    """Subgradient descent on exact max with Polyak-inspired adaptation."""
    n = len(a0)
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    
    best_obj = float('inf')
    best_a = a.copy()
    
    for it in range(n_iter):
        conv = fast_convolve(a)
        obj = 2.0 * n * conv.max()
        
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        grad, k_star, max_val = subgrad_max_conv(a)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Polyak-like step: lr * (current - lower_bound) / gnorm
        gap = obj - lower_bound
        if gap > 1e-10:
            lr_eff = min(lr, gap / (gnorm * gnorm + 1e-15) * 0.5)
        else:
            lr_eff = lr
        
        a_new = project_simplex(a - lr_eff * grad)
        a = a_new
        lr *= lr_decay
    
    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Gaussian bumps
    for sf in [0.2, 0.35, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (sf * n)) ** 2)
        g /= g.sum()
        inits['gauss_%.2f' % sf] = g
    
    # Triangle
    tri = np.minimum(x, n - 1 - x) + 1.0
    tri /= tri.sum()
    inits['triangle'] = tri
    
    # Bimodal (edge-heavy)
    bi = np.zeros(n)
    edge = max(1, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi
    
    # IFT-based
    inits['ert'] = ert_multi_init(n, n_trials=3, n_steps=80, rng=rng)
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # From parent
    if parent is not None and len(parent) > 0:
        pp = np.maximum(parent, 0)
        ps = pp.sum()
        if ps > 1e-10:
            pp = pp / ps
            if len(pp) == n:
                inits['parent'] = pp
            elif len(pp) > n:
                pp_ds = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(pp)), pp)
                pp_ds = np.maximum(pp_ds, 0)
                pds = pp_ds.sum()
                if pds > 1e-10:
                    inits['parent_ds'] = pp_ds / pds
            else:
                pp_up = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(pp)), pp)
                pp_up = np.maximum(pp_up, 0)
                pup = pp_up.sum()
                if pup > 1e-10:
                    inits['parent_up'] = pp_up / pup
    
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Combines IFT initialization, momentum/Adam gradient, subgradient refinement,
    multi-length exploration, and random restarts.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 60.0)
    
    best_sequence = None
    best_score = float('inf')
    
    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print("Parent available: length=%d, sum=%.2f" % (len(p), p.sum()))
            # Evaluate parent
            ps = float(evaluate_sequence(p.tolist()))
            print("Parent score: %.6f" % ps)
    except Exception as e:
        print("No parent: %s" % str(e))
    
    # Local helper (defined inside to capture nonlocal state)
    def score_and_update(a):
        nonlocal best_sequence, best_score
        if a is None:
            return float('inf')
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print("  New best: score=%.6f, n=%d (t=%.0fs)" % (score, len(a), time.monotonic() - t_start))
        return score if np.isfinite(score) else float('inf')
    
    # === Phase 1: IFT scan over lengths ===
    print("=== Phase 1: IFT Scan ===")
    lengths = [128, 192, 256, 320, 384, 512]
    phase1_deadline = t_start + 350.0
    
    for n in lengths:
        if time.monotonic() > phase1_deadline - 40:
            break
        inits = generate_initializations(n, parent, rng)
        
        priority = ['ert', 'parent', 'parent_ds', 'parent_up', 'gauss_0.35', 'uniform']
        init_names = [nm for nm in priority if nm in inits]
        init_names += [nm for nm in inits if nm not in priority]
        init_names = init_names[:4]
        
        for name in init_names:
            if time.monotonic() > phase1_deadline - 20:
                break
            remaining = phase1_deadline - time.monotonic() - 10
            if remaining < 5:
                break
            
            a0 = inits[name]
            n_iter = int(min(400, remaining * 60))
            a_opt, _ = nesterov_optimize(a0, beta=60.0, lr=0.06, n_iter=n_iter)
            score = score_and_update(a_opt)
    
    print("Phase 1 done. Best: %.6f" % best_score)
    
    # === Phase 2: Deep multi-beta optimization ===
    print("=== Phase 2: Deep Optimization ===")
    phase2_deadline = deadline - 250.0
    
    if best_sequence is not None:
        best_n = len(best_sequence)
    else:
        best_n = 256
    
    target_lengths = sorted(set([l for l in [best_n // 2, best_n * 3 // 4, best_n,
                                             best_n * 5 // 4, best_n * 2, 192, 320]
                                 if 64 <= l <= 1024]))
    
    for n in target_lengths:
        if time.monotonic() > phase2_deadline - 60:
            break
        
        inits = generate_initializations(n, parent, rng)
        
        # Add best sequence as init
        if best_sequence is not None:
            pp = np.maximum(best_sequence, 0)
            ps = pp.sum()
            if ps > 1e-10:
                pp = pp / ps
                if len(pp) == n:
                    inits['best'] = pp
                elif len(pp) > n:
                    pp_ds = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(pp)), pp)
                    pp_ds = np.maximum(pp_ds, 0)
                    pds = pp_ds.sum()
                    if pds > 1e-10:
                        inits['best_ds'] = pp_ds / pds
                else:
                    pp_up = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(pp)), pp)
                    pp_up = np.maximum(pp_up, 0)
                    pup = pp_up.sum()
                    if pup > 1e-10:
                        inits['best_up'] = pp_up / pup
        
        priority2 = ['best', 'best_ds', 'best_up', 'ert', 'parent', 'parent_ds', 'parent_up',
                     'gauss_0.35']
        init_names = [nm for nm in priority2 if nm in inits]
        init_names += [nm for nm in inits if nm not in priority2]
        init_names = init_names[:3]
        
        for name in init_names:
            if time.monotonic() > phase2_deadline - 40:
                break
            remaining = phase2_deadline - time.monotonic() - 25
            if remaining < 20:
                break
            
            a0 = inits[name]
            # Multi-beta annealing schedule
            betas = [100.0, 50.0, 25.0, 12.0, 6.0, 3.0, 1.5]
            a_cur = a0.copy()
            time_per_beta = remaining / len(betas)
            
            for beta in betas:
                if time.monotonic() > phase2_deadline - 20:
                    break
                t_single = min(time_per_beta, phase2_deadline - time.monotonic() - 10)
                if t_single < 3:
                    break
                n_iter = int(min(600, t_single * 80))
                
                if beta >= 30.0:
                    a_opt, _ = nesterov_optimize(a_cur, beta=beta, lr=0.05, n_iter=n_iter)
                elif beta >= 5.0:
                    a_opt, _ = adam_optimize(a_cur, beta=beta, lr=0.015, n_iter=n_iter)
                else:
                    a_opt, _ = adam_optimize(a_cur, beta=beta, lr=0.008, n_iter=n_iter)
                score = score_and_update(a_opt)
                a_cur = a_opt
    
    print("Phase 2 done. Best: %.6f" % best_score)
    
    # === Phase 3: Subgradient refinement (exact objective) ===
    print("=== Phase 3: Subgradient Refinement ===")
    if best_sequence is not None:
        n_best = len(best_sequence)
        remaining = deadline - time.monotonic() - 100.0
        
        if remaining > 30:
            a_cur = best_sequence / best_sequence.sum()
            
            # Multiple subgradient passes with decreasing step size
            for lr_start in [0.05, 0.03, 0.01, 0.005]:
                if time.monotonic() > deadline - 50:
                    break
                remaining = deadline - time.monotonic() - 30
                if remaining < 10:
                    break
                n_iter = int(min(5000, remaining * 150))
                lb = max(1.0, best_score - 0.5)
                a_opt, _ = subgradient_optimize(a_cur, lr=lr_start, n_iter=n_iter,
                                                lr_decay=0.9997, lower_bound=lb)
                score = score_and_update(a_opt)
                a_cur = a_opt
    
    print("Phase 3 done. Best: %.6f" % best_score)
    
    # === Phase 4: Final fine optimization ===
    print("=== Phase 4: Final Polish ===")
    if best_sequence is not None:
        n_best = len(best_sequence)
        remaining = deadline - time.monotonic() - 30.0
        
        if remaining > 20:
            a_cur = best_sequence / best_sequence.sum()
            t_single = remaining / 3
            n_iter = int(min(2000, t_single * 60))
            a_opt, _ = adam_optimize(a_cur, beta=1.5, lr=0.003, n_iter=n_iter)
            score = score_and_update(a_opt)
            
            remaining = deadline - time.monotonic() - 20.0
            if remaining > 15:
                n_iter = int(min(3000, remaining * 100))
                a_opt, _ = subgradient_optimize(a_opt, lr=0.008, n_iter=n_iter,
                                                lr_decay=0.9998, lower_bound=max(1.0, best_score - 0.3))
                score = score_and_update(a_opt)
    
    # === Phase 5: Random restarts ===
    print("=== Phase 5: Random Restarts ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 20.0
        n_best = len(best_sequence)
        
        n_restarts = 0
        while remaining > 40 and n_restarts < 5:
            n_restarts += 1
            # Try a perturbation of the best
            scale = rng.uniform(0.005, 0.02)
            noise = rng.standard_normal(n_best) * scale
            a_pert = project_simplex(best_sequence / best_sequence.sum() + noise)
            
            t_alloc = min(80.0, remaining / max(1, 5 - n_restarts + 1))
            if t_alloc < 10:
                break
            
            # Optimize from perturbation
            a_opt, _ = adam_optimize(a_pert, beta=10.0, lr=0.02, n_iter=int(min(500, t_alloc * 50)))
            score = score_and_update(a_opt)
            
            # Subgradient polish
            if score < best_score + 0.01 and remaining > 20:
                a_opt, _ = subgradient_optimize(a_opt, lr=0.01, n_iter=int(min(2000, remaining * 50)),
                                                lr_decay=0.9997, lower_bound=max(1.0, best_score - 0.3))
                score = score_and_update(a_opt)
            
            remaining = deadline - time.monotonic() - 15.0
    
    # === Final check ===
    if best_sequence is None:
        best_sequence = np.ones(128) / 128
    
    # Ensure all values are valid
    best_sequence = np.clip(best_sequence, 0.0, 1000.0)
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    
    print("\nFINAL SCORE: %.6f (n=%d)" % (best_score, len(best_sequence)))
    return [float(x) for x in best_sequence]
