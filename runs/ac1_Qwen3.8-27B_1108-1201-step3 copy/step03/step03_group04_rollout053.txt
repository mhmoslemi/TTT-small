def fast_convolve(a, b=None):

    if b is None:

    n = len(a)
    m = len(b)
    size = n + m - 1

    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def compute_convolution_and_gradient(p, beta):
    """
    Compute the convolution c = p*p and the gradient of
    (1/beta)*log(sum(exp(beta*c))) w.r.t. p.
    Returns (c, grad, lse_value).
    """
    n = len(p)
    c = fast_convolve(p)
    # Log-sum-exp
    max_c = c.max()
    exp_bc = np.exp(beta * (c - max_c))
    log_sum = np.log(exp_bc.sum())
    lse_val = max_c + log_sum / beta
    # Softmax weights
    w = exp_bc / exp_bc.sum()
    # Gradient: grad[j] = 2 * sum_k w[k] * p[k-j] = 2 * R[2n-2-j]
    # where R = convolve(flip(w), p)

    P = np.fft.rfft(p, fft_size)
    R = np.fft.irfft(W * P, fft_size)[:size]
    grad = 2.0 * R[(2*n-2) - np.arange(n)]
    # Multiply by 2n to get gradient of the full objective
    grad *= 2.0 * n
    return c, grad, lse_val

import numpy as np

from scipy.optimize import linprog

def fast_convolve(a, b=None):
    if b is None:
        b = a
    size = len(a) + len(b) - 1
    N = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    return np.fft.irfft(A * B, N)[:size]


def lse_gradient(p, beta):
    """Compute convolution and gradient of 2n * LSE_beta(p*p)."""

    c = fast_convolve(p)
    max_c = c.max()
    exp_bc = np.exp(beta * (c - max_c))
    w = exp_bc / exp_bc.sum()
    # Gradient via FFT cross-correlation
    w_rev = w[::-1]

    R = np.fft.irfft(W * P, N)[:size]

def build_conv_matrix(a):
    n = len(a)
    rows, cols, vals = [], [], []
    for k in range(2 * n - 1):
        j0 = max(0, k - n + 1)
        j1 = min(n - 1, k)
        for j in range(j0, j1 + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_oracle(a, M):
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)

                  options={'presolve': True, 'time_limit': 3.0})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None
