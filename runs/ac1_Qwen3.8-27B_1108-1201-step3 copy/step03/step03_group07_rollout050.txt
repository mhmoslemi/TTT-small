def project_simplex(v):
    n = len(v)

        return np.array([max(v[0], 0.0)])

    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)

        return np.ones(n) / n

    if rho == n - 1:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based iterative refinement
(HiGHS solver), frequency-domain initialization, and gradient polishing.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize
from scipy.fft import fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(n, f):
    """Build the A_ub matrix for the LP: (f*g)[k] = sum_j A_ub[k,j] * g[j] <= M."""
    # A_ub[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # This is a Toeplitz-like matrix
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        # g[j] appears in conv[k] with coefficient f[k-j]
        # j ranges over max(0, k-(n-1)) to min(n-1, k)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[k, j] = f[k - j]
    return A_ub


def solve_lp(f, M, method='highs'):
    """Solve: max sum(g) s.t. A_ub @ g <= M, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = build_lp_matrix(n, f)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method=method)
    if result.success:
        return result.x
    return None


def lp_refine(a0, n_iters=500, t_initial=1.0, t_min=0.1, patience=50):
    """LP-based iterative refinement."""
    n = len(a0)
    a = np.maximum(a0, 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    stall_count = 0
    
    A_ub = build_lp_matrix(n, a)  # Pre-build (changes as a changes)
    
    for it in range(n_iters):
        conv = fast_convolve(a)
        M = conv.max()
        
        # Rebuild LP matrix since a changed
        A_ub = build_lp_matrix(n, a)
        b_ub = np.full(2 * n - 1, M)
        
        result = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, 
                         bounds=[(0, None)] * n, method='highs')
        
        if not result.success:
            break
        
        g = np.maximum(result.x, 0)
        sg = g.sum()
        if sg < 1e-15:
            break
        
        g_norm = g / sg
        
        # Update with adaptive step size
        a_new = (1 - t_initial) * a + t_initial * g_norm
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        score_new = 2.0 * n * fast_convolve(a_new).max()
        
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            stall_count = 0
            # Can increase step size slightly
            t_initial = min(1.0, t_initial * 1.05)
        else:
            stall_count += 1
            t_initial = max(t_min, t_initial * 0.8)
        
        a = a_new
        
        if stall_count > patience:
            break
        
        if best_score < 1.503:
            # Getting very close, use smaller steps
            t_initial = 0.3
    
    return best_a, best_score

    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]

    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[k, j] = f[k - j]
    return A_ub


def smooth_max_and_weights(x, beta):
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -100, 100)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-30:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_shifted / total
    sm = x_max + np.log(total) / beta
    return sm, w


def compute_grad_smooth_conv(a, w):
    n = len(a)
    N = 1 << (len(w) + n - 1 + 63).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    grad = np.real(ifft(np.conj(W) * A, N))[:n] * 2.0
    return grad


def _smooth_obj(x, n, beta):
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    conv = fast_convolve(a)
    sm, _ = smooth_max_and_weights(conv, beta)
    return 2.0 * n * sm


def _smooth_jac(x, n, beta):
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    conv = fast_convolve(a)
    sm, w = smooth_max_and_weights(conv, beta)
    grad_a = compute_grad_smooth_conv(a, w)
    lagrange = np.dot(a, grad_a)
    return a * (grad_a - lagrange)


def optimize_lbfgsb(a0, beta, maxiter=300):
    n = len(a0)
    a0_clipped = np.maximum(a0, 1e-30)
    a0_clipped = a0_clipped / a0_clipped.sum()
    x0 = np.log(a0_clipped)
    result = minimize(_smooth_obj, x0, jac=_smooth_jac, args=(n, beta),
                      method='L-BFGS-B', options={'maxiter': maxiter, 'maxls': 50})
    x_opt = result.x
    a_opt = np.exp(x_opt - x_opt.max())
    a_opt = a_opt / a_opt.sum()
    return a_opt, result.fun

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based iterative refinement
(HiGHS solver) with adaptive step sizes, frequency-domain initialization,
L-BFGS-B gradient polishing, and multi-length search.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize
from scipy.fft import fft, ifft
from scipy.sparse import csr_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.argmax(~cond)
    if rho == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix_sparse(n, f):
    """Build sparse A_ub matrix for LP: (f*g)[k] = sum_j A_ub[k,j]*g[j]."""
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            data.append(f[k - j])
    return csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def smooth_max_and_weights(x, beta):
    """Compute smooth max and softmax weights."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -100, 100)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-30:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_shifted / total
    sm = x_max + np.log(total) / beta
    return sm, w


def compute_grad_smooth_conv(a, w):
    """Gradient of weighted sum of convolution w.r.t. a."""
    n = len(a)
    N = 1 << (len(w) + n - 1 + 63).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    grad = np.real(ifft(np.conj(W) * A, N))[:n] * 2.0
    return grad


def _smooth_obj(x, n, beta):
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    conv = fast_convolve(a)
    sm, _ = smooth_max_and_weights(conv, beta)
    return 2.0 * n * sm


def _smooth_jac(x, n, beta):
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a = a / a.sum()
    conv = fast_convolve(a)
    sm, w = smooth_max_and_weights(conv, beta)
    grad_a = compute_grad_smooth_conv(a, w)
    lagrange = np.dot(a, grad_a)
    return a * (grad_a - lagrange)


def optimize_lbfgsb(a0, beta, maxiter=300):
    """Optimize with L-BFGS-B using softmax reparametrization."""
    n = len(a0)
    a0_clipped = np.maximum(a0, 1e-30)
    a0_clipped = a0_clipped / a0_clipped.sum()
    x0 = np.log(a0_clipped)
    result = minimize(_smooth_obj, x0, jac=_smooth_jac, args=(n, beta),
                      method='L-BFGS-B', options={'maxiter': maxiter, 'maxls': 50})
    x_opt = result.x
    a_opt = np.exp(x_opt - x_opt.max())
    a_opt = a_opt / a_opt.sum()
    a_opt = np.maximum(a_opt, 0)
    s = a_opt.sum()
    if s > 1e-15:
        a_opt = a_opt / s
    return a_opt, result.fun


def flat_spectrum_init(n, rng, n_trials=40):
    """Initialize with approximately flat power spectrum."""
    fft_size = 1 << (n - 1).bit_length()
    best_a = np.ones(n) / n
    best_obj = float('inf')
    for _ in range(n_trials):
        A = np.zeros(fft_size, dtype=complex)
        A[0] = np.sqrt(n)
        half = fft_size // 2
        phases = rng.uniform(0, 2 * np.pi, half)
        for k in range(1, half):
            A[k] = np.exp(1j * phases[k - 1])
            A[fft_size - k] = np.exp(-1j * phases[k - 1])
        if fft_size % 2 == 0:
            A[half] = np.exp(1j * phases[-1])
        a = np.real(ifft(A, fft_size))[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            a = a / s
            obj = 2.0 * n * fast_convolve(a).max()
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
    return best_a


def lp_refine(a0, n_iters=2000, patience=150, verbose=False, t_init=1.0):
    """LP-based iterative refinement with adaptive step size."""
    n = len(a0)
    a = np.maximum(a0, 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a.copy(), float('inf')
    a = a / s
    # Ensure minimum positivity to avoid LP issues
    a = np.maximum(a, 1e-12)
    a = a / a.sum()

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    stall_count = 0
    t = t_init

    for it in range(n_iters):
        conv = fast_convolve(a)
        M = conv.max()

        f_clipped = np.maximum(a, 1e-10)
        A_ub = build_lp_matrix_sparse(n, f_clipped)
        b_ub = np.full(2 * n - 1, M)

        result = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub,
                         bounds=[(0, None)] * n, method='highs')

        if not result.success:
            break

        g = np.maximum(result.x, 0)
        sg = g.sum()
        if sg < 1e-15:
            break

        g_norm = g / sg
        a_new = (1.0 - t) * a + t * g_norm
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a_new = np.maximum(a_new, 1e-12)
        a_new = a_new / a_new.sum()

        score_new = 2.0 * n * fast_convolve(a_new).max()

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            stall_count = 0
            t = min(1.0, t * 1.03)
        else:
            stall_count += 1
            t = max(0.02, t * 0.85)

        a = a_new

        if verbose and it % 200 == 0:
            print("    LP iter %d: score=%.8f, t=%.4f" % (it, best_score, t))

        if stall_count > patience:
            break

    return best_a, best_score


def lp_refine_with_restarts(a0, total_iters=3000, patience=100, n_restarts=5, rng=None):
    """LP refinement with random restarts if stuck."""
    if rng is None:
        rng = np.random.default_rng(0)

    n = len(a0)
    a = a0.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    iters_per_restart = total_iters // (n_restarts + 1)

    for restart in range(n_restarts + 1):
        a_opt, score = lp_refine(a, n_iters=iters_per_restart, patience=patience)
        if score < best_score:
            best_score = score
            best_a = a_opt.copy()
        a = a_opt

        if restart < n_restarts:
            # Perturb for restart
            noise_scale = 0.05 / (restart + 1)
            noise = rng.standard_normal(n) * noise_scale
            a_pert = project_simplex(best_a + noise)
            a = np.maximum(a_pert, 1e-12)
            a = a / a.sum()

    return best_a, best_score
