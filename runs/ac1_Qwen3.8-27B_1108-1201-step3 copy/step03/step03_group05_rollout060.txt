def fast_convolve(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([target_sum])
    # First project onto {x >= 0, sum(x) = 1}, then scale
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target_sum)
    if not cond.any():
        return np.full(n, target_sum / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    Subgradient descent with Nesterov momentum on the simplex.
    Minimizes max_k (a*a)[k] where a is on the probability simplex.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    no_improve = 0
    iter_count = 0

    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_optimize(a0, time_limit, seed=0, lr_init=2.0, beta_init=0.92):
    """
    Fast subgradient descent with momentum on the simplex.
    Minimizes max_k (a*a)[k] where sum(a)=1, a>=0.
    Uses single-peak subgradient with periodic multi-peak smoothing.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    beta = beta_init
    last_improve = t0
    iter_count = 0
    perturb_count = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0

        # Choose K based on iteration pattern
        # Mostly K=1 for speed, K=3 every 500 iters for smoothing
        phase = iter_count % 1000
        if phase < 800:
            K_use = 1
        else:
            K_use = 3

        # Find top peaks
        if K_use < len(conv):
            top_k = np.argpartition(conv, -K_use)[-K_use:]
        else:
            top_k = np.arange(len(conv))

        # Weights
        if K_use == 1:
            k_star = int(top_k[0])
            g = np.zeros(n)
            i_start = max(0, k_star - n + 1)
            i_end = min(n - 1, k_star)
            if i_start <= i_end:
                idx = k_star - np.arange(i_start, i_end + 1)
                g[i_start:i_end + 1] = 2.0 * a[idx]
        else:
            top_k_sorted = top_k[np.argsort(conv[top_k])[::-1]]
            pv = conv[top_k_sorted]
            weights = np.exp((pv - pv[0]) * 5)
            weights /= weights.sum()
            g = np.zeros(n)
            for ki, wi in zip(top_k_sorted, weights):
                ki = int(ki)
                i_start = max(0, ki - n + 1)
                i_end = min(n - 1, ki)
                if i_start <= i_end:
                    idx = ki - np.arange(i_start, i_end + 1)
                    g[i_start:i_end + 1] += 2.0 * wi * a[idx]

        # Nesterov momentum step
        v = beta * v - lr * g
        a = project_simplex(a + v)

        # Update
        conv = fast_convolve(a)
        cur_obj = conv.max()
        iter_count += 1

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            last_improve = time.monotonic()

        # Stagnation: perturb
        if time.monotonic() - last_improve > 2.5:
            perturb_count += 1
            # Scale increases with consecutive perturbations
            base_scale = 0.02 + 0.02 * min(perturb_count, 5)
            # Structured perturbation: move mass from high to low regions
            threshold = np.median(a)
            high_mask = a > threshold
            low_mask = a <= threshold
            n_high = high_mask.sum()
            n_low = low_mask.sum()
            if n_high > 0 and n_low > 0:
                # Move a fraction of high-mass coordinates to random low ones
                move_frac = 0.1 + 0.05 * (perturb_count % 3)
                for _ in range(min(5, n_high)):
                    hi_idx = np.where(high_mask)[0]
                    lo_idx = np.where(low_mask)[0]
                    i = hi_idx[rng.integers(len(hi_idx))]
                    j = lo_idx[rng.integers(len(lo_idx))]
                    amt = a[i] * move_frac
                    a[i] -= amt
                    a[j] += amt
                a = project_simplex(a)
            else:
                noise = rng.standard_normal(n) * base_scale
                a = project_simplex(a + noise)

            v = np.zeros(n)
            lr = lr_init * max(0.3, 1.0 - 0.1 * perturb_count)
            conv = fast_convolve(a)
            if conv.max() < best_obj:
                best_obj = conv.max()
                best_a = a.copy()
            last_improve = time.monotonic()
            perturb_count = 0 if conv.max() < best_obj * 1.001 else perturb_count

        # Adaptive learning rate
        ratio = cur_obj / max(best_obj, 1e-15)
        if ratio > 1.05:
            lr = max(0.05, lr * 0.85)
        elif ratio < 1.001:
            lr = min(lr_init * 1.5, lr * 1.02)
            beta = min(0.96, beta + 0.001)
        else:
            lr = max(0.05, lr * 0.98)

    return best_a, best_obj


def coordinate_polish(a0, time_limit, seed=0):
    """
    Coordinate-wise local search: try moving small amounts of mass
    between pairs of coordinates to reduce the peak convolution.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        cur_max = conv.max()
        k_star = int(np.argmax(conv))

        # Find the pair with max contribution to the peak
        i_start = max(0, k_star - n + 1)
        i_end = min(n - 1, k_star)
        if i_start > i_end:
            break

        contributions = np.array([a[i] * a[k_star - i] for i in range(i_start, i_end + 1)])
        i_max = i_start + int(np.argmax(contributions))

        # Try moving mass from i_max to various other positions
        improved = False
        # Try moving to positions where a[k_star - j] is small
        target_scores = np.array([a[k_star - j] if 0 <= k_star - j < n else 0.0
                                  for j in range(n)])
        # Sort targets by score (ascending) - prefer moving to low-contribution positions
        sorted_targets = np.argsort(target_scores)

        for _ in range(10):
            j = int(sorted_targets[rng.integers(len(sorted_targets))])
            if j == i_max:
                continue
            for frac in [0.05, 0.1, 0.15, 0.25]:
                amt = a[i_max] * frac
                if amt < 1e-10:
                    break
                a_try = a.copy()
                a_try[i_max] -= amt
                a_try[j] += amt
                conv_try = fast_convolve(a_try)
                if conv_try.max() < cur_max - 1e-12:
                    a = a_try
                    cur_max = conv_try.max()
                    if cur_max < best_obj:
                        best_obj = cur_max
                        best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        # Also try random pair moves
        if not improved:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-8:
                for frac in [0.03, 0.07, 0.12]:
                    amt = a[i] * frac
                    a_try = a.copy()
                    a_try[i] -= amt
                    a_try[j] += amt
                    conv_try = fast_convolve(a_try)
                    if conv_try.max() < cur_max - 1e-12:
                        a = a_try
                        if conv_try.max() < best_obj:
                            best_obj = conv_try.max()
                            best_a = a.copy()
                        improved = True
                        break

        no_improve = 0 if improved else no_improve + 1
        if no_improve > 5000:
            # Perturb
            noise = rng.standard_normal(n) * 0.03
            a = project_simplex(a + noise)
            no_improve = 0

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Bimodal (edges)
    bi = np.zeros(n)
    edge = max(2, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Bimodal (quarter points)
    bi2 = np.zeros(n)
    q = n // 4
    bi2[q:2*q] = 1.0
    bi2[3*q:n] = 1.0
    bi2 /= bi2.sum()
    inits['bimodal_q'] = bi2

    # Bell (center peak)
    tri = 1.0 / (1.0 + 0.3 * ((x - n / 2) / (n / 5)) ** 2)
    tri /= tri.sum()
    inits['bell'] = tri

    # Raised cosine
    if n > 2:
        cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
        cos_w = np.maximum(cos_w, 1e-8)
        cos_w /= cos_w.sum()
        inits['raised_cos'] = cos_w

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.25 * n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g

    # Random Dirichlet
    inits['rand1'] = rng.dirichlet(np.ones(n) * 0.3)
    inits['rand2'] = rng.dirichlet(np.ones(n) * 1.0)
    inits['rand3'] = rng.dirichlet(np.ones(n) * 3.0)

    # Parent-based
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps
        # Reversed parent
        pp = np.maximum(parent[::-1], 0)
        ps = pp.sum()
        if ps > 1e-10:
            if len(pp) >= n:
                inits['parent_rev'] = pp[:n] / pp[:n].sum()
            else:
                pr = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(pp)), pp)
                pr = np.maximum(pr, 0)
                if pr.sum() > 1e-10:
                    inits['parent_rev'] = pr / pr.sum()

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search for minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Phase 1: Quick exploration across lengths and initializations.
    Phase 2: Deep subgradient optimization on top candidates.
    Phase 3: Coordinate-wise polish and final refinement.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            parent_score = evaluate_sequence(p.tolist())
            print(f"Parent: n={len(p)}, score={parent_score:.6f}", flush=True)
            if np.isfinite(parent_score):
                best_score = parent_score
                best_sequence = p.copy()
    except Exception as e:
        print(f"No parent: {e}", flush=True)

    # Phase 1: Exploration
    print("=== Phase 1: Exploration ===", flush=True)
    t1_start = time.monotonic()
    phase1_budget = min(300.0, (deadline - time.monotonic()) * 0.25)

    lengths = [192, 256, 320, 384, 448, 512]
    scan_results = []

    for n in lengths:
        if time.monotonic() > deadline - (deadline - t1_start) * 0.6:
            break

        inits = generate_initializations(n, parent, rng)
        # Priority ordering
        priority = ['parent', 'parent_ds', 'parent_up', 'parent_rev',
                    'bimodal', 'bimodal_q', 'bell', 'raised_cos',
                    'gauss', 'uniform', 'rand1', 'rand2', 'rand3']
        names = [nm for nm in priority if nm in inits]
        names = names[:7]

        n_per_init = max(3.0, phase1_budget / (len(lengths) * len(names)))
        for name in names:
            if time.monotonic() > deadline - (deadline - t1_start) * 0.55:
                break
            a0 = inits[name]
            a_opt, obj = subgradient_optimize(
                a0, min(n_per_init, 15.0), seed=int(rng.integers(0, 100000)))
            score = 2.0 * n * obj  # Since sum(a)=1, score = 2n * max(conv)
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n}, {name}: {score:.6f}", flush=True)

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 best: {best_score:.6f}", flush=True)

    # Phase 2: Deep optimization
    print("=== Phase 2: Deep Optimization ===", flush=True)

    top_configs = []
    seen_keys = set()
    for score, n, name, a_opt in scan_results:
        key = n
        if key not in seen_keys:
            top_configs.append((n, a_opt))
            seen_keys.add(key)
        if len(top_configs) >= 4:
            break

    # Include current best
    if best_sequence is not None:
        top_configs.insert(0, (len(best_sequence), best_sequence.copy()))

    # Add a couple random restarts at the best length
    if best_sequence is not None:
        best_n = len(best_sequence)
        top_configs.append((best_n, rng.dirichlet(np.ones(best_n) * 0.5)))
        top_configs.append((best_n, rng.dirichlet(np.ones(best_n) * 2.0)))

    remaining = deadline - time.monotonic() - 120.0
    n_configs = len(top_configs)
    per_config = max(30.0, remaining / n_configs)

    for i, (n, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 100:
            break
        t_alloc = min(per_config, 350.0)
        if t_alloc < 15:
            break

        # Split time: 70% subgradient, 30% coordinate polish
        t_sub = t_alloc * 0.70
        t_polish = t_alloc * 0.30

        # Subgradient optimization
        a_opt, obj = subgradient_optimize(
            a0, t_sub, seed=int(rng.integers(0, 1000000)),
            lr_init=2.0, beta_init=0.92)
        score = 2.0 * n * obj
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Config {i} (n={n}) sub: {score:.6f}", flush=True)

        # Coordinate polish
        if t_polish > 10 and best_sequence is not None:
            a_pol, obj_pol = coordinate_polish(
                best_sequence, min(t_polish, 60.0),
                seed=int(rng.integers(0, 1000000)))
            score_pol = 2.0 * len(a_pol) * obj_pol
            if np.isfinite(score_pol) and score_pol < best_score:
                best_score = score_pol
                best_sequence = a_pol.copy()
                print(f"  Config {i} polish: {score_pol:.6f}", flush=True)

    # Phase 3: Final polish
    print("=== Phase 3: Final Polish ===", flush=True)
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 10.0

        # Fine subgradient with low lr
        t_fg = remaining * 0.4
        if t_fg > 10:
            a_fg, obj_fg = subgradient_optimize(
                best_sequence, t_fg, seed=7777, lr_init=0.5, beta_init=0.95)
            score_fg = 2.0 * len(a_fg) * obj_fg
            if np.isfinite(score_fg) and score_fg < best_score:
                best_score = score_fg
                best_sequence = a_fg.copy()
                print(f"  Fine sub: {score_fg:.6f}", flush=True)

        # Coordinate polish
        remaining = deadline - time.monotonic() - 10.0
        t_cp = remaining * 0.4
        if t_cp > 10:
            a_cp, obj_cp = coordinate_polish(
                best_sequence, t_cp, seed=8888)
            score_cp = 2.0 * len(a_cp) * obj_cp
            if np.isfinite(score_cp) and score_cp < best_score:
                best_score = score_cp
                best_sequence = a_cp.copy()
                print(f"  Coord polish: {score_cp:.6f}", flush=True)

        # One more subgradient pass
        remaining = deadline - time.monotonic() - 5.0
        if remaining > 5:
            a_final, obj_final = subgradient_optimize(
                best_sequence, remaining, seed=9999, lr_init=0.3, beta_init=0.96)
            score_final = 2.0 * len(a_final) * obj_final
            if np.isfinite(score_final) and score_final < best_score:
                best_score = score_final
                best_sequence = a_final.copy()
                print(f"  Final sub: {score_final:.6f}", flush=True)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256

    # Verify with actual evaluator
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_sequence)})", flush=True)
    return [float(max(0.0, min(1000.0, x))) for x in best_sequence]
