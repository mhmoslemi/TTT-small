def simplex_project_old(a):
    """Projects onto the probability simplex {x | x ≥ 0, sum(x)=1}."""
    n = len(a)
    if n == 1:
        return np.array([max(a[0], 0.0)])
    c = np.sort(a, kind='quicksort', order=None)[::-1]
    p = np.arange(n)
    cssv = np.cumsum(c)
    cond = c * p > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.max(p[cond])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(a - theta, 0.0)
