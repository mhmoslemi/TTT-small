Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Adam optimization with softmax

peak-directed mutations, and multi-length search.

import numpy as np
import time
from scipy.fft import fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]

    """Compute smooth max (log-sum-exp) and softmax weights."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    total = exp_shifted.sum()
    if total < 1e-300:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = exp_shifted / total
    sm = x_max + np.log(total) / beta
    return sm, w


def compute_grad_conv(a, w):
    """Gradient of dot(w, conv(a)) w.r.t. a using FFT."""
    n = len(a)
    N = 1 << (len(w) + n - 1 + 63).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    grad = np.real(ifft(np.conj(W) * A, N))[:n] * 2.0
    return grad


def softmax(x):
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    return e / e.sum()


def adam_step(x, grad, state, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """In-place Adam update. State is dict with m, v, t."""
    state['t'] += 1
    t = state['t']
    if 'm' not in state:
        state['m'] = np.zeros_like(x)
        state['v'] = np.zeros_like(x)
    state['m'] = beta1 * state['m'] + (1 - beta1) * grad
    state['v'] = beta2 * state['v'] + (1 - beta2) * (grad ** 2)
    m_hat = state['m'] / (1 - beta1 ** t)
    v_hat = state['v'] / (1 - beta2 ** t)
    x -= lr * m_hat / (np.sqrt(v_hat) + eps)
    return x


def optimize_adam(a0, initial_beta, final_beta, n_iterations, lr=0.02,
                  beta1=0.9, beta2=0.999, patience=200, plateau_factor=0.5):
    """
    Adam optimization with multi-beta annealing and restarts.
    a0: initial point on simplex (will be converted to log-space)
    Returns best a found.
    """
    n = len(a0)
    a_init = np.maximum(a0, 1e-30)
    a_init = a_init / a_init.sum()
    x = np.log(a_init)

    best_obj = float('inf')
    best_x = x.copy()
    state = {'t': 0}

    # Annealing schedule: beta goes from initial_beta to final_beta
    n_anneal = n_iterations
    current_iter = 0
    no_improve = 0
    restarts = 0
    max_restarts = 30

    while current_iter < n_iterations and restarts < max_restarts:
        # Current beta (exponential annealing)
        progress = current_iter / max(1, n_anneal)
        log_beta = np.log(initial_beta) * (1 - progress) + np.log(final_beta) * progress
        beta = np.exp(log_beta)

        # Check for plateau
        if no_improve > patience and restarts < max_restarts:
            # Perturb and restart
            noise_scale = 0.1 * (1 + restarts * 0.1)
            x = best_x + noise_scale * np.random.standard_normal(n)
            state = {'t': 0}
            no_improve = 0
            restarts += 1
            # Reset learning rate
            lr = max(0.001, lr * plateau_factor)
            continue

        # Compute objective and gradient
        a = softmax(x)
        conv = fast_convolve(a)
        sm, w = smooth_max_and_weights(conv, beta)
        obj = 2.0 * n * sm

        if obj < best_obj:
            best_obj = obj
            best_x = x.copy()
            no_improve = 0
            # Decrease learning rate slightly on improvement
            lr = max(0.0005, lr * 0.9995)
        else:
            no_improve += 1

        # Compute gradient
        grad_a = compute_grad_conv(a, w)
        lagrange = np.dot(a, grad_a)
        grad_x = a * (grad_a - lagrange)

        # Adam step
        x = adam_step(x, grad_x, state, lr, beta1, beta2)

        current_iter += 1

    # Final: evaluate with true max
    a_best = softmax(best_x)
    return a_best, 2.0 * n * fast_convolve(a_best).max()


def optimize_lscg(a0, initial_beta=100.0, final_beta=3.0, n_phases=6):
    """
    Multi-phase optimization: start with smooth (high beta), end with sharp (low beta).
    Each phase uses Adam with appropriate settings.
    """
    n = len(a0)
    a_cur = np.maximum(a0, 1e-30).copy()
    a_cur = a_cur / a_cur.sum()

    best_obj = float('inf')
    best_a = a_cur.copy()

    for phase in range(n_phases):
        frac = (phase + 1) / n_phases
        beta = initial_beta * (final_beta / initial_beta) ** frac
        n_iter = int(500 + 1500 * (1 - frac))  # More iterations at low beta

        # Reduce learning rate as beta decreases
        lr = 0.03 * (0.7 ** phase)

        a_new, obj_phase = optimize_adam(
            a_cur, beta, beta * 0.5, n_iter, lr=lr,
            patience=100 + 50 * phase
        )

        if obj_phase < best_obj:
            best_obj = obj_phase
            best_a = a_new.copy()

        a_cur = best_a.copy()

    return best_a, 2.0 * n * fast_convolve(best_a).max()


def peak_directed_mutation(a, n_peaks=3, strength=0.05):
    """
    Mutate the sequence by reducing contributions to the top convolution peaks.
    """
    n = len(a)
    conv = fast_convolve(a)
    # Find top peaks (excluding the very center which is hardest to reduce)
    sorted_peaks = np.argsort(conv)[::-1][:n_peaks]

    a_new = a.copy()
    for k in sorted_peaks:
        # Elements contributing to conv[k]: a[i] * a[k-i] for valid i
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start >= i_end:
            continue
        # Reduce elements that contribute most to this peak
        contrib = np.zeros(n)
        for i in range(i_start, i_end + 1):
            j = k - i
            if 0 <= j < n:
                contrib[i] += a[j]
                if i != j:
                    contrib[j] += a[i]
        total = contrib.sum()
        if total > 1e-15:
            reduction = strength * (conv[k] - conv.min()) / total
            a_new -= reduction * contrib
    a_new = np.maximum(a_new, 1e-30)
    a_new = a_new / a_new.sum()
    return a_new


def random_bimodal_init(n, rng):
    """Generate a bimodal initialization."""
    center1 = rng.uniform(0.1, 0.4) * n
    center2 = rng.uniform(0.6, 0.9) * n
    width1 = rng.uniform(0.05, 0.15) * n
    width2 = rng.uniform(0.05, 0.15) * n
    x = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((x - center1) / width1) ** 2) + np.exp(-0.5 * ((x - center2) / width2) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def asymmetric_ramp_init(n, rng):
    """Asymmetric ramp initialization."""
    x = np.arange(n, dtype=np.float64)
    skew = rng.uniform(0.5, 2.0)
    a = np.exp(-x / (n * skew))
    if rng.uniform() < 0.5:
        a = a[::-1]
    a = np.maximum(a, 0)
    a /= a.sum()
    return a


def arctan_shape_init(n, rng):
    """Arc-tangent shaped initialization (flatter than uniform tail)."""
    x = np.arange(n, dtype=np.float64)
    cx = n * rng.uniform(0.3, 0.7)
    width = n * rng.uniform(0.3, 0.6)
    a = 1.0 / (1.0 + ((x - cx) / width) ** 2)
    a = np.maximum(a, 0)
    a /= a.sum()
    return a


def generate_diverse_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Asymmetric shapes
    inits['asym_ramp1'] = asymmetric_ramp_init(n, rng)
    inits['asym_ramp2'] = asymmetric_ramp_init(n, rng)
    inits['bimodal1'] = random_bimodal_init(n, rng)
    inits['bimodal2'] = random_bimodal_init(n, rng)
    inits['arctan1'] = arctan_shape_init(n, rng)
    inits['arctan2'] = arctan_shape_init(n, rng)

    # Power-law shapes
    for alpha in [1.5, 2.5, 4.0]:
        a = (x + 1) ** (-alpha)
        a /= a.sum()
        inits['powlaw_%.1f' % alpha] = a
        a_rev = a[::-1].copy()
        inits['powlaw_rev_%.1f' % alpha] = a_rev

    # Sine-based
    for freq in [1.0, 2.0, 3.0]:
        a = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x / n)
        a = np.maximum(a, 0)
        if a.sum() > 1e-10:
            a /= a.sum()
        inits['sine_%.0f' % freq] = a

    # Flat spectrum init
    fft_size = 1 << (n - 1).bit_length()
    for trial in range(5):
        A = np.zeros(fft_size, dtype=complex)
        phases = rng.uniform(0, 2 * np.pi, fft_size)
        for k in range(fft_size):
            A[k] = np.exp(1j * phases[k])
        A[fft_size - np.arange(1, fft_size // 2 + 1)] = np.conj(
            A[np.arange(1, fft_size // 2 + 1)])
        a = np.real(ifft(A, fft_size))[:n]
        a = np.maximum(a, 0)
        if a.sum() > 1e-10:
            a /= a.sum()
            inits['flatspec_%d' % trial] = a

    # From parent
    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            pp = np.maximum(parent, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)),
                           np.maximum(parent, 0))
            pp = np.maximum(pp, 1e-15)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_interp'] = pp / ps

    return inits


def coordinate_descent_polish(a, n_iters=100, n_candidates=5):
    """
    Coordinate descent: try adjusting one element at a time.
    """
    n = len(a)
    a = np.maximum(a, 1e-30).copy()
    a = a / a.sum()
    best_obj = 2.0 * n * fast_convolve(a).max()
    rng = np.random.default_rng(12345)

    for it in range(n_iters):
        improved = False
        # Try random subset of coordinates
        indices = rng.choice(n, size=min(n, 32), replace=False)
        for i in indices:
            old_val = a[i]
            # Try different values
            candidates = [old_val * 0.8, old_val * 0.9, old_val * 1.1, old_val * 1.2,
                          old_val + 0.01 / n, old_val - 0.01 / n,
                          old_val * 0.5, old_val * 1.5]
            for c in candidates:
                if c < 0:
                    continue
                a[i] = c
                a = np.maximum(a, 0)
                s = a.sum()
                if s < 1e-10:
                    a[i] = old_val
                    continue
                a = a / s
                obj = 2.0 * n * fast_convolve(a).max()
                if obj < best_obj:
                    best_obj = obj
                    improved = True
                    break
                a[i] = old_val
            if not any(a[i] != old_val for _ in [0]):
                a[i] = old_val

        if not improved and it > 50:
            break

    return a, best_obj


def evaluate_local(a):
    """Local evaluation (same as evaluate_sequence but for numpy arrays)."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    conv = fast_convolve(a)
    max_b = conv.max()
    return float(2 * n * max_b / (s ** 2))


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search: multi-phase optimization using Adam with multi-beta annealing,
    diverse asymmetric initializations, peak-directed mutations, and coordinate polish.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print("Parent: len=%d, sum=%.4f" % (len(p), p.sum()))
    except Exception:
        pass

    def score_and_update(a):
        nonlocal best_sequence, best_score
        if a is None:
            return float('inf')
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            elapsed = time.monotonic() - t_start
            print("  NEW BEST: %.6f (n=%d, t=%.0fs)" % (score, len(a), elapsed))
        return score if np.isfinite(score) else float('inf')

    # === Phase 1: Multi-start scan ===
    print("=== Phase 1: Multi-start Scan ===")
    lengths = [128, 192, 256, 320, 384]
    phase1_time = 400.0
    phase1_deadline = t_start + phase1_time

    scan_results = []

    for n in lengths:
        if time.monotonic() > phase1_deadline - 60:
            break
        print("  Scanning n=%d ..." % n)
        inits = generate_diverse_inits(n, parent, rng)

        # Score all inits quickly
        init_scores = []
        for name, a in inits.items():
            obj = evaluate_local(a)
            init_scores.append((obj, name, a))

        init_scores.sort(key=lambda x: x[0])
        # Take top 4 inits
        top_inits = init_scores[:4]
        print("    Top init: %s obj=%.6f" % (top_inits[0][1], top_inits[0][0]))

        for obj_init, name, a0 in top_inits:
            if time.monotonic() > phase1_deadline - 30:
                break
            # Quick Adam optimization
            a_opt, obj_opt = optimize_adam(a0, initial_beta=80.0, final_beta=15.0,
                                           n_iterations=400, lr=0.025)
            score = score_and_update(a_opt)
            scan_results.append((score, n, name, a_opt.copy()))

    print("Phase 1 complete. Best: %.6f" % best_score)

    # Sort and pick top candidates for deep optimization
    scan_results.sort(key=lambda x: x[0])

    # === Phase 2: Deep optimization ===
    print("=== Phase 2: Deep Optimization ===")
    phase2_deadline = deadline - 120.0

    # Pick diverse lengths to optimize deeply
    seen_lengths = set()
    deep_candidates = []
    for score, n, name, a in scan_results:
        if n not in seen_lengths:
            seen_lengths.add(n)
            deep_candidates.append((score, n, name, a))
        if len(deep_candidates) >= 3:
            break

    # Also include the best overall
    if scan_results:
        best_scan = scan_results[0]
        if best_scan[1] not in seen_lengths:
            deep_candidates.append(best_scan)

    for score_init, n, name, a0 in deep_candidates:
        if time.monotonic() > phase2_deadline - 60:
            break
        print("  Deep: n=%d, init=%s (score=%.6f)" % (n, name, score_init))

        # Multi-phase Adam with annealing
        a_opt, obj_opt = optimize_lscg(a0, initial_beta=200.0, final_beta=3.0,
                                       n_phases=5)
        score = score_and_update(a_opt)

        # Peak-directed mutation + re-optimize
        if time.monotonic() < phase2_deadline - 30:
            a_mut = peak_directed_mutation(a_opt, n_peaks=5, strength=0.03)
            a_opt2, obj_opt2 = optimize_adam(a_mut, initial_beta=50.0, final_beta=10.0,
                                             n_iterations=500, lr=0.015)
            score = score_and_update(a_opt2)

    print("Phase 2 complete. Best: %.6f" % best_score)

    # === Phase 3: Focus on best length, more diverse starts ===
    print("=== Phase 3: Refined Search ===")
    if best_sequence is not None:
        best_n = len(best_sequence)
    else:
        best_n = 256

    phase3_deadline = deadline - 60.0
    n_trials = 6
    trial_time = max(15.0, (phase3_deadline - time.monotonic()) / n_trials)

    for trial in range(n_trials):
        if time.monotonic() > phase3_deadline - 30:
            break

        # Create diverse starting points around the best
        if best_sequence is not None:
            a_base = best_sequence / best_sequence.sum()
            scale = 0.02 + 0.01 * trial
            noise = rng.standard_normal(len(a_base)) * scale
            a0 = np.maximum(a_base + noise, 1e-30)
            a0 = a0 / a0.sum()
        else:
            a0 = np.ones(best_n) / best_n

        # Run optimization with different parameters
        beta_start = 100.0 * (0.8 ** trial)
        beta_end = 5.0 * (0.9 ** trial)
        lr = 0.02 * (0.85 ** trial)
        n_iter = int(min(800, trial_time * 80))

        a_opt, _ = optimize_adam(a0, beta_start, beta_end, n_iter, lr=lr)
        score = score_and_update(a_opt)

        # Peak mutation variant
        if time.monotonic() < phase3_deadline - 20:
            a_mut = peak_directed_mutation(a_opt, n_peaks=trial % 5 + 2,
                                           strength=0.02 + 0.01 * trial)
            a_opt2, _ = optimize_adam(a_mut, beta_end, beta_end * 0.5,
                                      n_iterations=min(300, n_iter // 2), lr=lr * 0.7)
            score = score_and_update(a_opt2)

    print("Phase 3 complete. Best: %.6f" % best_score)

    # === Phase 4: Polish ===
    print("=== Phase 4: Polish ===")
    if best_sequence is not None:
        n_best = len(best_sequence)
        remaining = deadline - time.monotonic() - 20.0
        a_best = best_sequence / best_sequence.sum()

        # Coordinate descent
        if remaining > 30:
            a_cd, obj_cd = coordinate_descent_polish(a_best, n_iters=80)
            score = score_and_update(a_cd)
            remaining = deadline - time.monotonic() - 20.0

        # Final Adam polish with very low beta
        if remaining > 20:
            a_final, _ = optimize_adam(a_best, initial_beta=10.0, final_beta=2.0,
                                       n_iterations=min(500, int(remaining * 40)), lr=0.005)
            score = score_and_update(a_final)
            remaining = deadline - time.monotonic() - 20.0

        # Small random perturbations
        if remaining > 15:
            for _ in range(3):
                if remaining < 10:
                    break
                a_pert = np.maximum(a_best + 0.005 * rng.standard_normal(n_best), 1e-30)
                a_pert = a_pert / a_pert.sum()
                a_t, _ = optimize_adam(a_pert, initial_beta=20.0, final_beta=5.0,
                                       n_iterations=200, lr=0.01)
                score_and_update(a_t)
                remaining = deadline - time.monotonic() - 20.0

    # === Try a different length if time permits ===
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 15.0
        if remaining > 120:
            alt_n = 384 if len(best_sequence) == 256 else 256
            if alt_n != len(best_sequence):
                print("  Trying alt length n=%d" % alt_n)
                pp = np.interp(np.linspace(0, 1, alt_n),
                               np.linspace(0, 1, len(best_sequence)),
                               np.maximum(best_sequence, 0))
                pp = np.maximum(pp, 1e-15)
                pp = pp / pp.sum()
                a_alt, _ = optimize_lscg(pp, initial_beta=100.0, final_beta=5.0, n_phases=4)
                score_and_update(a_alt)

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(256) / 256.0

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print("\nFINAL: score=%.6f, n=%d" % (best_score, len(best_sequence)))
    return [float(x) for x in best_sequence]
