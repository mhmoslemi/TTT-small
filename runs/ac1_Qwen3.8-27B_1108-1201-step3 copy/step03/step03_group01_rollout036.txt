def lp_direction(a, M):
    """Solve LP: max sum(g) s.t. a*g <= M, g >= 0."""
    n = len(a)
    C = np.zeros((2*n-1, n))
    for i in range(n):
        C[i:i+n, i] = a
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        return result.x
    return None

def peak_shave(a, top_k=10, delta=0.01):
    """Directly reduce the top peaks by shaving contributing coordinates."""
    a = a.copy()
    n = len(a)
    conv = fast_convolve(a)
    top_positions = np.argsort(conv)[-top_k:]
    
    for k in top_positions:
        # Find which pairs (i, k-i) contribute most
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        contributions = np.array([a[i] * a[k-i] for i in range(i_start, i_end+1)])
        idx = np.arange(i_start, i_end+1)
        
        # Reduce the larger element in each high-contribution pair
        order = np.argsort(contributions)[::-1][:5]
        for oi in order:
            i = idx[oi]
            j = k - i
            if i != j:
                # Reduce the larger one
                if a[i] > a[j]:
                    a[i] *= (1 - delta)
                else:
                    a[j] *= (1 - delta)
            else:
                a[i] *= (1 - delta/2)
    
    return np.maximum(a, 0)

But this is just (a*g)_k <= (t + (a*a)_k)/2, which is similar to the original LP constraint.

I think the LP approach is already quite good. Let me focus on making it more effective through:
1. Better line search (parabolic interpolation)
2. Using the dual for targeted perturbations
3. Multiple LP solves with different M values (optimistic/pessimistic)
4. Periodic "reheating" with structured perturbations

**Let me also consider a completely different approach: direct search on the "active set."**

At the optimum, the max of conv(a,a) is achieved at a set of positions P (the "active set" of peaks). If we fix P, the problem becomes:
minimize (a*a)_k for k in P (all equal to the max)
subject to (a*a)_k <= max for k not in P
a >= 0, sum(a) = 1

This is a system of equations/inequalities. We could try to solve it directly using Newton's method or similar.

The key equations are: (a*a)_{k_1} = (a*a)_{k_2} = ... = (a*a)_{k_m} = t (all peaks equal)
And: (a*a)_k <= t for all other k.
And: sum(a) = 1, a >= 0.

This is a nonlinear system. The number of equations is m+1 (m peak equalities + sum constraint) and the unknowns are n + 1 (n coordinates + t). So we have more unknowns than equations, meaning there are many solutions. We want the one that minimizes t.

This is getting too complex for a practical implementation in 1800 seconds. Let me stick with the LP + L-BFGS approach but with key improvements.

**Final decision on the algorithm:**

1. Start with parent (n=512) and a few other initializations.
2. Main loop: alternate between LP-based optimization and L-BFGS-B soft-max optimization.
3. Use dual information for targeted perturbations.
4. Periodic structured perturbations (cyclic shifts, spectral modifications).
5. Track the best solution and return it.

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def objective_from_conv(conv, n):
    return 2.0 * n * conv.max()

def build_conv_matrix(a):
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C

def lp_solve(a, M):
    n = len(a)
    C = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        dual = None
        if hasattr(res, 'ineqlin') and res.ineqlin is not None:
            dual = res.ineqlin.marginals
        return res.x, dual
    return None, None

def soft_max_val_and_grad(a, alpha):
    n = len(a)
    conv = fast_convolve(a)
    cm = conv.max()
    if cm < 1e-30:
        return 0.0, np.zeros(n)
    scaled = alpha * (conv - cm)
    exp_s = np.exp(scaled)
    log_sum = cm + np.log(exp_s.sum()) / alpha
    w = exp_s / exp_s.sum()
    # grad[i] = 2 * sum_k w[k] * a[k-i] (valid range)
    a_flip = a[::-1]
    cross = np.convolve(a_flip, w)[:2*n-1]
    grad = 2.0 * cross[n-1:2*n-1]
    return log_sum, grad

    a = normalize(a0.copy())

        xn = normalize(x)
        val, _ = soft_max_val_and_grad(xn, alpha)
        return 2.0 * n * val

        xn = normalize(x)
        _, g = soft_max_val_and_grad(xn, alpha)
        return 2.0 * n * g

                   bounds=[(0, None)] * n,
                   options={'maxiter': maxiter, 'gtol': tol, 'ftol': 1e-10})
    return normalize(res.x)
