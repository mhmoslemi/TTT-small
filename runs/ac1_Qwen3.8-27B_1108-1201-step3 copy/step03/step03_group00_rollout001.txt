import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_max_and_grad(a, T, top_k=50):
    """
    Compute smooth max of conv(a,a) and its gradient using top-k peaks.
    Returns (smooth_max_value, gradient_vector).
    """
    n = len(a)
    conv = fast_convolve(a)
    L = len(conv)

    if top_k >= L:
        peaks = np.arange(L)
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]

    pv = conv[peaks]
    scaled = pv / T
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    smooth_val = (1.0 / T) * (T * pv.max() + np.log(ws))

    grad = np.zeros(n)
    for i in range(len(peaks)):
        wi = w[i]
        if wi < 1e-14:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * wi * a[k - idx]

    return smooth_val, grad


def peak_deflation_step(a, conv, n, top_peaks=5, shrink=0.3):
    """
    Peak deflation: identify coordinates contributing most to the convolution peak
    and reduce them (redistributing mass to less contributing positions).
    """
    a_new = a.copy()
    L = len(conv)
    peak_idx = np.argpartition(conv, -top_peaks)[-top_peaks:]
    peak_idx = peak_idx[np.argsort(conv[peak_idx])[::-1]]

    for k in peak_idx:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k - idx]
        # Find the coordinate with maximum contribution
        max_local = np.argmax(contrib)
        m = idx[max_local]
        # Reduce this coordinate and redistribute to a random "quiet" position
        reduction = a_new[m] * shrink
        if reduction < 1e-15:
            continue
        a_new[m] -= reduction
        # Redistribute to positions that don't contribute to the peak
        quiet_mask = np.ones(n, dtype=bool)
        for kp in peak_idx:
            js2 = max(0, kp - n + 1)
            je2 = min(n - 1, kp)
            for q in range(js2, je2 + 1):
                if 0 <= kp - q < n:
                    quiet_mask[q] = False
        quiet_positions = np.where(quiet_mask)[0]
        if len(quiet_positions) == 0:
            quiet_positions = np.arange(n)
        # Distribute mass to a few random quiet positions
        n_redistribute = min(5, len(quiet_positions))
        targets = rng_deflation.choice(quiet_positions, size=n_redistribute, replace=False)
        frac = reduction / len(targets)
        for t in targets:
            a_new[t] += frac

    # Project to simplex
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        a_new = np.ones(n) / n
    else:
        a_new /= s
    return a_new


rng_deflation = np.random.default_rng(12345)


def adaptive_sg_optimize(f0, time_limit, T_init=0.002, T_final=0.0001,
                         top_k=30, mom_init=0.9, mom_final=0.97,
                         deflation_freq=10, verbose=False, seed=0):
    """
    Adaptive subgradient optimization with:
    - Smooth max objective with geometric temperature schedule
    - Backtracking line search
    - Heavy-ball momentum with adaptive coefficient
    - Periodic peak deflation steps
    """
    global rng_deflation
    rng_deflation = np.random.default_rng(seed)

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    lr = 1.0 / (2.0 * np.sqrt(n))  # initial learning rate

    total_iters_expected = 20000
    T_ratio = T_final / T_init

    while time.monotonic() - t0 < time_limit:
        it += 1
        progress = min(1.0, it / total_iters_expected)
        T = T_init * (T_ratio ** progress)
        mom = mom_init + (mom_final - mom_init) * progress

        sv, grad = smooth_max_and_grad(f, T, top_k=top_k)

        # Backtracking line search
        alpha = lr
        f_base = f.copy()
        val_base = M  # use actual max, not smooth approx

        best_alpha = 0.0
        for _ in range(10):
            v_new = mom * v - alpha * grad
            f_trial = project_simplex(f_base + v_new)
            conv_trial = fast_convolve(f_trial)
            val_trial = conv_trial.max()
            if val_trial <= val_base + 1e-15:
                best_alpha = alpha
                break
            alpha *= 0.5

        if best_alpha > 0:
            v = mom * v - best_alpha * grad
            f = project_simplex(f_base + v)
            conv = fast_convolve(f)
            M = conv.max()
        else:
            # No descent found; try peak deflation
            f = peak_deflation_step(f, conv, n, top_peaks=3, shrink=0.2)
            conv = fast_convolve(f)
            M = conv.max()
            v = np.zeros(n)

        # Periodic peak deflation
        if deflation_freq > 0 and it % deflation_freq == 0:
            f_def = peak_deflation_step(f, conv, n, top_peaks=5, shrink=0.15)
            conv_def = fast_convolve(f_def)
            if conv_def.max() < M:
                f = f_def
                conv = conv_def
                M = conv.max()

        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Restart if stuck
        if no_improve > 200:
            rng_r = np.random.default_rng(it)
            scale = 0.005 * min(5.0, no_improve / 50.0)
            f = project_simplex(best_f + rng_r.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            v = np.zeros(n)
            no_improve = 0

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      it={it}, obj={best_obj:.10f}, T={T:.6f}, lr={best_alpha:.6f}, elapsed={elapsed:.1f}s")

    return best_f, best_obj


def sim_anneal_polish(f0, time_limit, seed=0, verbose=False):
    """
    Simulated annealing polish: randomly perturb small groups of coordinates
    and accept based on Metropolis criterion.
    """
    rng = np.random.default_rng(seed)
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    T = 0.01
    T_cool = 0.8

    while time.monotonic() - t0 < time_limit:
        # Random perturbation: pick 2-5 coordinates and swap/adjust
        n_perturb = rng.integers(2, 6)
        idx = rng.choice(n, size=n_perturb, replace=False)

        if rng.random() < 0.5:
            # Swap-based perturbation
            i, j = idx[0], idx[1]
            swap_amount = (f[i] - f[j]) * rng.uniform(0.1, 0.8)
            f[i] -= swap_amount
            f[j] += swap_amount
        else:
            # Local transfer
            i = idx[0]
            j = idx[1]
            transfer = f[i] * rng.uniform(0.05, 0.3)
            f[i] -= transfer
            f[j] += transfer

        f = np.maximum(f, 0)
        fs = f.sum()
        if fs < 1e-15:
            f = best_f.copy()
            continue
        f = f / fs

        conv = fast_convolve(f)
        M_new = conv.max()
        delta = M_new - M

        if delta < 0 or rng.random() < np.exp(-delta / T):
            M = M_new
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
        else:
            f = best_f.copy()

        T *= T_cool
        if T < 0.0005:
            T = 0.005  # reset temperature occasionally

    return best_f, best_obj


def coord_descent_polish(f0, time_limit, seed=0, verbose=False):
    """
    Coordinate descent: iterate over coordinates, find optimal value for each
    given the others fixed. Since conv is quadratic in each coordinate,
    the optimal value can be found analytically.
    """
    rng = np.random.default_rng(seed)
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        # Shuffle coordinate order
        order = rng.permutation(n)

        for m in order[::max(1, n // 200)]:  # subset per pass for speed
            # conv(a,a)[k] as function of a[m] (with others fixed):
            # For each k, conv[k] = a[m]*a[k-m] + a[k-m]*a[m] + rest = 2*a[m]*a[k-m] + rest_k
            # where rest_k = sum_{i+j=k, i!=m, j!=m} a[i]*a[j]
            # We want to minimize max_k (2*a[m]*a[k-m] + rest_k) over a[m] >= 0
            # subject to the simplex constraint (which changes when a[m] changes)
            # Simplified: just try a few candidate values for a[m]

            base_sum = f.sum() - f[m]
            best_f_m = f[m]
            best_local_M = conv.max()

            # Try reducing a[m]
            for frac in [0.5, 0.3, 0.1, 0.0]:
                f[m] = f[m] * frac
                s = f.sum()
                if s < 1e-15:
                    f[m] = best_f_m
                    continue
                f_norm = f / s
                c = fast_convolve(f_norm)
                cM = c.max()
                if cM < best_local_M - 1e-14:
                    best_local_M = cM
                    best_f_m = f[m] / s
                f[m] = f[m] * (base_sum + best_f_m) / (base_sum + f[m]) if f[m] > 0 else 0

            # Simpler: just try setting f[m] to 0, to f[m]*2, and to average
            candidates = [0.0, f[m], f[m] * 1.5, f[m] * 0.5, f[m] * 0.1]
            best_cand = f[m]
            best_cand_M = conv.max()

            for cv in candidates:
                if cv < 0:
                    cv = 0.0
                # Adjust to maintain sum=1
                new_sum = f.sum() - f[m] + cv
                if new_sum < 1e-15:
                    continue
                f[m] = cv
                f_trial = f / new_sum
                c = fast_convolve(f_trial)
                cM = c.max()
                if cM < best_cand_M - 1e-14:
                    best_cand_M = cM
                    best_cand = cv

            f[m] = best_cand
            f = f / f.sum()

        conv = fast_convolve(f)
        M = conv.max()
        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()
            no_improve = 0
            improved = True
        else:
            no_improve += 1

        if no_improve > 5:
            break

    return best_f, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # 1. Uniform
    inits.append(np.ones(n) / n)

    # 2. Edge-concentrated (inverse distance from center)
    dist = np.abs(x - 0.5 * (n - 1))
    edge_w = 1.0 / (dist + 1.0)
    inits.append(edge_w / edge_w.sum())

    # 3. Gaussian center peak
    sigma = 0.3 * n
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / sigma) ** 2)
    inits.append(g / g.sum())

    # 4. Cosine window
    c = np.cos(np.pi * x / n) ** 2 + 0.1
    inits.append(c / c.sum())

    # 5. Alternating (high-low pattern)
    alt = np.ones(n)
    alt[::2] = 2.0
    alt[1::2] = 0.5
    inits.append(alt / alt.sum())

    # 6. Quadratic residues (number-theoretic)
    if n >= 16:
        p = 1
        while p < n and (p * p % 4 != 1):
            p += 2
        # Use modular squares
        qr = np.zeros(n)
        for a_val in range(1, p):
            r = (a_val * a_val) % p
            if r < n:
                qr[r] += 1.0
        if qr.sum() > 0:
            inits.append(qr / qr.sum())

    # 7. Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    # 8. Parent resampled
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        if px.sum() > 1e-10:
            inits.append(px / px.sum())

    # 9. Two-peak (bimodal)
    peak1 = np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2)
    peak2 = np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2)
    bp = peak1 + peak2
    inits.append(bp / bp.sum())

    # 10. Symmetric with dip in center
    sym_dip = np.ones(n)
    center = n // 2
    dip_width = max(1, n // 10)
    sym_dip[max(0, center - dip_width):center + dip_width] = 0.3
    inits.append(sym_dip / sym_dip.sum())

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search using adaptive subgradient with
    peak deflation, simulated annealing, and coordinate descent polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        if a.sum() < 1e-15:
            return float('inf')
        a = a / a.sum()
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.6f}")
    except Exception:
        pass

    # Try parent first
    if parent is not None:
        try_seq(parent, "parent_raw")

    # === Phase 1: Adaptive SG on parent ===
    print("\n=== Phase 1: Adaptive SG Optimization ===")

    if parent is not None:
        n_p = len(parent)
        t1 = min(600, deadline - time.monotonic() - 300)
        if t1 > 60:
            print(f"  Adaptive SG on parent (n={n_p}), budget={t1:.0f}s")
            a_opt, obj = adaptive_sg_optimize(
                parent, time_limit=t1, T_init=0.003, T_final=0.00005,
                top_k=40, mom_init=0.85, mom_final=0.96,
                deflation_freq=8, seed=seed)
            try_seq(a_opt, "sg_parent")

    # === Phase 2: Try multiple lengths and initializations ===
    print("\n=== Phase 2: Multi-length Exploration ===")

    lengths_to_try = [384, 256, 512, 320, 448]
    if parent is not None:
        if len(parent) in lengths_to_try:
            lengths_to_try.remove(len(parent))
        lengths_to_try.insert(0, len(parent))

    remaining = deadline - time.monotonic() - 200
    total_inits_est = sum(len(generate_initializations(l, parent, rng)) for l in lengths_to_try)
    t_per_init = max(15, remaining / max(1, total_inits_est))

    for n in lengths_to_try:
        if time.monotonic() > deadline - 150:
            break
        inits = generate_initializations(n, parent, rng)
        for i, f_init in enumerate(inits):
            if time.monotonic() > deadline - 120:
                break
            t_alloc = min(t_per_init, deadline - time.monotonic() - 100)
            if t_alloc < 10:
                break
            a_opt, _ = adaptive_sg_optimize(
                f_init, time_limit=t_alloc, T_init=0.002, T_final=0.0001,
                top_k=30, mom_init=0.88, mom_final=0.95,
                deflation_freq=12, seed=seed + i * 100 + n)
            try_seq(a_opt, f"n{n}_init{i}")

    print(f"  After Phase 2: best={best_score:.8f}")

    # === Phase 3: Deep optimization on best ===
    print("\n=== Phase 3: Deep Optimization ===")

    if best_seq is not None:
        n_best = len(best_seq)
        remaining = deadline - time.monotonic() - 80

        # Run adaptive SG with different hyperparameters
        configs = [
            {"T_init": 0.001, "T_final": 0.00002, "top_k": 50,
             "mom_init": 0.90, "mom_final": 0.97, "deflation_freq": 5},
            {"T_init": 0.005, "T_final": 0.0001, "top_k": 20,
             "mom_init": 0.80, "mom_final": 0.93, "deflation_freq": 6},
            {"T_init": 0.002, "T_final": 0.00005, "top_k": 60,
             "mom_init": 0.92, "mom_final": 0.98, "deflation_freq": 4},
        ]

        for ci, cfg in enumerate(configs):
            rem = deadline - time.monotonic() - 60
            if rem < 30:
                break
            t_alloc = min(rem / (len(configs) - ci), 150)
            # Slight perturbation for diversity
            if ci == 0:
                f_init = best_seq.copy()
            elif ci == 1:
                noise = rng.standard_normal(n_best) * 0.004
                f_init = project_simplex(best_seq + noise)
            else:
                # Blend with symmetric version
                f_init = 0.6 * best_seq + 0.4 * best_seq[::-1]
                f_init = project_simplex(f_init)

            a_opt, _ = adaptive_sg_optimize(f_init, time_limit=t_alloc, seed=seed + 1000 + ci, **cfg)
            try_seq(a_opt, f"deep_cfg{ci}")

    # === Phase 4: Coordinate descent + Simulated Annealing polish ===
    print("\n=== Phase 4: Polish (Coord Descent + SA) ===")

    if best_seq is not None:
        n_best = len(best_seq)

        # Coordinate descent
        rem = deadline - time.monotonic() - 40
        if rem > 20:
            print(f"  Coordinate descent, budget={min(rem, 80):.0f}s")
            a_opt, _ = coord_descent_polish(best_seq, time_limit=min(rem, 80), seed=seed)
            try_seq(a_opt, "coord_desc")

        # Simulated annealing
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            print(f"  Simulated annealing, budget={min(rem, 60):.0f}s")
            a_opt, _ = sim_anneal_polish(best_seq, time_limit=min(rem, 60), seed=seed)
            try_seq(a_opt, "sa")

        # Another round of adaptive SG on the polished result
        rem = deadline - time.monotonic() - 10
        if rem > 15:
            a_opt, _ = adaptive_sg_optimize(
                best_seq, time_limit=min(rem, 40),
                T_init=0.0005, T_final=0.00001, top_k=80,
                mom_init=0.95, mom_final=0.99, deflation_freq=20,
                seed=seed + 5000)
            try_seq(a_opt, "sg_final")

        # Final SA on the very best
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            a_opt, _ = sim_anneal_polish(best_seq, time_limit=min(rem, 20), seed=seed + 7777)
            try_seq(a_opt, "sa_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  FALLBACK used: score={best_score:.6f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
