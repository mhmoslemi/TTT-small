def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)

    """Vectorized coordinate descent step."""
    n = len(a)
    T = a.sum() - a[i]
    
    if T < 1e-15:
        return a[i]
    
    c = conv.copy()
    b_arr = np.zeros(2*n-1)
    d_arr = np.zeros(2*n-1)
    
    # k range where 0 <= k-i < n
    k_start = i  # k-i >= 0
    k_end = min(2*n-2, i + n - 1)  # k-i <= n-1
    
    if k_end >= k_start:
        js = np.arange(k_start, k_end + 1) - i  # j = k - i, ranges from 0 to min(n-1, k_end-i)
        # Exclude j == i (i.e., k == 2i)
        mask = js != i
        # Apply to indices where mask is True
        ks_valid = js[mask]
        js_valid = js[mask]
        c[ks_valid] -= 2.0 * a[i] * a[js_valid]
        b_arr[ks_valid] = 2.0 * a[js_valid]
        # Handle k == 2i
        if k_start <= 2*i <= k_end:
            c[2*i] -= a[i] * a[i]
            d_arr[2*i] = 1.0
    
    def g(x):
        vals = c + b_arr * x + d_arr * x * x
        return 2.0 * n * vals.max() / ((T + x) ** 2)
    
    # Golden section search
    g0 = g(0.0)
    x_max = max(a[i] * 5.0, T / n * 5.0, 1.0)
    while g(x_max) < g0 and x_max < 1e8:
        x_max *= 2.0

    lo, hi = 0.0, x_max
    x1 = hi - phi * (hi - lo)
    x2 = lo + phi * (hi - lo)
    f1, f2 = g(x1), g(x2)
    
    for _ in range(45):
        if f1 < f2:

            x1 = hi - phi * (hi - lo)
            f1 = g(x1)
        else:
            lo, x1, f1 = x1, x2, f2
            x2 = lo + phi * (hi - lo)
            f2 = g(x2)

But wait, I realize there might be an issue. The golden section search assumes the function is unimodal on the search interval. Is g(x) unimodal?

g(x) = 2n * H(x) / (T+x)^2 where H(x) = max_k (c_k + b_k*x + d_k*x^2).

H(x) is convex (max of convex functions). (T+x)^2 is convex. The ratio of two convex functions is not necessarily unimodal.

However, in practice, H(x) is a piecewise quadratic convex function with "breaks" where the maximizing k changes. Between breaks, H(x) is a single quadratic, and g(x) = 2n * (quadratic) / (T+x)^2. This ratio can have at most 2 critical points (by taking the derivative and setting to zero, we get a cubic equation which has at most 3 real roots, but typically 1 or 2).

So g(x) might not be unimodal in general. Golden section search might fail.
