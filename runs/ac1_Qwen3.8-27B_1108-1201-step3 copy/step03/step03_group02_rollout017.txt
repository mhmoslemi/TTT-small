```python
import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_max_grad(a, t):
    """Compute smooth max of conv(a,a) and its gradient wrt a."""
    n = len(a)
    conv = fast_convolve(a)
    # Softmax
    scaled = t * conv
    scaled -= scaled.max()
    exp_vals = np.exp(scaled)
    p = exp_vals / exp_vals.sum()
    obj = (1.0/t) * (scaled.max() + np.log(exp_vals.sum()))
    # Gradient: grad[j] = sum_m a[m] * p[j+m]
    # Use FFT
    N = 1 << (3*n - 1).bit_length()
    A = np.fft.rfft(a, N)
    P = np.fft.rfft(p, N)
    # cross-correlation: ifft(fft(a) * conj(fft(p)))
    corr = np.fft.irfft(A * np.conj(P), N)
    grad = corr[:n]  # This gives sum_m a[m] * p[m+j] for j=0..n-1
    # Wait, I need to verify the indexing.
    # Actually, ifft(FFT(a) * conj(FFT(p)))[k] = sum_n a[n] * p[n+k] (circular)
    # With sufficient zero-padding, this equals sum_n a[n] * p[n+k] (linear) for k where n+k < len(p)
    # So grad[j] = corr[j] = sum_m a[m] * p[m+j] for j=0..n-1
    # But p has length 2n-1, and m goes from 0 to n-1, so m+j goes from j to j+n-1
    # We need j+n-1 <= 2n-2, i.e., j <= n-1. Good.
    return obj, grad
```

"""

1. Diverse parametric initialization and quick screening
2. Fast smoothed-max projected gradient with FFT-based gradients
3. LP-based refinement for precise local optimization

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_max_and_grad(a, t):
    """Compute smooth max of conv(a,a) and gradient via FFT cross-correlation."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = t * conv
    scaled -= scaled.max()
    exp_vals = np.exp(scaled)
    sum_exp = exp_vals.sum()
    p = exp_vals / sum_exp
    obj = (1.0 / t) * (scaled.max() + np.log(sum_exp))
    # Gradient: grad[j] = sum_m a[m] * p[j+m] for j=0..n-1
    # Use cross-correlation via FFT
    N = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, N)
    P = np.fft.rfft(p, N)
    corr = np.fft.irfft(A * np.conj(P), N)
    grad = corr[:n]
    return obj, grad, conv, p


def smooth_max_grad_topk(a, t, top_k):
    """Smooth max using only top-k peaks for more focused gradient."""
    n = len(a)
    conv = fast_convolve(a)
    if top_k >= len(conv):
        p_full = np.ones(len(conv)) / len(conv)
        p_top = np.ones(len(conv)) / len(conv)
    else:
        idx = np.argpartition(conv, -top_k)[-top_k:]
        vals = conv[idx]
        scaled = t * vals
        scaled -= scaled.max()
        exp_vals = np.exp(scaled)
        p_top = exp_vals / exp_vals.sum()
        # Compute obj from top-k
        obj_topk = (1.0 / t) * (scaled.max() + np.log(exp_vals.sum()))
        # For the full softmax over all (for proper gradient), use:
        # But we approximate: focus gradient on top peaks
        p_full = np.zeros(len(conv))
        p_full[idx] = p_top
    # Gradient
    N = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, N)
    P = np.fft.rfft(p_full, N)
    corr = np.fft.irfft(A * np.conj(P), N)
    grad = corr[:n]
    return obj_topk, grad, conv


def optimize_smooth(a0, time_limit, t_init=30.0, t_max=300.0,
                    lr_init=0.008, momentum=0.92, top_k=None,
                    anneal_rate=1.0003, restart_every=3000,
                    perturb_scale=0.005, verbose=False):
    """Projected subgradient with momentum for smooth max objective."""
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a))
    else:
        a /= s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    v = np.zeros(n)
    t = t_init
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    rng = np.random.default_rng(777)
    it = 0

    while time.monotonic() - t0 < time_limit:
        if top_k is not None:
            _, grad, conv = smooth_max_grad_topk(a, t, top_k)
        else:
            _, grad, conv, _ = smooth_max_and_grad(a, t)

        v = momentum * v - lr * grad
        a_new = project_simplex(a + v)

        conv = fast_convolve(a_new)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            a_new = a

        a = a_new
        t = min(t * anneal_rate, t_max)
        lr *= 0.99995

        if no_improve >= restart_every:
            noise = rng.standard_normal(n) * perturb_scale
            a = project_simplex(a + noise)
            v *= 0.3
            no_improve = 0
            restart_every = int(restart_every * 1.4)
            lr = max(lr_init * 0.3, lr * 0.6)

        it += 1
        if verbose and it % 5000 == 0:
            print(f"    iter={it}, obj={best_obj:.8f}, t={t:.1f}, lr={lr:.6f}")

    return best_a, best_obj


def build_conv_matrix(a):
    """Build matrix A where (A @ b)[k] = conv(a, b)[k] for k=0..2n-2."""
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A


def lp_step(a, M):
    """Solve: max sum(b) s.t. b >= 0, conv(a,b)[k] <= M for all k. Returns normalized b."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.status == 0:
        b = np.maximum(res.x, 0)
        s = b.sum()
        if s > 1e-15:
            return b / s
    return None


def optimize_lp(a0, time_limit, step_init=0.3, step_min=0.05, max_iters=200):
    """LP-based iterative refinement."""
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a))
    else:
        a /= s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    step = step_init
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        M = conv.max()
        b = lp_step(a, M)
        if b is None:
            break
        if np.linalg.norm(b - a) < 1e-10:
            break
        a_new = (1.0 - step) * a + step * b
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new /= s
        conv = fast_convolve(a_new)
        obj = 2.0 * n * conv.max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
            step = min(0.7, step * 1.05)
        else:
            no_improve += 1
            step = max(step_min, step * 0.9)
        a = a_new
        if no_improve > 50:
            step = step_init
            no_improve = 0
        if no_improve > 200:
            break

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent'] = pp / ps

    for freq in [1, 2, 3, 5]:
        for amp in [0.3, 0.6, 0.9]:
            a = 1.0 + amp * np.cos(2.0 * np.pi * freq * x / n)
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 0:
                inits[f'cos_{freq}_{amp:.1f}'] = a / s

    for sigma in [0.15, 0.25, 0.35]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sigma * n)) ** 2)
        s = g.sum()
        if s > 0:
            inits[f'gauss_{sigma:.2f}'] = g / s

    for alpha in [0.3, 0.7, 1.5, 3.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits[f'dirichlet_{alpha:.1f}'] = r

    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = ramp / ramp.sum()
    inits['anti_ramp'] = (ramp[::-1] / ramp.sum()).copy()

    # Triangular
    tri = 1.0 - 2.0 * np.abs(x / n - 0.5)
    tri = np.maximum(tri, 1e-6)
    inits['tri'] = tri / tri.sum()

    # Sinc-like (clipped)
    sinc_x = np.sin(np.pi * (x - 0.5 * n) / (0.5 * n)) / (np.pi * (x - 0.5 * n) / (0.5 * n) + 1e-10)
    sinc_x = np.maximum(sinc_x, 0)
    s = sinc_x.sum()
    if s > 0:
        inits['sinc'] = sinc_x / s

    return inits


def try_resample(parent, target_n, method='interp'):
    """Resample a sequence to a different length."""
    if len(parent) == target_n:
        return parent.copy()
    if method == 'interp':
        t_old = np.linspace(0, 1, len(parent))
        t_new = np.linspace(0, 1, target_n)
        resampled = np.interp(t_new, t_old, parent)
        resampled = np.maximum(resampled, 0)
        s = resampled.sum()
        if s > 1e-10:
            resampled /= s
        else:
            resampled = np.ones(target_n) / target_n
        return resampled
    else:
        # Nearest neighbor
        indices = np.round(np.linspace(0, len(parent) - 1, target_n)).astype(int)
        resampled = parent[indices]
        resampled = np.maximum(resampled, 0)
        s = resampled.sum()
        if s > 1e-10:
            resampled /= s
        else:
            resampled = np.ones(target_n) / target_n
        return resampled

                                       top_k=5)
            try_update(a_opt)

            # Tiny perturbation
            if time.monotonic() < phase5_deadline - 10:
                noise = rng.standard_normal(len(best_sequence)) * 0.001
                a_pert = project_simplex(best_sequence + noise)
                t_alloc2 = min(30.0, phase5_deadline - time.monotonic() - 5)
                if t_alloc2 > 3:
                    a_opt2, _ = optimize_smooth(a_pert, time_limit=t_alloc2,
                                                t_init=300, t_max=600,
                                                lr_init=0.0005, momentum=0.96)
                    try_update(a_opt2)

    print(f"Phase 5 done: best={best_score:.6f}, elapsed={time.monotonic()-t5_start:.0f}s")

    # Final fallback
    if best_sequence is None:
        if parent is not None:
            best_sequence = parent.copy()
        else:
            best_sequence = np.ones(512) / 512

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
