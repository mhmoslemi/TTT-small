    golden = (1 + 5**0.5) / 2
    ab = 1.0 / (golden * (golden - 1))  # ≈ 0.618

            c = d

LP-based block coordinate descent with golden section line search
for minimizing 2n*max(conv(a,a))/(sum(a))^2.

import numpy as np
import time
from scipy.optimize import linprog

def fast_conv(a):
    """Full linear convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def cross_conv(a, b):
    """Cross convolution a * b via FFT."""
    n = max(len(a), len(b))
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def objective_val(a):
    """Compute max(conv(a,a)) assuming sum(a)=1."""
    conv = fast_conv(a)
    return conv.max()

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp(a, M, n):
    """
    Solve: max sum(b) s.t. b >= 0, (a*b)[k] <= M for all k.
    Uses symmetry: (a*b)[k] = (a*b)[2n-2-k] only when a=b.
    For cross-correlation, we need all 2n-1 constraints.
    """
    # Actually for cross-correlation a*b, the symmetry doesn't hold.
    # We need all 2n-1 constraints.
    
    c_obj = -np.ones(n)
    
    # Build sparse constraint matrix
    # A[k, i] = a[k-i] if 0 <= k-i < n
    # This is a Toeplitz-like matrix
    
    # For efficiency, build it as a sparse matrix
    from scipy.sparse import csr_matrix
    
    rows = []
    cols = []
    data = []
    
    for i in range(n):
        k_start = i
        k_end = min(2 * n - 1, i + n)
        for k in range(k_start, k_end):
            rows.append(k)
            cols.append(i)
            data.append(a[k - i])
    
    A_ub = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, 
                     bounds=[(0, None)] * n,
                     method='highs',
                     options={'presolve': True, 'time_limit': 60})
    
    if not result.success or result.x is None:
        return None
    
    b = np.maximum(result.x, 0)
    if b.sum() < 1e-10:
        return None
    
    return b / b.sum()

"""
LP-based block coordinate descent with golden section line search for minimizing
2n*max(conv(a,a))/(sum(a))^2. Uses diverse initialization, multi-length exploration,
and iterative LP direction finding with symmetry-aware optimization.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_conv(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_conv(a, b):
    """Cross convolution a*b via FFT."""
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(a, n):
    """Build sparse constraint matrix for LP: A[k,i] = a[k-i] if 0<=k-i<n."""
    rows = []
    cols = []
    data = []
    for i in range(n):
        k_start = i
        k_end = min(2 * n - 1, i + n)
        a_slice = a[:k_end - k_start]
        rows.extend(range(k_start, k_end))
        cols.extend([i] * (k_end - k_start))
        data.extend(a_slice)
    A = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


def solve_lp_direction(a, M, n, A_ub_cache=None):
    """
    Solve LP: max sum(b) s.t. b >= 0, (a*b)[k] <= M for all k=0..2n-2.
    Returns b normalized to sum=1, or None.
    """
    if A_ub_cache is not None and np.allclose(A_ub_cache.data, build_lp_matrix(a, n).data):
        A_ub = A_ub_cache
    else:
        A_ub = build_lp_matrix(a, n)

    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': 30})
    except Exception:
        return None, None

    if not result.success or result.x is None:
        return None, A_ub

    b = np.maximum(result.x, 0.0)
    b_sum = b.sum()
    if b_sum < 1e-10:
        return None, A_ub

    return b / b_sum, A_ub


def golden_section_line_search(a, b, n_max_evals=18):
    """
    Find t in [0,1] minimizing max(conv((1-t)*a + t*b)).
    Uses golden section search since the objective is convex in t.
    """
    phi = (1.0 + 5.0**0.5) / 2.0
    inv_phi2 = 1.0 / (phi * phi)
    inv_phi = 1.0 / phi

    lo = 0.0
    hi = 1.0
    c = lo + inv_phi2 * (hi - lo)
    d = lo + inv_phi * (hi - lo)

    c_val = (1.0 - c) * a + c * b
    d_val = (1.0 - d) * a + d * b
    f_c = fast_conv(c_val).max()
    f_d = fast_conv(d_val).max()

    evals = 2
    for _ in range(n_max_evals - 2):
        if f_c < f_d:
            hi = d
            d = c
            f_d = f_c
            c = lo + inv_phi2 * (hi - lo)
            c_val = (1.0 - c) * a + c * b
            f_c = fast_conv(c_val).max()
            evals += 1
        else:
            lo = c
            c = d
            f_c = f_d
            d = lo + inv_phi * (hi - lo)
            d_val = (1.0 - d) * a + d * b
            f_d = fast_conv(d_val).max()
            evals += 1

    t = (lo + hi) / 2.0
    return t


def smoothed_objective(a, beta):
    """Log-sum-exp approximation of max(conv(a,a))."""
    conv = fast_conv(a)
    scaled = beta * (conv - conv.max())
    p = np.exp(scaled)
    return conv.max() + np.log(p.sum()) / beta


def smoothed_gradient(a, beta):
    """Gradient of logsumexp(beta*conv(a,a)/beta) w.r.t. a."""
    n = len(a)
    conv = fast_conv(a)
    scaled = beta * (conv - conv.max())
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p /= p_sum

    w = conv * p
    fft_size = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w, fft_size)
    cross = np.fft.irfft(A * np.conj(W), fft_size)
    return 2.0 * cross[:n]


def gradient_descent(a0, time_limit, beta_init=50.0, beta_end=300.0,
                     lr_init=0.01, momentum=0.9, seed=0):
    """Nesterov accelerated projected gradient descent on smoothed objective."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    conv = fast_conv(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    iters = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-6))
        beta = beta_init + (beta_end - beta_init) * frac

        y = a + momentum * v
        conv_y = fast_conv(y)
        grad = smoothed_gradient(y, beta)

        v = momentum * v - lr * grad
        a_new = project_simplex(a + v)

        conv_new = fast_conv(a_new)
        cur_obj = conv_new.max()
        iters += 1

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        if no_improve >= 200:
            noise = rng.standard_normal(n) * 0.003
            a = project_simplex(a + noise)
            v *= 0.2
            no_improve = 0
            lr *= 0.5

        lr = max(0.0005, lr_init * (1.0 - 0.5 * frac))

    return best_a, best_obj


def lp_optimize(a0, time_limit, seed=42):
    """
    LP-based block coordinate descent with golden section line search.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:

    n = len(a)

    conv = fast_conv(a)

    t0 = time.monotonic()
    A_ub_cache = None
    iters = 0

    while time.monotonic() - t0 < time_limit:
        conv_cur = fast_conv(cur_a)
        M = conv_cur.max()

        # Try LP with progressively relaxed M
        b_lp = None
        for factor in [1.0, 1.0001, 1.001, 1.005, 1.01]:
            M_try = M * factor
            b_lp, A_ub_cache = solve_lp_direction(cur_a, M_try, n, A_ub_cache)
            if b_lp is not None:
                break

        if b_lp is None:
            # Perturb and continue

            cur_a = project_simplex(cur_a + noise)
            iters += 1
            if iters > 50:
                break
            continue

        # Check if LP direction is actually different from current
        if np.allclose(b_lp, cur_a, atol=1e-6):
            # Stuck - perturb
            noise = rng.standard_normal(n) * 0.005

            iters += 1
            continue

        t_opt = golden_section_line_search(cur_a, b_lp, n_max_evals=15)

        new_a = (1.0 - t_opt) * cur_a + t_opt * b_lp
        new_a = np.maximum(new_a, 0.0)
        ns = new_a.sum()
        if ns < 1e-10:
            cur_a = project_simplex(cur_a + rng.standard_normal(n) * 0.01)
            iters += 1
            continue
        new_a = new_a / ns

        new_obj = fast_conv(new_a).max()
        iters += 1

        if new_obj < best_obj - 1e-12:
            best_obj = new_obj
            best_a = new_a.copy()
            cur_a = new_a.copy()
        elif new_obj < M * 0.9999:
            # Accept even if not new best (progress)
            cur_a = new_a.copy()
            if new_obj < best_obj:
                best_obj = new_obj
                best_a = new_a.copy()
        else:
            # Try smaller step
            for t_try in [0.05, 0.1, 0.15, 0.2]:
                trial = (1.0 - t_try) * cur_a + t_try * b_lp
                trial = np.maximum(trial, 0.0)
                ts = trial.sum()
                if ts < 1e-10:
                    continue
                trial = trial / ts
                trial_obj = fast_conv(trial).max()
                if trial_obj < M * 0.9999:
                    cur_a = trial
                    if trial_obj < best_obj:
                        best_obj = trial_obj
                        best_a = trial.copy()
                    break
            else:
                # No improvement - perturb
                if iters % 20 == 0:
                    noise = rng.standard_normal(n) * 0.003
                    cur_a = project_simplex(cur_a + noise)
                    A_ub_cache = None

    return best_a, best_obj


def peak_shaving(a0, time_limit, seed=0):
    """Targeted peak reduction via coordinate adjustments."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:

    n = len(a)

        max_val = conv_cur.max()
        peak_k = int(np.argmax(conv_cur))

        # Find top contributing pairs to the peak
        i_start = max(0, peak_k - n + 1)
        i_end = min(n - 1, peak_k)

        contributions = np.array([a[i] * a[peak_k - i] for i in range(i_start, i_end + 1)])
        if len(contributions) == 0:
            break

        # Try reducing the largest contributor
        sorted_idx = np.argsort(contributions)[::-1]
        improved = False

        for si in range(min(3, len(sorted_idx))):
            i_max = i_start + sorted_idx[si]
            a_val = a[i_max]
            if a_val < 1e-8:
                continue

            for _ in range(5):
                j = int(rng.integers(0, n))
                if j == i_max or abs(j - i_max) <= 1:
                    continue

                for frac in [0.03, 0.05, 0.08, 0.12]:
                    amt = a_val * frac
                    a_try = a.copy()
                    a_try[i_max] -= amt
                    a_try[j] += amt
                    obj_try = fast_conv(a_try).max()
                    if obj_try < best_obj - 1e-12:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Random pair adjustment
            i1 = int(rng.integers(0, n))
            i2 = int(rng.integers(0, n))
            if i1 != i2 and a[i1] > 1e-8:
                for frac in [0.02, 0.04, 0.06]:
                    amt = a[i1] * frac
                    a_try = a.copy()
                    a_try[i1] -= amt
                    a_try[i2] += amt
                    obj_try = fast_conv(a_try).max()
                    if obj_try < best_obj - 1e-12:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try.copy()
                        improved = True
                        break

            if not improved:
                # Try reducing peak by spreading mass from all contributors
                for i in range(i_start, i_end + 1):
                    if a[i] > 1e-6:
                        a[i] *= 0.99
                s2 = a.sum()
                if s2 > 0:
                    a = a / s2
                    obj_check = fast_conv(a).max()
                    if obj_check < best_obj:
                        best_obj = obj_check
                        best_a = a.copy()
                    else:
                        a = best_a.copy()

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n (all on simplex)."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Raised cosine
    if n > 2:
        rc = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n - 1)))
        rc = np.maximum(rc, 1e-8)
        rc /= rc.sum()
        inits['raised_cos'] = rc

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.3 * n)) ** 2)
    g /= g.sum()
    inits['gauss_center'] = g

    # Gaussian wide
    gw = np.exp(-0.5 * ((x - 0.5 * n) / (0.45 * n)) ** 2)
    gw /= gw.sum()
    inits['gauss_wide'] = gw

    # Bimodal
    bi = np.zeros(n)
    edge = max(2, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Tapered (linear fade at edges)
    taper = np.ones(n)
    fade = max(1, n // 8)
    taper[:fade] = np.linspace(0.1, 1.0, fade)
    taper[-fade:] = np.linspace(1.0, 0.1, fade)
    taper /= taper.sum()
    inits['tapered'] = taper

    # Random Dirichlet
    inits['rand1'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['rand2'] = rng.dirichlet(np.ones(n) * 1.0)
    inits['rand3'] = rng.dirichlet(np.ones(n) * 2.0)

    # Parent-based
    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            pp = np.maximum(parent, 0.0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        elif len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), np.maximum(parent, 0.0))
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_ds'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), np.maximum(parent, 0.0))
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps

    return inits

            if np.isfinite(score_fg) and score_fg < best_score:
                best_score = score_fg
                best_sequence = a_fg.copy()
                print(f"  FineGrad: {best_score:.6f}", flush=True)

        # Another LP pass on the polished result
        remaining = deadline - time.monotonic() - 10.0
        if remaining > 30:
            a_lp2, _ = lp_optimize(best_sequence, remaining * 0.5, seed=555)
            score_lp2 = float(evaluate_sequence(a_lp2.tolist()))
            if np.isfinite(score_lp2) and score_lp2 < best_score:
                best_score = score_lp2
                best_sequence = a_lp2.copy()
                print(f"  LP2: {best_score:.6f}", flush=True)

        # Final peak shave
        remaining = deadline - time.monotonic() - 5.0
        if remaining > 5:
            a_fs, _ = peak_shaving(best_sequence, remaining, seed=999)
            score_fs = float(evaluate_sequence(a_fs.tolist()))
            if np.isfinite(score_fs) and score_fs < best_score:
                best_score = score_fs
                best_sequence = a_fs.copy()
                print(f"  FinalShave: {best_score:.6f}", flush=True)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_sequence)})", flush=True)
    return [float(x) for x in best_sequence]
