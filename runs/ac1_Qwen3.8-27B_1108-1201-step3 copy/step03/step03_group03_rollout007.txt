"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using pressure-based iterative reduction,
adaptive multiplicative updates, multi-start search, and coordinate descent polishing.
Uses a novel approach that avoids expensive LP solves by directly computing
pressure from active peak constraints.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_pressure(a, conv, top_fraction=0.15, temperature=1.0):
    """
    Compute the pressure vector from peak convolution lags.
    
    For each lag k with high conv[k], the pressure on index i is
    proportional to a[k-i] (the gradient contribution).
    
    Returns a pressure vector of the same length as a.
    """
    n = len(a)
    M = conv.max()
    
    # Select lags above a threshold (top_fraction of max)
    threshold = M * (1.0 - top_fraction)
    peak_mask = conv >= threshold
    peak_idxs = np.where(peak_mask)[0]
    
    if len(peak_idxs) == 0:
        return np.zeros(n)
    
    # Weight lags by how close they are to the max
    peak_vals = conv[peak_idxs]
    weights = peak_vals / (temperature + peak_vals.max() * 0.01)
    weights = weights - weights.min()
    wsum = weights.sum()
    if wsum < 1e-15:
        weights = np.ones(len(peak_idxs)) / len(peak_idxs)
    else:
        weights = weights / wsum
    
    # Compute pressure
    pressure = np.zeros(n)
    for idx, k in enumerate(peak_idxs):
        w = weights[idx]
        if w < 1e-12:
            continue
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            pressure[j] += w * 2.0 * a[k - j]
    
    return pressure


def optimize_pressure(a0, time_limit, lr_init=0.01, decay=0.9995,
                      top_fraction=0.15, temperature=1.0,
                      restart_interval=100, noise_scale=0.005):
    """
    Pressure-based iterative optimization.
    
    Uses multiplicative updates driven by pressure from peak lags.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    rng = np.random.default_rng(12345)
    t0 = time.monotonic()
    lr = lr_init
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        pressure = compute_pressure(a, conv, top_fraction, temperature)
        
        # Multiplicative update: reduce where pressure is high
        # a_new = a * exp(-lr * pressure)
        # Use log for numerical stability
        log_a = np.log(np.maximum(a, 1e-30))
        log_a -= lr * pressure
        # Renormalize in log space
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-30:
            a_new = a.copy()
        else:
            a_new = a_new / a_new_sum
        
        a = a_new
        it += 1
        
        # Periodic noise injection
        if it % restart_interval == 0 and no_improve > restart_interval // 2:
            noise = rng.standard_normal(n) * noise_scale
            a = project_simplex(a + noise)
            no_improve = 0
        
        cur_obj = 2.0 * n * fast_convolve(a).max()  # need to recompute
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 500:
            # Strong restart
            noise = rng.standard_normal(n) * 0.02
            a = project_simplex(a + noise)
            no_improve = 0
            lr = min(lr * 2.0, 0.05)
        
        lr = max(0.001, lr * decay)
    
    return best_a, best_obj

def coordinate_descent_polish(a_in, max_rounds=100, step_frac=0.1):
    """Coordinate descent: try adjusting pairs of coordinates."""
    a = np.maximum(a_in.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    
    for rnd in range(max_rounds):
        improved = False
        # Try moving mass from high-pressure to low-pressure coordinates
        for i in range(0, n, max(1, n // 200)):  # sparse sampling for speed
            if a[i] < 1e-8:
                continue
            # Find best j to move mass to
            # Try a few candidate j's
            candidates = [0, n // 2, n - 1]
            # Add some random candidates
            candidates += list(np.random.default_rng(rnd * 1000 + i).choice(n, size=min(10, n), replace=False))
            
            for j in candidates:
                if j == i:
                    continue
                for frac in [0.02, 0.05, 0.1, 0.2]:
                    delta = a[i] * frac
                    if delta < 1e-10:
                        continue
                    a_test = a.copy()
                    a_test[i] -= delta
                    a_test[j] += delta
                    if a_test[i] < 0:
                        continue
                    conv_test = fast_convolve(a_test)
                    obj_test = 2.0 * n * conv_test.max()
                    if obj_test < best_obj - 1e-13:
                        a = a_test
                        best_obj = obj_test
                        conv = conv_test
                        improved = True
        
        if not improved:
            break
    
    return a

    # Edge-concentrated
    edge = np.zeros(n)
    e = max(1, n // 4)
    edge[:e] = 1.0
    edge[-e:] = 1.0
    inits['edge'] = edge / edge.sum()
    
    # Sine-modulated
    sine = 1.0 + 0.5 * np.sin(2 * np.pi * x / n)
    inits['sine'] = sine / sine.sum()
    
    # Multi-scale
    multi = 1.0
    for scale in [n//8, n//4, n//2]:
        multi += np.exp(-0.5 * (x - n//2)**2 / (2*(scale/3)**2))
    inits['multi'] = multi / multi.sum()
    
    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            p = np.maximum(parent, 0)
            ps = p.sum()
            if ps > 1e-10:
                inits['parent'] = p / ps
        elif len(parent) > n:
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)),

            ps = p.sum()
            if ps > 1e-10:
                inits['parent_ds'] = p / ps
        else:
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)),
                          np.maximum(parent, 0))
            ps = p.sum()
            if ps > 1e-10:
                inits['parent_up'] = p / ps
    
    return inits

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """
    Pressure-based multiplicative optimization with adaptive threshold.
    Fast iterations using direct gradient from active peak constraints.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    lr = lr_init
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        threshold = thresh_start + (thresh_end - thresh_start) * progress
        
        conv = fast_convolve(a)
        pressure = compute_pressure(a, conv, n, threshold_frac=threshold)
        
        if pressure.max() < 1e-15:
            break
        
        # Normalize and apply multiplicative update
        p_max = pressure.max()
        log_change = -np.clip(lr * pressure / p_max, -0.3, 0.3)
        a = a * np.exp(log_change)
        a = np.maximum(a, 1e-18)
        a = a / a.sum()
        
        it += 1
        
        if it % check_interval == 0:
            conv = fast_convolve(a)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj - 1e-13:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            if no_improve >= restart_count:
                noise_scale = 0.005 + 0.01 * (no_improve / restart_count)
                a = project_simplex(a + rng.standard_normal(n) * noise_scale)
                no_improve = 0
                lr = min(lr * 1.3, 0.04)
        
        lr = max(lr_min, lr * decay)
    
    return best_a, best_obj


def optimize_vol(a0, time_limit, seed=0):
    """
    Volume/multiplicative-weights algorithm: maintains a log-weight vector,
    updates based on the most violated constraint each iteration.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    log_w = np.zeros(n)
    lr = 0.025
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()
        
        # Get top 8 lags for robustness
        top_indices = np.argsort(conv)[-8:]
        
        grad = np.zeros(n)
        for k_raw in top_indices:
            k = int(k_raw)
            w = conv[k] / M
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                ki = k - j
                if ki == j:
                    grad[j] += w * a[j]
                else:
                    grad[j] += w * 2.0 * a[ki]
        
        log_w -= lr * grad
        log_w -= log_w.max()
        
        a_new = np.exp(log_w)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-30:
            a_new = a.copy()
            log_w = np.log(np.maximum(a, 1e-30))
            log_w -= log_w.max()
        else:
            a = a_new / a_new_sum
            log_w = np.log(np.maximum(a, 1e-30))
            log_w -= log_w.max()
        
        it += 1
        
        if it % 15 == 0:
            conv = fast_convolve(a)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj - 1e-13:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            if no_improve > 80:
                a = project_simplex(a + rng.standard_normal(n) * 0.008)
                log_w = np.log(np.maximum(a, 1e-30))
                log_w -= log_w.max()
                no_improve = 0
                lr = min(lr * 1.2, 0.06)
            
            if it % 300 == 0:
                lr = max(0.003, lr * 0.85)
    
    return best_a, best_obj


def coordinate_descent_polish(a_in, max_rounds=60, n_samples=200):
    """Coordinate descent: try moving mass between coordinate pairs."""
    a = np.maximum(a_in.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    
    rng = np.random.default_rng(777)
    
    for rnd in range(max_rounds):
        improved = False
        step = max(1, n // n_samples)
        for i in range(0, n, step):
            if a[i] < 1e-9:
                continue
            candidates = list(set([0, n // 4, n // 2, 3 * n // 4, n - 1] +
                                  list(rng.choice(n, size=min(15, n), replace=False))))
            
            for j in candidates:
                if j == i or j >= n:
                    continue
                for frac in [0.03, 0.08, 0.15]:
                    delta = a[i] * frac
                    if delta < 1e-12:
                        continue
                    a_test = a.copy()
                    a_test[i] -= delta
                    a_test[j] += delta
                    if a_test[i] < 0:
                        continue
                    conv_test = fast_convolve(a_test)
                    obj_test = 2.0 * n * conv_test.max()
                    if obj_test < best_obj - 1e-14:
                        a = a_test
                        best_obj = obj_test
                        conv = conv_test
                        improved = True
        
        if not improved:
            break
    
    return a


def gen_inits(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    tri = np.minimum(x + 1.0, n - x)
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()
    
    cos_b = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    cos_b = np.maximum(cos_b, 0.01)
    inits['cos'] = cos_b / cos_b.sum()
    
    bim = np.exp(-0.5 * ((x - 0.2 * n) / (0.08 * n)) ** 2) + \
          np.exp(-0.5 * ((x - 0.8 * n) / (0.08 * n)) ** 2)
    inits['bimodal'] = bim / bim.sum()
    
    inits['rand01'] = rng.dirichlet(np.ones(n) * 0.05)
    inits['rand1'] = rng.dirichlet(np.ones(n) * 1.0)
    inits['rand10'] = rng.dirichlet(np.ones(n) * 10.0)
    
    edge = np.zeros(n)
    e = max(1, n // 3)
    edge[:e] = 1.0
    edge[-e:] = 1.0
    inits['edge'] = edge / edge.sum()
    
    sine = 1.0 + 0.4 * np.sin(4 * np.pi * x / n)
    sine = np.maximum(sine, 0.01)
    inits['sine'] = sine / sine.sum()
    
    pw = (x + 1) ** 0.3
    inits['power'] = pw / pw.sum()
    
    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            p = np.maximum(parent, 0)
            ps = p.sum()
            if ps > 1e-10:
                inits['parent'] = p / ps
        else:
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)),
                          np.maximum(parent, 0))
            ps = p.sum()
            if ps > 1e-10:
                inits['parent_interp'] = p / ps
    
    return inits


def coarse_to_fine(coarse_a, target_n, time_limit, seed=0):
    """Upsample a coarse solution and refine at target resolution."""
    m = len(coarse_a)
    if m == target_n:
        return optimize_pressure(coarse_a, time_limit, seed=seed)
    
    # Interpolate to target_n
    if m < target_n:
        x_coarse = np.linspace(0, 1, m)
        x_fine = np.linspace(0, 1, target_n)
        fine = np.interp(x_fine, x_coarse, coarse_a)
    else:
        x_coarse = np.linspace(0, 1, m)
        x_fine = np.linspace(0, 1, target_n)
        fine = np.interp(x_fine, x_coarse, coarse_a)
    
    fine = np.maximum(fine, 0)
    fs = fine.sum()
    if fs < 1e-15:
        fine = np.ones(target_n) / target_n
    else:
        fine = fine / fs
    
    return optimize_pressure(fine, time_limit, seed=seed)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search: (1) broad pressure-based scan over lengths/inits,
    (2) evolutionary population with volume algorithm, (3) multi-scale
    refinement, (4) coordinate descent polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 35.0)
    
    best_sequence = None
    best_score = float('inf')
    
    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            par_score = float(evaluate_sequence(p.tolist()))
            print(f"Parent: n={len(p)}, score={par_score:.6f}")
    except Exception:
        pass
    
    # === Phase 1: Broad scan ===
    print("=== Phase 1: Broad Pressure Scan ===")
    lengths = [256, 384, 512, 640]
    
    results = []
    phase1_budget = min(550.0, (deadline - time.monotonic()) * 0.35)
    phase1_end = time.monotonic() + phase1_budget
    
    for n in lengths:
        if time.monotonic() > phase1_end - 25:
            break
        inits = gen_inits(n, parent, rng)
        prio = ['parent', 'parent_interp', 'uniform', 'triangle', 'bimodal',
                'cos', 'rand01', 'rand1', 'sine', 'edge', 'power', 'rand10']
        names = [nm for nm in prio if nm in inits][:5]
        
        t_per = max(8.0, phase1_budget / (len(lengths) * 5))
        
        for name in names:
            if time.monotonic() > phase1_end - 8:
                break
            t_alloc = min(t_per, phase1_end - time.monotonic() - 4)
            if t_alloc < 4:
                break
            
            a0 = inits[name]
            a_opt, obj_opt = optimize_pressure(a0, t_alloc, lr_init=0.015,
                                               seed=hash((n, name)) % 10000)
            # Quick second pass with lower lr
            t2 = min(4.0, t_alloc * 0.2)
            if t2 > 2:
                a2, obj2 = optimize_pressure(a_opt, t2, lr_init=0.004,
                                            lr_min=0.0001, decay=0.9998,
                                            thresh_start=0.85, thresh_end=0.97,
                                            seed=hash((n, name, 2)) % 10000)
                if obj2 < obj_opt:
                    a_opt, obj_opt = a2, obj2
            
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n} {name}: {score:.6f}")
    
    results.sort(key=lambda x: x[0])
    print(f"Phase 1 best: {best_score:.6f}")
    
    # === Phase 2: Evolutionary population with volume algorithm ===
    print("=== Phase 2: Evolutionary Volume Optimization ===")
    
    # Build initial population from top results
    population = []
    seen = set()
    for score, n, name, a_opt in results[:8]:
        key = (n, name)
        if key not in seen and n == len(population[0]) if population else True:
            population.append(a_opt.copy())
            seen.add(key)
        if len(population) >= 4:
            break
    
    if not population:
        n_pop = 512
        population = [np.ones(n_pop) / n_pop]
    
    n_pop_len = len(population[0])
    
    # Add diverse random members
    for i in range(3):
        r = rng.dirichlet(np.ones(n_pop_len) * rng.uniform(0.1, 5.0))
        population.append(r)
    
    # Cap population size
    population = population[:8]
    
    phase2_budget = min(600.0, (deadline - time.monotonic()) * 0.4)
    phase2_end = time.monotonic() + phase2_budget
    pop_size = len(population)
    
    # Evolutionary loop
    gen = 0
    while time.monotonic() < phase2_end:
        gen += 1
        t_per_member = max(3.0, (phase2_end - time.monotonic()) / (pop_size * 2))
        if t_per_member < 2:
            break
        
        # Optimize each member
        for i in range(len(population)):
            if time.monotonic() > phase2_end - 5:
                break
            t_alloc = min(t_per_member, phase2_end - time.monotonic() - 3)
            if t_alloc < 2:
                break
            
            # Alternate between pressure and volume methods
            if gen % 2 == 0 or i % 2 == 0:
                a_opt, obj = optimize_pressure(population[i], t_alloc,
                                              lr_init=0.008, lr_min=0.0001,
                                              decay=0.9994,
                                              thresh_start=0.75, thresh_end=0.95,
                                              seed=hash((gen, i)) % 10000)
            else:
                a_opt, obj = optimize_vol(population[i], t_alloc,
                                         seed=hash((gen, i)) % 10000)
            
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                population[i] = a_opt.copy()
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Gen {gen} member {i}: {score:.6f}")
        
        # Cross-over: breed new members from best ones
        if gen % 3 == 0 and len(population) > 4:
            # Sort by quality
            scored = []
            for i, a in enumerate(population):
                obj = 2.0 * len(a) * fast_convolve(a).max()
                scored.append((obj, i, a.copy()))
            scored.sort(key=lambda x: x[0])
            
            # Replace worst members with crosses of best
            for i in range(len(scored) - 1, max(2, len(scored) - 3), -1):
                parent1 = scored[0][2]
                parent2 = scored[1][2] if len(scored) > 1 else scored[0][2]
                alpha = rng.uniform(0.3, 0.7)
                child = alpha * parent1 + (1 - alpha) * parent2
                # Add noise
                child = child + rng.standard_normal(len(child)) * 0.01
                population[i] = project_simplex(np.maximum(child, 0))
    
    print(f"Phase 2 best: {best_score:.6f}")
    
    # === Phase 3: Multi-scale refinement ===
    print("=== Phase 3: Multi-scale Refinement ===")
    
    if best_sequence is not None:
        # Try different target lengths
        target_lengths = list(set([256, 384, 512, 640, 768]))
        target_lengths = [tl for tl in target_lengths if 128 <= tl <= 1024]
        
        phase3_budget = min(350.0, (deadline - time.monotonic()) * 0.4)
        phase3_end = time.monotonic() + phase3_budget
        
        for tl in target_lengths:
            if time.monotonic() > phase3_end - 10:
                break
            t_alloc = min(50.0, phase3_end - time.monotonic() - 8)
            if t_alloc < 8:
                break
            
            # Downsample best to get coarse, then upsample to target
            if len(best_sequence) != tl:
                coarse = np.interp(np.linspace(0, 1, min(tl, len(best_sequence) // 2)),
                                   np.linspace(0, 1, len(best_sequence)), best_sequence)
            else:
                coarse = best_sequence.copy()
            
            a_opt, obj_opt = coarse_to_fine(coarse, tl, t_alloc,
                                           seed=hash(tl) % 10000)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Multi-scale n={tl}: {score:.6f}")
        
        # Fine optimization at best found length
        n_best = len(best_sequence)
        remaining = phase3_end - time.monotonic()
        
        if remaining > 10:
            for lr_try in [0.003, 0.001, 0.0005]:
                if time.monotonic() > phase3_end - 5:
                    break
                t_c = min(remaining / 3.0, 40.0)
                if t_c < 3:
                    break
                a_opt, obj = optimize_pressure(best_sequence, t_c,
                                              lr_init=lr_try, lr_min=0.00005,
                                              decay=0.9999,
                                              thresh_start=0.88, thresh_end=0.98,
                                              seed=hash(lr_try) % 10000)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Fine lr={lr_try}: {score:.6f}")
    
    # === Phase 4: Coordinate descent + final polish ===
    print("=== Phase 4: Polish ===")
    
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 20.0
        
        # Coordinate descent
        if remaining > 15:
            t_cd = min(80.0, remaining * 0.4)
            a_cd = coordinate_descent_polish(best_sequence, max_rounds=min(80, int(t_cd * 2)))
            score_cd = float(evaluate_sequence(a_cd.tolist()))
            if np.isfinite(score_cd) and score_cd < best_score:
                best_score = score_cd
                best_sequence = a_cd.copy()
                print(f"  CD: {score_cd:.6f}")
        
        # Final volume algorithm pass
        remaining = deadline - time.monotonic() - 15.0
        if remaining > 8:
            t_vol = min(remaining * 0.5, 60.0)
            a_vol, _ = optimize_vol(best_sequence, t_vol, seed=99999)
            score_vol = float(evaluate_sequence(a_vol.tolist()))
            if np.isfinite(score_vol) and score_vol < best_score:
                best_score = score_vol
                best_sequence = a_vol.copy()
                print(f"  Final vol: {score_vol:.6f}")
        
        # Perturbation search
        remaining = deadline - time.monotonic() - 12.0
        if remaining > 10:
            n_b = len(best_sequence)
            for _ in range(4):
                if time.monotonic() > deadline - 10:
                    break
                ps = rng.uniform(0.002, 0.008)
                a_p = project_simplex(best_sequence + rng.standard_normal(n_b) * ps)
                t_p = min(8.0, remaining / 2)
                if t_p < 3:
                    break
                a_opt, _ = optimize_pressure(a_p, t_p, lr_init=0.005,
                                            seed=rng.integers(100000))
                score_p = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score_p) and score_p < best_score:
                    best_score = score_p
                    best_sequence = a_opt.copy()
                    print(f"  Perturb: {score_p:.6f}")
    
    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
    
    # Final validation
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if not np.isfinite(final_score):
        best_sequence = np.ones(256) / 256
        final_score = float(evaluate_sequence(best_sequence.tolist()))
    
    print(f"\nFINAL: score={final_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
