def fast_convolve(a):

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)

def upsample(a, new_n):
    """Upsample a sequence to a new length using linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    else:
        a_new = np.ones(new_n) / new_n
    return a_new

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-based accelerated projected
gradient descent with Nesterov momentum, mirror descent (multiplicative weights),
multi-scale optimization, and adaptive peak-weighting via softmax.
Key innovation: O(n log n) gradient computation via cross-correlation FFT.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient_fft(weights, a):
    """
    Compute gradient of sum_k weights[k]*b_k where b=conv(a,a).
    grad[m] = 2 * sum_l weights[m+l] * a[l]
    Uses FFT cross-correlation for O(n log n) speed.
    """
    n = len(a)
    N = 1 << (3 * n - 3).bit_length()
    if N < 3 * n - 2:
        N *= 2
    W = np.fft.rfft(weights, N)
    A = np.fft.rfft(a, N)
    # Cross-correlation: ifft(FFT(w) * conj(FFT(a)))
    corr = np.fft.irfft(W * np.conj(A), N)[:n]
    return 2.0 * corr


def softmax_weights(b, top_k=50, temperature=0.01):
    """Compute softmax weights over top-k peaks of b."""
    n = len(b)
    if top_k >= n:
        top_k = n
    if top_k == n:
        idx = np.arange(n)
        vals = b
    else:
        idx = np.argpartition(b, -top_k)[-top_k:]
        vals = b[idx]
    vmax = vals.max()
    if vmax < 1e-15:
        return np.zeros(n)
    scaled = (vals - vmax) / (temperature * vmax + 1e-30)
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        full_w = np.zeros(n)
        full_w[np.argmax(b)] = 1.0
        return full_w
    w /= w_sum
    full_w = np.zeros(n)
    full_w[idx] = w
    return full_w


def objective_val(a):
    """Compute 2*n*max(conv(a,a)) for normalized a."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def pgd_nesterov(a0, time_limit, lr_init=0.06, momentum=0.9, top_k=50,
                 temperature=0.01, use_mirror=False, verbose_every=500):
    """
    Nesterov-accelerated projected gradient on simplex.
    Minimizes 2n*max(conv(a,a)) with sum(a)=1, a>=0.
    Uses FFT-based O(n log n) gradient.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    y = a.copy()

    b = fast_convolve(a)
    best_obj = 2.0 * n * b.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    iteration = 0
    mu = 1.0 / (1.0 - momentum)  # for Nesterov

    while time.monotonic() - t0 < time_limit:
        # Compute gradient at extrapolated point y
        b_y = fast_convolve(y)
        pmax = b_y.max()
        if pmax < 1e-15:
            break

        w = softmax_weights(b_y, top_k=top_k, temperature=temperature)
        grad = compute_gradient_fft(w, y)

        # Nesterov update
        v = momentum * v
        a_new = y - mu * lr * grad
        a_new = project_simplex(a_new)

        # Evaluate
        b_new = fast_convolve(a_new)
        cur_obj = 2.0 * n * b_new.max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Update Nesterov state
        beta_t = momentum
        y = a_new + (beta_t - 1.0) / max(beta_t, 1e-8) * (a_new - a)

        a = a_new
        b = b_new
        iteration += 1

        # Adaptive step size
        if no_improve > 40:
            lr = max(0.001, lr * 0.75)
            v *= 0.5
            y = a.copy()
            no_improve = 0

        # Periodic restart
        if no_improve > 300:
            rng_local = np.random.default_rng(hash(iteration % 100000))
            noise = rng_local.standard_normal(n) * 0.003
            a = project_simplex(a + noise)
            v = np.zeros(n)
            y = a.copy()
            b = fast_convolve(a)
            no_improve = 0
            lr = max(0.003, lr * 0.8)

        if verbose_every > 0 and iteration % verbose_every == 0:
            elapsed = time.monotonic() - t0
            print(f"      PGD it={iteration}, obj={best_obj:.8f}, lr={lr:.5f}, t={elapsed:.1f}s")

    return best_a, best_obj


def mirror_descent(a0, time_limit, lr_init=0.08, top_k=50, temperature=0.01,
                   verbose_every=1000):
    """
    Mirror descent (multiplicative weights) on simplex.
    a_j <- a_j * exp(-lr * grad_j) / Z
    Naturally handles non-negativity.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    b = fast_convolve(a)
    best_obj = 2.0 * n * b.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    iteration = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        pmax = b.max()
        if pmax < 1e-15:
            break

        w = softmax_weights(b, top_k=top_k, temperature=temperature)
        grad = compute_gradient_fft(w, a)

        # Multiplicative update in log-space
        log_a = np.log(np.maximum(a, 1e-30))
        log_a_new = log_a - lr * grad
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-30:
            a_new = np.ones(n) / n
        else:
            a_new /= a_new_sum

        b_new = fast_convolve(a_new)
        cur_obj = 2.0 * n * b_new.max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 30:
            lr = max(0.001, lr * 0.7)
            no_improve = 0

        a = a_new
        b = b_new
        iteration += 1

        if no_improve > 250:
            rng_local = np.random.default_rng(hash(iteration % 100000))
            noise = rng_local.standard_normal(n) * 0.005
            a = project_simplex(a + noise)
            b = fast_convolve(a)
            no_improve = 0
            lr = max(0.005, lr * 0.8)

        if verbose_every > 0 and iteration % verbose_every == 0:
            elapsed = time.monotonic() - t0
            print(f"      MD it={iteration}, obj={best_obj:.8f}, lr={lr:.5f}, t={elapsed:.1f}s")

    return best_a, best_obj

def upsample_interp(a, new_n):
    """Upsample using linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.full(new_n, a[0])
    x_old = np.linspace(0.0, 1.0, n)
    x_new = np.linspace(0.0, 1.0, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    else:
        a_new = np.ones(new_n) / new_n
    return a_new

    q = max(2, n // 8)
    flat[q:n-q] = 1.0
    if flat.sum() > 0:
        inits['flat_mid'] = flat / flat.sum()

    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
            if len(parent) == n:
                inits['parent'] = p.copy()
            else:
                inits['parent_interp'] = upsample_interp(p, n)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search using FFT-accelerated gradient methods:
    1. Multi-scale: optimize at increasing lengths
    2. Deep PGD with Nesterov + Heavy Ball at best length
    3. Mirror descent polish
    4. Length exploration and random restarts
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            par_score = float(evaluate_sequence(p.tolist()))
            print(f"Parent: n={len(p)}, score={par_score:.6f}")
            best_score = min(best_score, par_score)
            best_sequence = (p / p.sum()).copy()
    except Exception:
        pass

    # Phase 1: Multi-scale optimization
    print("=== Phase 1: Multi-scale ===")
    scales = [32, 64, 128, 256, 384, 512, 640, 768]

    if parent is not None:
        par_n = len(parent)
        if par_n not in scales:
            scales.append(par_n)
            scales.sort()
        # Ensure parent length is included and we go up to it
        scales = [s for s in scales if s <= par_n + 256]
        scales.sort()
        # Make sure par_n is in the list
        if par_n not in scales:
            scales.insert(min(len(scales), scales.index(max(s for s in scales if s < par_n) if any(s < par_n for s in scales) else 0)), par_n)

    current_a = None

    for target_n in scales:
        if time.monotonic() > deadline - 80:
            break
        if target_n > 1024:
            continue

        if current_a is not None:
            if len(current_a) == target_n:
                a0 = current_a.copy()
            else:
                a0 = upsample_interp(current_a, target_n)
        else:
            inits = gen_inits(target_n, parent, rng)
            best_init_obj = float('inf')
            best_init = None
            for name, init in inits.items():
                obj = objective_val(init)
                if obj < best_init_obj:
                    best_init_obj = obj
                    best_init = init.copy()
            a0 = best_init
            print(f"  n={target_n} best init obj={best_init_obj:.4f}")

        # Time allocation
        remaining = deadline - time.monotonic() - 60.0
        t_alloc = min(100.0, remaining * 0.3)
        if t_alloc < 8:
            break

        # PGD Nesterov
        t_pgd = t_alloc * 0.5
        a_opt, obj_opt = pgd_nesterov(a0, t_pgd, lr_init=0.06, momentum=0.9,
                                       top_k=40, temperature=0.008)

        # Mirror descent
        remaining = deadline - time.monotonic() - 60.0
        t_md = min(25.0, remaining * 0.15)
        if t_md > 4:
            a_md, obj_md = mirror_descent(a_opt, t_md, lr_init=0.05,
                                           top_k=50, temperature=0.005)
            if obj_md < obj_opt:
                a_opt, obj_opt = a_md, obj_md

        if obj_opt < best_score - 1e-10:
            best_score = obj_opt
            best_sequence = a_opt.copy()
            print(f"  n={target_n}: new best {obj_opt:.8f}")

        current_a = a_opt

    # Phase 2: Deep optimization
    print("=== Phase 2: Deep Optimization ===")
    if best_sequence is not None:
        n_best = len(best_sequence)
        remaining = deadline - time.monotonic() - 50.0

        # Heavy ball with different params
        if remaining > 30:
            t_hb = min(80.0, remaining * 0.4)
            a_hb, obj_hb = pgd_heavy_ball(best_sequence, t_hb, lr_init=0.02,
                                           momentum=0.95, top_k=60, temperature=0.003)
            if obj_hb < best_score - 1e-10:
                best_score = obj_hb
                best_sequence = a_hb.copy()
                print(f"  HB: {obj_hb:.8f}")

        # Nesterov with different params
        remaining = deadline - time.monotonic() - 40.0
        if remaining > 20:
            t_nest = min(60.0, remaining * 0.4)
            a_nest, obj_nest = pgd_nesterov(best_sequence, t_nest, lr_init=0.03,
                                             momentum=0.93, top_k=30, temperature=0.004)
            if obj_nest < best_score - 1e-10:
                best_score = obj_nest
                best_sequence = a_nest.copy()
                print(f"  Nesterov: {obj_nest:.8f}")

        # Mirror descent
        remaining = deadline - time.monotonic() - 30.0
        if remaining > 15:
            t_md = min(40.0, remaining * 0.5)
            a_md, obj_md = mirror_descent(best_sequence, t_md, lr_init=0.03,
                                           top_k=80, temperature=0.003)
            if obj_md < best_score - 1e-10:
                best_score = obj_md
                best_sequence = a_md.copy()
                print(f"  MD deep: {obj_md:.8f}")

    # Phase 3: Fine-tuning with very small steps
    print("=== Phase 3: Fine-tuning ===")
    if best_sequence is not None and time.monotonic() < deadline - 25:
        remaining = deadline - time.monotonic() - 15.0

        # Tiny step PGD
        t_fine = min(20.0, remaining * 0.4)
        if t_fine > 3:
            a_fine, obj_fine = pgd_nesterov(best_sequence, t_fine, lr_init=0.003,
                                             momentum=0.97, top_k=10, temperature=0.001)
            if obj_fine < best_score - 1e-12:
                best_score = obj_fine
                best_sequence = a_fine.copy()
                print(f"  Fine PGD: {obj_fine:.8f}")

        # Tiny step MD
        remaining = deadline - time.monotonic() - 10.0
        t_md_fine = min(15.0, remaining * 0.5)
        if t_md_fine > 3:
            a_mdf, obj_mdf = mirror_descent(best_sequence, t_md_fine, lr_init=0.005,
                                             top_k=20, temperature=0.001)
            if obj_mdf < best_score - 1e-12:
                best_score = obj_mdf
                best_sequence = a_mdf.copy()
                print(f"  Fine MD: {obj_mdf:.8f}")

    # Phase 4: Length exploration
    print("=== Phase 4: Length exploration ===")
    if best_sequence is not None and time.monotonic() < deadline - 30:
        n_best = len(best_sequence)
        remaining = deadline - time.monotonic() - 20.0

        for offset in [64, -64, 128, -128, 32, -32]:
            new_n = n_best + offset
            if new_n < 32 or new_n > 1280:
                continue
            if remaining < 12:
                break

            a_trial = upsample_interp(best_sequence, new_n)
            t_alloc = min(15.0, remaining * 0.3)
            if t_alloc < 4:
                break

            a_opt, obj_opt = pgd_nesterov(a_trial, t_alloc, lr_init=0.04,
                                           momentum=0.9, top_k=40, temperature=0.005)
            if obj_opt < best_score - 1e-10:
                best_score = obj_opt
                best_sequence = a_opt.copy()
                print(f"  n={new_n}: {obj_opt:.8f}")
            remaining = deadline - time.monotonic() - 20.0

    # Phase 5: Random restarts around best
    print("=== Phase 5: Random restarts ===")
    if best_sequence is not None and time.monotonic() < deadline - 15:
        n_best = len(best_sequence)
        remaining = deadline - time.monotonic() - 10.0

        for noise_level in [0.003, 0.008, 0.015]:
            if remaining < 5:
                break
            for attempt in range(3):
                if remaining < 3:
                    break
                noise = rng.standard_normal(n_best) * noise_level
                a_pert = project_simplex(best_sequence + noise)
                t_alloc = min(8.0, remaining * 0.25)
                if t_alloc < 2:
                    break
                a_opt, obj_opt = pgd_nesterov(a_pert, t_alloc, lr_init=0.03,
                                               momentum=0.9, top_k=30, temperature=0.005)
                if obj_opt < best_score - 1e-12:
                    best_score = obj_opt
                    best_sequence = a_opt.copy()
                    print(f"  Perturb({noise_level}, {attempt}): {obj_opt:.8f}")
                remaining = deadline - time.monotonic() - 10.0

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256

    # Ensure validity
    best_sequence = np.maximum(best_sequence, 0.0)
    s = best_sequence.sum()
    if s < 0.01:
        best_sequence = np.ones(256) / 256

    # Final evaluation
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if not np.isfinite(final_score) or final_score > 2.0:
        best_sequence = np.ones(256) / 256
        final_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={final_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
