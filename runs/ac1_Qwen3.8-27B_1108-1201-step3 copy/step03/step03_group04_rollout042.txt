```python
"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses adaptive smoothed gradient descent with momentum on the simplex,
combined with LP oracle steps and peak-targeted adjustments.
Multiple initializations and multi-phase time allocation.

import numpy as np
import time
from scipy.sparse import csr_matrix
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def autocorr(a):
    """Compute conv(a, a)."""
    return fft_convolve(a, a)


def simplex_project(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_grad(a, c, t):
    """
    Compute gradient of 2*n*t*log(sum exp(c_k/t)).
    c = conv(a, a), shape (2n-1,).
    Returns gradient of shape (n,).
    """
    n = len(a)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / t)
    p = exp_c / exp_c.sum()
    # g_j = 2*n * sum_{m=0}^{n-1} a[m] * p[m+j]
    g = np.zeros(n)
    for j in range(n):
        g[j] = np.dot(a, p[j:j+n])
    g *= 2.0 * n
    return g


def compute_smoothed_grad_fast(a, c, t):
    """
    Faster version using the structure of the gradient.
    g_j = 2*n * sum_m a[m] * p[m+j]
    This is a cross-correlation that can be computed via FFT.
    """
    n = len(a)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / t)
    p = exp_c / exp_c.sum()
    
    # g_j = sum_m a[m] * p[m+j] for j = 0, ..., n-1
    # This is the cross-correlation of a and p, evaluated at lags 0, ..., n-1
    # cross_corr[j] = sum_m a[m] * p[m+j]
    # = conv(a_rev, p)[j + n - 1] where a_rev = a reversed

    g = np.zeros(n)
    for j in range(n):

def lp_oracle(a, M, time_limit=10.0):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0. Return b normalized to sum=1."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, method='highs', bounds=bounds,
                  options={'presolve': True, 'time_limit': time_limit})
    if res.success and res.x is not None:
        b = np.maximum(res.x, 0.0)
        sb = b.sum()
        if sb > 1e-15:
            return b / sb
    return None


def peak_targeted_step(a, c, alpha=0.1):
    """
    Reduce the contribution to the top peak by decreasing contributing elements
    and increasing others.
    """
    n = len(a)
    peak_k = c.argmax()
    # Elements contributing to peak: a[j] * a[peak_k - j] for valid j
    j_start = max(0, peak_k - n + 1)
    j_end = min(n - 1, peak_k)
    
    # Find the pair with maximum product
    best_j = -1
    best_prod = -1
    for j in range(j_start, j_end + 1):
        prod = a[j] * a[peak_k - j]
        if prod > best_prod:
            best_prod = prod
            best_j = j
    
    if best_j < 0 or best_prod < 1e-15:
        return a
    
    # Move mass from the two contributing elements to a "quiet" position
    # Find position with minimum convolution contribution
    # Simple heuristic: move to a position with low a value
    i1 = best_j
    i2 = peak_k - best_j
    
    if i1 == i2:
        # Single element contributes (a[i]^2)
        donor = i1
        amount = alpha * a[donor] * 0.5
        # redistribute to all other positions
        others = np.setdiff1d(np.arange(n), [donor])
        if len(others) == 0:
            return a
        # Give to positions with lowest a
        idx = others[np.argmin(a[others])]
        a_new = a.copy()
        a_new[donor] -= amount
        a_new[idx] += amount
    else:
        # Two elements contribute
        # Reduce both, increase others
        amount = alpha * min(a[i1], a[i2]) * 0.5
        a_new = a.copy()
        a_new[i1] -= amount
        a_new[i2] -= amount
        # Redistribute to positions with lowest a (excluding i1, i2)
        others = np.setdiff1d(np.arange(n), [i1, i2])
        if len(others) > 0:
            # Give half to each of the two lowest positions
            sorted_others = others[np.argsort(a[others])[:2]]
            a_new[sorted_others[0]] += amount
            if len(sorted_others) > 1:
                a_new[sorted_others[1]] += 0  # just add to one for simplicity
            else:
                a_new[sorted_others[0]] += amount
    
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a
    return a_new / s
```

`g_j = 2*n * sum_m a[m] * p[m+j]`

This is the cross-correlation of `a` and `p` at lag j. Using FFT:
- Pad `a` to length N (power of 2 >= 3n-2)
- Pad `p` to length N
- Compute `X = fft(a, N)`, `Y = fft(p, N)`
- `cross_corr = ifft(X * conj(Y))` (this gives circular cross-correlation)
- The first n elements (or a subset) give us what we want.

Actually, the cross-correlation of a (length n) and p (length 2n-1) has length n + (2n-1) - 1 = 3n-2. We need elements 0 through n-1.

With zero-padding to length N >= 3n-2, the circular cross-correlation equals the linear cross-correlation.

Let me implement this for speed:

```python
def compute_smoothed_grad_fft(a, c, t):
    n = len(a)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / t)
    p = exp_c / exp_c.sum()
    
    # cross_corr[j] = sum_m a[m] * p[m+j] for j = 0, ..., n-1
    # This is cross-correlation of a and p
    # Using FFT: cross_corr = ifft(fft(a, N) * conj(fft(p, N)))
    # But we need the "right" alignment
    
    # Actually, the standard cross-correlation is:
    # (a ⋆ p)[k] = sum_m a[m+k] * conj(p[m])
    # which is different from what we want.
    
    # What we want: g[j] = sum_m a[m] * p[m+j]
    # Let q[m] = p[m] (length 2n-1)
    # g[j] = sum_m a[m] * q[m+j]
    # This is the "anti-correlation" or we can write it as:
    # g[j] = sum_m a[m] * q[m+j] = (a * q_rev)[j] where q_rev is some reversal
    
    # Actually, let's just think of it as:
    # g = [sum_m a[m]*p[m], sum_m a[m]*p[m+1], ..., sum_m a[m]*p[m+n-1]]
    # = [a·p[0:n], a·p[1:n+1], ..., a·p[n-1:2n-1]]
    
    # This is a "sliding dot product" which is a 1D cross-correlation.
    # We can compute it as: g[j] = (a * p)[n-1+j] where * is convolution
    # Let's verify: (a*p)[k] = sum_i a[i] * p[k-i]
    # (a*p)[n-1+j] = sum_i a[i] * p[n-1+j-i]
    # Let m = i: sum_m a[m] * p[n-1+j-m]
    # We want sum_m a[m] * p[m+j]
    # These are different! 
    
    # Let's use the reversed a: let a_rev[i] = a[n-1-i]
    # (a_rev * p)[k] = sum_i a_rev[i] * p[k-i] = sum_i a[n-1-i] * p[k-i]
    # Let j' = n-1-i: = sum_j a[j] * p[k-n+1+j]
    # We want sum_j a[j] * p[j+lag]
    # So k-n+1+j = j+lag => k = n-1+lag
    # (a_rev * p)[n-1+lag] = sum_j a[j] * p[j+lag] = g[lag] / (2n)
    
    # So g = 2*n * (a_rev * p)[n-1 : n-1+n]
    
    N = 1 << (3*n).bit_length()  # power of 2 >= 3n
    A = np.fft.rfft(a[::-1], N)
    P = np.fft.rfft(p, N)
    conv = np.fft.irfft(A * P, N)
    g = 2.0 * n * conv[n-1 : 2*n-1]
    return g
```

```python
"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses adaptive smoothed gradient descent with momentum on the simplex,
FFT-accelerated gradient computation, LP oracle steps, and peak-targeted
adjustments. Multi-phase time allocation with diverse initializations.
"""

import numpy as np
import time
from scipy.sparse import csr_matrix
from scipy.optimize import linprog


def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def autocorr(a):
    """Compute conv(a, a) via FFT."""
    return fft_convolve(a, a)


def simplex_project(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_fft(a, c, t):
    """
    Compute gradient of 2*n*t*log(sum exp(c_k/t)) using FFT.
    c = conv(a,a) of shape (2n-1,). Returns gradient of shape (n,).
    """
    n = len(a)
    c_max = c.max()
    exp_c = np.exp((c - c_max) / t)
    p = exp_c / exp_c.sum()

    # g[j] = 2*n * sum_m a[m] * p[m+j] for j=0..n-1
    # = 2*n * conv(a_rev, p)[n-1+j]
    N = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a[::-1].astype(np.float64), N)
    P = np.fft.rfft(p, N)
    conv = np.fft.irfft(A * P, N)
    g = 2.0 * n * conv[n - 1:2 * n - 1]
    return g


def compute_grad_topk(a, c, K=30):
    """
    Compute gradient using only top-K peaks (sharper, faster).
    """
    n = len(a)
    if K < len(c):
        top_idx = np.argpartition(c, -K)[-K:]
    else:
        top_idx = np.arange(len(c))

    top_vals = c[top_idx]
    w = top_vals ** 3  # sharper weighting on highest peaks
    w_sum = w.sum()
    if w_sum < 1e-30:
        return np.zeros(n)
    w = w / w_sum

    # For each top peak k, it contributes to g[j] via a[k-j]
    # g[j] = 2*n * sum_{k in top} w_k * a[k-j] (valid j)
    # Equivalently: g = 2*n * conv(w_padded, a) at appropriate positions
    # Or directly: for each k, add w_k * a[k-j] to g[j] for valid j
    g = np.zeros(n)
    for idx, k in enumerate(top_idx):
        if w[idx] < 1e-15:
            continue
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        g[j_start:j_end + 1] += w[idx] * a[k - j_end:j_start - 1:-1] if j_start > 0 else w[idx] * a[0:j_end + 1][::-1]
    # Actually let me just do it more carefully:
    # For peak at position k: g[j] += w_k * a[k-j] for j in [max(0,k-n+1), min(n-1,k)]
    # a[k-j] where j ranges: a[k - j_start] down to a[k - j_end]
    # Let's redo:
    g = np.zeros(n)
    for idx, k in enumerate(top_idx):
        wk = w[idx]
        if wk < 1e-15:
            continue
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            g[j] += wk * a[k - j]
    g *= 2.0 * n
    return g


def lp_oracle(a, M, time_limit=10.0):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0. Return b/sum(b)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A_mat = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=A_mat, b_ub=b_ub, method='highs', bounds=bounds,
                  options={'presolve': True, 'time_limit': time_limit})
    if res.success and res.x is not None:
        b = np.maximum(res.x, 0.0)
        sb = b.sum()
        if sb > 1e-15:
            return b / sb
    return None


def optimize_smoothed(a0, time_limit, verbose=False, use_lp=False,
                      t_factor=0.05, lr_init=0.008, mom_init=0.90):
    """
    Smoothed gradient descent with momentum on the simplex.
    Optionally uses LP oracle for escape.
    """
    n = len(a0)
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    c = autocorr(a)
    M = c.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    v = np.zeros(n)
    t = M * t_factor
    t_min = M * 0.002
    lr = lr_init
    momentum = mom_init

    t0 = time.monotonic()
    it = 0
    stall = 0
    rng_local = np.random.default_rng(12345 + n)

    while time.monotonic() - t0 < time_limit:
        # Choose gradient method based on temperature
        if t > M * 0.02 and n <= 1024:
            g = compute_grad_fft(a, c, t)
        else:
            g = compute_grad_topk(a, c, K=max(10, n // 10))

        # Momentum update
        v = momentum * v - lr * g
        a_new = simplex_project(a + v)

        # Evaluate
        c_new = autocorr(a_new)
        M_new = c_new.max()
        obj_new = 2.0 * n * M_new

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            c = c_new
            M = M_new
            stall = 0
        else:
            a = a_new
            c = c_new
            M = M_new
            stall += 1

        # Adapt parameters
        t = max(t_min, t * 0.9997)
        if stall > 0 and stall % 50 == 0:
            lr *= 0.97
        if stall == 0:
            lr = min(lr_init * 1.5, lr * 1.005)

        # Escape mechanisms
        if stall > 300:
            if use_lp and stall % 200 == 0:
                b = lp_oracle(a, M, time_limit=min(5.0, max(1.0, time_limit - (time.monotonic() - t0))))
                if b is not None and not np.allclose(a, b, atol=1e-8):
                    a = simplex_project(0.7 * a + 0.3 * b)
                    c = autocorr(a)
                    M = c.max()
                    if 2.0 * n * M < best_obj - 1e-10:
                        best_obj = 2.0 * n * M
                        best_a = a.copy()
                    stall = 0
                    v = np.zeros(n)
            elif stall > 800:
                # Add structured noise
                noise = rng_local.standard_normal(n) * 0.003
                a = np.maximum(a + noise, 0.0)
                a = a / a.sum()
                c = autocorr(a)
                M = c.max()
                stall = 0
                v = np.zeros(n)
                lr = lr_init
                t = M * 0.03
            elif stall > 2000:
                break

        it += 1
        if verbose and it % 3000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    it={it}, obj={best_obj:.6f}, M={M:.4e}, t/t0={t/(M*t_factor):.3f}, "
                  f"lr={lr:.5f}, stall={stall}, time={elapsed:.0f}s")

    return best_a, best_obj


def optimize_lp_focused(a0, time_limit, verbose=False):
    """LP-focused optimization with line search."""
    n = len(a0)
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    c = autocorr(a)
    M = c.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    stall = 0
    rng_local = np.random.default_rng(999 + n)

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        b = lp_oracle(a, M, time_limit=min(10.0, remaining * 0.5))
        if b is None:
            break

        if np.allclose(a, b, atol=1e-9):
            stall += 1
        else:
            # Line search
            best_t = 0.0
            best_obj_try = best_obj
            for t_step in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                cand = (1.0 - t_step) * a + t_step * b
                cand = np.maximum(cand, 0.0)
                cs = cand.sum()
                if cs < 1e-15:
                    continue
                cand = cand / cs
                cc = autocorr(cand)
                obj_try = 2.0 * n * cc.max()
                if obj_try < best_obj_try - 1e-12:
                    best_obj_try = obj_try
                    best_t = t_step

            if best_t > 0:
                a = (1.0 - best_t) * a + best_t * b
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                c = autocorr(a)
                M = c.max()
                best_obj = best_obj_try
                best_a = a.copy()
                stall = 0
            else:
                stall += 1

        if stall > 5:
            # Perturb and reset
            scale = 0.001 * (1 + stall * 0.3)
            a = np.maximum(a + rng_local.standard_normal(n) * scale, 0.0)
            a = a / a.sum()
            c = autocorr(a)
            M = c.max()
            stall = 0

        it += 1
        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP it={it}, obj={best_obj:.6f}, M={M:.4e}, stall={stall}, time={elapsed:.0f}s")

    return best_a, best_obj


def generate_init(name, n, rng, parent=None):
    """Generate initialization of given type."""
    if name == 'uniform':
        return np.ones(n) / n
    elif name == 'random':
        x = rng.random(n) + 0.01
        return x / x.sum()
    elif name == 'random_sparse':
        x = rng.random(n)
        x[x < 0.4] = 0.0
        if x.sum() < 1e-10:
            x = rng.random(n) + 0.01
        return x / x.sum()
    elif name == 'gaussian':
        x = np.arange(n, dtype=np.float64)
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
        return g / g.sum()
    elif name == 'edge':
        e = max(1, n // 4)
        x = np.zeros(n)
        x[:e] = 1.0
        x[-e:] = 1.0
        return x / x.sum()
    elif name == 'ramp':
        r = np.arange(1, n + 1, dtype=np.float64)
        return r / r.sum()
    elif name == 'two_gaussians':
        x = np.arange(n, dtype=np.float64)
        g1 = np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2)
        g2 = np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2)
        g = g1 + g2
        return g / g.sum()
    elif name == 'sine_mod':
        x = np.arange(n, dtype=np.float64)
        s = 0.5 + 0.5 * np.sin(2 * np.pi * x / n)
        s = np.maximum(s, 0.02)
        return s / s.sum()
    elif name == 'random_smooth':
        # Random low-pass filtered
        x = rng.random(n)
        kernel = np.ones(5) / 5.0
        from numpy import convolve as np_conv
        x = np_conv(x, kernel, mode='same')
        x = np.maximum(x, 0.0)
        if x.sum() < 1e-10:
            x = rng.random(n) + 0.01
        return x / x.sum()
    elif name == 'parent':
        if parent is not None:
            p = np.maximum(parent, 0.0)
            ps = p.sum()
            if ps > 1e-10:
                return p / ps
        return np.ones(n) / n
    elif name == 'parent_interp':
        if parent is not None and len(parent) != n:
            p = np.maximum(parent, 0.0)
            p_int = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
            s = p_int.sum()
            if s > 1e-10:
                return p_int / s
        return np.ones(n) / n
    elif name == 'parent_perturb':
        if parent is not None:
            p = np.maximum(parent, 0.0).copy()
            ps = p.sum()
            if ps > 1e-10:
                p = p / ps
                noise = rng.standard_normal(n) * 0.01
                p = np.maximum(p + noise, 0.0)
                p = p / p.sum()
                return p
        return np.ones(n) / n
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    Phase 1: Quick scan over diverse initializations (smoothed gradient).
    Phase 2: Deep smoothed gradient optimization on best.
    Phase 3: LP-focused optimization for structural improvements.
    Phase 4: Fine polish with sharp gradient.
    Phase 5: Perturbation restarts for final improvement.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: n={len(p)}, sum={p.sum():.2f}, eval={evaluate_sequence(p.tolist()):.6f}")
    except Exception as e:
        print(f"Parent load failed: {e}")

    # Determine sequence length
    n = 384
    if parent is not None:
        parent_n = len(parent)
        if 200 <= parent_n <= 600:
            n = parent_n

    # === Phase 1: Quick scan (25% of time) ===
    print(f"=== Phase 1: Quick Scan (n={n}) ===")
    init_names = ['parent', 'parent_perturb', 'random', 'uniform', 'random_smooth',
                  'gaussian', 'edge', 'random_sparse', 'two_gaussians', 'sine_mod', 'ramp',
                  'parent_interp']

    scan_budget = (deadline - time.monotonic()) * 0.25
    time_per_init = max(10.0, scan_budget / len(init_names))

    for name in init_names:
        if time.monotonic() > deadline - scan_budget:
            break
        t_alloc = min(time_per_init, deadline - time.monotonic() - scan_budget)
        if t_alloc < 8:
            break

        a0 = generate_init(name, n, rng, parent)
        a_opt, obj_opt = optimize_smoothed(a0, time_limit=t_alloc, verbose=False, use_lp=False)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            print(f"  init={name:16s} score={score:.6f} (t={t_alloc:.0f}s)")
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()

    print(f"After Phase 1: best={best_score:.6f}")

    # === Phase 2: Deep smoothed gradient (30% of remaining) ===
    if best_sequence is not None:
        print("=== Phase 2: Deep Smoothed Gradient ===")
        remaining = deadline - time.monotonic() - 30.0
        t_deep = remaining * 0.30
        if t_deep > 10:
            a_opt, obj_opt = optimize_smoothed(best_sequence, time_limit=t_deep,
                                                verbose=True, use_lp=True,
                                                t_factor=0.03, lr_init=0.005, mom_init=0.92)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  Phase 2: score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    # === Phase 3: LP-focused optimization (25% of remaining) ===
    if best_sequence is not None and time.monotonic() < deadline - 30:
        print("=== Phase 3: LP-Focused ===")
        remaining = deadline - time.monotonic() - 20.0
        t_lp = remaining * 0.25
        if t_lp > 15:
            a_opt, obj_opt = optimize_lp_focused(best_sequence, time_limit=t_lp, verbose=True)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  Phase 3: score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    # === Phase 4: Fine polish with sharp gradient (15% of remaining) ===
    if best_sequence is not None and time.monotonic() < deadline - 15:
        print("=== Phase 4: Fine Polish ===")
        remaining = deadline - time.monotonic() - 10.0
        t_polish = remaining * 0.15
        if t_polish > 5:
            a_opt, obj_opt = optimize_smoothed(best_sequence, time_limit=t_polish,
                                                verbose=False, use_lp=False,
                                                t_factor=0.005, lr_init=0.002, mom_init=0.95)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                print(f"  Phase 4: score={score:.6f}")
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()

    # === Phase 5: Multi-restart perturbations (remaining time) ===
    if best_sequence is not None and time.monotonic() < deadline - 10:
        print("=== Phase 5: Perturbation Restarts ===")
        n_best = len(best_sequence)
        remaining = deadline - time.monotonic() - 5.0
        n_restarts = max(1, int(remaining // 12))
        n_restarts = min(n_restarts, 15)

        for i in range(n_restarts):
            if time.monotonic() > deadline - 5:
                break
            # Varying perturbation scales
            scales = [0.001, 0.002, 0.005, 0.003, 0.001, 0.004, 0.002, 0.001,
                      0.003, 0.002, 0.005, 0.001, 0.002, 0.003, 0.001]
            scale = scales[i % len(scales)]
            noise = rng.standard_normal(n_best) * scale
            a_pert = np.maximum(best_sequence + noise, 0.0)
            a_pert = a_pert / a_pert.sum()

            t_alloc = min(10.0, deadline - time.monotonic() - 5)
            if t_alloc < 3:
                break

            a_opt, obj_opt = optimize_smoothed(a_pert, time_limit=t_alloc,
                                                verbose=False, use_lp=False,
                                                t_factor=0.02, lr_init=0.004)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Restart {i}: score={score:.6f} (improved!)")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
```
