```python
def gs_flat_spectrum(n, n_iter=1000, seed=42):
    rng = np.random.default_rng(seed)
    N = 1 << (2*n-1).bit_length()  # FFT size for linear convolution
    # Initialize
    A = np.zeros(N, dtype=complex)
    a = rng.random(n)
    a = np.concatenate([a, np.zeros(N-n)])
    A = fft(a)
    
    for i in range(n_iter):
        # Project onto flat magnitude
        mag = np.abs(A)
        avg_mag = mag.mean()
        A = A / (mag + 1e-30) * avg_mag
        # Project onto time-domain constraint
        a_full = np.real(ifft(A))
        a_full[n:] = 0  # zero-pad constraint
        a_full = np.maximum(a_full, 0)  # non-negativity
        A = fft(a_full)
    
    a = np.real(ifft(A))[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-10:
        a /= s
    return a
```

```python
    n = len(a)
    n_k = 2 * n - 1
    
    # Objective: maximize sum(b) => minimize -sum(b)
    c = -np.ones(n)
    
    # Constraints: A_ub @ b <= b_ub
    # For each k in 0..2n-2: sum_i a[k-i] * b[i] <= M
    # where i ranges over 0..n-1 and 0 <= k-i <= n-1
    A_ub = np.zeros((n_k, n))
    for k in range(n_k):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            A_ub[k, i] = a[k - i]
    
    b_ub = np.full(n_k, M)
    
    # Bounds: b >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.status != 0:
        return None
    
    return result.x

            best_a = a_new.copy()
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Gerchberg-Saxton spectral flattening for initialization
2. LP-based iterative refinement (linear programming to find best descent direction)
3. L-BFGS-B with log-sum-exp smooth max for polishing
4. Multi-length, multi-start search with simulated annealing restarts
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.fft import fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = fft(a, fft_size)
    return np.real(ifft(A * A, fft_size))[:size]


def compute_objective(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def smooth_max_and_weights(x, beta):
    """Compute smooth max (log-sum-exp) and softmax weights."""
    x_max = x.max()
    shifted = beta * (x - x_max)
    shifted = np.clip(shifted, -50, 50)
    e = np.exp(shifted)
    total = e.sum()
    if total < 1e-30:
        w = np.zeros_like(x)
        w[np.argmax(x)] = 1.0
        return x_max, w
    w = e / total
    return x_max + np.log(total) / beta, w


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.argmax(~cond)
    if rho_val == n - 1:
        return np.ones(n) / n
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1)
    return np.maximum(v - theta, 0.0)


def gs_flat_spectrum(n, n_iter=1500, seed=42, n_trials=15):
    """Gerchberg-Saxton algorithm to find non-negative sequences with approximately flat power spectrum."""
    rng = np.random.default_rng(seed)
    N = 1 << (2 * n - 1).bit_length()
    best_a = None
    best_obj = np.inf

    for trial in range(n_trials):
        a = rng.random(n)
        a_full = np.zeros(N)
        a_full[:n] = a
        A = fft(a_full)

        for i in range(n_iter):
            # Project onto constant magnitude in frequency
            mag = np.abs(A)
            max_mag = mag.max()
            if max_mag < 1e-30:
                break
            target = mag.mean()
            A = A / (mag + 1e-30) * target

            # Project onto time-domain: non-negative, zero-padded
            a_full = np.real(ifft(A))
            a_full[n:] = 0.0
            a_full = np.maximum(a_full, 0.0)
            if a_full.sum() < 1e-30:
                break
            A = fft(a_full)

        a = np.real(ifft(A))[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-10:
            a /= s
            obj = compute_objective(a)
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()

    if best_a is None:
        best_a = np.ones(n) / n
        best_obj = compute_objective(best_a)
    return best_a, best_obj


def solve_lp_descent(a, M):
    """Solve LP: maximize sum(b) s.t. b>=0, (a*b)[k] <= M for all k."""
    n = len(a)
    n_k = 2 * n - 1

    c = -np.ones(n)

    # Build constraint matrix: row k has a[k-i] in column i where 0<=k-i<=n-1
    A_ub = np.zeros((n_k, n))
    for k in range(n_k):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        if i_max >= i_min:
            A_ub[k, i_min:i_max + 1] = a[k - i_max:k - i_min + 1][::-1]

    b_ub = np.full(n_k, M)
    bounds = [(0, None)] * n

    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True})
        if result.status == 0:
            return result.x
    except Exception:
        pass
    return None


def lp_refine(a, n_iter=200, t_init=0.4):
    """LP-based iterative refinement."""
    n = len(a)
    a = np.maximum(a, 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a /= s

    best_obj = compute_objective(a)
    best_a = a.copy()
    t = t_init

    for it in range(n_iter):
        conv = fast_convolve(a)
        M = conv.max()

        b = solve_lp_descent(a, M)
        if b is None:
            break

        b = np.maximum(b, 0.0)
        s_b = b.sum()
        if s_b < 1e-15:
            break
        b /= s_b

        a_new = (1.0 - t) * a + t * b
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new /= s_new

        obj_new = compute_objective(a_new)
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        t = max(0.03, t * 0.995)

    return best_a, best_obj


def objective_softmax(x, n, beta):
    """Objective for L-BFGS-B with softmax parametrization."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a /= a.sum()
    conv = fast_convolve(a)
    sm, _ = smooth_max_and_weights(conv, beta)
    return 2.0 * n * sm


def jacobian_softmax(x, n, beta):
    """Jacobian for L-BFGS-B with softmax parametrization."""
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a /= a.sum()
    conv = fast_convolve(a)
    sm, w = smooth_max_and_weights(conv, beta)

    # Gradient of smooth_max w.r.t. a
    N_fft = 1 << (len(w) + n - 1 + 63).bit_length()
    W = fft(w, N_fft)
    A = fft(a, N_fft)
    grad_a = np.real(ifft(np.conj(W) * A, N_fft))[:n] * 2.0

    # Convert to gradient w.r.t. log-parametrization
    lagrange = np.dot(a, grad_a)
    return a * (grad_a - lagrange)


def lbfgsb_optimize(a0, beta_schedule, maxiter_total=2000):
    """L-BFGS-B with multi-beta annealing schedule."""
    n = len(a0)
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()
    x = np.log(a)

    best_obj = compute_objective(a)
    best_a = a.copy()

    n_betas = len(beta_schedule)
    maxiter_per = max(50, maxiter_total // n_betas)

    for beta in beta_schedule:
        result = minimize(objective_softmax, x, jac=jacobian_softmax, args=(n, beta),
                          method='L-BFGS-B', options={'maxiter': maxiter_per, 'maxls': 40})
        x = result.x
        a = np.exp(x - x.max())
        a = a / a.sum()
        obj = compute_objective(a)
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    return best_a, best_obj


def generate_diverse_inits(n, parent, rng):

    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

                if ps_r > 1e-10:

def annealing_restart(a_best, rng, n_restarts=5, t=0.1):
    """Simulated annealing style restarts: perturb and re-optimize."""
    n = len(a_best)
    best_a = a_best.copy()
    best_obj = compute_objective(a_best)
    temperature = 0.3

    for r in range(n_restarts):
        # Perturb
        noise = rng.standard_normal(n) * temperature
        a_pert = project_simplex(a_best + noise)

        # Quick LP refinement
        a_new, obj_new = lp_refine(a_pert, n_iter=50, t_init=0.3)

        # Accept if better or with probability
        if obj_new < best_obj:
            best_a = a_new.copy()
            best_obj = obj_new
        elif np.random.random() < np.exp(-(obj_new - best_obj) / (temperature * 0.01)):
            best_a = a_new.copy()
            best_obj = obj_new

        temperature *= 0.5

    return best_a, best_obj

    # Add diverse inits for key lengths
    for n in [256, 320, 384]:
        inits = generate_diverse_inits(n, parent, rng)
        for name in ['gauss_0.15', 'gauss_0.2', 'bimodal', 'random']:
            if name in inits:
                candidates[('init_%s' % name, n)] = inits[name]

    # Run LP refinement on each candidate
    for key, a0 in sorted(candidates.items()):
        if time.monotonic() > phase2_deadline - 30:
            break
        t_alloc = min(120.0, phase2_deadline - time.monotonic() - 15)
        if t_alloc < 10:
            break

        n_iter_lp = int(min(300, max(50, t_alloc * 2)))
        a_lp, obj_lp = lp_refine(a0, n_iter=n_iter_lp, t_init=0.4)
        try_update(a_lp)
        candidates[key + '_lp'] = a_lp

    print("Phase 2 done. Best: %.6f" % best_score)

    # Phase 3: L-BFGS-B Polishing
    print("\n=== Phase 3: L-BFGS-B Polishing ===")
    phase3_deadline = deadline - 100.0

    if best_sequence is not None:
        n_best = len(best_sequence)
        a_norm = best_sequence / best_sequence.sum()
    else:
        n_best = 256
        a_norm = gs_results.get(256, np.ones(256) / 256)

    # Polish the best with L-BFGS-B
    beta_schedule = [50.0, 30.0, 20.0, 10.0, 5.0, 3.0]
    maxiter = int(min(3000, max(500, (phase3_deadline - time.monotonic()) * 3)))
    a_polished, obj_polished = lbfgsb_optimize(a_norm, beta_schedule, maxiter_total=maxiter)
    try_update(a_polished)

    # Polish other candidates too if time allows
    for key, a0 in candidates.items():
        if time.monotonic() > phase3_deadline - 20:
            break
        if 'lp' not in key:
            continue
        n_c = len(a0)
        if n_c != n_best and abs(n_c - n_best) / n_best > 0.25:
            continue
        t_alloc = min(60.0, phase3_deadline - time.monotonic() - 10)
        if t_alloc < 10:
            break
        a_c = a0 / a0.sum()
        if n_c == n_best:
            a_pol, _ = lbfgsb_optimize(a_c, [30.0, 10.0, 5.0], maxiter_total=500)
            try_update(a_pol)

    print("Phase 3 done. Best: %.6f" % best_score)

    # Phase 4: Further refinement with LP + L-BFGS-B alternation
    print("\n=== Phase 4: Alternating Refinement ===")
    if best_sequence is not None:
        a_cur = best_sequence / best_sequence.sum()
        n_cur = len(a_cur)

        for it in range(20):
            if time.monotonic() > deadline - 30:
                break

            # LP step
            a_lp, _ = lp_refine(a_cur, n_iter=50, t_init=0.2)
            a_cur = a_lp / a_lp.sum()
            obj_lp = compute_objective(a_cur)

            # L-BFGS-B step
            a_opt, _ = lbfgsb_optimize(a_cur, [20.0, 10.0, 5.0], maxiter_total=400)
            a_cur = a_opt

            obj_cur = compute_objective(a_cur)
            if obj_cur < best_score / (2.0 * n_cur):  # rough check
                try_update(a_cur)

            # Check convergence
            if abs(obj_cur - obj_lp) < 1e-8:
                break

    # Phase 5: Annealing restarts
    print("\n=== Phase 5: Annealing Restarts ===")
    if best_sequence is not None:
        a_for_anneal = best_sequence / best_sequence.sum()
        n_anneal = len(a_for_anneal)
        a_annealed, obj_annealed = annealing_restart(a_for_anneal, rng, n_restarts=5, t=0.15)
        try_update(a_annealed)

        if obj_annealed < compute_objective(best_sequence / best_sequence.sum()):
            # Polish the annealed result
            if time.monotonic() < deadline - 30:
                a_pol2, _ = lbfgsb_optimize(a_annealed, [20.0, 10.0, 5.0, 3.0], maxiter_total=800)
                try_update(a_pol2)

    # Phase 6: Try different lengths for the best structure
    print("\n=== Phase 6: Multi-length Search ===")
    if best_sequence is not None:
        n_best = len(best_sequence)
        alt_lengths = set()
        for factor in [0.5, 0.75, 1.0, 1.25, 1.5, 2.0]:
            alt_n = int(round(n_best * factor / 8)) * 8
            if 64 <= alt_n <= 1024:
                alt_lengths.add(alt_n)

        for alt_n in sorted(alt_lengths):
            if time.monotonic() > deadline - 30:
                break
            if alt_n == n_best:
                continue

            # Resample best to new length
            pp = np.interp(np.linspace(0, 1, alt_n), np.linspace(0, 1, n_best), best_sequence)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps < 1e-10:
                continue
            pp /= ps

            # Quick LP + L-BFGS-B
            a_qp, _ = lp_refine(pp, n_iter=30, t_init=0.3)
            a_qo, _ = lbfgsb_optimize(a_qp, [20.0, 10.0, 5.0], maxiter_total=500)
            try_update(a_qo)

    # Final: verify and return
    if best_sequence is None:
        # Fallback
        best_sequence = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final L-BFGS-B polish if time allows
    remaining = deadline - time.monotonic() - 5.0
    if remaining > 15:
        a_final = best_sequence / best_sequence.sum()
        beta_final = [10.0, 5.0, 3.0, 2.0]
        maxiter_final = int(min(1000, max(100, remaining * 5)))
        a_f, _ = lbfgsb_optimize(a_final, beta_final, maxiter_total=maxiter_final)
        try_update(a_f)

    # Ensure final answer is valid
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print("\nFINAL: score=%.6f, n=%d" % (best_score, len(best_sequence)))
    return [float(x) for x in best_sequence]
```
