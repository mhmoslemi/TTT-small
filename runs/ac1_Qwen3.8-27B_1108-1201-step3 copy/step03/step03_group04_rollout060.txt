"""

import numpy as np

from scipy.sparse import csr_matrix

# Preallocated space for FFT
def next_pow2(n):
    return 1 << (n - 1).bit_length()

def fast_convolve(a, b=None):
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = next_pow2(size)
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def compute_objective(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2 with sum normalization."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)

def compute_conv_max(a):
    """Assume sum(a) = 1, return max of conv."""
    conv = fast_convolve(a)
    return conv.max()

def multi_peak_subgradient(a, top_k=20, temperature=0.1):
    """
    Compute a weighted multi-peak subgradient of max(conv(a,a)).
    At peak k: grad a_j += 2 * a[k-j] (from the product rule on conv[k] = sum_i a_i a_{k-i})
    Actually, d/d a_j of conv[k] = a[k-j] + a[k-j] by symmetry... 
    Wait: conv[k] = sum_i a_i * a[k-i]. d/d a_j: 
    - j appears as a_i when i=j: contributes a[k-j]
    - j appears as a[k-i] when k-i=j, i.e., i=k-j: contributes a[k-j]
    So d conv[k] / d a_j = a[k-j] + a[k-j] = 2*a[k-j], when both j and k-j are in [0, n-1].
    If only one is in range, it's just a[k-j].
    """
    n = len(a)
    conv = fast_convolve(a)
    # Get top_k peaks
    k = min(top_k, len(conv))
    peak_indices = np.argpartition(conv, -k)[-k:]
    peak_vals = conv[peak_indices]
    
    # Weight by exponential of peak value (relative to max)
    max_val = peak_vals.max()
    if max_val < 1e-15:
        return np.zeros(n), conv
    # Temperature controls how many peaks we focus on
    weights = np.exp((peak_vals - max_val) / (temperature * max_val + 1e-15))
    weights /= weights.sum()
    
    g = np.zeros(n)
    for idx in range(len(peak_indices)):
        if weights[idx] < 1e-12:
            continue
        k_val = int(peak_indices[idx])
        w = weights[idx]
        # d conv[k_val] / d a_j for j in [0, n-1]:
        # j must satisfy: 0 <= k_val - j <= n-1, i.e., k_val-(n-1) <= j <= k_val
        j_start = max(0, k_val - n + 1)
        j_end = min(n - 1, k_val)
        if j_start > j_end:
            continue
        # For j in [j_start, j_end]: derivative is 2*a[k_val - j] 
        # (but if j == k_val - j, i.e., j = k_val/2, it's just a[j]^2 derivative = 2*a[j])

        g[indices] += w * 2.0 * a[k_val - indices]
    
    return g, conv

def bb_step_size(x_prev, x_curr, g_prev, g_curr):
    """Barzilai-Borwein step size."""
    s = x_curr - x_prev
    y = g_curr - g_prev
    ys = y @ s
    ss = s @ s
    yy = y @ y
    if ss < 1e-15:
        return 0.01
    if ys > 1e-10:
        return ss / ys
    elif yy > 1e-10:
        return ys / yy
    else:
        return 0.01

def lp_oracle(a, M, time_limit=3.0):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    # Build the convolution matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, method='highs', bounds=bounds,
                  options={'presolve': True, 'time_limit': time_limit})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def build_conv_matrix(a):
    """Build sparse matrix for convolution with a."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def generate_structured_inits(n, rng):

    inits = {}
    x = np.linspace(0, 1, n)

    inits['uniform'] = np.ones(n) / n

    g = np.exp(-50 * x**2)
    inits['gauss_center'] = g / g.sum()
    
    # Two Gaussians (at edges)
    g2 = np.exp(-50 * (x - 0.1)**2) + np.exp(-50 * (x - 0.9)**2)
    inits['gauss_two_edge'] = g2 / g2.sum()

    g4 = (np.exp(-80 * (x - 0.1)**2) + np.exp(-80 * (x - 0.4)**2) +
          np.exp(-80 * (x - 0.6)**2) + np.exp(-80 * (x - 0.9)**2))
    inits['w_4bump'] = g4 / g4.sum()

    w = max(1, n // 4)
    ec = np.zeros(n)
    ec[:w] = 1.0
    ec[-w:] = 1.0
    inits['edge_block'] = ec / ec.sum()
    
    # Ramp
    r = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = r / r.sum()

    inits['inv_ramp'] = r[::-1] / r.sum()

    tri = np.minimum(x, 1 - x) + 0.1
    inits['triangle'] = tri / tri.sum()
    
    # Cosine window
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x))
    inits['cosine'] = cos_w / cos_w.sum()
    
    # Sin^2 window (peaks in middle)

    inits['dirichlet'] = rng.dirichlet(np.ones(n) * 0.3)

4. Optimize the lifted sequence

This is like a multigrid approach. The smaller problem is easier to solve, and the solution gives a good initialization for the larger problem.

Let me also think about the LP formulation. The current LP is:
max sum(b) s.t. conv(a,b) <= M, b >= 0

This finds the b that maximizes total mass while keeping the cross-correlation bounded. The issue is that M is set to the current max of conv(a,a), which might not be the right bound.

Alternative: Use a smaller M (like 0.9 * current_max) to force more change.

Or: Use a different LP:
min max_k conv(b,b) s.t. b >= 0, sum(b) = 1
This is a min-max problem, which can be formulated as an LP:
min t s.t. conv(b,b)_k <= t for all k, b >= 0, sum(b) = 1

This directly minimizes the peak! This is actually a better formulation.

Let me reformulate:
Variables: b_0, ..., b_{n-1}, t
Objective: minimize t
Constraints:
- sum(b) = 1
- For each k in [0, 2n-2]: sum_i b_i * b_{k-i} <= t  (quadratic!)
- b_i >= 0

The quadratic constraints make this not an LP but a QP or SOCP... Actually, the constraint `sum_i b_i * b_{k-i} <= t` is quadratic in b. This is a QCQP (quadratically constrained quadratic program), which is hard in general.

However, if we fix b and linearize around it, we can solve an LP. This is essentially the Frank-Wolfe approach.

The linearization of `conv(b,b)[k]` at the current point `a` is:
`conv(a,b)[k] + conv(b-a, b-a)[k] ≈ conv(a,b)[k]` (first-order)

So the LP oracle that the previous program uses (max sum(b) s.t. conv(a,b) <= M) is the correct linearization! The issue is just in the implementation details and step size selection.

Let me think about another approach: **iterative water-filling**.

Algorithm:
1. Start with a distribution a (sum = 1)
2. Compute conv = conv(a, a)
3. Find the peak k* = argmax conv
4. Reduce a at positions that contribute most to conv[k*]
5. Renormalize
6. Repeat

Step 4: conv[k*] = sum_i a_i * a_{k*-i}. The "contributors" are the pairs (i, k*-i). We want to reduce a_i for those i that have large a_{k*-i} (and vice versa).

A natural update: a_i <- a_i * (1 - eta * a_{k*-i}) for all i where k*-i is in range.

This is like the "DPC" (dynamic power control) algorithm from communication theory, or the "power iteration" approach.

Let me also consider: **exp(-beta * conv)** approach.

Set a_i <- exp(-beta * (a*a_conv)_i) where (a*a_conv)_i is somehow the "local convolution pressure" at position i.

Actually, here's a cleaner formulation. Define the "pressure" at position j as:
P(j) = sum_k w_k * a[k-j]

where w_k are weights on the convolution peaks. Then update:
a_j <- a_j * exp(-eta * P(j))

This is like a multiplicative weight update. It naturally maintains non-negativity and we can renormalize.

This is similar to the "multiplicative weights" algorithm for games, which has good convergence guarantees.

Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines: (1) multiplicative weight updates inspired by power control,

def next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def fast_convolve(a, b=None):
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = next_pow2(size)
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def compute_obj_normalized(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2. Assumes a>=0."""
    n = len(a)

    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def mpd_update(a, conv, top_k=20, eta=0.1, temperature=0.1):
    """
    Multiplicative Pressure Descent update.
    For each peak k, the pressure on position j is a[k-j] (the partner).
    Update: a_j <- a_j * exp(-eta * pressure_j) / normalizer
    """
    n = len(a)
    max_val = conv.max()
    if max_val < 1e-15:
        return a.copy()
    
    k = min(top_k, len(conv))
    peak_idx = np.argpartition(conv, -k)[-k:]
    peak_vals = conv[peak_idx]
    
    # Soft-max weighting
    diff = (peak_vals - max_val) / (temperature * max_val + 1e-15)
    weights = np.exp(diff)
    weights /= weights.sum()
    
    pressure = np.zeros(n)
    for i in range(len(peak_idx)):
        if weights[i] < 1e-12:
            continue
        kv = int(peak_idx[i])
        j_start = max(0, kv - n + 1)
        j_end = min(n - 1, kv)
        if j_start > j_end:
            continue
        indices = np.arange(j_start, j_end + 1)
        pressure[indices] += weights[i] * a[kv - indices]
    
    # Multiplicative update
    log_a = np.log(np.maximum(a, 1e-30))
    log_a -= eta * pressure
    # Numerical stability
    log_a -= log_a.max()
    new_a = np.exp(log_a)
    s = new_a.sum()
    if s < 1e-30:
        return a.copy()
    return new_a / s

    """

def build_conv_matrix(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_solve(a, M, time_limit=5.0):
    """LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, method='highs', bounds=bounds,
                  options={'presolve': True, 'time_limit': time_limit})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def lp_polish(a0, time_limit, verbose=False, verbose_every=20):
    """LP-based Frank-Wolfe polishing."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    a = a / s
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        b = lp_solve(a, M, time_limit=min(5.0, time_limit - (time.monotonic() - t0) + t0 - 1.0))
        if b is None or b.sum() < 1e-15:
            break
        b = b / b.sum()
        
        if np.allclose(a, b, atol=1e-10):
            no_improve += 1
            if no_improve > 8:
                break
            # Perturb
            rng = np.random.default_rng(it + 31337)
            a = np.maximum(a + rng.standard_normal(n) * 0.003, 0.0)
            a = a / a.sum()
            conv = fast_convolve(a)
            M = conv.max()
            continue
        
        # Line search
        best_t = 0.0
        best_local = 2.0 * n * M
        for t in [0.05, 0.15, 0.3, 0.5, 0.7, 0.9]:

            if cur_obj < best_obj - 1e-12:

Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines: (1) Multiplicative Pressure Descent (MPD) with adaptive parameters,
(2) multi-peak subgradient with momentum, (3) LP oracle for stalling,

def next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def fast_convolve(a, b=None):
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = next_pow2(size)
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def compute_obj_normalized(a):
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def multi_peak_subgrad(a, top_k=20, temperature=0.1):
    n = len(a)
    conv = fast_convolve(a)
    max_val = conv.max()
    if max_val < 1e-15:
        return np.zeros(n), conv
    k = min(top_k, len(conv))
    peak_idx = np.argpartition(conv, -k)[-k:]
    peak_vals = conv[peak_idx]
    diff = (peak_vals - max_val) / (temperature * max_val + 1e-15)
    weights = np.exp(diff)
    weights /= weights.sum()
    g = np.zeros(n)
    for i in range(len(peak_idx)):
        if weights[i] < 1e-12:
            continue
        kv = int(peak_idx[i])
        j_start = max(0, kv - n + 1)
        j_end = min(n - 1, kv)
        if j_start > j_end:
            continue
        indices = np.arange(j_start, j_end + 1)
        g[indices] += weights[i] * 2.0 * a[kv - indices]

            a = np.maximum(a + noise, 1e-30)
            a = a / a.sum()
            no_improve = 0
            max_stall = int(max_stall * 1.35)
            eta = 0.25
        it += 1
        if verbose and it % verbose_every == 0:
            elapsed = time.monotonic() - t0
            print(f"    MPD it={it}, obj={best_obj:.6f}, eta={eta:.4f}, top_k={top_k}, temp={temperature:.3f}, elapsed={elapsed:.1f}s")
    return best_a, best_obj


def sg_optimize(a0, time_limit, verbose=False, verbose_every=3000):
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    a = a / s
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    lr = 0.008
    mom = 0.92
    v = np.zeros(n)
    no_improve = 0
    max_stall = 4000
    t0 = time.monotonic()
    it = 0
    while time.monotonic() - t0 < time_limit:
        top_k = min(40, 5 + it // 1000)
        temp = 0.05 + 0.1 * min(1.0, it / 8000.0)
        g, conv = multi_peak_subgrad(a, top_k=top_k, temperature=temp)
        v = mom * v + g
        a_new = a - lr * v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s_new
        a = a_new
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                lr = max(1e-5, lr * 0.9)
        if no_improve >= max_stall:
            rng = np.random.default_rng(it + 9999)
            noise = rng.standard_normal(n) * 0.003
            a = np.maximum(a + noise, 0.0)
            a = a / a.sum()
            no_improve = 0
            max_stall = int(max_stall * 1.3)
            lr = 0.008
            v = np.zeros(n)
        it += 1
        if verbose and it % verbose_every == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG it={it}, obj={best_obj:.6f}, lr={lr:.6f}, elapsed={elapsed:.1f}s")
    return best_a, best_obj


def build_conv_matrix(a):
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_solve(a, M, time_limit=5.0):
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, method='highs', bounds=bounds,
                  options={'presolve': True, 'time_limit': time_limit})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def lp_polish(a0, time_limit, verbose=False, verbose_every=20):
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    a = a / s
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    while time.monotonic() - t0 < time_limit - 2:
        remaining = time_limit - (time.monotonic() - t0)
        lp_tl = min(5.0, max(1.0, remaining - 2.0))
        b = lp_solve(a, M, time_limit=lp_tl)
        if b is None or b.sum() < 1e-15:
            break
        b = b / b.sum()
        if np.allclose(a, b, atol=1e-10):
            no_improve += 1
            if no_improve > 10:
                break
            rng = np.random.default_rng(it + 31337)
            a = np.maximum(a + rng.standard_normal(n) * 0.003, 0.0)
            a = a / a.sum()
            conv = fast_convolve(a)
            M = conv.max()
            continue
        best_t = 0.0
        best_local = 2.0 * n * M
        for t in [0.05, 0.1, 0.2, 0.35, 0.5, 0.7, 0.9]:
            cand = (1.0 - t) * a + t * b
            c = fast_convolve(cand)
            lo = 2.0 * n * c.max()
            if lo < best_local:
                best_local = lo
                best_t = t
        if best_t < 0.03:
            no_improve += 1
            if no_improve > 10:
                break
        else:
            a = (1.0 - best_t) * a + best_t * b
            conv = fast_convolve(a)
            M = conv.max()
            cur_obj = 2.0 * n * M
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        it += 1
        if verbose and it % verbose_every == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP it={it}, obj={best_obj:.6f}, elapsed={elapsed:.1f}s")
    return best_a, best_obj


def fourier_init(n, rng, mode='random_phase'):
    freq = np.fft.rfftfreq(n)
    if mode == 'random_phase':
        phase = rng.random(np.size(freq)) * 2 * np.pi
        spectrum = np.exp(1j * phase)
    elif mode == 'lowpass':
        cutoff = 0.3
        spectrum = np.ones(np.size(freq))
        spectrum[freq > cutoff] = 0.0
        phase = rng.random(np.size(freq)) * 2 * np.pi
        spectrum = spectrum * np.exp(1j * phase)
    elif mode == 'gaussian_spectral':
        sigma = 0.15
        amp = np.exp(-0.5 * (freq / sigma) ** 2)
        phase = rng.random(np.size(freq)) * 2 * np.pi
        spectrum = amp * np.exp(1j * phase)
    else:
        spectrum = np.ones(np.size(freq), dtype=complex)
    a = np.fft.irfft(spectrum, n)
    a = np.abs(a)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def generate_structured_inits(n, rng, parent=None):
    inits = {}
    x = np.linspace(0, 1, n)
    inits['uniform'] = np.ones(n) / n
    g = np.exp(-40 * x**2)
    inits['gauss_center'] = g / g.sum()
    g2 = np.exp(-60 * (x - 0.08)**2) + np.exp(-60 * (x - 0.92)**2)
    inits['gauss_two_edge'] = g2 / g2.sum()
    g4 = (np.exp(-80 * (x - 0.05)**2) + np.exp(-80 * (x - 0.35)**2) +
          np.exp(-80 * (x - 0.65)**2) + np.exp(-80 * (x - 0.95)**2))
    inits['w_4bump'] = g4 / g4.sum()
    w = max(1, n // 4)
    ec = np.zeros(n)
    ec[:w] = 1.0
    ec[-w:] = 1.0
    inits['edge_block'] = ec / ec.sum()
    r = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = r / r.sum()
    tri = np.minimum(x, 1 - x) + 0.05
    inits['triangle'] = tri / tri.sum()
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x))
    inits['cosine'] = cos_w
