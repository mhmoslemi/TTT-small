```python
def dikin_step(a, grad, lr):
    """Multiplicative weight update (Dikin step)."""
    # a_i_new = a_i * exp(-lr * grad_i)
    # Then normalize to simplex
    log_weights = -lr * grad
    log_weights -= log_weights.max()  # numerical stability
    weights = np.exp(log_weights)
    a_new = a * weights
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s
```

"""
Minimizes 2n*max(conv(a,a))/(sum(a))^2 using Adam-based multiplicative updates,
log-sum-exp smoothing with adaptive annealing, active-set LP, and peak shaving.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def fast_convolve(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    S = 1 << (size - 1).bit_length()
    A = rfft(a, S)
    return irfft(A * A, S)[:size]


def compute_gradient(a, p):
    """
    Compute grad[i] = sum_k p[k] * (a[k-i] + a[i-k]) for i=0..n-1.
    p has length 2n-1 (the softmax weights over convolution positions).
    """
    n = len(a)
    S = 1 << (3 * n).bit_length()
    A = rfft(a, S)
    P = rfft(p, S)
    
    # T2[i] = (p * a)[i] = sum_k p[k] a[i-k]
    T2 = irfft(P * A, S)[:n]
    
    # T1[i] = sum_k p[k] a[k-i] = sum_j a[j] p[j+i] = (a ⋆ p)[-i]
    cross = irfft(np.conj(A) * P, S)
    T1 = cross[(-np.arange(n)) % S]
    
    return T1 + T2


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u * np.arange(1, n + 1) > (cssv - 1))
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_objective_and_grad(a, beta):
    """Compute smoothed objective value and its gradient."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = beta * conv
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-300:
        obj = conv.max() / 1.0
        # Degenerate: subgradient
        max_idx = np.argmax(conv)
        p = np.zeros(2*n-1)
        p[max_idx] = 1.0
    else:
        obj = (scaled.max() + np.log(p_sum)) / beta
        p /= p_sum
    
    grad = compute_gradient(a, p)
    return obj, grad, conv, p


def adam_dikin_optimize(a0, time_limit, beta_start=5.0, beta_end=300.0,
                        lr=0.05, beta_adam=0.9, beta2_adam=0.999,
                        epsilon=1e-8, seed=42):
    """
    Adam optimizer with multiplicative (Dikin) updates on the simplex.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    t0 = time.monotonic()
    
    # Compute initial
    obj, grad, conv, p = smoothed_objective_and_grad(a, beta_start)
    best_obj = obj
    best_a = a.copy()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)
        beta = beta_start + (beta_end - beta_start) * (frac ** 0.7)
        
        # Adam update
        t += 1
        m = beta_adam * m + (1 - beta_adam) * grad
        v = beta2_adam * v + (1 - beta2_adam) * grad ** 2
        m_hat = m / (1 - beta_adam ** t)
        v_hat = v / (1 - beta2_adam ** t)
        
        # Multiplicative update: a_new ∝ a * exp(-lr * m_hat / sqrt(v_hat))
        update = m_hat / (np.sqrt(v_hat) + epsilon)
        log_weights = -lr * update
        log_weights -= log_weights.max()  # stability
        weights = np.exp(log_weights)
        a_new = a * weights
        a_new = np.maximum(a_new, 1e-30)
        a = a_new / a_new.sum()
        
        # Evaluate
        obj, grad, conv, p = smoothed_objective_and_grad(a, beta)
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Periodic perturbation
        if no_improve > 500:
            noise_scale = 0.01 * (1 - 0.5 * frac)
            noise = rng.standard_normal(n) * noise_scale
            a = project_simplex(a + noise)
            m *= 0.5
            v *= 0.5
            no_improve = 0
    
    return best_a, best_obj


def lp_direction_active_set(a, M, n, top_k=20, time_limit=10):
    """
    Solve LP with only top-k active constraints.
    max sum(b) s.t. b>=0, (a*b)[k] <= M for k in active_set.
    """
    conv_ab_info = None  # We'll build the constraint matrix for active constraints only
    
    # Compute current convolution to find active set
    conv = fast_convolve(a)
    active_set = np.argsort(conv)[-top_k:]
    
    # Build constraint matrix: for each active constraint k, the row is the
    # vector of a values that participate in conv_a*b[k]
    # (a*b)[k] = sum_i a[i] * b[k-i] (valid range)
    # = sum_i b[i] * a[k-i] (by symmetry of the constraint structure)
    
    A_ub = np.zeros((len(active_set), n))
    for idx, k in enumerate(active_set):
        k = int(k)
        # (a * b)[k] = sum_i a[k-i] * b[i] for valid i
        # a[k-i] is valid when 0 <= k-i < n, i.e., k-n+1 <= i <= k
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        A_ub[idx, i_start:i_end+1] = a[k - i_end:k - i_start + 1][::-1]
    
    c_obj = -np.ones(n)
    b_ub = np.full(len(active_set), M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': time_limit})
    except Exception:
        return None
    
    if not result.success:
        return None
    
    b = np.maximum(result.x, 0)
    b_sum = b.sum()
    if b_sum < 1e-10:
        return None
    
    return b / b_sum


def peak_shaving_targeted(a, time_limit, seed=0):
    """
    Targeted peak reduction: iteratively identify the max peak and move mass
    from contributing positions to non-contributing positions.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    iters = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        max_idx = int(np.argmax(conv))
        max_val = conv[max_val if False else max_idx]
        
        # Find contributing pairs for this peak
        i_start = max(0, max_idx - n + 1)
        i_end = min(n - 1, max_idx)
        
        contributions = np.array([a[i] * a[max_idx - i] for i in range(i_start, i_end + 1)])
        
        # Find the pair with max contribution
        max_c_idx = np.argmax(contributions)
        i_max = i_start + max_c_idx
        j_max = max_idx - i_max
        
        # Try to reduce by moving mass from i_max to a random position
        improved = False
        for _ in range(5):
            j_new = rng.integers(0, n)
            if j_new == i_max:
                continue
            
            for frac in [0.02, 0.05, 0.1, 0.15]:
                amt = a[i_max] * frac
                if amt < 1e-10:
                    break
                a_try = a.copy()
                a_try[i_max] -= amt
                a_try[j_new] += amt
                conv_try = fast_convolve(a_try)
                if conv_try.max() < best_obj - 1e-10:
                    best_obj = conv_try.max()
                    best_a = a_try.copy()
                    a = a_try.copy()
                    improved = True
                    break
        
        iters += 1
        if not improved and iters % 1000 == 0:
            # Random perturbation
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 1e-6:
                amt = a[i] * 0.1
                a[i] -= amt
                a[j] += amt
                a = np.maximum(a, 0)
                a = a / a.sum()
    
    return best_a, best_obj
