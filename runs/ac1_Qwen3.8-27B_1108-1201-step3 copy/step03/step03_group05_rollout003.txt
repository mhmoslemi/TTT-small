import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def gradient_max_conv(a, conv, beta):
    """Gradient of logsumexp(beta * conv) w.r.t. a, using FFT."""
    n = len(a)
    scaled = beta * conv - beta * conv.max()
    p = np.exp(scaled)
    p /= p.sum()
    w = conv * p
    fft_size = 1 << (3*n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w, fft_size)
    cross = np.fft.irfft(A * np.conj(W), fft_size)
    return 2.0 * cross[:n]

def gradient_pnorm(a, conv, p):
    """Gradient of (sum conv_k^p)^(1/p) = ||conv||_p."""
    n = len(a)
    conv_p = conv.copy()
    conv_p[conv_p < 0] = 0  # safety
    weighted = conv_p ** (p-1)
    weighted /= weighted.sum()  # normalize to avoid overflow
    w = conv_p ** (p-1) * conv_p  # = conv^p, but we only need the shape
    # Actually, grad of sum(conv_k^p) w.r.t. a_j = p * sum_k conv_k^(p-1) * a_{k-j}
    # So grad_j = p * 2 * sum_k conv_k^(p-1) * a_{k-j}
    # Let w_k = conv_k^(p-1), then grad_j = 2*p * (a * w)_j (cross-correlation)
    w = conv_p ** (p-1)
    fft_size = 1 << (3*n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w, fft_size)
    cross = np.fft.irfft(A * np.conj(W), fft_size)
    return 2.0 * p * cross[:n]

        beta = beta_init + (beta_end - beta_init) * frac
        lr = lr_init * (1.0 - 0.5 * frac)

        y = np.maximum(y, 0)
        sy = y.sum()

            a = project_simplex(a + noise)
            v *= 0.5
            no_improve = 0

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def grad_logsumexp(a, conv, beta):
    """Gradient of logsumexp(beta*conv)/1 w.r.t. a via FFT cross-correlation."""
    n = len(a)
    scaled = beta * conv
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p /= p_sum
    w = conv * p
    fft_size = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w, fft_size)
    cross = np.fft.irfft(A * np.conj(W), fft_size)
    return 2.0 * cross[:n]


def grad_pnorm(a, conv, p):
    """Gradient of ||conv||_p = (sum conv_k^p)^(1/p) w.r.t. a."""
    n = len(a)
    conv_pos = np.maximum(conv, 0)
    w = conv_pos ** (p - 1)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv_pos)] = 1.0
    else:
        w /= w_sum
    fft_size = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w, fft_size)
    cross = np.fft.irfft(A * np.conj(W), fft_size)
    return 2.0 * cross[:n]


def objective_value(a):

    n = len(a)
    conv = fast_convolve(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * n * conv.max() / (s * s)


def optimize_nesterov_smoothed(a0, time_limit, beta_init=50.0, beta_end=500.0,
                                lr_init=0.012, momentum=0.93, seed=0,

        current_beta = min(current_beta, beta_end)

        lr = lr_init * (1.0 - 0.4 * frac) + 0.0005

        # Nesterov lookahead
        y = a + momentum * v
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy > 1e-15:
            y /= sy
        else:
            y = a.copy()

        v = momentum * v - lr * grad
        a = project_simplex(a + v)

        conv = fast_convolve(a)
        cur_obj = conv.max()
        total_iters += 1

            no_improve = 0

            a = project_simplex(a + noise)

def peak_equalization(a0, time_limit, seed=0):
    """
    Iterative peak reduction: identify top peaks, transfer mass from
    high-sensitivity positions to low-sensitivity ones.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)

        threshold = max_val * 0.95

        if len(peak_indices) == 0:
            break

        for k in peak_indices:
            k = int(k)
            if k < 0 or k >= 2 * n - 1:
                continue
            # Sensitivity: d(conv_k)/da_j = 2*a_{k-j} (if valid index)
            sens = np.zeros(n)
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    sens[j] = 2.0 * a[idx]

            # Find the most sensitive position (largest contribution to this peak)
            max_sens_idx = np.argmax(sens)
            if sens[max_sens_idx] < 1e-10:
                continue

            # Find a low-sensitivity position to receive mass
            # Prefer positions that don't contribute to any current peak
            low_sens_candidates = np.where(sens < sens.mean())[0]
            if len(low_sens_candidates) == 0:
                low_sens_candidates = np.argsort(sens)[:max(1, n // 10)]

            # Try several transfer amounts and target positions
            best_transfer = None
            best_new_obj = max_val

            for target in low_sens_candidates[:5]:
                target = int(target)
                if target == max_sens_idx:
                    continue
                for frac in [0.05, 0.1, 0.15, 0.2, 0.3]:
                    amt = a[max_sens_idx] * frac
                    if amt < 1e-10:
                        break
                    a_try = a.copy()
                    a_try[max_sens_idx] -= amt
                    a_try[target] += amt
                    s_try = a_try.sum()
                    if s_try < 1e-10:
                        continue
                    a_try /= s_try
                    conv_try = fast_convolve(a_try)
                    obj_try = conv_try.max()
                    if obj_try < best_new_obj:
                        best_new_obj = obj_try
                        best_transfer = a_try.copy()

            if best_transfer is not None and best_new_obj < max_val - 1e-12:
                a = best_transfer
                if best_new_obj < best_obj:
                    best_obj = best_new_obj
                    best_a = a.copy()
                improved = True

        if not improved:
            # Random perturbation to escape
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 1e-6:
                for frac in [0.02, 0.05, 0.1, 0.2]:
                    amt = a[i] * frac
                    a_try = a.copy()
                    a_try[i] -= amt
                    a_try[j] += amt
                    a_try = np.maximum(a_try, 0)
                    s_try = a_try.sum()
                    if s_try < 1e-10:
                        continue
                    a_try /= s_try
                    conv_try = fast_convolve(a_try)
                    if conv_try.max() < max_val * 0.999:
                        a = a_try
                        if conv_try.max() < best_obj:
                            best_obj = conv_try.max()
                            best_a = a.copy()
                        break

    return best_a, best_obj


def lp_direction_fast(a, M, n, time_limit=5):
    """
    Solve LP: max sum(b) s.t. b>=0, (a*b)[k] <= M.
    Returns normalized b or None. Time-limited.
    """
    c_obj = -np.ones(n)

    # Build sparse constraint matrix
    # A_ub has shape (2n-1, n), A_ub[k, i] = a[k-i] if 0<=k-i<n
    from scipy.sparse import csr_matrix
    rows = []
    cols = []
    vals = []
    for i in range(n):
        k_start = i
        k_end = min(2 * n - 1, i + n)
        length = k_end - k_start
        rows.extend(range(k_start, k_end))
        cols.extend([i] * length)
        vals.extend(a[:length])

    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    t0 = time.monotonic()
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs',
                         options={'presolve': True, 'time_limit': max(1, time_limit - 0.5)})
    except Exception:
        return None

    if not result.success or result.x is None:
        return None

    b = np.maximum(result.x, 0)
    b_sum = b.sum()
    if b_sum < 1e-10:
        return None
    return b / b_sum

def spectral_init(n, parent=None, seed=0):
    """
    Generate initialization using spectral considerations.
    Aim for flat |FFT(a)|^2 spectrum (maximizes spectral flatness).
    """
    rng = np.random.default_rng(seed)

    # Start with a "complementary" pattern to uniform
    # Use inverse FFT of a target flat spectrum with random phases
    fft_size = 1 << (2 * n - 1).bit_length()
    # Target: approximately flat power spectrum
    target_power = np.ones(fft_size // 2 + 1)
    # Boost low frequencies slightly (to ensure non-negativity after IFFT)
    target_power[:max(1, fft_size // 20)] *= 2.0

    # Random phases
    phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    spectrum = np.sqrt(target_power) * np.exp(1j * phases)
    # Make Hermitian symmetric
    full_spectrum = np.zeros(fft_size, dtype=complex)
    full_spectrum[:fft_size // 2 + 1] = spectrum
    for k in range(1, fft_size // 2):
        full_spectrum[fft_size - k] = np.conj(full_spectrum[k])

    a_candidate = np.fft.irfft(full_spectrum, fft_size)[:n]
    a_candidate = np.maximum(a_candidate, 0)
    s = a_candidate.sum()
    if s < 1e-10:
        return np.ones(n) / n
    a_candidate /= s

    # Blend with parent if available
    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            a_candidate = 0.7 * a_candidate + 0.3 * parent
        else:
            parent_interp = np.interp(np.linspace(0, 1, n),
                                       np.linspace(0, 1, len(parent)), parent)
            a_candidate = 0.7 * a_candidate + 0.3 * parent_interp

    a_candidate = np.maximum(a_candidate, 0)
    s = a_candidate.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a_candidate / s


def generate_diverse_inits(n, parent, rng, num=6):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    bell /= bell.sum()
    inits['bell'] = bell

    # Edge-concentrated (bimodal)
    edge = max(3, n // 6)
    bi = np.zeros(n)
    bi[:edge] = 1.0

    bi /= bi.sum()
    inits['bimodal'] = bi

    # Raised cosine
    cos_w = 0.5 * (1 + np.cos(2 * np.pi * x / max(1, n - 1)))
    cos_w = np.maximum(cos_w, 1e-8)
    cos_w /= cos_w.sum()
    inits['raised_cos'] = cos_w

    # Dirichlet with low concentration
    inits['dirichlet_low'] = rng.dirichlet(np.ones(n) * 0.3)

    # Parent-based

        if len(parent) == n:
            inits['parent'] = parent.astype(np.float64).copy()
        elif len(parent) > n:
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
            p = np.maximum(p, 0)
            ps = p.sum()
            if ps > 1e-10:
                inits['parent_ds'] = p / ps
        else:
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
            p = np.maximum(p, 0)
            ps = p.sum()
            if ps > 1e-10:
                inits['parent_up'] = p / ps

    # Select top num
    keys = list(inits.keys())
    if len(keys) > num:
        # Prioritize parent and spectral
        priority = ['parent', 'parent_ds', 'parent_up', 'spectral', 'bell', 'raised_cos']
        keys = [k for k in priority if k in keys] + [k for k in keys if k not in priority]
        keys = keys[:num]

    return {k: inits[k] for k in keys}

            break

    # Ensure current best is included
    if best_sequence is not None:
        candidates.insert(0, (len(best_sequence), best_sequence.copy()))

    # Deduplicate by length
    deduped = []
    seen_lens = set()
    for n, a in candidates:
        if n not in seen_lens:
            deduped.append((n, a))
            seen_lens.add(n)
    candidates = deduped[:4]

    remaining = deadline - time.monotonic() - 200.0
    n_candidates = len(candidates)
    per_candidate = max(50.0, remaining / max(1, n_candidates))

    for ci, (n, a0) in enumerate(candidates):
        if time.monotonic() > deadline - 200:
            break
        t_alloc = min(per_candidate, 600.0)
        if t_alloc < 30:
            break

        print(f"  Candidate {ci}: n={n}", flush=True)

        # Strategy A: High-beta gradient (40% of time)
        t_a = t_alloc * 0.40
        if t_a > 10:
            a_a, obj_a = optimize_nesterov_smoothed(
                a0, t_a, beta_init=80, beta_end=600,
                lr_init=0.008, momentum=0.94,
                perturb_every=300, perturb_scale=0.002,
                seed=rng.integers(0, 100000))
            score_a = float(evaluate_sequence(a_a.tolist()))
            if np.isfinite(score_a) and score_a < best_score:
                best_score = score_a
                best_sequence = a_a.copy()
                print(f"    Grad(soft): {score_a:.6f}", flush=True)

        # Strategy B: p-norm gradient (25% of time)
        t_b = t_alloc * 0.25
        if t_b > 10:
            a_b, obj_b = optimize_pnorm(
                a_a, t_b, p_init=30, p_end=300,
                lr_init=0.006, momentum=0.95,
                seed=rng.integers(0, 100000))
            score_b = float(evaluate_sequence(a_b.tolist()))
            if np.isfinite(score_b) and score_b < best_score:
                best_score = score_b
                best_sequence = a_b.copy()
                print(f"    pNorm: {score_b:.6f}", flush=True)

        # Strategy C: Gradient + LP (20% of time)
        t_c = t_alloc * 0.20
        if t_c > 15:
            a_c, obj_c = optimize_with_lp(
                a_b, t_c, seed=rng.integers(0, 100000),
                lp_every=300, lp_timeout=3)
            score_c = float(evaluate_sequence(a_c.tolist()))
            if np.isfinite(score_c) and score_c < best_score:
                best_score = score_c
                best_sequence = a_c.copy()
                print(f"    Grad+LP: {score_c:.6f}", flush=True)

        # Strategy D: Peak equalization (15% of time)
        t_d = t_alloc * 0.15
        if t_d > 10:
            a_d, obj_d = peak_equalization(
                a_c, t_d, seed=rng.integers(0, 100000))
            score_d = float(evaluate_sequence(a_d.tolist()))
            if np.isfinite(score_d) and score_d < best_score:
                best_score = score_d
                best_sequence = a_d.copy()
                print(f"    PeakEq: {score_d:.6f}", flush=True)

    # Phase 3: Polish (200s)
    print("=== Phase 3: Polish ===", flush=True)
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 15.0
        if remaining > 30:
            # Polishing gradient with very high beta
            t_polish = remaining * 0.35
            if t_polish > 10:
                a_p, obj_p = optimize_nesterov_smoothed(
                    best_sequence, t_polish,
                    beta_init=200, beta_end=1000,
                    lr_init=0.003, momentum=0.96,
                    perturb_every=200, perturb_scale=0.001,
                    seed=9999)
                score_p = float(evaluate_sequence(a_p.tolist()))
                if np.isfinite(score_p) and score_p < best_score:
                    best_score = score_p
                    best_sequence = a_p.copy()
                    print(f"  PolishGrad: {best_score:.6f}", flush=True)

            # Peak equalization
            remaining = deadline - time.monotonic() - 15.0
            t_pe = remaining * 0.30
            if t_pe > 10:
                a_pe, obj_pe = peak_equalization(
                    best_sequence, t_pe, seed=7777)
                score_pe = float(evaluate_sequence(a_pe.tolist()))
                if np.isfinite(score_pe) and score_pe < best_score:
                    best_score = score_pe
                    best_sequence = a_pe.copy()
                    print(f"  PolishPeak: {best_score:.6f}", flush=True)

            # Final ultra-fine gradient
            remaining = deadline - time.monotonic() - 10.0
            t_fin = remaining * 0.35
            if t_fin > 5:
                a_f, obj_f = optimize_nesterov_smoothed(
                    best_sequence, t_fin,
                    beta_init=500, beta_end=2000,
                    lr_init=0.001, momentum=0.97,
                    perturb_every=100, perturb_scale=0.0005,
                    seed=4321)
                score_f = float(evaluate_sequence(a_f.tolist()))
                if np.isfinite(score_f) and score_f < best_score:
                    best_score = score_f
                    best_sequence = a_f.copy()
                    print(f"  FinalGrad: {best_score:.6f}", flush=True)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
        best_sequence = np.clip(best_sequence, 0, 1000)

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_sequence)})", flush=True)
    return [float(x) for x in best_sequence]
