def conv_peak_and_weights(convo, top_k=50, temperature=0.01):
    """Get peak and softmax weights over top-k peaks."""
    k = min(top_k, len(convo))
    idx = np.argpartition(convo, -k)[-k:]

    weights /= weights.sum()

    # Build constraint matrix
    # Number of constraints: 2n-1
    # Number of variables: n
    A_ub = np.zeros((2 * n - 1, n))

        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[k, j] = f[k - j]

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]
