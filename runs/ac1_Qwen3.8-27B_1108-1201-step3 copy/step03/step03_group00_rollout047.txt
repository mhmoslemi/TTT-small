import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns optimal g or None if infeasible/failed."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def golden_section_minimize(func, lo, hi, max_iter=50):
    """Minimize a unimodal function on [lo, hi] using golden section search."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_point = hi - gr * (hi - lo)
    b_point = lo + gr * (hi - lo)
    fa = func(a_point)
    fb = func(b_point)
    for _ in range(max_iter):
        if fa < fb:
            hi = b_point
            b_point = a_point
            fb = fa
            a_point = hi - gr * (hi - lo)
            fa = func(a_point)
        else:
            lo = a_point
            a_point = b_point
            fa = fb
            b_point = lo + gr * (hi - lo)
            fb = func(b_point)
    return (lo + hi) / 2.0


def coord_descent_optimize(f0, time_limit=60.0, max_sweeps=200):
    """Coordinate descent: optimize one coordinate at a time via golden section search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    sweep_no_improve = 0

    for sweep in range(max_sweeps):
        if time.monotonic() - t0 > time_limit:
            break

        sweep_improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[i]
            # Determine search range for coordinate i
            mean_val = 1.0 / n
            if old_val < 1e-12:
                lo = 0.0
                hi = 5.0 * mean_val
            else:
                lo = 0.0
                hi = max(3.0 * old_val, 5.0 * mean_val)

            def obj_at_t(t):
                f[i] = t
                s_new = f.sum()
                if s_new < 1e-15:
                    return 1e10
                f_norm = f / s_new
                return fast_convolve(f_norm).max()

            # Quick check: evaluate at a few points first
            candidates = [0.0, old_val, 0.5 * hi, hi]
            best_c = old_val
            best_cv = obj_at_t(old_val)
            for cv in candidates:
                val = obj_at_t(cv)
                if val < best_cv:
                    best_cv = val
                    best_c = cv

            # Golden section in the best bracket
            if best_c > 0 and best_c < hi:
                # Narrow the bracket
                m1 = max(0.0, best_c * 0.5)
                m2 = min(hi, best_c * 2.0)
                opt = golden_section_minimize(obj_at_t, m1, m2, max_iter=35)
                val_opt = obj_at_t(opt)
                if val_opt < best_cv:
                    best_c = opt
                    best_cv = val_opt
            else:
                opt = golden_section_minimize(obj_at_t, 0.0, hi, max_iter=35)
                val_opt = obj_at_t(opt)
                if val_opt < best_cv:
                    best_c = opt
                    best_cv = val_opt

            if best_c != old_val:
                f[i] = best_c
                sweep_improved = True

        # Normalize and check improvement
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s < 1e-15:
            break
        f = f / s
        conv = fast_convolve(f)
        co = conv.max()
        if co < best_obj - 1e-12:
            best_obj = co
            best_f = f.copy()
            sweep_no_improve = 0
        else:
            sweep_no_improve += 1
            if sweep_no_improve >= 5:
                break

    return best_f, best_obj


def sg_polish(f0, time_limit=60.0, lr=0.003, mom=0.94, top_k=48, temp=0.00005):
    """Subgradient descent with multi-peak softmax weighting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp

    while time.monotonic() - t0 < time_limit:
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-13:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9995
        cur_temp *= 1.004

        if no_improve > 500:
            rng = np.random.default_rng(int(time.monotonic() * 100000) % 1000000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0005)
            no_improve = 0

    return best_f, best_obj


def fw_lp_optimize(f0, time_limit=120.0, verbose=False):
    """Frank-Wolfe with LP oracle and adaptive constraint relaxation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    m_exp = 1.0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 2:
            break

        g = solve_lp_direction(f, M * m_exp)
        if g is None:
            m_exp *= 1.2
            if m_exp > 5.0:
                m_exp = 1.0
                continue
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 50:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue

        # Golden section line search
        def line_obj(alpha):
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                return 1e10
            ft = ft / st
            return fast_convolve(ft).max()

        best_alpha = golden_section_minimize(line_obj, 0.0, 1.0, max_iter=40)
        best_val = line_obj(best_alpha)

        if best_alpha > 1e-10 and best_val < M:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-13:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 50 == 0:
                    print(f"    FW-LP iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 4.0)

        if no_improve > 80:
            rng = np.random.default_rng(it * 7 + 13)
            scale = 0.002 + 0.004 * (no_improve // 100)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    x = np.arange(n, dtype=np.float64)
    # Edge-concentrated
    edge_w = np.minimum(x, n - 1 - x)
    edge_w = 1.0 / (edge_w + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits.append(g / g.sum())

    # Cosine structure
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.1
    inits.append(c1 / c1.sum())

    # Two-peak
    peak1 = n // 3
    peak2 = 2 * n // 3
    two_peak = np.exp(-0.5 * ((x - peak1) / (0.15 * n)) ** 2) + np.exp(-0.5 * ((x - peak2) / (0.15 * n)) ** 2)
    two_peak = two_peak / two_peak.sum()
    inits.append(two_peak)

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits

        n = 384
        if parent is not None:
            best_seq = parent.copy()
        else:
            best_seq = np.ones(n) / n
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
