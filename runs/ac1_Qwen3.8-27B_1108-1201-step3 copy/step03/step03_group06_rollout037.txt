```python
Hybrid optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses LP-based direction finding with line search, subgradient polishing,
and structured restarts. Focuses on the parent sequence length.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def obj_norm(a):
    """Objective with sum(a)=1: 2n * max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = int(np.max(rho[cond]))
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)

    """Build constraint matrix for conv(a,b)[k] <= M."""
    n = len(a)
    C = np.zeros((len(lags), n))
    for idx, k in enumerate(lags):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_min > i_max:
            continue
        idx_arr = np.arange(i_min, i_max + 1)
        C[idx, idx_arr] = a[k - idx_arr]
    return C

def solve_lp(a, M, lags=None):
    """Solve LP: max sum(b) s.t. b>=0, conv(a,b)[k]<=M."""
    n = len(a)
    if lags is None:
        lags = list(range(2 * n - 1))
    C = build_A_ub(a, lags)
    if C.shape[0] == 0:
        return None
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=C, b_ub=np.full(len(lags), M), bounds=bounds,
                      method='highs', options={'presolve': True, 'dual_feasibility_tolerance': 1e-10})
        if res.success and res.x is not None:
            b = res.x
            if b.sum() > 1e-15:
                return b / b.sum()
    except Exception:
        pass
    return None

def line_search(a, b, n_pts=30):
    """Line search along segment from a to b."""
    best_t = 0.0
    best_obj = obj_norm(a)
    best_a = a.copy()
    for t in np.linspace(0.05, 1.0, n_pts):
        a_t = (1 - t) * a + t * b
        a_t = np.maximum(a_t, 0.0)
        s = a_t.sum()
        if s < 1e-15:
            continue
        a_t = a_t / s
        o = obj_norm(a_t)
        if o < best_obj:
            best_obj = o
            best_t = t
            best_a = a_t
    return best_a, best_obj

def subgradient_step(a, step_size):
    """One subgradient step with exact max."""
    n = len(a)
    conv = fast_convolve(a)
    max_val = conv.max()
    active = np.where(conv >= max_val - 1e-12)[0]
    
    # Subgradient: average of gradients at active lags
    grad = np.zeros(n)
    for k in active:
        for i in range(n):
            ki = k - i
            if 0 <= ki < n:
                grad[i] += a[ki]
            ik = i - k
            if 0 <= ik < n:
                grad[i] += a[ik]
    grad /= len(active)
    
    # Projected subgradient step
    a_new = project_simplex(a - step_size * grad)
    return a_new, max_val

def smooth_obj_and_grad(a, beta=500.0):
    """Log-sum-exp smoothed objective and gradient."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = beta * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    
    obj = 2.0 * n * (conv.max() + np.log(w_sum) / beta)
    
    # Gradient: weighted sum of individual gradients
    grad = np.zeros(n)
    for k in range(len(conv)):
        if w[k] < 1e-15:
            continue
        for i in range(n):
            ki = k - i
            if 0 <= ki < n:
                grad[i] += w[k] * a[ki]
            ik = i - k
            if 0 <= ik < n:
                grad[i] += w[k] * a[ik]
    
    return obj, grad, conv
```

Wait, the gradient computation in the smooth version is O(n^2 * 2n) which is too slow for n=320. Let me use FFT.

The gradient of conv(a,a)[k] w.r.t. a[i] is:
- a[k-i] if 0 <= k-i < n (from the term a[i] * a[k-i])
- a[i-k] if 0 <= i-k < n (from the term a[i-k] * a[i])
- Plus a[i] if k-i = i (i.e., k = 2i), because of the a[i]^2 term

Wait, let me be more careful. conv(a,a)[k] = sum_j a[j] * a[k-j].

d/d(a[i]) conv(a,a)[k] = sum_j (da[j]/da[i]) * a[k-j] + sum_j a[j] * da[k-j]/da[i]
= a[k-i] (if 0 <= k-i < n) + a[i-k] (if 0 <= i-k < n)

Note: when i = k-i (i.e., k = 2i), both terms give a[i], so the gradient contribution is 2*a[i]. But that's already captured by the formula above: a[k-i] = a[i] and a[i-k] = a[i], so the total is 2*a[i]. Good.

So the gradient is:
grad[i] = sum_k w[k] * (a[k-i] * [0<=k-i<n] + a[i-k] * [0<=i-k<n])

This can be computed via FFT:
- First term: sum_k w[k] * a[k-i] for valid k,i. This is a cross-correlation.
- Second term: sum_k w[k] * a[i-k] for valid k,i. This is a convolution.

Let me use FFT for both:

```python
def smooth_obj_and_grad_fast(a, beta=500.0):

    fft_size = 1 << (2 * n).bit_length()
    A = np.fft.fft(a, fft_size)
    W = np.fft.fft(w, fft_size)
    
    # First term: sum_k w[k] * a[k-i] = cross-correlation
    # cross[i] = sum_k w[k] * a[k-i]
    # = ifft(conj(fft(w)) * fft(a))[i] ... let me verify
    # Cross-correlation: (w ⋆ a)[i] = sum_k w[k] * conj(a[k-i])
    # For real: sum_k w[k] * a[k-i]
    # In frequency: ifft(conj(W) * A)
    # Actually: DFT of cross-correlation = conj(W) * A
    cross = np.fft.ifft(np.conj(W) * A).real[:n]
    
    # Second term: sum_k w[k] * a[i-k] = convolution (w * a)[i]
    # In frequency: ifft(W * A)
    conv_wa = np.fft.ifft(W * A).real[:n]
    
    grad = cross + conv_wa
```

Wait, I need to be careful about the convolution vs cross-correlation definitions.

Let me define:
- (f * g)[n] = sum_k f[k] * g[n-k] (convolution)
- (f ⋆ g)[n] = sum_k f[k] * g[n+k] = (f * g_flipped)[n] (cross-correlation)

We want:
- Term 1: sum_k w[k] * a[k-i]. Let m = k-i, k = m+i. sum_m w[m+i] * a[m]. This is sum_m a[m] * w[m+i] = (a ⋆ w)[i] (cross-correlation of a and w at lag i).
  In FFT: ifft(A * conj(W))[i]

- Term 2: sum_k w[k] * a[i-k]. This is (w * a)[i] (convolution of w and a at position i).
  In FFT: ifft(W * A)[i]

So grad[i] = (a ⋆ w)[i] + (w * a)[i] = ifft(A * conj(W))[i] + ifft(W * A)[i]

Since a and w are real, conj(W) = W.conj(), and ifft(A * conj(W)) is the cross-correlation.

Let me verify with a small example. Let a = [1, 2, 3], w = [0.1, 0.2, 0.3, 0.4] (length 4, like conv output for n=3 which has length 5... hmm this gets complicated with different lengths).

Actually, for the auto-convolution of a length-n sequence, conv has length 2n-1. So w has length 2n-1 and a has length n. The FFT sizes need to accommodate both.

Let me use fft_size >= max(2n-1 + n - 1, ...) = 3n-2. Actually, for the cross-correlation of a (length n) and w (length 2n-1), the result has length n + 2n-1 - 1 = 3n-2. But we only need the first n values. So fft_size >= 3n-2 suffices.

For the convolution of w (length 2n-1) and a (length n), same thing: result length 3n-2.

So fft_size = next_power_of_2(3*n) should work.

Let me rewrite:

```python
def smooth_obj_and_grad_fast(a, beta=500.0):
```

This should be O(n log n) per evaluation.

Now, let me put together the full algorithm. Given the time budget of 1800 seconds, I'll allocate:

- Phase 1 (600s): LP-based optimization from parent with line search
- Phase 2 (400s): Smooth gradient descent with momentum (FISTA-like)
- Phase 3 (300s): Subgradient + LP alternation
- Phase 4 (200s): Try different lengths and restarts
- Phase 5 (180s): Final polish with very small steps

Let me also add a key trick: **warm-starting the LP**. After each iteration, the solution b is close to the previous b, so I can potentially speed up the LP. However, scipy's linprog doesn't support warm-starting directly. So I'll skip this.

Another trick: **constraint relaxation**. Instead of using all 2n-1 constraints, identify the "critical" ones (those closest to being violated) and only use those. This reduces the LP size.

Let me implement a "critical constraint" selection:
- At current a, compute conv(a,a). The constraints are conv(a,b)[k] <= M for all k.
- The "tightness" of constraint k depends on a. Constraints where a has mass are tighter.
- Heuristic: use the top 50% of lags by |conv(a,a)[k]|, or all lags where conv(a,a)[k] > 0.5*M.

Actually, for n=320, 639 constraints is fine for HiGHS. Let me just use all constraints.

```python
"""
Minimizes 2n*max(conv(a,a))/(sum(a))^2 via LP direction finding with line search,
smoothed gradient descent, and subgradient polishing.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def obj_norm(a):
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = int(np.max(rho[cond]))
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)

def solve_lp_direction(a, M):
    n = len(a)
    num_k = 2 * n - 1
    # Build constraint matrix
    C = np.zeros((num_k, n))
    for k in range(num_k):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_min <= i_max:
            idx = np.arange(i_min, i_max + 1)
            C[k, idx] = a[k - idx]
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=C, b_ub=np.full(num_k, M), bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.success:
            b = res.x
            if b is not None and b.sum() > 1e-15:
                return b / b.sum()
    except Exception:
        pass
    return None

    best_obj = obj_norm(a)

        if s < 1e-15:

        a_t = a_t / s
        o = obj_norm(a_t)
        if o < best_obj:
            best_obj = o
            best_t = t
            best_a = a_t
    return best_a, best_obj

def smooth_grad(a, beta=1000.0):
    n = len(a)
    conv = fast_convolve(a)
    scaled = beta * np.asarray(conv, dtype=np.float64)
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w = w / w_sum
    obj = 2.0 * n * (conv.max() + np.log(max(w_sum, 1e-300)) / beta)
    
    fft_size = 1
    while fft_size < 3 * n:
        fft_size *= 2
    A = np.fft.fft(np.asarray(a, dtype=np.float64), fft_size)
    W = np.fft.fft(w, fft_size)
    cross = np.fft.ifft(A * np.conj(W)).real[:n]
    conv_wa = np.fft.ifft(A * W).real[:n]
    grad = cross + conv_wa
    return obj, grad
```

```python
"""
Minimizes 2n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based direction finding with full constraint set and line search
2. Smoothed gradient descent with adaptive log-sum-exp parameter
3. Subgradient method with Polyak step size
4. Multi-phase search with structured restarts
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_norm(a):
    """Objective assuming sum(a)=1: 2n * max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = int(np.max(rho[cond]))
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(a, M, top_k=None):
    """Solve LP: max sum(b) s.t. b>=0, conv(a,b)[k]<=M."""
    n = len(a)
    num_k = 2 * n - 1

    if top_k is not None and top_k < num_k:
        conv_aa = fast_convolve(a)
        peaks = np.argpartition(conv_aa, -top_k)[-top_k:]
        lags = peaks.tolist()
    else:
        lags = list(range(num_k))

    C = np.zeros((len(lags), n))
    for idx, k in enumerate(lags):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_min <= i_max:
            idx_arr = np.arange(i_min, i_max + 1)
            C[idx, idx_arr] = a[k - idx_arr]

    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=C, b_ub=np.full(len(lags), M), bounds=bounds,
                      method='highs', options={'presolve': True,
                                               'dual_feasibility_tolerance': 1e-10})
        if res.success and res.x is not None:
            b = res.x
            b_sum = b.sum()
            if b_sum > 1e-15:
                return b / b_sum
    except Exception:
        pass
    return None


def line_search(a, b, n_pts=40):
    """Find best t in (0,1] minimizing obj along (1-t)*a + t*b."""
    best_obj = obj_norm(a)
    best_a = a.copy()
    for t in np.linspace(0.05, 1.0, n_pts):
        a_t = (1.0 - t) * a + t * b
        a_t = np.maximum(a_t, 0.0)
        s = a_t.sum()
        if s < 1e-15:
            continue
        a_t = a_t / s
        o = obj_norm(a_t)
        if o < best_obj:
            best_obj = o
            best_a = a_t
    return best_a, best_obj


def smooth_obj_grad(a, beta=1000.0):
    """Log-sum-exp smoothed objective and gradient via FFT."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = beta * np.asarray(conv, dtype=np.float64)
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w = w / w_sum

    obj = 2.0 * n * (conv.max() + np.log(max(w_sum, 1e-300)) / beta)

    fft_size = 1
    while fft_size < 3 * n:
        fft_size *= 2
    A = np.fft.fft(np.asarray(a, dtype=np.float64), fft_size)
    W = np.fft.fft(w, fft_size)
    cross = np.fft.ifft(A * np.conj(W)).real[:n]
    conv_wa = np.fft.ifft(A * W).real[:n]
    grad = cross + conv_wa
    return obj, grad


def subgradient_exact(a):
    """Exact subgradient of max conv(a,a) at active lags."""
    n = len(a)
    conv = fast_convolve(a)
    max_val = conv.max()
    active = np.where(conv >= max_val - 1e-12)[0]

    indicator = np.zeros(len(conv))
    indicator[active] = 1.0 / len(active)

    fft_size = 1
    while fft_size < 3 * n:
        fft_size *= 2
    A = np.fft.fft(np.asarray(a, dtype=np.float64), fft_size)
    I = np.fft.fft(indicator, fft_size)
    cross = np.fft.ifft(A * np.conj(I)).real[:n]
    conv_ia = np.fft.ifft(A * I).real[:n]
    grad = cross + conv_ia
    return grad, max_val

def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def lp_phase(a_init, time_limit, seed=0):
    """LP-based optimization with line search and adaptive control."""
    rng = np.random.default_rng(seed)
    a = normalize(a_init)
    n = len(a)

    track_best(a, {'score': float('inf'), 'seq': None})

    t0 = time.monotonic()
    step = 1.0
    no_improve = 0
    iter_count = 0

    best_local = {'score': float('inf'), 'seq': None}
    track_best(a, best_local)

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()

        b = solve_lp_direction(a, M)
        if b is None:
            b = solve_lp_direction(a, M * 1.005, top_k=max(100, n))
        if b is None:
            break

        # Line search
        a_new, obj_new = line_search(a, b, n_pts=30)
        obj_curr = obj_norm(a)

        if obj_new < obj_curr - 1e-13:
            a = a_new
            no_improve = 0
            step = min(1.0, step * 1.02)
        else:
            no_improve += 1
            step *= 0.8
            if no_improve > 40:
                # Stuck: perturb
                noise = rng.standard_normal(n) * 0.008
                a = project_simplex(a + noise)
                no_improve = 0
                step = 0.7

        iter_count += 1
        if iter_count % 200 == 0:
            track_best(a, best_local)

    track_best(a, best_local)
    return a, best_local['seq'] if best_local['seq'] is not None else a


def gradient_phase(a_init, time_limit, beta_init=500.0, lr_init=0.03,
                  momentum=0.9, seed=0):
    """Smooth gradient descent with momentum and adaptive beta."""
    rng = np.random.default_rng(seed)
    a = normalize(a_init)
    n = len(a)
    v = np.zeros(n)
    beta = beta_init
    lr = lr_init
    no_improve = 0

    t0 = time.monotonic()
    best_local = {'score': float('inf'), 'seq': None}
    track_best(a, best_local)

    while time.monotonic() - t0 < time_limit:
        obj, grad = smooth_obj_grad(a, beta)
        g_norm = np.linalg.norm(grad)
        if g_norm > 1e-15:
            grad = grad / g_norm

        v = momentum * v - lr * grad
        a_new = project_simplex(a + v)

        obj_new = obj_norm(a_new)
        obj_curr = obj_norm(a)

        if obj_new < obj_curr - 1e-13:
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 50 == 0:
                lr *= 0.85
                v *= 0.5
            if no_improve > 400:
                noise = rng.standard_normal(n) * 0.004
                a = project_simplex(a + noise)
                v = np.zeros(n)
                lr = lr_init
                beta = beta_init
                no_improve = 0

        beta = min(8000.0, beta * 1.001)

        if no_improve % 100 == 0:
            track_best(a, best_local)

    track_best(a, best_local)
    return a, best_local['seq'] if best_local['seq'] is not None else a


def subgrad_lp_phase(a_init, time_limit, seed=0):
    """Alternate between LP steps and exact subgradient steps."""
    rng = np.random.default_rng(seed)
    a = normalize(a_init)
    n = len(a)

    t0 = time.monotonic()
    best_local = {'score': float('inf'), 'seq': None}
    track_best(a, best_local)

    fstar = 2.0 * n / (2.0 * n - 1.0)

    while time.monotonic() - t0 < time_limit:
        # LP step
        conv = fast_convolve(a)
        M = conv.max()
        b = solve_lp_direction(a, M)
        if b is not None:
            a_ls, _ = line_search(a, b, n_pts=20)
            if obj_norm(a_ls) < obj_norm(a) - 1e-14:
                a = a_ls

        # Subgradient step with Polyak size
        grad, max_val = subgradient_exact(a)
        obj_curr = 2.0 * n * max_val
        gap = obj_curr - fstar
        g_sq = np.dot(grad, grad)
        if g_sq > 1e-15 and gap > 1e-10:
            t_step = min(gap / g_sq, 0.3)
        else:
            t_step = 0.005

        a_new = project_simplex(a - t_step * grad)
        if obj_norm(a_new) < obj_norm(a) - 1e-14:
            a = a_new

        # Occasional track
        if (time.monotonic() - t0) % 20 < 0.1:
            track_best(a, best_local)

    track_best(a, best_local)
    return a, best_local['seq'] if best_local['seq'] is not None else a
```
