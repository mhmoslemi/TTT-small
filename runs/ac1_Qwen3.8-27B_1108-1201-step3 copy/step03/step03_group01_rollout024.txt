Actually, the analytical gradient of the log-sum-exp of the convolution is:

$\frac{\partial}{\partial a_j} \text{LSE}(\beta \cdot (a*a)) = \frac{1}{\beta} \sum_k \frac{e^{\beta c_k}}{\sum_l e^{\beta c_l}} \cdot \frac{\partial c_k}{\partial a_j}$

where $c_k = (a*a)[k] = \sum_i a_i a_{k-i}$.

So $\frac{\partial c_k}{\partial a_j} = a_{k-j}$ if $0 \leq k-j \leq n-1$, else 0.

Therefore:
$\frac{\partial \text{LSE}}{\partial a_j} = \frac{1}{\beta} \sum_{k: 0 \leq k-j \leq n-1} \frac{e^{\beta c_k}}{\sum_l e^{\beta c_l}} \cdot a_{k-j}$

$= \frac{1}{\beta} \frac{1}{Z} \sum_{k=j}^{j+n-1} e^{\beta c_k} \cdot a_{k-j}$

where $Z = \sum_l e^{\beta c_l}$.

This is a cross-correlation between the "softmax weights" $w_k = e^{\beta c_k} / Z$ and $a$, evaluated at position $j$. We can compute this efficiently with FFT!

Specifically, if we define $w_k = e^{\beta c_k} / Z$ for $k = 0, \ldots, 2n-2$, then:
$\frac{\partial \text{LSE}}{\partial a_j} = \frac{1}{\beta} (w * a)[j]$

where $*$ here is the appropriate cross-correlation. Let me be more careful:

$\sum_{k=j}^{j+n-1} w_k \cdot a_{k-j} = \sum_{m=0}^{n-1} w_{j+m} \cdot a_m$

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
1. LP-based iterative optimization with warm starts and adaptive M
2. Smooth NLP polish using log-sum-exp approximation with L-BFGS-B
3. Final LP refinement and multi-n exploration
The NLP phase uses softmax parameterization for automatic constraint handling
and analytical gradients computed via FFT cross-correlation.

import numpy as np
import time
import math
from scipy.optimize import linprog, minimize
from scipy.signal import correlate as sig_correlate


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def objective_val(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2"""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def obj_normalized(a):
    """Objective assuming sum(a) = 1: returns 2n * max(conv(a,a))"""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def lp_solve(a, M):
    """
    Solve LP: max sum(g) s.t. a*g <= M (componentwise), g >= 0.
    Returns (g_normalized, g_sum).
    """
    n = len(a)
    C = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs',
                     options={'presolve': True})
    if result.success:
        g = result.x
        g_sum = g.sum()
        if g_sum > 1e-10:
            return g / g_sum, g_sum
    return None, 1.0


def line_search(a, direction, current_obj, n, max_evals=25):

            if evals >= max_evals:

            m1 = lo + 0.382 * (hi - lo)
            m2 = lo + 0.618 * (hi - lo)
            for t_try in [m1, m2]:
                a_try = a + t_try * direction
                a_try = np.maximum(a_try, 0.0)
                s = a_try.sum()
                if s < 1e-15:

def compute_lse_gradient(a, beta):
    """
    Compute gradient of log-sum-exp(beta * conv(a,a)) w.r.t. a.
    Uses FFT-based cross-correlation for efficiency.
    Returns array of shape (n,).
    """
    n = len(a)
    conv = fast_convolve(a)

    # Numerically stable softmax
    max_c = conv.max()
    shifted = beta * (conv - max_c)
    # Clip to prevent overflow
    shifted = np.clip(shifted, -50, 500)
    exp_shifted = np.exp(shifted)
    Z = exp_shifted.sum()
    if Z < 1e-300:
        w = np.zeros_like(conv)
        w[int(np.argmax(conv))] = 1.0
    else:
        w = exp_shifted / Z

    # Gradient: (1/beta) * (1/Z) * sum_k exp(beta*c_k) * a[k-j] for valid k-j
    # = (1/beta) * cross_correlation(a, w)
    # cross_correlation at position j: sum_m a[m] * w[j+m]
    grad = sig_correlate(a, w, mode='valid') / beta
    return grad, conv


def lse_objective_and_grad(x, beta, n):
    """
    Objective and gradient for L-BFGS-B optimization.
    x is the log-parameter (length n), a = softmax(x).
    Minimize 2n * LSE(beta * conv(a,a))
    """
    a = np.exp(x - x.max())
    a = a / a.sum()
    a = np.maximum(a, 1e-30)

    conv = fast_convolve(a)
    max_c = conv.max()
    shifted = beta * (conv - max_c)
    shifted = np.clip(shifted, -50, 500)
    exp_shifted = np.exp(shifted)
    Z = exp_shifted.sum()
    if Z < 1e-300:
        log_lse = max_c
        w = np.zeros_like(conv)
        w[int(np.argmax(conv))] = 1.0
    else:
        log_lse = max_c + np.log(Z) / beta
        w = exp_shifted / Z

    val = 2.0 * n * log_lse

    # Gradient w.r.t. a
    grad_a = sig_correlate(a, w, mode='valid') / beta

    # Gradient w.r.t. x: grad_x_j = a_j * (grad_a_j - sum_k grad_a_k * a_k)
    grad_a_dot_a = np.dot(grad_a, a)
    grad_x = a * (grad_a - grad_a_dot_a)

    return val, grad_x


def nlp_polish(a0, n, time_limit, verbose=False):
    """
    Smooth NLP polish using L-BFGS-B with log-sum-exp approximation.
    Uses softmax parameterization for automatic constraint handling.
    """
    a = np.maximum(a0, 1e-15)
    a = a / a.sum()

    # Initial log-parameter
    x0 = np.log(a + 1e-30)

    best_obj = obj_normalized(a)
    best_a = a.copy()
    t0 = time.monotonic()

    # Multi-scale beta schedule
    beta_schedule = [20, 50, 100, 200, 400, 800, 1500, 3000, 5000, 8000]

    for beta in beta_schedule:
        if time.monotonic() - t0 > time_limit - 30:
            break
        remaining = time_limit - (time.monotonic() - t0) - 20
        if remaining < 5:
            break

        t0_beta = time.monotonic()
        beta_time = min(remaining * 0.7, 120.0)

        def func_and_grad(x):
            val, grad = lse_objective_and_grad(x, beta, n)
            if not np.isfinite(val):
                return 1e10, np.zeros_like(x)
            return val, grad

        try:
            result = minimize(
                func_and_grad, x0, method='L-BFGS-B',
                jac=True,
                options={'maxiter': 2000, 'maxfun': 5000,
                         'ftol': 1e-15, 'gtol': 1e-12,
                         'max_ls': 50, 'iprint': 0},
                callback=lambda x: (time.monotonic() - t0_beta > beta_time)
            )
        except Exception as e:
            if verbose:
                print(f"    NLP beta={beta} failed: {e}")
            continue

        if result.success or result.fun < best_obj:
            x_opt = result.x
            a_new = np.exp(x_opt - x_opt.max())
            a_new = a_new / a_new.sum()
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s > 1e-15:
                a_new = a_new / s

            new_obj = obj_normalized(a_new)
            if new_obj < best_obj - 1e-14:
                best_obj = new_obj
                best_a = a_new.copy()
                x0 = x_opt
                if verbose:
                    print(f"    NLP beta={beta}: obj={best_obj:.8f}")
            else:
                # Still update x0 for the next beta
                x0 = x_opt
        else:
            x0 = x0  # Keep previous

    return best_a, best_obj

        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(a)
        M = conv.max()
        n_conv = len(conv)

        # Find active set: positions where conv is within 5% of max
        threshold = M * 0.97
        active = np.where(conv >= threshold)[0]

        if len(active) < 2:
            active = np.argpartition(conv, -3)[-3:]
            active = np.sort(active)

        S = len(active)

        # Set up Jacobian for the system:
        # Unknowns: a (n vars), M (1 var) -> n+1 unknowns
        # Equations: (a*a)[k_i] = M for i=1..S, sum(a) = 1,
        #            plus stationarity conditions (S-1 equations after removing one)
        #
        # Stationarity: for all j with a_j > 0:
        #   sum_{i=1}^{S} lam_i * a[k_i - j] = mu
        # So: sum_i lam_i * a[k_i - j] is constant across j.
        # This gives n-1 independent equations (set j=0 as reference):
        #   sum_i lam_i * (a[k_i - j] - a[k_i]) = 0 for j = 1, ..., n-1
        #
        # But we have S dual variables lam_i. The system is:
        #   F(a, M, lam) = 0
        #   (a*a)[k_i] - M = 0, i = 1..S
        #   sum(a) - 1 = 0
        #   sum_i lam_i * (a[k_i - j] - a[k_i]) = 0, j = 1..n-1
        #
        # Total equations: S + 1 + (n-1) = n + S
        # Unknowns: n (a) + 1 (M) + S (lam) = n + S + 1
        # One degree of freedom: we can fix sum(lam) = 1 or similar.
        # Let's fix lam_S = 1 - sum_{i<S} lam_i, reducing unknowns to n + S.
        #
        # Actually this system is complex. Let me use a simpler fixed-point approach:
        # Given active set S, solve for a that equalizes the active convolution values
        # while minimizing the max.

        # Simpler approach: use the fact that at optimum, the "pressure" 
        # from active constraints should be balanced.
        # Compute the "gradient" from active constraints:
        # g_j = sum_{k in active} w_k * a[k-j] (where w_k are weights)
        # At optimum, g_j should be constant (for j where a_j > 0).
        # Adjust a to make g more uniform.

        # Compute current "pressure" vector
        pressure = np.zeros(n)
        for k in active:
            for j in range(n):
                idx = k - j
                if 0 <= idx < n:
                    pressure[j] += a[idx]

        # Normalize
        p_mean = pressure.mean()
        p_std = pressure.std()
        if p_std < 1e-15:
            break

        # The "deficit" tells us where to move mass
        deficit = p_mean - pressure  # positive where pressure is low
        a_new = a + 0.01 * deficit
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()

        new_obj = obj_normalized(a_new)
        if new_obj < best_obj - 1e-14:
            best_obj = new_obj
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            a_new2 = a + 0.003 * deficit
            a_new2 = np.maximum(a_new2, 1e-15)
            a_new2 = a_new2 / a_new2.sum()
            new_obj2 = obj_normalized(a_new2)
            if new_obj2 < best_obj - 1e-14:
                best_obj = new_obj2
                best_a = a_new2.copy()
                a = a_new2
            else:
                break

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Various structured initializations
    tri = np.minimum(x, n - 1 - x) + 1.0
    inits['triangle'] = tri / tri.sum()

    # Ramp
    ramp = (x + 1.0)
    inits['ramp'] = ramp / ramp.sum()
    inits['ramp_rev'] = ramp[::-1] / ramp.sum()

    # Bimodal (edges)
    bi = np.zeros(n)
    edge = max(1, n // 5)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.2 * n)) ** 2)
    inits['gaussian'] = g / g.sum()

    # Sine
    sinv = (np.sin(np.pi * x / n) + 0.2)
    inits['sine'] = sinv / sinv.sum()

    # Two peaks
    two = np.zeros(n)
    two[n // 4] = 1.0
    two[3 * n // 4] = 1.0
    from scipy.ndimage import gaussian_filter1d
    two = gaussian_filter1d(two, sigma=n // 20)
    inits['two_peak'] = two / two.sum()

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase hybrid search:
    Phase 1: LP-based optimization across multiple initializations
    Phase 2: Smooth NLP polish (log-sum-exp + L-BFGS-B)
    Phase 3: LP refinement + active set Newton
    Phase 4: Multi-n exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: n={len(p)}, sum={p.sum():.2f}")
    except Exception:
        pass

    # Phase 1: LP-based scan
    print("=== Phase 1: LP Scan ===")
    t_start = time.monotonic()
    phase1_limit = 500.0
    phase1_deadline = t_start + phase1_limit

    lengths = [512]
    if parent is not None:
        lengths.insert(0, len(parent))
        if len(parent) != 512:
            lengths.append(512)

    # Remove duplicates, keep order
    seen_l = set()
    unique_lengths = []
    for l in lengths:
        if l not in seen_l and 64 <= l <= 2048:
            unique_lengths.append(l)
            seen_l.add(l)
    lengths = unique_lengths

    scan_results = []

    for n in lengths:
        if time.monotonic() > phase1_deadline - 30:
            break

        inits = generate_initializations(n, parent, rng)
        priority = ['parent', 'parent_interp', 'uniform', 'triangle', 'bimodal',
                    'sine', 'two_peak', 'ramp', 'gaussian', 'random']
        init_names = [nm for nm in priority if nm in inits]
        init_names = init_names[:4]

        t_per_init = max(10.0, (phase1_limit - 30.0) / (len(lengths) * len(init_names)))

        for name in init_names:
            if time.monotonic() > phase1_deadline - 20:
                break
            t_alloc = min(t_per_init, phase1_deadline - time.monotonic() - 10)
            if t_alloc < 8:
                break

            a0 = inits[name]
            a_opt, obj_opt = lp_optimize(a0, time_limit=t_alloc)
            score = float(evaluate_sequence(a_opt.tolist()))

            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Phase1 n={n}, {name}: {score:.6f}")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic() - t_start:.0f}s). Best: {best_score:.6f}")

    # Phase 2: NLP Polish
    print("=== Phase 2: NLP Polish ===")
    t_start2 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 200.0
        if remaining > 60:
            n_best = len(best_sequence)
            nlp_limit = min(remaining, 800.0)
            a_polished, obj_polished = nlp_polish(best_sequence, n_best, time_limit=nlp_limit, verbose=True)
            score = float(evaluate_sequence(a_polished.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_polished.copy()
                print(f"  NLP polish: {score:.6f}")

    print(f"Phase 2 done ({time.monotonic() - t_start2:.0f}s). Best: {best_score:.6f}")

    # Phase 3: LP Refinement on NLP result
    print("=== Phase 3: LP Refinement ===")
    t_start3 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 100.0
        if remaining > 30:
            lp_limit = min(remaining, 300.0)
            a_refined, obj_refined = lp_optimize(best_sequence, time_limit=lp_limit)
            score = float(evaluate_sequence(a_refined.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_refined.copy()
                print(f"  LP refine: {score:.6f}")

            # Second NLP pass on refined result
            remaining2 = deadline - time.monotonic() - 50.0
            if remaining2 > 30:
                a_polished2, _ = nlp_polish(best_sequence, len(best_sequence),
                                             time_limit=min(remaining2 * 0.6, 150.0))
                score2 = float(evaluate_sequence(a_polished2.tolist()))
                if np.isfinite(score2) and score2 < best_score:
                    best_score = score2
                    best_sequence = a_polished2.copy()
                    print(f"  NLP 2nd pass: {score:.6f}")

    print(f"Phase 3 done ({time.monotonic() - t_start3:.0f}s). Best: {best_score:.6f}")

    # Phase 4: Try other lengths
    print("=== Phase 4: Multi-n ===")
    t_start4 = time.monotonic()
    tried_n = set([len(s) for s in [best_sequence]] if best_sequence is not None else set())
    other_lengths = [256, 768, 1024, 384]
    other_lengths = [l for l in other_lengths if l not in tried_n]

    for n_try in other_lengths:
        if time.monotonic() > deadline - 60:
            break
        if n_try in tried_n:
            continue
        tried_n.add(n_try)

        remaining = deadline - time.monotonic() - 40
        if remaining < 30:
            break

        inits = generate_initializations(n_try, best_sequence, rng)
        # Use parent-interpolated as primary init
        a0 = inits.get('parent_interp', inits.get('uniform'))

        t_alloc = min(remaining * 0.4, 100.0)
        if t_alloc < 10:
            continue

        a_opt, obj_opt = lp_optimize(a0, time_limit=t_alloc)
        score = float(evaluate_sequence(a_opt.tolist()))

        if np.isfinite(score):
            # NLP polish on this
            if time.monotonic() < deadline - 30:
                a_pp, _ = nlp_polish(a_opt, n_try, time_limit=min(remaining * 0.2, 60.0))
                score = float(evaluate_sequence(a_pp.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_pp.copy()
                    print(f"  n={n_try}: {score:.6f}")

    print(f"Phase 4 done ({time.monotonic() - t_start4:.0f}s). Best: {best_score:.6f}")

    # Final: one more NLP + LP cycle on best
    if best_sequence is not None and time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic() - 5
        if remaining > 20:
            a_final, _ = nlp_polish(best_sequence, len(best_sequence), time_limit=remaining * 0.5)
            score_f = float(evaluate_sequence(a_final.tolist()))
            if np.isfinite(score_f) and score_f < best_score:
                best_score = score_f
                best_sequence = a_final.copy()
                print(f"  Final NLP: {score_f:.6f}")

            if time.monotonic() < deadline - 5:
                a_final2, _ = lp_optimize(best_sequence, time_limit=remaining * 0.3)
                score_f2 = float(evaluate_sequence(a_final2.tolist()))
                if np.isfinite(score_f2) and score_f2 < best_score:
                    best_score = score_f2
                    best_sequence = a_final2.copy()
                    print(f"  Final LP: {score_f2:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256.0
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
