"""
Step function optimization using L-BFGS-B with log-sum-exp smoothing,
Frank-Wolfe, and LP refinement to minimize 2*n*max(autoconv)/(sum)^2.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def compute_grad_a(a, beta=100.0):
    """Compute gradient of 2n*logsumexp(beta*conv)/beta w.r.t. a."""
    n = len(a)
    conv = fast_convolve(a)
    beta_conv = beta * conv
    m = beta_conv.max()
    w = np.exp(beta_conv - m)
    w_sum = w.sum()
    w = w / w_sum
    
    # cross[i] = sum_k w[k] * a[k-i] (a[m]=0 for m outside [0,n-1])
    fft_size = 1 << (2 * n - 1).bit_length()
    W = np.fft.fft(w, fft_size)
    A = np.fft.fft(a, fft_size)
    cross = np.fft.ifft(np.conj(A) * W).real[:n]
    
    grad_a = 4.0 * n * cross
    obj = 2.0 * n * (m + np.log(w_sum)) / beta
    return obj, grad_a, w, conv


def softmax(z):
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def lbfgsb_smoothed(a0, time_limit, beta_init=100.0, beta_max=5000.0,
                     beta_growth=1.02, lr_scale=1.0, seed=0, verbose=False):
    """
    L-BFGS-B optimization with iterative beta increase on log-sum-exp smoothed objective.
    Uses softmax parameterization to handle simplex constraint.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    # Initialize z from a: z_i = log(a_i) for a_i > 0, else some default
    z = np.full(n, -10.0)
    pos = a > 1e-10
    z[pos] = np.log(a[pos])
    
    t0 = time.monotonic()
    beta = beta_init
    best_obj = np.inf
    best_a = a.copy()
    
    def z_obj_grad(z):
        a_curr = softmax(z)
        obj_a, grad_a, w, conv = compute_grad_a(a_curr, beta)
        # Chain rule: grad_z = a * (grad_a - a @ grad_a)
        a_dot = np.dot(a_curr, grad_a)
        grad_z = a_curr * (grad_a - a_dot)
        return obj_a, grad_z
    
    # First evaluation
    obj, grad = z_obj_grad(z)
    if obj < best_obj:
        best_obj = obj
        a_curr = softmax(z)
        best_a = a_curr.copy()
    
    iteration = 0
    stuck_count = 0
    
    while time.monotonic() - t0 < time_limit:
        res = minimize(z_obj_grad, z, method='L-BFGS-B',

                                         'gtol': 1e-8, 'maxfun': 100})

    conv_final = fast_convolve(best_a)
    final_obj = 2.0 * n * conv_final.max()
    return best_a, final_obj


def frank_wolfe(a0, time_limit, beta=100.0, verbose=False, seed=0):

    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    t0 = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        beta_conv = beta * conv
        m = beta_conv.max()
        w = np.exp(beta_conv - m)
        w_sum = w.sum()
        w = w / w_sum
        obj = 2.0 * n * (m + np.log(w_sum)) / beta
        
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        # Gradient: grad[i] = 4*n * sum_k w[k] * a[k-i]
        fft_size = 1 << (2 * n - 1).bit_length()
        W = np.fft.fft(w, fft_size)
        A = np.fft.fft(a, fft_size)
        cross = np.fft.ifft(np.conj(A) * W).real[:n]
        grad = 4.0 * n * cross
        
        # FW direction: s = e_{argmin grad}
        s_idx = np.argmin(grad)
        s = np.zeros(n)
        s[s_idx] = 1.0
        
        # Line search (armijo)
        gap = np.dot(grad, a - s)  # FW gap
        if gap > -1e-12:
            break  # converged
        
        # Full step or half step
        t = 1.0 / (2.0 * (np.linalg.norm(a - s) + 1e-10))
        t = min(t, 1.0)
        
        a_new = (1.0 - t) * a + t * s
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
        
        # Check if it improved
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < obj:
            a = a_new
        else:
            # Try half step
            a_new = (1.0 - t/2) * a + (t/2) * s
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new > 1e-15:

def lp_refine(a0, time_limit, top_k=50, relax=1.0, step_init=0.8, seed=0):
    """LP-based refinement with top-k constraints."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    t0 = time.monotonic()
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    step = step_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()
        
        # Select top-k peaks
        if top_k is not None and top_k < len(conv):
            peaks = np.argpartition(-np.abs(conv), top_k)[:top_k]
        else:
            peaks = np.arange(len(conv))
        
        # Build LP: max sum(b) s.t. (a*b)[k] <= M*relax for k in peaks, b >= 0
        num_constraints = len(peaks)
        A_ub = np.zeros((num_constraints, n))
        for ci, k in enumerate(peaks):
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            idx = np.arange(i_min, i_max + 1)
            A_ub[ci, idx] = a[k - idx]
        
        c = -np.ones(n)
        bounds = [(0, None)] * n
        
        try:
            res = linprog(c, A_ub=A_ub, b_ub=np.full(num_constraints, M * relax),
                         bounds=bounds, method='highs',
                         options={'presolve': True})
            if res.success and res.x is not None:
                b = res.x
                b_sum = b.sum()
                if b_sum > 1e-15:
                    b = b / b_sum
                    a_new = (1.0 - step) * a + step * b
                    a_new = np.maximum(a_new, 0.0)
                    s_new = a_new.sum()
                    if s_new > 1e-15:
                        a_new = a_new / s_new
                    
                    conv_new = fast_convolve(a_new)
                    obj_new = 2.0 * n * conv_new.max()
                    
                    if obj_new < best_obj - 1e-12:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        a = a_new
                        no_improve = 0
                    else:
                        no_improve += 1
                        if no_improve > 5:
                            step = max(0.05, step * 0.8)
                            if no_improve > 20:
                                noise = rng.standard_normal(n) * 0.01
                                a = project_simplex(a + noise)
                                step = step_init * 0.5
                                no_improve = 0
        except Exception:
            break
    
    conv_final = fast_convolve(best_a)
    final_obj = 2.0 * n * conv_final.max()
    return best_a, final_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n (normalized to sum=1)."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Hamming-like window (raised cosine)
    ham = 0.54 - 0.46 * np.cos(2 * np.pi * x / (n - 1))
    ham = np.maximum(ham, 0.01)
    ham /= ham.sum()
    inits['hamming'] = ham
    
    # Hann window
    hann = 0.5 * (1 - np.cos(2 * np.pi * x / (n - 1)))
    hann = np.maximum(hann, 0.01)
    hann /= hann.sum()
    inits['hann'] = hann
    
    # Blackman window
    blackman = 0.42 - 0.5 * np.cos(2 * np.pi * x / (n - 1)) + 0.08 * np.cos(4 * np.pi * x / (n - 1))
    blackman = np.maximum(blackman, 0.01)
    blackman /= blackman.sum()
    inits['blackman'] = blackman
    
    # Bimodal (edges)
    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi
    
    # Gaussians
    for sf in [0.1, 0.15, 0.2, 0.3, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf:.2f}'] = g
    
    # Random
    inits['random1'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['random2'] = rng.dirichlet(np.ones(n) * 1.0)
    inits['random3'] = rng.dirichlet(np.ones(n) * 0.1)
    
    # Comb (periodic)
    for period in [3, 4, 5, 7]:
        comb = np.ones(n)
        comb[::period] = 2.0
        comb /= comb.sum()
        inits[f'comb_{period}'] = comb
    
    # From parent
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            pp /= pp.sum()
            inits['parent'] = pp
        else:
            pp = np.interp(np.linspace(0, 1, n),
                          np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent_up'] = pp
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                          np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent_ds'] = pp
    
    return inits

We want `cross[i] = sum_k w[k] * a[k-i]` where `a[m] = 0` for `m < 0` or `m >= n`.

Using FFT with zero-padding to `fft_size >= 2n-1`:
- Let `A = FFT(a, fft_size)` and `W = FFT(w, fft_size)`.
- The circular cross-correlation is `ifft(conj(A) * W)`.
- Specifically: `ifft(conj(A) * W)[i] = sum_j a[j] * w[(j+i) mod fft_size]`

For `i < n` and `j < n`, `j + i < 2n <= fft_size`, so there's no wrap-around. And `w[k] = 0` for `k >= len(w) = 2n-1` (since we zero-padded). Actually, `w` has length `2n-1`, so `w[k]` is nonzero for `k = 0, ..., 2n-2`.

So `ifft(conj(A) * W)[i] = sum_{j=0}^{n-1} a[j] * w[j+i]` (since `w[k]=0` for `k >= 2n-1`).

And we want `cross[i] = sum_k w[k] * a[k-i]`. Let `m = k-i`, so `k = m+i`:
`cross[i] = sum_m w[m+i] * a[m]` where `m` ranges such that `0 <= m < n` and `0 <= m+i < 2n-1`.

This is exactly `sum_{j=0}^{n-1} a[j] * w[j+i]` (with `w[k]=0` for `k >= 2n-1`).

So `cross = ifft(conj(FFT(a)) * FFT(w)).real[:n]` is correct!

"""
Step function optimization combining L-BFGS-B with log-sum-exp smoothing,
Frank-Wolfe, and LP direction finding to minimize 2*n*max(autoconv)/(sum)^2.
Uses softmax parameterization, iterative temperature increase, multi-scale
approach, and diverse initializations for robust global search.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    if size <= 1:
        return a.astype(np.float64) ** 2
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def compute_grad_a(a, beta):
    """Compute smooth objective value and gradient w.r.t. a on simplex."""
    n = len(a)
    conv = fast_convolve(a)
    beta_conv = beta * conv
    m = beta_conv.max()
    if m > 50:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
        obj = 2.0 * n * m / beta
    else:
        w = np.exp(beta_conv - m)
        w_sum = w.sum()
        w = w / w_sum
        obj = 2.0 * n * (m + np.log(w_sum)) / beta

    # cross[i] = sum_k w[k] * a[k-i] = ifft(conj(FFT(a)) * FFT(w)).real[:n]
    fft_size = 1 << (2 * n - 1).bit_length()
    W = np.fft.fft(w, fft_size)
    A = np.fft.fft(a, fft_size)
    cross = np.fft.ifft(np.conj(A) * W).real[:n]

    grad_a = 4.0 * n * cross
    return obj, grad_a, conv


def softmax_fn(z):
    """Stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def lbfgsb_optimize(a0, time_limit, beta_init=100.0, beta_max=5000.0,
                    beta_growth=1.03, seed=0, verbose=False):
    """
    L-BFGS-B on softmax-parameterized log-sum-exp smoothed objective.
    Iteratively increases beta for sharper max approximation.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    # Initialize z from a
    z = np.full(n, -8.0)
    pos = a > 1e-10
    z[pos] = np.log(a[pos])

    t0 = time.monotonic()
    beta = beta_init
    best_obj = np.inf
    best_a = a.copy()

    # Define objective and gradient in z-space
    def z_fn(z_curr):
        a_curr = softmax_fn(z_curr)
        obj, grad_a, _ = compute_grad_a(a_curr, beta)
        a_dot = np.dot(a_curr, grad_a)
        grad_z = a_curr * (grad_a - a_dot)
        return obj, grad_z

    # Initial evaluation
    obj_val, _ = z_fn(z)
    if obj_val < best_obj:
        best_obj = obj_val
        best_a = softmax_fn(z).copy()

    iteration = 0
    stuck = 0
    max_outer = 200

    while time.monotonic() - t0 < time_limit and iteration < max_outer:
        # Run L-BFGS-B for a few iterations
        max_iter_inner = max(10, min(80, int(500.0 / (1.0 + 0.1 * iteration))))
        try:
            res = minimize(z_fn, z, method='L-BFGS-B', jac=True,
                          options={'maxiter': max_iter_inner, 'ftol': 1e-12,
                                   'gtol': 1e-9, 'maxls': 30,
                                   'maxfun': max_iter_inner * 5})
        except Exception:
            break

        z = res.x
        a_new = softmax_fn(z)
        conv_new = fast_convolve(a_new)
        true_obj = 2.0 * n * conv_new.max()

        if true_obj < best_obj - 1e-10:
            best_obj = true_obj
            best_a = a_new.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 4:
                # Perturb in z-space
                z = z + rng.standard_normal(n) * 0.02 * np.sqrt(n)
                stuck = 0

        # Increase beta (sharper approximation)
        beta = min(beta_max, beta * beta_growth)
        iteration += 1

        if verbose and iteration % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LB iter {iteration}: obj={best_obj:.6f}, beta={beta:.0f}, t={elapsed:.1f}s")

    # Final true objective
    conv_final = fast_convolve(best_a)
    final_obj = 2.0 * n * conv_final.max()
    return best_a, final_obj


def frank_wolfe_optimize(a0, time_limit, beta=200.0, seed=0):
    """Frank-Wolfe (conditional gradient) with smooth max."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    t0 = time.monotonic()
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        beta_conv = beta * conv
        m = beta_conv.max()
        if m > 50:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w = np.exp(beta_conv - m)
            w_sum = w.sum()
            w = w / w_sum

        # Gradient
        fft_size = 1 << (2 * n - 1).bit_length()
        W = np.fft.fft(w, fft_size)
        A = np.fft.fft(a, fft_size)
        cross = np.fft.ifft(np.conj(A) * W).real[:n]
        grad = 4.0 * n * cross

        # FW direction: vertex with min gradient
        s_idx = np.argmin(grad)
        s_dir = np.zeros(n)
        s_dir[s_idx] = 1.0

        # FW gap
        gap = np.dot(grad, a - s_dir)
        if gap > -1e-10:
            break

        # Step size: use 2/(t+1) schedule or line search
        iter_count += 1
        t_lr = 2.0 / (iter_count + 2.0)

        a_new = (1.0 - t_lr) * a + t_lr * s_dir
        a_new = np.maximum(a_new, 0.0)
        a_new /= a_new.sum()

        # Also try full step
        a_full = s_dir.copy()
        conv_full = fast_convolve(a_full)
        obj_full = 2.0 * n * conv_full.max()

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
        a = a_new

    conv_final = fast_convolve(best_a)
    final_obj = 2.0 * n * conv_final.max()
    return best_a, final_obj


def lp_direction_optimize(a0, time_limit, top_k=50, relax=1.0,

    s = a.sum()

    else:
        a = a / s

    t0 = time.monotonic()
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    step = step_init
    no_improve = 0
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()

        # Select top-k peaks
        num_peaks = min(top_k if top_k is not None else len(conv), len(conv))
        if num_peaks < len(conv):
            peaks = np.argpartition(-conv, num_peaks)[:num_peaks]
        else:
            peaks = np.arange(len(conv))

        # Build LP
        A_ub = np.zeros((num_peaks, n))
        for ci, k in enumerate(peaks):
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            idx = np.arange(i_min, i_max + 1)
            A_ub[ci, idx] = a[k - idx]

        c_vec = -np.ones(n)
        bounds = [(0, None)] * n

        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=np.full(num_peaks, M * relax),
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 10})
            if res.success and res.x is not None:
                b = np.maximum(res.x, 0)
                b_sum = b.sum()
                if b_sum > 1e-15:
                    b = b / b_sum
                    a_new = (1.0 - step) * a + step * b
                    a_new = np.maximum(a_new, 0.0)
                    a_new /= a_new.sum()

                    conv_new = fast_convolve(a_new)
                    obj_new = 2.0 * n * conv_new.max()

                    if obj_new < best_obj - 1e-12:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        a = a_new
                        no_improve = 0
                    else:
                        no_improve += 1
                        if no_improve > 3:
                            step = max(0.05, step * 0.85)
                        if no_improve > 15:
                            noise = rng.standard_normal(n) * 0.01
                            a = project_simplex(a + noise)
                            step = step_init * 0.6
                            no_improve = 0
        except Exception:
            if time.monotonic() - t0 > time_limit * 0.3:
                break

        iter_count += 1
        if verbose and iter_count % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {iter_count}: obj={best_obj:.6f}, step={step:.3f}, t={elapsed:.1f}s")

    conv_final = fast_convolve(best_a)
    final_obj = 2.0 * n * conv_final.max()
    return best_a, final_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n (normalized to sum=1)."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Window functions
    ham = 0.54 - 0.46 * np.cos(2 * np.pi * x / max(1, n - 1))
    ham = np.maximum(ham, 0.01)
    ham /= ham.sum()
    inits['hamming'] = ham

    hann = 0.5 * (1.0 - np.cos(2.0 * np.pi * x / max(1, n - 1)))
    hann = np.maximum(hann, 0.01)
    hann /= hann.sum()
    inits['hann'] = hann

    # Bimodal
    edge = max(1, n // 5)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Edge-heavy (inverse Hann)
    inv_hann = 1.0 / (hann + 0.01)
    inv_hann /= inv_hann.sum()
    inits['inv_hann'] = inv_hann

    # Gaussians
    for sf in [0.1, 0.15, 0.2, 0.25, 0.35]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits['gauss_%s' % str(sf)] = g

    # Multi-modal
    if n >= 8:
        centers = np.array([0.15, 0.5, 0.85]) * n
        widths = 0.1 * n
        tri = np.zeros(n)
        for c in centers:
            tri += np.exp(-0.5 * ((x - c) / widths) ** 2)
        tri /= tri.sum()
        inits['triple_mod'] = tri

    # Comb patterns
    for period in [3, 4, 5]:
        comb = np.ones(n)
        comb[::period] = 1.5
        comb /= comb.sum()
        inits['comb_%d' % period] = comb

    # Random
    inits['rand_alpha1'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['rand_alpha2'] = rng.dirichlet(np.ones(n) * 1.0)
    inits['rand_alpha3'] = rng.dirichlet(np.ones(n) * 0.2)

    # From parent
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent'] = pp
        else:
            pp = np.interp(np.linspace(0, 1, n),
                          np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent_up'] = pp
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                          np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            if pp.sum() > 1e-10:
                pp /= pp.sum()
                inits['parent_ds'] = pp

    return inits


def two_scale_optimize(n, a_fine, time_coarse, time_fine, seed=0):
    """
    Two-scale approach: downsample to n/2, optimize, interpolate up, refine.
    """
    rng = np.random.default_rng(seed)
    n_half = max(16, n // 2)

    # Downsample via averaging
    a_coarse = np.zeros(n_half)
    ratio = n / n_half
    for i in range(n_half):
        start = int(i * ratio)
        end = int((i + 1) * ratio)
        end = min(end, n)
        if end > start:
            a_coarse[i] = a_fine[start:end].mean()
    a_coarse /= a_coarse.sum()

    # Optimize on coarse
    a_coarse_opt, _ = lbfgsb_optimize(a_coarse, time_limit=time_coarse,
                                       beta_init=100, beta_max=5000,
                                       beta_growth=1.05, seed=seed)
    # Interpolate back
    a_interp = np.interp(np.linspace(0, 1, n),
                         np.linspace(0, 1, n_half), a_coarse_opt)
    a_interp = np.maximum(a_interp, 0)
    a_interp /= a_interp.sum()

    # Fine optimization
    a_fine_opt, _ = lbfgsb_optimize(a_interp, time_limit=time_fine,
                                     beta_init=200, beta_max=10000,
                                     beta_growth=1.02, seed=seed + 1)
    return a_fine_opt


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase hybrid search: quick exploration -> deep L-BFGS-B -> LP refinement
    -> two-scale refinement -> final polish. Minimizes 2*n*max(autoconv)/(sum)^2.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    def try_update(a):
        nonlocal best_sequence, best_score
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print("Parent: len=%d, sum=%.2f" % (len(p), p.sum()))
    except (NameError, Exception):
        pass

    # ========== Phase 1: Quick Exploration (250s) ==========
    print("=== Phase 1: Quick Exploration ===")
    t1 = time.monotonic()
    phase1_time = 250.0

    # Determine lengths to try
    if parent is not None:
        p_len = len(parent)
        candidate_lengths = [p_len]
        for offset in [-64, -32, 0, 32, 64, 128]:
            l = p_len + offset
            if 32 <= l <= 800:
                candidate_lengths.append(l)
        # Add standard lengths
        for std in [64, 128, 192, 256, 320, 384, 512, 640]:
            candidate_lengths.append(std)
    else:
        candidate_lengths = [64, 128, 192, 256, 320, 384, 512, 640]

    candidate_lengths = sorted(set(candidate_lengths))
    # Limit to 7 lengths, prioritizing parent length
    if len(candidate_lengths) > 7:
        if parent is not None:
            p_len = len(parent)
            # Keep parent length and 3 on each side
            idx = candidate_lengths.index(p_len) if p_len in candidate_lengths else len(candidate_lengths) // 2
            keep = set(candidate_lengths[max(0, idx - 2):idx + 3])
            for l in candidate_lengths:
                if l not in keep and len(keep) < 7:
                    keep.add(l)
            candidate_lengths = sorted(keep)
        else:
            candidate_lengths = candidate_lengths[:7]

    scan_results = []

    for n in candidate_lengths:
        elapsed = time.monotonic() - t1
        if elapsed > phase1_time - 40:
            break

        inits = generate_initializations(n, parent, rng)
        priority = ['parent', 'parent_ds', 'parent_up', 'hamming', 'bimodal',
                    'inv_hann', 'gauss_0.2', 'rand_alpha1', 'uniform', 'hann']
        init_names = [nm for nm in priority if nm in inits][:4]
        if not init_names:
            init_names = list(inits.keys())[:3]

        remaining = phase1_time - elapsed - 10
        num_lengths_left = len(candidate_lengths) - candidate_lengths.index(n)
        time_per_init = max(10.0, remaining / max(1, num_lengths_left) / max(1, len(init_names)))

        for name in init_names:
            if time.monotonic() - t1 > phase1_time - 15:
                break
            t_alloc = min(time_per_init, phase1_time - (time.monotonic() - t1) - 5)
            if t_alloc < 8:
                break

            a0 = inits[name]

            # Quick FW
            a_fw, _ = frank_wolfe_optimize(a0, time_limit=min(t_alloc * 0.4, 25), beta=150)
            score = try_update(a_fw)
            scan_results.append((score, n, name, a_fw.copy()))

            # Quick L-BFGS-B
            a_lb, _ = lbfgsb_optimize(a0, time_limit=min(t_alloc * 0.6, 20),
                                       beta_init=80, beta_max=3000,
                                       beta_growth=1.08, seed=n + 7)
            score2 = try_update(a_lb)
            scan_results.append((score2, n, name + '_lb', a_lb.copy()))

            if score < 1.6 or score2 < 1.6:
                print("  n=%d %s: FW=%.6f LB=%.6f" % (n, name, score, score2))

    scan_results.sort(key=lambda x: x[0])
    print("Phase 1 best: %.6f" % best_score)

    # ========== Phase 2: Deep L-BFGS-B (600s) ==========
    print("=== Phase 2: Deep L-BFGS-B ===")
    t2 = time.monotonic()
    phase2_time = 600.0

    best_n = len(best_sequence) if best_sequence is not None else 320

    # Collect starting points
    deep_starts = []
    seen_keys = set()
    for score, n_val, name, a_opt in scan_results[:8]:
        key = (n_val, name.split('_')[0])
        if key not in seen_keys:
            deep_starts.append((n_val, a_opt.copy()))
            seen_keys.add(key)
        if len(deep_starts) >= 5:
            break

    # Add fresh at best_n
    inits_best = generate_initializations(best_n, parent, rng)
    for extra in ['hamming', 'inv_hann', 'bimodal', 'gauss_0.15']:
        if extra in inits_best:
            deep_starts.append((best_n, inits_best[extra].copy()))

    # L-BFGS-B configs for deep optimization
    lb_configs = [
        {'beta_init': 50, 'beta_max': 8000, 'beta_growth': 1.025},
        {'beta_init': 100, 'beta_max': 5000, 'beta_growth': 1.04},
        {'beta_init': 200, 'beta_max': 20000, 'beta_growth': 1.02},
    ]

    total_runs = len(deep_starts) * len(lb_configs)
    time_per_run = max(20.0, phase2_time / max(1, total_runs))

    run_i = 0
    for start_i, (n_start, a_start) in enumerate(deep_starts):
        for cfg_i, cfg in enumerate(lb_configs):
            elapsed = time.monotonic() - t2
            if elapsed > phase2_time - 30:
                break
            t_alloc = min(time_per_run, phase2_time - elapsed - 10)
            if t_alloc < 12:
                break

            a_opt, obj_opt = lbfgsb_optimize(a_start, time_limit=t_alloc,
                                             seed=run_i * 13 + 3, **cfg)
            score = try_update(a_opt)
            run_i += 1
            if score < best_score + 0.001:
                print("  P2 [%d,%d] n=%d: %.6f" % (start_i, cfg_i, n_start, score))

    print("Phase 2 best: %.6f" % best_score)

    # ========== Phase 3: LP Refinement (400s) ==========
    print("=== Phase 3: LP Refinement ===")
    t3 = time.monotonic()
    phase3_time = 400.0

    if best_sequence is not None:
        n_best = len(best_sequence)

        lp_configs = [
            {'top_k': max(30, n_best // 10), 'relax': 1.0, 'step_init': 0.8},
            {'top_k': max(50, n_best // 6), 'relax': 1.0, 'step_init': 0.5},
            {'top_k': max(20, n_best // 15), 'relax': 1.01, 'step_init': 1.0},
            {'top_k': None, 'relax': 1.0, 'step_init': 0.4},
        ]

        # Multiple starting points
        lp_starts = [best_sequence.copy()]
        for scale in [0.003, 0.008]:
            noise = rng.standard_normal(n_best) * scale
            lp_starts.append(project_simplex(best_sequence + noise))

        time_per_lp = max(25.0, phase3_time / max(1, len(lp_configs) * len(lp_starts)))

        for si, s_start in enumerate(lp_starts):
            for ci, cfg in enumerate(lp_configs):
                elapsed = time.monotonic() - t3
                if elapsed > phase3_time - 20:
                    break
                t_alloc = min(time_per_lp, phase3_time - elapsed - 5)
                if t_alloc < 15:
                    break

                a_opt, obj_opt = lp_direction_optimize(s_start, time_limit=t_alloc,
                                                       seed=si * 7 + ci, **cfg)
                score = try_update(a_opt)
                if score < best_score + 0.005:
                    print("  P3 s%dc%d: %.6f" % (si, ci, score))

    print("Phase 3 best: %.6f" % best_score)

    # ========== Phase 4: Two-Scale + Fine L-BFGS-B (300s) ==========
    print("=== Phase 4: Two-Scale + Fine Polish ===")
    t4 = time.monotonic()
    phase4_time = 300.0

    if best_sequence is not None:
        n_best = len(best_sequence)

        # Two-scale approach
        remaining_p4 = phase4_time - (time.monotonic() - t4)
        if remaining_p4 > 60:
            a_ts = two_scale_optimize(n_best, best_sequence,
                                      time_coarse=remaining_p4 * 0.15,
                                      time_fine=remaining_p4 * 0.25, seed=55)
            score = try_update(a_ts)
            if score < best_score + 0.005:
                print("  Two-scale: %.6f" % score)

        # Very fine L-BFGS-B (high beta, many iterations)
        remaining_p4 = phase4_time - (time.monotonic() - t4)
        if remaining_p4 > 40:
            a_fine, _ = lbfgsb_optimize(best_sequence, time_limit=min(remaining_p4 * 0.4, 80),
                                         beta_init=3000, beta_max=100000,
                                         beta_growth=1.005, seed=777)
            score = try_update(a_fine)
            if score < best_score + 0.005:
                print("  Fine LB: %.6f" % score)

        # Fine LP
        remaining_p4 = phase4_time - (time.monotonic() - t4)
        if remaining_p4 > 30:
            a_lp2, _ = lp_direction_optimize(best_sequence, time_limit=min(remaining_p4 * 0.3, 60),
                                              top_k=max(40, n_best // 5),
                                              relax=1.0, step_init=0.2, seed=666)
            score = try_update(a_lp2)
            if score < best_score + 0.005:
                print("  Fine LP: %.6f" % score)

        # FW with very high beta
        remaining_p4 = phase4_time - (time.monotonic() - t4)
        if remaining_p4 > 20:
            a_fw2, _ = frank_wolfe_optimize(best_sequence, time_limit=min(remaining_p4 * 0.3, 30),
                                             beta=2000)
            score = try_update(a_fw2)
            if score < best_score + 0.005:
                print("  Fine FW: %.6f" % score)

    print("Phase 4 best: %.6f" % best_score)

    # ========== Phase 5: Final Perturbation Search (remaining) ==========
    print("=== Phase 5: Final Perturbation ===")
    t5 = time.monotonic()
    phase5_time = deadline - time.monotonic()

    if best_sequence is not None:
        n_best = len(best_sequence)

        # Try different scales of perturbation
        scales = [0.0002, 0.0005, 0.001, 0.002, 0.003, 0.005, 0.01]
        remaining = phase5_time - 10
        max_tries = min(int(remaining / 2), 100)

        for i in range(max_tries):
            if time.monotonic() > deadline - 15:
                break
            scale = scales[i % len(scales)]
            for attempt in range(2):
                noise = rng.standard_normal(n_best) * scale
                a_pert = project_simplex(best_sequence + noise)
                score = try_update(a_pert)
                if score < best_score:
                    break

        # One final very fine L-BFGS-B if time allows
        remaining = deadline - time.monotonic() - 10
        if remaining > 15:
            a_final, _ = lbfgsb_optimize(best_sequence, time_limit=remaining,
                                         beta_init=5000, beta_max=200000,
                                         beta_growth=1.003, seed=12345)
            score = try_update(a_final)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128) / 128

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if not np.isfinite(final_score):
        best_sequence = np.ones(128) / 128
        final_score = float(evaluate_sequence(best_sequence.tolist()))

    # Scale to a reasonable range (avoid very small or very large values)
    s = best_sequence.sum()
    if s > 1e-10:
        scale_factor = np.sqrt(2.0 * len(best_sequence)) / s
        best_sequence = best_sequence * scale_factor

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if not np.isfinite(final_score):
        # Fallback to original
        best_sequence = np.ones(128) / 128
        final_score = float(evaluate_sequence(best_sequence.tolist()))

    print("\nFINAL: %.6f (n=%d)" % (final_score, len(best_sequence)))
    return [float(x) for x in best_sequence]
