"""
Hybrid LP + smooth gradient + multiplicative update optimization
to minimize 2*n*max(conv(a,a))/(sum(a))^2.
Improvements over previous: full LP constraints, better step scheduling,
larger n exploration, multiplicative updates, Nesterov momentum.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(a):
    """Build constraint matrix for LP: (a*b)[k] <= M for all k."""
    n = len(a)
    num_k = 2 * n - 1
    A_ub = np.zeros((num_k, n))
    for k in range(num_k):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)
        A_ub[k, idx] = a[k - idx]
    return A_ub


def solve_lp_direction(a, M):
    """Solve LP: max sum(b) s.t. b >= 0, (a*b)[k] <= M for all k."""
    n = len(a)
    num_k = 2 * n - 1
    A_ub = build_lp_matrix(a)
    b_ub = np.full(num_k, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def smooth_gradient(a, beta=500.0):
    """Compute smooth gradient of max(conv(a,a)) using softmax weighting."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = beta * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    
    # grad[i] = 2 * cross_correlate(w, a)[i]
    # = 2 * ifft(fft(w) * conj(fft(a))).real
    fft_size = 1 << (max(len(w), n) + n - 2 - 1).bit_length()
    W = np.fft.fft(w, fft_size)
    A = np.fft.fft(a, fft_size)
    corr = np.fft.ifft(W * np.conj(A)).real[:n]
    grad = 2.0 * corr
    return grad, conv


def lp_optimize(a0, time_limit, step_init=1.0, relax=1.0, seed=0):
    """LP-based optimization on the simplex."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    step = step_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()
        
        b = solve_lp_direction(a, M * relax)
        if b is None:
            break
        
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        
        b = b / b_sum
        
        a_new = (1.0 - step) * a + step * b
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s
        else:
            a_new = project_simplex(a_new)
        
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve < 10:
                step *= 0.9
            elif no_improve < 30:
                step *= 0.85
            else:
                # Perturb and reset

                a = project_simplex(a + noise)
                step = step_init * 0.5
                no_improve = 0
    
    conv_final = fast_convolve(best_a)
    final_obj = 2.0 * n * conv_final.max()
    return best_a, final_obj


def grad_optimize(a0, time_limit, beta_init=500.0, lr_init=0.05,
                  momentum=0.9, seed=0):
    """Smooth gradient descent with momentum on the simplex."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    v = np.zeros(n)
    
    t0 = time.monotonic()
    beta = beta_init
    lr = lr_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        grad, conv = smooth_gradient(a, beta)
        g_norm = np.linalg.norm(grad)
        if g_norm > 1e-15:
            grad = grad / g_norm
        
        v = momentum * v - lr * grad
        a_new = project_simplex(a + v)
        
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                lr *= 0.8
                v *= 0.5
            if no_improve > 300:
                noise = rng.standard_normal(n) * 0.01
                a = project_simplex(a + noise)
                v = np.zeros(n)
                lr = lr_init * 0.5
                beta = max(100, beta * 0.5)

Hybrid LP + smooth gradient + geometric interpolation optimization

Uses full-constraint LP, correct cross-correlation gradients,

import numpy as np

from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho_val = np.max(rho[cond])
    theta = (cssv[rho_val - 1] - 1.0) / rho_val
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(a):
    """Build constraint matrix for LP: (a*b)[k] <= M for all k."""
    n = len(a)
    num_k = 2 * n - 1
    A_ub = np.zeros((num_k, n))
    for k in range(num_k):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1)
        A_ub[k, idx] = a[k - idx]
    return A_ub


def solve_lp_direction(a, M):
    """Solve LP: max sum(b) s.t. b >= 0, (a*b)[k] <= M for all k."""
    n = len(a)
    num_k = 2 * n - 1
    A_ub = build_lp_matrix(a)
    b_ub = np.full(num_k, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def smooth_gradient(a, beta=500.0):
    """
    Compute smooth gradient of max(conv(a,a)) via softmax-weighted sum.
    Uses the identity: d(a*a)[k]/da[i] = 2*a[k-i] (with a=0 outside [0,n-1]).
    So grad[i] = 2 * sum_k w[k] * a[k-i] = 2 * cross_correlate(w, a)[i].
    """
    n = len(a)
    conv = fast_convolve(a)
    scaled = beta * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

    # Cross-correlation: (w star a)[i] = sum_m w[m+i] * a[m]
    # = ifft(fft(w) * conj(fft(a))).real
    fft_size = 1 << (max(len(w), n) + n - 2 - 1).bit_length()
    W = np.fft.fft(w, fft_size)
    A = np.fft.fft(a, fft_size)
    corr = np.fft.ifft(W * np.conj(A)).real[:n]
    grad = 2.0 * corr
    return grad, conv


def geometric_interpolate(a, b, t):
    """Geometric interpolation: a^(1-t) * b^t, then normalize."""
    eps = 1e-15
    a_safe = np.maximum(a, eps)
    b_safe = np.maximum(b, eps)
    result = a_safe ** (1.0 - t) * b_safe ** t
    s = result.sum()
    if s > 1e-15:
        result = result / s
    else:
        result = (a + b) / 2.0
        result = result / result.sum()
    return result

    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Cosine window
    if n > 1:
        cos_w = np.cos(np.pi * x / (n - 1)) ** 2
        cos_w = np.maximum(cos_w, 0.01)
        cos_w /= cos_w.sum()
        inits['cosine'] = cos_w

    # Triangle (peaked at center)
    tri = np.minimum(x, n - 1 - x).astype(np.float64) + 1.0
    tri /= tri.sum()
    inits['triangle_c'] = tri

    # Triangle (peaked at edges)
    tri_e = (n - 1 - np.minimum(x, n - 1 - x)).astype(np.float64) + 1.0
    tri_e /= tri_e.sum()
    inits['triangle_e'] = tri_e

    # Gaussians
    for sf in [0.15, 0.25, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5 * n) / (sf * n)) ** 2)
        g /= g.sum()
        inits[f'gauss_{sf:.2f}'] = g

    # Bimodal (two Gaussians at edges)
    bi = np.exp(-0.5 * (x / (0.15 * n)) ** 2) + np.exp(-0.5 * ((x - n) / (0.15 * n)) ** 2)
    bi /= bi.sum()
    inits['bimodal'] = bi

    # Palindromic random
    half_n = (n + 1) // 2
    half = rng.dirichlet(np.ones(half_n) * 0.5)
    pal = np.zeros(n)
    pal[:half_n] = half
    pal[half_n:] = half[:n - half_n][::-1]
    pal /= pal.sum()
    inits['palindromic'] = pal

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['random_sparse'] = rng.dirichlet(np.ones(n) * 0.1)

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
            if len(p) == n:
                inits['parent'] = p.copy()
            elif len(p) > n:
                pp = p[:n].copy()
                pp /= pp.sum()
                inits['parent_crop'] = pp
            # Interpolate to length n
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(p)), p)
            pp = np.maximum(pp, 0)
            ps2 = pp.sum()
            if ps2 > 1e-10:
                inits['parent_interp'] = pp / ps2

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase hybrid optimization to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    Phase 1: Quick LP exploration over multiple n values.
    Phase 2: Deep LP optimization with full constraints.
    Phase 3: Smooth gradient / multiplicative fine-tuning.
    Phase 4: Hybrid alternating LP + gradient.
    Phase 5: Final polish with tiny steps.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(60.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(p)}, sum={p.sum():.4f}")
    except (NameError, Exception):
        pass

    def try_update(a, label):
        nonlocal best_sequence, best_score
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return False
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
            return True
        return False

    # ========== Phase 1: Exploration ==========
    print("=== Phase 1: Exploration ===")
    t1 = time.monotonic()
    p1_time = 130.0

    lengths = [256, 320, 384, 512, 640, 768, 1024]
    candidates = []

    for li, n in enumerate(lengths):
        if time.monotonic() - t1 > p1_time - 20:
            break
        inits = generate_inits(n, parent, rng)
        init_names = list(inits.keys())
        priority = ['parent', 'parent_interp', 'parent_crop', 'cosine',
                    'palindromic', 'gauss_0.25', 'random', 'bimodal']
        init_names = [nm for nm in priority if nm in inits] + \
                     [nm for nm in init_names if nm not in priority]
        init_names = init_names[:4]

        remaining = p1_time - (time.monotonic() - t1) - 10
        num_left = max(1, len(lengths) - li)
        time_per = max(8.0, remaining / num_left / max(1, len(init_names)))

        for name in init_names:
            if time.monotonic() - t1 > p1_time - 10:
                break
            a0 = inits[name]
            a_opt, obj_opt = lp_optimize(a0, time_limit=time_per,
                                         step_init=0.8, relax=1.0,
                                         seed=n + li * 7)
            candidates.append((obj_opt, n, name, a_opt.copy()))
            try_update(a_opt, f"P1 n={n} {name}")

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t1:.1f}s). Best: {best_score:.6f}")

    # ========== Phase 2: Deep LP ==========
    print("=== Phase 2: Deep LP ===")
    t2 = time.monotonic()
    p2_time = 520.0

    deep_starts = []
    seen = set()
    for obj, n, name, a in candidates[:10]:
        key = (n, name)
        if key not in seen:
            deep_starts.append((n, a.copy()))
            seen.add(key)
        if len(deep_starts) >= 5:
            break
    if best_sequence is not None and len(best_sequence) not in [d[0] for d in deep_starts]:
        deep_starts.append((len(best_sequence), best_sequence.copy()))

    configs = [
        {'step_init': 1.0, 'relax': 1.0, 'geometric': False},
        {'step_init': 0.8, 'relax': 1.0, 'geometric': True},
        {'step_init': 1.0, 'relax': 1.005, 'geometric': False},
        {'step_init': 0.6, 'relax': 0.995, 'geometric': False},
        {'step_init': 1.0, 'relax': 1.0, 'geometric': True},
    ]

    total_runs = len(deep_starts) * len(configs)
    time_per_run = p2_time / max(1, total_runs)
    run_id = 0

    for i, (n_s, a_s) in enumerate(deep_starts):
        for j, cfg in enumerate(configs):
            if time.monotonic() - t2 > p2_time - 15:
                break
            t_alloc = min(time_per_run, p2_time - (time.monotonic() - t2) - 15)
            if t_alloc < 10:
                break
            a_opt, obj_opt = lp_optimize(a_s, time_limit=t_alloc,
                                         seed=run_id * 13 + 5, **cfg)
            try_update(a_opt, f"P2 n={n_s} cfg{j} run{run_id}")
            run_id += 1
            # Update start point if improved
            if best_sequence is not None and len(best_sequence) == n_s:
                a_s = best_sequence.copy()

    print(f"Phase 2 done ({time.monotonic()-t2:.1f}s). Best: {best_score:.6f}")

    # ========== Phase 3: Gradient / Multiplicative ==========
    print("=== Phase 3: Gradient Fine-tuning ===")
    t3 = time.monotonic()
    p3_time = 320.0

    if best_sequence is not None:
        start_a = best_sequence.copy()
        n_best = len(start_a)

        grad_configs = [
            {'beta_init': 300, 'lr_init': 0.04, 'momentum': 0.9, 'nesterov': False},
            {'beta_init': 800, 'lr_init': 0.02, 'momentum': 0.92, 'nesterov': False},
            {'beta_init': 500, 'lr_init': 0.03, 'momentum': 0.9, 'nesterov': True},
            {'beta_init': 1500, 'lr_init': 0.015, 'momentum': 0.93, 'nesterov': False},
            {'beta_init': 200, 'lr_init': 0.06, 'momentum': 0.85, 'nesterov': False},
        ]

        time_per = p3_time / (len(grad_configs) + 3)

        for i, cfg in enumerate(grad_configs):
            if time.monotonic() - t3 > p3_time - 10:
                break
            t_alloc = min(time_per, p3_time - (time.monotonic() - t3) - 10)
            if t_alloc < 10:
                break
            a_opt, obj_opt = grad_optimize(start_a, time_limit=t_alloc,
                                           seed=i * 17 + 3, **cfg)
            try_update(a_opt, f"P3 grad{i}")

        # Multiplicative
        for i in range(3):
            if time.monotonic() - t3 > p3_time - 10:
                break
            t_alloc = min(time_per, p3_time - (time.monotonic() - t3) - 10)
            if t_alloc < 10:
                break
            a_opt, obj_opt = multiplicative_optimize(
                start_a, time_limit=t_alloc,
                beta_init=400 + i * 300, eta_init=0.1 + i * 0.05,
                seed=50 + i * 11)
            try_update(a_opt, f"P3 mult{i}")

    print(f"Phase 3 done ({time.monotonic()-t3:.1f}s). Best: {best_score:.6f}")

    # ========== Phase 4: Hybrid Alternating ==========
    print("=== Phase 4: Hybrid ===")
    t4 = time.monotonic()
    p4_time = 400.0

    if best_sequence is not None:
        n_best = len(best_sequence)
        a = best_sequence.copy()
        v = np.zeros(n_best)

        hybrid_configs = [
            {'lp_step': 0.8, 'geo': False, 'gbeta': 800, 'glr': 0.025, 'gmom': 0.9},
            {'lp_step': 1.0, 'geo': True, 'gbeta': 500, 'glr': 0.03, 'gmom': 0.88},
            {'lp_step': 0.6, 'geo': False, 'gbeta': 1500, 'glr': 0.015, 'gmom': 0.92},
        ]

        time_per_hp = p4_time / len(hybrid_configs)

        for hci, hcfg in enumerate(hybrid_configs):
            if time.monotonic() - t4 > p4_time - 30:
                break
            t_alloc = min(time_per_hp, p4_time - (time.monotonic() - t4) - 30)
            if t_alloc < 30:
                break

            a_curr = a.copy()
            v_curr = np.zeros(n_best)
            best_phase = a.copy()
            best_phase_obj = 2.0 * n_best * fast_convolve(a).max()
            step = hcfg['lp_step']

            phase_t0 = time.monotonic()
            while time.monotonic() - phase_t0 < t_alloc:
                # LP step
                conv = fast_convolve(a_curr)
                M = conv.max()
                b = solve_lp_direction(a_curr, M)
                if b is not None:
                    b_sum = b.sum()
                    if b_sum > 1e-15:
                        b = b / b_sum
                        if hcfg['geo']:
                            a_curr = geometric_interpolate(a_curr, b, min(step, 1.0))
                        else:
                            a_curr = (1.0 - step) * a_curr + step * b
                            a_curr = np.maximum(a_curr, 0.0)
                            s = a_curr.sum()
                            if s > 1e-15:
                                a_curr = a_curr / s
                            else:
                                a_curr = project_simplex(a_curr)

                # Gradient step
                grad, conv = smooth_gradient(a_curr, hcfg['gbeta'])
                g_norm = np.linalg.norm(grad)
                if g_norm > 1e-15:
                    grad = grad / g_norm
                v_curr = hcfg['gmom'] * v_curr - hcfg['glr'] * grad
                a_curr = project_simplex(a_curr + v_curr)

                conv = fast_convolve(a_curr)
                obj = 2.0 * n_best * conv.max()
                if obj < best_phase_obj - 1e-13:
                    best_phase_obj = obj
                    best_phase = a_curr.copy()

            score = float(evaluate_sequence(best_phase.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = best_phase.copy()
                a = best_phase.copy()
                print(f"  P4 hybrid[{hci}]: {score:.6f}")

    print(f"Phase 4 done ({time.monotonic()-t4:.1f}s). Best: {best_score:.6f}")

    # ========== Phase 5: Final Polish ==========
    print("=== Phase 5: Polish ===")
    t5 = time.monotonic()
    p5_time = deadline - time.monotonic()

    if best_sequence is not None and p5_time > 20:
        n_best = len(best_sequence)

        # Fine LP
        t_alloc = min(p5_time * 0.35, 80.0)
        if t_alloc > 15:
            a_opt, _ = lp_optimize(best_sequence, time_limit=t_alloc,
                                   step_init=0.4, relax=1.0,
                                   geometric=True, seed=777)
            try_update(a_opt, "P5 LP polish")

        # Fine gradient
        t_alloc = min(p5_time * 0.3, 70.0)
        if t_alloc > 15:
            a_opt, _ = grad_optimize(best_sequence, time_limit=t_alloc,
                                     beta_init=3000, lr_init=0.008,
                                     momentum=0.95, nesterov=True, seed=888)
            try_update(a_opt, "P5 grad polish")

        # Tiny perturbations
        remaining = deadline - time.monotonic() - 15.0
        if remaining > 10:
            n_tries = int(remaining / 3)
            n_tries = min(n_tries, 30)
            for i in range(n_tries):
                if time.monotonic() > deadline - 15:
                    break
                improved = False
                for scale in [0.0005, 0.001, 0.002, 0.005]:
                    noise = rng.standard_normal(n_best) * scale
                    a_pert = project_simplex(best_sequence + noise)
                    s = float(evaluate_sequence(a_pert.tolist()))
                    if np.isfinite(s) and s < best_score:
                        best_score = s
                        best_sequence = a_pert.copy()
                        print(f"  P5 perturb[{i}] s={scale}: {s:.6f}")
                        improved = True
                        break
                if improved:
                    continue

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
