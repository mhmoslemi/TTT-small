Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a novel combination of:
1. L-BFGS optimization with log-sum-exp smoothing in softmax parameterization
2. LP-based direction finding for global guidance
3. Multi-start with diverse initializations and adaptive restarts
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - np.max(x)
    exp_x = np.exp(x_shifted)
    s = exp_x.sum()
    return exp_x / s


def compute_objective(a):
    """Compute the objective value for a normalized sequence."""
    n = len(a)
    conv = fast_convolve(a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def smooth_obj_and_grad(x, t):
    """
    Compute the smooth objective and gradient using log-sum-exp.
    x: pre-softmax parameters
    t: smoothing parameter
    Returns: (objective_value, gradient)
    """
    n = len(x)
    a = softmax(x)
    conv = fast_convolve(a)
    
    # Log-sum-exp computation
    t_conv = t * conv
    t_conv_max = t_conv.max()
    # For numerical stability
    exp_conv = np.exp(t_conv - t_conv_max)
    log_sum_exp = t_conv_max + np.log(exp_conv.sum())
    obj = 2.0 * n * log_sum_exp / t
    
    # Weights for the softmax over conv values
    weights = exp_conv / exp_conv.sum()
    
    # Compute grad_a[j] = sum_k weights[k] * a[k-j] for valid k
    n_conv = len(conv)  # 2n-1
    grad_a = np.zeros(n)
    for j in range(n):
        # k ranges from j to min(j + n - 1, n_conv - 1)
        k_start = j
        k_end = min(j + n, n_conv)
        if k_start < k_end:
            # grad_a[j] = sum_{k=k_start}^{k_end-1} weights[k] * a[k-j]
            grad_a[j] = np.dot(weights[k_start:k_end], a[:k_end - k_start])
    
    # Chain rule through softmax
    # d(a_j)/d(x_i) = a_j * (delta(j,i) - a_i)
    # df/dx_i = sum_j df/da_j * da_j/dx_i = a_i * (grad_a_i - sum_j a_j * grad_a_j)
    grad_a_dot = np.dot(grad_a, a)
    grad_x = 2.0 * n / t * a * (grad_a - grad_a_dot)
    
    return obj, grad_x

        a = a / s

    # x_i = log(a_i) works if a_i > 0

    x = np.log(np.maximum(a, 1e-30))
    
    best_a = a.copy()
    best_obj = compute_objective(a)

            smooth_obj_and_grad,
            x,

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS optimization with log-sum-exp smoothing in softmax parameterization
2. LP-based direction finding for global guidance
3. Mean-field steepest descent for global optimization
4. Multi-start with diverse initializations and adaptive restarts
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - np.max(x)
    exp_x = np.exp(x_shifted)
    s = exp_x.sum()
    return exp_x / s


def compute_objective(a):
    """Compute the objective value for a sequence."""
    n = len(a)
    conv = fast_convolve(a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * n * max_b / (sum_a ** 2)


def build_conv_matrix(a):
    """Build matrix C such that C @ g gives the linear convolution a * g."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def smooth_obj_and_grad(x, t):
    """
    Compute the smooth objective and gradient using log-sum-exp.
    x: pre-softmax parameters
    t: smoothing parameter (larger = closer to true max)
    Returns: (objective_value, gradient)
    """
    n = len(x)
    a = softmax(x)
    conv = fast_convolve(a)
    n_conv = len(conv)  # 2n-1

    # Log-sum-exp computation
    t_conv = t * conv
    t_conv_max = t_conv.max()
    exp_conv = np.exp(t_conv - t_conv_max)
    log_sum_exp = t_conv_max + np.log(exp_conv.sum())
    obj = 2.0 * n * log_sum_exp / t

    # Weights for the softmax over conv values
    weights = exp_conv / exp_conv.sum()

    # Compute G[j0] = sum_k weights[k] * d(conv[k])/d(a[j0])
    # d(conv[k])/d(a[j0]) = 2*a[k-j0] if 0<=k-j0<n and k!=2*j0
    #                      = 2*a[j0] if k==2*j0
    #                      = 2*a[k-j0] if k!=2*j0 (same formula)
    # Unified: 2*a[k-j0] when 0<=k-j0<n, but when k=2*j0 it's 2*a[j0]
    # Note: when k=2*j0, a[k-j0] = a[j0], so formula 2*a[k-j0] works in all cases!
    # EXCEPT: when k=2*j0, a[j0] appears only ONCE in the sum (as a[j0]*a[j0]),
    # so its derivative is 2*a[j0], not 2*a[k-j0] from two separate occurrences.
    # But 2*a[k-j0] = 2*a[j0] when k=2*j0, so the formula is the same!
    # The only issue is that when k=2*j0, it should be counted once, not twice.
    # In the sum over k, each k contributes independently, so for a fixed j0:
    # G[j0] = sum_k weights[k] * 2*a[k-j0] * I(0<=k-j0<n)
    # But when k=2*j0, this gives 2*a[j0] which is correct (derivative of a[j0]^2).
    # When k!=2*j0, this gives 2*a[k-j0] which is also correct.
    # So the unified formula is: G[j0] = sum_k weights[k] * 2*a[k-j0] * I(0<=k-j0<n)
    # which simplifies to: G[j0] = 2 * sum_k weights[k] * a[k-j0] * I(0<=k-j0<n)

    G = np.zeros(n)
    for j0 in range(n):
        # k ranges from j0 to min(j0+n-1, n_conv-1)
        k_start = j0
        k_end = min(j0 + n, n_conv)
        if k_start < k_end:
            G[j0] = 2.0 * np.dot(weights[k_start:k_end], a[:k_end - k_start])

    # Chain rule through softmax
    # df/dx_i = 2n * a[i] * (G[i] - sum_j G[j]*a[j])
    G_dot_a = np.dot(G, a)
    grad_x = 2.0 * n * a * (G - G_dot_a)

    return obj, grad_x


def mean_field_step(a, peaks, t=50.0, alpha=0.1):
    """
    Mean-field steepest descent step to minimize max(conv).
    Each coordinate decreases at rate proportional to its own value times the
    weighted peak contribution. This globally reduces the maximum.
    """
    n = len(a)
    conv = fast_convolve(a)
    n_conv = len(conv)

    # Compute weights from peaks using softmax
    peak_vals = conv[peaks]
    scaled = t * peak_vals
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(peak_vals))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w = w / w_sum

    # Compute G[j0] = sum_k weights[k] * 2*a[k-j0]
    # But we only need to sum over the peak indices
    G = np.zeros(n)
    for idx in range(len(peak_vals)):
        if w[idx] < 1e-12:
            continue
        k = int(peaks[idx])
        wi = w[idx]
        # For this k, G[j0] += 2 * wi * a[k-j0] for valid j0
        j0_start = max(0, k - n + 1)
        j0_end = min(n, k + 1)
        for j0 in range(j0_start, j0_end):
            G[j0] += 2.0 * wi * a[k - j0]

    # Mean-field update: each a_j decreases proportionally to a_j * G_j
    # and the total sum is preserved
    # a' = a * (1 - alpha * G) / (1 - alpha * dot(G, a))
    dot_G_a = np.dot(G, a)
    denom = 1.0 - alpha * dot_G_a
    if denom < 0.1:
        alpha = 0.01 / max(dot_G_a, 1e-10)
        denom = 1.0 - alpha * dot_G_a
    a_new = a * (1.0 - alpha * G) / denom
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new


def lp_direction(a, M):
    """
    Solve LP: maximize sum(g) s.t. conv(a,g) <= M, g >= 0.
    Returns the optimal g (normalized) or None.
    """
    n = len(a)
    C = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        result = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 30})
        if result.success:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum
    except Exception:
        pass
    return None


def line_search(a, d, best_obj):
    """
    Line search: find best t in [0,1] such that a + t*d improves the objective.
    """
    n = len(a)
    best_t = 0.0
    best_obj_local = best_obj

    candidates = [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05]
    for t in candidates:
        a_try = a + t * d
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-15:
            continue
        a_try = a_try / s
        obj_try = compute_objective(a_try)
        if obj_try < best_obj_local - 1e-12:
            best_obj_local = obj_try
            best_t = t

    # Refine
    if 0.05 < best_t < 0.95:
        step = 0.05
        for _ in range(4):
            improved = False
            for dt in [-step, step]:
                t_try = best_t + dt
                if t_try <= 0 or t_try >= 1:
                    continue
                a_try = a + t_try * d
                a_try = np.maximum(a_try, 0.0)
                s = a_try.sum()
                if s < 1e-15:
                    continue
                a_try = a_try / s
                obj_try = compute_objective(a_try)
                if obj_try < best_obj_local - 1e-12:
                    best_obj_local = obj_try
                    best_t = t_try
                    improved = True
            if not improved:
                break
            step *= 0.5

    if best_t > 0:
        a_new = a + best_t * d
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s
        return a_new, best_obj_local
    return a, best_obj


def optimize_lnbfgs(a0, t_start=20.0, t_end=500.0, n_anneal=5, verbose=False):
    """
    L-BFGS optimization with annealed log-sum-exp smoothing.
    """
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s

    n = len(a)
    best_a = a.copy()
    best_obj = compute_objective(a)

    # Initialize x
    x = np.log(np.maximum(a, 1e-30))

    # Anneal t
    t_values = np.geomspace(t_start, t_end, n_anneal)

    for ti, t in enumerate(t_values):
        max_iter = 300 if t < 100 else 100
        result = minimize(
            smooth_obj_and_grad,
            x,
            method='L-BFGS-B',
            jac=True,
            args=(t,),
            options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-9}
        )
        x = result.x
        a_new = softmax(x)
        obj_new = compute_objective(a_new)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            if verbose:
                print(f"    L-BFGS t={t:.0f}: obj={best_obj:.8f}")

    return best_a, best_obj


def optimize_lp_iterative(a0, time_limit, verbose=False):
    """
    LP-based iterative optimization.
    """
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:

    n = len(a)

    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    iteration = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        M_lp = M * 1.001
        g = lp_direction(a, M_lp)

        if g is None:
            no_improve += 10
        else:
            d = g - a
            norm_d = np.linalg.norm(d)
            if norm_d < 1e-10:
                no_improve += 1
            else:
                a_new, obj_new = line_search(a, d, best_obj)
                if obj_new < best_obj - 1e-12:
                    a = a_new
                    conv = fast_convolve(a)
                    M = conv.max()
                    best_obj = 2.0 * n * M
                    best_a = a.copy()
                    no_improve = 0
                    if verbose and iteration % 50 == 0:
                        print(f"    LP iter {iteration}: obj={best_obj:.8f}")
                else:
                    no_improve += 1

        iteration += 1

        if no_improve >= 100:
            # Perturb and restart
            rng = np.random.default_rng(iteration * 7 + 13)
            scale = 0.005 + 0.005 * min(no_improve // 100, 5)
            noise = rng.standard_normal(n) * scale
            a = best_a + noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0

            if no_improve > 500:
                break

def optimize_mean_field(a0, time_limit, verbose=False):
    """
    Global optimization using mean-field steepest descent.
    """
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    alpha = 0.15
    t_param = 30.0
    no_improve = 0
    top_k = 30

    while time.monotonic() - t0 < time_limit:
        # Batch of steps
        for _ in range(50):
            if time.monotonic() - t0 > time_limit:
                break

            conv = fast_convolve(a)
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            a = mean_field_step(a, peaks, t=t_param, alpha=alpha)

        alpha *= 0.995
        t_param *= 1.005
        top_k = min(50, top_k + 1)

            break

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Triangular (center-heavy)
    tri = np.minimum(x, n - 1 - x) + 1.0
    inits['triangle'] = tri / tri.sum()

    # Bimodal (edge-heavy)
    bi = np.zeros(n)
    edge = max(1, n // 4)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.2 * n)) ** 2)
    inits['gaussian'] = g / g.sum()

    # Ramp
    ramp = x + 1.0
    inits['ramp'] = ramp / ramp.sum()

    # Sine
    sig = np.sin(np.pi * x / n) + 0.05
    inits['sine'] = sig / sig.sum()

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            if len(p) == n:
                inits['parent'] = p / ps
            elif len(p) >= n:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_resample'] = pp / pps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_upsample'] = pp / pps

    return inits

                        best_sequence = a_opt2.copy()
                        print(f"  L-BFGS polish: {score2:.6f}")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")

    # Phase 3: LP-based iterative optimization
    print("=== Phase 3: LP Iterative ===")
    t_start3 = time.monotonic()
    if best_sequence is not None:
        remaining = min(300.0, deadline - time.monotonic() - 100.0)
        if remaining > 30:
            a_opt, obj_opt = optimize_lp_iterative(best_sequence, time_limit=remaining)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  LP: {score:.6f}")

            # L-BFGS polish
            if time.monotonic() < deadline - 60:
                t_alloc = min(100.0, deadline - time.monotonic() - 50)
                if t_alloc > 15:
                    a_opt2, obj_opt2 = optimize_lnbfgs(best_sequence, t_start=100, t_end=800, n_anneal=4)
                    score2 = float(evaluate_sequence(a_opt2.tolist()))
                    if np.isfinite(score2) and score2 < best_score:
                        best_score = score2
                        best_sequence = a_opt2.copy()
                        print(f"  L-BFGS after LP: {score2:.6f}")

    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")

    # Phase 4: Multi-start with perturbations + L-BFGS
    print("=== Phase 4: Multi-Start Perturbation ===")
    t_start4 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 50.0
        if remaining > 30:
            n_best = len(best_sequence)
            n_perturb = min(8, max(2, int(remaining / 40)))
            t_per_perturb = remaining / n_perturb

            for pi in range(n_perturb):
                if time.monotonic() > deadline - 30:
                    break
                if t_per_perturb < 10:
                    break

                # Different perturbation strategies
                scales = [0.005, 0.01, 0.003, 0.02, 0.008, 0.015, 0.004, 0.012]
                scale = scales[pi % len(scales)]
                noise = rng.standard_normal(n_best) * scale
                a_pert = best_sequence + noise
                a_pert = np.maximum(a_pert, 0.0)
                s = a_pert.sum()
                if s > 1e-15:
                    a_pert = a_pert / s

                # L-BFGS from perturbed
                t_alloc = min(t_per_perturb, deadline - time.monotonic() - 25)
                if t_alloc < 10:
                    break
                a_opt, obj_opt = optimize_lnbfgs(a_pert, t_start=50, t_end=400, n_anneal=3)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Perturb {pi} (scale={scale}): {score:.6f}")

    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")

    # Phase 5: Try different lengths
    print("=== Phase 5: Different Lengths ===")
    t_start5 = time.monotonic()
    if time.monotonic() < deadline - 60:
        for n_try in [256, 1024]:
            if time.monotonic() > deadline - 40:
                break
            t_alloc = min(120.0, deadline - time.monotonic() - 30)
            if t_alloc < 20:
                break

            inits_try = generate_initializations(n_try, parent, rng)
            # Use parent if available, else uniform
            if 'parent' in inits_try:
                a0 = inits_try['parent']
            elif 'parent_resample' in inits_try:
                a0 = inits_try['parent_resample']
            elif 'parent_upsample' in inits_try:
                a0 = inits_try['parent_upsample']
            else:
                a0 = inits_try['uniform']

            a_opt, obj_opt = optimize_lnbfgs(a0, t_start=30, t_end=300, n_anneal=4)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  n={n_try}: {score:.6f}")

    print(f"Phase 5 done ({time.monotonic()-t_start5:.0f}s). Best: {best_score:.6f}")

    # Phase 6: Final L-BFGS polish
    print("=== Phase 6: Final Polish ===")
    if best_sequence is not None and time.monotonic() < deadline - 10:
        t_alloc = min(60.0, deadline - time.monotonic() - 5.0)
        if t_alloc > 10:
            a_opt, obj_opt = optimize_lnbfgs(best_sequence, t_start=200, t_end=1000, n_anneal=3)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final: {score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
