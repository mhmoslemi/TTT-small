"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using coordinate descent with
golden-section 1D optimization, LP-based refinement, and stochastic
perturbation. Exploits the fact that varying one coordinate at a time
gives a 1D problem solvable via equilibration of top convolution peaks.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def objective_value(a):
    """Compute the objective 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    conv = fast_convolve(a)
    return float(2.0 * n * conv.max() / (S * S))


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A where (A @ b)[k] = conv(a, b)[k] for k=0..2n-2."""
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A


def lp_maximize_sum(a, M):
    """Solve: max sum(b) s.t. b >= 0, conv(a,b)[k] <= M for all k."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x, 0)
    return None


def lp_minimize_peak(a):
    """Solve: min M' s.t. sum(b)=1, b>=0, conv(a,b)[k] <= M'."""
    n = len(a)
    A = build_conv_matrix(a)
    num_rows = 2 * n - 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    A_ub_mat = np.zeros((num_rows, n + 1))
    A_ub_mat[:, :n] = A
    A_ub_mat[:, n] = -1.0
    b_ub = np.zeros(num_rows)
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub_mat, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x[:n], 0), res.x[n]
    return None, None

                    lo_gs = max(0.0, best_x - 0.01)
                    hi_gs = min(1.0, best_x + 0.01)

                c_gs = hi_gs - gphi * (hi_gs - lo_gs)
                d_gs = lo_gs + gphi * (hi_gs - lo_gs)
                fc = 2.0 * n * (conv_base + c_gs * d_vec).max()
                fd = 2.0 * n * (conv_base + d_gs * d_vec).max()

                for _ in range(25):
                    if fc < fd:
                        hi_gs = d_gs
                        d_gs = c_gs
                        fd = fc
                        c_gs = hi_gs - gphi * (hi_gs - lo_gs)
                        fc = 2.0 * n * (conv_base + c_gs * d_vec).max()
                    else:
                        lo_gs = c_gs
                        c_gs = d_gs
                        fc = fd
                        d_gs = lo_gs + gphi * (hi_gs - lo_gs)
                        fd = 2.0 * n * (conv_base + d_gs * d_vec).max()

                x_gs = (lo_gs + hi_gs) / 2.0
                obj_gs = 2.0 * n * (conv_base + x_gs * d_vec).max()
                if obj_gs < best_local_obj - 1e-14:
                    best_x = x_gs
                    best_local_obj = obj_gs

            # Apply update
            if abs(best_x - x) > 1e-13:
                delta = best_x - x
                conv += delta * d_vec
                a[i] = best_x
                any_change = True

                cur_obj = 2.0 * n * conv.max()
                if cur_obj < best_obj - 1e-12:
                    best_obj = cur_obj
                    best_a = a.copy()

        sweep += 1
        if not any_change:
            break

    return best_a


def lp_iterate(a_init, time_limit, method='max_sum', max_iters=300, step_init=0.3):
    """LP-based iterative refinement."""
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    step = step_init
    no_improve = 0
    cur_method = method

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        M = conv.max()

        if cur_method == 'max_sum':
            b = lp_maximize_sum(a, M)
            if b is None:
                break
            b_sum = b.sum()
            if b_sum < 1e-15:
                break
            b = b / b_sum
        else:
            b, _ = lp_minimize_peak(a)
            if b is None:
                break
            b_sum = b.sum()
            if b_sum < 1e-15:
                break
            b = b / b_sum

        if np.linalg.norm(b - a) < 1e-12:
            break

        a_new = (1.0 - step) * a + step * b
        a_new = np.maximum(a_new, 0.0)
        a_sum = a_new.sum()
        if a_sum < 1e-15:
            break
        a_new = a_new / a_sum

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            step = min(0.8, step * 1.05)
        else:
            no_improve += 1
            step = max(0.05, step * 0.93)

        a = a_new
        conv = conv_new

        if no_improve > 100:
            cur_method = 'min_peak' if cur_method == 'max_sum' else 'max_sum'
            step = step_init
            no_improve = 0
        if no_improve > 400:
            break

    return best_a


def stochastic_refine(a_init, time_limit, seed=123):
    """Stochastic perturbation + coordinate descent restarts."""
    n = len(a_init)
    a = np.maximum(a_init.copy(), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()

    perturbation_scale = 0.05
    cd_time_per_iter = min(30.0, time_limit / 50)

    while time.monotonic() - t0 < time_limit - 5:
        # Perturb
        a_pert = a + rng.standard_normal(n) * perturbation_scale
        a_pert = project_simplex(a_pert)

        # Coordinate descent on perturbed
        remain = time.monotonic() - t0
        t_alloc = min(cd_time_per_iter, time_limit - 10 - remain)
        if t_alloc < 5:
            break

        a_opt = coordinate_descent(a_pert, time_limit=t_alloc,
                                   max_sweeps=200, random_order=True,
                                   seed=rng.integers(0, 2**31))
        obj_opt = 2.0 * n * fast_convolve(a_opt).max()

        if obj_opt < best_obj - 1e-12:
            best_obj = obj_opt
            best_a = a_opt.copy()
            a = a_opt.copy()
            print(f"    Stochastic improvement: {obj_opt:.10f}")
        else:
            # Decay perturbation
            perturbation_scale *= 0.9
            if perturbation_scale < 0.001:
                perturbation_scale = 0.02  # Reset occasionally

    return best_a


def multi_length_search(parent, time_limit, budget_per_length=60):
    """Try different sequence lengths around the parent length."""
    n_parent = len(parent)
    candidates_n = []
    for offset in [-5, -3, -1, 1, 3, 5, 8, 16, 32, 64, 128]:
        n_try = n_parent + offset
        if 32 <= n_try <= 2048:
            candidates_n.append(n_try)

    candidates_n = sorted(set(candidates_n))

    best_a = None
    best_score = float('inf')

    for n_try in candidates_n:
        if time.monotonic() - time_limit + time_limit > 0 and (time.monotonic() - (time_limit - budget_per_length * len(candidates_n))) > time_limit:
            break
        # Interpolate parent to new length
        if n_try != n_parent:
            p = np.interp(np.linspace(0, 1, n_try),
                          np.linspace(0, 1, n_parent), parent)
            p = np.maximum(p, 0)
            s = p.sum()
            if s > 1e-10:
                p = p / s
            else:
                p = np.ones(n_try) / n_try
        else:
            p = parent.copy()

        # Quick coordinate descent
        t_cd = min(budget_per_length, max(10, time_limit))
        a_opt = coordinate_descent(p, time_limit=t_cd, max_sweeps=100,
                                   random_order=True, seed=n_try)
        obj = 2.0 * n_try * fast_convolve(a_opt).max()
        score = float(evaluate_sequence(a_opt.tolist()))

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a_opt.copy()
            print(f"    Length {n_try}: score={score:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization:
    Phase 1: Coordinate descent from parent (fast local optimization)
    Phase 2: LP-based refinement (principled direction finding)
    Phase 3: Stochastic refinement (escape local minima)
    Phase 4: Multi-length exploration
    Phase 5: Final polish
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    rng = np.random.default_rng(seed)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: len={len(p)}, score_estimate={2*len(p)*fast_convolve(parent/p.sum()).max():.8f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        print("No parent available, using uniform 512.")

    def try_update(a):
        nonlocal best_sequence, best_score
        if a is None or len(a) == 0:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print(f"  New best: {score:.8f} (n={len(a)})")

    # Phase 1: Coordinate descent (600s)
    print("=== Phase 1: Coordinate Descent ===")
    t1 = time.monotonic()
    t1_limit = min(600.0, (deadline - time.monotonic()) * 0.4)

    a_cd = coordinate_descent(parent, time_limit=t1_limit, max_sweeps=1000,
                              random_order=False, seed=42)
    try_update(a_cd)

    # Continue CD with random order
    remain_p1 = (deadline - time.monotonic()) * 0.35
    if remain_p1 > 30 and best_sequence is not None:
        a_cd2 = coordinate_descent(best_sequence, time_limit=remain_p1,
                                   max_sweeps=500, random_order=True, seed=99)
        try_update(a_cd2)

    print(f"Phase 1 done: best={best_score:.8f}, elapsed={time.monotonic()-t1:.0f}s")

    # Phase 2: LP-based refinement (400s)
    print("=== Phase 2: LP Refinement ===")
    t2 = time.monotonic()
    t2_limit = min(400.0, (deadline - time.monotonic()) * 0.25)

    if best_sequence is not None:
        for step_val in [0.3, 0.5, 0.15]:
            if time.monotonic() > deadline - 400:
                break
            t_alloc = min(t2_limit / 3, deadline - time.monotonic() - 350)
            if t_alloc < 10:
                break
            for method in ['max_sum', 'min_peak']:
                if time.monotonic() > deadline - 400:
                    break
                t_single = t_alloc / 2
                if t_single < 5:
                    break
                a_lp = lp_iterate(best_sequence, time_limit=t_single,
                                  method=method, max_iters=400,
                                  step_init=step_val)
                try_update(a_lp)
                # Polish with CD after LP
                if best_sequence is not None:
                    t_cd_polish = min(20.0, deadline - time.monotonic() - 350)
                    if t_cd_polish > 5:
                        a_pol = coordinate_descent(best_sequence,
                                                   time_limit=t_cd_polish,
                                                   max_sweeps=100)
                        try_update(a_pol)

    print(f"Phase 2 done: best={best_score:.8f}, elapsed={time.monotonic()-t2:.0f}s")

    # Phase 3: Stochastic refinement (400s)
    print("=== Phase 3: Stochastic Refinement ===")
    t3 = time.monotonic()
    t3_limit = min(400.0, (deadline - time.monotonic()) * 0.25)

    if best_sequence is not None:
        a_stoch = stochastic_refine(best_sequence, time_limit=t3_limit, seed=2024)
        try_update(a_stoch)

    print(f"Phase 3 done: best={best_score:.8f}, elapsed={time.monotonic()-t3:.0f}s")

    # Phase 4: Multi-length exploration (250s)
    print("=== Phase 4: Multi-Length ===")
    t4 = time.monotonic()
    t4_limit = min(250.0, deadline - time.monotonic() - 150)

    if t4_limit > 30:
        ref_parent = best_sequence if best_sequence is not None else parent
        a_ml, score_ml = multi_length_search(ref_parent, t0=time.monotonic(),
                                             budget_per_length=t4_limit / 8)
        if a_ml is not None:
            try_update(a_ml)

    print(f"Phase 4 done: best={best_score:.8f}, elapsed={time.monotonic()-t4:.0f}s")

    # Phase 5: Final polish (remaining time, reserve 60s)
    print("=== Phase 5: Final Polish ===")
    t5 = time.monotonic()
    t5_limit = max(10.0, deadline - time.monotonic() - 60)

    if best_sequence is not None and t5_limit > 20:
        # CD polish
        t_cd_final = t5_limit * 0.5
        a_final = coordinate_descent(best_sequence, time_limit=t_cd_final,
                                     max_sweeps=500, random_order=True, seed=777)
        try_update(a_final)

        # Short LP
        remain5 = deadline - time.monotonic() - 60
        if remain5 > 30:
            a_lp_fin = lp_iterate(best_sequence, time_limit=remain5 * 0.5,
                                  method='min_peak', max_iters=200, step_init=0.2)
            try_update(a_lp_fin)
            # CD after LP
            remain5b = deadline - time.monotonic() - 60
            if remain5b > 15:
                a_pol_fin = coordinate_descent(best_sequence, time_limit=remain5b,
                                               max_sweeps=200)
                try_update(a_pol_fin)

    print(f"Phase 5 done: best={best_score:.8f}, elapsed={time.monotonic()-t5:.0f}s")

    # Final fallback
    if best_sequence is None:
        best_sequence = parent

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        if final_score < best_score:
            best_score = final_score
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_sequence)}")
    else:
        # Fatal: return something valid
        best_sequence = np.ones(128) / 128
        best_score = float(evaluate_sequence(best_sequence.tolist()))
        print(f"FALLBACK: score={best_score:.8f}")

    return [float(x) for x in best_sequence]
