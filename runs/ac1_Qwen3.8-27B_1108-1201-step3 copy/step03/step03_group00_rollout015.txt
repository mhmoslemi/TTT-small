"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-peak subgradient descent
with annealed softmax weighting, bisection line search, structured initialization,
and pair-based local search.
"""

import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_score(a):
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max() / (s * s)

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_multipeak_grad(a, conv, top_k=30, temp=0.001):
    """Compute weighted subgradient of max conv using softmax over top peaks."""
    n = len(a)
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    pv = conv[peaks]
    sv = pv / temp
    sv -= sv.max()
    w = np.exp(sv)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-13:
            continue
        k = int(peaks[idx])
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        idx_arr = np.arange(i_lo, i_hi + 1)
        grad[i_lo:i_hi+1] += 2.0 * w[idx] * a[k - idx_arr]
    return grad

def line_search(a, direction, max_lr=0.1, n_bisect=10):
    """Bisection line search in the direction -direction (after projection)."""
    best_alpha = 0.0
    best_score = compute_score(a)
    
    # Coarse
    candidates = [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5]
    for alpha in candidates:
        a_try = project_simplex(a - alpha * direction)
        sc = compute_score(a_try)
        if sc < best_score:
            best_score = sc
            best_alpha = alpha
    
    # Fine bisection
    if best_alpha > 0:
        lo = max(0, best_alpha * 0.5)
        hi = min(1.0, best_alpha * 2.0)
        for _ in range(n_bisect):
            mid = (lo + hi) / 2
            a_try = project_simplex(a - mid * direction)
            sc = compute_score(a_try)
            if sc <= best_score:
                best_score = sc
                best_alpha = mid
                lo = mid
            else:
                hi = mid
    
    if best_alpha > 0:
        a_new = project_simplex(a - best_alpha * direction)
    else:
        a_new = a.copy()
    return a_new, best_score

def optimize_sequence(a0, time_limit, verbose=False):
    """Optimize a sequence using multi-peak subgradient + local search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_score = compute_score(a)
    if verbose:
        print(f"  Initial score: {best_score:.8f}")
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    temp = 0.0005
    base_lr = 0.008
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        phase_frac = (time.monotonic() - t0) / time_limit
        
        if phase_frac < 0.5:
            # Phase 1: Subgradient with annealing
            conv = fast_convolve(a)
            grad = compute_multipeak_grad(a, conv, top_k=30, temp=temp)
            
            # Line search
            a_new, new_score = line_search(a, grad, max_lr=base_lr, n_bisect=8)
            
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = a_new.copy()
                a = a_new
                no_improve = 0
            elif new_score < compute_score(a) + 1e-12:
                a = a_new
                no_improve += 1
            else:
                no_improve += 1
            
            # Anneal
            temp *= 1.003
            base_lr *= 0.9997
            
            if no_improve > 80:
                # Diversify
                noise_scale = 0.005 + 0.01 * min(1, (no_improve - 80) / 100)
                a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
                no_improve = 0
                temp = 0.0005
                base_lr = 0.008
                it = 0
        
        elif phase_frac < 0.8:
            # Phase 2: Local pair search
            conv = fast_convolve(a)
            top_peaks = np.argsort(conv)[-12:]
            current_score = compute_score(a)
            improved = False
            
            for k in top_peaks:
                k = int(k)
                i_lo = max(0, k - n + 1)
                i_hi = min(n - 1, k)
                if i_hi <= i_lo:
                    continue
                
                # Try spreading the top products
                prods = a[i_lo:i_hi+1] * a[k - np.arange(i_lo, i_hi+1)]
                top_pairs = np.argsort(prods)[-5:]
                
                for pi in top_pairs:
                    i = i_lo + int(pi)
                    j = k - i
                    if i == j:
                        continue
                    s_pair = a[i] + a[j]
                    if s_pair < 1e-10:
                        continue

                            a_try[i] = max(0, a[i] - t)
                            a_try[j] = max(0, a[j] + t)
                            ps = a_try[i] + a_try[j]
                            if ps > 1e-15:
                                a_try[i] *= s_pair / ps
                                a_try[j] *= s_pair / ps
                            a_try /= a_try.sum()
                            sc = compute_score(a_try)
                            if sc < best_score:
                                a = a_try
                                best_score = sc
                                best_a = a.copy()
                                improved = True
                            elif sc < current_score:
                                a = a_try
                                current_score = sc
                                improved = True
                                break
                        if improved:
                            break
                if improved:
                    break
            
            if not improved:
                # Random moves
                for _ in range(30):
                    i, j = rng.choice(n, 2, replace=False)

                    t = rng.uniform(-0.4 * s_pair, 0.4 * s_pair)
                    a_try = a.copy()
                    a_try[i] = max(0, a[i] - t)
                    a_try[j] = max(0, a[j] + t)
                    ps = a_try[i] + a_try[j]
                    if ps > 1e-15:
                        a_try[i] *= s_pair / ps
                        a_try[j] *= s_pair / ps
                    a_try /= a_try.sum()

                        improved = True

                        a = a_try
                        improved = True
                    if improved:
                        break
                
                if not improved:

                    a = project_simplex(best_a + rng.standard_normal(n) * 0.01)
        
        else:
            # Phase 3: Fine polish
            conv = fast_convolve(a)
            grad = compute_multipeak_grad(a, conv, top_k=50, temp=0.00005)
            a_new, new_score = line_search(a, grad, max_lr=0.003, n_bisect=10)
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                a = a_new
            elif new_score < compute_score(a):
                a = a_new
    
    return best_a, best_score
