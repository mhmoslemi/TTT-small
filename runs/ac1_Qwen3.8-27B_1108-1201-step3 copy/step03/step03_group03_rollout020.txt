def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 2 ** (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgrad_fft(a, w):
    """
    Compute softmax-weighted subgradient of max_k C_k using FFT.
    sg[i] = 2 * sum_j w[i+j] * a[j] (valid range)
    Uses cross-correlation via FFT for O(n log n) complexity.

    Softmax subgradient with temperature annealing and periodic restarts.
    """
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    if temp_init is None:
        temp_init = max_c_helper(a) * 0.1
    if temp_end is None:
        temp_end = max_c_helper(a) * 0.005
    
    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    temp = temp_init
    lr = lr_init
    no_improve = 0
    step_count = 0
    
    while time.monotonic() - t0 < time_limit:
        a, v, conv, cur_obj = softmax_subgrad_step(a, temp, lr, momentum, v)
        step_count += 1
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adapt temperature: increase when stuck (to explore broader)
        if no_improve > 100:
            temp = min(temp * 1.5, max_c_helper(a) * 0.5)
            lr = max(0.001, lr * 0.8)
            no_improve = 0
        elif no_improve > 50:
            temp = max(temp * 0.8, max_c_helper(a) * 0.001)
        
        # Periodic restart
        if step_count % restart_interval == 0 and no_improve > 50:
            v *= 0.1
            noise = np.random.default_rng(step_count).standard_normal(n) * restart_magnitude
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
        
        # Decay
        temp = max(temp_end, temp * temp_decay)
        lr *= lr_decay
        
        # Early stop if target reached
        if obj_target is not None and best_obj < obj_target:
            break
    
    return best_a, best_obj


def max_c_helper(a):
    """Helper to get max conv value."""
    conv = fast_convolve(a)
    return conv.max()


def spectral_shaping(a0, time_limit, alpha_init=0.7, alpha_end=0.3,
                     n_iters=None):
    """
    Iterative spectral shaping: compress |DFT(a)| to flatten autocorrelation.
    Steps: FFT -> compress magnitude -> IFFT -> project to non-negative simplex.
    """
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, 2.0 * len(a) * fast_convolve(a).max()
    a = a / s
    n = len(a)
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    
    # Use next power of 2 for FFT
    fft_n = 2 ** (n - 1).bit_length()
    if fft_n < n:
        fft_n *= 2
    
    step = 0
    total_expected = max(100, int((time_limit - 0.1) * 200))  # estimate
    alpha = alpha_init
    
    while time.monotonic() - t0 < time_limit:
        step += 1
        # Animate alpha
        progress = min(1.0, step / (total_expected * 0.7))
        alpha = alpha_init + (alpha_end - alpha_init) * progress
        
        # FFT
        A = np.fft.rfft(a, fft_n)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Compress magnitude: mag^alpha (reduces dynamic range)
        # Normalize to preserve total energy approximately
        mag_new = mag ** alpha
        # Scale to preserve DC component (sum)
        if mag_new[0] > 0:
            mag_new = mag_new * (mag[0] / mag_new[0])
        
        # IFFT
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_n)[:n]
        
        # Project to non-negative simplex
        a_new = project_simplex(a_new)
        
        # Check improvement
        conv = fast_convolve(a_new)
        obj = 2.0 * n * conv.max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_obj


def build_lp_matrix(n):
    """Build the sparse constraint structure for LP: (f*g)_k <= M."""
    # For the LP: max sum(g) s.t. (f*g)_k <= M for all k, g >= 0
    # (f*g)_k = sum_j f[j] * g[k-j] for valid j, k-j
    # So row k of A_ub has A_ub[k, j] = f[k-j] if 0 <= k-j < n
    return None  # We'll build A_ub on the fly


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)_k <= M for all k."""
    # Build A_ub: (2n-1) x n, where A_ub[k,j] = f[k-j] if 0 <= k-j < n
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[k, j] = f[k - j]
    
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': 30})
        if result.success:
            return np.maximum(result.x, 0.0)
    except Exception:
        pass
    return None


def optimize_lp_directional(a0, time_limit, blend_init=0.3, blend_min=0.005,
                            blend_halve_every=15):
    """LP-based directional optimization."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    blend = blend_init
    stall = 0
    uniform = np.ones(n) / n
    
    while time.monotonic() - t0 < time_limit:
        M = conv.max()
        if M < 1e-15:
            break
        
        g = solve_lp_direction(a, M, n)
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        # Blend to avoid sparsity
        if g.max() > 0.3:
            g = 0.7 * g + 0.3 * uniform
        
        a_new = (1.0 - blend) * a + blend * g
        a = project_simplex(a_new)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall >= blend_halve_every:
            blend = max(blend_min, blend * 0.5)
            stall = 0
            if blend < blend_min * 1.5:
                break
    
    return best_a, best_obj


def pair_exchange_search(a0, time_limit, n_rounds=None):
    """
    Local search: try moving small amounts of mass between pairs (i,j).
    Focus on pairs that most affect the peak.
    """
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    peak_pos = int(np.argmax(conv))
    
    t0 = time.monotonic()
    round_count = 0
    no_improve_rounds = 0
    
    # Pre-compute which pairs contribute to the peak
    # C_peak = sum_j a[j] * a[peak-j]
    # Moving mass from i to j changes C_peak by:
    # -delta * a[peak-i] + delta * (a[peak-j] - a[peak-i] + a[peak-j] * 0)
    # Actually it's more complex since other C_k also change.
    
    # Strategy: for the peak position, try reducing a[j] where a[j]*a[peak-j] is large
    # and increasing a[j'] where a[j']*a[peak-j'] is small.
    
    while time.monotonic() - t0 < time_limit:
        round_count += 1
        improved = False
        
        # Update peak
        peak_pos = int(np.argmax(conv))
        
        # Find top contributors to peak
        j_start = max(0, peak_pos - n + 1)
        j_end = min(n - 1, peak_pos)
        contrib = np.array([a[j] * a[peak_pos - j] for j in range(j_start, j_end + 1)])
        
        # Try moving mass from high contributors to low contributors
        # Pick top 5 contributors and try moving their mass to random other positions
        n_contrib = len(contrib)
        if n_contrib > 2:
            # Sort by contribution
            order = np.argsort(contrib)[::-1]
            top_indices = [j_start + order[i] for i in range(min(10, n_contrib))]
            
            # Find low-contribution positions (across all peaks)
            # For simplicity, try moving to positions that don't contribute to any peak
            all_peaks = np.argsort(conv)[::-1][:20]
            
            for i in top_indices:
                if a[i] < 1e-8:
                    continue
                # Try moving small fraction to various j
                delta = a[i] * 0.02
                for j in range(n):
                    if j == i or a[j] > 0.5:
                        continue
                    a_test = a.copy()
                    a_test[i] -= delta
                    a_test[j] += delta
                    a_test = np.maximum(a_test, 0.0)
                    st = a_test.sum()
                    if st < 1e-15:
                        continue
                    a_test /= st
                    conv_test = fast_convolve(a_test)
                    obj_test = 2.0 * n * conv_test.max()
                    if obj_test < best_obj - 1e-13:
                        a = a_test
                        best_obj = obj_test
                        conv = conv_test
                        improved = True
                        break
        
        # Also try small uniform redistribution
        if not improved and round_count % 5 == 0:
            # Find the "excess" at peaks and "deficit" at troughs
            avg_c = 1.0 / (2 * n - 1)
            for k in range(2 * n - 1):
                if conv[k] > avg_c * 1.5:
                    # Reduce contributions at this peak
                    j_start_k = max(0, k - n + 1)
                    j_end_k = min(n - 1, k)
                    for j in range(j_start_k, j_end_k + 1):
                        if a[j] < 1e-8:
                            continue
                        delta = a[j] * 0.01
                        # Move to a random position with low contribution
                        a_test = a.copy()
                        a_test[j] -= delta
                        a_test[np.argmin(conv)] += delta
                        a_test = np.maximum(a_test, 0.0)
                        st = a_test.sum()
                        if st < 1e-15:
                            continue
                        a_test /= st
                        conv_test = fast_convolve(a_test)
                        obj_test = 2.0 * n * conv_test.max()
                        if obj_test < best_obj - 1e-14:
                            a = a_test
                            best_obj = obj_test
                            conv = conv_test
                            improved =True
                            break
                    if improved:
                        break
        
        if not improved:
            no_improve_rounds += 1
            if no_improve_rounds > 50:
                break
        else:
            no_improve_rounds = 0
    
    return a


def gen_initializations(n, parent=None, rng=None):
    """Generate diverse initializations for length n."""
    if rng is None:
        rng = np.random.default_rng(123)
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Triangle
    tri = np.minimum(x + 1.0, n - x)
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()
    
    # Edge-concentrated
    edge = np.zeros(n)
    e = max(1, n // 4)
    edge[:e] = 1.0
    edge[-e:] = 1.0
    inits['edge'] = edge / edge.sum()
    
    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits['gauss'] = g / g.sum()
    
    # Bimodal
    g2 = (np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2) +
          np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2))
    inits['bimodal'] = g2 / g2.sum()
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits['ramp'] = ramp / ramp.sum()
    
    # Alternating (approximate)
    alt = np.ones(n)
    alt[::2] = 1.5
    alt[1::2] = 0.5
    inits['alternating'] = alt / alt.sum()
    
    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0)
        ps = p.sum()
        if ps > 1e-10:
            if len(parent) == n:
                inits['parent'] = p / ps
            else:
                p_interp = np.interp(np.linspace(0, 1, n),
                                     np.linspace(0, 1, len(parent)), p)
                inits['parent_interp'] = p_interp / p_interp.sum()
    
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search combining:
    - Softmax subgradient with temperature annealing
    - Spectral shaping
    - LP directional optimization
    - Pair-exchange local search
    Explores multiple lengths and initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_sequence = None
    best_score = float('inf')
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            par_score = float(evaluate_sequence(p.tolist()))
            print(f"Parent: n={len(p)}, score={par_score:.6f}")
            best_sequence = p / p.sum()
            best_score = par_score
    except Exception:
        pass
    
    def evaluate_and_update(a, label=""):
        nonlocal best_sequence, best_score
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print(f"  {label}: score={score:.6f}, n={len(a)}")
        return score
    
    # Phase 1: Broad exploration with subgradient
    print("=" * 60)
    print("Phase 1: Broad exploration (subgradient + spectral)")
    print("=" * 60)
    
    lengths = [480, 512, 544, 576, 640, 768]
    n_results = []
    phase1_budget = min(500.0, (deadline - time.monotonic()) * 0.30)
    phase1_end = time.monotonic() + phase1_budget
    
    for n in lengths:
        if time.monotonic() > phase1_end - 30:
            break
        inits = gen_initializations(n, parent, rng)
        # Priority: parent first, then structured inits
        prio = ['parent', 'parent_interp', 'bimodal', 'edge', 'alternating',
                'random', 'gauss', 'uniform', 'triangle', 'ramp']
        names = [nm for nm in prio if nm in inits][:4]
        
        t_per_init = max(15.0, (phase1_end - time.monotonic() - 20) / (len(names) + 0.1))
        
        for name in names:
            if time.monotonic() > phase1_end - 15:
                break
            t_alloc = min(t_per_init, phase1_end - time.monotonic() - 10)
            if t_alloc < 8:
                break
            
            a0 = inits[name].copy()
            
            # Subgradient with annealing
            t_sub = t_alloc * 0.6
            a_opt, obj_opt = optimize_subgrad_anneal(
                a0, t_sub,
                temp_init=max_c_helper(a0) * 0.15,
                temp_end=max_c_helper(a0) * 0.003,
                lr_init=0.02, momentum=0.9,
                temp_decay=0.9995, lr_decay=0.9999,
                restart_interval=200, restart_magnitude=0.002
            )
            score = evaluate_and_update(a_opt, f"subgrad n={n} {name}")
            
            # Quick spectral shaping
            t_spec = min(t_alloc * 0.3, 20.0)
            if time.monotonic() < phase1_end - 10:
                a_spec, obj_spec = spectral_shaping(a_opt, t_spec,
                                                     alpha_init=0.6, alpha_end=0.35)
                if obj_spec < obj_opt:
                    a_opt = a_spec
                    obj_opt = obj_spec
                    score = evaluate_and_update(a_opt, f"spectral n={n} {name}")
            
            n_results.append((score if np.isfinite(score) else obj_opt * 1, n, name, a_opt.copy()))
    
    n_results.sort(key=lambda x: x[0])
    print(f"Phase 1 best: {best_score:.6f}")
    
    # Phase 2: Deep optimization on top candidates
    print("=" * 60)
    print("Phase 2: Deep optimization")
    print("=" * 60)
    
    top_candidates = []
    seen = set()
    for score, n, name, a_opt in n_results[:10]:
        key = (n, name)
        if key not in seen:
            top_candidates.append((n, a_opt))
            seen.add(key)
        if len(top_candidates) >= 5:
            break
    if best_sequence is not None:
        top_candidates.append((len(best_sequence), best_sequence.copy()))
    
    phase2_budget = min(800.0, (deadline - time.monotonic()) * 0.45)
    phase2_end = time.monotonic() + phase2_budget
    t_per_cand = max(30.0, phase2_budget / max(1, len(top_candidates)))
    
    for n, a0 in top_candidates:
        if time.monotonic() > phase2_end - 50:
            break
        t_alloc = min(t_per_cand, phase2_end - time.monotonic() - 40)
        if t_alloc < 15:
            break
        
        # Extended subgradient with different parameters
        for params in [
            dict(temp_init=max_c_helper(a0) * 0.05, temp_end=max_c_helper(a0) * 0.001,
                 lr_init=0.01, momentum=0.92, restart_magnitude=0.001),
            dict(temp_init=max_c_helper(a0) * 0.2, temp_end=max_c_helper(a0) * 0.01,
                 lr_init=0.03, momentum=0.85, restart_magnitude=0.005),
        ]:
            if time.monotonic() > phase2_end - 30:
                break
            t_s = t_alloc / 2.0
            if t_s < 10:
                break
            a_opt, obj_opt = optimize_subgrad_anneal(a0, t_s, **params)
            score = evaluate_and_update(a_opt, f"deep_sub n={n}")
            if score < best_score:
                a0 = a_opt
        
        # Spectral shaping on best so far
        if time.monotonic() < phase2_end - 25:
            t_spec = min(30.0, phase2_end - time.monotonic() - 20)
            if t_spec > 5:
                a_spec, obj_spec = spectral_shaping(a0, t_spec,
                                                     alpha_init=0.5, alpha_end=0.25)
                score = evaluate_and_update(a_spec, f"deep_spec n={n}")
                if score < best_score:
                    a0 = a_spec
    
    # Phase 3: LP refinement
    print("=" * 60)
    print("Phase 3: LP refinement")
    print("=" * 60)
    
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 150.0
        # Only do LP if n is manageable (LP is O(n^2) in constraints)
        n_lp = len(best_sequence)
        if n_lp <= 600:
            for bi in [0.2, 0.1, 0.05]:
                if time.monotonic() > deadline - 130:
                    break
                t_lp = min(60.0, remaining / 3.0)
                if t_lp < 10:
                    break
                a_opt, obj_opt = optimize_lp_directional(
                    best_sequence, t_lp,
                    blend_init=bi, blend_min=0.003, blend_halve_every=20
                )
                score = evaluate_and_update(a_opt, f"LP bi={bi}")
                if score < best_score:
                    best_sequence = a_opt
    
    # Phase 4: Local search polish
    print("=" * 60)
    print("Phase 4: Local search polish")
    print("=" * 60)
    
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 50.0
        
        # Pair exchange
        t_pe = min(80.0, remaining * 0.4)
        if t_pe > 5:
            a_pe = pair_exchange_search(best_sequence, t_pe)
            score = evaluate_and_update(a_pe, "pair_exchange")
            if score < best_score:
                best_sequence = a_pe
        
        # Final subgradient with very small steps
        remaining = deadline - time.monotonic() - 30.0
        if remaining > 10:
            a_final, obj_final = optimize_subgrad_anneal(
                best_sequence, min(remaining, 60.0),
                temp_init=max_c_helper(best_sequence) * 0.02,
                temp_end=max_c_helper(best_sequence) * 0.001,
                lr_init=0.005, momentum=0.93,
                temp_decay=0.9999, lr_decay=0.99995,
                restart_interval=500, restart_magnitude=0.001
            )
            score = evaluate_and_update(a_final, "final_subgrad")
            if score < best_score:
                best_sequence = a_final
        
        # One more LP if time
        if time.monotonic() < deadline - 40 and len(best_sequence) <= 550:
            t_final_lp = min(30.0, deadline - time.monotonic() - 35)
            if t_final_lp > 8:
                a_lp, obj_lp = optimize_lp_directional(
                    best_sequence, t_final_lp,
                    blend_init=0.05, blend_min=0.002, blend_halve_every=25
                )
                score = evaluate_and_update(a_lp, "final_LP")
                if score < best_score:
                    best_sequence = a_lp
    
    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
    
    # Final validation
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if not np.isfinite(final_score):
        best_sequence = np.ones(256) / 256
        final_score = float(evaluate_sequence(best_sequence.tolist()))
    
    print(f"\n{'=' * 60}")
    print(f"FINAL: score={final_score:.6f}, n={len(best_sequence)}")
    print(f"{'=' * 60}")
    
    return [float(x) for x in best_sequence]
