import numpy as np
import time
from scipy.optimize import minimize

def fast_convolve(a):
    n = len(a)
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_smoothed_grad(a, beta):
    """Gradient of logsumexp(beta*conv) - 2*log(sum(a))"""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) * 2.0
    conv = fast_convolve(a)
    scaled = beta * conv
    scaled -= scaled.max()
    p = np.exp(scaled)
    p /= p.sum()
    
    # g[i] = 2*beta*sum_m a[m]*p[m+i] - 2/s
    N = 1 << (3 * n).bit_length()
    A = np.fft.fft(a, N)
    P = np.fft.fft(p, N)
    cross = np.fft.ifft(np.conj(A) * P).real[:n]
    
    return 2.0 * beta * cross - 2.0 / s

def smooth_obj_grad(a, beta):
    """Objective and gradient for L-BFGS-B."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.ones(n)
    conv = fast_convolve(a)
    scaled = beta * conv
    mj = scaled.max()
    ls = mj + np.log(np.exp(scaled - mj).sum())
    obj_val = ls - 2.0 * np.log(s) + 1e-8 * np.dot(a, a)
    
    # Gradient
    p = np.exp(scaled - mj)
    p /= p.sum()
    N = 1 << (3 * n).bit_length()
    A = np.fft.fft(a, N)
    P = np.fft.fft(p, N)
    cross = np.fft.ifft(np.conj(A) * P).real[:n]
    grad = 2.0 * beta * cross - 2.0 / s + 2e-8 * a
    
    return obj_val, grad

def lbfgs_optimize(a0, beta, maxiter=500, time_limit=60):
    """L-BFGS-B optimization on smoothed objective."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s  # normalize for numerical stability
    
    t0 = time.monotonic()
    
    def fun_grad(x):
        if time.monotonic() - t0 > time_limit:
            raise RuntimeError("time limit")
        x = np.maximum(x, 0.0)  # enforce non-negativity
        val, g = smooth_obj_grad(x, beta)
        return val, g
    
    bounds = [(0, 10.0)] * len(a)
    
    try:
        result = minimize(fun_grad, a, jac=True, method='L-BFGS-B',
                         bounds=bounds, options={'maxiter': maxiter,
                                                 'maxfun': 2000,
                                                 'maxls': 20})
        a_new = np.maximum(result.x, 0.0)
    except Exception as e:
        a_new = a
    
    s_new = a_new.sum()
    if s_new < 1e-10:
        a_new = a
    else:
        a_new = a_new / s_new
    
    return a_new

def subgradient_polish(a0, time_limit, lr_init=0.01, decay=0.999):
    """Subgradient descent on the true max objective (unsmoothed)."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    lr = lr_init
    gap = None
    
    t0 = time.monotonic()
    iters = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        max_val = conv.max()
        
        # Active set: indices where conv is close to max
        active = np.where(conv >= max_val - 1e-8 * max_val)[0]
        
        # Subgradient: average gradient of active constraints
        # d(conv_k)/d(a_i) = 2*a[k-i] (if valid and k≠2i) or 2*a[i] (if k=2i)
        # Using FFT: grad[i] = (2/|active|) * sum_{k in active} a[k-i] (with appropriate handling)
        
        # For efficiency, compute using the same FFT trick
        p = np.zeros(2 * n - 1)
        p[active] = 1.0
        p /= p.sum()
        
        N = 1 << (3 * n).bit_length()
        A = np.fft.fft(a, N)
        P = np.fft.fft(p, N)
        cross = np.fft.ifft(np.conj(A) * P).real[:n]
        grad = 2.0 * cross - 2.0 / a.sum()
        
        # Polyak step size if gap known, else fixed
        if gap is not None and gap > 1e-12:
            lr = (max_val - best_obj) ** 2 / (np.dot(grad, grad) + 1e-15)
            lr = min(lr, 0.1)
        else:
            lr = lr_init * (decay ** iters)
        
        a = a - lr * grad
        a = np.maximum(a, 0.0)
        s_new = a.sum()
        if s_new < 1e-10:
            a = best_a.copy()
            break
        a = a / s_new
        
        iters += 1
        
        conv_new = fast_convolve(a)
        obj_new = conv_new.max()
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a.copy()
            # Update gap estimate
            gap = (obj_new - 1.0 / n)  # crude lower bound
    
    return best_a, best_obj
