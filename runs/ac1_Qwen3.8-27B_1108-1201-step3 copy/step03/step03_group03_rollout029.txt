"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multiplicative relaxation

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def flat_top_window(n):
    """Flat-top window with very flat spectral response."""
    x = np.arange(n, dtype=np.float64)
    a0, a1, a2, a3, a4 = 0.21557895, 0.41663158, 0.277263158, 0.083578947, 0.006947368
    w = a0 - a1 * np.cos(2 * np.pi * x / (n - 1)) + a2 * np.cos(4 * np.pi * x / (n - 1)) \
        - a3 * np.cos(6 * np.pi * x / (n - 1)) + a4 * np.cos(8 * np.pi * x / (n - 1))
    w = np.maximum(w, 1e-10)
    return w / w.sum()

def compute_subgrad(a, conv, top_k=20, temperature=0.01):
    """Compute weighted subgradient from top-k peak positions."""
    n = len(a)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
        weights = np.ones(top_k)
    else:
        # Use softmax weighting over top peaks for smoothness
        idx = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[idx]
        # Soft weights
        scaled = (peak_vals - peak_vals.min()) / (peak_vals.max() - peak_vals.min() + 1e-15)
        weights = np.exp(scaled / max(temperature, 1e-6))
        weights /= weights.sum()
    
    sg = np.zeros(n)
    for i in range(len(idx)):
        if weights[i] < 1e-10:
            continue
        k = int(idx[i])
        w = 2.0 * weights[i]
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            sg[j] += w * a[k - j]
    return sg


def multiplicative_relax(a0, time_limit, lr_init=0.1, lr_decay=0.999,
                         top_k=30, temperature=0.05, min_lr=0.001):
    """Multiplicative relaxation: a *= exp(-lr * sg), renormalize."""
    a = np.maximum(a0.astype(np.float64), 1e-15)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    
    while time.monotonic() - t0 < time_limit:
        sg = compute_subgrad(a, conv, top_k=top_k, temperature=temperature)
        # Multiplicative update
        log_a = np.log(a) - lr * sg
        # Clip to prevent underflow
        log_a = np.clip(log_a, -30, 30)
        a = np.exp(log_a)
        a = a / a.sum()
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
        
        lr *= lr_decay
        if lr < min_lr:
            break
    
    return best_a, best_obj


def solve_lp_restricted(f, M, n, active_set=None):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)_k <= M for k in active_set (or all k)."""
    if active_set is None:
        active_set = np.arange(2 * n - 1)
    
    m = len(active_set)
    if m == 0:
        return None
    
    A_ub = np.zeros((m, n))
    for i, k in enumerate(active_set):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A_ub[i, j] = f[k - j]
    
    b_ub = np.full(m, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
        if result.success:
            return np.maximum(result.x, 0.0)
    except Exception:
        pass
    return None


def lp_direction_step(a, time_limit, top_k_constraints=50, blend_init=0.3,
                      blend_min=0.005, halve_every=10):
    """LP-based directional optimization with restricted constraints."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    blend = blend_init
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        M = conv.max()
        if M < 1e-15:
            break
        
        # Only use top-K constraints to speed up LP
        if top_k_constraints < 2 * n - 1:
            active = np.argpartition(conv, -top_k_constraints)[-top_k_constraints:]
        else:
            active = np.arange(2 * n - 1)
        
        g = solve_lp_restricted(a, M, n, active)
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        a_new = (1.0 - blend) * a + blend * g
        a = project_simplex(a_new)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall >= halve_every:
            blend = max(blend_min, blend * 0.5)
            stall = 0
            if blend < blend_min:
                break
    
    return best_a, best_obj

def coordinate_polish(a_in, max_rounds=30, step_frac=0.03):
    """Pair-swap coordinate descent for final polish."""
    a = np.maximum(a_in.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    
    for rnd in range(max_rounds):
        improved = False
        # Sort indices by value to get good swap candidates
        idx = np.argsort(a)
        # Try swapping mass from large to small (or vice versa)
        num_tries = min(n * 2, 500)
        for t in range(num_tries):
            i = int(idx[t % n])
            j = int(idx[(n - 1 - t) % n])
            if i == j:
                continue
            if a[i] < 1e-8:
                continue
            delta = a[i] * step_frac
            if delta < 1e-9:
                continue
            a_test = a.copy()
            a_test[i] -= delta
            a_test[j] += delta
            if a_test[i] < 0:
                continue
            st = a_test.sum()
            if st < 1e-15:
                continue
            a_test /= st
            conv_test = fast_convolve(a_test)
            obj_test = 2.0 * n * conv_test.max()
            if obj_test < best_obj - 1e-13:
                a = a_test
                best_obj = obj_test
                conv = conv_test
                improved = True
        if not improved:
            break
    
    return a

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-strategy approach:
1. Spectral-flattening initializations (flat-top, Blackman-Harris windows)
2. Multiplicative relaxation (mirror descent) with adaptive annealing
3. Periodic LP-based directional correction with restricted constraints
4. Simulated annealing for escaping local minima
5. Pair-swap coordinate descent for final polish
Explores multiple sequence lengths (256-2048) to find the global optimum.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1
    while fft_size < size:
        fft_size <<= 1
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def flat_top_window(n):

    x = np.arange(n, dtype=np.float64)
    a0, a1, a2, a3, a4 = 0.21557895, 0.41663158, 0.277263158, 0.083578947, 0.006947368
    w = (a0 - a1 * np.cos(2 * np.pi * x / (n - 1))
         + a2 * np.cos(4 * np.pi * x / (n - 1))
         - a3 * np.cos(6 * np.pi * x / (n - 1))
         + a4 * np.cos(8 * np.pi * x / (n - 1)))
    w = np.maximum(w, 1e-10)
    return w / w.sum()


def blackman_harris(n):
    """Blackman-Harris 4-term window."""
    x = np.arange(n, dtype=np.float64)
    a0, a1, a2, a3 = 0.35875, 0.48829, 0.14128, 0.01168
    w = (a0 - a1 * np.cos(2 * np.pi * x / (n - 1))
         + a2 * np.cos(4 * np.pi * x / (n - 1))
         - a3 * np.cos(6 * np.pi * x / (n - 1)))
    w = np.maximum(w, 1e-10)
    return w / w.sum()


def nuttall_window(n):
    """Nuttall window."""
    x = np.arange(n, dtype=np.float64)
    a0, a1, a2, a3 = 0.3635819, 0.4891775, 0.1365995, 0.0106411
    w = (a0 - a1 * np.cos(2 * np.pi * x / (n - 1))
         + a2 * np.cos(4 * np.pi * x / (n - 1))
         - a3 * np.cos(6 * np.pi * x / (n - 1)))
    w = np.maximum(w, 1e-10)
    return w / w.sum()


def cosine_u(n):
    """U-shaped window (large at edges)."""
    x = np.arange(n, dtype=np.float64)
    t = x / max(n - 1, 1)
    w = 0.3 + 0.7 * np.cos(np.pi * (2 * t - 1))
    w = np.maximum(w, 0.01)
    return w / w.sum()


def compute_subgrad_fast(a, conv, top_k, temperature):
    """Compute weighted subgradient from top-k peak positions (vectorized)."""
    n = len(a)
    conv_len = len(conv)
    if top_k >= conv_len:
        idx = np.arange(conv_len)
        weights = np.ones(top_k) / top_k
    else:
        idx = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[idx]
        vmin = peak_vals.min()
        vmax = peak_vals.max()
        if vmax - vmin < 1e-15:
            weights = np.ones(top_k) / top_k
        else:
            scaled = (peak_vals - vmin) / (vmax - vmin)
            weights = np.exp(scaled / max(temperature, 1e-8))
            weights /= weights.sum()

    sg = np.zeros(n)
    for i in range(len(idx)):
        if weights[i] < 1e-12:
            continue
        k = int(idx[i])
        w = 2.0 * weights[i]
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1)
        sg[js] += w * a[k - js]
    return sg


def multiplicative_relax(a0, time_limit, lr_init=0.12, lr_decay=0.9985,
                         top_k_init=30, temp_init=0.06, min_lr=0.002):
    """Pure multiplicative relaxation (mirror descent on simplex)."""
    a = np.maximum(a0.astype(np.float64), 1e-15)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    top_k = top_k_init
    temp = temp_init

    while time.monotonic() - t0 < time_limit:
        sg = compute_subgrad_fast(a, conv, top_k, temp)
        log_a = np.log(np.maximum(a, 1e-15)) - lr * sg
        log_a = np.clip(log_a, -28, 28)
        a = np.exp(log_a)
        a = a / a.sum()

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()

        lr *= lr_decay
        temp *= 0.9997
        top_k = min(120, int(top_k_init + (lr_init - lr) * 200))

        if lr < min_lr:
            break

    return best_a, best_obj


def solve_lp_restricted(f, M, n, active_indices):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)_k <= M for k in active_indices."""
    m = len(active_indices)
    if m == 0:
        return None

    A_ub = np.zeros((m, n))
    for i, k in enumerate(active_indices):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1)
        A_ub[i, js] = f[k - js]

    b_ub = np.full(m, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n

    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
        if result.success:
            return np.maximum(result.x, 0.0)
    except Exception:
        pass
    return None


def lp_direction_step(a_in, time_limit, top_k_constraints=40,
                      blend_init=0.25, blend_min=0.005, halve_every=10):
    """LP-based directional optimization with restricted constraints."""
    a = np.maximum(a_in.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    blend = blend_init
    stall = 0

    while time.monotonic() - t0 < time_limit:
        M = conv.max()
        if M < 1e-15:
            break

        k_use = min(top_k_constraints, 2 * n - 1)
        if k_use < 2 * n - 1:
            active = np.argpartition(conv, -k_use)[-k_use:]
        else:
            active = np.arange(2 * n - 1)

        g = solve_lp_restricted(a, M, n, active)
        if g is None:
            break

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        a_new = (1.0 - blend) * a + blend * g
        a = project_simplex(a_new)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall >= halve_every:
            blend = max(blend_min, blend * 0.5)
            stall = 0
            if blend < blend_min:
                break

    return best_a, best_obj


def combined_optimize(a0, time_limit, use_lp=True, lp_interval=40):
    """Combined: multiplicative relaxation + periodic LP correction."""
    a = np.maximum(a0.astype(np.float64), 1e-15)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = 0.12
    top_k = 35
    temp = 0.05
    count = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        sg = compute_subgrad_fast(a, conv, top_k, temp)
        log_a = np.log(np.maximum(a, 1e-15)) - lr * sg
        log_a = np.clip(log_a, -25, 25)
        a = np.exp(log_a)
        a = a / a.sum()

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        count += 1

        # Periodic LP
        if use_lp and count % lp_interval == 0 and n <= 2048:
            t_rem = time_limit - (time.monotonic() - t0)
            if t_rem > 1.5:
                k_use = min(40, 2 * n - 1)
                active = np.argpartition(conv, -k_use)[-k_use:]
                g = solve_lp_restricted(a, conv.max(), n, active)
                if g is not None and g.sum() > 1e-15:
                    g = g / g.sum()
                    blend = min(0.15, 3.0 / count)
                    a = project_simplex((1 - blend) * a + blend * g)
                    conv = fast_convolve(a)
                    cur_obj = 2.0 * n * conv.max()
                    if cur_obj < best_obj:
                        best_obj = cur_obj
                        best_a = a.copy()

        # Anneal
        lr *= 0.998
        temp *= 0.9996
        top_k = min(100, int(25 + count * 0.05))

        if lr < 0.003:
            break

        # Restart if stuck
        if stall > 150:
            rng_local = np.random.default_rng(count * 7919)
            a = project_simplex(a + rng_local.standard_normal(n) * 0.008)
            conv = fast_convolve(a)
            lr = max(lr, 0.04)
            stall = 0

    return best_a, best_obj


def simulated_annealing(a_in, time_limit, T_init=0.005, T_min=0.00005,
                        cool=0.9995, swap_frac_range=(0.01, 0.06)):
    """Simulated annealing with mass-transfer moves."""
    a = np.maximum(a_in.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    T = T_init
    t0 = time.monotonic()
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit and T > T_min:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        if a[i] < 1e-10:
            continue

        frac = rng.uniform(*swap_frac_range)
        delta = a[i] * frac

        a_test = a.copy()
        a_test[i] -= delta
        a_test[j] += delta
        st = a_test.sum()
        if st < 1e-15:
            continue
        a_test /= st

        conv_test = fast_convolve(a_test)
        obj_test = 2.0 * n * conv_test.max()
        cur_obj = 2.0 * n * conv.max()
        delta_obj = obj_test - cur_obj

        if delta_obj < 0 or rng.random() < np.exp(-delta_obj / max(T, 1e-10)):
            a = a_test
            conv = conv_test
            if obj_test < best_obj:
                best_obj = obj_test
                best_a = a.copy()

        T *= cool

    return best_a


def coordinate_polish(a_in, max_rounds=25, step_frac=0.025):
    """Pair-swap coordinate descent for final polish."""
    a = np.maximum(a_in.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    idx_sorted = np.argsort(a)

    for rnd in range(max_rounds):
        improved = False
        num_tries = min(n, 400)
        for t in range(num_tries):
            i = int(idx_sorted[t % n])
            j = int(idx_sorted[(n - 1 - (t % n)) % n])
            if i == j or a[i] < 1e-9:
                continue
            delta = a[i] * step_frac
            if delta < 1e-10:
                continue
            a_test = a.copy()
            a_test[i] -= delta
            a_test[j] += delta
            st = a_test.sum()
            if st < 1e-15:
                continue
            a_test /= st
            conv_test = fast_convolve(a_test)
            obj_test = 2.0 * n * conv_test.max()
            if obj_test < best_obj - 1e-13:
                a = a_test
                best_obj = obj_test
                conv = conv_test
                improved = True
                idx_sorted = np.argsort(a)
        if not improved:
            break

    return a


def gen_initializations(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(42)

    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['flat_top'] = flat_top_window(n)
    inits['bh'] = blackman_harris(n)
    inits['nuttall'] = nuttall_window(n)
    inits['cos_u'] = cosine_u(n)
    inits['uniform'] = np.ones(n) / n

    tri = np.minimum(x + 1.0, n - x)
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()

    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.22 * n)) ** 2)
    inits['gauss'] = g / g.sum()

    inits['random'] = rng.dirichlet(np.ones(n) * 0.4)

    g2 = (np.exp(-0.5 * ((x - 0.2 * n) / (0.1 * n)) ** 2)
          + np.exp(-0.5 * ((x - 0.8 * n) / (0.1 * n)) ** 2))
    inits['bimodal'] = g2 / g2.sum()

    # W-shaped (large at edges and middle)
    g3 = (np.exp(-0.5 * ((x - 0) / (0.08 * n)) ** 2)
          + 0.5 * np.exp(-0.5 * ((x - 0.5 * n) / (0.15 * n)) ** 2)
          + np.exp(-0.5 * ((x - n) / (0.08 * n)) ** 2))
    inits['w_shape'] = g3 / g3.sum()

    if parent is not None and len(parent) > 0:
        if len(parent) == n:
            p = np.maximum(parent, 0)
            ps = p.sum()
            if ps > 1e-10:
                inits['parent'] = p / ps
        else:
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)),
                          np.maximum(parent, 0))
            ps = p.sum()
            if ps > 1e-10:
                inits['parent_interp'] = p / ps

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search: broad exploration with spectral window initializations,
    deep optimization via combined MR+LP, simulated annealing, and coordinate
    descent polish. Explores sequence lengths 256-2048.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 35.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            par_score = float(evaluate_sequence(p.tolist()))
            print(f"Parent: n={len(p)}, score={par_score:.6f}")
            best_sequence = p.copy()
            best_score = par_score
    except Exception:
        pass

    # Phase 1: Broad exploration
    print("=== Phase 1: Broad Exploration ===")
    lengths = [256, 384, 512, 640, 768, 1024, 1536]

    p1_budget = min(700.0, (deadline - time.monotonic()) * 0.40)
    p1_end = time.monotonic() + p1_budget

    results = []
    prio_order = ['parent', 'parent_interp', 'flat_top', 'bh', 'nuttall',
                  'cos_u', 'w_shape', 'triangle', 'bimodal', 'gauss',
                  'random', 'uniform']

    for n in lengths:
        if time.monotonic() > p1_end - 40:
            break

        inits = gen_initializations(n, parent, rng)
        names = [nm for nm in prio_order if nm in inits][:4]

        t_per = max(10.0, p1_budget / (len(lengths) * 4))

        for name in names:
            if time.monotonic() > p1_end - 15:
                break
            t_alloc = min(t_per, p1_end - time.monotonic() - 8)
            if t_alloc < 6:
                break

            a0 = inits[name]
            a_opt, obj_opt = combined_optimize(a0, t_alloc, use_lp=(n <= 1024),
                                               lp_interval=35)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Phase1 n={n} {name}: {score:.6f}")

    results.sort(key=lambda x: x[0])
    print(f"Phase 1 best: {best_score:.6f}")

    # Phase 2: Deep optimization on top candidates
    print("=== Phase 2: Deep Optimization ===")
    top_list = []
    seen_n = set()
    for score, n, name, a_opt in results[:12]:
        if n not in seen_n:
            top_list.append((n, a_opt))
            seen_n.add(n)
        if len(top_list) >= 4:
            break
    if best_sequence is not None and len(best_sequence) not in seen_n:
        top_list.append((len(best_sequence), best_sequence.copy()))

    p2_budget = (deadline - time.monotonic()) * 0.35
    t_per = max(25.0, p2_budget / max(1, len(top_list) * 2))

    for n, a0 in top_list:
        if time.monotonic() > deadline - 200:
            break
        t_alloc = min(t_per, deadline - time.monotonic() - 180)
        if t_alloc < 12:
            break

        # Deep combined
        a_opt, obj_opt = combined_optimize(a0, t_alloc * 0.5,
                                           use_lp=(n <= 2048), lp_interval=30)

        # LP refinement
        if n <= 2048 and time.monotonic() < deadline - 160:
            t_lp = min(15.0, (deadline - time.monotonic() - 140) * 0.4)
            if t_lp > 5:
                a_lp, obj_lp = lp_direction_step(a_opt, t_lp,
                                                  top_k_constraints=min(45, 2 * n - 1),
                                                  blend_init=0.2, blend_min=0.008,
                                                  halve_every=8)
                if obj_lp < obj_opt:
                    a_opt, obj_opt = a_lp, obj_lp

        # Simulated annealing
        if time.monotonic() < deadline - 140:
            t_sa = min(20.0, (deadline - time.monotonic() - 120) * 0.3)
            if t_sa > 5:
                a_sa = simulated_annealing(a_opt, t_sa, T_init=0.003,
                                           T_min=0.00003, cool=0.9997)
                score_sa = 2.0 * n * fast_convolve(a_sa).max()
                if score_sa < obj_opt:
                    a_opt = a_sa
                    obj_opt = score_sa

        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Phase2 n={n}: {score:.6f}")

    # Phase 3: Polish
    print("=== Phase 3: Polish ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 35.0

        # Coordinate descent
        t_cd = min(100.0, remaining * 0.25)
        if t_cd > 5:
            a_cd = coordinate_polish(best_sequence, max_rounds=35, step_frac=0.02)
            score_cd = float(evaluate_sequence(a_cd.tolist()))
            if np.isfinite(score_cd) and score_cd < best_score:
                best_score = score_cd
                best_sequence = a_cd.copy()
                print(f"  CD: {score_cd:.6f}")

        # Final optimization cycles
        remaining = deadline - time.monotonic() - 25.0
        for cycle in range(5):
            if time.monotonic() > deadline - 20:
                break
            t_cyc = min(remaining / (5 - cycle), 50.0)
            if t_cyc < 5:
                break

            a_opt, obj_opt = combined_optimize(best_sequence, t_cyc * 0.5,
                                               use_lp=(len(best_sequence) <= 2048),
                                               lp_interval=25)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Cycle {cycle}: {score:.6f}")

            # SA between cycles
            if time.monotonic() < deadline - 15:
                a_sa = simulated_annealing(best_sequence, min(10.0, t_cyc * 0.3),
                                           T_init=0.002, T_min=0.00002,
                                           cool=0.9998)
                score_sa = float(evaluate_sequence(a_sa.tolist()))
                if np.isfinite(score_sa) and score_sa < best_score:
                    best_score = score_sa
                    best_sequence = a_sa.copy()
                    print(f"  SA cycle {cycle}: {score_sa:.6f}")

            # Quick CD
            a_cd2 = coordinate_polish(best_sequence, max_rounds=15, step_frac=0.015)
            score_cd2 = float(evaluate_sequence(a_cd2.tolist()))
            if np.isfinite(score_cd2) and score_cd2 < best_score:
                best_score = score_cd2
                best_sequence = a_cd2.copy()
                print(f"  CD cycle {cycle}: {score_cd2:.6f}")

        # Final random perturbations
        if time.monotonic() < deadline - 12:
            n_b = len(best_sequence)
            for ps in [0.002, 0.004, 0.008]:
                if time.monotonic() > deadline - 10:
                    break
                for _ in range(3):
                    if time.monotonic() > deadline - 8:
                        break
                    noise = rng.standard_normal(n_b) * ps
                    a_p = project_simplex(best_sequence + noise)
                    a_opt, _ = combined_optimize(a_p, min(8.0, deadline - time.monotonic() - 6),
                                                 use_lp=(n_b <= 1024), lp_interval=20)
                    score = float(evaluate_sequence(a_opt.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        print(f"  Perturb({ps}): {score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if not np.isfinite(final_score):
        best_sequence = np.ones(256) / 256
        final_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={final_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
