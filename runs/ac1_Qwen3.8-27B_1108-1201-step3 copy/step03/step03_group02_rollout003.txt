"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase strategy:
1. LP-based warm start from parent sequence
2. Smooth-max projected gradient with adaptive temperature scheduling
3. Peak-targeted subgradient with momentum for local minimum escape
4. LP-based fine-tuning refinement
5. Multi-restart perturbation and cross-phase re-optimization
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve_auto(a):
    """Compute full auto-convolution of a via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A where (A @ b)[k] = conv(a, b)[k] for k=0..2n-2."""
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A


def lp_maximize_sum(a, M):
    """Solve: max sum(b) s.t. b >= 0, conv(a,b)[k] <= M for all k."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x, 0)
    return None


def smooth_max_obj(a, t):
    """Compute 2*n*logsumexp(t*conv)/t assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve_auto(a)
    scaled = t * conv
    scaled -= scaled.max()
    exp_vals = np.exp(scaled)
    sum_exp = exp_vals.sum()
    obj = 2.0 * n * np.log(sum_exp) / t
    return obj, conv, exp_vals / sum_exp


def smooth_max_grad(a, p, t):
    """Compute gradient of 2*n*logsumexp(t*conv)/t w.r.t. a, given softmax weights p."""
    n = len(a)
    # grad[i] = (2n/t) * [sum_j a[j]*p[i+j] + p[2i]*a[i] (if 2i < 2n-1)]
    # First term via convolution with reversed a
    a_rev = a[::-1].copy()
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    F_a = np.fft.rfft(a_rev, fft_size)
    F_p = np.fft.rfft(p, fft_size)
    conv_ap = np.fft.irfft(F_a * F_p, fft_size)
    # grad_cross[i] = conv_ap[n-1+i] for i=0..n-1
    grad_cross = conv_ap[n - 1: n - 1 + n]

    # Diagonal correction: p[2i]*a[i] for valid 2i
    diag = np.zeros(n)
    two_i = 2 * np.arange(n)
    valid = two_i < (2 * n - 1)
    diag[valid] = p[two_i[valid]] * a[valid]

    grad = (2.0 * n / t) * (grad_cross + diag)
    return grad


def objective_value(a):
    """Compute the actual objective 2*n*max(conv)/(sum)^2."""
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    n = len(a)
    conv = fast_convolve_auto(a)
    return float(2.0 * n * conv.max() / (s * s))


def optimize_smooth_max(a0, time_limit, t_start=5.0, t_end=5000.0,
                        t_growth=1.15, lr_start=0.01, mom=0.85):
    """Projected gradient on smooth-max with adaptive temperature and step size."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve_auto(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    t = t_start
    lr = lr_start
    no_improve = 0
    patience = 200

    while time.monotonic() - t0 < time_limit:
        obj, conv, p = smooth_max_obj(a, t)
        grad = smooth_max_grad(a, p, t)

        v = mom * v - lr * grad
        a_new = project_simplex(a + v)

        conv_new = fast_convolve_auto(a_new)
        cur_obj = 2.0 * n * conv_new.max()

        if cur_obj < best_obj - 1e-9:
            best_obj = cur_obj
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
            conv = conv_new
        else:
            no_improve += 1
            a = a_new
            conv = conv_new

        if no_improve >= patience:
            t *= t_growth
            no_improve = 0
            patience = min(patience + 50, 1000)
            lr = max(0.0001, lr * 0.7)
            v *= 0.5

            if t > t_end:
                break

    return best_a, best_obj


def optimize_smooth_max_lbfgs(a0, time_limit, t_start=10.0, t_steps=20):
    """Use L-BFGS-B on smooth-max with temperature annealing."""
    from scipy.optimize import minimize as sc_minimize

    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve_auto(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    t = t_start

    for step in range(t_steps):
        if time.monotonic() - t0 > time_limit - 10:
            break

        def fun_and_grad(x):
            x = np.maximum(x, 0)
            sx = x.sum()
            if sx < 1e-15:
                return 1e10, np.zeros(len(x))
            x = x / sx
            obj, conv_v, p = smooth_max_obj(x, t)
            grad = smooth_max_grad(x, p, t)
            return obj, grad

        # Use bounds [0, 1] and let the solver handle it
        bounds = [(0.0, 1.0)] * n
        res = sc_minimize(fun_and_grad, a, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': 50,
                                                  'ftol': 1e-10,
                                                  'gtol': 1e-6})
        if res.x is not None:
            a_new = np.maximum(res.x, 0)
            sn = a_new.sum()
            if sn > 1e-15:
                a_new = a_new / sn
                conv_new = fast_convolve_auto(a_new)
                cur_obj = 2.0 * n * conv_new.max()
                if cur_obj < best_obj:
                    best_obj = cur_obj
                    best_a = a_new.copy()
                    a = a_new

        t *= 2.0

    return best_a, best_obj


def optimize_subgrad_peak(a0, time_limit, top_k=50, lr_start=0.02,
                          mom=0.9, temp=0.001):
    """Subgradient descent targeting top peaks of convolution."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve_auto(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    lr = lr_start
    no_improve = 0
    rng = np.random.default_rng(99)

    while time.monotonic() - t0 < time_limit:
        # Find top peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            peak_vals = conv.copy()
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            peak_vals = conv[peaks]

        # Weighted subgradient
        scaled = peak_vals / max(temp, 1e-10)
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            wi = w[i] * 2.0
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += wi * a[k - idx]

        v = mom * v - lr * sg
        a = project_simplex(a + v)
        conv = fast_convolve_auto(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-9:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9995
        temp *= 1.002

        if no_improve > 3000:
            noise = rng.standard_normal(n) * 0.003
            a = project_simplex(a + noise)
            conv = fast_convolve_auto(a)
            v *= 0.3
            no_improve = 0
            lr = lr_start * 0.5

    return best_a, best_obj


def optimize_lp_refine(a0, time_limit, method='max_sum', step_init=0.3,
                       max_iters=300, step_min=0.02):
    """LP-based iterative refinement."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve_auto(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    step = step_init
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        M = conv.max()

        if method == 'max_sum':
            b = lp_maximize_sum(a, M)
            if b is None:
                break
            b_sum = b.sum()
            if b_sum < 1e-15:
                break
            b = b / b_sum
        else:
            # min_peak variant: solve min M' s.t. sum(b)=1, conv(a,b)<=M', b>=0
            A = build_conv_matrix(a)
            num_rows = 2 * n - 1
            c = np.zeros(n + 1)
            c[-1] = 1.0
            A_eq = np.zeros((1, n + 1))
            A_eq[0, :n] = 1.0
            b_eq = np.array([1.0])
            A_ub = np.zeros((num_rows, n + 1))
            A_ub[:, :n] = A
            A_ub[:, n] = -1.0
            b_ub = np.zeros(num_rows)
            bounds = [(0, None)] * n + [(None, None)]
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.status != 0:
                break
            b = np.maximum(res.x[:n], 0)
            b_sum = b.sum()
            if b_sum < 1e-15:
                break
            b = b / b_sum

        if np.linalg.norm(b - a) < 1e-12:
            break

        a_new = (1.0 - step) * a + step * b
        a_new = np.maximum(a_new, 0.0)
        a_sum = a_new.sum()
        if a_sum < 1e-15:
            break
        a_new = a_new / a_sum

        conv_new = fast_convolve_auto(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            step = min(0.9, step * 1.03)
        else:
            no_improve += 1
            step = max(step_min, step * 0.95)

        a = a_new
        conv = conv_new

        if no_improve > 100:
            break

    return best_a, best_obj


def coordinate_descent(a0, time_limit, sweeps=50):
    """Coordinate-wise optimization: for each i, find best a[i] given rest."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve_auto(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    sweep = 0

    while time.monotonic() - t0 < time_limit and sweep < sweeps:
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit - 5:
                break

            # Try different values for a[i]
            rest_sum = 1.0 - a[i]
            if rest_sum < 1e-15:
                continue

            # The optimal a[i] is where the max convolution is minimized
            # conv[k] as function of a[i]:
            #   if k == 2i: a[i]^2 + c1 (quadratic)
            #   if i <= k <= 2i-1 (and k-i < n): 2*a[i]*a[k-i] + c2 (linear)
            #   if k < i or k-i >= n: constant

            # Bisect for the optimal a[i] in [0, rest_sum]
            lo = 0.0
            hi = rest_sum

            # Use ternary search since the max of convex functions is convex
            for _ in range(30):
                m1 = lo + (hi - lo) / 3
                m2 = hi - (hi - lo) / 3

                for val, test_i in [(m1, i), (m2, i)]:
                    a_test = a.copy()
                    a_test[i] = val
                    # Renormalize
                    st = a_test.sum()
                    if st > 1e-15:
                        a_test = a_test / st
                conv1 = fast_convolve_auto(a_test) if False else None
                # This is too slow for coordinate descent
                # Instead, compute the max convolution incrementally
                break

            # Simplified: just try a few candidate values
            candidates = [0.0, a[i] * 0.5, a[i] * 0.7, a[i] * 1.3,
                          a[i] * 1.5, rest_sum]
            best_val = a[i]
            best_local = conv.max()

            for cv in candidates:
                cv = max(0.0, min(cv, rest_sum))
                a_test = a.copy()
                a_test[i] = cv
                st = a_test.sum()
                if st > 1e-15:
                    a_test = a_test / st
                    conv_test = fast_convolve_auto(a_test)
                    if conv_test.max() < best_local:
                        best_local = conv_test.max()
                        best_val = cv

            if abs(best_val - a[i]) > 1e-8:
                a[i] = best_val
                st = a.sum()
                if st > 1e-15:
                    a = a / st
                improved = True

        if improved:
            conv = fast_convolve_auto(a)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj - 1e-10:
                best_obj = cur_obj
                best_a = a.copy()

        sweep += 1
        if not improved and sweep > 5:
            break

    return best_a, best_obj


def generate_starts(n, parent, rng):
    """Generate diverse starting points."""
    starts = {}
    x = np.arange(n, dtype=np.float64)

    starts['uniform'] = np.ones(n) / n

    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
        ps = pp.sum()
        if ps > 1e-10:
            starts['parent'] = pp / ps

    # Cosine variations
    for freq in [1, 2, 3, 4, 5]:
        a = 1.0 + 0.9 * np.cos(2.0 * np.pi * freq * x / n)
        a = np.maximum(a, 0)
        sa = a.sum()
        if sa > 1e-10:
            starts[f'cos_{freq}'] = a / sa

    # Gaussian centered
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.3 * n)) ** 2)
    starts['gauss'] = g / g.sum()

    # Two Gaussians
    g2 = np.exp(-0.5 * ((x - 0.3 * n) / (0.15 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.7 * n) / (0.15 * n)) ** 2)
    starts['gauss2'] = g2 / g2.sum()

    # Random Dirichlet
    starts['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # Ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    starts['ramp'] = ramp / ramp.sum()

    # Triangular
    tri = 1.0 - 2.0 * np.abs(x / n - 0.5)
    starts['tri'] = tri / tri.sum()

    # Sine squared
    ss = np.sin(np.pi * x / n) ** 2
    starts['sine2'] = ss / ss.sum()

    return starts

        ba_sum = ba.sum()
        if ba_sum > 1e-10:
            ba_n = np.interp(np.linspace(0, 1, n),
                             np.linspace(0, 1, len(ba)), ba)
            ba_n = np.maximum(ba_n, 0)
            sn = ba_n.sum()
            if sn > 1e-10:
                starts['phase1_best'] = ba_n / sn

    # Priority order
    priority = ['phase1_best', 'parent', 'uniform', 'cos_1', 'gauss', 'random',
                'sine2', 'cos_2', 'gauss2', 'tri']
    ordered_keys = [k for k in priority if k in starts] + \
                   [k for k in starts if k not in priority]

    # Run smooth-max on top candidates
    top_candidates = ordered_keys[:6]
    phase2_budget = 500.0
    per_candidate = phase2_budget / len(top_candidates)

    for key in top_candidates:
        if time.monotonic() > deadline - 600:
            break
        a0 = starts[key]
        t_alloc = min(per_candidate, deadline - time.monotonic() - 500)
        if t_alloc < 5:
            break
        print(f"  Smooth-max on init='{key}', alloc={t_alloc:.0f}s")
        a_opt, _ = optimize_smooth_max(a0, time_limit=t_alloc,
                                       t_start=5.0, t_end=5000.0,
                                       t_growth=1.2, lr_start=0.008, mom=0.85)
        try_update(a_opt)

    # Also try L-BFGS-B on the best so far
    if best_sequence is not None and time.monotonic() < deadline - 400:
        bs = np.maximum(best_sequence, 0)
        bs_sum = bs.sum()
        if bs_sum > 1e-10:
            bs = bs / bs_sum
            if len(bs) != n:
                bs = np.interp(np.linspace(0, 1, n),
                               np.linspace(0, 1, len(bs)), bs)
                bs = np.maximum(bs, 0)
                bs_sum2 = bs.sum()
                if bs_sum2 > 1e-10:
                    bs = bs / bs_sum2
            t_alloc = min(80.0, deadline - time.monotonic() - 350)
            if t_alloc > 10:
                print(f"  L-BFGS-B on best, alloc={t_alloc:.0f}s")
                a_opt, _ = optimize_smooth_max_lbfgs(bs, time_limit=t_alloc,
                                                     t_start=20.0, t_steps=10)
                try_update(a_opt)

    print(f"Phase 2 done: {time.monotonic()-t2:.0f}s, best={best_score:.6f}")

    # ============ PHASE 3: Peak-targeted subgradient ============
    print("\n=== Phase 3: Peak-Targeted Subgradient ===")
    t3 = time.monotonic()

    if best_sequence is not None:
        bs = np.maximum(best_sequence, 0)
        bs_sum = bs.sum()
        if bs_sum > 1e-10:
            bs = bs / bs_sum
            if len(bs) > n:
                bs = bs[:n]
                bs = bs / bs.sum()
            elif len(bs) < n:
                bs = np.interp(np.linspace(0, 1, n),
                               np.linspace(0, 1, len(bs)), bs)
                bs = np.maximum(bs, 0)
                bs = bs / bs.sum()

            phase3_budget = 300.0
            configs = [
                {'top_k': 30, 'lr_start': 0.015, 'temp': 0.0005},
                {'top_k': 50, 'lr_start': 0.025, 'temp': 0.001},
                {'top_k': 100, 'lr_start': 0.01, 'temp': 0.002},
                {'top_k': 20, 'lr_start': 0.02, 'temp': 0.0003},
            ]
            per_cfg = phase3_budget / len(configs)
            for cfg in configs:
                if time.monotonic() > deadline - 300:
                    break
                t_alloc = min(per_cfg, deadline - time.monotonic() - 250)
                if t_alloc < 5:
                    break
                a_opt, _ = optimize_subgrad_peak(bs, time_limit=t_alloc,
                                                 top_k=cfg['top_k'],
                                                 lr_start=cfg['lr_start'],
                                                 temp=cfg['temp'])
                try_update(a_opt)

    print(f"Phase 3 done: {time.monotonic()-t3:.0f}s, best={best_score:.6f}")

    # ============ PHASE 4: LP refinement on best ============
    print("\n=== Phase 4: LP Refinement ===")
    t4 = time.monotonic()

    if best_sequence is not None:
        bs = np.maximum(best_sequence, 0)
        bs_sum = bs.sum()
        if bs_sum > 1e-10:
            bs = bs / bs_sum
            n_b = len(bs)

            phase4_budget = 300.0
            lp_configs = [
                {'method': 'max_sum', 'step': 0.3},
                {'method': 'min_peak', 'step': 0.25},
                {'method': 'max_sum', 'step': 0.5},
                {'method': 'min_peak', 'step': 0.15},
                {'method': 'max_sum', 'step': 0.1},
            ]
            per_cfg = phase4_budget / len(lp_configs)
            for cfg in lp_configs:
                if time.monotonic() > deadline - 200:
                    break
                t_alloc = min(per_cfg, deadline - time.monotonic() - 150)
                if t_alloc < 5:
                    break
                a_opt, _ = optimize_lp_refine(bs, time_limit=t_alloc,
                                              method=cfg['method'],
                                              step_init=cfg['step'])
                try_update(a_opt)
                if a_opt is not None:
                    bs = np.maximum(a_opt, 0)
                    bs_sum = bs.sum()
                    if bs_sum > 1e-10:
                        bs = bs / bs_sum

    print(f"Phase 4 done: {time.monotonic()-t4:.0f}s, best={best_score:.6f}")

    # ============ PHASE 5: Final polishing with smooth-max at high temp ============
    print("\n=== Phase 5: Final Polishing ===")
    t5 = time.monotonic()

    if best_sequence is not None:
        bs = np.maximum(best_sequence, 0)
        bs_sum = bs.sum()
        if bs_sum > 1e-10:
            bs = bs / bs_sum

            # High-temperature smooth max for final polish
            t_alloc = min(100.0, deadline - time.monotonic() - 30)
            if t_alloc > 10:
                a_opt, _ = optimize_smooth_max(bs, time_limit=t_alloc,
                                               t_start=50.0, t_end=10000.0,
                                               t_growth=1.1, lr_start=0.003, mom=0.9)
                try_update(a_opt)

            # Another LP pass
            if time.monotonic() < deadline - 20:
                t_alloc = min(50.0, deadline - time.monotonic() - 10)
                if t_alloc > 5:
                    a_opt, _ = optimize_lp_refine(bs, time_limit=t_alloc,
                                                  method='max_sum', step_init=0.1)
                    try_update(a_opt)

    print(f"Phase 5 done: {time.monotonic()-t5:.0f}s, best={best_score:.6f}")

    # ============ PHASE 6: Multi-restart perturbation ============
    print("\n=== Phase 6: Multi-Restart Perturbation ===")
    t6 = time.monotonic()

    if best_sequence is not None:
        bs = np.maximum(best_sequence, 0)
        bs_sum = bs.sum()
        if bs_sum > 1e-10:
            bs = bs / bs_sum
            n_b = len(bs)

        pert_configs = [
            {'scale': 0.001, 'method': 'smooth', 't': 100.0},
            {'scale': 0.005, 'method': 'smooth', 't': 50.0},
            {'scale': 0.002, 'method': 'lp', 'step': 0.2},
            {'scale': 0.01, 'method': 'smooth', 't': 30.0},
            {'scale': 0.003, 'method': 'subgrad', 'top_k': 30},
        ]

        for pc in pert_configs:
            if time.monotonic() > deadline - 10:
                break
            noise = rng.standard_normal(n_b) * pc['scale']
            a_pert = project_simplex(bs + noise)

            t_alloc = min(30.0, deadline - time.monotonic() - 5)
            if t_alloc < 3:
                break

            if pc['method'] == 'smooth':
                a_opt, _ = optimize_smooth_max(a_pert, time_limit=t_alloc,
                                               t_start=pc['t'], t_end=5000.0,
                                               t_growth=1.15, lr_start=0.005,
                                               mom=0.88)
            elif pc['method'] == 'lp':
                a_opt, _ = optimize_lp_refine(a_pert, time_limit=t_alloc,
                                              method='max_sum',
                                              step_init=pc['step'])
            else:
                a_opt, _ = optimize_subgrad_peak(a_pert, time_limit=t_alloc,
                                                 top_k=pc['top_k'],
                                                 lr_start=0.01, temp=0.001)
            try_update(a_opt)

    print(f"Phase 6 done: {time.monotonic()-t6:.0f}s, best={best_score:.6f}")

    # Final check
    if best_sequence is None:
        best_sequence = np.ones(512) / 512

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"\n{'='*50}")
    print(f"FINAL: score={best_score:.6f}, n={len(best_sequence)}")
    print(f"{'='*50}")

    return [float(x) for x in best_sequence]
