"""
Hybrid optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses FFT-based convolution, LP-based Frank-Wolfe for key steps,
Riemannian subgradient descent for fast iterations, and multi-start

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_direction(f, M, n_constraints=None):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    For large n, use a subset of constraints for speed."""
    n = len(f)
    if n_constraints is not None and n_constraints < 2*n-1:
        # Use only the most important constraints (those closest to M)
        conv_f = fast_convolve(np.ones(n))  # not quite right
        # Actually, use all constraints but limit to a subset
        # For now, use all
        pass
    
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    if res.success:
        return res.x
    return None

def subgradient_step(f, lr, mom, temp, top_k, velocity):
    """One subgradient step with multi-peak softmax weighting."""
    n = len(f)
    conv = fast_convolve(f)
    
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-13:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k - idx]
    
    velocity = mom * velocity - lr * sg
    new_f = project_simplex(f + velocity)
    return new_f, velocity, conv.max()

def fw_lp_optimize(f0, time_limit, verbose=False, lp_every=1):
    """Frank-Wolfe with LP oracle, called every lp_every iterations."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    v = np.zeros(n)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        if it % lp_every == 1:
            # LP-based direction
            g = solve_lp_direction(f, M * m_exp)
            if g is None:
                m_exp *= 1.15
                if m_exp > 5.0:
                    m_exp = 1.0
                    continue
                continue
            
            g_sum = g.sum()
            if g_sum < 1e-15:
                break
            g = g / g_sum
            
            if np.max(np.abs(g - f)) < 1e-9:
                no_improve += 1
                if no_improve > 40:
                    break
                m_exp = min(m_exp * 1.05, 3.0)
                continue
            
            # Line search
            best_alpha = 0.0
            best_val = M
            for alpha in [0.1, 0.3, 0.5, 0.7, 0.9]:
                ft = (1.0 - alpha) * f + alpha * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    continue
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = alpha
            
            if best_alpha > 1e-10:
                f = (1.0 - best_alpha) * f + best_alpha * g
                f = np.maximum(f, 0.0)
                f = f / f.sum()
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                v = np.zeros(n)
                
                if M < best_obj - 1e-13:
                    best_obj = M
                    best_f = f.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
                m_exp = min(m_exp * 1.08, 4.0)
        else:
            # Subgradient step
            lr = 0.003 * (1.0 + 0.5 * (it % 100) / 100.0)
            f, v, M_new = subgradient_step(f, lr, 0.93, 0.00005, max(8, n//16), v)
            M = M_new
            if M < best_obj - 1e-13:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        if no_improve > 50:
            rng = np.random.default_rng(it * 7 + 13)
            scale = 0.005 + 0.005 * (no_improve // 50)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            v = np.zeros(n)
            no_improve = 0
    
    return best_f, best_obj

    # Cosine
    c = np.abs(np.cos(np.pi * x / n)) ** 3 + 0.05

    # Sine-based (try to make frequency response flat)
    s = 1.0 + 0.5 * np.cos(2 * np.pi * x / n) + 0.3 * np.cos(4 * np.pi * x / n)

        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    return inits
