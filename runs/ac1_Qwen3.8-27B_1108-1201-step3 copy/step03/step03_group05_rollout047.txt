"""
Hybrid optimizer combining spectral initialization, Nesterov smoothed gradient descent,
LP direction finding with warm starts, and peak-targeted coordinate descent.
Explores multiple sequence lengths to find the global optimum.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smoothed_gradient(a, conv, beta):
    n = len(a)
    scaled = beta * conv
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p /= p_sum
    w = conv * p
    fft_size = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w, fft_size)
    cross = np.fft.irfft(A * np.conj(W), fft_size)
    return 2.0 * cross[:n]

def spectral_flatten(a0, n_iter=50):
    """Iteratively flatten the power spectrum (like GS algorithm)."""
    a = a0.copy()
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    for _ in range(n_iter):
        # Compute power spectrum
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A)**2
        
        # Target: flat spectrum (with DC constraint from sum(a)=1)
        # DC bin should be (sum(a))^2 = 1
        # Other bins: target = (total power - DC) / (num_bins - 1)
        # Total power = sum(|A|^2) = fft_size * sum(a^2) by Parseval
        target_dc = 1.0  # (sum a)^2
        total_power = P.sum()
        target_flat = (total_power - target_dc) / (len(P) - 1) if len(P) > 1 else 0
        
        # New spectrum: keep DC, flatten the rest
        P_new = P.copy()
        P_new[0] = target_dc
        if len(P_new) > 1:
            P_new[1:] = target_flat
        
        # Take IFFT and project
        A_new = np.sqrt(P_new)  # magnitude
        # Keep original phase
        if np.abs(A).max() > 1e-30:
            phase = A / np.abs(A)
            phase[np.abs(A) < 1e-30] = 1.0
            A_new = A_new * phase
        else:
            A_new = P_new.copy()
        
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-10:
            a_new /= s
        else:
            a_new = a.copy()
        a = a_new
    
    return a

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_gradient(a, conv, beta):
    """Gradient of logsumexp(beta*conv)/beta w.r.t. a via FFT cross-correlation."""
    n = len(a)
    scaled = beta * conv
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p /= p_sum
    w = conv * p
    fft_size = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w, fft_size)
    cross = np.fft.irfft(A * np.conj(W), fft_size)
    return 2.0 * cross[:n]


def lp_direction(a, M, n, previous_b=None):
    """Solve LP for direction finding. Returns normalized b or None."""
    c_obj = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for i in range(n):
        k_start = i
        k_end = min(2 * n - 1, i + n)
        A_ub[k_start:k_end, i] = a[:k_end - k_start]
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    opts = {'presolve': True, 'time_limit': 20}
    if previous_b is not None:
        opts['presolve'] = True
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                         method='highs', options=opts)
    except Exception:
        return None
    if not result.success or result.x is None:
        return None
    b = np.maximum(result.x, 0)
    b_sum = b.sum()
    if b_sum < 1e-10:
        return None
    return b / b_sum


def spectral_init(n, rng, n_gs_iter=30):
    """Initialize with spectral flattening (Gerchberg-Saxton style)."""
    x = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((x - 0.5 * n) / (0.25 * n)) ** 2)
    a = np.maximum(a + rng.standard_normal(n) * 0.3, 0)
    a = a / a.sum()
    
    fft_size = 1 << (2 * n - 1).bit_length()
    for _ in range(n_gs_iter):
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        # Target: flatten non-DC bins
        total = P.sum()
        target = total / len(P)
        P_new = np.full_like(P, target)
        P_new[0] = P[0]  # Keep DC (preserves sum)
        mag = np.sqrt(P_new)
        phase = np.ones_like(A)
        mask = np.abs(A) > 1e-15
        phase[mask] = A[mask] / np.abs(A)[mask]
        A_new = mag * phase
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-10:
            a_new /= s
        a = a_new
    return a


def diverse_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Spectral
    inits['spectral'] = spectral_init(n, rng, 20)
    
    # Random with structure
    inits['rand1'] = rng.dirichlet(np.ones(n) * 0.3)
    inits['rand2'] = rng.dirichlet(np.ones(n) * 1.0)
    inits['rand3'] = rng.dirichlet(np.ones(n) * 3.0)
    
    # Alternating (checkerboard-like)
    alt = np.ones(n)
    alt[1::2] = 0.3
    inits['alternating'] = alt / alt.sum()
    
    # Two-scale
    coarse = rng.dirichlet(np.ones(max(2, n // 4)) * 0.5)
    fine = np.repeat(coarse, 4)[:n]
    fine = fine + rng.standard_normal(n) * 0.05
    fine = np.maximum(fine, 0)
    inits['two_scale'] = fine / fine.sum()
    
    # Cosine modulated
    cos_mod = 0.5 + 0.5 * np.cos(2 * np.pi * x / n * rng.integers(2, 8))
    cos_mod = np.maximum(cos_mod + rng.standard_normal(n) * 0.2, 0.01)
    inits['cos_mod'] = cos_mod / cos_mod.sum()
    
    # Parent-based
    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent'] = pp / ps
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_up'] = pp / ps
        if len(parent) > n:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
            ps = pp.sum()
            if ps > 1e-10:
                inits['parent_dn'] = pp / ps
    
    return inits


def nesterov_smoothed(a0, time_limit, beta_init=20.0, beta_end=300.0,
                       lr_init=0.012, momentum=0.92, seed=0):

    no_improve = 0
    iters = 0
    
    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / max(time_limit, 1e-6))
        beta = beta_init + (beta_end - beta_init) * (frac ** 0.7)

        y = a + momentum * v
        y = np.maximum(y, 0)
        sy = y.sum()

            y = y / sy

        obj_new = conv_new.max()
        
        if obj_new > conv.max() * 1.05:

            v *= 0.5
            a_new = project_simplex(a + 0.5 * v)
            conv_new = fast_convolve(a_new)

        conv = conv_new
        iters += 1
        
        if obj_new < best_obj - 1e-12:

            no_improve = 0
        else:
            no_improve += 1

            conv_flat = conv.copy()
            conv_flat -= conv_flat.min()
            conv_flat = conv_flat / (conv_flat.max() + 1e-15)

            for k in np.argpartition(conv, -5)[-5:]:

            contrib = contrib / (contrib.max() + 1e-15)
            noise = rng.standard_normal(n) * 0.003 * (1.0 - contrib)

                best_sequence = a_cur.copy()
                print(f"    Fine: {score_d:.6f}", flush=True)
    
    # === Phase 3: Intensive polish on best ===
    print("=== Phase 3: Polish ===", flush=True)
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 15.0
        
        # Alternating LP and gradient
        n_rounds = 4
        per_round = remaining / n_rounds
        
        for r in range(n_rounds):
            if time.monotonic() > deadline - 10:
                break
            t_r = min(per_round, 100.0)
            if t_r < 5:
                break
            
            # LP
            t_lp = t_r * 0.5
            if t_lp > 5:
                a_cur, _ = lp_refinement(best_sequence, t_lp, seed=1000 + r)
                score = float(evaluate_sequence(a_cur.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_cur.copy()
                    print(f"  Round {r} LP: {best_score:.6f}", flush=True)
            
            # Gradient
            t_g = t_r * 0.35
            if t_g > 5:
                a_cur, _ = nesterov_smoothed(best_sequence, t_g,
                                             beta_init=500, beta_end=2000,
                                             lr_init=0.001, momentum=0.96,
                                             seed=2000 + r)
                score = float(evaluate_sequence(a_cur.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_cur.copy()
                    print(f"  Round {r} Grad: {best_score:.6f}", flush=True)
            
            # Peak descent
            t_p = t_r * 0.15
            if t_p > 3:
                a_cur, _ = peak_coordinate_descent(best_sequence, t_p, seed=3000 + r)
                score = float(evaluate_sequence(a_cur.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_cur.copy()
                    print(f"  Round {r} Peak: {best_score:.6f}", flush=True)
    
    # === Phase 4: Final micro-polish ===
    remaining = deadline - time.monotonic() - 5.0
    if best_sequence is not None and remaining > 10:
        # Very fine gradient
        a_cur, _ = nesterov_smoothed(best_sequence, remaining * 0.6,
                                     beta_init=1000, beta_end=5000,
                                     lr_init=0.0005, momentum=0.97,
                                     seed=9999)
        score = float(evaluate_sequence(a_cur.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_cur.copy()
            print(f"  Micro: {best_score:.6f}", flush=True)
        
        # Final peak shave
        remaining2 = deadline - time.monotonic() - 3.0
        if remaining2 > 5:
            a_cur, _ = peak_coordinate_descent(best_sequence, remaining2, seed=4242)
            score = float(evaluate_sequence(a_cur.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_cur.copy()
                print(f"  FinalShave: {best_score:.6f}", flush=True)
    
    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
    
    # Final check
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    
    print(f"\nFINAL: {best_score:.6f} (n={len(best_sequence)})", flush=True)
    return [float(x) for x in best_sequence]
