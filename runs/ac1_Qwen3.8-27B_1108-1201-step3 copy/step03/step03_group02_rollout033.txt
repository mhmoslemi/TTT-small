"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive Adam optimization with
softmax smoothing, temperature annealing, LP-based direction resets, and
two-coordinate pair search. Multi-length exploration with restarts.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_grad(a, t, top_k=None):
    """Compute softmax-smoothed objective gradient w.r.t. a (normalized to sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    if top_k is not None and top_k < len(conv):
        # Use only top-k peaks for gradient
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        scaled = t * conv[peaks]
        scaled -= scaled.max()
        exp_vals = np.exp(scaled)
        p = exp_vals / exp_vals.sum()
        # Gradient: sum over active peaks of p_k * 2*a[k-j]
        grad = np.zeros(n)
        for idx in range(len(peaks)):
            if p[idx] < 1e-10:
                continue
            k = int(peaks[idx])
            wi = 2.0 * p[idx]
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idxs = np.arange(j_start, j_end + 1)
            grad[idxs] += wi * a[k - idxs]
    else:
        scaled = t * conv
        scaled -= scaled.max()
        exp_vals = np.exp(scaled)
        p = exp_vals / exp_vals.sum()
        grad = np.zeros(n)
        for k in range(len(conv)):
            if p[k] < 1e-10:
                continue
            wi = 2.0 * p[k]
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idxs = np.arange(j_start, j_end + 1)
            grad[idxs] += wi * a[k - idxs]
    # Project gradient to remove component along 1 (for simplex constraint)
    grad -= grad.mean()
    return grad, conv


def softmax_grad_fft(a, t):
    """Faster gradient computation using FFT for the full softmax case."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = t * conv
    scaled -= scaled.max()
    exp_vals = np.exp(scaled)
    p = exp_vals / exp_vals.sum()
    # grad[j] = 2 * sum_k p[k] * a[k-j] for valid k-j
    # This is 2 * conv(p, a)[j] but with appropriate range
    # conv(p, a)[j] = sum_k p[k] * a[j-k] for valid indices
    # We want sum_k p[k] * a[k-j] = conv_rev(p, a)
    # Actually: sum_k p[k] * a[k-j] where k-j in [0,n-1]
    # = sum_{k=j}^{j+n-1} p[k] * a[k-j] (with bounds)
    # This is the cross-correlation of p and a
    # = conv(p_flipped, a) shifted... let's just use direct
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    A = np.fft.rfft(a, fft_size)
    # sum_k p[k]*a[k-j] = (p * a_flipped)[j] circularly... 
    # Let's use: corr = ifft(fft(p) * conj(fft(a))) shifted
    # Actually simpler: grad[j] = 2 * sum_k p[k] * a[k-j] * 1_{0<=k-j<n}
    # = 2 * (p cross-correlated with a)[j]
    # In FFT: ifft(fft(p) * conj(fft(a_reversed)))... 
    # Let me just do it directly with the linear convolution structure
    # grad[j] = 2 * sum_{k: 0<=k-j<n} p[k] * a[k-j]
    # Let m = k-j, then k = m+j, m in [0, n-1], k in [j, j+n-1] ∩ [0, 2n-2]
    # grad[j] = 2 * sum_{m=0}^{min(n-1, 2n-2-j)} p[j+m] * a[m]
    # This is 2 * cross_correlation(p, a)[j]
    # Using FFT: cross_correlation = ifft(fft(p, size=4n) * conj(fft(a, size=4n)))
    N = 1 << (4 * n - 1).bit_length()
    P_full = np.fft.rfft(p, N)
    A_full = np.fft.rfft(a, N)
    corr = np.fft.irfft(P_full * np.conj(A_full), N)
    # corr[m] = sum_k p[k] * a[k-m] (circular, but N is large enough)
    # We want grad[j] = 2 * sum_m a[m] * p[j+m] = 2 * corr[j] (for j in 0..n-1)
    grad = 2.0 * corr[:n]
    grad -= grad.mean()
    return grad, conv


def build_conv_matrix(a):
    """Build matrix where (A @ b)[k] = conv(a, b)[k]."""
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A


def lp_maximize_sum(a, M):
    """Solve: max sum(b) s.t. b >= 0, conv(a,b)[k] <= M for all k."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if res.status == 0:
        return np.maximum(res.x, 0)
    return None


def lp_minimize_peak(a, time_limit=30):
    """Solve: min M' s.t. sum(b)=1, b>=0, conv(a,b)[k] <= M'."""
    n = len(a)
    A = build_conv_matrix(a)
    num_rows = 2 * n - 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    A_ub = np.zeros((num_rows, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(num_rows)
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': time_limit})
    if res.status == 0:
        return np.maximum(res.x[:n], 0), res.x[n]
    return None, None


def adam_simplex(a0, time_limit, lr_init=0.01, t_init=5.0, t_final=0.5,
                 beta1=0.9, beta2=0.999, eps=1e-8, top_k=None,
                 use_fft=True, restart_every=2000, restart_scale=0.01):
    """Adam optimizer with simplex projection and temperature annealing."""
    a = np.maximum(a0, 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    m = np.zeros(n)
    v = np.zeros(n)
    t0 = time.monotonic()
    total_time = time_limit
    step = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        step += 1
        progress = min(1.0, (time.monotonic() - t0) / total_time)
        # Temperature annealing: exponential decay
        t_cur = t_init * (t_final / t_init) ** progress
        # Learning rate: cosine decay
        lr = lr_init * 0.5 * (1.0 + np.cos(np.pi * progress))
        lr = max(lr, lr_init * 0.01)

        if use_fft:
            grad, conv = softmax_grad_fft(a, t_cur)
        else:
            grad, conv = softmax_grad(a, t_cur, top_k=top_k)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = project_simplex(a)

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-9:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Periodic restart
        if no_improve >= restart_every:
            rng = np.random.default_rng(hash((step, n)) % 2**31)
            a = project_simplex(a + rng.standard_normal(n) * restart_scale)
            m *= 0.1
            v *= 0.1
            no_improve = 0
            restart_every = int(restart_every * 1.5)

    return best_a, best_obj


def lp_refine(a0, time_limit, method='alternating', step_sizes=None):
    """LP-based iterative refinement."""
    if step_sizes is None:
        step_sizes = [0.4, 0.3, 0.2, 0.15, 0.1, 0.05]
    a = np.maximum(a0, 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    si = 0
    step = step_sizes[0]
    use_max_sum = (method == 'max_sum') or (method == 'alternating')

    while time.monotonic() - t0 < time_limit:
        # Check time for LP (reserve some)
        lp_time = min(60.0, time_limit - (time.monotonic() - t0) - 5)
        if lp_time < 3:
            break

        M = conv.max()
        if use_max_sum:
            b = lp_maximize_sum(a, M)
            if b is None:
                break
            b_sum = b.sum()
            if b_sum < 1e-15:
                break
            b = b / b_sum
        else:
            b, _ = lp_minimize_peak(a, time_limit=max(3, lp_time - 2))
            if b is None:
                break
            b_sum = b.sum()
            if b_sum < 1e-15:
                break
            b = b / b_sum

        if np.linalg.norm(b - a) < 1e-10:
            break

        a_new = (1.0 - step) * a + step * b
        a_new = np.maximum(a_new, 0.0)
        a_sum = a_new.sum()
        if a_sum < 1e-15:
            break
        a_new = a_new / a_sum

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            # Slightly increase step
            if step < 0.5:
                step = min(0.5, step * 1.05)
        else:
            step *= 0.85
            if step < 0.02:
                # Try different method
                use_max_sum = not use_max_sum
                step = step_sizes[0]
                si = (si + 1) % len(step_sizes)

        a = a_new
        conv = conv_new

    return best_a, best_obj


def two_coord_search(a0, time_limit, n_trials=5000):
    """Search over pairs of coordinates for optimal mass redistribution."""
    a = np.maximum(a0, 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    trial = 0

    while time.monotonic() - t0 < time_limit and trial < n_trials:
        trial += 1
        # Pick two random coordinates
        j, l = rng.integers(0, n, size=2)
        if j == l:
            continue

        # Current values
        aj, al = a[j], a[l]
        if aj + al < 1e-12:
            continue

        # Try redistributing mass between j and l
        # a[j] = alpha * (aj + al), a[l] = (1-alpha) * (aj + al)
        # Find alpha that minimizes max(conv)
        total = aj + al
        best_alpha = 0.5
        best_local = best_obj

        # Golden section search
        gr = (np.sqrt(5) - 1) / 2
        lo, hi = 0.0, 1.0
        for _ in range(20):
            m1 = hi - gr * (hi - lo)
            m2 = lo + gr * (hi - lo)
            a[j] = m1 * total
            a[l] = (1 - m1) * total
            c1 = fast_convolve(a)
            o1 = 2.0 * n * c1.max()

            a[j] = m2 * total
            a[l] = (1 - m2) * total
            c2 = fast_convolve(a)
            o2 = 2.0 * n * c2.max()

            if o1 < o2:
                hi = m2
            else:
                lo = m1

        best_alpha = (lo + hi) / 2
        a[j] = best_alpha * total
        a[l] = (1 - best_alpha) * total
        conv_new = fast_convolve(a)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a.copy()
        else:
            # Revert
            a[j] = aj
            a[l] = al
            conv = fast_convolve(a)

    return best_a, best_obj


def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent'] = pp / ps

    for freq in [1, 2, 3, 5]:
        a = 1.0 + 0.9 * np.cos(2.0 * np.pi * freq * x / n)
        a = np.maximum(a, 0)
        if a.sum() > 1e-10:
            a /= a.sum()
            inits[f'cos_{freq}'] = a

    # Raised cosine
    a = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    a = np.maximum(a, 0)
    if a.sum() > 1e-10:
        a /= a.sum()
        inits['raised_cos'] = a

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.3 * n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Two peaks
    p1, p2 = n // 3, 2 * n // 3
    a = np.exp(-0.5 * ((x - p1) / (n // 10)) ** 2) + np.exp(-0.5 * ((x - p2) / (n // 10)) ** 2)
    a /= a.sum()
    inits['two_peaks'] = a

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search: Adam optimization with annealing, LP refinement,
    two-coordinate search, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: len={len(p)}, sum={p.sum():.4f}")
    except Exception:
        pass

    def try_update(a):
        nonlocal best_sequence, best_score
        if a is None:
            return best_score
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return best_score
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print(f"  New best: {score:.6f} (n={len(a)}) [t={time.monotonic()-deadline-30:.0f}s]")
        return score

    # === Phase 1: Adam optimization on parent length ===
    print("=== Phase 1: Adam Optimization ===")
    t1_start = time.monotonic()

    n_main = len(parent) if parent is not None else 512
    if n_main < 32:
        n_main = 512

    # Evaluate parent first
    if parent is not None and n_main == len(parent):
        p_norm = np.maximum(parent, 0)
        p_norm = p_norm / p_norm.sum()
        try_update(p_norm)

    # Run Adam with different hyperparameters
    adam_configs = [
        {'lr_init': 0.02, 't_init': 8.0, 't_final': 0.3, 'top_k': None},
        {'lr_init': 0.01, 't_init': 4.0, 't_final': 0.5, 'top_k': 64},
        {'lr_init': 0.015, 't_init': 6.0, 't_final': 0.2, 'top_k': 128},
    ]

    inits = generate_diverse_inits(n_main, rng, parent)
    init_names = ['parent', 'uniform', 'random', 'cos_1', 'raised_cos']
    init_names = [nm for nm in init_names if nm in inits]

    p1_deadline = deadline - 500  # Reserve 500s for other phases
    n_configs = len(adam_configs)
    n_inits = len(init_names)
    time_per = max(10.0, (p1_deadline - time.monotonic()) / (n_configs * max(1, n_inits) - 2))

    # Start with parent init and best config
    for cfg_i, cfg in enumerate(adam_configs):
        for ini_i, name in enumerate(init_names):
            elapsed = time.monotonic() - t1_start
            remain = p1_deadline - time.monotonic()
            if remain < 15:
                break
            t_alloc = min(time_per * 1.5, remain * 0.8)
            if t_alloc < 8:
                continue
            if cfg_i > 0 and ini_i > 0:
                # Only do full search for a few combos
                if remain < 60:
                    break

            a0 = inits[name]
            print(f"  Adam: cfg={cfg_i}, init={name}, t={t_alloc:.0f}s")
            a_opt, _ = adam_simplex(a0, time_limit=t_alloc,
                                    lr_init=cfg['lr_init'],
                                    t_init=cfg['t_init'],
                                    t_final=cfg['t_final'],
                                    top_k=cfg['top_k'],
                                    use_fft=True)
            try_update(a_opt)

    print(f"Phase 1 done: best={best_score:.6f}, elapsed={time.monotonic()-t1_start:.0f}s")

    # === Phase 2: LP Refinement ===
    print("=== Phase 2: LP Refinement ===")
    t2_start = time.monotonic()

    if best_sequence is not None:
        p2_deadline = deadline - 250
        for method in ['alternating', 'max_sum', 'min_peak']:
            if time.monotonic() > p2_deadline:
                break
            t_alloc = min(120.0, p2_deadline - time.monotonic() - 5)
            if t_alloc < 10:
                break
            print(f"  LP: method={method}, t={t_alloc:.0f}s")
            a_opt, _ = lp_refine(best_sequence, time_limit=t_alloc, method=method)
            try_update(a_opt)

    print(f"Phase 2 done: best={best_score:.6f}, elapsed={time.monotonic()-t2_start:.0f}s")

    # === Phase 3: Two-coordinate search ===
    print("=== Phase 3: Two-Coordinate Search ===")
    t3_start = time.monotonic()

    if best_sequence is not None:
        p3_deadline = deadline - 150
        t_alloc = min(120.0, p3_deadline - time.monotonic() - 5)
        if t_alloc > 10:
            print(f"  2-coord search: t={t_alloc:.0f}s")
            a_opt, _ = two_coord_search(best_sequence, time_limit=t_alloc, n_trials=3000)
            try_update(a_opt)

    print(f"Phase 3 done: best={best_score:.6f}, elapsed={time.monotonic()-t3_start:.0f}s")

    # === Phase 4: Final Adam polish with low temperature ===
    print("=== Phase 4: Final Polish ===")
    t4_start = time.monotonic()

    if best_sequence is not None:
        p4_deadline = deadline - 30
        remaining = p4_deadline - time.monotonic()
        if remaining > 20:
            print(f"  Final Adam: t={remaining:.0f}s")
            a_opt, _ = adam_simplex(best_sequence, time_limit=remaining * 0.6,
                                    lr_init=0.005, t_init=2.0, t_final=0.1,
                                    restart_every=1000, restart_scale=0.005)
            try_update(a_opt)
            # One more with different params
            remaining2 = p4_deadline - time.monotonic()
            if remaining2 > 10:
                a_opt, _ = adam_simplex(best_sequence, time_limit=remaining2,
                                        lr_init=0.003, t_init=1.5, t_final=0.05,
                                        top_k=32, use_fft=False,
                                        restart_every=800, restart_scale=0.003)
                try_update(a_opt)

    print(f"Phase 4 done: best={best_score:.6f}, elapsed={time.monotonic()-t4_start:.0f}s")

    # === Phase 5: Try alternate lengths if time remains ===
    print("=== Phase 5: Alternate Lengths ===")
    t5_start = time.monotonic()

    alt_lengths = [448, 576, 640, 384]
    for alt_n in alt_lengths:
        if time.monotonic() > deadline - 20:
            break
        remain = deadline - time.monotonic() - 10
        if remain < 30:
            break
        if abs(alt_n - len(best_sequence)) < 50 if best_sequence is not None else False:
            continue
        print(f"  Trying n={alt_n}, t={remain:.0f}s")
        inits = generate_diverse_inits(alt_n, rng, parent)
        for name in ['parent', 'uniform', 'random']:
            if name not in inits:
                continue
            if time.monotonic() > deadline - 15:
                break
            t_alloc = remain * 0.4
            if t_alloc < 10:
                break
            a_opt, _ = adam_simplex(inits[name], time_limit=t_alloc,
                                    lr_init=0.015, t_init=5.0, t_final=0.3)
            try_update(a_opt)

    print(f"Phase 5 done: best={best_score:.6f}, elapsed={time.monotonic()-t5_start:.0f}s")

    # Final fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256

    # Verify final
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
