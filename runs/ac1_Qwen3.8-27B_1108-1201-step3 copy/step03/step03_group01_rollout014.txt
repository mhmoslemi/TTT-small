    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C

def subgradient_at_peaks(a, conv, n, threshold_frac=0.99):
    """Compute subgradient using all peaks above threshold_frac * max."""
    max_val = conv.max()
    if max_val < 1e-15:
        return np.zeros(n)
    active = np.where(conv >= threshold_frac * max_val)[0]
    # Weight by how close to max
    weights = conv[active] / max_val
    weights = weights ** 10  # sharpen
    weights /= weights.sum()
    
    sg = np.zeros(n)
    for w, k in zip(weights, active):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        sg[idx] += w * 2.0 * a[k - idx]
    return sg

def sg_optimize(a0, time_limit, lr_init=0.01, momentum=0.9, threshold_frac=0.99):

def subgradient_at_peaks(a, conv, n, threshold_frac=0.995, power=8.0):
    """Compute subgradient using all peaks above threshold with sharp weighting."""
    max_val = conv.max()
    if max_val < 1e-15:

        weights[0] = 1.0

        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)

            parent = p
            print(f"Parent loaded: n={len(p)}, sum={p.sum():.2f}")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is not None:
        parent_norm = parent / parent.sum()
        init_score = float(evaluate_sequence(parent.tolist()))
        if np.isfinite(init_score):
            update_best(parent_norm.copy(), init_score)

    # Phase 1: Quick subgradient from parent (120s)
    print("=== Phase 1: Quick SG Initialization ===")
    t_start = time.monotonic()
    p1_limit = min(120.0, (deadline - time.monotonic()) * 0.1)
    if p1_limit > 10:
        n_parent = len(parent) if parent is not None else 512
        a_init = parent_norm if parent is not None else np.ones(512) / 512
        a_opt, obj_opt = sg_optimize(a_init, time_limit=p1_limit, lr_init=0.01, momentum_init=0.95)
        score = float(evaluate_sequence(a_opt.tolist()))
        update_best(a_opt, score)
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # Phase 2: Deep LP optimization on the best (900s)
    print("=== Phase 2: Deep LP Optimization ===")
    t_start = time.monotonic()
    p2_limit = min(900.0, max(50.0, (deadline - time.monotonic()) * 0.5))
    if p2_limit > 20 and best_sequence is not None:
        a_opt, obj_opt = lp_optimize_deep(best_sequence, time_limit=p2_limit)
        score = float(evaluate_sequence(a_opt.tolist()))
        update_best(a_opt, score)
    print(f"Phase 2 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # Phase 3: Mixed LP+SG (300s)
    print("=== Phase 3: Mixed LP+SG ===")
    t_start = time.monotonic()
    p3_limit = min(300.0, max(30.0, (deadline - time.monotonic()) * 0.2))
    if p3_limit > 15 and best_sequence is not None:
        a_opt, obj_opt = mixed_optimize(best_sequence, time_limit=p3_limit)
        score = float(evaluate_sequence(a_opt.tolist()))
        update_best(a_opt, score)
    print(f"Phase 3 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # Phase 4: Multi-length exploration (200s)
    print("=== Phase 4: Multi-Length Exploration ===")
    t_start = time.monotonic()
    p4_limit = min(200.0, max(10.0, (deadline - time.monotonic()) * 0.15))
    if p4_limit > 20 and best_sequence is not None:
        n_best = len(best_sequence)
        other_ns = [384, 640, 768, 1024]
        # Filter to reasonable lengths
        other_ns = [n for n in other_ns if n != n_best and 128 <= n <= 1500]

        t_per_n = p4_limit / max(1, len(other_ns))
        for n_try in other_ns:
            if time.monotonic() > deadline - 30:
                break
            t_alloc = min(t_per_n, deadline - time.monotonic() - 25)
            if t_alloc < 10:
                break

            a_interp = interpolate_sequence(best_sequence, n_try)
            # Quick SG to adapt to new length
            a_opt, obj_opt = sg_optimize(a_interp, time_limit=min(t_alloc * 0.4, 30), lr_init=0.01, momentum_init=0.93)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                # Further optimize with LP
                remaining = t_alloc - (time.monotonic() - time.monotonic())
                if time.monotonic() < deadline - 30:
                    t_lp = min(t_alloc * 0.5, deadline - time.monotonic() - 25)
                    if t_lp > 10:
                        a_opt2, obj_opt2 = lp_optimize_deep(a_opt, time_limit=t_lp)
                        score2 = float(evaluate_sequence(a_opt2.tolist()))
                        update_best(a_opt2, score2)
                else:
                    update_best(a_opt, score)
            else:
                print(f"  n={n_try} didn't improve: {score:.6f}")

    print(f"Phase 4 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # Phase 5: Final polish with subgradient (remaining time)
    print("=== Phase 5: Final SG Polish ===")
    t_start = time.monotonic()
    remaining = deadline - time.monotonic() - 5.0
    if remaining > 15 and best_sequence is not None:
        a_opt, obj_opt = sg_optimize(best_sequence, time_limit=remaining, lr_init=0.003, momentum_init=0.94)
        score = float(evaluate_sequence(a_opt.tolist()))
        update_best(a_opt, score)

    print(f"Phase 5 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final clip and verify
    best_sequence = np.clip(best_sequence, 0.0, 1000.0)
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
