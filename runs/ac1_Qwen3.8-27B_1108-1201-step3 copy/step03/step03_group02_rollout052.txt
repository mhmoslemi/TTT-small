```python
def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)

        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
```

```python
def smooth_obj_and_grad_z(z, t, n):
    """Objective and gradient with respect to z (softmax parameterization)."""
    # Compute a = softmax(z)
    z_shift = z - z.max()
    exp_z = np.exp(z_shift)
    a = exp_z / exp_z.sum()
    
    # Compute convolution
    fft_size = 1 << (2*n-2).bit_length()  # need at least 2n-1
    A = np.fft.rfft(a, fft_size)
    power = A * np.conj(A)  # |A|^2
    conv = np.fft.irfft(power, fft_size)[:2*n-1].real
    
    # Compute softmax of t*conv
    tc = t * conv
    tc_shift = tc - tc.max()
    p = np.exp(tc_shift)
    p_sum = p.sum()
    p = p / p_sum
    
    # Objective: 2*n/t * logsumexp(t*conv) = 2*n/t * (log(p_sum) + tc.max())
    obj = 2.0 * n / t * (np.log(p_sum) + tc.max())
    
    # Gradient wrt a: grad_j = 2*t * sum_{k=j}^{j+n-1} p_k * a_{k-j}
    # This is 2*t * (p cross-correlation a)[j]
    P = np.fft.rfft(p, fft_size)
    corr = np.fft.irfft(P * np.conj(A), fft_size)[:n].real
    grad_a = 2.0 * t * corr
    
    # Gradient wrt z: dL/dz_j = a_j * (grad_a_j - sum_k a_k * grad_a_k)
    mean_grad = np.dot(a, grad_a)
    grad_z = a * (grad_a - mean_grad)
    
    return obj, grad_z
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-scale L-BFGS-B with
log-sum-exp smoothing (softmax parameterization), LP-based refinement,
and subgradient polishing. Key improvements over previous approach:
(1) L-BFGS-B on smooth objective converges faster than projected gradient,
(2) multi-scale search across sequence lengths 512/1024/2048,
(3) cooperative LP polishing after gradient convergence,
(4) adaptive temperature annealing schedule.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog


def fast_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size].real


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(z):
    """Numerically stable softmax."""
    z_shift = z - z.max()
    e = np.exp(z_shift)
    return e / e.sum()


def smooth_objgrad(z, t):
    """
    Log-sum-exp smoothed objective and gradient w.r.t. z (softmax params).
    a = softmax(z), obj ~= 2n/t * logsumexp(t * conv(a,a))
    """
    n = len(z)
    a = softmax_fn(z)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    power = A * np.conj(A)
    conv = np.fft.irfft(power, fft_size)[:size].real
    tc = t * conv
    tc_max = tc.max()
    exp_tc = np.exp(tc - tc_max)
    log_sum_exp = tc_max + np.log(exp_tc.sum())
    obj = 2.0 * n / t * log_sum_exp
    p = exp_tc / exp_tc.sum()
    P = np.fft.rfft(p, fft_size)
    corr = np.fft.irfft(P * np.conj(A), fft_size)[:n].real
    grad_a = 2.0 * t * corr
    mean_grad = np.dot(a, grad_a)
    grad_z = a * (grad_a - mean_grad)
    return obj, grad_z


def run_lbfgsb(a_init, time_limit, t_start, t_end):
    """
    L-BFGS-B with temperature annealing on the smoothed objective.
    Returns (best_a, best_true_obj).
    """
    n = len(a_init)
    a = np.maximum(a_init, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    eps = 1e-30
    z = np.log(np.maximum(a, eps))
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    n_temps = max(3, min(25, int(np.cbrt(max(time_limit, 10)))))
    temps = np.geomspace(t_start, t_end, n_temps)
    time_per = time_limit / n_temps

    for t in temps:
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        remaining = time_limit - elapsed
        if remaining < 3:
            break
        t_alloc = min(time_per * 1.5, remaining)
        # More iterations at higher t (sharper landscape)
        max_iter = max(100, int(500 * t / t_start))
        max_iter = min(max_iter, 5000)
        try:
            result = minimize(smooth_objgrad, z, method='L-BFGS-B',
                              args=(t,), jac=True,
                              options={'maxiter': max_iter, 'ftol': 1e-15,
                                       'gtol': 1e-10, 'maxfun': max_iter * 6,
                                       'maxls': 50})
        except Exception:
            break
        a_new = softmax_fn(result.x)
        conv = fast_convolve(a_new)
        obj_new = 2.0 * n * conv.max()
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
        z = result.x
    return best_a, best_obj


def build_conv_matrix(a):
    """Build matrix where (A @ b)[k] = conv(a,b)[k] for k=0..2n-2."""
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    return A


def lp_solve_max_sum(a, M):
    """LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x, 0)
    return None


def lp_iterative(a0, time_limit, step_init=0.35):
    """LP-based iterative refinement."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    step = step_init
    no_improve = 0
    while time.monotonic() - t0 < time_limit:
        M = conv.max()
        b = lp_solve_max_sum(a, M)
        if b is None:
            break
        b_sum = b.sum()
        if b_sum < 1e-15:
            break
        b = b / b_sum
        if np.linalg.norm(b - a) < 1e-10:
            break
        a_new = (1.0 - step) * a + step * b
        a_new = np.maximum(a_new, 0.0)
        a_sum = a_new.sum()
        if a_sum < 1e-15:
            break
        a_new = a_new / a_sum
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        if obj_new < best_obj - 1e-10:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            step = min(0.8, step * 1.04)
        else:
            no_improve += 1
            step = max(0.02, step * 0.93)
        a = a_new
        conv = conv_new
        if no_improve > 250:
            break
    return best_a, best_obj


def subgrad_optimize(a0, time_limit, lr_init=0.04, momentum=0.88, top_k=40):
    """Projected subgradient on top-k peaks with momentum."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    v = np.zeros(n)
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    rng_local = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        # Find top-k peaks
        if top_k < len(conv):
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            peak_vals = conv[peaks]
        else:
            peaks = np.arange(len(conv))
            peak_vals = conv.copy()
        # Soft weights
        temp = max(peak_vals.max() * 0.01, 1e-8)
        scaled = peak_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        # Subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            wi = 2.0 * w[i]
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += wi * a[k - idx]

        v = momentum * v - lr * sg
        a = project_simplex(a + v)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        lr *= 0.9995
        if no_improve > 3000:
            noise = rng_local.standard_normal(n) * 0.003
            a = project_simplex(a + noise)
            conv = fast_convolve(a)
            v *= 0.3
            no_improve = 0
            lr = max(lr_init * 0.3, lr * 0.7)
    return best_a, best_obj


def upsample(a, new_n):
    """Upsample a sequence using linear interpolation."""
    if len(a) == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, len(a))
    x_new = np.linspace(0, 1, new_n)
    result = np.interp(x_new, x_old, a)
    result = np.maximum(result, 0)
    s = result.sum()
    if s > 1e-15:
        result /= s
    return result


def generate_init(n, parent, rng, kind):
    """Generate a diverse initialization."""
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'parent':
        if parent is not None:
            p = np.maximum(parent, 0).astype(np.float64)
            ps = p.sum()
            if ps < 1e-15:
                return np.ones(n) / n
            p = p / ps
            if len(p) == n:
                return p
            return upsample(p, n)
        return np.ones(n) / n
    elif kind == 'cos':
        freq = int(rng.choice([1, 2, 3, 4, 5]))
        x = np.arange(n, dtype=np.float64)
        amp = rng.uniform(0.5, 0.9)
        a = 1.0 + amp * np.cos(2.0 * np.pi * freq * x / n)
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            return np.ones(n) / n
        return a / s
    elif kind == 'gauss':
        x = np.arange(n, dtype=np.float64)
        sigma = rng.uniform(0.12, 0.35)
        center = 0.5 * n
        a = np.exp(-0.5 * ((x - center) / (sigma * n)) ** 2)
        return a / a.sum()
    elif kind == 'random':
        alpha = rng.uniform(0.15, 0.8)
        return rng.dirichlet(np.ones(n) * alpha)
    elif kind == 'bimodal':
        a = np.zeros(n)
        c1, c2 = rng.choice(n, 2, replace=False)
        width = max(3, n // 12)
        for c in [c1, c2]:
            a[max(0, c - width):min(n, c + width + 1)] = 1.0
        s = a.sum()
        if s < 1e-15:
            return np.ones(n) / n
        return a / s
    elif kind == 'comb':
        x = np.arange(n, dtype=np.float64)
        period = int(rng.choice([4, 6, 8, 12]))
        a = 1.0 + 0.6 * np.cos(2.0 * np.pi * x / period)
        a = np.maximum(a, 0)
        return a / a.sum()
    else:
        return np.ones(n) / n
```
