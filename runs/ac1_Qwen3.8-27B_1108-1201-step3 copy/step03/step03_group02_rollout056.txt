"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
- LP-based iterative refinement with backtracking line search
- QP-based balancing (minimize variance of convolution values)
- Spectral flattening (iterative amplitude modification in frequency domain)
- Subgradient with momentum for final polish
Key improvements over previous: adaptive step sizes, QP directions, spectral method,
and more careful convergence management.

import numpy as np
import time
from scipy.optimize import linprog, minimize as sp_minimize


def fast_conv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(a):
    """Build matrix A where (A @ b)[k] = conv(a, b)[k] for k=0..2n-2."""
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        length = j_end - j_start + 1
        if length > 0:
            A[k, j_start:j_end+1] = a[k - j_end : k - j_start + 1]
    return A


def lp_maximize_sum(a, M):
    """Solve: max sum(b) s.t. b >= 0, conv(a,b)[k] <= M for all k."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x, 0)
    return None


def lp_minimize_peak(a):
    """Solve: min M' s.t. sum(b)=1, b>=0, conv(a,b)[k] <= M'."""
    n = len(a)
    A = build_conv_matrix(a)
    num_rows = 2 * n - 1
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    A_ub = np.zeros((num_rows, n + 1))
    A_ub[:, :n] = A
    A_ub[:, n] = -1.0
    b_ub = np.zeros(num_rows)
    bounds = [(0, None)] * n + [(None, None)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        return np.maximum(res.x[:n], 0), res.x[n]
    return None, None


def qp_balance_direction(a, M_target):
    """Find b minimizing ||conv(a,b) - M_target||^2 s.t. sum(b)=1, b>=0."""
    n = len(a)
    A = build_conv_matrix(a)
    ones = np.ones(2 * n - 1) * M_target

    def obj_and_grad(b):
        r = A @ b - ones
        val = r @ r
        grad = 2.0 * A.T @ r
        return val, grad

    b0 = np.ones(n) / n
    bounds = [(0, None)] * n
    from scipy.optimize import LinearConstraint
    cons = LinearConstraint(np.ones((1, n)), 1.0, 1.0)
    res = sp_minimize(obj_and_grad, b0, jac=None, method='L-BFGS-B',
                      bounds=bounds, constraints=[cons],
                      options={'maxiter': 300, 'ftol': 1e-14})
    if res.success or res.status == 0:
        b = np.maximum(res.x, 0)
        bs = b.sum()
        if bs > 1e-10:
            return b / bs
    return None


def spectral_flatten(a, alpha=0.5, n_iters=50):
    """Iteratively flatten the power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = np.maximum(a, 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    fft_size = 1 << (2 * n - 2).bit_length()

    for _ in range(n_iters):
        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = A / (mag + 1e-30)
        power = mag ** 2
        pmax = power.max()
        if pmax < 1e-30:
            break
        # Soft threshold: reduce bins above mean by factor
        threshold = power.mean() * 1.5
        reduction = np.where(power > threshold,
                            (power > threshold).astype(float) * (1.0 - alpha * (power - threshold) / pmax),
                            1.0)
        reduction = np.clip(reduction, 0.3, 1.0)
        mag_new = mag * np.sqrt(reduction)
        a_new = np.fft.irfft(mag_new * phase, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new = a_new / sn
        a = a_new

    return a


def compute_objective(a):
    """Compute 2n * max(conv(a,a)) for sum(a)=1."""
    n = len(a)
    c = fast_conv(a)
    return 2.0 * n * c.max(), c


def optimize_lp_backtrack(a0, time_limit, method='auto', max_no_improve=100):
    """LP-based iterative refinement with backtracking line search."""
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    c = fast_conv(a)
    best_obj = 2.0 * n * c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    current_method = method
    step_sizes = [1.0, 0.8, 0.65, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.05, 0.02]

    while time.monotonic() - t0 < time_limit:
        M = c.max()

        if current_method in ('max_sum', 'auto'):
            b_raw = lp_maximize_sum(a, M)
            if b_raw is None:
                current_method = 'min_peak' if current_method == 'auto' else 'max_sum'
                continue
            bs = b_raw.sum()
            if bs < 1e-15:
                current_method = 'min_peak'
                continue
            b = b_raw / bs
        else:
            b, _ = lp_minimize_peak(a)
            if b is None:
                current_method = 'max_sum'
                continue
            bs = b.sum()
            if bs < 1e-15:
                current_method = 'max_sum'
                continue
            b = b / bs

        if np.linalg.norm(b - a) < 1e-10:
            break

        # Backtracking line search
        improved = False
        for t in step_sizes:
            a_try = (1.0 - t) * a + t * b
            a_try = np.maximum(a_try, 0)
            st = a_try.sum()
            if st < 1e-15:
                continue
            a_try = a_try / st
            c_try = fast_conv(a_try)
            obj_try = 2.0 * n * c_try.max()

            if obj_try < best_obj - 1e-12:
                a = a_try
                c = c_try
                best_obj = obj_try
                best_a = a.copy()
                no_improve = 0
                improved = True
                break

        if not improved:
            no_improve += 1
            if no_improve >= 20 and current_method == 'max_sum':
                current_method = 'min_peak'
                no_improve = 0
            elif no_improve >= 20:
                current_method = 'max_sum'
                no_improve = 0

            if no_improve >= max_no_improve:
                break

    return best_a, best_obj

def optimize_subgrad(a0, time_limit, lr_init=0.02, momentum=0.9, top_k=50,
                     temp=0.0003, perturb_every=2000, perturb_scale=0.003):
    """Projected subgradient with momentum and periodic perturbation."""
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    c = fast_conv(a)
    best_obj = 2.0 * n * c.max()
    best_a = a.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    rng = np.random.default_rng(777)
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Subgradient: find top-k peaks, weight them, compute subgradient
        if top_k >= len(c):
            peaks = np.arange(len(c))
            peak_vals = c.copy()
        else:
            peaks = np.argpartition(c, -top_k)[-top_k:]
            peak_vals = c[peaks]

        # Soft weighting
        scaled = peak_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        w = w / w_sum

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]

        v = momentum * v - lr * sg
        a_new = project_simplex(a + v)
        c = fast_conv(a_new)
        cur_obj = 2.0 * n * c.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        lr *= 0.9998

        if no_improve >= perturb_every:
            noise = rng.standard_normal(n) * perturb_scale
            a = project_simplex(a + noise)
            c = fast_conv(a)
            v *= 0.3
            no_improve = 0
            perturb_every = int(perturb_every * 1.5)
            lr = max(lr_init * 0.2, lr * 0.5)

        if no_improve > 10000:
            break

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    if parent is not None and len(parent) > 0:
        if len(parent) >= n:
            pp = np.maximum(parent[:n], 0)
        else:
            pp = np.interp(np.linspace(0, 1, n),
                           np.linspace(0, 1, len(parent)), parent)
            pp = np.maximum(pp, 0)
        ps = pp.sum()
        if ps > 1e-10:
            inits['parent'] = pp / ps

    for freq in [1, 2, 3, 5]:
        a = 1.0 + 0.9 * np.cos(2.0 * np.pi * freq * x / n)
        a = np.maximum(a, 0)
        a /= a.sum()
        inits[f'cos_{freq}'] = a

    # Two-lobe
    a = 1.0 + 0.7 * np.cos(4.0 * np.pi * x / n)
    a = np.maximum(a, 0)
    a /= a.sum()
    inits['two_lobe'] = a

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.2 * n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g

    # Random (Dirichlet with concentration)
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    inits['random2'] = rng.dirichlet(np.ones(n) * 2.0)

    # Sinusoidal with different phase
    a = 1.0 + 0.8 * np.sin(2.0 * np.pi * 2.0 * x / n + 0.5)
    a = np.maximum(a, 0)
    a /= a.sum()
    inits['sin_phase'] = a

    # Triangular
    tri = 1.0 - 2.0 * np.abs(x / n - 0.5)
    tri /= tri.sum()
    inits['triangular'] = tri

    return inits

    # === Phase 6: Final LP polish (~100s) ===
    print("\n=== Phase 6: Final LP Polish ===")
    t6_start = time.monotonic()

    if best_sequence is not None:
        t_alloc = max(5.0, deadline - time.monotonic() - 10)
        if t_alloc > 10:
            print(f"  Final LP: t={t_alloc:.0f}s")
            a_opt, _ = optimize_lp_backtrack(best_sequence, time_limit=t_alloc,
                                             method='auto', max_no_improve=300)
            try_update(a_opt)

    print(f"Phase 6 done: best={best_score:.6f}, elapsed={time.monotonic()-t6_start:.0f}s")

    # Final fallback
    if best_sequence is None:
        best_sequence = np.ones(n) / n
        try_update(best_sequence)

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = min(best_score, final_score)

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
