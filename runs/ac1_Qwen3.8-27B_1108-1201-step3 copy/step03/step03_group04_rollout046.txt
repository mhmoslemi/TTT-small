"""

Combines smoothed gradient descent with subgradient polishing, coordinate

import numpy as np
import time

def fast_convolve(a, b=None):
    """Linear convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def compute_obj(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def compute_grad_conv(a, k):

def smooth_grad(a, conv, top_k=10, temperature=0.01):
    """Compute smoothed gradient using softmax over top peaks."""
    n = len(a)
    # Get top k peaks
    k = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k)[-k:]
    peak_vals = conv[peaks]
    # Softmax weights
    scaled = peak_vals / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    # Weighted gradient
    g = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-10:
            continue
        k_val = int(peaks[idx])
        gk = np.zeros(n)
        for j in range(n):
            kj = k_val - j
            if 0 <= kj < n:
                if j == kj:
                    gk[j] = 2.0 * a[j]
                else:
                    gk[j] = 2.0 * a[kj]
        g += w[idx] * gk
    return g


def project_simplex(a):
    """Project onto {x >= 0, sum(x) = 1}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def nesterov_smoothed_descent(a0, time_limit, top_k=10, temp_init=0.02, 
                                temp_decay=0.999, lr_init=0.01, momentum=0.9):

    lr = lr_init
    temp = temp_init

    it = 0
    stall = 0
    best_stall = 0
    
    while time.monotonic() - t0 < time_limit:

        # Nesterov update
        v = momentum * v - lr * g

        # Check objective

def coordinate_descent(a0, time_limit, n_dir=8):
    """Coordinate descent: try adjusting individual elements to reduce peak."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    a = a / a.sum()
    
    conv = fast_convolve(a)
    best_obj = compute_obj(a)
    best_a = a.copy()
    peak_val = conv.max()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Find current peak location
        k_peak = int(np.argmax(conv))
        
        # Try modifying elements that contribute to the peak
        # Elements contributing to conv[k_peak] are a[j] where a[k_peak-j] is also large
        contributors = []
        for j in range(n):
            kj = k_peak - j
            if 0 <= kj < n:
                weight = a[j] * a[kj]
                if weight > 0:
                    contributors.append((j, weight))
        
        # Sort by contribution (largest first), take top n_dir*3
        contributors.sort(key=lambda x: -x[1])
        candidates = [c[0] for c in contributors[:n_dir * 3]]
        
        improved = False
        for j in candidates[:n_dir]:
            # Try reducing a[j] by various amounts
            base_val = a[j]
            if base_val < 1e-10:
                continue
            for frac in [0.3, 0.5, 0.7, 0.9]:
                a[j] = base_val * (1.0 - frac)
                # Redistribute to maintain sum=1
                s = a.sum()
                if s < 1e-15:
                    a[j] = base_val
                    continue
                a = a / s
                conv = fast_convolve(a)
                obj = compute_obj(a)
                if obj < best_obj - 1e-14:
                    best_obj = obj
                    best_a = a.copy()
                    peak_val = conv.max()
                    improved = True
                    break
            if improved:
                break
            # Reset
            a[j] = base_val
            a = a / a.sum()
        
        # Also try increasing small elements (to spread mass)
        if not improved:
            small_idx = np.argsort(a)[:n_dir]
            for j in small_idx:
                base_val = a[j]
                for frac in [0.01, 0.05, 0.1]:
                    a[j] = base_val + frac / n
                    s = a.sum()
                    a = a / s
                    conv = fast_convolve(a)
                    obj = compute_obj(a)
                    if obj < best_obj - 1e-14:
                        best_obj = obj
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
                a[j] = base_val
                a = a / a.sum()
        
        if not improved:
            no_improve += 1
            if no_improve > 50:
                break
        else:
            no_improve = 0
        
        it += 1
        if it % 100 == 0:
            elapsed = time.monotonic() - t0

def build_conv_matrix_sparse(a):
    """Build sparse matrix for convolve(a, b) constraints."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def lp_direction(a, M, max_time=3.0):
    """Solve LP: max sum(b) s.t. conv(a,b) <= M, b >= 0."""
    n = len(a)
    A = build_conv_matrix_sparse(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, method='highs', bounds=bounds,
                      options={'presolve': True, 'time_limit': max_time})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lp_frank_wolfe(a0, time_limit, max_iter=200):
    """LP-based Frank-Wolfe with line search."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0.0)
    a = a / a.sum()
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = compute_obj(a)
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    stall = 0
    
    while time.monotonic() - t0 < time_limit and it < max_iter:
        b = lp_direction(a, M, max_time=2.0)
        if b is None or b.sum() < 1e-15:
            break
        b = b / b.sum()
        
        if np.allclose(a, b, atol=1e-12):
            stall += 1
            if stall > 20:
                break
        else:
            # Line search
            best_t = 0.0
            best_local = compute_obj(a)
            for t in [0.1, 0.3, 0.5, 0.7, 0.9]:
                cand = (1 - t) * a + t * b
                cand = np.maximum(cand, 0)
                s = cand.sum()
                if s < 1e-15:
                    continue
                cand = cand / s
                obj = compute_obj(cand)
                if obj < best_local:
                    best_local = obj
                    best_t = t
            
            if best_t > 0:
                a = (1 - best_t) * a + best_t * b
                a = np.maximum(a, 0)
                a = a / a.sum()
                conv = fast_convolve(a)
                M = conv.max()
                cur_obj = compute_obj(a)
                if cur_obj < best_obj - 1e-14:
                    best_obj = cur_obj
                    best_a = a.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                stall += 1
        
        if stall > 30:
            break
        it += 1
    
    return best_a, best_obj


def gen_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    inits['uniform'] = np.ones(n) / n
    
    # Edge concentrated
    edge_w = max(1, n // 4)
    ec = np.zeros(n)
    ec[:edge_w] = 1.0
    ec /= ec.sum()
    inits['left_edge'] = ec.copy()
    
    ec2 = np.zeros(n)
    ec2[-edge_w:] = 1.0
    ec2 /= ec2.sum()
    inits['right_edge'] = ec2.copy()
    
    # Linear ramp
    r = np.arange(1, n + 1, dtype=np.float64)
    r /= r.sum()
    inits['ramp'] = r.copy()
    
    # Quadratic decay from left
    x = np.arange(n, dtype=np.float64) / n
    qd = np.exp(-3.0 * x)
    qd /= qd.sum()
    inits['exp_decay'] = qd.copy()
    
    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5) / 0.25) ** 2)
    g /= g.sum()
    inits['gauss'] = g.copy()
    
    # W-shape (two peaks at edges)
    w = np.exp(-0.5 * ((x - 0.1) / 0.15) ** 2) + np.exp(-0.5 * ((x - 0.9) / 0.15) ** 2)
    w /= w.sum()
    inits['w_shape'] = w.copy()
    
    # Parent interpolation
    if parent is not None and len(parent) > 0:
        pp = np.maximum(parent, 0.0)
        ps = pp.sum()
        if ps > 1e-10:
            if len(parent) == n:
                inits['parent'] = pp / ps
            else:
                pp_interp = np.interp(np.linspace(0, 1, n),
                                       np.linspace(0, 1, len(pp)), pp)
                ps2 = pp_interp.sum()
                if ps2 > 1e-10:
                    inits['parent_interp'] = pp_interp / ps2
    
    return inits

        if t_cd > 5:
            a_opt, obj_opt = coordinate_descent(best_sequence, time_limit=t_cd, n_dir=6)
            score = update_best(a_opt, "CD")
            print(f"    local_obj={obj_opt:.8f}, score={score:.8f}")
    
    # === Phase 3: More smoothed gradient with lower temperature ===
    if best_sequence is not None and time.monotonic() < deadline - 60:
        print("=== Phase 3: Fine-tuned NSD ===")
        remaining = deadline - time.monotonic() - 40
        t_fine = min(remaining * 0.5, 180.0)
        if t_fine > 5:
            a_opt, obj_opt = nesterov_smoothed_descent(
                best_sequence, time_limit=t_fine, top_k=5, temp_init=0.005,
                temp_decay=0.9998, lr_init=0.005, momentum=0.94)
            score = update_best(a_opt, "Fine NSD")
            print(f"    local_obj={obj_opt:.8f}, score={score:.8f}")
    
    # === Phase 4: LP Frank-Wolfe ===
    if best_sequence is not None and time.monotonic() < deadline - 30:
        print("=== Phase 4: LP Frank-Wolfe ===")
        remaining = deadline - time.monotonic() - 15
        t_lp = min(remaining * 0.6, 180.0)
        if t_lp > 5:
            a_opt, obj_opt = lp_frank_wolfe(best_sequence, time_limit=t_lp, max_iter=100)
            score = update_best(a_opt, "LP-FW")
            print(f"    local_obj={obj_opt:.8f}, score={score:.8f}")
    
    # === Phase 5: Perturbation restarts ===
    if best_sequence is not None and time.monotonic() < deadline - 20:
        print("=== Phase 5: Perturbation Restarts ===")
        n_best = len(best_sequence)
        remaining = deadline - time.monotonic() - 10
        n_restarts = max(1, int(remaining // 20))
        n_restarts = min(n_restarts, 10)
        
        for i in range(n_restarts):
            if time.monotonic() > deadline - 8:
                break
            scales = [0.001, 0.003, 0.005, 0.01, 0.02]
            scale = scales[i % len(scales)]
            noise = rng.standard_normal(n_best) * scale
            a_pert = np.maximum(best_sequence + noise, 0.0)
            a_pert = a_pert / a_pert.sum()
            
            t_alloc = min(
                max(5.0, remaining / (n_restarts - i)),
                30.0
            )
            if t_alloc < 3:
                break
            
            # Quick NSD
            a_opt, obj_opt = nesterov_smoothed_descent(
                a_pert, time_limit=t_alloc, top_k=6, temp_init=0.01,
                temp_decay=0.999, lr_init=0.01, momentum=0.93)
            score = update_best(a_opt, f"Restart {i} scale={scale}")
            
            # If improved, do CD polish
            if score < best_score + 1e-6 and time.monotonic() < deadline - 10:
                a_opt2, obj_opt2 = coordinate_descent(a_opt, time_limit=10, n_dir=4)
                update_best(a_opt2, f"Restart {i} CD")
    
    # === Phase 6: Final coordinate descent ===
    if best_sequence is not None and time.monotonic() < deadline - 8:
        print("=== Phase 6: Final CD ===")
        remaining = deadline - time.monotonic() - 5
        if remaining > 3:
            a_opt, obj_opt = coordinate_descent(best_sequence, time_limit=remaining, n_dir=10)
            update_best(a_opt, "Final CD")
    
    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))
    
    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    
    print(f"\nFINAL: score={best_score:.8f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
