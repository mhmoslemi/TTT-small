def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lp_direction(a, M, use_min_peak=False):
    """Find LP direction to reduce peak auto-correlation.
    Solves: max sum(b) s.t. conv(a,b)[k] <= M, b >= 0
    Or: min M' s.t. sum(b)=1, b>=0, conv(a,b)[k] <= M'
    """
    n = len(a)
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    
    if use_min_peak:
        c = np.zeros(n + 1)
        c[-1] = 1.0
        A_eq = np.ones((1, n + 1))
        b_eq = np.array([1.0])
        A_ub = np.zeros((num_rows, n + 1))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = np.zeros(num_rows)
        bounds = [(0, None)] * n + [(None, None)]
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            return np.maximum(res.x[:n], 0.0), res.x[n]
        return None, None
    else:
        c = -np.ones(n)
        b_ub = np.full(num_rows, M)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return np.maximum(res.x, 0.0), M
        return None, None

    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a))
        s = 1.0
    a = a / s
    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve_total = 0
    
    for step in step_sizes:
        if time.monotonic() - t0 > time_limit:
            break
        t_sub = time_limit - (time.monotonic() - t0)
        if t_sub < 10:
            break
        no_improve = 0
        
        for it in range(max_iters_per_config):
            if time.monotonic() - t0 > time_limit:
                break
            if no_improve > 80:
                break
            
            M = conv.max()
            b, _ = lp_direction(a, M, use_min_peak=False)
            if b is None:
                break
            b_sum = b.sum()
            if b_sum < 1e-12:
                break
            b = b / b_sum
            
            if np.linalg.norm(b - a) < 1e-10:
                break
            
            a_new = (1.0 - step) * a + step * b
            a_new = np.maximum(a_new, 0.0)
            a_new_sum = a_new.sum()
            if a_new_sum < 1e-12:
                break
            a_new = a_new / a_new_sum
            
            conv_new = fast_convolve(a_new)
            obj_new = 2.0 * n * conv_new.max()
            
            if obj_new < best_obj - 1e-10:
                best_obj = obj_new
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1
                a_new = (1.0 - step * 0.5) * a + step * 0.5 * b
                a_new = np.maximum(a_new, 0.0)
                a_new_sum = a_new.sum()
                if a_new_sum > 1e-12:
                    a_new = a_new / a_new_sum
                    conv_new = fast_convolve(a_new)
                    obj_new = 2.0 * n * conv_new.max()
                    if obj_new < best_obj - 1e-10:
                        best_obj = obj_new
                        best_a = a_new.copy()
                        no_improve = 0
            
            a = a_new
            conv = conv_new
            step = min(0.9, step * 1.02)
        
        no_improve_total += no_improve
    
    return best_a, best_obj


def smoothed_grad_descend(a0, time_limit, t_init=20.0, t_rate=0.97, lr_init=0.01, momentum=0.9, seed=7):
    """Gradient descent on the log-sum-exp smoothed objective with simplex projection."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a))
        s = 1.0
    a = a / s
    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    v = np.zeros(n)
    t_cur = t_init
    lr = lr_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        scaled = t_cur * conv
        scaled -= scaled.max()
        exp_vals = np.exp(scaled)
        sum_exp = exp_vals.sum()
        p = exp_vals / sum_exp
        
        # Gradient of soft_max w.r.t. a
        # d(soft_max)/da_j = (2n) * sum_k p[k] * d(conv[k])/da[j]
        # d(conv[k])/da[j] = a[k-j] + a[j-(k)] ... = 2*a[k-j] for k!=2j, a[j] for k=2j... 
        # Actually: conv[k] = sum_j' a[j']*a[k-j']
        # d(conv[k])/da[j] = a[k-j] (if k-j in [0,n-1]) + a[j-(k)] (if j-k in [0,n-1] and j!=k-j)
        # = a[k-j] + a[j-k] if both valid and j != k-j
        # = a[k-j] * 2 if j != k-j, = a[j] if j = k-j (i.e., k=2j)
        
        # More simply: d(conv[k])/da[j] = a[k-j] if 0<=k-j<=n-1, plus a[j-k] if 0<=j-k<=n-1
        # But a[k-j] = a[j-k] when... no they're different indices.
        # d/d a[j] of sum_j' a[j']*a[k-j'] = a[k-j] (from j'=j) + a[j-k] (from k-j'=j, i.e., j'=k-j)
        # If j != k-j: two terms: a[k-j] + a[k-j] = 2*a[k-j]... wait no.
        # j'=j gives a[k-j]. j'=k-j gives a[k-(k-j)] = a[j]. 
        # So d(conv[k])/da[j] = a[k-j] (if valid) + a[j] (if k-j is valid, i.e., 0<=k-j<=n-1)
        # Wait that's not right either. Let me be very careful.
        
        # conv[k] = sum_{j'=max(0,k-n+1)}^{min(n-1,k)} a[j'] * a[k-j']
        # d(conv[k])/da[j] : which terms in the sum involve a[j]?
        # Term j'=j: a[j]*a[k-j] → derivative w.r.t. a[j] is a[k-j] (if j in valid range)
        # Term j'=k-j: a[k-j]*a[j] → derivative w.r.t. a[j] is a[k-j] (if k-j in valid range)
        # If j = k-j (i.e., k=2j): only one term a[j]*a[j], derivative is 2*a[j]
        # If j ≠ k-j: derivative is a[k-j] + a[k-j] = 2*a[k-j] (if both j and k-j are in valid range)
        # If only one is in valid range: derivative is a[k-j] or a[j]
        
        # For our purpose, let's compute gradient as:
        # grad_a[j] = 2n * sum_k p[k] * d(conv[k])/da[j]
        # = 2n * sum_k p[k] * (a[k-j] * I[0<=k-j<=n-1] + a[k-j] * I[0<=k-j<=n-1] - a[j]*I[k=2j])
        # Hmm this is getting complicated. Let me just use:
        # grad_a[j] = 2n * (2 * sum_{k: 0<=k-j<=n-1} p[k]*a[k-j] - p[2j]*a[j] if 2j valid)
        
        # Simpler: for each j, the affected k range is [j, j+n-1] intersect [0, 2n-2]
        grad_a = np.zeros(n)
        for j in range(n):
            k_lo = j
            k_hi = min(2 * n - 2, j + n - 1)
            pk = p[k_lo:k_hi + 1]
            akj = a[k_lo - j:k_hi - j + 1]  # a[k-j] for k in [k_lo, k_hi]
            grad_a[j] = 2.0 * n * (2.0 * np.dot(pk, akj))
            # Subtract the double-count at k=2j
            if k_lo <= 2 * j <= k_hi:
                grad_a[j] -= 2.0 * n * p[2 * j] * a[j]
        
        v = momentum * v - lr * grad_a
        a = project_simplex(a + v)
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 200:
            lr *= 0.7
            no_improve = 0
        if no_improve > 1000:
            break
        if t_cur < 500 and it % 500 == 0 if 'it' in dir() else False:
            t_cur *= 1.5
        t_cur = max(t_cur * 0.999, 2.0)
    
    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Sine perturbation
    for amp in [0.1, 0.3, 0.5]:
        a = 1.0 + amp * np.sin(2.0 * np.pi * x / n)
        a = np.maximum(a, 0.0)
        ss = a.sum()
        if ss > 0:
            inits[f'sine_{amp}'] = a / ss
    
    # Cosine (dip in middle)
    a = 1.0 - 0.3 * np.cos(2.0 * np.pi * x / n)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    inits['cos_dip'] = a
    
    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.2 * n)) ** 2)
    g /= g.sum()
    inits['gauss'] = g
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    
    # Two-bump
    a = np.zeros(n)
    a[:n//3] = 1.0
    a[2*n//3:] = 1.0
    a /= a.sum()
    inits['two_bump'] = a
    
    if parent is not None:
        p = np.maximum(parent[:n] if len(parent) >= n else np.interp(
            np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent), 0)
        ps = p.sum()
        if ps > 1e-10:
            inits['parent'] = p / ps
    
    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search: coordinate descent (main), LP refinement (secondary),
    and multi-start exploration to minimize the peak auto-correlation objective.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)
    
    best_sequence = None
    best_score = float('inf')
    
    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: len={len(p)}")
    except Exception:
        pass
    
    def try_update(a):
        nonlocal best_sequence, best_score
        if a is None:
            return best_score
        a_clipped = np.clip(a, 0, 1000)
        score = float(evaluate_sequence(a_clipped.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_clipped.copy()
            print(f"  New best: {score:.8f} (n={len(a)})")
        return score
    
    n = 512
    
    # === Phase 1: Coordinate descent from parent (600s) ===
    print("=== Phase 1: CD from parent ===")
    t1_start = time.monotonic()
    
    if parent is not None:
        a_parent = np.maximum(parent[:n] if len(parent) >= n else np.interp(
            np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent), 0)
        a_parent = a_parent / a_parent.sum()
        t_alloc = min(600.0, deadline - time.monotonic() - 800)
        if t_alloc > 30:
            a_opt, obj = coordinate_descent(a_parent, t_alloc, n_grid=200, seed=42)
            try_update(a_opt)
    
    # === Phase 2: CD from diverse inits (400s) ===
    print("=== Phase 2: CD from diverse inits ===")
    t2_start = time.monotonic()
    
    inits = generate_inits(n, parent, rng)
    init_names = [nm for nm in ['sine_0.3', 'sine_0.5', 'cos_dip', 'two_bump', 'random', 'gauss', 'sine_0.1'] if nm in inits]
    
    t_per_init = max(30.0, min(120.0, (400.0) / max(1, len(init_names))))
    
    for name in init_names:
        remain = deadline - time.monotonic() - 700
        if remain < 30:
            break
        t_alloc = min(t_per_init, remain)
        if t_alloc < 15:
            break
        a0 = inits[name]
        a_opt, obj = coordinate_descent(a0, t_alloc, n_grid=200, seed=seed + hash(name) % 1000)
        try_update(a_opt)
    
    # === Phase 3: LP refinement of best (400s) ===
    print("=== Phase 3: LP refinement ===")
    t3_start = time.monotonic()
    
    if best_sequence is not None:
        t_alloc = min(400.0, deadline - time.monotonic() - 350)
        if t_alloc > 30:
            a_opt, obj = lp_refine(best_sequence, t_alloc,
                                   step_sizes=(0.3, 0.5, 0.2, 0.15, 0.4))
            try_update(a_opt)
    
    # === Phase 4: CD polish on best (500s) ===
    print("=== Phase 4: CD polish ===")
    t4_start = time.monotonic()
    
    if best_sequence is not None:
        # Fine coordinate descent
        t_alloc = min(300.0, deadline - time.monotonic() - 250)
        if t_alloc > 30:
            a_opt, obj = coordinate_descent(best_sequence, t_alloc, n_grid=300, seed=999)
            try_update(a_opt)
        
        # LP again on the polished result
        if best_sequence is not None:
            t_alloc2 = min(150.0, deadline - time.monotonic() - 100)
            if t_alloc2 > 20:
                a_opt2, _ = lp_refine(best_sequence, t_alloc2,
                                      step_sizes=(0.2, 0.3, 0.4))
                try_update(a_opt2)
    
    # === Phase 5: Final CD polish (remaining time) ===
    print("=== Phase 5: Final polish ===")
    if best_sequence is not None:
        t_alloc = max(0, deadline - time.monotonic() - 20)
        if t_alloc > 10:
            a_opt, obj = coordinate_descent(best_sequence, t_alloc, n_grid=400, seed=7777)
            try_update(a_opt)
    
    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(n) / n
        best_score = float(evaluate_sequence(best_sequence.tolist()))
    
    print(f"\nFINAL: score={best_score:.8f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
