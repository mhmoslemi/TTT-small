Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-based smoothed gradient descent

for the log-sum-exp smoothed max, enabling 100x more iterations than LP-FW alone.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def smooth_max_gradient(f, T):
    """Compute gradient of (1/T)*log(sum_k exp(T*conv[k])) w.r.t. f using FFT."""
    n = len(f)
    conv = fast_convolve(f)
    scaled = T * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)

    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

        for _ in range(batch_size):
            t += 1
            grad, conv = smooth_max_gradient(f, T)

            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v_hat = beta2 * v_hat + (1 - beta2) * grad * grad
            m_hat = m / (1 - beta1 ** t)
            v_hat_hat = v_hat / (1 - beta2 ** t)

            f_new = f + lr * m_hat / (np.sqrt(v_hat_hat) + eps)
            f_new = project_simplex(f_new)

            # Evaluate
            new_obj = fast_convolve(f_new).max()
            f = f_new
            eval_count += 1

            if new_obj < best_obj - 1e-13:
                best_obj = new_obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1

        # Temperature schedule: grow T, but reset on restart
        if t % 200 == 0 and T < T_max:
            T = min(T * 1.05, T_max)

        # Restart strategy: if stuck long, do a large perturbation
        if no_improve > 3000:
            scale = 0.005 + 0.005 * min((no_improve - 3000) // 5000, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            m = np.zeros(n)
            v_hat = np.zeros(n)
            T = T_init
            no_improve = 0
            lr = max(0.001, lr * 1.5)

        # Periodic learning rate decay
        if t % 5000 == 0:
            lr *= 0.9

        if verbose and eval_count % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Smooth descent: obj={best_obj:.10f}, T={T:.1f}, "
                  f"lr={lr:.6f}, evals={eval_count}, elapsed={elapsed:.1f}s")

    return best_f, best_obj


def lp_fw_single_step(f, M):
    """Single LP-FW step: solve LP for direction g."""
    n = len(f)
    c = -np.ones(n)
    # Build constraint matrix: (f*g)[k] <= M for k=0..2n-2
    # (f*g)[k] = sum_j f[k-j] * g[j] (where 0<=k-j<n)
    # For efficiency, use sparse structure
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        js = max(0, j - n + 1)  # not needed, just k ranges
        # (f*g)[k] for k = j to j+n-1: f[k-j]*g[j]
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if res.success:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False, max_iters=200):
    """LP-based Frank-Wolfe with line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    m_exp = 1.0
    it = 0

    while time.monotonic() - t0 < time_limit and it < max_iters:
        it += 1

        g = lp_fw_single_step(f, M * m_exp)
        if g is None:
            m_exp *= 1.15
            if m_exp > 5.0:
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 20:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M

        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.85, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_exp = 1.0
            f = f_new

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 4.0)

        if verbose and it % 10 == 0:
            print(f"      LP-FW iter {it}: obj={best_obj:.10f}")

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent: try adjustments to each coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-10:
                continue

            cur_obj = fast_convolve(f).max()
            best_j_val = old_val
            best_j_obj = cur_obj

            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.6, 2.0]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            break

        if verbose:
            print(f"      CD: obj={best_obj:.10f}")

    return best_f, best_obj


def pairwise_swap(f0, time_limit, rng, n_swaps_per_sec=5000):
    """Random pairwise adjustment: pick two indices, redistribute mass."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        # Batch of random swaps
        for _ in range(n_swaps_per_sec):
            if time.monotonic() - t0 > time_limit:
                break

            i = rng.integers(n)
            j = rng.integers(n)
            if i == j or f[i] < 1e-10:
                continue

            # Try moving some mass from i to j
            for frac in [0.1, 0.3, 0.5, 0.7]:
                delta = f[i] * frac
                f_trial = f.copy()
                f_trial[i] -= delta
                f_trial[j] += delta
                if f_trial[i] < 0:
                    continue
                val = fast_convolve(f_trial).max()
                if val < best_obj - 1e-13:
                    f = f_trial
                    best_obj = val
                    best_f = f.copy()
                    break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def gaussian_init(n, rng, width_frac=0.6):
    """Gaussian-like initialization centered in the sequence."""
    x = np.arange(n)
    center = n / 2.0
    width = n * width_frac / 2.0
    a = np.exp(-0.5 * ((x - center) / width) ** 2)
    a += rng.standard_normal(n) * 0.01
    a = np.maximum(a, 0)
    a /= a.sum()
    return a


def cosine_init(n):
    """Cosine-tapered window."""
    x = np.arange(n)
    a = np.cos(np.pi * x / (2 * n)) ** 2
    a = np.maximum(a, 1e-10)
    a /= a.sum()
    return a


def tent_init(n):
    """Tent (triangular) shape - high at center, low at edges."""
    x = np.arange(n)
    center = (n - 1) / 2.0
    a = 1.0 - np.abs(x - center) / (center + 0.5)
    a = np.maximum(a, 0.01)
    a /= a.sum()
    return a


def u_shaped_init(n):
    """U-shaped: high at edges, low in middle."""
    x = np.arange(n)
    center = (n - 1) / 2.0
    a = np.abs(x - center) / (center + 0.5)
    a = a - a.min() + 0.1
    a /= a.sum()
    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining FFT-based smoothed gradient descent,
    LP-FW large moves, coordinate descent, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        ss = a.sum()
        if ss < 0.01:
            return float('inf')
        a = a / ss  # normalize to sum=1 for consistent evaluation
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {score:.6f} (n={len(a)}) [{time.monotonic():.1f}]")
        return score

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}, score={float(evaluate_sequence(parent.tolist())):.6f}")
    except Exception:
        pass

    if parent is None:
        parent = cosine_init(768)

    n_parent = len(parent)
    print(f"Starting with parent n={n_parent}")

    # ============================================================
    # Phase 1: Smooth descent on parent (fast, many iterations)
    # ============================================================
    print("\n=== Phase 1: Smooth descent on parent ===")
    t1 = min(300, deadline - time.monotonic() - 600)
    if t1 > 30:
        print(f"  Time: {t1:.0f}s, n={n_parent}")
        a_opt, obj = smooth_descent(parent, time_limit=t1, T_init=3.0,
                                    T_max=150.0, lr=0.004, verbose=True, seed=seed)
        try_seq(a_opt, "smooth_parent")

    # Second smooth descent pass with different params
    t1b = min(200, deadline - time.monotonic() - 500)
    if t1b > 20:
        base = best_seq if best_seq is not None else parent
        # Perturb and re-optimize
        noise = rng.standard_normal(len(base)) * 0.02
        f_pert = project_simplex(base + noise)
        print(f"  Time: {t1b:.0f}s, n={len(base)}, perturbed")
        a_opt, obj = smooth_descent(f_pert, time_limit=t1b, T_init=5.0,
                                    T_max=100.0, lr=0.003, verbose=False,
                                    seed=seed + 1)
        try_seq(a_opt, "smooth_perturbed")

    # ============================================================
    # Phase 2: Multi-length exploration
    # ============================================================
    print(f"\n=== Phase 2: Multi-length exploration (best={best_score:.6f}) ===")
    base_for_interp = best_seq if best_seq is not None else parent

    # Try several lengths
    lengths_to_try = []
    for target_n in [512, 768, 1024, 1280, 1536]:
        if target_n != n_parent:
            lengths_to_try.append(target_n)

    # Also try the parent length with different inits
    for target_n in lengths_to_try:
        rem = deadline - time.monotonic() - 300
        if rem < 40:
            break

        # Interpolate from best
        f_interp = interpolate_sequence(base_for_interp, target_n)
        t_alloc = min(rem * 0.35, 150)

        print(f"  Smooth descent n={target_n}, t={t_alloc:.0f}s")
        a_opt, obj = smooth_descent(f_interp, time_limit=t_alloc, T_init=3.0,
                                    T_max=120.0, lr=0.004, verbose=False,
                                    seed=seed + target_n)
        try_seq(a_opt, f"smooth_n{target_n}")

        # Also try different initialization for this length
        rem = deadline - time.monotonic() - 250
        if rem > 30:
            f_cos = cosine_init(target_n)
            f_tent = tent_init(target_n)
            # Blend with interpolated
            f_blend = 0.5 * f_interp + 0.3 * f_cos + 0.2 * f_tent
            f_blend = project_simplex(f_blend)
            t_alloc2 = min(rem * 0.4, 100)
            a_opt, obj = smooth_descent(f_blend, time_limit=t_alloc2, T_init=4.0,
                                        T_max=100.0, lr=0.003, verbose=False,
                                        seed=seed + target_n + 1000)
            try_seq(a_opt, f"blend_n{target_n}")

    # ============================================================
    # Phase 3: LP-FW refinement on best
    # ============================================================
    print(f"\n=== Phase 3: LP-FW refinement (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 200
        if rem > 30:
            t_lpfw = min(rem * 0.5, 180)
            print(f"  LP-FW n={n_best}, t={t_lpfw:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_lpfw,
                                        verbose=True, max_iters=200)
            try_seq(a_opt, "lpfw_refine")

            # LP-FW from perturbed start
            rem = deadline - time.monotonic() - 150
            if rem > 20:
                noise = rng.standard_normal(n_best) * 0.01
                f_pert = project_simplex(best_seq + noise)
                t_lpfw2 = min(rem * 0.5, 100)
                print(f"  LP-FW perturbed, t={t_lpfw2:.0f}s")
                a_opt, obj = lp_fw_optimize(f_pert, time_limit=t_lpfw2,
                                            verbose=False, max_iters=100)
                try_seq(a_opt, "lpfw_perturb")

    # ============================================================
    # Phase 4: Coordinate descent + pairwise swaps
    # ============================================================
    print(f"\n=== Phase 4: Coordinate descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_cd = min(rem * 0.5, 150)
            print(f"  CD n={n_best}, t={t_cd:.0f}s")
            a_opt, obj = coordinate_descent(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd")

            # Pairwise swaps
            rem = deadline - time.monotonic() - 60
            if rem > 15:
                rng_swap = np.random.default_rng(seed + 999)
                t_swap = min(rem * 0.6, 80)
                print(f"  Pairwise swaps, t={t_swap:.0f}s")
                a_opt, obj = pairwise_swap(best_seq, time_limit=t_swap,
                                           rng=rng_swap)
                try_seq(a_opt, "swap")

    # ============================================================
    # Phase 5: Final intensive polish
    # ============================================================
    print(f"\n=== Phase 5: Final polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Multiple short smooth descent passes with different T
        for restart in range(4):
            rem = deadline - time.monotonic() - 30
            if rem < 15:
                break
            t_polish = min(rem * 0.3, 60)

            if restart == 0:
                f_init = best_seq.copy()
                T_init = 50.0  # High T: focus on peak
                lr = 0.002
            elif restart == 1:
                f_init = best_seq.copy()
                T_init = 2.0  # Low T: explore
                lr = 0.005
            elif restart == 2:
                noise = rng.standard_normal(n_best) * 0.008
                f_init = project_simplex(best_seq + noise)
                T_init = 10.0
                lr = 0.003
            else:
                # Try the reversed sequence
                f_init = best_seq[::-1].copy()
                f_init = project_simplex(f_init)
                T_init = 10.0
                lr = 0.003

            a_opt, obj = smooth_descent(f_init, time_limit=t_polish,
                                        T_init=T_init, T_max=100.0,
                                        lr=lr, verbose=False,
                                        seed=seed + 10000 + restart)
            try_seq(a_opt, f"polish_{restart}")

        # Final tiny CD pass
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            a_opt, obj = coordinate_descent(best_seq, time_limit=min(rem, 40))
            try_seq(a_opt, "final_cd")

        # One more LP-FW if time allows
        rem = deadline - time.monotonic() - 10
        if rem > 15 and n_best <= 1024:
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=min(rem, 30),
                                        max_iters=30)
            try_seq(a_opt, "final_lpfw")

    # Fallback
    if best_seq is None:
        best_seq = cosine_init(768)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = cosine_init(384)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")
    return [float(x) for x in best_seq]
