"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses multiplicative weights (mirror descent) for fast bulk optimization,
LP-based Frank-Wolfe for refinement, and coordinate descent for polishing.
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def mul_weights_optimize(f0, time_limit, eta_init=0.5, decay=0.9999, top_k=20):
    """Multiplicative weights / mirror descent for the auto-convolution minimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    worst = conv.max()
    best_f = f.copy()
    best_obj = worst
    
    eta = eta_init
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Find top-k peaks
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        
        # Weight by softmax
        scaled = pv / (eta * 0.01)
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]
        
        # Multiplicative update
        f = f * np.exp(-eta * sg)
        fs = f.sum()
        if fs < 1e-15:
            break
        f = f / fs
        
        conv = fast_convolve(f)
        co = conv.max()
        
        if co < best_obj - 1e-12:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        eta *= decay
        
        if no_improve > 200:
            # Restart with perturbation
            rng = np.random.default_rng(it * 17 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            eta = max(0.05, eta_init * 0.5)
            no_improve = 0
    
    return best_f, best_obj

def solve_lp_direction_sparse(f, M, n):

    c = -np.ones(n)

    cols = []
    vals = []

        j_start = max(0, i - n + 1)

            rows.append(i)
            cols.append(j)
            vals.append(f[j])

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

    return None

        best_alpha = 0.0
        best_val = M

        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = float(fast_convolve(ft).max())
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if 0.0 < best_alpha < 1.0:
            lo = max(0.0, best_alpha - 0.12)
            hi = min(1.0, best_alpha + 0.12)
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            for _ in range(14):
                m1 = hi - gr * (hi - lo)
                m2 = lo + gr * (hi - lo)
                if hi - lo < 1e-6:
                    break
                f1 = f + m1 * (g - f)
                f1 = np.maximum(f1, 0.0)
                f1 = f1 / f1.sum()
                f2 = f + m2 * (g - f)
                f2 = np.maximum(f2, 0.0)
                f2 = f2 / f2.sum()
                v1 = float(fast_convolve(f1).max())
                v2 = float(fast_convolve(f2).max())
                if v1 < v2:
                    hi = m2
                else:
                    lo = m1
                mid = (lo + hi) / 2.0
                fm = f + mid * (g - f)
                fm = np.maximum(fm, 0.0)
                fm = fm / fm.sum()
                vm = float(fast_convolve(fm).max())
                if vm < best_val:
                    best_val = vm
                    best_alpha = mid

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = float(conv.max())
            m_exp = 1.0
            f = f_new

            if M < best_obj - 1e-13:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 20 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.08, 3.5)

    return best_f, best_obj


def mul_weights_optimize(f0, time_limit, eta_init=0.4, top_k=40):
    """Multiplicative weights / mirror descent with softmax-weighted peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    worst = float(conv.max())
    best_f = f.copy()
    best_obj = worst

    eta = eta_init
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]

        temp = 0.003 + 0.007 * (it % 100) / 100.0
        scaled = (pv - pv.max()) / temp
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        f = f * np.exp(-eta * sg)
        fs = f.sum()
        if fs < 1e-15:
            break
        f = f / fs

        conv = fast_convolve(f)
        co = float(conv.max())

        if co < best_obj - 1e-12:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        eta *= 0.99997

        if no_improve > 400:
            rng = np.random.default_rng(it * 13 + 5)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            conv = fast_convolve(f)
            eta = max(0.05, eta_init * 0.5)
            no_improve = 0

    return best_f, best_obj


def coordinate_descent(f0, time_limit):
    """Coordinate descent with multiplicative perturbations per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = float(fast_convolve(f).max())
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-12:
                continue

            cur_peak = float(fast_convolve(f).max())
            best_j_val = old_val
            best_j_obj = cur_peak

            multipliers = [0.4, 0.6, 0.8, 1.1, 1.3, 1.6, 2.0]
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                val = float(fast_convolve(f_trial).max())
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved and passes > 2:
            break

    return best_f, best_obj


def sg_polish(f0, time_limit, lr=0.002, mom=0.94, top_k=30, temp=0.00005):
    """Subgradient polishing with softmax-weighted peaks and momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = float(conv.max())
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp

    while time.monotonic() - t0 < time_limit:
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = float(conv.max())

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9996
        cur_temp *= 1.004

        if no_improve > 400:
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 1000000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0004)
            no_improve = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def try_and_update(a, best_state, label=""):
    """Evaluate sequence and update best state if improved.
    best_state is a list [best_array, best_score]."""
    a = np.maximum(a, 0.0)
    if len(a) == 0:
        return best_state
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_state[1]:
        best_state[1] = score
        best_state[0] = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_state


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Hybrid optimization: multiplicative weights + LP-FW + coordinate descent
    + subgradient polish, with multi-length exploration and adaptive restarts.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)

    # [best_array, best_score]
    best_state = [None, float('inf')]

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_state = [parent.copy(), p_score]
    except Exception:
        pass

    if parent is None:
        parent = np.ones(384) / 384
        p_score = float(evaluate_sequence(parent.tolist()))
        best_state = [parent.copy(), p_score]
        print(f"Default parent: n=384, score={p_score:.6f}")

    # === Phase 1: Multiplicative weights (fast, ~100s) ===
    print("\n=== Phase 1: Multi-weights on parent ===")
    rem = deadline - time.monotonic()
    t1 = min(100, rem * 0.08)
    if t1 > 15:
        print(f"  MW t={t1:.0f}s")
        a_opt, _ = mul_weights_optimize(parent, time_limit=t1, eta_init=0.5, top_k=50)
        best_state = try_and_update(a_opt, best_state, "mw_parent")

    # === Phase 2: Main LP-FW (~400s) ===
    print(f"\n=== Phase 2: LP-FW main (best={best_state[1]:.6f}) ===")
    base = best_state[0]
    rem = deadline - time.monotonic()
    t2 = min(420, rem * 0.32)
    if t2 > 40:
        print(f"  LP-FW n={len(base)}, t={t2:.0f}s")
        a_opt, obj = lp_fw_optimize(base, time_limit=t2, verbose=True)
        best_state = try_and_update(a_opt, best_state, "lpfw_main")

    # === Phase 3: Multi-weights + LP-FW on best (~150s) ===
    print(f"\n=== Phase 3: Refine best (best={best_state[1]:.6f}) ===")
    base = best_state[0]
    n_base = len(base)
    rem = deadline - time.monotonic()
    t3_mw = min(50, rem * 0.06)
    if t3_mw > 10:
        a_opt, _ = mul_weights_optimize(base, time_limit=t3_mw, eta_init=0.3, top_k=60)
        best_state = try_and_update(a_opt, best_state, "mw_refine")

    rem = deadline - time.monotonic()
    t3_lp = min(120, rem * 0.15)
    if t3_lp > 20:
        base = best_state[0]
        a_opt, _ = lp_fw_optimize(base, time_limit=t3_lp)
        best_state = try_and_update(a_opt, best_state, "lpfw_refine")

    # === Phase 4: Longer sequences ===
    print(f"\n=== Phase 4: Longer sequences (best={best_state[1]:.6f}) ===")
    base = best_state[0]
    for target_n in [1024, 1536]:
        rem = deadline - time.monotonic()
        if rem < 120:
            break
        f_interp = interpolate_sequence(base, target_n)
        t_mw = min(30, rem * 0.05)
        if t_mw > 10:
            a_opt, _ = mul_weights_optimize(f_interp, time_limit=t_mw, eta_init=0.4, top_k=50)
            best_state = try_and_update(a_opt, best_state, f"mw_n{target_n}")

        rem = deadline - time.monotonic()
        if rem < 80:
            continue
        base = best_state[0]
        f_interp = interpolate_sequence(base, target_n)
        t_lp = min(180, rem * 0.3)
        if t_lp > 25:
            print(f"  LP-FW n={target_n}, t={t_lp:.0f}s")
            a_opt, obj = lp_fw_optimize(f_interp, time_limit=t_lp, verbose=True)
            best_state = try_and_update(a_opt, best_state, f"lpfw_n{target_n}")

    # === Phase 5: LP-FW restarts ===
    print(f"\n=== Phase 5: LP-FW restarts (best={best_state[1]:.6f}) ===")
    if best_state[0] is not None:
        n_best = len(best_state[0])
        for p in range(4):
            rem = deadline - time.monotonic()
            if rem < 80:
                break
            t_alloc = min(180, rem * 0.25)

            if p == 0:
                f_init = best_state[0].copy()
            elif p == 1:
                noise = rng.standard_normal(n_best) * 0.006
                f_init = project_simplex(best_state[0] + noise)
            elif p == 2:
                f_init = 0.65 * best_state[0] + 0.35 * (np.ones(n_best) / n_best)
                f_init = project_simplex(f_init)
            else:
                # Reverse-blend
                f_init = 0.8 * best_state[0] + 0.2 * best_state[0][::-1]
                f_init = project_simplex(f_init)

            print(f"  Restart {p}, t={t_alloc:.0f}s")
            a_opt, _ = lp_fw_optimize(f_init, time_limit=t_alloc)
            best_state = try_and_update(a_opt, best_state, f"restart_{p}")

    # === Phase 6: Coordinate descent ===
    print(f"\n=== Phase 6: Coordinate Descent (best={best_state[1]:.6f}) ===")
    if best_state[0] is not None:
        rem = deadline - time.monotonic()
        if rem > 40:
            t_cd = min(100, rem * 0.3)
            print(f"  CD t={t_cd:.0f}s")
            a_opt, _ = coordinate_descent(best_state[0], time_limit=t_cd)
            best_state = try_and_update(a_opt, best_state, "cd")

    # === Phase 7: Subgradient polish ===
    print(f"\n=== Phase 7: SG Polish (best={best_state[1]:.6f}) ===")
    if best_state[0] is not None:
        n_best = len(best_state[0])
        rem = deadline - time.monotonic()
        if rem > 25:
            t_sg = min(60, rem * 0.5)
            a_opt, _ = sg_polish(best_state[0], time_limit=t_sg,
                                 lr=0.0015, mom=0.94,
                                 top_k=max(20, n_best // 6), temp=0.00004)
            best_state = try_and_update(a_opt, best_state, "sg_polish")

    # === Phase 8: Final LP-FW ===
    print(f"\n=== Phase 8: Final LP-FW (best={best_state[1]:.6f}) ===")
    if best_state[0] is not None:
        rem = deadline - time.monotonic()
        if rem > 20:
            t_final = min(60, rem * 0.9)
            a_opt, _ = lp_fw_optimize(best_state[0], time_limit=t_final)
            best_state = try_and_update(a_opt, best_state, "final_lpfw")

    # === Phase 9: Tiny perturbations ===
    if best_state[0] is not None:
        n_best = len(best_state[0])
        rem = deadline - time.monotonic()
        for _ in range(4):
            if rem < 15:
                break
            rem = deadline - time.monotonic()
            noise = rng.standard_normal(n_best) * 0.001
            f_pert = project_simplex(best_state[0] + noise)
            t_pert = min(20, rem * 0.7)
            a_opt, _ = sg_polish(f_pert, time_limit=t_pert,
                                 lr=0.001, mom=0.95,
                                 top_k=max(15, n_best // 8), temp=0.00002)
            best_state = try_and_update(a_opt, best_state, "perturb")

    # Fallback
    if best_state[0] is None:
        best_state[0] = np.ones(384) / 384
        best_state[1] = float(evaluate_sequence(best_state[0].tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_state[0].tolist()))
    if np.isfinite(fs):
        best_state[1] = fs

    print(f"\nFINAL SCORE: {best_state[1]:.6f} (n={len(best_state[0])})")
    return [float(x) for x in best_state[0]]
