def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

Now, how do we find such sequences?

The frequency-domain interpretation: conv(a,a) is the inverse DFT of |A(f)|^2 where A(f) is the (zero-padded) DFT of a. For the autocorrelation to be flat, |A(f)|^2 should be flat, i.e., |A(f)| should be constant (constant modulus).

This is the "constant modulus sequence" (CMS) problem, which is well-studied!

1. Start with an initial sequence
2. Compute its DFT
3. Adjust the magnitude to be constant (keep the phase)
4. Inverse DFT
5. Project back to the feasible set (real, non-negative, sum = 1)
6. Repeat

This is the Greedy Frequency Selection (GFS) algorithm or the CFO (Constant false-alarm rate) algorithm, adapted for our constraints.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using spectral shaping for initialization,
LP-based Frank-Wolfe for primary optimization, simulated annealing for escaping
local minima, and coordinate descent for final refinement.
Key innovation: constant-modulus spectral projection to generate high-quality
initializations, combined with multi-length exploration and adaptive restarts.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def spectral_shaping(a0, num_iters=100, alpha_start=0.7, alpha_end=0.1, seed=0):
    """
    Iterated constant-modulus projection in frequency domain.
    Blends the magnitude of the DFT towards a constant value while preserving phase,
    then projects back to non-negative simplex.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.copy(), 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    for it in range(num_iters):
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max(1, num_iters - 1))

        A = np.fft.rfft(a, fft_size)
        mag = np.abs(A)
        phase = A / (mag + 1e-30)

        # Blend towards constant magnitude
        target = np.full_like(mag, mag.mean())
        new_mag = alpha * mag + (1.0 - alpha) * target

        # Reconstruct
        new_A = new_mag * phase
        a_new = np.fft.irfft(new_A, fft_size)[:n]

        # Add small random perturbation to escape symmetries
        if it % 10 == 5:
            a_new += rng.standard_normal(n) * 0.001 * (1.0 - alpha)

        # Project to feasible set
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-10:
            a_new = a  # Keep previous
        else:
            a_new = a_new / sn

        conv = fast_convolve(a_new)
        obj = conv.max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a_new.copy()

        a = a_new

    return best_a


def build_conv_matrix(f):
    """Build B matrix where (B*g)[k] = (f*g)[k]."""
    n = len(f)
    B = np.zeros((2 * n - 1, n))
    for j in range(n):
        B[j:j + n, j] = f
    return B


def solve_linearized_lp(f, conv_f, B=None):
    """Solve linearized LP for Frank-Wolfe direction."""
    n = len(f)
    if B is None:
        B = build_conv_matrix(f)

    c = np.zeros(n + 1)
    c[0] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = 2.0 * B
    b_ub = -conv_f

    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)

    bounds = [(None, None)] + [(-f[i], None) for i in range(n)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[1:]
    return None


def solve_reduced_lp(f, conv_f, top_k=20):
    """Reduced LP using only top-k peak constraints."""
    n = len(f)
    peaks = np.argpartition(conv_f, -top_k)[-top_k:]
    B = build_conv_matrix(f)

    c = np.zeros(n + 1)
    c[0] = 1.0

    A_ub = np.zeros((top_k, n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = 2.0 * B[peaks, :]
    b_ub = -conv_f[peaks]

    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)

    bounds = [(None, None)] + [(-f[i], None) for i in range(n)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[1:]
    return None


def solve_maxsum_lp(f, M):
    """Max-sum LP oracle."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None and np.any(res.x > 0):
        return res.x
    return None


def frank_wolfe_optimize(f0, time_limit, mode="linearized", reduced_k=20, verbose=False):
    """Frank-Wolfe with LP oracle and line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(999)
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        current_obj = conv.max()

        if mode == "linearized":
            g = solve_reduced_lp(f, conv, top_k=reduced_k)
        elif mode == "maxsum":
            g_raw = solve_maxsum_lp(f, current_obj * m_exp)
            if g_raw is None:
                m_exp = min(m_exp * 1.1, 5.0)
                no_improve += 1
                if no_improve > 50:
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
                    m_exp = 1.0
                    no_improve = 0
                continue
            gs = g_raw.sum()
            if gs < 1e-15:
                break
            g = g_raw / gs - f  # Convert to direction form
            m_exp = 1.0
        else:
            break

        if g is None:
            no_improve += 3
            if no_improve > 80:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                no_improve = 0
            continue

        target = f + g
        target = np.maximum(target, 0.0)
        ts = target.sum()
        if ts < 1e-15:
            no_improve += 1
            continue
        target = target / ts

        if np.max(np.abs(target - f)) < 1e-10:
            no_improve += 1
            if no_improve > 40:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                no_improve = 0
            continue

        # Line search
        best_alpha = 0.0
        best_val = current_obj
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * target
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.08)
            hi = min(1.0, best_alpha + 0.08)
            for _ in range(6):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * target
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-8:
            f = (1.0 - best_alpha) * f + best_alpha * target
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            if best_val < best_obj - 1e-14:
                best_obj = best_val
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 200 == 0:
                    print(f"    FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 60:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            no_improve = 0

    return best_f, best_obj


def subgradient_optimize(f0, time_limit, lr=0.003, mom=0.93, top_k=30, temp=0.0001):
    """Subgradient descent with softmax-weighted multi-peak targeting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9998
        cur_temp *= 1.005

        if no_improve > 300:
            rng = np.random.default_rng((it * 31 + 17) % 100000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            cur_lr = max(lr * 0.6, 0.0005)
            no_improve = 0

    return best_f, best_obj


def simulated_annealing(f0, time_limit, rng, t_start=0.01, t_end=0.0001,
                        accept_prob_temp=50.0):
    """Simulated annealing with structured moves."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    cur_obj = best_obj

    t0 = time.monotonic()
    total_time = time_limit
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / total_time)
        temperature = t_start * (t_end / t_start) ** progress

        # Choose a move type
        move_type = rng.random()

        if move_type < 0.4:
            # Gaussian perturbation
            scale = temperature * 0.5
            f_new = project_simplex(f + rng.standard_normal(n) * scale)
        elif move_type < 0.7:
            # Swap two coordinates
            i, j = rng.integers(0, n, 2)
            f_new = f.copy()
            delta = (f_new[i] - f_new[j]) * rng.uniform(0.1, 0.5)
            f_new[i] -= delta
            f_new[j] += delta
            f_new = np.maximum(f_new, 0)
            f_new = f_new / f_new.sum()
        else:
            # Block move: perturb a contiguous block
            block_len = rng.integers(3, min(20, n // 4))
            start = rng.integers(0, n - block_len)
            f_new = f.copy()
            f_new[start:start + block_len] += rng.standard_normal(block_len) * temperature * 0.3
            f_new = np.maximum(f_new, 0)
            f_new = f_new / f_new.sum()

        conv_new = fast_convolve(f_new)
        new_obj = conv_new.max()
        delta_obj = new_obj - cur_obj

        # Acceptance criterion
        if delta_obj < 0 or rng.random() < np.exp(-delta_obj / (temperature * accept_prob_temp)):
            f = f_new
            cur_obj = new_obj
            if cur_obj < best_obj - 1e-14:
                best_obj = cur_obj
                best_f = f.copy()

    return best_f, best_obj


def coordinate_refinement(f0, time_limit, rng):
    """Coordinate-wise refinement: try small adjustments to individual coordinates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        peak_idx = int(np.argmax(conv))

        # Find coordinates contributing to the peak
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        active = list(range(js, je + 1))

        # Try perturbing each active coordinate
        improved = False
        rng_indices = rng.permutation(len(active))
        for idx_pos in rng_indices[:min(5, len(active))]:
            idx = active[idx_pos]
            for direction, step in [(-1, 0.002), (-1, 0.001), (1, 0.002), (1, 0.001)]:
                if f[idx] + direction * step < -1e-10:
                    continue
                f_trial = f.copy()
                f_trial[idx] += direction * step
                f_trial = np.maximum(f_trial, 0)
                f_trial = f_trial / f_trial.sum()
                c = fast_convolve(f_trial)
                if c.max() < best_obj - 1e-14:
                    best_obj = c.max()
                    best_f = f_trial.copy()
                    f = f_trial.copy()
                    improved = True
                    break

        if not improved:
            # Try a random coordinate
            idx = rng.integers(0, n)
            for step in [0.001, 0.002, -0.001, -0.002]:
                if f[idx] + step < -1e-10:
                    continue
                f_trial = f.copy()
                f_trial[idx] += step
                f_trial = np.maximum(f_trial, 0)
                f_trial = f_trial / f_trial.sum()
                c = fast_convolve(f_trial)
                if c.max() < best_obj - 1e-14:
                    best_obj = c.max()
                    best_f = f_trial.copy()
                    f = f_trial.copy()
                    improved = True
                    break

            if not improved:
                # Small random restart
                if rng.random() < 0.05:
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.002)

    return best_f, best_obj


def multi_start_spectral(n, rng, num_starts=8, iters_per_start=150):
    """Generate multiple spectral-shaped sequences and return the best."""
    best_a = None
    best_obj = float('inf')

    for i in range(num_starts):
        # Different initializations
        if i == 0:
            a_init = np.ones(n) / n
        elif i == 1:
            a_init = np.abs(rng.standard_normal(n)) + 0.01
            a_init = a_init / a_init.sum()
        elif i == 2:
            x = np.arange(n, dtype=np.float64)
            a_init = np.exp(-0.5 * ((x - n / 2) / (n * 0.3)) ** 2)
            a_init = a_init / a_init.sum()
        elif i == 3:
            x = np.arange(n, dtype=np.float64)
            a_init = np.abs(np.cos(np.pi * x / n * rng.uniform(1, 4))) + 0.01
            a_init = a_init / a_init.sum()
        else:
            a_init = np.abs(rng.standard_normal(n))
            # Apply random frequency mask
            fft_size = 1 << (2 * n - 2).bit_length()
            A = np.fft.rfft(a_init, fft_size)
            mask = rng.random(len(A)) > 0.3
            A[~mask] *= 0.1
            a_init = np.fft.irfft(A, fft_size)[:n]
            a_init = np.maximum(a_init, 0)
            a_init = a_init / (a_init.sum() + 1e-15)

        a = spectral_shaping(a_init, num_iters=iters_per_start,
                             alpha_start=0.8, alpha_end=0.1, seed=i * 7 + 1)
        obj = fast_convolve(a).max()
        if obj < best_obj:
            best_obj = obj
            best_a = a

    return best_a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search:
    Phase 1: Spectral shaping for diverse high-quality initializations
    Phase 2: LP Frank-Wolfe optimization
    Phase 3: Simulated annealing for escape
    Phase 4: Subgradient polishing
    Phase 5: Coordinate refinement
    Phase 6: Multi-length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}")
    except Exception:
        pass

    # Determine primary length
    if parent is not None:
        base_n = len(parent)
    else:
        base_n = 384

    print(f"Target: 1.5027, Starting n={base_n}")

    # === Phase 1: Spectral Shaping Initializations ===
    print("\n=== Phase 1: Spectral Shaping ===")
    t_p1 = time.monotonic()

    # Multi-start spectral shaping
    a_spec, _ = multi_start_spectral(base_n, rng, num_starts=6, iters_per_start=120)
    try_seq(a_spec, "spectral_multi")

    # More iterations on the best
    a_spec2 = spectral_shaping(a_spec, num_iters=200, alpha_start=0.6, alpha_end=0.05, seed=42)
    try_seq(a_spec2, "spectral_refined")

    # If we have parent, also shape it
    if parent is not None:
        a_pspec = spectral_shaping(parent, num_iters=100, alpha_start=0.5, alpha_end=0.1, seed=7)
        try_seq(a_pspec, "parent_spectral")

    # Evaluate parent directly
    if parent is not None:
        try_seq(parent, "parent")

    print(f"  Phase 1 best: {best_score:.8f} in {time.monotonic()-t_p1:.1f}s")

    # === Phase 2: LP Frank-Wolfe ===
    print("\n=== Phase 2: LP Frank-Wolfe ===")
    t_p2 = time.monotonic()
    remaining = deadline - time.monotonic() - 250
    n_best = len(best_seq)

    # Primary: linearized LP (reduced)
    t_fw1 = min(remaining * 0.35, 500)
    if t_fw1 > 30:
        a_opt, _ = frank_wolfe_optimize(best_seq, time_limit=t_fw1,
                                        mode="linearized", reduced_k=20, verbose=True)
        try_seq(a_opt, "fw_linearized")

    # Secondary: max-sum LP
    rem = deadline - time.monotonic() - 200
    t_fw2 = min(rem * 0.3, 300)
    if t_fw2 > 30:
        a_opt, _ = frank_wolfe_optimize(best_seq, time_limit=t_fw2,
                                        mode="maxsum", verbose=True)
        try_seq(a_opt, "fw_maxsum")

    print(f"  Phase 2 best: {best_score:.8f} in {time.monotonic()-t_p2:.1f}s")

    # === Phase 3: Simulated Annealing ===
    print("\n=== Phase 3: Simulated Annealing ===")
    t_p3 = time.monotonic()
    rem = deadline - time.monotonic() - 200
    t_sa = min(rem * 0.25, 350)
    if t_sa > 20:
        a_opt, _ = simulated_annealing(best_seq, time_limit=t_sa, rng=rng,
                                       t_start=0.008, t_end=0.00005,
                                       accept_prob_temp=30.0)
        try_seq(a_opt, "sa")

        # SA from spectral
        if best_seq is not None:
            rem = deadline - time.monotonic() - 150
            t_sa2 = min(rem * 0.2, 200)
            if t_sa2 > 15:
                a_opt2, _ = simulated_annealing(a_spec, time_limit=t_sa2, rng=rng,
                                                t_start=0.005, t_end=0.0001,
                                                accept_prob_temp=40.0)
                try_seq(a_opt2, "sa_spectral")

    print(f"  Phase 3 best: {best_score:.8f} in {time.monotonic()-t_p3:.1f}s")

    # === Phase 4: Subgradient Polish ===
    print("\n=== Phase 4: Subgradient Polish ===")
    t_p4 = time.monotonic()
    rem = deadline - time.monotonic() - 150
    t_sg = min(rem * 0.3, 300)
    if t_sg > 20:
        a_opt, _ = subgradient_optimize(best_seq, time_limit=t_sg,
                                        lr=0.002, mom=0.94, top_k=40, temp=0.00005)
        try_seq(a_opt, "sg_main")

        # Finer SG
        rem = deadline - time.monotonic() - 100
        t_sg2 = min(rem * 0.25, 150)
        if t_sg2 > 15:
            a_opt2, _ = subgradient_optimize(best_seq, time_limit=t_sg2,
                                             lr=0.0008, mom=0.96, top_k=60, temp=0.00002)
            try_seq(a_opt2, "sg_fine")

    print(f"  Phase 4 best: {best_score:.8f} in {time.monotonic()-t_p4:.1f}s")

    # === Phase 5: Coordinate Refinement ===
    print("\n=== Phase 5: Coordinate Refinement ===")
    t_p5 = time.monotonic()
    rem = deadline - time.monotonic() - 100
    t_cd = min(rem * 0.4, 250)
    if t_cd > 15:
        a_opt, _ = coordinate_refinement(best_seq, time_limit=t_cd, rng=rng)
        try_seq(a_opt, "coord")

    print(f"  Phase 5 best: {best_score:.8f} in {time.monotonic()-t_p5:.1f}s")

    # === Phase 6: Multi-length exploration ===
    print("\n=== Phase 6: Multi-length Exploration ===")
    t_p6 = time.monotonic()
    rem = deadline - time.monotonic() - 40

    # Try nearby lengths
    for n_try in [base_n + 64, base_n - 64, base_n + 128, 512, 640]:
        if n_try < 64 or n_try > 2000 or rem < 30:
            continue
        if n_try == len(best_seq):
            continue

        # Interpolate the best sequence to the new length
        px = np.interp(np.linspace(0, 1, n_try),
                       np.linspace(0, 1, len(best_seq)), best_seq)
        px = np.maximum(px, 0)
        px = px / px.sum()

        # Quick spectral shaping
        t_shape = min(rem * 0.2, 40)
        a_shaped = spectral_shaping(px, num_iters=50, alpha_start=0.5, alpha_end=0.1,
                                    seed=n_try % 100)
        try_seq(a_shaped, f"shape_n{n_try}")

        # Quick FW
        rem = deadline - time.monotonic() - 30
        t_fw = min(rem * 0.3, 60)
        if t_fw > 10:
            a_opt, _ = frank_wolfe_optimize(a_shaped, time_limit=t_fw,
                                            mode="linearized", reduced_k=15)
            try_seq(a_opt, f"fw_n{n_try}")

        rem = deadline - time.monotonic() - 25

    print(f"  Phase 6 best: {best_score:.8f} in {time.monotonic()-t_p6:.1f}s")

    # === Final: One more round of best optimizer on global best ===
    rem = deadline - time.monotonic() - 10
    if rem > 20 and best_seq is not None:
        print("\n=== Final Polish ===")
        t_final = min(rem, 60)
        a_opt, _ = frank_wolfe_optimize(best_seq, time_limit=t_final * 0.6,
                                        mode="linearized", reduced_k=25)
        try_seq(a_opt, "final_fw")

        rem = deadline - time.monotonic() - 5
        if rem > 10:
            a_opt2, _ = subgradient_optimize(best_seq, time_limit=rem,
                                             lr=0.0005, mom=0.97, top_k=80, temp=0.00001)
            try_seq(a_opt2, "final_sg")

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\n{'='*50}")
        print(f"FINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
        print(f"{'='*50}")
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.10f}")

    return [float(x) for x in best_seq]
