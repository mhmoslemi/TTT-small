import numpy as np
from scipy.optimize import linprog
from scipy.ndimage import gaussian_filter1d
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def solve_lp(a, M):
    """Solve LP: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0."""
    n = len(a)
    num_constraints = 2 * n - 1
    
    # Build constraint matrix A_ub: shape (2n-1, n)
    # A_ub[k, i] = a[k-i] if 0 <= k-i < n, else 0
    A_ub = np.zeros((num_constraints, n))
    for k in range(num_constraints):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_ub[k, j] = a[k - j]
    b_ub = np.full(num_constraints, M)
    c = np.full(n, -1.0)  # minimize -sum(b)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x, -res.fun
    return None, None

def lp_iteration(a, num_iters, t=0.5, verbose=False):
    """Run LP-based iteration."""
    a = np.maximum(a, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    n = len(a)
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    for it in range(num_iters):
        M = c.max()
        b, sum_b = solve_lp(a, M)
        if b is None:
            break
        if sum_b < 1e-10:
            break
        b = b / sum_b  # normalize to sum 1
        
        # Update
        a_new = (1 - t) * a + t * b
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        c_new = fast_convolve(a_new)
        obj_new = c_new.max()
        
        if obj_new < best_obj:
            best_obj = obj_new
            best_a = a_new.copy()
        
        a = a_new
        c = c_new
        
        if verbose and it % 100 == 0:
            print(f"    LP iter {it}: max_conv={obj_new:.10f}")
    
    return best_a, best_obj

    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1

        # Compute softmax-weighted subgradient
        if top_k < len(c_conv):
            indices = np.argpartition(c_conv, -top_k)[-top_k:]
            peek = c_conv[indices]
        else:
            indices = np.arange(len(c_conv))
            peek = c_conv.copy()

        scaled = peek / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(indices))
            w[np.argmax(peek)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for idx in range(len(indices)):
            if w[idx] < 1e-14:
                continue
            k = int(indices[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            sg[js:je + 1] += 2.0 * w[idx] * a[k - je:k - js + 1][::-1]

        sg_max = sg.max()
        if sg_max < 1e-15:
            break

        log_update = -eta * sg / sg_max
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-300)
        s_new = a_new.sum()
        if s_new < 1e-300:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        c_conv = fast_convolve(a_new)
        obj = c_conv.max()

        count += 1
        avg_a = (count * avg_a + a_new) / (count + 1)

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        eta *= eta_decay
        temp *= temp_decay

        if it % 500 == 0:
            c_avg = fast_convolve(avg_a)
            if c_avg.max() < best_obj - 1e-14:
                best_obj = c_avg.max()
                best_a = avg_a.copy()

        if no_improve > 2000:
            rng_local = np.random.default_rng(it * 17 + 3)
            a = best_a + rng_local.standard_normal(n) * 0.003
            a = np.maximum(a, 0)
            ps = a.sum()
            if ps > 1e-10:
                a = a / ps
            else:
                a = best_a.copy()
            c_conv = fast_convolve(a)
            no_improve = 0
            eta = max(eta, 0.1)
            temp = max(temp, temp0 * 0.1)
            avg_a = a.copy()
            count = 0

        if verbose and it % 1000 == 0:
            print(f"      MW iter {it}: obj={best_obj:.12f}, eta={eta:.6f}, temp={temp:.6f}")

    return best_a, best_obj


def exact_subgradient_mw(a0, time_limit, eta0=0.3, verbose=False):
    """Multiplicative weights using exact (argmax) subgradient."""
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    c_conv = fast_convolve(a)
    best_obj = c_conv.max()
    best_a = a.copy()

    eta = eta0
    deadline = time.monotonic() + time_limit
    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        it += 1
        k_star = int(np.argmax(c_conv))

        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        sg = np.zeros(n)
        sg[js:je + 1] = 2.0 * a[k_star - je:k_star - js + 1][::-1]

        sg_max = sg.max()
        if sg_max < 1e-15:
            break

        a_new = a.copy()
        a_new[js:je + 1] *= np.exp(-eta * sg[js:je + 1] / sg_max)
        a_new = np.maximum(a_new, 1e-300)
        s_new = a_new.sum()
        if s_new < 1e-300:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        c_conv = fast_convolve(a_new)
        obj = c_conv.max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        eta *= 0.99995

        if no_improve > 1000:
            eta = max(eta, 0.05)
            no_improve = 0

        if verbose and it % 2000 == 0:
            print(f"      ExactMW iter {it}: obj={best_obj:.12f}, eta={eta:.6f}")

    return best_a, best_obj


def coordinate_descent(a0, time_limit, verbose=False):
    """Coordinate descent: adjust individual coordinates to reduce max conv."""
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    n = len(a)

    c_conv = fast_convolve(a)
    best_obj = c_conv.max()
    best_a = a.copy()

    deadline = time.monotonic() + time_limit
    rng = np.random.default_rng(42)
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        improved = False

        # Try adjusting random coordinates
        n_try = max(10, n // 8)
        coords = rng.choice(n, size=n_try, replace=False)

        for i in coords:
            old_val = a[i]
            if old_val < 1e-12:
                continue

            # Try multiplicative perturbations
            for factor in [0.7, 0.85, 0.9, 0.95, 1.05, 1.1, 1.15, 1.3]:
                a[i] = old_val * factor
                if a[i] < 0:
                    a[i] = old_val
                    continue
                s2 = a.sum()
                a = a / s2
                c_conv = fast_convolve(a)
                obj = c_conv.max()
                if obj < best_obj - 1e-15:
                    best_obj = obj
                    best_a = a.copy()
                    improved = True
                    break
            if not improved:
                a = best_a.copy()

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                break

    return best_a, best_obj


def generate_inits(a0, rng, n):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Gaussian at center
    for sigma in [0.15, 0.25, 0.35, 0.5]:
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (sigma * n)) ** 2)
        inits.append(g / g.sum())

    # Exponential decay
    for beta in [0.5, 1.0, 2.0, 4.0]:
        e = np.exp(-beta * x / n)
        inits.append(e / e.sum())
        e2 = np.exp(-beta * (n - 1 - x) / n)
        inits.append(e2 / e2.sum())

    # Cosine
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.05
    inits.append(c1 / c1.sum())
    c2 = np.cos(np.pi * x / (2 * n)) ** 2 + 0.01
    inits.append(c2 / c2.sum())

    # Edge-concentrated
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0) ** 0.5
    inits.append(edge_w / edge_w.sum())

    # Random Dirichlet
    for alpha in [0.1, 0.5, 2.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits.append(r)

    # Perturbed parent
    if len(a0) == n:
        for scale in [0.005, 0.02, 0.05]:
            p = a0 + rng.standard_normal(n) * scale
            p = np.maximum(p, 0)
            ps = p.sum()
            if ps > 1e-10:
                inits.append(p / ps)

    # Smoothed random (low-pass filtered random)
    for _ in range(3):
        r = np.abs(rng.standard_normal(n))
        # Simple moving average smoothing
        kernel_size = max(3, n // 20)
        kernel = np.ones(kernel_size) / kernel_size
        r = np.convolve(r, kernel, mode='same')
        r = np.maximum(r, 0)
        rs = r.sum()
        if rs > 1e-10:
            inits.append(r / rs)

    return inits


def msequence_init(n):
    """Generate an m-sequence based initialization (good autocorrelation properties)."""
    # Use a simple LFSR to generate a ±1 sequence, then map to [0,1]
    # For n not of form 2^m - 1, we take the closest and mod out
    m = max(3, int(np.log2(n)) + 1)
    lfsr_len = (1 << m) - 1
    # Generate LFSR sequence (using a primitive polynomial)
    # Simple: use XOR feedback
    reg = [1] * m
    seq = []
    for _ in range(lfsr_len * 2):
        seq.append(1 if reg[-1] else -1)
        feedback = reg[0] ^ reg[2] ^ reg[5] ^ reg[m - 1] if m >= 6 else reg[0] ^ reg[m - 1]
        reg = reg[1:] + [int(feedback)]

    # Take first n * 2 samples and average to get n values in [0,1]
    s = np.array(seq[:n * 4], dtype=np.float64)
    # Resample to length n
    a = np.zeros(n)
    for i in range(n):
        start = int(i * len(s) / n)
        end = int((i + 1) * len(s) / n)
        if end > start:
            a[i] = 0.5 + 0.5 * s[start:end].mean()
        else:
            a[i] = 0.5

    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function combining LP-based iteration, smoothed multiplicative
    weights, exact subgradient MW, coordinate descent, and multi-start exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            return float('inf')
        a_norm = a / a.sum()
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a_norm)})", flush=True)
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={parent_score:.10f}", flush=True)
            if np.isfinite(parent_score):
                best_score = parent_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = np.ones(384) / 384
        print("No parent found, using uniform n=384", flush=True)
        best_score = float(evaluate_sequence(parent.tolist()))
        best_seq = parent.copy()

    n = len(parent)
    print(f"Starting: n={n}, best_score={best_score:.10f}", flush=True)

    # === Phase 1: LP-based iteration from parent (main workhorse) ===
    print(f"\n=== Phase 1: LP Iteration (n={n}) ===", flush=True)
    rem = deadline - time.monotonic()
    t1 = min(400, rem * 0.25)
    if t1 > 30:
        print(f"  Alloc={t1:.0f}s", flush=True)
        a_opt, obj = lp_optimize(parent, time_limit=t1, t_init=0.5, verbose=True)
        try_seq(a_opt, f"lp1_n{n}")

    # === Phase 2: Smoothed MW from best ===
    print(f"\n=== Phase 2: Smoothed MW ===", flush=True)
    print(f"  Best so far: {best_score:.10f}", flush=True)
    rem = deadline - time.monotonic()
    t2 = min(200, rem * 0.15)
    if t2 > 20 and best_seq is not None:
        print(f"  Alloc={t2:.0f}s", flush=True)
        a_opt, _ = smoothed_mw_optimize(best_seq, time_limit=t2,
                                        eta0=0.8, temp0=0.002,
                                        temp_decay=0.9995, eta_decay=0.9999,
                                        top_k_frac=0.15, verbose=True)
        try_seq(a_opt, "smw1")

    # === Phase 3: LP with different step sizes ===
    print(f"\n=== Phase 3: LP Variations ===", flush=True)
    print(f"  Best so far: {best_score:.10f}", flush=True)
    if best_seq is not None:
        for t_val in [0.2, 0.7, 0.9]:
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            t_alloc = min(150, rem * 0.1)
            if t_alloc < 15:
                break
            print(f"  LP t={t_val}, alloc={t_alloc:.0f}s", flush=True)
            a_opt, _ = lp_optimize(best_seq, time_limit=t_alloc, t_init=t_val, verbose=False)
            try_seq(a_opt, f"lp_t{t_val}")

    # === Phase 4: Exact subgradient MW ===
    print(f"\n=== Phase 4: Exact Subgradient MW ===", flush=True)
    print(f"  Best so far: {best_score:.10f}", flush=True)
    rem = deadline - time.monotonic()
    t4 = min(150, rem * 0.1)
    if t4 > 20 and best_seq is not None:
        print(f"  Alloc={t4:.0f}s", flush=True)
        a_opt, _ = exact_subgradient_mw(best_seq, time_limit=t4, eta0=0.4, verbose=True)
        try_seq(a_opt, "exactmw")

    # === Phase 5: Try different initializations at same length ===
    print(f"\n=== Phase 5: Multi-Init (n={n}) ===", flush=True)
    print(f"  Best so far: {best_score:.10f}", flush=True)
    if best_seq is not None:
        inits = generate_inits(best_seq, rng, n)
        for i, f_init in enumerate(inits[:8]):
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            t_alloc = min(40, rem / max(1, 8 - i) * 0.8)
            if t_alloc < 10:
                break
            a_opt, _ = lp_optimize(f_init, time_limit=t_alloc, t_init=0.5, verbose=False)
            try_seq(a_opt, f"init{i}_n{n}")

    # === Phase 6: Try different lengths ===
    print(f"\n=== Phase 6: Different Lengths ===", flush=True)
    print(f"  Best so far: {best_score:.10f}", flush=True)
    if best_seq is not None:
        alt_lengths = [n // 2, int(n * 1.25), int(n * 1.5), int(n * 2)]
        alt_lengths = [l for l in alt_lengths if 64 <= l <= 1024 and l != n]
        for n_new in alt_lengths:
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            t_alloc = min(60, rem * 0.12)
            if t_alloc < 10:
                break
            # Interpolate to new length
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_new)
            a_new = np.interp(x_new, x_old, best_seq)
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            print(f"  n={n_new}, alloc={t_alloc:.0f}s", flush=True)
            a_opt, _ = lp_optimize(a_new, time_limit=t_alloc, t_init=0.5, verbose=False)
            try_seq(a_opt, f"len_n{n_new}")

    # === Phase 7: Coordinate descent polish ===
    print(f"\n=== Phase 7: Coordinate Descent ===", flush=True)
    print(f"  Best so far: {best_score:.10f}", flush=True)
    rem = deadline - time.monotonic()
    t7 = min(80, rem * 0.1)
    if t7 > 10 and best_seq is not None:
        print(f"  Alloc={t7:.0f}s", flush=True)
        a_opt, _ = coordinate_descent(best_seq, time_limit=t7, verbose=True)
        try_seq(a_opt, "coord")

    # === Phase 8: Final LP polish on best ===
    print(f"\n=== Phase 8: Final LP Polish ===", flush=True)
    print(f"  Best so far: {best_score:.10f}", flush=True)
    rem = deadline - time.monotonic()
    t8 = min(100, rem * 0.2)
    if t8 > 10 and best_seq is not None:
        print(f"  Alloc={t8:.0f}s", flush=True)
        a_opt, _ = lp_optimize(best_seq, time_limit=t8, t_init=0.3, verbose=True)
        try_seq(a_opt, "lp_final")

    # === Phase 9: Smoothed MW final ===
    print(f"\n=== Phase 9: Final Smoothed MW ===", flush=True)
    print(f"  Best so far: {best_score:.10f}", flush=True)
    rem = deadline - time.monotonic()
    t9 = min(60, rem * 0.15)
    if t9 > 10 and best_seq is not None:
        print(f"  Alloc={t9:.0f}s", flush=True)
        a_opt, _ = smoothed_mw_optimize(best_seq, time_limit=t9,
                                        eta0=0.3, temp0=0.001,
                                        temp_decay=0.9998, eta_decay=0.99995,
                                        top_k_frac=0.3, verbose=False)
        try_seq(a_opt, "smw_final")

    # === Phase 10: M-sequence inspired init ===
    print(f"\n=== Phase 10: M-sequence Init ===", flush=True)
    print(f"  Best so far: {best_score:.10f}", flush=True)
    rem = deadline - time.monotonic()
    if rem > 30:
        t10 = min(60, rem * 0.15)
        a_m = msequence_init(n)
        print(f"  M-seq init, alloc={t10:.0f}s", flush=True)
        a_opt, _ = lp_optimize(a_m, time_limit=t10, t_init=0.5, verbose=False)
        try_seq(a_opt, "mseq")

    # === Final: coordinate descent one more time ===
    rem = deadline - time.monotonic()
    if rem > 10 and best_seq is not None:
        a_opt, _ = coordinate_descent(best_seq, time_limit=min(rem - 5, 30), verbose=False)
        try_seq(a_opt, "coord_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})", flush=True)
    return [float(x) for x in best_seq]
