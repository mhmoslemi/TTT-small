import numpy as np
from scipy.optimize import minimize
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lse_objective_and_grad(s, beta, n):
    """Compute LSE objective and gradient w.r.t. s (the log-parameter)."""
    a = np.exp(s - s.max())
    a = a / a.sum()
    
    # Compute convolution
    c = fast_convolve(a)
    
    # Compute softmax weights
    beta_c = beta * c
    beta_c_max = beta_c.max()
    exp_bc = np.exp(beta_c - beta_c_max)
    w = exp_bc / exp_bc.sum()
    
    # Objective
    obj = (1.0 / beta) * (np.log(exp_bc.sum()) + beta_c_max)
    
    # Compute cross-correlation: cross[j] = sum_k w[k] * a[k-j]
    # = conv(rev_a, w)[j + n - 1]
    rev_a = a[::-1]

    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    RA = np.fft.rfft(rev_a, fft_size)
    cross_full = np.fft.irfft(W * RA, fft_size)
    cross = cross_full[n-1 : n-1+n]  # length n
    
    # <w, c>
    wc = np.dot(w, c)
    
    # Gradient
    grad = 2.0 * a * (cross - wc)
    
    return obj, grad

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LSE-smoothed objective with L-BFGS optimization (softmax parameterization)
2. Direct peak-reduction post-processing
3. Multi-start with diverse initializations
4. Frequency-domain flattening for exploration

import numpy as np
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_grad_on_simplex(a, beta):
    """Compute LSE gradient w.r.t. a directly (for simplex-based methods).
    Returns (objective_value, gradient_of_length_n)
    """
    n = len(a)
    c = fast_convolve(a)
    
    # Softmax weights
    bc = beta * c
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    w = exp_bc / exp_bc.sum()
    
    obj = (1.0 / beta) * (np.log(exp_bc.sum()) + bc_max)
    
    # Gradient: grad_a[j] = 2 * sum_k w[k] * a[k-j]
    # = 2 * cross_corr(w, a)[j]
    # cross_corr[j] = sum_k w[k] * a[k-j]
    # Using FFT: conv(rev_a, w)[j+n-1]
    rev_a = a[::-1]
    total_len = n + (2*n - 1) - 1  # 3n - 2
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    RA = np.fft.rfft(rev_a, fft_size)
    cross_full = np.fft.irfft(W * RA, fft_size)
    cross = cross_full[n-1 : n-1+n]
    
    grad = 2.0 * cross
    return obj, grad


def softmax_param_objective_and_grad(s, beta, n):
    """LSE objective and gradient w.r.t. s where a = softmax(s)."""
    a = np.exp(s - s.max())
    a = a / a.sum()
    
    c = fast_convolve(a)
    
    bc = beta * c
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    w = exp_bc / exp_bc.sum()
    
    obj = (1.0 / beta) * (np.log(exp_bc.sum()) + bc_max)
    
    # dL/ds_j = 2 * a_j * (cross[j] - <w, c>)
    rev_a = a[::-1]
    total_len = 3 * n - 2
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    RA = np.fft.rfft(rev_a, fft_size)
    cross_full = np.fft.irfft(W * RA, fft_size)
    cross = cross_full[n-1 : n-1+n]
    
    wc = np.dot(w, c)
    grad = 2.0 * a * (cross - wc)
    
    return obj, grad


def lse_lbfgs_optimize(a0, time_limit, beta_start=8.0, beta_end=5000.0):
    """LSE + L-BFGS with softmax parameterization."""
    n = len(a0)
    
    a_init = np.maximum(a0, 1e-10)
    a_init = a_init / a_init.sum()
    s = np.log(a_init)
    
    c = fast_convolve(a_init)
    best_obj = c.max()
    best_a = a_init.copy()
    
    t0 = time.monotonic()
    beta = beta_start
    iteration = 0
    
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        elapsed = time.monotonic() - t0

        remaining = time_limit - elapsed
        max_iter = min(500, max(20, int(remaining * 50)))
        
        def objective(s):
            return softmax_param_objective_and_grad(s, beta, n)[0]
        
        def gradient(s):
            return softmax_param_objective_and_grad(s, beta, n)[1]
        
        try:
            result = minimize(
                fun=objective,
                x0=s.copy(),
                jac=gradient,
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'gtol': 1e-10, 'maxfun': max_iter * 3}
            )
            s = result.x
        except Exception:
            break
        
        a = np.exp(s - s.max())
        a = a / a.sum()

def peak_reduce(a0, time_limit, lr=0.001):

            best_obj = M
            best_a = a.copy()

        k = min(10, len(c))
        peak_indices = np.argpartition(c, -k)[-k:]
        peak_values = c[peak_indices]

        sg = np.zeros(n)

            js = max(0, idx - n + 1)

    """LSE gradient descent with Nesterov momentum on the simplex."""

    a = np.maximum(a0, 1e-10)
    a = a / a.sum()
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    t0 = time.monotonic()
    beta = beta_start
    lr = 2.0 / n
    momentum = 0.93
    v = np.zeros(n)
    no_improve = 0
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        obj, grad = lse_objective_and_grad_simplex(a, beta)
        v = momentum * v + grad
        a_new = project_simplex(a - lr * v)
        c_new = fast_convolve(a_new)
        obj_new = c_new.max()
        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            lr = min(lr * 1.0015, 5.0 / n)
        else:
            a = a_new
            no_improve += 1
            if no_improve >= 12:
                lr *= 0.93
                momentum = max(0.88, momentum - 0.002)
            if no_improve > 120:
                a = best_a.copy()
                v = np.zeros(n)
                lr = max(2.0 / n, 0.0005)
                beta = min(beta * 1.4, beta_end)
                momentum = 0.91
                no_improve = 0
        if beta < beta_end:
            beta *= 1.003
        if it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-Mom: it={it}, obj={best_obj:.10f}, beta={beta:.1f}, lr={lr:.6f}, t={elapsed:.0f}s")
    return best_a, best_obj


def peak_reduce(a0, time_limit, lr_init=0.002):
    """Direct peak reduction targeting top peaks via subgradient."""
    n = len(a0)
    a = np.maximum(a0, 0.0)
    a = a / a.sum()
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    t0 = time.monotonic()
    lr = lr_init
    it = 0
    stuck = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(a)
        M = c.max()
        if M < best_obj:
            best_obj = M
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
        k = min(16, len(c))
        peak_indices = np.argpartition(c, -k)[-k:]
        weights = c[peak_indices].copy()
        weights = weights / weights.sum()
        sg = np.zeros(n)
        for i in range(len(peak_indices)):
            if weights[i] < 1e-10:
                continue
            idx = int(peak_indices[i])
            js = max(0, idx - n + 1)
            je = min(n - 1, idx)
            j_range = np.arange(js, je + 1)
            sg[j_range] += 2.0 * weights[i] * a[idx - j_range]
        a_new = project_simplex(a - lr * sg)
        c_new = fast_convolve(a_new)
        if c_new.max() < M:
            a = a_new
            lr = min(lr * 1.005, 0.02)
        else:
            lr *= 0.94
            if lr < 1e-7:
                rng = np.random.default_rng(it * 31 + 7)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
                lr = lr_init * 0.5
                stuck = 0
        if stuck > 500:
            break
    return best_a, best_obj


def freq_flatten(a0, time_limit):
    """Frequency-domain flattening via iterative spectrum compression."""
    n = len(a0)
    a = np.maximum(a0, 1e-10)
    a = a / a.sum()
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    t0 = time.monotonic()
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        fft_size = 1 << (2 * n - 1).bit_length()
        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2
        c = np.fft.irfft(power, fft_size)[:2 * n - 1]
        c_max = c.max()
        if c_max < 1e-12:
            break
        alpha = 0.3
        c_compressed = c / (1.0 + alpha * c / c_max)
        power_new = np.real(np.fft.rfft(c_compressed, fft_size))
        power_new = np.maximum(power_new, 0)
        A_new = np.sqrt(power_new) * np.exp(1j * np.angle(A))
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = project_simplex(a_new)
        c_check = fast_convolve(a_new)
        if c_check.max() < best_obj:
            best_obj = c_check.max()
            best_a = a_new.copy()
        a = a_new
        if it % 100 == 0:
            print(f"    FreqFlat: it={it}, obj={best_obj:.10f}, t={time.monotonic()-t0:.0f}s")
    return best_a, best_obj


def make_raised_cosine(n):
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    a = a / a.sum()
    return a


def make_bell(n):
    x = np.linspace(0, 1, n)
    a = np.exp(-50 * (x - 0.5) ** 2)
    a = a / a.sum()
    return a


def make_bimodal(n):
    x = np.linspace(0, 1, n)
    a = np.exp(-80 * (x - 0.25) ** 2) + np.exp(-80 * (x - 0.75) ** 2)
    a = a / a.sum()
    return a


def try_update(a, best_tracker):
    a = np.maximum(a, 0.0)
    if a.sum() < 0.01:
        return
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  *** NEW BEST: {score:.8f} (n={len(a)})")
