Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using coordinate descent as the primary
local optimizer (making near-exact moves per coordinate), combined with LP-based

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_for_direction(f, M, time_limit=20):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns optimal g (unnormalized) or None."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            A_ub[k, j_start:j_end + 1] = f[k - j_start:k - j_start + (j_end - j_start + 1)]
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None


def coordinate_descent(a, time_limit, verbose=False):
    """Cyclic coordinate descent on the scale-invariant objective.
    Works with f(a) = 2*n*max(conv(a,a))/(sum(a))^2.
    Uses incremental convolution updates for speed.

    a = np.maximum(a.astype(np.float64), 0.0)

            S_others = 1.0 - old  # since sum = 1

            if S_others < 1e-15:
                continue

            k_lo = max(0, j)
            k_hi = min(2 * n - 2, j + n - 1)

            # Build the affected part
            affected_len = k_hi - k_lo + 1
            a_corr = a[k_lo - j: k_hi - j + 1]  # a[k-j] for k in [k_lo, k_hi]

            diag_pos = (2 * j - k_lo) if has_diag else -1

            if k_lo > 0 or k_hi < 2 * n - 2:
                # Unaffected is [0, k_lo-1] and [k_hi+1, 2n-2]
                unaff_parts = []
                if k_lo > 0:
                    unaff_parts.append(conv[:k_lo])
                if k_hi < 2 * n - 2:
                    unaff_parts.append(conv[k_hi + 1:])
                unaff_max = max(p.max() for p in unaff_parts) if unaff_parts else 0.0
            else:
                unaff_max = 0.0

            def g(t):
                S_new = S_others + t
                if S_new < 1e-15:
                    return 1e30
                new_aff = conv_aff + (t - old) * a_corr
                if has_diag:
                    new_aff[diag_pos] = diag_val - old * old + t * t

                    m = unaff_max
                return 2.0 * n * m / (S_new * S_new)

            lo = max(0.0, best_cand - span)
            hi = best_cand + span

            # Expand if function is still decreasing
            if g(hi) < g(lo) and hi < 1.0:
                lo = hi
                hi = min(1.0, hi * 2 + 0.01)

            gr = (np.sqrt(5) - 1) / 2
            a1 = lo + (1 - gr) * (hi - lo)
            a2 = lo + gr * (hi - lo)
            f1, f2 = g(a1), g(a2)

            n_iter = 30

                    break
                if f1 < f2:
                    hi = a2
                    a2 = a1
                    f2 = f1
                    a1 = lo + (1 - gr) * (hi - lo)
                    f1 = g(a1)
                else:
                    lo = a1
                    a1 = a2
                    f1 = f2
                    a2 = lo + gr * (hi - lo)
                    f2 = g(a2)

            t_opt = (lo + hi) / 2
            # Final comparison with 0 and current
            f0 = g(0.0)
            fopt = g(t_opt)
            fold = g(old)
            if f0 < fopt and f0 < fold:
                t_opt = 0.0
            elif fold < fopt and fold < f0:
                t_opt = old

                a[j] = max(0.0, t_opt)

def generate_inits(n, parent=None, rng=None):

    r = rng.dirichlet(np.ones(n) * 0.2)
    inits.append(r)

    r2 = rng.dirichlet(np.ones(n) * 5.0)

        for p in range(5):
            rem = deadline - time.monotonic() - 50
            if rem < 20:
                break
            t_alloc = min(rem // (5 - p), 100)

            if p == 0:
                f_init = a_best
            elif p == 1:
                f_init = project_simplex(a_best + rng.standard_normal(n_best) * 0.005)
            elif p == 2:
                f_init = project_simplex(0.9 * a_best + 0.1 * (a_best[::-1]))
            elif p == 3:
                f_init = project_simplex(a_best + rng.standard_normal(n_best) * 0.02)
            else:
                # Mix with a different structured init
                x = np.arange(n_best, dtype=np.float64)
                alt = np.exp(-((x - n_best/2) / (n_best/5))**2) + 0.01
                alt = alt / alt.sum()
                f_init = project_simplex(0.8 * a_best + 0.2 * alt)

            a_opt, _ = coordinate_descent(f_init, time_limit=t_alloc)
            try_seq(a_opt, f"deep_cd_{p}")

        # FW-LP deep runs
        for p in range(3):
            rem = deadline - time.monotonic() - 30
            if rem < 25:
                break
            t_alloc = min(rem // (3 - p), 80)

            if p == 0:
                f_init = a_best
            elif p == 1:
                f_init = project_simplex(a_best + rng.standard_normal(n_best) * 0.003)
            else:
                f_init = project_simplex(0.85 * a_best + 0.15 * (np.ones(n_best) / n_best))

            a_opt, _ = fw_lp_optimize(f_init, time_limit=t_alloc)
            try_seq(a_opt, f"deep_fw_{p}")
            # Follow with CD
            rem_cd = deadline - time.monotonic() - 20
            if rem_cd > 15:
                a_opt2, _ = coordinate_descent(a_opt, time_limit=min(rem_cd, 40))
                try_seq(a_opt2, f"deep_fwcd_{p}")

    # === Phase 5: Final polish ===
    print(f"\n=== Phase 5: Final Polish (best={best_score:.8f}) ===")

    if best_seq is not None:
        n_best = len(best_seq)
        a_best = best_seq / best_seq.sum()

        rem = deadline - time.monotonic() - 10
        if rem > 15:
            a_opt, _ = coordinate_descent(a_best, time_limit=min(rem, 60), verbose=False)
            try_seq(a_opt, "polish1")

        rem = deadline - time.monotonic() - 5
        if rem > 10:
            # Micro perturbations
            for _ in range(3):
                rem = deadline - time.monotonic() - 5
                if rem < 5:
                    break
                f_pert = project_simplex(a_best + rng.standard_normal(n_best) * 0.001)
                a_opt, _ = coordinate_descent(f_pert, time_limit=min(rem, 15))
                try_seq(a_opt, "micro")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(256)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
