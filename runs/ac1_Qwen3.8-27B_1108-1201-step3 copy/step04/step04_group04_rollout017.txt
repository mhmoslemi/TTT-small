import numpy as np

from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = conv(a, g) (full linear convolution).
    C has shape (2n-1, n) where n = len(a).
    """
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def compute_peak_gradient(a, conv, k_peak):
    """Compute exact gradient of conv[a*a][k_peak] w.r.t. a."""
    n = len(a)
    grad = np.zeros(n)
    for j in range(n):
        idx1 = k_peak - j
        idx2 = j - k_peak
        val = 0.0
        if 0 <= idx1 < n:
            val += a[idx1]
        if 0 <= idx2 < n and idx2 != idx1:
            val += a[idx2]
        elif 0 <= idx2 < n and idx2 == idx1:
            # When k_peak - j == j - k_peak, i.e., j = k_peak, we double count
            # Actually when idx1 == idx2, it means k_peak - j = j - k_peak => j = k_peak
            # In that case the term is 2*a[j] which is already handled
            pass
        grad[j] = val
    return grad


def compute_multi_peak_gradient(a, conv, peaks, weights):
    """Weighted sum of gradients at multiple peak lags."""
    n = len(a)
    grad = np.zeros(n)
    for k, w in zip(peaks, weights):
        if w < 1e-14:
            continue
        for j in range(n):
            idx1 = k - j
            idx2 = j - k
            val = 0.0
            if 0 <= idx1 < n:
                val += a[idx1]
            if 0 <= idx2 < n:
                val += a[idx2]
            grad[j] += w * val
    return grad


def soft_gradient(a, conv, top_k=30, temp=0.005):
    """Softmax-weighted gradient over top-k peaks."""
    n = len(a)
    if top_k >= len(conv):
        top_k = len(conv)
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peaks]
    scaled = peak_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum
    return compute_multi_peak_gradient(a, conv, peaks, w)


def lp_direction(a, M, C=None):
    """Solve LP: max sum(g) s.t. C@g <= M, g >= 0. Returns (g, success)."""
    n = len(a)
    if C is None:
        C = build_conv_matrix(a)
    c_vec = -np.ones(n)
    b_vec = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        result = linprog(c_vec, A_ub=C, b_ub=b_vec, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 30})
        if result.success and result.x is not None:
            g = np.maximum(result.x, 0)
            if g.sum() > 1e-10:
                return g, True
    except Exception:
        pass
    return None, False


def lp_step(a, conv, n, C_cache=None):
    """One LP-based step with binary-search step sizing. Returns (a_new, improved, C_cache)."""
    M = conv.max()
    if C_cache is None:
        C_cache = build_conv_matrix(a)
    else:
        # Update C_cache since a changed
        C_cache = build_conv_matrix(a)

    g, success = lp_direction(a, M * 1.00001, C_cache)

    if not success:
        return a.copy(), False, C_cache

    g_sum = g.sum()
    if g_sum <= 1.0 + 1e-6:
        return a.copy(), False, C_cache

    d = g - a
    norm_d = np.linalg.norm(d)
    if norm_d < 1e-14:
        return a.copy(), False, C_cache

    # Binary search for best step size
    best_a = a.copy()
    best_obj = 2.0 * n * M

    # First check t=1
    a_try = a + d
    a_try = np.maximum(a_try, 0.0)
    s_try = a_try.sum()
    if s_try > 1e-15:
        a_try_n = a_try / s_try
        conv_try = fast_convolve(a_try_n)
        obj_try = 2.0 * n * conv_try.max()
        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try_n.copy()
            # Try to go even further? d was g-a, so t=1 gives g (normalized)
            # We can't go beyond t=1 in this direction
            return best_a, True, C_cache

    # Binary search in [0, 1]
    lo, hi = 0.0, 1.0
    for _ in range(8):
        mid = (lo + hi) / 2
        a_try = a + mid * d
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            hi = mid
            continue
        a_try_n = a_try / s_try
        conv_try = fast_convolve(a_try_n)
        obj_try = 2.0 * n * conv_try.max()
        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try_n.copy()
            lo = mid
        else:
            hi = mid

    if np.linalg.norm(best_a - a) > 1e-14:
        return best_a, True, C_cache
    return a.copy(), False, C_cache


def lp_optimize(a0, time_limit, verbose=False):
    """LP-based iterative optimization with restarts and multi-M directions."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
        s = 1.0
    a = a / s
    n = len(a)

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    perturb_count = 0
    max_no_improve = 50
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        if elapsed > time_limit - 2:
            break

        # Try LP step with current M
        a_new, improved, _ = lp_step(a, conv, n)

        if improved:
            a = a_new
            conv = fast_convolve(a)
            M = conv.max()
            cur_obj = 2.0 * n * M
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()
            no_improve = 0
        else:
            # Try with slightly different M values for different directions
            found = False
            for m_scale in [1.001, 0.999, 1.002, 0.998]:
                C = build_conv_matrix(a)
                M_try = M * m_scale
                g, success = lp_direction(a, M_try, C)
                if not success:
                    continue
                d = g - a
                if np.linalg.norm(d) < 1e-14:
                    continue
                # Try a few step sizes
                for t in [0.3, 0.15, 0.07, 0.03]:
                    a_try = np.maximum(a + t * d, 0.0)
                    s_try = a_try.sum()
                    if s_try < 1e-15:
                        continue
                    a_try = a_try / s_try
                    conv_try = fast_convolve(a_try)
                    obj_try = 2.0 * n * conv_try.max()
                    if obj_try < best_obj - 1e-12:
                        a = a_try
                        conv = conv_try
                        M = conv.max()
                        best_obj = obj_try
                        best_a = a.copy()
                        no_improve = 0
                        found = True
                        break
                if found:
                    break

            if not found:
                no_improve += 1

        iteration += 1

        if verbose and iteration % 200 == 0:
            print(f"    LP iter {iteration}, obj={best_obj:.8f}, t={time.monotonic()-t0:.1f}s")

        # Handle stagnation with diverse perturbations
        if no_improve >= max_no_improve:
            scale = 0.002 + 0.004 * min(perturb_count, 10)
            noise = rng.standard_normal(n) * scale
            a = np.maximum(best_a + noise, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
            else:
                a = a / s
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            perturb_count += 1
            max_no_improve = min(400, 50 + 30 * perturb_count)

            if perturb_count > 60:
                break

    return best_a, best_obj


def gradient_optimize(a0, time_limit, lr_init=0.01, momentum=0.9, top_k=30, temp_init=0.002):
    """Nesterov-accelerated gradient descent with adaptive parameters."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    y = a.copy()
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    mom = momentum
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        y = a + mom * v
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y

        conv_y = fast_convolve(y)
        grad = soft_gradient(y, conv_y, top_k=top_k, temp=temp)

        v = mom * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.999
        temp *= 1.004
        mom = min(0.96, mom + 0.0003)

        if no_improve >= 300:
            break

    return best_a, best_obj


def peak_specific_optimize(a0, time_limit):
    """Directly target the peak lag with exact subgradient."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:

    best_a = a.copy()

    lr = 0.02
    no_improve = 0
    v = np.zeros(n)
    mom = 0.85

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        # Find all peaks within 1% of max
        max_val = conv.max()
        peak_threshold = max_val * 0.99
        peak_indices = np.where(conv >= peak_threshold)[0]
        
        if len(peak_indices) == 0:
            peak_indices = np.array([np.argmax(conv)])

        # Compute weighted gradient
        peak_vals = conv[peak_indices]
        weights = (peak_vals - peak_threshold + max_val * 0.01)
        weights = weights / weights.sum()

        grad = compute_multi_peak_gradient(a, conv, peak_indices, weights)

        # Nesterov update
        v = mom * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()
            v = np.zeros(n)

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.998
        mom = min(0.93, mom + 0.0002)

        if no_improve >= 200:
            break

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # Various structured inits
    inits['sine1'] = (np.sin(np.pi * x / n) ** 2 + 0.01)
    inits['sine1'] /= inits['sine1'].sum()

    inits['sine2'] = (np.sin(2 * np.pi * x / n) ** 2 + 0.01)
    inits['sine2'] /= inits['sine2'].sum()

    inits['sine3'] = (np.sin(3 * np.pi * x / n) ** 2 + 0.01)
    inits['sine3'] /= inits['sine3'].sum()

    # Bimodal
    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.25 * n)) ** 2)
    inits['gaussian'] = g / g.sum()

    # Alternating
    alt = np.ones(n)
    alt[::2] = 1.3
    alt[1::2] = 0.7
    inits['alternating'] = alt / alt.sum()

    # Power law
    pw = (x + 1.0) ** 0.5
    inits['power'] = pw / pw.sum()

    # Triangular (peak in middle)
    tri = 1.0 - np.abs(x - (n - 1) / 2.0) / ((n - 1) / 2.0 + 1e-10)
    tri = np.maximum(tri, 0.01)
    inits['triangular'] = tri / tri.sum()

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps

    return inits


def extend_sequence(a_short, n_target, rng, noise_scale=0.0):
    """Extend a shorter sequence to target length via interpolation + optional noise."""
    n_short = len(a_short)
    s = a_short.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    a_norm = a_short / s
    a_ext = np.interp(np.linspace(0, 1, n_target), np.linspace(0, 1, n_short), a_norm)
    if noise_scale > 0:
        a_ext += rng.standard_normal(n_target) * noise_scale * a_ext.mean()
    a_ext = np.maximum(a_ext, 0)
    ses = a_ext.sum()
    if ses > 1e-15:
        a_ext = a_ext / ses
    return a_ext


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Improved hybrid LP + gradient + peak-specific search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: n={len(p)}, sum={p.sum():.4f}")
    except Exception:
        pass

    # =========================================================
    # Phase 1: Quick scan across lengths with gradient methods
    # =========================================================
    print("=== Phase 1: Multi-scale Gradient Scan ===")
    t_start = time.monotonic()
    phase1_limit = 250.0

    # Include parent length and some larger sizes
    lengths = set()
    if parent is not None:
        lengths.add(len(parent))
    lengths.update([512, 768, 1024, 1536, 2048])
    lengths = sorted([l for l in lengths if l <= 2048])

    scan_results = []
    n_configs = min(5, len(lengths))
    n_inits_per_length = 3
    t_per = max(5.0, (phase1_limit - 10) / (n_configs * n_inits_per_length))

    tested = 0
    for n in lengths:
        if time.monotonic() > deadline - 600 or tested >= n_configs * n_inits_per_length:
            break
        inits = generate_initializations(n, parent, rng)

        priority = []
        if parent is not None:
            if 'parent' in inits:
                priority.append('parent')
            if 'parent_interp' in inits:
                priority.append('parent_interp')
        priority.extend(['uniform', 'sine1', 'random', 'alternating', 'bimodal', 'sine2', 'gaussian'])
        
        init_names = [nm for nm in priority if nm in inits][:n_inits_per_length]
        tested += len(init_names)

        for name in init_names:
            if time.monotonic() > deadline - 500:
                break
            t_alloc = min(t_per, deadline - time.monotonic() - 200)
            if t_alloc < 3:
                break

            a0 = inits[name]
            # Use peak-specific for speed
            a_opt, obj = peak_specific_optimize(a0, time_limit=t_alloc)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  PS n={n}, {name}: {score:.6f}")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 2: LP-based deep optimization
    # =========================================================
    print("=== Phase 2: LP Deep Optimization ===")
    t_start2 = time.monotonic()

    # Build candidate list: best from scan + parent at its native length
    top_configs = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        if (n, name) not in seen:
            top_configs.append((n, a_opt.copy(), score))
            seen.add((n, name))
        if len(top_configs) >= 5:
            break

    # Always include parent
    if parent is not None and len(parent) not in [c[0] for c in top_configs]:
        pn = len(parent)
        p_norm = parent / parent.sum()
        top_configs.insert(0, (pn, p_norm.copy(), float('inf')))

    remaining = deadline - time.monotonic() - 350.0
    n_cands = max(1, min(4, len(top_configs)))
    t_per_lp = max(50.0, remaining * 0.65 / n_cands)

    for i in range(n_cands):
        if time.monotonic() > deadline - 250:
            break
        n, a0, sc = top_configs[i]
        t_alloc = min(t_per_lp, deadline - time.monotonic() - 200)
        if t_alloc < 20:
            break

        print(f"  LP cand {i}: n={n}, init_score={sc:.6f}, t={t_alloc:.0f}s")
        a_opt, obj = lp_optimize(a0, time_limit=t_alloc, verbose=True)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  LP {i} (n={n}): {score:.6f} ***NEW BEST***")
            else:
                print(f"  LP {i} (n={n}): {score:.6f}")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 3: Hierarchical extension
    # =========================================================
    print("=== Phase 3: Hierarchical Extension ===")
    t_start3 = time.monotonic()

    if best_sequence is not None:
        n_best = len(best_sequence)
        # Try extending to 2x
        n_target = min(2048, 2 * n_best)
        if n_target > n_best:
            remaining = deadline - time.monotonic() - 200.0
            if remaining > 100:
                a_ext = extend_sequence(best_sequence, n_target, rng, noise_scale=0.01)
                t_ext = min(remaining * 0.4, 150.0)
                if t_ext > 20:
                    print(f"  Extending to n={n_target}...")
                    a_opt, obj = lp_optimize(a_ext, time_limit=t_ext)
                    score = float(evaluate_sequence(a_opt.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        print(f"  Extended n={n_target}: {score:.6f} ***NEW BEST***")
                    else:
                        print(f"  Extended n={n_target}: {score:.6f}")

                # Also try gradient on extended
                remaining = deadline - time.monotonic() - 100.0
                if remaining > 30:
                    a_opt, obj = gradient_optimize(a_ext, time_limit=min(remaining * 0.5, 60.0),
                                                    lr_init=0.008, top_k=50, temp_init=0.001)
                    score = float(evaluate_sequence(a_opt.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        print(f"  Grad ext n={n_target}: {score:.6f} ***NEW BEST***")

    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 4: Intensive LP polishing with multiple restarts
    # =========================================================
    print("=== Phase 4: Intensive LP Polish ===")
    t_start4 = time.monotonic()

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 100.0
        n_polish = 4
        t_per_polish = max(10.0, remaining * 0.7 / n_polish)

        for pi in range(n_polish):
            if time.monotonic() > deadline - 60:
                break
            t_alloc = min(t_per_polish, deadline - time.monotonic() - 50)
            if t_alloc < 10:
                break

            if pi == 0:
                a0 = best_sequence.copy()
            else:
                scale = 0.001 * pi
                noise = rng.standard_normal(len(best_sequence)) * scale
                a0 = np.maximum(best_sequence + noise, 0)
                s = a0.sum()
                if s > 1e-15:
                    a0 = a0 / s
                else:
                    a0 = best_sequence.copy()

            a_opt, obj = lp_optimize(a0, time_limit=t_alloc)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Polish {pi}: {score:.6f} ***NEW BEST***")
            else:
                print(f"  Polish {pi}: {score:.6f}")

    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 5: Peak-specific fine-tuning
    # =========================================================
    print("=== Phase 5: Peak-specific Fine-tuning ===")
    t_start5 = time.monotonic()

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 30.0
        if remaining > 20:
            t_ps = min(remaining * 0.6, 80.0)
            a_opt, obj = peak_specific_optimize(best_sequence, time_limit=t_ps)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  PeakSpec: {score:.6f} ***NEW BEST***")

            # Short gradient with very low learning rate
            remaining = deadline - time.monotonic() - 20.0
            if remaining > 10:
                a_opt, obj = gradient_optimize(best_sequence, time_limit=min(remaining * 0.8, 30.0),
                                                lr_init=0.002, momentum=0.95, top_k=80, temp_init=0.0005)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  FineGrad: {score:.6f} ***NEW BEST***")

    print(f"Phase 5 done ({time.monotonic()-t_start5:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 6: Final micro-LP
    # =========================================================
    if best_sequence is not None and time.monotonic() < deadline - 10:
        t_final = min(15.0, deadline - time.monotonic() - 5)
        if t_final > 5:
            a_opt, obj = lp_optimize(best_sequence, time_limit=t_final)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final LP: {score:.6f} ***NEW BEST***")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
