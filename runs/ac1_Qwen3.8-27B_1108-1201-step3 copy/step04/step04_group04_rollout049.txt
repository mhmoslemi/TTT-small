import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_sparse_conv_matrix(a):
    """Build sparse matrix C such that C @ g = full_convolve(a, g)."""
    n = len(a)
    num_rows = 2 * n - 1
    # For row k, the non-zero entries are at columns j where max(0, k-n+1) <= j <= min(n-1, k)
    # C[k, j] = a[k-j] for valid j
    rows = []
    cols = []
    vals = []
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    C = csr_matrix((vals, (rows, cols)), shape=(num_rows, n))
    return C


def lp_step(a, C, M_target):
    """Solve LP: max sum(g) s.t. C@g <= M_target, g >= 0."""
    n = len(a)
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    b = np.full(2 * n - 1, M_target)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c, A_ub=C, b_ub=b, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 10})
        if result.success and result.x is not None:
            g = np.maximum(result.x, 0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def soft_gradient(a, conv, top_k=50, temp=0.01):
    """Compute soft-max weighted gradient of the peak."""
    n = len(a)
    # Use top-k peaks with soft-max weighting
    if top_k >= len(conv):
        indices = np.arange(len(conv))
    else:
        indices = np.argpartition(conv, -top_k)[-top_k:]
    
    peak_vals = conv[indices]
    scaled = peak_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(indices))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum
    
    grad = np.zeros(n)
    for idx in range(len(indices)):
        if w[idx] < 1e-12:
            continue
        k = int(indices[idx])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idxs = np.arange(j_start, j_end + 1)
        grad[idxs] += w[idx] * a[k - idxs]
    return grad

        M_target = M * 1.001
        g = lp_step(a, C, M_target)
        
        if g is None:
            no_improve += 1
        else:
            g_sum = g.sum()
            if g_sum > 1.0 + 1e-6:
                d = g - a
                # Line search
                best_t = 0
                best_obj_try = best_obj
                for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.01]:
                    a_try = a + t * d
                    a_try = np.maximum(a_try, 0)

                    if s_try < 1e-15:

                    a_try = a_try / s_try

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Sparse-matrix LP-based direction finding (main workhorse)
2. Soft-max gradient descent with adaptive temperature annealing
3. Multi-length exploration with smart initialization
4. Iterative refinement with restarts and polishing

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_sparse_conv_matrix(a):
    """Build sparse matrix C such that C @ g = full_convolve(a, g)."""
    n = len(a)
    num_rows = 2 * n - 1
    rows = np.empty(num_rows * n, dtype=np.int32)
    cols = np.empty(num_rows * n, dtype=np.int32)
    vals = np.empty(num_rows * n, dtype=np.float64)
    ptr = 0
    for k in range(num_rows):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        count = j_end - j_start + 1
        rows[ptr:ptr+count] = k
        cols[ptr:ptr+count] = np.arange(j_start, j_end + 1)
        vals[ptr:ptr+count] = a[k - np.arange(j_start, j_end + 1)]
        ptr += count
    rows = rows[:ptr]
    cols = cols[:ptr]
    vals = vals[:ptr]
    return csr_matrix((vals, (rows, cols)), shape=(num_rows, n))


def solve_lp(a, C, M_target):
    """Solve LP: max sum(g) s.t. C@g <= M_target, g >= 0. Returns g or None."""
    n = len(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M_target)
    bounds = [(0, None)] * n
    try:
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True, 'time_limit': 15})
        if result.success and result.x is not None:
            g = np.maximum(result.x, 0.0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def compute_soft_gradient(a, conv, top_k=50, temp=0.01):
    """Compute soft-max weighted gradient of the peak autocorrelation."""
    n = len(a)
    num_peaks = min(top_k, len(conv))
    if num_peaks == len(conv):
        indices = np.arange(len(conv))
    else:
        indices = np.argpartition(conv, -num_peaks)[-num_peaks:]
    peak_vals = conv[indices]
    scaled = peak_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(indices))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum
    grad = np.zeros(n)
    for idx in range(len(indices)):
        if w[idx] < 1e-13:
            continue
        k = int(indices[idx])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idxs = np.arange(j_start, j_end + 1)
        grad[idxs] += 2.0 * w[idx] * a[k - idxs]
    return grad

    C = build_sparse_conv_matrix(a)
    conv = fast_convolve(a)
    M = conv.max()

    t0 = time.monotonic()
    iteration = 0
    no_improve = 0

    perturb_count = 0
    lp_fail_streak = 0

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:

        M_target = M * 1.0005
        g = solve_lp(a, C, M_target)
        if g is None:

            lp_fail_streak += 1
        else:
            g_sum = g.sum()
            if g_sum > 1.0 + 1e-5:
                d = g - a
                best_t = 0.0
                best_obj_try = best_obj
                for t in [0.4, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.01]:

                        continue
                    a_try = a_try / s_try
                    conv_try = fast_convolve(a_try)
                    obj_try = 2.0 * n * conv_try.max()
                    if obj_try < best_obj_try - 1e-10:
                        best_obj_try = obj_try
                        best_t = t
                        break
                if best_t > 0:
                    a = a + best_t * d
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    conv = fast_convolve(a)
                    M = conv.max()
                    C = build_sparse_conv_matrix(a)
                    cur_obj = 2.0 * n * M
                    if cur_obj < best_obj - 1e-12:
                        best_obj = cur_obj
                        best_a = a.copy()
                    no_improve = 0
                    lp_fail_streak = 0
                else:
                    no_improve += 1
                    lp_fail_streak += 1
            else:
                no_improve += 1
                lp_fail_streak += 1

        iteration += 1
        if verbose and iteration % 20 == 0:
            print(f"      LP iter {iteration}, obj={best_obj:.8f}, M={M:.8f}, t={time.monotonic()-t0:.1f}s")

        if no_improve >= max_no_improve:
            rng = np.random.default_rng(iteration * 13 + 7 + perturb_count * 1009)
            scale = 0.001 + 0.002 * min(perturb_count, 15)
            noise = rng.standard_normal(n) * scale
            a = best_a + noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
            else:
                a = a / s
            conv = fast_convolve(a)
            M = conv.max()
            C = build_sparse_conv_matrix(a)
            no_improve = 0
            perturb_count += 1
            max_no_improve = min(150, 40 + 15 * perturb_count)
            if perturb_count > 50:
                break

        if lp_fail_streak > 20:
            # Brief gradient escape
            a_g, obj_g = gradient_step(a, time_budget=min(3.0, remaining * 0.1),
                                        lr=0.005, top_k=40, temp=0.003)
            if obj_g < best_obj:
                best_obj = obj_g
                best_a = a_g.copy()
                a = a_g.copy()
                conv = fast_convolve(a)
                M = conv.max()
                C = build_sparse_conv_matrix(a)
            lp_fail_streak = 0
            no_improve = 0

    return best_a, best_obj


def gradient_step(a0, time_budget, lr=0.01, momentum=0.9, top_k=50, temp=0.01):
    """Nesterov-accelerated soft-max gradient descent."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    v = np.zeros(n)
    y = a.copy()
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    cur_lr = lr
    cur_temp = temp
    cur_mom = momentum
    no_improve = 0
    while time.monotonic() - t0 < time_budget:
        y = a + cur_mom * v
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y
        else:
            y = a.copy()
        conv_y = fast_convolve(y)
        grad = compute_soft_gradient(y, conv_y, top_k=top_k, temp=cur_temp)
        v = cur_mom * v - cur_lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()
            v = np.zeros(n)
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        cur_lr *= 0.999
        cur_temp *= 1.008
        cur_mom = min(0.97, cur_mom + 0.0002)
        if no_improve >= 300:
            break
    return best_a, best_obj


def generate_inits(n, parent, best_seq, rng):
    """Generate diverse initializations for length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    inits.append(('uniform', np.ones(n) / n))
    s1 = (np.sin(np.pi * x / n) ** 2 + 0.02)
    inits.append(('sine', s1 / s1.sum()))
    s2 = (np.sin(2 * np.pi * x / n) ** 2 + 0.02)
    inits.append(('sine2', s2 / s2.sum()))
    alt = np.ones(n)
    alt[::2] = 1.5
    alt[1::2] = 0.5
    inits.append(('alt', alt / alt.sum()))
    edge = max(1, n // 6)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    if bi.sum() > 0:
        inits.append(('bimodal', bi / bi.sum()))
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.15 * n)) ** 2)
    inits.append(('gauss', g / g.sum()))
    r = rng.dirichlet(np.ones(n) * 0.5)
    inits.append(('random', r))
    if parent is not None and parent.sum() > 1e-10:
        p = parent / parent.sum()
        if len(p) == n:
            inits.append(('parent', p.copy()))
        else:
            pi = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
            pi = np.maximum(pi, 0)
            if pi.sum() > 1e-10:
                inits.append(('parent_interp', pi / pi.sum()))
    if best_seq is not None and best_seq.sum() > 1e-10:
        b = best_seq / best_seq.sum()
        if len(b) != n:
            bi2 = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(b)), b)
            bi2 = np.maximum(bi2, 0)
            if bi2.sum() > 1e-10:
                inits.append(('best_interp', bi2 / bi2.sum()))
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Hybrid sparse-LP + gradient optimization for minimizing
    2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 35.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            parent_score = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(parent_score):
                best_score = parent_score
                best_sequence = parent.copy()
            print(f"Parent loaded: n={len(p)}, score={best_score:.6f}")
    except Exception:
        pass

    # ===================== PHASE 1: LP on parent =====================
    if best_sequence is not None:
        n_p = len(best_sequence)
        t_malloc = min(350.0, (deadline - time.monotonic()) * 0.25)
        print(f"\n=== Phase 1: LP optimize parent (n={n_p}, t={t_malloc:.0f}s) ===")
        a_opt, obj = lp_optimize(best_sequence, time_limit=t_malloc, verbose=True)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Phase 1: {score:.6f} *** NEW BEST ***")
            else:
                print(f"  Phase 1: {score:.6f} (no improvement)")

    # ===================== PHASE 2: Multi-length exploration =====================
    print("\n=== Phase 2: Multi-length exploration ===")
    lengths_to_try = []
    if best_sequence is not None:
        np_ = len(best_sequence)
        candidates = [np_, max(256, np_ // 2), min(1536, np_ + 256)]
        lengths_to_try = sorted(set(candidates))
    else:
        lengths_to_try = [512, 768, 1024]

    for n in lengths_to_try:
        if time.monotonic() > deadline - 150:
            break
        if n > 1536 or n < 128:
            continue
        inits = generate_inits(n, parent, best_sequence, rng)
        t_per = max(30.0, (deadline - time.monotonic() - 100) / (max(1, len(lengths_to_try)) * 3))
        print(f"  Length n={n}:")
        for name, a0 in inits:
            if time.monotonic() > deadline - 100:
                break
            t_alloc = min(t_per, deadline - time.monotonic() - 80)
            if t_alloc < 20:
                break
            # Quick gradient warm-up
            a_g, obj_g = gradient_step(a0, time_budget=min(15.0, t_alloc * 0.25),
                                        lr=0.01, top_k=40, temp=0.008)
            score_g = float(evaluate_sequence(a_g.tolist()))
            # LP refinement
            t_lp = t_alloc - min(15.0, t_alloc * 0.25) - 5
            if t_lp > 10:
                a_lp, obj_lp = lp_optimize(a_g, time_limit=t_lp)
                score_lp = float(evaluate_sequence(a_lp.tolist()))
            else:
                a_lp = a_g
                score_lp = score_g
            if np.isfinite(score_lp) and score_lp < best_score:
                best_score = score_lp
                best_sequence = a_lp.copy()
                print(f"    {name}: LP score={score_lp:.6f} *** NEW BEST ***")
            elif np.isfinite(score_g) and score_g < best_score:
                best_score = score_g
                best_sequence = a_g.copy()
                print(f"    {name}: Grad score={score_g:.6f} *** NEW BEST ***")

    # ===================== PHASE 3: Deep LP polish =====================
    if best_sequence is not None and time.monotonic() < deadline - 80:
        n_best = len(best_sequence)
        t_alloc = min(350.0, deadline - time.monotonic() - 50)
        print(f"\n=== Phase 3: Deep LP polish (n={n_best}, t={t_alloc:.0f}s) ===")
        a_opt, obj = lp_optimize(best_sequence, time_limit=t_alloc, verbose=True)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Phase 3: {score:.6f} *** NEW BEST ***")
        else:
            print(f"  Phase 3: {score:.6f}")

    # ===================== PHASE 4: Perturbed LP restarts =====================
    if best_sequence is not None and time.monotonic() < deadline - 60:
        n_best = len(best_sequence)
        print("\n=== Phase 4: Perturbed LP restarts ===")
        for pi in range(4):
            if time.monotonic() > deadline - 40:
                break
            t_alloc = min(120.0, (deadline - time.monotonic() - 30) / (4 - pi))
            if t_alloc < 20:
                break
            scale = 0.002 * (pi + 1)
            noise = rng.standard_normal(n_best) * scale
            a0 = np.maximum(best_sequence + noise, 0.0)
            s = a0.sum()
            if s > 1e-15:
                a0 = a0 / s
            else:
                a0 = best_sequence.copy()
            a_opt, obj = lp_optimize(a0, time_limit=t_alloc)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Restart {pi}: {score:.6f} *** NEW BEST ***")
            else:
                print(f"  Restart {pi}: {score:.6f}")

    # ===================== PHASE 5: Final gradient polish =====================
    if best_sequence is not None and time.monotonic() < deadline - 15:
        t_alloc = min(50.0, deadline - time.monotonic() - 10)
        if t_alloc > 10:
            print(f"\n=== Phase 5: Final gradient polish (t={t_alloc:.0f}s) ===")
            a_opt, obj = gradient_step(best_sequence, time_budget=t_alloc,
                                        lr=0.003, momentum=0.95, top_k=80, temp=0.001)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Phase 5: {score:.6f} *** NEW BEST ***")
            else:
                print(f"  Phase 5: {score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\n{'='*50}")
    print(f"FINAL: score={best_score:.8f}, n={len(best_sequence)}")
    print(f"Elapsed: {time.monotonic() - (deadline - float(budget_s) + 35.0):.1f}s")
    return [float(x) for x in best_sequence]
