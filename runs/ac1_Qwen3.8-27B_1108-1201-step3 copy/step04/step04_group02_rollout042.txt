```python
import numpy as np
from scipy.optimize import linprog, minimize
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_conv_matrix(f):
    """Build the convolution matrix A where A[k,j] = f[k-j] if valid."""
    n = len(f)
    size = 2 * n - 1
    A = np.zeros((size, n))
    for j in range(n):
        # (f*g)[k] includes term f[k-j] * g_j for valid k-j
        # k-j ranges from 0 to n-1, so k ranges from j to j+n-1
        ks = np.arange(j, j + n)
        A[ks, j] = f
    return A

def fw_lp_step(f, M, time_limit=10):
    """Solve one LP step of Frank-Wolfe."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return res.x
    except Exception:
        pass
    return None
```

```python
def qp_polish(f0, time_limit, top_k=20):
    """Polish using QP: min t s.t. (a*a)[k] <= t for top_k peaks, sum(a)=1, a>=0."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)
    
    conv = fast_convolve(f)
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    peaks = peaks[np.argsort(conv[peaks])[::-1]]  # sorted by value
    
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(best_f)
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        M_current = conv.max()
        
        # Define the QP
        # Variables: a (n) + t (1) = n+1
        # Objective: minimize t -> linear
        # Constraints: (a*a)[k] - t <= 0 for k in peaks
        #              sum(a) = 1
        #              a >= 0
        
        def objective(vars):
            a = vars[:n]
            t = vars[n]
            return t
        
        def objective_grad(vars):
            g = np.zeros(n + 1)
            g[n] = 1.0
            return g
        
        constraints = []
        for k in peaks:
            k = int(k)
            def make_constraint(k_val):
                def constraint(vars):
                    a = vars[:n]
                    t = vars[n]
                    # (a*a)[k_val] = sum_i a_i * a[k_val - i] for valid i
                    i_min = max(0, k_val - n + 1)
                    i_max = min(n - 1, k_val)
                    if i_min > i_max:
                        return t  # constraint always satisfied
                    sum_val = 0.0
                    for i in range(i_min, i_max + 1):
                        sum_val += a[i] * a[k_val - i]
                    return sum_val - t
                def constraint_jac(vars):
                    a = vars[:n]
                    jac = np.zeros(n + 1)
                    i_min = max(0, k_val - n + 1)
                    i_max = min(n - 1, k_val)
                    for i in range(i_min, i_max + 1):
                        # d/da_i [a_i * a[k_val-i]] = a[k_val-i]
                        # d/da_{k_val-i} [a_i * a[k_val-i]] = a[i]
                        # But if i == k_val-i, then it's 2*a[i]
                        if i == k_val - i:
                            jac[i] += 2.0 * a[i]
                        else:
                            jac[i] += a[k_val - i]
                            if k_val - i < n:
                                jac[k_val - i] += a[i]
                    jac[n] = -1.0
                    return jac
                return {'type': 'ineq', 'fun': constraint, 'jac': constraint_jac}
            constraints.append(make_constraint(k))
        
        # sum(a) = 1
        def eq_constraint(vars):
            return vars[:n].sum() - 1.0
        def eq_jac(vars):
            j = np.zeros(n + 1)
            j[:n] = 1.0
            return j
        constraints.append({'type': 'eq', 'fun': eq_constraint, 'jac': eq_jac})
        
        bounds = [(0, None)] * n + [(best_obj * 0.5, None)]
        
        x0 = np.concatenate([best_f, [M_current]])
        
        try:
            res = minimize(objective, x0, method='SLSQP', jac=objective_grad,
                         constraints=constraints, bounds=bounds,
                         options={'maxiter': 200, 'ftol': 1e-12})
            if res.success or (hasattr(res, 'fun') and res.fun < best_obj):
                a_new = np.maximum(res.x[:n], 0.0)
                a_new = a_new / a_new.sum()
                new_obj = fast_convolve(a_new).max()
                if new_obj < best_obj:
                    best_obj = new_obj
                    best_f = a_new
        except Exception:
            break
        
        if time.monotonic() - t0 > time_limit - 5:
            break
    
    return best_f, best_obj
```

```python
            break
        x1 = hi - gr * (hi - lo)
        x2 = lo + gr * (hi - lo)
        ft1 = np.maximum((1 - x1) * f + x1 * g, 0.0)
        ft1 = ft1 / ft1.sum()
        v1 = fast_convolve(ft1).max()
        ft2 = np.maximum((1 - x2) * f + x2 * g, 0.0)
        ft2 = ft2 / ft2.sum()
        v2 = fast_convolve(ft2).max()
        if v1 < v2:
            hi = x2
        else:
            lo = x1
        mid = (lo + hi) / 2.0
        ft = np.maximum((1 - mid) * f + mid * g, 0.0)
        ft = ft / ft.sum()
        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = mid

    return best_alpha, best_val


def fw_lp_optimize(f0, time_limit, verbose=False):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 3:
            break

        g = solve_fw_lp(f, M, time_limit=min(rem - 2, 20))
        if g is None:
            g = solve_fw_lp(f, M * 1.1, time_limit=min(rem - 2, 20))
            if g is None:
                break

        g_sum = g.sum()
        if g_sum < 1e-14:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 20:
                break
            continue

        alpha, best_val = line_search_alpha(f, g, M, max_evals=22)

        if alpha > 1e-10:
            f = (1.0 - alpha) * f + alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if verbose and it % 15 == 0:
            print(f"    FW it={it}, M={M:.10f}, best={best_obj:.10f}, alpha={alpha:.4f}, rem={rem:.0f}s")

        if no_improve > 40:
            rng = np.random.default_rng(it * 31 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0

    return best_f, best_obj


def compute_subgrad(f, conv, top_k=48, temp=0.001):
    n = len(f)
    size = len(conv)
    if top_k >= size:
        peaks = np.arange(size)
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    pv = conv[peaks]
    scaled = pv / max(temp, 1e-15)
    scaled -= scaled.max()
    w = np.exp(scaled)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= wsum

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            sg[js:je + 1] += w[i] * f[k - je:k - js:-1]
        js2 = max(k, 0)
        je2 = min(n - 1, k + n - 1)
        if js2 <= je2:
            sg[js2:je2 + 1] += w[i] * f[js2 - k:je2 - k + 1]
    return sg


def sg_polish(f0, time_limit, lr=0.003, mom=0.93, top_k=64, temp=0.0005):
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr
    cur_temp = temp

    while time.monotonic() - t0 < time_limit - 2:
        it += 1
        sg = compute_subgrad(f, conv, top_k=top_k, temp=cur_temp)
        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        obj = conv.max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        cur_lr *= 0.9997
        cur_temp *= 1.001
        if no_improve > 300:
            rng = np.random.default_rng(it + 500)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            conv = fast_convolve(f)
            v = np.zeros(n)
            cur_lr = lr * 0.5
            no_improve = 0
    return best_f, best_obj


def qp_polish(f0, time_limit, top_k=12):
    f = np.maximum(f0.astype(np.float64), 0.0)
    f = f / f.sum()
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit - 5:
        it += 1
        conv = fast_convolve(best_f)
        peaks = sorted(np.argpartition(conv, -top_k)[-top_k:].tolist())
        M_cur = conv.max()

        peak_data = []
        for k in peaks:
            i_min = max(0, k - n + 1)
            i_max = min(n - 1, k)
            if i_min <= i_max:
                i_arr = np.arange(i_min, i_max + 1)
                j_arr = k - i_arr
                peak_data.append((i_arr, j_arr))
        if not peak_data:
            break

        def qp_objective(vars_vec):
            return vars_vec[n]

        def qp_obj_grad(vars_vec):
            g = np.zeros(n + 1)
            g[n] = 1.0
            return g

        constraints = []
        for pd in peak_data:
            i_arr, j_arr = pd
            def make_ineq_ia(ia, ja):
                def con_fun(vars_vec):
                    a = vars_vec[:n]
                    return np.dot(a[ia], a[ja]) - vars_vec[n]
                def con_jac(vars_vec):
                    a = vars_vec[:n]
                    jac = np.zeros(n + 1)
                    np.add.at(jac, ia, a[ja])
                    np.add.at(jac, ja, a[ia])
                    jac[n] = -1.0
                    return jac
                return {'type': 'ineq', 'fun': con_fun, 'jac': con_jac}
            constraints.append(make_ineq_ia(i_arr, j_arr))

        def eq_fun(vars_vec):
            return vars_vec[:n].sum() - 1.0
        def eq_jac(vars_vec):
            j = np.zeros(n + 1)
            j[:n] = 1.0
            return j
        constraints.append({'type': 'eq', 'fun': eq_fun, 'jac': eq_jac})

        bounds = [(0.0, None)] * n + [(M_cur * 0.1, None)]
        x0 = np.concatenate([best_f, [M_cur]])

        try:
            res = minimize(qp_objective, x0, jac=qp_obj_grad, method='SLSQP',
                           constraints=constraints, bounds=bounds,
                           options={'maxiter': 200, 'ftol': 1e-14, 'maxls': 30})
            if res.x is not None:
                a_new = np.maximum(res.x[:n], 0.0)
                if a_new.sum() > 1e-10:
                    a_new = a_new / a_new.sum()
                    new_obj = fast_convolve(a_new).max()
                    if new_obj < best_obj - 1e-14:
                        best_obj = new_obj
                        best_f = a_new
                        print(f"    QP polish it={it}: {best_obj:.10f}")
        except Exception:
            pass

        if time.monotonic() - t0 > time_limit - 8:
            break

    return best_f, best_obj


def generate_inits(n, parent=None, rng=None):
    if rng is None:
        rng = np.random.default_rng(42)
    inits = []
    x = np.arange(n, dtype=np.float64)
    inits.append(np.ones(n) / n)
    ramp_d = np.arange(n, 0, -1, dtype=np.float64)
    inits.append(ramp_d / ramp_d.sum())
    ramp_i = np.arange(1, n + 1, dtype=np.float64)
    inits.append(ramp_i / ramp_i.sum())
    q = (x - n / 2) ** 2 + 1.0
    inits.append(q / q.sum())
    e1 = np.exp(-x / (n / 3.0))
    inits.append(e1 / e1.sum())
    e2 = np.exp(-x[::-1] / (n / 3.0))
    inits.append(e2 / e2.sum())
    c = np.cos(np.pi * (x / max(n - 1, 1) - 0.5)) ** 4 + 0.01
    inits.append(c / c.sum())
    sn = np.sin(np.pi * x / n) + 0.05
    inits.append(sn / sn.sum())
    r = rng.dirichlet(np.ones(n) * 0.5)
    inits.append(r)
    if parent is not None:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
            blended = 0.5 * (px / ps) + 0.5 * (ramp_d / ramp_d.sum())
            inits.append(blended / blended.sum())
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        a = np.minimum(a, 1000.0)
        if a.sum() < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            old = best_score
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}) (was {old:.8f})")
        return score

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
    except Exception:
        pass

    print("=== Phase 1: Parent FW-LP Optimization ===")
    if parent is not None:
        n_p = len(parent)
        t1 = min(350, deadline - time.monotonic() - 500)
        if t1 > 30:
            print(f"  FW-LP on parent (n={n_p}), t={t1:.0f}s")
            a_opt, obj = fw_lp_optimize(parent, time_limit=t1, verbose=True)
            try_seq(a_opt, "parent_fw")

            rem = deadline - time.monotonic() - 400
            if rem > 30:
                a_opt2, _ = qp_polish(a_opt, time_limit=min(rem, 120), top_k=10)
                try_seq(a_opt2, "parent_qp")

                rem2 = deadline - time.monotonic() - 200
                if rem2 > 20:
                    a_opt3, _ = sg_polish(a_opt2, time_limit=min(rem2, 60),
                                           lr=0.002, top_k=n_p // 4, temp=0.0002)
                    try_seq(a_opt3, "parent_sg")

    print(f"\n=== Phase 2: Multi-size Exploration ===")
    sizes_to_try = [432, 480, 512, 320, 256]
    if parent is not None:
        if len(parent) in sizes_to_try:
            sizes_to_try.remove(len(parent))

    for n in sizes_to_try:
        rem = deadline - time.monotonic() - 150
        if rem < 40:
            break
        t_alloc = min(rem // len(sizes_to_try), 120)
        if t_alloc < 20:
            t_alloc = 20
        print(f"  Trying n={n}, t={t_alloc:.0f}s")
        inits = generate_inits(n, parent, rng)
        best_for_n = None
        best_obj_n = float('inf')
        t_per_init = max(10, t_alloc / len(inits))

        for i, f_init in enumerate(inits[:4]):
            if time.monotonic() > deadline - 100:
                break
            t_i = min(t_per_init, deadline - time.monotonic() - 60)
            if t_i < 10:
                break
            a_opt, obj = fw_lp_optimize(f_init, time_limit=t_i)
            try_seq(a_opt, f"n{n}_i{i}")
            if obj < best_obj_n:
                best_obj_n = obj
                best_for_n = a_opt.copy()

        if best_for_n is not None and best_score - best_obj_n * 2 * n < 0.01:
            rem_qp = deadline - time.monotonic() - 80
            if rem_qp > 15:
                a_qp, _ = qp_polish(best_for_n, time_limit=min(rem_qp, 60), top_k=10)
                try_seq(a_qp, f"n{n}_qp")

    print(f"\n=== Phase 3: Deep Optimization on Best ===")
    print(f"  Current best: {best_score:.8f}")
    if best_seq is not None:
        n_best = len(best_seq)

        # Multiple FW-LP restarts
        for p in range(3):
            rem = deadline - time.monotonic() - 80
            if rem < 20:
                break
            t_alloc = min(rem // (3 - p), 100)
            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.003)
            else:
                f_init = project_simplex(0.9 * best_seq + 0.1 * (np.ones(n_best) / n_best))
            a_opt, obj = fw_lp_optimize(f_init, time_limit=t_alloc, verbose=(p == 0))
            try_seq(a_opt, f"deep_fw_{p}")

        # QP polish
        rem = deadline - time.monotonic() - 50
        if rem > 15:
            a_qp, _ = qp_polish(best_seq, time_limit=min(rem, 80), top_k=12)
            try_seq(a_qp, "deep_qp")

        # SG polish
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            a_sg, _ = sg_polish(best_seq, time_limit=min(rem, 50),
                                lr=0.001, mom=0.95, top_k=n_best // 3, temp=0.0001)
            try_seq(a_sg, "deep_sg")

    print(f"\n=== Phase 4: Final Micro-Optimization ===")
    print(f"  Current best: {best_score:.8f}")
    if best_seq is not None:
        n_best = len(best_seq)
        for trial in range(5):
            rem = deadline - time.monotonic() - 10
            if rem < 8:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.001)
            a_opt, _ = sg_polish(f_pert, time_limit=min(rem, 15),
                                 lr=0.0005, mom=0.96, top_k=n_best // 3, temp=0.00005)
            try_seq(a_opt, f"micro_{trial}")

        rem = deadline - time.monotonic() - 5
        if rem > 10:
            a_final, _ = fw_lp_optimize(best_seq, time_limit=min(rem, 30))
            try_seq(a_final, "final_fw")

    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
