import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_conv_matrix(a, n):
    """C such that C @ g = conv(a, g)"""
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C

def lp_direction(a, M, C=None):
    """Solve LP: max sum(g) s.t. C@g <= M, g >= 0"""
    n = len(a)
    if C is None:
        C = build_conv_matrix(a, n)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=C, b_ub=b, bounds=bounds, method='highs',
                         options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if result.success:
            g = result.x
            if g.sum() > 1e-10:
                return g, True
    except:
        pass
    return a.copy(), False

def dual_weights(a, M, C=None, n=None):
    """Get dual variables from LP to identify important constraints."""
    n = n or len(a)
    if C is None:
        C = build_conv_matrix(a, n)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=C, b_ub=b, bounds=bounds, method='highs',
                         options={'presolve': True})
        if result.success and result.ineqlin is not None:
            return result.ineqlin.slack  # or dual variables
    except:
        pass
    return None

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a, n):
    """Build matrix C such that C @ g = conv(a, g) (full linear convolution)."""
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def compute_subgradient(a, conv, top_k=15, temp=0.002):
    """Compute softmax-weighted subgradient of max autocorrelation."""
    n = len(a)
    size = 2 * n - 1
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peaks]
    scaled = peak_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum

    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-12:
            continue
        k = int(peaks[idx])
        # grad[i] += w * (a[max(0,k-i)] + a[min(n-1, k+i)] ... )
        # conv(a,a)[k] = sum_i a[i]*a[k-i] (valid indices)
        # d/d a[j] of conv(a,a)[k] = a[k-j] + a[j-k] (valid)
        for j in range(n):
            contrib = 0.0
            if k - j >= 0 and k - j < n:
                contrib += a[k - j]
            if j - k >= 0 and j - k < n:
                contrib += a[j - k]
            grad[j] += w[idx] * contrib
    return grad


def compute_subgradient_fast(a, conv, top_k=15, temp=0.002):
    """Faster subgradient using array operations."""
    n = len(a)

        k = int(peaks[idx])

        # conv(a,a)[k] = sum_{i} a[i]*a[k-i] where 0<=i<n and 0<=k-i<n
        # i ranges from max(0, k-(n-1)) to min(n-1, k)

        if j2_start <= j2_end:
            j2 = np.arange(j2_start, j2_end + 1)
            grad[j2] += wi * a[j2 - k]
    
    return grad


def lp_direction(a, M_val, C=None, n=None):
    """Solve LP: max sum(g) s.t. C@g <= M_val, g >= 0. Returns (g, success)."""
    if n is None:
        n = len(a)
    if C is None:
        C = build_conv_matrix(a, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M_val)
    bounds = [(0, None)] * n

    try:
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                                  'time_limit': 5.0})
        if result.success:
            g = np.maximum(result.x, 0)
            if g.sum() > 1e-10:
                return g, True
    except Exception:
        pass
    return a.copy(), False

def nesterov_optimize(a0, time_limit, lr_init=0.01, momentum=0.91, top_k=40, temp_init=0.002):
    """Nesterov-accelerated gradient descent."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    y = a.copy()
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    mom = momentum
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        y = a + mom * v
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y

        conv_y = fast_convolve(y)
        grad = compute_subgradient_fast(y, conv_y, top_k=top_k, temp=temp)
        grad = grad - grad.mean()  # Maintain sum

        v = mom * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9997
        temp *= 1.002
        mom = min(0.97, mom + 0.0001)

        if no_improve >= 300:
            break

    return best_a, best_obj


def coordinate_descent(a0, time_limit, seed=12345):

    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, 2.0 * len(a) * fast_convolve(a).max()
    a = a / s

    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    rng = np.random.default_rng(seed)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        batch = max(1, int(300000 / n))
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            for delta in [0.002, 0.005, 0.01, 0.02, -0.002, -0.005, -0.01, -0.02]:
                if a[i] + delta < 0 or a[j] - delta < 0:
                    continue
                a_try = a.copy()
                a_try[i] += delta
                a_try[j] -= delta
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()

                    a = a_try
                    best_obj = obj_try
                    best_a = a.copy()
                    no_improve = 0
                    break
            else:
                no_improve += 1

    tri = 1.0 - np.abs(x - 0.5 * n) / (0.5 * n)
    tri = np.maximum(tri, 0.01)
    inits['triangle'] = tri / tri.sum()

    # Cosine bell
    cosb = np.cos(np.pi * (x - 0.5 * n) / n) ** 2 + 0.01
    inits['cosbell'] = cosb / cosb.sum()

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps
                # Also add smoothed version
                from scipy.ndimage import uniform_filter1d
                pp_smooth = uniform_filter1d(pp, size=max(3, n // 30))
                pp_smooth = np.maximum(pp_smooth, 0)
                ppss = pp_smooth.sum()
                if ppss > 1e-10:
                    inits['parent_smooth'] = pp_smooth / ppss

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Hybrid LP + gradient + coordinate descent search
    with aggressive peak reduction and multi-n exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: n={len(p)}, score={evaluate_sequence(p.tolist()):.6f}")
    except Exception:
        pass

    # =========================================================
    # Phase 1: Quick gradient scan across lengths
    # =========================================================
    print("=== Phase 1: Multi-n Gradient Scan ===")
    t_start = time.monotonic()
    phase1_limit = 180.0

    # Choose n values to try
    n_candidates = [384, 512, 640, 768, 896, 1024]
    if parent is not None:
        pn = len(parent)
        # Add nearby values
        for dn in [-64, -32, 32, 64]:
            nn = pn + dn
            if nn >= 128 and nn not in n_candidates:
                n_candidates.append(nn)
        if pn not in n_candidates:
            n_candidates.append(pn)
        n_candidates.sort()

    n_candidates = n_candidates[:7]  # Limit to 7
    print(f"  Trying n values: {n_candidates}")

    scan_results = []
    t_per_n = max(15.0, (phase1_limit - 10) / len(n_candidates))

    for n in n_candidates:
        if time.monotonic() > deadline - 800:
            break
        inits = generate_inits(n, parent, rng)
        # Prioritize
        priority = ['parent', 'parent_interp', 'parent_smooth', 'gaussian_narrow',
                    'sine1', 'cosbell', 'random', 'triangle']
        init_names = [nm for nm in priority if nm in inits]
        init_names += [nm for nm in inits if nm not in priority]
        init_names = init_names[:3]  # Top 3 per n

        t_per_init = max(5.0, t_per_n / len(init_names))

        for name in init_names:
            if time.monotonic() > deadline - 700:
                break
            t_alloc = min(t_per_init, deadline - time.monotonic() - 600)
            if t_alloc < 4:
                break

            a0 = inits[name]
            a_opt, obj = nesterov_optimize(a0, time_limit=t_alloc,
                                           lr_init=0.012, momentum=0.92,
                                           top_k=50, temp_init=0.0015)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Grad n={n}, {name}: {score:.6f}")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 2: Aggressive LP deep optimization
    # =========================================================
    print("=== Phase 2: Aggressive LP Optimization ===")
    t_start2 = time.monotonic()

    # Select top candidates (up to 3, unique n values)
    top_configs = []
    seen_n = set()
    for score, n, name, a_opt in scan_results:
        if n not in seen_n:
            top_configs.append((n, a_opt))
            seen_n.add(n)
        if len(top_configs) >= 3:
            break

    # Always include the overall best
    if best_sequence is not None:
        bn = len(best_sequence)
        if bn not in seen_n:
            top_configs.append((bn, best_sequence.copy()))
            seen_n.add(bn)

    remaining = deadline - time.monotonic() - 200.0
    n_cand = max(1, len(top_configs))
    t_per_lp = max(40.0, remaining * 0.55 / n_cand)

    for i, (n, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 180:
            break
        t_alloc = min(t_per_lp, deadline - time.monotonic() - 150)
        if t_alloc < 15:
            break

        print(f"  LP candidate {i}: n={n}, t_alloc={t_alloc:.0f}s")
        a_opt, obj = lp_optimize_aggressive(a0, time_limit=t_alloc, verbose=True)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  LP {i} (n={n}): {score:.6f} *NEW BEST*")
            else:
                print(f"  LP {i} (n={n}): {score:.6f}")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 3: Nesterov polish on best
    # =========================================================
    print("=== Phase 3: Nesterov Polish ===")
    t_start3 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 120.0
        if remaining > 30:
            configs = [
                {'lr_init': 0.006, 'momentum': 0.93, 'top_k': 60, 'temp_init': 0.001},
                {'lr_init': 0.004, 'momentum': 0.95, 'top_k': 100, 'temp_init': 0.0005},
                {'lr_init': 0.01, 'momentum': 0.89, 'top_k': 30, 'temp_init': 0.003},
            ]
            t_per = max(10.0, remaining * 0.7 / len(configs))
            for gi, cfg in enumerate(configs):
                if time.monotonic() > deadline - 80:
                    break
                t_alloc = min(t_per, deadline - time.monotonic() - 60)
                if t_alloc < 5:
                    break
                a_opt, obj = nesterov_optimize(best_sequence, time_limit=t_alloc, **cfg)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Nesterov {gi}: {score:.6f} *NEW BEST*")

    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 4: Final LP polish (multiple runs)
    # =========================================================
    print("=== Phase 4: Final LP Polish ===")
    t_start4 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 40.0
        if remaining > 30:
            for pi in range(2):
                if time.monotonic() > deadline - 20:
                    break
                t_alloc = min(remaining / (2 - pi) * 0.8, deadline - time.monotonic() - 15)
                if t_alloc < 10:
                    break

                if pi == 0:
                    a0 = best_sequence.copy()
                else:
                    scale = 0.001
                    noise = rng.standard_normal(len(best_sequence)) * scale
                    a0 = np.maximum(best_sequence + noise, 0)
                    s = a0.sum()
                    if s > 1e-15:
                        a0 = a0 / s
                    else:
                        a0 = best_sequence.copy()

                a_opt, obj = lp_optimize_aggressive(a0, time_limit=t_alloc)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Final LP {pi}: {score:.6f} *NEW BEST*")

    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 5: Coordinate descent micro-polish
    # =========================================================
    print("=== Phase 5: Coord Descent Polish ===")
    if best_sequence is not None and time.monotonic() < deadline - 10:
        t_cd = min(20.0, deadline - time.monotonic() - 5)
        if t_cd > 5:
            a_opt, obj = coordinate_descent(best_sequence, time_limit=t_cd, seed=999)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  CoordDesc: {score:.6f} *NEW BEST*")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
