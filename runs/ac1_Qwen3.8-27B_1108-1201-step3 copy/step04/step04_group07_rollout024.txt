"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using improved smooth gradient (Nesterov)
with correct FFT-based gradient computation, periodic LP refinement, multi-length
exploration, adaptive temperature annealing, and structured perturbation for
escaping local minima. Key improvements: correct gradient (2*cross_corr/tau),
better initialization diversity, larger sequence lengths, and combined
LP+smooth-gradient optimization.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_gradient(a, conv, tau):
    """
    Compute gradient of soft_max(conv, tau) w.r.t. a.
    grad_i = (2/tau) * sum_j a[j] * p[j+i]
    where p = softmax(conv / tau)
    """
    n = len(a)
    z_max = conv.max()
    exp_z = np.exp((conv - z_max) / tau)
    Z = exp_z.sum()
    p = exp_z / Z

    # Cross-correlation: cross[m] = sum_j a[j] * p[j+m]
    N_fft = 1 << (2 * n).bit_length()
    A_fft = np.fft.rfft(a, N_fft)
    P_fft = np.fft.rfft(p, N_fft)
    cross = np.fft.irfft(np.conj(A_fft) * P_fft, N_fft)

    grad = (2.0 / tau) * cross[:n]
    return grad


def smooth_gradient_descent(a0, time_limit, tau_init=0.5, tau_end=0.0005,
                            lr_init=0.01, momentum=0.9, verbose=False):
    """
    Nesterov accelerated smooth gradient descent with adaptive temperature.
    Uses correct FFT-based gradient: grad = (2/tau) * cross_corr(a, softmax(conv/tau))
    """
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    y = a.copy()

    tau = tau_init
    lr = lr_init
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()

    t0 = time.monotonic()
    step = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        # Nesterov lookahead
        y = a + momentum * v

        # Clip y to ensure positivity for gradient computation
        y_pos = np.maximum(y, 0.0)
        s_y = y_pos.sum()
        if s_y > 1e-15:
            y_pos = y_pos / s_y

        # Compute gradient
        conv_y = fast_convolve(y_pos)
        grad = compute_gradient(y_pos, conv_y, tau)

        # Update
        v = momentum * v - lr * grad
        a_new = y_pos + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
        else:
            a_new = a.copy()

        v = a_new - a
        a = a_new

        # Evaluate
        conv_a = fast_convolve(a)
        cur_obj = 2.0 * n * conv_a.max()
        step += 1

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

            tau = max(tau_end, tau * 0.98)
            lr = max(0.001, lr * 0.995)

            a = best_a + noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            v = np.zeros(n)
            no_improve = 0

        if verbose and step % 500 == 0:
            print(f"    SGD step {step}, obj={best_obj:.8f}, tau={tau:.6f}, lr={lr:.6f}, t={time.monotonic()-t0:.1f}s")

def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def solve_lp_direction(a, M, A_ub):
    """Solve LP: maximize sum(g) s.t. a*g <= M, g >= 0."""
    n = len(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum, g_sum
    except Exception:
        pass
    return a.copy(), 1.0


def line_search(a, d, current_obj, n, max_evals=12):
    """Simple line search with geometric candidates."""
    best_t = 0.0
    best_obj = current_obj
    evals = 0

    for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
        if evals >= max_evals:
            break
        a_try = np.maximum(a + t * d, 0.0)
        s = a_try.sum()
        if s < 1e-15:
            continue
        a_try = a_try / s
        conv_try = fast_convolve(a_try)
        obj_try = 2.0 * n * conv_try.max()
        evals += 1
        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_t = t

    # Refine around best_t
    if 0.05 < best_t < 0.95 and evals < max_evals - 4:
        for frac in [0.8, 1.2, 0.6, 1.4]:
            t_try = best_t * frac
            if t_try <= 0 or t_try > 1.5 or evals >= max_evals:
                continue
            a_try = np.maximum(a + t_try * d, 0.0)
            s = a_try.sum()
            if s > 1e-15:
                a_try = a_try / s
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()
                evals += 1
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_t = t_try

    return best_t, best_obj

            rng = np.random.default_rng(iteration * 13 + 7)
            scale = 0.003 + 0.002 * min(no_improve // 60, 5)
            noise = rng.standard_normal(n) * scale
            a = best_a + noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            else:
                a = best_a.copy()
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            tightness = 1.002

    return best_a, best_obj


def combined_optimize(a0, time_limit, verbose=False):
    """
    Combined LP + smooth gradient optimization with alternating phases.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    phase = 0
    phase_time = 0
    phase_duration = 8.0  # seconds per mini-phase
    no_improve_total = 0

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.0:
            break

        # Alternate between LP and smooth gradient phases
        if phase % 2 == 0:
            # LP phase
            A_ub = build_conv_matrix(a)
            conv = fast_convolve(a)
            M = conv.max()
            tightness = 0.999
            for _ in range(min(5, int(remaining // 0.5))):
                g, g_sum = solve_lp_direction(a, M * tightness, A_ub)
                if g_sum > 1.0 + 1e-7:
                    d = g - a
                    if np.linalg.norm(d) > 1e-12:
                        t_best, obj_best = line_search(a, d, 2.0 * n * fast_convolve(a).max(), n)
                        if obj_best < 2.0 * n * fast_convolve(a).max() - 1e-13:
                            a_new = np.maximum(a + t_best * d, 0.0)
                            s_new = a_new.sum()
                            if s_new > 1e-15:
                                a = a_new / s_new
                                conv = fast_convolve(a)
                                M = conv.max()
                                if 2.0 * n * M < best_obj:
                                    best_obj = 2.0 * n * M
                                    best_a = a.copy()
                                    no_improve_total = 0
                else:
                    tightness *= 1.01
                if time.monotonic() - t0 > time_limit - 5:
                    break
        else:
            # Smooth gradient phase
            a_tmp, obj_tmp = smooth_gradient_descent(
                a, time_limit=min(15.0, remaining - 5),
                tau_init=0.1, tau_end=0.002, lr_init=0.005, momentum=0.92,
                verbose=False
            )
            if obj_tmp < best_obj:
                best_obj = obj_tmp
                best_a = a_tmp.copy()
                a = a_tmp.copy()
                no_improve_total = 0

        phase += 1
        no_improve_total += 1

        if no_improve_total > 10 and phase % 4 == 0:
            # Perturbation
            rng = np.random.default_rng(phase * 17 + 3)
            noise = rng.standard_normal(n) * 0.004
            a = best_a + noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            no_improve_total = 0

        if verbose and phase % 4 == 0:
            print(f"    Combined phase {phase}, obj={best_obj:.8f}, t={time.monotonic()-t0:.1f}s")

    return best_a, best_obj


def get_inits(n, parent_seq, rng_local):
    """Generate diverse initializations of length n."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    if parent_seq is not None:
        x_old = np.linspace(0, 1, len(parent_seq))
        x_new = np.linspace(0, 1, n)
        inits['parent'] = np.interp(x_new, x_old, parent_seq)
        inits['parent'] = np.maximum(inits['parent'], 0.0)
        s = inits['parent'].sum()
        if s > 1e-15:
            inits['parent'] = inits['parent'] / s

    # Various structured inits
    for freq, name in [(1, 'sine1'), (2, 'sine2'), (3, 'sine3'), (4, 'sine4')]:
        sig = 0.7 + 0.3 * np.sin(2.0 * np.pi * freq * x / n)
        inits[name] = sig / sig.sum()

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.15 * n)) ** 2)
    inits['gauss'] = g / g.sum()

    # Triangle
    tri = np.minimum(x, n - 1 - x) + 1.0
    inits['tri'] = tri / tri.sum()

    # Random
    inits['random'] = rng_local.dirichlet(np.ones(n) * 0.3)

    # Bimodal
    mode1 = np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2)
    mode2 = np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2)
    bim = mode1 + mode2 + 0.05 * np.ones(n)
    inits['bimodal'] = bim / bim.sum()

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    1. Quick multi-length smooth gradient exploration
    2. Deep LP + combined optimization on best candidates
    3. Fine smooth gradient polishing
    4. Final perturb + re-optimize
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_sequence = None
    best_score = float('inf')

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(p)}")
    except Exception:
        pass

    # ===== Phase 1: Quick exploration =====
    print("=== Phase 1: Quick Multi-length Exploration ===")
    t_start = time.monotonic()
    phase1_budget = 300.0
    phase1_deadline = t_start + phase1_budget

    lengths = [512, 768, 1024, 1536]
    init_names_priority = ['parent', 'uniform', 'sine1', 'sine2', 'gauss', 'random', 'bimodal']

    results = []

    for n in lengths:
        if time.monotonic() > phase1_deadline - 30:
            break

        inits = get_inits(n, parent, rng)
        available = [nm for nm in init_names_priority if nm in inits][:3]

        t_per = max(15.0, (phase1_budget - 20.0) / (len(lengths) * 3))

        for name in available:
            if time.monotonic() > phase1_deadline - 15:
                break
            t_alloc = min(t_per, phase1_deadline - time.monotonic() - 10)
            if t_alloc < 8:
                break

            a0 = inits[name]
            # Use smooth gradient for quick exploration
            a_opt, obj_opt = smooth_gradient_descent(
                a0, time_limit=t_alloc,
                tau_init=0.3, tau_end=0.005, lr_init=0.008, momentum=0.92
            )
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n}, {name}: {score:.6f}")

    results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # ===== Phase 2: Deep LP + Combined =====
    print("=== Phase 2: Deep LP + Combined ===")
    t_start2 = time.monotonic()

    # Select top candidates (diverse lengths)
    top_configs = []
    seen_lengths = set()
    for score, n, name, a_opt in results[:8]:
        if n not in seen_lengths:
            top_configs.append((n, a_opt.copy()))
            seen_lengths.add(n)
        if len(top_configs) >= 3:
            break

    if best_sequence is not None:
        bn = len(best_sequence)
        if bn not in seen_lengths:
            top_configs.append((bn, best_sequence.copy()))

    remaining_for_p2 = deadline - time.monotonic() - 350.0
    remaining_for_p2 = max(100.0, remaining_for_p2)
    t_per_deep = max(30.0, remaining_for_p2 / max(1, len(top_configs)))

    for i, (n, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 350:
            break
        t_alloc = min(t_per_deep, deadline - time.monotonic() - 330)
        if t_alloc < 15:
            break

        # Combined LP + smooth gradient
        a_opt, obj_opt = combined_optimize(a0, time_limit=t_alloc, verbose=True)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Deep {i} (n={n}): {score:.6f}")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")

    # ===== Phase 3: Fine Polish =====
    print("=== Phase 3: Fine Polish ===")
    t_start3 = time.monotonic()

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 100.0
        if remaining > 30:
            n_best = len(best_sequence)

            # Multiple polish rounds with different settings
            polish_configs = [
                {'tau_init': 0.05, 'tau_end': 0.001, 'lr_init': 0.003, 'momentum': 0.95},
                {'tau_init': 0.02, 'tau_end': 0.0005, 'lr_init': 0.002, 'momentum': 0.97},
                {'tau_init': 0.1, 'tau_end': 0.002, 'lr_init': 0.005, 'momentum': 0.93},
            ]

            t_per_polish = remaining / (len(polish_configs) + 2.0)

            for ci, cfg in enumerate(polish_configs):
                if time.monotonic() > deadline - 80:
                    break
                t_alloc = min(t_per_polish, deadline - time.monotonic() - 60)
                if t_alloc < 10:
                    break

                a_opt, obj_opt = smooth_gradient_descent(
                    best_sequence, time_limit=t_alloc, **cfg
                )
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Polish {ci}: {score:.6f}")

                # Short LP refinement after polish
                if time.monotonic() < deadline - 50:
                    t_lp = min(20.0, deadline - time.monotonic() - 40)
                    if t_lp > 8:
                        a_opt2, obj_opt2 = lp_optimize(best_sequence, time_limit=t_lp)
                        score2 = float(evaluate_sequence(a_opt2.tolist()))
                        if np.isfinite(score2) and score2 < best_score:
                            best_score = score2
                            best_sequence = a_opt2.copy()
                            print(f"  LP after Polish {ci}: {score:.6f}")

    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")

    # ===== Phase 4: Final Perturb + Re-optimize =====
    print("=== Phase 4: Final Refinement ===")
    t_start4 = time.monotonic()

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 10.0
        if remaining > 20:
            n_best = len(best_sequence)
            n_perturb = max(2, int(remaining / 25))
            t_per = remaining / n_perturb

            for pi in range(n_perturb):
                if time.monotonic() > deadline - 10:
                    break
                if t_per < 8:
                    break

                # Structured perturbation
                scale = 0.002 * (1.0 + 0.2 * pi)
                noise = rng.standard_normal(n_best) * scale
                # Add smooth component
                smooth_comp = 0.001 * np.sin(2 * np.pi * np.arange(n_best) / n_best * (pi + 1))
                a_pert = best_sequence + noise + smooth_comp
                a_pert = np.maximum(a_pert, 0.0)
                s = a_pert.sum()
                if s > 1e-15:
                    a_pert = a_pert / s

                t_alloc = min(t_per, deadline - time.monotonic() - 5)
                if t_alloc < 6:
                    break

                if pi % 2 == 0:
                    a_opt, obj_opt = combined_optimize(a_pert, time_limit=t_alloc)
                else:
                    a_opt, obj_opt = smooth_gradient_descent(
                        a_pert, time_limit=t_alloc,
                        tau_init=0.03, tau_end=0.001, lr_init=0.003, momentum=0.95
                    )

                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Final {pi}: {score:.6f}")

    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")

    # Fallback
    if best_sequence is None or not np.isfinite(best_score):
        best_sequence = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final check
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
        best_sequence = best_sequence.copy()

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
