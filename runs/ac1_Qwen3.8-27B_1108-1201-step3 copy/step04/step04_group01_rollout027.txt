import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_gradient(weights, f):
    """Compute gradient of LSE objective: 2 * cross_correlation(weights, f)."""
    n = len(f)
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]


def build_lp_matrix(f):
    """Build sparse constraint matrix for FW-LP."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend(f[k - js].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M_budget, A_cache):
    """Solve LP for FW direction: max sum(g) s.t. A@g <= M_budget, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M_budget)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 15})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

        # Adam update
        t_float = float(t)
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * grad ** 2
        m_hat = m / (1 - 0.9 ** t_float)
        v_hat = v / (1 - 0.999 ** t_float)
        update = m_hat / (np.sqrt(v_hat) + 1e-8)

        # Cosine annealing learning rate
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        lr_t = lr * 0.5 * (1 + np.cos(np.pi * progress))
        if no_improve > 30:
            lr_t *= 0.8

        f_new = project_simplex(f - lr_t * update)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                f = best_f.copy()
                m = np.zeros(n)
                v = np.zeros(n)
                no_improve = 0
                lr = min(lr * 1.2, 0.05)
            elif no_improve > 50:
                lr *= 0.95

        # Ramp up beta
        beta = min(beta * beta_decay, beta_end)

        if t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Adam-LSE it={t}, obj={best_obj:.10f}, beta={beta:.1f}, elapsed={elapsed:.0f}s")

    return best_f, best_obj


def fw_lp_optimize(f0, time_limit, top_peaks=3):
    """Frank-Wolfe with LP oracle using multiple active peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        # Build LP with multiple peak constraints
        A_cache = build_lp_matrix(f)
        # Use top peaks as constraints
        peak_indices = np.argsort(conv)[::-1][:top_peaks]
        M_budget = M * 1.01

        g = solve_lp_direction(f, M_budget, A_cache)
        if g is None:
            no_improve += 1
            if no_improve > 20:
                # Perturb and restart
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                no_improve = 0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 2
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        alphas = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
        for alpha in alphas:
            ft = np.maximum((1.0 - alpha) * f + alpha * g, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(6):
                mid = (lo + hi) / 2.0
                ft = np.maximum((1.0 - mid) * f + mid * g, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        f = np.maximum((1.0 - best_alpha) * f + best_alpha * g, 0.0)
        f = f / f.sum()
        conv = fast_convolve(f)
        M = conv.max()

        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 30:
            rng = np.random.default_rng(it * 7919)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0

    return best_f, best_obj


def peak_shaving(f0, time_limit,n_steps_per_round=50):
    """Local search that specifically targets the maximum peak for reduction."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        max_k = np.argmax(conv)
        max_val = conv[max_k]

        # Find coordinates that contribute most to the peak
        # Peak at k: C[k] = sum_j a[j]*a[k-j]
        # Contribution of a[i] to C[k] is a[i]*a[k-i]
        contributions = np.zeros(n)
        for i in range(n):
            j = max_k - i
            if 0 <= j < n:
                contributions[i] = f[i] * f[j]
            else:
                contributions[i] = 0.0

        # Try to reduce contribution by redistributing mass
        sorted_idx = np.argsort(contributions)[::-1]
        improved = False

        for _ in range(n_steps_per_round):
            if time.monotonic() - t0 > time_limit:
                break

            # Pick a high-contribution coordinate to reduce
            i = sorted_idx[rng.integers(0, min(20, n))]
            delta = f[i] * rng.uniform(0.01, 0.05)
            if delta > f[i] * 0.5:
                delta = f[i] * 0.1

            # Find a low-contribution coordinate to increase (far from peak)
            # Prefer coordinates where a[max_k - j] is small
            target_scores = np.zeros(n)
            for j in range(n):
                jj = max_k - j
                if 0 <= jj < n:
                    target_scores[j] = f[jj]
                else:
                    target_scores[j] = -1.0
            target_scores[i] = -1.0  # Don't pick same coordinate
            # Want low target_scores (small a[max_k-j])
            candidates = np.where(target_scores >= 0)[0]
            if len(candidates) == 0:
                break
            # Weight by inverse of target score (prefer low)
            weights = 1.0 / (target_scores[candidates] + 1e-8)
            weights /= weights.sum()
            j = candidates[rng.choice(len(candidates), p=weights)]

            f_try = f.copy()
            f_try[i] -= delta
            f_try[j] += delta
            f_try = np.maximum(f_try, 0.0)
            st = f_try.sum()
            f_try = f_try / st

            conv_try = fast_convolve(f_try)
            obj_try = conv_try.max()

            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_f = f_try.copy()
                f = f_try
                improved = True

        if not improved:
            # Try random perturbations
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            conv_test = fast_convolve(f)
            if conv_test.max() > best_obj * 1.01:
                f = best_f.copy()

    return best_f, best_obj


def gen_binary_start(n, seed):
    """Generate a binary sequence with good autocorrelation properties."""
    rng = np.random.default_rng(seed)
    # Use a simple LCG-based pseudo-random binary sequence
    weight = n // 2 + rng.integers(0, 3)
    idx = rng.choice(n, size=weight, replace=False)
    a = np.zeros(n)
    a[idx] = 1.0
    return a


def gen_tapered_random(n, seed):
    """Generate a random sequence with tapered ends."""
    rng = np.random.default_rng(seed)
    a = rng.random(n)
    # Apply raised-cosine taper
    taper = np.ones(n)
    taper_len = max(1, n // 8)
    for i in range(taper_len):
        taper[i] = 0.5 * (1 - np.cos(np.pi * i / taper_len))
        taper[n-1-i] = 0.5 * (1 - np.cos(np.pi * i / taper_len))
    a *= taper
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 0:
        a /= s
    return a


def gen_exponential_start(n, seed, rate_range=(0.01, 0.15)):
    """Generate an exponentially decaying/growing sequence."""
    rng = np.random.default_rng(seed)
    rate = rng.uniform(*rate_range)
    direction = rng.choice([-1, 1])
    a = np.exp(direction * rate * np.arange(n))
    a = np.maximum(a, 1e-10)
    s = a.sum()
    a /= s
    return a


def try_and_track(a, best_tracker):
    """Evaluate and update best tracker."""
    a = np.maximum(a, 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"    NEW BEST: {score:.8f} (n={len(a)})")
    return score


def optimize_candidate(f0, time_budget):
    """Run multi-phase optimization on a single candidate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t_total_start = time.monotonic()

    # Phase 1: Quick FW-LP for structure
    t1 = min(time_budget * 0.25, 120)
    if t1 > 15:
        result_f, result_obj = fw_lp_optimize(f, time_limit=t1, top_peaks=3)
        if result_obj < best_obj:
            best_obj = result_obj
            best_f = result_f

    # Phase 2: Adam-LSE main optimization
    t_rem = time_budget - (time.monotonic() - t_total_start)
    t2 = min(t_rem * 0.5, 250)
    if t2 > 15:
        result_f, result_obj = adam_lse_optimize(
            best_f, time_limit=t2,
            beta_start=8.0, beta_end=20000.0,
            lr_init=0.008)
        if result_obj < best_obj:
            best_obj = result_obj
            best_f = result_f

    # Phase 3: FW-LP refinement
    t_rem = time_budget - (time.monotonic() - t_total_start)
    t3 = min(t_rem * 0.25, 100)
    if t3 > 15:
        result_f, result_obj = fw_lp_optimize(best_f, time_limit=t3, top_peaks=5)
        if result_obj < best_obj:
            best_obj = result_obj
            best_f = result_f

    # Phase 4: High-beta Adam-LSE polish
    t_rem = time_budget - (time.monotonic() - t_total_start)
    t4 = min(t_rem * 0.3, 150)
    if t4 > 10:
        result_f, result_obj = adam_lse_optimize(
            best_f, time_limit=t4,
            beta_start=50.0, beta_end=80000.0,
            lr_init=0.003)
        if result_obj < best_obj:
            best_obj = result_obj
            best_f = result_f

    # Phase 5: Peak shaving
    t_rem = time_budget - (time.monotonic() - t_total_start)
    t5 = min(t_rem * 0.4, 80)
    if t5 > 10:
        result_f, result_obj = peak_shaving(best_f, time_limit=t5)
        if result_obj < best_obj:
            best_obj = result_obj
            best_f = result_f

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-start optimization with Adam-LSE, FW-LP, and peak shaving.
    Tries multiple initializations and lengths to find the best sequence.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    best_tracker = [float('inf'), None]
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}")
            try_and_track(parent, best_tracker)
    except Exception as e:
        print(f"No parent: {e}")

    if parent is None:
        parent = gen_tapered_random(448, 42)
        try_and_track(parent, best_tracker)

    n_parent = len(parent)

    # Define starting candidates
    candidates = []

    # 1. Parent
    candidates.append(("parent", parent.copy()))

    # 2. Tapered random (same length)
    candidates.append(("tapered_rand", gen_tapered_random(n_parent, seed + 1)))

    # 3. Exponential decay
    candidates.append(("exp_decay", gen_exponential_start(n_parent, seed + 2)))

    # 4. Binary sequence
    candidates.append(("binary", gen_binary_start(n_parent, seed + 3)))

    # 5. Different length: n_parent +/- some
    alt_n1 = min(512, n_parent + 64)
    if abs(alt_n1 - n_parent) > 10:
        x_old = np.linspace(0, 1, n_parent)
        x_new = np.linspace(0, 1, alt_n1)
        interp = np.interp(x_new, x_old, parent)
        interp = np.maximum(interp, 0)
        if interp.sum() > 1e-10:
            interp /= interp.sum()
        candidates.append((f"interp_{alt_n1}", interp))

    # 6. Different length: smaller
    alt_n2 = max(256, n_parent - 64)
    if abs(alt_n2 - n_parent) > 10:
        x_old = np.linspace(0, 1, n_parent)
        x_new = np.linspace(0, 1, alt_n2)
        interp2 = np.interp(x_new, x_old, parent)
        interp2 = np.maximum(interp2, 0)
        if interp2.sum() > 1e-10:
            interp2 /= interp2.sum()
        candidates.append((f"interp_{alt_n2}", interp2))

    # Time allocation: give parent and best candidates more time
    # Sort candidates by initial objective (best first)
    def sort_key(item):
        name, a = item
        conv = fast_convolve(a)
        return conv.max()

    candidates.sort(key=sort_key)

    # Allocate time: first 2 candidates get 40% each, rest share 20%
    base_time = (float(budget_s) - 100) / len(candidates)
    times = [base_time] * len(candidates)
    times[0] = min(times[0] * 2.0, 600)
    times[1] = min(times[1] * 1.5, 400)

    print(f"\nCandidates: {len(candidates)}, total budget: {budget_s}s")
    for name, _ in candidates:
        print(f"  - {name}")

    # Optimize each candidate
    for idx, (name, f_init) in enumerate(candidates):
        elapsed = time.monotonic() + (1800 - budget_s)
        remaining = deadline - time.monotonic()
        if remaining < 30:
            print(f"  [SKIP {name}] Not enough time")
            break

        t_alloc = min(times[idx], remaining - 30 * (len(candidates) - idx))
        if t_alloc < 25:
            print(f"  [SKIP {name}] Time too short ({t_alloc:.0f}s)")
            continue

        print(f"\n  Optimizing {name} (n={len(f_init)}, budget={t_alloc:.0f}s)...")
        t_start = time.monotonic()

        try:
            result_f, result_obj = optimize_candidate(f_init, time_budget=t_alloc)
            try_and_track(result_f, best_tracker)
            print(f"    Done {name}: obj={result_obj:.10f}, time={time.monotonic()-t_start:.0f}s")
        except Exception as e:
            print(f"    Error in {name}: {e}")

    # Final polish on the best found
    if best_tracker[1] is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            print(f"\n  Final polish: {remaining:.0f}s remaining")
            best_f = best_tracker[1].copy()

            # Very high beta LSE
            t_pol = min(remaining * 0.5, 200)
            if t_pol > 20:
                result_f, _ = adam_lse_optimize(
                    best_f, time_limit=t_pol,
                    beta_start=100.0, beta_end=100000.0,
                    lr_init=0.002)
                try_and_track(result_f, best_tracker)

            # Peak shaving
            remaining2 = deadline - time.monotonic()
            if remaining2 > 20:
                t_pk = min(remaining2 * 0.6, 100)
                if t_pk > 10:
                    result_f, _ = peak_shaving(best_tracker[1], time_limit=t_pk)
                    try_and_track(result_f, best_tracker)

    # Final verification
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.8f}, n={len(final_seq)}")
            return [float(max(0.0, x)) for x in final_seq]

    # Fallback
    fallback = np.ones(384) / 384
    print(f"\nFALLBACK: n=384")
    return [float(x) for x in fallback]
