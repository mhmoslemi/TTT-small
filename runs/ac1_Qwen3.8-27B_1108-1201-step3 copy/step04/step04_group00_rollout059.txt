```python
def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
```

```python
    A = np.fft.rfft(a, fft_size)
```

```python
def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]

    cond = u * np.arange(1, n + 1) > (cssv - 1)

    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def make_gaussian(n, center_frac, width_frac):
    """Create a Gaussian-shaped sequence."""
    x = np.arange(n, dtype=np.float64)
    center = center_frac * n
    width = width_frac * n
    a = np.exp(-0.5 * ((x - center) / width) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_raised_cosine(n):
    """Raised cosine window."""
    if n <= 1:
        return np.ones(max(n, 1)) / max(n, 1)
    x = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n - 1)))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_flat_top(n, taper_frac):
    """Flat-top with tapered edges."""
    a = np.ones(n, dtype=np.float64)
    taper = max(1, int(taper_frac * n))
    for i in range(taper):
        val = 0.5 * (1.0 + np.cos(np.pi * i / (taper + 1)))
        a[i] = val
        a[n - 1 - i] = val
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_two_blobs(n, blob_frac, sep_frac):
    """Two Gaussian blobs."""
    x = np.arange(n, dtype=np.float64)
    w = blob_frac * n
    c1 = (0.5 - sep_frac) * n
    c2 = (0.5 + sep_frac) * n
    a = np.exp(-0.5 * ((x - c1) / w) ** 2) + np.exp(-0.5 * ((x - c2) / w) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive M and restarts."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        g = solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.15
            if m_exp > 6.0:
                m_exp = 1.0
                # Reset with noise
                rng = np.random.default_rng(it * 13 + 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                conv = fast_convolve(f)
                M = conv.max()
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-11:
            no_improve += 1
            if no_improve > 60:
                break
            m_exp = min(m_exp * 1.06, 4.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M

        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Bisection refine
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.15)
            hi = min(1.0, best_alpha + 0.15)
            for _ in range(12):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_exp = 1.0
            f = f_new

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"      LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)

        if no_improve > 70:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.003 + 0.003 * min(no_improve // 70, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def peak_hammering(f0, time_limit, top_k=10):
    """Targeted peak reduction: repeatedly push down the top peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        current_max = conv.max()

        if current_max < best_obj:
            best_obj = current_max
            best_f = f.copy()

        # Get top-k peaks
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        peaks = peaks[np.argsort(conv[peaks])[::-1]]

        # For each peak, compute subgradient and take a step
        improved = False
        for k in peaks[:5]:
            if time.monotonic() - t0 > time_limit:
                break

            # Subgradient of conv[k] w.r.t. a
            sg = np.zeros(n)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] = f[k - idx]

            # Try reducing in the -sg direction
            for step in [0.01, 0.005, 0.002, 0.001]:
                f_trial = f - step * sg
                f_trial = project_simplex(f_trial)
                val = fast_convolve(f_trial).max()
                if val < current_max - 1e-15:
                    f = f_trial
                    current_max = val
                    improved = True
                    break

        if not improved:
            # Try coordinate descent on the peak-most elements
            peak_idx = int(peaks[0])
            js = max(0, peak_idx - n + 1)
            je = min(n - 1, peak_idx)

            for j in range(js, je + 1):
                if time.monotonic() - t0 > time_limit:
                    break
                if f[j] < 1e-10:
                    continue

                old = f[j]
                for frac in [0.5, 0.7, 0.85, 1.2, 1.5]:
                    new = old * frac
                    if new < 0:
                        continue
                    f_trial = f.copy()
                    f_trial[j] = new
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    denom = 1.0 - old
                    if denom < 1e-10:
                        continue
                    f_trial[mask] *= (1.0 - new) / denom
                    f_trial = f_trial / f_trial.sum()
                    val = fast_convolve(f_trial).max()
                    if val < current_max - 1e-15:
                        f = f_trial
                        current_max = val
                        improved = True
                        break

        if not improved:
            # Small noise to escape
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)

    return best_f, best_obj


def sg_refine(f0, time_limit, lr=0.003, mom=0.93, top_k=30, temp=0.0001):
    """Subgradient refinement with softmax-weighted peaks and Nesterov momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    y = f.copy()  # Nesterov lookahead
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        beta = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * it * it))

        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += w[i] * y[k - idx]

        # Nesterov update
        y_new = project_simplex(y - lr * sg)
        v_new = (it / beta) * (y_new - y)
        f_new = project_simplex(y_new - v_new / (beta - 1.0) if beta > 1 else y_new)

        f = f_new
        y = y_new
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve % 200 == 199:
            lr *= 0.95
        if no_improve > 800:
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            y = f.copy()
            v = np.zeros(n)
            conv = fast_convolve(f)
            no_improve = 0
            lr = max(lr, 0.001)

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def coordinate_descent_refine(f0, time_limit):
    """Fast coordinate descent that only modifies elements near the peak."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        current_max = conv.max()
        peak_k = int(np.argmax(conv))

        if current_max < best_obj:
            best_obj = current_max
            best_f = f.copy()

        # Only modify elements that contribute to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)

        improved_this_pass = False
        for j in range(js, je + 1):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-12:
                continue

            old = f[j]
            best_val = current_max
            best_frac = 1.0

            for frac in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                new = old * frac
                if new < 0 or new > 0.99:
                    continue
                denom = 1.0 - old
                if denom < 1e-10:
                    continue
                f_trial = f.copy()
                f_trial[j] = new
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= (1.0 - new) / denom
                f_trial = f_trial / f_trial.sum()
                val = fast_convolve(f_trial).max()
                if val < best_val:
                    best_val = val
                    best_frac = frac

            if best_frac != 1.0:
                new = old * best_frac
                denom = 1.0 - old
                if denom > 1e-10:
                    f[j] = new
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= (1.0 - new) / denom
                    f = f / f.sum()
                current_max = best_val
                improved_this_pass = True

        if not improved_this_pass:
            # Try a few random coordinates
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999983)
            for _ in range(5):
                j = rng.integers(0, n)
                if f[j] < 1e-12:
                    continue
                old = f[j]
                for frac in [0.7, 0.85, 1.15, 1.3]:
                    new = old * frac
                    if new < 0:
                        continue
                    denom = 1.0 - old
                    if denom < 1e-10:
                        continue
                    f_trial = f.copy()
                    f_trial[j] = new
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f_trial[mask] *= (1.0 - new) / denom
                    f_trial = f_trial / f_trial.sum()
                    val = fast_convolve(f_trial).max()
                    if val < current_max - 1e-15:
                        f = f_trial
                        current_max = val
                        improved_this_pass = True
                        break
            if not improved_this_pass:
                break

    return best_f, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining diverse initializations, LP-FW,
    peak hammering, subgradient refinement, and coordinate descent.
    Explores multiple sequence lengths to find the global optimum.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent sequence if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    # Phase 1: LP-FW on parent (if available) or good initializations
    print("=== Phase 1: Initial LP-FW ===")
    if parent is not None:
        rem = deadline - time.monotonic() - 600
        if rem > 60:
            print(f"  LP-FW on parent (n={len(parent)}), t={min(rem, 250):.0f}s")
            a_opt, _ = lp_fw_optimize(parent, time_limit=min(rem, 250), verbose=True)
            try_seq(a_opt, "lpfw_parent")
    else:
        # No parent: try multiple initializations at n=768
        n = 768
        inits = [
            make_gaussian(n, 0.5, 0.25),
            make_raised_cosine(n),
            make_flat_top(n, 0.2),
        ]
        for i, init in enumerate(inits):
            rem = deadline - time.monotonic() - 500
            if rem < 30:
                break
            print(f"  LP-FW init {i} (n={n}), t={min(rem, 150):.0f}s")
            a_opt, _ = lp_fw_optimize(init, time_limit=min(rem, 150), verbose=True)
            try_seq(a_opt, f"lpfw_init{i}")

    # Phase 2: Multi-length exploration
    print(f"\n=== Phase 2: Multi-length (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else np.ones(768) / 768

    target_lengths = [512, 768, 1024, 1280, 1536]
    # Filter: only try lengths different from what we already have
    base_n = len(base)
    target_lengths = [t for t in target_lengths if abs(t - base_n) > 64]
    if not target_lengths:
        target_lengths = [512, 1024]

    for target_n in target_lengths:
        rem = deadline - time.monotonic() - 200
        if rem < 50:
            break
        f_interp = interpolate_sequence(base, target_n)
        t_alloc = min(rem * 0.4, 180)
        print(f"  LP-FW n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = lp_fw_optimize(f_interp, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, f"lpfw_n{target_n}")

    # Phase 3: Diverse initializations at best length
    print(f"\n=== Phase 3: Diverse inits (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        inits = [
            make_gaussian(n_best, 0.5, 0.2),
            make_gaussian(n_best, 0.5, 0.35),
            make_raised_cosine(n_best),
            make_flat_top(n_best, 0.15),
            make_flat_top(n_best, 0.3),
            make_two_blobs(n_best, 0.15, 0.15),
            make_two_blobs(n_best, 0.2, 0.2),
        ]
        # Also try perturbed best
        inits.append(project_simplex(best_seq + rng.standard_normal(n_best) * 0.01))
        inits.append(0.5 * best_seq + 0.5 * best_seq[::-1])
        inits[-1] = project_simplex(inits[-1])

        for i, init in enumerate(inits):
            rem = deadline - time.monotonic() - 150
            if rem < 25:
                break
            t_alloc = min(rem * 0.3, 100)
            a_opt, _ = lp_fw_optimize(init, time_limit=t_alloc)
            try_seq(a_opt, f"diverse_{i}")

    # Phase 4: Deep LP-FW on current best
    print(f"\n=== Phase 4: Deep LP-FW (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(3):
            rem = deadline - time.monotonic() - 120
            if rem < 20:
                break
            t_alloc = min(rem * 0.4, 150)

            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.003)
            else:
                f_init = project_simplex(0.8 * best_seq + 0.2 * (np.ones(n_best) / n_best))

            a_opt, _ = lp_fw_optimize(f_init, time_limit=t_alloc, verbose=(p == 0))
            try_seq(a_opt, f"deep_{p}")

    # Phase 5: Peak hammering
    print(f"\n=== Phase 5: Peak Hammering (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 20:
            t_ph = min(rem * 0.6, 120)
            print(f"  Peak hammering t={t_ph:.0f}s")
            a_opt, _ = peak_hammering(best_seq, time_limit=t_ph, top_k=15)
            try_seq(a_opt, "peak_hammer")

    # Phase 6: Subgradient refinement
    print(f"\n=== Phase 6: SG Refine (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 60
        if rem > 15:
            t_sg = min(rem * 0.5, 80)
            print(f"  SG refine t={t_sg:.0f}s")
            a_opt, _ = sg_refine(best_seq, time_limit=t_sg,
                                  lr=0.002, mom=0.93, top_k=max(20, n_best // 6),
                                  temp=0.00005)
            try_seq(a_opt, "sg_refine")

    # Phase 7: Coordinate descent final polish
    print(f"\n=== Phase 7: CD Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 40
        if rem > 15:
            t_cd = min(rem * 0.6, 80)
            print(f"  CD t={t_cd:.0f}s")
            a_opt, _ = coordinate_descent_refine(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd_polish")

    # Phase 8: Final combined polish
    print(f"\n=== Phase 8: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        rem = deadline - time.monotonic() - 30
        if rem > 15:
            a_opt, _ = peak_hammering(best_seq, time_limit=min(rem, 40), top_k=5)
            try_seq(a_opt, "final_hammer")

        rem = deadline - time.monotonic() - 20
        if rem > 10:
            a_opt, _ = coordinate_descent_refine(best_seq, time_limit=min(rem, 30))
            try_seq(a_opt, "final_cd")

        rem = deadline - time.monotonic() - 15
        if rem > 10:
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=min(rem, 25))
            try_seq(a_opt, "final_lpfw")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  Fallback: {best_score:.6f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
