def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def simplex_project(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum} using Duchi et al. algorithm."""
    n = len(v)
    v = v.astype(np.float64).copy()
    u = np.sort(v)[::-1]
    cumsum = np.cumsum(u)
    rho_idx = np.where(u > (cumsum - target_sum) / np.arange(1, n + 1))[0]
    if len(rho_idx) == 0:
        return np.full(n, target_sum / n)
    rho = rho_idx[-1]
    theta = (cumsum[rho] - target_sum) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def compute_weights(conv, beta, top_k=None):
    """Compute softmax weights over convolution values."""

def lp_step(a, conv, n):
    """
    LP-based direction finding: max sum(g) s.t. conv(a,g) <= M, g >= 0.
    Returns (direction, success).
    """
    M = conv.max()
    
    # Build convolution matrix C such that C @ g = conv(a, g)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M * 1.00001)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True,
                                                   'time_limit': 10})
        if result.success:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                d = g - a
                if np.linalg.norm(d) > 1e-10:
                    return d, True
    except Exception:
        pass
    return None, False

    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    v = np.zeros(n)
    momentum = 0.9
    lr = 0.02
    no_improve = 0
    iteration = 0
    rng = np.random.default_rng(777)
    
    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break
        
        # Nesterov lookahead
        y = a + momentum * v
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y
        
        # Compute subgradient at y using top peaks
        conv_y = fast_convolve(y)
        top_k = min(30, len(conv_y))
        peak_idx = np.argpartition(conv_y, -top_k)[-top_k:]
        peak_vals = conv_y[peak_idx]
        
        # Soft weights over top peaks
        temp = 0.001
        scaled = peak_vals / temp - peak_vals.max() / temp
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum > 1e-300:
            w /= w_sum
        else:
            w = np.zeros(len(peak_idx))
            w[0] = 1.0
        
        # Gradient: sum of weighted correlation contributions
        grad = np.zeros(n)
        for idx, k in enumerate(peak_idx):
            if w[idx] < 1e-12:
                continue
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idxs = np.arange(j_start, j_end + 1)
            grad[idxs] += 2.0 * w[idx] * y[k - idxs]
        
        # Update
        v = momentum * v - lr * grad
        a_new = a + v
        a_new = simplex_project(a_new, 1.0)
        
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 5:
                lr *= 0.9
                a = best_a.copy()
                v = np.zeros(n)
        
        a = a_new
        momentum = min(0.95, momentum + 0.0005)
        iteration += 1
        
        if verbose and iteration % 100 == 0:
            print(f"    SG iter {iteration}, obj={best_obj:.8f}, lr={lr:.5f}")
        
        if no_improve > 300:
            # Perturb
            noise = rng.standard_normal(n) * 0.005
            a = best_a + noise
            a = simplex_project(a, 1.0)
            v = np.zeros(n)
            lr = 0.02
            no_improve = 0
    
    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Sine-based
    for freq in [1, 2, 3, 5]:
        name = f'sine{freq}'
        sig = (np.sin(np.pi * freq * x / n) ** 2 + 0.01)
        inits[name] = sig / sig.sum()
    
    # Bimodal
    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()
    
    # Trimodal
    t1 = max(1, n // 6)
    tri = np.zeros(n)
    tri[:t1] = 1.0
    tri[n//3 - t1//2:n//3 + t1//2] = 1.0
    tri[-t1:] = 1.0
    inits['trimodal'] = tri / tri.sum()
    
    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.15 * n)) ** 2)
    inits['gaussian'] = g / g.sum()
    
    # Alternating
    alt = np.ones(n)
    alt[::2] = 2.0
    alt[1::2] = 0.3
    inits['alternating'] = alt / alt.sum()
    
    # Power law
    for pw in [0.5, 1.0]:
        sig = (x + 1.0) ** pw
        inits[f'power{pw}'] = sig / sig.sum()
    
    # Random frequency comb
    num_peaks = rng.integers(3, 10)
    centers = rng.integers(0, n, size=num_peaks)
    widths = rng.uniform(0.02, 0.1, size=num_peaks) * n
    comb = np.zeros(n)
    for c, w in zip(centers, widths):
        comb += np.exp(-0.5 * ((x - c) % n / w) ** 2)
    comb += 0.01
    inits['comb'] = comb / comb.sum()
    
    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps
    
    return inits


def extend_sequence(a_short, n_target):
    """Extend a shorter sequence to target length via interpolation."""
    n_short = len(a_short)
    s = a_short.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    a_norm = a_short / s
    a_ext = np.interp(np.linspace(0, 1, n_target), np.linspace(0, 1, n_short), a_norm)
    a_ext = np.maximum(a_ext, 0)
    ses = a_ext.sum()
    if ses > 1e-15:
        a_ext = a_ext / ses
    return a_ext


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Hybrid log-sum-exp gradient + LP + subgradient search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_sequence = None
    best_score = float('inf')
    
    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: n={len(p)}, sum={p.sum():.4f}")
    except NameError:
        print("No parent sequence available.")
    except Exception as e:
        print(f"Parent loading error: {e}")
    
    def update_best(a, source_name):
        nonlocal best_sequence, best_score
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score):
            if score < best_score:
                best_score = score
                best_sequence = a.copy()
                print(f"  [{source_name}] NEW BEST: {score:.6f} (n={len(a)})")
        return score
    
    # =========================================================
    # Phase 1: Multi-scale gradient exploration
    # =========================================================
    print("=== Phase 1: Multi-scale Gradient Exploration ===")
    t_start = time.monotonic()
    
    # Try multiple lengths, prioritizing n=768 (parent size)
    lengths_to_try = [512, 768, 1024]
    if parent is not None:
        # Put parent length first
        pl = len(parent)
        if pl in lengths_to_try:
            lengths_to_try.remove(pl)
            lengths_to_try = [pl] + lengths_to_try
    
    results = []
    
    for n in lengths_to_try:
        if time.monotonic() > deadline - 400:
            break
        
        print(f"  Exploring n={n}...")
        inits = generate_initializations(n, parent, rng)
        
        # Prioritize: parent, uniform, sines, random
        priority = ['parent', 'parent_interp', 'uniform', 'sine1', 'sine2',
                    'sine5', 'random', 'alternating', 'gaussian', 'bimodal']
        init_names = [nm for nm in priority if nm in inits]
        init_names += [nm for nm in inits if nm not in priority]
        
        # Allocate time per init
        max_inits = min(5, len(init_names))
        t_per_init = max(15.0, (deadline - time.monotonic() - 400) / (len(lengths_to_try) * max_inits * 2.5))
        
        for name in init_names[:max_inits]:
            if time.monotonic() > deadline - 350:
                break
            t_alloc = min(t_per_init, deadline - time.monotonic() - 300)
            if t_alloc < 10:
                break
            
            a0 = inits[name]
            # Run gradient descent with different configs
            a_opt, obj = gradient_descent_ls(a0, time_limit=t_alloc,
                                             lr_init=0.08, beta_init=30.0,
                                             beta_end=3000.0, top_k=50,
                                             use_mirror=True)
            score = update_best(a_opt, f"GD-mirror n={n} {name}")
            results.append((score, n, name, a_opt.copy()))
            
            # Also try Euclidean projection
            if time.monotonic() < deadline - 350:
                a_opt2, obj2 = gradient_descent_ls(a0, time_limit=min(t_alloc * 0.7, 60.0),
                                                   lr_init=0.03, beta_init=50.0,
                                                   beta_end=2000.0, top_k=80,
                                                   use_mirror=False)
                score2 = update_best(a_opt2, f"GD-euc n={n} {name}")
                results.append((score2, n, name + '_euc', a_opt2.copy()))
    
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")
    
    # =========================================================
    # Phase 2: LP deep optimization on top candidates
    # =========================================================
    print("=== Phase 2: LP Deep Optimization ===")
    t_start2 = time.monotonic()
    
    # Get top unique candidates
    results.sort(key=lambda x: x[0])
    top_candidates = []
    seen = set()
    for score, n, name, a_opt in results:
        if not np.isfinite(score):
            continue
        key = n  # One per length
        if key not in seen:
            top_candidates.append((n, a_opt.copy()))
            seen.add(key)
        if len(top_candidates) >= 3:
            break
    
    # Ensure best is included
    if best_sequence is not None:
        bl = len(best_sequence)
        if bl not in seen:
            top_candidates.append((bl, best_sequence.copy()))
    
    remaining = deadline - time.monotonic() - 200.0
    n_cand = max(1, len(top_candidates))
    t_per_lp = max(30.0, remaining * 0.5 / n_cand)
    
    for i, (n, a0) in enumerate(top_candidates):
        if time.monotonic() > deadline - 180:
            break
        t_alloc = min(t_per_lp, deadline - time.monotonic() - 150)
        if t_alloc < 20:
            break
        
        print(f"  LP candidate {i}: n={n}, t={t_alloc:.0f}s")
        a_opt, obj = lp_optimize(a0, time_limit=t_alloc, verbose=True)
        update_best(a_opt, f"LP-{i} n={n}")
    
    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")
    
    # =========================================================
    # Phase 3: Combined LP + GD refinement
    # =========================================================
    print("=== Phase 3: Combined Refinement ===")
    t_start3 = time.monotonic()
    
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 150.0
        if remaining > 60:
            # Alternate between LP and gradient
            a = best_sequence.copy()
            n = len(a)
            conv = fast_convolve(a)
            cur_best = best_score
            cur_best_a = best_sequence.copy()
            
            t0_3 = time.monotonic()
            cycle = 0
            while time.monotonic() - t0_3 < remaining * 0.7:
                remaining_now = (t0_3 + remaining * 0.7) - time.monotonic()
                if remaining_now < 15:
                    break
                
                # LP step (up to 30s)
                t_lp = min(30.0, remaining_now * 0.4)
                if t_lp > 10:
                    a_lp, obj_lp = lp_optimize(a, time_limit=t_lp)
                    if obj_lp < cur_best:
                        cur_best = obj_lp
                        cur_best_a = a_lp.copy()
                    a = a_lp.copy()
                
                # GD step (up to 30s)
                t_gd = min(30.0, remaining_now * 0.4)
                if t_gd > 10:
                    beta = 100.0 + 50.0 * cycle
                    a_gd, obj_gd = gradient_descent_ls(a, time_limit=t_gd,
                                                       lr_init=0.02, beta_init=beta,
                                                       beta_end=beta * 10, top_k=40,
                                                       use_mirror=True)
                    if obj_gd < cur_best:
                        cur_best = obj_gd
                        cur_best_a = a_gd.copy()
                    a = a_gd.copy()
                
                # Subgradient polish (up to 20s)
                t_sg = min(20.0, remaining_now * 0.3)
                if t_sg > 5:
                    a_sg, obj_sg = subgradient_polish(a, time_limit=t_sg)
                    if obj_sg < cur_best:
                        cur_best = obj_sg
                        cur_best_a = a_sg.copy()
                    a = a_sg.copy()
                
                cycle += 1
                if cycle > 5:
                    break
            
            if cur_best < best_score:
                best_score = cur_best
                best_sequence = cur_best_a.copy()
                print(f"  Phase 3 improved: {best_score:.6f}")
    
    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")
    
    # =========================================================
    # Phase 4: Final subgradient polishing
    # =========================================================
    print("=== Phase 4: Final Polishing ===")
    t_start4 = time.monotonic()
    
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 20.0
        if remaining > 30:
            # Multiple subgradient runs with different params
            for pi in range(3):
                if time.monotonic() > deadline - 15:
                    break
                t_alloc = min(remaining / (3 - pi) * 0.7, 60.0)
                if t_alloc < 10:
                    break
                
                a0 = best_sequence.copy()
                if pi > 0:
                    noise = rng.standard_normal(len(a0)) * (0.001 * pi)
                    a0 = np.maximum(a0 + noise, 0)
                    s = a0.sum()
                    if s > 1e-15:
                        a0 = a0 / s
                
                a_opt, obj = subgradient_polish(a0, time_limit=t_alloc, verbose=True)
                update_best(a_opt, f"SG-polish-{pi}")
        
        # Final gradient micro-polish with very high beta
        remaining = deadline - time.monotonic() - 10.0
        if remaining > 15:
            a_opt, obj = gradient_descent_ls(best_sequence, time_limit=remaining * 0.6,
                                             lr_init=0.005, beta_init=500.0,
                                             beta_end=5000.0, top_k=20,
                                             use_mirror=True)
            update_best(a_opt, "Final-GD")
    
    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")
    
    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))
    
    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    
    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
