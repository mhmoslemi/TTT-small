"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LSE-smoothed Adam optimization
with multiple restarts, adaptive beta schedules, length exploration, and
systematic perturbation polish. Replaces the previous FW-LP approach with
faster per-iteration gradient methods to allow more diverse exploration.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_lse_grad(weights, f):
    """Compute gradient of sum_k weights[k]*conv[k] w.r.t. f.
    grad[i] = 2 * sum_{m=0}^{n-1} weights[i+m] * f[m]
    Uses FFT for efficiency.
    """
    n = len(f)
    f_rev = f[::-1].copy()
    total_len = n + (2 * n - 1) - 1
    fft_size = 1 << (total_len - 1).bit_length()
    F = np.fft.rfft(f_rev, fft_size)
    W = np.fft.rfft(weights, fft_size)
    conv = np.fft.irfft(F * W, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    t0 = time.monotonic()
    no_improve = 0
    lr_current = lr

    while time.monotonic() - t0 < time_limit:
        t += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)

        # Numerically stable softmax
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()

        # Gradient via FFT
        grad = compute_lse_grad(w, f)

        # Adam update
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        step = lr_current * m_hat / (np.sqrt(v_hat) + eps)

        f_new = project_simplex(f - step)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve == 50:
                lr_current *= 0.8
            elif no_improve == 100:
                lr_current *= 0.7
            elif no_improve == 200:
                # Restart from best with fresh Adam state
                f = best_f.copy()
                m = np.zeros(n)
                v = np.zeros(n)
                t = 0
                lr_current = lr
                beta = min(beta * 2.0, beta_end)
                no_improve = 0

        # Ramp up beta
        if beta < beta_end:
            beta *= beta_rate

        if t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-Adam t={t}, obj={best_obj:.10f}, beta={beta:.0f}, lr={lr_current:.6f}, elapsed={elapsed:.0f}s")

    return best_f, best_obj


def lse_momentum_optimize(f0, time_limit, beta_start=15.0, beta_end=8000.0,
                          lr_init=None, momentum=0.92):
    """LSE-smoothed momentum optimization (alternative to Adam)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    if lr_init is None:
        lr_init = 2.5 / np.sqrt(n)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = lr_init
    mom = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()

        grad = compute_lse_grad(w, f)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm

        mom = 0.92 * mom - lr * grad
        f_new = project_simplex(f + mom)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.001, 4.0 / np.sqrt(n))
        else:
            no_improve += 1
            if no_improve >= 20:
                lr *= 0.93
            if no_improve > 150:
                f = best_f.copy()
                mom = np.zeros(n)
                lr = max(lr_init * 0.5, 0.001)
                beta = min(beta * 1.8, beta_end)
                no_improve = 0

        if beta < beta_end:
            beta *= 1.004

    return best_f, best_obj


def power_sum_optimize(f0, time_limit, p_start=8.0, p_end=200.0,
                       lr=0.005, gamma=0.93):
    """Power-sum smoothed optimization (alternative smoothing to LSE)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    p = p_start
    mom = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        # Power-sum weights: w[k] = conv[k]^(p-1) / sum_j conv[j]^(p-1)
        conv_pos = np.maximum(conv, 1e-30)
        weights = conv_pos ** (p - 1)
        w_sum = weights.sum()
        if w_sum < 1e-30:
            break
        weights = weights / w_sum

        grad = compute_lse_grad(weights, f)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm

        mom = gamma * mom - lr * grad
        f_new = project_simplex(f + mom)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                lr *= 0.94
            if no_improve > 200:
                f = best_f.copy()
                mom = np.zeros(n)
                p = min(p * 1.5, p_end)
                no_improve = 0

        if p < p_end:
            p *= 1.005

    return best_f, best_obj


def perturbation_search(f0, time_limit, rng):
    """Systematic perturbation: move mass between pairs of positions."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()

    # Identify positions with mass to donate
    active = np.where(f > 1e-8)[0]
    if len(active) == 0:
        return best_f, best_obj

    # Sort by value to find natural donors (high) and recipients (low)
    sorted_idx = np.argsort(f)
    donors = sorted_idx[-max(10, n // 10):]
    recipients = sorted_idx[:max(10, n // 10)]

    attempt = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        attempt += 1
        if no_improve > 200:
            break

        # Pick a random donor and recipient
        d = rng.integers(len(donors))
        r = rng.integers(len(recipients))
        di = int(donors[d])
        ri = int(recipients[r])

        if f[di] < 1e-10:
            continue

        # Try moving a small fraction
        frac = rng.uniform(0.01, 0.15)
        delta = f[di] * frac

        f[di] -= delta
        f[ri] += delta
        f = np.maximum(f, 0.0)
        fs = f.sum()
        if fs < 1e-15:
            f = best_f.copy()
            continue
        f = f / fs

        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            f = best_f.copy()
            no_improve += 1

    return best_f, best_obj


def random_perturb_refine(f0, time_limit, rng, scale=0.002):
    """Add random perturbations and re-optimize with short LSE."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()

    n_tries = 0
    while time.monotonic() - t0 < time_limit:
        n_tries += 1
        # Perturb
        f_pert = best_f + rng.standard_normal(n) * scale
        f_pert = project_simplex(f_pert)

        # Quick LSE refinement (short time)
        t_alloc = min(5.0, time_limit - (time.monotonic() - t0) - 5)
        if t_alloc < 1.0:
            break
        f_opt, obj = lse_adam_optimize(f_pert, time_limit=t_alloc,
                                        beta_start=50.0, beta_end=1000.0,
                                        lr=0.005, beta_rate=1.01)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_opt.copy()
            scale = max(scale * 0.95, 0.0005)
        else:
            scale = max(scale * 0.9, 0.0005)

    return best_f, best_obj


def try_sequence(a, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.maximum(a, 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase LSE-Adam optimization with:
    1. Adam-based LSE on parent
    2. Momentum-based LSE with different schedule
    3. Power-sum smoothing
    4. Length exploration
    5. Perturbation + random restart polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_tracker = [float('inf'), None]

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
            try_sequence(parent, best_tracker)
    except Exception:
        print("No parent available, using uniform fallback")
        parent = np.ones(448) / 448
        try_sequence(parent, best_tracker)

    n_parent = len(parent)
    current_best = best_tracker[1].copy()

    # === Phase 1: LSE + Adam on parent ===
    print("\n=== Phase 1: LSE-Adam on parent ===")
    t_alloc = min(300, deadline - time.monotonic() - 500)
    if t_alloc > 30:
        print(f"  Allocating {t_alloc:.0f}s")
        a_opt, obj = lse_adam_optimize(current_best, time_limit=t_alloc,
                                        beta_start=20.0, beta_end=5000.0,
                                        lr=0.01, beta_rate=1.003)
        try_sequence(a_opt, best_tracker)

    # === Phase 2: LSE + Momentum (different optimizer, different path) ===
    print("\n=== Phase 2: LSE-Momentum ===")
    current_best = best_tracker[1].copy()
    t_alloc = min(250, deadline - time.monotonic() - 400)
    if t_alloc > 30:
        print(f"  Allocating {t_alloc:.0f}s")
        a_opt, obj = lse_momentum_optimize(current_best, time_limit=t_alloc,
                                            beta_start=10.0, beta_end=10000.0,
                                            momentum=0.94)
        try_sequence(a_opt, best_tracker)

    # === Phase 3: Power-sum smoothing (different objective landscape) ===
    print("\n=== Phase 3: Power-sum ===")
    current_best = best_tracker[1].copy()
    t_alloc = min(200, deadline - time.monotonic() - 300)
    if t_alloc > 30:
        print(f"  Allocating {t_alloc:.0f}s")
        a_opt, obj = power_sum_optimize(current_best, time_limit=t_alloc,
                                         p_start=6.0, p_end=300.0,
                                         lr=0.004, gamma=0.93)
        try_sequence(a_opt, best_tracker)

    # === Phase 4: Second Adam pass with higher beta start ===
    print("\n=== Phase 4: LSE-Adam (high beta) ===")
    current_best = best_tracker[1].copy()
    t_alloc = min(250, deadline - time.monotonic() - 200)
    if t_alloc > 30:
        print(f"  Allocating {t_alloc:.0f}s")
        a_opt, obj = lse_adam_optimize(current_best, time_limit=t_alloc,
                                        beta_start=100.0, beta_end=10000.0,
                                        lr=0.005, beta_rate=1.005)
        try_sequence(a_opt, best_tracker)

    # === Phase 5: Explore alternative lengths ===
    print("\n=== Phase 5: Alternative lengths ===")
    current_best = best_tracker[1].copy()
    n_best = len(current_best)

    alt_lengths = []
    for n_alt in [384, 400, 448, 480, 512, 256, 320]:
        if n_alt != n_best and n_alt not in alt_lengths:
            alt_lengths.append(n_alt)

    for n_alt in alt_lengths[:3]:
        t_remaining = deadline - time.monotonic() - 60
        if t_remaining < 40:
            break
        t_alloc = min(60, t_remaining / 2)
        if t_alloc < 20:
            break

        print(f"  Trying n={n_alt} ({t_alloc:.0f}s)")
        # Interpolate to new length
        x_old = np.linspace(0, 1, n_best)
        x_new = np.linspace(0, 1, n_alt)
        f_interp = np.interp(x_new, x_old, current_best)
        f_interp = np.maximum(f_interp, 0)
        if f_interp.sum() > 1e-10:
            f_interp = f_interp / f_interp.sum()

        a_opt, _ = lse_adam_optimize(f_interp, time_limit=t_alloc,
                                      beta_start=30.0, beta_end=3000.0,
                                      lr=0.008, beta_rate=1.004)
        try_sequence(a_opt, best_tracker)

    # === Phase 6: Perturbation search on best ===
    print("\n=== Phase 6: Perturbation search ===")
    current_best = best_tracker[1].copy()
    t_alloc = min(120, deadline - time.monotonic() - 30)
    if t_alloc > 10:
        print(f"  Allocating {t_alloc:.0f}s")
        a_opt, _ = perturbation_search(current_best, time_limit=t_alloc, rng=rng)
        try_sequence(a_opt, best_tracker)

    # === Phase 7: Random perturb + refine ===
    print("\n=== Phase 7: Random perturb + refine ===")
    current_best = best_tracker[1].copy()
    t_alloc = min(100, deadline - time.monotonic() - 20)
    if t_alloc > 15:
        print(f"  Allocating {t_alloc:.0f}s")
        a_opt, _ = random_perturb_refine(current_best, time_limit=t_alloc,
                                          rng=rng, scale=0.003)
        try_sequence(a_opt, best_tracker)

    # === Phase 8: Final high-precision LSE polish ===
    print("\n=== Phase 8: Final LSE polish ===")
    current_best = best_tracker[1].copy()
    t_alloc = min(100, deadline - time.monotonic() - 10)
    if t_alloc > 15:
        print(f"  Allocating {t_alloc:.0f}s")
        a_opt, _ = lse_adam_optimize(current_best, time_limit=t_alloc,
                                      beta_start=500.0, beta_end=20000.0,
                                      lr=0.002, beta_rate=1.008)
        try_sequence(a_opt, best_tracker)

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
        else:
            final_seq = np.ones(448) / 448
            print(f"\nFINAL fallback: n=448")
    else:
        final_seq = np.ones(448) / 448
        print(f"\nFINAL fallback: n=448")

    return [float(x) for x in final_seq]
