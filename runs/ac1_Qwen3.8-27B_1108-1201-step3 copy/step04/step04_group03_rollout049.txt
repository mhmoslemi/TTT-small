import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_grad(a, w, tau):
    """
    Compute gradient of logsumexp(conv(a,a)/tau) w.r.t. a using FFT.
    w = softmax(b/tau) where b = conv(a,a)
    Returns grad of shape (n,).
    """
    n = len(a)
    size = 2 * n - 1
    a_rev = a[::-1]
    conv_len = size + n - 1  # 3n-2
    fft_size = 1 << (conv_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    AR = np.fft.rfft(a_rev, fft_size)
    c = np.fft.irfft(W * AR, fft_size)
    # grad[i] = (2/tau) * c[n-1+i]
    return (2.0 / tau) * c[n - 1:n - 1 + n]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(x, tau):
    """Stable softmax with temperature."""
    x = x / tau
    x -= x.max()
    e = np.exp(x)
    s = e.sum()
    if s < 1e-300:
        return np.zeros_like(x)
    return e / s


def peak_deflate(a, b, top_k=3, strength=0.5):
    """
    Direct peak deflation: for the top peaks, reduce the contributing positions.
    Returns modified a.
    """
    n = len(a)
    a_new = a.copy()
    
    # Get top peak positions
    peak_idx = np.argpartition(b, -top_k)[-top_k:]
    peak_idx = peak_idx[np.argsort(b[peak_idx])[::-1]]
    
    for k in peak_idx:
        # Positions contributing to b[k]: a[j] * a[k-j] for valid j
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        contrib = a[js:je + 1] * a[k - je:k - js + 1][::-1]
        # Total contribution
        total = contrib.sum()
        if total < 1e-15:
            continue
        # Reduce positions proportional to their contribution
        reduction = strength * contrib / total * min(1.0, total / (a.sum() + 1e-15))
        a_new[js:je + 1] -= reduction * a[js:je + 1] * 0  # We'll do multiplicative
        # Actually, do a multiplicative reduction
        a_new[js:je + 1] *= (1.0 - strength * contrib / (total + 1e-15) * 0.1)
    
    # Renormalize
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    else:
        a_new = a / a.sum()
    
    return np.maximum(a_new, 0)


def grad_flow_optimize(a0, time_limit, tau_init=None, lr_init=0.01, name=""):
    """
    Main gradient flow optimizer with adaptive temperature.
    Uses logsumexp smoothing, FFT-based gradient, multiplicative updates.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    if tau_init is None:
        b_init = fast_convolve(a)
        tau_init = b_init.max() * 0.1

    tau = tau_init
    lr = lr_init
    mom = 0.85
    v = np.zeros(n)

    b = fast_convolve(a)
    best_obj = b.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    tau_min = b.max() * 0.005
    tau_max = b.max() * 0.5

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(a)
        cur_max = b.max()

        # Adapt tau: start wide, narrow down
        target_tau = best_obj * (0.01 + 0.09 * np.exp(-it / 2000.0))
        tau = max(tau_min, min(tau_max, tau * 0.9995 + target_tau * 0.0005))

        # Compute softmax weights
        w = softmax(b, tau)

        # Compute gradient via FFT
        grad = fast_grad(a, w, tau)

        # Momentum update
        v = mom * v + grad

        # Multiplicative update (Bregman / mirror descent)
        # a_new = a * exp(-lr * v)
        scale = lr * (1.0 + 0.5 * no_improve / 50.0)
        a_new = a * np.exp(-scale * v)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        b_new = fast_convolve(a_new)
        obj_new = b_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
            # Slightly increase lr on improvement
            lr = min(lr * 1.002, lr_init * 2.0)
        else:
            no_improve += 1
            a = a_new  # Still update, just not "best"
            lr = max(lr * 0.999, lr_init * 0.1)

        # Peak deflation every 50 steps if stuck
        if no_improve > 50 and it % 20 == 0:
            a_def = peak_deflate(best_a, fast_convolve(best_a), top_k=3,
                                  strength=0.3 + 0.3 * min(1.0, no_improve / 200.0))
            b_def = fast_convolve(a_def)
            if b_def.max() < best_obj:
                best_obj = b_def.max()
                best_a = a_def.copy()
                a = a_def.copy()
                no_improve = 0

        # Restart if very stuck
        if no_improve > 300:
            rng = np.random.default_rng(hash((it, n, seed_global)) % (2**32))
            noise_scale = 0.01 + 0.02 * min(1.0, no_improve / 1000.0)
            perturbed = best_a * np.exp(rng.standard_normal(n) * noise_scale)
            perturbed = np.maximum(perturbed, 0)
            a = perturbed / perturbed.sum()
            v = np.zeros(n)
            tau = best_obj * 0.2
            lr = lr_init * 0.8
            no_improve = 0

    return best_a, best_obj


def alternating_peak_opt(a0, time_limit):
    """
    Alternating optimization: alternate between reducing each top peak.
    For peak k, the gradient w.r.t. a[i] is a[k-i]. To reduce peak k,
    decrease a[i] where a[k-i] is large (i.e., where a contributes most to peak k).
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    b = fast_convolve(a)
    best_obj = b.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    lr = 0.02
    top_k = 5

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(a)
        cur_max = b.max()

        # Get top peaks
        peak_idx = np.argpartition(b, -top_k)[-top_k:]
        peak_idx = peak_idx[np.argsort(b[peak_idx])[::-1]]

        # For each top peak, compute a "blame" vector
        blame = np.zeros(n)
        for k in peak_idx:
            ks = max(0, k - n + 1)
            ke = min(n - 1, k)
            idx = np.arange(ks, ke + 1)
            # blame[i] += a[k-i] (the contribution of position i to peak k)
            # Weight by how much this peak exceeds average
            weight = max(0, b[k] - b.mean())
            blame[idx] += weight * a[k - idx]

        # Multiplicative update: reduce high-blame positions
        blame_norm = blame / (blame.max() + 1e-15)
        a_new = a * np.exp(-lr * blame_norm)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        b_new = fast_convolve(a_new)
        obj_new = b_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            no_improve += 1
            a = a_new
            lr = max(lr * 0.998, 0.002)

        if no_improve > 200:
            rng = np.random.default_rng(hash((it, n)) % (2**32))
            a = best_a * np.exp(rng.standard_normal(n) * 0.01)
            a = np.maximum(a, 0)
            a = a / a.sum()
            lr = 0.015
            no_improve = 0

    return best_a, best_obj


def fine_polish(a0, time_limit):
    """
    Very fine polish using low-temperature gradient flow.
    Focuses exclusively on the single highest peak.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    b = fast_convolve(a)
    best_obj = b.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    lr = 0.005
    tau = best_obj * 0.01
    v = np.zeros(n)

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(a)

        # Use very low temperature (essentially just the top peak)
        w = softmax(b, max(tau, b.max() * 0.005))
        grad = fast_grad(a, w, max(tau, b.max() * 0.005))

        v = 0.9 * v + grad
        a_new = a * np.exp(-lr * v)
        a_new = np.maximum(a_new, 1e-30)
        a_new = a_new / a_new.sum()

        b_new = fast_convolve(a_new)
        obj_new = b_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            no_improve += 1
            a = a_new

        if no_improve > 500:
            rng = np.random.default_rng(hash((it, 999)) % (2**32))
            a = best_a * np.exp(rng.standard_normal(n) * 0.003)
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
            lr = 0.003
            no_improve = 0

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    x = np.arange(n, dtype=np.float64)
    # Edge-weighted (favor edges)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0) ** 0.5
    inits.append(edge_w / edge_w.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())

    # Cosine
    c = np.abs(np.cos(np.pi * x / n)) + 0.02
    inits.append(c / c.sum())

    # Two-lobe
    lo = np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2)
    inits.append(lo / lo.sum())

    # Random Fourier (random phases, flat magnitude)
    phases = rng.uniform(0, 2 * np.pi, n)
    mag = np.ones(n) / np.sqrt(n)
    freq = np.fft.fftshift(mag * np.exp(1j * phases))
    a_fourier = np.abs(np.fft.ifft(np.fft.ifftshift(freq)))
    a_fourier = np.maximum(a_fourier, 0)
    if a_fourier.sum() > 1e-10:
        inits.append(a_fourier / a_fourier.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    # Sine-modulated
    sm = (1 + 0.5 * np.sin(2 * np.pi * x / n * 3)) 
    inits.append(np.maximum(sm, 0.01) / np.maximum(sm, 0.01).sum())

    return inits


seed_global = 42


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase FFT-accelerated search:
    1. Quick multi-init exploration on candidate lengths
    2. Deep gradient flow on best candidate
    3. Alternating peak optimization
    4. Fine polish
    5. Diversification with restarts
    """
    global seed_global
    seed_global = seed
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return float('inf')
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={parent_score:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    # Candidate lengths to try
    candidate_lengths = [384, 512, 768]
    if parent is not None:
        candidate_lengths.insert(0, len(parent))
    # Remove duplicates while preserving order
    seen = set()
    unique_lengths = []
    for L in candidate_lengths:
        if L not in seen and 64 <= L <= 2048:
            seen.add(L)
            unique_lengths.append(L)

    # === Phase 1: Quick multi-init exploration (~300s) ===
    print("\n=== Phase 1: Quick Exploration ===")
    t_phase1 = min(300, deadline - time.monotonic() - 1200)
    if t_phase1 < 30:
        t_phase1 = 30

    if parent is not None:
        # Quick gradient flow on parent
        t1 = min(100, t_phase1 - 50)
        if t1 > 10:
            print(f"  GradFlow on parent (n={len(parent)}), t={t1:.0f}s")
            a_opt, _ = grad_flow_optimize(parent, time_limit=t1, name="parent")
            try_seq(a_opt, "gf_parent")

    # Try different inits and lengths
    for n_try in unique_lengths:
        rem = deadline - time.monotonic() - 1000
        if rem < 20:
            break
        t_alloc = min(rem, 60)
        if t_alloc < 10:
            continue

        inits = generate_inits(n_try, parent if parent is not None else None, rng)
        best_local = None
        best_local_score = float('inf')

        for ii, init in enumerate(inits[:4]):
            rem = deadline - time.monotonic() - 900
            if rem < 10:
                break
            t_i = min(rem, 20)
            if t_i < 5:
                break
            a_opt, obj = grad_flow_optimize(init, time_limit=t_i, name=f"init_{ii}")
            sc = try_seq(a_opt, f"n{n_try}_init{ii}")
            if sc < best_local_score:
                best_local_score = sc
                best_local = a_opt

    print(f"  Phase 1 best: {best_score:.8f}")

    # === Phase 2: Deep optimization (~600s) ===
    print("\n=== Phase 2: Deep Gradient Flow ===")

    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 700
        if rem > 20:
            t_alloc = min(rem, 400)
            print(f"  Deep GradFlow (n={n_best}), t={t_alloc:.0f}s")
            a_opt, _ = grad_flow_optimize(best_seq, time_limit=t_alloc, name="deep")
            try_seq(a_opt, "deep_gf")

        # Alternating peak optimization
        rem = deadline - time.monotonic() - 400
        if rem > 20:
            t_alloc = min(rem, 200)
            print(f"  AltPeak (n={n_best}), t={t_alloc:.0f}s")
            a_opt, _ = alternating_peak_opt(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "alt_peak")

    # Try a different length if time permits
    rem = deadline - time.monotonic() - 350
    if rem > 30 and len(unique_lengths) > 1:
        # Pick a length we haven't deeply optimized
        tried_n = len(best_seq) if best_seq is not None else 0
        alt_n = None
        for L in unique_lengths:
            if L != tried_n:
                alt_n = L
                break
        if alt_n is not None and rem > 30:
            print(f"  Trying alt length n={alt_n}")
            inits = generate_inits(alt_n, best_seq, rng)
            t_alloc = min(rem, 100)
            for ii, init in enumerate(inits[:3]):
                rem = deadline - time.monotonic() - 250
                if rem < 10:
                    break
                a_opt, _ = grad_flow_optimize(init, time_limit=min(rem, 40), name=f"alt_{ii}")
                try_seq(a_opt, f"alt_n{alt_n}_{ii}")

    print(f"  Phase 2 best: {best_score:.8f}")

    # === Phase 3: Fine polish (~300s) ===
    print("\n=== Phase 3: Fine Polish ===")

    if best_seq is not None:
        n_best = len(best_seq)

        # Fine gradient flow with very low temperature
        rem = deadline - time.monotonic() - 150
        if rem > 15:
            t_alloc = min(rem, 150)
            print(f"  Fine Polish, t={t_alloc:.0f}s")
            a_opt, _ = fine_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "fine_polish")

        # One more pass of alternating peaks
        rem = deadline - time.monotonic() - 100
        if rem > 15:
            t_alloc = min(rem, 80)
            print(f"  AltPeak finish, t={t_alloc:.0f}s")
            a_opt, _ = alternating_peak_opt(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "alt_finish")

        # Tiny perturbations + quick refine
        rem = deadline - time.monotonic() - 50
        if rem > 15:
            for _ in range(4):
                rem = deadline - time.monotonic() - 30
                if rem < 10:
                    break
                noise = rng.standard_normal(n_best) * 0.003
                perturbed = np.maximum(best_seq * np.exp(noise), 0)
                perturbed = perturbed / perturbed.sum()
                a_opt, _ = fine_polish(perturbed, time_limit=min(rem, 20))
                try_seq(a_opt, "perturb_fine")

    print(f"  Phase 3 best: {best_score:.8f}")

    # === Phase 4: Final diversification (remaining time) ===
    print("\n=== Phase 4: Final Diversification ===")

    if best_seq is not None:
        n_best = len(best_seq)
        while time.monotonic() < deadline - 30:
            rem = deadline - time.monotonic() - 20
            if rem < 10:
                break

            # Try a random perturbation + short optimize
            noise_scale = 0.005 + 0.01 * rng.random()
            perturbed = np.maximum(best_seq * np.exp(rng.standard_normal(n_best) * noise_scale), 0)
            s = perturbed.sum()
            if s < 1e-15:
                break
            perturbed = perturbed / s

            method = rng.random()
            if method < 0.5:
                a_opt, _ = grad_flow_optimize(perturbed, time_limit=min(rem, 30), name="div_gf")
            else:
                a_opt, _ = alternating_peak_opt(perturbed, time_limit=min(rem, 30))
            try_seq(a_opt, "diversify")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
