from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def fast_cross_corr(w, a):
    """Compute cross-correlation: (w ⋆ a)[m] = sum_j w[j+m]*a[j]"""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    # Cross-correlation via FFT: ifft(conj(W) * A)
    result = np.fft.irfft(np.conj(W) * A, fft_size)[:size]
    return result

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def softmax(z):

    exp_z = np.exp(z - z_max)
    return exp_z / exp_z.sum()

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LSE-smoothed gradient with Adam optimizer in softmax(z) parameterization
2. Nesterov-accelerated projected gradient on the simplex
3. LP-based Frank-Wolfe oracle for global direction finding
4. Multi-scale exploration with diverse structured initializations
5. Adaptive temperature annealing and periodic restarts

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_cross_corr(w, a):
    """Cross-correlation: result[m] = sum_j w[j+m]*a[j] for valid j."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    result = np.fft.irfft(np.conj(W) * A, fft_size)[:size]
    return result


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_func(z):
    """Numerically stable softmax."""
    z_max = z.max()
    exp_z = np.exp(z - z_max)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def compute_lse_grad_a(a, conv, temperature):
    """
    Compute gradient of LSE(conv, temperature) w.r.t. a.
    Returns grad_a of length n.
    """
    n = len(a)
    scaled = conv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
    else:
        w /= wsum

    # grad_LSE_da[m] = sum_k w[k] * (2 - [k==2m]) * a[k-m]
    # = 2 * cross_corr(w, a)[m] - w[2m]*a[m] (diagonal correction)
    cc = fast_cross_corr(w, a)
    grad_a = 2.0 * cc[:n]
    # Diagonal correction: for m=0..n-1, k=2m is always in [0, 2n-2]
    grad_a -= w[2 * np.arange(n)] * a

    return grad_a, w


def compute_grad_z(z, temperature):
    """Compute gradient of 2*n*LSE(conv(softmax(z),softmax(z)), t) w.r.t. z."""
    n = len(z)
    a = softmax_func(z)
    conv = fast_convolve(a)
    grad_a, w = compute_lse_grad_a(a, conv, temperature)

    # dLSE/dz_m = a[m] * (grad_a[m] - sum_i a[i]*grad_a[i])
    correction = np.dot(a, grad_a)
    grad_z = a * (grad_a - correction)
    grad_z *= 2.0 * n
    return a, conv, grad_z


def adam_softmax(z_init, time_limit, n, t0_temp=0.005, t_end=0.00005,
                 lr=0.005, beta1=0.9, beta2=0.999, eps=1e-8,
                 restart_interval=300, noise_scale=0.05):
    """Adam optimization in z-space with softmax parameterization."""
    rng = np.random.default_rng(hash(z_init.tobytes()) % (2**31))
    z = z_init.astype(np.float64).copy()
    m_hat = np.zeros_like(z)
    v_hat = np.zeros_like(z)
    t = 0

    best_a = softmax_func(z)
    best_conv = fast_convolve(best_a)
    best_obj = best_conv.max() * 2.0 * n

    temperature = t0_temp
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        t += 1

        a, conv, grad_z = compute_grad_z(z, temperature)

        m_hat = beta1 * m_hat + (1 - beta1) * grad_z
        v_hat = beta2 * v_hat + (1 - beta2) * grad_z ** 2
        m_hat_corr = m_hat / (1 - beta1 ** t)
        v_hat_corr = v_hat / (1 - beta2 ** t)
        z = z - lr * m_hat_corr / (np.sqrt(v_hat_corr) + eps)

        a_new = softmax_func(z)
        conv_new = fast_convolve(a_new)
        obj = conv_new.max() * 2.0 * n

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Anneal temperature logarithmically
        frac = min(1.0, t / (time_limit * 60))
        temperature = t0_temp * (t_end / t0_temp) ** (frac ** 0.7)

        # Restart if stuck
        if no_improve > restart_interval:
            z = np.log(best_a + 1e-12) + rng.standard_normal(n) * noise_scale
            m_hat = np.zeros_like(z)
            v_hat = np.zeros_like(z)
            t = 0
            temperature = t0_temp * 0.5
            no_improve = 0
            noise_scale *= 0.8

        # LR schedule
        if t % 1000 == 0:
            lr *= 0.85

    return best_a, best_obj


def nesterov_simplex(f_init, time_limit, n, t0_temp=0.005, t_end=0.0001,
                     lr0=0.05, restart_every=400):
    """Nesterov-accelerated projected gradient on simplex with LSE smoothing."""
    rng = np.random.default_rng(hash(f_init.tobytes()) % (2**31))
    f = f_init.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    f_prev = f.copy()
    t = 1
    best_f = f.copy()
    best_obj = fast_convolve(f).max() * 2.0 * n

    temperature = t0_temp
    t0 = time.monotonic()
    no_improve = 0
    lr = lr0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 1.0:
            break

        # Nesterov lookahead
        momentum = (t - 1) / (t + 2)
        f_nest = f + momentum * (f - f_prev)
        f_nest = project_simplex(f_nest)

        # Compute gradient
        conv = fast_convolve(f_nest)
        grad_a, _ = compute_lse_grad_a(f_nest, conv, temperature)

        # Update
        f_new = project_simplex(f_nest - lr / (t + 1) * grad_a * (t + 1))
        # Simplified: f_new = project(f_nest - lr * grad_a)
        f_new = project_simplex(f_nest - lr * grad_a)

        f_prev = f
        f = f_new
        t += 1

        obj = fast_convolve(f).max() * 2.0 * n
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Anneal
        frac = min(1.0, t / 5000)
        temperature = t0_temp * (t_end / t0_temp) ** frac
        lr *= 0.9995

        # Restart
        if no_improve > restart_every:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.02)
            f_prev = f.copy()
            t = 1
            temperature = t0_temp * 0.5
            lr = lr0 * 0.5
            no_improve = 0

    return best_f, best_obj


def solve_lp_fast(f, M, time_limit=15):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    if n > 600:
        return None
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def fw_escape(f, best_obj, time_limit=30):
    """Frank-Wolfe escape using LP oracle."""
    n = len(f)
    f = f.astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        return f, best_obj
    f = f / s

    conv = fast_convolve(f)
    M = conv.max()
    t0 = time.monotonic()
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        g = solve_lp_fast(f, M * m_exp, time_limit=10)
        if g is None:
            m_exp *= 1.5
            if m_exp > 10:
                break
            continue

        gs = g.sum()
        if gs < 1e-15:
            break
        g = g / gs

        # Line search
        best_alpha = 0.0
        best_val = fast_convolve(f).max()
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = project_simplex((1 - alpha) * f + alpha * g)
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            f = project_simplex((1 - best_alpha) * f + best_alpha * g)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M * 2 * n < best_obj:
                best_obj = M * 2 * n
        else:
            m_exp *= 1.3

    return f, best_obj


def generate_inits(n, parent=None, rng=None):
    """Generate diverse structured initializations on the simplex."""
    if rng is None:
        rng = np.random.default_rng(123)
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Symmetric ramp (peaked at center)
    center = (n - 1) / 2.0
    sym_ramp = 1.0 / (1.0 + 0.1 * (x - center) ** 2 / (n * 0.1))
    inits.append(sym_ramp / sym_ramp.sum())

    # Cosine window
    c = 0.5 * (1 + np.cos(2 * np.pi * x / n)) + 0.01
    inits.append(c / c.sum())

    # Exponential concentration at center
    e = np.exp(-((x - center) ** 2) / (2 * (n / 6) ** 2))
    inits.append(e / e.sum())

    # Inverse distance (peaked at edges)
    d = np.minimum(x, n - 1 - x) + 1
    inv_d = 1.0 / np.sqrt(d)
    inits.append(inv_d / inv_d.sum())

    # Sine
    sn = np.sin(np.pi * (x + 0.5) / n) + 0.1
    inits.append(sn / sn.sum())

    # Quadratic peak
    q = 1.0 - ((x - center) / center) ** 2
    q = np.maximum(q, 0.01)
    inits.append(q / q.sum())

    # Random on simplex
    r = rng.dirichlet(np.ones(n) * 0.5)
    inits.append(r)

    # Symmetric random
    half_n = n // 2
    rh = rng.dirichlet(np.ones(half_n) * 0.5)
    sym = np.concatenate([rh, rh[::-1] if n % 2 == 0 else np.concatenate([rh[::-1], [0]])[:n - half_n]])
    sym = np.maximum(sym, 0)
    ss = sym.sum()
    if ss > 1e-10:
        inits.append(sym / ss)

    # Parent-based
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.copy())
            inits.append(project_simplex(parent + rng.standard_normal(n) * 0.02))
            inits.append(project_simplex(0.7 * parent + 0.3 * (np.ones(n) / n)))
        else:
            px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
            px = np.maximum(px, 0)
            ps = px.sum()
            if ps > 1e-10:
                px = px / ps
                inits.append(px)
                inits.append(project_simplex(px + rng.standard_normal(n) * 0.03))

    return inits

            t_rem = deadline - time.monotonic() - 40
            t_alloc = min(t_rem, 60)
            if t_alloc < 10:
                break

            if round_num == 0:
                f_start = f_best
                label = "deep_adam"
            elif round_num == 1:
                f_start = project_simplex(f_best + rng.standard_normal(n_best) * 0.01)
                label = "deep_perturb"
            elif round_num == 2:
                f_start = project_simplex(0.8 * f_best + 0.2 * (np.ones(n_best) / n_best))
                label = "deep_blend"
            elif round_num == 3:
                f_start = project_simplex(f_best + rng.standard_normal(n_best) * 0.03)
                label = "deep_noise"
            else:
                f_start = f_best
                label = "deep_polish"

            if round_num == 4:
                a_opt, _ = fine_polish(f_start, time_limit=t_alloc, n=n_best)
            else:
                a_opt, _ = trial_and_refine(f_start, time_limit=t_alloc, n=n_best)
            try_seq(a_opt, label)

        # FW escape on the best
        if time.monotonic() < deadline - 40 and n_best <= 500:
            t_rem = min(40, deadline - time.monotonic() - 30)
            if t_rem > 10:
                a_fw, _ = fw_escape(f_best, best_score, time_limit=t_rem)
                try_seq(a_fw, "deep_fw")

    # === Phase 4: Try different length near best ===
    print(f"\n=== Phase 4: Adjacent Lengths ===")

    if best_seq is not None:
        n_best = len(best_seq)
        adjacent = [n_best - 16, n_best - 8, n_best + 8, n_best + 16,
                    n_best - 32, n_best + 32]
        adjacent = [n for n in adjacent if n >= 64 and n <= 2048 and n != n_best]
        adjacent = sorted(set(adjacent))[:4]

        for n_adj in adjacent:
            if time.monotonic() > deadline - 40:
                break
            # Interpolate best to new length
            f_interp = np.interp(np.arange(n_adj),
                                 np.linspace(0, 1, n_best), best_seq)
            f_interp = np.maximum(f_interp, 0)
            f_interp = f_interp / f_interp.sum()

            t_alloc = min(40, deadline - time.monotonic() - 30)
            if t_alloc < 10:
                break

            a_opt, _ = trial_and_refine(f_interp, time_limit=t_alloc, n=n_adj)
            try_seq(a_opt, f"adj_n{n_adj}")

    # === Phase 5: Final polish ===
    print(f"\n=== Phase 5: Final Polish ===")
    print(f"  Current best: {best_score:.8f}")

    if best_seq is not None:
        n_fin = len(best_seq)
        f_fin = best_seq / best_seq.sum()

        # Ultra-fine polish
        t_rem = deadline - time.monotonic() - 15
        if t_rem > 10:
            a_pol, _ = fine_polish(f_fin, time_limit=min(30, t_rem), n=n_fin)
            try_seq(a_pol, "final_polish")

        # Micro perturbations
        t_rem = deadline - time.monotonic() - 10
        if t_rem > 10:
            for _ in range(3):
                t_rem = deadline - time.monotonic() - 8
                if t_rem < 5:
                    break
                f_pert = project_simplex(f_fin + rng.standard_normal(n_fin) * 0.005)
                a_p, _ = fine_polish(f_pert, time_limit=min(10, t_rem), n=n_fin)
                try_seq(a_p, "micro")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
