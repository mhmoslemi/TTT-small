import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_reduced(f, M, active_indices):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in active_indices, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((len(active_indices), n))
    for i, k in enumerate(active_indices):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        A_ub[i, idx] = f[k - idx]
    b_ub = np.full(len(active_indices), M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def lp_frank_wolfe(a0, time_limit):
    """Frank-Wolfe with LP oracle and gradient descent fallback."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = M
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 3:
            break

        # Adaptive active set: top peaks
        fraction = max(0.1, 0.5 * M / max(conv.std() + 1e-15, 1e-15))
        k_active = max(n // 4, int(fraction * (2 * n - 1)))
        k_active = min(k_active, 2 * n - 1)
        active = np.argpartition(conv, -k_active)[-k_active:]

        g = solve_lp_reduced(a, M * 1.01, active)
        if g is None:
            # Fallback: use subgradient direction
            peak = np.argmax(conv)
            js = max(0, peak - n + 1)
            je = min(n - 1, peak)
            grad = np.zeros(n)
            grad[js:je+1] = 2.0 * a[peak - np.arange(js, je + 1)]
            g = project_simplex(a - 0.01 * grad)

        g = np.maximum(g, 0)
        gs = g.sum()
        if gs < 1e-15:
            break
        g = g / gs

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            at = (1 - alpha) * a + alpha * g
            at = np.maximum(at, 0)
            st = at.sum()
            if st < 1e-15:
                continue
            at = at / st
            val = fast_convolve(at).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-8:
            a = (1 - best_alpha) * a + best_alpha * g
            a = np.maximum(a, 0)
            a = a / a.sum()
            conv = fast_convolve(a)
            M = conv.max()
            if M < best_obj - 1e-13:

            no_improve = 0

def spectral_flatten(a0, n_iters=100, alpha=0.7):
    """Gerchberg-Saxton style spectral flattening."""
    a = np.maximum(a0, 0).copy()
    a = a / a.sum()
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    for _ in range(n_iters):
        A = np.fft.rfft(a, fft_size)
        P = np.abs(A) ** 2
        # Flatten: P_new = P^alpha (compresses dynamic range)
        P_flat = P ** alpha
        # Preserve total power
        P_flat = P_flat * (P.sum() / max(P_flat.sum(), 1e-30))
        # Reconstruct with original phase
        phase = np.angle(A)
        A_new = np.sqrt(P_flat) * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-15:
            a_new = a_new / s
            obj = fast_convolve(a_new).max()
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
            a = a_new
        else:
            break

    return best_a, best_obj


def subgradient_refine(a0, time_limit, lr=0.003, top_k=64):

    a = np.maximum(a0, 0).copy()

    n = len(a)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    v = np.zeros(n)
    mom = 0.9
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = lr

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        # Compute subgradient (softmax-weighted over top peaks)

        else:

        temp = 0.001

        w = np.exp(scaled)

        sg = np.zeros(n)

            k = int(peaks[i])

            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]

    return best_a, best_obj

    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_score = float('inf')
    best_seq = None

    best_state = [float('inf'), None]

    def try_seq(a, label=""):
        nonlocal best_score, best_seq
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        a = np.minimum(a, 1000.0)
        if a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
    except Exception:
        pass

    # === Phase 1: Quick initial optimization on parent ===
    print("=== Phase 1: Quick Parent Optimization ===")
    if parent is not None:
        n_p = len(parent)
        try_seq(parent, "parent_init")

        # LP-FW quick run
        t1 = min(60, deadline - time.monotonic() - 500)
        if t1 > 10:
            a_opt, _ = lp_frank_wolfe(parent, time_limit=t1)
            try_seq(a_opt, "parent_fw")

        # Coordinate descent quick run
        rem = deadline - time.monotonic() - 400
        if rem > 10:
            a_opt, _ = coordinate_descent_optimize(a_opt if best_seq is not None and len(a_opt) == n_p else parent,
                                                   time_limit=min(rem, 60))
            try_seq(a_opt, "parent_cd")

    # If no parent, create one
    if best_seq is None or best_score > 2.0:
        print("  Creating initial sequence...")
        n_init = 384
        a_init = np.ones(n_init) / n_init
        try_seq(a_init, "uniform_384")
        # Quick CD
        a_opt, _ = coordinate_descent_optimize(a_init, time_limit=30)
        try_seq(a_opt, "cd_384")

    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # === Phase 2: Main optimization loop ===
    print(f"\n=== Phase 2: Main Optimization (best={best_score:.8f}) ===")
    n_best = len(best_seq)
    a_work = best_seq.copy()
    a_work = a_work / a_work.sum()

    phase2_deadline = deadline - 200
    method = 0
    no_phase_improve = 0

    while time.monotonic() < phase2_deadline:
        rem = phase2_deadline - time.monotonic()
        if rem < 5:
            break

        # Cycle through methods
        t_alloc = min(rem, 60)

        if method == 0:
            # Coordinate descent (fastest, do most of the work here)
            t_cd = min(rem, 120)
            a_opt, obj = coordinate_descent_optimize(a_work, time_limit=t_cd)
            try_seq(a_opt, "cd_main")
            if obj < best_score / (2 * n_best) - 1e-8:
                a_work = a_opt / a_opt.sum()
                no_phase_improve = 0
            else:
                no_phase_improve += 1
        elif method == 1:
            # LP Frank-Wolfe
            t_lp = min(rem, 30)
            a_opt, obj = lp_frank_wolfe(a_work, time_limit=t_lp)
            try_seq(a_opt, "lp_fw")
            a_work = a_opt / a_opt.sum()
            no_phase_improve = 0
        elif method == 2:
            # Spectral flattening + CD
            a_sf, _ = spectral_flatten(a_work, n_iters=50, alpha=0.8)
            a_cd, _ = coordinate_descent_optimize(a_sf, time_limit=min(rem, 30))
            try_seq(a_cd, "spectral_cd")
            a_work = a_cd / a_cd.sum()
            no_phase_improve += 1
        elif method == 3:
            # Subgradient refinement
            a_opt, obj = subgradient_refine(a_work, time_limit=min(rem, 30),
                                            lr=0.002, top_k=n_best // 8)
            try_seq(a_opt, "sg_refine")
            a_work = a_opt
            no_phase_improve += 1

        method = (method + 1) % 4
        # Spend more time on CD
        if method == 0 and no_phase_improve > 2:
            # Try a perturbation to escape
            if rem > 30:
                a_pert = project_simplex(a_work + rng.standard_normal(n_best) * 0.005)
                a_cd, _ = coordinate_descent_optimize(a_pert, time_limit=min(rem, 60))
                try_seq(a_cd, "perturb_cd")
                a_work = a_cd / a_cd.sum()
                no_phase_improve = 0

        if no_phase_improve > 8 and rem > 60:
            # Major restart with different structure
            print(f"  Phase 2: restarting with perturbation (no improve for {no_phase_improve} cycles)")
            a_restart = project_simplex(best_seq / best_seq.sum() + rng.standard_normal(n_best) * 0.02)
            a_cd, _ = coordinate_descent_optimize(a_restart, time_limit=min(rem, 90))
            try_seq(a_cd, "restart_cd")
            a_work = a_cd / a_cd.sum()
            no_phase_improve = 0

    # === Phase 3: Deep refinement ===
    print(f"\n=== Phase 3: Deep Refinement (best={best_score:.8f}) ===")
    n_best = len(best_seq)
    a_cur = best_seq / best_seq.sum()

    for p in range(5):
        rem = deadline - time.monotonic() - 30
        if rem < 10:
            break
        t_alloc = min(rem, 120)

        # Alternate CD and LP
        if p % 2 == 0:
            a_opt, _ = coordinate_descent_optimize(a_cur, time_limit=t_alloc)
        else:
            a_opt, _ = lp_frank_wolfe(a_cur, time_limit=t_alloc // 2)
            a_opt, _ = coordinate_descent_optimize(a_opt, time_limit=t_alloc // 2)

        try_seq(a_opt, f"deep_{p}")
        a_cur = a_opt / a_opt.sum()

    # === Phase 4: Try alternative lengths ===
    print(f"\n=== Phase 4: Alternative Lengths (best={best_score:.8f}) ===")

    alt_lengths = [320, 448, 512, 640]
    # Remove current length
    alt_lengths = [n for n in alt_lengths if n != n_best]

    for n_alt in alt_lengths:
        rem = deadline - time.monotonic() - 20
        if rem < 30:
            break

        # Create initial sequence of new length using interpolation from best
        if best_seq is not None:
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_alt)
            a_interp = np.interp(x_new, x_old, best_seq)
            a_interp = np.maximum(a_interp, 0)
            a_interp = a_interp / a_interp.sum()
        else:
            a_interp = np.ones(n_alt) / n_alt

        # Quick CD
        t_alloc = min(rem // max(1, len(alt_lengths) + 1), 60)
        a_opt, _ = coordinate_descent_optimize(a_interp, time_limit=t_alloc)
        try_seq(a_opt, f"alt_n{n_alt}")

        # LP-FW
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            a_opt, _ = lp_frank_wolfe(a_opt, time_limit=min(rem, 25))
            try_seq(a_opt, f"alt_n{n_alt}_lp")

    # === Phase 5: Final polish ===
    print(f"\n=== Phase 5: Final Polish (best={best_score:.8f}) ===")
    n_best = len(best_seq)

    # Extended CD
    rem = deadline - time.monotonic() - 10
    if rem > 10:
        a_opt, _ = coordinate_descent_optimize(best_seq, time_limit=min(rem, 60))
        try_seq(a_opt, "final_cd")

    # Subgradient polish
    rem = deadline - time.monotonic() - 5
    if rem > 5:
        a_opt, _ = subgradient_refine(best_seq, time_limit=min(rem, 30),
                                      lr=0.001, top_k=max(16, n_best // 10))
        try_seq(a_opt, "final_sg")

    # Micro perturbations
    rem = deadline - time.monotonic() - 3
    if rem > 3:
        for _ in range(3):
            rem = deadline - time.monotonic() - 2
            if rem < 2:
                break
            a_pert = project_simplex(best_seq / best_seq.sum() + rng.standard_normal(n_best) * 0.002)
            a_opt, _ = coordinate_descent_optimize(a_pert, time_limit=min(rem, 10))
            try_seq(a_opt, "micro")

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        else:
            best_seq = np.ones(256) / 256
            best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
