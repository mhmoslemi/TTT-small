def fast_autoconv(a):

def project_simplex(v, s=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - s)
    if not cond.any():
        return np.full(n, s / n)
    rho_max = np.nonzero(cond)[0][-1] + 1
    theta = (cssv[rho_max - 1] - s) / rho_max
    return np.maximum(v - theta, 0.0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. L-BFGS with soft-max smoothing and softmax parameterization (fast, second-order)
2. FW-LP (Frank-Wolfe with LP oracle) for local refinement
3. Multi-length exploration and perturbation-based restarts
Key innovation: smooth optimization via soft-max enables much faster convergence
than the subgradient methods used previously, complemented by LP-based local search.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time


def fast_autoconv(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, s=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = s}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - s)
    if not cond.any():
        return np.full(n, s / n)
    rho_max = np.nonzero(cond)[0][-1] + 1
    theta = (cssv[rho_max - 1] - s) / rho_max
    return np.maximum(v - theta, 0.0)


def softmax_fn(x):
    """Numerically stable softmax."""
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()


def obj_and_grad(x, n, t):
    """
    Compute objective and gradient for L-BFGS on softmax-parameterized variables.
    Objective: 2n * soft_max(t, conv(softmax(x), softmax(x)))
    """
    a = softmax_fn(x)
    conv = fast_autoconv(a)

    # Soft-max
    tc = t * conv
    tc_max = tc.max()
    exp_tc = np.exp(tc - tc_max)
    sum_exp = exp_tc.sum()
    log_sum_exp = tc_max + np.log(sum_exp)
    obj = 2.0 * n * log_sum_exp / t

    # Weights
    w = exp_tc / sum_exp

    # Gradient: d(obj)/d(a_j) = 4n * sum_{m} w[j+m] * a[m]
    # Compute cross-correlation via FFT
    fft_size = 1 << (3 * n - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    cc = np.fft.irfft(W * np.conj(A), fft_size)
    grad_a = 4.0 * n * cc[:n]

    # Chain rule for softmax: d(obj)/d(x_i) = grad_a[i]*a[i] - a[i]*sum_j(grad_a[j]*a[j])
    dot_ga_a = np.dot(grad_a, a)
    grad_x = grad_a * a - a * dot_ga_a

    return obj, grad_x


def run_lbfgs(x0, n, t_schedule, max_iter_per_t=300):
    """Run L-BFGS with annealed temperature schedule."""
    x = x0.copy()
    best_a = None
    best_exact = float('inf')

    for t in t_schedule:
        res = minimize(obj_and_grad, x, args=(n, t), method='L-BFGS-B', jac=True,
                       options={'maxiter': max_iter_per_t, 'ftol': 1e-14, 'gtol': 1e-10})
        x = res.x
        a = softmax_fn(x)
        exact_obj = 2.0 * n * fast_autoconv(a).max()
        if exact_obj < best_exact:
            best_exact = exact_obj
            best_a = a.copy()

    return best_a, best_exact


def solve_conv_lp(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    inits.append(np.ones(n) / n)

    # Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    # Edge-weighted
    x = np.arange(n, dtype=np.float64)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits.append(g / g.sum())

    # Cosine
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.05
    inits.append(c1 / c1.sum())

    # Sine squared
    s1 = np.sin(np.pi * (x + 1) / (n + 1)) ** 2
    inits.append(s1 / s1.sum())

    # Two-peak
    p1 = np.exp(-0.5 * ((x - 0.3 * n) / (0.1 * n)) ** 2)
    p2 = np.exp(-0.5 * ((x - 0.7 * n) / (0.1 * n)) ** 2)
    two_peak = p1 + p2
    inits.append(two_peak / two_peak.sum())

    # From parent (interpolated)
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits
