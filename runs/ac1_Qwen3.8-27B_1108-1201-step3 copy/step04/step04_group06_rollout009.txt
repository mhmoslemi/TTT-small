method with momentum, adaptive step size, Fourier-domain initialization,

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_objective(a):
    """Compute the objective value for a normalized sequence."""
    n = len(a)
    c = fast_convolve(a)
    return c.max()


def eg_optimize_with_momentum(a0, time_limit, eta0=1.0, temp0=0.005,
                               temp_decay=0.9997, eta_decay=0.99995,
                               top_k_frac=0.3, momentum=0.8,
                               use_adaptive_eta=True, verbose=False):
    """EG with momentum and adaptive step size."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    top_k = max(10, int(n * top_k_frac))

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    # Momentum buffer
    momentum_buffer = np.zeros(n)
    
    avg_a = a.copy().astype(np.float64)
    count = 0
    eta = eta0
    temp = temp0
    prev_obj = best_obj

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Compute subgradient
        if top_k < len(c):
            indices = np.argpartition(c, -top_k)[-top_k:]
            peek = c[indices]
        else:
            indices = np.arange(len(c))
            peek = c.copy()

        scaled = peek / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(indices))
            w[np.argmax(peek)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for idx in range(len(indices)):
            if w[idx] < 1e-15:
                continue
            k = int(indices[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            sg[js:je + 1] += 2.0 * w[idx] * a[k - je:k - js + 1][::-1]

        sg_max = sg.max()
        if sg_max < 1e-15:
            break

        # Compute update direction with momentum
        log_update = -eta * sg / sg_max
        momentum_buffer = momentum * momentum_buffer + (1 - momentum) * log_update
        log_update = momentum_buffer
        
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-300)
        s_new = a_new.sum()
        if s_new < 1e-300:
            a_new = a.copy()
            s_new = 1.0
        a_new = a_new / s_new

        c_new = fast_convolve(a_new)
        obj_new = c_new.max()

        count += 1
        avg_a = (count * avg_a + a_new) / (count + 1)

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0

                print(f"    EG iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        # Adaptive eta
        if use_adaptive_eta:
            if obj_new < prev_obj:
                eta = min(eta * 1.01, 5.0)  # increase step if improving
            else:
                eta = eta * 0.99  # decrease step if not improving
            prev_obj = obj_new

        a = a_new
        c = c_new

        if it % 500 == 0:
            c_avg = fast_convolve(avg_a)
            if c_avg.max() < best_obj - 1e-12:
                best_obj = c_avg.max()
                best_a = avg_a.copy()

        eta *= eta_decay
        temp *= temp_decay
        temp = max(temp, 0.0001)

        if no_improve > 2000:

            no_improve = 0
            eta = eta0 * 0.5
            temp = max(temp, temp0 * 0.1)
            avg_a = a.copy()
            count = 0
            momentum_buffer = np.zeros(n)

def fourier_init(n, rng, mode='flat_power'):
    """Generate initialization with approximately flat power spectrum."""
    if mode == 'flat_power':
        # Start with flat power spectrum, add random phase
        P = np.ones(n)
        # Add some structure to help with non-negativity
        # Use a low-pass filter on the power
        freqs = np.fft.rfftfreq(n)
        lowpass = np.exp(-5 * freqs * n)
        P = P * (1 + 0.5 * lowpass)
        
        # Random phase
        phases = rng.uniform(0, 2 * np.pi, size=n // 2 + 1)
        # Enforce conjugate symmetry
        A = np.zeros(n // 2 + 1, dtype=np.complex128)
        A = np.sqrt(P[:n//2+1]) * np.exp(1j * phases)
        
        a = np.fft.irfft(A, n)
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            a = a / s
        else:
            a = np.ones(n) / n
        return a
    elif mode == 'gaussian_phase':
        # Gaussian in frequency with random phase
        freqs = np.arange(n // 2 + 1)
        sigma = n * 0.1
        P = np.exp(-0.5 * ((freqs - n//4) / sigma) ** 2)
        phases = rng.standard_normal(n // 2 + 1)
        A = np.sqrt(P) * np.exp(1j * phases)
        a = np.fft.irfft(A, n)
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-10:
            a = a / s
        else:
            a = np.ones(n) / n
        return a
    else:
        return np.ones(n) / n


def local_search(a0, time_limit, max_rounds=500):
    """Enhanced local search with multiple move types."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 100:
        improved = False

        # Move type 1: single coordinate scale
        i = rng.integers(0, n)
        if a[i] > 1e-10:
            for factor in [0.5, 0.7, 0.8, 1.0/0.8, 1.0/0.7, 1.0/0.5,
                           0.6, 1.0/0.6, 0.85, 1.0/0.85, 0.9, 1.0/0.9]:
                old = a[i]
                a[i] = old * factor
                s2 = a.sum()
                if s2 < 1e-15:
                    a[i] = old
                    continue
                a_norm = a / s2
                c2 = fast_convolve(a_norm)
                obj2 = c2.max()
                if obj2 < best_obj - 1e-12:
                    best_obj = obj2
                    best_a = a_norm.copy()
                    a = a_norm.copy()
                    improved = True
                    break
                a[i] = old
            if improved:
                no_improve = 0
                continue
            a = best_a.copy()

        # Move type 2: pair swap (redistribute mass between two coordinates)
        i, j = rng.choice(n, size=2, replace=False)
        if a[i] > 1e-10 and a[j] > 1e-10:
            for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.8, 0.9]:
                transfer = min(a[i] * alpha, a[j] * (1-alpha))
                a[i] -= transfer
                a[j] += transfer
                s2 = a.sum()
                a_norm = a / s2
                c2 = fast_convolve(a_norm)
                obj2 = c2.max()
                if obj2 < best_obj - 1e-12:
                    best_obj = obj2
                    best_a = a_norm.copy()
                    a = a_norm.copy()
                    improved = True
                    break
            if improved:
                no_improve = 0
                continue
            a = best_a.copy()

        # Move type 3: block perturbation
        block_size = max(2, n // 32)
        start = rng.integers(0, n - block_size)
        old_block = a[start:start+block_size].copy()
        perturb = rng.standard_normal(block_size) * 0.01
        a[start:start+block_size] = np.maximum(old_block + perturb, 0)
        s2 = a.sum()
        if s2 > 1e-15:
            a_norm = a / s2
            c2 = fast_convolve(a_norm)
            obj2 = c2.max()
            if obj2 < best_obj - 1e-12:
                best_obj = obj2
                best_a = a_norm.copy()
                a = a_norm.copy()
                improved = True

        if not improved:
            no_improve += 1
            a = best_a.copy()

    return best_a.copy(), best_obj

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_subgradient(a, c, temperature, top_k):
    """Compute softmax-weighted subgradient of max(c)."""
    n = len(a)
    len_c = len(c)
    if top_k is not None and top_k < len_c and top_k > 0:
        indices = np.argpartition(c, -top_k)[-top_k:]
        peek = c[indices]
    else:
        indices = np.arange(len_c)
        peek = c.copy()
    scaled = peek / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(indices))
        w[np.argmax(peek)] = 1.0
    else:
        w /= ws
    sg = np.zeros(n)
    for idx in range(len(indices)):
        if w[idx] < 1e-15:
            continue
        k = int(indices[idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        sg[js:je + 1] += 2.0 * w[idx] * a[k - je:k - js + 1][::-1]
    return sg

def gen_diverse_inits(n, best_a, rng, n_inits=20):
    """Generate diverse initializations."""
    inits = [best_a.copy()]
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Various Beta distributions
    for alpha, beta in [(2, 5), (5, 2), (0.5, 0.5), (1, 3), (3, 1)]:
        b = rng.beta(alpha, beta, size=n)
        inits.append(b / b.sum())

    # Cosine variants
    for freq in [1, 2, 3]:
        c = np.abs(np.cos(np.pi * freq * x / n)) + 0.1
        inits.append(c / c.sum())

    # Edge-concentrated
    for p in [0.5, 1.0, 2.0]:
        edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0) ** p
        inits.append(edge / edge.sum())

    # Gaussian
    for sigma_frac in [0.1, 0.2, 0.3, 0.5]:
        g = np.exp(-0.5 * ((x - n / 2) / (sigma_frac * n)) ** 2)
        inits.append(g / g.sum())

    # Random Dirichlet
    for conc in [0.1, 0.5, 2.0, 5.0]:
        r = rng.dirichlet(np.ones(n) * conc)
        inits.append(r)

    # Fourier-domain: flat power with random phase
    for _ in range(3):
        P = 1.0 + 0.3 * np.exp(-np.linspace(0, 1, n // 2 + 1) ** 2)
        phases = rng.uniform(0, 2 * np.pi, size=n // 2 + 1)
        A = np.sqrt(P) * np.exp(1j * phases)
        a_f = np.fft.irfft(A, n)
        a_f = np.maximum(a_f, 0)
        sf = a_f.sum()
        if sf > 1e-10:
            inits.append(a_f / sf)
        else:
            inits.append(np.ones(n) / n)

    # Perturbed best
    for scale in [0.01, 0.05, 0.1, 0.2]:
        p = best_a + rng.standard_normal(n) * scale
        p = np.maximum(p, 0)
        ps = p.sum()
        if ps > 1e-10:
            inits.append(p / ps)

    return inits[:n_inits]


def crossover(a1, a2, rng, alpha=None):
    """Crossover between two sequences."""
    if alpha is None:
        alpha = rng.uniform(0.3, 0.7)
    a = alpha * a1 + (1 - alpha) * a2
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-10:
        a = a / s
    return a

            i = rng.integers(0, n)
            if a[i] > 1e-12:
                for factor in [0.5, 0.7, 0.8, 0.9, 1/0.9, 1/0.8, 1/0.7, 1/0.5,
                               0.6, 1/0.6, 0.85, 1/0.85]:
                    old = a[i]
                    a[i] = old * factor
                    s2 = a.sum()
                    if s2 > 1e-15:
                        a_n = a / s2
                        c2 = fast_convolve(a_n)
                        if c2.max() < best_obj - 1e-13:
                            best_obj = c2.max()
                            best_a = a_n.copy()
                            a = a_n.copy()
                            improved = True
                            break
                    a[i] = old
                if not improved:
                    a = best_a.copy()

        elif move_type == 1:
            # Pair redistribution
            i, j = rng.choice(n, size=2, replace=False)
            if a[i] > 1e-12 and a[j] > 1e-12:
                for alpha in [0.1, 0.2, 0.3, 0.5, 0.7]:
                    transfer = a[i] * alpha
                    if transfer > a[j] * 0.5:
                        continue
                    a[i] -= transfer
                    a[j] += transfer
                    s2 = a.sum()
                    if s2 > 1e-15:
                        a_n = a / s2
                        c2 = fast_convolve(a_n)
                        if c2.max() < best_obj - 1e-13:
                            best_obj = c2.max()
                            best_a = a_n.copy()
                            a = a_n.copy()
                            improved = True
                            break
                if not improved:
                    a = best_a.copy()

        elif move_type == 2:
            # Gaussian block perturbation
            block_size = max(3, n // 16)
            start = rng.integers(0, n - block_size + 1)
            old_block = a[start:start + block_size].copy()
            scale = step_scale * np.std(a)
            perturbation = rng.standard_normal(block_size) * scale
            a[start:start + block_size] = np.maximum(old_block + perturbation, 0)
            s2 = a.sum()
            if s2 > 1e-15:
                a_n = a / s2
                c2 = fast_convolve(a_n)
                if c2.max() < best_obj - 1e-13:
                    best_obj = c2.max()
                    best_a = a_n.copy()
                    a = a_n.copy()
                    improved = True
                    step_scale = min(step_scale * 1.2, 0.5)
            if not improved:
                a = best_a.copy()
                step_scale = max(step_scale * 0.9, 0.01)

        elif move_type == 3:
            # Mirror perturbation: perturb a[i] and a[n-1-i] together
            i = rng.integers(0, n // 2)
            j = n - 1 - i
            delta = rng.standard_normal() * step_scale * np.std(a)
            old_i, old_j = a[i], a[j]
            a[i] = max(0, old_i + delta)
            a[j] = max(0, old_j - delta * 0.5)
            s2 = a.sum()
            if s2 > 1e-15:
                a_n = a / s2
                c2 = fast_convolve(a_n)
                if c2.max() < best_obj - 1e-13:
                    best_obj = c2.max()
                    best_a = a_n.copy()
                    a = a_n.copy()
                    improved = True
            if not improved:
                a = best_a.copy()

        if improved:
            no_improve = 0
        else:
            no_improve += 1

    return best_a.copy(), best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search: diverse multi-strategy optimization combining EG with momentum,
    basin hopping, crossover, multi-length exploration, and local search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')
    all_good = []  # collection of good solutions for crossover

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score):
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
            if len(all_good) < 20 and score < best_score + 0.05:
                all_good.append(a.copy())
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = evaluate_sequence(parent.tolist())
            print(f"Parent: n={len(parent)}, score={parent_score:.8f}")
            if np.isfinite(parent_score):
                best_score = parent_score
                best_seq = parent.copy()
                all_good.append(parent.copy())
    except Exception:
        pass

    if parent is None:
        parent = np.ones(384) / 384
        best_score = evaluate_sequence(parent.tolist())
        best_seq = parent.copy()
        all_good.append(parent.copy())

    n = len(parent)
    print(f"Starting: n={n}, best_score={best_score:.8f}")

    # === Phase 1: Diverse short EG runs (global exploration) ===
    print("\n=== Phase 1: Diverse Short EG ===")
    t_remaining = deadline - time.monotonic()
    t1 = t_remaining * 0.30
    n_runs = 15
    t_per_run = t1 / n_runs

    inits = gen_diverse_inits(n, best_seq, rng, n_inits=n_runs)
    for i, ini in enumerate(inits):
        rem = deadline - time.monotonic() - t_remaining * 0.45
        if rem < 10:
            break
        t_alloc = min(t_per_run, rem)
        if t_alloc < 8:
            break
        # Vary hyperparameters per run
        eta0 = [1.0, 1.5, 0.5, 2.0, 3.0][i % 5]
        temp0 = [0.005, 0.01, 0.002, 0.008, 0.003][i % 5]
        mom = [0.0, 0.5, 0.7, 0.3, 0.8][i % 5]
        nest = (i % 3 == 0)

        a_opt, obj = eg_run(ini, time_limit=t_alloc, eta0=eta0, temp0=temp0,
                            temp_decay=0.999, eta_decay=0.9999,
                            top_k_frac=0.3, momentum=mom, nesterov=nest,
                            reheat_interval=800, seed=i * 100 + 7)
        try_seq(a_opt, f"div{i}")

    print(f"  After Phase 1: best={best_score:.8f}")

    # === Phase 2: Basin hopping ===
    print(f"\n=== Phase 2: Basin Hopping ===")
    t_remaining = deadline - time.monotonic()
    t2 = t_remaining * 0.25
    n_hops = 8
    t_per_hop = t2 / n_hops

    for h in range(n_hops):
        rem = deadline - time.monotonic() - t_remaining * 0.25
        if rem < 15:
            break
        t_alloc = min(t_per_hop, rem)
        if t_alloc < 10:
            break

        # Perturb the current best
        pert_scale = 0.01 + 0.02 * h / n_hops
        a_perturbed = best_seq + rng.standard_normal(n) * pert_scale
        a_perturbed = np.maximum(a_perturbed, 0)
        a_perturbed = a_perturbed / a_perturbed.sum()

        a_opt, obj = eg_run(a_perturbed, time_limit=t_alloc, eta0=1.0,
                            temp0=0.003, temp_decay=0.9997, eta_decay=0.99995,
                            top_k_frac=0.3, momentum=0.6, nesterov=True,
                            reheat_interval=1000, seed=1000 + h * 13)
        try_seq(a_opt, f"hop{h}")

    print(f"  After Phase 2: best={best_score:.8f}")

    # === Phase 3: Crossover between diverse solutions ===
    print(f"\n=== Phase 3: Crossover ===")
    t_remaining = deadline - time.monotonic()
    t3 = t_remaining * 0.15
    if len(all_good) >= 2:
        n_cross = min(6, len(all_good) * (len(all_good) - 1) // 2)
        t_per_cross = t3 / max(1, n_cross)
        cross_count = 0
        for i in range(min(len(all_good), 5)):
            for j in range(i + 1, min(len(all_good), 5)):
                if cross_count >= n_cross:
                    break
                rem = deadline - time.monotonic() - t_remaining * 0.20
                if rem < 15:
                    break
                t_alloc = min(t_per_cross, rem)
                if t_alloc < 10:
                    break

                for alpha in [0.3, 0.5, 0.7]:
                    a_cross = crossover(all_good[i], all_good[j], rng, alpha)
                    if len(a_cross) == n:
                        a_opt, obj = eg_run(a_cross, time_limit=min(t_alloc / 2, 30),
                                            eta0=1.0, temp0=0.005,
                                            temp_decay=0.9997, eta_decay=0.99995,
                                            top_k_frac=0.3, momentum=0.5,
                                            seed=cross_count * 7 + 1)
                        try_seq(a_opt, f"cross{cross_count}_{alpha}")
                cross_count += 1

    print(f"  After Phase 3: best={best_score:.8f}")

    # === Phase 4: Long fine EG on best ===
    print(f"\n=== Phase 4: Long Fine EG ===")
    t_remaining = deadline - time.monotonic()
    t4 = t_remaining * 0.20
    if best_seq is not None:
        a_opt, obj = eg_run(best_seq, time_limit=min(t4, 300), eta0=0.5,
                            temp0=0.001, temp_decay=0.9999, eta_decay=0.99999,
                            top_k_frac=0.5, momentum=0.8, nesterov=True,
                            reheat_interval=2000, reheat_factor=3.0, seed=999)
        try_seq(a_opt, "fine_eg")

        # Another with different params
        rem = deadline - time.monotonic() - t_remaining * 0.15
        if rem > 30:
            a_opt, obj = eg_run(best_seq, time_limit=min(rem, 200), eta0=0.3,
                                temp0=0.0005, temp_decay=0.99995, eta_decay=0.99999,
                                top_k_frac=0.8, momentum=0.9, nesterov=True,
                                reheat_interval=3000, seed=888)
            try_seq(a_opt, "fine_eg2")

    print(f"  After Phase 4: best={best_score:.8f}")

    # === Phase 5: Try different lengths ===
    print(f"\n=== Phase 5: Multi-Length ===")
    t_remaining = deadline - time.monotonic()
    t5 = t_remaining * 0.15
    alt_lengths = [n + 32, n - 32, n + 64, n // 2 + 32, n * 3 // 2]
    alt_lengths = [l for l in alt_lengths if 64 <= l <= 1024 and l != n]

    for n_new in alt_lengths[:4]:
        rem = deadline - time.monotonic() - t_remaining * 0.10
        if rem < 20:
            break
        # Interpolate
        x_old = np.linspace(0, 1, len(best_seq))
        x_new = np.linspace(0, 1, n_new)
        a_new = np.interp(x_new, x_old, best_seq)
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-10:
            a_new = a_new / s
            t_alloc = min(40, rem)
            if t_alloc > 15:
                a_opt, obj = eg_run(a_new, time_limit=t_alloc, eta0=1.0,
                                    temp0=0.005, temp_decay=0.9997,
                                    eta_decay=0.99995, momentum=0.5,
                                    nesterov=True, seed=n_new)
                try_seq(a_opt, f"len{n_new}")

    print(f"  After Phase 5: best={best_score:.8f}")

    # === Phase 6: Local search polish ===
    print(f"\n=== Phase 6: Local Search ===")
    t_remaining = deadline - time.monotonic()
    t6 = t_remaining * 0.15
    if best_seq is not None:
        a_opt, obj = local_search_v2(best_seq, time_limit=min(t6, 180), seed=777)
        try_seq(a_opt, "local1")

        # Run local search again on the result
        rem = deadline - time.monotonic() - 10
        if rem > 20:
            a_opt, obj = local_search_v2(best_seq, time_limit=min(rem, 100), seed=888)
            try_seq(a_opt, "local2")

    print(f"  After Phase 6: best={best_score:.8f}")

    # === Phase 7: Final ultra-fine EG ===
    print(f"\n=== Phase 7: Final Ultra-Fine ===")
    rem = deadline - time.monotonic() - 5
    if rem > 10 and best_seq is not None:
        a_opt, obj = eg_run(best_seq, time_limit=min(rem, 60), eta0=0.1,
                            temp0=0.0003, temp_decay=0.99998, eta_decay=0.99999,
                            top_k_frac=1.0, momentum=0.95, nesterov=True,
                            reheat_interval=0, seed=12345)
        try_seq(a_opt, "ultra_fine")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
