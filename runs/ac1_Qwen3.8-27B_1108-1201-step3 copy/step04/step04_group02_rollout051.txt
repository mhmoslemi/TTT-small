"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS optimization of a 
smoothed (logsumexp) objective with softmax parameterization.
Multiple sequence lengths and initializations are explored with adaptive
time allocation and iterative beta refinement.
"""
import numpy as np
from scipy.optimize import minimize
import time

def fft_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def softmax(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def smooth_obj(z, beta):
    """Smooth approximation to max(conv(softmax(z)))."""
    a = softmax(z)
    conv = fft_convolve(a)
    bc = beta * conv
    bc_max = bc.max()
    return (1.0 / beta) * (bc_max + np.log(np.exp(bc - bc_max).sum()))

def smooth_grad(z, beta):
    """Gradient of smooth objective w.r.t. z."""
    a = softmax(z)
    n = len(a)
    conv = fft_convolve(a)
    bc = beta * conv
    bc_max = bc.max()
    w = np.exp(bc - bc_max)
    w /= w.sum()
    
    # g_a = 2 * conv(w_rev, a)[:n]
    w_rev = w[::-1].copy()
    fft_size = 1 << (2 * n - 2).bit_length()
    W = np.fft.rfft(w_rev, fft_size)
    A = np.fft.rfft(a, fft_size)
    g_a = np.fft.irfft(W * A, fft_size)[:n] * 2.0
    
    mu = np.dot(g_a, a)
    return a * (g_a - mu)

def smooth_obj_grad(z, beta):
    """Combined objective and gradient."""
    a = softmax(z)
    n = len(a)
    conv = fft_convolve(a)
    bc = beta * conv
    bc_max = bc.max()
    obj = (1.0 / beta) * (bc_max + np.log(np.exp(bc - bc_max).sum()))
    w = np.exp(bc - bc_max)
    w /= w.sum()
    
    w_rev = w[::-1].copy()
    fft_size = 1 << (2 * n - 2).bit_length()
    W = np.fft.rfft(w_rev, fft_size)
    A = np.fft.rfft(a, fft_size)
    g_a = np.fft.irfft(W * A, fft_size)[:n] * 2.0
    
    mu = np.dot(g_a, a)
    grad = a * (g_a - mu)
    return obj, grad

def true_obj(a):
    """True max of auto-convolution."""
    return fft_convolve(a).max()

def optimize_from_z(z_init, betas, max_iter=200, gtol=1e-10, time_limit=30.0):
    """Run L-BFGS with increasing beta values."""
    t0 = time.monotonic()
    best_true = float('inf')
    best_z = z_init.copy()
    
    for beta in betas:
        if time.monotonic() - t0 > time_limit:
            break
        try:
            res = minimize(
                lambda z: smooth_obj_grad(z, beta)[0],
                best_z.copy(),
                jac=lambda z: smooth_obj_grad(z, beta)[1],
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'gtol': gtol, 'maxls': 50}
            )
            a = softmax(res.x)
            t = true_obj(a)
            if t < best_true:
                best_true = t
                best_z = res.x.copy()
        except Exception as e:
            pass
    
    a = softmax(best_z)
    return a, best_true

def a_to_z(a, eps=1e-15):
    """Convert probability vector to log-parameterization."""
    a = np.maximum(a, eps)
    a = a / a.sum()
    return np.log(a)

def generate_inits(n, parent=None, rng=None):
    """Generate diverse initializations as log-space vectors."""
    if rng is None:
        rng = np.random.default_rng(42)
    x = np.arange(n, dtype=np.float64)
    inits = []
    
    # Uniform
    inits.append(np.zeros(n))
    
    # Linear ramp decreasing
    ramp_d = np.arange(n, 0, -1, dtype=np.float64)
    inits.append(np.log(ramp_d / ramp_d.sum()))
    
    # Linear ramp increasing
    ramp_i = np.arange(1, n + 1, dtype=np.float64)
    inits.append(np.log(ramp_i / ramp_i.sum()))
    
    # Quadratic (U-shaped)
    q = (x - n / 2) ** 2 + 1.0
    inits.append(np.log(q / q.sum()))
    
    # Exponential decay from left
    e1 = np.exp(-x / (n / 3.0))
    inits.append(np.log(e1 / e1.sum()))
    
    # Exponential decay from right
    e2 = np.exp(-x[::-1] / (n / 3.0))
    inits.append(np.log(e2 / e2.sum()))
    
    # Cosine window
    c = np.cos(np.pi * (x / (n - 1) - 0.5)) ** 4 + 0.01
    inits.append(np.log(c / c.sum()))
    
    # Power law
    pw = (x + 1.0) ** 0.3
    inits.append(np.log(pw / pw.sum()))
    
    # Inverse power law
    ipw = 1.0 / (x + 1.0) ** 0.3
    inits.append(np.log(ipw / ipw.sum()))
    
    # Sine
    sn = np.sin(np.pi * x / n) + 0.05
    inits.append(np.log(sn / sn.sum()))
    
    # Random Dirichlet (various concentrations)
    for alpha in [0.1, 0.5, 1.0, 5.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits.append(np.log(np.maximum(r, 1e-15)))
    
    # Parent-based
    if parent is not None:
        px = np.maximum(parent, 1e-15)
        px = px / px.sum()
        inits.append(np.log(px))
        # Blend with uniform
        for t in [0.3, 0.5, 0.7]:
            blended = t * px + (1 - t) * (np.ones(n) / n)
            inits.append(np.log(np.maximum(blended, 1e-15)))
        # Reversed
        inits.append(np.log(np.maximum(px[::-1], 1e-15)))
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS optimization of a
logsumexp-smoothed objective with softmax (log-space) parameterization.
The gradient is computed efficiently via FFT cross-correlation.
Multiple sequence lengths, initializations, and beta schedules are explored
with adaptive time allocation and iterative refinement.
"""
import numpy as np
from scipy.optimize import minimize
import time


def fft_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def true_max_conv(a):
    """True max of auto-convolution."""
    return fft_convolve(a).max()


def smooth_obj_grad(z, beta):
    """
    Compute smooth objective (logsumexp approximation to max conv) and its
    gradient w.r.t. z (log-space parameterization).
    
    Objective: (1/beta) * log(sum_k exp(beta * conv_k))
    Parameterization: a_i = exp(z_i) / sum_j exp(z_j)
    """
    a = softmax(z)
    n = len(a)
    conv = fft_convolve(a)
    bc = beta * conv
    bc_max = bc.max()
    shifted = bc - bc_max
    obj = (1.0 / beta) * (bc_max + np.log(np.exp(shifted).sum()))
    w = np.exp(shifted)
    w /= w.sum()

    # g_a[i] = 2 * sum_j w[i+j] * a[j]  (cross-correlation)
    # = 2 * (u * w)[i + n - 1] where u = a[::-1]
    u = a[::-1].copy()
    # Linear convolution of u (len n) and w (len 2n-1) has length 3n-2
    total_len = 3 * n - 2
    fft_size = 1 << (total_len - 1).bit_length()
    U = np.fft.rfft(u, fft_size)
    W = np.fft.rfft(w, fft_size)
    conv_uw = np.fft.irfft(U * W, fft_size)[:total_len]

    g_a = 2.0 * conv_uw[n - 1:n - 1 + n]

    # Gradient w.r.t. z: g_z[j] = a[j] * (g_a[j] - mu)
    mu = np.dot(g_a, a)
    grad = a * (g_a - mu)
    return obj, grad


def optimize_lbfgs(z_init, betas, max_iter=200, gtol=1e-10, time_limit=30.0, maxls=50):
    """
    Run L-BFGS on the smooth objective with increasing beta values.
    Returns (best probability vector, best true max_conv).
    """
    t0 = time.monotonic()
    best_true = float('inf')
    best_z = z_init.copy()

    for beta in betas:
        if time.monotonic() - t0 > time_limit:
            break
        try:
            res = minimize(
                lambda z: smooth_obj_grad(z, beta)[0],
                best_z.copy(),
                jac=lambda z: smooth_obj_grad(z, beta)[1],
                method='L-BFGS-B',
                options={'maxiter': max_iter, 'gtol': gtol, 'maxls': maxls}
            )
            a = softmax(res.x)
            t = true_max_conv(a)
            if t < best_true:
                best_true = t
                best_z = res.x.copy()
        except Exception:
            pass

    a = softmax(best_z)
    return a, best_true


def a_to_z(a, eps=1e-15):
    """Convert probability vector to log-space parameterization."""
    a = np.maximum(a, eps)
    a = a / a.sum()
    return np.log(a)


def generate_inits_z(n, parent=None, rng=None):
    """Generate diverse initializations in log-space."""
    if rng is None:
        rng = np.random.default_rng(42)
    x = np.arange(n, dtype=np.float64)
    inits = []

    # Uniform
    inits.append(np.zeros(n))

    # Linear ramp decreasing
    ramp_d = np.arange(n, 0, -1, dtype=np.float64)
    inits.append(np.log(ramp_d / ramp_d.sum()))

    # Linear ramp increasing
    ramp_i = np.arange(1, n + 1, dtype=np.float64)
    inits.append(np.log(ramp_i / ramp_i.sum()))

    # Quadratic (U-shaped)
    q = (x - n / 2.0) ** 2 + 1.0
    inits.append(np.log(q / q.sum()))

    # Exponential decay from left
    e1 = np.exp(-x / (n / 3.0))
    inits.append(np.log(e1 / e1.sum()))

    # Exponential decay from right
    e2 = np.exp(-x[::-1] / (n / 3.0))
    inits.append(np.log(e2 / e2.sum()))

    # Cosine window (Hann-like)
    c = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1))) + 0.01
    inits.append(np.log(c / c.sum()))

    # Power law varying exponent
    for pw_exp in [0.2, 0.5, 1.0]:
        pw = (x + 1.0) ** pw_exp
        inits.append(np.log(pw / pw.sum()))

    # Inverse power law
    for pw_exp in [0.3, 0.7]:
        ipw = 1.0 / (x + 1.0) ** pw_exp
        inits.append(np.log(ipw / ipw.sum()))

    # Sine
    sn = np.abs(np.sin(np.pi * x / n)) + 0.02
    inits.append(np.log(sn / sn.sum()))

    # Random Dirichlet with various concentrations
    for alpha in [0.1, 0.3, 0.5, 1.0, 2.0, 5.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits.append(np.log(np.maximum(r, 1e-15)))

    # Random smooth (spline-like)
    for _ in range(3):
        knots = rng.random(8) * np.log(1e5)
        z_rand = np.interp(x, np.linspace(0, n - 1, 8), knots)
        ar = softmax(z_rand)
        inits.append(np.log(np.maximum(ar, 1e-15)))

    # Parent-based
    if parent is not None:
        px = np.maximum(parent, 1e-15)
        px = px / px.sum()
        inits.append(np.log(px))
        # Reversed
        inits.append(np.log(np.maximum(px[::-1], 1e-15)))
        # Blends with uniform
        for t in [0.3, 0.5, 0.7]:
            blended = t * px + (1 - t) * (np.ones(n) / n)
            inits.append(np.log(np.maximum(blended, 1e-15)))
        # Blends with ramp
        for t in [0.5, 0.7]:
            ramp = np.arange(n, 0, -1, dtype=np.float64)
            ramp = ramp / ramp.sum()
            blended = t * px + (1 - t) * ramp
            inits.append(np.log(np.maximum(blended, 1e-15)))

    return inits


def multiplicative_weights_step(a, grad, eta):
    """Multiplicative weights (mirror descent) step on simplex."""
    # grad should be the gradient of the objective w.r.t. a
    # a_new[i] = a[i] * exp(-eta * grad[i]) / Z
    log_a = np.log(np.maximum(a, 1e-300)) - eta * grad
    log_a -= log_a.max()
    a_new = np.exp(log_a)
    s = a_new.sum()
    if s < 1e-300:
        return a.copy()
    return a_new / s


def mw_optimize(a_init, time_limit, eta_init=0.01, top_k=32):
    """Multiplicative weights descent with adaptive step size."""
    a = np.maximum(a_init, 1e-15).astype(np.float64)
    a = a / a.sum()
    n = len(a)
    conv = fft_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    eta = eta_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        # Compute subgradient-like direction using top-k peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            w = np.ones(len(peaks)) / len(peaks)
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            pv = conv[peaks]
            w = pv / pv.sum()

        # g_a[i] = 2 * sum_k w_k * a_{k-i}
        # = 2 * (u * w_full)[i + n - 1]
        w_full = np.zeros(2 * n - 1)
        w_full[peaks] = w
        u = a[::-1].copy()
        total_len = 3 * n - 2
        fft_size = 1 << (total_len - 1).bit_length()
        U = np.fft.rfft(u, fft_size)
        WF = np.fft.rfft(w_full, fft_size)
        conv_uw = np.fft.irfft(U * WF, fft_size)[:total_len]
        g_a = 2.0 * conv_uw[n - 1:n - 1 + n]

        # MW step
        a_new = multiplicative_weights_step(a, g_a, eta)
        conv_new = fft_convolve(a_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            conv = conv_new
            no_improve = 0
            eta = min(eta * 1.05, 0.5)
        else:
            no_improve += 1
            eta *= 0.9
            if no_improve > 100:
                a = best_a + rng.standard_normal(n) * 0.001
                a = np.maximum(a, 0)
                a = a / a.sum()
                conv = fft_convolve(a)
                eta = eta_init
                no_improve = 0

    return best_a, best_obj
