def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_value(f):
    """Compute objective for normalized f (sum=1). Returns max(conv)."""
    conv = fast_convolve(f)
    return conv.max()


def normalize(f):
    """Normalize to sum=1, clip negatives."""
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return np.ones(len(f)) / len(f)
    return f / s


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """Multiple passes of coordinate descent using sparse updates."""
    n = len(f)
    f = normalize(f.copy())
    conv = fast_convolve(f)
    M = conv.max()
    best_f = f.copy()
    best_M = M
    t0 = time.monotonic()
    
    for p in range(num_passes):
        # Compute M_out for each position (max of conv excluding the affected window)
        # M_out[i] = max(conv[0:i], conv[i+n:2n-1])
        # We can precompute prefix and suffix max
        conv_len = 2 * n - 1
        prefix_max = np.maximum.accumulate(conv)
        suffix_max = np.maximum.accumulate(conv[::-1])[::-1]
        
        order = np.arange(n)
        rng = np.random.default_rng(42 + p)
        rng.shuffle(order)
        
        for i in order:
            lo = i
            hi = min(i + n, conv_len)
            # M_out = max of conv outside [lo, hi)
            m_left = prefix_max[lo - 1] if lo > 0 else 0.0
            m_right = suffix_max[hi] if hi < conv_len else 0.0
            M_out = max(m_left, m_right)
            
            f_new, M_new = sparse_coord_update(f, i, conv, M_out)
            if M_new < M - 1e-14:
                f = f_new
                conv = fast_convolve(f)
                M = conv.max()
                if M < best_M - 1e-14:
                    best_M = M
                    best_f = f.copy()
            else:
                # Small random nudge to escape
                if rng.random() < 0.1:
                    f[i] *= (1.0 + rng.standard_normal() * 0.02)
                    f = normalize(f)
                    conv = fast_convolve(f)
                    M = conv.max()
                # Update conv lazily - it might be slightly stale
                # Just update the affected part
                # For simplicity, skip and let next iteration handle it
        
        if verbose:
            print(f"    Pass {p+1}: M={M:.10f}, best={best_M:.10f}, t={time.monotonic()-t0:.1f}s")
    
    return best_f, best_M


def multiplicative_descent(f, time_limit, lr_init=0.02, top_k=20, temp_init=0.0003, verbose=False):
    """Multiplicative weight descent with adaptive parameters."""
    n = len(f)
    f = normalize(f.copy())
    conv = fast_convolve(f)
    best_M = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    cur_lr = lr_init
    cur_temp = temp_init
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        # Select top-k peaks with softmax weighting
        ck = min(top_k, len(conv))
        peaks = np.argpartition(conv, -ck)[-ck:]
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(ck)
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        # Compute pressure on each coordinate
        pressure = np.zeros(n)
        for i in range(ck):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            pressure[idx] += 2.0 * w[i] * f[k - idx]
        
        # Multiplicative update
        f = f * np.exp(-cur_lr * pressure)
        f = np.maximum(f, 1e-30)
        f = f / f.sum()
        
        conv = fast_convolve(f)
        new_M = conv.max()
        
        if new_M < best_M - 1e-15:
            best_M = new_M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        cur_lr *= 0.9997
        cur_temp *= 1.002
        
        if no_improve > 100:
            rng = np.random.default_rng(hash(f.tobytes()) % (2**31))
            f = normalize(best_f * np.exp(rng.standard_normal(n) * 0.008))
            cur_lr = max(lr_init * 0.5, 0.005)
            cur_temp = temp_init * 3.0
            no_improve = 0
    
    return best_f, best_M


def genetic_algorithm(f_init, time_limit, pop_size=24, verbose=False):
    """Genetic algorithm with diverse operators."""
    n = len(f_init)
    rng = np.random.default_rng(12345)
    
    # Initialize population
    pop = [normalize(f_init.copy())]
    # Add diverse individuals
    inits = [
        np.ones(n) / n,
        rng.dirichlet(np.ones(n) * 0.3),
        rng.dirichlet(np.ones(n) * 2.0),
    ]
    # Structured inits
    x = np.arange(n, dtype=np.float64)
    inits.append((np.abs(np.sin(np.pi * x / n)) + 0.1) / (np.abs(np.sin(np.pi * x / n)) + 0.1).sum())
    inits.append((np.abs(np.cos(2 * np.pi * x / n * 3)) + 0.1) / (np.abs(np.cos(2 * np.pi * x / n * 3)) + 0.1).sum())
    # Perturbed parent
    inits.append(normalize(f_init * np.exp(rng.standard_normal(n) * 0.1)))
    inits.append(normalize(f_init * np.exp(rng.standard_normal(n) * 0.3)))
    
    for ini in inits:
        if len(pop) < pop_size // 2:
            pop.append(normalize(ini))
    
    # Fill rest with perturbations
    while len(pop) < pop_size:
        base = pop[rng.integers(0, len(pop))]
        pop.append(normalize(base * np.exp(rng.standard_normal(n) * 0.05)))
    
    # Evaluate
    scores = [obj_value(ind) for ind in pop]
    best_f = pop[np.argmin(scores)].copy()
    best_score = min(scores)
    
    t0 = time.monotonic()
    gen = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        gen += 1
        offspring = []
        
        # Elitism: keep top 2
        order = np.argsort(scores)
        for idx in order[:2]:
            offspring.append(pop[idx].copy())
        
        # Generate rest of population
        while len(offspring) < pop_size:
            # Tournament selection
            t1 = rng.integers(0, 5)
            t2 = rng.integers(0, 5)
            i1 = t1 if scores[t1] < scores[t2] else t2
            
            strategy = rng.random()
            if strategy < 0.4:
                # Blend crossover (Lerp)
                j1 = rng.integers(0, 5)
                j2 = j1 if scores[j1] < scores[j2] else j2
                alpha = rng.random()
                child = alpha * pop[i1] + (1 - alpha) * pop[j2]
                child = normalize(child)
            elif strategy < 0.7:
                # Uniform crossover
                j1 = rng.integers(0, 5)
                j2 = j1 if scores[j1] < scores[j2] else j2
                mask = rng.random(n) < 0.5
                child = np.where(mask, pop[i1], pop[j2])
                child = normalize(child)
            else:
                # Mutation only
                child = pop[i1].copy()
            
            # Mutation
            mut_type = rng.random()
            if mut_type < 0.5:
                # Gaussian mutation
                sigma = 0.02 + 0.03 * min(1.0, gen / 50.0)
                child = child * np.exp(rng.standard_normal(n) * sigma)
            elif mut_type < 0.7:
                # Coordinate mutation (change a few coordinates significantly)
                num_mut = rng.integers(2, 8)
                mut_idx = rng.integers(0, n, num_mut)
                for mi in mut_idx:
                    child[mi] *= rng.uniform(0.3, 2.0)
            elif mut_type < 0.85:
                # Block swap
                b1 = rng.integers(0, n - 10)
                b2 = rng.integers(0, n - 10)
                blen = rng.integers(3, min(15, n // 4))
                seg1 = child[b1:b1+blen].copy()
                blen2 = min(blen, n - b2)
                child[b1:b1+blen2] = child[b2:b2+blen2]
                child[b2:b2+blen2] = seg1[:blen2]
            else:
                # Mirror perturbation
                half = n // 2
                if rng.random() < 0.5:
                    child[half:] = child[:half][::-1] * rng.uniform(0.7, 1.3)
                else:
                    child[:half] = child[half:][::-1] * rng.uniform(0.7, 1.3)
            
            child = normalize(child)
            offspring.append(child)
        
        # Evaluate offspring
        scores = [obj_value(ind) for ind in offspring]
        pop = offspring
        
        curr_best = min(scores)
        if curr_best < best_score - 1e-12:
            best_score = curr_best
            best_f = offspring[np.argmin(scores)].copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Periodically inject diversity
        if no_improve > 20:
            for _ in range(3):
                base = best_f * np.exp(rng.standard_normal(n) * 0.15)
                scores.append(obj_value(normalize(base)))
                pop.append(normalize(base))
            if len(pop) > pop_size:
                order = np.argsort(scores)
                pop = [pop[i] for i in order[:pop_size]]
                scores = [scores[i] for i in order[:pop_size]]
            no_improve = 0
        
        if verbose and gen % 10 == 0:
            print(f"    GA gen {gen}: best={best_score:.10f}, t={time.monotonic()-t0:.1f}s")
    
    return best_f, best_score


def simulated_annealing(f_init, time_limit, t_init=0.005, t_min=0.0001, cool=0.9997, verbose=False):
    """Simulated annealing with structured moves."""
    n = len(f_init)
    rng = np.random.default_rng(999)
    f = normalize(f_init.copy())
    conv = fast_convolve(f)
    curr_M = conv.max()
    best_f = f.copy()
    best_M = curr_M
    T = t_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit and T > t_min:
        it += 1
        move_type = rng.random()
        f_prop = f.copy()
        
        if move_type < 0.4:
            # Perturb single coordinate
            i = rng.integers(0, n)
            f_prop[i] *= rng.uniform(0.5, 1.5)
        elif move_type < 0.6:
            # Perturb few coordinates
            num = rng.integers(2, 6)
            idx = rng.integers(0, n, num)
            f_prop[idx] *= rng.uniform(0.6, 1.4, num)
        elif move_type < 0.75:
            # Block rotation
            b = rng.integers(0, n - 10)
            blen = rng.integers(3, min(12, n // 3))
            shift = rng.integers(1, blen)
            f_prop[b:b+blen] = np.roll(f_prop[b:b+blen], shift)
        elif move_type < 0.85:
            # Swap two coordinates
            i, j = rng.integers(0, n, 2)
            f_prop[i], f_prop[j] = f_prop[j], f_prop[i]
        else:
            # Gaussian perturbation (bigger)
            f_prop = f * np.exp(rng.standard_normal(n) * 0.03)
        
        f_prop = normalize(f_prop)
        new_conv = fast_convolve(f_prop)
        new_M = new_conv.max()
        
        delta = new_M - curr_M
        if delta < 0 or rng.random() < np.exp(-delta / T):
            f = f_prop
            curr_M = new_M
            if curr_M < best_M - 1e-14:
                best_M = curr_M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        T *= cool
        
        # Restart if stuck too long
        if no_improve > 500:
            f = normalize(best_f * np.exp(rng.standard_normal(n) * 0.05))
            curr_M = obj_value(f)
            T = max(T, t_init * 0.3)
            no_improve = 0
    
    return best_f, best_M


def fine_polish(f_init, time_limit, verbose=False):
    """Very fine coordinate descent for final polish."""
    n = len(f_init)
    f = normalize(f_init.copy())
    conv = fast_convolve(f)
    best_M = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        order = np.random.default_rng(int(time.monotonic() * 1000) % 100000).permutation(n)
        
        for i in order:
            # Try small multiplicative changes
            for factor in [0.98, 0.99, 1.01, 1.02]:
                f_try = f.copy()
                f_try[i] *= factor
                if f_try[i] < 0:
                    continue
                f_try = normalize(f_try)
                new_M = obj_value(f_try)
                if new_M < best_M - 1e-15:
                    f = f_try
                    conv = fast_convolve(f)
                    new_M = conv.max()
                    if new_M < best_M - 1e-15:
                        best_M = new_M
                        best_f = f.copy()
                    improved = True
                    break
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 5:
                # Small random restart
                rng = np.random.default_rng(int(time.monotonic() * 100000) % 100000)
                f = normalize(best_f * np.exp(rng.standard_normal(n) * 0.003))
                no_improve = 0
    
    return best_f, best_M


def try_different_lengths(f_best, score_best, time_limit, n_candidates=None, verbose=False):
    """Try different sequence lengths with quick optimization."""
    if n_candidates is None:
        n_candidates = [256, 512, 768]
    
    best_f = f_best.copy()
    best_score = score_best
    t0 = time.monotonic()
    
    for n_try in n_candidates:
        if time.monotonic() - t0 > time_limit:
            break
        rem = time_limit - (time.monotonic() - t0)
        if rem < 30:
            break
        
        # Resample from best to new length
        if len(f_best) != n_try:
            x_old = np.linspace(0, 1, len(f_best))
            x_new = np.linspace(0, 1, n_try)
            f_try = np.interp(x_new, x_old, f_best)
            f_try = normalize(f_try)
        else:
            f_try = f_best.copy()
        
        # Quick multiplicative descent
        alloc = min(rem * 0.5, 120)
        f_opt, m = multiplicative_descent(f_try, time_limit=alloc, lr_init=0.03, top_k=15, temp_init=0.0004)
        
        score = 2.0 * n_try * m
        if score < best_score:
            best_score = score
            best_f = f_opt
            if verbose:
                print(f"  New length {n_try}: score={score:.8f}")
        
        # Quick coordinate descent
        rem2 = time_limit - (time.monotonic() - t0)
        if rem2 > 20:
            f_opt2, m2 = full_coord_descent(f_opt, num_passes=2)
            score2 = 2.0 * n_try * m2
            if score2 < best_score:
                best_score = score2
                best_f = f_opt2
                if verbose:
                    print(f"  Length {n_try} after CD: score={score2:.8f}")
    
    return best_f, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search combining coordinate descent,
    genetic algorithm, simulated annealing, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_f = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_f, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_f = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score
    
    # Load parent
    print("=== Loading parent ===")
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={ps:.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")
        parent = np.ones(384) / 384
        try_seq(parent, "uniform_384")
    
    if best_f is None:
        best_f = parent
    n0 = len(best_f)
    
    # === Phase 1: Coordinate Descent (~400s) ===
    print("\n=== Phase 1: Coordinate Descent ===")
    rem = deadline - time.monotonic() - 1000
    if rem > 30:
        alloc = min(rem, 400)
        print(f"  Allocating {alloc:.0f}s")
        f_opt, m = full_coord_descent(best_f, num_passes=15, verbose=True)
        try_seq(f_opt, "coord_desc")
    
    # === Phase 2: Multiplicative Descent (~300s) ===
    print("\n=== Phase 2: Multiplicative Descent ===")
    rem = deadline - time.monotonic() - 700
    if rem > 20:
        alloc = min(rem, 300)
        # Run with different configs
        configs = [
            {'lr_init': 0.02, 'top_k': 15, 'temp_init': 0.0003},
            {'lr_init': 0.04, 'top_k': 25, 'temp_init': 0.0006},
            {'lr_init': 0.01, 'top_k': 10, 'temp_init': 0.0001},
        ]
        per = max(10, alloc / len(configs))
        for ci, cfg in enumerate(configs):
            rem = deadline - time.monotonic() - 400
            if rem < 10:
                break
            t_a = min(per, rem)
            print(f"  MW config {ci}, t={t_a:.0f}s")
            f_opt, m = multiplicative_descent(best_f, time_limit=t_a, **cfg)
            try_seq(f_opt, f"mw_{ci}")
    
    # === Phase 3: Genetic Algorithm (~350s) ===
    print("\n=== Phase 3: Genetic Algorithm ===")
    rem = deadline - time.monotonic() - 350
    if rem > 30:
        alloc = min(rem, 350)
        print(f"  Allocating {alloc:.0f}s, n={n0}")
        f_opt, m = genetic_algorithm(best_f, time_limit=alloc, pop_size=20, verbose=True)
        try_seq(f_opt, "ga")
    
    # === Phase 4: Simulated Annealing (~200s) ===
    print("\n=== Phase 4: Simulated Annealing ===")
    rem = deadline - time.monotonic() - 150
    if rem > 20:
        alloc = min(rem, 200)
        print(f"  Allocating {alloc:.0f}s")
        f_opt, m = simulated_annealing(best_f, time_limit=alloc, t_init=0.004, t_min=0.00005)
        try_seq(f_opt, "sa")
    
    # === Phase 5: Different Lengths (~200s) ===
    print("\n=== Phase 5: Different Lengths ===")
    rem = deadline - time.monotonic() - 100
    if rem > 30:
        alloc = min(rem, 200)
        print(f"  Allocating {alloc:.0f}s")
        f_opt, sc = try_different_lengths(best_f, best_score, time_limit=alloc,
                                           n_candidates=[512, 256, 768], verbose=True)
        if sc < best_score:
            best_score = sc
            best_f = f_opt
            try_seq(best_f, "new_length")
    
    # === Phase 6: Fine Polish (~100s) ===
    print("\n=== Phase 6: Fine Polish ===")
    rem = deadline - time.monotonic() - 30
    if rem > 15:
        n_best = len(best_f)
        # Coordinate descent polish
        alloc1 = min(rem * 0.5, 50)
        if alloc1 > 10:
            print(f"  Coord polish, t={alloc1:.0f}s")
            f_opt, m = full_coord_descent(best_f, num_passes=8)
            try_seq(f_opt, "polish_cd")
        
        # Fine multiplicative
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            alloc2 = min(rem, 40)
            print(f"  Fine MW, t={alloc2:.0f}s")
            f_opt, m = multiplicative_descent(best_f, time_limit=alloc2,
                                              lr_init=0.005, top_k=8, temp_init=0.00005)
            try_seq(f_opt, "polish_mw")
        
        # Tiny SA
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            f_opt, m = simulated_annealing(best_f, time_limit=min(rem, 30),
                                            t_init=0.001, t_min=0.00002, cool=0.9995)
            try_seq(f_opt, "polish_sa")
    
    # === Final: Additional coord descent passes ===
    rem = deadline - time.monotonic() - 10
    if rem > 10:
        f_opt, m = full_coord_descent(best_f, num_passes=3)
        try_seq(f_opt, "final_cd")
    
    # Fallback
    if best_f is None:
        best_f = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_f.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_f.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_f)})")
    return [float(x) for x in best_f]
