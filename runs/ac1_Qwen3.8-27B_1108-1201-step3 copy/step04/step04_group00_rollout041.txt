"""

import numpy as np

import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_cross_corr(weights, a):
    """Compute out[i] = sum_j a[j] * weights[i+j] for i=0..n-1.
    weights has length >= n, a has length n.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    n_w = len(weights)
    N = 1 << (n_w + n - 1 - 1).bit_length()
    a_rev = a[::-1]
    A = np.fft.rfft(a_rev, N)
    W = np.fft.rfft(weights, N)
    conv_rw = np.fft.irfft(A * W, N)
    return conv_rw[n - 1:2 * n - 1]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, M, n):
    """Solve LP: max 1^T g s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """Standard LP-based Frank-Wolfe with line search and restarts."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        g = solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.12
            if m_exp > 5.0:
                f = project_simplex(best_f + np.random.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
                continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 60:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue

        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.85, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_exp = 1.0
            f = f_new

            if M < best_obj - 1e-14:

                best_f = f.copy()
                no_improve = 0
                if verbose and it % 30 == 0:

                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.08, 4.0)

        if no_improve > 80:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.002 + 0.004 * min(no_improve // 100, 5)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

def smooth_grad_optimize(f0, time_limit, lr=0.003, mom=0.9, temp=0.001,
                         top_k=None, verbose=False):
    """FFT-based smoothed subgradient with momentum and adaptive temperature."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp

    while time.monotonic() - t0 < time_limit:
        # Compute smoothed gradient
        if top_k is not None and top_k < len(conv):
            k_actual = min(top_k, len(conv))
            peaks = np.argpartition(conv, -k_actual)[-k_actual:]
            w = np.zeros(len(conv))
            pv = conv[peaks]
            scaled = pv / cur_temp
            scaled -= scaled.max()
            exp_w = np.exp(scaled)
            ws = exp_w.sum()
            if ws > 1e-15:
                w[peaks] = exp_w / ws
            else:
                w[np.argmax(conv)] = 1.0
        else:
            scaled = conv / cur_temp
            scaled -= scaled.max()
            w = np.exp(scaled)
            ws = w.sum()
            if ws < 1e-15:
                w = np.zeros(len(conv))
                w[np.argmax(conv)] = 1.0
            else:
                w = w / ws

        # Gradient via FFT cross-correlation
        sg = fast_cross_corr(w, f)

        # Momentum update
        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)

        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
            if verbose and no_improve == 0 and int(time.monotonic() - t0) % 5 == 0:
                print(f"    SG iter obj={best_obj:.10f}, temp={cur_temp:.6f}")
        else:
            no_improve += 1

        cur_lr *= 0.9995
        cur_temp *= 1.005

        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0003)
            no_improve = 0

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with local search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in [0.3, 0.5, 0.7, 1.3, 1.7, 2.5]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()
                
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved and passes > 2:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def geometric_init(n, r):
    """Create a geometric sequence initialized on the simplex."""
    a = r ** np.arange(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def sech_init(n):
    """Create a sech-shaped sequence on the simplex."""
    x = np.linspace(-1, 1, n)
    a = 1.0 / np.cosh(2.5 * x)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def lopsided_init(n, alpha=0.7):
    """Create a lopsided sequence (more mass at one end)."""
    a = np.ones(n)
    decay = np.linspace(1.0, alpha, n)
    a = a * decay
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining AFW, smoothed gradient, coordinate descent,
    and multi-length exploration with structured initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        a = [float(x) for x in a]
        score = float(evaluate_sequence(a))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = np.array(a, dtype=np.float64)
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # === Load parent ===
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = sech_init(768)

    # === Phase 1: Quick LP-FW on parent to establish strong baseline ===
    print("\n=== Phase 1: LP-FW on parent ===")
    rem = deadline - time.monotonic() - 600
    if rem > 60:
        t1 = min(rem, 180)
        print(f"  LP-FW on parent (n={len(parent)}), t={t1:.0f}s")
        a_opt, _ = lp_fw_optimize(parent, time_limit=t1, verbose=True)
        try_seq(a_opt, "lpfw_parent")

    # === Phase 2: AFW on best ===
    print(f"\n=== Phase 2: AFW on best (score={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 500
        if rem > 40:
            t2 = min(rem, 150)
            print(f"  AFW (n={len(best_seq)}), t={t2:.0f}s")
            a_opt, _ = afgw_optimize(best_seq, time_limit=t2, verbose=True)
            try_seq(a_opt, "afw")

    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-length exploration (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else parent
    n_base = len(base)

    target_lengths = []
    if n_base <= 512:
        target_lengths = [768, 1024, 1536]
    elif n_base <= 768:
        target_lengths = [1024, 1536, 2048]
    elif n_base <= 1024:
        target_lengths = [1536, 2048]
    else:
        target_lengths = [1024, 768]

    for target_n in target_lengths:
        rem = deadline - time.monotonic() - 300
        if rem < 50:
            break
        f_interp = interpolate_sequence(base, target_n)
        t_alloc = min(rem * 0.5, 120)
        print(f"  LP-FW n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = lp_fw_optimize(f_interp, time_limit=t_alloc, verbose=True)
        try_seq(a_opt, f"lpfw_n{target_n}")

    # === Phase 4: Structured initializations ===
    print(f"\n=== Phase 4: Structured inits (best={best_score:.6f}) ===")
    n_try = len(best_seq) if best_seq is not None else 768

    inits = []
    # Geometric with various ratios
    for r in [0.95, 0.97, 0.98, 0.99]:
        inits.append(geometric_init(n_try, r))
    # Sech
    inits.append(sech_init(n_try))
    # Lopsided
    inits.append(lopsided_init(n_try, 0.5))
    inits.append(lopsided_init(n_try, 0.7))
    # Reversed best
    if best_seq is not None:
        inits.append(best_seq[::-1].copy())
    # Blend with uniform
    if best_seq is not None:
        inits.append(project_simplex(0.7 * best_seq + 0.3 * np.ones(n_try) / n_try))

    for idx, f_init in enumerate(inits):
        rem = deadline - time.monotonic() - 200
        if rem < 20:
            break
        t_alloc = min(rem / max(1, len(inits) - idx), 60)
        if t_alloc < 10:
            break
        a_opt, _ = lp_fw_optimize(f_init, time_limit=t_alloc, verbose=False)
        try_seq(a_opt, f"init_{idx}")

    # === Phase 5: Smoothed gradient polishing ===
    print(f"\n=== Phase 5: Smoothed gradient polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_sg = min(rem * 0.6, 120)
            print(f"  SG polish n={n_best}, t={t_sg:.0f}s")
            a_opt, _ = smooth_grad_optimize(
                best_seq, time_limit=t_sg,
                lr=0.002, mom=0.92, temp=0.0005,
                top_k=max(20, n_best // 6), verbose=True
            )
            try_seq(a_opt, "sg_polish")

            # Second pass with different params
            rem = deadline - time.monotonic() - 60
            if rem > 15:
                a_opt, _ = smooth_grad_optimize(
                    best_seq, time_limit=min(rem, 60),
                    lr=0.001, mom=0.95, temp=0.0002,
                    top_k=max(10, n_best // 4), verbose=False
                )
                try_seq(a_opt, "sg_fine")

    # === Phase 6: Coordinate descent ===
    print(f"\n=== Phase 6: Coordinate descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t_cd = min(rem * 0.7, 100)
            print(f"  CD t={t_cd:.0f}s")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd")

    # === Phase 7: Final LP-FW + perturbations ===
    print(f"\n=== Phase 7: Final optimization (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # One more LP-FW round
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=min(rem, 60), verbose=True)
            try_seq(a_opt, "final_lpfw")

        # Perturbations with SG polish
        for p in range(4):
            rem = deadline - time.monotonic() - 15
            if rem < 10:
                break
            noise = rng.standard_normal(n_best) * (0.001 + 0.002 * p)
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = smooth_grad_optimize(
                f_pert, time_limit=min(rem, 30),
                lr=0.001, mom=0.93, temp=0.0003,
                top_k=max(15, n_best // 5)
            )
            try_seq(a_opt, f"perturb_{p}")

        # Interpolate to slightly different length and optimize
        for scale in [0.9, 1.1]:
            rem = deadline - time.monotonic() - 10
            if rem < 10:
                break
            new_n = int(n_best * scale)
            if new_n < 64 or new_n > 4096:
                continue
            f_interp = interpolate_sequence(best_seq, new_n)
            a_opt, _ = lp_fw_optimize(f_interp, time_limit=min(rem, 30))
            try_seq(a_opt, f"interp_n{new_n}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence([float(x) for x in best_seq]))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
