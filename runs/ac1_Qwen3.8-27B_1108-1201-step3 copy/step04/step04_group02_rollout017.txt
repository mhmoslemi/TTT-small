    f = f.copy()
    f = f / f.sum()
    n = len(f)
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

def coordinate_descent(f0, time_limit, verbose=False):

    it = 0

                break

            spread = max(x0 * 0.5, 1.0 / n)
            candidates = np.unique(np.concatenate([
                [0.0, x0, 1.0/n],
                np.linspace(max(0, x0 - spread), min(0.5, x0 + spread), 8)

                    best_x = x_new
            
            f[i] = best_x

        if obj < best_obj - 1e-14:

def sg_descent(f0, time_limit, initial_lr=0.005, mom=0.9, top_k=32, temp=0.001):
    """Subgradient descent with momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = initial_lr
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        
        # Compute subgradient with softmax over top peaks
        k_eff = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_eff)[-k_eff:]
        pv = conv[peaks]
        scaled = pv / max(temp, 1e-10)
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        
        sg = np.zeros(n)
        for i, k in enumerate(peaks):
            if w[i] < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += w[i] * 2.0 * f[k - idx]
        
        v = mom * v - cur_lr * sg
        f_new = project_simplex(f + v)
        if f_new.sum() > 1e-15:
            f = f_new
            f = f / f.sum()
        
        conv = fast_convolve(f)
        obj = conv.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        cur_lr *= 0.999
        temp *= 1.002
        
        if no_improve > 200:
            rng = np.random.default_rng(it * 7 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            cur_lr = initial_lr * 0.5
            no_improve = 0
    
    return best_f, best_obj

def generate_inits(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    inits.append(np.ones(n) / n)
    
    # Ramp
    ramp = np.arange(n, 0, -1, dtype=np.float64)
    inits.append(ramp / ramp.sum())
    
    # Gaussian
    g = np.exp(-0.5 * ((x - n/2) / (n/4))**2)
    inits.append(g / g.sum())
    
    # Cosine
    c = 0.5 + 0.5 * np.cos(2 * np.pi * x / n)
    inits.append(c / c.sum())
    
    # Two peaks
    p1 = np.exp(-0.5 * ((x - n*0.3) / (n/8))**2)
    p2 = np.exp(-0.5 * ((x - n*0.7) / (n/8))**2)
    inits.append((p1 + p2) / (p1 + p2).sum())
    
    # Random
    r = rng.dirichlet(np.ones(n) * 0.5)
    inits.append(r)
    
    if parent is not None:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    return inits

min max_k (f*g)[k] s.t. g >= 0, sum(g) = 1

min t

     sum(g) = 1
     g >= 0

The optimal t is the minimum possible max cross-convolution. If t < M (current max auto-conv), then the direction is useful.

    # Minimize: t
    # Constraints: A@g - t <= 0 (i.e., A@g <= t), sum(g) = 1, g >= 0, t >= 0

            g = np.maximum(res.x[:n], 0)
            t = res.x[-1]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Frank-Wolfe with minmax LP oracle (more principled direction finding)
2. Fast coordinate descent with grid search for polishing
3. Subgradient descent with adaptive parameters
4. Multi-phase strategy: explore -> converge -> polish

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f):
    """Build A where A[k,j] = f[k-j] if 0<=k-j<n, else 0."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_minmax(f, A=None, time_limit=5.0):
    """Solve: min t s.t. A@g <= t, sum(g)=1, g>=0. Returns (g, t)."""
    n = len(f)
    if A is None:
        A = build_lp_matrix(f)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.hstack([A, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(0, None)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            g = np.maximum(res.x[:n], 0.0)
            t_val = res.x[-1]
            gs = g.sum()
            if gs > 1e-15:
                g = g / gs
            return g, t_val
    except Exception:
        pass
    return None, None


def solve_lp_maxsum(f, M, A=None, time_limit=5.0):
    """Solve: max sum(g) s.t. A@g <= M, g>=0."""
    n = len(f)
    if A is None:
        A = build_lp_matrix(f)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M), bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            if g.sum() > 1e-15:
                return g / g.sum()
    except Exception:
        pass
    return None


def fw_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with minmax LP oracle + backtracking line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    A = build_lp_matrix(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 3:
            break

        # Rebuild A periodically
        if it % 20 == 0:
            A = build_lp_matrix(f)

        # Try minmax LP first
        g, t_lp = solve_lp_minmax(f, A, time_limit=min(5.0, rem / 3))

        if g is None:
            # Fallback: max-sum LP with current M
            M_try = M * (1.0 + 0.05 * (it % 5))
            g = solve_lp_maxsum(f, M_try, A, time_limit=min(4.0, rem / 3))
            if g is None:
                # Gradient fallback
                k_star = np.argmax(conv)
                js = max(0, k_star - n + 1)
                je = min(n - 1, k_star)
                idx = np.arange(js, je + 1)
                sg = np.zeros(n)
                sg[idx] = 2.0 * f[k_star - idx]
                f = project_simplex(f - 0.005 * sg)
                conv = fast_convolve(f)
                M = conv.max()
                continue

        # Check similarity to f
        if np.linalg.norm(g - f) < 1e-9:
            no_improve += 1
            if no_improve > 40:
                rng = np.random.default_rng(it * 11 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                A = build_lp_matrix(f)
                no_improve = 0
            continue

        # Backtracking line search (geometric)
        best_alpha = 0.0
        best_val = M
        alpha = 1.0
        while alpha > 1e-5:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                alpha *= 0.5
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
            alpha *= 0.5

        # Fine search around best
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(8):
                if hi - lo < 1e-7:
                    break
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-8:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if verbose and it % 50 == 0:
            print(f"    FW it={it}, M={M:.10f}, best={best_obj:.10f}, alpha={best_alpha:.4f}")

    return best_f, best_obj


def coordinate_descent(f0, time_limit, n_grid=24, verbose=False):
    """Coordinate descent with dense grid search and renormalization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(789)
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break

        improved_this_round = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - t0 > time_limit - 0.2:
                return best_f, best_obj

            x0 = f[i]
            if x0 < 1e-10:
                continue

            # Search range: [0, max(2*x0, 3/n)]
            hi = min(0.5, max(2.5 * x0, 3.0 / n, 1.5 / n))
            lo = 0.0

            # Create grid (denser near x0)
            grid = np.unique(np.concatenate([
                [0.0, x0, hi],
                np.linspace(lo, hi, n_grid),
                np.linspace(max(0, x0 - 0.3 * x0), min(hi, x0 + 0.3 * x0), 8)
            ]))

            best_x = x0
            best_m = best_obj

            for x_new in grid:
                f[i] = x_new
                ss = f.sum()
                if ss < 1e-15:
                    f[i] = x0
                    continue
                fn = f / ss
                m = fast_convolve(fn).max()
                if m < best_m:
                    best_m = m
                    best_x = x_new

            if best_x != x0:
                f[i] = best_x
                f = f / f.sum()
                improved_this_round = True
                if best_m < best_obj - 1e-15:
                    best_obj = best_m
                    best_f = f.copy()

        if not improved_this_round:
            # Try wider perturbations
            if it < 20:
                f_pert = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                conv_p = fast_convolve(f_pert)
                if conv_p.max() < best_obj:
                    f = f_pert
                    best_obj = conv_p.max()
                    best_f = f.copy()
                continue
            else:
                break

        if verbose and it % 5 == 0:
            print(f"    CD round={it}, obj={best_obj:.10f}")

    return best_f, best_obj


def sg_descent(f0, time_limit, initial_lr=0.005, mom=0.9, top_k=32, temp=0.001):
    """Subgradient descent with momentum and softmax peak weighting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = initial_lr
    cur_temp = temp

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.3:
            break

        k_eff = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_eff)[-k_eff:]
        pv = conv[peaks]
        scaled = pv / max(cur_temp, 1e-10)
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

        sg = np.zeros(n)
        for i, k in enumerate(peaks):
            if w[i] < 1e-12:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += w[i] * 2.0 * f[k - idx]

        v = mom * v - cur_lr * sg
        f_new = project_simplex(f + v)
        f = f_new
        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9995
        cur_temp *= 1.003

        if no_improve > 300:
            rng = np.random.default_rng(it * 13 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            v = np.zeros(n)
            cur_lr = initial_lr * 0.6
            no_improve = 0

    return best_f, best_obj


def generate_inits(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = []
    x = np.arange(n, dtype=np.float64)

    inits.append(np.ones(n) / n)

    ramp = np.arange(n, 0, -1, dtype=np.float64)
    inits.append(ramp / ramp.sum())

    g = np.exp(-0.5 * ((x - n / 2) / (n / 4)) ** 2)
    inits.append(g / g.sum())

    c = 0.5 + 0.5 * np.cos(2 * np.pi * x / n)
    inits.append(c / c.sum())

    # Two Gaussian peaks
    p1 = np.exp(-0.5 * ((x - n * 0.25) / (n / 10)) ** 2)
    p2 = np.exp(-0.5 * ((x - n * 0.75) / (n / 10)) ** 2)
    inits.append((p1 + p2) / (p1 + p2).sum())

    # Power law
    pw = (x + 1) ** 0.7
    inits.append(pw / pw.sum())

    r = rng.dirichlet(np.ones(n) * 0.3)
    inits.append(r)

    if parent is not None:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits
