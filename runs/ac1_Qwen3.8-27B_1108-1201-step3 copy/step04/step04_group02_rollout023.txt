Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid of:
1. Exact coordinate descent (each coordinate solved via bisection on the objective level)
2. LP-based Frank-Wolfe for global escape from local minima
3. Structured initialization and diverse restarts

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, conv, top_k=64, temp=0.001):
    """Compute subgradient of max(conv) with softmax weighting over top peaks."""
    n = len(f)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]

    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= wsum

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k - idx]

    return sg


def solve_lp(f, M, time_limit=30):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def coord_descent_sweep(a, max_sweeps=50, time_limit=60, tol=1e-12):
    """
    Full coordinate descent: sweep through all coordinates, updating each.
    Returns (best_a, best_obj_normalized).
    """
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return a, float('inf')
    
    conv = fast_convolve(a)
    best_obj = conv.max() / (S * S)
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    
    for sweep in range(max_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        
        improved = False
        for i in range(n):
            old_val = a[i]
            new_val = coord_descent_step(a, conv, i, S)
            if abs(new_val - old_val) > 1e-14:
                a[i] = new_val
                S = S - old_val + new_val
                conv = fast_convolve(a)  # Recompute (could be optimized)
                obj = conv.max() / (S * S)
                if obj < best_obj - 1e-13:
                    best_obj = obj
                    best_a = a.copy()
                improved = True
        
        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
        else:
            no_improve = 0
    
    return best_a, best_obj


def fw_lp_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 3:
            break

        g = solve_lp(f, M * 1.0, time_limit=min(rem - 2, 20))
        if g is None:
            # Try with slightly larger M
            g = solve_lp(f, M * 1.05, time_limit=min(rem - 2, 20))
            if g is None:
                no_improve += 1
                if no_improve > 30:
                    break
                continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 50:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

    return best_f, best_obj


def generate_inits(n, rng=None):

    ramp_i = np.arange(1, n + 1, dtype=np.float64)

    c = np.cos(np.pi * (x / (n - 1) - 0.5)) ** 4 + 0.01
    inits.append(c / c.sum())
    
    sn = np.sin(np.pi * x / n) + 0.05
    inits.append(sn / sn.sum())

    tri = np.minimum(x, n - 1 - x) + 1.0
    inits.append(tri / tri.sum())

    st = (np.minimum(x, n - 1 - x) + 1.0) ** 2

    bi = np.exp(-((x - n*0.25)/5)**2) + np.exp(-((x - n*0.75)/5)**2) + 0.01
    inits.append(bi / bi.sum())
    
    return inits

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp(f, M, time_limit=20):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

    inits.append(np.ones(n) / n)

    # Ramps
    rd = np.arange(n, 0, -1, dtype=np.float64)
    inits.append(rd / rd.sum())
    ri = np.arange(1, n + 1, dtype=np.float64)
    inits.append(ri / ri.sum())

    # Triangular
    tri = np.minimum(x, n - 1 - x) + 1.0
    inits.append(tri / tri.sum())

    # Super-triangular
    st = (np.minimum(x, n - 1 - x) + 1.0) ** 2
    inits.append(st / st.sum())

    # Cosine window
    c = np.cos(np.pi * (x / (n - 1) - 0.5)) ** 2 + 0.01
    inits.append(c / c.sum())

    # Exponential
    e1 = np.exp(-x / (n / 4.0))
    inits.append(e1 / e1.sum())

    # Sine
    sn = np.sin(np.pi * x / n) + 0.05
    inits.append(sn / sn.sum())

    # Bimodal
    bi = np.exp(-((x - n * 0.3) / (n * 0.15)) ** 2) + np.exp(-((x - n * 0.7) / (n * 0.15)) ** 2) + 0.01
    inits.append(bi / bi.sum())

    # Random Dirichlet
    r = rng.dirichlet(np.ones(n) * 0.5)
    inits.append(r)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search: coordinate descent + LP-Frank-Wolfe hybrid with
    diverse initialization and adaptive time allocation.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        a = np.minimum(a, 1000.0)
        if a.sum() < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            old = best_score
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}) (was {old:.8f})", flush=True)
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={parent_score:.8f}", flush=True)
    except (NameError, Exception):
        pass

    # === Phase 1: Coordinate descent on parent (fast local optimization) ===
    print("=== Phase 1: Coord Descent on Parent ===", flush=True)
    if parent is not None:
        rem = deadline - time.monotonic() - 400
        t1 = min(150, max(30, rem))
        print(f"  n={len(parent)}, t={t1:.0f}s", flush=True)
        a_opt, obj = coord_descent_sweep(parent, max_sweeps=300, time_limit=t1)
        try_seq(a_opt, "parent_cd")

    # === Phase 2: LP-FW for global improvement ===
    print(f"\n=== Phase 2: LP-Frank-Wolfe ===", flush=True)
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        t2 = min(200, max(30, rem))
        print(f"  t={t2:.0f}s", flush=True)
        a_opt, obj = fw_lp_optimize(best_seq, time_limit=t2)
        try_seq(a_opt, "fw")

        # Follow with CD
        rem = deadline - time.monotonic() - 250
        if rem > 20:
            a_opt2, _ = coord_descent_sweep(a_opt, max_sweeps=200, time_limit=min(rem, 80))
            try_seq(a_opt2, "fw_cd")

    # === Phase 3: Explore diverse initializations ===
    print(f"\n=== Phase 3: Diverse Exploration ===", flush=True)
    if best_seq is not None:
        current_n = len(best_seq)
    else:
        current_n = 384

    lengths = [current_n, 384, 256, 512, 320]
    # Remove duplicates, keep order
    seen = set()
    unique_lengths = []
    for l in lengths:
        if l not in seen and 50 <= l <= 1024:
            seen.add(l)
            unique_lengths.append(l)

    for n in unique_lengths:
        if time.monotonic() > deadline - 250:
            break

        inits = generate_inits(n, rng)
        rem = deadline - time.monotonic() - 200
        t_per = max(15, rem // (len(inits) + 1))
        t_per = min(t_per, 40)

        for i, f_init in enumerate(inits):
            if time.monotonic() > deadline - 150:
                break
            t_alloc = min(t_per, deadline - time.monotonic() - 130)
            if t_alloc < 10:
                break

            a_opt, obj = coord_descent_sweep(f_init, max_sweeps=80, time_limit=t_alloc)
            try_seq(a_opt, f"n{n}_i{i}")

    print(f"  Phase 3 best: {best_score:.8f}", flush=True)

    # === Phase 4: Deep alternating optimization ===
    print(f"\n=== Phase 4: Deep Optimization ===", flush=True)
    if best_seq is not None:
        for rnd in range(8):
            rem = deadline - time.monotonic() - 80
            if rem < 20:
                break
            t_alloc = min(rem // 2, 60)

            if rnd % 2 == 0:
                a_opt, _ = coord_descent_sweep(best_seq, max_sweeps=200, time_limit=t_alloc)
                try_seq(a_opt, f"deep_cd_{rnd}")
            else:
                a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
                try_seq(a_opt, f"deep_fw_{rnd}")
                # Quick CD follow-up
                rem2 = deadline - time.monotonic() - 60
                if rem2 > 15:
                    a_opt2, _ = coord_descent_sweep(a_opt, max_sweeps=100, time_limit=min(rem2, 30))
                    try_seq(a_opt2, f"deep_fwcd_{rnd}")

    # === Phase 5: Final intense polishing ===
    print(f"\n=== Phase 5: Final Polish ===", flush=True)
    print(f"  Current best: {best_score:.8f}", flush=True)

    if best_seq is not None:
        # Long CD
        rem = deadline - time.monotonic() - 20
        if rem > 20:
            a_opt, _ = coord_descent_sweep(best_seq, max_sweeps=500, time_limit=min(rem, 90))
            try_seq(a_opt, "polish_cd_long")

        # FW
        rem = deadline - time.monotonic() - 15
        if rem > 15:
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=min(rem, 50))
            try_seq(a_opt, "polish_fw")

        # Final CD
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            a_opt, _ = coord_descent_sweep(best_seq, max_sweeps=200, time_limit=min(rem, 30))
            try_seq(a_opt, "polish_cd_final")

        # Micro perturbations + CD
        rem = deadline - time.monotonic() - 3
        if rem > 8:
            for p in range(3):
                rem = deadline - time.monotonic() - 3
                if rem < 5:
                    break
                perturb = best_seq.copy()
                mask = rng.random(len(perturb)) < 0.1
                perturb[mask] *= (1 + rng.standard_normal(mask.sum()) * 0.02)
                perturb = np.maximum(perturb, 0)
                ps = perturb.sum()
                if ps > 1e-10:
                    perturb = perturb / ps
                a_opt, _ = coord_descent_sweep(perturb, max_sweeps=100, time_limit=min(rem, 15))
                try_seq(a_opt, f"micro_{p}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})", flush=True)
    return [float(x) for x in best_seq]
