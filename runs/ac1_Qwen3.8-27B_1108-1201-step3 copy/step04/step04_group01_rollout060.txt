"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f):
    """Build sparse constraint matrix for the LP.
    A[k, j] = f[k-j] if 0 <= k-j < n, else 0.
    Shape: (2n-1, n)
    """
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            for j in js:
                rows.append(k)
                cols.append(int(j))
                vals.append(float(f[k - j]))
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, A=None):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(f)
    if A is None:
        A = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 30})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def fw_lp_step(f, time_limit, relaxation=1.0):
    """One FW-LP iteration with line search."""
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    
    A = build_lp_matrix(f)
    M_eff = M * relaxation
    g = solve_lp_direction(f, M_eff, A)
    
    if g is None:
        return f, M, relaxation * 1.1
    if g.sum() < 1e-15:
        return f, M, relaxation * 1.1
    
    g = np.maximum(g, 0.0)
    g = g / g.sum()
    
    # Line search
    best_alpha = 0.0
    best_val = M
    for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        ft = np.maximum((1.0 - alpha) * f + alpha * g, 0.0)
        st = ft.sum()
        if st < 1e-15:
            continue
        ft = ft / st
        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = alpha
    
    # Bisection refinement
    if best_alpha > 0:
        lo = max(0.0, best_alpha - 0.05)
        hi = min(1.0, best_alpha + 0.05)
        for _ in range(8):
            mid = (lo + hi) / 2.0
            ft = np.maximum((1.0 - mid) * f + mid * g, 0.0)
            st = ft.sum()
            if st < 1e-15:
                break
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = mid
                lo = mid
            else:
                hi = mid
    
    if best_alpha > 1e-10:
        f_new = np.maximum((1.0 - best_alpha) * f + best_alpha * g, 0.0)
        f_new = f_new / f_new.sum()
        return f_new, best_val, 1.0
    return f, M, relaxation * 1.05

    lr = 3.0 / n
    momentum = 0.92
    v = np.zeros(n)

    t0 = time.monotonic()

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. FW-LP for global structural optimization
2. LSE (log-sum-exp) gradient descent with momentum for smooth polishing
3. Coordinate descent for fast local refinement

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f):
    """Build sparse constraint matrix for the LP. Shape: (2n-1, n)."""
    n = len(f)
    total_entries = 0
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        total_entries += max(0, j_end - j_start + 1)
    
    rows = np.empty(total_entries, dtype=np.int32)
    cols = np.empty(total_entries, dtype=np.int32)
    vals = np.empty(total_entries, dtype=np.float64)
    idx = 0
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = f[k - j]
            idx += 1
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, A):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 25})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

            no_improve += 1
            relaxation = min(relaxation * 1.06, 5.0)
        
        if no_improve > 40:
            rng = np.random.default_rng(it * 17 + 42)
            scale = 0.002 + 0.003 * min(1.0, (no_improve - 40) / 40.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            best_obj = conv.max()
            best_f = f.copy()
            relaxation = 1.0
            no_improve = 0
        
        if it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW-LP it={it}, obj={best_obj:.12f}, elapsed={elapsed:.0f}s")
    
    return best_f, best_obj


def compute_lse_gradient(weights, f):
    """Compute gradient of LSE objective: 2 * cross_correlation(weights, f)."""
    n = len(f)
    f_flipped = f[::-1].copy()
    total_len = len(weights) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]


def lse_optimize(f0, time_limit, beta_start=20.0, beta_end=5000.0):
    """Log-sum-exp smoothed gradient descent with momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    beta = beta_start
    lr = 3.0 / n
    momentum = 0.90
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(f)
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()
        
        grad = compute_lse_gradient(w, f)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            no_improve += 100
            break
        grad = grad / grad_norm
        
        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.002, 6.0 / n)
        else:
            no_improve += 1
            if no_improve >= 12:
                lr *= 0.93
                momentum = max(0.85, momentum - 0.004)
            if no_improve > 90:
                f = best_f.copy()
                v = np.zeros(n)
                lr = max(2.0 / n, 0.001)
                beta = min(beta * 1.4, beta_end)
                momentum = 0.88
                no_improve = 0
        
        if beta < beta_end:
            beta *= 1.004
        
        if it % 400 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE it={it}, obj={best_obj:.12f}, beta={beta:.1f}, lr={lr:.6f}, elapsed={elapsed:.0f}s")
    
    return best_f, best_obj


def cd_optimize(f0, time_limit):
    """Coordinate descent: optimize one coordinate at a time.
    Minimizes max(conv(k)) with incremental updates. Normalizes after each sweep.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    sweep = 0
    
    while time.monotonic() - t0 < time_limit:
        sweep += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        improved = False
        for i in range(n):
            old = f[i]
            k_min = i
            k_max = min(i + n - 1, 2 * n - 2)
            
            # Precompute the "base" conv values for relevant k's
            # conv_new[k] = conv[k] - old_contrib + new_contrib
            # For k = 2i: conv_new = conv[k] - old^2 + t^2
            # For k != 2i: conv_new = conv[k] - 2*old*f[k-i] + 2*t*f[k-i]
            
            # Find optimal t via ternary/golden search on [0, t_hi]
            t_hi = max(2.0 * old + 1.0 / n, 2.0 / n)
            
            # Golden section search (faster than bisection for unimodal)
            gr = (np.sqrt(5.0) - 1.0) / 2.0  # ~0.618
            a_lo = 0.0
            a_hi = t_hi
            c_candidate = a_hi - gr * (a_hi - a_lo)
            d_candidate = a_lo + gr * (a_hi - a_lo)
            
            def eval_t(t):
                m = 0.0
                for k in range(k_min, k_max + 1):
                    if k == 2 * i:
                        val = conv[k] - old * old + t * t
                    else:
                        j2 = k - i
                        if 0 <= j2 < n:
                            val = conv[k] - 2.0 * old * f[j2] + 2.0 * t * f[j2]
                        else:
                            val = conv[k]
                    if val > m:
                        m = val
                # Also check non-relevant
                non_rel_max = conv[0:k_min].max() if k_min > 0 else 0.0
                if k_max < 2 * n - 2:
                    nr2 = conv[k_max + 1:].max()
                    if nr2 > non_rel_max:
                        non_rel_max = nr2
                if non_rel_max > m:
                    m = non_rel_max
                return m
            
            c_val = eval_t(c_candidate)
            d_val = eval_t(d_candidate)
            
            for _ in range(20):
                if c_val < d_val:
                    a_hi = d_candidate
                    d_candidate = c_candidate
                    d_val = c_val
                    c_candidate = a_hi - gr * (a_hi - a_lo)
                    c_val = eval_t(c_candidate)
                else:
                    a_lo = c_candidate
                    c_candidate = d_candidate
                    c_val = d_val
                    d_candidate = a_lo + gr * (a_hi - a_lo)
                    d_val = eval_t(d_candidate)
            
            t_opt = (a_lo + a_hi) / 2.0
            t_opt = max(0.0, t_opt)
            
            if abs(t_opt - old) > 1e-10:
                # Update conv incrementally
                for k in range(k_min, k_max + 1):
                    if k == 2 * i:
                        conv[k] = conv[k] - old * old + t_opt * t_opt
                    else:
                        j2 = k - i
                        if 0 <= j2 < n:
                            conv[k] = conv[k] - 2.0 * old * f[j2] + 2.0 * t_opt * f[j2]
                f[i] = t_opt
                improved = True
        
        # Normalize
        s = f.sum()
        if s < 1e-15:
            break
        f = f / s
        conv = fast_convolve(f)
        obj = conv.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if not improved:
                no_improve += 5
        
        if no_improve > 30:
            break
        
        if sweep % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD sweep={sweep}, obj={best_obj:.12f}, elapsed={elapsed:.0f}s")
    
    return best_f, best_obj


def subgrad_polish(f0, time_limit, n_peaks=16):
    """Subgradient descent weighting only the top peaks of conv.
    More focused than full LSE when near convergence.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    lr = 1.5 / n
    momentum = 0.93
    v = np.zeros(n)
    temp = 0.0001
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(f)
        
        # Select top peaks
        if n_peaks >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -n_peaks)[-n_peaks:]
        
        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / w_sum
        
        # Subgradient: sum of weighted contributions
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx_arr = np.arange(js, je + 1)
            sg[idx_arr] += 2.0 * w[idx] * f[k - idx_arr]
        
        v = momentum * v - lr * sg
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.001, 4.0 / n)
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                lr *= 0.95
            if no_improve > 200:
                rng = np.random.default_rng(it * 13 + 5)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                v = np.zeros(n)
                lr = max(lr * 0.5, 0.0005)
                no_improve = 0
        
        temp *= 1.003
        
        if it % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG it={it}, obj={best_obj:.12f}, lr={lr:.6f}, elapsed={elapsed:.0f}s")
    
    return best_f, best_obj


def try_sequence(a, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.maximum(a, 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase optimization:
    Phase 1: FW-LP structural optimization
    Phase 2: LSE gradient polishing  
    Phase 3: Coordinate descent local refinement
    Phase 4: FW-LP refinement on CD result
    Phase 5: Subgradient polish + final LSE
    Phase 6: Alternative length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    
    best_tracker = [float('inf'), None]
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}")
            try_sequence(parent, best_tracker)
    except Exception as e:
        print(f"No parent available: {e}")
    
    if parent is None:
        n = 448
        x = np.linspace(0, 1, n)
        parent = 0.5 * (1.0 + np.cos(2.0 * np.pi * x))
        parent = np.maximum(parent, 1e-8)
        parent = parent / parent.sum()
        print(f"Fallback init: n={n}")
        try_sequence(parent, best_tracker)
    
    n_parent = len(parent)
    
    # === Phase 1: FW-LP ===
    t_rem = deadline - time.monotonic()
    t_alloc = min(320, t_rem * 0.19)
    t_alloc = max(t_alloc, 60)
    print(f"\n=== Phase 1: FW-LP ({t_alloc:.0f}s) ===")
    a_opt, _ = fw_lp_optimize(parent, time_limit=t_alloc)
    try_sequence(a_opt, best_tracker)
    
    # === Phase 2: LSE ===
    t_rem = deadline - time.monotonic()
    current = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc = min(420, t_rem * 0.26)
    t_alloc = max(t_alloc, 60)
    print(f"\n=== Phase 2: LSE ({t_alloc:.0f}s) ===")
    a_opt, _ = lse_optimize(current, time_limit=t_alloc,
                            beta_start=15.0, beta_end=3000.0)
    try_sequence(a_opt, best_tracker)
    
    # === Phase 3: CD ===
    t_rem = deadline - time.monotonic()
    current = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc = min(220, t_rem * 0.22)
    t_alloc = max(t_alloc, 40)
    print(f"\n=== Phase 3: CD ({t_alloc:.0f}s) ===")
    a_opt, _ = cd_optimize(current, time_limit=t_alloc)
    try_sequence(a_opt, best_tracker)
    
    # === Phase 4: FW-LP refinement ===
    t_rem = deadline - time.monotonic()
    current = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc = min(220, t_rem * 0.22)
    t_alloc = max(t_alloc, 40)
    print(f"\n=== Phase 4: FW-LP Refine ({t_alloc:.0f}s) ===")
    a_opt, _ = fw_lp_optimize(current, time_limit=t_alloc)
    try_sequence(a_opt, best_tracker)
    
    # === Phase 5: SG + LSE final ===
    t_rem = deadline - time.monotonic()
    current = best_tracker[1] if best_tracker[1] is not None else parent
    
    # Subgradient polish
    t_alloc = min(130, t_rem * 0.25)
    t_alloc = max(t_alloc, 30)
    print(f"\n=== Phase 5a: SG Polish ({t_alloc:.0f}s) ===")
    a_opt, _ = subgrad_polish(current, time_limit=t_alloc,
                               n_peaks=max(8, len(current) // 28))
    try_sequence(a_opt, best_tracker)
    
    # Final LSE with high beta
    t_rem = deadline - time.monotonic()
    current = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc = min(120, t_rem * 0.25)
    t_alloc = max(t_alloc, 25)
    print(f"\n=== Phase 5b: LSE High Beta ({t_alloc:.0f}s) ===")
    a_opt, _ = lse_optimize(current, time_limit=t_alloc,
                            beta_start=80.0, beta_end=10000.0)
    try_sequence(a_opt, best_tracker)
    
    # Final CD
    t_rem = deadline - time.monotonic()
    current = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc = min(80, t_rem * 0.3)
    t_alloc = max(t_alloc, 20)
    print(f"\n=== Phase 5c: Final CD ({t_alloc:.0f}s) ===")
    a_opt, _ = cd_optimize(current, time_limit=t_alloc)
    try_sequence(a_opt, best_tracker)
    
    # === Phase 6: Alternative lengths ===
    t_rem = deadline - time.monotonic()
    print(f"\n=== Phase 6: Alt Lengths (remaining ~{t_rem:.0f}s) ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    
    alt_lengths = []
    for delta in [-2, -1, 1, 2, 3]:
        n_alt = n_parent + delta
        if 384 <= n_alt <= 512:
            alt_lengths.append(n_alt)
    
    for n_alt in alt_lengths[:2]:
        t_rem = deadline - time.monotonic()
        if t_rem < 60:
            break
        t_alloc = min(120, t_rem - 30)
        
        # Interpolate
        x_old = np.linspace(0, 1, len(current_best))
        x_new = np.linspace(0, 1, n_alt)
        f_interp = np.interp(x_new, x_old, current_best)
        f_interp = np.maximum(f_interp, 0)
        if f_interp.sum() > 1e-10:
            f_interp = f_interp / f_interp.sum()
        
        print(f"  Trying n={n_alt} with {t_alloc:.0f}s")
        
        # CD then LSE
        t1 = t_alloc * 0.45
        t2 = t_alloc * 0.55
        a_opt, _ = cd_optimize(f_interp, time_limit=t1)
        try_sequence(a_opt, best_tracker)
        
        current_alt = best_tracker[1] if best_tracker[1] is not None else f_interp
        if len(current_alt) == n_alt:
            a_opt, _ = lse_optimize(current_alt, time_limit=t2,
                                    beta_start=50.0, beta_end=5000.0)
            try_sequence(a_opt, best_tracker)
    
    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.8f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]
    
    # Fallback
    final_seq = np.ones(384) / 384
    score = float(evaluate_sequence(final_seq.tolist()))
    print(f"\nFALLBACK: score={score:.8f}")
    return [float(x) for x in final_seq]
