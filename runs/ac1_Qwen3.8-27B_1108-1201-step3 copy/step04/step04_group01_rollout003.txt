import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    ind = np.nonzero(cond)[0][-1]
    theta = (cssv[ind] - 1.0) / (ind + 1.0)
    return np.maximum(v - theta, 0.0)


def build_A_sparse(f):
    """Build sparse matrix A where A[k,j] = f[k-j] if 0<=k-j<n.
    Shape: (2n-1, n)
    """
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            js = np.arange(j_min, j_max + 1)
            fs = f[k - js]
            rows.extend(np.full(len(js), k).tolist())
            cols.extend(js.tolist())
            vals.extend(fs.tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_minM(f, A_cache=None):
    """Solve LP: min M s.t. A@g <= M, sum(g)=1, g>=0.
    Returns (g, M) or (None, None).
    """

    if A_cache is None:
        A_cache = build_A_sparse(f)
    
    # Variables: g[0..n-1], M
    # Objective: minimize M -> c = [0,...,0,1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: A@g - M <= 0
    # A has shape (2n-1, n), we need to extend with -1 in the M column
    A_ub = A_cache[:, :-1]  # This doesn't work directly with sparse...
    
    # Let me build it differently
    # A_ub shape: (2n-1, n+1)
    # A_ub[k, 0:n] = A[k, :], A_ub[k, n] = -1
    n_ineq = 2 * n - 1
    rows = []
    cols = []
    vals = []
    
    # Add A part
    rows.extend(A_cache.row.tolist())
    cols.extend(A_cache.col.tolist())
    vals.extend(A_cache.data.tolist())
    
    # Add -M column
    rows.extend(range(n_ineq))
    cols.extend([n] * n_ineq)
    vals.extend([-1.0] * n_ineq)
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(n_ineq, n + 1))
    b_ub = np.zeros(n_ineq)

def compute_lse_grad(weights, f):
    """Compute gradient of LSE approximation.
    grad[i] = 2 * sum_{m=0}^{n-1} weights[i+m] * f[m] (cross-correlation)
    """
    n = len(f)
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]
