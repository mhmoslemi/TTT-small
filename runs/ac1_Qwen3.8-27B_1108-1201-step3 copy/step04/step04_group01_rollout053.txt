"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses FW-LP with aggressive line search, LSE gradient polishing with adaptive
beta up to very high values, and alternating refinement phases.
Key improvements: tighter line search, adaptive M management, ultra-high-beta LSE,
and multiple restart strategies to escape local minima.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_grad(weights, f):
    """Compute gradient: 2 * cross_correlation(weights, f) of length n."""
    n = len(f)
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]


def build_lp_matrix(f):
    """Build sparse constraint matrix for the LP. Shape: (2n-1, n)"""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend(f[k - js].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_for_direction(f, M, A_cache=None):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(f)
    if A_cache is None:
        A_cache = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 60})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def fw_line_search(f, g, M_current, n_evals=24, n_bisect=14):
    """Aggressive line search for FW step. Returns (best_alpha, best_f, best_val)."""
    best_alpha = 0.0
    best_val = M_current

    # Grid search
    alphas = np.linspace(0.0, 1.0, n_evals + 1)[1:]
    best_grid_alpha = 0.0
    for alpha in alphas:
        ft = (1.0 - alpha) * f + alpha * g
        ft = np.maximum(ft, 0.0)
        st = ft.sum()
        if st < 1e-15:
            continue
        ft = ft / st
        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_grid_alpha = alpha
            best_alpha = alpha

    if best_alpha == 0.0:
        return 0.0, f.copy(), M_current

    # Bisection refinement
    lo = max(0.0, best_grid_alpha - 1.0 / n_evals)
    hi = min(1.0, best_grid_alpha + 1.0 / n_evals)
    for _ in range(n_bisect):
        if hi - lo < 1e-10:
            break
        mid1 = lo + (hi - lo) / 3.0
        mid2 = hi - (hi - lo) / 3.0
        ft1 = np.maximum((1.0 - mid1) * f + mid1 * g, 0.0)
        st1 = ft1.sum()
        ft1 = ft1 / st1 if st1 > 1e-15 else ft1
        val1 = fast_convolve(ft1).max() if st1 > 1e-15 else np.inf

        ft2 = np.maximum((1.0 - mid2) * f + mid2 * g, 0.0)
        st2 = ft2.sum()
        ft2 = ft2 / st2 if st2 > 1e-15 else ft2
        val2 = fast_convolve(ft2).max() if st2 > 1e-15 else np.inf

        if val1 < val2:
            hi = mid2
            if val1 < best_val:
                best_val = val1
                best_alpha = mid1
        else:
            lo = mid1
            if val2 < best_val:
                best_val = val2
                best_alpha = mid2

    ft = np.maximum((1.0 - best_alpha) * f + best_alpha * g, 0.0)
    st = ft.sum()
    if st > 1e-15:
        ft = ft / st
    return best_alpha, ft, best_val


def fw_lp_optimize(f0, time_limit, n_grid=24, n_bisect=14):
    """Frank-Wolfe with LP oracle and adaptive constraint relaxation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        A_cache = build_lp_matrix(f)
        g = solve_lp_for_direction(f, M * m_exp, A_cache)
        if g is None:
            m_exp *= 1.15
            if m_exp > 8.0:
                m_exp = 1.0
            no_improve += 1
            if no_improve > 10:
                # Restart with perturbation
                rng = np.random.default_rng(it * 31 + 7)
                scale = 0.001 + 0.002 * min(1.0, no_improve / 50.0)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                rng = np.random.default_rng(it * 13 + 3)
                scale = 0.002 + 0.003 * min(1.0, no_improve / 60.0)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            continue

        # Line search
        alpha, f_new, val = fw_line_search(f, g, M, n_grid=n_grid, n_bisect=n_bisect)

        if alpha > 1e-10 and val < M:
            f = f_new
            M = val
            m_exp = 1.0
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.08, 6.0)
            if no_improve > 50:
                rng = np.random.default_rng(it * 17 + 42)
                scale = 0.002 + 0.004 * min(1.0, (no_improve - 50) / 50.0)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0

        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW it={it}, obj={best_obj:.12f}, M={M:.12f}, m_exp={m_exp:.2f}, elapsed={elapsed:.0f}s")

    return best_f, best_obj


def lse_optimize(f0, time_limit, beta_start=20.0, beta_end=50000.0, lr_start=None):
    """Log-sum-exp smoothed gradient descent with adaptive parameters."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    if lr_start is None:
        lr = 3.0 / n
    else:
        lr = lr_start
    momentum = 0.92
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)

        # Softmax weights (numerically stable)
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w_sum = exp_bc.sum()
        if w_sum < 1e-300:
            w = np.zeros(len(conv))
            w[np.argmax(conv)] = 1.0
        else:
            w = exp_bc / w_sum

        # Gradient via FFT
        grad = compute_lse_grad(w, f)

        # Normalize gradient for stable step size
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            no_improve += 100
            break
        grad = grad / grad_norm

        # Momentum update
        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.003, 6.0 / n)
        else:
            no_improve += 1
            if no_improve >= 8:
                lr *= 0.93
                momentum = max(0.85, momentum - 0.003)
            if no_improve > 100:
                # Restart
                f = best_f.copy()
                v = np.zeros(n)
                lr = max(3.0 / n, 0.001)
                beta = min(beta * 2.0, beta_end)
                momentum = 0.90
                no_improve = 0
                # Small perturbation
                if no_improve > 150:
                    rng = np.random.default_rng(it * 7 + 1)
                    f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                    no_improve = 0

        # Ramp up beta
        if beta < beta_end:
            beta *= 1.004

        if it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE it={it}, obj={best_obj:.12f}, beta={beta:.1f}, lr={lr:.6f}, elapsed={elapsed:.0f}s")

    return best_f, best_obj


def subgrad_polish(f0, time_limit, top_k=64, temp_init=0.0001):
    """Subgradient descent using top-k peak weighting for final polishing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = 1.5 / n
    cur_temp = temp_init
    v = np.zeros(n)
    cur_mom = 0.95
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)

        # Select top-k peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / w_sum

        # Subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = cur_mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                cur_lr *= 0.96
            if no_improve > 400:
                rng = np.random.default_rng(it * 23 + 5)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.0008)
                v = np.zeros(n)
                cur_lr = max(1.0 / n, 0.0003)
                no_improve = 0

        cur_temp *= 1.003

        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG it={it}, obj={best_obj:.12f}, temp={cur_temp:.6f}, lr={cur_lr:.6f}, elapsed={elapsed:.0f}s")

    return best_f, best_obj
