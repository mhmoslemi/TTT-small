"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines FW-LP with spectral initialization, LSE polishing with adaptive
optimizers, and multi-length exploration.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, lil_matrix
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_lse_grad(weights, f):
    """Compute gradient of LSE approximation."""
    n = len(f)
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]

def build_lp_sparse(f):
    """Build sparse constraint matrix for LP: A[k,j] = f[k-j] if valid."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend(f[k - js].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def solve_lp(f, M, A_sparse):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_sparse, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

def spectral_init(n, rng, num_iters=50):
    """Gerchberg-Saxton style spectral initialization."""
    f = rng.dirichlet(np.ones(n))
    for _ in range(num_iters):
        F = np.fft.fft(f)
        mag = np.abs(F)
        mag[mag < 1e-10] = 1e-10
        target = np.sqrt(n) * np.ones(n)
        F_new = target * F / mag
        f_new = np.real(np.fft.ifft(F_new))
        f = np.maximum(f_new, 0.0)
        s = f.sum()
        if s < 1e-15:
            f = rng.dirichlet(np.ones(n))
        else:
            f = f / s
    return f

def lse_optimize(f0, time_limit, beta_start=30.0, beta_end=5000.0):
    """LSE gradient descent with AdaGrad optimizer."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = 1.0 / np.sqrt(n)
    v = np.zeros(n)  # AdaGrad accumulator
    beta_v = 0.99   # Bias correction for beta
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    best_lr = lr

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()

        grad = compute_lse_grad(w, f)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            no_improve += 100
            break
        grad = grad / grad_norm

        # AdaGrad update
        v = v + grad * grad
        step = lr / np.sqrt(v + 1e-8)
        f_new = project_simplex(f - step)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 20 == 0:
                lr *= 0.9
            if no_improve > 100:
                f = best_f.copy()
                v = np.zeros(n)
                beta = min(beta * 1.3, beta_end)
                lr = max(lr * 0.5, 0.0001)
                no_improve = 0

        if beta < beta_end:
            beta *= 1.002

    return best_f, best_obj

def subgrad_polish(f0, time_limit, top_k=32, lr_init=0.003):
    """Subgradient descent at the active peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    lr = lr_init
    momentum = 0.93
    v = np.zeros(n)
    temp = 0.0001

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        
        # Get top-k peaks with softmax weighting
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        
        pv = conv[peaks]
        scaled = (pv - pv.max()) / (temp + 1e-10)
        w = np.exp(scaled)
        w = w / w.sum()

        # Compute subgradient
        sg = np.zeros(n)
        for i, k in enumerate(peaks):
            if w[i] < 1e-12:
                continue
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = momentum * v - lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                lr *= 0.95
            if no_improve > 200:

                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                v = np.zeros(n)
                lr = max(lr * 0.5, 0.0003)
                no_improve = 0

        temp *= 1.001

    return best_f, best_obj

Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses accelerated Frank-Wolfe with LP oracle, Adam-based LSE polishing,
spectral initialization, and multi-length exploration.
Key improvements: momentum-accelerated FW, adaptive LP constraints,

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_grad(weights, f):
    """Compute gradient of LSE: 2 * cross_correlation(weights, f)."""
    n = len(f)
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]


def build_lp_sparse(f):
    """Build sparse constraint matrix: A[k,j] = f[k-j] if 0<=k-j<n."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend(f[k - js].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_full(f, M, A_sparse):
    """Solve full LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_sparse, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def solve_lp_reduced(f, M, A_sparse, prev_g, num_keep=200):
    """Solve LP with reduced constraint set, verify against full."""
    n = len(f)
    num_rows = 2 * n - 1

        return solve_lp_full(f, M, A_sparse)
    binds = A_sparse @ prev_g
    ratios = binds / max(M, 1e-10)
    top_idx = np.argsort(ratios)[::-1][:num_keep]

    A_red = A_sparse[top_idx, :]
    c = -np.ones(n)
    b_ub = np.full(len(top_idx), M)
    bounds = [(0.0, None)] * n
    try:

            if np.all(full_binds <= M * 1.0001):

    except Exception:
        pass
    return solve_lp_full(f, M, A_sparse)


def spectral_init(n, rng, num_iters=80):
    """Gerchberg-Saxton style spectral initialization for flat spectrum."""
    f = rng.dirichlet(np.ones(n))
    for _ in range(num_iters):
        F = np.fft.fft(f)
        mag = np.abs(F)
        mag[mag < 1e-12] = 1e-12
        target = np.sqrt(n) * np.ones(n)
        F_new = target * F / mag
        f_new = np.real(np.fft.ifft(F_new))
        f = np.maximum(f_new, 0.0)
        s = f.sum()
        if s < 1e-15:
            f = rng.dirichlet(np.ones(n))
        else:
            f = f / s
    return f


def fw_lp_optimize(f0, time_limit, m_expand=1.12, use_reduced_lp=True, reduce_keep=200):
    """Accelerated Frank-Wolfe with LP oracle and adaptive line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    prev_g = None
    rng = np.random.default_rng(98765)

    # Momentum for accelerated FW
    momentum = 0.0
    v_fw = np.zeros(n)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        A_sparse = build_lp_sparse(f)

        if use_reduced_lp and it > 5:
            g = solve_lp_reduced(f, M * m_exp, A_sparse, prev_g, reduce_keep)
        else:
            g = solve_lp_full(f, M * m_exp, A_sparse)

        if g is None or g.sum() < 1e-15:
            m_exp = min(m_exp * m_expand, 10.0)
            if m_exp >= 10.0:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.008)
                m_exp = 1.0
                conv = fast_convolve(f)
                M = conv.max()
                prev_g = None
            continue

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        g = g / g_sum
        prev_g = g

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 35:
                scale = 0.004 + 0.004 * min(1.0, (no_improve - 35) / 80.0)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                m_exp = 1.0
                conv = fast_convolve(f)
                M = conv.max()
                prev_g = None
                no_improve = 0
            continue

        # Line search: 20-step bisection
        best_alpha = 0.0
        best_val = M

        lo, hi = 0.0, 1.0
        for _ in range(20):
            mid = (lo + hi) / 2.0
            ft = (1.0 - mid) * f + mid * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                hi = mid
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = mid
                lo = mid
            else:
                hi = mid

        if best_alpha > 1e-10:
            # Apply step with optional momentum
            f_new = (1.0 - best_alpha) * f + best_alpha * g
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()

            # Try a slightly overshooting step (acceleration)
            if it % 30 == 0 and best_alpha > 0.3:
                alpha_overshoot = min(1.0, best_alpha * 1.2)
                f_test = (1.0 - alpha_overshoot) * f + alpha_overshoot * g
                f_test = np.maximum(f_test, 0.0)
                st = f_test.sum()
                if st > 1e-15:
                    f_test = f_test / st
                    val_test = fast_convolve(f_test).max()
                    if val_test < best_val:
                        best_val = val_test
                        best_alpha = alpha_overshoot
                        f_new = f_test

            f = f_new
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.04, 6.0)

        if no_improve > 60:
            scale = 0.005 + 0.005 * min(1.0, (no_improve - 60) / 100.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            prev_g = None
            no_improve = 0

    return best_f, best_obj


def lse_adam_optimize(f0, time_limit, beta_start=50.0, beta_end=15000.0, lr_init=0.5):
    """LSE gradient descent with Adam optimizer."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = lr_init / np.sqrt(n)
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    adam_t = 0
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()

        grad = compute_lse_grad(w, f)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-16:
            no_improve += 200
            break
        grad = grad / grad_norm

        adam_t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** adam_t)
        v_hat = v / (1 - beta2 ** adam_t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        f_new = project_simplex(f - step)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 15 == 0:
                lr *= 0.92
            if no_improve > 80:
                f = best_f.copy()
                m = np.zeros(n)
                v = np.zeros(n)
                adam_t = 0
                beta = min(beta * 1.4, beta_end)
                lr = max(lr * 0.6, 0.01 / np.sqrt(n))
                no_improve = 0

        if beta < beta_end:
            beta *= 1.003

    return best_f, best_obj


def subgrad_polish(f0, time_limit, top_k=32, lr_init=0.003):
    """Subgradient descent at active peaks with momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    lr = lr_init
    momentum = 0.92
    v = np.zeros(n)
    temp = 0.00005

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        scaled = (pv - pv.max()) / (temp + 1e-10)
        w = np.exp(scaled)
        w = w / w.sum()

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = momentum * v - lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 25 == 0:
                lr *= 0.94
            if no_improve > 150:
                rng2 = np.random.default_rng((it + 7919) % 1000003)
                f = project_simplex(best_f + rng2.standard_normal(n) * 0.003)
                v = np.zeros(n)
                lr = max(lr * 0.5, 0.0003)
                no_improve = 0

        temp *= 1.002

    return best_f, best_obj


def try_sequence(a, best_tracker):
    """Evaluate and update best."""
    a = np.maximum(a, 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score


def propose_candidate(seed=42, budget_s=1800):
    """
    Main multi-phase optimization:
    1. FW-LP initialization on parent
    2. Extended accelerated FW-LP (main workhorse)
    3. LSE + Adam polishing
    4. Subgradient fine-tuning
    5. FW-LP refinement on polished result
    6. Multi-length exploration
    7. Final alternating refinement
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    print(f"Budget: {budget_s}s, deadline in {float(budget_s)}s")

    best_tracker = [float('inf'), None]

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_sequence(parent.tolist())
            print(f"Parent: n={len(parent)}, score={ps:.8f}")
            try_sequence(parent, best_tracker)
    except Exception:
        pass

    if parent is None:
        parent = np.ones(448) / 448
        try_sequence(parent, best_tracker)

    current_best = best_tracker[1].copy() if best_tracker[1] is not None else parent
    n_best = len(current_best)

    # === Phase 1: Quick FW-LP init (100s) ===
    print(f"\n=== Phase 1: FW-LP Init (n={n_best}) ===")
    t1 = min(100, deadline - time.monotonic() - 1450)
    if t1 > 15:
        a_opt, obj = fw_lp_optimize(current_best, time_limit=t1)
        try_sequence(a_opt, best_tracker)
        current_best = best_tracker[1].copy() if best_tracker[1] is not None else current_best
    print(f"  Phase 1 done: {best_tracker[0]:.8f}, t_remaining={deadline - time.monotonic():.0f}s")

    # === Phase 2: Extended FW-LP (600s) ===
    print(f"\n=== Phase 2: Extended FW-LP ===")
    t2 = min(600, deadline - time.monotonic() - 850)
    if t2 > 30:
        a_opt, obj = fw_lp_optimize(current_best, time_limit=t2,
                                    m_expand=1.10, use_reduced_lp=True, reduce_keep=250)
        try_sequence(a_opt, best_tracker)
        current_best = best_tracker[1].copy() if best_tracker[1] is not None else current_best
    print(f"  Phase 2 done: {best_tracker[0]:.8f}, t_remaining={deadline - time.monotonic():.0f}s")

    # === Phase 3: LSE + Adam (350s) ===
    print(f"\n=== Phase 3: LSE + Adam ===")
    t3 = min(350, deadline - time.monotonic() - 500)
    if t3 > 20:
        a_opt, obj = lse_adam_optimize(current_best, time_limit=t3,
                                       beta_start=30.0, beta_end=10000.0,
                                       lr_init=0.8)
        try_sequence(a_opt, best_tracker)
        current_best = best_tracker[1].copy() if best_tracker[1] is not None else current_best
    print(f"  Phase 3 done: {best_tracker[0]:.8f}, t_remaining={deadline - time.monotonic():.0f}s")

    # === Phase 4: Subgradient (120s) ===
    print(f"\n=== Phase 4: Subgradient ===")
    t4 = min(120, deadline - time.monotonic() - 330)
    if t4 > 10:
        n_cur = len(current_best)
        a_opt, obj = subgrad_polish(current_best, time_limit=t4,
                                    top_k=max(16, n_cur // 14), lr_init=0.002)
        try_sequence(a_opt, best_tracker)
        current_best = best_tracker[1].copy() if best_tracker[1] is not None else current_best
    print(f"  Phase 4 done: {best_tracker[0]:.8f}, t_remaining={deadline - time.monotonic():.0f}s")

    # === Phase 5: FW-LP refinement (200s) ===
    print(f"\n=== Phase 5: FW-LP Refinement ===")
    t5 = min(200, deadline - time.monotonic() - 200)
    if t5 > 15:
        a_opt, obj = fw_lp_optimize(current_best, time_limit=t5,
                                    m_expand=1.08, use_reduced_lp=True, reduce_keep=300)
        try_sequence(a_opt, best_tracker)
        current_best = best_tracker[1].copy() if best_tracker[1] is not None else current_best
    print(f"  Phase 5 done: {best_tracker[0]:.8f}, t_remaining={deadline - time.monotonic():.0f}s")

    # === Phase 6: Multi-length exploration ===
    print(f"\n=== Phase 6: Multi-length ===")
    alt_lengths = [384, 512, 320, 480]
    if n_best in alt_lengths:
        alt_lengths.remove(n_best)

    for n_alt in alt_lengths[:3]:
        t_rem = deadline - time.monotonic() - 80
        if t_rem < 50:
            break
        t_alloc = min(60, t_rem // max(1, 3 - alt_lengths.index(n_alt)))
        if t_alloc < 10:
            break
        print(f"  Trying n={n_alt}, budget={t_alloc:.0f}s")

        x_old = np.linspace(0, 1, len(current_best))
        x_new = np.linspace(0, 1, n_alt)
        f_interp = np.interp(x_new, x_old, current_best)
        f_interp = np.maximum(f_interp, 0)
        if f_interp.sum() > 1e-10:
            f_interp = f_interp / f_interp.sum()

        t_fwl = max(8, t_alloc * 0.6)
        t_lse = max(5, t_alloc * 0.4)
        a_opt, _ = fw_lp_optimize(f_interp, time_limit=t_fwl)
        try_sequence(a_opt, best_tracker)

        cur_alt = best_tracker[1].copy() if best_tracker[1] is not None else f_interp
        if len(cur_alt) == n_alt:
            a_opt, _ = lse_adam_optimize(cur_alt, time_limit=t_lse,
                                         beta_start=30.0, beta_end=3000.0)
            try_sequence(a_opt, best_tracker)

    print(f"  Phase 6 done: {best_tracker[0]:.8f}, t_remaining={deadline - time.monotonic():.0f}s")

    # === Phase 7: Final alternating refinement ===
    print(f"\n=== Phase 7: Final Refinement ===")
    current_best = best_tracker[1].copy() if best_tracker[1] is not None else parent
    t_rem = deadline - time.monotonic() - 5
    if t_rem > 20:
        n_rounds = max(1, int(t_rem // 25))
        beta_cursor = 200.0
        for r in range(n_rounds):
            t_rem = deadline - time.monotonic() - 5
            if t_rem < 8:
                break
            # Short FW burst
            t_fw = min(20, t_rem * 0.5)
            a_opt, _ = fw_lp_optimize(current_best, time_limit=t_fw)
            try_sequence(a_opt, best_tracker)
            current_best = best_tracker[1].copy() if best_tracker[1] is not None else current_best

            t_rem = deadline - time.monotonic() - 5
            if t_rem < 8:
                break
            # LSE burst with increasing beta
            t_lse = min(18, t_rem * 0.5)
            a_opt, _ = lse_adam_optimize(current_best, time_limit=t_lse,
                                         beta_start=beta_cursor, beta_end=beta_cursor * 20)
            try_sequence(a_opt, best_tracker)
            current_best = best_tracker[1].copy() if best_tracker[1] is not None else current_best
            beta_cursor *= 3.0

    print(f"\n=== COMPLETE ===")
    print(f"  Final best: {best_tracker[0]:.8f}")

    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"  VERIFIED: {final_score:.8f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    fallback = np.ones(448) / 448
    return [float(x) for x in fallback]
