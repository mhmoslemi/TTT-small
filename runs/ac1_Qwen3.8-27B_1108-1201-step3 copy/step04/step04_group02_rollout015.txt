def fast_convolve(a):

    n = len(a)
    size = 2 * n - 1

    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_smooth_grad(a, temp, n_peaks=256):

    Compute gradient of (1/temp) * log(sum_k exp(temp * (a*a)[k])).
    Returns (gradient, conv).

        pv = conv[peaks] * temp
        pv -= pv.max()

def solve_lp_oracle(f, M, time_limit=10):
    """
    Solve: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns g or None.
    """
    n = len(f)
    c = -np.ones(n)
    
    # Build constraint matrix: (f*g)[k] = sum_j f[max(0,k-n+1):min(n,k+1)]_reversed * g
    # A_ub[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # This is a banded/tropical matrix
    
    A = lil_matrix((2*n-1, n))
    for j in range(n):
        # (f*g)[k] includes f[j] * g[k-j] for valid k-j
        # So A[k, j] = f[j] for k such that 0 <= k-j < n, i.e., j <= k <= j+n-1
        k_start = j
        k_end = j + n - 1
        A[k_start:k_end+1, j] = f[:n]
    
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A.tocsr(), b_ub=b_ub, bounds=bounds, 
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Smooth optimization (log-sum-exp surrogate + Adam) for fast local improvement
2. LP-based Frank-Wolfe for global direction finding
3. Multi-length exploration (n=384, 512, 768)
4. Pairwise transfer local search for final polishing
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csc_matrix
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_grad_smooth(a, temp, n_peaks=512):
    n = len(a)
    conv = fast_convolve(a)
    if n_peaks >= len(conv):
        peaks = np.arange(len(conv))
        w = np.ones(len(peaks)) / len(peaks)
    else:
        peaks = np.argpartition(conv, -n_peaks)[-n_peaks:]
        pv = conv[peaks] * temp
        pv -= pv.max()
        w = np.exp(pv)
        w /= w.sum()
    
    grad = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-15:
            continue
        k = int(peaks[i])
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        g = 2.0 * a[k - idx]
        if k % 2 == 0:
            mid = k // 2
            if lo <= mid <= hi:
                g[mid - lo] = a[mid]
        grad[lo:hi+1] += w[i] * g
    
    return grad, conv

def solve_lp_oracle(f, M, time_limit=15):
    n = len(f)
    c = -np.ones(n)
    A = lil_matrix((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = f
    b_ub = np.full(2*n-1, M) * 1.0  # small slack to avoid infeasibility
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A.tocsr(), b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def adam_smooth_optimize(f0, time_limit, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8,
                         init_temp=0.5, temp_growth=1.002, n_peaks=512):
    """Adam optimization on the log-sum-exp surrogate with simplex projection."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = 2 * n * conv.max()
    best_f = f.copy()
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    t0 = time.monotonic()
    temp = init_temp
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        if time.monotonic() - t0 > time_limit - 1:
            break
        
        grad, conv = compute_grad_smooth(f, temp, n_peaks)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        lr_t = lr * np.sqrt(1 - beta2**t) / (1 - beta1**t)
        f_new = project_simplex(f - lr_t * m_hat / (np.sqrt(v_hat) + eps))
        
        f = f_new
        conv = fast_convolve(f)
        obj = 2 * n * conv.max()
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        temp *= temp_growth
        
        if no_improve > 2000:
            # Reset
            rng = np.random.default_rng(t)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            conv = fast_convolve(f)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            temp = init_temp * 0.5
            no_improve = 0
    
    return best_f, best_obj / (2 * n)  # return max_conv

def fpw_optimize(f0, time_limit, n_peaks=256, init_temp=0.1):
    """Frank-Wolfe with LP oracle and backtracking line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    M_factor = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 8:
            break
        
        # Solve LP oracle
        g = solve_lp_oracle(f, M * M_factor, time_limit=min(15, rem * 0.3))
        if g is None:
            M_factor = min(M_factor * 1.3, 5.0)
            if M_factor > 5.0:
                # Reset with noise
                rng = np.random.default_rng(it * 7)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                conv = fast_convolve(f)
                M = conv.max()
                M_factor = 1.0
                no_improve = 0
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            M_factor = min(M_factor * 1.05, 3.0)
            if no_improve > 100:
                rng = np.random.default_rng(it * 11)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
                conv = fast_convolve(f)
                M = conv.max()
                M_factor = 1.0
                no_improve = 0
            continue
        
        # Backtracking line search
        best_alpha = 0.0
        best_val = M
        alpha = 1.0
        for _ in range(30):
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                alpha *= 0.7
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
            alpha *= 0.7
            if alpha < 0.001:
                break
        
        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            M_factor = 1.0
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            M_factor = min(M_factor * 1.2, 5.0)
        
        if no_improve > 60:
            rng = np.random.default_rng(it * 13 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            conv = fast_convolve(f)
            M = conv.max()
            M_factor = 1.0
            no_improve = 0
    
    return best_f, best_obj

def pairwise_transfer_search(f0, time_limit, n_trials=None):
    """Local search via pairwise mass transfers."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(12345)
    
    if n_trials is None:
        n_trials = 10**9
    
    while time.monotonic() - t0 < time_limit and it < n_trials:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        
        # Pick random pair
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        
        # Transfer t from j to i: a_i += t, a_j -= t
        # t in [-(a_j), a_i + a_j - min_val] but we just need 0 <= a_i+t and 0 <= a_j-t
        # So t in [-a_j, a_j]... wait, a_i + t >= 0 => t >= -a_i, a_j - t >= 0 => t <= a_j
        # So t in [-a_i, a_j]
        
        # Try a few values of t
        vals = [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5]
        best_t = 0.0
        best_val = best_obj
        
        for sign in [1, -1]:
            for t_val in vals:
                t = sign * t_val
                if t < -f[i] or t > f[j]:
                    continue
                ft = f.copy()
                ft[i] += t
                ft[j] -= t
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_t = t
        
        if best_t != 0.0:
            f[i] += best_t
            f[j] -= best_t
            conv = fast_convolve(f)
            obj = conv.max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 500:
            break
    
    return best_f, best_obj

def generate_inits(n, parent=None, rng=None):
    if rng is None:
        rng = np.random.default_rng(42)
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    inits.append(np.ones(n) / n)
    
    # Symmetric triangular
    tri = np.minimum(x, n-1-x) + 1
    inits.append(tri / tri.sum())
    
    # Symmetric cosine
    c = np.cos(np.pi * x / (n - 1)) ** 2 + 0.01
    inits.append(c / c.sum())
    
    # Exponential
    e = np.exp(-x / (n/4)) + np.exp(-(n-1-x) / (n/4)) + 0.01
    inits.append(e / e.sum())
    
    # Random
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    
    # Parent-based
    if parent is not None:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    return inits
