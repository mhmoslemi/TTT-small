```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-accelerated log-sum-exp
smoothed gradient descent with adaptive temperature, peak-directed subgradient
methods, and coordinate-wise refinement. Key innovations: closed-form gradient
via cross-correlation FFT, adaptive peak focusing, and multi-scale restarts.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_weights(conv, t):
    """Compute softmax weights for LSE approximation."""
    scaled = conv / t
    scaled -= scaled.max()
    w = np.exp(scaled)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= s
    return w


def compute_lse_gradient(a, w):
    """Compute gradient of LSE using cross-correlation via FFT.
    grad_i = 2 * sum_k w_k * a_{k-i} (valid indices only)
    This equals 2 * convolve(w, flip(a))[n-1 : 2n-1]
    """
    n = len(a)
    flip_a = a[::-1]
    # Linear convolution of w (length 2n-1) and flip_a (length n)
    result_len = (2 * n - 1) + n - 1  # = 3n - 2
    fft_size = 1 << (result_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    Af = np.fft.rfft(flip_a, fft_size)
    full_conv = np.fft.irfft(W * Af, fft_size)[:result_len]
    grad = 2.0 * full_conv[n - 1:2 * n - 1]
    return grad


def compute_topk_gradient(a, conv, top_k, temperature):
    """Weighted gradient focusing on top-k peaks."""
    n = len(a)
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    pv = conv[peaks]
    scaled = pv / temperature
    scaled -= scaled.max()
    exw = np.exp(scaled)
    sw = exw.sum()
    if sw < 1e-300:
        exw = np.zeros(len(peaks))
        exw[np.argmax(pv)] = 1.0
    else:
        exw /= sw

    # Build weight vector
    w_vec = np.zeros(len(conv))
    w_vec[peaks] = exw

    return compute_lse_gradient(a, w_vec)


def lse_gradient_descent(a0, time_limit, t_init=None, t_final=None,
                         lr_init=0.01, momentum=0.9, verbose=False):
    """Log-sum-exp smoothed gradient descent on the simplex.
    Adapts temperature, learning rate, and momentum over time.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    if t_init is None:
        conv0 = fast_convolve(a)
        t_init = max(conv0.max() * 0.3, 1e-6)
    if t_final is None:
        t_final = t_init * 0.02

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    v = np.zeros(n)
    lr = lr_init
    no_improve = 0
    it = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(elapsed / time_limit, 1.0)

        # Anneal temperature
        t = t_init * (t_final / t_init) ** progress
        t = max(t, conv.max() * 0.005)

        # Adaptive learning rate
        lr = lr_init * (1.0 - 0.7 * progress) + 0.001

        # Compute gradient
        w = compute_lse_weights(conv, t)
        grad = compute_lse_gradient(a, w)

        # Momentum update
        v = momentum * v - lr * grad
        a_new = a + v

        # Project onto simplex
        a_new = project_simplex(a_new)

        # Evaluate
        conv = fast_convolve(a_new)
        obj = conv.max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"    LSE iter {it}: obj={obj:.10f}, t={t:.6f}, lr={lr:.6f}")
        else:
            no_improve += 1

        a = a_new

        # Restart if stuck
        if no_improve > 200:
            scale = 0.003 + 0.002 * (no_improve // 200)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            v = np.zeros(n)
            no_improve = 0
            conv = fast_convolve(a)

    return best_a, best_obj


def peak_cycling_descent(a0, time_limit, lr_init=0.005, top_k=5, verbose=False):
    """Peak-cycling: focus gradient on top-k peaks, cycle through them."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    it = 0
    rng = np.random.default_rng(888)

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(elapsed / time_limit, 1.0)
        lr = lr_init * (1.0 - 0.5 * progress) + 0.0005

        # Compute top-k weighted gradient
        temp = conv.max() * 0.1
        grad = compute_topk_gradient(a, conv, top_k, temp)

        # Step
        a_new = a - lr * grad
        a_new = project_simplex(a_new)

        conv = fast_convolve(a_new)
        obj = conv.max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"    PeakCyc iter {it}: obj={obj:.10f}")
        else:
            no_improve += 1

        a = a_new

        if no_improve > 150:
            scale = 0.002 + 0.001 * min(no_improve // 150, 5)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            no_improve = 0
            conv = fast_convolve(a)

    return best_a,
