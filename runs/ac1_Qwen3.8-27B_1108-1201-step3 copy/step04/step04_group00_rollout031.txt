Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth optimization (log-sum-exp surrogate + projected gradient with momentum)
2. LP-based Frank-Wolfe for global corrections
3. Coordinate descent for final polishing
4. Multi-length exploration with adaptive time allocation

import numpy as np
from scipy.optimize import linprog
import time

# ============ Core utilities ============

def fast_convolve(a):
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2*n-1]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n+1)
    cond = (u - (cssv - 1.0) / rho) > 0
    if not cond.any():
        return np.ones(n) / n
    i = np.max(np.where(cond))
    theta = (cssv[i] - 1.0) / (i + 1.0)
    return np.maximum(v - theta, 0.0)

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

# ============ Smooth optimization ============

def smooth_step(f, t):
    """One step of gradient descent on the log-sum-exp objective."""
    n = len(f)
    c = fast_convolve(f)
    tc = t * c
    tc -= tc.max()
    exp_tc = np.exp(tc)
    w = exp_tc / exp_tc.sum()
    
    # Gradient: 2 * sum_k w[k] * f[k-j]
    # = 2 * (convolve(w, f_rev))[n-1 : 2n-1]
    f_rev = f[::-1]
    cw = np.convolve(w, f_rev)
    grad = 2.0 * cw[n-1:2*n-1]
    
    return grad

def smooth_optimize(f0, time_limit, t_init=200.0, t_max=20000.0, verbose=False):

    best_f = f.copy()

    v = np.zeros(n)
    mu = 0.85  # momentum

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        grad = smooth_step(f, t)

        v = mu * v - lr * grad
        f_new = project_simplex(f + v)

        c = fast_convolve(f_new)
        peak = c.max()

            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        t = min(t * 1.003, t_max)

    return best_f, best_peak

# ============ LP-Frank-Wolfe ============

def solve_lp_direction(f, M, n):
    c_vec = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j+n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def lp_fw_optimize(f0, time_limit, verbose=False):

    conv = fast_convolve(f)
    M = conv.max()

    no_improve = 0
    m_exp = 1.0

def coordinate_descent(f0, time_limit):
    f = normalize(f0.copy())
    n = len(f)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue
            
            cur_peak = fast_convolve(f).max()
            best_j_val = old_val
            best_j_obj = cur_peak
            
            for mult in [0.3, 0.5, 0.7, 1.4, 2.0]:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 1.0:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()
                
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved and passes > 2:
            break
    
    return best_f, best_obj

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
1. Mirror descent (exponentiated gradient) on the simplex with log-sum-exp smooth surrogate
2. LP-based Frank-Wolfe for global corrections
3. Coordinate descent for final polishing
4. Multi-length exploration with adaptive time allocation and multi-start
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2 * n - 1]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = (u - (cssv - 1.0) / rho) > 0
    if not cond.any():
        return np.ones(n) / n
    i = np.max(np.where(cond))
    theta = (cssv[i] - 1.0) / (i + 1.0)
    return np.maximum(v - theta, 0.0)


def normalize(a):
    """Clamp to non-negative and normalize to sum=1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def smooth_gradient(f, t):
    """Compute gradient of (1/t)*log(sum_k exp(t*(f*f)[k])) w.r.t. f."""
    n = len(f)
    c = fast_convolve(f)
    tc = t * c
    tc -= tc.max()
    exp_tc = np.exp(tc)
    w = exp_tc / exp_tc.sum()
    # Gradient: 2 * sum_k w[k] * f[k-j]
    f_rev = f[::-1]
    cw = np.convolve(w, f_rev)
    grad = 2.0 * cw[n - 1:2 * n - 1]
    return grad, c.max()


def mirror_descent_optimize(f0, time_limit, t_init=100.0, t_max=50000.0, verbose=False):
    """Mirror descent (exponentiated gradient) on simplex with log-sum-exp surrogate."""
    f = normalize(f0.copy())
    n = len(f)
    best_f = f.copy()
    best_peak = fast_convolve(f).max()

    t = t_init
    lr = 1.0 / t
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        grad, _ = smooth_gradient(f, t)

        # Mirror descent step: f_new = f * exp(-lr * grad) / Z
        scaled = -lr * grad
        scaled -= scaled.max()
        exp_scaled = np.exp(scaled)
        f_new = f * exp_scaled
        f_new_sum = f_new.sum()
        if f_new_sum < 1e-15:
            f_new = np.ones(n) / n
        else:
            f_new /= f_new_sum

        # Evaluate actual peak
        peak = fast_convolve(f_new).max()

        if peak < best_peak - 1e-13:
            best_peak = peak
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new

        # Anneal temperature (make approximation sharper)
        t = min(t * 1.004, t_max)
        # Adjust learning rate inversely with t
        lr = 1.0 / t

        # Adaptive restart
        if no_improve > 200:
            rng = np.random.default_rng(it * 7919 + 3)
            scale = 0.003 + 0.003 * min(no_improve // 300, 3)
            pert = rng.standard_normal(n) * scale
            f = normalize(best_f + pert)
            t = max(t_init, t * 0.4)
            lr = 1.0 / t
            no_improve = 0
            if verbose:
                print(f"    MD restart at it={it}, t={t:.0f}, peak={best_peak:.10f}")

        # Periodic check
        if verbose and it % 5000 == 0:
            print(f"    MD iter {it}: t={t:.0f}, peak={best_peak:.10f}, obj={2*n*best_peak:.6f}")

    return best_f, best_peak


def projected_gradient_optimize(f0, time_limit, t_init=50.0, t_max=30000.0, verbose=False):
    """Projected gradient with momentum on simplex."""
    f = normalize(f0.copy())
    n = len(f)
    best_f = f.copy()
    best_peak = fast_convolve(f).max()

    t = t_init
    lr = 2.0 / (t * np.max(f) * (2 * n - 1) + 1e-10)
    v = np.zeros(n)
    mu = 0.8
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        grad, _ = smooth_gradient(f, t)

        # Heavy-ball momentum
        v = mu * v - lr * grad
        f_new = project_simplex(f + v)

        peak = fast_convolve(f_new).max()

        if peak < best_peak - 1e-13:
            best_peak = peak
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new
        t = min(t * 1.003, t_max)
        lr = 2.0 / (t * np.max(f_new) * (2 * n - 1) + 1e-10)
        mu = min(0.95, 0.8 + 0.0005 * it)

        if no_improve > 150:
            rng = np.random.default_rng(it * 104729 + 11)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            v = np.zeros(n)
            t = max(t_init, t * 0.3)
            lr = 2.0 / (t * np.max(f) * (2 * n - 1) + 1e-10)
            no_improve = 0

    return best_f, best_peak


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c_vec = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with line search."""
    f = normalize(f0.copy())
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        g = solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp = min(m_exp * 1.1, 5.0)
            if m_exp > 4.0:
                no_improve += 5
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 40:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = project_simplex(f + alpha * (g - f))
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            lo = max(0.0, best_alpha - 0.12)
            hi = min(1.0, best_alpha + 0.12)
            for _ in range(10):
                mid = (lo + hi) / 2.0
                ft = project_simplex(f + mid * (g - f))
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

            f_new = project_simplex(f + best_alpha * (g - f))
            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new
            m_exp = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 20 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.08, 4.0)

        if no_improve > 60:
            rng = np.random.default_rng(it * 31 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def coordinate_descent(f0, time_limit):
    """Coordinate descent with local search per coordinate."""
    f = normalize(f0.copy())
    n = len(f)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue

            cur_peak = fast_convolve(f).max()
            if cur_peak < best_obj:
                best_obj = cur_peak
                best_f = f.copy()

            best_j_val = old_val
            best_j_obj = cur_peak

            for mult in [0.2, 0.4, 0.6, 0.8, 1.25, 1.5, 2.0, 3.0]:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved and passes > 2:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
