import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_value(a):
    """Compute objective: 2n * max(a*a) / (sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def build_partial_conv_matrix(a, lags):
    """Build matrix C such that C @ g = (a * g)[k] for k in lags."""
    n = len(a)
    C = np.zeros((len(lags), n))
    for row, k in enumerate(lags):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        # (a*g)[k] = sum_j a[k-j] * g[j] for valid j
        # C[row, j] = a[k-j]
        for j in range(j_start, j_end + 1):
            C[row, j] = a[k - j]
    return C


def lp_direction(a, M, lags=None, C_full=None):
    """
    Solve LP: max sum(g) s.t. (a*g)[k] <= M for k in lags, g >= 0.
    Returns (g, g_sum, success).
    """
    n = len(a)
    if lags is None:

    if C_full is None and lags == list(range(2 * n - 1)):
        C = build_conv_matrix(a)
    else:
        C = build_partial_conv_matrix(a, lags)

    c = -np.ones(n)
    b = np.full(len(lags), M)
    bounds = [(0, None)] * n

                         options={'presolve': True})

            g_sum = g.sum()
            if g_sum > 1e-10:
                return g, g_sum, True
    except Exception:
        pass
    return None, 0.0, False

        conv = fast_convolve(a_try)
        obj = 2.0 * n * conv.max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_try.copy()
            best_t = t

    return best_a, best_obj, best_t


def soft_topk_lags(conv, threshold_frac=0.95, max_lags=50):
    """Find lags where conv[k] >= threshold_frac * max(conv)."""
    M = conv.max()
    if M < 1e-15:
        return [0]
    threshold = M * threshold_frac
    lags = np.where(conv >= threshold)[0]
    if len(lags) > max_lags:
        # Keep the ones closest to the peak
        peak_idx = conv.argmax()
        lags = lags[np.argsort(np.abs(lags - peak_idx))[:max_lags]]
    return lags.tolist()


def gradient_optimize(a0, time_limit, lr_init=0.01, momentum=0.92, top_k=40, temp_init=0.002):
    """Nesterov gradient descent with soft-max gradient."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    y = a.copy()
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    mom = momentum
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        y = a + mom * v
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy

        conv_y = fast_convolve(y)
        peaks = np.argpartition(conv_y, -top_k)[-top_k:]
        peak_vals = conv_y[peaks]
        scaled = peak_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum > 1e-15:
            w /= w_sum
        else:
            w = np.zeros(top_k)
            w[np.argmax(peak_vals)] = 1.0

        grad = np.zeros(n)
        for idx in range(top_k):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idxs = np.arange(j_start, j_end + 1)
            grad[idxs] += w[idx] * y[k - idxs]

        v = mom * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9992
        temp *= 1.004
        mom = min(0.97, mom + 0.0001)

        if no_improve >= 300:
            break

    return best_a, best_obj


def lp_optimize(a0, time_limit, verbose=False):
    """
    Enhanced LP-based optimization with active set, adaptive M, and line search.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    C_full = build_conv_matrix(a)

    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    max_no_improve = 60
    perturb_count = 0
    m_reduction = 1.0  # factor to multiply M by
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit:
        M = conv.max()

        # Determine active lags
        active_lags = soft_topk_lags(conv, threshold_frac=0.97, max_lags=40)

        # Try LP with slightly reduced M to push for improvement
        M_eff = M * m_reduction

        g, g_sum, success = lp_direction(a, M_eff, lags=active_lags, C_full=None)

        if success and g_sum > 1.0 + 1e-6:
            # Normalize g and compute direction
            g_norm = g / g_sum
            d = g_norm - a
            norm_d = np.linalg.norm(d)
            if norm_d > 1e-12:
                a_new, obj_new, t_used = line_search(a, d, n)
                if t_used > 0:
                    a = a_new
                    conv = fast_convolve(a)
                    cur_obj = 2.0 * n * conv.max()
                    if cur_obj < best_obj - 1e-12:
                        best_obj = cur_obj
                        best_a = a.copy()
                    no_improve = 0
                    # Try to reduce M further
                    m_reduction = max(0.995, m_reduction * 0.998)
                    C_full = build_conv_matrix(a)
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            # LP didn't help - relax M and try full constraint set
            if m_reduction < 1.0:
                m_reduction = min(1.0, m_reduction + 0.005)
                no_improve += 1
            else:
                # Try with all lags constrained
                g, g_sum, success = lp_direction(a, M * 1.0001, lags=None)
                if success and g_sum > 1.0 + 1e-6:
                    g_norm = g / g_sum
                    d = g_norm - a
                    if np.linalg.norm(d) > 1e-12:
                        a_new, obj_new, t_used = line_search(a, d, n)
                        if t_used > 0:
                            a = a_new
                            conv = fast_convolve(a)
                            cur_obj = 2.0 * n * conv.max()
                            if cur_obj < best_obj - 1e-12:
                                best_obj = cur_obj
                                best_a = a.copy()
                            no_improve = 0
                            C_full = build_conv_matrix(a)
                        else:
                            no_improve += 1
                    else:
                        no_improve += 1
                else:
                    no_improve += 1

        iteration += 1

        if verbose and iteration % 200 == 0:
            print(f"    LP iter {iteration}, obj={best_obj:.8f}, M_red={m_reduction:.4f}, t={time.monotonic()-t0:.1f}s")

        # Handle stagnation
        if no_improve >= max_no_improve:
            if perturb_count < 50:
                scale = 0.002 + 0.003 * min(perturb_count, 10)
                noise = rng.standard_normal(n) * scale
                a = best_a + noise
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 1e-15:
                    a = a / s
                else:
                    a = best_a.copy()
                conv = fast_convolve(a)
                C_full = build_conv_matrix(a)
            else:
                # Try gradient escape
                a_esc, obj_esc = gradient_optimize(best_a, time_limit=3.0, lr_init=0.005, top_k=60, temp_init=0.001)
                if obj_esc < best_obj:
                    best_obj = obj_esc
                    best_a = a_esc.copy()
                    a = best_a.copy()
                    conv = fast_convolve(a)
                C_full = build_conv_matrix(a)
                if time.monotonic() - t0 > time_limit * 0.9:
                    break

            no_improve = 0
            perturb_count += 1
            max_no_improve = min(200, 60 + 10 * perturb_count)
            m_reduction = 1.0

    return best_a, best_obj


def coordinate_polish(a0, time_limit, rng=None):
    """
    Coordinate descent: try pairwise mass transfers targeting peak lags.
    """
    if rng is None:
        rng = np.random.default_rng(999)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, obj_value(a)
    a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        # Find peak lag
        peak_k = int(conv.argmax())
        # Find contributing pairs
        j_start = max(0, peak_k - n + 1)
        j_end = min(n - 1, peak_k)
        pairs = []
        for j in range(j_start, j_end + 1):
            contrib = a[j] * a[peak_k - j]
            pairs.append((contrib, j, peak_k - j))
        pairs.sort(reverse=True)

        # Try reducing top contributing pairs
        improved = False
        for contrib, j1, j2 in pairs[:20]:
            if contrib < 1e-10:
                break
            # Try moving mass from the larger element to a smaller one
            if a[j1] >= a[j2]:
                src, dst = j1, j2
            else:
                src, dst = j2, j1

            for frac in [0.01, 0.03, 0.05, 0.1]:
                delta = a[src] * frac
                if delta < 1e-12:
                    continue
                a_try = a.copy()
                a_try[src] -= delta
                a_try[dst] += delta
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj - 1e-11:
                    a = a_try
                    conv = conv_try
                    best_obj = obj_try
                    best_a = a.copy()
                    improved = True
                    break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            # Random pair swap
            for _ in range(5):
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue
                for delta in [0.002, -0.002, 0.005, -0.005, 0.01, -0.01]:
                    if a[i] + delta < 0 or a[j] - delta < 0:
                        continue
                    a_try = a.copy()
                    a_try[i] += delta
                    a_try[j] -= delta
                    conv_try = fast_convolve(a_try)
                    obj_try = 2.0 * n * conv_try.max()
                    if obj_try < best_obj - 1e-11:
                        a = a_try
                        conv = conv_try
                        best_obj = obj_try
                        best_a = a.copy()
                        break

        if no_improve > 500:
            break

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # Various structured inits
    inits['sine'] = (1.0 + np.sin(np.pi * x / n)) ** 2
    inits['sine'] /= inits['sine'].sum()

    inits['cos2'] = (1.0 + 0.3 * np.cos(2 * np.pi * 2 * x / n))
    inits['cos2'] = np.maximum(inits['cos2'], 0.01)
    inits['cos2'] /= inits['cos2'].sum()

    # Edge-weighted
    edge = max(1, n // 6)
    bi = np.ones(n) * 0.5
    bi[:edge] = 1.5
    bi[-edge:] = 1.5
    inits['bimodal'] = bi / bi.sum()

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.25 * n)) ** 2)
    inits['gaussian'] = g / g.sum()

    # Alternating with modulation
    alt = np.ones(n)
    alt[::2] = 1.3
    alt[1::2] = 0.7
    alt *= (1 + 0.3 * np.sin(np.pi * x / n))
    inits['alternating'] = alt / alt.sum()

    # Parent interpolation
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0.01)
                inits['parent_interp'] = pp / pp.sum()

    return inits


def extend_sequence(a_short, n_target):
    """Extend a shorter sequence to target length via interpolation."""
    n_short = len(a_short)
    s = a_short.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    a_norm = a_short / s
    a_ext = np.interp(np.linspace(0, 1, n_target), np.linspace(0, 1, n_short), a_norm)
    a_ext = np.maximum(a_ext, 0.01)
    a_ext = a_ext / a_ext.sum()
    return a_ext


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Enhanced LP + gradient + coordinate descent hybrid search.
    Uses active-set LP with adaptive M reduction, multi-length exploration,
    and structured stagnation handling.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            parent_score = float(evaluate_sequence(p.tolist()))
            print(f"Parent loaded: n={len(p)}, score={parent_score:.6f}")
            if np.isfinite(parent_score):
                best_score = parent_score
                best_sequence = p.copy()
    except Exception as e:
        print(f"Parent load failed: {e}")

    # =========================================================
    # Phase 1: Quick gradient scan across lengths
    # =========================================================
    print("=== Phase 1: Gradient Scan ===")
    t_start = time.monotonic()
    phase1_limit = 150.0

    lengths = [512, 768, 1024]
    scan_results = []

    for n in lengths:
        if time.monotonic() > deadline - 600:
            break
        inits = generate_inits(n, parent, rng)

        priority = ['parent', 'parent_interp', 'uniform', 'sine', 'random', 'cos2']
        init_names = [nm for nm in priority if nm in inits]
        init_names += [nm for nm in inits if nm not in priority]
        init_names = init_names[:4]

        t_per = max(8.0, (phase1_limit - 5) / (len(lengths) * 4))

        for name in init_names:
            if time.monotonic() > deadline - 500:
                break
            t_alloc = min(t_per, deadline - time.monotonic() - 400)
            if t_alloc < 4:
                break

            a0 = inits[name]
            a_opt, obj = gradient_optimize(a0, time_limit=t_alloc,
                                           lr_init=0.01, momentum=0.91,
                                           top_k=50, temp_init=0.002)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Grad n={n}, {name}: {score:.6f}")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 2: LP-based deep optimization
    # =========================================================
    print("=== Phase 2: LP Deep Optimization ===")
    t_start2 = time.monotonic()

    # Select top candidates
    top_configs = []
    seen_lengths = set()
    for score, n, name, a_opt in scan_results:
        if n not in seen_lengths:
            top_configs.append((n, a_opt, score))
            seen_lengths.add(n)
        if len(top_configs) >= 3:
            break

    # Always include best overall
    if best_sequence is not None:
        if len(best_sequence) not in seen_lengths:
            top_configs.append((len(best_sequence), best_sequence.copy(), best_score))
            seen_lengths.add(len(best_sequence))

    # Include parent at its native length
    if parent is not None:
        pn = len(parent)
        if pn not in seen_lengths:
            top_configs.append((pn, parent.copy(), best_score))
            seen_lengths.add(pn)

    top_configs.sort(key=lambda x: x[2])

    remaining = deadline - time.monotonic() - 200.0
    n_cands = max(1, len(top_configs))
    t_per_lp = max(20.0, remaining * 0.65 / n_cands)

    for i, (n, a0, sc) in enumerate(top_configs):
        if time.monotonic() > deadline - 180:
            break
        t_alloc = min(t_per_lp, deadline - time.monotonic() - 150)
        if t_alloc < 10:
            break

        print(f"  LP cand {i}: n={n}, t_alloc={t_alloc:.0f}s")
        a_opt, obj = lp_optimize(a0, time_limit=t_alloc, verbose=True)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  LP {i} (n={n}): {score:.6f} *NEW BEST*")
            else:
                print(f"  LP {i} (n={n}): {score:.6f}")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 3: Extended LP polish on best with multiple restarts
    # =========================================================
    print("=== Phase 3: Extended LP Polish ===")
    t_start3 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 100.0
        if remaining > 40:
            n_restarts = 4
            t_per_restart = remaining / n_restarts * 0.85

            for ri in range(n_restarts):

                    break
                t_alloc = min(t_per_restart, deadline - time.monotonic() - 50)
                if t_alloc < 8:
                    break

                if ri == 0:
                    a0 = best_sequence.copy()
                elif ri == 1:
                    # Small perturbation
                    noise = rng.standard_normal(len(best_sequence)) * 0.001
                    a0 = np.maximum(best_sequence + noise, 0)
                    a0 = a0 / a0.sum()
                elif ri == 2:
                    # Structural perturbation: smooth
                    from scipy.ndimage import uniform_filter1d
                    a0 = uniform_filter1d(best_sequence, size=5)
                    a0 = np.maximum(a0, 0.01)
                    a0 = a0 / a0.sum()
                else:
                    # Larger perturbation
                    noise = rng.standard_normal(len(best_sequence)) * 0.005
                    a0 = np.maximum(best_sequence + noise, 0)
                    a0 = a0 / a0.sum()

                a_opt, obj = lp_optimize(a0, time_limit=t_alloc)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score):
                    if score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        print(f"  Polish {ri}: {score:.6f} *NEW BEST*")
                    else:
                        print(f"  Polish {ri}: {score:.6f}")

    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 4: Coordinate descent + gradient micro-tuning
    # =========================================================
    print("=== Phase 4: Fine-tuning ===")
    t_start4 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 20.0

        # Coordinate polish
        t_cd = min(remaining * 0.4, 45.0)
        if t_cd > 3:
            a_opt, obj = coordinate_polish(best_sequence, time_limit=t_cd, rng=rng)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  CoordDesc: {score:.6f} *NEW BEST*")
            else:
                print(f"  CoordDesc: {score:.6f}")

        # Short gradient runs with different configs
        remaining = deadline - time.monotonic() - 15.0
        if remaining > 5:
            configs = [
                {'lr_init': 0.003, 'momentum': 0.95, 'top_k': 80, 'temp_init': 0.001},
                {'lr_init': 0.008, 'momentum': 0.90, 'top_k': 30, 'temp_init': 0.003},
            ]
            for gi, cfg in enumerate(configs):
                if time.monotonic() > deadline - 10:
                    break
                t_sg = min(remaining / (len(configs) - gi) * 0.7, 20.0)
                if t_sg < 3:
                    break
                a_opt, obj = gradient_optimize(best_sequence, time_limit=t_sg, **cfg)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  GradFine {gi}: {score:.6f} *NEW BEST*")

    # =========================================================
    # Phase 5: Final LP micro-polish
    # =========================================================
    if best_sequence is not None and time.monotonic() < deadline - 5:

        if t_final > 3:

        else:

        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence([float(x) for x in best_sequence]))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
