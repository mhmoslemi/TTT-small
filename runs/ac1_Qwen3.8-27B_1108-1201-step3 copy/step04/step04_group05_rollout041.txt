```python
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using iterative water-filling,
smooth log-sum-exp projected gradient, and L-BFGS in log-space.
Different approach from LP-based methods: uses multiplicative updates

import numpy as np
from scipy.optimize import minimize

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def cross_corr_via_fft(w, a):
    """Compute cross-correlation R[j] = sum_k w[k]*a[k-j] for j=0..len(a)-1."""
    n = len(a)
    L = 1 << (3 * n - 1).bit_length()
    w_fft = np.fft.rfft(w, L)
    a_fft = np.fft.rfft(a, L)
    return np.fft.irfft(w_fft * np.conj(a_fft), L)[:n]

def water_filling(a, num_iters=100, peak_frac=0.3, eta=0.5, lse_tau=100):
    """Iterative water-filling in log domain."""
    n = len(a)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return a / n
    a = a / s
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    for it in range(num_iters):
        c = fast_convolve(a)
        cur_obj = c.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()
        
        # Select peak lags
        threshold = c.max() * (1.0 - peak_frac)
        K = np.where(c >= threshold)[0]
        
        # Compute load
        indicator = np.zeros(len(c))
        indicator[K] = 1.0
        load = cross_corr_via_fft(indicator, a)
        
        # Multiplicative update
        max_load = load.max()
        if max_load < 1e-15:
            break
        a = a * np.exp(-eta * load / max_load)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s
    
    return best_a
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Iterative multiplicative water-filling (log-domain updates targeting convolution peaks)
2. Smooth log-sum-exp projected gradient with tau annealing
3. L-BFGS optimization in log-space (softmax parameterization)
4. Multi-length exploration and local search

Key difference from LP-based approaches: uses spectral-domain (FFT) gradient
computation and multiplicative updates for fast, targeted peak reduction.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Projection onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def cross_corr_via_fft(w, a):
    """Compute R[j] = sum_k w[k]*a[k-j] for j=0..len(a)-1 via FFT."""
    n = len(a)
    L = 1 << (3 * n - 1).bit_length()
    w_fft = np.fft.rfft(w, L)
    a_fft = np.fft.rfft(a, L)
    return np.fft.irfft(w_fft * np.conj(a_fft), L)[:n]


def water_filling(a0, num_iters=200, peak_frac=0.3, eta=0.5, momentum=0.0):
    """Iterative multiplicative water-filling in log domain."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    v = np.zeros(n)

    for it in range(num_iters):
        c = fast_convolve(a)
        cur_obj = c.max()
        if cur_obj < best_obj:
            best_obj = cur_obj
            best_a = a.copy()

        cmax = c.max()
        if cmax < 1e-18:
            break

        threshold = cmax * (1.0 - peak_frac)
        K = np.where(c >= threshold)[0]
        if len(K) == 0:
            K = np.array([np.argmax(c)])

        indicator = np.zeros(len(c))
        indicator[K] = 1.0
        load = cross_corr_via_fft(indicator, a)

        max_load = load.max()
        if max_load < 1e-15:
            break

        grad = load / max_load
        v = momentum * v - eta * grad
        a_new = a * np.exp(v)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new

    return best_a, best_obj


def smooth_proj_grad(a0, time_limit, tau_init=50.0, tau_max=5000.0,
                     lr_init=0.005, momentum=0.9, anneal_rate=1.05):
    """Smooth log-sum-exp projected gradient with tau annealing."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    v = np.zeros(n)
    lr = lr_init
    tau = tau_init
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        c = fast_convolve(a)
        m = c.max()
        scaled = tau * (c - m)
        max_s = scaled.max()
        exp_s = np.exp(scaled - max_s)
        sum_exp = exp_s.sum()
        if sum_exp < 1e-300:
            break
        w = exp_s / sum_exp

        grad_a = 2.0 * cross_corr_via_fft(w, a)

        v = momentum * v - lr * grad_a
        a_new = a + v
        a_new = project_simplex(a_new)

        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
        a = a_new

        tau = min(tau * anneal_rate, tau_max)
        lr *= 0.9995
        if lr < 1e-7:
            lr = 1e-7

    return best_a, best_obj


def _lbfgs_fun(b, tau):
    """Objective and gradient for L-BFGS in log-space."""
    b_shifted = b - b.max()
    a_s = np.exp(b_shifted)
    a_sum = a_s.sum()
    if a_sum < 1e-300:
        return 1e10, np.zeros(len(b))
    a_s = a_s / a_sum

    c = fast_convolve(a_s)
    m = c.max()
    scaled = tau * (c - m)
    max_s = scaled.max()
    exp_s = np.exp(scaled - max_s)
    sum_exp = exp_s.sum()
    if sum_exp < 1e-300:
        return 1e10, np.zeros(len(b))
    log_sum = max_s + np.log(sum_exp)
    obj = m + log_sum / tau

    w = exp_s / sum_exp
    grad_a = 2.0 * cross_corr_via_fft(w, a_s)
    dot_ga = np.dot(grad_a, a_s)
    grad_b = a_s * (grad_a - dot_ga)
    return obj, grad_b


def _lbfgs_run(b0, tau, max_iter, maxfun):
    """Run L-BFGS-B with given initial point and parameters."""
    res = minimize(_lbfgs_fun, b0, args=(tau,), method='L-BFGS-B',
                   jac=True, options={'maxiter': max_iter, 'maxfun': maxfun,
                                      'gtol': 1e-12, 'ftol': 1e-15})
    if res.x is not None:
        b_opt = res.x
        a_opt = np.exp(b_opt - b_opt.max())
        a_sum = a_opt.sum()
        if a_sum > 1e-300:
            a_opt = a_opt / a_sum
            return a_opt
    return None


def lbfgs_multi_tau(a0, time_limit):
    """Run L-BFGS at increasing tau values, each starting from best so far."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    tau_schedule = [100, 300, 700, 1500, 3000, 7000, 15000, 30000]

    for tau in tau_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2.0:
            break

        b0 = np.log(np.maximum(a, 1e-15))
        # Estimate iterations based on time
        # Each iteration ~ O(n log n) for gradient, plus L-BFGS overhead
        est_time_per_iter = 0.015 * (n / 384.0)
        max_iter = max(50, min(400, int(remaining * 0.8 / est_time_per_iter)))
        maxfun = max_iter * 4

        t_start = time.monotonic()
        a_res = _lbfgs_run(b0, tau, max_iter, maxfun)
        t_elapsed = time.monotonic() - t_start

        if a_res is not None:
            obj_res = fast_convolve(a_res).max()
            if obj_res < best_obj - 1e-14:
                best_obj = obj_res
                best_a = a_res.copy()
                print(f"    L-BFGS tau={tau}: obj={obj_res:.10f} (took {t_elapsed:.1f}s)")

    return best_a, best_obj


def local_perturb_search(a0, time_limit, rng):
    """Random perturbation + quick optimization."""
    n = len(a0)
    best_a = a0.astype(np.float64).copy()
    best_obj = fast_convolve(best_a).max()

    t0 = time.monotonic()
    scale = 0.003
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Try several perturbations
        for _ in range(10):
            if time.monotonic() - t0 > time_limit:
                break
            noise = rng.standard_normal(n) * scale
            a_pert = project_simplex(best_a + noise)
            obj = fast_convolve(a_pert).max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_pert.copy()
                # Quick L-BFGS polish
                b0 = np.log(np.maximum(a_pert, 1e-15))
                a_res = _lbfgs_run(b0, 2000, 50, 200)
                if a_res is not None:
                    obj2 = fast_convolve(a_res).max()
                    if obj2 < best_obj:
                        best_obj = obj2
                        best_a = a_res.copy()

        scale *= 0.92
        if scale < 1e-5:
            scale = 0.005

    return best_a, best_obj


def gen_gaussian(n, center=0.5, width=0.2):
    """Generate a Gaussian-shaped sequence."""
    x = np.arange(n, dtype=np.float64) / n
    g = np.exp(-0.5 * ((x - center) / width) ** 2)
    return g / g.sum()


def gen_cosine(n, freq=1.0):
    """Generate a cosine-shaped sequence."""
    x = np.arange(n, dtype=np.float64) / n
    c = np.abs(np.cos(np.pi * freq * x)) + 0.05
    return c / c.sum()


def gen_beta(n, alpha=2.0, beta=2.0):
    """Generate a Beta-like shaped sequence."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    b = x ** (alpha - 1) * (1 - x) ** (beta - 1) + 1e-10
    return b / b.sum()


def gen_u_shape(n, sharpness=2.0):
    """Generate a U-shaped sequence."""
    x = np.arange(n, dtype=np.float64) / n
    u = (x - 0.5) ** 2 * sharpness + 0.1
    u = 1.0 / u
    return u / u.sum()


def gen_random_dirichlet(n, concentration, rng):
    """Generate a random Dirichlet sequence."""
    return rng.dirichlet(np.ones(n) * concentration)

        if ps > 1e-10:
            cands.append(px / ps)

    return cands


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase optimization:
    Phase 1: Multi-length initialization with quick water-filling
    Phase 2: Iterative water-filling with parameter sweep
    Phase 3: Smooth projected gradient with tau annealing
    Phase 4: L-BFGS in log-space with increasing tau (key phase)
    Phase 5: Local search and final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')
    best_seq_norm = None  # normalized to sum=1

    def try_seq(a, label=""):
        nonlocal best_seq, best_score, best_seq_norm
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        if a.sum() < 1e-10:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            best_seq_norm = a / a.sum()
            print(f"  NEW BEST [{label}]: score={score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
    except Exception:
        pass

    # ============ Phase 1: Multi-length initialization ============
    print("=== Phase 1: Multi-length Initialization ===")
    t_start = time.monotonic()

    lengths_to_try = [256, 320, 384, 448, 512]
    if parent is not None:
        lengths_to_try.insert(0, len(parent))
        # Remove duplicates
        lengths_to_try = sorted(set(lengths_to_try))

    phase1_budget = 80.0
    per_length_time = phase1_budget / len(lengths_to_try)

    for n in lengths_to_try:
        if time.monotonic() > deadline - 400:
            break
        t_n0 = time.monotonic()
        cands = generate_candidates(n, parent, rng)

        local_best = None
        local_best_obj = float('inf')

        for i, cand in enumerate(cands):
            if time.monotonic() - t_n0 > per_length_time:
                break
            # Quick water-filling (10 iterations)
            wf_a, wf_obj = water_filling(cand, num_iters=15, peak_frac=0.3, eta=0.5)
            if wf_obj < local_best_obj:
                local_best_obj = wf_obj
                local_best = wf_a

        if local_best is not None:
            try_seq(local_best, f"init_n{n}")

    if best_seq is None:
        best_seq = np.ones(384) / 384
        try_seq(best_seq, "fallback_uniform")

    n_best = len(best_seq)
    print(f"  Phase 1 best: score={best_score:.8f} (n={n_best})")
    print(f"  Phase 1 time: {time.monotonic() - t_start:.1f}s")

    # ============ Phase 2: Water-filling optimization ============
    print("\n=== Phase 2: Water-filling Optimization ===")
    t2_start = time.monotonic()
    phase2_budget = 350.0

    a_work = best_seq_norm.copy()
    params_2d = [
        (0.1, 0.5, 0.0), (0.1, 0.7, 0.0), (0.2, 0.5, 0.0),
        (0.2, 0.7, 0.2), (0.3, 0.5, 0.0), (0.3, 0.8, 0.3),
        (0.4, 0.5, 0.0), (0.4, 1.0, 0.3), (0.5, 0.7, 0.0),
        (0.3, 0.5, 0.5), (0.2, 1.0, 0.5), (0.5, 0.5, 0.3),
    ]

    per_param_time = phase2_budget / len(params_2d)

    for pf, eta, mom in params_2d:
        if time.monotonic() > deadline - 400:
            break
        t_p0 = time.monotonic()
        wf_a, wf_obj = water_filling(a_work, num_iters=500,
                                     peak_frac=pf, eta=eta, momentum=mom)
        if wf_obj < fast_convolve(a_work).max():
            a_work = wf_a
            try_seq(wf_a, f"wf_{pf}_{eta}_{mom}")

        # Also try from best_seq_norm directly
        wf_a2, _ = water_filling(best_seq_norm, num_iters=200,
                                 peak_frac=pf, eta=eta, momentum=mom)
        try_seq(wf_a2, f"wf2_{pf}_{eta}")
        if fast_convolve(wf_a2).max() < fast_convolve(a_work).max():
            a_work = wf_a2

    # Use a_work for subsequent phases
    try_seq(a_work, "wf_final")
    print(f"  Phase 2 best: score={best_score:.8f}")
    print(f"  Phase 2 time: {time.monotonic() - t2_start:.1f}s")

    # ============ Phase 3: Smooth projected gradient ============
    print("\n=== Phase 3: Smooth Projected Gradient ===")
    t3_start = time.monotonic()
    phase3_budget = 250.0

    sg_configs = [
        (50.0, 5000.0, 0.005, 0.90, 1.05),
        (100.0, 3000.0, 0.003, 0.93, 1.03),
        (50.0, 10000.0, 0.008, 0.85, 1.08),
        (200.0, 5000.0, 0.002, 0.95, 1.02),
        (50.0, 5000.0, 0.005, 0.90, 1.05),  # repeat with different start
    ]

    per_sg_time = phase3_budget / len(sg_configs)
    start_src = [a_work, best_seq_norm, a_work, best_seq_norm, a_work]

    for i, (t0, t1, lr, mom, ann) in enumerate(sg_configs):
        if time.monotonic() > deadline - 300:
            break
        src = start_src[i % len(start_src)]
        sg_a, sg_obj = smooth_proj_grad(src, time_limit=per_sg_time,
                                        tau_init=t0, tau_max=t1,
                                        lr_init=lr, momentum=mom,
                                        anneal_rate=ann)
        try_seq(sg_a, f"sg_{i}")
        if sg_obj < fast_convolve(a_work).max():
            a_work = sg_a

    try_seq(a_work, "sg_final")
    print(f"  Phase 3 best: score={best_score:.8f}")
    print(f"  Phase 3 time: {time.monotonic() - t3_start:.1f}s")

    # ============ Phase 4: L-BFGS in log-space (KEY PHASE) ============
    print("\n=== Phase 4: L-BFGS Log-Space Optimization ===")
    t4_start = time.monotonic()
    phase4_budget = 500.0

    # Run L-BFGS multiple times from different starting points
    start_points = [a_work, best_seq_norm]
    # Add a perturbed version
    pert = project_simplex(a_work + rng.standard_normal(n_best) * 0.005)
    start_points.append(pert)

    per_start_time = phase4_budget / len(start_points)

    for i, sp in enumerate(start_points):
        if time.monotonic() > deadline - 200:
            break
        lb_a, lb_obj = lbfgs_multi_tau(sp, time_limit=per_start_time)
        try_seq(lb_a, f"lbfgs_{i}")
        if lb_obj < fast_convolve(a_work).max():
            a_work = lb_a

    # Additional L-BFGS runs at high tau for fine tuning
    if time.monotonic() < deadline - 100:
        remaining_lbg = min(150.0, max(0.0, deadline - time.monotonic() - 80))
        if remaining_lbg > 10:
            b0 = np.log(np.maximum(a_work, 1e-15))
            for hi_tau in [5000, 10000, 20000, 50000]:
                if time.monotonic() > deadline - 80:
                    break
                t_sub = min(30.0, max(5.0, (deadline - time.monotonic() - 80) / 3.0))
                est_iter = max(20, min(200, int(t_sub / 0.02)))
                a_res = _lbfgs_run(b0, hi_tau, est_iter, est_iter * 4)
                if a_res is not None:
                    obj_res = fast_convolve(a_res).max()
                    if obj_res < fast_convolve(a_work).max() - 1e-14:
                        a_work = a_res
                        try_seq(a_work, f"lbfgs_hi_tau_{hi_tau}")
                        b0 = np.log(np.maximum(a_work, 1e-15))

    print(f"  Phase 4 best: score={best_score:.8f}")
    print(f"  Phase 4 time: {time.monotonic() - t4_start:.1f}s")

    # ============ Phase 5: Local search and final polish ============
    print("\n=== Phase 5: Local Search & Final Polish ===")
    t5_start = time.monotonic()

    remaining_5 = max(0.0, deadline - time.monotonic() - 30)
    if remaining_5 > 10:
        # Local perturbation search
        ls_time = min(remaining_5 * 0.4, 80.0)
        ls_a, ls_obj = local_perturb_search(a_work, time_limit=ls_time, rng=rng)
        try_seq(ls_a, "local_search")
        if ls_obj < fast_convolve(a_work).max():
            a_work = ls_a

        # Final L-BFGS polish
        rem = max(0.0, deadline - time.monotonic() - 20)
        if rem > 5:
            b0 = np.log(np.maximum(a_work, 1e-15))
            for final_tau in [10000, 30000]:
                if time.monotonic() > deadline - 15:
                    break
                a_res = _lbfgs_run(b0, final_tau, 100, 400)
                if a_res is not None:
                    obj_res = fast_convolve(a_res).max()
                    if obj_res < fast_convolve(a_work).max() - 1e-14:
                        a_work = a_res
                        try_seq(a_work, f"final_lbfgs_{final_tau}")
                        b0 = np.log(np.maximum(a_work, 1e-15))

        # Try a slightly different length
        rem = max(0.0, deadline - time.monotonic() - 15)
        if rem > 10:
            n_try = n_best + 16
            px = np.interp(np.linspace(0, 1, n_try),
                           np.linspace(0, 1, n_best), a_work)
            px = np.maximum(px, 0)
            ps = px.sum()
            if ps > 1e-10:
                px = px / ps
                wf_px, _ = water_filling(px, num_iters=50, peak_frac=0.2, eta=0.5)
                try_seq(wf_px, f"len_{n_try}_wf")
                if time.monotonic() < deadline - 15:
                    b0p = np.log(np.maximum(wf_px, 1e-15))
                    a_res = _lbfgs_run(b0p, 5000, 50, 200)
                    if a_res is not None:
                        try_seq(a_res, f"len_{n_try}_lbg")

    print(f"  Phase 5 best: score={best_score:.8f}")
    print(f"  Phase 5 time: {time.monotonic() - t5_start:.1f}s")

    # ============ Final verification ============
    if best_seq is not None:
        final_score = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.10f}")

    return [float(x) for x in best_seq]
```
