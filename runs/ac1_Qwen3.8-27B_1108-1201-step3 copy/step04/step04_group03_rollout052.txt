- If i = k-i (k=2i): these are the same term, f_i^2 → derivative is 2*f_i

So db_k/df_i = f_{k-i} + f_{k-i} = 2*f_{k-i} if 0 <= k-i < n and k-i != i
              = 2*f_i if k-i = i (i.e., k = 2i)
              = 0 otherwise

In all cases where 0 <= k-i < n: db_k/df_i = 2*f_{k-i}
(When k-i = i, this gives 2*f_i, which is correct.)

So: db_k/df_i = 2*f_{k-i} if 0 <= k-i < n, else 0.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth L-BFGS on log-sum-exp approximation
2. Exact 1D coordinate descent (ternary search on convex subproblem)
3. Pair coordinate descent for local minimum escape
4. Multi-length exploration (384, 512, 768)
5. Peak-targeted subgradient with sensitivity weighting
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_ls_exp(f, t):
    """Compute log-sum-exp of conv(f,f)/t and its gradient."""
    n = len(f)
    b = fast_convolve(f)
    scaled = b / t
    max_s = scaled.max()
    exp_s = np.exp(scaled - max_s)
    log_sum = t * (max_s + np.log(exp_s.sum()))
    p = exp_s / exp_s.sum()
    # grad[i] = 2 * sum_j f[j] * p[j+i]
    grad = np.zeros(n)
    for i in range(n):
        grad[i] = 2.0 * np.dot(p[i:i + n], f)
    return log_sum, grad, b, p


def lbfgs_smooth_phase(f0, time_limit, t_init=0.05):
    """L-BFGS on log-sum-exp approximation with decreasing temperature."""
    f = f0.copy()
    n = len(f)
    f = np.maximum(f, 0)
    f = f / f.sum()
    
    t0 = time.monotonic()
    t = t_init
    best_f = f.copy()
    best_b = fast_convolve(f).max()
    
    while time.monotonic() - t0 < time_limit:
        phase_limit = min(40, time_limit - (time.monotonic() - t0))
        
        def obj_func(x):
            val, grad, _, _ = compute_ls_exp(x, t)
            return val, grad
        
        res = minimize(obj_func, f, method='L-BFGS-B',
                       bounds=[(0, None)] * n,
                       options={'maxiter': 300, 'maxfun': 3000, 'gtol': 1e-8})
        
        if res.x is not None and res.fun < obj_func(f)[0]:
            f = np.maximum(res.x, 0)
            s = f.sum()
            if s > 1e-15:
                f = f / s
        
        cur_b = fast_convolve(f).max()
        if cur_b < best_b:
            best_b = cur_b
            best_f = f.copy()
        
        t *= 0.6
        if t < 0.001:
            break
    
    return best_f


def coord_descent(f0, time_limit, max_passes=500):
    """Exact 1D coordinate descent with ternary search."""
    f = f0.copy()
    n = len(f)
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    
    for pass_num in range(max_passes):
        if time.monotonic() - t0 > time_limit:
            break
        
        for i in range(n):
            old = f[i]
            # Compute base conv (f[i]=0)
            f_base = f.copy()
            f_base[i] = 0.0
            base = fast_convolve(f_base)
            
            # Linear coefficients: line[k] = 2*f[k-i] if valid and k-i != i
            line = np.zeros(2 * n - 1)
            quad_at = 2 * i  # position of quadratic term
            
            for k in range(2 * n - 1):
                ki = k - i
                if 0 <= ki < n and ki != i:
                    line[k] = 2.0 * f[ki]
            
            # Ternary search for x in [0, hi]
            # g(x) = max_k (base[k] + line[k]*x + (x^2 if k==quad_at else 0))
            lo = 0.0
            hi = max(old * 2.0, 0.01)
            
            # First check boundaries
            def eval_g(x):
                vals = base + line * x
                if quad_at < 2 * n - 1:
                    vals[quad_at] += x * x
                return np.max(vals)
            
            g_lo = eval_g(lo)
            g_hi = eval_g(hi)
            
            if g_lo <= g_hi:
                hi = old  # minimum is in [0, old]
                hi = max(hi, 1e-10)
            else:
                lo = old
                hi = max(old * 2, old + 0.01)
            
            # Ternary search
            for _ in range(40):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                g1 = eval_g(m1)
                g2 = eval_g(m2)
                if g1 < g2:
                    hi = m2
                else:
                    lo = m1
            
            f[i] = (lo + hi) / 2.0
        
        # Renormalize
        f = np.maximum(f, 0)
        s = f.sum()
        if s < 1e-15:
            f = np.ones(n) / n
        else:
            f = f / s
        
        conv = fast_convolve(f)
        cur_obj = conv.max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 30:
            break
    
    return best_f


def pair_coord_descent(f0, time_limit, max_passes=100):
    """Pair coordinate descent: optimize 2 coordinates at a time."""
    f = f0.copy()
    n = len(f)
    f = np.maximum(f, 0)
    f = f / f.sum()
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    
    for pass_num in range(max_passes):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Random pair ordering
        pairs = [(i, (i + 1) % n) for i in range(0, n - 1, 2)]
        if n % 2 == 1:
            pairs.append((n - 1, 0))
        rng.shuffle(pairs)
        
        for (i, j) in pairs:
            old_i, old_j = f[i], f[j]
            
            # Remove contributions of f[i] and f[j]
            f_base = f.copy()
            f_base[i] = 0.0
            f_base[j] = 0.0
            base = fast_convolve(f_base)
            
            # Contribution of f[i]=x and f[j]=y to b_k:
            # From f[i]: 2*x*f[k-i] (if valid, k-i!=j, k-i!=i) or x^2 (if k=2i)
            #                          or 2*x*f[k-i] handles the cross with other elements
            # From f[j]: similar
            # Cross term: 2*x*y if k-i == j or k-j == i
            
            # Simplified: just evaluate on a small grid
            xi = old_i
            xj = old_j
            
            def eval_g2(x, y):
                vals = base.copy()
                # Add contribution of f[i]=x
                for k in range(2 * n - 1):
                    ki = k - i
                    if 0 <= ki < n:
                        if ki == i:
                            vals[k] += x * x
                        elif ki == j:
                            vals[k] += 0  # cross handled below
                        else:
                            vals[k] += 2 * x * (f[ki] if ki != j else 0 * y)
                # Actually this is getting complicated. Let me just do direct eval.
                return np.max(fast_convolve(np.where(np.arange(n) == i, x,
                                        np.where(np.arange(n) == j, y, f))))
            
            # Small grid around (old_i, old_j)
            delta = max(old_i, old_j) * 0.3 + 0.001
            best_x, best_y, best_val = old_i, old_j, eval_g2(old_i, old_j)
            
            for dx in [-1, -0.5, 0.5, 1]:
                for dy in [-1, -0.5, 0.5, 1]:
                    x = max(0, old_i + dx * delta)
                    y = max(0, old_j + dy * delta)
                    v = eval_g2(x, y)
                    if v < best_val:
                        best_val = v
                        best_x, best_y = x, y
            
            # Refine
            if (best_x, best_y) != (old_i, old_j):
                delta2 = delta * 0.3
                for dx in [-1, 0, 1]:
                    for dy in [-1, 0, 1]:
                        x = max(0, best_x + dx * delta2)
                        y = max(0, best_y + dy * delta2)
                        v = eval_g2(x, y)
                        if v < best_val:
                            best_val = v
                            best_x, best_y = x, y
            
            f[i] = best_x
            f[j] = best_y
        
        f = np.maximum(f, 0)
        f = f / f.sum()
        conv = fast_convolve(f)
        if conv.max() < best_obj - 1e-14:
            best_obj = conv.max()
            best_f = f.copy()
    
    return best_f


def peak_subgradient(f0, time_limit, lr_init=0.002, decay=0.9997):
    """Peak-targeted subgradient with sensitivity weighting."""
    f = f0.copy()
    n = len(f)
    f = np.maximum(f, 0)
    f = f / f.sum()
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    v = np.zeros(n)
    lr = lr_init
    momentum = 0.9
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()
        
        # Find peaks (top fraction)
        n_peaks = max(5, n // 10)
        peaks = np.argpartition(conv, -n_peaks)[-n_peaks:]
        
        # Weight by excess over mean
        mean_b = conv.mean()
        excess = np.maximum(conv[peaks] - mean_b, 0)
        w = excess / (excess.sum() + 1e-15)
        
        # Compute subgradient
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            k = int(peaks[idx])
            # db_k/df_i = 2*f[k-i] for valid k-i
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_range = np.arange(js, je + 1)
            sg[idx_range] += w[idx] * 2.0 * f[k - idx_range]
        
        v = momentum * v - lr * sg
        f_new = f + v
        f_new = project_simplex(f_new)
        
        conv_new = fast_convolve(f_new)
        if conv_new.max() < best_obj - 1e-15:
            best_obj = conv_new.max()
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
        
        lr *= decay
        
        if no_improve > 200:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            lr = lr_init * 0.5
            no_improve = 0
    
    return best_f


def smooth_polish(f0, time_limit):
    """Final polish using very low temperature L-BFGS + coordinate descent."""
    f = f0.copy()
    n = len(f)
    t0 = time.monotonic()
    
    # Very fine smooth optimization
    rem = time_limit - (time.monotonic() - t0)
    if rem > 20:
        f = lbfgs_smooth_phase(f, time_limit=min(rem, 30), t_init=0.01)
    
    # Fine coordinate descent
    rem = time_limit - (time.monotonic() - t0)
    if rem > 30:
        f = coord_descent(f, time_limit=rem)
    
    return f


def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    # Uniform
    inits.append(np.ones(n) / n)
    # Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    # Edge-weighted
    x = np.arange(n, dtype=np.float64)
    edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge / edge.sum())
    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits.append(g / g.sum())
    # Cosine
    c = np.abs(np.cos(np.pi * x / n)) + 0.1
    inits.append(c / c.sum())
    # Interpolated parent
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    return inits
