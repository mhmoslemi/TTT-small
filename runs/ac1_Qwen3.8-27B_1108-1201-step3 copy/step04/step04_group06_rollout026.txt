    """Gerchberg-Saxton / HIO for flat spectrum."""
    n = len(a0)
    L = 1 << (2 * n).bit_length()
    
    a = a0.copy()
    for it in range(n_iters):
        A = np.fft.rfft(a, L)
        mag = np.abs(A)
        target_mag = np.sqrt(L)  # flat target
        A_new = target_mag * A / (mag + 1e-300)
        a_new = np.fft.irfft(A_new, L)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-10:
            a_new = a_new / s
        else:
            a_new = a / a.sum()
        a = a_new
    
    return a

def gen_chirp_init(n, rng):
    """Generate initialization from a chirp-like pattern."""
    x = np.arange(n, dtype=np.float64)
    # Chirp: cos(2*pi*f*t^2) made non-negative
    phase = 2 * np.pi * np.outer(x, x) / n
    chirp = np.cos(phase)
    chirp = np.maximum(chirp, 0)
    s = chirp.sum()
    if s > 1e-10:
        chirp = chirp / s
    else:
        chirp = np.ones(n) / n
    return chirp

def gen_random_phase_init(n, rng):
    """Generate from random phase with flat magnitude spectrum."""
    L = 1 << (2 * n).bit_length()
    phase = rng.uniform(0, 2 * np.pi, L // 2 + 1)
    # Make symmetric for real signal
    A = np.sqrt(L) * np.exp(1j * phase)
    # Enforce Hermitian symmetry
    A_sym = A.copy()
    # For real signal: A[k] = conj(A[L-k])
    # rfft gives [0, 1, ..., L/2], so we just need A[0] and A[L/2] to be real
    A_sym[0] = np.abs(A_sym[0])
    if L % 2 == 0:
        A_sym[-1] = np.abs(A_sym[-1])
    a = np.fft.irfft(A_sym, L)[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-10:
        a = a / s
    else:
        a = np.ones(n) / n
    return a

def generate_inits(best_a, rng, n):
    """Generate diverse initializations."""
    inits = []
    
    # Parent (if same length)
    if best_a is not None and len(best_a) == n:
        inits.append(best_a.copy())
        # Perturbed parent
        for scale in [0.01, 0.05, 0.1]:
            p = best_a + rng.standard_normal(n) * scale / best_a.sum()
            p = np.maximum(p, 0)
            ps = p.sum()
            if ps > 1e-10:
                inits.append(p / ps)
    
    # Chirp
    inits.append(gen_chirp_init(n, rng))
    
    # Random phase
    for _ in range(3):
        inits.append(gen_random_phase_init(n, rng))
    
    # Edge-concentrated
    x = np.arange(n, dtype=np.float64)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())
    
    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())
    
    # Cosine
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.01
    inits.append(c1 / c1.sum())
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Random Dirichlet
    for _ in range(2):
        inits.append(rng.dirichlet(np.ones(n) * 0.1))
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Exponentiated gradient (multiplicative weights) with adaptive softmax subgradients
2. Variance-based pre-optimization for smoother landscape navigation
3. Gerchberg-Saxton phase retrieval for diverse initializations
4. Incremental-update local search for final polishing
5. Multi-length exploration with interpolation
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def eg_optimize(a0, time_limit, eta0=2.0, temp0=0.005, temp_decay=0.9995,
                eta_decay=0.9999, top_k_frac=0.5, mode='max', adaptive_temp=False):
    """Exponentiated gradient with adaptive features.
    mode: 'max' for max(conv), 'var' for sum(conv^2)
    Returns optimized a (sum=1, non-negative).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    c = fast_convolve(a)
    if mode == 'max':
        best_obj = c.max()
    else:
        best_obj = np.sum(c * c)
    best_a = a.copy()

    avg_a = a.copy()
    count = 0
    eta = eta0
    temp = temp0
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1

        if adaptive_temp:
            c_range = c.max() - c.min()
            if c_range > 1e-15:
                temp = min(c_range / 5.0, 0.1)
                temp = max(temp, 1e-7)

        if mode == 'max':
            top_k = max(5, int(n * top_k_frac))
            if top_k < len(c):
                indices = np.argpartition(c, -top_k)[-top_k:]
                peek = c[indices]
            else:
                indices = np.arange(len(c))
                peek = c.copy()

            scaled = peek / temp
            scaled -= scaled.max()
            w = np.exp(scaled)
            ws = w.sum()
            if ws < 1e-300:
                w = np.zeros(len(indices))
                w[np.argmax(peek)] = 1.0
            else:
                w /= ws

            sg = np.zeros(n)
            for idx in range(len(indices)):
                if w[idx] < 1e-15:
                    continue
                k = int(indices[idx])
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                sg[js:je + 1] += 2.0 * w[idx] * a[k - je:k - js + 1][::-1]

            sg_max = sg.max()
            if sg_max < 1e-15:
                break
            sg = sg / sg_max

        else:
            # Gradient of sum(c^2): sg[i] = 2 * sum_k c[k] * a[k-i]
            L = 1 << (len(c) + n).bit_length()
            C = np.fft.rfft(c, L)
            A = np.fft.rfft(a, L)
            cross = np.fft.irfft(C * np.conj(A), L)
            sg = 2.0 * cross[:n]
            sg_max = sg.max()
            if sg_max < 1e-15:
                break
            sg = sg / sg_max

        # Multiplicative update
        log_update = -eta * sg
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-300)
        s_new = a_new.sum()
        if s_new < 1e-300:
            pass
        else:
            a_new = a_new / s_new
            c_new = fast_convolve(a_new)

            if mode == 'max':
                obj_new = c_new.max()
            else:
                obj_new = np.sum(c_new * c_new)

            count += 1
            avg_a = (count * avg_a + a_new) / (count + 1)

            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1

            a = a_new
            c = c_new

        if it % 500 == 0:
            c_avg = fast_convolve(avg_a)
            if mode == 'max':
                obj_avg = c_avg.max()
            else:
                obj_avg = np.sum(c_avg * c_avg)
            if obj_avg < best_obj - 1e-14:
                best_obj = obj_avg
                best_a = avg_a.copy()

        eta *= eta_decay
        if not adaptive_temp:
            temp *= temp_decay

        if no_improve > 2000:
            rng_local = np.random.default_rng(it * 31 + 7)
            a = best_a + rng_local.standard_normal(n) * 0.002
            a = np.maximum(a, 0)
            ps = a.sum()
            if ps > 1e-10:
                a = a / ps
            else:
                a = best_a.copy()
            c = fast_convolve(a)
            no_improve = 0
            eta = eta0 * 0.5
            if not adaptive_temp:
                temp = max(temp, temp0 * 0.01)
            avg_a = a.copy()
            count = 0

    return best_a.copy()


def gio_optimize(a0, n_iters=2000):
    """Gerchberg-Saxton / HIO for flat magnitude spectrum."""
    n = len(a0)
    L = 1 << (2 * n).bit_length()

    a = a0.copy()
    for it in range(n_iters):
        A = np.fft.rfft(a, L)
        mag = np.abs(A)
        target_mag = np.sqrt(L)
        A_new = target_mag * A / (mag + 1e-300)
        a_new = np.fft.irfft(A_new, L)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s > 1e-10:
            a_new = a_new / s
        else:
            a_new = a / a.sum()
        a = a_new

    return a


def local_polish(a0, time_limit):
    """Coordinate descent local search with FFT-based evaluation."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    no_improve = 0

    factors = [0.5, 0.75, 0.9, 1.0 / 0.9, 1.0 / 0.75, 1.0 / 0.5]

    while time.monotonic() - t0 < time_limit:
        i = rng.integers(n)
        old_val = a[i]
        if old_val < 1e-15:
            no_improve += 1
            if no_improve > 2000:
                break
            continue

        improved = False
        for factor in factors:
            a[i] = old_val * factor
            s2 = a.sum()
            if s2 < 1e-15:
                a[i] = old_val
                continue
            a_norm = a / s2
            c2 = fast_convolve(a_norm)
            obj2 = c2.max()
            if obj2 < best_obj - 1e-15:
                best_obj = obj2
                best_a = a_norm.copy()
                a = a_norm.copy()
                c = c2
                improved = True
                break
            elif factor > 1.0:
                a[i] = old_val
                break
            else:
                a[i] = old_val

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 3000:
                # Random batch perturbation
                batch = max(1, n // 8)
                coords = rng.choice(n, size=batch, replace=False)
                for ci in coords:
                    if a[ci] > 1e-15:
                        a[ci] *= rng.uniform(0.6, 1.6)
                s2 = a.sum()
                if s2 > 1e-10:
                    a = a / s2
                    c2 = fast_convolve(a)
                    if c2.max() < best_obj:
                        best_obj = c2.max()
                        best_a = a.copy()
                        no_improve = 0

    return best_a


def gen_random_phase_init(n, rng):
    """Generate from random phase with flat magnitude spectrum."""
    L = 1 << (2 * n).bit_length()
    half = L // 2 + 1
    phase = rng.uniform(0, 2 * np.pi, half)
    A = np.sqrt(L) * np.exp(1j * phase)
    A[0] = np.abs(A[0])
    if L % 2 == 0:
        A[-1] = np.abs(A[-1])
    a = np.fft.irfft(A, L)[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-10:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def gen_chirp_init(n, rng):
    """Generate from chirp pattern made non-negative."""
    x = np.arange(n, dtype=np.float64)
    for _ in range(5):
        f = rng.uniform(0.5, 2.0)
        phase = 2 * np.pi * f * x * x / n
        chirp = np.cos(phase)
        chirp = np.maximum(chirp, 0)
        s = chirp.sum()
        if s > 1e-10:
            chirp = chirp / s
            return chirp
    return np.ones(n) / n


def generate_inits(best_a, rng, n):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    if best_a is not None and len(best_a) == n:
        inits.append(best_a.copy())
        for scale in [0.01, 0.05, 0.15]:
            p = best_a + rng.standard_normal(n) * scale / max(best_a.sum(), 1e-10)
            p = np.maximum(p, 0)
            ps = p.sum()
            if ps > 1e-10:
                inits.append(p / ps)

    inits.append(gen_chirp_init(n, rng))
    for _ in range(4):
        inits.append(gen_random_phase_init(n, rng))

    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())

    c1 = np.abs(np.cos(np.pi * x / n)) + 0.01
    inits.append(c1 / c1.sum())

    inits.append(np.ones(n) / n)

    for _ in range(2):
        inits.append(rng.dirichlet(np.ones(n) * 0.1))

    return inits
