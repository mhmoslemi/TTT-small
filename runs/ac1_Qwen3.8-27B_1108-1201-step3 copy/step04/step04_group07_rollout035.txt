    # And m ranges from k - j_max to k - j_min = m_start to m_end
    if m_start <= m_end:
        a_slice = a[j_min:j_max+1]  # a[k-m] for m from m_start to m_end
        # m = m_start corresponds to k - j_max (the last element of a_slice)
        # m = m_end corresponds to k - j_min (the first element of a_slice)
        # So g[m_start:m_end+1] = 2 * reverse(a_slice)
        g[m_start:m_end+1] = 2.0 * a_slice[::-1]
    return g

import numpy as np
import time
from scipy.optimize import linprog

GLOBAL_BEST = [None]  # To track the best solution globally

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_obj_normalized(a):
    """Compute 2n * max(conv) for normalized a (sum=1, a>=0)."""
    n = len(a)
    conv = fast_convolve(a)
    max_c = conv.max()
    return 2.0 * n * max_c

def logsumexp_smooth(a, tau, top_k=80):
    """Compute log-sum-exp approximation of max(conv)."""
    conv = fast_convolve(a)
    if top_k < len(conv):
        top_idx = np.argpartition(conv, -top_k)[-top_k:]
        top_vals = conv[top_idx]
    else:
        top_idx = np.arange(len(conv))
        top_vals = conv
    m = top_vals.max()
    lse = m + np.log(np.sum(np.exp(top_vals - m)))
    return lse, conv, top_idx

def lse_gradient(a, conv, top_idx, tau):
    """Gradient of log-sum-exp objective w.r.t. a."""
    n = len(a)
    top_vals = conv[top_idx]
    m = top_vals.max()
    w = np.exp(top_vals - m)
    w /= w.sum()
    
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        wi = w[i]
        if wi < 1e-14:
            continue
        # grad[m] += wi * 2 * a[k-m] for valid m
        m_start = max(0, k - n + 1)
        m_end = min(n - 1, k)
        if m_start <= m_end:
            j_min = k - m_end
            j_max = k - m_start
            grad[m_start:m_end+1] += 2.0 * wi * a[j_min:j_max+1][::-1]
    return grad

def adam_step(params, a, grad, beta1, beta2, lr, t):
    """One Adam step. params stores m, v, t."""
    m = beta1 * params['m'] + (1 - beta1) * grad
    v = beta2 * params['v'] + (1 - beta2) * grad**2
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    params['m'] = m
    params['v'] = v
    params['t'] = t + 1
    return a_new

def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C

def solve_lp(a, M, top_k=None):
    """Solve LP: maximize sum(g) s.t. conv(a,g) <= M, g >= 0.
    If top_k given, only constrain the top_k peaks."""
    n = len(a)
    C_full = build_conv_matrix(a)
    
    if top_k is not None and top_k < 2*n-1:
        conv_a = fast_convolve(a)
        top_idx = np.argpartition(conv_a, -top_k)[-top_k:]
        C = C_full[top_idx, :]
        b_ub = np.full(top_k, M)
    else:
        C = C_full
        b_ub = np.full(2*n-1, M)
    
    c = -np.ones(n)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum, g_sum
    except Exception:
        pass
    return a.copy(), 1.0

def adam_optimize(a0, time_limit, lr=0.01, tau_start=0.01, tau_end=0.001,

    else:

    n = len(a)
    
    params = {'m': np.zeros(n), 'v': np.zeros(n), 't': 0}
    t0 = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    step = 0
    
    while time.monotonic() - t0 < time_limit:

def resample_sequence(seq, new_n):
    """Resample a sequence to a different length."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s > 1e-15:
        new_seq = new_seq / s
    return new_seq
