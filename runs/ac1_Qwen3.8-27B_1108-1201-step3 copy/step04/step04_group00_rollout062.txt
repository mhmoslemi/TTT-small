s.t. `(a*a)[k] <= M` for all `k = 0, ..., 2n-2`
     `sum(a) = 1`
     `a >= 0`

But this is non-convex because of the `(a*a)[k] <= M` constraints (they're quadratic).

**SDP relaxation:**
Let `X = a * a^T` (rank-1, PSD). Then `(a*a)[k] = sum_{i-j=k} X[i,j]`.

The SDP relaxation:
`min M`
s.t. `sum_{i-j=k} X[i,j] <= M` for all `k`
     `sum_i X[i,i] >= (sum_i a_i)^2 = 1` (by Cauchy-Schwarz, actually `sum_i a_i^2 <= 1` and `(sum a_i)^2 = 1`)
     
Hmm, this doesn't directly work because we need `X` to be rank-1 for it to correspond to a valid `a`.

Let me think differently. Actually, `(a*a)[k] = sum_j a[j] * a[k-j]` is a quadratic form in `a`. The constraint `(a*a)[k] <= M` defines a convex set in `a` (it's an ellipsoid). The intersection of all these ellipsoids with the simplex is our feasible set.

We can use an **ellipsoid-cutting** approach or **separation oracle** based method:
1. Start with a point in the simplex
2. Check if `max_k (a*a)[k] <= M`
3. If not, the violated constraint gives a "cut"
4. Use the cutting plane to find a better point

But this is essentially what the LP-FW approach does.

Let me try yet another angle: **Power iteration / multiplicative weight update**.

The idea: if `conv[k]` is large for some `k`, we want to reduce the "correlation" at lag `k`. We can do this by reducing the product `a[j]*a[k-j]` for the relevant pairs.

A natural update: `a[j] <- a[j] * exp(-eta * sum_k p[k] * (a[k-j] + a[j-k]))`

This is exactly the gradient step with the LSE-smoothed objective! So we're back to the subgradient method.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LSE-smoothed FISTA, LP-Frank-Wolfe,
coordinate descent with exact 1D optimization, and multi-length multi-restart search.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_compute_grad(y, p):
    """Compute gradient of LSE approximation using FFT.
    
    grad[i] = sum_k p[k] * (y[k-i] + y[i-k])
    
    Uses cross-correlation and convolution via FFT.
    """
    n = len(y)
    L = 1 << (3 * n - 2 + 1).bit_length()
    P_fft = np.fft.rfft(p, L)
    Y_fft = np.fft.rfft(y, L)
    # Cross-correlation: (p ⋆ y)[i] = sum_k p[k]*y[k-i]
    f1 = np.fft.irfft(P_fft * np.conj(Y_fft), L)[:n]
    # Convolution: (p * y)[i] = sum_k p[k]*y[i-k]
    f2 = np.fft.irfft(P_fft * Y_fft, L)[:n]
    return f1 + f2


def lse_fista(f0, time_limit, verbose=False, seed=0):
    """LSE-smoothed FISTA (Nesterov accelerated gradient) with adaptive temperature."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    x = f.copy()
    x_prev = f.copy()
    t = 1.0

    # Temperature: controls smoothing level
    temp_base = 0.02 / np.sqrt(n)
    temp_decay = 0.99995
    temp_min = 1e-8

    # Step size
    step = 1.0 / (2.0 * n)
    step_min = 1e-10
    step_decay = 0.95

    rng = np.random.default_rng(seed)
    no_improve = 0
    t_start = time.monotonic()
    it = 0

    while time.monotonic() - t_start < time_limit:
        it += 1

        # Nesterov momentum
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        beta = (t - 1.0) / t_new
        y = x + beta * (x - x_prev)
        x_prev = x
        t = t_new

        # Project y to simplex for gradient evaluation
        y_proj = project_simplex(y)

        # Compute LSE gradient
        conv = fast_convolve(y_proj)
        temp = max(temp_base * (temp_decay ** it), temp_min)

        scaled = conv / temp
        scaled -= scaled.max()
        p = np.exp(scaled)
        ps = p.sum()
        if ps < 1e-300:
            p = np.zeros_like(conv)
            p[np.argmax(conv)] = 1.0
        else:
            p /= ps

        grad = lse_compute_grad(y_proj, p)

        # Gradient step
        x_new = project_simplex(y - step * grad)

        # Evaluate
        conv_new = fast_convolve(x_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = x_new.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"      LSE-FISTA it={it}: obj={best_obj:.10f}, temp={temp:.2e}, step={step:.2e}")
        else:
            no_improve += 1

        # Adaptive step size
        if no_improve > 100:
            step *= 0.7
            no_improve = 0
            if step < step_min:
                step = step_min
        elif no_improve == 0:
            step = min(step * 1.05, 1.0 / (2.0 * n))

        x = x_new

        # Periodic restart
        if no_improve > 5000:
            noise = rng.standard_normal(n) * 0.003
            x = project_simplex(best_f + noise)
            x_prev = x.copy()
            t = 1.0
            no_improve = 0
            step = 1.0 / (2.0 * n)

        # Time check for verbose
        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t_start
            print(f"      LSE-FISTA it={it}: elapsed={elapsed:.1f}s, best={best_obj:.10f}")

    return best_f, best_obj


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False, seed=0):
    """LP-based Frank-Wolfe with adaptive M relaxation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        g = solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.15
            if m_exp > 8.0:
                m_exp = 1.0
                no_improve += 10
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                m_exp = min(m_exp * 1.05, 3.0)
                if m_exp > 3.0:
                    break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M

        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.85, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_exp = 1.0
            f = f_new

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 10 == 0:
                    print(f"      LP-FW it={it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.08, 5.0)

        if no_improve > 40:
            rng2 = np.random.default_rng(it * 31 + 7 + seed)
            scale = 0.001 + 0.002 * min(no_improve // 50, 5)
            noise = rng2.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with exact 1D golden-section search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0
    no_sweep_improve = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        sweep_improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-14:
                continue

            one_minus_old = 1.0 - old_val
            if one_minus_old < 1e-14:
                continue

            # b[i] = f[i] / (1 - f[j]) for i != j
            b = f.copy()
            b[j] = 0.0
            b /= one_minus_old

            # Precompute conv_b = b * b
            conv_b = fast_convolve(b)

            # Define q_k(t) for all k
            # q_k(t) = (1-t)^2 * c_k + 2*t*(1-t)*d_k + t^2 * e_k
            # where c_k = conv_b[k] - (terms with j), d_k = b[k-j] (if valid), e_k = 1 if k=2j
            
            # Compute c_k, d_k, e_k arrays
            ck = conv_b.copy()
            dk = np.zeros(2 * n - 1)
            ek = np.zeros(2 * n - 1)

            # Subtract terms involving j
            # conv_b[k] includes b[j]*b[k-j] and b[k-j]*b[j] terms (but b[j]=0 now)
            # Actually conv_b is computed with b[j]=0, so it already excludes j terms.
            # So c_k = conv_b[k] directly.
            
            # d_k = b[k-j] if 0 <= k-j < n (and k-j != j, but b[j]=0 so it doesn't matter)
            for k in range(2 * n - 1):
                kj = k - j
                if 0 <= kj < n:
                    dk[k] = b[kj]  # b[j] = 0 so this handles k=2j case too
                ek[k] = 1.0 if k == 2 * j else 0.0

            # q_k(t) = (1-t)^2 * ck[k] + 2*t*(1-t)*dk[k] + t^2 * ek[k]
            #        = ck[k]*(1 - 2t + t^2) + 2*dk[k]*(t - t^2) + ek[k]*t^2
            #        = ck[k] + t*(-2*ck[k] + 2*dk[k]) + t^2*(ck[k] - 2*dk[k] + ek[k])
            
            A_k = ck - 2.0 * dk + ek
            B_k = -2.0 * ck + 2.0 * dk
            C_k = ck

            # Minimize max_k (A_k * t^2 + B_k * t + C_k) over t in [0, 1]
            # Using golden section search
            
            def obj_1d(t):
                vals = A_k * t * t + B_k * t + C_k
                return vals.max()

            # Golden section search on [0, 1]
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a_lo = 0.0
            a_hi = 1.0
            c1 = a_hi - gr * (a_hi - a_lo)
            c2 = a_lo + gr * (a_hi - a_lo)
            f1 = obj_1d(c1)
            f2 = obj_1d(c2)

            for _ in range(40):
                if f1 < f2:
                    a_hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = a_hi - gr * (a_hi - a_lo)
                    f1 = obj_1d(c1)
                else:
                    a_lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = a_lo + gr * (a_hi - a_lo)
                    f2 = obj_1d(c2)

            t_opt = (a_lo + a_hi) / 2.0
            obj_at_opt = obj_1d(t_opt)

            if obj_at_opt < best_obj - 1e-13:
                # Apply the update
                f[j] = t_opt
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f[mask] = b[mask] * (1.0 - t_opt)
                f = f / f.sum()
                best_obj = obj_at_opt
                best_f = f.copy()
                sweep_improved = True

        if not sweep_improved:
            no_sweep_improve += 1
            if no_sweep_improve >= 2:
                break
        else:
            no_sweep_improve = 0

        if verbose and passes % 5 == 0:
            print(f"      CD pass={passes}: obj={best_obj:.10f}")

    return best_f, best_obj


def sg_polish(f0, time_limit, lr=0.002, mom=0.94, top_k=50, temp=0.00003, seed=0):
    """Subgradient polishing with softmax-weighted peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp

    while time.monotonic() - t0 < time_limit:
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9997
        cur_temp *= 1.004

        if no_improve > 400:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0003)
            no_improve = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return float('inf')
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = try_seq(parent, "parent")
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(512) / 512
        try_seq(parent, "uniform_512")

    n_parent = len(parent)

    # === Phase 1: LSE-FISTA for quick improvement ===
    print(f"\n=== Phase 1: LSE-FISTA (n={n_parent}) ===")
    rem = deadline - time.monotonic()
    t1 = min(200, rem * 0.15)
    if t1 > 20:
        print(f"  Budget: {t1:.0f}s")
        a_opt, _ = lse_fista(parent, time_limit=t1, verbose=True, seed=seed)
        try_seq(a_opt, "lse_fista_1")

    # === Phase 2: LP-FW on best ===
    print(f"\n=== Phase 2: LP-FW (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t2 = min(500, rem * 0.35)
        if t2 > 30:
            print(f"  Budget: {t2:.0f}s")
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t2, verbose=True, seed=seed)
            try_seq(a_opt, "lp_fw_1")

    # === Phase 3: Coordinate descent ===
    print(f"\n=== Phase 3: Coordinate Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t3 = min(300, rem * 0.25)
        if t3 > 20:
            print(f"  Budget: {t3:.0f}s")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t3, verbose=True)
            try_seq(a_opt, "cd_1")

    # === Phase 4: SG polish ===
    print(f"\n=== Phase 4: SG Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        t4 = min(120, rem * 0.12)
        if t4 > 10:
            print(f"  Budget: {t4:.0f}s")
            a_opt, _ = sg_polish(best_seq, time_limit=t4,
                                  lr=0.001, mom=0.95,
                                  top_k=max(20, n_best // 8),
                                  temp=0.00002, seed=seed)
            try_seq(a_opt, "sg_1")

    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length (best={best_score:.6f}) ===")
    if best_seq is not None:
        base = best_seq
        for target_n in [1024, 512, 768]:
            if target_n == len(base):
                continue
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            t_alloc = min(rem * 0.2, 180)
            if t_alloc < 20:
                break
            f_interp = interpolate_sequence(base, target_n)
            print(f"  Exploring n={target_n}, budget={t_alloc:.0f}s")

            # LSE-FISTA
            t_sub = t_alloc * 0.5
            if t_sub > 10:
                a_opt, _ = lse_fista(f_interp, time_limit=t_sub, seed=seed + target_n)
                try_seq(a_opt, f"lse_n{target_n}")
                if best_seq is not None and len(best_seq) == target_n:
                    f_interp = best_seq

            # LP-FW
            rem = deadline - time.monotonic()
            t_lp = min(rem * 0.15, 100)
            if t_lp > 15:
                a_opt, _ = lp_fw_optimize(f_interp, time_limit=t_lp, seed=seed + target_n)
                try_seq(a_opt, f"lpfw_n{target_n}")

    # === Phase 6: Deep polish on best ===
    print(f"\n=== Phase 6: Deep Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()

        if rem > 30:
            # Another CD round
            t_cd = min(rem * 0.3, 150)
            if t_cd > 15:
                print(f"  CD round 2, budget={t_cd:.0f}s")
                a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd)
                try_seq(a_opt, "cd_2")

        rem = deadline - time.monotonic()
        if rem > 20:
            # LP-FW round 2
            t_lp = min(rem * 0.3, 120)
            if t_lp > 15:
                print(f"  LP-FW round 2, budget={t_lp:.0f}s")
                a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_lp, seed=seed + 999)
                try_seq(a_opt, "lpfw_2")

        rem = deadline - time.monotonic()
        if rem > 15:
            # LSE-FISTA round 2 (fresh)
            t_lse = min(rem * 0.3, 80)
            if t_lse > 10:
                print(f"  LSE-FISTA round 2, budget={t_lse:.0f}s")
                a_opt, _ = lse_fista(best_seq, time_limit=t_lse, seed=seed + 777)
                try_seq(a_opt, "lse_2")

        rem = deadline - time.monotonic()
        if rem > 10:
            # Final SG
            t_sg = min(rem * 0.4, 40)
            if t_sg > 5:
                a_opt, _ = sg_polish(best_seq, time_limit=t_sg,
                                      lr=0.0005, mom=0.96,
                                      top_k=max(10, n_best // 4),
                                      temp=0.00001, seed=seed + 555)
                try_seq(a_opt, "sg_final")

    # === Phase 7: Multi-restart with perturbations ===
    print(f"\n=== Phase 7: Multi-restart (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for restart in range(5):
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            t_r = min(rem * 0.15, 40)
            if t_r < 8:
                break
            noise_scale = [0.001, 0.002, 0.003, 0.005, 0.008][restart]
            noise = rng.standard_normal(n_best) * noise_scale
            f_pert = project_simplex(best_seq + noise)
            # Quick LSE
            a_opt, _ = lse_fista(f_pert, time_limit=t_r, seed=seed + 100 + restart)
            try_seq(a_opt, f"restart_{restart}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
