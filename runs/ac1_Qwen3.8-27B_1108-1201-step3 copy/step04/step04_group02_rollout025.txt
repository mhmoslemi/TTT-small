Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using convex optimization with

coordinate descent polishing.

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def softmax_obj_grad(z, beta, n):
    """Compute objective and gradient for softmax-parameterized problem."""
    # a = softmax(z)
    z_shift = z - z.max()
    exp_z = np.exp(z_shift)
    a = exp_z / exp_z.sum()
    
    # conv = auto-convolution of a
    conv = fast_convolve(a)
    
    # Smooth max via log-sum-exp
    shifted = beta * conv
    shifted -= shifted.max()
    exp_shifted = np.exp(shifted)
    w_sum = exp_shifted.sum()
    w = exp_shifted / w_sum
    
    # Objective
    obj = 2.0 * n * (conv.max() + np.log(w_sum) / beta)
    
    # Gradient of smooth max w.r.t. conv: 2n * w
    # Gradient of conv w.r.t. a: cross-correlation
    # df/da_i = 4n * sum_j a[j] * w[j+i]
    grad_da = np.zeros(n)
    for i in range(n):
        j_max = min(n - 1, 2*n - 2 - i)
        grad_da[i] = 4.0 * n * np.dot(a[:j_max+1], w[i:i+j_max+1])
    
    # Chain rule for softmax: df/dz = a * (df/da - a * (a^T df/da))
    S = np.dot(a, grad_da)
    grad_z = a * (grad_da - a * S)
    
    return obj, grad_z

def exact_obj(a):
    """Compute the exact objective value."""
    n = len(a)
    conv = fast_convolve(a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 1e-15:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)
