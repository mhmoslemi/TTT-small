import numpy as np

def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of non-negative coefficients and returns a float score.
    The score is defined as: 2*n*max(convolve(sequence, sequence, mode='full'))/(sum(sequence))**2,
    with normalization and defensive clipping (max 0, min 1000) applied.
    NaN or infinity returns np.inf.
    """
    n = len(sequence)
    # First, convert to floats and ensure non-negative with 0.01 minimum sum
    try:
        arr = np.array(sequence, dtype=np.float64)
    except TypeError:
        return np.inf
    arr = np.maximum(arr.astype(np.float64), 0)
    total_sum = arr.sum()
    if total_sum < 0.01:
        return np.inf
    
    # Automatically handle ndarray or list input
    if isinstance(arr, np.ndarray):
        if arr.ndim != 1:
            return np.inf  # Ensure 1D arrays
    elif isinstance(arr, list):
        if not all(isinstance(x, (int, float)) for x in arr):
            return np.inf
    
    # Perform full convolution using FFT
    try:
        from numpy.fft import rfft, irfft
        size = 2 * n - 1
        fft_size = 1 << (size - 1).bit_length()
        A = rfft(arr, fft_size)
        conv = irfft(A * A, fft_size)
        conv = conv[:size]
        max_val = np.max(conv)
    except Exception as e:
        print(f"Convolution error: {e}")
        return np.inf
    
    score = float(2 * n * max_val / (total_sum ** 2))
    return score

class StepFunctionOptimizer:
    def __init__(self, n, alpha=0.999, beta=0.1, temperature=0.001):
        """
        Optimizes a step function sequence of fixed length n.
        
        Parameters:
        n (int): Length of the sequence
        alpha (float): Maximum allowed gradient magnitude
        beta (float): Learning rate (initial)
        temperature (float): The smallest peak height to consider
        """
        self.n = n
        self.alpha = alpha  # Your threshold for peak contribution
        self.beta = beta  # Starting learning rate
        self.temperature = temperature  # Smallest detectable peak
        self.decay_lr = 0.9995  # lr decay per iteration
        self.decay_temp = 1.002  # temperature decay per iteration
        
    def project_simplex(self, vec):
        """
        Projects coefficients to the feasible simplex (non-negative and sum=1)
        """
        n_vec = len(vec)
        if n_vec == 1:
            return np.array([max(vec[0], 0.0)])
        
        # For 1D numpy arrays or lists
        if isinstance(vec, np.ndarray) and vec.ndim != 1:
            return np.ones(n_vec) / n_vec
        
        vec = np.sort(vec)[::-1]
        cum_ex = np.cumsum(vec)
        cond = vec * np.arange(1, n_vec+1) > (cum_ex - 1)
        if not cond.any():
            return np.ones(n_vec) / n_vec
        
        cutoff = (np.nonzero(cond)[0][-1] + 1)
        theta = (cum_ex[cutoff-1] - 1) / cutoff
        return np.maximum(vec - theta, 0)
    
    def fast_convolve(self, a):
        """
        Returns the convolution for the current vector using FFT. orizeulated for speed.
        """
        size = 2 * self.n - 1
        fft_size = 1 << (size - 1).bit_length()
        A = np.fft.rfft(a.astype(np.float64), fft_size)
        conv = np.fft.irfft(A * A, fft_size)
        return conv[:size]
    
    def evaluate_objective(self, sequence):
        """Adapted to always call the external evaluator without direct access to arr коэффициентов for indexчёт."""
        seq_copy = sequence.copy()
        score = evaluate_sequence(seq_copy)
        if np.isnan(score) or score == np.inf:
            return float('inf')
        return score
    
    def optimize(self, initial_sequence, time_budget=600):
        """
        Optimizes the sequence using Multiplicative Weights Update Method (MWUM).
        
        Parameters:
        initial_sequence (list[float]): Starting coefficients
        time_budget (float): Time available in seconds
        
        Returns:
        dict: Contains best_sequence and its score
        """
        best_score_at = None
        best_sequence = None
        start_time = time.time()
        current_sequence = initial_sequence
        
        while time.time() - start_time < time_budget:
            # Project to any feasible non-negative sequence with sum > 0.01
            if isinstance(current_sequence, list) and any(x < 0 for x in current_sequence):
                continue
                
            conv = self.fast_convolve(np.array(current_sequence))
            current_score = np.max(conv) / (np.array(current_sequence).sum() ** 2)
            
            # Find top-k peaks across the convolution
            top_k_index = np.arg.partition(conv, -self.n)[-self.n:]
            sorted_indices = np.argsort(conv)[::-1][:self.n]
            
            # Compute gradient scores and allocate change
            gradient = np.zeros(self.n)
            total_weight = 0
            
            # Ranking-based importance allocation (rank 0 has highest weight)
            for i, idx in enumerate(sorted_indices):
                if i >= self.n:
                    break
                contribution = conv[idx] / self.temperature
                if contribution > 0:   # Ignore if not meaningful
                    # Compute gradient contribution term
                    grad_term = 2 * current_sequence[idx - (idx // self.temperature)]
                    gradient[idx] += grad_term * (contribution / (current_sequence[0] + 1))
                    total_weight += contribution
                else:
                    break
            
            # Normalize gradient
            if total_weight < 1e-5:
                break
            gradient = gradient / total_weight
            
            # Update the current sequence by exponential cooling
            update_factor = np.exp(-self.beta * gradient)
            self.beta *= self.decay_lr
            self.temperature *= self.decay_temp
            
            new_sequence = [x * update_factor for x in current_sequence]
            new_sequence_reproj = self.project_simplex(new_sequence)
            
            # Re-normalize to achieve valid sequence bound checks
            if self.evaluate_objective(new_sequence_reproj) < current_score + 1e-6:
                best_sequence = new_sequence_reproj
                best_score_at = current_score
    
        return best_sequence, best_score_at
