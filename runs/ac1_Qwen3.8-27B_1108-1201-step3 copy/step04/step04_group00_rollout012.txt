"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(f, conv, top_k=20, temp=0.01):
    """Compute weighted subgradient of max_k conv[k] using softmax over top peaks."""
    n = len(f)
    conv_len = len(conv)
    k_actual = min(top_k, conv_len)
    
    # Get top k peaks
    peak_indices = np.argpartition(conv, -k_actual)[-k_actual:]
    peak_values = conv[peak_indices]
    
    # Softmax weights
    scaled = peak_values / temp
    scaled -= scaled.max()
    weights = np.exp(scaled)
    weights /= weights.sum()
    
    # Compute subgradient
    sg = np.zeros(n)
    for i in range(len(peak_indices)):
        if weights[i] < 1e-15:
            continue
        k = int(peak_indices[i])
        # d/d f[j] of sum_l f[l]*f[k-l] = f[k-j] (for valid j)
        # The subgradient contribution from lag k is: for each j, f[k-j] if k-j is valid
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += weights[i] * f[k - idx]
    
    # Factor of 2 for the quadratic (since d/dx of x^T B x = 2Bx for symmetric B)
    # Actually, (a*a)[k] = sum_j a_j * a_{k-j}, so d/da_j = a_{k-j}
    # The max is over k, so subgradient is from active k's
    return sg  # Note: no factor of 2 needed since we're taking subgradient of the max

def subgradient_optimize(f0, time_limit, top_k=30, temp_init=0.01, lr_init=0.01, verbose=False):
    """Subgradient descent on the simplex with adaptive parameters."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        sg = compute_subgradient(f, conv, top_k=top_k, temp=temp)
        
        # Mirror descent step (multiplicative weights on simplex)
        # f_new[j] = f[j] * exp(-lr * sg[j]) / sum(...)
        log_step = -lr * sg
        log_step -= log_step.max()  # numerical stability
        f_new = f * np.exp(log_step)
        f_new_sum = f_new.sum()
        if f_new_sum < 1e-15:
            f_new = np.ones(n) / n
        else:
            f_new /= f_new_sum
        
        conv = fast_convolve(f_new)
        cur_obj = conv.max()
        
        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Don't update f, or update with smaller step
            if no_improve > 20:
                # Try smaller step
                lr *= 0.8
                f_new = f  # keep old f
            else:
                f = f_new
        
        # Adaptive temp increase (softmax becomes more uniform)
        temp *= 1.002
        
        # Periodic restart
        if no_improve > 200:
            rng = np.random.default_rng(it * 17 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            lr = lr_init * 0.5
            no_improve = 0
            temp = temp_init * 0.5
        
        if verbose and it % 100 == 0:
            print(f"    SG iter {it}: obj={best_obj:.10f}, lr={lr:.6f}, temp={temp:.6f}")
    
    return best_f, best_obj

def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        g = solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.1
            if m_exp > 5.0:
                m_exp = 1.0
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_exp = 1.0
            f = f_new
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.08, 4.0)
        
        if no_improve > 50:
            rng = np.random.default_rng(it * 31 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0
    
    return best_f, best_obj

def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent polishing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-12:
                continue
            
            conv = fast_convolve(f)
            cur_peak = conv.max()
            best_j_val = old_val
            best_j_obj = cur_peak
            
            multipliers = [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0]
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()
                
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved and passes > 3:
            break
    
    return best_f, best_obj

def generate_initializations(n, seed=42):
    """Generate diverse initializations of length n."""
    rng = np.random.default_rng(seed)
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Hamming window
    hamming = 0.54 - 0.46 * np.cos(2 * np.pi * np.arange(n) / (n - 1))
    inits.append(hamming / hamming.sum())
    
    # Hann window
    hann = 0.5 * (1 - np.cos(2 * np.pi * np.arange(n) / (n - 1)))
    inits.append(hann / hann.sum())
    
    # U-shaped (inverted parabola)
    x = np.linspace(-1, 1, n)
    u_shape = x**2 + 0.1
    inits.append(u_shape / u_shape.sum())
    
    # Cosine bell (concentrated in middle)
    cos_bell = np.cos(np.pi * x)**2 + 0.01
    inits.append(cos_bell / cos_bell.sum())
    
    # Two peaks
    two_peak = np.exp(-((x - 0.5)**2) / 0.05) + np.exp(-((x + 0.5)**2) / 0.05)
    inits.append(two_peak / two_peak.sum())
    
    # Random
    rand = rng.random(n) + 0.01
    inits.append(rand / rand.sum())
    
    # Exponential from edges
    exp_edge = np.exp(-np.abs(x) * 2) + 0.01
    inits.append(exp_edge / exp_edge.sum())
    
    # Random smooth (low-frequency random)
    freqs = rng.choice([1, 2, 3, 4, 5], size=3, replace=False)
    phases = rng.uniform(0, 2*np.pi, size=3)
    smooth = np.zeros(n)
    for i, (f_freq, ph) in enumerate(zip(freqs, phases)):
        smooth += np.sin(2*np.pi*f_freq*x + ph)
    smooth = np.abs(smooth) + 0.1
    inits.append(smooth / smooth.sum())
    
    # Step-like
    step = np.concatenate([np.ones(n//3), np.ones(n//3)*0.3, np.ones(n - 2*(n//3))])
    inits.append(step / step.sum())
    
    return inits

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
