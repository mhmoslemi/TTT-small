import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def softmax(z):
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()

def adam_optimizer(z, lr, z_top_k_frac, max_iters, time_limit):
    """
    Adam optimization in log-softmax domain.
    z is the log-parameters, a = softmax(z).
    """
    n = len(z)
    a = softmax(z)
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    
    t0 = time.monotonic()
    no_improve = 0
    
    top_k = max(1, int(n * z_top_k_frac))
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        
        # Find top K peaks
        if top_k < len(c):
            peak_indices = np.argpartition(c, -top_k)[-top_k:]
        else:
            peak_indices = np.arange(len(c))
        
        # Compute subgradient s_i = (1/K) * sum_{k in peaks} a_{k-i} * [0<=k-i<n]
        s = np.zeros(n)
        for k in peak_indices:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            s[js:je+1] += a[k-je:k-js+1][::-1]
        s /= len(peak_indices)
        
        # z-gradient: 2 * a_k * (s_k - <s, a>)
        sa = np.dot(s, a)
        grad_z = 2.0 * a * (s - sa)
        
        # Numerical stability
        grad_max = np.abs(grad_z).max()
        if grad_max < 1e-15:
            break
        grad_z = grad_z / grad_max  # normalize
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * grad_z ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Compute new a and objective
        a = softmax(z)
        c = fast_convolve(a)
        obj = c.max()
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 5000:
            break

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def safe_softmax(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def compute_z_gradient(a, c, top_k):
    """Compute the gradient of max(c) w.r.t. z in log-softmax parameterization.
    
    Returns (grad_z, peak_indices) where grad_z = 2*a*(s - <s,a>)
    and s_i = (1/K) * sum_{k in peaks} a_{k-i} * [valid]
    """
    n = len(a)
    len_c = len(c)
    
    if top_k < len_c and top_k > 0:
        peak_indices = np.argpartition(c, -top_k)[-top_k:]
    else:
        peak_indices = np.arange(len_c)
    
    s = np.zeros(n)
    for k in peak_indices:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        s[js:je + 1] += a[k - je:k - js + 1][::-1]
    s /= len(peak_indices)
    
    sa = np.dot(s, a)
    grad_z = 2.0 * a * (s - sa)
    return grad_z, peak_indices

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    v = np.zeros(n)
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8
    t_count = 0

    no_improve = 0
    it = 0

        grad_z = grad_z / grad_max

        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * grad_z ** 2
        m_hat = m / (1 - beta1 ** t_count)
        v_hat = v / (1 - beta2 ** t_count)

        lr_t = lr * (0.9995 ** it)

        c = fast_convolve(a)

            no_improve = 0
            m = np.zeros(n)
            v = np.zeros(n)
            t_count = 0

def coordinate_descent(a0, time_limit, n_rounds=200):
    """Coordinate descent: try multiplicative perturbations on random coordinates."""
    a = np.maximum(a0.astype(np.float64), 1e-300)
    a = a / a.sum()

    it = 0
    no_improve = 0

               0.75, 1.0/0.75, 0.9, 1.0/0.9, 0.6, 1.0/0.6]
    
    while time.monotonic() - t0 < time_limit and it < n_rounds:
        it += 1
        improved = False
        
        n_perturb = max(1, n // 8)
        coords = rng.choice(n, size=n_perturb, replace=False)
        
        for i in coords:
            old_val = a[i]
            if old_val < 1e-15:
                continue
            
            best_factor = None
            best_obj_local = best_obj
            
            for factor in factors:
                a[i] = old_val * factor
                a = np.maximum(a, 1e-300)
                s2 = a.sum()
                if s2 < 1e-15:
                    a[i] = old_val
                    continue
                a = a / s2
                c2 = fast_convolve(a)
                obj2 = c2.max()
                if obj2 < best_obj_local - 1e-12:
                    best_obj_local = obj2
                    best_factor = factor
            
            if best_factor is not None:
                a[i] = old_val * best_factor
                a = np.maximum(a, 1e-300)
                a = a / a.sum()
                best_obj = best_obj_local
                best_a = a.copy()
                improved = True
                # Recompute c for the updated a
                c = fast_convolve(a)
            else:
                a[i] = old_val
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 40:
                break
    
    return best_a.copy(), best_obj


def generate_diverse_inits(rng, n, best_a=None):

    inits.append(g / g.sum())
    
    # Cosine variants
    for freq in [1, 2, 3]:
        c1 = np.abs(np.cos(np.pi * freq * x / n)) + 0.05
        inits.append(c1 / c1.sum())
    
    # Triangular
    tri = 1.0 - 2.0 * np.abs(x - 0.5 * (n - 1)) / n
    inits.append(tri / tri.sum())
    
    # Random Dirichlet
    for alpha in [0.2, 0.5, 1.0, 2.0]:
        r = rng.dirichlet(np.ones(n) * alpha)

    return inits


def resample_sequence(a, new_n):
    """Resample a sequence to a new length via linear interpolation."""
    old_n = len(a)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 1e-10)
    return a_new / a_new.sum()

        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = evaluate_sequence(parent.tolist())
            print(f"Parent: n={len(parent)}, score={parent_score:.10f}")
            if np.isfinite(parent_score):
                best_score = parent_score
                best_seq = parent.copy()
    except Exception:
        pass
    
    if parent is None:
        parent = np.ones(384) / 384
        print("No parent found, using uniform n=384")
        best_score = evaluate_sequence(parent.tolist())
        best_seq = parent.copy()
    
    n = len(parent)
    print(f"Starting: n={n}, best_score={best_score:.10f}")
    
    # === Phase 1: Adam optimization on parent (primary) ===
    print(f"\n=== Phase 1: Adam Opt (n={n}) ===")
    t1 = min(500, deadline - time.monotonic() - 500)
    if t1 > 60:
        print(f"  Adam on parent, alloc={t1:.0f}s")
        a_opt, _ = adam_optimize(parent, time_limit=t1, lr=0.05,
                                  top_k_start=1, top_k_end=n // 3,
                                  restart_every=4000, verbose=True, label="adam1")
        try_seq(a_opt, "adam1")
    
    # === Phase 2: EG optimization on best ===
    print(f"\n=== Phase 2: EG Opt ===")
    print(f"  Best so far: {best_score:.10f}")
    
    if best_seq is not None:
        configs = [
            {"eta0": 1.5, "temp0": 0.003, "temp_decay": 0.9996, "eta_decay": 0.99995, "top_k_frac": 0.3},
            {"eta0": 0.5, "temp0": 0.01, "temp_decay": 0.9998, "eta_decay": 0.9999, "top_k_frac": 0.5},
            {"eta0": 2.0, "temp0": 0.002, "temp_decay": 0.9995, "eta_decay": 0.9999, "top_k_frac": 0.15},
        ]
        for ci, cfg in enumerate(configs):
            rem = deadline - time.monotonic() - 300
            if rem < 30:
                break
            t_alloc = min(100, rem / max(1, len(configs) - ci))
            a_opt, _ = eg_optimize(best_seq, time_limit=t_alloc, **cfg, label=f"eg{ci}")
            try_seq(a_opt, f"eg{ci}")
    
    # === Phase 3: Adam with different hyperparameters ===
    print(f"\n=== Phase 3: Adam Variations ===")
    print(f"  Best so far: {best_score:.10f}")
    
    if best_seq is not None:
        adam_configs = [
            {"lr": 0.02, "top_k_start": 1, "top_k_end": n // 8},
            {"lr": 0.1, "top_k_start": 5, "top_k_end": n // 2},
            {"lr": 0.05, "top_k_start": 1, "top_k_end": n // 4},
        ]
        for ai, cfg in enumerate(adam_configs):
            rem = deadline - time.monotonic() - 250
            if rem < 30:
                break
            t_alloc = min(80, rem / max(1, len(adam_configs) - ai))
            a_opt, _ = adam_optimize(best_seq, time_limit=t_alloc,
                                      restart_every=3000, label=f"adam{ai+2}", **cfg)
            try_seq(a_opt, f"adam{ai+2}")
    
    # === Phase 4: Multi-initialization exploration ===
    print(f"\n=== Phase 4: Multi-Init ===")
    print(f"  Best so far: {best_score:.10f}")
    
    if best_seq is not None:
        inits = generate_diverse_inits(rng, n, best_seq)
        n_tried = 0
        for i, f_init in enumerate(inits):
            rem = deadline - time.monotonic() - 200
            if rem < 20:
                break
            t_alloc = min(40, rem / max(1, len(inits) - i))
            if t_alloc < 10:
                break
            n_tried += 1
            a_opt, _ = adam_optimize(f_init, time_limit=t_alloc, lr=0.03,
                                      top_k_start=1, top_k_end=n // 4,
                                      restart_every=2000, label=f"init{i}")
            try_seq(a_opt, f"init{i}")
            if n_tried >= 8:
                break
    
    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-Length ===")
    print(f"  Best so far: {best_score:.10f}")
    
    if best_seq is not None:
        alt_lengths = [383, 389, 448, 512, 640]
        for n_alt in alt_lengths:
            rem = deadline - time.monotonic() - 100
            if rem < 30:
                break
            t_alloc = min(60, rem / max(1, len(alt_lengths)))
            if t_alloc < 15:
                break
            a_resampled = resample_sequence(best_seq, n_alt)
            a_opt, _ = adam_optimize(a_resampled, time_limit=t_alloc, lr=0.04,
                                      top_k_start=1, top_k_end=n_alt // 4,
                                      restart_every=2000, label=f"len{n_alt}")
            try_seq(a_opt, f"len{n_alt}")
    
    # === Phase 6: Coordinate descent polish ===
    print(f"\n=== Phase 6: Coord Descent ===")
    print(f"  Best so far: {best_score:.10f}")
    
    if best_seq is not None:
        rem = deadline - time.monotonic() - 50
        if rem > 20:
            a_opt, _ = coordinate_descent(best_seq, time_limit=min(rem, 80), n_rounds=150)
            try_seq(a_opt, "coord")
    
    # === Phase 7: Final fine Adam ===
    print(f"\n=== Phase 7: Final Polish ===")
    print(f"  Best so far: {best_score:.10f}")
    
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 20:
            a_opt, _ = adam_optimize(best_seq, time_limit=min(rem, 60),
                                      lr=0.01, top_k_start=1, top_k_end=max(5, len(best_seq) // 8),
                                      restart_every=3000, label="final")
            try_seq(a_opt, "final_adam")
        
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            a_opt, _ = coordinate_descent(best_seq, time_limit=min(rem, 30), n_rounds=50)
            try_seq(a_opt, "final_coord")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
