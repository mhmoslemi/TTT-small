"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using projected gradient descent with
Nesterov momentum, adaptive soft-max temperature, FFT-accelerated gradients,
peak-shaving corrections, and multi-length exploration. Key improvements over
previous approaches: (1) O(n log n) gradient via FFT enables 100x more iterations,
(2) full-spectrum soft-max weighting instead of top-k, (3) explicit peak-shaving
steps using linearized quadratic structure, (4) wider length exploration.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient_fft(a, w):
    """
    Compute grad[j] = 2 * sum_k w[k] * a[k-j] (valid range) using FFT.
    w has length 2n-1, a has length n. Returns vector of length n.
    """
    n = len(a)
    a_flipped = a[::-1]
    # We need conv(w, a_flipped)[n-1+j] for j=0..n-1
    # conv length = (2n-1) + n - 1 = 3n-2
    size = 1 << (3 * n - 1).bit_length()
    W = np.fft.rfft(w, size)
    Af = np.fft.rfft(a_flipped, size)
    conv = np.fft.irfft(W * Af, size)
    return 2.0 * conv[n - 1:n - 1 + n]


def compute_gradient_direct(a, w):
    """Direct gradient computation (for verification or small n)."""
    n = len(a)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * wk * a[k - idx]
    return grad


def obj_value(f):
    """Compute 2*n*max(auto_conv) for sum(f)=1."""
    n = len(f)
    return 2.0 * n * fast_convolve(f).max()


def peak_shave_step(f, c, k_star, epsilon):
    """
    Compute a step direction that reduces c[k_star] by approximately epsilon.
    Returns (f_new, success).
    """
    n = len(f)
    # Partner vector: p[i] = a[k_star - i] for valid i
    # c[k_star] = sum_i f[i] * f[k_star - i]
    # d(c[k_star])/d(f[i]) = 2 * f[k_star - i]
    # To reduce: step in direction -f[k_star - i], projected to maintain sum=1
    
    # Partner vector (only valid entries)
    p = np.zeros(n)
    for i in range(n):
        k_minus_i = k_star - i
        if 0 <= k_minus_i < n:
            p[i] = f[k_minus_i]
    
    # Center the direction (to maintain sum)
    p_centered = p - p.mean()
    p_norm_sq = np.dot(p_centered, p_centered)
    if p_norm_sq < 1e-15:
        return f, False
    
    # Step size: reduce c[k_star] by epsilon
    # Delta c ≈ 2 * sum_i f[i] * d[k_star-i] ... 
    # Actually: d(c[k_star])/d(f[i]) = 2*p[i], so step -alpha*p_centered gives
    # Delta c ≈ -2*alpha * sum_i p[i]*p_centered[i] = -2*alpha * (sum p^2 - n*mean(p)^2)
    # = -2*alpha * p_norm_sq (since p_norm_sq = ||p - mean||^2)
    alpha = epsilon / (2.0 * p_norm_sq)
    
    # Ensure non-negativity
    max_neg = np.min(f - alpha * p_centered)
    if max_neg < 0:
        # Scale down
        neg_idx = (f - alpha * p_centered) < 0
        if neg_idx.any():
            ratio = np.min(f[neg_idx] / np.maximum(p_centered[neg_idx], 1e-15))
            alpha = alpha * ratio * 0.9
    
    f_new = f - alpha * p_centered
    f_new = project_simplex(f_new)
    return f_new, True


def nesterov_pgd(f0, time_limit, tau0=30.0, tau_growth=1.0008,
                 lr0=0.008, lr_decay=0.99998, mom0=0.92,
                 shave_interval=200, shave_eps=0.0001,
                 verbose=False):
    """
    Nesterov accelerated projected gradient descent with periodic peak shaving.
    Optimizes max(auto_conv(f)) subject to sum(f)=1, f>=0.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    c = fast_convolve(f)
    best_peak = c.max()
    best_f = f.copy()
    vel = np.zeros(n)
    
    t0 = time.monotonic()
    lr = lr0
    tau = tau0
    mu = mom0
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Nesterov lookahead
        f_y = f - mu * vel
        f_y = np.maximum(f_y, 0.0)
        sy = f_y.sum()
        if sy > 1e-15:
            f_y = f_y / sy
        else:
            f_y = f
        
        # Compute soft-max weights over all lags
        c = fast_convolve(f_y)
        scaled = tau * c
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws > 1e-15:
            w /= ws
        else:
            w = np.zeros(len(c))
            w[np.argmax(c)] = 1.0
        
        # Gradient via FFT
        grad = compute_gradient_fft(f_y, w)
        
        # Update
        vel = mu * vel + grad
        f_new = f - lr * vel
        f_new = project_simplex(f_new)
        
        # Periodic peak shaving
        if it % shave_interval == 0:
            c_new = fast_convolve(f_new)
            k_star = int(np.argmax(c_new))
            f_shaved, _ = peak_shave_step(f_new, c_new, k_star, shave_eps)
            obj_shave = fast_convolve(f_shaved).max()
            obj_new = c_new.max()
            if obj_shave < obj_new:
                f_new = f_shaved
        
        peak = fast_convolve(f_new).max()
        
        if peak < best_peak - 1e-14:
            best_peak = peak
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        f = f_new
        lr *= lr_decay
        tau *= tau_growth
        
        # Adaptive adjustments when stuck
        if no_improve > 50:
            mu = max(0.7, mu * 0.995)
            tau *= 1.005
            lr = max(lr0 * 0.4, 0.0003)
            no_improve = 0
        
        if no_improve > 500:
            # Larger restart
            rng = np.random.default_rng(it * 7 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            vel = np.zeros(n)
            mu = mom0
            lr = lr0 * 0.5
            no_improve = 0
        
        if verbose and it % 500 == 0:
            print(f"    PGD it={it}, peak={best_peak:.8f}, obj={2*n*best_peak:.8f}")
    
    return best_f, 2.0 * n * best_peak


def multi_peak_pgd(f0, time_limit, tau_schedule=None, lr0=0.006,
                   mom0=0.9, periodic_shave=True, verbose=False):
    """
    PGD with periodic emphasis on different peak subsets.
    Cycles through: all peaks, top-50, top-20, top-5, then repeats.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    best_peak = fast_convolve(f).max()
    best_f = f.copy()
    vel = np.zeros(n)
    
    t0 = time.monotonic()
    lr = lr0
    mu = mom0
    no_improve = 0
    it = 0
    
    # Peak focus cycle: (tau, top_k) pairs
    focus_cycle = [
        (20.0, None),    # all peaks, low temp
        (50.0, None),    # all peaks, medium temp
        (100.0, 50),     # top 50, high temp
        (200.0, 20),     # top 20, very high temp
        (500.0, 5),      # top 5, extremely focused
        (30.0, None),    # back to broad
    ]
    focus_idx = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Determine current focus
        cur_tau, cur_topk = focus_cycle[focus_idx % len(focus_cycle)]
        
        # Switch focus every 200 iterations
        if it % 200 == 0:
            focus_idx += 1
        
        # Compute weights
        c = fast_convolve(f)
        if cur_topk is not None and cur_topk < len(c):
            peaks = np.argpartition(c, -cur_topk)[-cur_topk:]
            w_full = np.zeros(len(c))
            pv = c[peaks]
            scaled = cur_tau * pv - cur_tau * pv.max()
            wp = np.exp(scaled)
            wp /= wp.sum()
            w_full[peaks] = wp
        else:
            scaled = cur_tau * c
            scaled -= scaled.max()
            w_full = np.exp(scaled)
            ws = w_full.sum()
            if ws > 1e-15:
                w_full /= ws
            else:
                w_full = np.zeros(len(c))
                w_full[np.argmax(c)] = 1.0
        
        # Gradient
        grad = compute_gradient_fft(f, w_full)
        
        # Momentum update
        vel = mu * vel + grad
        f_new = f - lr * vel
        f_new = project_simplex(f_new)
        
        # Peak shaving
        if periodic_shave and it % 100 == 0:
            c_new = fast_convolve(f_new)
            k_star = int(np.argmax(c_new))
            f_sh, _ = peak_shave_step(f_new, c_new, k_star, 0.00005)
            if fast_convolve(f_sh).max() < c_new.max():
                f_new = f_sh
        
        peak = fast_convolve(f_new).max()
        
        if peak < best_peak - 1e-14:
            best_peak = peak
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        f = f_new
        lr *= 0.99999
        mu = min(0.95, mu * 1.0001)  # slowly increase momentum
        
        if no_improve > 300:
            rng = np.random.default_rng(it * 13 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            vel = np.zeros(n)
            lr = lr0 * 0.6
            no_improve = 0
            focus_idx = 0
        
        if verbose and it % 1000 == 0:
            print(f"    MultiPGD it={it}, peak={best_peak:.8f}, obj={2*n*best_peak:.8f}")
    
    return best_f, 2.0 * n * best_peak


def coordinate_descent_polish(f0, time_limit, verbose=False):
    """
    Cyclic coordinate descent: optimize pairs of coordinates (i, n-1-i) 
    to reduce the center peak, then single coordinates for off-center peaks.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    best_peak = fast_convolve(f).max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    pass_count = 0
    
    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        c = fast_convolve(f)
        k_star = int(np.argmax(c))
        
        # For the peak at k_star, try to reduce it by adjusting pairs
        # c[k_star] = sum_i f[i] * f[k_star - i]
        # For each i, the "partner" is f[k_star - i]
        # We want to make products f[i]*f[k_star-i] more uniform
        
        improved = False
        for i in range(n // 2):
            j = k_star - i
            if j < 0 or j >= n or j == i:
                continue
            # Current product: f[i] * f[j]
            # Sum: f[i] + f[j] = s_ij (fixed)
            # Product is maximized when f[i]=f[j]=s_ij/2
            # To reduce product, make them more unequal
            s_ij = f[i] + f[j]
            if s_ij < 1e-15:
                continue
            # Move toward making one larger, other smaller
            # But we need to be careful not to hurt other peaks
            cur_prod = f[i] * f[j]
            # Try a small asymmetric move
            delta = 0.01 * s_ij
            if f[i] >= f[j]:
                f_new_i = f[i] + delta
                f_new_j = max(0, f[j] - delta)
            else:
                f_new_i = max(0, f[i] - delta)
                f_new_j = f[j] + delta
            
            f[i] = f_new_i
            f[j] = f_new_j
        
        # Renormalize
        fs = f.sum()
        if fs > 1e-15:
            f = f / fs
        
        peak = fast_convolve(f).max()
        if peak < best_peak - 1e-14:
            best_peak = peak
            best_f = f.copy()
            improved = True
        
        if not improved and pass_count > 20:
            # Try the reverse: equalize pairs (for off-center peaks)
            f = best_f.copy()
        
        if pass_count % 500 == 0 and verbose:
            print(f"    CD pass={pass_count}, peak={best_peak:.8f}")
    
    return best_f, 2.0 * n * best_peak


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    inits.append(np.ones(n) / n)
    
    # Various structured inits
    x = np.arange(n, dtype=np.float64)
    center = 0.5 * (n - 1)
    
    for sigma_frac in [0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
        sigma = sigma_frac * n
        g = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        inits.append(g / g.sum())
    
    # Cosine window
    for freq in [1, 2, 3]:
        c = np.abs(np.cos(np.pi * freq * x / n)) + 0.01
        inits.append(c / c.sum())
    
    # Two-peak (bipolar)
    for sep_frac in [0.2, 0.3, 0.4]:
        sep = sep_frac * n
        g1 = np.exp(-0.5 * ((x - center + sep) / (0.1 * n)) ** 2)
        g2 = np.exp(-0.5 * ((x - center - sep) / (0.1 * n)) ** 2)
        two = g1 + g2
        inits.append(two / two.sum())
    
    # U-shape (edge concentrated)
    edge = 1.0 / (0.5 * np.abs(x - center) / n + 0.05)
    inits.append(edge / edge.sum())
    
    # Random
    for _ in range(3):
        r = rng.dirichlet(np.ones(n) * 0.3)
        inits.append(r)
    
    # Interpolated parent
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    return inits

            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * noise_scale)
            a_opt, _ = nesterov_pgd(f_pert, time_limit=t_each * 0.6,
                                    tau0=100.0, tau_growth=1.001,
                                    lr0=0.003, lr_decay=0.9999, mom0=0.88)
            try_eval(a_opt, f"restart_{attempt}")
            rem = deadline - time.monotonic() - 25
    
    # === Phase 6: Try different lengths ===
    print("\n=== Phase 6: Length Exploration ===")
    rem = deadline - time.monotonic() - 20
    if rem > 20 and best_seq is not None:
        for n_try in [n_best + 32, n_best + 64, n_best - 32]:
            if n_try < 64 or rem < 15:
                continue
            px = np.interp(np.linspace(0, 1, n_try),
                          np.linspace(0, 1, n_best), best_seq)
            px = np.maximum(px, 0)
            ps = px.sum()
            if ps > 1e-10:
                px = px / ps
                t_opt = min(rem * 0.3, 40)
                a_opt, _ = nesterov_pgd(px, time_limit=t_opt,
                                        tau0=50.0, tau_growth=1.001,
                                        lr0=0.008, lr_decay=0.99995, mom0=0.92)
                try_eval(a_opt, f"len{n_try}")
                rem = deadline - time.monotonic() - 15
    
    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.10f}")
    
    return [float(x) for x in best_seq]
