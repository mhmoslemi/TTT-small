import numpy as np

from scipy.optimize import linprog
from scipy.linalg import toeplitz


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_convolve(a, g):
    """Cross-convolution a*g via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    G = np.fft.rfft(g, fft_size)
    return np.fft.irfft(A * G, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C (2n-1 x n) such that C @ g = a * g (cross-convolution)."""
    n = len(a)
    first_col = np.concatenate([a, np.zeros(n - 1)])
    first_row = np.array([a[0]] + [0.0] * (n - 1))
    return toeplitz(first_col, first_row)


def compute_obj(a):
    """Compute objective 2n*max(conv) for normalized a (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv


def solve_lp(a, M):
    """Solve LP: max sum(g) s.t. a*g <= M, g >= 0. Returns normalized g."""
    n = len(a)
    C = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        g = res.x
        gs = g.sum()
        if gs > 1e-12:
            dual = None
            try:
                dual = res.ineqlin.marginals
            except Exception:
                pass
            return g / gs, gs, dual
    return a.copy(), 1.0, None


def line_search(a, d, best_obj, n, max_steps=12):

def subgradient_polish(a0, time_limit, lr_init=0.005, momentum=0.95,
                       top_k=80, temp=0.00015):
    """Subgradient descent with momentum for fine-tuning."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    _, conv = compute_obj(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    step_count = 0

    while time.monotonic() - t0 < time_limit:
        # Compute subgradient at top peaks
        top_k_eff = min(top_k, len(conv))
        peaks = np.argpartition(conv, -top_k_eff)[-top_k_eff:]
        peak_vals = conv[peaks]
        scaled = peak_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum

        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            j_idx = np.arange(j_start, j_end + 1)
            sg[j_idx] += w[idx] * 2.0 * a[k - j_idx]

        v = momentum * v - lr * sg
        a = np.maximum(a + v, 0.0)
        s_a = a.sum()
        if s_a > 1e-15:
            a = a / s_a
        _, conv = compute_obj(a)
        cur_obj = 2.0 * n * conv.max()
        step_count += 1

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve % 40 == 0 and no_improve > 0:
            lr *= 0.999
            temp *= 1.003

        if no_improve >= 350:
            rng = np.random.default_rng(step_count * 5 + 13)
            noise = rng.standard_normal(n) * 0.003
            a = best_a + noise
            a = np.maximum(a, 0.0)
            s_a = a.sum()
            if s_a > 1e-15:
                a = a / s_a
            v = np.zeros(n)
            _, conv = compute_obj(a)
            no_improve = 0

    return best_a, best_obj


def coordinate_descent(a0, time_limit, coords_per_step=5):
    """Coordinate descent: optimize a few coordinates at a time."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)

    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        # Pick random coordinates to optimize
        idx = rng.choice(n, size=min(coords_per_step, n), replace=False)
        improved = False
        for j in idx:
            # Try a range of values for a[j]
            base = a.copy()
            base_val = base[j]
            if base_val < 1e-10:
                candidates = [0.001, 0.002, 0.005, 0.01] / n
            else:
                candidates = [base_val * f for f in [0.5, 0.7, 1.3, 1.5, 0.8, 1.2]]

            best_t = base_val
            best_local = best_obj
            for t in candidates:
                base[j] = t
                s_base = base.sum()
                if s_base < 1e-15:
                    continue
                base_norm = base / s_base
                obj_try = compute_obj(base_norm)[0]
                if obj_try < best_local:
                    best_local = obj_try
                    best_t = t
            base[j] = best_t
            s_base = base.sum()
            if s_base > 1e-15:
                a = base / s_base
                obj_new = compute_obj(a)[0]
                if obj_new < best_obj - 1e-15:
                    best_obj = obj_new
                    best_a = a.copy()
                    improved = True
        # Recompute conv periodically
        _, conv = compute_obj(a)

    return best_a, best_obj


def resample_sequence(seq, new_n):
    """Resample to different length via linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s > 1e-15:
        new_seq = new_seq / s
    return new_seq


def get_inits(n, parent_seq, rng_local):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    inits['uniform'] = np.ones(n) / n

    if parent_seq is not None:
        inits['parent'] = resample_sequence(parent_seq, n)

    # Various structured inits
    inits['sine'] = (0.7 + 0.3 * np.sin(2 * np.pi * x / n))
    inits['sine'] /= inits['sine'].sum()

    lobe = 0.6 + 0.4 * np.cos(4 * np.pi * x / n)
    inits['tlobe'] = lobe / lobe.sum()

    lobe4 = 0.7 + 0.3 * np.cos(8 * np.pi * x / n)
    inits['flobe'] = lobe4 / lobe4.sum()

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.25 * n)) ** 2)
    inits['gauss'] = g / g.sum()

    # Two-bump
    b1 = np.exp(-0.5 * ((x - 0.25 * n) / (0.15 * n)) ** 2)
    b2 = np.exp(-0.5 * ((x - 0.75 * n) / (0.15 * n)) ** 2)
    two_bump = b1 + b2
    inits['twobump'] = two_bump / two_bump.sum()

    # Random
    inits['random'] = rng_local.dirichlet(np.ones(n) * 0.5)

    # Flat with edges suppressed
    edge = np.minimum(x, n - 1 - x)
    flat_edge = np.ones(n) * 0.5 + edge / n
    inits['flat_edge'] = flat_edge / flat_edge.sum()

    return inits

    return [float(max(0.0, min(1000.0, x))) for x in best_sequence]
