        sg[m_lo:m_hi + 1] += 2.0 * weights[i] * a[idx_lo:idx_hi + 1][::-1]

    return sg


def gs_preshape(a, n_iter=50):
    """Gerchberg-Saxton spectral pre-shaping to flatten the auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    a_curr = a.copy()

    for _ in range(n_iter):
        A = np.fft.rfft(a_curr, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)

        # Target: constant magnitude for non-DC bins
        # DC bin (0) corresponds to sum(a) = 1
        # For non-DC bins, target magnitude based on desired flat convolution
        # Flat conv value = 1/(2n-1), so target |A|^2 total = fft_size/(2n-1)
        # DC contributes 1, so non-DC total = fft_size/(2n-1) - 1
        non_dc_count = len(mag) - 1
        target_total = fft_size / (2 * n - 1)
        non_dc_energy = max(target_total - 1.0, 0.01)
        target_mag = np.sqrt(non_dc_energy / max(non_dc_count, 1))

        new_mag = np.full_like(mag, target_mag)
        new_mag[0] = 1.0

        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, fft_size)[:n]
        a_curr = project_simplex(a_new)

    return a_curr


def subgradient_optimize(a0, time_limit, top_k=5, temp_init=0.002,
                         temp_max=0.01, lr_init=0.008, lr_min=0.0002,
                         momentum=0.93, verbose=False):
    """Adaptive subgradient descent on the simplex."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    v = np.zeros(n)
    lr = lr_init
    temp = temp_init
    no_improve = 0
    it = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        it += 1

        conv = fast_convolve(a)
        cur_obj = conv.max()

        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Compute subgradient
        sg = compute_subgrad_fast(a, conv, top_k, temp)
        sg_norm = np.sum(sg * a)  # Directional derivative on simplex

        # Momentum update
        v = momentum * v - lr * sg
        a_new = project_simplex(a + v)

        # Symmetry: try averaging with reversal
        if it % 50 == 0:
            a_sym = 0.5 * (a_new + a_new[::-1])
            a_sym = project_simplex(a_sym)
            conv_sym = fast_convolve(a_sym)
            if conv_sym.max() < fast_convolve(a_new).max():
                a_new = a_sym

        a = a_new
        lr = max(lr * 0.9997, lr_min)
        temp = min(temp * 1.003, temp_max)
        momentum = min(0.97, momentum + 0.0001)

        # Restart if stuck
        if no_improve > 200:
            scale = 0.003 + 0.002 * min(no_improve / 100, 3.0)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            v = np.zeros(n)
            lr = max(lr_init * 0.5, lr_min * 3)
            temp = temp_init
            no_improve = 0

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SG iter {it}: obj={best_obj:.10f}, lr={lr:.6f}, "
                  f"temp={temp:.6f}, elapsed={elapsed:.1f}s")

    return best_a, best_obj


def local_perturbation_search(a0, time_limit, rng, base_noise=0.002):
    """Fine-grained local search with adaptive noise scale."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    noise_scale = base_noise
    improvements = 0
    total_tries = 0

    while time.monotonic() - t0 < time_limit:
        # Batch of perturbations
        batch_size = 20
        for _ in range(batch_size):
            if time.monotonic() - t0 > time_limit:
                break
            total_tries += 1
            noise = rng.standard_normal(n) * noise_scale
            a_pert = project_simplex(best_a + noise)
            conv_pert = fast_convolve(a_pert)
            obj_pert = conv_pert.max()
            if obj_pert < best_obj - 1e-15:
                best_obj = obj_pert
                best_a = a_pert.copy()
                improvements += 1
                noise_scale *= 1.02  # Slightly increase to continue exploring

        # Decay noise scale
        noise_scale *= 0.9
        if noise_scale < 1e-5:
            # Reset with a jump
            noise = rng.standard_normal(n) * 0.01
            a_jump = project_simplex(best_a + noise)
            conv_jump = fast_convolve(a_jump)
            if conv_jump.max() < best_obj:
                best_obj = conv_jump.max()
                best_a = a_jump.copy()
            noise_scale = base_noise * 0.5

        if time.monotonic() - t0 % 10 < 0.1:
            print(f"      Local search: obj={best_obj:.10f}, scale={noise_scale:.6f}, "
                  f"improvements={improvements}/{total_tries}")

    return best_a, best_obj


def spectral_refinement(a0, time_limit, n_gs=30, n_sg=200):
    """Combine GS pre-shaping with subgradient for refinement."""
    n = len(a0)
    t0 = time.monotonic()

    # GS pre-shape
    remaining = time_limit - (time.monotonic() - t0)
    gs_budget = remaining * 0.3
    if gs_budget > 1:
        a_gs = gs_preshape(a0, n_iter=n_gs)
        conv_gs = fast_convolve(a_gs)
        obj_gs = conv_gs.max()
    else:
        a_gs = a0.copy()
        obj_gs = fast_convolve(a_gs).max()

    # Subgradient from both starting points
    sg_budget = remaining * 0.6
    if sg_budget > 2:
        a_opt, obj_opt = subgradient_optimize(
            a_gs, time_limit=sg_budget, top_k=3,
            temp_init=0.001, lr_init=0.005)

        # Also try from original
        rem = time_limit - (time.monotonic() - t0)
        if rem > 2:
            a_opt2, obj_opt2 = subgradient_optimize(
                a0, time_limit=rem, top_k=3,
                temp_init=0.001, lr_init=0.005)
            if obj_opt2 < obj_opt:
                a_opt, obj_opt = a_opt2, obj_opt2
    else:
        a_opt, obj_opt = a_gs, obj_gs

    return a_opt, obj_opt


def generate_initialization(n, rng, parent=None):
    """Generate a diverse initialization of length n."""
    x = np.arange(n, dtype=np.float64)

    options = []
    options.append(np.ones(n) / n)

    # Gaussian bumps at different positions
    for center_frac in [0.3, 0.5, 0.7]:
        sigma = 0.2 * n
        g = np.exp(-0.5 * ((x - center_frac * (n - 1)) / sigma) ** 2)
        options.append(g / g.sum())

    # Cosine window
    c = 0.5 * (1 + np.cos(2 * np.pi * x / (2 * n))) + 0.01
    options.append(c / c.sum())

    # Beta-like
    b = (x + 1) ** 1.5 * (n - x) ** 1.5
    options.append(b / b.sum())

    # Random Dirichlet
    options.append(rng.dirichlet(np.ones(n) * 0.3))

    # Parent interpolation
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            options.append(px / ps)

    return options


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Multi-phase approach:
    Phase 1: Quick exploration of multiple initializations and lengths
    Phase 2: Intensive subgradient optimization on best candidates
    Phase 3: Spectral refinement (GS + SG)
    Phase 4: Local perturbation search
    Phase 5: Final fine-tuning
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(10.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a_norm)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.10f}")
    except Exception:
        pass

    # === Phase 1: Quick exploration ===
    print("=== Phase 1: Quick Exploration ===")
    t_phase1_end = time.monotonic() + min(120, (deadline - time.monotonic()) * 0.1)

    lengths_to_try = [384]
    if parent is not None:
        lengths_to_try.insert(0, len(parent))
    # Add some other lengths
    for n_try in [320, 448, 512, 640]:
        if n_try not in lengths_to_try:
            lengths_to_try.append(n_try)

    candidates = []  # (sequence, label)

    for n in lengths_to_try:
        if time.monotonic() > t_phase1_end:
            break
        inits = generate_initialization(n, rng, parent)
        for idx, ini in enumerate(inits):
            if time.monotonic() > t_phase1_end:
                break
            label = f"n{n}_init{idx}"
            score = try_seq(ini, label)
            if score < best_score:
                candidates.append((ini.copy(), label))

    # Quick GS on a few candidates
    print("  Running quick GS pre-shaping...")
    for cand, label in candidates[:6]:
        if time.monotonic() > t_phase1_end - 10:
            break
        n = len(cand)
        a_gs = gs_preshape(cand, n_iter=30)
        try_seq(a_gs, f"gs_{label}")

    print(f"  Phase 1 best: {best_score:.10f} (n={len(best_seq) if best_seq is not None else 0})")

    # === Phase 2: Intensive subgradient optimization ===
    print("\n=== Phase 2: Intensive Subgradient ===")
    n_best = len(best_seq)

    # Run subgradient with different parameters
    sg_configs = [
        {'top_k': 3, 'temp_init': 0.001, 'lr_init': 0.008, 'momentum': 0.93},
        {'top_k': 5, 'temp_init': 0.005, 'lr_init': 0.005, 'momentum': 0.95},
        {'top_k': 10, 'temp_init': 0.01, 'lr_init': 0.004, 'momentum': 0.96},
        {'top_k': 2, 'temp_init': 0.0005, 'lr_init': 0.01, 'momentum': 0.92},
    ]

    remaining = deadline - time.monotonic() - 100
    time_per_config = remaining / max(len(sg_configs), 1)

    for i, cfg in enumerate(sg_configs):
        if time.monotonic() > deadline - 200:
            break
        alloc = min(time_per_config, 400)
        if alloc < 10:
            break
        print(f"  SG config {i} (top_k={cfg['top_k']}, lr={cfg['lr_init']}), "
              f"budget={alloc:.0f}s")
        a_opt, obj = subgradient_optimize(
            best_seq, time_limit=alloc, **cfg)
        try_seq(a_opt, f"sg_cfg{i}")

        # Also try from a perturbed version
        rem = deadline - time.monotonic() - 200
        if rem > 15:
            noise = rng.standard_normal(n_best) * 0.01
            a_pert = project_simplex(best_seq + noise)
            a_opt2, obj2 = subgradient_optimize(
                a_pert, time_limit=min(rem * 0.5, alloc * 0.4), **cfg)
            try_seq(a_opt2, f"sg_pert{i}")

    print(f"  Phase 2 best: {best_score:.10f}")

    # === Phase 3: Spectral refinement ===
    print("\n=== Phase 3: Spectral Refinement ===")
    remaining = deadline - time.monotonic() - 80
    if remaining > 30:
        alloc_sr = min(remaining * 0.4, 300)
        print(f"  Spectral refinement budget: {alloc_sr:.0f}s")
        a_opt, obj = spectral_refinement(best_seq, time_limit=alloc_sr)
        try_seq(a_opt, "spectral")

        # Try on a perturbed version
        rem = deadline - time.monotonic() - 80
        if rem > 20:
            noise = rng.standard_normal(n_best) * 0.005
            a_pert = project_simplex(best_seq + noise)
            a_opt2, obj2 = spectral_refinement(a_pert, time_limit=min(rem * 0.6, 150))
            try_seq(a_opt2, "spectral_pert")

    print(f"  Phase 3 best: {best_score:.10f}")

    # === Phase 4: Local perturbation search ===
    print("\n=== Phase 4: Local Perturbation ===")
    remaining = deadline - time.monotonic() - 50
    if remaining > 20:
        alloc_ls = min(remaining * 0.5, 250)
        print(f"  Local search budget: {alloc_ls:.0f}s")
        a_opt, obj = local_perturbation_search(
            best_seq, time_limit=alloc_ls, rng=rng, base_noise=0.003)
        try_seq(a_opt, "local")

        # Finer local search
        rem = deadline - time.monotonic() - 50
        if rem > 15:
            a_opt2, obj2 = local_perturbation_search(
                best_seq, time_limit=min(rem * 0.6, 100), rng=rng, base_noise=0.001)
            try_seq(a_opt2, "local_fine")

    print(f"  Phase 4 best: {best_score:.10f}")

    # === Phase 5: Final fine-tuning ===
    print("\n=== Phase 5: Final Fine-Tuning ===")
    remaining = deadline - time.monotonic() - 20
    if remaining > 10:
        # Very fine subgradient
        alloc_final = remaining * 0.6
        print(f"  Final SG budget: {alloc_final:.0f}s")
        a_opt, obj = subgradient_optimize(
            best_seq, time_limit=alloc_final, top_k=2,
            temp_init=0.0003, lr_init=0.002, momentum=0.97)
        try_seq(a_opt, "final_sg")

        # One more local search
        rem = deadline - time.monotonic() - 15
        if rem > 5:
            a_opt2, obj2 = local_perturbation_search(
                best_seq, time_limit=min(rem, 30), rng=rng, base_noise=0.0005)
            try_seq(a_opt2, "final_local")

    # === Try different lengths for the best shape ===
    print("\n=== Extra: Length exploration ===")
    remaining = deadline - time.monotonic() - 10
    if remaining > 20 and best_seq is not None:
        for n_try in [n_best - 32, n_best + 32, n_best + 64]:
            if n_try < 64 or n_try > 1024 or remaining < 10:
                continue
            px = np.interp(np.linspace(0, 1, n_try),
                           np.linspace(0, 1, n_best), best_seq)
            px = np.maximum(px, 0)
            ps = px.sum()
            if ps > 1e-10:
                px = px / ps
                # Quick optimize
                alloc = min(remaining * 0.4, 60)
                a_opt, obj = subgradient_optimize(
                    px, time_limit=alloc, top_k=3,
                    temp_init=0.002, lr_init=0.005)
                try_seq(a_opt, f"len{n_try}")
                remaining = deadline - time.monotonic() - 10

    # Final verification
    if best_seq is not None:
        final_score = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.10f}")

    return [float(x) for x in best_seq]
