        j_end = min(n - 1, k)
        idxs = np.arange(j_start, j_end + 1)
        grad[idxs] += w[idx] * a[k - idxs]
    return grad


def lp_solve(a, M, C=None):
    """Solve LP: max sum(g) s.t. C@g <= M, g >= 0."""
    n = len(a)
    if C is None:
        C = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        result = linprog(c, A_ub=C, b_ub=b, bounds=bounds, method='highs',
                         options={'presolve': True, 'dual_feasibility_tolerance': 1e-9,
                                  'primal_feasibility_tolerance': 1e-9})
        if result.success:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g, g_sum, True
    except Exception:
        pass
    return None, 1.0, False


def bisection_line_search(a, d, M_target, n, max_iters=40):
    """
    Find maximum t in [0,1] such that after normalization,
    max(conv(a_new, a_new)) <= M_target * (sum(a_new))^2 / (sum(a))^2.
    We work in normalized space: a has sum=1, so we need max(conv(normalized)) <= M_target.
    """
    # Check t=0
    # Check t=1
    a_try = a + d
    a_try = np.maximum(a_try, 0.0)
    s_try = a_try.sum()
    if s_try < 1e-15:
        return 0.0
    a_try = a_try / s_try
    conv_try = fast_convolve(a_try)
    if conv_try.max() <= M_target:
        return 1.0

    # Bisection
    t_lo = 0.0
    t_hi = 1.0
    for _ in range(max_iters):
        t_mid = 0.5 * (t_lo + t_hi)
        a_try = a + t_mid * d
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            t_hi = t_mid
            continue
        a_try = a_try / s_try
        conv_try = fast_convolve(a_try)
        if conv_try.max() <= M_target:
            t_lo = t_mid
        else:
            t_hi = t_mid
        if t_hi - t_lo < 1e-8:
            break
    return t_lo


def lp_step_improved(a, conv, n, C, M_factor=1.0, use_bisect=True):
    """One LP step with bisection line search."""
    M = conv.max()

    g, g_sum, success = lp_solve(a, M * M_factor, C)
    if not success or g is None or g_sum <= 1.0 + 1e-8:
        return a.copy(), False, 0.0

    d = g - a
    if np.linalg.norm(d) < 1e-14:
        return a.copy(), False, 0.0

    if use_bisect:
        t = bisection_line_search(a, d, M * M_factor, n)
    else:
        t = 0.2

    if t < 1e-10:
        return a.copy(), False, 0.0

    a_new = a + t * d
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy(), False, 0.0
    a_new = a_new / s
    return a_new, True, t


def nesterov_gradient_optimize(a0, time_limit, lr_init=0.01, momentum=0.9, top_k=30, temp_init=0.003):
    """Nesterov-accelerated gradient descent with adaptive parameters."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    y = a.copy()
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    mom = momentum
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        y = a + mom * v
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y

        conv_y = fast_convolve(y)
        grad = compute_soft_gradient(y, conv_y, top_k=top_k, temp=temp)

        v = mom * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9995
        temp *= 1.003
        mom = min(0.97, mom + 0.0002)

        if no_improve >= 300:
            break

    return best_a, best_obj


def lp_optimize_v2(a0, time_limit, verbose=False, m_reduction_rate=0.999):
    """
    LP-based iterative optimization with bisection line search,
    direction momentum, and adaptive M scheduling.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    perturb_count = 0
    max_no_improve = 60
    m_factor = 1.0
    dir_momentum = np.zeros(n)
    momentum_beta = 0.7

    while time.monotonic() - t0 < time_limit:
        # Rebuild C since a changed
        C = build_conv_matrix(a)

        a_new, improved, t_used = lp_step_improved(a, conv, n, C, M_factor=m_factor)

        if improved:
            # Update direction momentum
            d_used = a_new - a
            dir_momentum = momentum_beta * dir_momentum + (1 - momentum_beta) * d_used
            a = a_new
            conv = fast_convolve(a)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj - 1e-14:
                best_obj = cur_obj
                best_a = a.copy()
            no_improve = 0
            # Slightly reduce M_factor to force more aggressive steps over time
            m_factor = max(0.95, m_factor * m_reduction_rate)
        else:
            no_improve += 1
            # Reset m_factor if stuck
            m_factor = min(1.05, m_factor * 1.01)

        iteration += 1

        if verbose and iteration % 200 == 0:
            print(f"    LP iter {iteration}, obj={best_obj:.8f}, t_used={t_used:.4f}, "
                  f"m_factor={m_factor:.4f}, time={time.monotonic()-t0:.1f}s")

        # Handle stagnation
        if no_improve >= max_no_improve:
            rng = np.random.default_rng(iteration * 13 + perturb_count * 7 + 1)

            # Structured perturbation: shift mass from center to one side (breaks symmetry)
            scale = 0.002 + 0.003 * min(perturb_count, 10)
            if perturb_count % 3 == 0:
                # Bias toward asymmetry
                x = np.arange(n)
                bias = np.exp(-0.5 * ((x - 0.3 * n) / (0.25 * n)) ** 2)
                bias -= bias.mean()
                a = best_a + scale * bias
            elif perturb_count % 3 == 1:
                # Random perturbation
                noise = rng.standard_normal(n) * scale
                a = best_a + noise
            else:
                # Peak-targeted perturbation: reduce at peak-correlated positions
                peak_idx = int(np.argmax(conv))
                j_start = max(0, peak_idx - n + 1)
                j_end = min(n - 1, peak_idx)
                peak_positions = np.arange(j_start, j_end + 1)
                a = best_a.copy()
                a[peak_positions] *= (1 - scale * 2)
                # Add mass to non-peak positions
                non_peak = np.setdiff1d(np.arange(n), peak_positions)
                if len(non_peak) > 0:
                    a[non_peak] += scale * 0.5
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
            else:
                a = a / s
            conv = fast_convolve(a)
            no_improve = 0
            perturb_count += 1
            max_no_improve = min(200, 60 + 15 * perturb_count)
            dir_momentum = np.zeros(n)
            m_factor = 1.0

            if perturb_count > 60:
                break

    return best_a, best_obj


def coordinate_descent(a0, time_limit, window=20):
    """Randomized coordinate descent: try adjusting pairs."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, 2.0 * len(a) * fast_convolve(a).max()
    a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(99999)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        batch_size = max(1, int(300000 / n))
        for _ in range(batch_size):
            if time.monotonic() - t0 > time_limit - 0.3:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            for delta in [0.002, 0.005, 0.01, -0.002, -0.005, -0.01]:
                if a[i] + delta < 0 or a[j] - delta < 0:
                    continue
                a_try = a.copy()
                a_try[i] += delta
                a_try[j] -= delta
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj - 1e-10:
                    a = a_try
                    best_obj = obj_try
                    best_a = a.copy()
                    conv = conv_try
                    no_improve = 0
                    break
            else:
                no_improve += 1

        if no_improve > 100:
            break

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations emphasizing asymmetry."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Asymmetric Gaussian (mass shifted left)
    for center_frac in [0.25, 0.35, 0.4]:
        key = f'asym_g_{int(center_frac*100)}'
        g = np.exp(-0.5 * ((x - center_frac * n) / (0.2 * n)) ** 2)
        inits[key] = g / g.sum()

    # Bimodal (mass at edges)
    edge = max(1, n // 5)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    # Asymmetric bimodal (more mass on one side)
    bi_a = np.zeros(n)
    bi_a[:edge * 2] = 1.5
    bi_a[-edge:] = 0.5
    inits['asym_bimodal'] = bi_a / bi_a.sum()

    # Power law (decaying)
    pw = (n - x) ** 0.5
    inits['power_dec'] = pw / pw.sum()

    # Power law (increasing)
    pw2 = (x + 1.0) ** 0.5
    inits['power_inc'] = pw2 / pw2.sum()

    # Sine-based
    inits['sine1'] = (np.sin(np.pi * x / n) ** 2 + 0.01)
    inits['sine1'] /= inits['sine1'].sum()

    # Triangle peak at 1/3
    tri = 1.0 - np.abs(x / n - 1.0 / 3) / (1.0 / 3)
    tri = np.maximum(tri, 0.01)
    inits['tri_13'] = tri / tri.sum()

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps

    return inits


def extend_sequence(a_short, n_target):
    """Extend a shorter sequence to target length via interpolation."""
    n_short = len(a_short)
    s = a_short.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    a_norm = a_short / s
    a_ext = np.interp(np.linspace(0, 1, n_target), np.linspace(0, 1, n_short), a_norm)
    a_ext = np.maximum(a_ext, 0)
    ses = a_ext.sum()
    if ses > 1e-15:
        a_ext = a_ext / ses
    return a_ext
