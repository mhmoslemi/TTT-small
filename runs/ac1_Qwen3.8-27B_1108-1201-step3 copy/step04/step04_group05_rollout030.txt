import time


def fast_auto_conv(a):
    """Auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:size]


def cross_corr_lambda_a(lam, a):
    """Compute (lam ⊛ a)[j] = sum_k lam[k]*a[k-j] for j=0..n-1 via FFT."""
    n = len(a)
    N = 1 << (2 * (2 * n - 1) - 1).bit_length()
    L = np.fft.rfft(lam, N)
    A = np.fft.rfft(a, N)
    corr = np.fft.irfft(np.conj(L) * A, N)
    return corr[:n]


def compute_smooth_gradient(a, conv, T):
    """Compute gradient of T*log(sum(exp(conv/T)))."""
    n = len(a)
    # Softmax (stable)
    scaled = conv / T
    scaled -= scaled.max()
    exp_scaled = np.exp(scaled)
    lam = exp_scaled / exp_scaled.sum()
    grad = 2.0 * cross_corr_lambda_a(lam, a)
    return grad


def project_simplex(v):
    """Euclidean projection onto simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def mirror_descent_optimize(a0, time_limit, T_start=0.01, T_end=0.001,
                            eta=0.3, verbose=True, rng=None):

    if s < 1e-15:

    else:
        a = a / s
    n = len(a)
    
    conv = fast_auto_conv(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    T = T_start
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        grad = compute_smooth_gradient(a, conv, T)

        # Clip for numerical stability

            no_improve = 0

            no_improve += 1

        conv = conv_new
