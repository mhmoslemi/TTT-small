```python
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(rev_w, fft_size)
    c = np.fft.irfft(A * W, fft_size)
    grad = 2.0 * c[2 * n - 2 : n - 1 : -1]
    return grad


def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    a = np.maximum(a, 0.0)

    if s < 1e-15:

        else:

        pv = conv[peaks]
        # Use rank-based weights

```

Wait, the subgradient computation in the loop is O(n * top_k) which for n=384 and top_k=10 is 3840 operations. That's fine.

But I can make it faster using the FFT trick! The subgradient at the argmax position k* is:
`sg[j] = 2 * a[k*-j]` for valid j.

This is just a shifted/reversed version of a. For multiple peaks with weights, it's a weighted sum of such shifts, which is again a convolution!

`sg = 2 * conv(a, weights_at_peaks)` where `weights_at_peaks` is a sparse vector with the weights at the peak positions.

Let me optimize this:

```python
def compute_sg_fft(a, peaks, weights):
    """Compute subgradient using FFT. peaks and weights are arrays of same length."""
    n = len(a)
    # Create a sparse weight vector in the convolution domain
    w_full = np.zeros(2 * n - 1)
    for i in range(len(peaks)):
        w_full[int(peaks[i])] = weights[i]
    
    # sg[j] = 2 * sum_k w_full[k] * a[k-j] for valid k-j
    # = 2 * conv(a, rev(w_full))[2n-2-j]
    rev_w = w_full[::-1]
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(rev_w, fft_size)
    c = np.fft.irfft(A * W, fft_size)
    return 2.0 * c[2 * n - 2 : n - 1 : -1]
```

This is much faster for large top_k.

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-phase smooth optimization.
Uses FFT-based gradient computation (O(n log n) per iteration) with:
Phase 1: LSE (log-sum-exp) with high temperature for coarse shaping
Phase 2: LSE with decreasing temperature for medium refinement
Phase 3: Power-sum smooth max with increasing p for fine optimization
Phase 4: Direct subgradient peak-chasing
Phase 5: Ultra-fine polish with very small steps
Includes momentum, adaptive learning rate, and periodic restarts.
"""

import numpy as np
import time


def fast_auto_conv(a):
    """Compute auto-convolution a * a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v -

theta, 0.0)


def normalize_positive(a):
    """Clamp to non-negative and normalize to sum=1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def lse_grad_fft(a, conv, T):
    """Compute gradient of T*log(sum(exp(conv/T))) w.r.t. a using FFT.
    grad[j] = 2 * sum_k softmax(conv/T)[k] * a[k-j] (valid indices only)."""
    n = len(a)
    x = conv / T
    x -= x.max()
    exp_x = np.exp(x)
    w = exp_x / exp_x.sum()
    # grad[j] = 2 * conv(a, rev(w))[2n-2-j]
    rev_w = w[::-1]
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(rev_w, fft_size)
    c = np.fft.irfft(A * W, fft_size)
    grad = 2.0 * c[2 * n - 2: n - 1: -1]
    return grad


def power_sum_grad_fft(a, conv, p):
    """Compute gradient of (sum(conv^p))^(1/p) w.r.t. a using FFT."""
    n = len(a)
    max_c = conv.max()
    if max_c < 1e-30:
        return np.zeros(n)
    norm = conv / max_c
    # w_unnorm[k] = norm[k]^(p-1)
    w_unnorm = norm ** (p - 1)
    sum_p = (norm ** p).sum()
    scale = sum_p ** (1.0 / p - 1.0)
    rev_w = w_unnorm[::-1]
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(rev_w, fft_size)
    c = np.fft.irfft(A * W, fft_size)
    grad = 2.0 * scale * c[2 * n - 2: n - 1: -1]
    return grad


def sg_grad_fft(a, peaks, weights):
    """Compute weighted subgradient using FFT.
    peaks: array of peak indices, weights: corresponding weights."""
    n = len(a)
    w_full = np.zeros(2 * n - 1)
    for i in range(len(peaks)):
        k = int(peaks[i])
        if 0 <= k < 2 * n - 1:
            w_full[k] = weights[i]
    rev_w = w_full[::-1]
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(rev_w, fft_size)
    c = np.fft.irfft(A * W, fft_size)
    return 2.0 * c[2 * n - 2: n - 1: -1]


def smooth_phase(a0, time_limit, mode, T_start, T_end, lr_init, decay_lr,
                 mom=0.9, restart_every=300, verbose=True):
    """Generic smooth optimization phase.
    mode: 'lse' or 'power_sum'
    For lse: T_start/T_end are temperatures.
    For power_sum: T_start/T_end are p values (exponent).
    """
    a = normalize_positive(a0.copy())
    n = len(a)
    conv = fast_auto_conv(a)
    best_obj = conv.max()
    best_a = a.copy()

    v = np.zeros(n)
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    step = 0

    while time.monotonic() - t0 < time_limit:
        step += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / max(time_limit, 1e-10))

        if mode == 'lse':
            T = T_start * (T_end / T_start) ** progress
            conv = fast_auto_conv(a)
            grad = lse_grad_fft(a, conv, T)
        else:  # power_sum
            p = T_start + (T_end - T_start) * progress
            conv = fast_auto_conv(a)
            grad = power_sum_grad_fft(a, conv, p)

        v = mom * v - lr * grad
        a_new = project_simplex(a + v)

        conv_new = fast_auto_conv(a_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        lr *= decay_lr

        if no_improve > restart_every:
            rng = np.random.default_rng((step * 7919 + n) % (2**31))
            scale = 0.002 + 0.002 * min(1.0, no_improve / (2 * restart_every))
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            v = np.zeros(n)
            lr = max(lr_init * 0.4, 1e-5)
            no_improve = 0

        if verbose and step % 2000 == 0:
            cur_T = T if mode == 'lse' else p
            print(f"      [{mode}] step={step}, T/p={cur_T:.4f}, "
                  f"max={conv.max():.8f}, best={best_obj:.8f}, "
                  f"lr={lr:.6f}, t={elapsed:.0f}s")

    return best_a, best_obj

        n_restarts = max(2, int(rem / 40))
        for ri in range(n_restarts):
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            t_alloc = min(rem - 15, 35)
            t_alloc = max(5, t_alloc)
            # Different perturbation scales
            scales = [0.005, 0.003, 0.01, 0.002, 0.008]
            sc = scales[ri % len(scales)]
            perturbed = project_simplex(best_seq + rng.standard_normal(n) * sc)
            # Quick smooth optimization on perturbed
            a_opt, obj = smooth_phase(perturbed, t_alloc, mode='lse',
                                      T_start=0.02, T_end=0.005,
                                      lr_init=0.02, decay_lr=0.9998,
                                      mom=0.92, restart_every=150,
                                      verbose=False)
            try_seq(a_opt, f"explore_{ri}")

    # === Final: One more fine polish on the best ===
    rem = deadline - time.monotonic()
    if rem > 30:
        t_final = min(rem - 5, 50)
        if t_final > 10:
            print(f"\n=== Final Polish ({t_final:.0f}s) ===")
            a_opt, obj = fine_polish(best_seq, t_final,
                                     lr_init=2e-4, mom=0.96,
                                     top_k=50, restart_every=500,
                                     verbose=False)
            try_seq(a_opt, "final_polish")

    # Tiny perturbations at the very end
    rem = deadline - time.monotonic()
    if rem > 10:
        for _ in range(3):
            rem = deadline - time.monotonic()
            if rem < 5:
                break
            tiny = best_seq * (1 + rng.standard_normal(n) * 5e-5)
            tiny = project_simplex(tiny)
            try_seq(tiny, "tiny_perturb")

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs) and fs < best_score:
            best_score = fs

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'='*50}")
    return [float(x) for x in best_seq]
```
