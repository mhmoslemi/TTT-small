def coordinate_descent(a_init, time_limit, n_ternary=50, verbose=True):
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    a = a / a.sum()
    c_len = 2 * n - 1
    conv = np.convolve(a, a)
    
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    sweep = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        sweep += 1
        order = rng.permutation(n)
        
        for i in order:
            # Sensitivity (using numpy slicing)
            s1 = np.zeros(c_len)
            s1[i:i+n] = 2.0 * a
            s2 = np.zeros(c_len)
            s2[2*i] = 1.0
            
            # Quick check: is d=0 optimal?
            # Subgradient at d=0 for max_k (conv[k] + s1[k]*d + s2[k]*d^2):
            # = s1[k*] + 2*0*s2[k*] = s1[k*] where k* = argmax(conv)
            # If s1[k*] >= 0 for all k* in argmax set, d should be negative
            # If s1[k*] <= 0 for all k* in argmax set, d should be positive
            # Actually for the max function, the subgradient is s1[idx] for any idx in argmax
            # We want to find d where 0 is in the subgradient
            
            # Simple approach: just do ternary search
            lo = -a[i]
            hi = 0.01
            
            # Check if hi=0 is better than hi=0.01
            f0 = conv.max()  # = max(conv + s1*0 + s2*0)
            f_hi = np.max(conv + s1 * 0.01 + s2 * 0.0001)
            if f_hi > f0:
                hi = 0.0
            
            # Ternary search
            for _ in range(n_ternary):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = np.max(conv + s1 * m1 + s2 * m1 * m1)
                f2 = np.max(conv + s1 * m2 + s2 * m2 * m2)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            
            d_opt = (lo + hi) / 2.0
            
            # Update
            if abs(d_opt) > 1e-12:
                a[i] = max(0.0, a[i] + d_opt)
                conv += s1 * d_opt + s2 * d_opt * d_opt
                conv = np.maximum(conv, 0.0)
            
            cur = conv.max()
            if cur < best_obj:
                best_obj = cur
                best_a = a.copy()
                no_improve = 0
            # else: no_improve += 1  # don't increment per coordinate
        
        # Re-normalize
        s = a.sum()
        if s > 1e-10:
            a = a / s
            conv = np.convolve(a, a)
        
        cur_best = conv.max()
        if cur_best < best_obj - 1e-14:
            best_obj = cur_best
            best_a = a.copy()
        else:
            no_improve += 1
        
        if verbose and sweep % 50 == 0:
            print(f"  CD sweep {sweep}: obj={best_obj:.10f}, no_improve={no_improve}")
        
        # Restart if stuck
        if no_improve > 50:
            a = best_a + rng.standard_normal(n) * 0.001
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            conv = np.convolve(a, a)
            no_improve = 0
    
    return best_a, best_obj

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using coordinate descent with exact
1D minimization (ternary search), combined with FW-LP for diversity.

Key innovations:
- Exact coordinate descent: for each coordinate, find the exact 1D optimum
- Sensitivity computed via numpy slicing (O(n) per coordinate)
- Random coordinate order to avoid systematic bias
- Periodic re-normalization to maintain numerical precision
- Random restarts to escape local minima
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp(f, M):
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def fw_lp_optimize(f0, time_limit):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    m_exp = 1.0
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        g = solve_lp(f, M * m_exp)
        if g is None:
            m_exp = min(m_exp * 1.15, 5.0)
            if m_exp > 4.5:
                m_exp = 1.0
                no_improve += 5
            continue
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        if np.max(np.abs(g - f)) < 1e-9:
            no_improve += 1
            if no_improve > 30:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.08, 4.0)

            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0
    return best_f, best_obj
