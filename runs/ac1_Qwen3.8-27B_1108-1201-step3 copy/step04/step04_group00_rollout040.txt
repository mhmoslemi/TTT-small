    f = normalize(f0)
    n = len(f)
    best_f = f.copy()
    best_obj = conv_max(f)
    
    lr = 0.1  # initial learning rate
    temp = 0.01 * best_obj  # softmax temperature
    top_k = min(n, 50)  # number of peaks to consider
    
    no_improve = 0
    while time_remaining > 0:
        c = fast_convolve(f)
        M = c.max()
        
        # Select top-k peaks
        k_eff = min(top_k, len(c))
        peaks = argpartition(c, -k_eff)[-k_eff:]
        
        # Softmax weights
        pv = c[peaks]
        w_scaled = (pv - pv.max()) / temp
        w = exp(w_scaled) / sum(exp(w_scaled))
        
        # Create weight vector
        w_full = zeros(2n-1)
        w_full[peaks] = w
        
        # Compute cost
        cost = 2 * correlate(w_full, f, 'full')[:n][::-1]
        
        # Mass transfer: move from high cost to low cost
        src = argmax(cost)
        dst = argmin(cost)
        
        # Line search for transfer amount
        # Try moving fraction alpha of f[src] to f[dst]
        best_alpha = 0
        best_val = M
        for alpha in [0.01, 0.05, 0.1, 0.2, 0.3, 0.5]:
            f_trial = f.copy()
            f_trial[src] *= (1 - alpha)
            f_trial[dst] += alpha * f[src]  # careful: use original f[src]
            # Actually: transfer alpha * f[src] from src to dst
            transfer = alpha * f[src]
            f_trial = f.copy()
            f_trial[src] -= transfer
            f_trial[dst] += transfer
            f_trial = max(f_trial, 0)
            f_trial /= f_trial.sum()
            val = fast_convolve(f_trial).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 0:
            # Apply
            transfer = best_alpha * f[src]
            f[src] -= transfer
            f[dst] += transfer
            f = max(f, 0)
            f /= f.sum()
            
            if best_val < best_obj - 1e-14:
                best_obj = best_val
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        # Adaptive: increase top_k or temperature if stuck
        if no_improve > 20:
            top_k = min(top_k * 2, n)
            temp *= 2
            no_improve = 0
        
        if no_improve > 100:

polishing. Key innovation: O(n log n) subgradient computation via FFT-based
cross-correlation, enabling millions of iterations within the time budget.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_cost_fft(w, f):
    """
    Compute cost[j] = 2 * sum_k w[k] * f[k-j] for j=0..n-1.
    Uses FFT-based convolution for O(n log n) complexity.
    """
    n = len(f)
    f_rev = f[::-1].copy()
    size = len(w) + n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    F = np.fft.rfft(f_rev, fft_size)
    conv_full = np.fft.irfft(W * F, fft_size)
    # cost[j] = 2 * conv_full[n-1+j] for j=0..n-1
    return 2.0 * conv_full[n - 1:2 * n - 1]


def mult_weight_optimize(f0, time_limit, rng, verbose=False):
    """
    Fast multiplicative weight descent with adaptive top-k peaks.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    lr = 0.5
    temp = 0.01 * best_obj
    top_k = min(30, n)
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    best_lr = lr
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        # Select top-k peaks with softmax weighting
        k_eff = min(top_k, 2 * n - 1)
        if k_eff == 1:
            peaks = np.array([np.argmax(conv)])
            w_full = np.zeros(2 * n - 1)
            w_full[peaks[0]] = 1.0
        else:
            peaks = np.argpartition(conv, -k_eff)[-k_eff:]
            pv = conv[peaks]
            scaled = (pv - pv.max()) / temp
            scaled = np.clip(scaled, -50, 0)
            w = np.exp(scaled)
            ws = w.sum()
            if ws < 1e-15:
                w = np.zeros(len(peaks))
                w[np.argmax(pv)] = 1.0
            else:
                w /= ws
            w_full = np.zeros(2 * n - 1)
            w_full[peaks] = w
        
        # Compute cost via FFT
        cost = compute_cost_fft(w_full, f)
        
        # Multiplicative update: f[j] *= exp(-lr * cost[j] / max_cost)
        max_cost = cost.max()
        if max_cost < 1e-15:
            break
        normed_cost = cost / max_cost
        f_new = f * np.exp(-lr * normed_cost)
        
        # Project onto simplex
        f_new = project_simplex(f_new)
        
        if f_new.max() < 1e-15:
            break
        
        val = fast_convolve(f_new).max()
        
        if val < best_obj - 1e-14:
            best_obj = val
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            best_lr = lr
            if verbose and it % 10000 == 0:
                print(f"    MW iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            # Try smaller step
            if no_improve > 50:
                lr *= 0.7
                if lr < 0.001:
                    # Try larger top_k
                    top_k = min(top_k * 2, n)
                    temp *= 1.5
                    lr = 0.1
                if no_improve > 200:
                    # Reset
                    noise = rng.standard_normal(n) * 0.005
                    f = project_simplex(best_f + noise)
                    no_improve = 0
                    top_k = min(30, n)
                    temp = 0.01 * fast_convolve(f).max()
                    lr = 0.5
                if no_improve > 500:
                    break
        
        # Slowly increase top_k and temperature for diversity
        if it % 1000 == 0:
            top_k = min(top_k + 2, n)
            temp *= 1.01
    
    return best_f, best_obj


def pairwise_transfer(f0, time_limit, rng, verbose=False):
    """
    Pairwise mass transfer optimization.
    Move mass from high-cost to low-cost coordinates.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    alpha = 0.05  # transfer fraction
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        # Compute cost for active peaks
        k_eff = min(20, 2 * n - 1)
        peaks = np.argpartition(conv, -k_eff)[-k_eff:]
        w_full = np.zeros(2 * n - 1)
        # Uniform weights on active peaks
        w_full[peaks] = 1.0 / k_eff
        
        cost = compute_cost_fft(w_full, f)
        
        # Find best pair to transfer
        # Source: high cost, positive value
        # Dest: low cost
        src_candidates = np.where(cost > cost.mean())[0]
        dst_candidates = np.where(cost < cost.mean())[0]
        
        if len(src_candidates) == 0 or len(dst_candidates) == 0:
            no_improve += 1
            if no_improve > 100:
                noise = rng.standard_normal(n) * 0.003
                f = project_simplex(best_f + noise)
                no_improve = 0
            continue
        
        # Try a few random pairs
        best_alpha_found = 0
        best_val = M
        best_src = -1
        best_dst = -1
        
        n_tries = min(5, len(src_candidates) * len(dst_candidates))
        for _ in range(n_tries):
            si = src_candidates[rng.integers(len(src_candidates))]
            di = dst_candidates[rng.integers(len(dst_candidates))]
            if si == di or f[si] < 1e-10:
                continue
            
            # Try transferring fraction alpha of f[si] to f[di]
            for a in [0.01, 0.05, 0.1, 0.2]:
                transfer = a * f[si]
                f_trial = f.copy()
                f_trial[si] -= transfer
                f_trial[di] += transfer
                f_trial = np.maximum(f_trial, 0)
                st = f_trial.sum()
                if st < 1e-15:
                    continue
                f_trial /= st
                val = fast_convolve(f_trial).max()
                if val < best_val:
                    best_val = val
                    best_alpha = a
                    best_src = si
                    best_dst = di
                    best_alpha_found = a
        
        if best_src >= 0:
            transfer = best_alpha_found * f[best_src]
            f[best_src] -= transfer
            f[best_dst] += transfer
            f = np.maximum(f, 0)
            f /= f.sum()
            
            if best_val < best_obj - 1e-14:
                best_obj = best_val
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"    Pairwise iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 200:
            noise = rng.standard_normal(n) * 0.003
            f = project_simplex(best_f + noise)
            no_improve = 0
            alpha = 0.05
    
    return best_f, best_obj


def coordinate_descent_polish(f0, time_limit, verbose=False):
    """Coordinate descent with local search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)

    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            if f[j] < 1e-12:
                continue
            
            conv = fast_convolve(f)
            cur_peak = conv.max()
            
            # Try different values for f[j]
            best_j_obj = cur_peak
            best_j_val = f[j]
            
            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                new_val = f[j] * mult
                if new_val < 0 or new_val > 1.0:
                    continue
                # Redistribute the difference among other coordinates proportionally
                diff = new_val - f[j]
                others_sum = 1.0 - f[j]
                if others_sum < 1e-10:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] -= diff * (f[mask] / others_sum)
                f_trial = np.maximum(f_trial, 0)
                st = f_trial.sum()
                if st < 1e-10:
                    continue
                f_trial /= st
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != f[j]:
                diff = best_j_val - f[j]
                others_sum = 1.0 - f[j]
                if others_sum > 1e-10:
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] -= diff * (f[mask] / others_sum)
                    f = np.maximum(f, 0)
                    f /= f.sum()
                
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Linear interpolation to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast multiplicative weight
descent with FFT-based subgradient computation, pairwise mass transfer,
adaptive active set tracking, coordinate descent polishing, and
multi-length exploration. Key innovation: O(n log n) per-iteration cost
via FFT cross-correlation enables millions of optimization steps.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_cost_fft(w, f):
    """
    Compute cost[j] = 2 * sum_k w[k] * f[k-j] for j=0..n-1.
    Uses FFT-based convolution for O(n log n) complexity.
    w has length 2n-1, f has length n.
    """
    n = len(f)
    f_rev = f[::-1].copy()
    size = len(w) + n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    F = np.fft.rfft(f_rev, fft_size)
    conv_full = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv_full[n - 1:2 * n - 1]


def mult_weight_optimize(f0, time_limit, rng, verbose=False):
    """
    Fast multiplicative weight descent with adaptive top-k peaks,
    softmax weighting, and periodic restarts.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    lr = 0.4
    temp = 0.005 * best_obj
    top_k = min(20, n)

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    patience = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Select top-k peaks with softmax weighting
        k_eff = min(top_k, 2 * n - 1)
        if M < 1e-15:
            break

        if k_eff <= 1:
            peak_idx = np.argmax(conv)
            w_full = np.zeros(2 * n - 1)
            w_full[peak_idx] = 1.0
        else:
            peaks = np.argpartition(conv, -k_eff)[-k_eff:]
            pv = conv[peaks]
            max_pv = pv.max()
            scaled = (pv - max_pv) / temp
            scaled = np.clip(scaled, -60, 0)
            w = np.exp(scaled)
            ws = w.sum()
            if ws < 1e-15:
                w = np.zeros(len(peaks))
                w[np.argmax(pv)] = 1.0
            else:
                w /= ws
            w_full = np.zeros(2 * n - 1)
            w_full[peaks] = w

        # Compute cost via FFT
        cost = compute_cost_fft(w_full, f)
        max_cost = cost.max()
        if max_cost < 1e-15:
            break

        # Multiplicative update
        normed_cost = cost / max_cost
        f_new = f * np.exp(-lr * normed_cost)
        f_new = project_simplex(f_new)

        if f_new.max() < 1e-15:
            break

        val = fast_convolve(f_new).max()

        if val < best_obj - 1e-14:
            best_obj = val
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            patience = 0
            if verbose and it % 50000 == 0:
                print(f"    MW iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            patience += 1

            if no_improve > 30:
                lr *= 0.8
                if lr < 0.01:
                    top_k = min(top_k + 5, n)
                    temp *= 1.3
                    lr = 0.2
            if patience > 300:
                # Restart with noise
                scale = 0.003 * min(patience // 300, 5)
                noise = rng.standard_normal(n) * scale
                f = project_simplex(best_f + noise)
                no_improve = 0
                patience = 0
                top_k = min(20, n)
                temp = 0.005 * fast_convolve(f).max()
                lr = 0.4
            if patience > 2000:
                break

        # Gradually increase diversity
        if it % 20000 == 0:
            top_k = min(top_k + 3, n)
            temp *= 1.02

    return best_f, best_obj


def pairwise_transfer(f0, time_limit, rng, verbose=False):
    """
    Pairwise mass transfer: move mass from high-cost to low-cost coordinates.
    More precise than multiplicative updates for fine-tuning.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    top_k = min(15, n)

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        k_eff = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -k_eff)[-k_eff:]
        w_full = np.zeros(2 * n - 1)
        w_full[peaks] = 1.0 / k_eff

        cost = compute_cost_fft(w_full, f)

        # Find source (high cost) and destination (low cost)
        cost_mean = cost.mean()
        src_mask = cost > cost_mean
        dst_mask = cost < cost_mean

        if not src_mask.any() or not dst_mask.any():
            no_improve += 1
            if no_improve > 200:
                noise = rng.standard_normal(n) * 0.002
                f = project_simplex(best_f + noise)
                no_improve = 0
            continue

        src_indices = np.where(src_mask)[0]
        dst_indices = np.where(dst_mask)[0]

        best_val = M
        best_src = -1
        best_dst = -1
        best_amt = 0.0

        # Try multiple candidate pairs
        n_tries = min(8, len(src_indices) * len(dst_indices))
        for _ in range(n_tries):
            si = src_indices[rng.integers(len(src_indices))]
            di = dst_indices[rng.integers(len(dst_indices))]
            if si == di or f[si] < 1e-12:
                continue

            for frac in [0.02, 0.05, 0.1, 0.15, 0.25]:
                amt = frac * f[si]
                f_trial = f.copy()
                f_trial[si] -= amt
                f_trial[di] += amt
                f_trial = np.maximum(f_trial, 0)
                st = f_trial.sum()
                if st < 1e-15:
                    continue
                f_trial /= st
                val = fast_convolve(f_trial).max()
                if val < best_val - 1e-15:
                    best_val = val
                    best_src = si
                    best_dst = di
                    best_amt = amt

        if best_src >= 0:
            f[best_src] -= best_amt
            f[best_dst] += best_amt
            f = np.maximum(f, 0)
            f /= f.sum()

            if best_val < best_obj - 1e-14:
                best_obj = best_val
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 500 == 0:
                    print(f"    Pairwise iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 300:
            noise = rng.standard_normal(n) * 0.002
            f = project_simplex(best_f + noise)
            no_improve = 0

    return best_f, best_obj


def coordinate_descent_polish(f0, time_limit, verbose=False):
    """Coordinate descent with local search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    no_full_pass = 0

    while time.monotonic() - t0 < time_limit:
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            if f[j] < 1e-13:
                continue

            conv = fast_convolve(f)
            cur_peak = conv.max()
            best_j_obj = cur_peak
            best_j_val = f[j]

            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.6]:
                new_val = f[j] * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                others_sum = 1.0 - f[j]
                if others_sum < 1e-10:
                    continue
                diff = new_val - f[j]
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] -= diff * (f[mask] / others_sum)
                f_trial = np.maximum(f_trial, 0)
                st = f_trial.sum()
                if st < 1e-10:
                    continue
                f_trial /= st
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != f[j]:
                others_sum = 1.0 - f[j]
                if others_sum > 1e-10:
                    diff = best_j_val - f[j]
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] -= diff * (f[mask] / others_sum)
                    f = np.maximum(f, 0)
                    f /= f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved:
            no_full_pass += 1
            if no_full_pass >= 2:
                break
        else:
            no_full_pass = 0

    return best_f, best_obj


def sg_polish(f0, time_limit, rng, lr=0.005, mom=0.9, top_k=30, temp=0.0001):
    """Subgradient polishing with momentum and softmax-weighted peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp

    while time.monotonic() - t0 < time_limit:
        k_actual = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = (pv - pv.max()) / cur_temp
        scaled = np.clip(scaled, -50, 0)
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        w_full = np.zeros(2 * n - 1)
        w_full[peaks] = w

        cost = compute_cost_fft(w_full, f)
        cost /= max(cost.max(), 1e-15)

        v = mom * v - cur_lr * cost
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9995
        cur_temp *= 1.005

        if no_improve > 500:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0005)
            no_improve = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Linear interpolation to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
