**Algorithm Design:**

1. **Initialization**: Start from parent sequence, also try a few other inits.

2. **Softmax gradient descent with momentum**: 
   - Compute conv, softmax weights, gradient
   - Apply momentum update
   - Project to non-negative, renormalize
   - Adaptive learning rate

3. **Periodic LP refinement**:
   - Every K iterations, solve LP to get a good direction
   - Use line search along that direction

4. **Coordinate descent for fine-tuning**:
   - In the final phase, try to improve individual coordinates
   - Focus on coordinates that contribute to the top peaks

5. **Boundary optimization**:
   - The first and last few elements can be optimized more aggressively
   - They contribute to fewer conv terms, so we can compute their optimal values more precisely

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive softmax gradient descent,
LP-based direction finding, and coordinate descent fine-tuning.
Key innovations: smooth softmax-based gradient (vs. hard subgradient), 
FFT-accelerated gradient computation, boundary-aware coordinate descent,
and adaptive learning rate scheduling.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def obj_value(a, n):
    """Compute 2n * max(conv(a,a)) and the convolution."""
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv


def compute_grad(a, w, n):
    """Compute gradient: grad[i] = 2 * sum_k w[k] * a[k-i]."""
    corr = np.correlate(w, a, mode='full')
    return 2.0 * corr[n-1 : 2*n-1]


def softmax_grad_step(a, conv, lr, beta, n, v, momentum):
    """One softmax gradient descent step with momentum."""
    scaled = beta * (conv - conv.max())
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= w_sum
    
    grad = compute_grad(a, w, n)
    
    v = momentum * v - lr * grad
    a_new = a + v
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    return a_new, v


def solve_lp_direction(a, M):
    """Solve LP: maximize sum(g) s.t. a*g <= M, g >= 0."""
    n = len(a)
    A_ub = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum, g_sum
    except Exception:
        pass
    return a.copy(), 1.0


def line_search(a, d, current_obj, n, max_steps=15):
    """Simple line search along direction d."""
    best_t = 0.0
    best_obj = current_obj
    
    candidates = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]
    for t in candidates:
        a_try = np.maximum(a + t * d, 0.0)
        s = a_try.sum()
        if s < 1e-15:
            continue
        a_try = a_try / s
        obj_try, _ = obj_value(a_try, n)
        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_t = t
    
    # Small refinement around best_t
    if 0.03 < best_t < 0.97:
        for frac in [0.8, 1.0, 1.2]:
            t_try = best_t * frac
            if 0 < t_try < 1:
                a_try = np.maximum(a + t_try * d, 0.0)
                s = a_try.sum()
                if s > 1e-15:
                    a_try = a_try / s
                    obj_try, _ = obj_value(a_try, n)
                    if obj_try < best_obj - 1e-14:
                        best_obj = obj_try
                        best_t = t_try
    
    return best_t, best_obj
```

Let me optimize the coordinate descent by doing incremental updates:

For coordinate i, changing a[i] from old_val to new_val:
- The change in conv[k] is:
  - For k in [i, i+n-1] (clamped to [0, 2n-2]):
    - The term a[i]*a[k-i] changes by (new_val - old_val) * a[k-i]
    - The term a[k-i]*a[i] changes by a[k-i] * (new_val - old_val) [if k-i != i]
    - If k == 2i: a[i]*a[i] changes by new_val^2 - old_val^2

So total change in conv[k]:
- If k == 2i: (new_val^2 - old_val^2) = (new_val - old_val)(new_val + old_val)
- If k != 2i and 0 <= k-i < n: 2 * (new_val - old_val) * a[k-i]
- Otherwise: 0

This is O(n) per coordinate update (to update the affected conv values). Then finding the max is also O(n). So each coordinate update is O(n), and a full pass is O(n^2). For n=768, that's ~590K operations per pass, which is very fast.

"""
Minimizes 2n*max(conv(a,a))/(sum(a))^2 using adaptive softmax gradient descent
with FFT-accelerated gradients, periodic LP refinement, and analytical coordinate
descent for fine-tuning. Key improvements: smooth gradient signal, adaptive 
temperature scheduling, and O(n) coordinate updates via incremental convolution.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def obj_value(a, n):
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv


def compute_grad(a, w, n):
    """grad[i] = 2 * sum_k w[k] * a[k-i] using correlation."""
    corr = np.correlate(w, a, mode='full')
    return 2.0 * corr[n-1 : 2*n-1]


def softmax_step(a, conv, lr, beta, n, v, momentum):
    scaled = beta * (conv - conv.max())
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        w /= ws
    grad = compute_grad(a, w, n)
    v = momentum * v - lr * grad
    a_new = np.maximum(a + v, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new /= s
    return a_new, v


def solve_lp(a, M):
    n = len(a)
    A_ub = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            g = res.x
            gs = g.sum()
            if gs > 1e-10:
                return g / gs, gs
    except:
        pass
    return a.copy(), 1.0


def line_search(a, d, cur_obj, n):
    best_t, best_o = 0.0, cur_obj
    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
        at = np.maximum(a + t * d, 0.0)
        s = at.sum()
        if s < 1e-15:
            continue
        at /= s
        o, _ = obj_value(at, n)
        if o < best_o - 1e-14:
            best_o, best_t = o, t
    return best_t, best_o
