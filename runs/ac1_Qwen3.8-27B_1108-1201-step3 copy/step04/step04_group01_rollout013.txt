    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

FW-LP with exact line search, peak-shaving, and multi-start exploration.

from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_lob(x, beta):
    """Smoothed max using log-sum-exp."""
    m = x.max()
    return m + np.log(np.exp(x - m).sum()) / beta

def lse_gradient(f, beta):
    """Compute gradient of the beta-smoothed max of conv(f,f)."""
    n = len(f)
    conv = fast_convolve(f)
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    w = exp_bc / exp_bc.sum()
    # grad[j] = 2 * sum_k w[k] * f[k-j] for valid k-j
    return compute_cross_grad(w, f)

    result[j] = 2 * sum_k w[k] * f[k-j] where 0<=k-j<n

            rows.extend(k for _ in js)
            cols.extend(js.tolist())
            vals.extend((f[ks] for ks in (k - js)))

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FISTA (Nesterov-accelerated) LSE,
FW-LP with golden-section line search, peak-shaving subgradient, and multi-start
random exploration. Key improvements over prior approach: proper FISTA momentum,
adaptive Lipschitz estimation, aggressive beta annealing, and targeted peak reduction.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_cross_grad(w, f):
    """Compute 2 * (w * f) cross-correlation of length n.
    result[j] = 2 * sum_k w[k] * f[k-j] where 0 <= k-j < n.
    """
    n = len(f)
    f_flipped = f[::-1].copy()
    total_len = len(w) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]


def build_lp_matrix(f):
    """Build sparse constraint matrix A of shape (2n-1, n) where (A@g)[k] = (f*g)[k]."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            for j in js:
                rows.append(k)
                cols.append(j)
                vals.append(f[k - j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0. Returns g or None."""
    n = len(f)
    A = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 15})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

        if time.monotonic() - t0 >= time_budget:
            break

        g = solve_lp_direction(f, M * m_exp)

        if g is None or g.sum() < 1e-15:
            m_exp = min(m_exp * 1.15, 10.0)
            if m_exp >= 10.0:
                rng = np.random.default_rng(it * 7919 + 3)
                g = rng.dirichlet(np.ones(n) * 0.5)
                m_exp = 1.0
            else:
                continue

        gs = g.sum()
        if gs < 1e-15:
            break
        g = g / gs

        if np.max(np.abs(g - f)) < 1e-13:
            no_improve += 1
            if no_improve > 40:
                break
            m_exp = min(m_exp * 1.03, 5.0)
            continue

        # Golden section line search
        phi = (np.sqrt(5.0) - 1.0) / 2.0
        lo, hi = 0.0, 1.0
        c_pt = hi - phi * (hi - lo)
        d_pt = lo + phi * (hi - lo)

        def eval_fn(alpha):
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                return float('inf')
            ft = ft / st
            return fast_convolve(ft).max()

        fc = eval_fn(c_pt)
        fd = eval_fn(d_pt)
        for _ in range(25):
            if time.monotonic() - t0 >= time_budget:
                break
            if fc < fd:
                hi = d_pt
                d_pt = c_pt
                fd = fc
                c_pt = hi - phi * (hi - lo)
                fc = eval_fn(c_pt)
            else:
                lo = c_pt
                c_pt = d_pt
                fc = fd
                d_pt = lo + phi * (hi - lo)
                fd = eval_fn(d_pt)

        best_alpha = (lo + hi) / 2.0
        f_new = (1.0 - best_alpha) * f + best_alpha * g
        f_new = np.maximum(f_new, 0.0)
        f_new = f_new / f_new.sum()
        conv_new = fast_convolve(f_new)
        M_new = conv_new.max()

        if M_new < best_obj - 1e-14:
            best_obj = M_new
            best_f = f_new.copy()
            f = f_new
            M = M_new
            m_exp = 1.0
            no_improve = 0
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.07, 8.0)

        if no_improve > 50:
            rng = np.random.default_rng(it * 104729 + 11)
            scale = 0.002 + 0.004 * min(1.0, (no_improve - 50) / 60.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

        if it % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW-LP it={it}, obj={best_obj:.10f}, alpha={best_alpha:.4f}, elapsed={elapsed:.0f}s")

    return best_f, best_obj


def peak_shave(f0, time_budget, top_k=16):
    """Subgradient method targeting the top-k convolution peaks with momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    lr = 0.004
    v = np.zeros(n)
    mom = 0.92
    it = 0

    while time.monotonic() - t0 < time_budget:
        it += 1
        if time.monotonic() - t0 >= time_budget:
            break

        conv = fast_convolve(f)
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        # Weight peaks by their excess over the mean
        vals = conv[peaks]
        excess = np.maximum(vals - vals.mean(), 0.0)
        w_peak = excess / (excess.sum() + 1e-15)

        w = np.zeros(len(conv))
        w[peaks] = w_peak

        grad = compute_cross_grad(w, f)
        v = mom * v - lr * grad
        f_new = project_simplex(f + v)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            if no_improve == 0 and it % 200 == 0:
                lr = min(lr * 1.002, 0.015)
        else:
            no_improve += 1
            if no_improve % 40 == 0:
                lr *= 0.94
            if no_improve > 250:
                rng = np.random.default_rng(it * 37 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                v = np.zeros(n)
                lr = 0.005
                no_improve = 0

        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave it={it}, obj={best_obj:.12f}, lr={lr:.6f}, elapsed={elapsed:.0f}s")

    return best_f, best_obj


def try_eval(a, best_tracker):
    """Evaluate sequence and update best tracker."""
    a = np.maximum(a, 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  *** NEW BEST: {score:.10f} (n={len(a)}) ***")
    return score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization:
    Phase 1: FW-LP for global structure
    Phase 2: FISTA-LSE for smooth polishing
    Phase 3: Peak shaving for targeted reduction
    Phase 4: FW-LP refinement
    Phase 5: High-beta FISTA-LSE for final precision
    Phase 6: Multi-start random exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)

    best_tracker = [float('inf'), None]

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, initial score={evaluate_sequence(parent.tolist()):.10f}")
            try_eval(parent, best_tracker)
    except Exception as e:
        print(f"No parent available: {e}")

    if parent is None:
        n = 448
        parent = np.ones(n) / n
        try_eval(parent, best_tracker)

    # Phase 1: FW-LP
    print("\n=== Phase 1: FW-LP (structure) ===")
    t_rem = deadline - time.monotonic()
    t1 = min(350, t_rem * 0.22)
    if t1 > 30:
        print(f"  Budget: {t1:.0f}s")
        a_opt, _ = fw_lp_optimize(parent, time_budget=t1)
        try_eval(a_opt, best_tracker)

    # Phase 2: FISTA-LSE
    print("\n=== Phase 2: FISTA-LSE (polish) ===")
    cur = best_tracker[1] if best_tracker[1] is not None else parent
    t_rem = deadline - time.monotonic()
    t2 = min(450, t_rem * 0.28)
    if t2 > 30:
        print(f"  Budget: {t2:.0f}s")
        a_opt, _ = lse_fista(cur, time_budget=t2, beta_start=20.0, beta_end=10000.0)
        try_eval(a_opt, best_tracker)

    # Phase 3: Peak shaving
    print("\n=== Phase 3: Peak Shaving ===")
    cur = best_tracker[1] if best_tracker[1] is not None else parent
    t_rem = deadline - time.monotonic()
    t3 = min(180, t_rem * 0.12)
    if t3 > 20:
        print(f"  Budget: {t3:.0f}s")
        nk = max(8, len(cur) // 30)
        a_opt, _ = peak_shave(cur, time_budget=t3, top_k=nk)
        try_eval(a_opt, best_tracker)

    # Phase 4: FW-LP refinement
    print("\n=== Phase 4: FW-LP (refine) ===")
    cur = best_tracker[1] if best_tracker[1] is not None else parent
    t_rem = deadline - time.monotonic()
    t4 = min(250, t_rem * 0.18)
    if t4 > 20:
        print(f"  Budget: {t4:.0f}s")
        a_opt, _ = fw_lp_optimize(cur, time_budget=t4)
        try_eval(a_opt, best_tracker)

    # Phase 5: High-beta FISTA-LSE
    print("\n=== Phase 5: Final FISTA-LSE (high beta) ===")
    cur = best_tracker[1] if best_tracker[1] is not None else parent
    t_rem = deadline - time.monotonic()
    t5 = min(300, t_rem * 0.20)
    if t5 > 20:
        print(f"  Budget: {t5:.0f}s")
        a_opt, _ = lse_fista(cur, time_budget=t5, beta_start=200.0, beta_end=50000.0)
        try_eval(a_opt, best_tracker)

    # Phase 6: Multi-start exploration
    print("\n=== Phase 6: Multi-Start Exploration ===")
    cur = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(cur)
    t_rem = deadline - time.monotonic() - 30

    n_restarts = min(8, max(2, int(t_rem / 80)))
    for r in range(n_restarts):
        t_rem = deadline - time.monotonic() - 30
        if t_rem < 40:
            break
        t_each = min(120, t_rem / max(1, n_restarts - r))
        if t_each < 30:
            break

        rng_r = np.random.default_rng(seed + 9999 + r * 131)
        # Different perturbation scales
        scales = [0.003, 0.006, 0.01, 0.015, 0.025, 0.04, 0.06, 0.1]
        scale = scales[r % len(scales)]
        f_pert = project_simplex(cur + rng_r.standard_normal(n_cur) * scale)

        # Mix FISTA and peak shave based on restart
        if r % 2 == 0:
            a_opt, _ = lse_fista(f_pert, time_budget=t_each,
                                 beta_start=30.0 + r * 20.0, beta_end=15000.0)
        else:
            a_opt, _ = peak_shave(f_pert, time_budget=t_each, top_k=max(6, n_cur // 40))
        try_eval(a_opt, best_tracker)

    # Final verification
    print("\n=== Final ===")
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"FINAL: score={final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    # Fallback
    fallback = np.ones(448) / 448
    print(f"FALLBACK: score={evaluate_sequence(fallback.tolist()):.10f}")
    return [float(x) for x in fallback]
