def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_grad_smooth(conv, f, beta):
    """Compute LSE gradient: 2 * cross_correlation(softmax(beta*conv), f)."""
    n = len(f)
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    w = exp_bc / exp_bc.sum()
    
    # Gradient: grad[i] = 2 * sum_k w[k] * f[k-i] (for valid k-i)
    # This is 2 * cross-correlation(w, f) truncated to length n
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv_result = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv_result[n - 1:n - 1 + n]


def build_lp_matrix(f):
    """Build sparse constraint matrix for the LP.
    A[k, j] = f[k-j] if 0 <= k-j < n, else 0.

    rows = []
    cols = []

            js = np.arange(j_start, j_end + 1)
            rows.extend(k for _ in js)
            cols.extend(js.tolist())
            vals.extend(f[k - j] for j in js)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

        if res.success:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None

    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_gradient(conv, f, beta):
    """Compute LSE gradient via FFT cross-correlation."""
    n = len(f)
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    w = exp_bc / exp_bc.sum()
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv_result = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv_result[n - 1:n - 1 + n]


def build_lp_matrix(f):
    """Build sparse constraint matrix for the LP. Shape: (2n-1, n)."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend(k for _ in js)
            cols.extend(js.tolist())
            vals.extend(f[k - j] for j in js)
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, A_cache=None):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(f)
    if A_cache is None:
        A_cache = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 30})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def fw_lp_optimize(f0, time_limit):
    """Frank-Wolfe with LP oracle and adaptive relaxation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    A_cache = build_lp_matrix(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        it += 1

        A_cache = build_lp_matrix(f)
        g = solve_lp_direction(f, M * m_exp, A_cache)
        if g is None:
            m_exp *= 1.15
            if m_exp > 8.0:
                m_exp = 1.0
            continue

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                break
            m_exp = min(m_exp * 1.05, 4.0)
            continue

        best_alpha = 0.0
        best_val = M
        alphas = [0.02, 0.04, 0.06, 0.08, 0.1, 0.15, 0.2, 0.25, 0.3,
                  0.4, 0.5, 0.6, 0.7, 0.8, 0.9, 1.0]
        for alpha in alphas:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.04)
            hi = min(1.0, best_alpha + 0.04)
            for _ in range(9):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.08, 6.0)

        if no_improve > 50:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.003 + 0.005 * min(1.0, (no_improve - 50) / 100.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def lse_optimize(f0, time_limit, beta_start=10.0, beta_end=10000.0):
    """LSE smoothed gradient descent with piecewise beta schedule."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = 3.0 / n
    momentum = 0.92
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    total_iters_est = max(1, int(time_limit * n / 50.0))

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        it += 1

        conv = fast_convolve(f)
        grad = compute_lse_gradient(conv, f, beta)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            no_improve += 100
            break
        grad = grad / grad_norm

        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.003, 6.0 / n)
        else:
            no_improve += 1
            if no_improve >= 8:
                lr *= 0.93
                momentum = max(0.85, momentum - 0.003)
            if no_improve > 100:
                f = best_f.copy()
                v = np.zeros(n)
                lr = max(3.0 / n, 0.001)
                beta = min(beta * 2.0, beta_end)
                momentum = 0.88
                no_improve = 0

        # Piecewise beta schedule: slow ramp early, faster in middle, slow at end
        frac = it / max(total_iters_est, 1)
        if frac < 0.3:
            beta *= 1.004
        elif frac < 0.7:
            beta *= 1.008
        else:
            beta *= 1.005
        beta = min(beta, beta_end)

        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE it={it}, obj={best_obj:.12f}, beta={beta:.1f}, lr={lr:.6f}, el={elapsed:.0f}s")

    return best_f, best_obj


def multiplicative_weights_optimize(f0, time_limit, top_k=24):
    """Multiplicative weights with multi-peak softmax targeting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    f = np.maximum(f, 1e-12)
    f = f / f.sum()

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    eta = 0.4
    temperature = 0.5
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        it += 1

        conv = fast_convolve(f)

        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            peak_vals = conv.copy()
        else:
            idx = np.argpartition(-conv, top_k)[:top_k]
            peaks = idx
            peak_vals = conv[idx]

        scaled = (peak_vals - peak_vals.max()) / temperature
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w = w / w_sum

        # Compute gradient via FFT
        w_full = np.zeros(2 * n - 1)
        for i in range(len(peaks)):
            w_full[int(peaks[i])] = w[i]

        f_flipped = f[::-1].copy()
        total_len = (2 * n - 1) + n - 1
        fft_size = 1 << (total_len - 1).bit_length()
        W = np.fft.rfft(w_full, fft_size)
        F = np.fft.rfft(f_flipped, fft_size)
        conv_result = np.fft.irfft(W * F, fft_size)
        grad = 2.0 * conv_result[n - 1:n - 1 + n]

        grad_max = grad.max()
        if grad_max < 1e-15:
            no_improve += 100
            break

        f_new = f * np.exp(-eta * grad / grad_max)
        f_new = np.maximum(f_new, 1e-12)
        f_new = f_new / f_new.sum()

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            eta = min(eta * 1.005, 1.5)
        else:
            f = f_new
            no_improve += 1
            if no_improve % 30 == 0:
                eta *= 0.92
            if no_improve > 250:
                rng = np.random.default_rng(it * 17 + 3)
                f = np.maximum(best_f + rng.standard_normal(n) * 0.003, 1e-12)
                f = f / f.sum()
                eta = 0.3
                no_improve = 0

        temperature *= 1.006

        if it % 300 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MW it={it}, obj={best_obj:.12f}, eta={eta:.4f}, temp={temperature:.4f}, el={elapsed:.0f}s")

    return best_f, best_obj


def spectral_perturb(f, strength=0.001):
    """Perturb in frequency domain while preserving non-negativity."""
    n = len(f)
    fft_size = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(f, fft_size)
    freq = np.arange(len(A))
    # Dampen high frequencies, add small random perturbation
    rng = np.random.default_rng(int(time.monotonic() * 10000) % 999983)
    perturbation = rng.standard_normal(len(A)) * strength
    perturbation[0] = 0  # Preserve DC component (sum)
    # Only perturb mid-band frequencies
    low_cut = max(1, n // 32)
    high_cut = len(A) - n // 32
    mask = np.zeros(len(A))
    mask[low_cut:high_cut] = 1.0
    A_perturbed = A + perturbation * mask
    f_new = np.fft.irfft(A_perturbed, fft_size)[:n]
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-15:
        return f.copy()
    return f_new / s


def try_sequence(a, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.maximum(a, 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score
