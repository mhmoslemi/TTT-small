            best_obj = om
            best_t = mid
    
    return best_t, best_obj


def adam_init(n):
    """Initialize Adam optimizer state."""
    return {
        'm': np.zeros(n),
        'v': np.zeros(n),
        't': 0
    }


def adam_step(state, grad, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """Update Adam state and return new state."""
    state['t'] += 1
    t = state['t']
    state['m'] = beta1 * state['m'] + (1 - beta1) * grad
    state['v'] = beta2 * state['v'] + (1 - beta2) * grad ** 2
    m_hat = state['m'] / (1 - beta1 ** t)
    v_hat = state['v'] / (1 - beta2 ** t)
    step = -lr * m_hat / (np.sqrt(v_hat) + eps)
    return state, step

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using dual-guided LP optimization,
FFT-accelerated weighted subgradient with Adam optimizer, and multi-length
exploration. Key innovations: LP dual variables as optimal subgradient weights,
cross-convolution minimizing LP for better directions, Adam with aggressive
LR scheduling, and adaptive perturbation for escaping local minima.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_autoconv(a):
    """Fast full auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_norm(a):
    """Objective for normalized (sum=1) sequence."""
    n = len(a)
    conv = fast_autoconv(a)
    return 2.0 * n * conv.max()


def normalize(a):
    """Normalize to sum=1, clip to non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_matrix(a):
    """Build matrix C such that C @ g = conv(a, g)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def solve_lp_maxsum(a, M):
    """LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns (g, g_sum, dual)."""
    n = len(a)
    C = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(C.shape[0], M)
    bounds = [(0.0, None)] * n
    result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        g = result.x
        g_sum = float(g.sum())
        dual = None
        try:
            dual = np.abs(result.ineqlin.marginals)
        except Exception:
            pass
        if g_sum > 1e-10:
            return g, g_sum, dual
    return None, 0.0, None


def solve_lp_minpeak(a):
    """LP: min t s.t. conv(a,g) <= t, sum(g)=1, g >= 0. Returns (g, t_star, dual)."""
    n = len(a)
    C = build_conv_matrix(a)
    # Variables: g[0..n-1], t
    # Minimize t
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    # Constraints: C @ g - t <= 0 for each row
    A_ub = np.hstack([C, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0.0, None)] * n + [(None, None)]
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    if result.success:
        g = result.x[:n]
        t_star = result.x[-1]
        dual = None
        try:
            dual = np.abs(result.ineqlin.marginals)
        except Exception:
            pass
        return g, t_star, dual
    return None, np.inf, None


def compute_sg_fft(a, weights):
    """Compute weighted subgradient of max(autoconv(a)) using FFT. O(n log n)."""
    n = len(a)
    w = weights / (weights.sum() + 1e-30)
    # sg[j] = 2 * sum_i a[i]*w[i+j] - w[2j]*a[j]
    # First term via cross-correlation: (a * w_rev)[2n-2-j]
    w_rev = w[::-1]
    total_len = n + (2 * n - 1) - 1  # = 3n-2
    fft_size = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w_rev, fft_size)
    conv = np.fft.irfft(A * W, fft_size)[:total_len]
    # corr[j] = conv[2n-2-j] for j=0,...,n-1
    corr = conv[2 * n - 2: n - 2: -1]
    sg = 2.0 * corr - w[0:2 * n - 1:2] * a
    return sg


def adam_step(m, v, t, grad, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """One Adam step. Returns (m, v, t, step)."""
    t += 1
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad ** 2
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    step = -lr * m_hat / (np.sqrt(v_hat) + eps)
    return m, v, t, step


def line_search(a, direction, max_iter=25):
    """Line search: find best t in [0,1] for a + t*direction."""
    best_t = 0.0
    best_obj = obj_norm(a)
    # Coarse
    for t in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
        if max_iter <= 0:
            break
        a_t = normalize(a + t * direction)
        o = obj_norm(a_t)
        max_iter -= 1
        if o < best_obj - 1e-14:
            best_obj = o
            best_t = t
    # Fine golden section
    if 0.05 < best_t < 0.95 and max_iter > 6:
        lo = max(0.01, best_t - 0.08)
        hi = min(0.99, best_t + 0.08)
        gr = (np.sqrt(5) - 1) / 2
        for _ in range(5):
            if max_iter <= 2:
                break
            t1 = lo + (1 - gr) * (hi - lo)
            t2 = lo + gr * (hi - lo)
            o1 = obj_norm(normalize(a + t1 * direction))
            o2 = obj_norm(normalize(a + t2 * direction))
            max_iter -= 2
            if o1 < o2:
                hi, t2, o2 = t2, t1, o1
            else:
                lo, t1, o1 = t1, t2, o2
            if hi - lo < 0.005:
                break
        mid = (lo + hi) / 2.0
        om = obj_norm(normalize(a + mid * direction))
        max_iter -= 1
        if om < best_obj:
            best_obj = om
            best_t = mid
    return best_t, best_obj

    v = np.zeros(n)
    t_adam = 0

        if no_improve % 80 == 0 and no_improve > 0:
            lr *= 0.92
        if no_improve >= 400:
            # Perturb
            rng = np.random.default_rng(hash((iteration, perturb_count)) % (2**31))
            scale = 0.008 + 0.005 * min(perturb_count, 5)
            noise = rng.standard_normal(n) * scale
            # Add smooth component
            x = np.arange(n)
            smooth = (np.sin(2 * np.pi * x / n * rng.integers(2, 12)) *
                      np.exp(-0.5 * ((x - n / 2) / (0.3 * n)) ** 2)) * scale * 0.3
            a = normalize(best_a + noise + smooth)
            m = np.zeros(n)
            v = np.zeros(n)
            t_adam = 0
            lr = 0.08
            no_improve = 0
            perturb_count += 1
            last_lp_dual = None
            M_margin = 1.0
            conv = fast_autoconv(a)
            M = conv.max()

        iteration += 1

        if verbose and iteration % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    iter={iteration}, obj={best_obj:.8f}, t={elapsed:.1f}s, "
                  f"lr={lr:.4f}, no_imp={no_improve}, perturb={perturb_count}")

    return best_a, best_obj


def resample_seq(seq, new_n):
    """Resample sequence to new length via linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    return normalize(new_seq)


def get_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    if parent is not None:
        inits['parent'] = resample_seq(parent, n)

    # Modulated patterns
    for freq, name in [(1, 'sine1'), (2, 'sine2'), (3, 'sine3'), (4, 'sine4'), (6, 'sine6')]:
        sig = 0.5 + 0.5 * np.cos(2.0 * np.pi * freq * x / n)
        inits[name] = normalize(sig)

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.15 * n)) ** 2)
    inits['gauss'] = normalize(g)

    # Triangular
    tri = np.minimum(x, n - 1 - x) + 0.1
    inits['tri'] = normalize(tri)

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Bimodal
    mode1 = np.exp(-0.5 * ((x - 0.3 * n) / (0.1 * n)) ** 2)
    mode2 = np.exp(-0.5 * ((x - 0.7 * n) / (0.1 * n)) ** 2)
    inits['bimodal'] = normalize(mode1 + mode2)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses LP-guided direction finding with dual-weighted Adam subgradient.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.6f}")
            ps = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(ps):
                best_score = ps
                best_sequence = parent.copy()
    except Exception:
        pass

    # ===== Phase 1: Quick multi-length scout =====
    print("=== Phase 1: Multi-length Scout ===")
    t_start = time.monotonic()
    phase1_budget = 300.0
    phase1_end = t_start + phase1_budget

    lengths = [512, 768, 1024]
    results = []

    for n in lengths:
        if time.monotonic() > phase1_end - 30:
            break
        inits = get_initializations(n, parent, rng)
        # Priority: parent first, then diverse
        init_order = ['parent', 'uniform', 'sine2', 'sine4', 'gauss', 'random', 'bimodal']
        init_order = [nm for nm in init_order if nm in inits][:4]

        t_per = max(15.0, (phase1_budget - 20) / (len(lengths) * 4))

        for name in init_order:
            if time.monotonic() > phase1_end - 10:
                break
            t_alloc = min(t_per, phase1_end - time.monotonic() - 5)
            if t_alloc < 8:
                break
            a0 = inits[name]
            a_opt, obj_opt = optimize_main(a0, time_limit=t_alloc)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  n={n}, {name}: score={score:.6f}")

    results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # ===== Phase 2: Deep optimization on top candidates =====
    print("=== Phase 2: Deep Optimization ===")
    t_start2 = time.monotonic()

    # Pick top candidates (max 3, different lengths preferred)
    top_candidates = []
    seen_n = set()
    for score, n, name, a_opt in results:
        if n not in seen_n:
            top_candidates.append((n, a_opt.copy()))
            seen_n.add(n)
        if len(top_candidates) >= 3:
            break

    # Always include the current best
    if best_sequence is not None:
        bn = len(best_sequence)
        if bn not in seen_n and len(top_candidates) < 4:
            top_candidates.append((bn, best_sequence.copy()))

    remaining = deadline - time.monotonic() - 250.0  # reserve 250s for phases 3-4
    if remaining < 100:
        remaining = 100
    t_per_deep = max(30.0, remaining / max(1, len(top_candidates)))

    for i, (n, a0) in enumerate(top_candidates):
        if time.monotonic() > deadline - 250:
            break
        t_alloc = min(t_per_deep, deadline - time.monotonic() - 220)
        if t_alloc < 15:
            break
        a_opt, obj_opt = optimize_main(a0, time_limit=t_alloc, verbose=True)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Deep {i} (n={n}): score={score:.6f}")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")

    # ===== Phase 3: Subgradient-only fine tuning =====
    print("=== Phase 3: Fine Tuning ===")
    t_start3 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 120.0
        if remaining > 30:
            # Multiple short runs with different Adam configs
            n_best = len(best_sequence)
            n_runs = max(1, int(remaining / 40))
            t_per = remaining / n_runs

            for ri in range(n_runs):
                if time.monotonic() > deadline - 100:
                    break
                t_alloc = min(t_per, deadline - time.monotonic() - 80)
                if t_alloc < 10:
                    break

                # Slight perturbation for diversity
                scale = 0.003 * (1 + 0.2 * ri)
                noise = rng.standard_normal(n_best) * scale
                a_start = normalize(best_sequence + noise)

                a_opt, obj_opt = optimize_main(a_start, time_limit=t_alloc)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Fine {ri}: score={score:.6f}")

    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")

    # ===== Phase 4: Final perturb & quick refine =====
    print("=== Phase 4: Final Exploration ===")
    t_start4 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 15.0
        if remaining > 20:
            n_best = len(best_sequence)
            n_perturb = max(1, int(remaining / 25))
            t_per = max(5.0, remaining / n_perturb)

            for pi in range(n_perturb):
                if time.monotonic() > deadline - 10:
                    break
                t_alloc = min(t_per, deadline - time.monotonic() - 8)
                if t_alloc < 4:
                    break

                # Varied perturbation strategies
                strategy = pi % 4
                if strategy == 0:
                    noise = rng.standard_normal(n_best) * 0.005
                    a_pert = normalize(best_sequence + noise)
                elif strategy == 1:
                    # Smooth perturbation
                    x = np.arange(n_best)
                    f = rng.integers(3, 15)
                    noise = np.sin(2 * np.pi * f * x / n_best + rng.uniform(0, 2*np.pi)) * 0.005
                    a_pert = normalize(best_sequence + noise)
                elif strategy == 2:
                    # Localized perturbation
                    center = rng.integers(n_best // 4, 3 * n_best // 4)
                    width = rng.integers(n_best // 8, n_best // 4)
                    x = np.arange(n_best)
                    gauss_noise = np.exp(-0.5 * ((x - center) / width) ** 2)
                    noise = gauss_noise * rng.standard_normal(n_best) * 0.01
                    a_pert = normalize(best_sequence + noise)
                else:
                    # Resample to different length and back
                    new_n = n_best + rng.integers(-n_best//8, n_best//8 + 1)
                    new_n = max(256, min(2048, new_n))
                    a_resampled = resample_seq(best_sequence, new_n)
                    a_pert = resample_seq(a_resampled, n_best)

                a_opt, obj_opt = optimize_main(a_pert, time_limit=t_alloc)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Perturb {pi} (n={len(a_opt)}): score={score:.6f}")

    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final check
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
