"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses: (1) soft-max gradient with backtracking, (2) LP direction finding,
(3) peak-targeted coordinate descent, (4) multi-scale refinement.

import numpy as np
import time
from scipy.optimize import linprog

# --- Helper functions (top-level, no closures) ---

def fast_conv(a):
    """Full linear convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def obj_value(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fast_conv(a)
    return 2.0 * n * c.max() / (s * s)

def build_conv_matrix(n):
    """Matrix C such that C @ g = conv(a, g) for any a of length n.
    Actually this depends on a, so we need: C(a) @ g = conv(a, g)
    C(a) has rows that are shifted versions of a.
    """
    C = np.zeros((2*n-1, n))
    for j in range(n):
        C[j:j+n, j] = np.flip(a)  # Wait, this isn't right either
    # Let me think... conv(a, g)[k] = sum_i a[i] * g[k-i]
    # = sum_j g[j] * a[k-j]
    # So C[k, j] = a[k-j] if 0 <= k-j <= n-1, else 0
    C = np.zeros((2*n-1, n))
    for j in range(n):
        for k in range(j, j+n):
            C[k, j] = a[k-j]
    return C

def compute_gradient(a, c, top_k=50, temp=0.01):
    """Compute gradient of max(c) using soft-max weighting.
    c = conv(a, a).
    grad_m = sum_k w_k * (a_m + a_{k-m}) for valid k-m.
    """
    n = len(a)
    # Get top peaks
    peak_indices = np.argpartition(c, -top_k)[-top_k:]
    peak_values = c[peak_indices]
    
    # Soft-max weights
    scaled = peak_values / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        # All equal, use hard max
        w = np.zeros(len(peak_indices))
        w[peak_values.argmax()] = 1.0
    else:
        w /= w_sum
    
    # Compute gradient
    grad = np.zeros(n)
    for idx in range(len(peak_indices)):
        if w[idx] < 1e-12:
            continue
        k = int(peak_indices[idx])
        wi = w[idx]
        # For each m in valid range: grad[m] += wi * (a[m] + a[k-m])
        m_start = max(0, k - n + 1)
        m_end = min(n - 1, k)
        ms = np.arange(m_start, m_end + 1)
        k_ms = k - ms
        grad[ms] += wi * (a[ms] + a[k_ms])
    return grad

def armijo_line_search(a0, direction, f0, c=fast_conv, alpha=0.5, beta=0.5, max_iter=30):
    """Armijo backtracking line search.
    Find t > 0 such that f(a0 + t*d) <= f0 + alpha * t * grad^T d
    Here we use a simpler condition: f(a0 + t*d) < f0 (just decreasing).
    """
    n = len(a0)
    t = 1.0
    for _ in range(max_iter):
        a_try = a0 + t * direction
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-15:
            t *= beta
            continue
        a_try = a_try / s  # Renormalize
        f_try = obj_value(a_try)
        if f_try < f0 - 1e-14:
            return a_try, f_try, t
        t *= beta
    return a0, f0, 0.0

def gradient_step(a, f0, lr, top_k, temp):
    """One gradient descent step with line search."""
    c = fast_conv(a)
    grad = compute_gradient(a, c, top_k, temp)
    
    # Scale gradient
    direction = -grad / (np.linalg.norm(grad) + 1e-15)
    
    # Line search
    a_new, f_new, t = armijo_line_search(a, lr * direction, f0)
    return a_new, f_new, t > 0

def lp_direction(a, M, C=None):
    """Solve LP: max sum(g) s.t. C @ g <= M, g >= 0."""
    n = len(a)
    if C is None:
        C = np.zeros((2*n-1, n))
        for j in range(n):
            for k in range(j, j+n):
                C[k, j] = a[k-j]
    
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, 
                        method='highs', options={'presolve': True})
        if result.success:
            g = result.x
            if g.sum() > 1.0 + 1e-10:  # Only useful if sum increased
                return g, True
    except Exception:
        pass
    return None, False

def lp_step(a, f0, verbose=False):
    """One LP-based step."""
    n = len(a)
    c = fast_conv(a)
    M = c.max()
    
    g, success = lp_direction(a, M * 1.001)
    if not success:
        return a.copy(), f0, False
    
    direction = g - a
    norm_dir = np.linalg.norm(direction)
    if norm_dir < 1e-12:
        return a.copy(), f0, False
    
    # Line search on the direction
    a_new, f_new, t = armijo_line_search(a, direction, f0)
    if t > 0 and f_new < f0:
        return a_new, f_new, True
    return a.copy(), f0, False

def soft_gradient_optimize(a0, time_limit, initial_lr=0.05, top_k=50, temp=0.01, momentum=0.9):
    """Nesterov-accelerated gradient descent with adaptive parameters."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a))
    else:
        a = a / s
    n = len(a)
    
    f0 = obj_value(a)
    best_a = a.copy()
    best_f = f0
    
    v = np.zeros(n)
    y = a.copy()
    lr = initial_lr
    cur_temp = temp
    mom = momentum
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Nesterov lookahead
        y = a + mom * v
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy
        
        cy = fast_conv(y)
        grad = compute_gradient(y, cy, top_k, cur_temp)
        
        # Update
        v = mom * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_try = a_new / s_new
        else:
            a_try = a
        
        f_try = obj_value(a_try)
        
        if f_try < best_f - 1e-13:
            a = a_try
            best_f = f_try
            best_a = a.copy()
            no_improve = 0
        elif f_try < f0:
            a = a_try
            no_improve += 1
        else:
            a = a  # Revert
            v = v * 0.5  # Dampen
            no_improve += 1
        
        f0 = obj_value(a)
        
        # Adaptive updates
        lr *= 0.999
        cur_temp *= 1.005
        mom = min(0.98, mom + 0.0001)
        
        if no_improve >= 300:
            break
    
    return best_a, best_f

def peak_targeted_search(a0, time_limit):

"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines: (1) Nesterov gradient descent with adaptive soft-max peak weighting,
(2) LP-based direction finding with tightened peak constraints,
(3) Frank-Wolfe/conditional gradient for escaping local minima,
(4) peak-targeted coordinate search, (5) multi-phase time allocation.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_conv(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def obj_value(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    c = fast_conv(a)
    return float(2.0 * n * c.max() / (s * s))


def build_conv_matrix(a):
    """Build matrix C such that C @ g = conv(a, g) (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        start = j
        end = j + n
        C[start:end, j] = a
    return C


def compute_gradient(a, c, top_k=50, temp=0.01):
    """Compute soft-max weighted gradient of peak autocorrelation."""
    n = len(a)
    peak_indices = np.argpartition(c, -top_k)[-top_k:]
    peak_vals = c[peak_indices]
    scaled = peak_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(peak_indices))
        w[peak_vals.argmax()] = 1.0
    else:
        w /= w_sum

    grad = np.zeros(n)
    for idx in range(len(peak_indices)):
        if w[idx] < 1e-12:
            continue
        k = int(peak_indices[idx])
        wi = w[idx]
        m_start = max(0, k - n + 1)
        m_end = min(n - 1, k)
        ms = np.arange(m_start, m_end + 1)
        k_ms = k - ms
        grad[ms] += wi * (a[ms] + a[k_ms])
    return grad


def compute_gradient_hard(a, c, top_k=5):
    """Compute gradient using hard max over top_k peaks (equal weight)."""
    n = len(a)
    peak_indices = np.argpartition(c, -top_k)[-top_k:]
    grad = np.zeros(n)
    for k in peak_indices:
        k = int(k)
        m_start = max(0, k - n + 1)
        m_end = min(n - 1, k)
        ms = np.arange(m_start, m_end + 1)
        k_ms = k - ms
        grad[ms] += (a[ms] + a[k_ms]) / top_k
    return grad


def armijo_line_search(a, direction, f0, alpha=0.1, beta=0.5, max_iter=40):
    """Backtracking line search: find t such that f(a + t*d) < f0."""
    n = len(a)
    t = 1.0
    for _ in range(max_iter):
        a_try = a + t * direction
        a_try = np.maximum(a_try, 0.0)
        s = a_try.sum()
        if s < 1e-15:
            t *= beta
            continue
        a_try = a_try / s
        f_try = obj_value(a_try)
        if f_try < f0 - 1e-14:
            return a_try, f_try, t
        t *= beta
    return a, f0, 0.0


def lp_solve_direction(a, C, epsilon=0.0):
    """Solve LP: max sum(g) s.t. C@g <= M*(1-eps at peak), g >= 0."""
    n = len(a)
    c = fast_conv(a)
    M = c.max()
    k_peak = int(np.argmax(c))

    b_ub = np.full(2 * n - 1, M * 1.001)
    if epsilon > 0:
        b_ub[k_peak] = M * (1.0 - epsilon)

    c_obj = -np.ones(n)
    bounds = [(0, None)] * n

    try:
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
        if result.success:
            g = result.x
            if g.sum() > 1.0 + 1e-10:
                return g, True
    except Exception:
        pass
    return None, False


def gradient_optimize(a0, time_limit, initial_lr=0.03, top_k=50, temp_init=0.01,
                      momentum=0.92, temp_rate=0.9995, lr_rate=0.9995):

    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()

    else:
        a = a / s
    n = len(a)

    f0 = obj_value(a)
    best_a = a.copy()
    best_f = f0

    y = a.copy()

        iter_count += 1

    return best_a, best_f


def lp_optimize(a0, time_limit, rng, tight_eps=0.005):
    """LP-based iterative optimization with tightened peak constraints."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    a = a / s
    n = len(a)

    f0 = obj_value(a)
    best_a = a.copy()
    best_f = f0

    C = build_conv_matrix(a)

    t0 = time.monotonic()
    no_improve = 0
    perturb_count = 0

    while time.monotonic() - t0 < time_limit:
        c = fast_conv(a)
        M = c.max()

        # Try LP with tightened constraint at peak
        g, success = lp_solve_direction(a, C, epsilon=tight_eps)

        if not success:
            # Try without tightening
            g, success = lp_solve_direction(a, C, epsilon=0.0)

        if success:
            direction = g - a
            a_new, f_new, t = armijo_line_search(a, direction, f0)
            if t > 0 and f_new < f0 - 1e-13:
                a = a_new
                f0 = f_new
                if f_new < best_f - 1e-13:
                    best_f = f_new
                    best_a = a.copy()
                C = build_conv_matrix(a)
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve >= 40:
            # Perturb
            scale = 0.001 + 0.003 * min(perturb_count, 15)
            noise = rng.standard_normal(n) * scale
            a = np.maximum(a + noise, 0)
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
            else:
                a = a / s
            f0 = obj_value(a)
            C = build_conv_matrix(a)
            no_improve = 0
            perturb_count += 1

            if perturb_count > 50:
                break

        if time.monotonic() - t0 > time_limit * 0.5 and no_improve >= 80:
            # Switch to gradient for a while
            a_g, f_g = gradient_optimize(a, time_limit=min(5.0, time_limit - (time.monotonic() - t0)),
                                          initial_lr=0.01, top_k=30, temp_init=0.005)
            if f_g < best_f:
                best_f = f_g
                best_a = a_g.copy()
            a = best_a.copy()
            f0 = obj_value(a)
            C = build_conv_matrix(a)
            no_improve = 0
            perturb_count = 0

    return best_a, best_f


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase hybrid search combining gradient descent,
    LP direction finding, Frank-Wolfe, and peak-targeted local search.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: n={len(p)}, sum={p.sum():.4f}")
            score_p = evaluate_sequence(parent.tolist())
            if np.isfinite(score_p):
                best_score = score_p
                best_sequence = parent.copy()
                print(f"Initial score: {best_score:.8f}")
    except Exception as e:
        print(f"Parent not available: {e}")

    if best_sequence is None:
        best_sequence = np.ones(512)
        best_score = evaluate_sequence(best_sequence.tolist())
        print(f"Default init: score={best_score:.8f}")

    # ============================================================
    # Phase 1: Gradient refinement (250s)
    # ============================================================
    print("\n=== Phase 1: Gradient Refinement ===")
    t0 = time.monotonic()
    t_phase1 = min(250.0, deadline - t0 - 1000.0)
    if t_phase1 > 30:
        print(f"  Allocating {t_phase1:.0f}s")
        a_opt, f_opt = gradient_optimize(
            best_sequence, time_limit=t_phase1,
            initial_lr=0.04, top_k=40, temp_init=0.008,
            momentum=0.93, temp_rate=0.9998, lr_rate=0.9997
        )
        score = evaluate_sequence(a_opt.tolist())
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Phase 1: {score:.8f} *IMPROVED*")
        else:
            print(f"  Phase 1: {score:.8f}")

    # ============================================================
    # Phase 2: LP-based deep optimization (600s)
    # ============================================================
    print("\n=== Phase 2: LP Optimization ===")
    t0 = time.monotonic()
    t_phase2 = min(600.0, deadline - t0 - 500.0)
    if t_phase2 > 30:
        print(f"  Allocating {t_phase2:.0f}s")
        a_opt, f_opt = lp_optimize(best_sequence, time_limit=t_phase2, rng=rng, tight_eps=0.005)
        score = evaluate_sequence(a_opt.tolist())
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Phase 2: {score:.8f} *IMPROVED*")
        else:
            print(f"  Phase 2: {score:.8f}")

    # ============================================================
    # Phase 3: Peak-targeted search (250s)
    # ============================================================
    print("\n=== Phase 3: Peak-Targeted Search ===")
    t0 = time.monotonic()
    t_phase3 = min(250.0, deadline - t0 - 300.0)
    if t_phase3 > 20:
        print(f"  Allocating {t_phase3:.0f}s")
        a_opt, f_opt = peak_targeted_search(best_sequence, time_limit=t_phase3, rng=rng)
        score = evaluate_sequence(a_opt.tolist())
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Phase 3: {score:.8f} *IMPROVED*")
        else:
            print(f"  Phase 3: {score:.8f}")

    # ============================================================
    # Phase 4: Frank-Wolfe for exploration (150s)
    # ============================================================
    print("\n=== Phase 4: Frank-Wolfe Exploration ===")
    t0 = time.monotonic()
    t_phase4 = min(150.0, deadline - t0 - 200.0)
    if t_phase4 > 20:
        print(f"  Allocating {t_phase4:.0f}s")
        a_opt, f_opt = fp_optimize(best_sequence, time_limit=t_phase4)
        score = evaluate_sequence(a_opt.tolist())
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Phase 4: {score:.8f} *IMPROVED*")
        else:
            print(f"  Phase 4: {score:.8f}")

    # ============================================================
    # Phase 5: Fine gradient polish (remaining time)
    # ============================================================
    print("\n=== Phase 5: Fine Gradient Polish ===")
    t0 = time.monotonic()
    t_phase5 = max(20.0, deadline - t0 - 15.0)
    if t_phase5 > 15:
        print(f"  Allocating {t_phase5:.0f}s")

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Phase 5: {score:.8f} *IMPROVED*")
        else:
            print(f"  Phase 5: {score:.8f}")

    # Phase 6: Final LP micro-polish with tighter constraints
    # ============================================================
    print("\n=== Phase 6: Final LP Polish ===")
    t0 = time.monotonic()
    t_phase6 = max(10.0, deadline - t0 - 5.0)
    if t_phase6 > 10:
        print(f"  Allocating {t_phase6:.0f}s")
        a_opt, f_opt = lp_optimize(best_sequence, time_limit=t_phase6, rng=rng, tight_eps=0.01)

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Phase 6: {score:.8f} *IMPROVED*")

    # Final verification
    final_score = evaluate_sequence([float(x) for x in best_sequence])
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.8f}, n={len(best_sequence)}")
    print(f"Total time: {time.monotonic() - t_start:.1f}s")

    return [float(x) for x in best_sequence]
