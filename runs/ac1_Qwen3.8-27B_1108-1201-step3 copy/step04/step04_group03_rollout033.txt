def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    r = np.nonzero(cond)[0][-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)

def enforce_palindrome(a):
    """Enforce palindromic symmetry: a[j] = a[n-1-j]."""
    n = len(a)
    half = (n + 1) // 2
    left = a[:half]
    if n % 2 == 0:
        right = left[::-1]
    else:
        right = left[:-1][::-1]
    return np.concatenate([left, right])

def subgrad_max_optimize(f0, time_limit, lr_init=0.002, momentum=0.9, verbose=False):
    """Subgradient method for min max_k conv[k]."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    f = project_palindrome_simplex(f)
    n = len(f)
    
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        # Subgradient: pick argmax, gradient is 2*f[k*-j] for each j
        # For palindromic sequences, use average of all argmax positions
        peak_indices = np.where(conv >= M - 1e-10)[0]
        
        grad = np.zeros(n)
        for k in peak_indices:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * f[k - idx]
        grad /= len(peak_indices)
        
        # Palindromize
        grad = 0.5 * (grad + grad[::-1])
        
        v = momentum * v - cur_lr * grad
        f_new = project_palindrome_simplex(f + v)
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-12:
            f = f_new
            best_obj = obj_new
            best_f = f.copy()
            no_improve = 0
            cur_lr = min(cur_lr * 1.002, lr_init * 2)
        else:
            no_improve += 1
            cur_lr *= 0.98
            if no_improve > 300:
                rng = np.random.default_rng(int(time.monotonic() * 10000) % 1000000)
                scale = 0.001 + 0.002 * min(1.0, no_improve / 500.0)
                f = project_palindrome_simplex(best_f + rng.standard_normal(n) * scale)
                v = np.zeros(n)
                cur_lr = max(lr_init * 0.5, 0.0003)

        cur_lr *= 0.9995
    
    return best_f, best_obj

def generate_inits(n, parent, rng):

        a = g / g.sum()

        c = 0.5 + 0.5 * np.cos(2 * np.pi * freq * x / n)
        a = c / c.sum()
        inits.append(project_palindrome_simplex(a))
    
    # Random Dirichlet

    # Parent interpolation

        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:

    return inits

"""
Multi-stage optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses L2 flattening, log-sum-exp soft-max with annealing, and subgradient
methods with palindromic symmetry, spectral initialization, and adaptive
step sizes. Key innovation: smooth surrogate objectives (L2, log-sum-exp)
that provide gradient flow to the non-smooth max objective.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    r = np.nonzero(cond)[0][-1]
    theta = (cssv[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)


def enforce_palindrome(a):
    """Enforce palindromic symmetry."""
    n = len(a)
    half = (n + 1) // 2
    left = a[:half].copy()
    if n % 2 == 0:
        return np.concatenate([left, left[::-1]])
    else:
        return np.concatenate([left, left[:-1][::-1]])


def project_palindrome(v):
    """Project onto palindromic non-negative simplex (approximate)."""
    v_sym = 0.5 * (v + v[::-1])
    v_sym = np.maximum(v_sym, 0.0)
    s = v_sym.sum()
    if s < 1e-15:
        return np.ones(len(v)) / len(v)
    return v_sym / s


def spectral_init(n, rng):
    """Initialize via random phases in Fourier domain (flat magnitude heuristic)."""
    m = 2 * n - 1
    fft_size = 1 << (m - 1).bit_length()
    phases = rng.uniform(0, 2 * np.pi, fft_size // 2 + 1)
    mag = np.ones(fft_size // 2 + 1)
    A = mag * np.exp(1j * phases)
    A[-1] = np.conj(A[1]) if fft_size > 2 else A[-1]
    a = np.fft.irfft(A, fft_size)[:m]
    a = a[:n]
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return project_palindrome(a)

def subgrad_max(f0, time_limit, lr_init=0.003, momentum=0.92):

        a = w / w.sum()
        inits.append(project_palindrome(a))

    for width in [0.15, 0.25, 0.4, 0.6]:
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (width * n)) ** 2)
        a = g / g.sum()
        inits.append(project_palindrome(a))

    for freq in [2, 3, 4, 6, 8, 12]:
        c = 0.5 + 0.5 * np.cos(2.0 * np.pi * freq * x / n)
        a = c / c.sum()
        inits.append(project_palindrome(a))

    for _ in range(3):
        a = rng.dirichlet(np.ones(n) * 0.5)
        inits.append(project_palindrome(a))

    for _ in range(2):
        a = spectral_init(n, rng)
        inits.append(a)

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0.0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(project_palindrome(px / ps))

    return inits


def evaluate_safe(seq_list):
    """Safely evaluate a sequence."""
    try:
        val = float(evaluate_sequence(seq_list))
        return val if np.isfinite(val) else float('inf')
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-stage search:
    Phase 1: Quick L2 exploration on diverse inits
    Phase 2: Deep L2 flattening
    Phase 3: Log-sum-exp soft-max with annealing
    Phase 4: Subgradient max optimization
    Phase 5: Fine polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    def update_best(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 1e-15:
            return
        a_norm = a / s
        score = evaluate_safe(a_norm.tolist())
        if score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})", flush=True)

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_safe(parent.tolist())
            print(f"Parent: n={len(parent)}, score={ps:.8f}", flush=True)
            if ps < best_score:
                best_score = ps
                best_seq = parent.copy()
    except Exception:
        pass

    # Determine lengths to try
    lengths = []
    if parent is not None:
        lengths.append(len(parent))
    for nl in [384, 448, 512, 256, 640]:
        if nl not in lengths:
            lengths.append(nl)

    # === Phase 1: Quick Exploration ===
    print("\n=== Phase 1: Quick Exploration ===", flush=True)

    for n in lengths[:4]:
        rem = deadline - time.monotonic() - 700
        if rem < 30:
            break

        inits = gen_inits(n, parent, rng)
        t_per = max(3, min(20, rem / max(1, min(len(inits), 10))))

        for ii in range(min(len(inits), 10)):
            rem = deadline - time.monotonic() - 650
            if rem < 5:
                break
            t_alloc = min(t_per, rem)
            a_opt, _ = l2_flatten(inits[ii], time_limit=t_alloc)
            update_best(a_opt, f"l2_{n}_{ii}")

    print(f"  After Phase 1: {best_score:.8f}", flush=True)

    # === Phase 2: Deep L2 ===
    print("\n=== Phase 2: Deep L2 Flattening ===", flush=True)

    if best_seq is not None:
        rem = deadline - time.monotonic() - 500
        if rem > 20:
            t_alloc = min(rem, 250)
            print(f"  L2 deep (n={len(best_seq)}), t={t_alloc:.0f}s", flush=True)
            a_opt, _ = l2_flatten(best_seq, time_limit=t_alloc)
            update_best(a_opt, "l2_deep")

            # Also try a different n
            rem = deadline - time.monotonic() - 450
            if rem > 20:
                n_alt = len(best_seq) + 64
                inits_alt = gen_inits(n_alt, best_seq, rng)
                a_opt2, _ = l2_flatten(inits_alt[0], time_limit=min(rem, 100))
                update_best(a_opt2, "l2_alt_n")

    print(f"  After Phase 2: {best_score:.8f}", flush=True)

    # === Phase 3: Log-Sum-Exp ===
    print("\n=== Phase 3: Log-Sum-Exp Refinement ===", flush=True)

    if best_seq is not None:
        lse_configs = [
            (0.3, 40.0, 0.004),
            (1.0, 60.0, 0.003),
            (0.1, 30.0, 0.005),
        ]
        rem = deadline - time.monotonic() - 250
        t_per = max(10, rem / len(lse_configs))

        for ci, (Ti, Tf, lr) in enumerate(lse_configs):
            rem = deadline - time.monotonic() - 200
            if rem < 10:
                break
            t_alloc = min(t_per, rem)
            print(f"  LSE {ci}: T={Ti}->{Tf}, lr={lr}, t={t_alloc:.0f}s", flush=True)
            a_opt, _ = logsumexp_opt(best_seq, time_limit=t_alloc,
                                     T_init=Ti, T_final=Tf, lr_init=lr)
            update_best(a_opt, f"lse_{ci}")

    print(f"  After Phase 3: {best_score:.8f}", flush=True)

    # === Phase 4: Subgradient Max ===
    print("\n=== Phase 4: Subgradient Max ===", flush=True)

    if best_seq is not None:
        sg_configs = [
            (0.003, 0.92),
            (0.0015, 0.95),
            (0.005, 0.88),
        ]
        rem = deadline - time.monotonic() - 100
        t_per = max(10, rem / len(sg_configs))

        for si, (lr, mom) in enumerate(sg_configs):
            rem = deadline - time.monotonic() - 60
            if rem < 10:
                break
            t_alloc = min(t_per, rem)
            print(f"  SG {si}: lr={lr}, mom={mom}, t={t_alloc:.0f}s", flush=True)
            a_opt, _ = subgrad_max(best_seq, time_limit=t_alloc,
                                   lr_init=lr, momentum=mom)
            update_best(a_opt, f"sg_{si}")

    print(f"  After Phase 4: {best_score:.8f}", flush=True)

    # === Phase 5: Fine Polish ===
    print("\n=== Phase 5: Fine Polish ===", flush=True)

    if best_seq is not None:
        # Fine subgradient
        rem = deadline - time.monotonic() - 40
        if rem > 10:
            t_alloc = min(rem, 50)
            print(f"  Fine SG, t={t_alloc:.0f}s", flush=True)
            a_opt, _ = fine_polish(best_seq, time_limit=t_alloc,
                                   lr_init=0.0005, momentum=0.96)
            update_best(a_opt, "fine_sg")

        # Short LSE with high T
        rem = deadline - time.monotonic() - 25
        if rem > 8:
            a_opt, _ = logsumexp_opt(best_seq, time_limit=min(rem, 25),
                                     T_init=5.0, T_final=100.0, lr_init=0.001)
            update_best(a_opt, "fine_lse")

        # Tiny perturbations
        rem = deadline - time.monotonic() - 10
        for _ in range(5):
            rem = deadline - time.monotonic() - 5
            if rem < 3:
                break
            noise = rng.standard_normal(len(best_seq)) * 0.0005
            f_pert = project_palindrome(best_seq + noise)
            a_opt, _ = fine_polish(f_pert, time_limit=min(rem, 8),
                                   lr_init=0.0003, momentum=0.96)
            update_best(a_opt, "perturb")

        # Try slightly different length
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            n_try = len(best_seq) + 32
            inits_try = gen_inits(n_try, best_seq, rng)
            a_opt, _ = subgrad_max(inits_try[-1], time_limit=min(rem, 10),
                                   lr_init=0.002, momentum=0.9)
            update_best(a_opt, "diff_n")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = evaluate_safe(best_seq.tolist())

    # Final check
    fs = evaluate_safe(best_seq.tolist())
    if fs < best_score:
        best_score = fs

    print(f"\nFINAL: {best_score:.10f} (n={len(best_seq)})", flush=True)
    return [float(x) for x in best_seq]
