def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(f, conv, top_k=10, temp=0.001):
    """Compute weighted subgradient of max(conv) using softmax weighting on top peaks."""
    n = len(f)
    # Find top peaks
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    # Subgradient of conv[k] w.r.t. f[j] is 2*f[k-j] (for valid j)
    sg = np.zeros(n)
    for i in range(len(peaks)):

        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k - idx]
    
    return sg

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_conv_gradient(f, k):
    """Compute d conv[k] / d f[j] for all j."""
    n = len(f)
    g = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    for j in range(js, je + 1):
        km = k - j
        if 0 <= km < n:
            if j == km:
                g[j] = f[j]
            else:
                g[j] = 2.0 * f[km]
    return g


def compute_weighted_subgrad(f, conv, top_k=30, temp=0.0005):
    """Compute weighted subgradient of max(conv) using softmax over top peaks."""
    n = len(f)
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-14:
            continue
        k = int(peaks[i])
        g = compute_conv_gradient(f, k)
        sg += w[i] * g
    return sg


def softmax_from_t(t):
    """Convert parameter vector to probability vector via softmax."""
    t_shifted = t - t.max()
    a = np.exp(t_shifted)
    s = a.sum()
    if s < 1e-300:
        return np.ones(len(t)) / len(t)
    return a / s


def t_gradient(sg, a):
    """Convert subgradient in a-space to t-space via softmax chain rule.
    d conv/dt[m] = a[m] * (sg[m] - dot(sg, a))
    """
    dot = np.dot(sg, a)
    return a * (sg - dot)


def adam_softmax_optimize(t_init, time_limit, lr=0.05, beta1=0.9, beta2=0.999,
                          eps=1e-8, top_k=40, temp=0.0005, verbose=False):
    """Adam optimizer with softmax parameterization for minimizing max(conv(a,a))."""
    t = t_init.astype(np.float64).copy()
    n = len(t)

    m = np.zeros(n)
    v = np.zeros(n)
    step = 0

    a = softmax_from_t(t)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    best_t = t.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_top_k = top_k
    cur_temp = temp

    while time.monotonic() - t0 < time_limit:
        step += 1

        sg = compute_weighted_subgrad(a, conv, top_k=cur_top_k, temp=cur_temp)
        grad_t = t_gradient(sg, a)

        m = beta1 * m + (1 - beta1) * grad_t
        v = beta2 * v + (1 - beta2) * grad_t ** 2
        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)

        t = t - cur_lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = softmax_from_t(t)
        conv_new = fast_convolve(a_new)
        obj = conv_new.max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a_new.copy()
            best_t = t.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        conv = conv_new

        if verbose and step % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam iter {step}: obj={best_obj:.10f}, elapsed={elapsed:.1f}s")

        if no_improve > 200:
            rng = np.random.default_rng(step * 7919 + 13)
            noise_scale = 0.02 + 0.01 * min(no_improve // 200, 5)
            t = best_t + rng.standard_normal(n) * noise_scale
            a = softmax_from_t(t)
            conv = fast_convolve(a)
            m = np.zeros(n)
            v = np.zeros(n)
            step = 0
            no_improve = 0
            cur_lr = max(cur_lr * 0.7, 0.001)
            cur_temp = min(cur_temp * 1.5, 0.01)

        if step % 1000 == 0:
            cur_lr *= 0.95
            cur_temp = min(cur_temp * 1.1, 0.01)
            cur_top_k = min(cur_top_k + 5, 200)

    return best_a, best_obj


def mirror_descent_optimize(f_init, time_limit, eta=0.01, top_k=30, temp=0.0005,
                            verbose=False):
    """Mirror descent (exponential weights) on the simplex."""
    f = np.maximum(f_init.astype(np.float64), 1e-15)
    f = f / f.sum()
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_eta = eta
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        sg = compute_weighted_subgrad(f, conv, top_k=top_k, temp=temp)

        # Mirror descent update: f_new_j = f_j * exp(-eta * sg_j) / Z
        log_f = np.log(f)
        log_f_new = log_f - cur_eta * sg
        log_f_new -= log_f_new.max()
        f_new = np.exp(log_f_new)
        f_new /= f_new.sum()

        conv_new = fast_convolve(f_new)
        obj = conv_new.max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD iter {it}: obj={best_obj:.10f}, elapsed={elapsed:.1f}s")

        if no_improve > 300:
            rng = np.random.default_rng(it * 31 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.01)
            no_improve = 0
            cur_eta *= 0.5

        if it % 500 == 0:
            cur_eta *= 0.92
            temp = min(temp * 1.05, 0.01)

    return best_f, best_obj


def coordinate_descent(f_init, time_limit, verbose=False):
    """Coordinate descent on the simplex: adjust one coordinate at a time."""
    f = np.maximum(f_init.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f_init)) / len(f_init)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-14:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            multipliers = [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.6, 2.0]
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-16:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()

                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved and passes > 2:
            break

    return best_f, best_obj


def pair_swap_descent(f_init, time_limit, verbose=False):
    """Try swapping mass between pairs of coordinates."""
    f = np.maximum(f_init.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f_init)) / len(f_init)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)

    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break

        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # Try redistributing mass between i and j
        total_ij = f[i] + f[j]
        if total_ij < 1e-14:
            continue

        for frac in [0.1, 0.25, 0.4, 0.5, 0.6, 0.75, 0.9]:
            f_trial = f.copy()
            f_trial[i] = total_ij * frac
            f_trial[j] = total_ij * (1.0 - frac)
            val = fast_convolve(f_trial).max()
            if val < best_obj - 1e-16:
                best_obj = val
                best_f = f_trial.copy()
                f = f_trial.copy()
                break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_smooth_init(n, seed=0):
    """Create a smooth initial sequence using cosine window with random perturbation."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.sin(np.pi * x) ** 2
    a += 0.1 * rng.standard_normal(n)
    a = np.maximum(a, 0)
    a /= a.sum()
    return a


def make_bump_init(n, center=0.5, width=0.3, seed=0):
    """Create a bump-like initial sequence."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - center) / width) ** 2)
    a += 0.05 * rng.standard_normal(n)
    a = np.maximum(a, 0)
    a /= a.sum()
    return a


def make_multi_bump_init(n, n_bumps=3, seed=0):
    """Create a multi-bump initial sequence."""
    rng = np.random.default_rng(seed)
    x = np.linspace(0, 1, n)
    a = np.zeros(n)
    for _ in range(n_bumps):
        c = rng.uniform(0.2, 0.8)
        w = rng.uniform(0.05, 0.2)
        a += np.exp(-((x - c) / w) ** 2)
    a += 0.02 * rng.standard_normal(n)
    a = np.maximum(a, 0)
    a /= a.sum()
    return a
