"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FISTA (accelerated proximal gradient)
with log-sum-exp smoothing, symmetric constraint exploitation, fast FFT-based
convolution and gradient computation via stride tricks, multi-length exploration,
and active-set polishing. Key innovation: uses the spectral structure of the
convolution Hessian (bounded by 2/tau) to justify large step sizes, enabling
very fast convergence with Nesterov acceleration.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def proj_simplex(w):
    """Project w onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(w)
    if n == 1:
        return np.array([max(0.0, float(w[0]))])
    u = np.sort(w)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(w - theta, 0.0)


def compute_gradient_fast(a, c, tau):
    """Compute gradient of (1/tau)*log(sum_k exp(tau*c_k)) using stride tricks."""
    n = len(a)
    scaled = tau * c
    scaled -= scaled.max()
    w = np.exp(scaled)
    p = w / w.sum()
    # Build P[i,j] = p[i+j] via stride tricks (no copy)
    P = np.lib.stride_tricks.as_strided(p, (n, n), (p.strides[0], p.strides[0]))
    return 2.0 * (P @ a), p


def fista_optimize(a0, tau=50.0, max_iters=50000, use_sym=True,

    """
    FISTA (Nesterov accelerated proximal gradient) on the log-sum-exp smoothed

    if s < 1e-15:

    if use_sym:
        a = 0.5 * (a + a[::-1])
        a /= a.sum()

    a_prev = a.copy()
    t_k = 1.0

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    no_improve = 0

    alpha = alpha_factor * tau / 2.0

    for it in range(max_iters):
        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0

def multiplicative_weights_optimize(a0, eta0=1.0, eta_decay=0.9999,

        if ys < 1e-15:

            y = 0.5 * (y + y[::-1])
            y /= y.sum()

        c_y = fast_convolve(y)
        indices = np.argpartition(c_y, -top_k)[-top_k:]
        peek = c_y[indices]
        scaled = peek / tau
        scaled -= scaled.max()

        ws = w.sum()

        else:

            je = min(n - 1, k)

            no_improve += 1

        a = a_new
        t_k = t_new

def active_set_polish(a0, max_rounds=2000, verbose=False):
    """
    Active-set style polishing: identify the top peaks, and solve the
    KKT system approximately by iteratively equalizing the active peaks
    while maintaining the simplex constraint.
    """
    n = len(a0)
    a = np.maximum(a0, 0.0).astype(np.float64)
    a /= a.sum()

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    for rnd in range(max_rounds):
        # Find the top peaks
        num_peaks = max(3, min(50, n // 4))
        top_indices = np.argpartition(c, -num_peaks)[-num_peaks:]
        sort_order = np.argsort(c[top_indices])[::-1]
        top_indices = top_indices[sort_order]

        # Compute the current peak values
        peak_vals = c[top_indices]
        max_peak = peak_vals[0]

        # Subgradient at the max peak (equal weight on top peaks)
        weights = np.ones(num_peaks) / num_peaks

        sg = np.zeros(n)
        for idx in range(num_peaks):
            k = int(top_indices[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            sg[js:je + 1] += 2.0 * weights[idx] * a[k - je:k - js + 1][::-1]

        # Multiplicative update with small step
        step = 0.05
        log_update = -step * sg
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new

        c_new = fast_convolve(a_new)
        obj_new = c_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            c = c_new
        else:
            # Try smaller step
            for step2 in [0.02, 0.01, 0.005]:
                log_update = -step2 * sg
                log_update -= log_update.max()
                a_try = a * np.exp(log_update)
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                c_try = fast_convolve(a_try)
                if c_try.max() < best_obj - 1e-15:
                    best_obj = c_try.max()
                    best_a = a_try.copy()
                    a = a_try
                    c = c_try
                    break
            else:
                if verbose and rnd % 100 == 0:
                    print(f"      AS round {rnd}: obj={best_obj:.12f}")
                if rnd > 200:
                    break

    return best_a.copy(), best_obj


def coordinate_polish(a0, time_limit=60, verbose=False):
    """Random coordinate-wise multiplicative perturbations."""
    n = len(a0)
    a = np.maximum(a0, 0.0).astype(np.float64)
    a /= a.sum()

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    rng = np.random.default_rng(42)
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        n_perturb = max(10, n // 8)
        coords = rng.choice(n, size=n_perturb, replace=False)

        for i in coords:
            old_val = a[i]
            if old_val < 1e-15:
                continue
            for factor in [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]:
                a[i] = old_val * factor
                s2 = a.sum()
                if s2 < 1e-15:
                    a[i] = old_val
                    break
                a_norm = a / s2
                c2 = fast_convolve(a_norm)
                obj2 = c2.max()
                if obj2 < best_obj - 1e-14:
                    best_obj = obj2
                    best_a = a_norm.copy()
                    a = a_norm
                    no_improve = 0
                    break
                else:
                    a = best_a.copy()
                    break

        no_improve += 1
        if no_improve > 50:
            break

    return best_a.copy(), best_obj


def generate_inits(seed, n, parent=None):
    """Generate diverse initializations for length n."""
    rng = np.random.default_rng(seed)
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    inits.append(rc / rc.sum())

    # Gaussian centered
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())

    # Trapped (edge-concentrated)
    edge = np.minimum(x, n - 1 - x)
    tw = 1.0 / (edge + 2.0)
    inits.append(tw / tw.sum())

    # Exponential decay (both directions)
    for beta in [0.5, 1.5, 3.0]:
        e = np.exp(-beta * x / n)
        inits.append(e / e.sum())
        e2 = np.exp(-beta * (n - 1 - x) / n)
        inits.append(e2 / e2.sum())

    # Perturbed parent (if available)
    if parent is not None:
        p = parent
        if len(p) != n:
            p = np.interp(x / (n - 1), np.linspace(0, 1, len(p)), p)
        p = np.maximum(p, 0)
        p /= p.sum()
        inits.append(p)
        for scale in [0.005, 0.02, 0.05]:
            pert = p + rng.standard_normal(n) * scale
            pert = np.maximum(pert, 0)
            ps = pert.sum()
            if ps > 1e-10:
                inits.append(pert / ps)

    # Random Dirichlet
    for alpha in [0.5, 1.0, 2.0]:
        d = rng.dirichlet(np.ones(n) * alpha)
        inits.append(d)

    # Symmetric random
    half = n // 2
    for _ in range(3):
        h = rng.dirichlet(np.ones(half) * 1.0)
        sym = np.zeros(n)
        sym[:half] = h
        sym[half:] = h[::-1]
        if n % 2 == 1:
            sym[half] = h[half - 1]
        sym = np.maximum(sym, 0)
        ss = sym.sum()
        if ss > 1e-10:
            inits.append(sym / ss)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search: multi-phase optimization combining FISTA (large tau, fast
    convergence), multiplicative weights with momentum, active-set polishing,
    and coordinate search. Explores multiple sequence lengths and diverse
    initializations.
    """
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    best_seq = None
    best_score = float('inf')

    def try_seq(arr, label=""):
        nonlocal best_seq, best_score
        arr = np.maximum(arr, 0.0)
        if arr.sum() < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(arr.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = arr.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(arr)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = evaluate_sequence(parent.tolist())
            print(f"Parent: n={len(parent)}, score={parent_score:.8f}")
            if np.isfinite(parent_score):
                best_score = parent_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = np.ones(384) / 384
        print("No parent found, using uniform n=384")
        best_score = evaluate_sequence(parent.tolist())
        best_seq = parent.copy()

    print(f"Starting: best_score={best_score:.8f}")

    # === Phase 1: FISTA on parent length ===
    print("\n=== Phase 1: FISTA on parent ===")
    n_parent = len(parent)
    rem = deadline - time.monotonic()
    t1 = min(400, rem * 0.2)
    if t1 > 20:
        print(f"  FISTA n={n_parent}, alloc={t1:.0f}s")
        t_start = time.monotonic()
        a_opt, _ = fista_optimize(parent, tau=100.0, max_iters=100000,
                                  use_sym=True, alpha_factor=1.0,
                                  verbose=True, verbose_every=5000)
        try_seq(a_opt, "fista_p1")
        elapsed = time.monotonic() - t_start
        print(f"  FISTA p1 done in {elapsed:.1f}s, best={best_score:.8f}")

    # Continue FISTA with refined settings
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t2 = min(300, rem * 0.15)
        if t2 > 20:
            print(f"  FISTA refined n={len(best_seq)}, alloc={t2:.0f}s")
            a_opt, _ = fista_optimize(best_seq, tau=50.0, max_iters=200000,
                                      use_sym=True, alpha_factor=0.8,
                                      verbose=True, verbose_every=10000)
            try_seq(a_opt, "fista_p1b")

    # === Phase 2: Multiplicative weights with momentum ===
    print(f"\n=== Phase 2: MW with momentum ===")
    print(f"  Best so far: {best_score:.8f}")

    if best_seq is not None:
        rem = deadline - time.monotonic()
        configs = [
            {"eta0": 1.0, "eta_decay": 0.9998, "tau0": 0.01, "tau_decay": 0.9996},
            {"eta0": 2.0, "eta_decay": 0.9995, "tau0": 0.005, "tau_decay": 0.9994},
            {"eta0": 0.5, "eta_decay": 0.9999, "tau0": 0.02, "tau_decay": 0.9998},
        ]
        for ci, cfg in enumerate(configs):
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            t_alloc = min(180, rem / max(1, len(configs) - ci))
            if t_alloc < 15:
                break
            print(f"  MW config {ci}, alloc={t_alloc:.0f}s")
            a_opt, _ = multiplicative_weights_optimize(best_seq, use_sym=True, **cfg)
            try_seq(a_opt, f"mw_{ci}")

    # === Phase 3: Active set polishing ===
    print(f"\n=== Phase 3: Active set polish ===")
    print(f"  Best so far: {best_score:.8f}")

    if best_seq is not None:
        a_opt, _ = active_set_polish(best_seq, max_rounds=5000)
        try_seq(a_opt, "active_set")

    # === Phase 4: FISTA on different lengths ===
    print(f"\n=== Phase 4: Different lengths ===")
    print(f"  Best so far: {best_score:.8f}")

    lengths_to_try = []
    if best_seq is not None:
        n_base = len(best_seq)
        for nf in [0.5, 0.75, 1.5, 2.0, 0.6, 1.25, 1.5]:
            n_new = int(n_base * nf)
            n_new = max(64, min(2048, n_new))
            if n_new not in lengths_to_try and n_new != n_base:
                lengths_to_try.append(n_new)

    for n_try in lengths_to_try[:6]:
        rem = deadline - time.monotonic()
        if rem < 20:
            break
        t_alloc = min(60, rem / max(1, len(lengths_to_try)))
        if t_alloc < 10:
            break

        # Interpolate best_seq to new length
        x_old = np.linspace(0, 1, len(best_seq))
        x_new = np.linspace(0, 1, n_try)
        a_init = np.interp(x_new, x_old, best_seq)
        a_init = np.maximum(a_init, 0)
        a_init /= a_init.sum()

        a_opt, _ = fista_optimize(a_init, tau=50.0, max_iters=50000,
                                   use_sym=True, alpha_factor=1.0)
        try_seq(a_opt, f"fista_n{n_try}")

        # Follow up with MW
        if time.monotonic() < deadline - 30:
            a_opt2, _ = multiplicative_weights_optimize(a_opt, eta0=0.5,
                                                         eta_decay=0.9999,
                                                         tau0=0.01,
                                                         tau_decay=0.9997,
                                                         max_iters=20000,
                                                         use_sym=True)
            try_seq(a_opt2, f"mw_n{n_try}")

    # === Phase 5: Multi-init exploration on best length ===
    print(f"\n=== Phase 5: Multi-init ===")
    print(f"  Best so far: {best_score:.8f}")

    if best_seq is not None:
        n_best = len(best_seq)
        inits = generate_inits(seed + 100, n_best, parent=best_seq)
        rng = np.random.default_rng(seed)

        for i, init in enumerate(inits):
            rem = deadline - time.monotonic()
            if rem < 25:
                break
            t_alloc = min(40, rem / max(1, len(inits) - i))
            if t_alloc < 8:
                break

            a_opt, _ = fista_optimize(init, tau=50.0, max_iters=30000,
                                       use_sym=True, alpha_factor=1.0)
            try_seq(a_opt, f"init_fista_{i}")

            if time.monotonic() < deadline - 15:
                a_opt2, _ = multiplicative_weights_optimize(a_opt, eta0=1.0,
                                                             eta_decay=0.9998,
                                                             tau0=0.005,
                                                             tau_decay=0.9995,
                                                             max_iters=10000,
                                                             use_sym=True)
                try_seq(a_opt2, f"init_mw_{i}")

    # === Phase 6: Coordinate polish ===
    print(f"\n=== Phase 6: Coordinate polish ===")
    print(f"  Best so far: {best_score:.8f}")

    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 10:
            t_alloc = min(rem * 0.3, 120)
            a_opt, _ = coordinate_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "coord_p1")

    # === Phase 7: Final FISTA fine-tune ===
    print(f"\n=== Phase 7: Final fine-tune ===")
    print(f"  Best so far: {best_score:.8f}")

    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 20:
            t_alloc = min(rem * 0.3, 180)
            a_opt, _ = fista_optimize(best_seq, tau=200.0, max_iters=200000,
                                       use_sym=True, alpha_factor=0.5,
                                       verbose=True, verbose_every=20000)
            try_seq(a_opt, "fista_fine")

        # Active set again
        if time.monotonic() < deadline - 10:
            a_opt, _ = active_set_polish(best_seq, max_rounds=5000)
            try_seq(a_opt, "active_final")

    # === Phase 8: Final coordinate polish ===
    print(f"\n=== Phase 8: Final coordinate polish ===")
    print(f"  Best so far: {best_score:.8f}")

    if best_seq is not None:
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_opt, _ = coordinate_polish(best_seq, time_limit=min(rem, 60))
            try_seq(a_opt, "coord_final")

    # === Phase 9: One more round of FISTA + MW on final best ===
    if best_seq is not None:
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            a_opt, _ = fista_optimize(best_seq, tau=100.0, max_iters=100000,
                                       use_sym=True, alpha_factor=0.7)
            try_seq(a_opt, "fista_veryfine")

        rem = deadline - time.monotonic() - 3
        if rem > 5:
            a_opt, _ = multiplicative_weights_optimize(best_seq, eta0=0.3,
                                                        eta_decay=0.99995,
                                                        tau0=0.002,
                                                        tau_decay=0.9999,
                                                        max_iters=20000,
                                                        use_sym=True)
            try_seq(a_opt, "mw_veryfine")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
