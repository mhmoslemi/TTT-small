                if f_val < best_f:
                    best_f = f_val
                    best_t = t_c

            # Golden section refinement around best_t
            if best_t != v:
                lo = max(0.0, best_t * 0.5)
                hi = best_t * 1.5
                if lo == 0.0 and hi == 0.0:
                    hi = mean_val
                for _ in range(8):
                    m1 = hi - golden_ratio * (hi - lo)
                    m2 = lo + golden_ratio * (hi - lo)

                    for t_c in [m1, m2]:
                        d_c = t_c - v
                        new_S = S_old + d_c
                        if new_S < 1e-12:
                            continue
                        max_c = (conv + d_c * L).max()
                        if has_quad:
                            val_q = conv[q_idx] + d_c * L[q_idx] + d_c * d_c
                            if val_q > max_c:
                                max_c = val_q
                        f_val = 2.0 * n * max_c / (new_S * new_S)
                        if f_val < best_f:
                            best_f = f_val
                            best_t = t_c

                    # Update bracket
                    f_m1_num = 2.0 * n * (conv + (m1 - v) * L).max()
                    if has_quad:
                        vq = conv[q_idx] + (m1 - v) * L[q_idx] + (m1 - v) ** 2
                        if vq > f_m1_num:
                            f_m1_num = vq
                    f_m2_num = 2.0 * n * (conv + (m2 - v) * L).max()
                    if has_quad:
                        vq = conv[q_idx] + (m2 - v) * L[q_idx] + (m2 - v) ** 2
                        if vq > f_m2_num:
                            f_m2_num = vq
                    S1 = S_old + (m1 - v)
                    S2 = S_old + (m2 - v)
                    if S1 < 1e-12 or S2 < 1e-12:
                        break
                    if f_m1_num / (S1 * S1) > f_m2_num / (S2 * S2):
                        lo = m1
                    else:
                        hi = m2

            # Apply update
            if best_t != v and best_f < best_obj:
                d_final = best_t - v
                a[i] = best_t
                S = S_old + d_final
                conv = conv + d_final * L
                if has_quad:
                    conv[q_idx] += d_final * d_final
                best_obj = best_f
                best_a = a.copy()
                best_S = S
                improved = True

        sweep += 1
        if verbose and sweep % 200 == 0:
            print(f"      CD sweep {sweep}: obj={best_obj:.8f}")

        if not improved:
            no_improve_sweeps += 1
            if no_improve_sweeps > 20:
                # Restart with perturbation
                a = np.maximum(best_a + rng.standard_normal(n) * (0.01 * best_S / n), 0.0)
                S = a.sum()
                conv = fast_convolve(a)
                no_improve_sweeps = 0
                best_obj = 2.0 * n * conv.max() / (S * S)
                if best_obj > 2.0 * n * fast_convolve(best_a).max() / (best_S * best_S) + 1e-10:
                    a = best_a.copy()
                    S = best_S
                    conv = fast_convolve(a)
                    best_obj = 2.0 * n * conv.max() / (S * S)
        else:
            no_improve_sweeps = 0

    # Final normalization
    final_S = best_a.sum()
    if final_S > 1e-15:
        best_a = best_a / final_S
        obj = 2.0 * n * fast_convolve(best_a).max()
    else:
        obj = float('inf')

    if obj < best_obj:
        best_obj = obj

    return best_a, best_obj


def mult_weight_refine(a, time_limit, rng, verbose=False):
    """
    Multiplicative weight (mirror descent) optimization.
    Naturally preserves positivity via exponential updates.
    """
    f = np.maximum(a.astype(np.float64), 0.0)
    S = f.sum()
    if S < 1e-15:
        f = np.ones(len(f)) / len(f)
        S = 1.0
    n = len(f)

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max() / (S * S)
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = 0.03
    cur_temp = 0.0003
    top_k = max(10, n // 20)

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        S = f.sum()
        if S < 1e-15:
            break

        # Select top-k peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        # Softmax weights
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute gradient signal
        log_update = np.zeros(n)
        for pi in range(len(peaks)):
            if w[pi] < 1e-14:
                continue
            k = int(peaks[pi])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_contrib = 2.0 * w[pi] * f[k - idx]
            log_update[idx] -= cur_lr * grad_contrib

        # Apply multiplicative update
        f = f * np.exp(log_update)
        f = np.maximum(f, 1e-25)
        f = f / f.sum()

        conv = fast_convolve(f)
        obj = 2.0 * n * conv.max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9997
        cur_temp *= 1.001

        if no_improve > 100:
            f = best_f * np.exp(rng.standard_normal(n) * 0.005)
            f = f / f.sum()
            cur_lr = max(0.01, cur_lr * 1.5)
            cur_temp = 0.0005
            no_improve = 0

    conv = fast_convolve(best_f)
    obj = 2.0 * n * conv.max()
    if obj < best_obj:
        best_obj = obj

    return best_f, best_obj


def gen_inits(n, parent=None, rng=None):
    """Generate diverse initializations for length n."""
    if rng is None:
        rng = np.random.default_rng(0)
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    x = np.arange(n, dtype=np.float64)
    # Edge-concentrated
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits.append(g / g.sum())

    # Cosine
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.05
    inits.append(c1 / c1.sum())

    # Two-peak (edges)
    peaks = np.zeros(n)
    peaks[0] = 1.0
    peaks[n - 1] = 1.0
    peaks[n // 4] = 0.5
    peaks[3 * n // 4] = 0.5
    inits.append(peaks / peaks.sum())

    # Exponential decay from edges
    e = np.exp(-x / (n * 0.3)) + np.exp(-(n - 1 - x) / (n * 0.3))
    inits.append(e / e.sum())

    # Periodic
    p1 = 0.5 + 0.3 * np.cos(2 * np.pi * 2 * x / n)
    p1 = np.maximum(p1, 0.01)
    inits.append(p1 / p1.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits


def try_and_track(seq, best_state, label=""):
    """Try a sequence and update best state if improved."""
    a = np.maximum(np.asarray(seq, dtype=np.float64), 0.0)
    S = a.sum()
    if S < 1e-15 or len(a) == 0:
        return best_state, float('inf')
    a = a / S
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_state[1]:
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return (a.copy(), score)
    return best_state, score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search:
    Phase 1: Fast coordinate descent on parent and multiple inits (~400s)
    Phase 2: Multi-length exploration with CD (~400s)
    Phase 3: MW refinement (~250s)
    Phase 4: More CD with restarts (~400s)
    Phase 5: Final polish (~150s)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(10.0, float(budget_s) - 15.0)

    best_state = (None, float('inf'))

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}")
            best_state, _ = try_and_track(parent, best_state, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(384) / 384
        best_state, _ = try_and_track(parent, best_state, "uniform_384")

    # === Phase 1: Coordinate descent on best (~400s) ===
    print("\n=== Phase 1: Coordinate Descent ===")
    rem = deadline - time.monotonic()
    t1 = min(350, rem * 0.25)
    n_best = len(best_state[0])
    print(f"  CD on n={n_best}, t={t1:.0f}s")
    a_opt, obj = coord_descent(best_state[0], time_limit=t1, rng=rng, verbose=True)
    best_state, _ = try_and_track(a_opt, best_state, "cd_phase1")

    # Try a few other inits briefly
    for ii in range(3):
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        inits = gen_inits(n_best, parent, rng)
        t_alloc = min(rem * 0.1, 60)
        if t_alloc < 5:
            break
        print(f"  CD on init {ii}, t={t_alloc:.0f}s")
        a_opt, obj = coord_descent(inits[ii + 1], time_limit=t_alloc, rng=rng)
        best_state, _ = try_and_track(a_opt, best_state, f"cd_init_{ii}")

    print(f"  Phase 1 best: {best_state[1]:.8f}")

    # === Phase 2: Multi-length exploration (~400s) ===
    print("\n=== Phase 2: Multi-Length Exploration ===")
    lengths_to_try = [256, 320, 384, 448, 512]
    rem = deadline - time.monotonic()
    t_per_len = min(rem * 0.22, 80)

    for nl in lengths_to_try:
        rem = deadline - time.monotonic()
        if rem < 20:
            break
        n_best = len(best_state[0])
        if nl == n_best:
            # Already tried, do another CD pass
            t_alloc = min(t_per_len, rem * 0.3)
        else:
            # Interpolate parent to new length
            p_interp = np.interp(np.linspace(0, 1, nl), np.linspace(0, 1, n_best), best_state[0])
            p_interp = np.maximum(p_interp, 0)
            ps = p_interp.sum()
            if ps > 1e-10:
                p_interp = p_interp / ps
            t_alloc = min(t_per_len, rem * 0.3)

        if t_alloc < 5:
            break
        print(f"  CD on n={nl}, t={t_alloc:.0f}s")
        if nl == n_best:
            a_start = best_state[0].copy()
        else:
            a_start = p_interp
        a_opt, obj = coord_descent(a_start, time_limit=t_alloc, rng=rng, verbose=True)
        best_state, _ = try_and_track(a_opt, best_state, f"cd_n{nl}")

    print(f"  Phase 2 best: {best_state[1]:.8f}")

    # === Phase 3: MW Refinement (~250s) ===
    print("\n=== Phase 3: MW Refinement ===")
    rem = deadline - time.monotonic()
    t3 = min(250, rem * 0.2)

    mw_configs = [
        {'lr_init': 0.03, 'top_k': 15, 'temp_init': 0.0003},
        {'lr_init': 0.05, 'top_k': 20, 'temp_init': 0.0005},
        {'lr_init': 0.02, 'top_k': 30, 'temp_init': 0.0002},
    ]

    for ci, cfg in enumerate(mw_configs):
        rem = deadline - time.monotonic()
        if rem < 15:
            break
        t_alloc = min(t3 / len(mw_configs), rem * 0.4)
        print(f"  MW config {ci}, t={t_alloc:.0f}s")
        a_opt, obj = mult_weight_refine(best_state[0], time_limit=t_alloc, rng=rng)
        best_state, _ = try_and_track(a_opt, best_state, f"mw_{ci}")

        # Follow up with CD
        rem = deadline - time.monotonic()
        if rem > 10:
            t_cd = min(30, rem * 0.3)
            a_opt2, obj2 = coord_descent(a_opt, time_limit=t_cd, rng=rng)
            best_state, _ = try_and_track(a_opt2, best_state, f"mw{ci}_cd")

    print(f"  Phase 3 best: {best_state[1]:.8f}")

    # === Phase 4: Extended CD with random restarts (~400s) ===
    print("\n=== Phase 4: Extended CD with Restarts ===")
    rem = deadline - time.monotonic()
    t4 = min(400, rem * 0.35)
    n_best = len(best_state[0])

    # Multiple CD runs with different restart strategies
    for ri in range(5):
        rem = deadline - time.monotonic()
        if rem < 15:
            break
        t_alloc = min(t4 / 5, rem * 0.4)

        if ri == 0:
            a_start = best_state[0].copy()
        else:
            noise_scale = 0.003 + 0.003 * ri
            a_pert = best_state[0] * np.exp(rng.standard_normal(n_best) * noise_scale)
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            a_start = a_pert

        print(f"  CD restart {ri}, t={t_alloc:.0f}s")
        a_opt, obj = coord_descent(a_start, time_limit=t_alloc, rng=rng, verbose=True)
        best_state, _ = try_and_track(a_opt, best_state, f"cd_restart_{ri}")

    print(f"  Phase 4 best: {best_state[1]:.8f}")

    # === Phase 5: Final polish (~150s) ===
    print("\n=== Phase 5: Final Polish ===")
    rem = deadline - time.monotonic()

    # CD with very fine control
    if rem > 30:
        t5 = min(80, rem * 0.5)
        print(f"  Final CD, t={t5:.0f}s")
        a_opt, obj = coord_descent(best_state[0], time_limit=t5, rng=rng, verbose=True)
        best_state, _ = try_and_track(a_opt, best_state, "final_cd")

    # Tiny perturbations
    rem = deadline - time.monotonic()
    if rem > 20:
        for _ in range(5):
            rem = deadline - time.monotonic()
            if rem < 5:
                break
            noise = rng.standard_normal(n_best) * 0.001
            a_pert = np.maximum(best_state[0] + noise, 0)
            a_pert = a_pert / a_pert.sum()
            t_alloc = min(10, rem * 0.5)
            a_opt, obj = coord_descent(a_pert, time_limit=t_alloc, rng=rng)
            best_state, _ = try_and_track(a_opt, best_state, "perturb")

    # Final MW with very small lr
    rem = deadline - time.monotonic()
    if rem > 10:
        print(f"  Final MW, t={min(rem-5, 30):.0f}s")
        a_opt, obj = mult_weight_refine(best_state[0], time_limit=min(rem - 5, 30), rng=rng)
        best_state, _ = try_and_track(a_opt, best_state, "final_mw")

    # Fallback
    if best_state[0] is None:
        best_seq = np.ones(384) / 384
        best_state = (best_seq, float(evaluate_sequence(best_seq.tolist())))

    # Final verification
    final_seq = best_state[0]
    final_score = float(evaluate_sequence(final_seq.tolist()))
    if np.isfinite(final_score) and final_score < best_state[1]:
        best_state = (final_seq, final_score)

    print(f"\nFINAL SCORE: {best_state[1]:.8f} (n={len(best_state[0])})")
    return [float(x) for x in best_state[0]]
