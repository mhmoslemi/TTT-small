import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def projection_simplex(v, sum_target=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = sum_target}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u * np.arange(1, n + 1) - (cssv - sum_target)) > 0
    if not cond.any():
        return np.full(n, sum_target / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - sum_target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

= 2 * Σ_{m=0}^{n-1} w[j+m] * p[m] (where j+m ≤ 2n-2)

This is exactly the cross-correlation of w and p at lag j:
(w ⋆ p)[j] = Σ_m w[j+m] p[m]

In FFT terms, the cross-correlation can be computed as:
ifft(conj(FFT(w, N)) * FFT(p, N))

where N ≥ (2n-1) + n - 1 = 3n-2 for the linear cross-correlation.

Actually, let me be more careful. For linear cross-correlation:
(x ⋆ y)[m] = Σ_n x[n+m] y[n]

If x has length Lx and y has length Ly, then the result has length Lx + Ly - 1.
For FFT: we need N ≥ Lx + Ly - 1.
ifft(conj(FFT(x, N)) * FFT(y, N)) gives the circular cross-correlation, which equals the linear cross-correlation when N ≥ Lx + Ly - 1.

In our case, Lx = 2n-1 (for w) and Ly = n (for p).
So we need N ≥ 3n-2.

Let me use N = next power of 2 ≥ 3n-2.

Actually wait, I think there might be an issue with the FFT approach for cross-correlation. Let me use a simpler approach:

g[j] = 2 * Σ_{m=0}^{min(n-1, 2n-2-j)} w[j+m] * p[m]

This is a sum over m for each j. For all j, this is a convolution-like operation.

Actually, let me think of it differently. Define w' as the reversed w: w'[k] = w[2n-2-k].
Then (w ⋆ p)[j] = Σ_m w[j+m] p[m] = Σ_m w'[2n-2-j-m] p[m] = (w' * p)[2n-2-j]

where * is convolution. So we can compute this via FFT convolution!

(conv(w', p))[2n-2-j] = Σ_m w'[2n-2-j-m] p[m] = Σ_m w[j+m] p[m] ✓

So:
1. Reverse w: w' = w[::-1]
2. Compute conv = fft_convolve(w', p) (this gives a result of length (2n-1) + n - 1 = 3n-2)
3. g[j] = 2 * conv[2n-2-j] for j = 0, ..., n-1

w' = [w4, w3, w2, w1, w0]

conv(w', p) has length 5+3-1 = 7:

conv[5] = w1*p2 + w0*p1
conv[6] = w0*p2

g[j] = 2 * conv[2n-2-j] = 2 * conv[4-j]:
g[0] = 2 * conv[4] = 2*(w2*p2 + w1*p1 + w0*p0)
g[1] = 2 * conv[3] = 2*(w3*p2 + w2*p1 + w1*p0)
g[2] = 2 * conv[2] = 2*(w4*p2 + w3*p1 + w2*p0)

Check: g[0] should be 2 * Σ_k w[k] * p[k-0] = 2 * (w0*p0 + w1*p1 + w2*p2 + w3*p3 + w4*p4)

g[1] should be 2 * Σ_k w[k] * p[k-1] = 2*(w1*p0 + w2*p1 + w3*p2) ✓

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smoothed max descent
with FFT-based gradient computation, adaptive temperature, momentum,
and diverse structured initializations.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def projection_simplex(v, sum_target=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = sum_target}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u * np.arange(1, n + 1) - (cssv - sum_target)) > 0
    if not cond.any():
        return np.full(n, sum_target / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - sum_target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_and_conv(p, tau, top_k=None):
    """Compute gradient of smoothed max of conv(p,p) and the convolution itself."""
    n = len(p)
    c = fast_convolve(p)
    
    # Compute softmax weights
    if top_k is not None and top_k < len(c):
        peaks = np.argpartition(c, -top_k)[-top_k:]
        w = np.zeros(len(c))
        pv = c[peaks]
        scaled = pv / tau
        scaled -= scaled.max()
        wpeaks = np.exp(scaled)
        wsum = wpeaks.sum()
        if wsum > 1e-30:
            w[peaks] = wpeaks / wsum
        else:
            w[np.argmax(pv)] = 1.0
    else:
        scaled = c / tau
        scaled -= scaled.max()
        w = np.exp(scaled)
        wsum = w.sum()
        if wsum > 1e-30:
            w /= wsum
        else:
            w[np.argmax(c)] = 1.0
    
    # Compute gradient via cross-correlation
    # g[j] = 2 * sum_m w[j+m] * p[m]
    # = 2 * conv(w_reversed, p)[2n-2-j]
    w_rev = w[::-1]
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    WR = np.fft.rfft(w_rev, fft_size)
    P = np.fft.rfft(p, fft_size)
    conv_wp = np.fft.irfft(WR * P, fft_size)
    
    # g[j] = 2 * conv_wp[2n-2-j] for j = 0, ..., n-1
    grad = 2.0 * conv_wp[2*n-2 : n-1 : -1]
    
    return grad, c


def smooth_max_descent(p_init, time_limit, tau_init=0.005, tau_decay=0.999,
                       lr_init=0.01, momentum=0.9, top_k=None,
                       use_adaptive_lr=True, verbose=False):
    """Smoothed max descent on the simplex with momentum."""
    p = np.maximum(p_init.astype(np.float64), 0.0)
    s = p.sum()
    if s < 1e-15:
        p = np.ones(len(p)) / len(p)
    else:
        p = p / s
    
    n = len(p)
    if top_k is None:
        top_k = max(32, n // 4)
    
    tau = tau_init
    lr = lr_init
    
    c = fast_convolve(p)
    best_obj = c.max()
    best_p = p.copy()
    
    v = np.zeros(n)  # momentum
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break
        
        grad, c = compute_grad_and_conv(p, tau, top_k)
        
        # Momentum update
        v = momentum * v - lr * grad
        p_new = projection_simplex(p + v)
        
        # Evaluate
        c_new = fast_convolve(p_new)
        obj_new = c_new.max()
        
        if use_adaptive_lr and obj_new > best_obj * 1.01:
            # Step too large, reduce and retry
            lr *= 0.5
            v = momentum * v - lr * grad
            p_new = projection_simplex(p + v)
            c_new = fast_convolve(p_new)
            obj_new = c_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_p = p_new.copy()
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"    it={it}, obj={best_obj:.8f}, tau={tau:.6f}, lr={lr:.6f}")
        else:
            no_improve += 1
        
        p = p_new
        c = c_new
        
        # Decay parameters
        tau = max(tau_decay * tau, 1e-6)
        if no_improve == 0:
            lr *= 1.001  # slight increase on success
        else:
            lr *= 0.999
        
        # Reset if stuck
        if no_improve > 100:
            rng = np.random.default_rng(it * 7919 + 13)
            perturb = 0.001 + 0.003 * (no_improve / 500.0)
            p = projection_simplex(best_p + rng.standard_normal(n) * perturb)
            c = fast_convolve(p)
            v = np.zeros(n)
            no_improve = 0
            lr = lr_init * 0.5
            tau = max(tau_init * 0.5, 0.001)
    
    return best_p, best_obj


def generate_inits(n, parent=None, rng=None):
    """Generate diverse structured initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Cosine window (raised)
    cw = 0.5 * (1 + np.cos(2 * np.pi * x / n)) + 0.1
    inits.append(cw / cw.sum())
    
    # Gaussian
    center = n / 2
    g = np.exp(-(x - center)**2 / (2 * (n/4)**2)) + 0.01
    inits.append(g / g.sum())
    
    # U-shaped (inverse Gaussian)
    ig = 1.0 / (np.exp(-(x - center)**2 / (2 * (n/4)**2)) + 0.01)
    inits.append(ig / ig.sum())
    
    # Linear ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits.append(ramp / ramp.sum())
    
    # Triangular
    tri = np.minimum(x + 1, n - x)
    inits.append(tri / tri.sum())
    
    # Power law
    pw = (x + 1) ** 0.5
    inits.append(pw / pw.sum())
    
    # Sine
    sn = np.sin(np.pi * (x + 1) / (n + 1)) + 0.1
    inits.append(sn / sn.sum())
    
    # Raised cosine (Hann)
    hann = 0.5 * (1 - np.cos(2 * np.pi * (x + 0.5) / n))
    inits.append(hann / hann.sum())
    
    # Blackman
    blackman = 0.42 - 0.5 * np.cos(2 * np.pi * (x + 0.5) / n) + 0.08 * np.cos(4 * np.pi * (x + 0.5) / n)
    blackman = np.maximum(blackman, 0.01)
    inits.append(blackman / blackman.sum())
    
    # Flat top
    flat = 0.2156 - 0.416 * np.cos(2 * np.pi * (x + 0.5) / n) + 0.278 * np.cos(4 * np.pi * (x + 0.5) / n) - 0.084 * np.cos(6 * np.pi * (x + 0.5) / n) + 0.0087 * np.cos(8 * np.pi * (x + 0.5) / n)
    flat = np.maximum(flat, 0.01)
    inits.append(flat / flat.sum())
    
    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    
    # Parent-based
    if parent is not None:
        parent_norm = parent / parent.sum()
        inits.append(parent_norm.copy())
        # Slightly perturbed parent
        inits.append(projection_simplex(parent_norm + rng.standard_normal(n) * 0.01))
        # Blend with uniform
        inits.append(0.7 * parent_norm + 0.3 * (np.ones(n) / n))
        # Blend with cosine
        cw2 = 0.5 * (1 + np.cos(2 * np.pi * x / n)) + 0.1
        inits.append(0.6 * parent_norm + 0.4 * cw2 / cw2.sum())
    
    return inits


def local_search(p_init, time_limit, step_size=0.001, n_trials=5000):
    """Simple local search: try small perturbations and keep improvements."""
    p = np.maximum(p_init.astype(np.float64), 0.0)
    s = p.sum()
    if s < 1e-15:
        p = np.ones(len(p)) / len(p)
    else:
        p = p / s
    
    n = len(p)
    c = fast_convolve(p)
    best_obj = c.max()
    best_p = p.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        # Random perturbation
        idx = rng.integers(0, n, size=min(10, n))
        delta = rng.standard_normal(len(idx)) * step_size
        p_new = p.copy()
        p_new[idx] += delta
        p_new = projection_simplex(p_new)
        
        c_new = fast_convolve(p_new)
        obj_new = c_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_p = p_new.copy()
            p = p_new
            step_size *= 1.01
        else:
            step_size *= 0.99
    
    return best_p, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smoothed max descent
with FFT-based gradient computation, adaptive temperature cooling,
momentum, and diverse structured initializations.
The key innovation is using a smooth approximation (log-sum-exp) of the
non-smooth max in the auto-convolution, enabling efficient gradient-based
optimization with O(n log n) per iteration via FFT cross-correlation.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def projection_simplex(v, sum_target=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = sum_target}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u * np.arange(1, n + 1) - (cssv - sum_target)) > 0
    if not cond.any():
        return np.full(n, sum_target / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - sum_target) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad_conv(p, tau, top_k=None):
    """Compute gradient of tau*log(sum(exp(c/tau))) and the convolution."""
    n = len(p)
    c = fast_convolve(p)

    if top_k is not None and top_k < len(c) and top_k > 0:
        peaks = np.argpartition(c, -top_k)[-top_k:]
        w = np.zeros(len(c))
        pv = c[peaks]
        scaled = pv / tau
        scaled -= scaled.max()
        wpeaks = np.exp(scaled)
        wsum = wpeaks.sum()
        if wsum > 1e-30:
            w[peaks] = wpeaks / wsum
        else:
            w[np.argmax(pv)] = 1.0
    else:
        scaled = c / tau
        scaled -= scaled.max()
        w = np.exp(scaled)
        wsum = w.sum()
        if wsum > 1e-30:
            w /= wsum
        else:
            w[np.argmax(c)] = 1.0

    # Gradient: g[j] = 2 * sum_m w[j+m] * p[m]
    # Computed via: conv(w_reversed, p)[2n-2-j]
    w_rev = w[::-1]
    fft_size = 1 << (3 * n - 2).bit_length()
    WR = np.fft.rfft(w_rev, fft_size)
    P = np.fft.rfft(p, fft_size)
    conv_wp = np.fft.irfft(WR * P, fft_size)

    grad = 2.0 * conv_wp[2 * n - 2: n - 1: -1]
    return grad, c


def smooth_max_descent(p_init, time_limit, tau_init=0.005, tau_decay=0.999,
                       lr_init=0.005, momentum=0.9, top_k=None,
                       adapt_lr=True, restart_every=150):
    """Smoothed max descent on the simplex with momentum and adaptive LR."""
    p = np.maximum(p_init.astype(np.float64), 0.0)
    s = p.sum()
    if s < 1e-15:
        p = np.ones(len(p)) / len(p)
    else:
        p = p / s

    n = len(p)
    if top_k is None:
        top_k = max(32, n // 5)

    tau = tau_init
    lr = lr_init

    c = fast_convolve(p)
    best_obj = c.max()
    best_p = p.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        grad, c = compute_grad_conv(p, tau, top_k)

        v = momentum * v - lr * grad
        p_new = projection_simplex(p + v)

        c_new = fast_convolve(p_new)
        obj_new = c_new.max()

        if adapt_lr and obj_new > best_obj * 1.05:
            lr *= 0.5
            v = momentum * v - lr * grad
            p_new = projection_simplex(p + v)
            c_new = fast_convolve(p_new)
            obj_new = c_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_p = p_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        p = p_new
        c = c_new

        tau = max(tau * tau_decay, 1e-7)
        if no_improve == 0:
            lr = min(lr * 1.002, lr_init * 2.0)
        else:
            lr *= 0.998

        if no_improve > restart_every:
            rng = np.random.default_rng(it * 7919 + 13 + n)
            perturb_scale = 0.001 + 0.005 * min(1.0, no_improve / (restart_every * 3.0))
            p = projection_simplex(best_p + rng.standard_normal(n) * perturb_scale)
            c = fast_convolve(p)
            v = np.zeros(n)
            no_improve = 0
            lr = lr_init * 0.4
            tau = max(tau_init * 0.5, 0.002)

    return best_p, best_obj


def smooth_max_descent_nesterov(p_init, time_limit, tau_init=0.003, tau_decay=0.9995,
                                 lr_init=0.003, momentum=0.91, top_k=None):
    """Nesterov-accelerated smoothed max descent."""
    p = np.maximum(p_init.astype(np.float64), 0.0)
    s = p.sum()
    if s < 1e-15:
        p = np.ones(len(p)) / len(p)
    else:
        p = p / s

    n = len(p)
    if top_k is None:
        top_k = max(32, n // 4)

    tau = tau_init
    lr = lr_init

    c = fast_convolve(p)
    best_obj = c.max()
    best_p = p.copy()

    y = p.copy()
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    beta = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.05:
            break

        beta = (1.0 + np.sqrt(1.0 + 4.0 * beta * beta)) / 2.0
        alpha = (1.0 - 1.0 / beta) / momentum

        new_y = p + (p - y) * alpha
        y = projection_simplex(new_y)

        grad, c = compute_grad_conv(y, tau, top_k)
        v = momentum * v + grad
        p_new = projection_simplex(y - lr * v)

        c_new = fast_convolve(p_new)
        obj_new = c_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_p = p_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 50 == 0:
                lr *= 0.95

        p = p_new
        c = c_new
        tau = max(tau * tau_decay, 1e-7)

        if no_improve > 200:
            rng = np.random.default_rng(it * 31 + 7 + n)
            p = projection_simplex(best_p + rng.standard_normal(n) * 0.003)
            y = p.copy()
            v = np.zeros(n)
            c = fast_convolve(p)
            no_improve = 0
            lr = lr_init
            beta = 1.0
            tau = max(tau_init, 0.002)

    return best_p, best_obj


def peak_directed_descent(p_init, time_limit, n_iters_per_peak=200):
    """Directly target and reduce the highest peaks in the convolution."""
    p = np.maximum(p_init.astype(np.float64), 0.0)
    s = p.sum()
    if s < 1e-15:
        p = np.ones(len(p)) / len(p)
    else:
        p = p / s

    n = len(p)
    c = fast_convolve(p)
    best_obj = c.max()
    best_p = p.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.1:
            break

        c = fast_convolve(p)
        peak_val = c.max()

        if peak_val < best_obj - 1e-15:
            best_obj = peak_val
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Find top contributing pairs for the peak
        peak_idx = np.argmax(c)
        # c[peak_idx] = sum_i p[i] * p[peak_idx - i]
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        idx = np.arange(js, je + 1)
        products = p[idx] * p[peak_idx - idx]

        # Find the largest contributing pairs
        n_top = min(20, len(idx))
        top_local = np.argpartition(products, -n_top)[-n_top:]
        top_pairs = idx[top_local]

        # Reduce the larger element of each top pair, add to a "safe" position
        # Find positions that don't contribute to the peak
        safe_mask = np.ones(n, dtype=bool)
        for j in top_pairs:
            k = peak_idx - j
            if 0 <= k < n:
                safe_mask[j] = False
                safe_mask[k] = False

        if safe_mask.any():
            safe_pos = np.where(safe_mask)[0]
            # Redistribute: reduce top pairs, add to random safe positions
            total_reduce = 0.0
            for j in top_pairs:
                k = peak_idx - j
                if 0 <= k < n:
                    reduce_amt = 0.3 * min(p[j], p[k])
                    reduce_amt = min(reduce_amt, p[j] * 0.5, p[k] * 0.5)
                    p[j] -= reduce_amt * (p[j] / (p[j] + p[k] + 1e-30))
                    p[k] -= reduce_amt * (p[k] / (p[j] + p[k] + 1e-30))
                    total_reduce += reduce_amt

            if total_reduce > 1e-10:
                n_add = len(safe_pos)
                add_amt = total_reduce / n_add
                p[safe_pos] += add_amt
                p = np.maximum(p, 0.0)
                p = p / p.sum()

        if no_improve > 100:
            rng = np.random.default_rng(it * 13 + n)
            p = projection_simplex(best_p + rng.standard_normal(n) * 0.002)
            no_improve = 0

    return best_p, best_obj


def coordinate_descent_polish(p_init, time_limit, n_coords=50):
    """Coordinate descent: try adjusting individual (or small groups of) coordinates."""
    p = np.maximum(p_init.astype(np.float64), 0.0)
    s = p.sum()
    if s < 1e-15:
        p = np.ones(len(p)) / len(p)
    else:
        p = p / s

    n = len(p)
    c = fast_convolve(p)
    best_obj = c.max()
    best_p = p.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        # Pick a random coordinate and try to adjust it
        j = rng.integers(0, n)
        # Try moving mass from j to a random other position
        k = rng.integers(0, n)
        if j == k:
            continue

        # Try several transfer amounts
        max_transfer = p[j] * 0.5
        for frac in [0.1, 0.25, 0.5, 0.75]:
            transfer = frac * max_transfer
            if transfer < 1e-15:
                continue
            p_try = p.copy()
            p_try[j] -= transfer
            p_try[k] += transfer
            p_try = np.maximum(p_try, 0.0)
            p_try = p_try / p_try.sum()

            c_try = fast_convolve(p_try)
            if c_try.max() < best_obj - 1e-15:
                best_obj = c_try.max()
                best_p = p_try.copy()
                p = p_try
                c = c_try
                break

        if (it := 0) == 0:
            pass

    return best_p, best_obj


def generate_inits(n, parent=None, rng=None):
    """Generate diverse structured initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Raised cosine
    cw = 0.5 * (1 + np.cos(2 * np.pi * x / n)) + 0.1
    inits.append(cw / cw.sum())

    # Gaussian (centered)
    center = n / 2.0
    g = np.exp(-(x - center) ** 2 / (2 * (n / 4.0) ** 2)) + 0.01
    inits.append(g / g.sum())

    # U-shaped (inverse of Gaussian)
    ig = 1.0 / (np.exp(-(x - center) ** 2 / (2 * (n / 4.0) ** 2)) + 0.05)
    inits.append(ig / ig.sum())

    # Linear ramp
    ramp = np.arange(1, n + 1, dtype=np.float64)
    inits.append(ramp / ramp.sum())

    # Triangular
    tri = np.minimum(x + 1, n - x)
    inits.append(tri / tri.sum())

    # Power law
    pw = (x + 1) ** 0.5
    inits.append(pw / pw.sum())

    # Sine
    sn = np.sin(np.pi * (x + 1) / (n + 1)) + 0.1
    inits.append(sn / sn.sum())

    # Hann window
    hann = 0.5 * (1 - np.cos(2 * np.pi * (x + 0.5) / n))
    inits.append(hann / hann.sum())

    # Blackman
    bm = 0.42 - 0.5 * np.cos(2 * np.pi * (x + 0.5) / n) + 0.08 * np.cos(4 * np.pi * (x + 0.5) / n)
    bm = np.maximum(bm, 0.01)
    inits.append(bm / bm.sum())

    # Flat top
    ft = (0.2156 - 0.416 * np.cos(2 * np.pi * (x + 0.5) / n)
          + 0.278 * np.cos(4 * np.pi * (x + 0.5) / n)
          - 0.084 * np.cos(6 * np.pi * (x + 0.5) / n)
          + 0.0087 * np.cos(8 * np.pi * (x + 0.5) / n))
    ft = np.maximum(ft, 0.01)
    inits.append(ft / ft.sum())

    # Two Gaussian bumps
    two_g = (np.exp(-((x - n * 0.25)) ** 2 / (2 * (n / 8.0) ** 2))
             + np.exp(-((x - n * 0.75)) ** 2 / (2 * (n / 8.0) ** 2))) + 0.01
    inits.append(two_g / two_g.sum())

    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    # Another random
    inits.append(rng.dirichlet(np.ones(n) * 0.2))

    if parent is not None:
        pn = parent / parent.sum()
        inits.append(pn.copy())
        inits.append(projection_simplex(pn + rng.standard_normal(n) * 0.01))
        inits.append(0.7 * pn + 0.3 * (np.ones(n) / n))
        cw2 = 0.5 * (1 + np.cos(2 * np.pi * x / n)) + 0.1
        inits.append(0.6 * pn + 0.4 * cw2 / cw2.sum())

    return inits


def smooth_perturbation(p, sigma, rng):
    """Apply smooth (band-limited) perturbation."""
    n = len(p)
    # Generate random frequency-domain perturbation
    fft_size = 1 << (n - 1).bit_length()
    freqs = np.arange(fft_size // 2 + 1)
    # Low-pass filter
    cutoff = int(fft_size * 0.3)
    mask = np.exp(-freqs ** 2 / (2 * (cutoff / 3.0) ** 2))
    noise_freq = (rng.standard_normal(fft_size // 2 + 1) + 1j * rng.standard_normal(fft_size // 2 + 1)) * mask
    noise = np.fft.irfft(noise_freq, fft_size)[:n]
    return projection_simplex(p + noise * sigma)

                 top_k=n_best // 2, adapt_lr=True),
            dict(tau_init=0.003, tau_decay=0.9997, lr_init=0.0025, momentum=0.90,
                 top_k=n_best // 4, adapt_lr=True),
        ]

        for d, cfg in enumerate(deep_configs):
            if time.monotonic() > t3_end - 20:
                break
            t_alloc = min(100, t3_end - time.monotonic() - 20)
            if t_alloc < 15:
                break

            if d == 0:
                f_init = p_best.copy()
            elif d == 1:
                f_init = projection_simplex(p_best + rng.standard_normal(n_best) * 0.002)
            else:
                f_init = smooth_perturbation(p_best, 0.003, rng)

            a_opt, obj = smooth_max_descent(f_init, time_limit=t_alloc, **cfg)
            _try(a_opt, f"P3_deep{d}")

        # Nesterov on best
        if time.monotonic() < t3_end - 20:
            t_alloc = min(60, t3_end - time.monotonic() - 10)
            if t_alloc > 15:
                a_opt, obj = smooth_max_descent_nesterov(
                    p_best, time_limit=t_alloc,
                    tau_init=0.002, tau_decay=0.9997,
                    lr_init=0.0015, momentum=0.93, top_k=n_best // 3)
                _try(a_opt, "P3_nest")

        # Peak directed
        if time.monotonic() < t3_end - 15:
            t_alloc = min(40, t3_end - time.monotonic() - 10)
            if t_alloc > 10:
                a_opt, obj = peak_directed_descent(p_best, time_limit=t_alloc)
                _try(a_opt, "P3_peak")

    print(f"  Phase 3 best: {best_score:.8f}")

    # === Phase 4: Final polish (~remaining time) ===
    print("\n=== Phase 4: Final Polish ===")

    if best_seq is not None:
        n_best = len(best_seq)
        p_best = best_seq / best_seq.sum()

        polish_configs = [
            dict(tau_init=0.0005, tau_decay=0.999, lr_init=0.0008, momentum=0.95,
                 top_k=n_best // 2, adapt_lr=True),
            dict(tau_init=0.001, tau_decay=0.9995, lr_init=0.001, momentum=0.94,
                 top_k=n_best // 2, adapt_lr=True),
            dict(tau_init=0.0003, tau_decay=0.999, lr_init=0.0005, momentum=0.96,
                 top_k=n_best, adapt_lr=True),
        ]

        for pi, cfg in enumerate(polish_configs):
            if time.monotonic() > deadline - 10:
                break
            t_alloc = min(30, deadline - time.monotonic() - 10)
            if t_alloc < 8:
                break

            if pi == 0:
                f_init = p_best.copy()
            elif pi == 1:
                f_init = projection_simplex(p_best + rng.standard_normal(n_best) * 0.001)
            else:
                f_init = p_best.copy()

            a_opt, obj = smooth_max_descent(f_init, time_limit=t_alloc, **cfg)
            _try(a_opt, f"P4_polish{pi}")

        # Nesterov final
        if time.monotonic() < deadline - 15:
            t_alloc = min(25, deadline - time.monotonic() - 10)
            if t_alloc > 8:
                a_opt, obj = smooth_max_descent_nesterov(
                    p_best, time_limit=t_alloc,
                    tau_init=0.0008, tau_decay=0.9995,
                    lr_init=0.001, momentum=0.94, top_k=n_best // 2)
                _try(a_opt, "P4_nest_final")

    # Fallback
    if best_seq is None:
        n_fb = 384
        best_seq = np.ones(n_fb) / n_fb
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"  Fallback score: {best_score:.8f}")

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
