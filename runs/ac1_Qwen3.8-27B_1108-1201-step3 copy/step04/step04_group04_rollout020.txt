"""
Hybrid optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses: LP-based alternating direction, FFT-accelerated smoothed gradient descent
with multiplicative weights update, peak-focused coordinate moves, and multi-length search.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = conv(a, g)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def obj_value(a):
    """Compute objective 2n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def compute_smooth_grad(a, conv, top_k=30, temp=0.005):
    """Compute gradient of smoothed max autocorrelation using FFT cross-correlation."""
    n = len(a)
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peaks]
    scaled = peak_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum

    # Use FFT cross-correlation: grad[i] = 2 * sum_k w_full[k] * a[k-i]
    fft_size = 1 << (4 * n).bit_length()
    w_full = np.zeros(fft_size)
    for idx, k in enumerate(peaks):
        if w[idx] > 1e-15:
            w_full[int(k)] = w[idx]

    a_padded = np.zeros(fft_size)
    a_padded[:n] = a

    W = np.fft.fft(w_full)
    A = np.fft.fft(a_padded)
    grad = 2.0 * np.fft.ifft(W * np.conj(A)).real[:n]
    # Clip tiny negatives from FFT roundoff
    grad = np.maximum(grad, 0.0)
    return grad


def lp_direction(a, M, C=None):
    """Solve LP: max 1^T g s.t. conv(a,g) <= M, g >= 0. Return g or None."""
    n = len(a)
    if C is None:
        C = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=C, b_ub=b, bounds=bounds, method='highs',
                         options={'presolve': True, 'max_iter': 5000})
        if result.success and result.x is not None:
            g = result.x
            if g.sum() > a.sum() * (1 + 1e-8):
                return g, C
    except Exception:
        pass
    return None, C


def step_size_search(a, d, n, max_iter=8):
    """Find best step size t in (0,1] for a + t*d using golden section on obj."""
    best_t = 0.0
    best_obj = obj_value(a)

    # Try a range of candidates
    candidates = [0.8, 0.5, 0.4, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]
    for t in candidates:
        trial = a + t * d
        trial = np.maximum(trial, 0.0)
        s = trial.sum()
        if s < 1e-15:
            continue
        obj = obj_value(trial)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_t = t

    return best_t, best_obj


def lp_optimize(a0, time_limit, verbose=False):
    """LP-based alternating optimization with step sizing and restarts."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0))
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    stagnant_threshold = 60
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        # Solve LP for direction
        result = lp_direction(a, M * 1.00005)
        if result[0] is None:
            no_improve += 1
        else:
            g, _ = result
            d = g - a
            norm_d = np.linalg.norm(d)
            if norm_d < 1e-12:
                no_improve += 1
            else:
                best_t, trial_obj = step_size_search(a, d, n)
                if best_t > 0:
                    a_new = a + best_t * d
                    a_new = np.maximum(a_new, 0.0)
                    s_new = a_new.sum()
                    a_new = a_new / s_new
                    if trial_obj < best_obj - 1e-12:
                        a = a_new
                        conv = fast_convolve(a)
                        M = conv.max()
                        best_obj = 2.0 * n * M
                        best_a = a.copy()
                        no_improve = 0
                    else:
                        no_improve += 1
                else:
                    no_improve += 1

        # Stagnation handling
        if no_improve >= stagnant_threshold:
            scale = 0.002 + 0.003 * min(no_improve - stagnant_threshold, 20) / 20.0
            noise = rng.standard_normal(n) * scale
            a = best_a + noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
            else:
                a = a / s
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            stagnant_threshold = min(150, stagnant_threshold + 10)
            if stagnant_threshold > 150:
                break

        if verbose and no_improve % 30 == 0:
            print(f"    LP: obj={best_obj:.8f}, no_improve={no_improve}, t={time.monotonic()-t0:.1f}s")

    return best_a, best_obj


def mirror_descent_optimize(a0, time_limit, lr_init=0.5, top_k=30, temp_init=0.003, verbose=False):
    """Multiplicative weights / mirror descent on the simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        grad = compute_smooth_grad(a, conv, top_k=top_k, temp=temp)

        # Multiplicative update: a <- a * exp(-lr * grad) normalized
        log_update = -lr * grad
        # Clip to avoid underflow
        log_update = np.clip(log_update, -50, 50)
        a_new = a * np.exp(log_update)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # If no improvement, take a smaller step
            a_new = a + 0.1 * (a_new - a)
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            conv_new = fast_convolve(a_new)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                no_improve = 0

        a = a_new
        lr *= 0.998
        temp *= 1.002
        it += 1

        if no_improve > 200:
            break
        if verbose and it % 1000 == 0:
            print(f"    MD: obj={best_obj:.8f}, lr={lr:.4f}, t={time.monotonic()-t0:.1f}s")

    return best_a, best_obj


def gradient_descent_optimize(a0, time_limit, lr_init=0.01, momentum=0.9, top_k=30, temp_init=0.003):
    """Nesterov-accelerated gradient descent with soft-max smoothing."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    v = np.zeros(n)
    y = a.copy()
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    mom = momentum
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        y = a + mom * v
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y

        conv_y = fast_convolve(y)
        grad = compute_smooth_grad(y, conv_y, top_k=top_k, temp=temp)

        v = mom * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9995
        temp *= 1.003
        mom = min(0.97, mom + 0.0001)

        if no_improve >= 300:
            break

    return best_a, best_obj


def peak_peel_optimize(a0, time_limit, verbose=False):
    """Iteratively identify top peaks and reduce them via targeted mass redistribution."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()

        # Find top 5 peaks
        top_peaks = np.argpartition(conv, -5)[-5:]
        top_peaks = top_peaks[np.argsort(conv[top_peaks])[::-1]]

        improved = False
        for k in top_peaks:
            k = int(k)
            # Find contributing pairs (i, k-i)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            contributing = list(range(i_start, i_end + 1))
            if not contributing:
                continue

            # Find positions NOT contributing to any top peak
            in_peak = set()
            for kp in top_peaks:
                kp = int(kp)
                ist = max(0, kp - n + 1)
                ien = min(n - 1, kp)
                in_peak.update(range(ist, ien + 1))

            free_positions = [i for i in range(n) if i not in in_peak]
            if not free_positions:
                continue

            # Move mass from the largest contributor to a random free position
            contrib_vals = [(a[i], i, k - i) for i in contributing]
            contrib_vals.sort(reverse=True)

            if not contrib_vals:
                continue

            src_i, src_j = contrib_vals[0][1], contrib_vals[0][2]
            dst = rng.choice(free_positions)

            # Try several transfer amounts
            for frac in [0.1, 0.05, 0.02, 0.01]:
                delta = a[src_i] * frac
                if delta < 1e-10:
                    continue
                a_try = a.copy()
                a_try[src_i] -= delta
                if a_try[src_i] < 0:
                    continue
                a_try[dst] += delta
                s_try = a_try.sum()
                if abs(s_try - 1.0) > 1e-10:
                    a_try = a_try / s_try
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj - 1e-10:
                    a = a_try
                    best_obj = obj_try
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            no_improve += 1
            # Perturb to escape
            if no_improve > 50:
                a = best_a + 0.005 * rng.standard_normal(n)
                a = np.maximum(a, 0)
                a = a / a.sum()
                no_improve = 0
                if no_improve > 200:
                    break
        else:
            no_improve = 0

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # Sine structures
    inits['sine1'] = (np.sin(np.pi * x / n) ** 2 + 0.01)
    inits['sine1'] /= inits['sine1'].sum()
    inits['sine2'] = (np.sin(2 * np.pi * x / n) ** 2 + 0.01)
    inits['sine2'] /= inits['sine2'].sum()
    inits['sine3'] = (np.sin(3 * np.pi * x / n) ** 2 + 0.01)
    inits['sine3'] /= inits['sine3'].sum()

    # Bimodal
    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    # Three clusters
    c = n // 3
    tri = np.zeros(n)
    tri[:c//2] = 1.0
    tri[n//3:2*n//3] = 1.0
    tri[-c//2:] = 1.0
    if tri.sum() > 0:
        inits['trimodal'] = tri / tri.sum()

    # Alternating
    alt = np.ones(n)
    alt[::2] = 2.0
    alt[1::2] = 0.5
    inits['alternating'] = alt / alt.sum()

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.25 * n)) ** 2)
    inits['gaussian'] = g / g.sum()

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps

    # Random spikes
    spike = np.ones(n) * 0.1
    n_spikes = max(3, n // 16)
    spike[rng.choice(n, n_spikes, replace=False)] = 1.0
    inits['spikes'] = spike / spike.sum()

    return inits

            else:
                print(f"  GD {i} (n={n}): {score:.6f}")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 3: LP deep optimization (slowest but most effective)
    # =========================================================
    print("=== Phase 3: LP Deep Optimization ===")
    t_start3 = time.monotonic()

    # Use the current best for LP
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 200.0
        n_best = len(best_sequence)

        # Split time across a few LP runs with different starting points
        n_lp_runs = 3
        t_per_lp = max(20.0, remaining * 0.7 / n_lp_runs)

        for pi in range(n_lp_runs):
            if time.monotonic() > deadline - 150:
                break
            t_alloc = min(t_per_lp, deadline - time.monotonic() - 100)
            if t_alloc < 15:
                break

            if pi == 0:
                a0 = best_sequence.copy()
            else:
                scale = 0.003 * pi
                noise = rng.standard_normal(n_best) * scale
                a0 = np.maximum(best_sequence + noise, 0)
                a0 = a0 / a0.sum()

            print(f"  LP run {pi}: n={n_best}, t_alloc={t_alloc:.0f}s")
            a_opt, obj_val = lp_optimize(a0, time_limit=t_alloc, verbose=True)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  LP {pi}: {score:.6f} *NEW BEST*")
                else:
                    print(f"  LP {pi}: {score:.6f}")

    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 4: Peak peeling + mirror descent final polish
    # =========================================================
    print("=== Phase 4: Final Polish ===")
    t_start4 = time.monotonic()

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 30.0

        # Peak peeling
        t_pp = min(remaining * 0.4, 80.0)
        if t_pp > 10:
            a_opt, obj_val = peak_peel_optimize(best_sequence, time_limit=t_pp)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  PeakPeel: {score:.6f} *NEW BEST*")
            else:
                print(f"  PeakPeel: {score:.6f}")

        # Short mirror descent
        remaining = deadline - time.monotonic() - 20.0
        t_md = min(remaining * 0.4, 50.0)
        if t_md > 5:
            a_opt, obj_val = mirror_descent_optimize(best_sequence, time_limit=t_md,
                                                      lr_init=0.3, top_k=50, temp_init=0.001)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  MD polish: {score:.6f} *NEW BEST*")
            else:
                print(f"  MD polish: {score:.6f}")

        # Very short LP micro-polish
        remaining = deadline - time.monotonic() - 15.0
        t_lp = min(remaining * 0.5, 30.0)
        if t_lp > 8:
            a_opt, obj_val = lp_optimize(best_sequence, time_limit=t_lp)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  LP micro: {score:.6f} *NEW BEST*")
            else:
                print(f"  LP micro: {score:.6f}")

    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Fallback
    # =========================================================
    if best_sequence is None:
        best_sequence = np.ones(256)

    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(256)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
