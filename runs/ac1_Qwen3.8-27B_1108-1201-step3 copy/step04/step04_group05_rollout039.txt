```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using softmax-parameterized L-BFGS-B
with homotopy (temperature annealing) on the smooth max of auto-convolution.
Combines multi-length search, multi-start, and coordinate descent polishing.

import numpy as np
from scipy.optimize import minimize
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax(x):
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()


def smooth_max_val(conv, T):
    """Compute log-sum-exp smooth max of conv with temperature T."""
    scaled = conv / T
    scaled -= scaled.max()
    return T * (np.log(np.sum(np.exp(scaled))) + conv.max() / T)


def smooth_max_grad_wrt_a(a, T, top_k=50):
    """Compute gradient of smooth max of conv(a,a) w.r.t. a."""
    n = len(a)
    conv = fast_convolve(a)
    # softmax weights over the peaks
    scaled = conv / T
    scaled -= scaled.max()
    w = np.exp(scaled)
    wsum = w.sum()
    w = w / wsum
    
    # grad_a[m] = 2 * sum_k w[k] * a[k-m] (for valid k-m)
    # = 2 * cross_correlation(w, a)[m]
    # Using FFT: c[m] = sum_j w[j+m]*a[j] = ifft(conj(FFT(w)) * FFT(a))[m]
    fft_size = 1 << (len(conv) + n - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(np.conj(W) * A, fft_size)
    return 2.0 * c[:n]


def obj_and_grad_logspace(x, T, n):
    """Objective and gradient in log-space (x parameterizes a = softmax(x))."""
    a = softmax(x)
    conv = fast_convolve(a)
    obj = smooth_max_val(conv, T)
    g = smooth_max_grad_wrt_a(a, T)
    # d obj / dx = a * (g - dot(a, g))
    dot_ag = np.dot(a, g)
    grad = a * (g - dot_ag)
    return obj, grad


def homotopy_l_bfgsb(a_init, n, T_start, T_end, max_iter_per_stage, time_limit, seed):
    """Run L-BFGS-B with temperature annealing."""
    a = a_init.copy()
    a = project_simplex(a)
    # Initialize x in log-space
    a_safe = np.maximum(a, 1e-15)
    x = np.log(a_safe)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    T = T_start
    rng = np.random.default_rng(seed)
    
    # Annealing schedule
    n_stages = 20
    T_schedule = np.geomspace(T_start, T_end, n_stages)
    
    for stage_T in T_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        
        def fun(x):
            a_s = softmax(x)
            conv = fast_convolve(a_s)
            return smooth_max_val(conv, stage_T)
        
        def grad(x):
            a_s = softmax(x)
            g = smooth_max_grad_wrt_a(a_s, stage_T)
            dot_ag = np.dot(a_s, g)
            return a_s * (g - dot_ag)
        
        res = minimize(fun, x, jac=grad, method='L-BFGS-B',
                       options={'maxiter': max_iter_per_stage, 'gtol': 1e-8})
        
        if res.success or res.fun < fun(x):
            x = res.x
            a = softmax(x)
            conv = fast_convolve(a)
            actual_obj = conv.max()
            if actual_obj < best_obj:
                best_obj = actual_obj
                best_a = a.copy()
    
    return best_a, best_obj


def coordinate_descent_polish(a, time_limit, max_passes=100):
    """Coordinate descent: for each coordinate, find optimal value given others."""
    n = len(a)
    a = project_simplex(a.copy())
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    
    for _ in range(max_passes):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            # Try to optimize a[i] while keeping sum = 1
            # a[i] can range from 0 to 1 - sum(a[j], j != i)
            others_sum = 1.0 - a[i]
            # Objective as function of a[i]:
            # b[k] changes only for k where a[i] appears
            # b[k] includes term 2*a[i]*a[k-i] (for k-i valid and != i) + a[i]^2 (for k=2i)
            
            # Simple approach: try a few values of a[i]
            # Current value
            cur = a[i]
            # Try: 0, cur*0.5, cur*1.5, others_sum
            candidates = [0.0, cur * 0.5, cur * 1.5, others_sum]
            # Also try the analytical optimum if the peak is at a specific lag
            # For now, just try a few values
            
            best_val = best_obj
            best_ai = cur
            for ai in candidates:
                ai = np.clip(ai, 0.0, others_sum)
                a[i] = ai
                # Renormalize (should already be sum=1)
                s = a.sum()
                if s < 1e-15:
                    a[i] = cur
                    break
                if abs(s - 1.0) > 1e-10:
                    a = a / s
                conv = fast_convolve(a)
                val = conv.max()
                if val < best_val:
                    best_val = val
                    best_ai = ai
            a[i] = best_ai
            s = a.sum()
            if abs(s - 1.0) > 1e-10:
                a = a / s
            if best_val < best_obj - 1e-12:
                best_obj = best_val
                improved = True
        
        if not improved:
            break
    
    return a, best_obj


def projected_gradient_polish(a, time_limit, lr_init=0.01, top_k=30):
    """Projected gradient descent on the simplex."""
    n = len(a)
    a = project_simplex(a.copy())
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    lr = lr_init
    v = np.zeros(n)
    mom = 0.9
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        w = pv / pv.sum()
        
        # Gradient
        g = np.zeros(n)
        for i in range(len(peaks)):
            k = int(peaks[i])
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            for m in range(lo, hi + 1):
                g[m] += 2.0 * w[i] * a[k - m]
        
        v = mom * v - lr * g
        a_new = project_simplex(a + v)
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                lr *= 0.5
                v = np.zeros(n)
                no_improve = 0
        
        lr *= 0.999
    
    return a, best_obj


def generate_initializations(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(0)
    
    inits = []
    
    # Uniform
    inits.append(('uniform', np.ones(n) / n))
    
    # Gaussian centered
    x = np.arange(n, dtype=np.float64)
    for sigma_frac in [0.2, 0.3, 0.4, 0.5]:
        sigma = sigma_frac * n
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / sigma) ** 2)
        inits.append((f'gauss_{sigma_frac}', g / g.sum()))
    
    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    inits.append(('raised_cos', rc / rc.sum()))
    
    # Flat-top approximation
    ft = np.cos(np.pi * x / (n + 1)) ** 2
    inits.append(('flattop', ft / ft.sum()))
    
    # U-shape (edge concentration)
    u = 1.0 / (0.1 + np.abs(x - 0.5 * (n - 1)) / n)
    inits.append(('u_shape', u / u.sum()))
    
    # Dirichlet random
    for alpha_val in [0.5, 1.0, 2.0]:
        d = rng.dirichlet(np.ones(n) * alpha_val)
        inits.append((f'dirichlet_{alpha_val}', d))
    
    # Parent interpolation
    if parent is not None:
        p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        p = np.maximum(p, 0)
        ps = p.sum()
        if ps > 1e-10:
            inits.append(('parent_interp', p / ps))
    
    # Symmetric random
    half = rng.dirichlet(np.ones((n + 1) // 2))
    if n % 2 == 0:
        sym = np.concatenate([half, half[::-1]])
    else:
        sym = np.concatenate([half, half[-2::-1]])
    inits.append(('sym_random', sym / sym.sum()))
    
    return inits


def evaluate_quick(a):
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using softmax-parameterized L-BFGS-B
with homotopy temperature annealing on the smooth max of auto-convolution,
combined with multi-length multi-start search and projected gradient polishing.
Key innovation: working in log-space (x where a=softmax(x)) removes constraints,
and multi-temperature gradient averaging provides more balanced peak targeting.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_fn(x):
    """Numerically stable softmax."""
    x = x - x.max()
    e = np.exp(x)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def smooth_max_and_weights(conv, T):
    """Compute smooth max value and softmax weights."""
    scaled = conv / T
    scaled_max = scaled.max()
    exp_s = np.exp(scaled - scaled_max)
    log_sum = np.log(exp_s.sum())
    val = T * (log_sum + scaled_max)
    w = exp_s / exp_s.sum()
    return val, w


def grad_wrt_a(a, w):
    """Compute gradient of weighted sum of convolution peaks w.r.t. a.
    grad[m] = 2 * sum_k w[k] * a[k-m] for valid indices.
    Uses FFT for efficiency."""
    n = len(a)
    fft_size = 1 << (len(w) + n).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(np.conj(W) * A, fft_size)
    return 2.0 * c[:n]


def multi_temp_obj_grad(x, temps, weights):
    """Objective and gradient in log-space using multi-temperature averaging."""
    a = softmax_fn(x)
    dot_term = 0.0
    grad_a_total = np.zeros(len(a))
    for T, wt in zip(temps, weights):
        conv = fast_convolve(a)
        val, w_peaks = smooth_max_and_weights(conv, T)
        dot_term += wt * val
        grad_a_total += wt * grad_wrt_a(a, w_peaks)
    dot_ag = np.dot(a, grad_a_total)
    grad_x = a * (grad_a_total - dot_ag)
    return dot_term, grad_x


def omega_l_bfgsb(a_init, T_starts, T_ends, temp_weights, max_iter_per_stage, time_limit, seed):
    """L-BFGS-B with homotopy in log-space. T_starts and T_ends are lists of temps
    for each temperature channel, annealed together."""
    n = len(a_init)
    a_safe = np.maximum(a_init, 1e-12)
    x = np.log(a_safe)
    
    best_a = a_init.copy()
    a_norm = best_a / best_a.sum()
    best_obj = fast_convolve(a_norm).max()
    
    t0 = time.monotonic()
    n_stages = 15
    anneal = np.linspace(0, 1, n_stages) ** 1.5
    
    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        # Interpolate temperatures
        cur_temps = [T_starts[i] * (1 - anneal[stage]) + T_ends[i] * anneal[stage]
                     for i in range(len(T_starts))]
        
        def fun(xi):
            a_s = softmax_fn(xi)
            total = 0.0
            for T, wt in zip(cur_temps, temp_weights):
                conv = fast_convolve(a_s)
                val, _ = smooth_max_and_weights(conv, T)
                total += wt * val
            return total
        
        def grad_fn(xi):
            a_s = softmax_fn(xi)
            grad_total = np.zeros(n)
            for T, wt in zip(cur_temps, temp_weights):
                conv = fast_convolve(a_s)
                _, w_p = smooth_max_and_weights(conv, T)
                grad_total += wt * grad_wrt_a(a_s, w_p)
            dot_ag = np.dot(a_s, grad_total)
            return a_s * (grad_total - dot_ag)
        
        res = minimize(fun, x, jac=grad_fn, method='L-BFGS-B',
                       options={'maxiter': max_iter_per_stage, 'gtol': 1e-9,
                                'maxfun': 5000})
        if res.x is not None:
            x = res.x
            a = softmax_fn(x)
            conv = fast_convolve(a)
            actual = conv.max()
            if actual < best_obj:
                best_obj = actual
                best_a = a.copy()
    
    return best_a, best_obj


def pg_polish(a_init, time_limit, lr_init=0.008, top_k=40, mom=0.92):
    """Projected gradient on simplex with multi-peak targeting."""
    n = len(a_init)
    a = project_simplex(a_init.copy())
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    lr = lr_init
    v = np.zeros(n)
    no_improve = 0
    T = 0.001
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-30:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
        else:
            w /= ws
        
        g = grad_wrt_a(a, w)
        v = mom * v - lr * g
        a_new = project_simplex(a + v)
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                lr *= 0.5
                v = np.zeros(n)
                no_improve = 0
        
        lr *= 0.9995
        T *= 1.002
    
    return best_a, best_obj

        if i == j:
            continue
        cur_i, cur_j = a[i], a[j]
        if cur_i < 1e-10 and cur_j < 1e-10:
            continue
        # Try redistributing between i and j
        total = cur_i + cur_j
        for frac in [0.1, 0.3, 0.5, 0.7, 0.9]:
            a[i] = total * frac
            a[j] = total * (1.0 - frac)
            conv = fast_convolve(a)
            if conv.max() < best_obj - 1e-14:
                best_obj = conv.max()
            else:
                a[i] = cur_i
                a[j] = cur_j
                break
        else:
            continue
    
    return a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    inits.append((np.ones(n) / n, 'uniform'))
    
    for sf in [0.15, 0.2, 0.3, 0.4, 0.5]:
        sigma = sf * n
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / sigma) ** 2)
        inits.append((g / g.sum(), f'gauss_{sf}'))
    
    # Two-lobe
    for dark in [0.1, 0.15, 0.2]:
        d = dark * n
        t1 = np.exp(-0.5 * ((x - 0.3 * n) / (0.15 * n)) ** 2)
        t2 = np.exp(-0.5 * ((x - 0.7 * n) / (0.15 * n)) ** 2)
        t = (t1 + t2) + dark
        inits.append((t / t.sum(), f'twolo_{dark}'))
    
    # Raised cosine variants
    for beta in [1.0, 2.0, 3.0]:
        rc = 0.5 * (1 + np.cos(np.pi * x / (n // 2 + 1))) ** beta
        if rc.sum() > 1e-10:
            inits.append((rc / rc.sum(), f'rc_{beta}'))
    
    # Flat top
    ft = np.cos(np.pi * x / (n + 2)) ** 4
    if ft.sum() > 1e-10:
        inits.append((ft / ft.sum(), 'flattop'))
    
    # Edge concentration
    for alpha in [2.0, 4.0, 8.0]:
        e = ((x + 1) / (n + 1)) ** alpha + ((n - x) / (n + 1)) ** alpha
        inits.append((e / e.sum(), f'edge_{alpha}'))
    
    # Dirichlet
    for av in [0.3, 0.5, 1.0, 2.0]:
        d = rng.dirichlet(np.ones(n) * av)
        inits.append((d, f'dir_{av}'))
    
    # Symmetric Dirichlet
    half_n = (n + 1) // 2
    h = rng.dirichlet(np.ones(half_n) * 0.7)
    if n % 2 == 0:
        sym = np.concatenate([h, h[::-1]])
    else:
        sym = np.concatenate([h, h[-2::-1]])
    inits.append((sym / sym.sum(), 'sym_dir'))
    
    # Parent interpolation
    if parent is not None and len(parent) > 0:
        p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        p = np.maximum(p, 0)
        ps = p.sum()
        if ps > 1e-10:
            inits.append((p / ps, 'parent'))
        # Scaled parent with slight perturbation
        pp = p + rng.standard_normal(n) * 0.01
        pp = np.maximum(pp, 0)
        pps = pp.sum()
        if pps > 1e-10:
            inits.append((pp / pps, 'parent_pert'))
    
    return inits


def update_best(best_seq, best_score, new_seq, label):
    """Evaluate and update best if improved."""
    a = np.maximum(np.asarray(new_seq, dtype=np.float64), 0.0)
    if a.size == 0 or a.sum() < 0.01:
        return best_seq, best_score
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return best_seq, best_score
    if np.isfinite(score) and score < best_score:
        best_seq = a.copy()
        best_score = score
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry: multi-phase search using softmax L-BFGS-B homotopy,
    multi-length exploration, projected gradient, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)
    
    best_seq = None
    best_score = float('inf')
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={parent_score:.8f}")
            best_seq, best_score = update_best(best_seq, best_score, parent, "parent")
    except Exception:
        pass
    
    # === Phase 1: Quick initialization scan ===
    print("\n=== Phase 1: Initialization Scan ===")
    t1_start = time.monotonic()
    t1_budget = min(120.0, (deadline - time.monotonic()) * 0.1)
    
    lengths = sorted(set([192, 256, 320, 384, 448, 512, 640]))
    if parent is not None:
        lengths = sorted(set(lengths + [len(parent)]))
    
    all_candidates = []
    for n_try in lengths:
        if time.monotonic() - t1_start > t1_budget:
            break
        inits = generate_inits(n_try, parent, rng)
        for a, label in inits:
            a_n = a / a.sum()
            conv = fast_convolve(a_n)
            score = 2.0 * n_try * conv.max()
            all_candidates.append((score, n_try, label, a_n))
    
    all_candidates.sort(key=lambda x: x[0])
    
    # Evaluate top initializations
    for score, n_try, label, a in all_candidates[:30]:
        if time.monotonic() - t1_start > t1_budget:
            break
        best_seq, best_score = update_best(best_seq, best_score, a, f"init_{label}_n{n_try}")
    
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_seq, best_score = update_best(best_seq, best_score, best_seq, "fallback")
    
    print(f"  Phase 1 best: {best_score:.8f} (n={len(best_seq)})")
    
    # === Phase 2: L-BFGS-B Homotopy on top candidates ===
    print("\n=== Phase 2: L-BFGS-B Homotopy ===")
    t2_start = time.monotonic()
    t2_budget = min(600.0, (deadline - time.monotonic()) * 0.45)
    
    # Choose which candidates to optimize
    to_optimize = []
    seen_lengths = set()
    for score, n_try, label, a in all_candidates[:25]:
        if len(to_optimize) >= 10:
            break
        to_optimize.append((n_try, label, a))
        seen_lengths.add(n_try)
    # Ensure best_seq length is included
    if len(best_seq) not in seen_lengths:
        to_optimize.append((len(best_seq), "best", best_seq.copy()))
    
    # Temperature settings for multi-channel
    # Channel 1: broad (high T), Channel 2: focused (low T)
    T1_start, T1_end = 0.05, 5e-5
    T2_start, T2_end = 0.01, 1e-5
    tw = [0.6, 0.4]
    
    time_per = max(10.0, t2_budget / max(1, len(to_optimize)))
    
    for idx, (n_try, label, a_init) in enumerate(to_optimize):
        if time.monotonic() - t2_start > t2_budget:
            break
        alloc = min(time_per, t2_budget - (time.monotonic() - t2_start))
        if alloc < 5:
            break
        print(f"  [LBFGSB] n={n_try} [{label}], budget={alloc:.0f}s")
        
        a_opt, obj = omega_l_bfgsb(
            a_init,
            T_starts=[T1_start, T2_start],
            T_ends=[T1_end, T2_end],
            temp_weights=tw,
            max_iter_per_stage=80,
            time_limit=alloc,
            seed=seed + idx
        )
        best_seq, best_score = update_best(best_seq, best_score, a_opt, f"lbfgsb_{label}_n{n_try}")
    
    print(f"  Phase 2 best: {best_score:.8f} (n={len(best_seq)})")
    
    # === Phase 3: Projected Gradient Polish ===
    print("\n=== Phase 3: PG Polish ===")
    t3_start = time.monotonic()
    t3_budget = min(250.0, (deadline - time.monotonic()) * 0.2)
    
    n_cur = len(best_seq)
    if t3_budget > 10:
        a_opt, obj = pg_polish(best_seq, time_limit=t3_budget * 0.5, lr_init=0.006, top_k=50, mom=0.93)
        best_seq, best_score = update_best(best_seq, best_score, a_opt, "pg1")
        
        rem = t3_budget - (time.monotonic() - t3_start)
        if rem > 10:
            a_opt, obj = pg_polish(best_seq, time_limit=rem * 0.7, lr_init=0.002, top_k=80, mom=0.95)
            best_seq, best_score = update_best(best_seq, best_score, a_opt, "pg2")
        
        rem = t3_budget - (time.monotonic() - t3_start)
        if rem > 5:
            a_opt, obj = coord_descent_polish(best_seq, time_limit=rem * 0.8)
            best_seq, best_score = update_best(best_seq, best_score, a_opt, "coord_cd")
    
    print(f"  Phase 3 best: {best_score:.8f} (n={len(best_seq)})")
    
    # === Phase 4: Multi-restart from perturbed best ===
    print("\n=== Phase 4: Multi-Restart ===")
    t4_start = time.monotonic()
    t4_budget = min(300.0, (deadline - time.monotonic()) * 0.2)
    
    n_restarts = max(2, min(8, int(t4_budget / 30)))
    time_per_r = t4_budget / n_restarts
    
    for r in range(n_restarts):
        if time.monotonic() - t4_start > t4_budget:
            break
        alloc = min(time_per_r, t4_budget - (time.monotonic() - t4_start))
        if alloc < 5:
            break
        noise_scale = 0.003 + 0.002 * r
        a_pert = project_simplex(best_seq + rng.standard_normal(n_cur) * noise_scale)
        
        # Use tighter temperatures for polishing
        a_opt, obj = omega_l_bfgsb(
            a_pert,
            T_starts=[0.02, 0.005],
            T_ends=[2e-5, 5e-6],
            temp_weights=[0.5, 0.5],
            max_iter_per_stage=60,
            time_limit=alloc,
            seed=seed + 500 + r
        )
        best_seq, best_score = update_best(best_seq, best_score, a_opt, f"restart_{r}")
    
    print(f"  Phase 4 best: {best_score:.8f} (n={len(best_seq)})")
    
    # === Phase 5: Try different lengths with interpolation ===
    print("\n=== Phase 5: Length Exploration ===")
    t5_start = time.monotonic()
    t5_budget = min(150.0, (deadline - time.monotonic()) * 0.15)
    
    n_best = len(best_seq)
    length_tries = sorted(set([n_best - 32, n_best - 16, n_best + 16, n_best + 32,
                                n_best - 64, n_best + 64]))
    length_tries = [lt for lt in length_tries if 64 <= lt <= 1024]
    
    for n_try in length_tries[:4]:
        if time.monotonic() - t5_start > t5_budget:
            break
        rem = t5_budget - (time.monotonic() - t5_start)
        if rem < 10:
            break
        # Interpolate best to new length
        interp = np.interp(np.linspace(0, 1, n_try), np.linspace(0, 1, n_best), best_seq)
        interp = np.maximum(interp, 0)
        interp = interp / interp.sum()
        
        alloc = min(30.0, rem * 0.7)
        a_opt, obj = omega_l_bfgsb(
            interp,
            T_starts=[0.03, 0.01],
            T_ends=[3e-5, 1e-5],
            temp_weights=[0.6, 0.4],
            max_iter_per_stage=60,
            time_limit=alloc,
            seed=seed + 900 + n_try
        )
        best_seq, best_score = update_best(best_seq, best_score, a_opt, f"len{n_try}")
    
    print(f"  Phase 5 best: {best_score:.8f} (n={len(best_seq)})")
    
    # === Phase 6: Final fine polish ===
    print("\n=== Phase 6: Final Polish ===")
    remaining = deadline - time.monotonic() - 15
    if remaining > 10:
        a_opt, obj = pg_polish(best_seq, time_limit=remaining * 0.6, lr_init=0.001, top_k=100, mom=0.96)
        best_seq, best_score = update_best(best_seq, best_score, a_opt, "final_pg")
        
        rem2 = deadline - time.monotonic() - 10
        if rem2 > 8:
            a_opt, obj = coord_descent_polish(best_seq, time_limit=rem2 * 0.8)
            best_seq, best_score = update_best(best_seq, best_score, a_opt, "final_cd")
    
    # Final verification
    if best_seq is not None:
        final_score = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(final_score) and final_score < best_score:
            best_score = final_score
            best_seq = np.maximum(best_seq, 0.0)
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK: {best_score:.10f}")
    
    return [float(x) for x in best_seq]
```
