```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L-BFGS optimization on a
log-sum-exp smoothed objective in log-space (softmax parameterization),
combined with exponentiated gradient polishing, multi-start exploration,
and multi-length search. Key innovation: analytic gradient of the smoothed
max-convolution enables efficient quasi-Newton optimization.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def smoothed_obj_and_grad(x, T, c=None, a=None):
    """
    Compute smoothed objective F(x) = 2n*T*log(sum_k exp(c_k/T)) and its gradient,
    where a = softmax(x) and c = conv(a, a).
    Returns (value, gradient).
    """
    n = len(x)
    # Numerically stable softmax
    x_shifted = x - x.max()
    exp_x = np.exp(x_shifted)
    a = exp_x / exp_x.sum()

    if c is None:
        c = fast_convolve(a)

    # Log-sum-exp of c/T
    c_scaled = c / T
    c_max = c_scaled.max()
    log_sum = c_max + np.log(np.exp(c_scaled - c_max).sum())
    value = 2.0 * n * T * log_sum

    # Weights w_k = exp(c_k/T) / sum_l exp(c_l/T)
    w = np.exp(c_scaled - c_max)
    w /= w.sum()

    # W = sum_k w_k * c_k
    W = np.dot(w, c)

    # b_i = sum_k w_k * a_{k-i} (with a_j = 0 if j out of [0,n-1])
    # This is np.correlate(w, a, mode='valid')
    b = np.correlate(w, a, mode='valid')

    # grad_i = 4n * a_i * (b_i - W)
    grad = 4.0 * n * a * (b - W)

    return value, grad


def lbfgs_optimize(a0, time_limit, T_start=0.005, T_end=0.0005, n_annealing=4):
    """
    L-BFGS optimization with temperature annealing.
    a0: initial probability vector (sum=1, non-negative)
    Returns best probability vector found.
    """
    n = len(a0)
    a = np.maximum(a0, 1e-15).astype(np.float64)
    a = a / a.sum()

    x0 = np.log(a)

    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()

    t0 = time.monotonic()

    # Anneal temperature
    for stage in range(n_annealing):
        T = T_start * (T_end / T_start) ** (stage / max(1, n_annealing - 1))
        stage_time = (time_limit - (time.monotonic() - t0)) / (n_annealing - stage)

        # Internal function for scipy
        def fun(x_inner):
            val, _ = smoothed_obj_and_grad(x_inner, T)
            return val

        def fun_and_grad(x_inner):
            val, g = smoothed_obj_and_grad(x_inner, T)
            return val, g

        try:
            res = minimize(fun_and_grad, x0, method='L-BFGS-B',
                           jac=True, options={'maxiter': 500, 'maxls': 50,
                                              'ftol': 1e-10, 'gtol': 1e-8,
                                              'maxfun': 2000})
        except Exception:
            break

        if res.x is not None:
            a_new = np.exp(res.x - res.x.max())
            a_new = a_new / a_new.sum()
            score = 2.0 * n * fast_convolve(a_new).max()
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
            x0 = res.x

        if time.monotonic() - t0 > time_limit:
            break

    return best_a, best_score


def eg_optimize(a0, time_limit, eta0=1.0, temp0=0.003, temp_decay=0.9996,
                eta_decay=0.99995, top_k=200, verbose=False):
    """Exponentiated gradient / multiplicative weights on the simplex."""
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    avg_a = a.copy().astype(np.float64)
    count = 0
    eta = eta0
    temp = temp0

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if it > 200000:
            break

        # Subgradient: focus on top peaks
        if top_k < len(c):
            indices = np.argpartition(c, -top_k)[-top_k:]
            peek = c[indices]
        else:
            indices = np.arange(len(c))
            peek = c.copy()

        scaled = peek / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(indices))
            w[np.argmax(peek)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for idx in range(len(indices)):
            if w[idx] < 1e-15:
                continue
            k = int(indices[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            sg[js:je + 1] += 2.0 * w[idx] * a[k - je:k - js + 1][::-1]

        sg_max = sg.max()
        if sg_max < 1e-15:
            break

        log_update = -eta * sg / sg_max
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-300)
        s_new = a_new.sum()
        a_new = a_new / s_new

        c_new = fast_convolve(a_new)
        obj_new = c_new.max()

        count += 1
        avg_a = (count * avg_a + a_new) / (count + 1)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Check running average periodically
        if it % 500 == 0:
            c_avg = fast_convolve(avg_a)
            if c_avg.max() < best_obj - 1e-14:
                best_obj = c_avg.max()
                best_a = avg_a.copy()
                no_improve = 0

        a = a_new
        c = c_new
        eta *= eta_decay
        temp *= temp_decay

        # Stagnation restart
        if no_improve > 2000:
            rng_local = np.random.default_rng(it * 17 + 3)
            a = best_a + rng_local.standard_normal(n) * 0.001
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            c = fast_convolve(a)
            no_improve = 0
            eta = max(eta0 * 0.5, 0.01)
            temp = max(temp, temp0 * 0.05)
            avg_a = a.copy()
            count = 0

    return best_a, best_obj

                s2 = a.sum()
                a = a / s2
                c2 = fast_convolve(a)
                obj2 = c2.max()
                if obj2 < best_obj - 1e-13:
                    best_obj = obj2
                    best_a = a.copy()
                    improved = True
                    break
                else:
                    a = best_a.copy()
                    break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                break

    return best_a, best_obj


def geometric_init(n, alpha):
    """Geometric sequence with ratio r = 1 - alpha/n, normalized to sum=1."""
    x = np.arange(n, dtype=np.float64)
    r = 1.0 - alpha / n
    if r <= 0:
        r = 0.5
    if r >= 1:
        r = 0.99
    a = r ** x
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def diverse_init(n, rng, alpha_range=None):
    """Generate a diverse initialization."""
    x = np.arange(n, dtype=np.float64)
    choice = rng.integers(0, 10)
    if choice == 0:
        return np.ones(n) / n
    elif choice == 1:
        return geometric_init(n, rng.uniform(0.5, 3.0))
    elif choice == 2:
        a = np.exp(-2.0 * x / n)
        return a / a.sum()
    elif choice == 3:
        a = np.exp(-2.0 * (n - 1 - x) / n)
        return a / a.sum()
    elif choice == 4:
        a = np.abs(np.cos(np.pi * x / n)) + 0.1
        return a / a.sum()
    elif choice == 5:
        edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
        return edge_w / edge_w.sum()
    elif choice == 6:
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
        return g / g.sum()
    elif choice == 7:
        return rng.dirichlet(np.ones(n) * 0.5)
    elif choice == 8:
        a = np.ones(n)
        a[:n//4] *= 0.5
        a[3*n//4:] *= 0.5
        return a / a.sum()
    else:
        return rng.dirichlet(np.ones(n) * 0.2)


def propose_candidate(seed=42, budget_s=1800):
    """
    Main search function combining L-BFGS with temperature annealing,
    exponentiated gradient, local search, and multi-start exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
        elif np.isfinite(score):
            pass
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = evaluate_sequence(parent.tolist())
            print(f"Parent: n={len(parent)}, score={parent_score:.8f}")
            if np.isfinite(parent_score):
                best_score = parent_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        n_init = 384
        parent = geometric_init(n_init, 1.5)
        best_score = try_seq(parent, "parent_geometric")
        best_seq = parent.copy()
        print(f"No parent, using geometric n={n_init}, score={best_score:.8f}")

    n = len(parent)
    print(f"Starting: n={n}, best_score={best_score:.8f}")
    print(f"Time budget: {budget_s}s, deadline in {deadline - time.monotonic():.0f}s")

    # === Phase 1: L-BFGS on parent with temperature annealing ===
    print("\n=== Phase 1: L-BFGS on parent ===")
    rem = deadline - time.monotonic() - 600
    t1 = min(400, max(60, rem))
    if t1 > 30:
        print(f"  L-BFGS (n={n}), alloc={t1:.0f}s")
        a_opt, _ = lbfgs_optimize(parent, time_limit=t1, T_start=0.008, T_end=0.0003, n_annealing=5)
        try_seq(a_opt, "lbfgs_p1")

    # === Phase 2: EG polish ===
    print(f"\n=== Phase 2: EG Polish ===")
    print(f"  Best so far: {best_score:.8f}")
    rem = deadline - time.monotonic() - 400
    t2 = min(200, max(30, rem))
    if t2 > 15 and best_seq is not None:
        n_best = len(best_seq)
        top_k = max(50, n_best // 2)
        a_opt, _ = eg_optimize(best_seq, time_limit=t2, eta0=0.5, temp0=0.002,
                               temp_decay=0.9997, eta_decay=0.99998, top_k=top_k)
        try_seq(a_opt, "eg_p2")

    # === Phase 3: Multi-length L-BFGS exploration ===
    print(f"\n=== Phase 3: Multi-length L-BFGS ===")
    print(f"  Best so far: {best_score:.8f}")

    candidate_ns = [192, 256, 320, 384, 448, 512, 640]
    # Filter to reasonable range
    candidate_ns = [nn for nn in candidate_ns if 64 <= nn <= 800]
    # Sort by proximity to current best n
    candidate_ns.sort(key=lambda x: abs(x - n))
    # Ensure current n is first
    if n in candidate_ns:
        candidate_ns.remove(n)
        candidate_ns.insert(0, n)

    for idx, n_try in enumerate(candidate_ns):
        rem = deadline - time.monotonic() - 150
        if rem < 30:
            break
        t_alloc = min(120, rem / max(1, len(candidate_ns) - idx))
        if t_alloc < 15:
            break

        # Interpolate or create init
        if n_try == n and best_seq is not None:
            a_init = best_seq.copy()
        elif best_seq is not None:
            x_old = np.linspace(0, 1, len(best_seq))
            x_new = np.linspace(0, 1, n_try)
            a_init = np.interp(x_new, x_old, best_seq)
            a_init = np.maximum(a_init, 1e-10)
            a_init = a_init / a_init.sum()
        else:
            a_init = geometric_init(n_try, 1.5)

        # Also try a geometric init for diversity
        a_init2 = geometric_init(n_try, rng.uniform(0.8, 2.5))

        print(f"  n={n_try}: L-BFGS alloc={t_alloc:.0f}s")
        t_each = t_alloc / 2.5
        a_opt1, _ = lbfgs_optimize(a_init, time_limit=t_each, T_start=0.005, T_end=0.0005, n_annealing=3)
        try_seq(a_opt1, f"lbfgs_n{n_try}")

        if time.monotonic() < deadline - 100:
            a_opt2, _ = lbfgs_optimize(a_init2, time_limit=t_each, T_start=0.005, T_end=0.0005, n_annealing=3)
            try_seq(a_opt2, f"lbgfs_geo_n{n_try}")

    # === Phase 4: More L-BFGS refinement on best ===
    print(f"\n=== Phase 4: L-BFGS Refinement ===")
    print(f"  Best so far: {best_score:.8f}")

    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        t4 = min(300, max(30, rem))
        if t4 > 20:
            print(f"  L-BFGS refine (n={len(best_seq)}), alloc={t4:.0f}s")
            a_opt, _ = lbfgs_optimize(best_seq, time_limit=t4, T_start=0.002, T_end=0.0001, n_annealing=6)
            try_seq(a_opt, "lbfgs_refine")

    # === Phase 5: EG with different hyperparameters ===
    print(f"\n=== Phase 5: EG Variations ===")
    print(f"  Best so far: {best_score:.8f}")

    if best_seq is not None:
        configs = [
            {"eta0": 0.3, "temp0": 0.001, "temp_decay": 0.9999, "eta_decay": 0.99999, "top_k": len(best_seq)},
            {"eta0": 2.0, "temp0": 0.01, "temp_decay": 0.9995, "eta_decay": 0.9999, "top_k": max(20, len(best_seq) // 4)},
            {"eta0": 0.8, "temp0": 0.005, "temp_decay": 0.9998, "eta_decay": 0.99995, "top_k": max(50, len(best_seq) // 2)},
        ]
        for ci, cfg in enumerate(configs):
            rem = deadline - time.monotonic() - 60
            if rem < 20:
                break
            t_alloc = min(100, rem / max(1, len(configs) - ci))
            if t_alloc < 10:
                break
            print(f"  EG cfg{ci}, alloc={t_alloc:.0f}s")
            a_opt, _ = eg_optimize(best_seq, time_limit=t_alloc, **cfg)
            try_seq(a_opt, f"eg_v{ci}")

    # === Phase 6: Local search polish ===
    print(f"\n=== Phase 6: Local Search ===")
    print(f"  Best so far: {best_score:.8f}")

    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            t6 = min(60, rem)
            a_opt, _ = local_search(best_seq, time_limit=t6, rng=rng)
            try_seq(a_opt, "local")

    # === Phase 7: Final L-BFGS + EG combo ===
    print(f"\n=== Phase 7: Final Combo ===")
    print(f"  Best so far: {best_score:.8f}")

    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 15:
            t7a = rem * 0.6
            t7b = rem * 0.4 - 5
            if t7a > 10:
                a_opt, _ = lbfgs_optimize(best_seq, time_limit=t7a, T_start=0.001, T_end=0.0001, n_annealing=4)
                try_seq(a_opt, "final_lbfgs")
            if best_seq is not None and t7b > 5:
                a_opt, _ = eg_optimize(best_seq, time_limit=t7b, eta0=0.1, temp0=0.0005,
                                       temp_decay=0.99995, eta_decay=0.99999,
                                       top_k=len(best_seq))
                try_seq(a_opt, "final_eg")

    # === Phase 8: One more local search ===
    if best_seq is not None:
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_opt, _ = local_search(best_seq, time_limit=min(rem, 20), rng=rng)
            try_seq(a_opt, "final_local")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
```
