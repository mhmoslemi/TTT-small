    conv = fast_convolve(a)

    c = conv - conv.max()

    # np.correlate(w, a, 'valid')[k] = sum_j w[k+j]*a[j], k=0..n-1
    corr = np.correlate(w, a, mode='valid')
    return 2.0 * corr

def objective_and_grad(b, T):
    """Objective and gradient in b-space where a = softmax(b)."""
    # Compute a = softmax(b)
    b_max = b.max()
    exp_b = np.exp(b - b_max)
    a = exp_b / exp_b.sum()
    
    # Compute convolution
    conv = fast_convolve(a)
    
    # Compute smooth max and weights
    val = smooth_max_value(conv, T)
    w = smooth_max_weight(conv, T)
    
    # Gradient w.r.t. a
    grad_a = gradient_smooth_max(w, a)
    
    # Gradient w.r.t. b (chain rule through softmax)
    # d(a_i)/d(b_j) = a_i * (delta_ij - a_j)
    # grad_b_j = sum_i grad_a_i * d(a_i)/d(b_j) = sum_i grad_a_i * a_i * (delta_ij - a_j)
    #          = a_j * grad_a_j - a_j * sum_i grad_a_i * a_i
    grad_b = a * (grad_a - np.dot(grad_a, a))
    
    return val, grad_b

def lbfgs_optimize(a0, T_schedule, max_iter=5000, tol=1e-10):
    """L-BFGS optimization with adaptive temperature schedule."""
    a = a0.astype(np.float64).copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    # Initialize b from a
    b = np.log(a + 1e-30)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    for T in T_schedule:
        # Optimize at this temperature
        current_b = b.copy()
        
        def func(b_vec):
            val, _ = objective_and_grad(b_vec, T)
            return val
        
        def grad(b_vec):
            _, g = objective_and_grad(b_vec, T)
            return g
        
        res = minimize(func, current_b, jac=grad, method='L-BFGS-B',
                       options={'maxiter': max_iter, 'ftol': tol, 'gtol': 1e-8,
                               'maxls': 50})
        
        if res.success or res.fun < func(current_b):
            current_b = res.x
            a_new = np.exp(current_b - current_b.max())
            a_new = a_new / a_new.sum()
            obj = fast_convolve(a_new).max()
            if obj < best_obj:
                best_obj = obj
                best_a = a_new.copy()
        
        b = current_b
    
    return best_a, best_obj

def coordinate_descent(a0, time_limit, num_passes_max=50):
    """
    Coordinate descent for minimizing max(conv(a,a)) with sum(a)=1 constraint.
    For each coordinate i, minimize F(x) = max(conv after setting a[i]=x and renormalizing).
    """
    a = a0.astype(np.float64).copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return a0, fast_convolve(a0).max()
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    
    for p in range(num_passes_max):
        if time.monotonic() - t0 > time_limit:
            break
        
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            v = a[i]
            if v < 1e-12:
                # Try setting it to a small positive value
                x_try = 0.01 / n
            else:
                x_try = v
            
            # Minimize F(x) = c(x)^2 * max(conv_modified)
            # where c(x) = 1/(1-v+x), conv_modified[k] = conv[k] + delta_k(x)
            
            # Sample a range of x values
            # x ranges from 0 to maybe 5*v (or more)
            x_max = max(5.0 * v, 0.1 / n) if v > 0 else 0.1 / n
            
            # Sample
            x_samples = np.array([0.0, 0.1*x_max, 0.3*x_max, 0.5*x_max, 
                                  0.7*x_max, 0.9*x_max, x_max, 1.5*x_max, 2.0*x_max])
            
            best_x = v
            best_F = best_obj  # current value (at x=v)
            
            for x in x_samples:
                if x < 0:
                    continue
                c = 1.0 / (1.0 - v + x)
                # Compute modified conv max
                # Affected positions: k in [i, i+n-1] and k = 2i
                # conv_mod[k] = conv[k] + 2*(x-v)*a[k-i] for k in [i, i+n-1], k≠2i
                # conv_mod[2i] = conv[2i] - v^2 + x^2
                
                # Compute max efficiently
                # Unaffected max
                affected_mask = np.zeros(2*n-1, dtype=bool)
                affected_mask[i:min(i+n, 2*n-1)] = True
                unaffected_max = np.max(conv[~affected_mask]) if np.any(~affected_mask) else 0.0
                
                # Affected positions
                quad_val = conv[2*i] - v*v + x*x
                affected_vals = np.array([quad_val])
                
                k_range = np.arange(i, min(i+n, 2*n-1))
                k_range = k_range[k_range != 2*i]
                if len(k_range) > 0:
                    lin_vals = conv[k_range] + 2.0*(x-v)*a[k_range - i]
                    affected_vals = np.concatenate([affected_vals, lin_vals])
                
                mod_max = max(unaffected_max, np.max(affected_vals))
                F = c*c * mod_max
                
                if F < best_F:
                    best_F = F
                    best_x = x
            
            # Refine around best_x with bisection
            if best_x > 0:
                lo = max(0, best_x * 0.5)
                hi = best_x * 1.5
                for _ in range(20):
                    mid1 = lo + (hi - lo) / 3
                    mid2 = hi - (hi - lo) / 3
                    
                    def eval_F(x):
                        if x < 0:
                            return float('inf')
                        c = 1.0 / (1.0 - v + x)
                        quad_val = conv[2*i] - v*v + x*x
                        k_range = np.arange(i, min(i+n, 2*n-1))
                        k_range = k_range[k_range != 2*i]
                        affected_vals = np.array([quad_val])
                        if len(k_range) > 0:
                            lin_vals = conv[k_range] + 2.0*(x-v)*a[k_range - i]
                            affected_vals = np.concatenate([affected_vals, lin_vals])
                        affected_mask = np.zeros(2*n-1, dtype=bool)
                        affected_mask[i:min(i+n, 2*n-1)] = True
                        unaffected_max = np.max(conv[~affected_mask]) if np.any(~affected_mask) else 0.0
                        mod_max = max(unaffected_max, np.max(affected_vals))
                        return c*c * mod_max
                    
                    F1 = eval_F(mid1)
                    F2 = eval_F(mid2)
                    
                    if F1 < F2:
                        hi = mid2
                    else:
                        lo = mid1
                    
                    if hi - lo < 1e-10:
                        break
                
                best_x = (lo + hi) / 2
            
            # Update
            if abs(best_x - v) > 1e-10 and best_x >= 0:
                a[i] = best_x
                # Renormalize
                s = a.sum()
                if s < 1e-15:
                    a = best_a.copy()
                    break
                a = a / s
                conv = fast_convolve(a)
                if conv.max() < best_obj:
                    best_obj = conv.max()
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_a, best_obj

def simulated_annealing(a0, time_limit, rng):
    """Simulated annealing with transfer moves."""
    a = a0.astype(np.float64).copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    current_obj = conv.max()
    best_obj = current_obj
    best_a = a.copy()
    
    t0 = time.monotonic()
    temp = 0.01
    cool_rate = 0.999
    
    while time.monotonic() - t0 < time_limit:
        # Transfer move: move mass from i to j
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        
        # Amount to transfer (fraction of a[i])
        frac = rng.uniform(0.01, 0.3)
        delta = frac * a[i]
        
        if a[j] + delta < 0 or a[i] - delta < 0:
            continue
        
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        a_new = a_new / a_new.sum()
        
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()
        
        delta_obj = obj_new - current_obj
        if delta_obj < 0 or rng.random() < np.exp(-delta_obj / temp):
            a = a_new
            current_obj = obj_new
            if current_obj < best_obj:
                best_obj = current_obj
                best_a = a.copy()
        
        temp *= cool_rate
        if temp < 1e-8:
            temp = 0.005
            # Reset near best
            a = best_a + rng.standard_normal(n) * 0.002
            a = np.maximum(a, 0)
            a = a / a.sum()
            current_obj = fast_convolve(a).max()
    
    return best_a, best_obj
