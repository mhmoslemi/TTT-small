def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(f):
    """Build sparse constraint matrix for the LP. Shape: (2n-1, n)"""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend(f[k - js].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, A_cache=None, top_k=None):
    """Solve LP: max sum(g) s.t. (A@g)[top peaks] <= M, g >= 0.
    If top_k is specified, only constrain the top-k lag positions."""
    n = len(f)
    if A_cache is None:
        A_cache = build_lp_matrix(f)
    c = -np.ones(n)
    
    if top_k is not None and top_k < 2 * n - 1:
        # Only constrain top-k positions from current convolution
        conv = fast_convolve(f)
        top_idx = np.argpartition(conv, -top_k)[-top_k:]
        top_idx = top_idx[top_idx >= 0]  # safety
        A_sub = A_cache[top_idx, :]
        b_ub = np.full(len(top_idx), M)
    else:
        A_sub = A_cache
        b_ub = np.full(2 * n - 1, M)
    
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_sub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 20})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def power_sum_gradient(f, p):
    """Compute gradient of sum(b_k^p) where b = conv(f,f).
    grad_i = 2*p * sum_k b_k^(p-1) * f_{k-i}"""
    n = len(f)
    conv = fast_convolve(f)
    conv_p1 = conv ** (p - 1)  # b_k^(p-1)
    
    # grad[i] = 2*p * sum_m conv_p1[i+m] * f[m] for valid m
    f_flipped = f[::-1].copy()
    total_len = len(conv_p1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    C = np.fft.rfft(conv_p1, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    cross = np.fft.irfft(C * F, fft_size)
    grad = 2.0 * p * cross[n - 1:n - 1 + n]
    return grad, conv

    lr = 3.0 / n
    momentum = 0.92

    t0 = time.monotonic()
    no_improve = 0
    it = 0

        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        if use_power_sum:
            p = 2.0 + beta / 10.0  # Map beta to p
            grad, conv = power_sum_gradient(f, p)
        else:
            conv = fast_convolve(f)
            bc = beta * conv
            bc_max = bc.max()
            exp_bc = np.exp(bc - bc_max)
            w = exp_bc / exp_bc.sum()
            # Gradient: 2 * cross_correlation(w, f)
            f_flipped = f[::-1].copy()
            total_len = (2 * n - 1) + n - 1

        f_new = project_simplex(f + v)

def make_chirp(n, chirp_rate=1.0, window='hann'):
    """Create a chirp-based initialization."""
    t = np.arange(n) / n
    phase = 2 * np.pi * chirp_rate * t ** 2
    sig = np.abs(np.exp(1j * phase))  # |chirp| is flat
    # Better: use the real part absolute value with a window
    sig = np.cos(np.pi * chirp_rate * t * (t - 1)) ** 2
    if window == 'hann':
        sig *= 0.5 * (1 + np.cos(2 * np.pi * t))
    elif window == 'hamming':
        sig *= 0.54 + 0.46 * np.cos(2 * np.pi * t)
    sig = np.maximum(sig, 0.0)
    s = sig.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return sig / s


def make_periodic(n, period, duty=0.5):
    """Create a periodic pattern."""
    sig = np.zeros(n)
    for k in range(period):
        if k < period * duty:
            sig[k::period] = 1.0
    s = sig.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return sig / s

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)

    best_tracker = [float('inf'), None]

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
            try_sequence(parent, best_tracker)
    except Exception as e:
        print(f"No parent: {e}")

    if parent is None:
        parent = make_chirp(448, chirp_rate=2.0)
        try_sequence(parent, best_tracker)

    n_parent = len(parent)
    print(f"Starting optimization, parent n={n_parent}, best={best_tracker[0]:.8f}")

    # === Phase 1: FW-LP on parent (top-k mode) ===
    print("\n=== Phase 1: FW-LP (top-k) on parent ===")
    t_alloc = min(300, deadline - time.monotonic() - 600)
    if t_alloc > 30:
        print(f"  Time: {t_alloc:.0f}s")
        a_opt, _ = fw_lp_optimize(parent, time_limit=t_alloc, use_top_k=True, top_k_frac=0.25)
        try_sequence(a_opt, best_tracker)

    # === Phase 2: Standard FW-LP (full constraints) ===
    print("\n=== Phase 2: FW-LP (full) ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc = min(200, deadline - time.monotonic() - 500)
    if t_alloc > 20:
        print(f"  Time: {t_alloc:.0f}s")
        a_opt, _ = fw_lp_optimize(current_best, time_limit=t_alloc, use_top_k=False)
        try_sequence(a_opt, best_tracker)

    # === Phase 3: LSE polishing ===
    print("\n=== Phase 3: LSE Polishing ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc = min(300, deadline - time.monotonic() - 400)
    if t_alloc > 20:
        print(f"  Time: {t_alloc:.0f}s")
        a_opt, _ = lse_optimize(current_best, time_limit=t_alloc,
                                beta_start=8.0, beta_end=4000.0)
        try_sequence(a_opt, best_tracker)

    # === Phase 4: Peak shaving LP ===
    print("\n=== Phase 4: Peak Shaving ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc = min(150, deadline - time.monotonic() - 300)
    if t_alloc > 15:
        print(f"  Time: {t_alloc:.0f}s")
        a_opt, _ = peak_shave_lp(current_best, time_limit=t_alloc, num_peaks=8)
        try_sequence(a_opt, best_tracker)

    # === Phase 5: Power-sum gradient polish ===
    print("\n=== Phase 5: Power-Sum Gradient ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc = min(150, deadline - time.monotonic() - 200)
    if t_alloc > 15:
        print(f"  Time: {t_alloc:.0f}s")
        a_opt, _ = lse_optimize(current_best, time_limit=t_alloc,
                                beta_start=30.0, beta_end=2000.0, use_power_sum=True)
        try_sequence(a_opt, best_tracker)

    # === Phase 6: Explore alternative lengths ===
    print("\n=== Phase 6: Alternative Lengths ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_best = len(current_best) if current_best is not None else n_parent

    alt_lengths = []
    for n_alt in [256, 320, 384, 448, 512, 576, 640]:
        if n_alt != n_best:
            alt_lengths.append(n_alt)
    
    # Sort by proximity to current best length (tend to work well nearby)
    alt_lengths.sort(key=lambda x: abs(x - n_best))
    
    for n_alt in alt_lengths[:4]:
        t_remaining = deadline - time.monotonic() - 30
        if t_remaining < 50:
            break
        t_alloc = min(100, (t_remaining - 30) / max(1, len(alt_lengths[:4]) - alt_lengths.index(n_alt)))
        if t_alloc < 25:
            break

        print(f"  Trying n={n_alt}, time={t_alloc:.0f}s")
        
        # Interpolate from current best
        x_old = np.linspace(0, 1, len(current_best))
        x_new = np.linspace(0, 1, n_alt)
        f_interp = np.interp(x_new, x_old, current_best)
        f_interp = np.maximum(f_interp, 0)
        if f_interp.sum() > 1e-10:
            f_interp = f_interp / f_interp.sum()
        else:
            f_interp = make_chirp(n_alt, chirp_rate=2.0)

        try_sequence(f_interp, best_tracker)

        # Quick optimization
        t1 = min(t_alloc * 0.5, 50)
        a_opt, _ = fw_lp_optimize(f_interp, time_limit=t1, use_top_k=True, top_k_frac=0.3)
        try_sequence(a_opt, best_tracker)

        t2 = min(t_alloc * 0.4, 40)
        current_alt = best_tracker[1] if best_tracker[1] is not None and len(best_tracker[1]) == n_alt else a_opt
        a_opt, _ = lse_optimize(current_alt, time_limit=t2,
                                beta_start=15.0, beta_end=3000.0)
        try_sequence(a_opt, best_tracker)

    # === Phase 7: Final aggressive LSE on best found ===
    print("\n=== Phase 7: Final LSE ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_remaining = deadline - time.monotonic() - 10
    if t_remaining > 20:
        print(f"  Time: {t_remaining:.0f}s, n={len(current_best)}")
        a_opt, _ = lse_optimize(current_best, time_limit=t_remaining,
                                beta_start=50.0, beta_end=10000.0)
        try_sequence(a_opt, best_tracker)

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.8f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]
    
    # Fallback
    fallback = np.ones(384) / 384
    print(f"\nFALLBACK: n=384")
    return [float(x) for x in fallback]
