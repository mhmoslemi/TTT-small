Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a two-phase smooth optimization:
Phase 1: L2 norm of convolution (fully smooth, no hyperparameters)
Phase 2: LogSumExp approximation of max with adaptive temperature and exact FFT-based gradients.
Uses multiplicative updates (exponentiated gradient) with backtracking line search,
diverse Fourier-structured initializations, and multi-length exploration.

import numpy as np
import time


def next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def fft_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = next_pow2(size)
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:size]


def compute_grad_l2(a, c):
    """
    Gradient of ||conv(a,a)||_2^2 w.r.t. a.
    grad_i = 2 * [sum_k c_k * a_{k-i} + sum_k c_k * a_{i-k}]
    Computed via FFT.
    """
    n = len(a)
    size_c = len(c)  # 2n-1
    conv_len = n + size_c - 1  # 3n-2
    N = next_pow2(conv_len)

    A = np.fft.rfft(a, N)
    C = np.fft.rfft(c, N)

    # Q_i = sum_k c_k * a_{i-k} = (c * a)_i = ifft(C * A)_i
    Q = np.fft.irfft(C * A, N)[:n]

    # P_i = sum_k c_k * a_{k-i}
    # = (a * c_rev)[2n-2-i] where c_rev = c[::-1]
    C_rev = np.fft.rfft(c[::-1], N)
    P_full = np.fft.irfft(A * C_rev, N)
    P = P_full[2 * n - 2 : n - 1 : -1]  # indices 2n-2, 2n-3, ..., n-1

    return 2.0 * (P + Q)


def compute_grad_logsumexp(a, c, beta):
    """
    Gradient of (1/beta)*log(sum(exp(beta*c))) w.r.t. a.
    grad_i = (1/beta) * [sum_k w_k * a_{k-i} + sum_k w_k * a_{i-k}]
    where w = softmax(beta * c).
    """
    n = len(a)
    size_c = len(c)  # 2n-1
    conv_len = n + size_c - 1  # 3n-2
    N = next_pow2(conv_len)

    # Softmax weights
    bc = beta * c
    bc_shifted = bc - bc.max()
    w = np.exp(bc_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(size_c)
        w[np.argmax(bc)] = 1.0
    else:
        w /= w_sum

    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)

    # Q_i = sum_k w_k * a_{i-k} = (w * a)_i
    Q = np.fft.irfft(W * A, N)[:n]

    # P_i = sum_k w_k * a_{k-i} = (a * w_rev)[2n-2-i]
    W_rev = np.fft.rfft(w[::-1], N)
    P_full = np.fft.irfft(A * W_rev, N)
    P = P_full[2 * n - 2 : n - 1 : -1]

    return (P + Q) / beta

        return np.inf
    c = fft_convolve(a)
    return 2.0 * n * c.max() / (s * s)


def optimize_smooth(a0, time_limit, phase='l2', beta0=100.0, beta_final=5000.0,
                    eta0=0.5, eta_decay=0.9999, verbose=False):
    """
    Optimize using multiplicative updates with backtracking line search.
    phase='l2': minimize ||conv(a,a)||_2^2
    phase='lse': minimize (1/beta)*logsumexp(beta*conv(a,a)) with adaptive beta
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    c = fft_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    if phase == 'lse':
        beta = beta0
    else:
        beta = 1.0

    eta = eta0
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)

        # Adaptive beta for LSE phase
        if phase == 'lse':
            beta = beta0 * (beta_final / beta0) ** frac

        # Compute gradient
        if phase == 'l2':
            grad = compute_grad_l2(a, c)
        else:
            grad = compute_grad_logsumexp(a, c, beta)

        # Center the gradient (remove mean component since we renormalize)
        grad_centered = grad - (a * grad).sum()

        # Multiplicative update with backtracking
        obj_current = c.max()
        improved = False
        for _ls in range(10):
            a_cand = a * np.exp(-eta * grad_centered)
            a_cand = np.maximum(a_cand, 1e-300)
            sc = a_cand.sum()
            if sc < 1e-300:
                break
            a_cand = a_cand / sc
            c_cand = fft_convolve(a_cand)
            obj_cand = c_cand.max()

            if obj_cand < obj_current - 1e-15:
                a = a_cand
                c = c_cand
                obj_current = obj_cand
                improved = True
                if obj_cand < best_obj - 1e-12:
                    best_obj = obj_cand
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1
                break
            else:
                eta *= 0.5

        if not improved:
            eta *= 0.5
            no_improve += 1

        # Decay eta slowly
        eta *= eta_decay

        # Stagnation: perturb and restart
        if no_improve > 2000:
            rng = np.random.default_rng(it * 13 + 7)
            a = best_a + rng.standard_normal(n) * 0.003
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            c = fft_convolve(a)
            best_obj = c.max()
            no_improve = 0
            eta = eta0 * 0.3

        if verbose and it % 5000 == 0:
            print(f"      iter={it}, obj={best_obj:.10f}, beta={beta:.1f}, eta={eta:.6f}")

    return best_a.copy(), best_obj


def optimize_l2_lse(a0, time_limit, verbose=False):
    """Two-phase: first L2, then LSE with increasing temperature."""
    t_l2 = time_limit * 0.4
    t_lse_1 = time_limit * 0.3
    t_lse_2 = time_limit * 0.3

    # Phase 1: L2 objective
    a_mid, obj_l2 = optimize_smooth(a0, time_limit=t_l2, phase='l2',
                                     eta0=0.8, eta_decay=0.99995, verbose=False)
    if verbose:
        print(f"    After L2: max_conv={obj_l2:.10f}")

    # Phase 2a: LSE with moderate temperature
    a_mid2, obj_lse1 = optimize_smooth(a_mid, time_limit=t_lse_1, phase='lse',
                                        beta0=50.0, beta_final=500.0,
                                        eta0=0.3, eta_decay=0.9999, verbose=False)
    if verbose:
        print(f"    After LSE1: max_conv={obj_lse1:.10f}")

    # Phase 2b: LSE with high temperature (sharp)
    a_final, obj_lse2 = optimize_smooth(a_mid2, time_limit=t_lse_2, phase='lse',
                                         beta0=200.0, beta_final=10000.0,
                                         eta0=0.1, eta_decay=0.99995, verbose=False)
    if verbose:
        print(f"    After LSE2: max_conv={obj_lse2:.10f}")

    return a_final.copy()
