"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Analytic softmax-gradient with FFT-accelerated cross-correlation
2. Simplex projection for non-negativity and normalization
3. Multi-scale (coarse-to-fine) optimization
4. Adaptive temperature for softmax (from smooth to sharp)
5. Simulated annealing restarts to escape local minima
6. Coordinate descent polishing
"""

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.argmax(1 + u * np.arange(1, n + 1) > cv)
    theta = (cv[rho] - target_sum) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_gradient(a, conv, tau):
    """
    Compute gradient of soft_max(conv, tau) w.r.t. a using FFT cross-correlation.
    soft_max = (1/tau) * log(sum(exp(tau*c)))
    d(soft_max)/da[j] = 2 * sum_k p[k] * a[k-j]
    where p[k] = exp(tau*c[k]) / sum(exp(tau*c))
    """
    n = len(a)
    c_len = len(conv)  # 2n-1
    
    # Compute softmax weights
    scaled = tau * conv
    scaled -= scaled.max()
    exp_vals = np.exp(scaled)
    p = exp_vals / exp_vals.sum()
    
    # Compute cross-correlation: corr[j] = sum_k p[k] * a[k-j]
    # Using FFT: corr = IFFT(conj(FFT(p)) * FFT(a))
    fft_size = 1 << (c_len + n - 1 - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    A = np.fft.rfft(a, fft_size)
    corr = np.fft.irfft(np.conj(P) * A, fft_size)[:n]
    
    return 2.0 * corr

def soft_gradient_descent(a0, time_limit, tau_init=0.1, tau_final=10.0, 
                          lr_init=0.01, lr_final=0.0001, 
                          n_iter_max=None, seed=0):
    """
    Gradient descent with simplex projection and adaptive parameters.
    """
    a = project_simplex(a0.copy(), 1.0)
    n = len(a)
    
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0
    max_no_improve = 50
    
    if n_iter_max is None:
        n_iter_max = 100000
    
    for it in range(n_iter_max):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Progress from 0 to 1
        progress = min(1.0, it / (2000.0 if it < 2000 else it))
        
        # Adaptive parameters (exponential schedule)
        tau = tau_init * (tau_final / tau_init) ** progress
        lr = lr_init * (lr_final / lr_init) ** progress
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Compute gradient
        grad = compute_gradient(a, conv, tau)
        
        # Gradient step
        a_new = a - lr * grad
        
        # Project onto simplex
        a = project_simplex(a_new, 1.0)
        
        # Simulated annealing: occasionally accept worse solutions
        if no_improve > max_no_improve and rng.random() < 0.1:
            noise_scale = 0.01 * (1 + no_improve / max_no_improve)
            a = project_simplex(a + rng.standard_normal(n) * noise_scale, 1.0)
            no_improve = 0
        
        if no_improve > 500:
            break
    
    return best_a, best_obj

def multi_scale_optimize(n_target, time_limit, seed=0):
    """
    Multi-scale: start small, optimize, upsample, optimize larger.
    """
    # Start with n=32, double each time until n_target
    n = 32
    a = np.ones(n) / n
    t0 = time.monotonic()
    
    while n < n_target and time.monotonic() - t0 < time_limit - 30:
        remaining = time_limit - (time.monotonic() - t0)
        t_alloc = remaining * 0.3  # 30% of remaining time for this scale
        t_alloc = max(5.0, t_alloc)
        
        a, obj = soft_gradient_descent(a, time_limit=t_alloc, seed=seed + n)
        
        # Upsample to 2n
        n_new = min(2 * n, n_target)
        if n_new > n:
            a_new = np.zeros(n_new)
            for i in range(n):
                # Linear interpolation
                pos = i * (n_new - 1) / (n - 1) if n > 1 else 0
                i0 = int(pos)
                i1 = min(i0 + 1, n_new - 1)
                frac = pos - i0
                a_new[i0:i1+1] += a[i] * [1-frac, frac] if i1 > i0 else [1.0]
            a = project_simplex(a_new, 1.0)
        
        n = len(a)
    
    # Final optimization at target size
    a = project_simplex(a[:n_target], 1.0)
    remaining = time_limit - (time.monotonic() - t0)
    if remaining > 10:
        a, obj = soft_gradient_descent(a, time_limit=remaining, 
                                         tau_init=1.0, tau_final=50.0,
                                         lr_init=0.005, lr_final=0.00001,
                                         seed=seed + 1000)
    
    return a, obj

def coordinate_polish(a0, time_limit, seed=0):
    """
    Coordinate descent: try adjusting one element at a time.
    For each element j, find the optimal value that minimizes the objective
    (given all other elements fixed).
    """
    a = project_simplex(a0.copy(), 1.0)
    n = len(a)
    best_a = a.copy()
    best_obj = 2.0 * n * fast_convolve(a).max()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    
    while time.monotonic() - t0 < time_limit:
        # Randomly permute coordinates
        order = rng.permutation(n)
        improved = False
        
        for j in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            # Try several values for a[j]
            current_val = a[j]
            remain_sum = 1.0 - current_val
            
            # Candidate values
            candidates = [current_val * 0.5, current_val * 0.8, current_val * 1.2, 
                         current_val * 1.5, current_val * 0.1, 0.0,
                         remain_sum * 0.1, remain_sum * 0.01]
            candidates = [max(0.0, min(c, remain_sum)) for c in candidates]
            candidates.append(current_val)  # Don't change
            candidates = list(set(round(c, 10) for c in candidates))
            
            best_try = current_val
            best_try_obj = 2.0 * n * fast_convolve(a).max()
            
            for val in candidates:
                if abs(val - current_val) < 1e-12:
                    continue
                a[j] = val
                # Re-normalize
                a = project_simplex(a, 1.0)
                conv = fast_convolve(a)
                obj = 2.0 * n * conv.max()
                if obj < best_try_obj - 1e-12:
                    best_try_obj = obj
                    best_try = a[j]
            
            a[j] = best_try
            a = project_simplex(a, 1.0)
            
            if best_try_obj < best_obj - 1e-12:
                best_obj = best_try_obj
                best_a = a.copy()
                improved = True
        
        if not improved:
            break
    
    return best_a, best_obj

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Analytic softmax-gradient with FFT-accelerated cross-correlation  
2. Simplex projection for non-negativity and normalization
3. Adaptive temperature schedule (smooth to sharp)
4. Backtracking line search
5. Multi-start with diverse initializations
6. Coordinate descent polishing
7. Simulated annealing restarts
"""

import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v, s=1.0):
    n = len(v)
    if n == 0:
        return v.copy()
    # Standard simplex projection algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(1 + u * np.arange(1, n+1) > cssv)[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_gradient_fft(a, conv, tau):
    """Analytic gradient of soft_max(conv, tau) with respect to a."""
    n = len(a)
    c_len = len(conv)
    
    # Numerically stable softmax
    scaled = tau * conv
    scaled -= scaled.max()
    exp_s = np.exp(scaled)
    p = exp_s / exp_s.sum()
    
    # Cross-correlation via FFT: (p * a)[j] = sum_k p[k] * a[k-j]
    # = IFFT(conj(FFT(p)) * FFT(a))
    fft_size = 1 << (c_len + n - 1 - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    A = np.fft.rfft(a, fft_size)
    corr = np.fft.irfft(np.conj(P) * A, fft_size)[:n]
    
    return 2.0 * corr

def objective(a):
    """Compute 2n*max(conv)/(sum)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)

def soft_max_objective(a, tau):
    """Smooth version of the objective."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    scaled = tau * conv
    scaled -= scaled.max()
    sm = (1.0/tau) * np.log(np.sum(np.exp(scaled)))
    return 2.0 * n * sm / (s * s)

def gd_optimize(a0, time_limit, tau_start=0.1, tau_end=50.0,
                lr_start=0.05, lr_end=0.0001, seed=0,
                anneal_start_iter=500, anneal_prob=0.05):
    """
    Gradient descent with simplex projection, adaptive tau and lr,
    and simulated annealing restarts.
    """
    a = project_simplex(a0.astype(np.float64).copy(), 1.0)
    n = len(a)
    
    best_a = a.copy()
    best_obj = objective(a)
    
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Progress 0->1 over ~3000 iterations
        prog = min(1.0, it / 3000.0)
        
        # Adaptive parameters
        tau = tau_start * (tau_end / tau_start) ** prog
        lr = lr_start * (lr_end / lr_start) ** prog
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Compute gradient
        grad = compute_gradient_fft(a, conv, tau)
        
        # Gradient step with simplex projection
        a_new = a - lr * grad
        a_new = project_simplex(a_new, 1.0)
        
        # Check improvement
        conv_new = fast_convolve(a_new)
        obj_new = 2.0 * n * conv_new.max()
        
        if obj_new < cur_obj:
            a = a_new
        elif it > anneal_start_iter and rng.random() < anneal_prob * min(1.0, (it - anneal_start_iter) / 1000.0):
            # Simulated annealing: accept or explore
            a = a_new  # Accept even if worse (will revert if too bad)
            if objective(a) > best_obj * 1.5:
                a = best_a.copy()  # Revert if too bad
        else:
            a = a_new
        
        # Periodic restart from best
        if no_improve > 300 and rng.random() < 0.3:
            noise = rng.standard_normal(n) * 0.005
            a = project_simplex(best_a + noise, 1.0)
            no_improve = 0
        
        if no_improve > 1000:
            break
    
    return best_a, best_obj

def random_restart_gd(a0, n_restarts, time_per_restart, seed=0, **gd_kwargs):
    """Try multiple random restarts around a0."""
    best_a = a0.copy()
    best_obj = objective(a0)
    rng = np.random.default_rng(seed)
    n = len(a0)
    
    for r in range(n_restarts):
        # Perturb
        scale = 0.01 + 0.02 * r
        a_start = project_simplex(a0 + rng.standard_normal(n) * scale, 1.0)
        
        a_opt, obj = gd_optimize(a_start, time_limit=time_per_restart,
                                  seed=seed + r * 100, **gd_kwargs)
        if obj < best_obj:
            best_obj = obj
            best_a = a_opt.copy()
    
    return best_a, best_obj

def coord_descent(a0, time_limit, seed=0):
    """
    Coordinate descent: for each coordinate, try to optimize it
    by trying several values.
    """
    a = project_simplex(a0.astype(np.float64).copy(), 1.0)
    n = len(a)
    best_a = a.copy()
    best_obj = objective(a)
    
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    pass_count = 0
    
    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        # Shuffle coordinates
        order = rng.permutation(n)
        improved = False
        
        for j in order:
            if time.monotonic() - t0 > time_limit - 0.1:
                break
            
            save_val = a[j]
            remain = 1.0 - save_val
            
            # Try geometrically spaced values
            if save_val > 1e-10:
                ratios = [0.3, 0.5, 0.7, 1.3, 1.5, 2.0, 0.1, 3.0]
                candidates = [save_val * r for r in ratios]
            else:
                candidates = [remain * 0.01, remain * 0.05, remain * 0.1, remain * 0.3]
            
            # Also try 0 and remaining
            candidates.append(0.0)
            candidates.append(remain * 0.9)
            
            best_val = save_val
            best_local_obj = best_obj  # Current is baseline
            
            for cval in candidates:
                cval = max(0.0, min(cval, remain + save_val))  # Can't exceed 1
                if cval < 0:
                    continue
                a[j] = cval
                a = project_simplex(a, 1.0)
                conv = fast_convolve(a)
                obj = 2.0 * n * conv.max()
                if obj < best_local_obj - 1e-13:
                    best_local_obj = obj
                    best_val = a[j]
            
            a[j] = best_val
            a = project_simplex(a, 1.0)
            
            if best_local_obj < best_obj - 1e-13:
                best_obj = best_local_obj
                best_a = a.copy()
                improved = True
        
        if not improved and pass_count > 3:
            break
    
    return best_a, best_obj

def generate_diverse_inits(n, parent, rng):
    """Generate diverse initial sequences."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits.append(('uniform', np.ones(n) / n))
    
    # Sine
    s = np.sin(np.pi * x / n) ** 2 + 0.01
    inits.append(('sine', s / s.sum()))
    
    # Bimodal
    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits.append(('bimodal', bi / bi.sum()))
    
    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5*n) / (0.15*n)) ** 2)
    inits.append(('gaussian', g / g.sum()))
    
    # Alternating
    alt = np.ones(n)
    alt[::2] = 1.5
    alt[1::2] = 0.5
    inits.append(('alternating', alt / alt.sum()))
    
    # Random (Dirichlet)
    r = rng.dirichlet(np.ones(n) * 0.1)
    inits.append(('random', r))
    
    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
            if len(p) == n:
                inits.append(('parent', p.copy()))
            else:
                pi = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pi = np.maximum(pi, 0)
                pis = pi.sum()
                if pis > 1e-10:
                    inits.append(('parent_interp', pi / pis))
    
    return inits
