        # Restart
        if no_improve > patience:
            rng = np.random.default_rng(int(time.monotonic() * 100000) % 1000000)
            scale = 0.001 + 0.003 * min(1.0, no_improve / (2 * patience))
            f = np.maximum(best_f * np.exp(rng.standard_normal(n) * scale), 1e-15)
            f = f / f.sum()
            cur_lr = max(lr_init * 0.5, 0.005)
            cur_T = T_init * 2
            no_improve = 0
            if verbose:
                print(f"    RESTART at it={it}, obj={best_obj:.8f}")
    
    return best_f, best_obj

def gradient_descent(f0, time_limit, T_init=200, T_growth=1.002,
                     lr_init=1.0, lr_decay=0.999,
                     patience=200, verbose=True):
    """Subgradient-like method with LSE smoothing and simplex projection."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_T = T_init
    cur_lr = lr_init
    it = 0
    momentum = np.zeros(n)
    mom_factor = 0.9
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        scaled = cur_T * conv
        scaled -= scaled.max()
        p = np.exp(scaled)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p /= p_sum
        
        grad = 2.0 * cross_corr_fft(p, f)
        
        # Nesterov-like momentum
        momentum = mom_factor * momentum - cur_lr * grad
        f_new = f + momentum
        f_new = project_simplex(f_new)
        
        conv_new = fast_convolve(f_new)
        obj = conv_new.max()
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
        else:
            no_improve += 1
            f = f_new  # still update to explore
        
        cur_T = min(cur_T * T_growth, 3000.0)
        cur_lr *= lr_decay
        cur_lr = max(cur_lr, 0.01)
        
        if no_improve > patience:
            rng = np.random.default_rng(int(time.monotonic() * 100000) % 1000000)
            scale = 0.002 + 0.004 * min(1.0, no_improve / (2*patience))
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            momentum = np.zeros(n)
            cur_lr = max(lr_init * 0.6, 0.05)
            cur_T = T_init
            no_improve = 0
    
    return best_f, best_obj

def coord_descent(f0, time_limit, verbose=True):
    """
    Coordinate descent: for each coordinate j, optimize a_j while keeping others fixed.
    The max of conv as a function of a_j is the max of a set of quadratic functions.
    We use ternary search or direct minimization.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        improved = False
        # Random order of coordinates
        order = np.random.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit - 2:
                break
            
            # Fix all a_i for i != j, optimize a_j
            # conv_k = C_k + 2*a_j*a_{k-j} - a_j^2 * 1[k=2j]
            # where C_k = conv_k - 2*a_j*a_{k-j} + a_j^2*1[k=2j]
            
            a_j = f[j]
            others_sum = 1.0 - a_j  # since sum = 1
            
            # Compute C_k (convolution without the a_j terms)
            # C_k = conv_k - 2*a_j*a_{k-j} + a_j^2*1[k=2j]
            conv_old = fast_convolve(f)
            
            # For the new a_j = t, the new convolution at position k is:
            # new_conv_k = C_k + 2*t*a_{k-j} - t^2*1[k=2j]
            # = conv_old_k - 2*a_j*a_{k-j} + a_j^2*1[k=2j] + 2*t*a_{k-j} - t^2*1[k=2j]
            # = conv_old_k + 2*(t-a_j)*a_{k-j} + (a_j^2-t^2)*1[k=2j]
            
            # We want to minimize max_k new_conv_k over t in [0, others_sum]
            # (since sum must remain 1, and a_j can range from 0 to 1-sum(others))
            
            # This is a 1D minimization of max of linear/quadratic functions.
            # Use a simple grid + refinement.
            
            # The function is convex in t (max of convex functions), so we can use ternary search.
            # But it's piecewise quadratic, so let's just use a fine grid.
            
            lo, hi = 0.0, others_sum
            # First, coarse grid
            best_t = a_j
            best_val = conv_old.max()
            
            n_grid = 20
            for t in np.linspace(lo, hi, n_grid):
                # Compute new conv (only need to check the max)
                # new_conv_k = conv_old_k + 2*(t-a_j)*a_{k-j} + (a_j^2-t^2)*1[k=2j]
                # The change is small if t is close to a_j, so let's compute the max efficiently
                delta = 2.0 * (t - a_j)
                # a_{k-j} for k=0..2n-2 is: 0 for k<j, a_{k-j} for j<=k<=j+n-1, 0 otherwise
                a_shifted = np.zeros(2*n-1)
                a_shifted[j:j+n] = f
                new_conv = conv_old + delta * a_shifted + (a_j**2 - t**2) * np.zeros(2*n-1)
                # Handle the k=2j term
                if 2*j < 2*n-1:
                    new_conv[2*j] += (a_j**2 - t**2)
                # Wait, I already included that. Let me redo.
                # new_conv_k = conv_old_k - 2*a_j*a_{k-j} + a_j^2*1[k=2j] + 2*t*a_{k-j} - t^2*1[k=2j]
                # = conv_old_k + 2*(t-a_j)*a_{k-j} + (a_j^2-t^2)*1[k=2j]
                
                val = new_conv.max()
                if val < best_val:
                    best_val = val
                    best_t = t
            
            # Refine around best_t
            if best_t > lo and best_t < hi:
                lo2 = max(lo, best_t - (hi-lo)/n_grid)
                hi2 = min(hi, best_t + (hi-lo)/n_grid)
                for t in np.linspace(lo2, hi2, 20):
                    delta = 2.0 * (t - a_j)
                    a_shifted = np.zeros(2*n-1)
                    a_shifted[j:j+n] = f

                        best_t = t
            
            if best_val < conv_old.max() - 1e-15:
                f[j] = best_t
                f = f / f.sum()  # renormalize (should be ~1 already)

        obj = conv.max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
            if verbose and it % 10 == 0:

        elif not improved:

            rng = np.random.default_rng(it)

    return best_f, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-accelerated optimization:
- Log-sum-exp surrogate with adaptive temperature for smooth gradients
- Mirror descent (multiplicative weight) updates preserving positivity
- Momentum-based gradient descent with simplex projection
- Coordinate descent for final polish
- Periodic Frank-Wolfe with LP oracle for escaping local minima
- Multi-phase schedule with diverse restarts
Key advantage over previous approach: O(n log n) gradient computation via FFT
allows 10-100x more iterations in the same time budget.
"""

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT. O(n log n)."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def cross_corr_conv(p, a):
    """
    Compute xc[j] = sum_k p[k] * a[k-j] for j=0..n-1
    where len(p) = 2n-1, len(a) = n.
    Uses FFT: xc = conv(p, a[::-1])[n-1 : 2n-1]
    """
    n = len(a)
    a_flipped = a[::-1]
    fft_size = 1 << (3 * n - 3).bit_length()
    P = np.fft.rfft(p, fft_size)
    B = np.fft.rfft(a_flipped, fft_size)
    result = np.fft.irfft(P * B, fft_size)
    return result[n - 1: 2 * n - 1]

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.sum(cond)) - 1
    if rho < 0:
        return np.ones(n) / n
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def softmax_weights(conv, T):
    """Compute softmax(conv / T) -> normalized weights over convolution positions."""
    scaled = T * conv
    scaled -= scaled.max()
    p = np.exp(scaled)
    s = p.sum()
    if s < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p /= s
    return p

def compute_gradient(a, T):
    """Compute gradient of LSE_T(conv(a,a)) w.r.t. a using FFT. O(n log n)."""
    n = len(a)
    conv = fast_convolve(a)
    p = softmax_weights(conv, T)
    grad = 2.0 * cross_corr_conv(p, a)
    return conv, p, grad

def lse_mirror_descent(f0, time_limit, T_init=150, T_max=5000, T_growth=1.004,
                       lr_init=0.15, lr_min=0.002, lr_decay=0.9992,
                       patience=400, verbose=True):
    """
    Mirror descent on LSE surrogate. Update: a <- a * exp(-lr * grad), renormalize.
    Naturally preserves positivity. O(n log n) per iteration.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = float(conv.max())
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_T = T_init
    cur_lr = lr_init
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv, p, grad = compute_gradient(f, cur_T)
        
        # Multiplicative update
        # To avoid underflow, subtract max of lr*grad
        log_mult = -cur_lr * grad
        log_mult -= log_mult.max()  # normalize for numerical stability
        f = f * np.exp(log_mult)
        f = np.maximum(f, 1e-30)
        f = f / f.sum()
        
        conv = fast_convolve(f)
        obj = float(conv.max())
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 2000 == 0:
                print(f"    MD it={it}, obj={obj:.8f}, T={cur_T:.0f}, lr={cur_lr:.5f}")
        else:
            no_improve += 1
        
        cur_T = min(cur_T * T_growth, T_max)
        cur_lr = max(cur_lr * lr_decay, lr_min)
        
        if no_improve > patience:
            rng = np.random.default_rng(int(time.monotonic() * 100000) % 1000000)
            scale = 0.002 + 0.004 * min(1.0, no_improve / (2.0 * patience))
            f = np.maximum(best_f * np.exp(rng.standard_normal(n) * scale), 1e-15)
            f = f / f.sum()
            cur_lr = max(lr_init * 0.6, lr_min * 3)
            cur_T = T_init * 2.0
            no_improve = 0
            if verbose:
                print(f"    MD RESTART it={it}, best={best_obj:.8f}")
    
    return best_f, best_obj

def lse_momentum_grad(f0, time_limit, T_init=200, T_max=3000, T_growth=1.003,
                      lr_init=0.5, lr_min=0.01, lr_decay=0.999,
                      mom=0.85, patience=300, verbose=True):
    """
    Gradient descent with momentum on LSE surrogate + simplex projection.
    O(n log n) per iteration via FFT gradients.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = float(conv.max())
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_T = T_init
    cur_lr = lr_init
    momentum = np.zeros(n)
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv, p, grad = compute_gradient(f, cur_T)
        
        # Momentum update
        momentum = mom * momentum - cur_lr * grad
        f_new = f + momentum
        f_new = project_simplex(f_new)
        
        conv_new = fast_convolve(f_new)
        obj = float(conv_new.max())
        
        f = f_new
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 1000 == 0:
                print(f"    MG it={it}, obj={obj:.8f}, T={cur_T:.0f}, lr={cur_lr:.5f}")
        else:
            no_improve += 1
        
        cur_T = min(cur_T * T_growth, T_max)
        cur_lr = max(cur_lr * lr_decay, lr_min)
        
        if no_improve > patience:
            rng = np.random.default_rng(int(time.monotonic() * 100000) % 1000000)
            scale = 0.002 + 0.003 * min(1.0, no_improve / (2.0 * patience))
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            momentum = np.zeros(n)
            cur_lr = max(lr_init * 0.5, lr_min * 2)
            cur_T = T_init
            no_improve = 0
            if verbose:
                print(f"    MG RESTART it={it}, best={best_obj:.8f}")
    
    return best_f, best_obj

def solve_lp_oracle(f, M):
    """LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def fw_lp(f0, time_limit, verbose=True):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = float(conv.max())
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        g = solve_lp_oracle(f, M * m_exp)
        if g is None:
            m_exp *= 1.2
            if m_exp > 10:
                m_exp = 1.0
            continue
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 50:
                break
            m_exp = min(m_exp * 1.05, 5.0)
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = float(fast_convolve(ft).max())
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 0:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = float(conv.max())
            m_exp = 1.0
            if M < best_obj - 1e-13:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 5.0)
        
        if no_improve > 30:
            rng = np.random.default_rng(it * 7 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            M = float(conv.max())
            m_exp = 1.0
            no_improve = 0
    
    return best_f, best_obj

def coord_descent_polish(f0, time_limit, n_grid=30, verbose=True):
    """
    Coordinate descent: for each j, find optimal a_j given others fixed.
    The objective max_k conv_k as a function of a_j is convex (max of quadratics).
    Use golden section or fine grid search.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    best_obj = float(fast_convolve(f).max())
    best_f = f.copy()
    
    t0 = time.monotonic()
    sweep = 0
    total_improvement = 0.0
    
    while time.monotonic() - t0 < time_limit:
        sweep += 1
        order = np.random.permutation(n)
        sweep_improved = False
        
        for j in order:
            if time.monotonic() - t0 > time_limit - 1:
                break
            
            a_j = f[j]
            # Range for a_j: [0, 1 - sum(others)] = [0, 1 - (1 - a_j)] = [0, 1]
            # But more tightly: [0, 1] since all others are non-negative and sum to 1-a_j
            
            # Precompute the "fixed" part of the convolution
            # conv_k = C_k + 2*t * a[k-j] - t^2 * 1[k=2j] where t is the new a_j
            # C_k = conv_k - 2*a_j*a[k-j] + a_j^2*1[k=2j]
            
            conv_old = fast_convolve(f)
            
            # a[k-j] for k = 0..2n-2:
            #   a[k-j] is defined for 0 <= k-j < n, i.e., j <= k <= j+n-1
            a_shifted = np.zeros(2 * n - 1)
            a_shifted[j:j+n] = f
            
            # Try a range of values for t
            # Focus on [max(0, a_j - 0.5*mean_a), min(1, a_j + 0.5*mean_a)]
            mean_a = 1.0 / n
            lo = max(0.0, a_j - 2.0 * mean_a)
            hi = min(1.0, a_j + 2.0 * mean_a)
            
            if hi - lo < 1e-15:
                continue
            
            best_t = a_j
            best_val = float(conv_old.max())
            
            # Coarse grid
            for t in np.linspace(lo, hi, n_grid):
                new_conv = conv_old + 2.0 * (t - a_j) * a_shifted
                if 2 * j < 2 * n - 1:
                    new_conv[2 * j] += (a_j * a_j - t * t)
                val = float(new_conv.max())
                if val < best_val:
                    best_val = val
                    best_t = t
            
            # Fine grid around best_t
            step = (hi - lo) / n_grid
            lo2 = max(0.0, best_t - step)
            hi2 = min(1.0, best_t + step)
            for t in np.linspace(lo2, hi2, n_grid // 2):
                new_conv = conv_old + 2.0 * (t - a_j) * a_shifted
                if 2 * j < 2 * n - 1:
                    new_conv[2 * j] += (a_j * a_j - t * t)
                val = float(new_conv.max())
                if val < best_val:
                    best_val = val
                    best_t = t
            
            if best_val < float(conv_old.max()) - 1e-14:
                f[j] = best_t
                f[j] = max(f[j], 0.0)
                s = f.sum()
                if s < 1e-15:
                    f = np.ones(n) / n
                else:
                    f = f / s
                sweep_improved = True
        
        if sweep_improved:
            obj = float(fast_convolve(f).max())
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f.copy()
                total_improvement += 1
                if verbose and sweep % 5 == 0:
                    print(f"    CD sweep={sweep}, obj={obj:.8f}")
    
    return best_f, best_obj
