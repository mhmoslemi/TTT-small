import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.flatnonzero(cond)[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    A_eq = np.ones((1, n))
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})

    return None

    A_eq = np.ones((1, n))
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None

    n = len(f)

        f = np.ones(n) / n
    else:
        f = f / s
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    sweep = 0

        sweep += 1
        improved = False
        
        for i in range(n):
            old_val = f[i]

            # Contribution of f[i] to conv[k]:
            #   k = 2*i: f[i]^2
            #   0 <= k-i < n, k != 2*i: 2*f[i]*f[k-i]
            
            # Vectorized computation
            base = conv.copy()
            
            # Remove f[i]^2 at position 2*i
            if 2 * i < 2 * n - 1:

            k_end = min(2 * n - 2, i + n - 1)
            if k_start <= k_end:
                ks = np.arange(k_start, k_end + 1)
                mask = ks != 2 * i

            # Precompute coefficients
            # h(t) = max(base + contrib(t))
            # contrib(t)[k] = t^2 at k=2i; 2t*f[k-i] for valid k != 2i

            hi = max(old_val * 3.0, 4.0 / n)

            for _ in range(3):
                h_hi = h_at_0  # start
                # Compute h(hi) quickly
                total_hi = base + 0.0  # base + contrib(hi)
                # Actually let me just evaluate h at hi
                # This is getting complex, let me just do a simple search

            test_points = np.concatenate([
                [0.0],
                np.linspace(0.01/n, hi, 8)
            ])
            
            h_vals = np.empty(len(test_points))
            for idx, t in enumerate(test_points):
                # h(t) = max of (base[k] + contrib_k(t))
                val = base.copy()
                if 2 * i < 2 * n - 1:
                    val[2 * i] += t * t
                if k_start <= k_end:
                    if mask.any():
                        val[ks[mask]] += 2.0 * t * f[ks[mask] - i]
                h_vals[idx] = val.max()

            best_idx = np.argmin(h_vals)
            best_t = test_points[best_idx]
            best_h = h_vals[best_idx]

                lo = test_points[max(0, best_idx - 1)]
            else:
                lo = 0.0
            if best_idx < len(test_points) - 1:
                hi = test_points[min(len(test_points)-1, best_idx + 1)]
            else:
                hi = test_points[-1]
            
            for _ in range(8):
                mid1 = lo + (hi - lo) / 3
                mid2 = hi - (hi - lo) / 3
                
                # h(mid1)
                val1 = base.copy()
                if 2 * i < 2 * n - 1:
                    val1[2 * i] += mid1 * mid1
                if k_start <= k_end and mask.any():
                    val1[ks[mask]] += 2.0 * mid1 * f[ks[mask] - i]
                h1 = val1.max()
                
                # h(mid2)
                val2 = base.copy()
                if 2 * i < 2 * n - 1:
                    val2[2 * i] += mid2 * mid2
                if k_start <= k_end and mask.any():
                    val2[ks[mask]] += 2.0 * mid2 * f[ks[mask] - i]

                    hi = mid2
                else:
                    lo = mid1
                if hi - lo < 1e-15:
                    break
            
            best_t = (lo + hi) / 2
            
            # Evaluate and update
            if best_t != old_val:
                f[i] = best_t

                # k=2i: new_val^2 - old_val^2
                # valid k != 2i: 2*(new_val - old_val)*f[k-i]
                delta = best_t - old_val

                if k_start <= k_end and mask.any():
                    conv[ks[mask]] += 2.0 * delta * f[ks[mask] - i]
            
            # Check time

            best_f = f.copy()
            no_improve = 0

            no_improve += 1

            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            conv = fast_convolve(f)
            no_improve = 0
    
    return best_f, best_obj

Wait, there's an issue with my conv update. When I update f[i] from old_val to best_t, the change in conv[k] is:

conv_new[k] - conv_old[k] = (contribution of new_val) - (contribution of old_val)

For k = 2i: best_t^2 - old_val^2
For valid k ≠ 2i: 2*(best_t - old_val)*f[k-i]

This is correct. But I need to be careful: after updating f[i], when I later update f[j] (j≠i), the "base" computation uses the current conv which already includes the updated f[i]. So the incremental update approach is correct.

However, there's a subtlety: when I compute base for coordinate i, I'm subtracting f[i]'s contribution from the current conv. This is correct because conv already reflects all previous updates.

Let me also think about vectorizing the inner loop more. The main bottleneck is the O(n) work per coordinate (computing base and evaluating h). For n=384, a full sweep is 384 * 384 ≈ 147K operations (plus the ternary search overhead). With ~15 evaluations per coordinate, that's ~2.2M per sweep. At ~100 operations per clock cycle (rough estimate for numpy), that's ~22 microseconds per sweep... no, that's too optimistic. Let's say ~1ms per sweep. So we can do ~1000 sweeps in 1 second. That's plenty.

Actually, the real bottleneck is the `base = conv.copy()` which is O(n) memory allocation. For 384 coordinates × 384 values × 15 evaluations = ~2.2M float operations per sweep, plus memory allocations. In numpy, this should be fast.

Let me also think about whether I can avoid the copy. I could pre-allocate a buffer:

Or even better, compute h(t) without explicitly forming the full array:

h(t) = max(base.max(), base[2*i] + t^2, (base[ks[mask]] + 2*t*f[ks[mask]-i]).max())

This avoids the copy entirely! Let me use this.

def evaluate_h(base, t, i, f, n, ks, mask):
    """Compute h(t) = max of conv when f[i]=t."""
    result = base.max()
    if 2 * i < 2 * n - 1:

        if v > result:
            result = v
    if mask.any():

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast vectorized coordinate
descent with LP-based Frank-Wolfe escapes and multi-scale exploration.
Key innovation: highly optimized 1D coordinate updates that avoid full
convolution recomputation, enabling many fast sweeps.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.flatnonzero(cond)[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, M, time_limit=15):
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    A_eq = np.ones((1, n))
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            x = np.maximum(res.x, 0.0)
            s = x.sum()
            if s > 1e-15:
                return x / s
    except Exception:
        pass
    return None


def run_cd(f, time_limit, verbose=False):
    """Fast coordinate descent minimizing max(auto-conv) subject to sum=1, f>=0."""
    n = len(f)
    f = np.maximum(f.astype(np.float64), 0.0)
    f = f / f.sum()
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    sweep = 0
    
    while time.monotonic() - t0 < time_limit:
        sweep += 1
        
        for i in range(n):
            old_val = f[i]
            if old_val < 1e-18:
                continue
                
            # Compute base (conv without f[i]'s contribution)
            # We'll compute h(t) = max conv value when f[i] = t
            
            # The contribution of f[i]=t to conv[k]:
            #   k = 2i: t^2
            #   0 <= k-i < n, k != 2i: 2t * f[k-i]
            #   else: 0
            
            # base[k] = conv[k] - (contribution at old_val)
            # = conv[k] - old_val^2 (at k=2i)
            # - 2*old_val*f[k-i] (at valid k != 2i)
            
            k_start = max(0, i)
            k_end = min(2 * n - 2, i + n - 1)
            
            # Precompute the slice indices
            ks = np.arange(k_start, k_end + 1)
            mask = ks != 2 * i
            
            # base at k=2i
            base_2i = conv[2 * i] - old_val * old_val if 2 * i < 2 * n - 1 else 0.0
            # base at other relevant k
            if mask.any():
                base_ks = conv[ks[mask]] - 2.0 * old_val * f[ks[mask] - i]
                f_ks = f[ks[mask] - i]
            else:
                base_ks = np.array([])
                f_ks = np.array([])
            
            # Also need base.max() for k not in our range
            # conv[k] for k not in [k_start, k_end] and k != 2i is unchanged
            # So base.max() over those = conv.max() excluding our range
            # For safety, just use conv.max() (it's an upper bound on base.max())
            # Actually, base[k] = conv[k] for k not in our range, so:
            # base.max() = max(conv.max() excluding [k_start,k_end] and 2i, base_2i, base_ks.max())
            # This is hard to compute efficiently. Let's just use:
            # h(t) = max(base_2i + t^2, (base_ks + 2t*f_ks).max(), other_max)
            # where other_max = max of conv[k] for k not in our range
            # For simplicity, upper bound other_max by conv.max()
            # (This is slightly conservative but OK for our purposes)
            
            other_max = conv.max()  # upper bound
            
            def h(t):
                result = other_max
                v = base_2i + t * t
                if v > result:
                    result = v
                if len(base_ks) > 0:
                    v = (base_ks + 2.0 * t * f_ks).max()
                    if v > result:
                        result = v
                return result
            
            # Find optimal t
            # Convex function, use ternary search
            lo = 0.0
            hi = max(old_val * 2.5, 5.0 / n)
            
            # Ensure hi is above the minimum
            for _ in range(3):
                if h(hi / 2.0) < h(hi):
                    break
                hi *= 2.0
            
            # Ternary search
            for _ in range(12):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                if h(m1) < h(m2):
                    hi = m2
                else:
                    lo = m1
                if hi - lo < 1e-16:
                    break
            
            new_val = (lo + hi) / 2.0
            
            # Update f and conv
            delta = new_val - old_val
            if abs(delta) > 1e-18:
                f[i] = new_val
                if 2 * i < 2 * n - 1:
                    conv[2 * i] += new_val * new_val - old_val * old_val
                if mask.any():
                    conv[ks[mask]] += 2.0 * delta * f_ks
        
        # End of sweep
        new_obj = conv.max()
        if new_obj < best_obj - 1e-15:
            best_obj = new_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 30:
            rng = np.random.default_rng(hash((sweep, n)) % (2**32))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            no_improve = 0
        
        if time.monotonic() - t0 > time_limit:
            break
    
    return best_f, best_obj


def fw_step(f, time_limit=30):
    """One or a few Frank-Wolfe steps with LP oracle."""
    n = len(f)
    f = np.maximum(f.astype(np.float64), 0.0)
    f = f / f.sum()
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        g = solve_lp_direction(f, M, time_limit=min(15, time_limit - (time.monotonic() - t0)))
        if g is None:
            # Try relaxing M
            g = solve_lp_direction(f, M * 1.01, time_limit=min(15, time_limit - (time.monotonic() - t0)))
            if g is None:
                break
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            s = ft.sum()
            if s < 1e-15:
                continue
            ft = ft / s
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-10:
            f = (1 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
        else:
            break
    
    return best_f, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast vectorized coordinate
descent with LP-based Frank-Wolfe escapes and multi-scale exploration.
Key innovation: highly optimized 1D coordinate updates with incremental
convolution tracking, enabling many fast sweeps within the time budget.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.flatnonzero(cond)[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, M, time_limit=15):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0, sum(g)=1."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    A_eq = np.ones((1, n))
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            x = np.maximum(res.x, 0.0)
            s = x.sum()
            if s > 1e-15:
                return x / s
    except Exception:
        pass
    return None


def run_cd(f_init, time_limit, verbose=False):
    """
    Fast coordinate descent minimizing max(auto-convolution) subject to sum(f)=1, f>=0.
    Uses incremental convolution updates and ternary search per coordinate.
    """
    n = len(f_init)
    f = np.maximum(f_init.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    sweep = 0
    full_size = 2 * n - 1

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        prev_best = conv.max()

        for i in range(n):
            old_val = f[i]

            # Determine affected convolution indices
            # k = 2i: contribution is f[i]^2
            # k in [i, i+n-1] intersect [0, 2n-2], k != 2i: contribution is 2*f[i]*f[k-i]
            k_start = max(0, i)
            k_end = min(full_size - 1, i + n - 1)

            # Build affected mask
            affected = np.zeros(full_size, dtype=bool)
            has_2i = (2 * i < full_size)
            if has_2i:
                affected[2 * i] = True
            if k_start <= k_end:
                affected[k_start:k_end + 1] = True

            # other_max: max of conv at unaffected positions
            if affected.all():
                other_max = 0.0
            else:
                other_max = conv[~affected].max()

            # base values at affected positions
            base_2i = conv[2 * i] - old_val * old_val if has_2i else 0.0

            ks = np.arange(k_start, k_end + 1) if k_start <= k_end else np.array([], dtype=int)
            mask = ks != 2 * i if len(ks) > 0 else np.array([], dtype=bool)

            if mask.any():
                base_ks = conv[ks[mask]] - 2.0 * old_val * f[ks[mask] - i]
                f_ks = f[ks[mask] - i]
            else:
                base_ks = np.array([])
                f_ks = np.array([])

            # Define h(t) = max conv value when f[i] = t
            # h(t) = max(other_max, base_2i + t^2 [if has_2i], max(base_ks + 2t*f_ks) [if mask.any()])

            # Quick evaluation for bracketing
            h_0 = other_max
            if has_2i:
                v = base_2i
                if v > h_0:
                    h_0 = v
            if len(base_ks) > 0:
                v = base_ks.max()
                if v > h_0:
                    h_0 = v

            # Find optimal t via ternary search
            lo = 0.0
            hi = max(old_val * 2.0, 4.0 / n)

            # Evaluate h at hi to check if we need to expand
            h_hi = other_max
            if has_2i:
                v = base_2i + hi * hi
                if v > h_hi:
                    h_hi = v
            if len(base_ks) > 0:
                v = (base_ks + 2.0 * hi * f_ks).max()
                if v > h_hi:
                    h_hi = v

            # Expand if still decreasing
            for _ in range(3):
                h_mid = other_max
                mid_check = hi * 0.5
                if has_2i:
                    v = base_2i + mid_check * mid_check
                    if v > h_mid:
                        h_mid = v
                if len(base_ks) > 0:
                    v = (base_ks + 2.0 * mid_check * f_ks).max()
                    if v > h_mid:
                        h_mid = v
                if h_mid >= h_hi:
                    break
                hi = hi * 2.0
                h_hi = h_mid
                h_mid = other_max
                if has_2i:
                    v = base_2i + hi * hi
                    if v > h_mid:
                        h_mid = v
                if len(base_ks) > 0:
                    v = (base_ks + 2.0 * hi * f_ks).max()
                    if v > h_mid:
                        h_mid = v
                h_hi = h_mid

            # Ternary search (convex function)
            n_iter = 14 if sweep < 10 else 10
            for _ in range(n_iter):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0

                # h(m1)
                h1 = other_max
                if has_2i:
                    v = base_2i + m1 * m1
                    if v > h1:
                        h1 = v
                if len(base_ks) > 0:
                    v = (base_ks + 2.0 * m1 * f_ks).max()
                    if v > h1:
                        h1 = v

                # h(m2)
                h2 = other_max
                if has_2i:
                    v = base_2i + m2 * m2
                    if v > h2:
                        h2 = v
                if len(base_ks) > 0:
                    v = (base_ks + 2.0 * m2 * f_ks).max()
                    if v > h2:
                        h2 = v

                if h1 < h2:
                    hi = m2
                else:
                    lo = m1
                if hi - lo < 1e-14:
                    break

            new_val = (lo + hi) / 2.0

            # Update f[i] and conv incrementally
            delta = new_val - old_val
            if abs(delta) > 1e-16:
                f[i] = new_val
                if has_2i:
                    conv[2 * i] += new_val * new_val - old_val * old_val
                if mask.any():
                    conv[ks[mask]] += 2.0 * delta * f_ks

            # Time check every 64 coordinates
            if (i & 63) == 0 and time.monotonic() - t0 > time_limit:
                break

        # End of sweep
        new_obj = conv.max()
        if new_obj < best_obj - 1e-15:
            best_obj = new_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and sweep % 5 == 0:
            print(f"    CD sweep {sweep}: obj={best_obj:.10f}, no_improve={no_improve}")

        if no_improve > 40:
            rng = np.random.default_rng((sweep * 7919 + n) % (2**31))
            noise_scale = 0.002 + 0.001 * (no_improve // 40)
            f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
            conv = fast_convolve(f)
            no_improve = 0

    return best_f, best_obj


def fw_lp_optimize(f_init, time_limit):
    """Frank-Wolfe with LP oracle for global direction finding."""
    n = len(f_init)
    f = np.maximum(f_init.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    m_exp = 1.0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time.limit if False else time_limit - (time.monotonic() - t0)
        if rem < 5:
            break

        g = solve_lp_direction(f, M * m_exp, time_limit=min(20, rem - 2))
        if g is None:
            m_exp *= 1.15
            if m_exp > 6.0:
                m_exp = 1.0
                no_improve += 1
            continue

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 3.0)
            if no_improve > 30:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        alphas = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        for alpha in alphas:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 4.0)

        if no_improve > 50:
            rng = np.random.default_rng(it * 31 + 13)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def sg_descent(f_init, time_limit, initial_lr=0.005, mom=0.93, top_k=64, temp=0.001):
    """Subgradient descent with momentum as a complementary method."""
    n = len(f_init)
    f = np.maximum(f_init.astype(np.float64), 0.0)
    f = f / f.sum()

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    cur_lr = initial_lr

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 1:
            break

        # Compute subgradient using softmax over top peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= wsum

        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idxs = np.arange(js, je + 1)
            sg[idxs] += 2.0 * w[idx] * f[k - idxs]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9998
        temp *= 1.002

        if no_improve > 200:
            rng = np.random.default_rng(it * 17 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            v = np.zeros(n)
            cur_lr = initial_lr
            no_improve = 0

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse structured initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    inits.append(np.ones(n) / n)

    ramp_d = np.arange(n, 0, -1, dtype=np.float64)
    inits.append(ramp_d / ramp_d.sum())

    ramp_i = np.arange(1, n + 1, dtype=np.float64)
    inits.append(ramp_i / ramp_i.sum())

    q = (x - n / 2) ** 2 + 1.0
    inits.append(q / q.sum())

    e1 = np.exp(-x / (n / 3.0))
    inits.append(e1 / e1.sum())

    c = np.cos(np.pi * (x / max(n - 1, 1) - 0.5)) ** 4 + 0.01
    inits.append(c / c.sum())

    sn = np.sin(np.pi * x / n) + 0.05
    inits.append(sn / sn.sum())

    r = rng.dirichlet(np.ones(n) * 0.5)
    inits.append(r)

    if parent is not None:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits

                f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.001)
                a_opt, _ = run_cd(f_pert, time_limit=min(rem, 15))
                try_seq(a_opt, f"micro_{attempt}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
