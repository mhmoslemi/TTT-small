def fast_convolve(a):
    """Linear auto-convolution via FFT."""

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1) FFT-accelerated entropic smoothing for fast gradient computation
2) Nesterov-momentum projected gradient on the simplex
3) Two-point coordinate descent (mass transfer) for polishing
4) Multi-start exploration with diverse initializations and lengths
Key innovation: O(n log n) gradient via FFT cross-correlation, enabling
thousands of iterations within the time budget, combined with adaptive
temperature annealing and smart coordinate selection.

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_grad_fft(a, w):
    """
    Compute grad[i] = sum_k w[k]*(a[k-i]+a[i-k]) for valid indices, using FFT.
    w has length 2n-1 (the softmax weights over convolution values).
    """
    n = len(a)
    conv_len = 2 * n - 1
    N = 1 << (n + conv_len - 1).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    first_term = np.fft.irfft(np.conj(A) * W, N)[:n]
    second_term = np.fft.irfft(A * W, N)[:n]
    return first_term + second_term


def smooth_objective_grad(a, tau):
    """Compute smoothed objective and gradient via FFT."""
    n = len(a)
    conv = fast_convolve(a)
    m = conv.max()
    shifted = (conv - m) / tau
    exp_shifted = np.exp(shifted)
    log_sum = m / tau + np.log(exp_shifted.sum())
    f_val = tau * log_sum
    w = exp_shifted / exp_shifted.sum()
    grad = smooth_grad_fft(a, w)
    return f_val, grad, w


def nesterov_smoothed_optimize(a0, time_limit, tau_init_frac=20.0, tau_end_frac=200.0,
                                lr_init=0.002, momentum=0.85, verbose=False):
    """
    Nesterov-momentum projected gradient on smoothed objective.
    a0: initial point on simplex (sum=1, all >=0)
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    conv = fast_convolve(a)
    max_conv = conv.max()
    tau = max_conv / tau_init_frac
    tau_end = max_conv / tau_end_frac

    v = np.zeros(n)
    a_prev = a.copy()
    t_count = 0
    best_obj = max_conv
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    lr = lr_init

    while time.monotonic() - t0 < time_limit:
        t_count += 1

        # Nesterov lookahead
        beta = (1.0 + np.sqrt(1.0 + 4.0 * (t_count - 1) ** 2)) / 2.0
        beta = min(beta / (t_count + 1), 1.0)  # cap momentum
        y = a + beta * (a - a_prev)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys < 1e-15:
            y = a.copy()
            ys = 1.0
        else:
            y = y / ys

        # Compute gradient at y
        f_val, grad, w = smooth_objective_grad(y, tau)

        # Gradient step
        v = momentum * v - lr * grad
        a_new = project_simplex(a + v)

        # Evaluate true objective (max of convolution)
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            if verbose and t_count % 500 == 0:
                elapsed = time.monotonic() - t0
                print(f"      Nesterov iter {t_count}: obj={best_obj:.10f}, "
                      f"tau={tau:.6f}, elapsed={elapsed:.0f}s")
        else:
            no_improve += 1

        a_prev = a.copy()
        a = a_new

        # Adaptive step size: decrease if stuck
        if no_improve > 200:
            lr *= 0.7
            no_improve = 0
            # Also anneal temperature
            tau = max(tau * 0.85, tau_end)

        # Slow temperature annealing
        progress = min((time.monotonic() - t0) / time_limit, 1.0)
        tau = max(max_conv / (tau_init_frac + (tau_end_frac - tau_init_frac) * progress ** 0.5), tau_end)

        # Restart if completely stuck
        if no_improve > 500:
            rng = np.random.default_rng(t_count * 7 + 13)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            a_prev = a.copy()
            v = np.zeros(n)
            lr = max(lr, 0.0005)
            no_improve = 0

        # Check time
        if time.monotonic() - t0 > time_limit * 0.95:
            break

    # Final: use best_a
    # Re-normalize
    s = best_a.sum()
    if s > 1e-15:
        best_a = best_a / s
    return best_a, best_obj


def coordinate_descent_polish(a0, time_limit, top_k=5, verbose=False):
    """
    Two-point coordinate descent: repeatedly transfer mass from high-gradient
    positions to low-gradient positions using the softmax-weighted gradient.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    conv = fast_convolve(a)
    max_conv = conv.max()
    best_obj = max_conv
    best_a = a.copy()
    tau = max_conv / 30.0
    step_size = max_conv * 0.001

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        m = conv.max()
        shifted = (conv - m) / tau
        exp_shifted = np.exp(shifted)
        w = exp_shifted / exp_shifted.sum()

        # Compute gradient
        grad = smooth_grad_fft(a, w)

        # Find best pair: transfer mass from argmax(grad) to argmin(grad)
        # But avoid transferring from positions with a[j] = 0
        positive_mask = a > 1e-10
        grad_pos = grad.copy()
        grad_pos[~positive_mask] = np.inf  # can't decrease these
        j_star = np.argmin(grad_pos)  # source (highest gradient among positive)
        # Wait, we want to DECREASE positions with high gradient and INCREASE low gradient
        # So source = argmax(grad) among positive positions
        #    sink = argmin(grad) (any position)
        
        grad_source = grad.copy()
        grad_source[~positive_mask] = -np.inf
        source = np.argmax(grad_source)

        sink = np.argmin(grad)

        if source == sink:
            no_improve += 1
            if no_improve > 100:
                break
            continue

        # Try multiple swap sizes
        a_new = a.copy()
        best_local_obj = conv.max()
        best_eps = 0.0

        for frac in [0.1, 0.3, 0.5, 0.7, 1.0]:
            eps = step_size * frac
            if eps >= a[source]:
                eps = a[source] * 0.5
            if eps < 1e-15:
                continue
            a_try = a.copy()
            a_try[source] -= eps
            a_try[sink] += eps
            conv_try = fast_convolve(a_try)
            obj_try = conv_try.max()
            if obj_try < best_local_obj:
                best_local_obj = obj_try
                best_eps = eps

        if best_eps > 1e-15:
            a[source] -= best_eps
            a[sink] += best_eps
            if best_local_obj < best_obj - 1e-14:
                best_obj = best_local_obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # Adaptive: also try top-k sources and sinks
        if it % 10 == 0:
            top_sources = np.argsort(grad)[::-1][:top_k]
            top_sinks = np.argsort(grad)[:top_k]
            for src in top_sources:
                if a[src] < 1e-10:
                    continue
                for snk in top_sinks:
                    if snk == src:
                        continue
                    eps = min(step_size * 0.3, a[src] * 0.3)
                    a_try = a.copy()
                    a_try[src] -= eps
                    a_try[snk] += eps
                    conv_try = fast_convolve(a_try)
                    if conv_try.max() < best_obj:
                        a = a_try
                        best_obj = conv_try.max()
                        best_a = a.copy()
                        no_improve = 0
                        break

        # Animate step size and temperature
        if no_improve > 50:
            step_size *= 0.7
            tau = max(tau * 0.9, max_conv / 200.0)
            no_improve = 0

        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"      CD iter {it}: obj={best_obj:.10f}, elapsed={elapsed:.0f}s")

    s = best_a.sum()
    if s > 1e-15:
        best_a = best_a / s
    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    center = 0.5 * (n - 1)

    # Uniform
    inits.append(np.ones(n) / n)

    # Dirichlet
    inits.append(rng.dirichlet(np.full(n, 0.3)))
    inits.append(rng.dirichlet(np.full(n, 0.8)))
    inits.append(rng.dirichlet(np.full(n, 2.0)))

    # Gaussian (various widths and centers)
    for sigma_frac in [0.1, 0.15, 0.2, 0.3, 0.5]:
        for center_frac in [0.3, 0.5, 0.7]:
            sigma = sigma_frac * n
            c = center_frac * (n - 1)
            g = np.exp(-0.5 * ((x - c) / sigma) ** 2)
            inits.append(g / g.sum())

    # Cosine
    for freq in [1, 2, 3]:
        c = np.abs(np.cos(np.pi * freq * x / n)) + 0.1
        inits.append(c / c.sum())

    # Linearly decreasing (asymmetric)
    for slope in [0.5, 1.0, 2.0, 3.0]:
        lin = np.exp(-slope * x / n)
        inits.append(lin / lin.sum())

    # V-shaped (centered)
    v_shape = np.abs(x - center) / n + 0.1
    inits.append(v_shape / v_shape.sum())

    # Two bumps
    for c1_f, c2_f in [(0.25, 0.75), (0.2, 0.8), (0.3, 0.6)]:
        b1 = np.exp(-0.5 * ((x - c1_f * (n - 1)) / (0.1 * n)) ** 2)
        b2 = np.exp(-0.5 * ((x - c2_f * (n - 1)) / (0.1 * n)) ** 2)
        two = b1 + b2
        inits.append(two / two.sum())

    # Power law
    for p in [0.5, 1.0, 2.0, 4.0]:
        pw = (x + 1) ** (-p)
        inits.append(pw / pw.sum())

    # Sinc-like
    for width in [5, 10, 20]:
        sinc = np.sinc((x - center) / width)
        sinc = np.maximum(sinc, 0.01)
        inits.append(sinc / sinc.sum())

    # Parent interpolation
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
            # Perturbed parent
            for scale in [0.01, 0.03, 0.05]:
                pp = px / ps + rng.standard_normal(n) * scale
                pp = np.maximum(pp, 0)
                pp = pp / pp.sum()
                inits.append(pp)

    return inits


def evaluate_quick(a):

    return 2.0 * len(a) * conv.max()

    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        a_norm = a / s  # normalize for score computation
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            elapsed = time.monotonic()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={parent_score:.10f}")
    except Exception:
        pass

    # === Phase 1: Initialization and screening ===
    print("=== Phase 1: Initialization ===")
    t_phase1_end = time.monotonic() + 40.0

    lengths_to_try = [256, 320, 384, 448, 512, 640]
    if parent is not None:
        n_parent = len(parent)
        if n_parent not in lengths_to_try:
            lengths_to_try.insert(0, n_parent)

    best_init = None
    best_init_obj = float('inf')
    best_init_n = 384

    for n in lengths_to_try:
        if time.monotonic() > t_phase1_end:
            break
        inits = generate_initializations(n, parent, rng)
        for i, ini in enumerate(inits):
            if time.monotonic() > t_phase1_end:
                break
            obj = evaluate_quick(ini)
            if obj < best_init_obj:
                best_init_obj = obj
                best_init = ini.copy()
                best_init_n = n

    print(f"  Best init: n={best_init_n}, smooth_obj~{best_init_obj:.6f}")

    # Evaluate best init formally
    try_seq(best_init, "init_best")

    # Also evaluate parent if available
    if parent is not None:
        try_seq(parent, "parent")
        if best_seq is not None and len(best_seq) != best_init_n:
            # Keep the better one
            pass

    # Work with the best so far
    work_a = best_seq.astype(np.float64).copy() if best_seq is not None else best_init.copy()
    work_n = len(work_a)
    work_a = work_a / work_a.sum()

    # === Phase 2: Smoothed Nesterov (main optimization) ===
    print(f"\n=== Phase 2: Smoothed Nesterov (n={work_n}) ===")
    t_phase2_end = time.monotonic() + min(600, (deadline - time.monotonic()) * 0.45)

    # Run Nesterov on the best
    a_opt, obj_opt = nesterov_smoothed_optimize(
        work_a, time_limit=min(400, t_phase2_end - time.monotonic()),
        tau_init_frac=15.0, tau_end_frac=150.0,
        lr_init=0.003, momentum=0.82, verbose=True
    )
    try_seq(a_opt, "nesterov1")

    # Run again from the result (continue optimization)
    rem = t_phase2_end - time.monotonic()
    if rem > 30:
        best_a = best_seq.astype(np.float64).copy() / best_seq.sum()
        a_opt, obj_opt = nesterov_smoothed_optimize(
            best_a, time_limit=min(200, rem),
            tau_init_frac=30.0, tau_end_frac=300.0,
            lr_init=0.001, momentum=0.88, verbose=True
        )
        try_seq(a_opt, "nesterov2")

    # Try a different length
    for n_alt in [work_n + 64, max(128, work_n - 64)]:
        rem = t_phase2_end - time.monotonic()
        if rem < 60:
            break
        # Interpolate to new length
        x_old = np.linspace(0, 1, len(best_seq))
        x_new = np.linspace(0, 1, n_alt)
        a_alt = np.interp(x_new, x_old, best_seq)
        a_alt = np.maximum(a_alt, 0)
        a_alt = a_alt / a_alt.sum()
        a_opt_n, _ = nesterov_smoothed_optimize(
            a_alt, time_limit=min(100, rem - 10),
            tau_init_frac=20.0, tau_end_frac=200.0,
            lr_init=0.002, momentum=0.85
        )
        try_seq(a_opt_n, f"nesterov_n{n_alt}")
        rem = t_phase2_end - time.monotonic()

    # === Phase 3: Coordinate Descent Polishing ===
    print(f"\n=== Phase 3: Coordinate Descent (n={len(best_seq)}) ===")
    t_phase3_end = time.monotonic() + min(350, (deadline - time.monotonic()) * 0.25)

    best_a = best_seq.astype(np.float64).copy() / best_seq.sum()
    a_opt, obj_opt = coordinate_descent_polish(
        best_a, time_limit=min(250, t_phase3_end - time.monotonic()),
        top_k=5, verbose=True
    )
    try_seq(a_opt, "cd1")

    # Second CD pass with different params
    rem = t_phase3_end - time.monotonic()
    if rem > 30:
        best_a = best_seq.astype(np.float64).copy() / best_seq.sum()
        a_opt, obj_opt = coordinate_descent_polish(
            best_a, time_limit=min(120, rem),
            top_k=10, verbose=False
        )
        try_seq(a_opt, "cd2")

    # === Phase 4: Multi-restart exploration ===
    print(f"\n=== Phase 4: Multi-restart ===")
    t_phase4_end = time.monotonic() + min(300, (deadline - time.monotonic()) * 0.25)

    best_a = best_seq.astype(np.float64).copy() / best_seq.sum()
    n_best = len(best_a)

    for restart in range(8):
        rem = t_phase4_end - time.monotonic()
        if rem < 30:
            break
        # Perturb and optimize
        noise_scale = 0.005 + restart * 0.003
        a_pert = project_simplex(best_a + rng.standard_normal(n_best) * noise_scale)
        t_nest = min(60, rem - 10)
        a_opt, _ = nesterov_smoothed_optimize(
            a_pert, time_limit=t_nest,
            tau_init_frac=15.0 + restart * 5,
            tau_end_frac=150.0 + restart * 30,
            lr_init=0.002, momentum=0.85
        )
        try_seq(a_opt, f"restart{restart}")

        # Quick CD polish on the result
        rem = t_phase4_end - time.monotonic()
        if rem > 20:
            a_cd, _ = coordinate_descent_polish(
                a_opt / a_opt.sum(), time_limit=min(30, rem - 5), top_k=3
            )
            try_seq(a_cd, f"restart{restart}_cd")

    # === Phase 5: Final refinement ===
    print(f"\n=== Phase 5: Final Refinement ===")
    rem_total = deadline - time.monotonic()

    if best_seq is not None and rem_total > 30:
        best_a = best_seq.astype(np.float64).copy() / best_seq.sum()

        # Final Nesterov with very low learning rate
        t_final = min(100, rem_total * 0.4)
        a_opt, _ = nesterov_smoothed_optimize(
            best_a, time_limit=t_final,
            tau_init_frac=50.0, tau_end_frac=500.0,
            lr_init=0.0005, momentum=0.9
        )
        try_seq(a_opt, "final_nest")

        # Final CD
        rem = deadline - time.monotonic()
        if rem > 20:
            best_a = best_seq.astype(np.float64).copy() / best_seq.sum()
            a_opt, _ = coordinate_descent_polish(
                best_a, time_limit=min(60, rem - 10), top_k=5
            )
            try_seq(a_opt, "final_cd")

        # Try a few different lengths around the best
        rem = deadline - time.monotonic()
        if rem > 20:
            n_cur = len(best_seq)
            for n_try in [n_cur + 32, n_cur - 32]:
                if n_try < 64 or rem < 15:
                    continue
                x_old = np.linspace(0, 1, n_cur)
                x_new = np.linspace(0, 1, n_try)
                a_try = np.interp(x_new, x_old, best_seq)
                a_try = np.maximum(a_try, 0) / np.maximum(a_try.sum(), 1e-15)
                a_opt_l, _ = nesterov_smoothed_optimize(
                    a_try, time_limit=min(30, rem - 5),
                    tau_init_frac=20.0, tau_end_frac=200.0,
                    lr_init=0.001, momentum=0.85
                )
                try_seq(a_opt_l, f"len{n_try}")
                rem = deadline - time.monotonic()

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.10f}")

    return [float(x) for x in best_seq]
