"""
Peak-flow optimization with exact line search for minimizing
2*n*max(auto-conv)/(sum)^2. Uses the reflection structure of
auto-convolution to compute exact 1D quadratic line searches,
combined with multi-peak weighting, coordinate descent fallback,
and simulated annealing restarts.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def peak_flow_step(a, peak_indices, peak_weights, n_steps=20):
    """
    Perform one peak-flow step with line search.
    a: normalized sequence (sum=1)
    peak_indices: array of peak positions
    peak_weights: weights for each peak
    Returns updated a.
    """
    n = len(a)
    
    # Build direction: d[j] = sum_i w_i * (mean(r_i) - r_i[j])
    d = np.zeros(n)
    for i, k in enumerate(peak_indices):
        w = peak_weights[i]
        if w < 1e-15:
            continue
        # r[j] = a[k-j] with a[x]=0 for x outside [0,n-1]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        r = np.zeros(n)
        # r[j] = a[k-j] for j in [js, je]
        r[js:je+1] = a[k - np.arange(js, je+1)]
        d += w * (r.mean() - r)
    
    # Ensure d is in tangent space (sum = 0)
    d -= d.mean()
    
    # Check if direction is meaningful
    if np.max(np.abs(d)) < 1e-15:
        return a.copy()
    
    # Line search: minimize max conv along a + t*d
    # Use golden section or parabolic fit
    best_t = 0.0
    best_obj = fast_convolve(a).max()
    
    # First, find a good range
    # Try a few candidate step sizes
    max_d = np.max(np.abs(d))
    if max_d < 1e-15:
        return a.copy()
    
    # Scale: t should be O(1) relative to the scale of a
    scale = 1.0 / max_d
    
    candidates = [0.01/scale, 0.03/scale, 0.05/scale, 0.1/scale, 
                  0.2/scale, 0.3/scale, 0.5/scale, 0.7/scale, 1.0/scale]
    
    best_t = 0.0
    best_val = best_obj
    
    for t in candidates:
        a_new = a + t * d
        if np.any(a_new < -1e-10):
            continue
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-10:
            continue
        a_new = a_new / a_new.sum()
        val = fast_convolve(a_new).max()
        if val < best_val:
            best_val = val
            best_t = t
    
    # Refine with golden section
    if best_t > 0:
        lo = max(0, best_t * 0.5)
        hi = best_t * 1.5
        for _ in range(15):
            m1 = lo + (hi - lo) * 0.382
            m2 = lo + (hi - lo) * 0.618
            a1 = np.maximum(a + m1 * d, 0.0)
            a2 = np.maximum(a + m2 * d, 0.0)
            if a1.sum() > 1e-10:
                a1 = a1 / a1.sum()
                v1 = fast_convolve(a1).max()
            else:
                v1 = np.inf
            if a2.sum() > 1e-10:
                a2 = a2 / a2.sum()
                v2 = fast_convolve(a2).max()
            else:
                v2 = np.inf
            if v1 < v2:
                hi = m2
            else:
                lo = m1
        best_t = (lo + hi) / 2
        a_new = np.maximum(a + best_t * d, 0.0)
        if a_new.sum() > 1e-10:
            a_new = a_new / a_new.sum()
            best_val = fast_convolve(a_new).max()
    
    if best_t > 0:
        a_new = np.maximum(a + best_t * d, 0.0)
        a_new = a_new / a_new.sum()
        return a_new
    return a.copy()


def coordinate_descent(a, time_limit, n_trials_per_coord=5):
    """
    Coordinate descent: for each coordinate, try small perturbations
    and keep the best one.
    """
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        # Random subset of coordinates for speed
        n_coords = min(n, max(10, n // 4))
        coords = rng.choice(n, size=n_coords, replace=False)
        
        for j in coords:
            if time.monotonic() - t0 > time_limit:
                break
            a_j = a[j]
            if a_j < 1e-10:
                continue
            
            # Try different perturbation sizes
            for frac in [0.01, 0.02, 0.05, 0.1, 0.2, 0.05, 0.1]:
                # Try increasing a[j] (and proportionally decreasing others)
                delta = a_j * frac
                a_new = a.copy()
                a_new[j] += delta
                # Distribute the increase proportionally to others
                others = np.ones(n)
                others[j] = 0
                others = others * a
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_new -= delta * others / others_sum
                a_new = np.maximum(a_new, 0)
                if a_new.sum() < 1e-15:
                    continue
                a_new = a_new / a_new.sum()
                obj = fast_convolve(a_new).max()
                if obj < best_obj - 1e-14:
                    a = a_new
                    best_obj = obj
                    improved = True
                    break
                # Also try decreasing
                delta = -a_j * frac
                a_new = a.copy()
                a_new[j] += delta
                if a_new[j] < 0:

                others = np.ones(n)
                others[j] = 0
                others = others * a
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_new += (-delta) * others / others_sum
                a_new = np.maximum(a_new, 0)
                if a_new.sum() < 1e-15:
                    continue
                a_new = a_new / a_new.sum()
                obj = fast_convolve(a_new).max()
                if obj < best_obj - 1e-14:
                    a = a_new
                    best_obj = obj
                    improved = True
                    break

"""
Peak-flow optimization with exact line search for minimizing
2*n*max(auto-conv)/(sum)^2. Uses the reflection structure of
auto-convolution to compute efficient line searches along
directions that reduce peak values, combined with adaptive
peak selection, coordinate descent, and simulated annealing.
"""

import numpy as np
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_flow_direction(a, peaks, weights):
    """Compute the flow direction to reduce the specified peaks."""
    n = len(a)
    d = np.zeros(n)
    for i, k in enumerate(peaks):
        w = weights[i]
        if w < 1e-15:
            continue
        # r[j] = a[k-j] for valid j
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        r = np.zeros(n)
        if je >= js:
            r[js:je+1] = a[k - np.arange(js, je+1)]
        r_mean = r.mean()
        d += w * (r_mean - r)
    d -= d.mean()
    return d


def line_search_peak_flow(a, d, max_fevals=30):
    """Line search along direction d from point a, minimizing max conv."""
    n = len(a)
    if np.max(np.abs(d)) < 1e-15:
        return a, fast_convolve(a).max()
    
    base_conv = fast_convolve(a)
    base_val = base_conv.max()
    
    # Find initial feasible range
    # a + t*d >= 0 => t >= -a[j]/d[j] for d[j] < 0
    neg_mask = d < -1e-15
    if neg_mask.any():
        t_max = np.min(-a[neg_mask] / d[neg_mask]) * 0.99
    else:
        t_max = 10.0
    t_max = min(t_max, 5.0)
    
    if t_max < 1e-10:
        return a, base_val
    
    # Golden section search in [0, t_max]
    best_t = 0.0
    best_val = base_val
    fevals = 1  # we already computed base
    
    # Try a few spread-out points first
    t_try = np.linspace(0.01 * t_max, t_max, 8)
    
    for t in t_try:
        if fevals >= max_fevals:
            break
        a_new = a + t * d
        if np.any(a_new < 0):
            continue
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-15:
            continue
        a_new = a_new / a_new_sum
        val = fast_convolve(a_new).max()
        fevals += 1
        if val < best_val:
            best_val = val
            best_t = t
    
    # Refine with golden section around best_t
    if best_t > 0 and fevals < max_fevals:
        lo = max(0, best_t * 0.5)
        hi = min(t_max, best_t * 1.5)
        for _ in range(12):
            if fevals >= max_fevals - 2:
                break
            m1 = lo + (hi - lo) * 0.382
            m2 = lo + (hi - lo) * 0.618
            a1 = a + m1 * d
            a2 = a + m2 * d
            if np.any(a1 < 0):
                v1 = np.inf
            else:
                s1 = a1.sum()
                if s1 < 1e-15:
                    v1 = np.inf
                else:
                    v1 = fast_convolve(a1 / s1).max()
                    fevals += 1
            if np.any(a2 < 0):
                v2 = np.inf
            else:
                s2 = a2.sum()
                if s2 < 1e-15:
                    v2 = np.inf
                else:
                    v2 = fast_convolve(a2 / s2).max()
                    fevals += 1
            if v1 < v2:
                hi = m2
            else:
                lo = m1
        mid = (lo + hi) / 2
        a_mid = a + mid * d
        if not np.any(a_mid < 0):
            s_mid = a_mid.sum()
            if s_mid > 1e-15:
                a_mid = a_mid / s_mid
                v_mid = fast_convolve(a_mid).max()
                if v_mid < best_val:
                    best_val = v_mid
                    best_t = mid
    
    if best_t > 0:
        a_new = a + best_t * d
        if not np.any(a_new < 0):
            return a_new / a_new.sum(), best_val
    return a, base_val

    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()

        # Strategy 1: Always include the max peak
        # Strategy 2: Include top-k with softmax weighting
        # Strategy 3: Occasionally target a secondary peak
        
        if no_improve < 50:

            k_max = n - 1
            top_k = max(3, min(20, n // 16))

            # Sharp softmax (low temperature)

            scaled = pv / temp
            scaled -= scaled.max()
            w = np.exp(scaled)
            w_sum = w.sum()
            if w_sum > 0:

                w[np.argmax(pv)] = 1.0

def coord_descent_step(a, n_trials=3, rng=None):
    """One pass of coordinate descent trying random coordinates."""
    n = len(a)
    if rng is None:
        rng = np.random.default_rng(999)
    
    best_obj = fast_convolve(a).max()
    
    # Try a batch of random coordinate changes
    batch_size = min(n, 50)
    coords = rng.choice(n, size=batch_size, replace=False)
    
    for j in coords:
        a_j = a[j]
        if a_j < 1e-12:
            continue
        
        for frac in [0.05, 0.1, 0.2]:
            # Move mass from j to a random other coordinate
            j2 = rng.integers(0, n)
            if j2 == j:
                j2 = (j + 1) % n
            
            delta = a_j * frac
            a_new = a.copy()
            a_new[j] -= delta
            a_new[j2] += delta
            
            if a_new[j] < 0:
                continue
            a_new = a_new / a_new.sum()
            obj = fast_convolve(a_new).max()
            if obj < best_obj - 1e-14:
                a = a_new.copy()
                best_obj = obj
                break
    
    return a, best_obj

"""
Peak-flow optimization with exact line search for minimizing
2*n*max(auto-conv)/(sum)^2. Exploits the reflection structure of
auto-convolution: for peak at k, conv[k] = <a, reflect_k(a)>.
Moves mass from high-contribution to low-contribution positions via
weighted flow directions, with adaptive peak selection, line search,
coordinate descent, and simulated annealing for escape.
"""

import numpy as np
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_flow_direction(a, peaks, weights):
    """
    Compute flow direction to reduce specified peaks.
    For peak at k: conv[k] = sum_j a[j]*a[k-j].
    Direction: d[j] = sum_i w_i * (mean(r_i) - r_i[j]) where r_i[j] = a[k_i - j].
    This moves mass from positions contributing heavily to peaks to those contributing little.
    """
    n = len(a)
    d = np.zeros(n)
    for i in range(len(peaks)):
        w = weights[i]
        if w < 1e-15:
            continue
        k = peaks[i]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        r = np.zeros(n)
        if je >= js:
            idx = np.arange(js, je + 1)
            r[idx] = a[k - idx]
        d += w * (r.mean() - r)
    d -= d.mean()
    return d


def line_search_flow(a, d, max_fevals=30):
    """
    Golden-section line search along direction d from point a.
    Minimizes max(conv(a + t*d)) over t >= 0 with a + t*d >= 0.
    """
    n = len(a)
    if np.max(np.abs(d)) < 1e-15:
        return a.copy(), fast_convolve(a).max()

    base_val = fast_convolve(a).max()

    # Find maximum feasible t
    neg_mask = d < -1e-15
    if neg_mask.any():
        t_max = np.min(-a[neg_mask] / d[neg_mask]) * 0.98
    else:
        t_max = 5.0
    t_max = min(t_max, 10.0)

    if t_max < 1e-12:
        return a.copy(), base_val

    best_t = 0.0
    best_val = base_val
    fevals = 0

    # Spread out initial evaluations
    n_try = min(8, max_fevals)
    t_candidates = np.geomspace(0.01, t_max, n_try)
    for t in t_candidates:
        if fevals >= max_fevals - 10:
            break
        a_new = a + t * d
        if np.any(a_new < -1e-14):
            continue
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            continue
        a_new = a_new / s
        val = fast_convolve(a_new).max()
        fevals += 1
        if val < best_val:
            best_val = val
            best_t = t

    # Golden section refinement
    if best_t > 0 and fevals < max_fevals - 2:
        lo = max(0.0, best_t * 0.6)
        hi = min(t_max, best_t * 1.4)
        for _ in range(12):
            if fevals >= max_fevals - 2:
                break
            if hi - lo < 1e-12 * t_max:
                break
            m1 = lo + (hi - lo) * 0.382
            m2 = lo + (hi - lo) * 0.618
            a1 = a + m1 * d
            a2 = a + m2 * d
            if np.any(a1 < -1e-14):
                v1 = np.inf
            else:
                s1 = a1.sum()
                v1 = fast_convolve(np.maximum(a1, 0) / s1).max() if s1 > 1e-15 else np.inf
                fevals += 1
            if np.any(a2 < -1e-14):
                v2 = np.inf
            else:
                s2 = a2.sum()
                v2 = fast_convolve(np.maximum(a2, 0) / s2).max() if s2 > 1e-15 else np.inf
                fevals += 1
            if v1 < v2:
                hi = m2
            else:
                lo = m1
        mid = (lo + hi) / 2.0
        a_mid = a + mid * d
        if not np.any(a_mid < -1e-14):
            s_mid = a_mid.sum()
            if s_mid > 1e-15:
                a_mid = np.maximum(a_mid, 0) / s_mid
                v_mid = fast_convolve(a_mid).max()
                fevals += 1
                if v_mid < best_val:
                    best_val = v_mid
                    best_t = mid

    if best_t > 0:
        a_new = np.maximum(a + best_t * d, 0.0)
        s = a_new.sum()
        if s > 1e-15:
            return a_new / s, best_val
    return a.copy(), base_val


def peak_flow_optimize(a0, time_limit, verbose=False):
    """
    Main peak-flow optimization loop with adaptive peak selection
    and escape mechanisms.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_obj = fast_convolve(a).max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    rng = np.random.default_rng(314)
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)

        # Adaptive peak selection based on stagnation
        if no_improve < 30:
            # Sharp focus on top peaks
            top_k = max(3, min(15, n // 20))
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            pv = conv[peaks]
            temp = 0.005 * pv.max() + 1e-10
            scaled = pv / temp - (pv.max() / temp)
            w = np.exp(scaled)
            ws = w.sum()
            w = w / ws if ws > 0 else np.zeros(len(peaks))
            if w.sum() < 1e-15:
                w = np.zeros(len(peaks))
                w[np.argmax(pv)] = 1.0
        elif no_improve < 100:
            # Focus on single max peak
            peaks = np.array([int(np.argmax(conv))])
            w = np.array([1.0])
        elif no_improve < 200:
            # Max + second peak
            conv_c = conv.copy()
            k1 = int(np.argmax(conv_c))
            conv_c[k1] = -1
            k2 = int(np.argmax(conv_c))
            peaks = np.array([k1, k2])
            w = np.array([0.6, 0.4])
        else:
            # Random secondary peak exploration
            conv_c = conv.copy()
            k1 = int(np.argmax(conv_c))
            conv_c[k1] = -1
            # Pick a random high peak
            top_indices = np.argpartition(conv_c, -5)[-5:]
            k2 = top_indices[rng.integers(0, len(top_indices))]
            peaks = np.array([k1, k2])
            w = np.array([0.5, 0.5])

        d = compute_flow_direction(a, peaks, w)

        if np.max(np.abs(d)) < 1e-13:
            no_improve += 5
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            continue

        a_new, new_obj = line_search_flow(a, d, max_fevals=25)

        if new_obj < best_obj - 1e-14:
            a = a_new
            best_obj = new_obj
            best_a = a.copy()
            no_improve = 0
        elif new_obj <= best_obj + 1e-12:
            a = a_new
            no_improve += 1
        else:
            no_improve += 1

        # Escape mechanisms
        if no_improve > 250:
            noise_scale = 0.003 + 0.008 * min(1.0, (no_improve - 250) / 300.0)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            no_improve = 0
        elif no_improve > 100 and it % 50 == 0:
            # Smaller perturbation
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    iter {it}, obj={best_obj:.10f}, score~{2*n*best_obj:.6f}, elapsed={elapsed:.1f}s")

    return best_a, best_obj


def coord_descent_optimize(a0, time_limit):
    """
    Coordinate descent: try pairwise mass transfers between random
    coordinate pairs with various fractions.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    rng = np.random.default_rng(777)

    best_obj = fast_convolve(a).max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        improved = False

        # Try a batch of random pairwise transfers
        batch = min(30, n // 3)
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit:
                break
            j1 = rng.integers(0, n)
            j2 = rng.integers(0, n)
            if j1 == j2:
                continue
            a_j1 = a[j1]
            if a_j1 < 1e-12:
                continue

            for frac in [0.05, 0.1, 0.2, 0.3]:
                delta = a_j1 * frac
                a_new = a.copy()
                a_new[j1] -= delta
                a_new[j2] += delta
                if a_new[j1] < 0:
                    break
                a_new = a_new / a_new.sum()
                obj = fast_convolve(a_new).max()
                if obj < best_obj - 1e-14:
                    a = a_new.copy()
                    best_obj = obj
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            no_improve += 1
            if no_improve > 15:
                noise = 0.004 + 0.003 * min(1.0, no_improve / 50.0)
                a = project_simplex(best_a + rng.standard_normal(n) * noise)
                no_improve = 0
                if fast_convolve(a).max() < best_obj:
                    best_obj = fast_convolve(a).max()
                    best_a = a.copy()

    return best_a, best_obj


def multi_peak_equalize(a0, peaks, time_limit):
    """
    Peak-flow with fixed equal-weight peaks (for targeted equalization).
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    t0 = time.monotonic()
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    no_improve = 0
    rng = np.random.default_rng(555)

    w = np.ones(len(peaks)) / len(peaks)

    while time.monotonic() - t0 < time_limit:
        d = compute_flow_direction(a, peaks, w)
        if np.max(np.abs(d)) < 1e-13:
            no_improve += 3
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            continue

        a_new, new_obj = line_search_flow(a, d, max_fevals=20)
        if new_obj < best_obj - 1e-14:
            a = a_new
            best_obj = new_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 40:
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            no_improve = 0

    return best_a, best_obj


def spectral_init(n, parent):
    """
    Generate a spectral-motivated initialization:
    Try to make the auto-convolution flat by designing a sequence
    whose DFT has specific properties.
    """
    # Start from parent interpolated
    if parent is not None and len(parent) != n:
        x_new = np.linspace(0, 1, n)
        x_old = np.linspace(0, 1, len(parent))
        a = np.interp(x_new, x_old, parent)
    elif parent is not None:
        a = parent.copy()
    else:
        a = np.ones(n) / n

    # Apply a few iterations of "anti-peak" smoothing
    a = np.maximum(a, 0)
    a = a / a.sum()

    # Add subtle structured perturbation
    x = np.arange(n, dtype=np.float64)
    for freq in [2, 3, 5, 7]:
        a += 0.001 * np.cos(2 * np.pi * freq * x / n)

    a = np.maximum(a, 0)
    a = a / a.sum()
    return a
