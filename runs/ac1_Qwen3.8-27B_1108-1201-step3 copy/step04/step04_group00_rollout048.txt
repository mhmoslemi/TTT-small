import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with pairs of coordinates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-12:
                continue

            cur_conv = fast_convolve(f)
            cur_peak = cur_conv.max()
            best_j_ratio = 1.0
            best_j_obj = cur_peak

            for ratio in [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 0.3, 2.0]:
                new_val = f[j] * ratio
                if new_val < 0 or new_val > 0.99:
                    continue
                # Renormalize others
                others_sum = 1.0 - f[j]
                if others_sum < 1e-10:
                    continue
                scale = (1.0 - new_val) / others_sum
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_ratio = ratio

            if best_j_ratio != 1.0:
                others_sum = 1.0 - f[j]
                if others_sum > 1e-10:
                    new_val = f[j] * best_j_ratio
                    scale = (1.0 - new_val) / others_sum
                    f[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved and passes > 2:
            break

    return best_f, best_obj


def sg_polish(f0, time_limit, lr=0.003, mom=0.93, top_k=30, temp=0.00005):
    """Subgradient polishing with adaptive temperature."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp

    while time.monotonic() - t0 < time_limit:
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_temp *= 1.005

            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            cur_lr = max(lr * 0.4, 0.0003)

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def cosine_init(n, width=1.0):
    """Cosine-tapered initialization."""
    x = np.arange(n)
    t = (x - n / 2) / (n * width / 2)
    t = np.clip(t, -1, 1)
    a = np.cos(t * np.pi / 2) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def gaussian_init(n, sigma_frac=0.25):
    """Gaussian initialization."""
    x = np.arange(n)
    center = (n - 1) / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def supergaussian_init(n, p=3.0):
    """Super-Gaussian (flat-top) initialization."""
    x = np.arange(n)
    center = (n - 1) / 2.0
    width = n * 0.3
    t = (x - center) / width
    t = np.clip(t, -1.2, 1.2)
    a = np.exp(-np.abs(t) ** p)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a

"""

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def sg_polish(f0, time_limit, lr=0.003, mom=0.93, top_k=30, temp=0.00005):
    """Subgradient polishing with softmax-weighted peaks and adaptive params."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp

    while time.monotonic() - t0 < time_limit:
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()

            no_improve += 1

        cur_temp *= 1.006

            v = np.zeros(n)
            cur_lr = max(lr * 0.4, 0.0003)

def pair_swap_cd(f0, time_limit, verbose=False):
    """Coordinate descent via pair swaps of mass between coordinates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            if f[j] < 1e-13:
                continue

            # Try scaling just this coordinate
            cur_peak = fast_convolve(f).max()
            best_ratio = 1.0
            best_obj_j = cur_peak

            for ratio in [0.5, 0.7, 0.85, 1.15, 1.3, 1.6, 0.3, 2.0, 1.05, 0.95]:
                new_val = f[j] * ratio
                if new_val < 0 or new_val > 0.999:
                    continue
                others = 1.0 - f[j]
                if others < 1e-10:
                    continue
                scale = (1.0 - new_val) / others
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()
                val = fast_convolve(f_trial).max()
                if val < best_obj_j - 1e-15:
                    best_obj_j = val
                    best_ratio = ratio

            if best_ratio != 1.0:
                others = 1.0 - f[j]
                if others > 1e-10:
                    new_val = f[j] * best_ratio
                    scale = (1.0 - new_val) / others
                    f[j] = new_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                    if best_obj_j < best_obj - 1e-14:
                        best_obj = best_obj_j
                        best_f = f.copy()
                        improved = True

        if not improved and passes > 2:
            break
        if verbose and passes % 5 == 0:
            print(f"    CD pass {passes}: obj={best_obj:.10f}")

    return best_f, best_obj


def cosine_init(n, width=1.0):
    """Raised-cosine window initialization."""
    x = np.arange(n)
    t = (x - n / 2.0) / (n * width / 2.0)
    t = np.clip(t, -1.0, 1.0)
    a = np.cos(t * np.pi / 2.0) ** 2
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def gaussian_init(n, sigma_frac=0.25):
    """Gaussian initialization."""
    x = np.arange(n)
    center = (n - 1) / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def supergaussian_init(n, p=4.0, width_frac=0.35):
    """Super-Gaussian (near flat-top) initialization."""
    x = np.arange(n)
    center = (n - 1) / 2.0
    width = n * width_frac
    t = (x - center) / width
    t = np.clip(t, -1.5, 1.5)
    a = np.exp(-np.abs(t) ** p)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

    init_candidates['gauss0.2'] = gaussian_init(n_parent, 0.2)
    init_candidates['gauss0.25'] = gaussian_init(n_parent, 0.25)
    init_candidates['gauss0.3'] = gaussian_init(n_parent, 0.3)
    init_candidates['sg4'] = supergaussian_init(n_parent, 4.0, 0.35)
    init_candidates['sg6'] = supergaussian_init(n_parent, 6.0, 0.30)
    init_candidates['sg8'] = supergaussian_init(n_parent, 8.0, 0.25)

    best_init = parent.copy()
    best_init_mconv = float('inf')
    for name, init in init_candidates.items():
        mc = fast_convolve(init).max()
        if mc < best_init_mconv:
            best_init_mconv = mc
            best_init = init.copy()
            print(f"  Best init: {name}, max_conv={mc:.12f}")

    # === Phase 2: LP-FW on best init at parent length ===
    print(f"\n=== Phase 2: LP-FW n={n_parent} ===")
    rem = deadline - time.monotonic() - 600
    if rem > 60:
        t1 = min(rem, 350)
        print(f"  Time alloc: {t1:.0f}s")
        a_opt, obj = lp_fw_optimize(best_init, time_limit=t1, verbose=True)
        sc, better = evaluate_and_track(a_opt, [best_score], "lpfw_parent")
        if better:
            best_score = sc
            best_seq = a_opt.copy()

    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-length (best={best_score:.8f}) ===")
    base = best_seq if best_seq is not None else best_init
    target_lengths = []
    for tl in [1024, 896, 512, 640]:
        if tl != n_parent:
            target_lengths.append(tl)

    for idx, target_n in enumerate(target_lengths):
        rem = deadline - time.monotonic() - (len(target_lengths) - idx) * 40 - 250
        if rem < 50:
            break
        t_alloc = min(rem, 180)
        f_interp = interpolate_sequence(base, target_n)
        print(f"  LP-FW n={target_n}, t={t_alloc:.0f}s")
        a_opt, obj = lp_fw_optimize(f_interp, time_limit=t_alloc, verbose=(idx == 0))
        sc, better = evaluate_and_track(a_opt, [best_score], f"lpfw_n{target_n}")
        if better:
            best_score = sc
            best_seq = a_opt.copy()
            base = best_seq

    # === Phase 4: Deep LP-FW with restarts ===
    print(f"\n=== Phase 4: Deep LP-FW (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        restart_strategies = [
            ("direct", best_seq.copy()),
            ("small_noise", project_simplex(best_seq + rng.standard_normal(n_best) * 0.002)),
            ("med_noise", project_simplex(best_seq + rng.standard_normal(n_best) * 0.005)),
            ("mirror", project_simplex(0.5 * best_seq + 0.5 * best_seq[::-1])),
            ("cos_blend", project_simplex(0.85 * best_seq + 0.15 * cosine_init(n_best, 0.9))),
            ("gauss_blend", project_simplex(0.8 * best_seq + 0.2 * gaussian_init(n_best, 0.25))),
        ]

        for rname, f_init in restart_strategies:
            rem = deadline - time.monotonic() - 120
            if rem < 25:
                break
            t_alloc = min(rem, 100)
            a_opt, obj = lp_fw_optimize(f_init, time_limit=t_alloc, verbose=False)
            sc, better = evaluate_and_track(a_opt, [best_score], f"deep_{rname}")
            if better:
                best_score = sc
                best_seq = a_opt.copy()

    # === Phase 5: Coordinate Descent ===
    print(f"\n=== Phase 5: Coordinate Descent (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 30:
            t_cd = min(rem * 0.7, 150)
            print(f"  CD time: {t_cd:.0f}s")
            a_opt, obj = pair_swap_cd(best_seq, time_limit=t_cd, verbose=True)
            sc, better = evaluate_and_track(a_opt, [best_score], "cd")
            if better:
                best_score = sc
                best_seq = a_opt.copy()

    # === Phase 6: Subgradient Polishing ===
    print(f"\n=== Phase 6: SG Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Coarse SG
        rem = deadline - time.monotonic() - 80
        if rem > 20:
            t_sg = min(rem, 100)
            a_opt, obj = sg_polish(best_seq, time_limit=t_sg,
                                   lr=0.002, mom=0.93, top_k=max(20, n_best // 6),
                                   temp=0.00003)
            sc, better = evaluate_and_track(a_opt, [best_score], "sg_coarse")
            if better:
                best_score = sc
                best_seq = a_opt.copy()

        # Fine SG
        rem = deadline - time.monotonic() - 40
        if rem > 15:
            t_sg = min(rem, 60)
            a_opt, obj = sg_polish(best_seq, time_limit=t_sg,
                                   lr=0.001, mom=0.95, top_k=max(10, n_best // 12),
                                   temp=0.00001)
            sc, better = evaluate_and_track(a_opt, [best_score], "sg_fine")
            if better:
                best_score = sc
                best_seq = a_opt.copy()

        # Very fine SG
        rem = deadline - time.monotonic() - 25
        if rem > 10:
            t_sg = min(rem, 35)
            a_opt, obj = sg_polish(best_seq, time_limit=t_sg,
                                   lr=0.0005, mom=0.96, top_k=max(5, n_best // 20),
                                   temp=0.000005)
            sc, better = evaluate_and_track(a_opt, [best_score], "sg_veryfine")
            if better:
                best_score = sc
                best_seq = a_opt.copy()

    # === Phase 7: Final LP-FW polish ===
    print(f"\n=== Phase 7: Final LP-FW (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 15
        if rem > 15:
            t_final = min(rem, 60)
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_final)
            sc, better = evaluate_and_track(a_opt, [best_score], "final_lpfw")
            if better:
                best_score = sc
                best_seq = a_opt.copy()

        # One more tiny perturbation + SG
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            n_best = len(best_seq)
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.0005)
            a_opt, obj = sg_polish(f_pert, time_limit=min(rem, 20),
                                   lr=0.0003, mom=0.96, top_k=max(5, n_best // 30),
                                   temp=0.000003)
            sc, better = evaluate_and_track(a_opt, [best_score], "final_perturb")
            if better:
                best_score = sc
                best_seq = a_opt.copy()

    # Fallback
    if best_seq is None:
        best_seq = cosine_init(768, 1.0)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
