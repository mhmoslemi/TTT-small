```python
"""
Search minimizer for
    2*n*max(conv(a,a)) / (sum(a))**2
with nonnegative coefficients.

Strategy:
- Use a fast FFT-based evaluation internally.
- Start from many smooth / irregular initial shapes, including the provided parent.
- Optimize with a mirror-descent / multiplicative-weight method on the smoothed
  objective logsumexp(conv(a,a)).
- Occasionally refine with a Frank-Wolfe LP oracle on smaller sequences.
- Try upsampled versions of the best sequence and final low-temperature polishing.
"""

import numpy as np
import time

try:
    from scipy.optimize import linprog
except Exception:
    linprog = None


def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def normalize_sequence(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.reshape(-1)
    a = np.maximum(a, 0.0)
    if a.size == 0:
        return np.array([1.0], dtype=np.float64)
    if not np.all(np.isfinite(a)):
        a = np.zeros_like(a, dtype=np.float64)
    s = float(np.sum(a))
    if s <= 1e-30:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def convolve_fast(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([0.0], dtype=np.float64)
    size = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, size)
    b = np.fft.irfft(fa * fa, size)[:2 * n - 1]
    b = np.array(b, dtype=np.float64, copy=True)
    b[np.abs(b) < 1e-15] = 0.0
    b[b < 0.0] = 0.0
    return b


def internal_score(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1 or a.size == 0:
        return float("inf")
    a = np.maximum(a, 0.0)
    if not np.all(np.isfinite(a)):
        return float("inf")
    s = float(np.sum(a))
    n = a.size
    if s <= 1e-12:
        return float("inf")
    a = a / s
    b = convolve_fast(a)
    if b.size == 0:
        return float("inf")
    m = float(np.max(b))
    if not np.isfinite(m):
        return float("inf")
    return float(2.0 * n * m)


def gradient_from_weights(w, a):
    """
    For b = conv(a, a), grad_j = 2 * sum_r w_{j+r} a_r.
    """
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    L = 2 * n - 1
    if w.size != L:
        # Fallback: only use valid overlap.
        ww = np.zeros(L, dtype=np.float64)
        m = min(L, w.size)
        ww[:m] = w[:m]
        w = ww
    size = next_pow2(L + n - 1)
    W = np.fft.rfft(w, size)
    Ar = np.fft.rfft(a[::-1], size)
    c = np.fft.irfft(W * Ar, size)
    g = 2.0 * c[n - 1:n - 1 + n]
    g = np.array(g, dtype=np.float64, copy=True)
    g[g < 0.0] = 0.0
    return g


def up_sample_sequence(a, n):
    a = normalize_sequence(a)
    if a.size == int(n):
        return a
    if a.size < 1:
        return np.ones(int(n), dtype=np.float64) / int(n)
    old_x = np.linspace(0.0, 1.0, a.size)
    new_x = np.linspace(0.0, 1.0, int(n))
    y = np.interp(new_x, old_x, a)
    return normalize_sequence(y)


def project_simplex(v):
    v = np.asarray(v, dtype=np.float64).reshape(-1)
    n = v.size
    if n == 1:
        return np.array([max(float(v[0]), 0.0)], dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / float(n)
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / float(rho + 1)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s <= 1e-30:
        return np.ones(n, dtype=np.float64) / float(n)
    return out / s


def solve_lp(f, M):
    if linprog is None:
        return None
    n = len(f)
    if n == 0:
        return None
    c = -np.ones(n, dtype=np.float64)
    A_ub = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, float(M), dtype=np.float64)
    bounds = [(0.0, None)] * n
    res = None
    last_exc = None
    for meth in ("highs", "highs-ds", "highs-ipm"):
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method=meth)
            break
        except Exception as exc:
            last_exc = exc
            res = None
    if res is None:
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds)
        except Exception:
            return None
    if getattr(res, "success", False):
        x = res.x
        x = np.asarray(x, dtype=np.float64)
        x = np.nan_to_num(x, nan=0.0, posinf=0.0, neginf=0.0)
        return x
    return None


def fw_lp_optimize(f0, time_limit, seed=0):
    f = normalize_sequence(f0)
    n = f.size
    if n <= 1:
        return f, internal_score(f)

    conv = convolve_fast(f)
    M = float(conv.max())
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    m_exp = 1.0
    rng = np.random.default_rng(seed)
    alphas = [0.02, 0.05, 0.08, 0.10, 0.15, 0.20, 0.30, 0.40, 0.50, 0.60, 0.70, 0.80, 0.90, 1.00]

    while time.monotonic() - t0 < max(1e-6, float(time_limit)):
        it += 1
        g = solve_lp(f, M * m_exp)
        if g is None:
            m_exp *= 1.2
            if m_exp > 10.0:
                m_exp = 1.0
            continue

        g = normalize_sequence(g)
        if np.max(np.abs(g - f)) < 1e-9:
            no_improve += 1
            m_exp = min(4.0, m_exp * 1.05)
            if no_improve > 40:
                break
            continue

        best_alpha = 0.0
        best_val = M
        for alpha in alphas:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = float(ft.sum())
            if st <= 1e-15:
                continue
            ft = ft / st
            val = float(convolve_fast(ft).max())
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0.0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(8):
                mid = 0.5 * (lo + hi)
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = float(ft.sum())
                if st <= 1e-15:
                    break
                ft = ft / st
                val = float(convolve_fast(ft).max())
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            s = float(f.sum())
            if s <= 1e-15:
                f = best_f.copy()
                break
            f = f / s
            conv = convolve_fast(f)
            M = float(conv.max())
            m_exp = 1.0
            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(4.0, m_exp * 1.08)

        if no_improve > 50:
            f = best_f + rng.standard_normal(n) * 0.002
            f = project_simplex(f)
            conv = convolve_fast(f)
            M = float(conv.max())
            m_exp = 1.0
            no_improve = 0

    return best_f, float(best_obj)


def mirror_descent(a0, time_limit, lr0=0.08, tau_scale0=0.25,
                   min_tau_scale=0.015, max_tau_scale=0.6,
                   max_log=1.0, patience=250, seed=0, verbose=False):
    a = normalize_sequence(a0)
    n = a.size
    if n <= 1:
        return a, internal_score(a)

    floor = 1e-12 / float(n)
    a = np.maximum(a, floor)
    a = a / a.sum()

    b = convolve_fast(a)
    cur_score = float(2.0 * n * b.max())
    best_score = cur_score
    best_a = a.copy()

    lr = float(lr0)
    tau_scale = float(tau_scale0)
    no_improve = 0
    it = 0
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < max(1e-6, float(time_limit)):
        it += 1
        maxb = float(b.max())
        if maxb <= 1e-300:
            break

        tau = max(1e-30, tau_scale * maxb)
        d = (b - maxb) / tau
        d = np.clip(d, -700.0, 0.0)
        w = np.exp(d)
        wsum = float(w.sum())
        if not np.isfinite(wsum) or wsum <= 1e-30:
            w = np.zeros(b.size, dtype=np.float64)
            w[int(np.argmax(b))] = 1.0
            wsum = 1.0
        else:
            w = w / wsum

        grad = gradient_from_weights(w, a)
        gs = float(grad.sum())
        if gs <= 1e-300:
            break

        eta = lr * float(n) / gs
        gmax = float(grad.max())
        if gmax > 1e-300 and eta * gmax > max_log:
            eta = max_log / gmax
        if eta <= 1e-300:
            break

        logfac = -eta * grad
        logfac = np.clip(logfac, -50.0, 0.0)
        aw = np.maximum(a, floor)
        a_new = aw * np.exp(logfac)
        a_new = np.maximum(a_new, 0.0)
        ns = float(a_new.sum())
        if ns <= 1e-30:
            a = best_a.copy()
            b = convolve_fast(a)
            cur_score = float(2.0 * n * b.max())
            break
        a_new = a_new / ns

        b_new = convolve_fast(a_new)
        new_score = float(2.0 * n * b_new.max())

        accepted = False
        if np.isfinite(new_score) and new_score < cur_score + 1e-5:
            a = a_new
            b = b_new
            cur_score = new_score
            accepted = True

        if accepted:
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = a.copy()
                no_improve = 0
                tau_scale = max(min_tau_scale, tau_scale * 0.999)
                lr = min(max(lr, 0.01), lr * 1.02)
                if verbose and it % 200 == 0:
                    print(f"      mirror new best: {best_score:.8f} (it={it})")
            else:
                no_improve += 1
                if no_improve > 30:
                    tau_scale = min(max_tau_scale, tau_scale * 1.003)
        else:
            no_improve += 1
            lr = max(0.01, lr * 0.55)
            tau_scale = min(max_tau_scale, tau_scale * 1.20)

        lr = max(0.01, lr * 0.9999)
        tau_scale = min(max_tau_scale, max(min_tau_scale, tau_scale * 1.0002))

        if no_improve > patience:
            noise = rng.standard_normal(n) * 0.045
            a = best_a * np.exp(noise)
            a = np.maximum(a, floor)
            s = float(a.sum())
            if s <= 1e-30:
                a = best_a.copy()
            else:
                a = a / s
            b = convolve_fast(a)
            cur_score = float(2.0 * n * b.max())
            lr = float(lr0)
            tau_scale = min(0.5, float(tau_scale0) * 1.4)
            no_improve = 0

            if verbose:
                print(f"      mirror restart: it={it}, score={cur_score:.8f}")

    return best_a, float(best_score)


def consider_candidate(seq, score_internal, best_seq, best_score, tag=""):
    if seq is None:
        return best_seq, best_score
    seq = np.asarray(seq, dtype=np.float64)
    if seq.ndim != 1:
        seq = seq.reshape(-1)
    if seq.size == 0 or not np.all(np.isfinite(seq)):
        return best_seq, best_score
    seq = np.maximum(seq, 0.0)
    s = float(seq.sum())
    if s <= 1e-12:
        return best_seq, best_score
    seq = seq / s

    if not np.isfinite(score_internal):
        score_internal = internal_score(seq)
    if not np.isfinite(score_internal) or score_internal >= best_score:
        return best_seq, best_score

    try:
        ext = float(evaluate_sequence(seq.tolist()))
    except Exception:
        ext = float(score_internal)

    if np.isfinite(ext) and ext < best_score:
        print(f"  NEW BEST [{tag}]: ext={ext:.8f}, internal={score_internal:.8f}, n={seq.size}")
        return seq.copy(), float(ext)

    return best_seq, best_score


def generate_initializations(n, parent, rng):
    n = int(n)
    inits = []
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)

    inits.append(np.ones(n, dtype=np.float64))

    if parent is not None:
        p = normalize_sequence(parent)
        if p.size != n:
            p = up_sample_sequence(p, n)
        inits.append(p)

    for beta in (0.5, 1.0, 2.0, 4.0):
        inits.append(np.exp(-beta * x))
        inits.append(np.exp(-beta * (1.0 - x)))

    for pw in (0.25, 0.5, 1.0, 2.0):
        inits.append(np.power(x + 1e-12, pw))
        inits.append(np.power(1.0 - x + 1e-12, pw))

    for c in (0.2, 0.5, 1.0, 1.5):
        a = 1.0 + c * (2.0 * x - 1.0) ** 2
        inits.append(np.maximum(a, 1e-8))

    for mode in (1, 2, 3, 4, 5, 6, 8):
        for phase in (0.0, np.pi / (2.0 * mode)):
            a = 1.0 + 0.8 * np.cos(2.0 * np.pi * mode * x + phase)
            inits.append(np.maximum(a, 1e-8))

    bases = [
        np.ones(n, dtype=np.float64),
        np.exp(-x),
        np.maximum(1.0 + 0.8 * (2.0 * x - 1.0) ** 2, 1e-8),
    ]
    for base in bases:
        noise = rng.standard_normal(n) * 0.45
        a = base * np.exp(noise)
        inits.append(np.maximum(a, 1e-8))

    for p in (0.3, 0.5, 0.7):
        r = (rng.random(n) < p).astype(np.float64)
        ker = np.ones(5, dtype=np.float64) / 5.0
        a = np.convolve(r, ker, mode="same")
        a = 0.55 * a + 0.45 * np.ones(n, dtype=np.float64)
        inits.append(np.maximum(a, 1e-8))

    for alpha in (0.4, 1.0, 3.0):
        try:
            inits.append(rng.dirichlet(np.full(n, alpha, dtype=np.float64)))
        except Exception:
            pass

    cleaned = []
    for a in inits:
        a = normalize_sequence(a)
        if a.size == n and np.all(np.isfinite(a)):
            cleaned.append(a)
    return cleaned


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    start = time.monotonic()
    B = max(60.0, float(budget_s))
    deadline = start + B - 30.0
    rng = np.random.default_rng(seed)

    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        parent = np.clip(parent, 0.0, 1000.0)
        if parent.ndim != 1 or parent.size == 0:
            parent = None
        elif not np.all(np.isfinite(parent)):
            parent = None
        elif parent.sum() < 1e-8:
            parent = None
    except Exception:
        parent = None

    best_seq = None
    best_score = float("inf")

    # Parent preprocessing.
    if parent is not None:
        pnorm = normalize_sequence(parent)
        psc = internal_score(pnorm)
        best_seq, best_score = consider_candidate(pnorm, psc, best_seq, best_score, "parent_initial")

        if pnorm.size <= 600:
            t = max(0.0, deadline - time.monotonic() - 1200.0)
            t = min(50.0, t)
            if t > 5.0:
                try:
                    a_opt, sc = mirror_descent(pnorm, t, lr0=0.10, tau_scale0=0.30,
                                               patience=70, seed=int(seed) + 11, verbose=False)
                    best_seq, best_score = consider_candidate(a_opt, sc, best_seq, best_score, "parent_mirror")
                except Exception:
                    pass

            t = max(0.0, deadline - time.monotonic() - 1100.0)
            t = min(70.0, t)
            if t > 10.0:
                try:
                    a_opt, sc = fw_lp_optimize(pnorm, t, seed=int(seed) + 12)
                    best_seq, best_score = consider_candidate(a_opt, sc, best_seq, best_score, "parent_fw")
                except Exception:
                    pass

            t = max(0.0, deadline - time.monotonic() - 1000.0)
            t = min(35.0, t)
            if t > 5.0 and best_seq is not None:
                try:
                    a_opt, sc = mirror_descent(best_seq, t, lr0=0.06, tau_scale0=0.12,
                                               patience=80, seed=int(seed) + 13, verbose=False)
                    best_seq, best_score = consider_candidate(a_opt, sc, best_seq, best_score, "post_fw_mirror")
                except Exception:
                    pass

    # Choose a main search length.
    if B < 700.0:
        n_main = 512
    else:
        n_main = 1024

    # Initialize and screen on n_main.
    try:
        inits = generate_initializations(n_main, parent, rng)
    except Exception:
        inits = [np.ones(n_main, dtype=np.float64)]

    scored = []
    for i, a in enumerate(inits):
        a = normalize_sequence(a)
        if a.size != n_main:
            continue
        sc = internal_score(a)
        if np.isfinite(sc):
            scored.append((sc, i, a))

    scored.sort()

    candidates = []
    if parent is not None:
        candidates.append(up_sample_sequence(parent, n_main))

    for item in scored:
        if len(candidates) >= 12:
            break
        if len(candidates) == 0 or (item[1] % 2 == 0):
            candidates.append(item[2])

    if not candidates:
        candidates.append(np.ones(n_main, dtype=np.float64) / n_main)

    # Candidate screening with short mirror runs.
    rem_screen = deadline - time.monotonic() - 1000.0
    if rem_screen > 20.0:
        total_screen = min(150.0, max(20.0, rem_screen))
        per = max(2.0, total_screen / max(1, len(candidates)))
        for idx, cand in enumerate(candidates):
            rem = deadline - time.monotonic() - 900.0
            if rem < 5.0:
                break
            t = min(per, 10.0, rem)
            if t < 2.0:
                continue
            try:
                a_opt, sc = mirror_descent(cand, t, lr0=0.12, tau_scale0=0.35,
                                           patience=50, seed=int(seed) + 100 + idx, verbose=False)
                best_seq, best_score = consider_candidate(a_opt, sc, best_seq, best_score, f"screen_{idx}")
            except Exception:
                pass

    # Deep mirror on the current best.
    if best_seq is not None:
        rem = deadline - time.monotonic() - 450.0
        if rem > 30.0:
            t = min(500.0, rem)
            try:
                a_opt, sc = mirror_descent(best_seq, t, lr0=0.08, tau_scale0=0.25,
                                           patience=250, seed=int(seed) + 1000, verbose=True)
                best_seq, best_score = consider_candidate(a_opt, sc, best_seq, best_score, "deep_mirror_1")
            except Exception:
                pass

    # Focused mirror on the current best.
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300.0
        if rem > 40.0:
            t = min(200.0, rem)
            try:
                a_opt, sc = mirror_descent(best_seq, t, lr0=0.05, tau_scale0=0.10,
                                           patience=320, seed=int(seed) + 1001, verbose=True)
                best_seq, best_score = consider_candidate(a_opt, sc, best_seq, best_score, "deep_mirror_2")
            except Exception:
                pass

    # Exploratory mirror on the current best.
    if best_seq is not None:
        rem = deadline - time.monotonic() - 220.0
        if rem > 50.0:
            t = min(110.0, rem)
            try:
                a_opt, sc = mirror_descent(best_seq, t, lr0=0.16, tau_scale0=0.50,
                                           patience=120, seed=int(seed) + 1002, verbose=False)
                best_seq, best_score = consider_candidate(a_opt, sc, best_seq, best_score, "explore_mirror")
            except Exception:
                pass

    # Upsample and refine at larger length.
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200.0
        if rem > 60.0:
            n2 = 2048
            a2 = up_sample_sequence(best_seq, n2)
            sc2 = internal_score(a2)
            best_seq, best_score = consider_candidate(a2, sc2, best_seq, best_score, "upsample_2048")

            t = min(200.0, max(0.0, deadline - time.monotonic() - 100.0))
            if t > 15.0:
                try:
                    a_opt, sc = mirror_descent(a2, t, lr0=0.08, tau_scale0=0.25,
                                               patience=220, seed=int(seed) + 2000, verbose=True)
                    best_seq, best_score = consider_candidate(a_opt, sc, best_seq, best_score, "upsample_mirror")
                except Exception:
                    pass

    # LP refinement on small / downsampled sequences.
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100.0
        if rem > 30.0:
            small_seq = None
            tag = ""
            if best_seq.size <= 576:
                small_seq = normalize_sequence(best_seq)
                tag = "fw_best_small"
            else:
                small_seq = up_sample_sequence(best_seq, 384)
                tag = "fw_downsample_384"

            sc_small = internal_score(small_seq)
            best_seq, best_score = consider_candidate(small_seq, sc_small, best_seq, best_score, "fw_start")

            t = min(100.0, max(0.0, deadline - time.monotonic() - 40.0))
            if t > 10.0:
                try:
                    a_opt, sc = fw_lp_optimize(small_seq, t, seed=int(seed) + 3000)
                    best_seq, best_score = consider_candidate(a_opt, sc, best_seq, best_score, tag)
                except Exception:
                    pass

                t = min(35.0, max(0.0, deadline - time.monotonic() - 20.0))
                if t > 5.0:
                    try:
                        a_opt, sc = mirror_descent(best_seq, t, lr0=0.04, tau_scale0=0.07,
                                                   patience=180, seed=int(seed) + 3001, verbose=False)
                        best_seq, best_score = consider_candidate(a_opt, sc, best_seq, best_score, "fw_post_mirror")
                    except Exception:
                        pass

    # Threshold / sparsification attempts from the best sequence.
    if best_seq is not None:
        base = normalize_sequence(best_seq)
        for pt in (10, 30, 50, 70, 85):
            rem = deadline - time.monotonic() - 20.0
            if rem < 5.0:
                break
            thr = float(np.percentile(base, pt))
            mask = (base > thr).astype(np.float64)
            if int(mask.sum()) < 5:
                continue
            a = 0.65 * base + 0.35 * mask
            a = normalize_sequence(a)
            sc = internal_score(a)
            best_seq, best_score = consider_candidate(a, sc, best_seq, best_score, f"thresh_{pt}")

            t = min(4.5, max(0.0, deadline - time.monotonic() - 10.0))
            if t > 1.0:
                try:
                    a_opt, sc = mirror_descent(a, t, lr0=0.10, tau_scale0=0.20,
                                               patience=35, seed=int(seed) + 4000 + pt, verbose=False)
                    best_seq, best_score = consider_candidate(a_opt, sc, best_seq, best_score, f"thresh_{pt}_mirror")
                except Exception:
                    pass

    # Final fine polish.
    if best_seq is not None:
        rem = deadline - time.monotonic()
        if rem > 15.0:
            t = max(5.0, min(60.0, rem - 5.0))
            try:
                a_opt, sc = mirror_descent(best_seq, t, lr0=0.03, tau_scale0=0.05,
                                           min_tau_scale=0.01, patience=800,
                                           seed=int(seed) + 5000, verbose=False)
                best_seq, best_score = consider_candidate(a_opt, sc, best_seq, best_score, "final_polish")
            except Exception:
                pass

    # Fallback.
    if best_seq is None:
        best_seq = np.ones(1024, dtype=np.float64) / 1024.0
        best_score = internal_score(best_seq)

    best_seq = normalize_sequence(best_seq)

    # Final external verification and safety fallback.
    final_ext = float("inf")
    try:
        final_ext = float(evaluate_sequence(best_seq.tolist()))
    except Exception:
        final_ext = float("inf")

    if not np.isfinite(final_ext):
        fallback = np.ones(1024, dtype=np.float64) / 1024.0
        try:
            fb_ext = float(evaluate_sequence(fallback.tolist()))
        except Exception:
            fb_ext = float("inf")
        if np.isfinite(fb_ext):
            best_seq = fallback
            final_ext = fb_ext

    print(f"FINAL: score={final_ext:.8f}, n={best_seq.size}")

    return [float(x) for x in best_seq]
```
