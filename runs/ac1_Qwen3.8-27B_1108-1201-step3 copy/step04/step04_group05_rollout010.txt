import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 1 << (2 * n - 1 + 1).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2 * n - 1]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient_fft(w, f):
    """
    Compute grad[j] = 2 * sum_k w[k] * f[k-j] using FFT.
    w has length 2n-1 (convolution weights), f has length n.
    Returns array of length n.
    """
    n = len(f)
    S = 1 << (3 * n).bit_length()
    W = np.fft.rfft(w, S)
    F = np.fft.rfft(f, S)
    corr = np.fft.irfft(W * F.conj(), S)
    return 2.0 * corr[:n]


def softmax_weights(conv, temperature, top_k=None):
    """Compute softmax weights over convolution values."""
    if top_k is not None and top_k < len(conv):
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        vals = conv[peaks]
        w = np.zeros(len(conv))
        scaled = vals / temperature
        scaled -= scaled.max()
        exp_vals = np.exp(scaled)
        w[peaks] = exp_vals / exp_vals.sum()
    else:
        scaled = conv / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
    return w


def build_conv_matrix(f):
    """Build B matrix where (B*g)[k] = (f*g)[k]."""
    n = len(f)
    B = np.zeros((2 * n - 1, n))
    for j in range(n):
        B[j:j + n, j] = f
    return B


def solve_linearized_lp(f, conv_f, B=None):
    """Solve: min t s.t. t >= conv_f[k] + 2*(B*g)[k] for all k, g >= -f, sum(g)=0."""
    n = len(f)
    if B is None:
        B = build_conv_matrix(f)
    c = np.zeros(n + 1)
    c[0] = 1.0
    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = 2.0 * B
    b_ub = -conv_f
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)
    bounds = [(None, None)] + [(-f[i], None) for i in range(n)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[1:]
    return None


def solve_maxsum_lp(f, M):
    """Solve: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None and np.any(res.x > 0):
        return res.x
    return None


def line_search(f, target, current_obj, max_alpha=1.0):
    """Find best alpha for f_new = (1-alpha)*f + alpha*target."""
    best_alpha = 0.0
    best_obj = current_obj
    alphas = np.linspace(0.05, max_alpha, 15)
    for alpha in alphas:
        f_new = (1.0 - alpha) * f + alpha * target
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-15:
            continue
        f_new = f_new / s
        obj = fast_convolve(f_new).max()
        if obj < best_obj:
            best_obj = obj
            best_alpha = alpha
    if best_alpha > 0:
        lo = max(0.0, best_alpha - 0.08)
        hi = min(1.0, best_alpha + 0.08)
        for _ in range(8):
            mid = (lo + hi) / 2.0
            f_new = (1.0 - mid) * f + mid * target
            f_new = np.maximum(f_new, 0.0)
            s = f_new.sum()
            if s < 1e-15:
                break
            f_new = f_new / s
            obj = fast_convolve(f_new).max()
            if obj < best_obj:
                best_obj = obj
                best_alpha = mid
                lo = mid
            else:
                hi = mid
    return best_alpha, best_obj


def adam_optimize(f0, time_limit, lr=0.002, beta1=0.9, beta2=0.999,
                  temp_init=0.002, temp_decay=1.0005, top_k=None,
                  reheat_every=500, reheat_factor=1.5, verbose=False):
    """
    Adam-based optimization with adaptive temperature.
    Minimizes max_k (f*f)[k] subject to f>=0, sum(f)=1.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    if top_k is None:
        top_k = max(5, n // 8)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    temperature = temp_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        t += 1

        conv = fast_convolve(f)
        w = softmax_weights(conv, temperature, top_k=top_k)
        grad = compute_gradient_fft(w, f)

        # Remove mean component (tangent space projection for sum constraint)
        grad -= grad.mean()

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        f_new = f - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        f_new = project_simplex(f_new)

        obj = fast_convolve(f_new).max()

            best_f = f_new.copy()
            no_improve = 0
            if verbose and it % 5000 == 0:
                elapsed = time.monotonic() - t0
                print(f"    Adam iter {it}: obj={best_obj:.10f}, temp={temperature:.6f}, elapsed={elapsed:.1f}s")
        else:
            no_improve += 1

        f = f_new
        temperature *= temp_decay

        # Reheating
        if no_improve == reheat_every:
            temperature *= reheat_factor
            lr *= 1.2
            if verbose:
                print(f"    Reheat at iter {it}: temp={temperature:.6f}, lr={lr:.6f}")

        if no_improve > reheat_every * 3:
            # Larger restart
            rng = np.random.default_rng((it * 7 + 3) % 100000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            temperature = temp_init * 0.5
            no_improve = 0

        m_hat = m / (1 - 0.9 ** t)
        v_hat = v / (1 - 0.999 ** t)

        f_new = f - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        f_new = project_simplex(f_new)

        obj = fast_convolve(f_new).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
            if verbose and it % 3000 == 0:
                print(f"    PeakAdam iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        f = f_new

        if no_improve > 800:
            rng = np.random.default_rng((it * 13 + 7) % 100000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            no_improve = 0

    return best_f, best_obj


def generate_diverse_inits(n, parent, rng):
    """Generate diverse initializations including creative patterns."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    xm = 0.5 * (n - 1)

    # Standard
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    # Gaussians
    for sigma_frac in [0.1, 0.15, 0.2, 0.3, 0.4]:
        sigma = sigma_frac * n
        g = np.exp(-0.5 * ((x - xm) / sigma) ** 2)
        inits.append(g / g.sum())

    # Bimodal
    for sep_frac in [0.2, 0.3, 0.4]:
        sep = sep_frac * n
        b = np.exp(-0.5 * ((x - xm + sep / 2) / (n * 0.08)) ** 2) + \
            np.exp(-0.5 * ((x - xm - sep / 2) / (n * 0.08)) ** 2)
        inits.append(b / b.sum())

    # U-shape
    edge_w = 1.0 / (0.3 * np.abs(x - xm) / n + 0.05)
    inits.append(edge_w / edge_w.sum())

    # Cosine
    for freq in [1, 2, 3]:
        c = np.abs(np.cos(np.pi * freq * x / n)) + 0.02
        inits.append(c / c.sum())

    # Periodic (sawtooth-like)
    for period in [8, 16, 32]:
        p = np.exp(-2.0 * ((x % period) / period - 0.5) ** 2)
        inits.append(p / p.sum())

    # Power law
    for alpha_p in [0.5, 1.0, 1.5, 2.0]:
        pl = (1.0 / (np.abs(x - xm) / n + 0.05)) ** alpha_p
        inits.append(pl / pl.sum())

    # Block pattern
    for block_size in [8, 16, 32, 64]:
        nb = n // block_size
        blocks = rng.dirichlet(np.ones(nb) * 2.0)
        bp = np.repeat(blocks, block_size)[:n]
        if len(bp) < n:
            bp = np.concatenate([bp, np.zeros(n - len(bp))])
        bs = bp.sum()
        if bs > 1e-10:
            inits.append(bp / bs)

    # Staircase
    for direction in [1, -1]:
        st = np.arange(1, n + 1, dtype=np.float64)
        if direction == -1:
            st = st[::-1]
        inits.append(st / st.sum())

    # Symmetric peaks at thirds
    for frac in [0.25, 0.33]:
        tp = np.exp(-0.5 * ((x - n * frac) / (n * 0.05)) ** 2) + \
             np.exp(-0.5 * ((x - n * (1 - frac)) / (n * 0.05)) ** 2) + \
             np.exp(-0.5 * ((x - xm) / (n * 0.08)) ** 2)
        inits.append(tp / tp.sum())

    # Parent interpolation
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
        # Perturbed parent
        pp = px + rng.standard_normal(n) * 0.01
        pp = np.maximum(pp, 0)
        pp = pp / pp.sum()
        inits.append(pp)

    return inits


def refine_length(f_best, n_target, rng):
    """Interpolate best solution to target length and do quick optimization."""
    x_src = np.linspace(0, 1, len(f_best))
    x_dst = np.linspace(0, 1, n_target)
    f_interp = np.interp(x_dst, x_src, f_best)
    f_interp = np.maximum(f_interp, 0)
    f_interp = f_interp / f_interp.sum()
    return f_interp


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase hybrid search:
    Phase 1: Diverse initialization and broad exploration
    Phase 2: Adam-based fast optimization (bulk)
    Phase 3: LP Frank-Wolfe (precise direction finding)
    Phase 4: Peak-targeting Adam (fine polishing)
    Phase 5: Local search and alternating refinement
    Phase 6: Length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            pscore = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={pscore:.8f}")
    except Exception:
        pass

    n_p = len(parent) if parent is not None else 384

    # === Phase 1: Diverse Initialization ===
    print("=== Phase 1: Diverse Initialization ===")
    t_start = time.monotonic()
    t_alloc_p1 = 40

    candidates = []
    inits = generate_diverse_inits(n_p, parent, rng)
    for i, ini in enumerate(inits):
        candidates.append((f"init_{i}", ini))

    # Different lengths
    for n_try in [320, 360, 400, 448, 480]:
        if n_try == n_p:
            continue
        ini = np.interp(np.linspace(0, 1, n_try), np.linspace(0, 1, n_p),
                        parent if parent is not None else np.ones(n_p) / n_p)
        ini = np.maximum(ini, 0)
        ini = ini / ini.sum()
        candidates.append((f"len{n_try}", ini))

    # Evaluate all candidates
    for label, cand in candidates:
        elapsed = time.monotonic() - t_start
        if elapsed > t_alloc_p1 or time.monotonic() > deadline - 400:
            break
        try_seq(cand, f"init_{label}")

    if best_seq is None:
        best_seq = np.ones(384) / 384
        try_seq(best_seq, "fallback")

    n_best = len(best_seq)
    print(f"  Phase 1 best: {best_score:.8f} (n={n_best})")

    # === Phase 2: Adam-based fast optimization ===
    print(f"\n=== Phase 2: Adam Optimization (n={n_best}) ===")
    rem = deadline - time.monotonic() - 350
    t_alloc_p2 = max(60, min(rem * 0.4, 500))

    # Run Adam with different hyperparameters
    configs = [
        {"lr": 0.003, "temp_init": 0.003, "temp_decay": 1.0004, "top_k": n_best // 6},
        {"lr": 0.001, "temp_init": 0.001, "temp_decay": 1.0008, "top_k": n_best // 4},
        {"lr": 0.005, "temp_init": 0.005, "temp_decay": 1.0003, "top_k": n_best // 10},
    ]

    for ci, cfg in enumerate(configs):
        rem = deadline - time.monotonic() - (350 - ci * 80)
        t_each = max(30, min(rem / max(1, len(configs) - ci), t_alloc_p2 / len(configs)))
        if t_each < 20:
            break
        print(f"  Adam config {ci}: lr={cfg['lr']}, temp={cfg['temp_init']}, budget={t_each:.0f}s")
        a_opt, _ = adam_optimize(best_seq, time_limit=t_each, **cfg)
        try_seq(a_opt, f"adam_{ci}")

    print(f"  Phase 2 best: {best_score:.8f}")

    # === Phase 3: LP Frank-Wolfe ===
    print(f"\n=== Phase 3: LP Frank-Wolfe (n={n_best}) ===")
    rem = deadline - time.monotonic() - 250
    t_alloc_p3 = max(40, min(rem * 0.5, 400))

    # Linearized LP
    t_half = t_alloc_p3 * 0.55
    print(f"  Linearized LP-FW, budget={t_half:.0f}s")
    a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_half, use_maxsum=False, verbose=True)
    try_seq(a_opt, "lpfw_lin")

    # Max-sum LP
    rem = deadline - time.monotonic() - 200
    t_rem = max(30, min(rem * 0.5, t_alloc_p3 * 0.4))
    if t_rem > 20:
        print(f"  Max-sum LP-FW, budget={t_rem:.0f}s")
        a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_rem, use_maxsum=True)
        try_seq(a_opt, "lpfw_max")

    # Perturbed LP-FW
    rem = deadline - time.monotonic() - 180
    if rem > 30:
        noise = rng.standard_normal(n_best) * 0.008
        f_pert = project_simplex(best_seq + noise)
        t_each = min(rem * 0.4, 80)
        a_opt, _ = lp_fw_optimize(f_pert, time_limit=t_each, use_maxsum=False)
        try_seq(a_opt, "lpfw_pert")

    print(f"  Phase 3 best: {best_score:.8f}")

    # === Phase 4: Peak-targeting Adam ===
    print(f"\n=== Phase 4: Peak-Targeting Adam (n={n_best}) ===")
    rem = deadline - time.monotonic() - 120
    t_alloc_p4 = max(30, min(rem * 0.6, 250))

    print(f"  Peak-target Adam, budget={t_alloc_p4 * 0.6:.0f}s")
    a_opt, _ = peak_target_adam(best_seq, time_limit=t_alloc_p4 * 0.6, lr=0.001)
    try_seq(a_opt, "peak_adam")

    # Fine Adam on the result
    rem = deadline - time.monotonic() - 80
    if rem > 20:
        t_fine = min(rem * 0.5, 100)
        a_opt, _ = adam_optimize(best_seq, time_limit=t_fine,
                                 lr=0.0005, temp_init=0.0005,
                                 temp_decay=1.001, top_k=n_best // 2)
        try_seq(a_opt, "adam_fine")

    print(f"  Phase 4 best: {best_score:.8f}")

    # === Phase 5: Alternating refinement ===
    print(f"\n=== Phase 5: Alternating Refinement ===")
    rem = deadline - time.monotonic() - 50
    t_alloc_p5 = max(20, min(rem * 0.7, 150))

    # Alternating: small perturbation + quick Adam + quick LP
    for attempt in range(5):
        rem = deadline - time.monotonic() - 40
        if rem < 15:
            break
        t_each = min(rem / max(1, 5 - attempt), 30)
        noise_scale = 0.002 * (attempt + 1)
        f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * noise_scale)

        t_adam = t_each * 0.5
        t_lp = t_each * 0.4
        a_opt, _ = adam_optimize(f_pert, time_limit=t_adam,
                                 lr=0.002, temp_init=0.002,
                                 temp_decay=1.001, top_k=n_best // 5)
        try_seq(a_opt, f"alt_adam_{attempt}")

        rem = deadline - time.monotonic() - 35
        if rem > 10:
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_lp, use_maxsum=False)
            try_seq(a_opt, f"alt_lp_{attempt}")

    # Local search
    rem = deadline - time.monotonic() - 30
    if rem > 10:
        print("  Local search...")
        f = best_seq.copy()
        n = len(f)
        conv = fast_convolve(f)
        local_best = conv.max()
        scale = 0.003
        t_ls0 = time.monotonic()

        while time.monotonic() - t_ls0 < min(rem, 50):
            for _ in range(200):
                if time.monotonic() - t_ls0 > min(rem, 50) or time.monotonic() > deadline - 25:
                    break
                noise = rng.standard_normal(n) * scale
                f_p = project_simplex(best_seq + noise)
                obj = fast_convolve(f_p).max()
                if obj < local_best - 1e-14:
                    local_best = obj
                    best_seq_new = f_p.copy()
                    try_seq(best_seq_new, "local")
                    best_seq = best_seq_new
            scale *= 0.97
            if scale < 1e-5:
                scale = 0.002

    print(f"  Phase 5 best: {best_score:.8f}")

    # === Phase 6: Length exploration ===
    print(f"\n=== Phase 6: Length Exploration ===")
    rem = deadline - time.monotonic() - 25
    if rem > 15 and best_seq is not None:
        for n_try in [n_best - 16, n_best + 16, n_best - 32, n_best + 32]:
            if n_try < 64 or n_try > 600 or rem < 8:
                continue
            f_interp = refine_length(best_seq, n_try, rng)
            t_each = min(rem * 0.3, 20)
            a_opt, _ = adam_optimize(f_interp, time_limit=t_each * 0.6,
                                     lr=0.002, temp_init=0.002,
                                     temp_decay=1.001, top_k=n_try // 6)
            try_seq(a_opt, f"len{n_try}")
            rem = deadline - time.monotonic() - 20

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.10f}")

    return [float(x) for x in best_seq]
