import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_subgradient_for_k(a, k):
    """Compute 2*Q_k*a, the subgradient of (a*a)_k."""
    n = len(a)
    sg = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    # (Q_k a)_j = a_{k-j} for j in [js, je]
    sg[js:je+1] = 2.0 * a[k-je:k-js+1][::-1]
    return sg

def softmax(theta):
    theta_shifted = theta - theta.max()
    exp_theta = np.exp(theta_shifted)
    return exp_theta / exp_theta.sum()

                           top_k=100, verbose=False):

    theta = theta0.copy().astype(np.float64)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    a = softmax(theta)
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    # Running average
    avg_a = a.copy()
    count = 0
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        
        # Compute subgradient

                g += w[idx] * compute_subgradient_for_k(a, int(indices[idx]))
