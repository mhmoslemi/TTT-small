import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def mirror_descent_optimize(f0, time_limit, lr_init=0.01, momentum=0.9, top_k=30, beta_init=1000.0, beta_end=50.0, verbose=False):
    """
    Mirror descent (multiplicative weights) for minimizing max(conv(a,a)) 
    subject to a >= 0, sum(a) = 1.
    """
    f = np.maximum(f0.astype(np.float64), 1e-15)
    f = f / f.sum()
    n = len(f)
    
    best_f = f.copy()
    conv = fast_convolve(f)
    best_obj = conv.max()
    
    v_mom = np.zeros(n)
    lr = lr_init
    beta = beta_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = elapsed / max(time_limit, 1e-10)
        
        # Adaptive beta: start high (focus on top peak), gradually lower (spread attention)
        beta = beta_init * (beta_end / beta_init) ** progress if progress > 0 else beta_init
        
        # Adaptive lr: decrease over time
        lr = lr_init * (1.0 / (1.0 + elapsed / (0.3 * time_limit)))
        
        # Compute convolution
        conv = fast_convolve(f)
        cur_obj = conv.max()
        
        # Find top-k peaks
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        
        # Softmax weights on peaks
        pv = conv[peaks]
        scaled = pv * beta
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= w_sum
        
        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            # sg[j] += 2 * w[i] * f[k-j] (where 0 <= k-j < n)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]
        
        # Momentum update in log-space
        v_mom = momentum * v_mom + sg
        
        # Mirror descent step: f[j] <- f[j] * exp(-lr * v_mom[j]) / Z
        log_f = np.log(f)
        log_f_new = log_f - lr * v_mom
        # Normalize
        log_f_new -= log_f_new.max()  # numerical stability
        f_new = np.exp(log_f_new)
        f_new = f_new / f_new.sum()
        
        f = f_new
        conv = fast_convolve(f)
        obj = conv.max()
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 1000 == 0:
                print(f"    MD iter {it}: obj={best_obj:.10f} (lr={lr:.6f}, beta={beta:.1f})")
        else:
            no_improve += 1
        
        # Restart if stuck
        if no_improve > 10000:
            rng = np.random.default_rng(it)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.001, 1e-15)
            f = f / f.sum()
            v_mom = np.zeros(n)
            no_improve = 0
            lr = lr_init * 0.5
    
    return best_f, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast mirror descent (multiplicative weights)
with momentum, adaptive peak weighting, periodic LP-FW refinement, coordinate descent
polishing, and multi-length/multi-restart exploration.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive M relaxation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        g = solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.15
            if m_exp > 5.0:
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 100:
                break
            m_exp = min(m_exp * 1.03, 3.0)
            continue

        best_alpha = 0.0
        best_val = M
        for alpha in [0.03, 0.07, 0.15, 0.3, 0.5, 0.7, 0.85, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-8:
            lo = max(0.0, best_alpha - 0.08)
            hi = min(1.0, best_alpha + 0.08)
            for _ in range(12):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_exp = 1.0
            f = f_new

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 20 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.08, 4.0)

        if no_improve > 80:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.002 + 0.003 * min(no_improve // 100, 5)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def mirror_descent_optimize(f0, time_limit, lr_init=0.005, momentum=0.85, top_k=40,
                            beta_start=5000.0, beta_end=20.0, verbose=False):
    """
    Mirror descent (softmax/multiplicative weights) for minimizing max(conv(a,a))
    subject to a >= 0, sum(a) = 1. Much faster per iteration than LP-FW.
    """
    f = np.maximum(f0.astype(np.float64), 1e-15)
    f = f / f.sum()
    n = len(f)

    best_f = f.copy()
    conv = fast_convolve(f)
    best_obj = conv.max()

    log_f = np.log(f)
    v_mom = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    total_iters = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        total_iters += 1
        elapsed = time.monotonic() - t0
        progress = min(elapsed / max(time_limit, 1e-10), 1.0)

        # Adaptive beta: focus on top peak early, spread later
        beta = beta_start * (beta_end / beta_start) ** (progress ** 0.5)

        # Adaptive learning rate: start moderate, decay
        lr = lr_init / (1.0 + elapsed / (0.4 * time_limit))
        lr = max(lr, lr_init * 0.05)

        # Compute convolution
        conv = fast_convolve(f)
        cur_obj = conv.max()

        # Find top-k peaks
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        # Softmax weights
        pv = conv[peaks]
        scaled = pv * beta
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= w_sum

        # Compute subgradient: sg[j] = 2 * sum_i w_i * f[k_i - j]
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        # Momentum
        v_mom = momentum * v_mom + sg

        # Mirror descent step in log-space
        log_f = log_f - lr * v_mom
        # Numerical stabilization
        log_max = log_f.max()
        if np.isfinite(log_max):
            log_f = log_f - log_max
        f_new = np.exp(log_f)
        f_sum = f_new.sum()
        if f_sum < 1e-30 or not np.isfinite(f_sum):
            f = project_simplex(best_f + np.random.default_rng(it).standard_normal(n) * 0.001)
            log_f = np.log(np.maximum(f, 1e-15))
            v_mom = np.zeros(n)
            no_improve = 0
            continue
        f_new = f_new / f_sum
        f = f_new

        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 5000 == 0:
                print(f"    MD iter {it}: obj={best_obj:.10f} (lr={lr:.6f}, beta={beta:.0f})")
        else:
            no_improve += 1

        # Restart mechanism
        if no_improve > 20000:
            rng = np.random.default_rng(total_iters * 13 + 5)
            scale = 0.001 * (1 + no_improve / 50000.0)
            f = np.maximum(best_f + rng.standard_normal(n) * scale, 1e-15)
            f = f / f.sum()
            log_f = np.log(f)
            v_mom = np.zeros(n)
            no_improve = 0

    return best_f, best_obj


def sg_polish(f0, time_limit, lr=0.003, mom=0.94, top_k=50, temp=0.00005):
    """Subgradient polishing with softmax-weighted peaks (Euclidean step)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        cur_lr = lr * max(0.3, 1.0 - elapsed / time_limit)
        cur_temp = temp * (1.0 + elapsed / (time_limit * 5.0))

        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-15:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 3000:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 1000000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            v = np.zeros(n)
            no_improve = 0

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with fine local search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-12:
                continue

            conv = fast_convolve(f)
            cur_peak = conv.max()
            best_j_val = old_val
            best_j_obj = cur_peak

            multipliers = [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0]
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved and passes > 5:
            break
        if verbose and passes % 10 == 0:
            print(f"    CD pass {passes}: obj={best_obj:.10f}")

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def symmetric_optimize(f0_full, time_limit, verbose=False):

    def half_to_full(fh):
        if n % 2 == 0:
            return np.concatenate([fh, fh[::-1]])
        else:
            return np.concatenate([fh, fh[-2::-1]])

    def full_obj(fh):
        ff = half_to_full(fh)
        ff = np.maximum(ff, 0)
        s = ff.sum()
        if s < 1e-15:
            return 1e10
        ff = ff / s
        return fast_convolve(ff).max()

    # Normalize half
    ff_init = half_to_full(f_half)
    s_init = ff_init.sum()
    if s_init > 1e-15:
        f_half = f_half / s_init * len(ff_init)  # rough

    # Mirror descent on the half
    best_fh = f_half.copy()
    best_obj = full_obj(f_half)

    log_fh = np.log(np.maximum(f_half, 1e-15))
    v_mom = np.zeros(half)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    lr_init = 0.003
    momentum = 0.85

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(elapsed / max(time_limit, 1e-10), 1.0)
        lr = lr_init / (1.0 + elapsed / (0.5 * time_limit))
        lr = max(lr, lr_init * 0.05)

        ff = half_to_full(f_half)
        ff = np.maximum(ff, 0)
        s = ff.sum()
        if s < 1e-15:
            break
        ff = ff / s
        conv = fast_convolve(ff)
        cur_obj = conv.max()

        top_k = min(30, len(conv))
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        beta = 3000.0 * (10.0 / 3000.0) ** (progress ** 0.5)
        pv = conv[peaks]
        scaled = pv * beta
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-30:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Subgradient w.r.t. full sequence
        sg_full = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            sg_full[idx] += 2.0 * w[i] * ff[k - idx]

        # Project to half (by symmetry)
        sg_half = sg_full[:half].copy()
        if n % 2 == 0:
            sg_half += sg_full[n - 1 : n - half : -1]
        else:
            sg_half[-1] += 0  # center element
            sg_half[:-1] += sg_full[n - 1 : n - half : -1]

        v_mom = momentum * v_mom + sg_half
        log_fh = log_fh - lr * v_mom
        log_fh -= log_fh.max()
        f_new_half = np.exp(log_fh)
        f_new_half = np.maximum(f_new_half, 1e-15)
        f_new_half = f_new_half / f_new_half.sum()
        f_half = f_new_half

        obj = full_obj(f_half)
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_fh = f_half.copy()
            no_improve = 0
            if verbose and it % 2000 == 0:
                print(f"    SymMD iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        if no_improve > 15000:
            rng = np.random.default_rng(it * 7 + 3)
            f_half = np.maximum(best_fh + rng.standard_normal(half) * 0.001, 1e-15)
            f_half = f_half / f_half.sum()
            log_fh = np.log(f_half)
            v_mom = np.zeros(half)
            no_improve = 0

    best_full = half_to_full(best_fh)
    best_full = np.maximum(best_full, 0)
    best_full = best_full / best_full.sum()
    return best_full, full_obj(best_fh)


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining fast mirror descent, LP-FW refinement,
    coordinate descent, symmetric optimization, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return float('inf')
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Initialize from parent if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(384) / 384
        p_score = float(evaluate_sequence(parent.tolist()))
        print(f"Using uniform parent: n=384, score={p_score:.8f}")

    # === Phase 1: Fast mirror descent on parent (main workhorse) ===
    print("\n=== Phase 1: Mirror Descent on parent ===")
    t1 = min(600, deadline - time.monotonic() - 800)
    if t1 > 30:
        print(f"  MD n={len(parent)}, t={t1:.0f}s")
        a_opt, _ = mirror_descent_optimize(parent, time_limit=t1,
                                           lr_init=0.004, momentum=0.88,
                                           top_k=50, beta_start=8000.0, beta_end=15.0,
                                           verbose=True)
        try_seq(a_opt, "md_parent")

    # Second MD run with different params
    rem = deadline - time.monotonic() - 700
    if rem > 30 and best_seq is not None:
        print(f"  MD v2 n={len(best_seq)}, t={min(rem, 300):.0f}s")
        a_opt, _ = mirror_descent_optimize(best_seq, time_limit=min(rem, 300),
                                           lr_init=0.002, momentum=0.92,
                                           top_k=80, beta_start=2000.0, beta_end=5.0,
                                           verbose=False)
        try_seq(a_opt, "md_parent_v2")

    # === Phase 2: LP-FW refinement ===
    print(f"\n=== Phase 2: LP-FW (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            t_lpfw = min(rem * 0.6, 200)
            print(f"  LP-FW n={len(best_seq)}, t={t_lpfw:.0f}s")
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=t_lpfw, verbose=True)
            try_seq(a_opt, "lpfw")

    # === Phase 3: Symmetric optimization ===
    print(f"\n=== Phase 3: Symmetric Optimization (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            t_sym = min(rem * 0.5, 150)
            print(f"  SymMD n={len(best_seq)}, t={t_sym:.0f}s")
            a_opt, _ = symmetric_optimize(best_seq, time_limit=t_sym, verbose=True)
            try_seq(a_opt, "symmetric")

    # === Phase 4: Multi-length exploration ===
    print(f"\n=== Phase 4: Multi-Length (best={best_score:.8f}) ===")
    base = best_seq if best_seq is not None else parent

    for target_n in [512, 1024, 640, 896]:
        rem = deadline - time.monotonic() - 200
        if rem < 30:
            break
        f_interp = interpolate_sequence(base, target_n)
        t_alloc = min(rem * 0.4, 120)
        print(f"  MD n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = mirror_descent_optimize(f_interp, time_limit=t_alloc,
                                           lr_init=0.003, momentum=0.90,
                                           top_k=40, beta_start=5000.0, beta_end=10.0,
                                           verbose=False)
        try_seq(a_opt, f"md_n{target_n}")

    # === Phase 5: Coordinate descent polishing ===
    print(f"\n=== Phase 5: Coordinate Descent (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_cd = min(rem * 0.6, 120)
            print(f"  CD n={len(best_seq)}, t={t_cd:.0f}s")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd, verbose=True)
            try_seq(a_opt, "cd")

    # === Phase 6: SG polish ===
    print(f"\n=== Phase 6: SG Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 15:
            t_sg = min(rem * 0.7, 80)
            n_best = len(best_seq)
            print(f"  SG n={n_best}, t={t_sg:.0f}s")
            a_opt, _ = sg_polish(best_seq, time_limit=t_sg,
                                 lr=0.001, mom=0.95,
                                 top_k=max(20, n_best // 6), temp=0.00002)
            try_seq(a_opt, "sg")

    # === Phase 7: Final LP-FW + perturbations ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        rem = deadline - time.monotonic() - 30
        if rem > 10:
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=min(rem, 40))
            try_seq(a_opt, "final_lpfw")

        # Tiny perturbations + MD
        rem = deadline - time.monotonic() - 15
        for p in range(4):
            rem = deadline - time.monotonic() - 10
            if rem < 8:
                break
            noise = rng.standard_normal(n_best) * 0.0008
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = mirror_descent_optimize(f_pert, time_limit=min(rem, 15),
                                               lr_init=0.001, momentum=0.95,
                                               top_k=100, beta_start=500.0, beta_end=5.0)
            try_seq(a_opt, f"perturb_{p}")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
