def fast_convolve(a):

    return np.fft.irfft(A * A, fft_size)[:size]


def compute_soft_gradient(a, conv, K=50, temp=0.001):

Hybrid optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Combines Adam gradient descent, spectral balancing, LP direction finding,
and coordinate polishing. Uses multi-scale initialization and SGLD exploration.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def soft_gradient(a, conv, K=50, temp=0.001):
    """
    Compute the (negative) gradient direction for minimizing soft-max of top-K conv values.
    Returns gradient to subtract from a.
    """
    n = len(a)
    K = min(K, len(conv))
    # Get top K indices
    if K == len(conv):
        idx = np.arange(len(conv))
    else:
        idx = np.argpartition(conv, -K)[-K:]
    vals = conv[idx]
    
    # Softmax weights
    scaled = vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(idx))
        w[np.argmax(vals)] = 1.0
    else:
        w /= w_sum
    
    # Compute gradient of weighted sum of conv[k]
    # d(conv[k])/d(a[i]) = a[k-i] for valid i
    grad = np.zeros(n)
    for j in range(len(idx)):
        if w[j] < 1e-15:
            continue
        k = int(idx[j])
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start > i_end:
            continue
        i_range = np.arange(i_start, i_end + 1)
        grad[i_range] += w[j] * a[k - i_range]
    
    # Subtract mean to make gradient tangent to simplex
    grad -= np.mean(grad)
    return grad


def build_conv_matrix(a):
    """Matrix C such that C @ d = conv(a, d) (full linear convolution)."""
    n = len(a)
    m = 2 * n - 1
    C = np.zeros((m, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def lp_find_direction(a, conv):
    """
    LP: max sum(g) s.t. conv(a, g) <= M*1.1, g >= 0.
    Returns normalized g (sum=1) or None.
    """
    n = len(a)
    M = conv.max() * 1.1
    C = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 5})
        if res.success and res.x is not None and np.isfinite(res.x).all():
            g = np.maximum(res.x, 0)
            gs = g.sum()
            if gs > 1e-10:
                return g / gs
    except Exception:
        pass
    return None


def adam_optimize(a0, time_limit, lr=0.005, K_init=50, temp_init=0.001,
                  beta1=0.9, beta2=0.999, eps=1e-8, seed=0, 
                  sgle_interval=500, sgle_sigma=0.002):
    """Adam optimizer with soft-peak gradient and SGLD exploration."""
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    no_improve = 0
    
    t0 = time.monotonic()
    cur_lr = lr
    temp = temp_init
    K = K_init
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(a)
        
        # Adaptive parameters
        K = min(2 * n - 1, K_init + t // 500)
        temp = temp_init * (1.0 + 0.3 * t / max(1, int(time_limit / 5.0)))
        
        grad = soft_gradient(a, conv, K=K, temp=temp)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        a_new = a - cur_lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()
            no_improve += 10  # Penalize
        
        # Periodic check (every 10 steps)
        if t % 10 == 0:
            conv_check = fast_convolve(a)
            obj_check = 2.0 * n * conv_check.max()
            if obj_check < best_obj - 1e-10:
                best_obj = obj_check
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        
        # Adaptive LR
        if no_improve > 300:
            cur_lr *= 0.9
            no_improve = 150  # Reset partially
        if no_improve > 2000:
            break
        
        # SGLD perturbation
        if t % sgle_interval == 0 and no_improve > sgle_interval // 2:
            sigma = sgle_sigma * min(3.0, no_improve / 100.0)
            a = np.maximum(a + rng.standard_normal(n) * sigma, 0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            m *= 0.5  # Decay momentum after perturbation
            v *= 0.5
    
    return best_a, best_obj
