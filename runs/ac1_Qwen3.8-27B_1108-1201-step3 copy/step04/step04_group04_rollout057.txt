    n = len(a0)
    a = a0.copy()
    conv_len = 2 * n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    
    for _ in range(n_iter):
        # Forward: compute auto-convolution
        A = np.fft.fft(a, fft_size)
        C = np.fft.ifft(A * A).real[:conv_len]
        
        # Project C onto feasible set: max(C) <= M, C >= 0, sum(C) = 1
        C_proj = np.minimum(C, M)
        C_proj = np.maximum(C_proj, 0)
        C_sum = C_proj.sum()
        if C_sum < 1e-15:
            break
        C_proj = C_proj / C_sum  # normalize to sum = 1
        
        # Inverse: find a such that a*a ≈ C_proj
        Cp_fft = np.fft.fft(C_proj, fft_size)
        # We want A_new such that A_new^2 = Cp_fft (in magnitude)
        # Keep phase from current A
        magnitude = np.sqrt(np.abs(Cp_fft))
        phase = np.angle(A)
        A_new = magnitude * np.exp(1j * phase)
        
        # Ensure Hermitian symmetry (for real a)
        # Actually, since C_proj is real, Cp_fft is Hermitian, so magnitude is symmetric
        # and we need A_new to be Hermitian too
        a_new = np.fft.ifft(A_new).real[:n]
        
        # Project to a >= 0, sum(a) = 1
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        a = a_new  # or relax: a = 0.5 * a + 0.5 * a_new
        
    return a

    # FFT size for convolution

    best_a = a.copy()

    while time.monotonic() - t0 < time_limit:

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

            best_a = a.copy()
            no_improve = 0

            no_improve += 1
        
        # Compute softmax weights
        scaled = conv / T
        scaled -= scaled.max()
        weights = np.exp(scaled)
        weights /= weights.sum()

        grad = 2.0 * C[n-1:2*n-1]  # positions n-1 to 2n-2
        
        # Nesterov update
        t_new = (1 + np.sqrt(1 + 4 * t * t)) / 2

        a_prev = a.copy()

    return 2.0 * C[n-1:2*n-1]  # indices n-1 to 2n-2
