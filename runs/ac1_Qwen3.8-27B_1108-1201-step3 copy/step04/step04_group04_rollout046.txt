"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Asymmetric initializations exploiting the asymmetry of self-convolution
2. Sparse-matrix LP-based refinement for fast direction finding
3. Nesterov-accelerated subgradient descent with adaptive peak weighting
4. Peak-targeted coordinate descent that directly reduces the dominant lags
5. Multi-length exploration with hierarchical extension
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def sparse_conv_matrix(a):
    """Build sparse matrix C where C @ g = convolve(a, g)."""
    n = len(a)
    C = lil_matrix((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        C[j:j + n, j] = a
    return C.tocsr()


def compute_gradient(a, conv, top_k=50, temp=0.002):
    """Compute softmax-weighted subgradient of max(conv)."""
    n = len(a)
    idx = np.argpartition(conv, -top_k)[-top_k:]
    vals = conv[idx]
    scaled = (vals - vals.max()) / temp
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(idx))
        w[np.argmax(vals)] = 1.0
    else:
        w /= w_sum

    grad = np.zeros(n)
    for ww, k in zip(w, idx):
        if ww < 1e-13:
            continue
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        j_range = np.arange(j_start, j_end + 1)
        grad[j_range] += ww * a[k - j_range]
    return grad


def nesterov_optimize(a0, time_limit, lr_init=0.01, momentum=0.92, top_k=50, temp_init=0.002):
    """Nesterov-accelerated subgradient descent."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    y = a.copy()
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    mom = momentum
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        y = a + mom * v
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy

        conv_y = fast_convolve(y)
        grad = compute_gradient(y, conv_y, top_k=top_k, temp=temp)

        v = mom * v - lr * grad
        a_new = np.maximum(a + v, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9992
        temp *= 1.008
        mom = min(0.97, mom + 0.0003)

        if no_improve >= 300:
            break

    return best_a, best_obj


def lp_step(a, M, C):
    """Solve one LP: max sum(g) s.t. C@g <= M, g >= 0."""
    n = len(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'time_limit': 30})
        if res.success:
            g = np.maximum(res.x, 0)
            gs = g.sum()
            if gs > 1e-10:
                return g, gs
    except Exception:
        pass
    return None, 0


def lp_optimize(a0, time_limit, verbose=False):
    """LP-based iterative optimization with sparse matrices."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    C = sparse_conv_matrix(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    perturb_count = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()

        g, gs = lp_step(a, M * (1 + 1e-9), C)

        if g is None or gs <= a.sum() * (1 + 1e-8):
            no_improve += 1
        else:
            d = g - a
            best_t = 0.0
            best_trial = None
            for t in [0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]:
                a_try = np.maximum(a + t * d, 0.0)
                st = a_try.sum()
                if st < 1e-15:
                    continue
                a_try = a_try / st
                obj_try = 2.0 * n * fast_convolve(a_try).max()
                if obj_try < best_obj - 1e-11:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    best_t = t
                    break

            if best_t > 0:
                a = best_a.copy()
                C = sparse_conv_matrix(a)
                no_improve = 0
            else:
                no_improve += 1

        iteration += 1
        if verbose and iteration % 50 == 0:
            print(f"    LP iter {iteration}, obj={best_obj:.8f}, t={time.monotonic()-t0:.1f}s")

        if no_improve >= 40:
            rng = np.random.default_rng(iteration * 31 + 17 + perturb_count * 7)
            scale = 0.003 + 0.004 * min(perturb_count, 10)
            noise = rng.standard_normal(n) * scale
            a = np.maximum(best_a + noise, 0.0)
            sa = a.sum()
            if sa < 1e-15:
                a = best_a.copy()
            else:
                a = a / sa
            C = sparse_conv_matrix(a)
            no_improve = 0
            perturb_count += 1
            if perturb_count > 50:
                break

    return best_a, best_obj


def peak_targeted_descent(a0, time_limit):
    """Coordinate descent that targets the peak lags directly."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a0, 2.0 * len(a0)
    a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(99)
    no_improve = 0
    iters_per_check = max(1, int(10000 / n))

    while time.monotonic() - t0 < time_limit:
        for _ in range(iters_per_check):
            if time.monotonic() - t0 > time_limit - 0.5:
                break

            # Find the peak lag
            peak_k = int(np.argmax(conv))

            # Find contributing pairs at the peak
            j_start = max(0, peak_k - n + 1)
            j_end = min(n - 1, peak_k)
            j_range = np.arange(j_start, j_end + 1)
            products = a[j_range] * a[peak_k - j_range]

            if products.max() < 1e-15:
                no_improve += 1
                break

            # Target the largest product: reduce one of the pair
            best_idx = j_range[np.argmax(products)]
            partner = peak_k - best_idx

            # Try reducing the larger of the pair
            if a[best_idx] >= a[partner]:
                target = best_idx
                other = partner
            else:
                target = partner
                other = best_idx

            # Try several reduction amounts
            improved = False
            for frac in [0.02, 0.05, 0.1, 0.15, 0.2]:
                delta = a[target] * frac
                if delta < 1e-12:
                    continue
                a_try = a.copy()
                a_try[target] -= delta
                a_try[other] += delta * 0.5  # Partial redistribution
                st = a_try.sum()
                if st > 1e-15:
                    a_try = a_try / st
                    conv_try = fast_convolve(a_try)
                    obj_try = 2.0 * n * conv_try.max()
                    if obj_try < best_obj - 1e-11:
                        a = a_try
                        conv = conv_try
                        best_obj = obj_try
                        best_a = a.copy()
                        improved = True
                        break

            if improved:
                no_improve = 0
            else:
                no_improve += 1

        if no_improve > 500:
            break

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse asymmetric initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n)

    # Asymmetric exponential decays (key insight: asymmetry reduces conv peak)
    for alpha in [1.0, 2.0, 3.0, 5.0]:
        key = f'exp_a{alpha}'
        arr = np.exp(-alpha * x / n)
        inits[key] = arr

    # Power law decays
    for beta in [0.5, 1.0, 2.0]:
        key = f'pow_b{beta}'
        arr = (1.0 - x / (n - 1)) ** beta + 0.01
        inits[key] = arr

    # Rising asymmetric
    inits['rising'] = x + 1.0

    # Asymmetric sine
    inits['asym_sine'] = np.sin(np.pi * (x + 0.3 * n) / (1.3 * n)) ** 2 + 0.01

    # Bimodal asymmetric (more mass on left)
    left_w = np.exp(-((x - 0.2 * n) / (0.15 * n)) ** 2)
    right_w = np.exp(-((x - 0.8 * n) / (0.1 * n)) ** 2) * 0.5
    inits['asym_bimodal'] = left_w + right_w

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5) * n

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
            if len(p) == n:
                inits['parent'] = p
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps * n

    return inits


def evaluate_obj(a):
    """Quick objective evaluation (scale-invariant, assume sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (a.sum() ** 2)

                best_sequence = a_opt.copy()
                print(f"  Grad polish {gi}: {score:.6f} *NEW BEST*")

        # Final LP micro-polish
        remaining = deadline - time.monotonic() - 20.0
        if remaining > 30:
            t_final = min(60.0, remaining - 15)
            if t_final > 15:
                a_opt, obj = lp_optimize(best_sequence, time_limit=t_final)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Final LP: {score:.6f} *NEW BEST*")

    # =========================================================
    # Phase 4: Try extending to larger n if time permits
    # =========================================================
    if best_sequence is not None and time.monotonic() < deadline - 100:
        print("\n=== Phase 4: Extended Length ===")
        n_best = len(best_sequence)
        n_ext = n_best * 2 if n_best < 2048 else n_best + 512
        if n_ext <= 4096:
            t_ext = min(80.0, deadline - time.monotonic() - 40)
            if t_ext > 20:
                a_ext = np.interp(np.linspace(0, 1, n_ext),
                                  np.linspace(0, 1, n_best),
                                  best_sequence / best_sequence.sum())
                a_ext = np.maximum(a_ext, 0)
                aes = a_ext.sum()
                if aes > 1e-15:
                    a_ext = a_ext / aes
                    a_opt, obj = nesterov_optimize(a_ext, time_limit=min(30, t_ext * 0.6),
                                                   lr_init=0.008, momentum=0.93,
                                                   top_k=60, temp_init=0.001)
                    score = float(evaluate_sequence(a_opt.tolist()))
                    if np.isfinite(score) and score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        print(f"  Extended n={n_ext}: {score:.6f} *NEW BEST*")
                    else:
                        print(f"  Extended n={n_ext}: {score:.6f}")

    print(f"\nPhase 3/4 done. Best: {best_score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(256)
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
