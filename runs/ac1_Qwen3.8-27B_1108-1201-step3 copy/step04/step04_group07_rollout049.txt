"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP-based direction search, smooth softmax L-BFGS optimization,
coordinate descent peak-shaving, and multi-length exploration.
Key innovations: smooth homotopy approach to escape LP local minima,
efficient gradient computation via FFT, and adaptive length selection.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize


def fast_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax(x):
    """Numerically stable softmax."""
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()


def build_conv_matrix(a):
    """Build matrix C such that C @ g gives full linear convolution a * g."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def obj_normalized(a):
    """Compute objective for normalized (sum=1) sequence."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv


def solve_lp_direction(a, M):
    """Solve LP: maximize sum(g) s.t. a*g <= M, g >= 0. Returns normalized g and dual."""
    n = len(a)
    C = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    result = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        g = result.x
        g_sum = g.sum()
        dual = None
        try:
            if hasattr(result, 'ineqlin') and result.ineqlin is not None:
                dual = result.ineqlin.marginals
        except Exception:
            pass
        if g_sum > 1e-10:
            return g / g_sum, g_sum, dual
    return a.copy(), 1.0, None


def line_search(a, d, current_obj, n, max_evals=15):
    """Find best step size along direction d."""
    best_t = 0.0
    best_obj = current_obj
    evals = 0

    for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
        if evals >= max_evals:
            break
        a_try = np.maximum(a + t * d, 0.0)
        s = a_try.sum()
        if s < 1e-15:
            continue
        a_try = a_try / s
        obj_try, _ = obj_normalized(a_try)
        evals += 1
        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_t = t

    if best_t > 0.05 and best_t < 0.95 and evals < max_evals - 3:
        gr = (np.sqrt(5) - 1) / 2
        lo = max(0.01, best_t - 0.06)
        hi = min(0.99, best_t + 0.06)
        d1 = lo + (1.0 - gr) * (hi - lo)
        d2 = lo + gr * (hi - lo)
        for _ in range(3):
            if evals >= max_evals - 1:
                break
            a1 = np.maximum(a + d1 * d, 0.0)
            s1 = a1.sum()
            obj1 = obj_normalized(a1 / s1)[0] if s1 > 1e-15 else current_obj
            evals += 1
            a2 = np.maximum(a + d2 * d, 0.0)
            s2 = a2.sum()
            obj2 = obj_normalized(a2 / s2)[0] if s2 > 1e-15 else current_obj
            evals += 1
            if obj1 < obj2:
                hi, d2, obj2 = d2, d1, obj1
                d1 = lo + (1.0 - gr) * (hi - lo)
            else:
                lo, d1, obj1 = d1, d2, obj2
                d2 = lo + gr * (hi - lo)
            if hi - lo < 0.005:
                break
        mid = (lo + hi) / 2.0
        a_mid = np.maximum(a + mid * d, 0.0)
        sm = a_mid.sum()
        if sm > 1e-15:
            obj_mid, _ = obj_normalized(a_mid / sm)
            evals += 1
            if obj_mid < best_obj:
                best_obj = obj_mid
                best_t = mid

    return best_t, best_obj


def lp_optimize(a0, time_limit):
    """LP-based iterative optimization."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    _, conv = obj_normalized(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    tightness = 0.9995
    perturb_count = 0

    while time.monotonic() - t0 < time_limit:
        M_lp = M * tightness
        g, g_sum, _ = solve_lp_direction(a, M_lp)

        if g_sum <= 1.0 + 1e-7 or np.linalg.norm(g - a) < 1e-12:
            tightness = min(1.003, tightness * 1.004)
            no_improve += 1
        else:
            d = g - a
            t_best, obj_best = line_search(a, d, best_obj, n)
            if obj_best < best_obj - 1e-13:
                a_new = np.maximum(a + t_best * d, 0.0)
                s_new = a_new.sum()
                if s_new > 1e-15:
                    a = a_new / s_new
                    _, conv = obj_normalized(a)
                    M = conv.max()
                    best_obj = 2.0 * n * M
                    best_a = a.copy()
                    no_improve = 0
                    tightness = max(0.999, tightness - 0.0003)
                else:
                    a = best_a.copy()
                    no_improve += 1
            else:
                no_improve += 1

        iteration += 1

        if no_improve >= 60:
            rng = np.random.default_rng(iteration * 11 + perturb_count * 1000)
            scale = 0.001 + 0.0015 * min(perturb_count, 10)
            a = best_a + rng.standard_normal(n) * scale
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
            else:
                a = a / s
            _, conv = obj_normalized(a)
            M = conv.max()
            no_improve = 0
            perturb_count += 1
            tightness = 1.001
            if perturb_count > 40:
                break

    return best_a, best_obj


def smooth_objective_and_grad(x, beta):
    """
    Compute objective and gradient of sum_k exp(beta * conv_k(softmax(x))).
    Returns (objective_value, gradient).
    """
    a = softmax(x)
    n = len(a)
    conv = fast_convolve(a)

    # Compute log-sum-exp for stability
    max_bc = beta * conv.max()
    log_p_unnorm = beta * conv - max_bc
    p_unnorm = np.exp(log_p_unnorm)
    p_sum = p_unnorm.sum()
    p = p_unnorm / p_sum

    # Compute g_i = sum_k p[k] * (a[k-i] + a[i-k])
    # Using FFT: cross-correlation
    fft_size = 1 << (4 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p, fft_size)

    cross1 = np.fft.irfft(np.conj(P) * A, fft_size)[:n]  # (p ⊛ a)[i]
    cross2 = np.fft.irfft(np.conj(A) * P, fft_size)[:n]  # (a ⊛ p)[i]
    g = cross1 + cross2

    # Gradient: df/dx_j = beta * a_j * (g_j - sum_i g_i * a_i)
    weighted_sum = np.dot(g, a)
    grad = beta * a * (g - weighted_sum)

    # Objective value (for monitoring, use the actual sum of exps)
    obj_val = np.log(p_sum) + max_bc / beta  # This is log(sum exp(beta*conv))
    # Actually just return the normalized peak as a proxy
    return float(conv.max()), grad


def smooth_optimize(a0, time_limit, beta_start=20, beta_end=80, n_betas=None):
    """Smooth optimization using L-BFGS on softmax parametrization with homotopy."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_conv_max = fast_convolve(a).max()
    best_a = a.copy()

    t0 = time.monotonic()
    if n_betas is None:
        n_betas = 5

    betas = np.linspace(beta_start, beta_end, n_betas)
    time_per_beta = time_limit / n_betas

    x = np.zeros(n)  # softmax(0) = uniform
    # Initialize from current a
    a_pos = np.maximum(a, 1e-15)
    x = np.log(a_pos) - np.log(a_pos.sum())

    for bi, beta in enumerate(betas):
        beta_time_remaining = min(time_per_beta, time_limit - (time.monotonic() - t0))
        if beta_time_remaining < 5:
            break

        best_x_local = x.copy()
        best_obj_local = best_conv_max

        def objective_and_grad(x_cur):
            _, grad = smooth_objective_and_grad(x_cur, beta)
            # For monitoring, compute actual conv max
            a_cur = softmax(x_cur)
            conv_max = fast_convolve(a_cur).max()
            return conv_max, grad

        result = minimize(objective_and_grad, x, method='L-BFGS-B',
                          jac=True, options={'maxiter': 200, 'gtol': 1e-8,
                                             'maxfun': 500, 'iprint': 0})

        if result.x is not None:
            a_new = softmax(result.x)
            conv_new = fast_convolve(a_new).max()
            if conv_new < best_conv_max - 1e-10:
                best_conv_max = conv_new
                best_a = a_new.copy()
                best_obj_local = conv_new
            x = result.x.copy()

        # Small random perturbation between beta stages
        rng = np.random.default_rng(bi * 7 + 42)
        x += rng.standard_normal(n) * 0.01

    return best_a, best_conv_max


def coordinate_descent(a0, time_limit):
    """Coordinate descent: optimize one coordinate at a time via 1D search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    full_passes = 0

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            v = a[j]
            if v < 1e-15:
                # Try small positive values
                candidates = [1e-4, 5e-4, 1e-3]
            else:
                candidates = [v * 0.3, v * 0.5, v * 0.7, v, v * 1.3, v * 1.5, v * 2.0,
                              v + max(a) * 0.1, v - max(a) * 0.1]
                candidates = [max(0, c) for c in candidates]

            best_t = v
            best_obj_j = 2.0 * n * conv.max() / s ** 2

            for t in candidates:
                if t == v:
                    continue
                # Compute new convolution efficiently
                # conv_k changes: for k != 2j: +2*(t-v)*a[k-j]
                # for k = 2j: (t-v)*(t+v)
                new_conv = conv.copy()
                # Apply changes
                k_range = np.arange(2 * n - 1)
                # For k != 2j: new_conv[k] = conv[k] + 2*(t-v)*a[k-j]
                # a[k-j] is nonzero when 0 <= k-j < n, i.e., j <= k <= j+n-1
                k_lo = max(0, j)
                k_hi = min(2 * n - 2, j + n - 1)
                if k_lo <= k_hi:
                    new_conv[k_lo:k_hi + 1] += 2.0 * (t - v) * a[0:k_hi - k_lo + 1]
                # For k = 2j: additional correction
                k2j = 2 * j
                if 0 <= k2j <= 2 * n - 2:
                    # Already added 2*(t-v)*a[0] for this position (since k-j = j, a[j-j]=a[0]... 
                    # Wait, a[k-j] when k=2j is a[j], not a[0]. Let me redo.
                    # For k = 2j: the formula gives 2*(t-v)*a[2j-j] = 2*(t-v)*a[j]
                    # But the actual change is (t^2 - v^2) = (t-v)*(t+v)
                    # So we need to correct: add (t-v)*(t+v) - 2*(t-v)*a[j] = (t-v)*(t+v-2*a[j])
                    # Since v = a[j]: (t-v)*(t+v) - 2*(t-v)*v = (t-v)*(t+v-2v) = (t-v)*(t-v) = (t-v)^2
                    new_conv[k2j] += (t - v) ** 2

                new_s = s + (t - v)
                if new_s < 1e-15:
                    continue
                new_obj = 2.0 * n * new_conv.max() / (new_s ** 2)

                if new_obj < best_obj_j - 1e-14:
                    best_obj_j = new_obj
                    best_t = t

            if best_t != v and abs(best_t - v) > 1e-12:
                # Apply the change
                a_new = a.copy()
                a_new[j] = best_t
                s_new = a_new.sum()
                if s_new > 1e-15:
                    a = a_new / s_new
                    conv = fast_convolve(a)
                    s = 1.0
                    cur_obj = 2.0 * n * conv.max()
                    if cur_obj < best_obj - 1e-13:
                        best_obj = cur_obj
                        best_a = a.copy()
                        no_improve = 0
                    improved = True

        full_passes += 1
        if not improved:
            no_improve += 1
            if no_improve >= 5:
                # Perturb and restart
                rng = np.random.default_rng(full_passes * 13)
                a = best_a + rng.standard_normal(n) * 0.002
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 1e-15:
                    a = a / s
                    conv = fast_convolve(a)
                no_improve = 0
                if full_passes > 50:
                    break
        else:
            no_improve = 0

    return best_a, best_obj


def upsample_sequence(seq, new_n):
    """Upsample a sequence to a new length using cubic-like interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    new_seq = np.interp(x_new, x_old, seq, left=0, right=0)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s > 1e-15:
        new_seq = new_seq / s
    return new_seq


def get_diverse_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    if parent is not None:
        inits['parent'] = upsample_sequence(parent, n)

    # Various structured inits
    for freq, name in [(1, 'sine1'), (2, 'sine2'), (3, 'sine3'), (4, 'sine4')]:
        vals = 0.6 + 0.4 * np.cos(2.0 * np.pi * freq * x / n)
        inits[name] = vals / vals.sum()

    # Triangular
    tri = np.minimum(x, n - 1 - x) + 1.0
    inits['tri'] = tri / tri.sum()

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits['gauss'] = g / g.sum()

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Ramp
    ramp = np.linspace(0.1, 1.0, n)
    inits['ramp'] = ramp / ramp.sum()

    return inits


def subgradient_polish(a0, time_limit, lr_init=0.005, momentum=0.94, top_k=80, temp=0.0002):
    """Momentum subgradient descent for fine-tuning."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    step_count = 0

    while time.monotonic() - t0 < time_limit:
        # Compute subgradient from top-k peaks
        top_k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -top_k_actual)[-top_k_actual:]
        peak_vals = conv[peaks]
        scaled = peak_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx = np.arange(j_start, j_end + 1)
            sg[idx] += w[i] * 2.0 * a[k - idx]

        v = momentum * v - lr * sg
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        step_count += 1

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve % 30 == 0:
            lr *= 0.997
            temp *= 1.003

        if no_improve >= 300:
            rng = np.random.default_rng(step_count * 3 + 7)
            noise_scale = 0.001 + 0.001 * (no_improve // 300)
            a = best_a + rng.standard_normal(n) * noise_scale
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            v = np.zeros(n)
            conv = fast_convolve(a)
            no_improve = 0

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase optimization combining LP, smooth L-BFGS, coordinate descent,
    and subgradient polishing across multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(p)}, score={evaluate_sequence(parent.tolist()):.6f}")
    except Exception as e:
        print(f"No parent available: {e}")

    def try_update(a_candidate):
        nonlocal best_sequence, best_score
        if a_candidate is None:
            return
        score = float(evaluate_sequence([float(x) for x in a_candidate]))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = np.asarray(a_candidate, dtype=np.float64).copy()
            print(f"  New best: {score:.6f} (n={len(a_candidate)})")

    # ===== Phase 1: LP from parent (n=768) =====
    print("=== Phase 1: LP from parent ===")
    t_start = time.monotonic()
    p1_budget = 250.0

    if parent is not None:
        n_p = len(parent)
        t_alloc = min(p1_budget, deadline - time.monotonic() - 200.0)
        if t_alloc > 20:
            a_opt, _ = lp_optimize(parent, time_limit=t_alloc)
            try_update(a_opt)
    else:
        # No parent, try a few inits at n=512
        for name in ['uniform', 'sine1', 'gauss']:
            a0 = np.ones(512) / 512 if name == 'uniform' else \
                 (0.6 + 0.4 * np.cos(2 * np.pi * np.arange(512) / 512)) / 512
            t_alloc = min(60.0, deadline - time.monotonic() - 200.0)
            if t_alloc > 15:
                a_opt, _ = lp_optimize(a0, time_limit=t_alloc)
                try_update(a_opt)

    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s), best={best_score:.6f}")

    # ===== Phase 2: Smooth L-BFGS optimization =====
    print("=== Phase 2: Smooth L-BFGS ===")
    t_start = time.monotonic()
    p2_budget = 250.0

    if best_sequence is not None:
        t_alloc = min(p2_budget, deadline - time.monotonic() - 250.0)
        if t_alloc > 30:
            a_opt, _ = smooth_optimize(best_sequence, time_limit=t_alloc,
                                       beta_start=15, beta_end=60, n_betas=4)
            try_update(a_opt)

            # LP polish after smooth
            if time.monotonic() < deadline - 100:
                t_alloc2 = min(60.0, deadline - time.monotonic() - 80.0)
                if t_alloc2 > 15:
                    a_opt2, _ = lp_optimize(a_opt, time_limit=t_alloc2)
                    try_update(a_opt2)

    print(f"Phase 2 done ({time.monotonic()-t_start:.0f}s), best={best_score:.6f}")

    # ===== Phase 3: Multi-length exploration =====
    print("=== Phase 3: Multi-length exploration ===")
    t_start = time.monotonic()
    p3_budget = 400.0

    lengths_to_try = [768, 1024, 1280, 1536]
    # Remove current best length from priority (already tried)
    if best_sequence is not None:
        best_n = len(best_sequence)

    for n_target in lengths_to_try:
        if time.monotonic() > deadline - 200.0:
            break

        # Get initialization
        if parent is not None and n_target != len(parent):
            a0 = upsample_sequence(parent, n_target)
        elif best_sequence is not None:
            a0 = upsample_sequence(best_sequence, n_target)
        else:
            a0 = np.ones(n_target) / n_target

        t_alloc = min(120.0, deadline - time.monotonic() - 180.0)
        if t_alloc < 20:
            break

        print(f"  Trying n={n_target}, alloc={t_alloc:.0f}s")
        a_opt, _ = lp_optimize(a0, time_limit=t_alloc)
        try_update(a_opt)

        # Short smooth polish
        if time.monotonic() < deadline - 120.0:
            t_alloc2 = min(40.0, deadline - time.monotonic() - 100.0)
            if t_alloc2 > 15:
                a_opt2, _ = smooth_optimize(a_opt, time_limit=t_alloc2,
                                            beta_start=30, beta_end=50, n_betas=2)
                try_update(a_opt2)

    print(f"Phase 3 done ({time.monotonic()-t_start:.0f}s), best={best_score:.6f}")

    # ===== Phase 4: Coordinate descent on best =====
    print("=== Phase 4: Coordinate descent ===")
    t_start = time.monotonic()
    p4_budget = 250.0

    if best_sequence is not None:
        t_alloc = min(p4_budget, deadline - time.monotonic() - 100.0)
        if t_alloc > 20:
            a_opt, _ = coordinate_descent(best_sequence, time_limit=t_alloc)
            try_update(a_opt)

            # Follow up with LP
            if time.monotonic() < deadline - 60.0:
                t_alloc2 = min(40.0, deadline - time.monotonic() - 50.0)
                if t_alloc2 > 10:
                    a_opt2, _ = lp_optimize(a_opt, time_limit=t_alloc2)
                    try_update(a_opt2)

    print(f"Phase 4 done ({time.monotonic()-t_start:.0f}s), best={best_score:.6f}")

    # ===== Phase 5: Subgradient polish + final refinement =====
    print("=== Phase 5: Final refinement ===")
    t_start = time.monotonic()

    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 10.0

        if remaining > 30:
            # Subgradient polish
            t_sg = min(80.0, remaining * 0.4)
            if t_sg > 15:
                a_opt, _ = subgradient_polish(best_sequence, time_limit=t_sg,
                                              lr_init=0.004, momentum=0.95,
                                              top_k=100, temp=0.00015)
                try_update(a_opt)

            remaining = deadline - time.monotonic() - 10.0

            # Structured perturbations + short LP
            if remaining > 20:
                n_best = len(best_sequence)
                n_perturb = max(1, int(remaining / 25))
                t_per = remaining / n_perturb

                for pi in range(n_perturb):
                    if time.monotonic() > deadline - 10:
                        break
                    if t_per < 8:
                        break

                    scale = 0.001 * (1.0 + 0.5 * pi)
                    noise = rng.standard_normal(n_best) * scale
                    # Add smooth component
                    smooth = np.sin(np.pi * np.arange(n_best) / n_best) * scale * 0.3
                    a_pert = best_sequence + noise + smooth
                    a_pert = np.maximum(a_pert, 0.0)
                    s = a_pert.sum()
                    if s > 1e-15:
                        a_pert = a_pert / s

                    t_alloc = min(t_per, deadline - time.monotonic() - 5)
                    if t_alloc < 5:
                        break

                    if pi % 2 == 0:
                        a_opt, _ = lp_optimize(a_pert, time_limit=t_alloc)
                    else:
                        a_opt, _ = subgradient_polish(a_pert, time_limit=t_alloc,
                                                      lr_init=0.003, momentum=0.95,
                                                      top_k=70, temp=0.0002)
                    try_update(a_opt)

    print(f"Phase 5 done ({time.monotonic()-t_start:.0f}s), best={best_score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final check
    final_score = float(evaluate_sequence([float(x) for x in best_sequence]))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_sequence = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
