"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive sparse-constraint LP
with FFT-accelerated subgradient hybrid, multi-length exploration (512-2048),
and constraint-adaptive peeling strategy. Key innovations: top-K constraint
selection for fast LP, peak-weighted subgradient via FFT, and adaptive
tightness scheduling.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def buildSparseConvRows(a, active_indices):
    """Build rows of convolution matrix for only active constraint indices."""
    n = len(a)
    K = len(active_indices)
    C = np.zeros((K, n))
    for idx, k in enumerate(active_indices):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            C[idx, j_start:j_end+1] = a[k - j_end:k - j_start + 1][::-1]
    return C


def solve_lp_sparse(a, M, active_indices):
    """Solve LP: maximize sum(g) s.t. (a*g)[k] <= M for active k, g >= 0."""
    n = len(a)
    A_ub = buildSparseConvRows(a, active_indices)
    b_ub = np.full(len(active_indices), M)
    c = -np.ones(n)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success and result.x is not None:
            g = result.x
            g_sum = float(g.sum())
            if g_sum > 1e-12:
                return g / g_sum, g_sum
    except Exception:
        pass
    return a.copy(), 1.0


def compute_obj(a):
    """Compute objective value for normalized sequence (sum=1)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv


def sg_direction(a, conv, top_k, temperature):
    """Compute weighted peak subgradient direction via FFT."""
    n = len(a)
    top_k = min(top_k, len(conv))
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peaks]
    scaled = peak_vals / max(temperature, 1e-15)
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum

    weight_vec = np.zeros(len(conv))
    for i in range(len(peaks)):
        weight_vec[int(peaks[i])] = w[i]

    a_padded = np.zeros(len(conv) + n - 1)
    a_padded[:n] = a
    w_padded = np.zeros(len(conv) + n - 1)
    w_padded[:len(conv)] = weight_vec
    fft_sz = 1 << (2 * (len(conv) + n - 1) - 2).bit_length()
    sg = np.fft.irfft(np.fft.rfft(w_padded, fft_sz) * np.fft.rfft(a_padded, fft_sz), fft_sz)
    sg = sg[:len(conv) + n - 1]

    result = np.zeros(n)
    for j in range(n):
        idx = j + n - 1
        if 0 <= idx < len(sg):
            result[j] = sg[idx]
    return result


def line_search(a, d, current_obj, n, max_steps=12):
    """Adaptive line search: coarse scan + golden section."""
    best_t = 0.0
    best_obj = current_obj

    candidates = [0.15, 0.3, 0.45, 0.6, 0.75, 0.9, 1.0]
    for t in candidates:
        a_try = np.maximum(a + t * d, 0.0)
        s = a_try.sum()
        if s < 1e-15:
            continue
        a_try = a_try / s
        obj_try, _ = compute_obj(a_try)
        if obj_try < best_obj - 1e-14:
            best_obj = obj_try
            best_t = t

    if 0.05 < best_t < 0.95:
        gr = (np.sqrt(5) - 1) / 2
        lo = max(0.0, best_t - 0.1)
        hi = min(1.0, best_t + 0.1)
        d1 = lo + (1.0 - gr) * (hi - lo)
        d2 = lo + gr * (hi - lo)
        for _ in range(4):
            a1 = np.maximum(a + d1 * d, 0.0)
            s1 = a1.sum()
            obj1 = compute_obj(a1 / s1)[0] if s1 > 1e-15 else current_obj
            a2 = np.maximum(a + d2 * d, 0.0)
            s2 = a2.sum()
            obj2 = compute_obj(a2 / s2)[0] if s2 > 1e-15 else current_obj
            if obj1 < obj2:
                hi, d2, obj2 = d2, d1, obj1
                d1 = lo + (1.0 - gr) * (hi - lo)
            else:
                lo, d1, obj1 = d1, d2, obj2
                d2 = lo + gr * (hi - lo)
            if abs(hi - lo) < 0.003:
                break
        mid = (lo + hi) / 2.0
        a_mid = np.maximum(a + mid * d, 0.0)
        s_mid = a_mid.sum()
        if s_mid > 1e-15:
            obj_mid, _ = compute_obj(a_mid / s_mid)
            if obj_mid < best_obj:
                best_obj = obj_mid
                best_t = mid

    return best_t, best_obj


def lp_optimize_adaptive(a0, time_limit, K_init=30, K_max=200, verbose=False):

    a = np.maximum(a0.astype(np.float64), 0.0)

    else:
        a = a / s
    n = len(a)

    M = conv.max()

    iteration = 0
    no_improve = 0
    K = K_init

        M_lp = M * tightness
        g, g_sum = solve_lp_sparse(a, M_lp, active)

            if np.linalg.norm(d) < 1e-12:

                if obj_best < best_obj - 1e-13:

                    s_new = a_new.sum()
                    a = a_new / s_new if s_new > 1e-15 else best_a.copy()
                    new_conv = fast_convolve(a)
                    M = new_conv.max()
                    conv = new_conv
                    best_obj = 2.0 * n * M
                    best_a = a.copy()
                    no_improve = 0

    v = np.zeros(n)
    _, conv = compute_obj(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    step = 0

    while time.monotonic() - t0 < time_limit:
        sg = sg_direction(a, conv, top_k, temp)
        v = momentum * v - lr * sg
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        a = a_new / s_new if s_new > 1e-15 else best_a.copy()
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        step += 1

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve % 30 == 0:
            lr *= 0.995
            temp *= 1.003

        if no_improve >= 300:
            rng = np.random.default_rng(step * 3 + 11)
            a = best_a + rng.standard_normal(n) * 0.002
            a = np.maximum(a, 0.0)
            s = a.sum()
            a = a / s if s > 1e-15 else best_a.copy()
            conv = fast_convolve(a)
            v = np.zeros(n)
            no_improve = 0

    return best_a, best_obj


def resample_seq(seq, new_n):
    """Resample sequence to new length via interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    return new_seq / s if s > 1e-15 else new_seq


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    inits['uniform'] = np.ones(n) / n
    if parent is not None:
        inits['parent'] = resample_seq(parent, n)
    sine = 0.7 + 0.3 * np.sin(2.0 * np.pi * x / n)
    inits['sine'] = sine / sine.sum()
    lobe = 0.5 + 0.5 * np.cos(4.0 * np.pi * x / n)
    inits['tlobe'] = lobe / lobe.sum()
    tri = np.minimum(x + 1, n - x).astype(float)
    inits['tri'] = tri / tri.sum()
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.15 * n)) ** 2)
    inits['gauss'] = g / g.sum()
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: adaptive sparse LP + FFT subgradient hybrid.
    Explores lengths 512-2048, uses top-K constraint peeling, and alternating
    LP/SG refinement with adaptive hyperparameters.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(p)}, score~1.5059")
    except Exception:
        pass

    # ===== Phase 1: Multi-length LP Scout (250s) =====
    print("=== Phase 1: Multi-length Scout ===")
    t_start = time.monotonic()
    phase1_budget = 250.0
    phase1_deadline = t_start + phase1_budget

    lengths = [768, 1024, 1536]
    results = []

    for n in lengths:
        if time.monotonic() > phase1_deadline - 30:
            break
        inits = generate_inits(n, parent, rng)
        init_names = []
        if 'parent' in inits:
            init_names.append('parent')
        init_names.extend(['uniform', 'sine', 'random'])
        init_names = init_names[:3]

        t_per = max(15.0, (phase1_budget - 20.0) / (len(lengths) * 3))

        for name in init_names:
            if time.monotonic() > phase1_deadline - 10:
                break
            t_alloc = min(t_per, phase1_deadline - time.monotonic() - 5)
            if t_alloc < 8:
                break

            a0 = inits[name]
            a_opt, obj_opt = lp_optimize_adaptive(a0, time_limit=t_alloc, K_init=25, K_max=120)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                print(f"  n={n} {name}: {score:.6f} ({time.monotonic()-t_start:.0f}s)")

    results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # ===== Phase 2: Deep LP Optimization (800s) =====
    print("=== Phase 2: Deep LP ===")
    t_start2 = time.monotonic()

    top_configs = []
    seen = set()
    for score, n, name, a_opt in results:
        if n not in seen:
            top_configs.append((n, a_opt.copy()))
            seen.add(n)
        if len(top_configs) >= 3:
            break

    if best_sequence is not None:
        bn = len(best_sequence)
        if bn not in seen:
            top_configs.append((bn, best_sequence.copy()))

    remain_for_p2 = deadline - time.monotonic() - 450.0
    if remain_for_p2 < 100:
        remain_for_p2 = 100
    t_per_deep = max(40.0, remain_for_p2 / max(1, len(top_configs)))

    for i, (n, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 450:
            break
        t_alloc = min(t_per_deep, deadline - time.monotonic() - 420)
        if t_alloc < 15:
            break

        a_opt, obj_opt = lp_optimize_adaptive(a0, time_limit=t_alloc, K_init=20, K_max=220, verbose=True)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Deep LP {i} (n={n}): {score:.6f}")

        # Quick SG burst after LP
        if time.monotonic() < deadline - 430:
            t_sg = min(30.0, deadline - time.monotonic() - 420)
            if t_sg > 10:
                a_sg, _ = sg_polish(a_opt, time_limit=t_sg, lr_init=0.004, momentum=0.95, top_k=100, temp=0.0001)
                s_sg = float(evaluate_sequence(a_sg.tolist()))
                if np.isfinite(s_sg) and s_sg < best_score:
                    best_score = s_sg
                    best_sequence = a_sg.copy()
                    print(f"  SG-after-LP {i}: {s_sg:.6f}")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")

    # ===== Phase 3: Intensive Hybrid Refinement (450s) =====
    print("=== Phase 3: Hybrid Refinement ===")
    t_start3 = time.monotonic()

    if best_sequence is not None:
        n_best = len(best_sequence)
        remain = deadline - time.monotonic() - 120.0
        if remain > 50:
            # Alternating LP/SG cycles
            cycle_time = min(60.0, remain / 8)
            n_cycles = max(3, int(remain / (cycle_time * 2)))

            for ci in range(n_cycles):
                if time.monotonic() > deadline - 100:
                    break

                # LP phase
                t_lp = min(cycle_time, deadline - time.monotonic() - 80)
                if t_lp > 10:
                    a_lp, _ = lp_optimize_adaptive(best_sequence, time_limit=t_lp, K_init=30, K_max=250)
                    s_lp = float(evaluate_sequence(a_lp.tolist()))
                    if np.isfinite(s_lp) and s_lp < best_score:
                        best_score = s_lp
                        best_sequence = a_lp.copy()

                # SG phase
                t_sg = min(cycle_time, deadline - time.monotonic() - 60)
                if t_sg > 10:
                    a_sg, _ = sg_polish(best_sequence, time_limit=t_sg, lr_init=0.003, momentum=0.96, top_k=120, temp=0.00008)
                    s_sg = float(evaluate_sequence(a_sg.tolist()))
                    if np.isfinite(s_sg) and s_sg < best_score:
                        best_score = s_sg
                        best_sequence = a_sg.copy()

                # Perturbed restart every 3 cycles
                if (ci + 1) % 3 == 0 and n_cycles - ci > 2:
                    rng2 = np.random.default_rng(ci * 100 + 42)
                    a_pert = best_sequence + rng2.standard_normal(n_best) * 0.001
                    a_pert = np.maximum(a_pert, 0.0)
                    s_p = a_pert.sum()
                    a_pert = a_pert / s_p if s_p > 1e-15 else best_sequence.copy()

                    t_restart = min(25.0, deadline - time.monotonic() - 50)
                    if t_restart > 8:
                        a_r, _ = lp_optimize_adaptive(a_pert, time_limit=t_restart, K_init=25, K_max=150)
                        s_r = float(evaluate_sequence(a_r.tolist()))
                        if np.isfinite(s_r) and s_r < best_score:
                            best_score = s_r
                            best_sequence = a_r.copy()
                            print(f"  Restart {ci}: {s_r:.6f}")

                if ci % 2 == 0:
                    print(f"  Cycle {ci}: best={best_score:.6f} ({time.monotonic()-t_start3:.0f}s)")

    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")

    # ===== Phase 4: Final Fine Polish (120s) =====
    print("=== Phase 4: Final Polish ===")
    t_start4 = time.monotonic()

    if best_sequence is not None:
        # Very fine SG
        remain = deadline - time.monotonic() - 30.0
        if remain > 20:
            a_fine, _ = sg_polish(best_sequence, time_limit=remain * 0.5,
                                  lr_init=0.002, momentum=0.97, top_k=150, temp=0.00005)
            s_f = float(evaluate_sequence(a_fine.tolist()))
            if np.isfinite(s_f) and s_f < best_score:
                best_score = s_f
                best_sequence = a_fine.copy()

        # Final LP
        remain = deadline - time.monotonic() - 15.0
        if remain > 15:
            a_final, _ = lp_optimize_adaptive(best_sequence, time_limit=remain, K_init=50, K_max=300)
            s_f = float(evaluate_sequence(a_final.tolist()))
            if np.isfinite(s_f) and s_f < best_score:
                best_score = s_f
                best_sequence = a_final.copy()

    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
