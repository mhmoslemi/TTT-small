Since the objective is homogeneous of degree 0 (scale-invariant), this equals 2*n*max(conv(a',a'))/(1-a_i+t)^2.

- For k where a_i doesn't appear: same as conv(a,a)_k
- For k where a_i appears: the contribution of a_i changes from a_i to t

Specifically, if k = 2i: conv(a',a')_{2i} = conv(a,a)_{2i} - a_i^2 + t^2
If k in [i, i+n-1], k ≠ 2i: conv(a',a')_k = conv(a,a)_k - 2*a_i*a_{k-i} + 2*t*a_{k-i}

- L_k(t) = conv[k] - 2*a_i*a_{k-i} + 2*t*a_{k-i} for k in [i, i+n-1], k≠2i

f(t) = max(C, Q(t), max_k L_k(t)) / (1-a_i+t)^2

The numerator is convex in t (max of convex functions: constant, convex quadratic, linear functions).

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FW-LP for global structure
2. LSE gradient descent with Adam for medium polishing
3. Coordinate descent with 1D golden-section search for final precision
4. Multi-length exploration
The key new addition is coordinate descent which exploits the 1D structure
of each coordinate's contribution to the autocorrelation peak.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_gradient(conv, f, beta):
    """Compute gradient of log-sum-exp smoothed max of conv w.r.t. f.
    Returns gradient of shape (n,) for f of shape (n,).
    """
    n = len(f)
    bc = beta * conv
    bc_max = bc.max()
    exp_bc = np.exp(bc - bc_max)
    w = exp_bc / exp_bc.sum()

    # Gradient: grad[i] = 2 * sum_m w[m] * f contribution
    # conv[m] = sum_j f[j]*f[m-j], d(conv[m])/df[i] = 2*f[m-i] (with care at m=2i)
    # Use FFT cross-correlation
    f_flipped = f[::-1]
    total_len = len(w) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv_out = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv_out[n - 1:n - 1 + n]


def build_lp_matrix(f):
    """Build sparse constraint matrix for the LP. Shape: (2n-1, n)."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend(k for _ in js)
            cols.extend(js.tolist())
            vals.extend(float(f[ks]) for ks in (k - js))
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. conv(f,g)_k <= M for all k, g >= 0."""
    n = len(f)
    A = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 30})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def fw_lp_optimize(f0, time_limit):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        g = solve_lp_oracle(f, M * m_exp)
        if g is None or g.sum() < 1e-15:
            m_exp *= 1.15
            if m_exp > 8.0:
                m_exp = 1.0
            no_improve += 1
            if no_improve > 30:
                break
            continue
        m_exp = 1.0

        g = g / g.sum()

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        alphas = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
        for alpha in alphas:
            ft = np.maximum((1.0 - alpha) * f + alpha * g, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = np.maximum((1.0 - mid) * f + mid * g, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = np.maximum((1.0 - best_alpha) * f + best_alpha * g, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 50:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.003 + 0.005 * min(1.0, (no_improve - 50) / 60.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0

    return best_f, best_obj


def lse_adam_optimize(f0, time_limit, beta_start=10.0, beta_end=5000.0):
    """LSE smoothed optimization with Adam optimizer."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = 3.0 / n
    m = np.zeros(n)
    v = np.zeros(n)
    t_step = 0
    b1, b2, eps = 0.85, 0.999, 1e-8

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        t_step += 1

        conv = fast_convolve(f)
        grad = compute_lse_gradient(conv, f, beta)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            no_improve += 10
            if no_improve > 200:
                break
            continue

        # Adam update
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad ** 2
        m_hat = m / (1 - b1 ** t_step)
        v_hat = v / (1 - b2 ** t_step)
        update = m_hat / (np.sqrt(v_hat) + eps)

        f_new = f - lr * update
        f_new = project_simplex(f_new)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            f = f_new  # still accept (helps exploration)
            no_improve += 1
            if no_improve % 20 == 0:
                lr *= 0.93
            if no_improve > 300:
                f = best_f.copy()
                m = np.zeros(n)
                v = np.zeros(n)
                t_step = 0
                lr = max(3.0 / n, lr * 0.5)
                beta = min(beta * 2.0, beta_end)
                no_improve = 0

        if beta < beta_end:
            beta *= 1.004

        if it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-Adam it={it}, obj={best_obj:.12f}, beta={beta:.1f}, lr={lr:.6f}, t={elapsed:.0f}s")

    return best_f, best_obj


def coordinate_descent_polish(f0, time_limit, n_sweeps_max=500):
    """Coordinate descent with 1D optimization for final polishing.
    For each coordinate, find the optimal value via grid + golden section search.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    sweep = 0

    while time.monotonic() - t0 < time_limit and sweep < n_sweeps_max:
        sweep += 1
        improved = False

        conv = fast_convolve(f)

        for i in range(n):
            # Determine affected conv indices: k in [i, i+n-1]
            k_lo = i
            k_hi = min(i + n - 1, 2 * n - 2)

            # Compute base conv values (removing a_i contribution)
            # conv[k] for k in [k_lo, k_hi] includes terms with a_i
            # For k = 2i: includes a_i^2
            # For k != 2i: includes 2*a_i*a[k-i]
            ai = f[i]
            if ai < 1e-15:
                # If a_i is essentially 0, try small positive values
                t_candidates = [0.0, 1e-6, 5e-6, 1e-5, 5e-5]
            else:
                t_max = max(5.0 / n, ai * 3.0)
                # Grid of candidate values
                t_candidates = np.linspace(0.0, t_max, 40)

            best_t = ai
            best_obj_i = conv.max()

            for t in t_candidates:
                if t < 0:
                    continue
                # Update affected conv values
                new_vals = conv.copy()
                for k in range(k_lo, k_hi + 1):
                    # Original contribution of a_i to conv[k]:
                    if k == 2 * i:
                        old_contrib = ai * ai
                    else:
                        j = k - i
                        if 0 <= j < n:
                            old_contrib = 2.0 * ai * f[j]
                        else:
                            old_contrib = 0.0
                    # New contribution with t:
                    if k == 2 * i:
                        new_contrib = t * t
                    else:
                        j = k - i
                        if 0 <= j < n:
                            new_contrib = 2.0 * t * f[j]
                        else:
                            new_contrib = 0.0
                    new_vals[k] = conv[k] - old_contrib + new_contrib

                # Also need to account for normalization
                new_sum = 1.0 - ai + t
                if new_sum < 1e-10:
                    continue
                # Objective = max(new_vals) / new_sum^2
                obj_i = new_vals.max() / (new_sum * new_sum)
                if obj_i < best_obj_i - 1e-15:
                    best_obj_i = obj_i
                    best_t = t

            # Golden section refinement around best_t
            if best_t > 0:
                lo = max(0.0, best_t - 0.3 / n)
                hi = best_t + 0.3 / n
                # Verify lo and hi bracket the minimum
                obj_lo = _coord_obj(f, i, lo)
                obj_hi = _coord_obj(f, i, hi)
                if obj_lo < best_obj_i:
                    hi = best_t
                    best_t = lo
                    best_obj_i = obj_lo
                if obj_hi < best_obj_i:
                    lo = best_t
                    best_t = hi
                    best_obj_i = obj_hi

                # Golden section search
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c = hi - gr * (hi - lo)
                d = lo + gr * (hi - lo)
                fc = _coord_obj(f, i, c)
                fd = _coord_obj(f, i, d)
                for _ in range(20):
                    if fc < fd:
                        hi = d
                        d = c
                        fd = fc
                        c = hi - gr * (hi - lo)
                        fc = _coord_obj(f, i, c)
                    else:
                        lo = c
                        c = d
                        fc = fd
                        d = lo + gr * (hi - lo)
                        fd = _coord_obj(f, i, d)
                best_t = (lo + hi) / 2.0
                best_obj_i = _coord_obj(f, i, best_t)

            if abs(best_t - ai) > 1e-12:
                f[i] = best_t
                # Renormalize
                f = f / f.sum()
                improved = True

        if improved:
            conv = fast_convolve(f)
            obj = conv.max()
            if obj < best_obj - 1e-15:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            if no_improve > 3:
                # Try perturbation
                rng = np.random.default_rng(sweep * 13 + 5)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                no_improve = 0

        if sweep % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD sweep={sweep}, obj={best_obj:.12f}, t={elapsed:.0f}s")

    return best_f, best_obj


def _coord_obj(f, i, t):
    """Compute the objective if f[i] were set to t (before renormalization)."""
    n = len(f)
    ai = f[i]
    new_sum = 1.0 - ai + t
    if new_sum < 1e-10:
        return 1e10

    k_lo = i
    k_hi = min(i + n - 1, 2 * n - 2)

    # We need max of all conv values after updating
    # Only conv[k] for k in [k_lo, k_hi] change
    # Get current max from unaffected regions
    if k_lo > 0:
        conv_affected = fast_convolve(f)
    else:
        conv_affected = fast_convolve(f)

    # Compute new max
    max_val = 0.0
    for k in range(2 * n - 1):
        if k_lo <= k <= k_hi:
            orig = conv_affected[k]
            if k == 2 * i:
                old_c = ai * ai
                new_c = t * t
            else:
                j = k - i
                if 0 <= j < n:
                    old_c = 2.0 * ai * f[j]
                    new_c = 2.0 * t * f[j]
                else:
                    old_c = 0.0
                    new_c = 0.0
            val = orig - old_c + new_c
        else:
            val = conv_affected[k]
        if val > max_val:
            max_val = val

    return max_val / (new_sum * new_sum)


def _coord_obj_fast(f, i, t, conv_cache):
    """Faster version using conv cache."""
    n = len(f)
    ai = f[i]
    new_sum = 1.0 - ai + t
    if new_sum < 1e-10:
        return 1e10

    k_lo = i
    k_hi = min(i + n - 1, 2 * n - 2)

    # Max of unaffected conv values
    if k_lo > 0 and k_hi < 2 * n - 2:
        unaffected = np.max(np.concatenate([conv_cache[:k_lo], conv_cache[k_hi+1:]]))
    elif k_lo == 0:
        unaffected = conv_cache[k_hi+1:].max() if k_hi < 2*n-2 else 0.0
    else:
        unaffected = conv_cache[:k_lo].max()

    # Max of affected conv values
    max_affected = 0.0
    for k in range(k_lo, k_hi + 1):
        orig = conv_cache[k]
        if k == 2 * i:
            old_c = ai * ai
            new_c = t * t
        else:
            j = k - i
            if 0 <= j < n:
                old_c = 2.0 * ai * f[j]
                new_c = 2.0 * t * f[j]
            else:
                old_c = 0.0
                new_c = 0.0
        val = orig - old_c + new_c
        if val > max_affected:
            max_affected = val

    max_val = max(unaffected, max_affected)
    return max_val / (new_sum * new_sum)

            # Handle separately

            new_sum_arr = 1.0 - ai + t_vals  # shape (n_grid,)

            # Compute objective for each t
            best_t = ai
            best_obj_t = current_max  # current max is upper bound

            # For each t, the new conv values are:
            # For affected k (non-quad): conv[k] + lin_c*(t) + const_shift
            # For affected k (quad): conv[k] + t^2 + const_shift (const_shift = -ai^2)
            # For unaffected k: conv[k] (unchanged)

            max_unaffected = 0.0
            mask_unaffected = np.ones(n_conv, dtype=bool)
            mask_unaffected[ks_valid] = False
            if mask_unaffected.any():
                max_unaffected = conv[mask_unaffected].max()

            # Vectorized evaluation
            # For each t_j, obj_j = max(max_unaffected, max_affected(j)) / new_sum(j)^2
            # max_affected(j) = max over k of [conv[k] + lin_c_k * t_j + const_shift_k, or t_j^2 + conv[2i] - ai^2]

            affected_base = conv[ks_valid].copy()
            # Non-quad part: affected_base + lin_c * t + const_shift
            # Quad part: affected_base + t^2 (const_shift for quad is -ai^2)

            non_quad_mask = ~quad_mask
            if non_quad_mask.any():
                base_nq = affected_base[non_quad_mask] + const_shift[non_quad_mask]
                lc_nq = lin_c[non_quad_mask]
            else:
                base_nq = np.array([0.0])
                lc_nq = np.array([0.0])

            if quad_mask.any():
                base_q = affected_base[quad_mask] + const_shift[quad_mask]
                idx_q = np.where(quad_mask)[0]
            else:
                base_q = np.array([0.0])
                idx_q = np.array([], dtype=int)

            # Evaluate on grid
            objs = np.full(n_grid, 1e10)
            for j_idx in range(n_grid):
                t = t_vals[j_idx]
                new_s = 1.0 - ai + t
                if new_s < 1e-10:
                    continue
                # Non-quad affected:
                if len(base_nq) > 0:
                    vals_nq = base_nq + lc_nq * t
                    max_nq = vals_nq.max()
                else:
                    max_nq = 0.0
                # Quad affected:
                if len(base_q) > 0:
                    vals_q = base_q + t * t
                    max_q = vals_q.max()
                else:
                    max_q = 0.0
                max_aff = max(max_nq, max_q)
                max_total = max(max_unaffected, max_aff)
                objs[j_idx] = max_total / (new_s * new_s)

            # Find best on grid
            min_idx = np.argmin(objs)
            if objs[min_idx] < best_obj_t - 1e-14:
                best_t = t_vals[min_idx]
                best_obj_t = objs[min_idx]

            # Golden section refinement
            if 0 < min_idx < n_grid - 1:
                lo = t_vals[max(0, min_idx - 1)]
                hi = t_vals[min(n_grid - 1, min_idx + 1)]
                gr = (np.sqrt(5.0) - 1.0) / 2.0
                c = hi - gr * (hi - lo)
                d = lo + gr * (hi - lo)

                def eval_at(t_val):
                    new_s = 1.0 - ai + t_val
                    if new_s < 1e-10:
                        return 1e10
                    if len(base_nq) > 0:
                        max_nq = (base_nq + lc_nq * t_val).max()
                    else:
                        max_nq = 0.0
                    if len(base_q) > 0:
                        max_q = (base_q + t_val * t_val).max()
                    else:
                        max_q = 0.0
                    max_total = max(max_unaffected, max_nq, max_q)
                    return max_total / (new_s * new_s)

                fc = eval_at(c)
                fd = eval_at(d)
                for _ in range(25):
                    if fc < fd:
                        hi = d
                        d = c
                        fd = fc
                        c = hi - gr * (hi - lo)
                        fc = eval_at(c)
                    else:
                        lo = c
                        c = d
                        fc = fd
                        d = lo + gr * (hi - lo)
                        fd = eval_at(d)
                t_mid = (lo + hi) / 2.0
                obj_mid = eval_at(t_mid)
                if obj_mid < best_obj_t:
                    best_t = t_mid
                    best_obj_t = obj_mid

            if abs(best_t - ai) > 1e-13:
                f[i] = best_t
                f = f / f.sum()
                improved = True

        if improved:
            conv = fast_convolve(f)
            obj = conv.max()
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            if no_improve > 5:
                rng = np.random.default_rng(sweep * 17 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                no_improve = 0

        if sweep % 30 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD2 sweep={sweep}, obj={best_obj:.14f}, t={elapsed:.0f}s")

    return best_f, best_obj


def try_sequence(a, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.maximum(a, 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.10f} (n={len(a)})")
    return score


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase optimization:
    Phase 1: FW-LP on parent
    Phase 2: LSE-Adam polishing
    Phase 3: Coordinate descent v2 (vectorized 1D search)
    Phase 4: Alternative lengths + final CD polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_tracker = [float('inf'), None]

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.10f}")
            try_sequence(parent, best_tracker)
    except Exception as e:
        print(f"No parent: {e}")

    if parent is None:
        parent = np.ones(448) / 448
        try_sequence(parent, best_tracker)

    n_parent = len(parent)

    # === Phase 1: FW-LP ===
    remaining = deadline - time.monotonic()
    t1 = min(300, remaining * 0.25)
    print(f"\n=== Phase 1: FW-LP ({t1:.0f}s) ===")
    if t1 > 30:
        a_opt, _ = fw_lp_optimize(parent, time_limit=t1)
        try_sequence(a_opt, best_tracker)

    # === Phase 2: LSE-Adam ===
    remaining = deadline - time.monotonic()
    t2 = min(350, remaining * 0.25)
    print(f"\n=== Phase 2: LSE-Adam ({t2:.0f}s) ===")
    if t2 > 30:
        current = best_tracker[1] if best_tracker[1] is not None else parent
        a_opt, _ = lse_adam_optimize(current, time_limit=t2,
                                     beta_start=8.0, beta_end=8000.0)
        try_sequence(a_opt, best_tracker)

    # === Phase 3: Coordinate Descent V2 ===
    remaining = deadline - time.monotonic()
    t3 = min(400, remaining * 0.30)
    print(f"\n=== Phase 3: CD V2 ({t3:.0f}s) ===")
    if t3 > 30:
        current = best_tracker[1] if best_tracker[1] is not None else parent
        a_opt, _ = coordinate_descent_v2(current, time_limit=t3, n_sweeps_max=200)
        try_sequence(a_opt, best_tracker)

    # === Phase 4: More LSE + CD alternating ===
    remaining = deadline - time.monotonic()
    t4a = min(200, remaining * 0.25)
    t4b = min(250, remaining - t4a - 200)
    print(f"\n=== Phase 4: Alternating ({t4a:.0f}s + {t4b:.0f}s) ===")
    if t4a > 20:
        current = best_tracker[1] if best_tracker[1] is not None else parent
        a_opt, _ = lse_adam_optimize(current, time_limit=t4a,
                                     beta_start=200.0, beta_end=20000.0)
        try_sequence(a_opt, best_tracker)

    if t4b > 20:
        current = best_tracker[1] if best_tracker[1] is not None else parent
        a_opt, _ = coordinate_descent_v2(current, time_limit=t4b, n_sweeps_max=300)
        try_sequence(a_opt, best_tracker)

    # === Phase 5: Alternative lengths ===
    remaining = deadline - time.monotonic()
    print(f"\n=== Phase 5: Alternative lengths ({remaining:.0f}s remaining) ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_best = len(current_best)

    alt_lengths = [384, 512, 320, 576]
    alt_lengths = [n_alt for n_alt in alt_lengths if n_alt != n_best]

    for n_alt in alt_lengths[:2]:
        t_remaining = deadline - time.monotonic() - 30
        if t_remaining < 60:
            break
        t_alloc = min(150, t_remaining - 60)
        if t_alloc < 30:
            break

        print(f"  Trying n={n_alt} with {t_alloc:.0f}s")
        # Interpolate
        x_old = np.linspace(0, 1, len(current_best))
        x_new = np.linspace(0, 1, n_alt)
        f_interp = np.interp(x_new, x_old, current_best)
        f_interp = np.maximum(f_interp, 0)
        if f_interp.sum() > 1e-10:
            f_interp = f_interp / f_interp.sum()

        # Quick optimization
        t_fw = t_alloc * 0.3
        t_cd = t_alloc * 0.7
        a_opt, _ = fw_lp_optimize(f_interp, time_limit=t_fw)
        try_sequence(a_opt, best_tracker)

        cur_alt = best_tracker[1] if best_tracker[1] is not None else f_interp
        if len(cur_alt) == n_alt:
            a_opt, _ = coordinate_descent_v2(cur_alt, time_limit=t_cd, n_sweeps_max=100)
            try_sequence(a_opt, best_tracker)

    # === Final CD polish on best ===
    remaining = deadline - time.monotonic() - 10
    if remaining > 20 and best_tracker[1] is not None:
        print(f"\n=== Final Polish ({remaining:.0f}s) ===")
        a_opt, _ = coordinate_descent_v2(best_tracker[1], time_limit=remaining,
                                         n_sweeps_max=200)
        try_sequence(a_opt, best_tracker)

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
    else:
        final_seq = np.ones(448) / 448
        final_score = float(evaluate_sequence(final_seq.tolist()))
        print(f"\nFINAL (fallback): score={final_score:.10f}")

    return [float(x) for x in final_seq]
