"""

import numpy as np
import time
import math
from scipy.optimize import linprog


def fft_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_lse_gradient(a, conv, tau):
    """
    Compute gradient of tau*log(sum(exp(conv[k]/tau))) w.r.t. a.
    Gradient[m] = a[m] + sum_k w[k]*a[k-m] where w = softmax(conv/tau).
    Uses FFT for the cross-correlation term.
    """
    n = len(a)
    conv_scaled = conv / tau
    conv_scaled -= conv_scaled.max()
    exp_vals = np.exp(conv_scaled)
    w = exp_vals / exp_vals.sum()

    # cross[m] = sum_k w[k] * a[k-m] = (w * a_rev)[m + n - 1]
    a_rev = a[::-1]
    conv_len = (2 * n - 1) + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    AR = np.fft.rfft(a_rev, fft_size)
    conv_full = np.fft.irfft(W * AR, fft_size)
    cross = conv_full[n - 1: 2 * n - 1]

    grad = a + cross
    return grad


def lse_objective(conv, tau):
    """Compute tau * log(sum(exp(conv/tau)))."""
    c = conv / tau
    c -= c.max()
    return tau * math.log(np.exp(c).sum() + 1e-300)


def nesterov_lse_optimize(a0, time_limit, tau_init=0.01, tau_min=0.001,
                          lr_init=0.05, momentum_init=0.9, seed=0):
    """
    Nesterov-accelerated gradient descent on the LSE-smoothed objective.
    Returns (best_a, best_score) where best_score = 2*n*max(conv)/(sum)^2.
    """
    rng = np.random.default_rng(seed)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    y = a.copy()
    tau = tau_init
    lr = lr_init
    mom = momentum_init

    conv_y = fft_convolve(y)
    best_obj = 2.0 * n * conv_y.max()
    best_a = a.copy()
    no_improve = 0
    t0 = time.monotonic()
    iter_count = 0

    while time.monotonic() - t0 < time_limit:
        # Nesterov lookahead
        y = a + mom * v
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y

        conv_y = fft_convolve(y)
        grad = compute_lse_gradient(y, conv_y, tau)

        # Gradient step
        v = mom * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()
            v = np.zeros(n)
            continue

        conv = fft_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-10:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive updates

            else:
                a = best_a.copy()
            v = np.zeros(n)
            no_improve = 0
            if no_improve > 500:
                break

    return best_a, best_obj


def newton_refine(a0, time_limit, max_iters=200):
    """
    Damped Newton refinement using conjugate gradient for Hessian solve.
    H = I + sum_k w_k * M_k (Hankel structure).
    Hessian-vector product: H @ x = x + cross_corr(w, x) (same structure as gradient shift).
    """
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a0, 2.0 * len(a0) * fft_convolve(a0).max()
    a = a / s
    n = len(a)

    tau = 0.005
    best_obj = 2.0 * n * fft_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fft_convolve(a)
        grad = compute_lse_gradient(a, conv, tau)

        # CG to solve (I + K) x = -grad where K is the Hankel part
        # K @ x = cross_corr(w, x)
        # Full H @ x = x + K @ x

        conv_scaled = conv / tau
        conv_scaled -= conv_scaled.max()
        w = np.exp(conv_scaled)
        w /= w.sum()

        a_rev = a[::-1]  # not needed for H, but for K we need w
        # K @ x = (w * x_rev)[n-1 : 2n-1] where x_rev = x[::-1]
        conv_len = (2 * n - 1) + n - 1
        fft_size = 1 << (conv_len - 1).bit_length()
        W = np.fft.rfft(w, fft_size)

        def hvp(x):
            """H @ x = x + K @ x"""
            x_rev = x[::-1]
            AR = np.fft.rfft(x_rev, fft_size)
            conv_full = np.fft.irfft(W * AR, fft_size)
            Kx = conv_full[n - 1: 2 * n - 1]
            return x + Kx

        # Conjugate gradient to solve H p = -grad
        p = -grad.copy()
        r = -grad - hvp(np.zeros(n))  # r = -grad - H@0 = -grad
        # Actually r = b - H*p where b = -grad, p=0: r = -grad
        r = -grad.copy()
        rz = r @ r
        d = r.copy()

        cg_iter = 0
        max_cg = min(30, n)
        for _ in range(max_cg):
            if time.monotonic() - t0 > time_limit:
                break
            Hd = hvp(d)
            dHd = d @ Hd
            if dHd < 1e-15:
                break
            alpha = rz / dHd
            p += alpha * d
            r -= alpha * Hd
            rz_new = r @ r
            if rz_new < 1e-12 * rz:
                break
            d = r + (rz_new / rz) * d
            rz = rz_new
            cg_iter += 1

        # Line search along p
        best_alpha = 0.0
        best_trial = best_obj
        a_trial = a.copy()
        for alpha in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
            a_try = a + alpha * p
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            conv_try = fft_convolve(a_try)
            obj_try = 2.0 * n * conv_try.max()
            if obj_try < best_trial:
                best_trial = obj_try
                best_alpha = alpha
                a_trial = a_try.copy()

        if best_alpha > 0:
            a = a_trial
            if best_obj < best_obj:
                pass
            if best_trial < best_obj - 1e-12:
                best_obj = best_trial
                best_a = a.copy()
            tau = max(0.001, tau * 0.95)
        else:
            tau = max(0.001, tau * 0.9)
            if it > 50:
                break

    return best_a, best_obj


def lp_step(a, M_multiplier=1.0001):
    """Single LP-based step: find direction g that maximizes sum(g) s.t. conv(a,g) <= M*M_mult."""
    n = len(a)
    conv = fft_convolve(a)
    M = conv.max() * M_multiplier

    # Build convolution matrix C where (C @ g)[k] = sum_j C[k,j]*g[j] = conv(a,g)[k]
    # C[k,j] = a[k-j] if valid
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = max(0, j - n + 1)
        hi = min(2 * n - 2, j + n - 1)
        C[lo:hi + 1, j] = a[lo - j: n] if lo < j else a[:hi - j + 1]

    # Actually, let me build it more carefully
    # conv(a,g)[k] = sum_j a[j] * g[k-j] (with zero padding)
    # So C[k,j] = a[k-j] if 0 <= k-j < n, else 0
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        for k in range(max(0, j), min(2 * n - 1, j + n)):
            C[k, j] = a[k - j]

    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
        if result.success and result.x is not None:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g
    except Exception:
        pass
    return None


def lp_optimize(a0, time_limit, max_lp_iters=50):
    """LP-based iterative optimization."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a0, float('inf')
    a = a / s
    n = len(a)

    conv = fft_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0

    for it in range(max_lp_iters):
        if time.monotonic() - t0 > time_limit:
            break

        g = lp_step(a)
        if g is None:
            no_improve += 1
            if no_improve > 10:
                break
            continue

        # Line search: a + t*(g-a)
        d = g - a
        best_t = 0.0
        best_trial = best_obj
        a_trial = a.copy()

        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.01]:
            a_try = a + t * d
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            conv_try = fft_convolve(a_try)
            obj_try = 2.0 * n * conv_try.max()
            if obj_try < best_trial:
                best_trial = obj_try
                best_t = t
                a_trial = a_try.copy()

        if best_t > 0:
            a = a_trial
            if best_trial < best_obj - 1e-12:
                best_obj = best_trial
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 15:
            break

    return best_a, best_obj


def coordinate_descent_polish(a0, time_limit, n_trials_per_batch=500):
    """Random coordinate descent: swap small amounts between pairs."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    n = len(a)

    best_obj = 2.0 * n * fft_convolve(a).max()
    best_a = a.copy()
    rng = np.random.default_rng(999)
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        for _ in range(n_trials_per_batch):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            delta = rng.choice([0.005, 0.01, 0.02, 0.05, -0.005, -0.01, -0.02, -0.05])
            if a[i] + delta < -1e-10 or a[j] - delta < -1e-10:
                continue

            # Quick check: only compute conv if it might help
            a[i] += delta
            a[j] -= delta
            conv = fft_convolve(a)
            obj = 2.0 * n * conv.max()
            if obj < best_obj - 1e-10:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
            else:
                a[i] -= delta
                a[j] += delta
                no_improve += 1

        if no_improve > 2000:
            break

    return best_a, best_obj


def spectral_init(n, rng, mode='flat_spectrum'):
    """Generate initialization based on spectral considerations."""
    if mode == 'flat_spectrum':
        # Try to create a sequence whose DFT has roughly flat magnitude (except DC)
        # Start with random phases, enforce non-negativity
        for attempt in range(50):
            phases = rng.uniform(0, 2 * np.pi, n)
            magnitudes = np.ones(n) / math.sqrt(n)
            magnitudes[0] = 1.0  # DC component
            A = magnitudes * np.exp(1j * phases)
            a = np.fft.irfft(A, n) if n <= 2 else np.fft.irfft(np.fft.rfftfreq(n) * 0 + A[:n // 2 + 1] * 2 - A[n // 2 + 1:] * 0, n)
            # Simpler: just take real part of IFFT
            a = np.fft.ifft(A).real
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 0.5 * n / n:  # enough non-negative mass
                a = a / s
                return a
        return None
    elif mode == 'barker_like':
        # Based on Barker code structure (low autocorrelation)
        # For even n, use a pseudo-random +/-1 pattern, then take absolute value
        Barker_lengths = [4, 5, 7, 11, 13]
        # Just use random signs
        signs = rng.choice([-1, 1], n)
        a = np.abs(signs).astype(float)  # all 1s, not useful
        # Better: use a modulated pattern
        freq = rng.uniform(0.1, 0.4)
        x = np.arange(n)
        a = (1 + np.cos(2 * np.pi * freq * x)) / 2
        a = np.maximum(a, 0.01)
        a = a / a.sum()
        return a
    elif mode == 'gaussian_ring':
        # Ring-shaped: mass away from center
        x = np.arange(n) - n / 2
        sigma = n * 0.25
        a = np.exp(-((np.abs(x) - n * 0.3) ** 2) / (2 * sigma ** 2))
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
        return a
    return np.ones(n) / n


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # Sine-based
    for freq in [1, 2, 3]:
        key = f'sine{freq}'
        a = (np.sin(np.pi * freq * x / n) ** 2 + 0.05)
        inits[key] = a / a.sum()

    # Bimodal
    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    # Smooth edges
    for i in range(min(5, edge)):
        bi[i] *= i / float(edge)
        bi[n - 1 - i] *= i / float(edge)
    inits['bimodal'] = bi / bi.sum()

    # Gaussian
    g = np.exp(-0.5 * ((x - n / 2) / (n * 0.15)) ** 2)
    inits['gaussian'] = g / g.sum()

    # Power law
    pw = (x + 1.0) ** 0.5
    inits['power'] = pw / pw.sum()

    # Spectral inits
    sp = spectral_init(n, rng, 'flat_spectrum')
    if sp is not None:
        inits['spectral'] = sp

    gr = spectral_init(n, rng, 'gaussian_ring')
    if gr is not None:
        inits['ring'] = gr

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps

    return inits

            top_candidates.append((n, a_opt))

        if len(top_candidates) >= 4:
            break

    # Ensure best is included
    if best_sequence is not None:
        bn = len(best_sequence)
        if bn not in seen_lengths:
            top_candidates.insert(0, (bn, best_sequence.copy()))

    t_per_newton = max(20.0, phase2_budget / max(1, len(top_candidates)))

    for i, (n, a0) in enumerate(top_candidates):
        if time.monotonic() > deadline - 300:
            break
        t_alloc = min(t_per_newton, deadline - time.monotonic() - 200)
        if t_alloc < 10:
            break

        print(f"  Newton candidate {i}: n={n}, t_alloc={t_alloc:.0f}s")
        a_opt, obj = newton_refine(a0, time_limit=t_alloc, max_iters=300)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Newton {i} (n={n}): {score:.6f} *NEW BEST*")
            else:
                print(f"  Newton {i} (n={n}): {score:.6f}")

    print(f"Phase 2 done ({time.monotonic()-t_phase2_start:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 3: Additional Nesterov runs with different params
    # =========================================================
    print("=== Phase 3: Additional Gradient Runs ===")
    t_phase3_start = time.monotonic()
    if best_sequence is not None:
        n_best = len(best_sequence)
        remaining3 = min(300.0, deadline - time.monotonic() - 200)
        if remaining3 > 30:
            # Run gradient with different hyperparameters
            configs = [
                {'tau_init': 0.02, 'tau_min': 0.005, 'lr_init': 0.03, 'momentum_init': 0.92},
                {'tau_init': 0.008, 'tau_min': 0.001, 'lr_init': 0.06, 'momentum_init': 0.88},
                {'tau_init': 0.015, 'tau_min': 0.003, 'lr_init': 0.04, 'momentum_init': 0.93},
                {'tau_init': 0.005, 'tau_min': 0.0005, 'lr_init': 0.08, 'momentum_init': 0.85},
            ]
            t_per_run = remaining3 / len(configs)
            for ci, cfg in enumerate(configs):
                if time.monotonic() > deadline - 150:
                    break
                t_alloc = min(t_per_run, deadline - time.monotonic() - 100)
                if t_alloc < 10:
                    break

                # Slightly perturb the best
                scale = 0.005 * (ci + 1)
                noise = rng.standard_normal(n_best) * scale
                a0 = np.maximum(best_sequence + noise, 0)
                s = a0.sum()
                if s > 1e-15:
                    a0 = a0 / s
                else:
                    a0 = best_sequence.copy()

                a_opt, obj = nesterov_lse_optimize(a0, time_limit=t_alloc, seed=seed + 100 + ci, **cfg)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score):
                    if score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                        print(f"  GradCfg {ci}: {score:.6f} *NEW BEST*")
                    else:
                        print(f"  GradCfg {ci}: {score:.6f}")

    print(f"Phase 3 done ({time.monotonic()-t_phase3_start:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 4: LP-based refinement (limited)
    # =========================================================
    print("=== Phase 4: LP Refinement ===")
    t_phase4_start = time.monotonic()
    if best_sequence is not None:
        n_best = len(best_sequence)
        remaining4 = min(200.0, deadline - time.monotonic() - 100)
        if remaining4 > 30 and n_best <= 1024:  # LP is expensive for large n
            a_opt, obj = lp_optimize(best_sequence, time_limit=remaining4, max_lp_iters=30)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  LP: {score:.6f} *NEW BEST*")
                else:
                    print(f"  LP: {score:.6f}")
        elif remaining4 > 30:
            # For large n, do a few Nesterov runs instead
            for ri in range(2):
                if time.monotonic() > deadline - 80:
                    break
                t_alloc = min(60.0, deadline - time.monotonic() - 60)
                if t_alloc < 10:
                    break
                noise = rng.standard_normal(n_best) * 0.003
                a0 = np.maximum(best_sequence + noise, 0)
                s = a0.sum()
                if s > 1e-15:
                    a0 = a0 / s
                else:
                    a0 = best_sequence.copy()
                a_opt, obj = nesterov_lse_optimize(a0, time_limit=t_alloc,
                                                    tau_init=0.005, tau_min=0.001,
                                                    lr_init=0.05, momentum_init=0.91,
                                                    seed=seed + 200 + ri)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Large-N Grad {ri}: {score:.6f} *NEW BEST*")

    print(f"Phase 4 done ({time.monotonic()-t_phase4_start:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 5: Coordinate descent polish
    # =========================================================
    print("=== Phase 5: Coordinate Descent Polish ===")
    t_phase5_start = time.monotonic()
    if best_sequence is not None:
        remaining5 = min(80.0, deadline - time.monotonic() - 20)
        if remaining5 > 10:
            a_opt, obj = coordinate_descent_polish(best_sequence, time_limit=remaining5,
                                                    n_trials_per_batch=200)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  CoordDesc: {score:.6f} *NEW BEST*")
                else:
                    print(f"  CoordDesc: {score:.6f}")

    print(f"Phase 5 done ({time.monotonic()-t_phase5_start:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 6: Final micro-gradient polish
    # =========================================================
    if best_sequence is not None and time.monotonic() < deadline - 10:
        remaining = min(30.0, deadline - time.monotonic() - 5)
        if remaining > 5:
            a_opt, obj = nesterov_lse_optimize(
                best_sequence, time_limit=remaining,
                tau_init=0.003, tau_min=0.0005,
                lr_init=0.02, momentum_init=0.93,
                seed=seed + 999
            )
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final polish: {score:.6f} *NEW BEST*")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
