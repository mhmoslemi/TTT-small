def fast_convolve(a):

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u * np.arange(1, n + 1)) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

            obj = fast_convolve(ini).max()
            if obj < best_local_obj:
                best_local_obj = obj
                best_local = ini.copy()
        if best_local is not None:
            # Quick FW (10s budget)
            f_opt, obj_opt = fw_optimize(best_local, time_limit=min(8.0, (deadline - 400 - time.monotonic()) * 0.1),
                                         top_k=15, temp=0.005, use_paired=False, verbose=False)
            score = 2.0 * n_try * obj_opt / (f_opt.sum() ** 2)
            quick_results.append((score, n_try, f_opt))
            print(f"  n={n_try}: quick score={score:.8f}")

    # Sort and keep best
    quick_results.sort(key=lambda x: x[0])
    if quick_results:
        for score_q, n_q, f_q in quick_results[:3]:
            try_seq(f_q, f"phase1_n{n_q}")

    if best_seq is None:
        best_seq = np.ones(384) / 384
        try_seq(best_seq, "fallback")

    print(f"  Phase 1 best: {best_score:.8f} (n={len(best_seq)})")

    # === Phase 2: Main FW optimization ===
    print("\n=== Phase 2: Main FW Optimization ===")
    n_best = len(best_seq)
    remaining = deadline - time.monotonic() - 200

    # Standard FW (long run)
    t_fw1 = min(remaining * 0.4, 500)
    if t_fw1 > 20:
        print(f"  Standard FW (n={n_best}), budget={t_fw1:.0f}s, top_k=20, temp=0.002")
        f_opt, obj_opt = fw_optimize(best_seq, time_limit=t_fw1, top_k=20, temp=0.002,
                                     use_paired=False, verbose=True)
        try_seq(f_opt, "fw_standard")

    # Paired FW
    remaining = deadline - time.monotonic() - 150
    t_fw2 = min(remaining * 0.3, 300)
    if t_fw2 > 20:
        print(f"  Paired FW (n={n_best}), budget={t_fw2:.0f}s")
        f_opt, obj_opt = fw_optimize(best_seq, time_limit=t_fw2, top_k=25, temp=0.001,
                                     use_paired=True, verbose=True)
        try_seq(f_opt, "fw_paired")

    # FW with different parameters on best
    remaining = deadline - time.monotonic() - 120
    t_fw3 = min(remaining * 0.3, 250)
    if t_fw3 > 20:
        print(f"  FW fine-tune (n={n_best}), budget={t_fw3:.0f}s, top_k=40, temp=0.0005")
        f_opt, obj_opt = fw_optimize(best_seq, time_limit=t_fw3, top_k=40, temp=0.0005,
                                     use_paired=False, verbose=True)
        try_seq(f_opt, "fw_fine")

    print(f"  Phase 2 best: {best_score:.8f} (n={len(best_seq)})")

    # === Phase 3: Local search polishing ===
    print("\n=== Phase 3: Local Search Polish ===")
    remaining = deadline - time.monotonic() - 80
    t_ls = min(remaining * 0.4, 150)
    if t_ls > 10:
        print(f"  Local search, budget={t_ls:.0f}s")
        f_opt, obj_opt = local_search_polish(best_seq, time_limit=t_ls, rng=rng)
        try_seq(f_opt, "local_search")

    # === Phase 4: FW from perturbed best ===
    remaining = deadline - time.monotonic() - 60
    n_perturb = 3
    if remaining > 30:
        t_each = min(remaining / n_perturb, 50)
        for p in range(n_perturb):
            if time.monotonic() > deadline - 30:
                break
            scale = 0.003 * (p + 1)
            f_pert = project_simplex(best_seq + rng.standard_normal(len(best_seq)) * scale)
            f_opt, obj_opt = fw_optimize(f_pert, time_limit=t_each, top_k=20, temp=0.001,
                                         use_paired=False, verbose=False)
            try_seq(f_opt, f"perturb_{p}")

    # === Phase 5: Try nearby lengths ===
    remaining = deadline - time.monotonic() - 25
    n_best = len(best_seq)
    if remaining > 30:
        for n_try in [n_best - 16, n_best + 16, n_best - 32, n_best + 32]:
            if n_try < 64 or time.monotonic() > deadline - 10:
                continue
            px = np.interp(np.linspace(0, 1, n_try), np.linspace(0, 1, n_best), best_seq)
            px = np.maximum(px, 0)
            ps = px.sum()
            if ps > 1e-10:
                px = px / ps
                t_try = min((deadline - time.monotonic() - 10) * 0.4, 30)
                if t_try > 5:
                    f_opt, obj_opt = fw_optimize(px, time_limit=t_try, top_k=20, temp=0.001,
                                                 use_paired=False, verbose=False)
                    try_seq(f_opt, f"len{n_try}")

    # Final verification
    if best_seq is not None:
        final_score = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.10f}")

    return [float(x) for x in best_seq]
