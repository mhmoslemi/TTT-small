"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2:
- Fast subgradient descent with momentum (no LP needed)
- Periodic LP-FW for structured improvements
- Multi-length and multi-start exploration
- Spectral initialization (Hann-like windows)
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def subgradient_step(a, lr, momentum, v):
    """One subgradient step with momentum."""
    c = fast_convolve(a)
    k_star = np.argmax(c)
    n = len(a)
    # Subgradient: sg[i] = 2*a[k_star - i] if 0 <= k_star - i < n
    sg = np.zeros(n)
    for i in range(n):
        j = k_star - i
        if 0 <= j < n:
            sg[i] = 2.0 * a[j]
    v = momentum * v - lr * sg
    a_new = project_simplex(a + v)
    return a_new, v

def sg_optimize(a0, time_limit, lr_init=0.01, momentum=0.9, decay=0.9999):
    """Subgradient descent with momentum."""
    a = project_simplex(a0.copy())
    n = len(a)
    v = np.zeros(n)
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Batch of steps
        for _ in range(100):
            if time.monotonic() - t0 > time_limit:
                break
            a_new, v = subgradient_step(a, lr, momentum, v)
            c = fast_convolve(a_new)
            obj = c.max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1
            a = a_new
            lr *= decay
        
        if no_improve > 500:
            # Restart
            noise = np.random.default_rng(no_improve).standard_normal(n) * 0.01
            a = project_simplex(best_a + noise)
            v = np.zeros(n)
            lr = lr_init
            no_improve = 0
    
    return best_a, best_obj

"""
Hybrid optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses constraint-generation LP-FW, subgradient pre-optimization,
multi-length search, and spectral initialization for improved performance.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_constraint_matrix_banded(n, k_list, f):
    """Build the A_ub matrix for selected constraints using banded structure."""
    K = len(k_list)
    A = np.zeros((K, n))
    for idx, k in enumerate(k_list):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A[idx, j] = f[k - j]
    return A


def solve_lp_direction(f, M, n, active_lags=None, max_constraints=200):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for selected lags, g >= 0."""
    conv = fast_convolve(f)
    if active_lags is None:
        # Use constraint generation: top constraints
        max_constraints = min(max_constraints, 2 * n - 1)
        active_lags = np.argsort(conv)[::-1][:max_constraints]
    
    A_ub = build_constraint_matrix_banded(n, active_lags, f)
    b_ub = np.full(len(active_lags), M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        M = conv.max()
        
        # Select active constraints: lags where conv is close to peak
        threshold = 0.5 * M
        active_mask = conv >= threshold
        active_lags = np.where(active_mask)[0]
        if len(active_lags) > max_constraints:
            active_lags = conv.argsort()[::-1][:max_constraints]
            active_lags = np.sort(active_lags)
        
        g = solve_lp_direction(f, M * m_exp, n, active_lags=active_lags,
                               max_constraints=max_constraints)

            m_exp *= 1.15
            if m_exp > 5.0:

                # Try with more relaxed threshold

                if g is None:

            continue

        g_sum = g.sum()
        if g_sum < 1e-15:

                no_improve = 0
                m_exp = 1.0
                continue
            m_exp = min(m_exp * 1.06, 3.0)
            continue

        # Line search with golden section
        best_alpha = 0.0
        best_val = M

        # Coarse grid
        for alpha in [0.03, 0.07, 0.12, 0.2, 0.3, 0.45, 0.6, 0.75, 0.9, 1.0]:

            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:

            if val < best_val:

                best_alpha = alpha

        # Fine bisection around best

            lo = max(0.0, best_alpha - 0.08)
            hi = min(1.0, best_alpha + 0.08)
            for _ in range(12):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv_new = fast_convolve(f_new)
            M_new = conv_new.max()
            f = f_new
            m_exp = 1.0

                best_f = f.copy()
                no_improve = 0
                if verbose and it % 30 == 0:

            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.08, 4.0)

    return best_f, best_obj


def sg_pre_optimize(f0, time_limit, lr_init=0.02, momentum=0.85, decay=0.9998):
    """Fast subgradient pre-optimization with momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if it % 1000 == 0 and time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        k_star = np.argmax(conv)

        # Subgradient: sg[i] = 2*f[k_star - i] if valid
        sg = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        for j in range(js, je + 1):
            sg[j] = 2.0 * f[k_star - j]

        v = momentum * v - lr * sg
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        obj = conv_new.max()

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new
        lr *= decay

        if no_improve > 800:
            rng = np.random.default_rng(it * 7 + 1)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            v = np.zeros(n)
            lr = lr_init * 0.7
            no_improve = 0

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with local best-step per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-12:
                continue

            best_j_val = old_val
            best_j_obj = fast_convolve(f).max()

            multipliers = [0.3, 0.5, 0.7, 1.3, 1.6, 2.2]
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved and passes > 4:
            break

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def hann_window(n):
    """Hann window of length n, normalized to sum 1."""
    w = 0.5 * (1 + np.cos(2 * np.pi * np.arange(n) / (n - 1))) if n > 1 else np.ones(1)
    w = np.maximum(w, 0)
    s = w.sum()
    if s > 0:
        w /= s
    return w


def blackman_window(n):
    """Blackman window of length n, normalized to sum 1."""
    if n == 1:
        return np.ones(1)
    w = np.zeros(n)
    for i in range(n):
        w[i] = 0.42 - 0.5 * np.cos(2 * np.pi * i / (n - 1)) + 0.08 * np.cos(4 * np.pi * i / (n - 1))
    w = np.maximum(w, 0)
    s = w.sum()
    if s > 0:
        w /= s
    return w


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase hybrid optimization:
    1. Subgradient pre-optimization (fast, many iterations)
    2. LP-FW with constraint generation (structured improvements)
    3. Multi-length exploration (interpolation + re-optimization)
    4. Coordinate descent polishing
    5. Final perturb-and-refine
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        a = np.minimum(a, 1000.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
    except Exception:
        pass

    if parent is None:
        parent = hann_window(768)
        print("No parent, using Hann window n=768")

    # === Phase 1: Subgradient pre-optimization on parent ===
    print("\n=== Phase 1: SG Pre-optimization ===")
    t1 = deadline - time.monotonic()
    t_alloc = min(45.0, t1 * 0.05)
    if t_alloc > 5:
        a_sg, _ = sg_pre_optimize(parent, time_limit=t_alloc, lr_init=0.015,
                                  momentum=0.88, decay=0.9999)
        try_seq(a_sg, "sg_parent")

    base = best_seq if best_seq is not None else parent

    # === Phase 2: LP-FW with constraint generation on parent length ===
    print(f"\n=== Phase 2: LP-FW n={len(base)} (score={best_score:.8f}) ===")
    t2 = deadline - time.monotonic()
    t_alloc = min(500.0, t2 * 0.35)
    if t_alloc > 30:
        a_opt, _ = lp_fw_optimize(base, time_limit=t_alloc, verbose=True,
                                  max_constraints=120)
        try_seq(a_opt, "lpfw_parent")

    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-length (best={best_score:.8f}) ===")
    base = best_seq if best_seq is not None else parent
    base_n = len(base)

    target_lengths = [512, 1024, 1536]
    # Filter to avoid too similar to base
    target_lengths = [tl for tl in target_lengths if abs(tl - base_n) > 50]
    # Add a length close to base but different
    if base_n not in target_lengths:
        target_lengths.append(base_n + base_n // 8)
        target_lengths = [tl for tl in target_lengths if tl != base_n]

    for target_n in target_lengths[:3]:
        rem = deadline - time.monotonic() - 400
        if rem < 50:
            break

        # Initialize with interpolation + window blend
        f_interp = interpolate_sequence(base, target_n)
        w_hann = hann_window(target_n)
        f_init = 0.7 * f_interp + 0.3 * w_hann
        f_init = project_simplex(f_init)

        t_alloc = min(rem * 0.45, 200)
        print(f"  LP-FW n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = lp_fw_optimize(f_init, time_limit=t_alloc, verbose=False,
                                  max_constraints=100)
        try_seq(a_opt, f"lpfw_n{target_n}")

    # === Phase 4: Deep LP-FW on best so far ===
    print(f"\n=== Phase 4: Deep LP-FW (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        
        for p in range(5):
            rem = deadline - time.monotonic() - 150
            if rem < 30:
                break
            t_alloc = min(rem / max(1, 5 - p), 180)

            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                noise = rng.standard_normal(n_best) * 0.005
                f_init = project_simplex(best_seq + noise)
            elif p == 2:
                # Blend with reversed
                f_init = 0.55 * best_seq + 0.45 * best_seq[::-1]
                f_init = project_simplex(f_init)
            elif p == 3:
                # Blend with Hann
                w_h = hann_window(n_best)
                f_init = 0.8 * best_seq + 0.2 * w_h
                f_init = project_simplex(f_init)
            else:
                # Blend with Blackman
                w_b = blackman_window(n_best)
                f_init = 0.75 * best_seq + 0.25 * w_b
                f_init = project_simplex(f_init)

            a_opt, _ = lp_fw_optimize(f_init, time_limit=t_alloc,
                                      verbose=(p == 0), max_constraints=130)
            try_seq(a_opt, f"deep_{p}")

    # === Phase 5: Coordinate descent polishing ===
    print(f"\n=== Phase 5: Coordinate Descent (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 30:
            t_cd = min(rem * 0.6, 200)
            print(f"  CD t={t_cd:.0f}s")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd")

    # === Phase 6: Final refinement ===
    print(f"\n=== Phase 6: Final Refinement (best={best_score:.8f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Another LP-FW pass with tighter constraints
        rem = deadline - time.monotonic() - 40
        if rem > 20:
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=min(rem, 100),
                                      max_constraints=80)
            try_seq(a_opt, "final_lpfw")

        # SG polish
        rem = deadline - time.monotonic() - 20
        if rem > 15:
            a_opt, _ = sg_pre_optimize(best_seq, time_limit=min(rem, 60),
                                       lr_init=0.005, momentum=0.92, decay=0.99995)
            try_seq(a_opt, "sg_final")

        # Tiny perturbations + quick LP-FW
        for _ in range(4):
            rem = deadline - time.monotonic() - 10
            if rem < 10:
                break
            noise = rng.standard_normal(n_best) * 0.002
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = lp_fw_optimize(f_pert, time_limit=min(rem, 30),
                                      max_constraints=60)
            try_seq(a_opt, f"perturb_{_}")

        # One final CD pass
        rem = deadline - time.monotonic() - 5
        if rem > 15:
            a_opt, _ = coordinate_descent(best_seq, time_limit=min(rem, 40))
            try_seq(a_opt, "cd_final")

    # Fallback
    if best_seq is None:
        best_seq = hann_window(768)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
