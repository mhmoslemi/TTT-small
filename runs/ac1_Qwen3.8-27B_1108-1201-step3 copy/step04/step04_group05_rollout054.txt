Looking at this problem, I need to minimize `2*n*max(conv(a,a))/(sum(a))^2`. The key insight is that `np.convolve(a,a)` is an auto-convolution (not auto-correlation), and we want it as flat as possible. Since `sum_k conv(a,a)[k] = (sum(a))^2`, the average convolution value is fixed, so we want to minimize the peak-to-average ratio.

I'll use a fundamentally different approach from the previous LP-based method:

2. **Two update schemes**: multiplicative (mirror descent) and additive (momentum + simplex projection)

4. **Diverse initializations** including U-shaped, edge-concentrated, and parent-based

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smoothed gradient with FFT-based gradient computation
2. Multiplicative (mirror descent) and additive (Nesterov) update schemes
3. Adaptive temperature annealing
4. Diverse initializations, restarts, and local search polishing
"""
import numpy as np
import time


def fast_convolve(a):
    """Compute full auto-convolution using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(A * A, fft_size)
    return conv[:size]


def project_simplex(v):
    """Euclidean projection onto probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_grad_fft(p, a, fft_size_grad):
    """
    Compute gradient of (1/t)*log(sum_k exp(t*c[k])) w.r.t. a using FFT.
    grad[j] = 2 * sum_{m=0}^{n-1} p[j+m] * a[m]
    Computed as: 2 * (p * a_rev)[n-1 : 2n-1] via FFT convolution.
    """
    n = len(a)
    a_rev = a[::-1]
    P = np.fft.rfft(p, fft_size_grad)
    Ar = np.fft.rfft(a_rev, fft_size_grad)
    conv = np.fft.irfft(P * Ar, fft_size_grad)
    return 2.0 * conv[n - 1:2 * n - 1]


def get_softmax_weights(c, t):
    """Compute softmax weights exp(t*c)/sum(exp(t*c))."""
    tc = t * c
    tc -= tc.max()
    e = np.exp(tc)
    s = e.sum()
    if s < 1e-300:
        # Fallback: uniform over top elements
        n = len(c)
        k = min(10, n)
        idx = np.argpartition(c, -k)[-k:]
        w = np.zeros(n)
        w[idx] = 1.0 / k
        return w
    return e / s


def optimize_multiplicative(a0, time_limit, t_start=5.0, t_end=500.0,
                            lr_init=0.08, total_iters=30000, seed=0):
    """
    Mirror descent (multiplicative update) with log-sum-exp smoothed objective.
    Update: a[j] *= exp(-lr * grad[j]), then normalize.
    Temperature increases from t_start to t_end (annealing from smooth to peaked).
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    grad_conv_len = (2 * n - 1) + n - 1
    fft_size_grad = 1 << (grad_conv_len - 1).bit_length()

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    t0_time = time.monotonic()
    it = 0
    no_improve = 0
    cur_lr = lr_init

    while time.monotonic() - t0_time < time_limit:
        it += 1

        # Temperature: quadratic ramp up
        progress = min(1.0, it / float(total_iters))
        t = t_start + (t_end - t_start) * progress * progress

        c = fast_convolve(a)
        p = get_softmax_weights(c, t)
        grad = compute_smoothed_grad_fft(p, a, fft_size_grad)

        # Multiplicative update
        # Clip gradient to avoid overflow
        grad_clipped = np.clip(grad, -50.0, 50.0)
        update = np.exp(-cur_lr * grad_clipped)
        a = a * update
        s = a.sum()
        if s < 1e-15 or not np.isfinite(s):
            a = best_a.copy()
            cur_lr *= 0.5
            no_improve += 10
            continue
        a = a / s
        a = np.maximum(a, 0.0)
        s2 = a.sum()
        if s2 < 1e-15:
            a = best_a.copy()
            cur_lr *= 0.5
            continue
        a = a / s2

        # Check objective periodically
        if it % 3 == 0:
            c = fast_convolve(a)
            obj = c.max()
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

        # Adaptive learning rate
        if no_improve > 150:
            cur_lr *= 0.85
            if cur_lr < 0.0005:
                break
        elif no_improve == 0 and cur_lr < lr_init:
            cur_lr = min(cur_lr * 1.02, lr_init)

    return best_a, best_obj


def optimize_momentum(a0, time_limit, t_start=5.0, t_end=500.0,
                      lr_init=0.03, momentum=0.85, total_iters=30000, seed=0):
    """
    Heavy-ball momentum gradient descent with simplex projection.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    grad_conv_len = (2 * n - 1) + n - 1
    fft_size_grad = 1 << (grad_conv_len - 1).bit_length()

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    v = np.zeros(n)
    t0_time = time.monotonic()
    it = 0
    no_improve = 0
    cur_lr = lr_init
    cur_mom = momentum

    while time.monotonic() - t0_time < time_limit:
        it += 1

        progress = min(1.0, it / float(total_iters))
        t = t_start + (t_end - t_start) * progress * progress

        c = fast_convolve(a)
        p = get_softmax_weights(c, t)
        grad = compute_smoothed_grad_fft(p, a, fft_size_grad)

        # Momentum update
        v = cur_mom * v + cur_lr * grad
        a_new = a - v
        a = project_simplex(a_new)

        if it % 3 == 0:
            c = fast_convolve(a)
            obj = c.max()
            if obj < best_obj - 1e-13:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

        # Adaptive
        if no_improve > 100:
            cur_lr *= 0.9
            cur_mom = min(cur_mom * 1.01, 0.95)
            if cur_lr < 0.0003:
                break
        elif no_improve == 0:
            cur_lr = min(cur_lr * 1.01, lr_init)

    return best_a, best_obj


def local_search_polish(a0, time_limit, seed=42):
    """
    Local search: try random perturbations of varying scale, keep best.
    Uses coordinate flips and small Gaussian noise.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    rng = np.random.default_rng(seed)
    t0_time = time.monotonic()
    scale = 0.003
    it = 0

    while time.monotonic() - t0_time < time_limit:
        it += 1

        # Try several perturbations
        for _ in range(20):
            if time.monotonic() - t0_time > time_limit:
                break

            # Mix of small Gaussian and coordinate-level changes
            r = rng.random()
            if r < 0.6:
                # Small Gaussian perturbation
                noise = rng.standard_normal(n) * scale
                a_try = project_simplex(best_a + noise)
            elif r < 0.8:
                # Coordinate pair swap
                i, j = rng.integers(0, n, size=2)
                if i == j:
                    continue
                delta = (best_a[i] - best_a[j]) * rng.uniform(0.1, 0.5)
                a_try = best_a.copy()
                a_try[i] = max(0.0, a_try[i] - delta)
                a_try[j] = max(0.0, a_try[j] + delta)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try
            else:
                # Block perturbation
                block_len = rng.integers(5, max(6, n // 10))
                start = rng.integers(0, n - block_len)
                noise = rng.standard_normal(block_len) * scale * 2
                a_try = best_a.copy()
                a_try[start:start + block_len] = np.maximum(0, a_try[start:start + block_len] + noise)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try = a_try / s_try

            c_try = fast_convolve(a_try)
            obj_try = c_try.max()
            if obj_try < best_obj - 1e-14:
                best_obj = obj_try
                best_a = a_try.copy()
                scale = min(scale * 1.1, 0.01)

        if it % 50 == 0:
            scale *= 0.95
            if scale < 0.0005:
                scale = 0.003  # Reset

    return best_a, best_obj


def annealed_local_search(a0, time_limit, seed=99):
    """
    Coordinate descent style: pick the coordinate most responsible for the peak,
    and try to reduce it while maintaining sum=1.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    rng = np.random.default_rng(seed)
    t0_time = time.monotonic()

    while time.monotonic() - t0_time < time_limit:
        c = fast_convolve(a)
        obj = c.max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()

        # Find peak location
        peak_k = int(np.argmax(c))

        # The gradient contribution to the peak comes from a[k-j] for valid j
        # Reduce coordinates that contribute most to the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        contributors = a[js:je + 1]
        # The contribution to c[peak_k] from a[j] is a[peak_k-j]
        contrib_weights = a[max(0, peak_k - je):min(n, peak_k - js + 1)]
        # Actually: c[peak_k] = sum_{j=js}^{je} a[j] * a[peak_k - j]
        # To reduce c[peak_k], we should decrease a[j] where a[peak_k-j] is large
        a_at_peak_k_minus_j = a[max(0, peak_k - je):min(n, peak_k - js + 1)][::-1]

        # Find the j that contributes most
        contribs = contributors * a_at_peak_k_minus_j
        if contribs.max() < 1e-15:
            # Random perturbation instead
            noise = rng.standard_normal(n) * 0.002
            a = project_simplex(a + noise)
            continue

        max_idx = js + int(np.argmax(contribs))

        # Try reducing a[max_idx] and distributing to others
        reduce_frac = rng.uniform(0.05, 0.2)
        amount = a[max_idx] * reduce_frac
        if amount < 1e-15:
            continue

        a_try = a.copy()
        a_try[max_idx] -= amount
        # Distribute to random other coordinates weighted by current value
        others = np.arange(n)
        others = others[others != max_idx]
        distrib = a_try[others].copy()
        distrib = np.maximum(distrib, 0)
        distrib_sum = distrib.sum()
        if distrib_sum < 1e-15:
            distrib = np.ones(len(others))
            distrib_sum = distrib.sum()
        distrib = distrib / distrib_sum
        a_try[others] += amount * distrib
        a_try = np.maximum(a_try, 0)
        s_try = a_try.sum()
        if s_try < 1e-10:
            continue
        a_try = a_try / s_try

        c_try = fast_convolve(a_try)
        if c_try.max() < obj:
            a = a_try

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    x = np.linspace(0, 1, n)

    # Uniform
    inits.append(("uniform", np.ones(n) / n))

    # U-shaped (edge-concentrated)
    for beta in [2.0, 4.0, 8.0]:
        u = np.exp(-beta * np.sin(np.pi * x) ** 2)
        inits.append((f"u_{beta}", u / u.sum()))

    # Gaussian centered
    for sigma in [0.15, 0.25, 0.35]:
        g = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        inits.append((f"gauss_{sigma}", g / g.sum()))

    # Cosine
    c = np.abs(np.cos(np.pi * x)) + 0.01
    inits.append(("cosine", c / c.sum()))

    # Two peaks (anti-center)
    p2 = np.exp(-0.5 * ((x - 0.2) / 0.1) ** 2) + np.exp(-0.5 * ((x - 0.8) / 0.1) ** 2)
    inits.append(("two_peak", p2 / p2.sum()))

    # Power law (heavier edges)
    dist_from_center = np.abs(x - 0.5) * 2
    pw = dist_from_center ** 2 + 0.05
    inits.append(("power", pw / pw.sum()))

    # Random Dirichlet
    inits.append(("dirichlet", rng.dirichlet(np.ones(n) * 0.3)))

    # Interpolated parent
    if parent is not None:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(("parent_interp", px / ps))

    # Sine-squared modulated
    for freq in [2, 3]:
        sm = (1 + np.sin(2 * np.pi * freq * x)) ** 2 + 0.1
        inits.append((f"sine{freq}", sm / sm.sum()))

    return inits
