"""
Multi-phase optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def obj_normalized(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1. Returns (obj, conv)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv


def solve_lp(a, M_target):
    """Solve LP: maximize sum(g) s.t. conv(a,g) <= M_target, g >= 0.
    Returns (g_normalized, sum_g, dual_vars)."""
    n = len(a)
    C = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M_target)
    bounds = [(0.0, None)] * n
    result = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        g = result.x
        g_sum = g.sum()
        dual = None
        try:
            dual = result.ineqlin.marginals
        except Exception:
            pass
        if g_sum > 1e-12:
            return g / g_sum, g_sum, dual
    return None


def nesterov_sg_step(a, conv, lr, momentum, v, top_k, temp, gamma=0.9):
    """Nesterov-accelerated subgradient step."""
    n = len(a)
    top_k = min(top_k, len(conv))
    
    # Select top peaks with softmax weighting
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peaks]
    scaled = peak_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum

    # Compute subgradient
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        sg[idx] += w[i] * 2.0 * a[k - idx]

    # Nesterov: use velocity in gradient computation
    a_lookahead = a + momentum * v
    a_lookahead = np.maximum(a_lookahead, 0.0)
    s_la = a_lookahead.sum()
    if s_la > 1e-15:
        a_lookahead = a_lookahead / s_la
    else:
        a_lookahead = a

    # Recompute subgradient at lookahead point (optional, expensive)
    # For efficiency, use the original subgradient but with Nesterov update
    v_new = gamma * momentum * v - lr * sg
    a_new = a + v_new
    a_new = np.maximum(a_new, 0.0)
    s_new = a_new.sum()
    if s_new > 1e-15:
        a_new = a_new / s_new
    else:
        a_new = a
    return a_new, v_new

def sg_polish_v2(a0, time_limit, lr_init=0.005, momentum=0.95, top_k=80, temp=0.0001, nesterov_gamma=0.9):

        a = a / s
    n = len(a)

    best_obj = 2.0 * n * conv.max()

def coordinate_descent(a0, time_limit, ng_max=50):
    """Coordinate descent: try adjusting individual pairs of coordinates."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    a = a / s
    n = len(a)
    _, conv = obj_normalized(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        # Pick a random pair of indices
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # Try moving mass from i to j (and vice versa)
        for direction in [1, -1]:
            for delta in [0.001, 0.005, 0.01, 0.02, 0.0005, 0.002]:
                a_try = a.copy()
                a_try[i] -= direction * delta * a[i]
                a_try[j] += direction * delta * a[i]
                a_try = np.maximum(a_try, 0.0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj - 1e-15:
                    best_obj = obj_try
                    best_a = a_try.copy()
                    a = a_try.copy()
                    break

        if (int((time.monotonic() - t0) * 10)) % ng_max == 0:
            # Occasionally try a block update
            block_size = rng.integers(2, 8)
            start = rng.integers(0, max(1, n - block_size))
            block = a[start:start + block_size].copy()
            # Randomly redistribute the block
            rng_block = rng.dirichlet(np.ones(block_size) * 0.5)
            total = block.sum()
            a_try = a.copy()
            a_try[start:start + block_size] = rng_block * total
            s_try = a_try.sum()
            a_try = a_try / s_try
            conv_try = fast_convolve(a_try)
            obj_try = 2.0 * n * conv_try.max()
            if obj_try < best_obj - 1e-15:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()

    return best_a, best_obj


def resample_sequence(seq, new_n):
    """Resample a sequence to a different length via linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s > 1e-15:
        new_seq = new_seq / s
    return new_seq


def spectral_init(n, rng, mode='flat'):
    """Generate a spectrally-flat initialization."""
    # Start with random phases (for complex), take real part and rectify
    if mode == 'flat':
        # Try to create a sequence with flat power spectrum
        # Start with random, iterate: FFT -> flatten magnitude -> IFFT
        a = rng.standard_normal(n) + 1.0
        a = np.maximum(a, 0.0)
        for _ in range(50):
            A = np.fft.rfft(a)
            mag = np.abs(A)
            # Flatten: set magnitude to target
            target_mag = np.full_like(mag, 1.0 / np.sqrt(n))
            # Blend to avoid complete flattening
            mag_new = 0.5 * mag + 0.5 * target_mag * (mag.max() / (mag.mean() + 1e-15))
            A_new = mag_new * np.exp(1j * np.angle(A))
            a = np.fft.irfft(A_new, n)
            a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-15:
            a = a / s
        else:
            a = np.ones(n) / n
        return a
    elif mode == 'sine':
        x = np.arange(n, dtype=np.float64)
        freq = rng.uniform(0.5, 3.0)
        a = 0.6 + 0.4 * np.sin(2 * np.pi * freq * x / n + rng.uniform(0, 2 * np.pi))
        a = np.maximum(a, 0.0)
        return a / a.sum()
    elif mode == 'tlobe':
        x = np.arange(n, dtype=np.float64)
        a = 0.5 + 0.5 * np.cos(4 * np.pi * x / n)
        a = np.maximum(a, 0.0)
        return a / a.sum()
    else:
        return np.ones(n) / n

"""
Multi-phase optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Combines LP direction finding with dual-guided subgradient (Nesterov momentum),

Key improvements: aggressive LP tightness scheduling, Nesterov acceleration,

import numpy as np

from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def obj_normalized(a):
    """Compute 2n * max(conv(a,a)) assuming sum(a)=1. Returns (obj, conv)."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max(), conv


def solve_lp(a, M_target):
    """Solve LP: maximize sum(g) s.t. conv(a,g) <= M_target, g >= 0."""
    n = len(a)
    C = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M_target)
    bounds = [(0.0, None)] * n
    result = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
    if result.success:
        g = result.x
        g_sum = g.sum()
        dual = None
        try:
            dual = result.ineqlin.marginals
        except Exception:
            pass
        if g_sum > 1e-12:
            return g / g_sum, g_sum, dual
    return None


def nesterov_sg_step(a, conv, lr, momentum, v, top_k, temp):
    """Nesterov-accelerated subgradient step with softmax peak selection."""
    n = len(a)
    top_k = min(top_k, len(conv))
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peaks]
    scaled = peak_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        sg[idx] += w[i] * 2.0 * a[k - idx]

    # Nesterov lookahead for gradient
    a_la = a + momentum * v
    a_la = np.maximum(a_la, 0.0)
    s_la = a_la.sum()
    if s_la > 1e-15:
        a_la = a_la / s_la

    # Recompute subgradient at lookahead (expensive but more accurate)
    if s_la > 1e-15:
        conv_la = fast_convolve(a_la)
        peaks_la = np.argpartition(conv_la, -top_k)[-top_k:]
        pv_la = conv_la[peaks_la]
        sc_la = pv_la / temp
        sc_la -= sc_la.max()
        w_la = np.exp(sc_la)
        ws_la = w_la.sum()
        if ws_la < 1e-15:
            w_la = np.zeros(len(peaks_la))
            w_la[np.argmax(pv_la)] = 1.0
        else:
            w_la /= ws_la

        sg_la = np.zeros(n)
        for i in range(len(peaks_la)):
            if w_la[i] < 1e-12:

            k = int(peaks_la[i])

def coordinate_descent(a0, time_limit):
    """Coordinate descent with pair updates and block redistribution."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    a = a / s
    n = len(a)
    _, conv = obj_normalized(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        mode = it % 4

        if mode == 0 or mode == 1:
            # Pair update
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            for direction in [1, -1]:
                for delta in [0.002, 0.005, 0.01, 0.001, 0.008, 0.015]:
                    a_try = a.copy()
                    amount = direction * delta * a[i]
                    if a_try[i] + amount < 0:
                        continue
                    a_try[i] += amount
                    a_try[j] -= amount
                    a_try = np.maximum(a_try, 0.0)
                    s_try = a_try.sum()
                    if s_try < 1e-15:
                        continue
                    a_try = a_try / s_try
                    conv_try = fast_convolve(a_try)
                    obj_try = 2.0 * n * conv_try.max()
                    if obj_try < best_obj - 1e-15:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        a = a_try.copy()
                        break
        else:
            # Block update
            block_size = rng.integers(3, 10)
            start = rng.integers(0, max(1, n - block_size))
            block = a[start:start + block_size].copy()
            total = block.sum()
            if total < 1e-15:
                continue
            alpha = rng.uniform(0.1, 0.8)
            block_new = block * (1 - alpha) + total * rng.dirichlet(np.ones(block_size)) * alpha
            a_try = a.copy()
            a_try[start:start + block_size] = block_new
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            conv_try = fast_convolve(a_try)
            obj_try = 2.0 * n * conv_try.max()
            if obj_try < best_obj - 1e-15:
                best_obj = obj_try
                best_a = a_try.copy()
                a = a_try.copy()

    return best_a, best_obj


def resample_sequence(seq, new_n):
    """Resample a sequence to a different length via linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s > 1e-15:
        new_seq = new_seq / s
    return new_seq


def spectral_flat_init(n, rng, n_iters=100):
    """Generate a spectrally-flat non-negative sequence via iterative spectral flattening."""
    a = rng.standard_normal(n) + 2.0
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_peak = fast_convolve(a).max()

    for it in range(n_iters):
        A = np.fft.rfft(a)
        mag = np.abs(A)
        target = mag.mean()
        blend = 0.05 + 0.4 * (it / n_iters)
        mag_new = (1.0 - blend) * mag + blend * target
        A_new = mag_new * np.exp(1j * np.angle(A))
        a_new = np.fft.irfft(A_new, n)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        peak = fast_convolve(a_new).max()
        if peak < best_peak:
            best_peak = peak
            best_a = a_new.copy()

    return best_a


def diverse_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = {}
    if parent is not None:
        inits['parent'] = resample_sequence(parent, n)
    inits['uniform'] = np.ones(n) / n
    inits['flat'] = spectral_flat_init(n, rng)

    x = np.arange(n, dtype=np.float64)
    freq1 = rng.uniform(0.5, 3.0)
    sine = 0.6 + 0.4 * np.sin(2.0 * np.pi * freq1 * x / n + rng.uniform(0, 6.28))
    inits['sine'] = np.maximum(sine, 0.0)
    inits['sine'] /= inits['sine'].sum()

    tlobe = 0.5 + 0.5 * np.cos(4.0 * np.pi * x / n)
    inits['tlobe'] = np.maximum(tlobe, 0.0)
    inits['tlobe'] /= inits['tlobe'].sum()

    gauss = np.exp(-0.5 * ((x - 0.5 * n) / (0.25 * n)) ** 2)
    inits['gauss'] = gauss / gauss.sum()

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: LP scout -> multi-length exploration ->
    intensive SG+LP alternating -> coordinate descent -> final refinement.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(p)}, score={parent_score:.6f}")
            best_sequence = parent.copy()
            best_score = parent_score
    except Exception:
        pass

    if best_sequence is None:
        best_sequence = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_sequence.tolist()))
        print(f"Fallback uniform: score={best_score:.6f}")

    # ===== Phase 1: LP on parent =====
    print("=== Phase 1: LP on parent ===")
    t1 = time.monotonic()
    n_par = len(best_sequence)
    t_alloc = min(180.0, deadline - time.monotonic() - 600.0)
    a_opt, obj_opt = lp_optimize_v2(best_sequence, time_limit=max(10, t_alloc))
    score = float(evaluate_sequence(a_opt.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_sequence = a_opt.copy()
        print(f"  LP parent (n={n_par}): {score:.6f}")
    else:
        a_opt2, _ = sg_polish_v2(best_sequence, time_limit=max(5, t_alloc * 0.3),
                                  lr_init=0.004, momentum=0.95, top_k=80, temp=0.0001)
        score2 = float(evaluate_sequence(a_opt2.tolist()))
        if np.isfinite(score2) and score2 < best_score:
            best_score = score2
            best_sequence = a_opt2.copy()
            print(f"  SG parent: {score2:.6f}")
    print(f"Phase 1 done ({time.monotonic()-t1:.0f}s). Best: {best_score:.6f}")

    # ===== Phase 2: Multi-length exploration =====
    print("=== Phase 2: Multi-length exploration ===")
    t2 = time.monotonic()
    p2_deadline = deadline - 400.0

    n_par = len(best_sequence)
    lengths = sorted(set([
        n_par,
        max(384, n_par - 128),
        n_par + 128,
        896,
        1024,
        1280,
    ]))
    lengths = [l for l in lengths if 384 <= l <= 2048]
    lengths = sorted(set(lengths))

    for n in lengths:
        if time.monotonic() > p2_deadline - 20:
            break

        inits = diverse_inits(n, parent, rng)
        # Priority: parent > flat > uniform > others
        priority = ['parent', 'flat', 'uniform', 'sine', 'tlobe', 'gauss']
        init_names = [nm for nm in priority if nm in inits][:4]

        remaining_inits = max(1, len(lengths) - lengths.index(n))
        t_per = max(12.0, (p2_deadline - time.monotonic() - 10) / (remaining_inits * 3))

        for name in init_names:
            if time.monotonic() > p2_deadline - 8:
                break
            t_a = min(t_per, p2_deadline - time.monotonic() - 5)
            if t_a < 6:
                break

            a0 = inits[name]
            # For the parent length, do more aggressive LP
            if n == n_par:
                a_opt, _ = lp_optimize_v2(a0, time_limit=max(8, t_a))
            else:
                a_opt, _ = lp_optimize_v2(a0, time_limit=max(5, t_a * 0.7))

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  n={n}, {name}: {score:.6f}")

    print(f"Phase 2 done ({time.monotonic()-t2:.0f}s). Best: {best_score:.6f}")

    # ===== Phase 3: Intensive SG+LP alternating on best =====
    print("=== Phase 3: Intensive polishing ===")
    t3 = time.monotonic()
    p3_deadline = deadline - 120.0

    n_best = len(best_sequence)
    configs = [
        {'lr_init': 0.003, 'momentum': 0.95, 'top_k': 100, 'temp': 0.00006},
        {'lr_init': 0.005, 'momentum': 0.94, 'top_k': 60, 'temp': 0.00015},
        {'lr_init': 0.002, 'momentum': 0.96, 'top_k': 150, 'temp': 0.00004},
        {'lr_init': 0.004, 'momentum': 0.95, 'top_k': 80, 'temp': 0.0001},
    ]

    t_per = (p3_deadline - time.monotonic()) / len(configs)

    for ci, cfg in enumerate(configs):
        if time.monotonic() > p3_deadline - 10:
            break
        t_a = min(t_per, p3_deadline - time.monotonic() - 5)
        if t_a < 5:
            break

        # SG phase
        a_opt, _ = sg_polish_v2(best_sequence, time_limit=max(3, t_a * 0.5), **cfg)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  SG-{ci}: {score:.6f}")

        # LP phase after SG
        if time.monotonic() < p3_deadline - 8:
            t_lp = min(t_a * 0.4, p3_deadline - time.monotonic() - 5)
            if t_lp > 3:
                a_opt, _ = lp_optimize_v2(best_sequence, time_limit=t_lp)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  LP-after-SG-{ci}: {score:.6f}")

    print(f"Phase 3 done ({time.monotonic()-t3:.0f}s). Best: {best_score:.6f}")

    # ===== Phase 4: Coordinate descent + final refinements =====
    print("=== Phase 4: Coordinate descent + final ===")
    t4 = time.monotonic()
    remaining = deadline - time.monotonic()

    # Coordinate descent
    t_cd = min(remaining * 0.4, 80.0)
    if t_cd > 5:
        a_opt, _ = coordinate_descent(best_sequence, time_limit=t_cd)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  CD: {score:.6f}")

    # Final short LP+SG rounds
    remaining = deadline - time.monotonic()
    n_rounds = max(1, int(remaining / 15))
    n_best = len(best_sequence)

    for ri in range(n_rounds):
        if time.monotonic() > deadline - 5:
            break
        t_a = min(12.0, deadline - time.monotonic() - 3)
        if t_a < 3:
            break

        # Small perturbation
        scale = 0.0008 + 0.0003 * ri
        noise = rng.standard_normal(n_best) * scale
        a_pert = best_sequence + noise
        a_pert = np.maximum(a_pert, 0.0)
        s = a_pert.sum()
        if s > 1e-15:
            a_pert = a_pert / s

        # Quick LP
        a_opt, _ = lp_optimize_v2(a_pert, time_limit=max(2, t_a * 0.5))
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Final-{ri}: {score:.6f}")

        # Quick SG
        if time.monotonic() < deadline - 3:
            a_opt2, _ = sg_polish_v2(best_sequence, time_limit=max(2, t_a * 0.3),
                                     lr_init=0.002, momentum=0.96, top_k=120, temp=0.00005)
            score2 = float(evaluate_sequence(a_opt2.tolist()))
            if np.isfinite(score2) and score2 < best_score:
                best_score = score2
                best_sequence = a_opt2.copy()
                print(f"  Final-SG-{ri}: {score2:.6f}")

    # Final coordinate descent if time
    remaining = deadline - time.monotonic()
    if remaining > 10:
        a_opt, _ = coordinate_descent(best_sequence, time_limit=min(remaining - 5, 40.0))
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  Final CD: {score:.6f}")

    print(f"Phase 4 done ({time.monotonic()-t4:.0f}s). Best: {best_score:.6f}")

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
