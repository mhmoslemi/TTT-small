    c = c - c.max()
    a = np.exp(c)
    return a / a.sum()

def lse_objective_and_grad_c(c, tau):
    """Objective and gradient in log-parameterization."""
    n = len(c)
    a = softmax(c)
    conv = fast_convolve(a)

    cn = conv / tau
    cn_max = cn.max()
    e = np.exp(cn - cn_max)
    e_sum = e.sum()
    lse = cn_max + tau * np.log(e_sum)
    obj = 2.0 * n * lse

    grad = 4.0 * n * a * (S - T)
    
    return obj, grad

import numpy as np
import time
from scipy.optimize import minimize, linprog
from scipy.fft import rfft, irfft, fft, ifft


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax(c):
    """Numerically stable softmax."""
    c = c - c.max()
    e = np.exp(c)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(c)) / len(c)
    return e / s


def lse_obj_grad_c(c, tau):
    """
    Compute LSE-smoothed objective and gradient in log-parameterization.
    a = softmax(c), objective = 2n * LSE(conv(a))
    """
    n = len(c)
    a = softmax(c)
    conv = fast_convolve(a)

    # LSE with numerical stability
    cn = conv / tau
    cn_max = cn.max()
    e = np.exp(cn - cn_max)
    e_sum = e.sum()
    if e_sum < 1e-300:
        lse = cn_max * tau
        w = np.zeros(len(conv))
        w[np.argmax(conv)] = 1.0
    else:
        lse = cn_max * tau + tau * np.log(e_sum)
        w = e / e_sum

    obj = 2.0 * n * lse

    # Gradient: grad_c[j] = 4n * a_j * (S_j - T)
    # S_j = sum_k w_k * a_{k-j} = cross-correlation of w and a at lag j
    # T = sum_k w_k * conv_k

    # Cross-correlation via FFT
    N = 1 << (3 * n - 2 - 1).bit_length()
    W = fft(w, N)
    A = fft(a, N)
    S = ifft(np.conj(W) * A, N).real[:n]

    T = np.dot(w, conv)

    grad = 4.0 * n * a * (S - T)

    return obj, grad


def lse_lbfgs_optimize(a0, time_limit, tau_init=0.05, tau_final=0.001, tau_decay=0.97,

    else:
        a0 = a0 / s

    # Initial log-parameterization
    c = np.log(a0)
    t0 = time.monotonic()

    best_a = a0.copy()
    best_obj = 2.0 * n * fast_convolve(a0).max()

    tau = tau_init
    total_iter = 0

    while time.monotonic() - t0 < time_limit:
        # Run L-BFGS-B for this tau
        started = False
        for it in range(max_iter_per_tau):
            if time.monotonic() - t0 > time_limit - 2:
                break
            if started:
                break
            obj, grad = lse_obj_grad_c(c, tau)
            if not np.isfinite(obj):
                break
            started = True

        # Do the actual L-BFGS-B call with a time budget
        remaining = time_limit - (time.monotonic() - t0) - 1
        if remaining < 2:
            break

        max_time_this_tau = min(remaining, max(5.0, time_limit * 0.15))

        # Count iterations via callback
        state = {'count': 0}

        def _callback():
            state['count'] += 1
            return state['count'] > max_iter_per_tau or \
                   (state['count'] % 10 == 0 and
                    time.monotonic() - t0 > time_limit - 5)

        try:
            res = minimize(
                lambda c_: lse_obj_grad_c(c_, tau)[0],
                c,

                callback=_callback,

            c = res.x

        if tau < tau_final:
            tau = tau_final
            # Check convergence
            if total_iter > 200:
                break

        total_iter += 1

    # Final: also try a few steps with exact max objective (subgradient)
    if time.monotonic() - t0 < time_limit - 5:

def build_conv_matrix(a):
    """Build matrix C such that C @ g = conv(a, g)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def lp_step(a, conv, M=None):
    """One LP-based refinement step. Returns (a_new, improved)."""
    n = len(a)
    if M is None:
        M = conv.max()
    C = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M * 1.0001)
    bounds = [(0, None)] * n

    try:
        result = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True,
                                                   'time_limit': 30})
        if result.success:
            g = result.x
            g_sum = g.sum()
            s_a = a.sum()
            if g_sum > s_a * (1.0 + 1e-8):
                d = g - a
                norm_d = np.linalg.norm(d)
                if norm_d < 1e-12:
                    return a.copy(), False
                # Step sizing
                best_a = a.copy()
                best_obj = 2.0 * n * conv.max()
                for t in [0.5, 0.3, 0.2, 0.15, 0.1, 0.05, 0.03, 0.01]:
                    a_try = a + t * d
                    a_try = np.maximum(a_try, 0.0)
                    s_try = a_try.sum()
                    if s_try < 1e-15:
                        continue
                    a_try = a_try / s_try
                    conv_try = fast_convolve(a_try)
                    obj_try = 2.0 * n * conv_try.max()
                    if obj_try < best_obj - 1e-12:
                        best_obj = obj_try
                        best_a = a_try.copy()
                        break
                if np.linalg.norm(best_a - a) > 1e-12:
                    return best_a, True
    except Exception:
        pass
    return a.copy(), False

    perturb_count = 0

        if improved:

            conv = fast_convolve(a)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj - 1e-12:

        else:
            no_improve += 1

            noise = rng.standard_normal(n) * scale
            a = best_a + noise

            s = a.sum()
            if s < 1e-15:

            else:
                a = a / s
            conv = fast_convolve(a)
            no_improve = 0
            perturb_count += 1
            max_no_improve = min(200, 60 + 20 * perturb_count)

def coordinate_descent(a0, time_limit, n_trials_per_iter=200):
    """Randomized coordinate descent: move mass between pairs."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, 2.0 * len(a) * fast_convolve(a).max()
    a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    no_improve = 0
    deltas = [0.002, 0.005, 0.01, 0.02, 0.003, 0.007, 0.015]

    while time.monotonic() - t0 < time_limit:
        for _ in range(n_trials_per_iter):
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            delta = deltas[rng.integers(0, len(deltas))]
            sign = 1 if rng.random() < 0.5 else -1
            d = sign * delta

            if a[i] + d < 0 or a[j] - d < 0:
                continue
            a[i] += d
            a[j] -= d
            conv_new = fast_convolve(a)
            obj_new = 2.0 * n * conv_new.max()
            if obj_new < best_obj - 1e-10:
                best_obj = obj_new
                best_a = a.copy()
                conv = conv_new
                no_improve = 0
            else:
                # Revert
                a[i] -= d
                a[j] += d
                no_improve += 1

        if no_improve > 5000:
            # Perturb
            noise = rng.standard_normal(n) * 0.003
            a = best_a + noise
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            no_improve = 0

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations for optimization."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n

    # Bimodal (edges)
    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    # Smooth the edges
    for i in range(min(10, edge)):
        w = 0.5 + 0.5 * i / min(10, edge)
        bi[i] = w
        bi[n - 1 - i] = w
    inits['bimodal'] = bi / bi.sum()

    # Sine squared
    inits['sine'] = (np.sin(np.pi * x / n) ** 2 + 0.01)
    inits['sine'] /= inits['sine'].sum()

    # Chirp-based
    c = 1.0
    inits['chirp'] = (1.0 + np.cos(2.0 * np.pi * c * x * x / n)) / 2.0 + 0.01
    inits['chirp'] /= inits['chirp'].sum()

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.25 * n)) ** 2)
    inits['gaussian'] = g / g.sum()

    # Random Dirichlet
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # Periodic
    for freq in [3, 5]:
        key = f'periodic_{freq}'
        inits[key] = (1.0 + 0.5 * np.cos(2.0 * np.pi * freq * x / n))
        inits[key] /= inits[key].sum()

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n),
                               np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent'] = pp / pps
                # Also add noise version
                noise = rng.standard_normal(n) * 0.05
                pp2 = np.maximum(pp / pps + noise, 0)
                pp2s = pp2.sum()
                if pp2s > 1e-10:
                    inits['parent_noisy'] = pp2 / pp2s

    return inits


def extend_sequence(a_short, n_target, rng, noise_scale=0.02):
    """Extend a shorter sequence to target length via interpolation + noise."""
    n_short = len(a_short)
    s = a_short.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    a_norm = a_short / s
    a_ext = np.interp(np.linspace(0, 1, n_target),
                      np.linspace(0, 1, n_short), a_norm)
    if noise_scale > 0:
        a_ext = a_ext + rng.standard_normal(n_target) * noise_scale
    a_ext = np.maximum(a_ext, 0)
    ses = a_ext.sum()
    if ses > 1e-15:
        a_ext = a_ext / ses
    return a_ext


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase optimization:
    Phase 1: Broad LSE-LBFGS search across lengths and initializations
    Phase 2: Deep LSE-LBFGS on best candidates
    Phase 3: LP-based refinement
    Phase 4: Coordinate descent + final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            score_p = float(evaluate_sequence(p.tolist()))
            if np.isfinite(score_p):
                best_score = score_p
                best_sequence = p.copy()
                print(f"Parent: n={len(p)}, score={score_p:.6f}")
    except Exception:
        pass

    # =========================================================
    # Phase 1: Broad LSE-LBFGS search (250s)
    # =========================================================
    print("=== Phase 1: Broad LSE-LBFGS Search ===")
    t1_start = time.monotonic()
    phase1_limit = 250.0

    lengths = [384, 512, 768, 1024, 1280]
    results = []

    for n in lengths:
        if time.monotonic() > deadline - 900:
            break
        inits = generate_initializations(n, parent, rng)

        # Prioritize init
        priority = ['parent', 'parent_noisy', 'bimodal', 'chirp',
                    'sine', 'uniform', 'random', 'periodic_3', 'periodic_5', 'gaussian']
        init_names = [nm for nm in priority if nm in inits]
        init_names = init_names[:4]  # Limit to 4 per length

        t_per = max(8.0, (phase1_limit - 10) / (len(lengths) * 4))

        for name in init_names:
            if time.monotonic() > deadline - 850:
                break
            t_alloc = min(t_per, deadline - time.monotonic() - 800)
            if t_alloc < 5:
                break

            a0 = inits[name]
            try:
                a_opt, obj = lse_lbfgs_optimize(a0, time_limit=t_alloc,
                                                 tau_init=0.05, tau_final=0.005,
                                                 tau_decay=0.95)
            except Exception:
                continue
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  P1 n={n} {name}: {score:.6f}")

    results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t1_start:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 2: Deep LSE optimization on top candidates (700s)
    # =========================================================
    print("=== Phase 2: Deep LSE Optimization ===")
    t2_start = time.monotonic()

    # Collect top unique candidates
    top_candidates = []
    seen = set()
    for score, n, name, a_opt in results:
        key = n
        if key not in seen:
            top_candidates.append((n, a_opt))
            seen.add(key)
        if len(top_candidates) >= 4:
            break

    # Always include current best
    if best_sequence is not None:
        if len(best_sequence) not in seen:
            top_candidates.append((len(best_sequence), best_sequence.copy()))
            seen.add(len(best_sequence))

    # Also try extending best short to longer
    if len(top_candidates) > 0:
        best_n, best_a = top_candidates[0]
        for ext_n in [best_n * 2, best_n + 256]:
            if ext_n <= 2048 and ext_n not in seen:
                a_ext = extend_sequence(best_a, ext_n, rng, noise_scale=0.01)
                top_candidates.append((ext_n, a_ext))
                seen.add(ext_n)

    remaining = deadline - time.monotonic() - 500.0
    n_cand = max(1, len(top_candidates))
    t_per_deep = max(30.0, remaining * 0.85 / n_cand)

    for i, (n, a0) in enumerate(top_candidates[:4]):
        if time.monotonic() > deadline - 450:
            break
        t_alloc = min(t_per_deep, deadline - time.monotonic() - 400)
        if t_alloc < 15:
            break

        print(f"  P2 cand {i}: n={n}, t={t_alloc:.0f}s")

        # Multi-tau LSE optimization
        a_opt, obj = lse_lbfgs_optimize(a0, time_limit=t_alloc,
                                         tau_init=0.03, tau_final=0.001,
                                         tau_decay=0.96)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  P2 {i} (n={n}): {score:.6f} *NEW BEST*")
            else:
                print(f"  P2 {i} (n={n}): {score:.6f}")

    print(f"Phase 2 done ({time.monotonic()-t2_start:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 3: LP-based refinement (500s)
    # =========================================================
    print("=== Phase 3: LP Refinement ===")
    t3_start = time.monotonic()

    if best_sequence is not None and time.monotonic() < deadline - 200:
        remaining = deadline - time.monotonic() - 200.0
        n_best = len(best_sequence)

        # Multiple LP runs with perturbation
        n_lp_runs = max(1, min(4, int(remaining // 100)))
        t_per_lp = remaining / n_lp_runs * 0.85

        for pi in range(n_lp_runs):
            if time.monotonic() > deadline - 150:
                break
            t_alloc = min(t_per_lp, deadline - time.monotonic() - 120)
            if t_alloc < 10:
                break

            if pi == 0:
                a0 = best_sequence.copy()
            else:
                scale = 0.002 * (pi + 1)
                noise = rng.standard_normal(n_best) * scale
                a0 = np.maximum(best_sequence + noise, 0)
                s = a0.sum()
                if s > 1e-15:
                    a0 = a0 / s
                else:
                    a0 = best_sequence.copy()

            a_opt, obj = lp_refine(a0, time_limit=t_alloc)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  P3 LP {pi}: {score:.6f} *NEW BEST*")
                else:
                    print(f"  P3 LP {pi}: {score:.6f}")

    print(f"Phase 3 done ({time.monotonic()-t3_start:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 4: Final polish (150s)
    # =========================================================
    print("=== Phase 4: Final Polish ===")
    t4_start = time.monotonic()

    if best_sequence is not None and time.monotonic() < deadline - 20:
        remaining = deadline - time.monotonic() - 20.0
        n_best = len(best_sequence)

        # Coordinate descent
        t_cd = min(remaining * 0.4, 60.0)
        if t_cd > 5:
            a_opt, obj = coordinate_descent(best_sequence, time_limit=t_cd)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  P4 CoordDesc: {score:.6f} *NEW BEST*")

        # Short LSE runs with very small tau
        remaining = deadline - time.monotonic() - 20.0
        if remaining > 15:
            for gi in range(3):
                if time.monotonic() > deadline - 15:
                    break
                t_sg = min(remaining / (3 - gi) * 0.7, 40.0)
                if t_sg < 5:
                    break
                a_opt, obj = lse_lbfgs_optimize(best_sequence, time_limit=t_sg,
                                                 tau_init=0.01, tau_final=0.0005,
                                                 tau_decay=0.93)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  P4 LSE {gi}: {score:.6f} *NEW BEST*")

        # Final micro LP
        remaining = deadline - time.monotonic() - 15.0
        if remaining > 10:
            a_opt, obj = lp_refine(best_sequence, time_limit=min(remaining, 30.0))
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  P4 Final LP: {score:.6f} *NEW BEST*")

    print(f"Phase 4 done ({time.monotonic()-t4_start:.0f}s). Best: {best_score:.6f}")

    # Fallback
    if best_sequence is None or best_score >= 1.9:
        if parent is not None:
            best_sequence = parent.copy()
        else:
            best_sequence = np.ones(256) / 256

    # Final validation
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
