            best_obj = obj
            best_x = x.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300:
            cur_lr *= 0.85
            no_improve = 0
            if cur_lr < 1e-7:
                break

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Adam iter {t}, elapsed={elapsed:.0f}s, obj={obj:.8f}, best={best_obj:.8f}")

    return best_x, best_obj


def lbfgs_optimize(x0, n, beta, time_limit, max_iter=5000):
    """L-BFGS optimization in log-parameterized space with time check."""
    t0 = time.monotonic()
    state = {'time_exceeded': False}

    def fun_and_grad(x):
        if time.monotonic() - t0 > time_limit:
            state['time_exceeded'] = True
            obj, _ = smooth_obj_and_grad(x, n, beta)
            return obj, np.zeros_like(x)
        return smooth_obj_and_grad(x, n, beta)

    x_best = np.array(x0, dtype=np.float64)
    best_val = smooth_obj_and_grad(x_best, n, beta)[0]

    try:
        result = minimize(fun_and_grad, x_best, method='L-BFGS-B',
                          jac=True, options={'maxiter': max_iter, 'ftol': 1e-18,
                                            'gtol': 1e-12, 'maxls': 30})
        if result.fun < best_val:
            best_val = result.fun
            x_best = result.x
    except Exception:
        pass

    return x_best, best_val


def subgradient_polish(a0, n, time_limit):
    """Subgradient method on the exact non-smooth objective with momentum."""
    a = np.array(a0, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    v = np.zeros(n)
    b = fast_convolve(a)
    best_obj = b.max()
    best_a = a.copy()
    lr = 0.02
    mom = 0.85
    no_improve = 0

    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        M = b.max()

        # Use peaks within 0.5% of max for a richer subgradient
        threshold = M * 0.995
        active = np.where(b >= threshold)[0]

        sg = np.zeros(n)
        for k in active:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * a[k - idx]

        sg_norm = np.linalg.norm(sg)
        if sg_norm > 1e-15:
            sg /= sg_norm

        v = mom * v - lr * sg
        a = project_simplex(a + v)

        b = fast_convolve(a)
        obj = b.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9997

        if no_improve > 150:
            rng = np.random.default_rng(int(time.monotonic() * 99991) % 2**31)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            lr = 0.008
            no_improve = 0

    return best_a, best_obj


def nesterov_accelerated(x0, n, beta, time_limit, lr=0.02):
    """Nesterov accelerated gradient on log-parameterized space."""
    x = np.array(x0, dtype=np.float64)
    x_prev = x.copy()
    momentum = 1.0

    obj, _ = smooth_obj_and_grad(x, n, beta)
    best_obj = obj
    best_x = x.copy()
    no_improve = 0
    t = 0
    cur_lr = lr

    t0 = time.monotonic()
    while time.monotonic() - t0 < time_limit:
        t += 1
        # Nesterov lookahead
        x_lookahead = x + momentum * (x - x_prev)
        obj_la, grad_la = smooth_obj_and_grad(x_lookahead, n, beta)

        x_prev = x.copy()
        x = x_lookahead - cur_lr * grad_la

        # Evaluate at actual position
        obj, _ = smooth_obj_and_grad(x, n, beta)

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_x = x.copy()
            no_improve = 0
        else:
            no_improve += 1

        beta_nest = (1 + np.sqrt(1 + 4 * momentum ** 2)) / 2
        momentum = (1 - beta_nest) / beta_nest

        if no_improve > 200:
            cur_lr *= 0.9
            no_improve = 0
            if cur_lr < 1e-7:
                break

    return best_x, best_obj


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase optimization:
    Phase 1: Adam with smooth max annealing (coarse to fine beta)
    Phase 2: L-BFGS refinement at high beta
    Phase 3: Multi-start perturbations with Adam
    Phase 4: Nesterov accelerated gradient
    Phase 5: Subgradient polish on exact objective
    Phase 6: Ultra-fine final perturbations
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    def update_best(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        a = a / s
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return float('inf')
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
            update_best(parent, "parent_init")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(384) / 384
        update_best(parent, "uniform")

    n = len(parent)

    # === Phase 1: Smooth Max + Adam with Beta Annealing ===
    print("\n=== Phase 1: Smooth Max Annealing (Adam) ===")

    x = np.log(np.maximum(parent, 1e-15))

    beta_schedule = [
        (5, 0.15), (10, 0.15), (30, 0.12), (100, 0.10),
        (300, 0.08), (1000, 0.08), (3000, 0.07), (10000, 0.07),
    ]

    for beta, lr_frac in beta_schedule:
        rem = deadline - time.monotonic() - 200
        if rem < 5:
            break
        t_alloc = max(5, min(rem, (lr_frac / 0.87) * min(budget_s - 100, rem)))
        lr = 0.08 if beta < 50 else (0.04 if beta < 500 else 0.02)
        print(f"  beta={beta}, lr={lr}, t={t_alloc:.0f}s")
        x_opt, obj_opt = adam_optimize(x, n, beta, t_alloc, lr=lr)
        a_opt = softmax(x_opt)
        update_best(a_opt, f"adam_b{beta}")
        x = np.log(np.maximum(a_opt, 1e-15))

    print(f"  Phase 1 best: {best_score:.8f}")

    # === Phase 2: L-BFGS at high beta ===
    print("\n=== Phase 2: L-BFGS Refinement ===")

    if best_seq is not None:
        x_cur = np.log(np.maximum(best_seq, 1e-15))
        for beta in [3000, 10000, 30000]:
            rem = deadline - time.monotonic() - 400
            if rem < 5:
                break
            t_alloc = min(rem, 150)
            print(f"  L-BFGS beta={beta}, t={t_alloc:.0f}s")
            x_opt, _ = lbfgs_optimize(x_cur, n, beta, t_alloc, max_iter=2000)
            a_opt = softmax(x_opt)
            update_best(a_opt, f"lbfgs_b{beta}")
            x_cur = np.log(np.maximum(a_opt, 1e-15))

    print(f"  Phase 2 best: {best_score:.8f}")

    # === Phase 3: Multi-Start Perturbations ===
    print("\n=== Phase 3: Multi-Start Perturbations ===")

    if best_seq is not None:
        n_best = len(best_seq)
        x_base = np.log(np.maximum(best_seq, 1e-15))
        b_curr = fast_convolve(best_seq)

        for trial in range(12):
            rem = deadline - time.monotonic() - 200
            if rem < 5:
                break
            t_alloc = min(rem, 100)

            if trial % 4 == 0:
                # Gaussian perturbation, increasing scale
                scale = 0.005 + 0.003 * (trial // 4)
                x_pert = x_base + rng.standard_normal(n_best) * scale
            elif trial % 4 == 1:
                # Targeted: reduce elements contributing to top peak
                peak_k = int(b_curr.argmax())
                js = max(0, peak_k - n_best + 1)
                je = min(n_best - 1, peak_k)
                x_pert = x_base.copy()
                x_pert[js:je + 1] -= 0.015
                # Boost random other positions
                other = list(set(range(n_best)) - set(range(js, je + 1)))
                boost_idx = rng.choice(other, size=min(20, len(other)), replace=False)
                x_pert[boost_idx] += 0.008
            elif trial % 4 == 2:
                # Block perturbation: shift mass from one region to another
                block_size = n_best // 8
                src = rng.integers(0, n_best - block_size)
                dst = rng.integers(0, n_best - block_size)
                x_pert = x_base.copy()
                x_pert[src:src + block_size] -= 0.02
                x_pert[dst:dst + block_size] += 0.02
            else:
                # Antisymmetric perturbation (breaks any symmetry)
                half = n_best // 2
                x_pert = x_base.copy()
                pert = rng.standard_normal(half) * 0.01
                x_pert[:half] += pert
                x_pert[half:] -= pert[::-1]

            beta = 5000 + 1000 * trial
            lr = 0.015
            x_opt, _ = adam_optimize(x_pert, n_best, beta, t_alloc, lr=lr)
            a_opt = softmax(x_opt)
            score = update_best(a_opt, f"multi_{trial}")
            if score < best_score + 1e-10:
                x_base = np.log(np.maximum(a_opt, 1e-15))
                b_curr = fast_convolve(a_opt)
                best_seq = a_opt.copy()

    print(f"  Phase 3 best: {best_score:.8f}")

    # === Phase 4: Nesterov Accelerated Gradient ===
    print("\n=== Phase 4: Nesterov Acceleration ===")

    if best_seq is not None:
        n_best = len(best_seq)
        x_cur = np.log(np.maximum(best_seq, 1e-15))

        for beta, lr in [(5000, 0.01), (15000, 0.005), (30000, 0.003)]:
            rem = deadline - time.monotonic() - 100
            if rem < 5:
                break
            t_alloc = min(rem, 120)
            print(f"  Nesterov beta={beta}, lr={lr}, t={t_alloc:.0f}s")
            x_opt, _ = nesterov_accelerated(x_cur, n_best, beta, t_alloc, lr=lr)
            a_opt = softmax(x_opt)
            update_best(a_opt, f"nest_b{beta}")
            x_cur = np.log(np.maximum(a_opt, 1e-15))

    print(f"  Phase 4 best: {best_score:.8f}")

    # === Phase 5: Subgradient Polish ===
    print("\n=== Phase 5: Subgradient Polish ===")

    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 40
        if rem > 10:
            t_alloc = min(rem, 200)
            print(f"  SG polish t={t_alloc:.0f}s")
            a_opt, _ = subgradient_polish(best_seq, n_best, t_alloc)
            update_best(a_opt, "sg_polish")

            # Second round with perturbed start
            rem = deadline - time.monotonic() - 20
            if rem > 10:
                x_pert = np.log(np.maximum(best_seq, 1e-15)) + rng.standard_normal(n_best) * 0.003
                a_pert = softmax(x_pert)
                a_opt2, _ = subgradient_polish(a_pert, n_best, min(rem, 80))
                update_best(a_opt2, "sg_polish2")

    print(f"  Phase 5 best: {best_score:.8f}")

    # === Phase 6: Ultra-fine perturbations ===
    print("\n=== Phase 6: Ultra-fine Perturbations ===")

    if best_seq is not None:
        n_best = len(best_seq)
        for _ in range(5):
            rem = deadline - time.monotonic() - 5
            if rem < 3:
                break
            scale = 0.001 + 0.001 * (_ % 3)
            x_pert = np.log(np.maximum(best_seq, 1e-15)) + rng.standard_normal(n_best) * scale
            # Quick Adam at very high beta
            x_opt, _ = adam_optimize(x_pert, n_best, 50000, min(rem, 40), lr=0.005)
            a_opt = softmax(x_opt)
            update_best(a_opt, f"ultra_{_}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = evaluate_sequence(best_seq.tolist())

    # Final verification
    try:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs) and fs < best_score:
            best_score = fs
    except Exception:
        pass

    print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
