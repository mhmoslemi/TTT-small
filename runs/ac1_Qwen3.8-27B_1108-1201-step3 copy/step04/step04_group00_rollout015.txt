import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(a, conv, T, top_k=50):
    """Compute weighted subgradient of smoothed max of auto-convolution."""
    n = len(a)
    k_idx = np.argpartition(conv, -top_k)[-top_k:]
    pv = conv[k_idx]
    scaled = pv / T
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(len(k_idx))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    # Subgradient: s[j] = 2 * sum_k w[k] * a[k-j] (for valid k-j)
    s = np.zeros(n)
    for i, k in enumerate(k_idx):
        if w[i] < 1e-15:
            continue
        k = int(k)
        # a[k-j] is valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        s[js:je+1] += 2.0 * w[i] * a[k-je:k-js:-1]
    
    return s

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(a, conv, T, top_k):
    """Compute smoothed subgradient of max auto-convolution."""
    n = len(a)
