def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def lp_direction(f, M):
    """Solve the LP to find the best direction b."""
    n = len(f)
    # Maximize sum(b) subject to b >= 0 and f*b <= M
    # f*b has 2n-1 components
    num_constraints = 2 * n - 1
    
    # Build constraint matrix A: A[k, j] = f[k-j] if 0 <= k-j < n
    # This is a (2n-1) x n matrix
    # For efficiency, build it as a sparse-ish structure
    A = np.zeros((num_constraints, n))
    for k in range(num_constraints):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A[k, j] = f[k - j]
    
    # c = -ones (we minimize -sum(b) = maximize sum(b))
    c = -np.ones(n)
    b_vec = M * np.ones(num_constraints)
    
    res = linprog(c, A_ub=A, b_ub=b_vec, bounds=[(0, None)] * n, method='highs')
    
    if res.success:
        return res.x
    return None

  max sum(b) s.t. b >= 0, conv(f,b) <= M
at each iteration, where M = max(conv(f,f)). The optimal b provides a
direction that reduces the peak of the auto-convolution. Combined with

Additional components:

import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def fast_cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def lp_find_direction(f, M):
    """
    Solve LP: max sum(b) s.t. b >= 0, conv(f,b) <= M.
    Returns the optimal b.
    """
    n = len(f)
    num_c = 2 * n - 1
    
    # Build constraint matrix: A[k, j] = f[k-j] if 0 <= k-j < n
    A = np.zeros((num_c, n))
    for k in range(num_c):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            A[k, js:je+1] = f[k-je:k-js+1]
    
    c = -np.ones(n)  # minimize -sum(b)
    b_ub = np.full(num_c, M)
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    
    if res.success:
        return np.maximum(res.x, 0)
    return None

def lp_iteration(f, M, line_search=True):
    """
    One iteration of the LP-based method.
    Returns new f and improvement.
    """
    b = lp_find_direction(f, M)
    if b is None:
        return f, 0.0
    
    S = b.sum()
    if S < 1e-10:
        return f, 0.0
    
    g = b / S  # normalize to sum = 1
    
    if line_search:
        best_t = 0.0
        best_obj = M
        for t in [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1, 0.05]:
            f_new = (1 - t) * f + t * g
            f_new = np.maximum(f_new, 0)
            sn = f_new.sum()
            if sn < 1e-10:
                continue
            f_new = f_new / sn
            c_new = fast_convolve(f_new)
            obj_new = c_new.max()
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_t = t
        if best_t > 0:
            f_new = (1 - best_t) * f + best_t * g
            f_new = np.maximum(f_new, 0)
            f_new = f_new / f_new.sum()
            return f_new, M - best_obj
        return f, 0.0
    else:
        return g, M - fast_convolve(g).max()

def exp_gradient_optimize(a0, time_limit, eta0=1.0, temp0=0.005,
                          temp_decay=0.9997, eta_decay=0.99995,
                          top_k_frac=0.3, verbose=False):
    """Exponentiated gradient / multiplicative weights with softmax subgradients."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    top_k = max(10, int(n * top_k_frac))
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    avg_a = a.copy().astype(np.float64)
    count = 0
    eta = eta0
    temp = temp0
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute softmax-weighted subgradient
        if top_k < len(c):
            indices = np.argpartition(c, -top_k)[-top_k:]
        else:
            indices = np.arange(len(c))
        
        peek = c[indices]
        scaled = peek / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(indices))
            w[np.argmax(peek)] = 1.0
        else:
            w /= ws
        
        sg = np.zeros(n)
        for idx in range(len(indices)):
            if w[idx] < 1e-15:
                continue
            k = int(indices[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            sg[js:je + 1] += 2.0 * w[idx] * a[k - je:k - js + 1][::-1]
        
        sg_max = sg.max()
        if sg_max < 1e-15:
            break
        
        log_update = -eta * sg / sg_max
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-300)
        s_new = a_new.sum()
        if s_new < 1e-300:
            a_new = a.copy()
            s_new = 1.0
        a_new = a_new / s_new
        
        c_new = fast_convolve(a_new)
        obj_new = c_new.max()
        
        count += 1
        avg_a = (count * avg_a + a_new) / (count + 1)
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        c = c_new
        
        if it % 500 == 0:
            c_avg = fast_convolve(avg_a)
            if c_avg.max() < best_obj - 1e-12:
                best_obj = c_avg.max()
                best_a = avg_a.copy()
        
        eta *= eta_decay
        temp *= temp_decay
        
        if no_improve > 2000:
            rng_local = np.random.default_rng(it * 31 + 7)
            a = best_a + rng_local.standard_normal(n) * 0.001
            a = np.maximum(a, 0)
            ps = a.sum()
            if ps > 1e-10:
                a = a / ps
            else:
                a = best_a.copy()
            c = fast_convolve(a)
            best_obj = c.max()
            no_improve = 0
            eta = eta0 * 0.5
            temp = max(temp, temp0 * 0.01)
            avg_a = a.copy()
            count = 0
    
    return best_a.copy(), best_obj

def lp_optimize(a0, time_limit, verbose=False):
    """
    LP-based optimization with line search.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
        s = 1.0
    a = a / s
    n = len(a)
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        M = best_obj  # use the current best M
        
        b = lp_find_direction(a, M)
        if b is None:
            break
        
        S = b.sum()
        if S < 1e-10:
            break
        
        g = b / S
        
        # Line search
        improved = False
        for t in [1.0, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02]:
            a_new = (1 - t) * a + t * g
            a_new = np.maximum(a_new, 0)
            sn = a_new.sum()
            if sn < 1e-10:
                continue
            a_new = a_new / sn
            c_new = fast_convolve(a_new)
            obj_new = c_new.max()
            if obj_new < best_obj - 1e-13:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                improved = True
                break
        
        if not improved:
            no_improve += 1
            if no_improve > 50:
                # Try with a slightly smaller M to get a different direction
                M_reduced = M * 0.99
                b = lp_find_direction(a, M_reduced)
                if b is not None:
                    S = b.sum()
                    if S > 1e-10:
                        g = b / S
                        a_new = 0.5 * a + 0.5 * g
                        a_new = np.maximum(a_new, 0)
                        a_new = a_new / a_new.sum()
                        c_new = fast_convolve(a_new)
                        if c_new.max() < best_obj:
                            best_obj = c_new.max()
                            best_a = a_new.copy()
                            a = a_new
                            no_improve = 0
                            continue
                break
        else:
            no_improve = 0
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {it}: obj={best_obj:.10f}, elapsed={elapsed:.1f}s")
    
    return best_a.copy(), best_obj

def solve_qcqp_cvxpy(a0, time_limit=120):
    """Solve the QCQP using CVXPY."""
    import cvxpy as cp
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return None
    a = a / s
    n = len(a)
    
    if n > 300:
        return None
    
    t_start = time.monotonic()
    
    try:
        x = cp.Variable(n)
        t = cp.Variable()
        
        constraints = [
            cp.sum(x) == 1,
            x >= 0
        ]
        
        for k in range(2 * n - 1):
            expr = cp.Constant(0)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                expr = expr + x[j] * x[k - j]
            constraints.append(expr <= t)
        
        prob = cp.Problem(cp.Minimize(t), constraints)
        
        solved = False
        for solver_name in ['MOSEK', 'CLARABEL', 'SCS']:
            try:
                prob.solve(solver=solver_name)
                if prob.status in ['optimal', 'optimal_inaccurate']:
                    solved = True
                    break
            except Exception:
                continue
        
        if solved and x.value is not None:
            result = np.maximum(x.value, 0)
            rs = result.sum()
            if rs > 1e-10:
                result = result / rs
                return result
    except Exception:
        pass
    
    return None

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using an LP-based iterative method
as the primary optimizer, combined with exponentiated gradient (multiplicative
weights) for polishing, and multi-scale exploration.

The LP-based method solves at each iteration:
  max sum(b) s.t. b >= 0, conv(f,b) <= M
where M = max(conv(f,f)). The optimal b provides a direction that
reduces the auto-convolution peak. Combined with line search and adaptive
step sizes, this creates a fast local optimizer that makes steady progress.

The EG method uses softmax-weighted subgradients for fine-grained polishing
near the optimum.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_constraint_matrix(f):
    """Build the matrix A where (A @ b)[k] = (conv(f, b))[k]."""
    n = len(f)
    num_c = 2 * n - 1
    A = np.zeros((num_c, n))
    for k in range(num_c):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            A[k, js:je+1] = f[k-je:k-js+1][::-1]
    return A


def lp_find_direction(f, M):
    """Solve LP: max sum(b) s.t. b >= 0, conv(f,b) <= M."""
    n = len(f)
    A = build_conv_constraint_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    
    if res.success:
        return np.maximum(res.x, 0)
    return None


def lp_optimize(a0, time_limit, verbose=False):
    """LP-based optimization with line search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    M_factor = 1.0  # starts at 1.0, decreases over time
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        M = best_obj * M_factor
        
        b = lp_find_direction(a, M)
        if b is None:
            M_factor *= 0.99
            if M_factor < 0.9:
                M_factor = 1.0
            continue
        
        S = b.sum()
        if S < 1e-10:
            break
        
        g = b / S
        
        # Line search
        improved = False
        for t in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05]:
            a_new = (1.0 - t) * a + t * g
            a_new = np.maximum(a_new, 0)
            sn = a_new.sum()
            if sn < 1e-10:
                continue
            a_new = a_new / sn
            c_new = fast_convolve(a_new)
            obj_new = c_new.max()
            if obj_new < best_obj - 1e-13:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
                c = c_new
                improved = True
                break
        
        if improved:
            no_improve = 0
            # Gradually decrease M_factor to get sharper directions
            M_factor = max(0.95, M_factor - 0.0001)
        else:
            no_improve += 1
            if no_improve > 20:
                # Reduce M to force a different direction
                M_factor = max(0.9, M_factor - 0.01)
                if M_factor <= 0.9:
                    # Try perturbation
                    rng_local = np.random.default_rng(it * 13 + 5)
                    a = best_a + rng_local.standard_normal(n) * 0.001
                    a = np.maximum(a, 0)
                    ps = a.sum()
                    if ps > 1e-10:
                        a = a / ps
                    else:
                        a = best_a.copy()
                    c = fast_convolve(a)
                    best_obj = c.max()
                    M_factor = 1.0
                    no_improve = 0
                continue
            if no_improve > 100:
                break
        
        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {it}: obj={best_obj:.10f}, M_factor={M_factor:.4f}, elapsed={elapsed:.1f}s")
    
    return best_a.copy(), best_obj


def eg_optimize(a0,
