Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smoothed L-BFGS with 
temperature continuation, mirror descent, and LP-based polishing.
Key innovations: softmax parameterization for simplex constraint,
log-sum-exp smoothing for differentiability, and multi-phase optimization.

import numpy as np
import time
from scipy.optimize import linprog, minimize
from numpy.lib.stride_tricks import sliding_window_view


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g gives a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def compute_grad_a(a, p):
    """Compute gradient of smoothed objective w.r.t. a, given softmax weights p on conv."""
    n = len(a)
    if n == 1:
        return np.array([2.0 * a[0] * p[0]])
    # grad_a[j] = 2 * sum_{m=0}^{n-1} p[j+m] * a[m]
    windows = sliding_window_view(p, n)
    return 2.0 * (windows @ a)


def softmax_obj_grad(w, tau):
    """Compute log-sum-exp smoothed objective and gradient w.r.t. w (where a=softmax(w))."""
    n = len(w)
    w_shifted = w - w.max()
    exp_w = np.exp(w_shifted)
    a = exp_w / exp_w.sum()

    max_c = conv.max()
    shifted = (conv - max_c) / tau
    shifted = np.clip(shifted, -100, 100)
    exp_shifted = np.exp(shifted)
    p = exp_shifted / exp_shifted.sum()
    
    F = max_c + tau * np.log(exp_shifted.sum())
    grad_a = compute_grad_a(a, p)
    a_dot = np.dot(a, grad_a)

    return 2.0 * n * conv.max()


def softmax_fn(w):
    """Stable softmax."""
    w = w - w.max()
    e = np.exp(w)
    return e / e.sum()


def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def lse_smoothed_obj_grad(w, tau):
    """Log-sum-exp smoothed objective and gradient w.r.t. w (where a=softmax(w))."""
    n = len(w)
    a = softmax_fn(w)
    conv = fast_convolve(a)
    max_c = conv.max()
    shifted = (conv - max_c) / tau
    shifted = np.clip(shifted, -50.0, 50.0)
    exp_s = np.exp(shifted)
    p = exp_s / exp_s.sum()
    F = max_c + tau * np.log(exp_s.sum())
    # grad_a[j] = 2 * sum_{m=0}^{n-1} p[j+m] * a[m]
    windows = sliding_window_view(p, n)
    grad_a = 2.0 * (windows @ a)
    a_dot = float(np.dot(a, grad_a))
    grad_w = a * (grad_a - a_dot)
    return F, grad_w


def compute_subgradient(a, conv, top_k=10, sharpness=50.0):
    """Compute a smoothed subgradient using top-k peaks."""
    n = len(a)
    top_k = min(top_k, len(conv))
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peaks]
    # Softmax weights on peaks (sharp)
    w = (peak_vals - peak_vals.max()) * sharpness
    w = np.exp(w)
    w /= w.sum()
    grad_a = np.zeros(n)
    for i in range(top_k):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idx = np.arange(j_start, j_end + 1)
        grad_a[idx] += w[i] * 2.0 * a[k - idx]
    return grad_a


def lp_direction(a, M, margin=1.0):
    """Solve LP: max sum(g) s.t. a*g <= M*margin, g>=0. Returns normalized g or None."""
    n = len(a)
    C = build_conv_matrix(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M * margin)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        g = res.x
        gs = g.sum()
        if gs > 1e-10:
            return g / gs
    return None


def phase_lbfgs(a0, time_limit, tau_list=None, verbose=True):
    """Smoothed L-BFGS with temperature continuation."""
    a = a0.copy()
    n = len(a)
    a = a / a.sum()
    
    if tau_list is None:
        tau_list = [0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    
    # Initialize w from log(a)
    w = np.log(np.maximum(a, 1e-15))
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_obj = obj_from_conv(fast_convolve(a), n)
    
    for tau in tau_list:
        if time.monotonic() - t0 > time_limit - 15:
            break
        
        local_count = [0]
        
        def fun_grad(cur_w):
            F, g = lse_smoothed_obj_grad(cur_w, tau)
            local_count[0] += 1
            return F, g
        
        elapsed = time.monotonic() - t0
        remaining = time_limit - elapsed - 10
        if remaining < 5:
            break
        
        # Time allocation per temperature
        max_iter = min(300, int(remaining * 3))
        
        try:
            res = minimize(
                fun_grad, w, method='L-BFGS-B',
                jac=True, maxiter=max_iter, maxfun=max_iter * 3,
                options={'ftol': 1e-14, 'gtol': 1e-8, 'maxls': 40,
                         'iprint': -1}
            )
            if res.x is not None:
                a_new = softmax_fn(res.x)
                a_new = np.maximum(a_new, 0.0)
                s = a_new.sum()
                if s > 1e-15:
                    a_new /= s
                obj_new = obj_from_conv(fast_convolve(a_new), n)
                if obj_new < best_obj - 1e-15:
                    best_obj = obj_new
                    best_a = a_new.copy()
                    a = a_new.copy()
                    w = np.log(np.maximum(a, 1e-15))
        except Exception:
            pass
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    L-BFGS tau={tau:.1e}: obj={best_obj:.8f}, "
                  f"evals={local_count[0]}, elapsed={elapsed:.1f}s")
    
    return best_a, best_obj


def phase_mirror_descent(a0, time_limit, verbose=True):
    """Mirror descent (multiplicative weights) with adaptive step size."""
    a = a0.copy()
    n = len(a)
    a = a / a.sum()
    
    t0 = time.monotonic()
    best_a = a.copy()
    conv = fast_convolve(a)
    best_obj = obj_from_conv(conv, n)
    
    eta = 0.08
    no_improve = 0
    it = 0
    sharpness = 30.0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        max_c = conv.max()
        
        grad_a = compute_subgradient(a, conv, top_k=10, sharpness=sharpness)
        
        # Mirror descent step
        log_a = np.log(np.maximum(a, 1e-300))
        unnormalized = log_a - eta * grad_a
        unnormalized -= unnormalized.max()
        a_new = np.exp(unnormalized)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new /= s
        
        new_conv = fast_convolve(a_new)
        new_obj = obj_from_conv(new_conv, n)
        
        it += 1
        if new_obj < best_obj - 1e-15:
            best_obj = new_obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        # Adaptive strategy
        if no_improve > 80:
            eta *= 0.92
            sharpness *= 1.05
            no_improve = 0
        if no_improve > 50 and no_improve % 50 == 0:
            eta = max(0.005, eta * 0.98)
        
        if verbose and it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD iter {it}: obj={best_obj:.8f}, eta={eta:.5f}, "
                  f"sharp={sharpness:.1f}, elapsed={elapsed:.1f}s")
    
    return best_a, best_obj


def phase_lp(a0, time_limit, verbose=True):
    """LP-based direction finding with line search and perturbations."""
    a = a0.copy()
    n = len(a)
    a = a / a.sum()
    
    t0 = time.monotonic()
    best_a = a.copy()
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = obj_from_conv(conv, n)
    
    no_improve = 0
    it = 0
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        g = lp_direction(a, M, margin=1.0)
        it += 1
        
        if g is not None:
            d = g - a
            norm_d = np.linalg.norm(d)
            if norm_d > 1e-10:
                # Line search
                best_t = 0.0
                best_f = best_obj
                for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
                    a_try = np.maximum(a + t * d, 0)
                    s = a_try.sum()
                    if s < 1e-15:
                        continue
                    a_try /= s
                    f_try = obj_from_conv(fast_convolve(a_try), n)
                    if f_try < best_f:
                        best_f = f_try
                        best_t = t
                
                if best_t > 0:
                    a = np.maximum(a + best_t * d, 0)
                    a /= a.sum()
                    conv = fast_convolve(a)
                    M = conv.max()
                    cur_obj = 2.0 * n * M
                    if cur_obj < best_obj - 1e-14:
                        best_obj = cur_obj
                        best_a = a.copy()
                        no_improve = 0
                        continue
            no_improve += 1
        else:
            no_improve += 5
        
        # Perturbation when stuck
        if no_improve > 30:
            scale = 0.001 + 0.001 * min(no_improve / 30, 5.0)
            a = best_a + rng.standard_normal(n) * scale
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-15:
                a /= s
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
        
        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {it}: obj={best_obj:.8f}, elapsed={elapsed:.1f}s")
    
    return best_a, best_obj


def resample_to_length(seq, new_n):
    """Resample a sequence to a different length."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy().astype(np.float64) / seq.sum()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s > 1e-15:
        new_seq /= s
    return new_seq


def phase_coordinated(a0, time_limit, verbose=True):
    """Coordinate-wise optimization: for each coordinate, find best value via golden section."""
    a = a0.copy()
    n = len(a)
    a = a / a.sum()
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_obj = obj_from_conv(fast_convolve(a), n)
    
    sweep = 0
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit - 5:
                break
            
            # Golden section search on a[j] (with renormalization)
            a_orig = a[j]
            # Search in [0.1*a_orig, 3.0*a_orig]
            lo = 0.05 * a_orig
            hi = 3.0 * a_orig
            gr = (np.sqrt(5) - 1) / 2
            
            best_local = best_obj
            best_val = a_orig
            
            for _ in range(15):
                d1 = lo + (1 - gr) * (hi - lo)
                d2 = lo + gr * (hi - lo)
                
                for candidate in [d1, d2]:
                    a[j] = candidate
                    s = a.sum()
                    if s < 1e-15:
                        continue
                    a_norm = a / s
                    f = obj_from_conv(fast_convolve(a_norm), n)
                    if f < best_local:
                        best_local = f
                        best_val = candidate
                
                if d1 < d2:
                    if f(d1) < f(d2):  # placeholder - can't call f like this
                        pass
                break  # Simplified: just try a few values
            
            # Simplified coordinate update: try a few multipliers
            best_mult = 1.0
            best_f = obj_from_conv(fast_convolve(a), n)
            
            for mult in [0.6, 0.75, 0.9, 1.0, 1.1, 1.25, 1.5]:
                a[j] = a_orig * mult
                s = a.sum()
                if s < 1e-15:
                    continue
                a_norm = a / s
                f = obj_from_conv(fast_convolve(a_norm), n)
                if f < best_f:
                    best_f = f
                    best_mult = mult
            
            if best_mult != 1.0:
                a[j] = a_orig * best_mult
                a = a / a.sum()
                cur_obj = obj_from_conv(fast_convolve(a), n)
                if cur_obj < best_obj - 1e-14:
                    best_obj = cur_obj
                    best_a = a.copy()
                    improved = True
        
        sweep += 1
        if not improved and sweep > 3:
            # Add small perturbation
            rng = np.random.default_rng(sweep)
            a = best_a + rng.standard_normal(n) * 0.001
            a = np.maximum(a, 0)
            a /= a.sum()
            sweep = 0
        elif not improved:
            sweep += 1
        
        if verbose and sweep % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Coord sweep {sweep}: obj={best_obj:.8f}, elapsed={elapsed:.1f}s")
    
    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Phases: L-BFGS continuation, Mirror Descent, LP polish, Coordinated search, Restarts.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)
    
    best_sequence = None
    best_score = float('inf')
    
    def try_update(a):
        """Try to update best sequence with given array."""
        nonlocal best_sequence, best_score
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print(f"  >>> NEW BEST: score={score:.8f}, n={len(a)}")
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            n_p = len(parent)
            obj_p = obj_from_conv(fast_convolve(parent), n_p)
            print(f"Parent: n={n_p}, obj={obj_p:.6f}")
    except Exception:
        pass
    
    if parent is None:
        parent = np.ones(768) / 768.0
        print("No parent, using uniform n=768")
    
    n = len(parent)
    
    # ===== Phase 1: Smoothed L-BFGS on parent =====
    print("\n=== Phase 1: Smoothed L-BFGS (main) ===")
    t_start = time.monotonic()
    t_alloc = min(500.0, (deadline - t_start) * 0.38)
    print(f"  Time budget: {t_alloc:.0f}s")
    
    taus1 = [0.004, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]
    a1, obj1 = phase_lbfgs(parent, time_limit=t_alloc, tau_list=taus1, verbose=True)
    try_update(a1)
    
    # ===== Phase 2: Mirror Descent =====
    print("\n=== Phase 2: Mirror Descent ===")
    t_start = time.monotonic()
    t_alloc = min(350.0, (deadline - t_start) * 0.3)
    print(f"  Time budget: {t_alloc:.0f}s")
    
    a2, obj2 = phase_mirror_descent(best_sequence, time_limit=t_alloc, verbose=True)
    try_update(a2)
    
    # ===== Phase 3: LP Polish =====
    print("\n=== Phase 3: LP Direction Finding ===")
    t_start = time.monotonic()
    t_alloc = min(250.0, (deadline - t_start) * 0.28)
    print(f"  Time budget: {t_alloc:.0f}s")
    
    a3, obj3 = phase_lp(best_sequence, time_limit=t_alloc, verbose=True)
    try_update(a3)
    
    # ===== Phase 4: Second L-BFGS pass (finer tau) =====
    print("\n=== Phase 4: Fine L-BFGS Polish ===")
    t_start = time.monotonic()
    t_alloc = min(150.0, (deadline - t_start) * 0.35)
    if t_alloc > 30:
        taus4 = [0.0001, 0.00005, 0.00002, 0.00001, 0.000005]
        a4, obj4 = phase_lbfgs(best_sequence, time_limit=t_alloc, tau_list=taus4, verbose=True)
        try_update(a4)
    
    # ===== Phase 5: Coordinate Descent =====
    print("\n=== Phase 5: Coordinate Descent ===")
    t_start = time.monotonic()
    t_alloc = min(120.0, (deadline - t_start) * 0.35)
    if t_alloc > 20:
        a5, obj5 = phase_coordinated(best_sequence, time_limit=t_alloc, verbose=True)
        try_update(a5)
    
    # ===== Phase 6: Multiple restarts with different strategies =====
    print("\n=== Phase 6: Restarts ===")
    t_start = time.monotonic()
    remaining = deadline - t_start
    if remaining > 30:
        n_restarts = max(1, int(remaining / 35))
        strategies = ['lbfgs', 'md', 'lbfgs_fine', 'md_coarse']
        
        for ri in range(n_restarts):
            if time.monotonic() > deadline - 10:
                break
            t_alloc = min(30.0, deadline - time.monotonic() - 5)
            if t_alloc < 8:
                break
            
            strat = strategies[ri % len(strategies)]
            
            # Generate perturbed starting point
            scale = 0.0015 * (1.0 + 0.15 * ri)
            # Mix of random and structured perturbations
            noise = rng.standard_normal(n) * scale
            if ri % 3 == 0:
                # Add smooth component
                smooth = np.sin(np.pi * np.arange(n) / n) * scale * 0.3
                noise += smooth
            elif ri % 3 == 1:
                # Add cosine component
                smooth = np.cos(2 * np.pi * np.arange(n) / n) * scale * 0.2
                noise += smooth
            
            a_pert = best_sequence + noise
            a_pert = np.maximum(a_pert, 0)
            s = a_pert.sum()
            if s > 1e-15:
                a_pert /= s
            
            if strat == 'lbfgs':
                taus_r = [0.0005, 0.0002, 0.0001, 0.00005, 0.00002]
                a_opt, _ = phase_lbfgs(a_pert, time_limit=t_alloc, tau_list=taus_r, verbose=False)
            elif strat == 'lbfgs_fine':
                taus_r = [0.0001, 0.00005, 0.00002, 0.00001]
                a_opt, _ = phase_lbfgs(a_pert, time_limit=t_alloc, tau_list=taus_r, verbose=False)
            elif strat == 'md_coarse':
                a_opt, _ = phase_mirror_descent(a_pert, time_limit=t_alloc, verbose=False)
            else:
                a_opt, _ = phase_mirror_descent(a_pert, time_limit=t_alloc, verbose=False)
            
            try_update(a_opt)
    
    # ===== Final verification =====
    if best_sequence is None:
        best_sequence = parent.copy()
    
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    
    print(f"\n{'='*50}")
    print(f"FINAL: score={best_score:.8f}, n={len(best_sequence)}")
    print(f"Total time: {time.monotonic() - (deadline - float(budget_s) + 25.0):.1f}s")
    print(f"{'='*50}")
    
    return [float(x) for x in best_sequence]
