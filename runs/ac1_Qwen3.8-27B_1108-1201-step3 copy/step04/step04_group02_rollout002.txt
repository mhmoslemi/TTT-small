"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, conv, top_k=64, temp=0.001):
    """Compute subgradient of max(conv) with softmax weighting over top peaks."""
    n = len(f)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]

    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= wsum

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k - idx]

    return sg


def solve_lp_min_m(a, time_limit=30):
    """
    Solve LP: min M s.t. (a*g)[k] <= M for all k, sum(g)=1, g>=0.
    Returns (g_opt, M_opt).
    """
    n = len(a)
    num_conv = 2 * n - 1
    
    # Variables: [g_0, ..., g_{n-1}, M]
    # Minimize M
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    
    # Constraints: (a*g)[k] - M <= 0 for k = 0, ..., 2n-2
    # (a*g)[k] = sum_j a[j]*g[k-j] (for valid j)
    # So: sum_j a[j]*g[k-j] - M <= 0
    A_ub = np.zeros((num_conv, n + 1))
    for k in range(num_conv):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A_ub[k, j] = a[idx]
        A_ub[k, n] = -1.0  # -M
    
    b_ub = np.zeros(num_conv)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            g = res.x[:n]
            M = res.x[n]
            return g, M
    except Exception:
        pass
    return None, None


def solve_lp_max_sum(a, M, time_limit=30):
    """
    Solve LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Returns g_opt.
    """
    n = len(a)
    num_conv = 2 * n - 1
    
    c = -np.ones(n)
    
    A_ub = np.zeros((num_conv, n))
    for k in range(num_conv):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A_ub[k, j] = a[idx]
    
    b_ub = np.full(num_conv, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def mirror_descent(a0, time_limit, eta0=0.01, decay=0.9999, top_k=32, temp0=0.001, temp_decay=1.002):
    """Exponentiated gradient / mirror descent for minimizing max conv."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    eta = eta0
    temp = temp0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        
        sg = compute_subgradient(a, conv, top_k=top_k, temp=temp)
        
        # Exponentiated gradient update
        update = np.exp(-eta * sg)
        update[~np.isfinite(update)] = 1.0
        update = np.maximum(update, 1e-30)
        
        a_new = a * update
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-30:
            a_new = a
        else:
            a_new = a_new / sn
        
        conv = fast_convolve(a_new)
        obj = conv.max()
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive parameters
        eta *= decay
        temp *= temp_decay
        
        # Restart if stuck
        if no_improve > 200:
            rng = np.random.default_rng(it * 17 + 3)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(a)
            eta = eta0 * 0.5
            temp = temp0 * 0.5
            no_improve = 0
        else:
            a = a_new
    
    return best_a, best_obj

def generate_inits(n, parent=None, rng=None):
    """Generate diverse structured initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = []
    x = np.arange(n, dtype=np.float64)
    xm = x - n / 2
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Sine window (raised cosine)
    c = np.cos(np.pi * xm / (n/2 + 1)) ** 2 + 0.01
    inits.append(np.maximum(c, 0) / np.maximum(c, 0).sum())
    
    # Gaussian
    g = np.exp(-xm**2 / (2 * (n/4)**2))
    inits.append(g / g.sum())
    
    # Two Gaussians (bimodal)
    g2 = np.exp(-(xm - n/4)**2 / (2 * (n/8)**2)) + np.exp(-(xm + n/4)**2 / (2 * (n/8)**2))
    inits.append(g2 / g2.sum())
    
    # Power law decay from center
    pw = 1.0 / (np.abs(xm) + 1.0) ** 0.5
    inits.append(pw / pw.sum())
    
    # Linear taper
    lt = 1.0 - 2.0 * np.abs(xm) / n
    lt = np.maximum(lt, 0.01)
    inits.append(lt / lt.sum())
    
    # Cosine^4
    c4 = np.cos(np.pi * xm / (n/2 + 1)) ** 4 + 0.01
    inits.append(np.maximum(c4, 0) / np.maximum(c4, 0).sum())
    
    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    
    # Random Dirichlet (concentrated)
    inits.append(rng.dirichlet(np.ones(n) * 0.1))
    
    # Parent-based
    if parent is not None:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
            # Scaled versions
            for scale in [0.5, 1.5, 2.0]:
                x2 = np.arange(int(n/scale), dtype=np.float64)
                p_interp = np.interp(x2, np.linspace(0, 1, len(parent)), parent)
                p_interp = np.maximum(p_interp, 0)
                # Tile or pad to length n
                p_tiled = np.tile(p_interp / p_interp.sum(), n // len(p_interp) + 1)[:n]
                p_tiled = p_tiled / p_tiled.sum()
                inits.append(p_tiled)
    
    return inits

        t1 = min(200, deadline - time.monotonic() - 600)
        if t1 > 20:
            print(f"  Running FW-LP (n={n_parent}), budget={t1:.0f}s")
            a_opt, obj = fw_lp_optimize(parent, time_limit=t1, verbose=True)
            try_seq(a_opt, "phase1_fw")
    else:
        a_opt = np.ones(384) / 384
        try_seq(a_opt, "initial_uniform")

    print("\n=== Phase 2: Multi-start optimization ===")

    lengths = [384]

    n_best = len(best_f)

    if parent is not None:
        if n_parent not in lengths:
            lengths.insert(0, n_parent)
    
    # Add some variation
    for extra in [256, 512, 768]:
        if extra not in lengths and len(lengths) < 4:
            lengths.append(extra)
    
    rem = deadline - time.monotonic() - 500
    n_lengths = len(lengths)
    t_per_length = max(50, rem / n_lengths)
    
    for n in lengths:
        if time.monotonic() > deadline - 400:
            break
        
        inits = generate_inits(n, parent, rng)
        
        # Select top 4-6 inits based on initial objective
        scored_inits = []
        for fi in inits:
            fi = np.maximum(fi, 0)
            fi = fi / fi.sum()
            c = fast_convolve(fi).max()
            scored_inits.append((2 * n * c, fi))
        scored_inits.sort()
        top_inits = [fi for _, fi in scored_inits[:min(5, len(scored_inits))]]
        
        rem2 = deadline - time.monotonic() - 400
        t_per_init = max(20, rem2 / (len(top_inits) + 1))

        if best_seq is not None and len(best_seq) != n:
            bp = np.interp(np.arange(n), np.linspace(0, 1, len(best_seq)), best_seq)
            bp = np.maximum(bp, 0)
            bp = bp / bp.sum()
            top_inits.insert(0, bp)

                break
            t_alloc = min(t_per_init, deadline - time.monotonic() - 300)
            if t_alloc < 15:

                a_opt, obj = sg_descent_v2(fi, time_limit=t_alloc, eta0=0.005, top_k=max(32, n//8))

                try_seq(a_opt, f"p2_combined_n{n}")

    print(f"\n=== Phase 3: Deep Combined Optimization ===")
    
    if best_seq is not None:
        n_best = len(best_seq)
        
        # Multiple combined runs with perturbations
        for p in range(5):
            rem = deadline - time.monotonic() - 250
            if rem < 20:
                break
            t_alloc = min(rem // (5 - p), 100)
            
            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:

            elif p == 2:
                f_init = project_simplex(0.9 * best_seq + 0.1 * (np.ones(n_best) / n_best))

    print(f"\n=== Phase 4: Final Polish ===")
    print(f"  Current best: {best_score:.8f}")
    
    if best_seq is not None:
        n_best = len(best_seq)
        
        # Mirror descent polish
        rem = deadline - time.monotonic() - 30
        if rem > 10:

                                      eta0=0.003, top_k=n_best//3, temp0=0.0003)

        # Combined polish
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            a_opt, _ = combined_optimize(best_seq, time_limit=min(rem, 40), verbose=False)
            try_seq(a_opt, "p4_combined_polish")
        
        # Micro perturbations
        rem = deadline - time.monotonic() - 10
        if rem > 5:
            for _ in range(6):
                rem = deadline - time.monotonic() - 8
                if rem < 4:
                    break
                f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.001)
                a_opt, _ = mirror_descent(f_pert, time_limit=min(rem, 10),
                                          eta0=0.001, top_k=n_best//3, temp0=0.0001)
                try_seq(a_opt, "p4_micro")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
