import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:size]

def compute_lse_gradient(a, b, t):
    """Compute the gradient of t*log(sum exp(b/t)) w.r.t. a.
    
    Returns the gradient vector of length n.
    """
    n = len(a)
    # w = softmax(b / t)
    bt = b / t
    bt -= bt.max()
    w = np.exp(bt)
    ws = w.sum()
    w /= ws
    
    # forward[i] = sum_m a[m] * w[m+i] for i = 0, ..., n-1
    N = 1 << (3 * n - 2 - 1).bit_length()
    An = np.fft.rfft(a, N)
    Wn = np.fft.rfft(w, N)
    forward = np.fft.irfft(np.conj(An) * Wn, N)[:n]
    
    # grad[i] = 2n * (2*forward[i] + correction)
    # correction: (w[2*i] - 2*w[i]) * a[i] if 2*i < 2n-1 else -2*w[i]*a[i]
    correction = -2.0 * w[:n] * a  # w[i] for i = 0, ..., n-1
    idx_2i = 2 * np.arange(n)
    valid = idx_2i < 2 * n - 1
    correction[valid] += w[idx_2i[valid]] * a[valid]
    
    grad = 2.0 * n * (2.0 * forward + correction)
    return grad

    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    b = fast_convolve(a)
    max_b = b.max()
    best_obj = max_b
    best_a = a.copy()
    
    avg_a = a.copy().astype(np.float64)
    count = 0
    
    t = max_b * t0_frac
    eta = eta0
    
    t_start = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t_start < time_limit:
        it += 1
        
        grad = compute_lse_gradient(a, b, t)
        gmax = grad.max()
        if gmax < 1e-15:
            break
        
        # Multiplicative update
        log_update = -eta * grad / gmax
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 1e-300)
        s_new = a_new.sum()
        a_new = a_new / s_new
        
        b_new = fast_convolve(a_new)
        obj_new = b_new.max()
        
        count += 1
        avg_a = (count * avg_a + a_new) / (count + 1)

            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

            if b_avg.max() < best_obj - 1e-12:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using exponentiated gradient (multiplicative
weights) with log-sum-exp smoothed subgradients computed via FFT-based cross-correlation.
Combined with projected subgradient, coordinate-pair local search, multi-scale

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:size]


def compute_lse_gradient(a, b, t):
    """Compute gradient of LSE(b/t)*t w.r.t. a using FFT cross-correlation.
    
    grad[i] = 2n * (2*forward[i] + (w[2i] - 2*w[i])*a[i])
    where forward[i] = sum_m a[m]*w[m+i], w = softmax(b/t)
    """
    n = len(a)
    # Softmax weights
    bt = b / t
    bt -= bt.max()
    w = np.exp(bt)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(b))
        w[np.argmax(b)] = 1.0
    else:
        w /= ws

    # Cross-correlation: forward[i] = sum_m a[m] * w[m+i] for i=0..n-1
    N = 1 << (3 * n - 2 - 1).bit_length()
    An = np.fft.rfft(a, N)
    Wn = np.fft.rfft(w, N)
    forward = np.fft.irfft(np.conj(An) * Wn, N)[:n]

    # Correction term
    idx2 = 2 * np.arange(n)
    correction = (w[idx2] - 2.0 * w[:n]) * a

    grad = 2.0 * n * (2.0 * forward + correction)
    return grad


def lse_eg_optimize(a0, time_limit, t0_frac=0.1, t_decay=0.9992, eta0=1.0,
                    eta_decay=0.9999, restart_threshold=2000, verbose=False):
    """Exponentiated gradient with log-sum-exp smoothed subgradient."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    b = fast_convolve(a)
    max_b = b.max()
    best_obj = max_b
    best_a = a.copy()

    avg_a = a.copy().astype(np.float64)
    count = 0
    t = max(1e-6, max_b * t0_frac)
    eta = eta0

    t_start = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t_start < time_limit:
        it += 1

        grad = compute_lse_gradient(a, b, t)
        gmax = np.abs(grad).max()
        if gmax < 1e-18:
            break

        # Multiplicative update with numerical stability
        log_update = -eta * grad / gmax
        log_update -= log_update.max()
        exp_update = np.exp(log_update)
        a_new = a * exp_update
        a_new = np.maximum(a_new, 1e-300)
        s_new = a_new.sum()
        if s_new < 1e-300:
            a_new = a.copy()
            s_new = 1.0
        a_new = a_new / s_new

        b_new = fast_convolve(a_new)
        obj_new = b_new.max()

        count += 1
        avg_a = (count * avg_a + a_new) / (count + 1)

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            if verbose and it % 5000 == 0:
                print(f"    LSE-EG iter {it}: obj={best_obj:.10f}, t={t:.6f}, eta={eta:.6f}")
        else:
            no_improve += 1

        a = a_new
        b = b_new

        # Check running average periodically
        if it % 2000 == 0:
            b_avg = fast_convolve(avg_a)
            if b_avg.max() < best_obj - 1e-14:
                best_obj = b_avg.max()
                best_a = avg_a.copy()

        t *= t_decay
        eta *= eta_decay

        # Stagnation: restart with perturbation
        if no_improve > restart_threshold:
            rng = np.random.default_rng(it * 13 + 7)
            scale = 0.001 * (1 + np.log1p(no_improve / restart_threshold))
            a = best_a + rng.standard_normal(n) * scale
            a = np.maximum(a, 0)
            ps = a.sum()
            if ps > 1e-10:
                a = a / ps
            else:
                a = best_a.copy()
            b = fast_convolve(a)
            best_obj = b.max()
            no_improve = 0
            eta = max(eta0 * 0.3, eta)
            t = max(t, max_b * t0_frac * 0.1)
            avg_a = a.copy()
            count = 0

    return best_a.copy(), best_obj


def projected_subgradient_optimize(a0, time_limit, t0_frac=0.05, t_decay=0.999,
                                   eta0=0.01, eta_decay=0.9995, verbose=False):
    """Projected subgradient method on the simplex with LSE smoothing."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    b = fast_convolve(a)
    max_b = b.max()
    best_obj = max_b
    best_a = a.copy()

    t = max(1e-6, max_b * t0_frac)
    eta = eta0
    t_start = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t_start < time_limit:
        it += 1

        grad = compute_lse_gradient(a, b, t)
        grad -= grad.mean()  # Center for simplex projection

        a_new = a - eta * grad
        # Project onto simplex {x >= 0, sum(x) = 1}
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()

        else:
            no_improve += 1

        a = a_new
        b = b_new
        t *= t_decay

            else:
                a = best_a.copy()
            b = fast_convolve(a)
            no_improve = 0
            eta = eta0 * 0.5

    return best_a.copy(), best_obj


def local_search_polish(a0, time_limit, max_rounds=500, rng_seed=42):

    no_improve = 0

    while time.monotonic() - t0 < time_limit and it < max_rounds:
        it += 1
        improved = False

        # Pick a pair of coordinates to perturb
        if n > 1:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:

            # Binary search for best delta in a[i]+δ, a[j]-δ
            best_delta = 0.0
            best_local_obj = best_obj

            for _ in range(20):  # Binary search iterations

                hi = best_a[j] * 0.9

                        a_candidate = best_a.copy()
                        a_candidate[i] += delta

                            improved = True
                            break

    return best_a.copy(), best_obj


def generate_inits(a0, rng, n):
    """Generate diverse initializations for a given length n."""
    inits = []

    # Scaled parent
    if len(a0) > 0:
        if len(a0) == n:
            inits.append(a0.copy())
        else:
            x_old = np.linspace(0, 1, len(a0))
            x_new = np.linspace(0, 1, n)
            interp = np.interp(x_new, x_old, a0)
            interp = np.maximum(interp, 0)
            if interp.sum() > 1e-10:
                inits.append(interp / interp.sum())

    # Uniform
    inits.append(np.ones(n) / n)

    x = np.arange(n, dtype=np.float64)

    # Exponential decay
    for beta in [0.5, 1.0, 2.0, 5.0]:
        e = np.exp(-beta * x / n)
        inits.append(e / e.sum())
        e2 = np.exp(-beta * (n - 1 - x) / n)
        inits.append(e2 / e2.sum())

    # Edge-concentrated
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Strong edge
    edge_strong = 1.0 / (np.minimum(x, n - 1 - x) + 0.1)
    inits.append(edge_strong / edge_strong.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())

    # Cosine
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.02
    inits.append(c1 / c1.sum())

    # Diamond (V-shape)
    d = 1.0 - np.abs(x - 0.5 * (n - 1)) / (0.5 * (n - 1) + 1)
    d = np.maximum(d, 0.01)
    inits.append(d / d.sum())

    # Two-peak
    peak1 = np.exp(-0.5 * ((x - 0.2 * n) / (0.05 * n)) ** 2)
    peak2 = np.exp(-0.5 * ((x - 0.8 * n) / (0.05 * n)) ** 2)
    two = peak1 + peak2
    inits.append(two / two.sum())

    # Random Dirichlet
    for alpha in [0.1, 0.5, 2.0]:
        r = rng.dirichlet(np.ones(n) * alpha)
        inits.append(r)

    # Perturbed best
    if len(a0) == n and a0.sum() > 1e-10:
        a0_norm = a0 / a0.sum()
        for scale in [0.001, 0.005, 0.02, 0.05]:
            p = a0_norm + rng.standard_normal(n) * scale
            p = np.maximum(p, 0)
            ps = p.sum()
            if ps > 1e-10:
                inits.append(p / ps)

    return inits

            n_alt = int(n_curr * factor)
            if 64 <= n_alt <= 1024:
                alt_lengths.add(n_alt)

        for n_alt in sorted(alt_lengths):
            rem = deadline - time.monotonic() - 50
            if rem < 20:
                break
            if n_alt == len(best_seq):
                continue
            a_alt = try_length_change(best_seq, n_alt, rng)
            t_alloc = min(40, rem / 2)
            if t_alloc < 10:
                break
            a_opt, _ = lse_eg_optimize(a_alt, time_limit=t_alloc, t0_frac=0.1,
                                       t_decay=0.9992, eta0=1.0, eta_decay=0.9999)
            try_seq(a_opt, f"len_n{n_alt}")

    # === Phase 7: Final fine-tuning with very low temperature ===
    print(f"\n=== Phase 7: Final Fine-tune (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            t_alloc = min(60, rem)
            a_opt, _ = lse_eg_optimize(best_seq, time_limit=t_alloc, t0_frac=0.02,
                                       t_decay=0.9999, eta0=0.3, eta_decay=0.99995,
                                       restart_threshold=5000, verbose=False)
            try_seq(a_opt, "fine1")

            # Even finer
            rem = deadline - time.monotonic() - 10
            if rem > 5:
                a_opt2, _ = lse_eg_optimize(best_seq, time_limit=min(rem, 30),
                                            t0_frac=0.01, t_decay=0.99995,
                                            eta0=0.1, eta_decay=0.99998,
                                            restart_threshold=10000, verbose=False)
                try_seq(a_opt2, "fine2")

        # Final local search
        rem = deadline - time.monotonic() - 5
        if rem > 5 and best_seq is not None:
            a_opt, _ = local_search_polish(best_seq, time_limit=min(rem, 20),
                                           max_rounds=50, rng_seed=789)
            try_seq(a_opt, "local_final")

    # === Phase 8: One more full pass on best found ===
    print(f"\n=== Phase 8: Final Polish (best={best_score:.8f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 5
        if rem > 15:
            a_opt, _ = lse_eg_optimize(best_seq, time_limit=min(rem, 40),
                                       t0_frac=0.03, t_decay=0.9998,
                                       eta0=0.5, eta_decay=0.99995, verbose=False)
            try_seq(a_opt, "final_eg")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
