            return x[n]

            return g

            return x[:n].sum() - 1.0

            g = np.zeros(n + 1)
            g[:n] = 1.0
            return g

    factors = [0.7, 0.8, 0.9, 0.95, 1.05, 1.1, 1.2, 1.3]

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            old_val = best_a[i]
            if old_val < 1e-12:
                continue
            for f in factors:
                candidate = best_a.copy()
                candidate[i] = old_val * f
                if candidate[i] < 0:
                    continue
                cs = candidate.sum()
                if cs < 1e-15:
                    continue
                candidate = candidate / cs
                obj = compute_score(candidate)
                if obj < best_obj - 1e-13:
                    best_a = candidate
                    best_obj = obj
                    old_val = candidate[i]
                    improved = True
                    break
        if not improved and sweep > 5:
            break

    return best_a.copy(), best_obj


def generate_inits(n, rng, a_best=None):
    """Generate diverse initializations for length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine modulated (various frequencies)
    for freq in [1, 2, 3, 4, 5, 7]:
        for alpha in [0.3, 0.5, 0.7]:
            c = 1.0 + alpha * np.cos(2.0 * np.pi * freq * x / n)
            c = np.maximum(c, 0.01)
            inits.append(c / c.sum())

    # Edge concentrated
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())

    # Bimodal (two peaks)
    for frac in [0.25, 0.33]:
        g1 = np.exp(-0.5 * ((x - frac * (n - 1)) / (0.1 * n)) ** 2)
        g2 = np.exp(-0.5 * ((x - (1 - frac) * (n - 1)) / (0.1 * n)) ** 2)
        b = g1 + g2
        inits.append(b / b.sum())

    # Difference-set inspired: periodic pattern
    for period in [3, 4, 5]:
        pattern = np.ones(period)
        pattern[0] = 1.5
        pat = np.tile(pattern, n // period + 1)[:n]
        inits.append(pat / pat.sum())

    # Random Dirichlet
    for alpha_val in [0.1, 0.5, 2.0]:
        r = rng.dirichlet(np.ones(n) * alpha_val)
        inits.append(r)

    # From best (if available and same length)
    if a_best is not None and len(a_best) == n:
        for scale in [0.001, 0.005, 0.01, 0.02, 0.05]:
            p = a_best + rng.standard_normal(n) * scale
            p = np.maximum(p, 0)
            ps = p.sum()
            if ps > 1e-10:
                inits.append(p / ps)

    return inits


def interpolate_to_length(a, n_new):
    """Interpolate a sequence to a new length."""
    n_old = len(a)
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    else:
        a_new = np.ones(n_new) / n_new
    return a_new


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Uses:
    - EG with FISTA momentum as primary optimizer
    - Active-set QP polishing via scipy
    - Coordinate descent for fine polishing
    - Multi-scale, multi-init exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-10:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = evaluate_sequence(parent.tolist())
            print(f"Parent: n={len(parent)}, score={parent_score:.8f}")
            if np.isfinite(parent_score):
                best_score = parent_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = np.ones(384) / 384
        print("No parent, using uniform n=384")
        best_score = evaluate_sequence(parent.tolist())
        best_seq = parent.copy()

    n = len(parent)
    print(f"Starting: n={n}, best_score={best_score:.8f}")
    print(f"Time budget: {budget_s}s, deadline in {deadline - time.monotonic():.0f}s")

    # === Phase 1: Primary EG on parent with FISTA momentum ===
    print("\n=== Phase 1: EG with momentum on parent ===")
    rem = deadline - time.monotonic() - 500
    t1 = min(400, rem)
    if t1 > 30:
        print(f"  EG n={n}, alloc={t1:.0f}s")
        a_opt, _ = eg_optimize(parent, time_limit=t1, eta0=1.2, temp0=0.003,
                               temp_decay=0.9995, eta_decay=0.99995,
                               top_k_frac=0.3, use_momentum=True,
                               verbose=True, rand_seed=seed)
        try_seq(a_opt, "eg_p1_mom")

        # Run again with different params
        rem = deadline - time.monotonic() - 400
        if rem > 60:
            a_opt2, _ = eg_optimize(best_seq, time_limit=min(100, rem),
                                     eta0=0.5, temp0=0.008,
                                     temp_decay=0.9998, eta_decay=0.9999,
                                     top_k_frac=0.4, use_momentum=True,
                                     rand_seed=seed + 1)
            try_seq(a_opt2, "eg_p1_v2")

    # === Phase 2: QP Polish ===
    print(f"\n=== Phase 2: QP Polish ===")
    print(f"  Best: {best_score:.8f}")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 350
        if rem > 30:
            print(f"  QP polish, alloc={min(rem, 200):.0f}s")
            a_qp, _ = active_set_qp_polish(best_seq, time_limit=min(rem, 200))
            try_seq(a_qp, "qp_p2")

    # === Phase 3: Coordinate polish ===
    print(f"\n=== Phase 3: Coordinate Polish ===")
    print(f"  Best: {best_score:.8f}")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 20:
            a_coord, _ = coordinate_polish(best_seq, time_limit=min(rem, 60), n_sweeps=40)
            try_seq(a_coord, "coord_p3")

    # === Phase 4: Diverse initializations ===
    print(f"\n=== Phase 4: Diverse Inits ===")
    print(f"  Best: {best_score:.8f}")

    inits = generate_inits(n, rng, best_seq)
    for i, f_init in enumerate(inits):
        rem = deadline - time.monotonic() - 150
        if rem < 15:
            break
        t_alloc = min(40, rem / max(1, len(inits) - i))
        if t_alloc < 8:
            break
        a_opt, _ = eg_optimize(f_init, time_limit=t_alloc,
                               eta0=1.0, temp0=0.005,
                               temp_decay=0.9996, eta_decay=0.99995,
                               top_k_frac=0.3, use_momentum=True,
                               rand_seed=seed + 100 + i)
        try_seq(a_opt, f"init{i}_n{n}")

    # === Phase 5: QP polish on best ===
    print(f"\n=== Phase 5: Second QP Polish ===")
    print(f"  Best: {best_score:.8f}")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 20:
            a_qp2, _ = active_set_qp_polish(best_seq, time_limit=min(rem, 150), top_k=40, outer_iters=8)
            try_seq(a_qp2, "qp_p5")

    # === Phase 6: Different lengths ===
    print(f"\n=== Phase 6: Different Lengths ===")
    print(f"  Best: {best_score:.8f}")

    if best_seq is not None:
        alt_lengths = [n // 2, n * 2, int(n * 0.75), int(n * 1.5),
                       256, 512, 768, 200, 448]
        alt_lengths = [x for x in alt_lengths if 64 <= x <= 1200 and x != n]
        # Remove duplicates
        seen = set()
        alt_lengths_unique = []
        for x in alt_lengths:
            if x not in seen:
                seen.add(x)
                alt_lengths_unique.append(x)
        alt_lengths = alt_lengths_unique

        for li, n_alt in enumerate(alt_lengths):
            rem = deadline - time.monotonic() - 60
            if rem < 20:
                break
            a_alt = interpolate_to_length(best_seq, n_alt)
            t_alloc = min(35, rem / max(1, len(alt_lengths) - li))
            if t_alloc < 10:
                break
            a_opt, _ = eg_optimize(a_alt, time_limit=t_alloc,
                                   eta0=1.0, temp0=0.004,
                                   temp_decay=0.9996, eta_decay=0.99995,
                                   top_k_frac=0.3, use_momentum=True,
                                   rand_seed=seed + 200 + li)
            try_seq(a_opt, f"len{n_alt}_eg")

            # Quick QP on the new length if it improved
            if best_seq is not None and len(best_seq) == n_alt:
                rem2 = deadline - time.monotonic() - 40
                if rem2 > 15:
                    a_qp, _ = active_set_qp_polish(best_seq, time_limit=min(rem2, 60), top_k=25, outer_iters=3)
                    try_seq(a_qp, f"len{n_alt}_qp")

    # === Phase 7: Final EG with fine params ===
    print(f"\n=== Phase 7: Final Fine EG ===")
    print(f"  Best: {best_score:.8f}")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            a_opt, _ = eg_optimize(best_seq, time_limit=min(rem, 60),
                                   eta0=0.3, temp0=0.0003,
                                   temp_decay=0.99998, eta_decay=0.99999,
                                   top_k_frac=0.5, use_momentum=True,
                                   rand_seed=seed + 500)
            try_seq(a_opt, "eg_final_fine")

    # === Phase 8: Final QP + coordinate ===
    print(f"\n=== Phase 8: Final Polish ===")
    print(f"  Best: {best_score:.8f}")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            a_qp, _ = active_set_qp_polish(best_seq, time_limit=min(rem, 40), top_k=50, outer_iters=10)
            try_seq(a_qp, "qp_final")

        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_coord, _ = coordinate_polish(best_seq, time_limit=min(rem, 20), n_sweeps=30)
            try_seq(a_coord, "coord_final")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
