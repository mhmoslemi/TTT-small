def fast_convolve(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def compute_subgradient_topk(a, c, temperature, top_k=None):

    len_c = len(c)

        # Use top-k peaks
        top_indices = np.argpartition(c, -top_k)[-top_k:]
        top_vals = c[top_indices]
    else:
        top_indices = np.arange(len_c)
        top_vals = c
    
    scaled = top_vals / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros(len(top_indices))
        w[np.argmax(top_vals)] = 1.0
    else:
        w /= ws

    for idx in range(len(top_indices)):
        if w[idx] < 1e-12:
            continue
        k = int(top_indices[idx])

        je = min(n - 1, k)
        sg[js:je + 1] += 2.0 * w[idx] * a[k - je:k - js + 1][::-1]
    
    return sg

    """Compute softmax-weighted subgradient of max(c) using top-k peaks."""
    n = len(a)

        w[np.argmax(top_vals)] = 1.0

        je = min(n - 1, k)

def freq_domain_init(n, rng):
    """Generate random initialization from frequency domain."""
    phases = rng.uniform(-np.pi, np.pi, n)
    freqs = np.arange(n, dtype=np.float64)
    envelope = np.exp(-freqs / (n * 0.4))
    magnitudes = envelope * (0.5 + 0.5 * rng.random(n))
    spectrum = magnitudes * np.exp(1j * phases)
    a = np.fft.ifft(spectrum).real
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return a / s


def symmetrize(a):
    """Force palindromic symmetry."""
    n = len(a)
    a_sym = (a + a[::-1]) / 2.0
    a_sym = np.maximum(a_sym, 0)
    s = a_sym.sum()
    if s < 1e-10:
        return a / a.sum()
    return a_sym / s


def peak_shave_local_search(a0, time_limit, n_trials=20000, seed=999):
    """Local search that specifically targets the peak of the convolution."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    n = len(a)

    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    trial = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit and trial < n_trials:
        trial += 1

        k_peak = int(np.argmax(c))
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)

        products = a[js:je + 1] * a[k_peak - je:k_peak - js + 1][::-1]
        n_prod = len(products)

        if n_prod > 3:
            top_few = np.argpartition(products, -min(5, n_prod))[-min(5, n_prod):]
            top_idx = top_few[int(rng.integers(0, len(top_few)))]
        else:
            top_idx = int(rng.integers(0, n_prod))

        j = js + top_idx
        other_j = k_peak - j
        if other_j < 0 or other_j >= n:
            continue

        # Also try reducing other coordinates
        if rng.random() < 0.3:
            j = int(rng.integers(0, n))
            other_j = j

        factor = 0.7 + 0.3 * rng.random()
        if j < n:
            a[j] *= factor
        if other_j != j and other_j < n and other_j >= 0:
            a[other_j] *= (0.85 + 0.15 * rng.random())

        a = np.maximum(a, 0)
        s2 = a.sum()
        if s2 < 1e-10:
            a = best_a.copy()
            c = fast_convolve(a)
            continue
        a = a / s2

        c = fast_convolve(a)
        obj = c.max()

        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            a = best_a.copy()
            c = fast_convolve(a)
            no_improve += 1
            if no_improve > 2000:
                break

    return best_a.copy(), best_obj


def generate_diverse_inits(best_a, rng, n, n_inits=6):
    """Generate diverse initializations."""
    inits = []
    inits.append(best_a.copy() if len(best_a) == n else np.ones(n) / n)
    inits.append(np.ones(n) / n)

    x = np.arange(n, dtype=np.float64)

        inits.append(e / e.sum())

    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)

    inits.append(g / g.sum())

    for _ in range(2):
        r = rng.dirichlet(np.ones(n) * 0.3)
        inits.append(r)

    return inits[:n_inits + 2]

    print(f"\n=== Phase 3: EG Variations ===")
    print(f"  Best so far: {best_score:.8f}")

    configs = [
        {"eta0": 0.5, "temp0": 0.01, "temp_decay": 0.9999, "eta_decay": 0.9999, "top_k_frac": 0.3, "adaptive_temp": True},
        {"eta0": 3.0, "temp0": 0.001, "temp_decay": 0.9995, "eta_decay": 0.9999, "top_k_frac": 0.1, "adaptive_temp": True},
        {"eta0": 1.0, "temp0": 0.005, "temp_decay": 0.99985, "eta_decay": 0.99997, "top_k_frac": 0.2, "poly_decay": True},
        {"eta0": 2.0, "temp0": 0.002, "temp_decay": 0.9999, "eta_decay": 0.99993, "top_k_frac": 0.5, "adaptive_temp": True},
    ]

    for ci, cfg in enumerate(configs):
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t_alloc = min(100, rem / (len(configs) - ci + 2))
        if t_alloc < 15:
            break
        a_opt, _ = eg_optimize(best_seq, time_limit=t_alloc, **cfg)
        do_try(a_opt, f"eg_cfg{ci}")

    # === Phase 4: L-BFGS again on improved seq ===
    print(f"\n=== Phase 4: L-BFGS Round 2 ===")
    print(f"  Best so far: {best_score:.8f}")
    rem = deadline - time.monotonic()
    if rem > 30:
        t_alloc = min(200, rem * 0.15)
        a_opt, _ = l_bfgs_smoothed(best_seq, time_limit=t_alloc,
                                   mu_start=0.02, mu_end=0.0003, n_phases=5)
        do_try(a_opt, "lbfgs_p4")

    # === Phase 5: Diverse initializations ===
    print(f"\n=== Phase 5: Diverse Inits ===")
    print(f"  Best so far: {best_score:.8f}")

    inits = generate_diverse_inits(best_seq, rng, n, n_inits=5)
    for i, f_init in enumerate(inits[1:]):
        rem = deadline - time.monotonic()
        if rem < 20:
            break
        t_alloc = min(50, rem / max(1, len(inits) - 1 - i))
        if t_alloc < 8:
            break
        a_opt, _ = eg_optimize(f_init, time_limit=t_alloc,
                               eta0=1.0, temp0=0.01, temp_decay=0.9998,
                               eta_decay=0.99995, adaptive_temp=True)
        do_try(a_opt, f"init{i}")

    # Frequency domain inits
    for fi in range(2):
        rem = deadline - time.monotonic()
        if rem < 15:
            break
        a_init = freq_domain_init(n, rng)
        t_alloc = min(40, rem / 3)
        if t_alloc < 8:
            break
        a_opt, _ = eg_optimize(a_init, time_limit=t_alloc,
                               eta0=1.5, temp0=0.005, temp_decay=0.9998,
                               eta_decay=0.99995, adaptive_temp=True)
        do_try(a_opt, f"freq{fi}")

    # Symmetric
    rem = deadline - time.monotonic()
    if rem > 20:
        a_sym = symmetrize(best_seq + rng.standard_normal(n) * 0.005)
        t_alloc = min(40, rem / 2)
        a_opt, _ = eg_optimize(a_sym, time_limit=t_alloc,
                               eta0=1.0, temp0=0.005, temp_decay=0.9998,
                               eta_decay=0.99995, adaptive_temp=True)
        do_try(a_opt, "symmetric")

    # === Phase 6: Peak-shaving local search ===
    print(f"\n=== Phase 6: Peak Shaving ===")
    print(f"  Best so far: {best_score:.8f}")
    rem = deadline - time.monotonic()
    if rem > 10:
        t_alloc = min(80, rem * 0.15)
        a_opt, _ = peak_shave_local_search(best_seq, time_limit=t_alloc, n_trials=30000, seed=42)
        do_try(a_opt, "peak_shave_1")

    # === Phase 7: EG polish on peak-shaved result ===
    print(f"\n=== Phase 7: EG Polish ===")
    print(f"  Best so far: {best_score:.8f}")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_alloc = min(100, rem * 0.2)
        a_opt, _ = eg_optimize(best_seq, time_limit=t_alloc,
                               eta0=0.5, temp0=0.001, temp_decay=0.99997,
                               eta_decay=0.99998, top_k_frac=0.1,
                               adaptive_temp=True)
        do_try(a_opt, "eg_polish")

    # === Phase 8: L-BFGS final ===
    print(f"\n=== Phase 8: L-BFGS Final ===")
    print(f"  Best so far: {best_score:.8f}")
    rem = deadline - time.monotonic()
    if rem > 20:
        t_alloc = min(150, rem * 0.2)
        a_opt, _ = l_bfgs_smoothed(best_seq, time_limit=t_alloc,
                                   mu_start=0.01, mu_end=0.0002, n_phases=4)
        do_try(a_opt, "lbfgs_final")

    # === Phase 9: Try different lengths ===
    print(f"\n=== Phase 9: Different Lengths ===")
    print(f"  Best so far: {best_score:.8f}")
    rem = deadline - time.monotonic()

    for n_new in [n * 2, max(64, n // 2), n + 96, max(64, n - 96)]:
        if n_new < 32 or n_new > 2000:
            continue
        rem = deadline - time.monotonic()
        if rem < 15:
            break
        x_old = np.linspace(0, 1, n)
        x_new = np.linspace(0, 1, n_new)
        a_new = np.interp(x_new, x_old, best_seq)
        a_new = np.maximum(a_new, 0)
        s_val = a_new.sum()
        if s_val > 1e-10:
            a_new = a_new / s_val
            t_alloc = min(40, rem / 4)
            if t_alloc < 8:
                break
            a_opt, _ = eg_optimize(a_new, time_limit=t_alloc,
                                   eta0=1.0, temp0=0.005, temp_decay=0.9998,
                                   eta_decay=0.99995, adaptive_temp=True)
            do_try(a_opt, f"len_{n_new}")

    # === Phase 10: Final peak shaving ===
    print(f"\n=== Phase 10: Final Peak Shave ===")
    print(f"  Best so far: {best_score:.8f}")
    rem = deadline - time.monotonic()
    if rem > 5:
        t_alloc = min(40, rem * 0.4)
        a_opt, _ = peak_shave_local_search(best_seq, time_limit=t_alloc, n_trials=20000, seed=777)
        do_try(a_opt, "peak_shave_final")

    # === Phase 11: Ultra-fine EG ===
    rem = deadline - time.monotonic()
    if rem > 5:
        t_alloc = min(20, rem * 0.5)
        a_opt, _ = eg_optimize(best_seq, time_limit=t_alloc,
                               eta0=0.1, temp0=0.0005, temp_decay=0.99999,
                               eta_decay=0.99999, top_k_frac=0.05,
                               adaptive_temp=False)
        do_try(a_opt, "eg_ultra_fine")

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"Fallback score: {best_score:.8f}")

    return [float(x) for x in best_seq]
