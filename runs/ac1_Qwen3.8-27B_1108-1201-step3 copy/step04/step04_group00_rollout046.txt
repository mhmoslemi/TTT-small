"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multiplicative-weights mirror descent,
LP-based Frank-Wolfe, and coordinate descent with multi-length exploration.

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def mirror_descent(f0, time_limit, eta0=0.1, eta_decay=0.9999, n_peaks=5, verbose=False):
    """Multiplicative weights mirror descent for minimizing max auto-convolution."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    eta = eta0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Find top peaks
        k_actual = min(n_peaks, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        
        # Weight by softmax
        pv = conv[peaks]
        max_pv = pv.max()
        if max_pv < 1e-15:
            break
        w = np.exp(-(max_pv - pv) / (0.001 * max_pv))
        w /= w.sum()
        
        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            # sg[j] += 2 * w_i * f[k-j] for valid j
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]
        
        # Multiplicative update
        f = f * np.exp(-eta * sg)
        s = f.sum()
        if s < 1e-15:
            f = best_f.copy()
            eta *= 0.5
            continue
        f = f / s
        
        conv = fast_convolve(f)
        M = conv.max()
        
        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        eta *= eta_decay
        
        if no_improve > 200:
            # Restart with perturbation
            rng = np.random.default_rng(it * 13 + 7)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.001, 0)
            f = f / f.sum()
            eta = eta0 * 0.5
            no_improve = 0
        
        if verbose and it % 1000 == 0:
            print(f"    MD iter {it}: obj={best_obj:.10f}, eta={eta:.6f}")
    
    return best_f, best_obj

def solve_lp_direction(f, M, n):
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def lp_fw_optimize(f0, time_limit, verbose=False):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()

        f = np.ones(len(f0)) / len(f0)

    it = 0
    m_exp = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        g = solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.15
            if m_exp > 5.0:
                m_exp = 1.0
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue
        
        # Line search (coarse + fine)
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        
        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            f = f_new
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 4.0)
        
        if no_improve > 50:
            rng = np.random.default_rng(it * 31 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0
    
    return best_f, best_obj

def coordinate_descent(f0, time_limit):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue
            
            cur_conv = fast_convolve(f)
            cur_peak = cur_conv.max()
            best_j_val = old_val
            best_j_obj = cur_peak
            
            # Try a few multipliers
            for mult in [0.5, 0.7, 0.85, 1.15, 1.3, 1.7]:
                new_val = old_val * mult
                if new_val <= 0 or new_val >= 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved and passes > 2:
            break
    
    return best_f, best_obj

def cosine_window(n):
    """Generate a cosine window."""
    x = np.arange(n)
    w = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    w = np.maximum(w, 1e-10)
    return w / w.sum()

def gaussian_window(n, sigma_frac=0.3):
    """Generate a Gaussian window."""
    x = np.arange(n)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    w = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    return w / w.sum()

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multiplicative-weights mirror descent,
LP-based Frank-Wolfe, coordinate descent, and multi-length exploration with adaptive
restarts. Combines fast subgradient iterations with expensive LP steps for global
progress and local polish.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def mirror_descent(f0, time_limit, eta0=0.05, eta_decay=0.99995, n_peaks=8, verbose=False):
    """Multiplicative weights mirror descent for minimizing max auto-convolution."""
    f = np.maximum(f0.astype(np.float64), 0.0)

    else:
        f = f / s
    n = len(f)

    M = conv.max()
    best_obj = M
    best_f = f.copy()

    it = 0
    no_improve = 0
    eta = eta0

        it += 1

        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        max_pv = pv.max()

            break
        temp = 0.002 * max_pv
        w = np.exp(-(max_pv - pv) / temp)

            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute subgradient (convex combination of peak subgradients)
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

            f = best_f.copy()
            eta *= 0.5
            continue
        f = f / s

            no_improve = 0

def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def coordinate_descent(f0, time_limit):

    f = np.maximum(f0.astype(np.float64), 0.0)

        improved = False

            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue

            cur_conv = fast_convolve(f)
            cur_peak = cur_conv.max()
            best_j_val = old_val
            best_j_obj = cur_peak

            for mult in [0.5, 0.7, 0.85, 1.15, 1.35, 1.7]:
                new_val = old_val * mult
                if new_val <= 0 or new_val >= 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved and passes > 2:
            break

    return best_f, best_obj


def cosine_window(n):
    """Generate a cosine (Hann) window normalized to sum 1."""
    x = np.arange(n)
    w = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    w = np.maximum(w, 1e-10)
    return w / w.sum()


def gaussian_window(n, sigma_frac=0.28):
    """Generate a Gaussian window."""
    x = np.arange(n)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    w = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    return w / w.sum()


def interpolate_sequence(f, new_n):
    """Linear interpolation to change sequence length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def projected_gradient_step(f, sg, eta):
    """Projected gradient (additive) step onto the simplex."""
    f_new = f - eta * sg
    f_new = project_simplex(f_new)
    return f_new

                print(f"  n={target_n} LP: {best_score:.6f}")

    # === Phase 4: Deep LP-FW with diverse inits ===
    print(f"\n=== Phase 4: Deep LP-FW (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for p in range(6):
            rem = deadline - time.monotonic() - 100
            if rem < 15:
                break
            t_alloc = min(rem * 0.25, 90)

            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                noise = rng.standard_normal(n_best) * 0.003
                f_init = project_simplex(best_seq + noise)
            elif p == 2:
                f_init = 0.5 * best_seq + 0.5 * best_seq[::-1]
                f_init = project_simplex(f_init)
            elif p == 3:
                cw = cosine_window(n_best)
                f_init = 0.6 * best_seq + 0.4 * cw
                f_init = project_simplex(f_init)
            elif p == 4:
                gw = gaussian_window(n_best, 0.25)
                f_init = 0.7 * best_seq + 0.3 * gw
                f_init = project_simplex(f_init)
            else:
                f_init = 0.85 * best_seq + 0.15 * (np.ones(n_best) / n_best)
                f_init = project_simplex(f_init)

            a_opt, _ = lp_fw_optimize(f_init, time_limit=t_alloc)
            sc = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  Deep {p}: {best_score:.6f}")

    # === Phase 5: Coordinate descent ===
    print(f"\n=== Phase 5: Coordinate Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 50
        if rem > 20:
            a_opt, _ = coordinate_descent(best_seq, time_limit=min(rem, 100))
            sc = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  CD: {best_score:.6f}")

    # === Phase 6: Final fine-tuning ===
    print(f"\n=== Phase 6: Final Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        # Fine mirror descent
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            a_opt, _ = mirror_descent(best_seq, time_limit=min(rem, 50),
                                       eta0=0.008, eta_decay=0.99999, n_peaks=5, verbose=False)
            sc = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  MD final: {best_score:.6f}")

        # Final LP-FW
        rem = deadline - time.monotonic() - 10
        if rem > 8:
            a_opt, _ = lp_fw_optimize(best_seq, time_limit=min(rem, 30))
            sc = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_opt.copy()
                print(f"  LPFW final: {best_score:.6f}")

        # Tiny perturbations
        rem = deadline - time.monotonic() - 5
        if rem > 8 and best_seq is not None:
            n_best = len(best_seq)
            for _ in range(3):
                rem = deadline - time.monotonic() - 3
                if rem < 5:
                    break
                noise = rng.standard_normal(n_best) * 0.0008
                f_pert = project_simplex(best_seq + noise)
                a_opt, _ = mirror_descent(f_pert, time_limit=min(rem, 15),
                                           eta0=0.005, eta_decay=0.99999, n_peaks=3)
                sc = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = a_opt.copy()
                    print(f"  Perturb: {best_score:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = cosine_window(384)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
