"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth-max mirror descent
with temperature annealing, combined with FW-LP and peak equalization.
"""

import numpy as np
from scipy.optimize import linprog
import time
import sys

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_max_grad(conv, t):
    """Compute the softmax weights w_k = exp(conv[k]/t) / sum exp(conv[k']/t)"""
    m = conv.max()
    exp_vals = np.exp((conv - m) / t)
    w = exp_vals / exp_vals.sum()
    return w

def compute_grad_a(a, w):
    """Compute grad_a[j] = sum_k w[k] * a[k-j] for valid k-j."""
    n = len(a)
    grad_a = np.zeros(n)
    for j in range(n):
        grad_a[j] = np.dot(a, w[j:j+n])
    return grad_a

def smooth_max_mirror_descent(f0, time_limit, t_init=0.05, t_min=0.001,
                                t_decay=0.999, lr_init=0.1, lr_decay=0.9995,
                                verbose=True):
    """
    Mirror descent (multiplicative weights) with smooth max objective.
    Temperature annealing from t_init to t_min.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    cur_t = t_init
    cur_lr = lr_init
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        
        # Compute smooth max weights
        w = smooth_max_grad(conv, cur_t)
        
        # Compute gradient in a-space
        grad_a = compute_grad_a(f, w)
        
        # Multiplicative weight update: f_new = f * exp(-lr * grad_a)
        # Then normalize to simplex
        scale = cur_lr * (grad_a - grad_a.mean())  # center for stability
        f_new = f * np.exp(-scale)
        f_new = np.maximum(f_new, 1e-30)
        f_new = f_new / f_new.sum()
        
        # Line search: try a few step sizes
        best_f_ls = f_new
        best_obj_ls = conv.max()
        for alpha_mult in [0.25, 0.5, 1.0, 2.0]:
            scale_ls = cur_lr * alpha_mult * (grad_a - grad_a.mean())
            f_ls = f * np.exp(-scale_ls)
            f_ls = np.maximum(f_ls, 1e-30)
            f_ls = f_ls / f_ls.sum()
            obj_ls = fast_convolve(f_ls).max()
            if obj_ls < best_obj_ls:
                best_obj_ls = obj_ls
                best_f_ls = f_ls
        
        f = best_f_ls
        conv = fast_convolve(f)
        obj = conv.max()
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"    SM-MD it={it}: obj={best_obj:.8f}, t={cur_t:.6f}, lr={cur_lr:.6f}")
        else:
            no_improve += 1
        
        # Anneal
        cur_t = max(t_min, cur_t * t_decay)
        cur_lr *= lr_decay
        
        # Restart if stuck
        if no_improve > 100:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 1000000)
            noise_scale = 0.002 + 0.003 * min(1.0, no_improve / 500.0)
            f = best_f * np.exp(rng.standard_normal(n) * noise_scale)
            f = f / f.sum()
            cur_lr = max(lr_init * 0.5, 0.01)
            no_improve = 0
    
    return best_f, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth-max L-BFGS with temperature annealing (main optimizer)
2. FW-LP refinement (Frank-Wolfe with LP oracle)
3. Mirror descent with smooth max gradient (polish)
4. Fine subgradient polish
The smooth-max L-BFGS approach is the key innovation: it provides a differentiable
objective that L-BFGS can exploit with approximate second-order information,
leading to faster convergence than pure subgradient methods.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp(f, M):
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def fw_lp_optimize(f0, time_limit, verbose=False):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        g = solve_lp(f, M * m_exp)
        if g is None:
            m_exp = min(m_exp * 1.2, 5.0)
            if m_exp > 5.0:
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 40:
                m_exp = min(m_exp * 1.05, 3.0)
                continue

        best_alpha = 0.0
        best_val = M
        for alpha in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(15):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

def smooth_max_lbfgs(f0, time_limit, t_init=0.1, t_min=0.003, t_decay=0.9995,
                     max_inner_iter=200, verbose=True):
    """
    Smooth max optimization using L-BFGS-B with softmax parameterization.
    Temperature annealing from t_init to t_min.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    # Initialize x from f
    f_safe = np.maximum(f, 1e-12)
    x = np.log(f_safe)
    x = x - x.max()  # center for numerical stability
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    cur_t = t_init
    
    def make_objective(t_val):
        def obj_and_grad(x_val):
            x_shifted = x_val - x_val.max()
            a = np.exp(x_shifted)
            a_sum = a.sum()
            a = a / a_sum
            
            conv_local = fast_convolve(a)
            m = conv_local.max()
            exp_vals = np.exp((conv_local - m) / t_val)
            obj = t_val * np.log(exp_vals.sum())
            w = exp_vals / exp_vals.sum()
            
            # Compute grad_a[j] = sum_d a[d] * w[j+d]
            grad_a = np.empty(n)
            for j in range(n):
                grad_a[j] = np.dot(a, w[j:j+n])
            
            # Chain rule for softmax
            mean_grad = np.dot(grad_a, a)
            grad_x = a * (grad_a - mean_grad)
            
            return obj, grad_x
        return obj_and_grad
    
    it_outer = 0
    while time.monotonic() - t0 < time_limit:
        it_outer += 1
        
        if verbose and it_outer % 10 == 1:
            print(f"    SM-LBFGS outer={it_outer}, t={cur_t:.6f}, best={best_obj:.8f}")
        
        obj_func = make_objective(cur_t)
        
        # Run L-BFGS-B from current x
        try:
            res = minimize(obj_func, x, method='L-BFGS-B',
                          options={'maxiter': max_inner_iter, 'gtol': 1e-10})
            if res.x is not None:
                x = res.x
        except Exception:
            pass
        
        # Convert back to a
        x_shifted = x - x.max()
        a = np.exp(x_shifted)
        a_sum = a.sum()
        if a_sum < 1e-15:
            break
        a = a / a_sum
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = a.copy()
        
        # Anneal temperature
        cur_t = max(t_min, cur_t * t_decay)
        
        # If temperature is very small and we've converged, try to continue
        if cur_t <= t_min * (1 + 1e-6):
            # Try to continue with even smaller t for a bit
            cur_t = max(0.0001, cur_t * 0.9)
    
    return best_f, best_obj


def smooth_max_mirror_descent(f0, time_limit, t_init=0.05, t_min=0.002,
                               t_decay=0.9995, lr_init=0.05, lr_decay=0.9998,
                               verbose=True):
    """Mirror descent with smooth max and adaptive line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    cur_t = t_init
    cur_lr = lr_init
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        m = conv.max()
        exp_vals = np.exp((conv - m) / cur_t)
        w = exp_vals / exp_vals.sum()
        
        # Compute gradient
        grad_a = np.empty(n)
        for j in range(n):
            grad_a[j] = np.dot(f, w[j:j+n])
        
        # Center gradient for numerical stability
        grad_centered = grad_a - np.dot(grad_a, f)
        
        # Try multiple step sizes (line search)
        best_f_new = f.copy()
        best_obj_new = best_obj
        
        for alpha_mult in [0.5, 1.0, 2.0, 0.25]:
            lr_try = cur_lr * alpha_mult
            f_try = f * np.exp(-lr_try * grad_centered)
            f_try = np.maximum(f_try, 1e-30)
            f_try = f_try / f_try.sum()
            obj_try = fast_convolve(f_try).max()
            if obj_try < best_obj_new:
                best_obj_new = obj_try
                best_f_new = f_try.copy()
        
        f = best_f_new
        obj = fast_convolve(f).max()
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        cur_t = max(t_min, cur_t * t_decay)
        cur_lr *= lr_decay
        
        if no_improve > 100:
            rng = np.random.default_rng(hash((it, int(time.monotonic()*100))) % 2**31)
            noise_scale = 0.003 + 0.005 * min(1.0, no_improve / 400.0)
            f = best_f * np.exp(rng.standard_normal(n) * noise_scale)
            f = f / f.sum()
            cur_lr = max(lr_init * 0.5, 0.005)
            no_improve = 0
    
    return best_f, best_obj


def fine_sg_polish(f0, time_limit, lr=0.001, mom=0.95, top_k=30, temp=0.00005):

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()

    cur_lr = lr
    
    t0 = time.monotonic()
    no_improve = 0

        conv = fast_convolve(f)
        m = conv.max()
        exp_vals = np.exp((conv - m) / cur_t)
        w = exp_vals / exp_vals.sum()

        if top_k < len(conv):
            top_idx = np.argpartition(conv, -top_k)[-top_k:]
            w_sparse = np.zeros_like(w)
            w_sparse[top_idx] = w[top_idx]

            w_sparse = w

        for j in range(n):
            sg[j] = np.dot(f, w_sparse[j:j+n])
        
        sg_centered = sg - np.dot(sg, f)
        v = mom * v - cur_lr * sg_centered
        f = project_simplex(f + v)

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()

            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0002)
            no_improve = 0
    
    return best_f, best_obj

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Length exploration with quick FW-LP

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp(f, M):
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):

    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 4.0)

        if no_improve > 60:
            rng = np.random.default_rng(it * 31 + 17 + int(time.monotonic() * 1000) % 100000)
            scale = 0.002 + 0.004 * min(1.0, no_improve / 200.0)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

        if verbose and it % 50 == 0:
            print(f"    FW-LP it={it}, M={M:.10f}, best={best_obj:.10f}")

    return best_f, best_obj


def smooth_max_obj_grad(x, n, t_val):
    """Compute smooth max objective and gradient for L-BFGS-B.
    x is the log-parameter (softmax gives a).
    Returns (objective, gradient).
    """
    x_shifted = x - x.max()
    a = np.exp(x_shifted)
    a_sum = a.sum()
    if a_sum < 1e-300:
        return 1e10, np.zeros(n)
    a = a / a_sum

    conv = fast_convolve(a)
    m = conv.max()
    exp_vals = np.exp((conv - m) / t_val)
    obj = t_val * np.log(exp_vals.sum())
    w = exp_vals / exp_vals.sum()

    # grad_a[j] = sum_d a[d] * w[j+d]
    grad_a = np.empty(n)
    for j in range(n):
        grad_a[j] = np.dot(a, w[j:j + n])

    # Chain rule for softmax: d(a)/d(x_j) = a_j * (e_j - a)
    mean_grad = np.dot(grad_a, a)
    grad_x = a * (grad_a - mean_grad)

    return obj, grad_x


def smooth_max_lbfgs(f0, time_limit, t_init=0.1, t_min=0.003, t_decay=0.9995,
                     max_inner_iter=300, verbose=True):
    """Smooth max L-BFGS-B with temperature annealing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    f_safe = np.maximum(f, 1e-12)
    x = np.log(f_safe)
    x = x - x.max()

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    cur_t = t_init
    it_outer = 0

    while time.monotonic() - t0 < time_limit:
        it_outer += 1
        if verbose and it_outer % 5 == 1:
            print(f"    SM-LBFGS outer={it_outer}, t={cur_t:.6f}, best={best_obj:.10f}")

        try:
            res = minimize(smooth_max_obj_grad, x, args=(n, cur_t),
                           method='L-BFGS-B', jac=True,
                           options={'maxiter': max_inner_iter, 'gtol': 1e-12,
                                     'ftol': 1e-14, 'maxfun': max_inner_iter * 3})
            if res.x is not None and np.isfinite(res.x).all():
                x = res.x
        except Exception as e:
            if verbose:
                print(f"    L-BFGS error: {e}")

        # Convert back
        x_shifted = x - x.max()
        a = np.exp(x_shifted)
        a_sum = a.sum()
        if a_sum < 1e-15:
            break
        a = a / a_sum
        a = np.maximum(a, 0.0)
        a = a / a.sum()

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = a.copy()

        cur_t = max(t_min, cur_t * t_decay)

        # If at minimum temperature, try even smaller
        if cur_t <= t_min * (1 + 1e-6):
            cur_t = max(0.0001, cur_t * 0.95)

    return best_f, best_obj


def smooth_max_mirror_descent(f0, time_limit, t_init=0.05, t_min=0.001,
                              t_decay=0.9995, lr_init=0.05, lr_decay=0.9999,
                              verbose=True):
    """Mirror descent (multiplicative weights) with smooth max gradient."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    cur_t = t_init
    cur_lr = lr_init

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        m = conv.max()
        exp_vals = np.exp((conv - m) / cur_t)
        w = exp_vals / exp_vals.sum()

        grad_a = np.empty(n)
        for j in range(n):
            grad_a[j] = np.dot(f, w[j:j + n])

        grad_centered = grad_a - np.dot(grad_a, f)

        # Line search over step sizes
        best_f_new = f.copy()
        best_obj_new = m
        for alpha_mult in [0.5, 1.0, 2.0, 0.25, 4.0]:
            lr_try = cur_lr * alpha_mult
            f_try = f * np.exp(-lr_try * grad_centered)
            f_try = np.maximum(f_try, 1e-30)
            f_try = f_try / f_try.sum()
            obj_try = fast_convolve(f_try).max()
            if obj_try < best_obj_new:
                best_obj_new = obj_try
                best_f_new = f_try.copy()

        f = best_f_new
        obj = fast_convolve(f).max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_t = max(t_min, cur_t * t_decay)
        cur_lr *= lr_decay

        if verbose and it % 200 == 0:
            print(f"    SM-MD it={it}, obj={best_obj:.10f}, t={cur_t:.6f}")

        if no_improve > 150:
            rng = np.random.default_rng(hash((it, n)) % (2**31))
            noise_scale = 0.003 + 0.005 * min(1.0, no_improve / 500.0)
            f = best_f * np.exp(rng.standard_normal(n) * noise_scale)
            f = f / f.sum()
            cur_lr = max(lr_init * 0.5, 0.005)
            no_improve = 0

    return best_f, best_obj


def fine_sg_polish(f0, time_limit, lr=0.001, mom=0.95, top_k=40, temp=0.00005):
    """Fine subgradient polish with smooth max weighting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    cur_t = temp
    cur_lr = lr

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        m = conv.max()
        exp_vals = np.exp((conv - m) / cur_t)
        w = exp_vals / exp_vals.sum()

        sg = np.zeros(n)
        for j in range(n):
            sg[j] = np.dot(f, w[j:j + n])

        sg_centered = sg - np.dot(sg, f)
        v = mom * v - cur_lr * sg_centered
        f = project_simplex(f + v)

        obj = fast_convolve(f).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9997
        cur_t *= 1.0003

        if no_improve > 300:
            rng = np.random.default_rng(hash((no_improve, n)) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0002)
            no_improve = 0

    return best_f, best_obj


def generate_structured_init(n, rng):
    """Generate a structured initialization."""
    x = np.arange(n, dtype=np.float64)
    # Edge-weighted
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    a = edge_w + 0.1 * rng.standard_normal(n)
    a = np.maximum(a, 0)
    a = a / a.sum()
    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search combining smooth-max L-BFGS,
    FW-LP, mirror descent, and fine subgradient polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}")
            try_seq(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(384) / 384

    n_p = len(parent)

    # === Phase 0: Quick length exploration (~80s) ===
    print("\n=== Phase 0: Length Exploration ===")
    rem = deadline - time.monotonic()
    exp_n = [n_p]
    if n_p <= 512:
        exp_n.append(512)
    if n_p <= 640:
        exp_n.append(640)

    for n_try in exp_n:
        rem = deadline - time.monotonic()
        if rem < 20:
            break
        if n_try == n_p and best_seq is not None and len(best_seq) == n_p:
            continue  # Already have parent at this length
        t_alloc = min(25, rem - 500)
        if t_alloc < 10:
            continue
        print(f"  Exploring n={n_try}, t={t_alloc:.0f}s")
        init = np.ones(n_try) / n_try
        if len(parent) != n_try:
            # Interpolate parent
            px = np.interp(np.linspace(0, 1, n_try),
                           np.linspace(0, 1, len(parent)), parent)
            px = np.maximum(px, 0)
            if px.sum() > 1e-10:
                init = px / px.sum()
        a_opt, _ = fw_lp_optimize(init, time_limit=t_alloc)
        try_seq(a_opt, f"explore_n{n_try}")

    # Use best_seq for rest of optimization
    if best_seq is None:
        best_seq = parent
    n_best = len(best_seq)
    print(f"  Proceeding with n={n_best}, score={best_score:.8f}")

    # === Phase 1: Smooth-max L-BFGS main (~500s) ===
    print("\n=== Phase 1: SM-LBFGS Main ===")
    rem = deadline - time.monotonic()
    t1 = min(500, rem * 0.35)
    if t1 > 60:
        print(f"  SM-LBFGS, t={t1:.0f}s")
        a_opt, _ = smooth_max_lbfgs(best_seq, time_limit=t1,
                                     t_init=0.1, t_min=0.005, t_decay=0.9995,
                                     max_inner_iter=200)
        try_seq(a_opt, "sm_lbfgs_p1")

    # === Phase 2: FW-LP deep (~400s) ===
    print("\n=== Phase 2: FW-LP Deep ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t2 = min(400, rem * 0.3)
        if t2 > 30:
            print(f"  FW-LP, t={t2:.0f}s")
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=t2, verbose=True)
            try_seq(a_opt, "fw_lp_p2")

    # === Phase 3: SM-LBFGS round 2 (~250s) ===
    print("\n=== Phase 3: SM-LBFGS Round 2 ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t3 = min(250, rem * 0.25)
        if t3 > 30:
            print(f"  SM-LBFGS r2, t={t3:.0f}s")
            a_opt, _ = smooth_max_lbfgs(best_seq, time_limit=t3,
                                          t_init=0.05, t_min=0.002, t_decay=0.9995,
                                          max_inner_iter=300)
            try_seq(a_opt, "sm_lbfgs_p3")

    # === Phase 4: SM Mirror Descent (~200s) ===
    print("\n=== Phase 4: SM Mirror Descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t4 = min(200, rem * 0.2)
        if t4 > 20:
            print(f"  SM-MD, t={t4:.0f}s")
            a_opt, _ = smooth_max_mirror_descent(best_seq, time_limit=t4,
                                                   t_init=0.03, t_min=0.001,
                                                   lr_init=0.04, lr_decay=0.9999)
            try_seq(a_opt, "sm_md_p4")

    # === Phase 5: Fine polish (~150s) ===
    print("\n=== Phase 5: Fine Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t5 = min(100, rem * 0.4)
        if t5 > 15:
            print(f"  Fine SG, t={t5:.0f}s")
            a_opt, _ = fine_sg_polish(best_seq, time_limit=t5,
                                       lr=0.0005, mom=0.96, top_k=n_best // 3,
                                       temp=0.00003)
            try_seq(a_opt, "fine_p5")

        # Micro perturbations + quick polish
        rem = deadline - time.monotonic()
        if rem > 20:
            n_perturb = max(3, int(rem / 12))
            for pi in range(n_perturb):
                rem = deadline - time.monotonic()
                if rem < 8:
                    break
                noise_scale = 0.0005 + 0.001 * (pi % 3)
                noise = rng.standard_normal(n_best) * noise_scale
                f_pert = project_simplex(best_seq + noise)
                a_opt, _ = fine_sg_polish(f_pert, time_limit=min(rem, 12),
                                            lr=0.0003, mom=0.97,
                                            top_k=n_best // 2, temp=0.00002)
                try_seq(a_opt, f"micro_{pi}")

    # === Final: try structured inits if time remains ===
    rem = deadline - time.monotonic()
    if rem > 40 and best_seq is not None:
        print("\n=== Final: Structured Inits ===")
        for si in range(3):
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            init = generate_structured_init(n_best, rng)
            a_opt, _ = fine_sg_polish(init, time_limit=min(rem, 12),
                                       lr=0.001, mom=0.95, top_k=n_best // 3,
                                       temp=0.0001)
            try_seq(a_opt, f"struct_{si}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
