import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = (u * np.arange(1, n + 1)) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_grad(weights, f):
    """Compute gradient of LSE objective. weights has length 2n-1, f has length n."""
    n = len(f)
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]

        M = conv.max()

        g = solve_lp_direction(f, M)
        if g is None or g.sum() < 1e-15:
            no_improve += 1
            if no_improve > 30:
                break
            continue

        g = g / g.sum()

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                break
            continue

        # Line search: evaluate on grid then golden section
        best_alpha = 0.0
        best_val = M
        alphas = np.linspace(0.02, 1.0, 20)
        for alpha in alphas:
            ft = np.maximum((1.0 - alpha) * f + alpha * g, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Golden section refinement
        if 0 < best_alpha < 1.0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            for _ in range(15):
                m1 = hi - gr * (hi - lo)
                m2 = lo + gr * (hi - lo)
                ft1 = np.maximum((1.0 - m1) * f + m1 * g, 0.0)
                ft1 = ft1 / ft1.sum()
                v1 = fast_convolve(ft1).max()
                ft2 = np.maximum((1.0 - m2) * f + m2 * g, 0.0)
                ft2 = ft2 / ft2.sum()
                v2 = fast_convolve(ft2).max()
                if v1 < best_val:
                    best_val = v1
                    best_alpha = m1
                if v2 < best_val:
                    best_val = v2
                    best_alpha = m2
                if v1 < v2:
                    hi = m2
                else:
                    lo = m1

        if best_alpha > 1e-10:
            f = np.maximum((1.0 - best_alpha) * f + best_alpha * g, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 50:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.001 + 0.002 * min(1.0, (no_improve - 50) / 50.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0

        if it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW-LP it={it}, obj={best_obj:.12f}, elapsed={elapsed:.0f}s")

    return best_f, best_obj


def lse_nesterov(f0, time_limit, beta_start=25.0, beta_end=10000.0):
    """Nesterov-accelerated LSE gradient descent with adaptive beta."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = 1.0 / np.sqrt(n)
    t_param = 1.0
    y = f.copy()
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    beta_rate = 1.004

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param)) / 2.0
        momentum = (t_param - 1.0) / t_new
        y = project_simplex(f + momentum * (f - y + lr * v))
        # Actually, let's use the standard Nesterov formulation
        # y = f + (t_old - 1)/(t_new) * (f - f_prev)
        # But simpler: just use heavy-ball momentum which works well in practice
        y = f  # fallback to standard momentum

        conv_y = fast_convolve(y)
        bc = beta * conv_y
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()

        grad = compute_lse_grad(w, y)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-18:
            no_improve += 100
            break
        grad = grad / grad_norm

        # Adaptive learning rate based on objective curvature estimate
        v = 0.9 * v - lr * grad
        f_new = project_simplex(y + v)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.001, 3.0 / np.sqrt(n))
        else:
            no_improve += 1
            if no_improve >= 15:
                lr *= 0.93
            if no_improve > 100:
                f = best_f.copy()
                v = np.zeros(n)
                lr = max(0.5 / np.sqrt(n), 0.001)
                beta = min(beta * 1.3, beta_end)
                no_improve = 0

        # Ramp up beta
        if beta < beta_end:
            beta *= beta_rate
            # Increase beta rate as we get closer to convergence
            if no_improve > 50:
                beta_rate = 1.008

        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE it={it}, obj={best_obj:.12f}, beta={beta:.1f}, lr={lr:.6f}, elapsed={elapsed:.0f}s")

    return best_f, best_obj


def peak_targeting(f0, time_limit, lr_init=0.005, n_peaks=16):
    """Explicitly target and reduce the top convolution peaks via subgradient steps."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        it += 1

        conv = fast_convolve(f)
        peaks = np.argpartition(conv, -n_peaks)[-n_peaks:]

        # Compute weighted subgradient from top peaks
        pv = conv[peaks]
        # Use softmax weighting over peaks
        scaled = (pv - pv.max()) * 50.0
        w = np.exp(scaled)
        w = w / w.sum()

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            # subgradient at position k: sg[j] += 2*w[i]*(f[k-j] + f[j-k]) for valid indices
            # More carefully: d/d(f_j) of conv(f,f)_k = f[k-j] (if 0<=k-j<n) + f[j-k] (if 0<=j-k<n)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            for j in idx:
                term = 0.0
                k_minus_j = k - j
                if 0 <= k_minus_j < n:
                    term += f[k_minus_j]
                j_minus_k = j - k
                if 0 <= j_minus_k < n:
                    term += f[j_minus_k]
                sg[j] += 2.0 * w[i] * term

        # Normalize and step
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-18:
            break
        sg = sg / sg_norm

        f_new = project_simplex(f - lr * sg)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 20 == 0:
                lr *= 0.95
            if no_improve > 200:
                rng = np.random.default_rng(it * 13 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.0005)
                no_improve = 0
                lr = lr_init * 0.5

        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Peak it={it}, obj={best_obj:.12f}, lr={lr:.6f}, elapsed={elapsed:.0f}s")

    return best_f, best_obj


def try_sequence(a, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.maximum(a, 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Improved multi-phase optimization:
    Phase 1: FW-LP (no relaxation) for global structure
    Phase 2: LSE Nesterov polishing
    Phase 3: Peak targeting for precision
    Phase 4: FW-LP refinement
    Phase 5: Final LSE + peak targeting
    Phase 6: Alternative length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_tracker = [float('inf'), None]

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, raw score={evaluate_sequence(parent.tolist()):.8f}")
            try_sequence(parent, best_tracker)
    except Exception as e:
        print(f"No parent available: {e}")

    if parent is None:
        parent = np.ones(448) / 448
        try_sequence(parent, best_tracker)

    n_parent = len(parent)
    print(f"Starting optimization with n={n_parent}")

    # === Phase 1: FW-LP ===
    t1_start = time.monotonic()
    t1_alloc = min(200, deadline - t1_start - 800)
    if t1_alloc > 30:
        print(f"\n=== Phase 1: FW-LP ({t1_alloc:.0f}s) ===")
        a_opt, obj = fw_lp_optimize(parent, time_limit=t1_alloc)
        try_sequence(a_opt, best_tracker)

    # === Phase 2: LSE Polishing ===
    current = best_tracker[1] if best_tracker[1] is not None else parent
    t2_start = time.monotonic()
    t2_alloc = min(400, deadline - t2_start - 500)
    if t2_alloc > 30:
        print(f"\n=== Phase 2: LSE Nesterov ({t2_alloc:.0f}s) ===")
        a_opt, obj = lse_nesterov(current, time_limit=t2_alloc,
                                  beta_start=20.0, beta_end=8000.0)
        try_sequence(a_opt, best_tracker)

    # === Phase 3: Peak Targeting ===
    current = best_tracker[1] if best_tracker[1] is not None else parent
    t3_start = time.monotonic()
    t3_alloc = min(150, deadline - t3_start - 350)
    if t3_alloc > 20:
        print(f"\n=== Phase 3: Peak Targeting ({t3_alloc:.0f}s) ===")
        a_opt, obj = peak_targeting(current, time_limit=t3_alloc,
                                    lr_init=0.003, n_peaks=24)
        try_sequence(a_opt, best_tracker)

    # === Phase 4: FW-LP Refinement ===
    current = best_tracker[1] if best_tracker[1] is not None else parent
    t4_start = time.monotonic()
    t4_alloc = min(150, deadline - t4_start - 250)
    if t4_alloc > 20:
        print(f"\n=== Phase 4: FW-LP Refinement ({t4_alloc:.0f}s) ===")
        a_opt, obj = fw_lp_optimize(current, time_limit=t4_alloc)
        try_sequence(a_opt, best_tracker)

    # === Phase 5: Final LSE with very high beta ===
    current = best_tracker[1] if best_tracker[1] is not None else parent
    t5_start = time.monotonic()
    t5_alloc = min(200, deadline - t5_start - 100)
    if t5_alloc > 20:
        print(f"\n=== Phase 5: Final LSE ({t5_alloc:.0f}s) ===")
        a_opt, obj = lse_nesterov(current, time_limit=t5_alloc,
                                  beta_start=200.0, beta_end=20000.0)
        try_sequence(a_opt, best_tracker)

    # === Phase 6: Peak targeting final ===
    current = best_tracker[1] if best_tracker[1] is not None else parent
    t6_start = time.monotonic()
    t6_alloc = min(80, deadline - t6_start - 60)
    if t6_alloc > 10:
        print(f"\n=== Phase 6: Final Peak Targeting ({t6_alloc:.0f}s) ===")
        a_opt, obj = peak_targeting(current, time_limit=t6_alloc,
                                    lr_init=0.001, n_peaks=32)
        try_sequence(a_opt, best_tracker)

    # === Phase 7: Alternative lengths ===
    print(f"\n=== Phase 7: Alternative Lengths ===")
    current = best_tracker[1] if best_tracker[1] is not None else parent
    alt_lengths = [384, 512, 640]
    if n_parent in alt_lengths:
        alt_lengths.remove(n_parent)

    for n_alt in alt_lengths[:2]:
        t_remaining = deadline - time.monotonic() - 30
        if t_remaining < 50:
            break
        t_alloc = min(100, t_remaining / 2.0)
        if t_alloc < 25:
            break

        print(f"  Trying n={n_alt}, budget={t_alloc:.0f}s")
        # Interpolate
        x_old = np.linspace(0, 1, len(current))
        x_new = np.linspace(0, 1, n_alt)
        f_interp = np.interp(x_new, x_old, current)
        f_interp = np.maximum(f_interp, 0)
        if f_interp.sum() > 1e-10:
            f_interp = f_interp / f_interp.sum()
        try_sequence(f_interp, best_tracker)

        # Quick FW-LP + LSE
        t_fw = t_alloc * 0.4
        t_lse = t_alloc * 0.5
        a_opt, _ = fw_lp_optimize(f_interp, time_limit=t_fw)
        try_sequence(a_opt, best_tracker)

        cur_alt = best_tracker[1] if (best_tracker[1] is not None and len(best_tracker[1]) == n_alt) else f_interp
        a_opt, _ = lse_nesterov(cur_alt, time_limit=t_lse,
                                beta_start=30.0, beta_end=5000.0)
        try_sequence(a_opt, best_tracker)

    # === Final: alternate between LSE and peak targeting on best ===
    current = best_tracker[1] if best_tracker[1] is not None else parent
    t_final = deadline - time.monotonic() - 10
    if t_final > 20:
        print(f"\n=== Final Polish ({t_final:.0f}s) ===")
        half = t_final / 2.0
        if half > 10:
            a_opt, _ = lse_nesterov(current, time_limit=half,
                                    beta_start=500.0, beta_end=30000.0)
            try_sequence(a_opt, best_tracker)
        current = best_tracker[1] if best_tracker[1] is not None else parent
        t_left = deadline - time.monotonic() - 10
        if t_left > 10:
            a_opt, _ = peak_targeting(current, time_limit=t_left,
                                      lr_init=0.0005, n_peaks=40)
            try_sequence(a_opt, best_tracker)

    # Final verification
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
            return [float(x) for x in final_seq]

    # Fallback
    fallback = np.ones(448) / 448
    score = float(evaluate_sequence(fallback.tolist()))
    print(f"\nFALLBACK: score={score:.8f}, n=448")
    return [float(x) for x in fallback]
