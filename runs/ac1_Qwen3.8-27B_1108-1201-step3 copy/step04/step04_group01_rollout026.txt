def p_norm_and_grad(conv, p):
    """Compute (sum(conv^p))^(1/p) and its gradient w.r.t. conv."""
    conv = np.maximum(conv, 1e-18)
    log_cp = (p - 1.0) * np.log(conv)
    log_sum = np.logaddexp.reduce(log_cp)
    # log(val) = (1/p) * log_sum
    # grad_k = val / sum(conv^p) * conv_k^(p-1) = conv_k^(p-1) / (n_k * val) ... 
    # Actually: d/dconv_k of (sum c^p)^(1/p) = c_k^(p-1) / (sum c^p)^(1-1/p)
    # = c_k^(p-1) * (sum c^p)^(1/p - 1)
    # Let's compute carefully:
    # val = (S)^(1/p) where S = sum c_k^p
    # d(val)/d(c_k) = (1/p) * S^(1/p - 1) * p * c_k^(p-1) = S^(1/p-1) * c_k^(p-1)
    # In log space: log(S^(1/p-1)) = (1/p - 1) * log(S) = (1/p - 1) * log_sum
    # log(c_k^(p-1)) = (p-1) * log(c_k)
    # So log(grad_k) = (1/p - 1)*log_sum + (p-1)*log(c_k)
    log_grad = (1.0/p - 1.0) * log_sum + (p - 1.0) * np.log(conv)
    grad = np.exp(log_grad)
    val = np.exp(log_sum / p)
    return val, grad

        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    p = p_start
    lr = 1.5 / np.sqrt(n)
    t0 = time.monotonic()
    global_patience = 0

    while time.monotonic() - t0 < time_limit and p < p_end:
        patience = 0
        max_pat = 3000
        lr_cur = lr
        it = 0

        while patience < max_pat and time.monotonic() - t0 < time_limit:
            it += 1
            conv = fast_convolve(f)
            _, grad_conv = p_norm_and_grad(conv, p)
            grad = grad_through_conv(f, grad_conv)

            gn = np.linalg.norm(grad)
            if gn < 1e-15:
                break
            grad = grad / gn

            f_new = mirror_step(f, grad, lr_cur)
            conv_new = fast_convolve(f_new)
            obj_new = conv_new.max()

            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_f = f_new.copy()
                f = f_new
                patience = 0
                global_patience = 0
                lr_cur = min(lr_cur * 1.003, 3.0 / np.sqrt(n))
            else:
                patience += 1
                global_patience += 1
                if patience % 80 == 0:
                    lr_cur *= 0.92
                if patience > 500:
                    rng = np.random.default_rng((it * 31 + int(time.monotonic() * 1e4)) % 999983)
                    f = best_f + rng.standard_normal(n) * 0.003
                    f = np.maximum(f, 0)
                    fs = f.sum()
                    if fs < 1e-15:
                        f = best_f.copy()
                    else:
                        f = f / fs
                    lr_cur = lr
                    patience = 0

        # Increase p
        p *= p_growth
        lr = max(lr * 0.85, 0.3 / np.sqrt(n))

        if global_patience > 5000:
            break

    return best_f, best_obj


def peak_target_optimize(f0, time_limit, use_mirror=True):
    """Subgradient descent targeting the current maximum lag."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = 2.0 / n
    momentum = 0.88
    v = np.zeros(n)
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        k_star = int(np.argmax(conv))

        # Subgradient: sg[i] = 2 * a[k_star - i] for valid i
        sg = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        if js <= je:
            idx = np.arange(js, je + 1)
            sg[idx] = 2.0 * f[k_star - idx]

        # Normalize
        sn = np.linalg.norm(sg)
        if sn < 1e-15:
            break
        sg = sg / sn

        # Momentum update
        v = momentum * v - lr * sg

        if use_mirror:
            f_new = mirror_step(f, v, 1.0)
        else:
            f_new = f + v
            f_new = np.maximum(f_new, 0)
            fns = f_new.sum()
            if fns < 1e-15:
                f_new = best_f.copy()
            else:
                f_new = f_new / fns

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            if no_improve == 0:
                lr = min(lr * 1.002, 4.0 / n)
        else:
            no_improve += 1
            if no_improve % 40 == 0:
                lr *= 0.94
                momentum = max(0.80, momentum - 0.003)
            if no_improve > 250:
                rng = np.random.default_rng((it * 13 + 7) % 999983)
                f = best_f + rng.standard_normal(n) * 0.002
                f = np.maximum(f, 0)
                fs = f.sum()
                if fs < 1e-15:
                    f = best_f.copy()
                else:
                    f = f / fs
                v = np.zeros(n)
                lr = 2.0 / n
                momentum = 0.88
                no_improve = 0

    return best_f, best_obj


def mixed_lag_optimize(f0, time_limit, num_lags=5):
    """Optimize using weighted subgradient from top-k lag peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = 1.5 / n
    momentum = 0.90
    v = np.zeros(n)
    no_improve = 0
    temperature = 0.5 / n
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)

        # Get top num_lags peaks with softmax weighting
        if num_lags >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -num_lags)[-num_lags:]

        pv = conv[peaks]
        scaled = (pv - pv.max()) / temperature
        w = np.exp(scaled)
        w = w / w.sum()

        # Combined subgradient
        sg = np.zeros(n)
        for ii in range(len(peaks)):
            if w[ii] < 1e-14:
                continue
            k = int(peaks[ii])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                idx = np.arange(js, je + 1)
                sg[idx] += w[ii] * 2.0 * f[k - idx]

        sn = np.linalg.norm(sg)
        if sn < 1e-15:
            break
        sg = sg / sn

        v = momentum * v - lr * sg
        f_new = mirror_step(f, v, 1.0)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 50 == 0:
                lr *= 0.95
            if no_improve > 300:
                rng = np.random.default_rng((it * 17 + 3) % 999983)
                f = best_f + rng.standard_normal(n) * 0.002
                f = np.maximum(f, 0)
                fs = f.sum()
                if fs < 1e-15:
                    f = best_f.copy()
                else:
                    f = f / fs
                v = np.zeros(n)
                lr = 1.5 / n
                no_improve = 0

        temperature *= 1.004

    return best_f, best_obj


def coordinate_polish(f0, time_limit, num_sweeps_target=200):
    """Cyclic coordinate descent exploiting the 1D structure of each coordinate update."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        # For each coordinate, try to improve
        for i in range(0, n, max(1, n // num_sweeps_target)):
            # Try a few candidate values for f[i]
            current_val = f[i]
            # Candidate: reduce (to minimize b[2i] = f[i]^2 component)
            # and increase (to maintain sum)
            candidates = [current_val * 0.8, current_val * 0.6, current_val * 1.2, current_val * 1.5]
            if current_val < 1e-10:
                candidates = [1.0 / n, 2.0 / n, current_val * 2]

            best_cand = current_val
            best_cobj = best_obj

            for cv in candidates:
                if cv < -1e-10:
                    continue
                cv = max(0.0, cv)
                f[i] = cv
                # Renormalize
                fs = f.sum()
                if fs < 1e-15:
                    f[i] = current_val
                    continue
                f_norm = f / fs
                conv_c = fast_convolve(f_norm)
                obj_c = conv_c.max()
                if obj_c < best_cobj - 1e-12:
                    best_cobj = obj_c
                    best_cand = cv

            f[i] = best_cand
            fs = f.sum()
            if fs > 1e-15:
                f = f / fs

        # Recompute
        conv = fast_convolve(f)
        obj = conv.max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 20:
                break

    return best_f, best_obj


def antisymmetric_organize(f0):
    """Map sequence to enhance antisymmetric structure.
    For pairs (j, n-1-j), make one larger and one smaller."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    n = len(f)
    s = f.sum()
    if s < 1e-15:
        return np.ones(n) / n
    f = f / s

    # For each pair (j, n-1-j), redistribute mass to make one bigger, one smaller
    for j in range(n // 2):
        pair_sum = f[j] + f[n - 1 - j]
        if pair_sum < 1e-15:
            continue
        # Make one 70%, other 30% (or alternating)
        if j % 2 == 0:
            f[j] = 0.65 * pair_sum
            f[n - 1 - j] = 0.35 * pair_sum
        else:
            f[j] = 0.35 * pair_sum
            f[n - 1 - j] = 0.65 * pair_sum
        # Keep endpoints small
    f[0] = min(f[0], 0.005 / n)
    f[n - 1] = min(f[n - 1], 0.005 / n)

    f = np.maximum(f, 0)
    fs = f.sum()
    if fs < 1e-15:
        return np.ones(n) / n
    return f / fs


def try_sequence(a, best_tracker):
    """Evaluate and track best."""
    a = np.maximum(a, 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score
