def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_conv_matrix(a, k=None):
    """Build matrix C such that C @ g = conv(a, g)[k] for specified k values, or all if k=None."""
    n = len(a)
    if k is None:
        k = np.arange(2*n-1)
    else:
        k = np.asarray(k)
    C = np.zeros((len(k), n))
    for row, kk in enumerate(k):
        kk = int(kk)
        j_start = max(0, kk - n + 1)
        j_end = min(n - 1, kk)
        for j in range(j_start, j_end + 1):
            C[row, j] = a[kk - j]
    return C

def logsumexp_gradient(a, conv, K=50, num_iters=30):
    """
    Compute gradient of log-sum-exp of top K peaks.
    Uses iterative reweighting.
    """
    n = len(a)
    # Get top K peaks
    top_k_idx = np.argpartition(conv, -K)[-K:]
    
    grad = np.zeros(n)
    total_mass = np.zeros(n)
    
    for _ in range(num_iters):
        # Compute weights based on peakedness
        peak_vals = conv[top_k_idx]
        # Use softmax with temperature based on the gap
        if peak_vals.max() > 0:
            temp = peak_vals.max() * 0.01
            scaled = peak_vals / temp
            scaled -= scaled.max()
            w = np.exp(scaled)
            w /= w.sum()
        else:
            w = np.ones(K) / K
        
        # For each weighted peak, add its contribution to gradient
        for idx in range(K):
            if w[idx] < 1e-15:

            k = int(top_k_idx[idx])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                grad[j] += w[idx] * a[k - j]
                total_mass[j] += w[idx]
    
    return grad

def compute_virtual_grad(a, conv, K=50, n_iter=25):

def multi_peak_lp(a, conv, K=30):
    """
    LP: minimize t s.t. conv(a,g)[k] <= t for top K peaks, sum(g)=1, g>=0.
    Returns (g, t_value, success).
    """
    n = len(a)
    top_idx = np.argpartition(conv, -K)[-K:]
    
    C = np.zeros((K, n))
    for row, kk in enumerate(top_idx):
        kk = int(kk)
        j_start = max(0, kk - n + 1)
        j_end = min(n - 1, kk)
        for j in range(j_start, j_end + 1):
            C[row, j] = a[kk - j]
    
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = C
    A_ub[:, -1] = -1.0
    b_ub = np.zeros(K)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method='highs')
        if result.success:
            g = np.maximum(result.x[:n], 0)
            t_val = result.x[-1]
            gs = g.sum()
            if gs > 1e-10:
                g = g / gs
            return g, t_val, True
    except Exception:
        pass
    return a.copy(), conv.max(), False

        # Find contributing pairs
        j_start = max(0, peak_idx - n + 1)
        j_end = min(n - 1, peak_idx)
        j_range = np.arange(j_start, j_end + 1)
        contributions = a_work[j_range] * a_work[peak_idx - j_range]
        
        # Find the largest contributor
        max_contrib_idx = int(np.argmax(contributions))
        j = int(j_range[max_contrib_idx])
        partner = peak_idx - j
        
        # Try reducing the larger of the two
        if a_work[j] >= a_work[partner]:
            target = j
            other = partner
        else:
            target = partner
            other = j
        
        # Reduce target, redistribute to other or to random positions
        for frac in [0.02, 0.05, 0.01]:
            amount = a_work[target] * frac
            if amount < 1e-12:
                continue
            a_try = a_work.copy()
            a_try[target] -= amount
            # Redistribute to the partner (increases the cross term but reduces self term)
            a_try[other] += amount
            
            cv = fast_convolve(a_try)
            obj = 2.0 * n * cv.max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a_try.copy()
                a_work = a_try
                conv_work = cv
                break
    
    return best_a, best_obj


def optimize_gradient(a0, time_limit, top_k=60, lr_init=0.01, momentum=0.91):
    """Nesterov-accelerated gradient descent on virtual peak."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    y = a.copy()
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    mom = momentum
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        y = a + mom * v
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy
        
        conv_y = fast_convolve(y)
        grad = compute_virtual_grad(y, conv_y, K=top_k, n_iter=20)
        
        # Normalize gradient for stable step
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            grad = grad / gnorm
        
        v = mom * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()
            v = np.zeros(n)
            lose = 50
            break
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        lr *= 0.9992
        mom = min(0.97, mom + 0.0003)
        
        if no_improve >= 250:
            break
    
    return best_a, best_obj


def optimize_lp_iter(a0, time_limit, K=25):
    """Iterative LP-based optimization with multi-peak targeting."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    iteration = 0
    
    while time.monotonic() - t0 < time_limit:
        g, t_val, success = multi_peak_lp(a, conv, K=K)
        
        if success:
            d = g - a
            best_t = 0.0
            for t in [0.4, 0.25, 0.15, 0.1, 0.06, 0.03, 0.01]:
                a_try = a + t * d
                a_try = np.maximum(a_try, 0.0)
                st = a_try.sum()
                if st < 1e-15:
                    continue
                a_try = a_try / st
                cv = fast_convolve(a_try)
                obj = 2.0 * n * cv.max()
                if obj < best_obj - 1e-12:
                    a = a_try
                    conv = cv
                    if obj < best_obj:
                        best_obj = obj
                        best_a = a.copy()
                    best_t = t
                    no_improve = 0
                    break
            if best_t == 0.0:
                no_improve += 1
        else:
            no_improve += 1
        
        iteration += 1
        
        if no_improve >= 40:
            rng = np.random.default_rng(iteration * 17 + 3)
            scale = 0.002 + 0.004 * min(no_improve // 40, 8)
            noise = rng.standard_normal(n) * scale
            a = np.maximum(best_a + noise, 0.0)
            sa = a.sum()
            if sa > 1e-15:
                a = a / sa
            else:
                a = best_a.copy()
            conv = fast_convolve(a)
            no_improve = 0
            if no_improve > 400:
                break
    
    return best_a, best_obj


def optimize_deflation(a0, time_limit):
    """Peak deflation loop."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        a_new, obj = peak_deflation(a, conv, n_peaks=5, n_rounds=20)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            conv = fast_convolve(a)
        else:
            # Try with more aggressive deflation
            a_perturbed = a + np.random.default_rng(int(time.monotonic()*1000)).standard_normal(n) * 0.001
            a_perturbed = np.maximum(a_perturbed, 0)
            sp = a_perturbed.sum()
            if sp > 1e-15:
                a_perturbed = a_perturbed / sp
            a_new, obj = peak_deflation(a_perturbed, fast_convolve(a_perturbed), n_peaks=5, n_rounds=30)
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a_new.copy()
                a = a_new
                conv = fast_convolve(a)
            else:
                break
    
    return best_a, best_obj


def optimize_hybrid(a0, time_limit, phase_time=20.0):
    """
    Hybrid: alternates between gradient, LP, and deflation phases.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    method_cycle = 0  # 0=grad, 1=lp, 2=deflation
    
    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        t_alloc = min(phase_time, remaining)
        
        if method_cycle == 0:
            a_opt, obj = optimize_gradient(a, t_alloc, top_k=80, lr_init=0.009, momentum=0.92)
        elif method_cycle == 1:
            a_opt, obj = optimize_lp_iter(a, t_alloc, K=30)
        else:
            a_opt, obj = optimize_deflation(a, t_alloc)
        
        if obj < best_obj:
            best_obj = obj
            best_a = a_opt.copy()
        
        a = a_opt
        method_cycle = (method_cycle + 1) % 3
    
    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # Sine-squared (reduces edge peaks)
    s1 = np.sin(np.pi * x / n) ** 2 + 0.005
    inits['sine1'] = s1 / s1.sum()
    
    # Power ramp (more mass at one end)
    ramp_up = (x + 1.0) ** 0.5
    inits['ramp_up'] = ramp_up / ramp_up.sum()
    ramp_dn = (n - x) ** 0.5
    inits['ramp_dn'] = ramp_dn / ramp_dn.sum()
    
    # Quadratic bowl
    bowl = (x - 0.5 * n) ** 2 + 1.0
    inits['bowl'] = bowl / bowl.sum()
    
    # Bimodal
    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()
    
    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps
    
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry: hybrid multi-phase search for optimal step function heights.
    Uses smoothed gradient, multi-peak LP, and peak deflation in alternating phases.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 35.0)
    
    best_sequence = None
    best_score = float('inf')
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}, score_est={2*len(parent)*fast_convolve(parent).max():.6f}")
    except Exception as e:
        print(f"No parent: {e}")
    
    def try_sequence(a):
        nonlocal best_sequence, best_score
        try:
            score = float(evaluate_sequence(a.tolist()))
        except Exception:
            return
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print(f"  NEW BEST: {score:.6f} (n={len(a)})")
    
    # ============================================================
    # Phase 1: Quick gradient scan across lengths and initializations
    # ============================================================
    print("=== Phase 1: Quick Scan ===")
    t_p1 = time.monotonic()
    p1_budget = 180.0
    
    lengths = [512, 640, 768, 896, 1024]
    scan_results = []
    
    for n in lengths:
        if time.monotonic() > deadline - 500:
            break
        inits = generate_inits(n, parent, rng)
        
        priority = ['parent', 'parent_interp', 'uniform', 'sine1', 'ramp_up', 'random', 'bimodal']
        init_names = [nm for nm in priority if nm in inits][:4]
        
        t_per = max(8.0, (p1_budget - 10) / (len(lengths) * 4))
        
        for name in init_names:
            if time.monotonic() > deadline - 400:
                break
            t_alloc = min(t_per, deadline - time.monotonic() - 200)
            if t_alloc < 5:
                break
            
            a0 = inits[name]
            a_opt, obj = optimize_gradient(a0, time_limit=t_alloc, top_k=50, lr_init=0.012, momentum=0.90)
            try_sequence(a_opt)
            scan_results.append((obj, n, name, a_opt.copy()))
    
    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_p1:.0f}s). Best: {best_score:.6f}")
    
    # ============================================================
    # Phase 2: Hybrid deep optimization on top candidates
    # ============================================================
    print("=== Phase 2: Hybrid Deep Optimization ===")
    t_p2 = time.monotonic()
    
    # Select top candidates (diverse by length)
    candidates = []
    seen_n = set()
    for obj, n, name, a in scan_results:
        if n not in seen_n:
            candidates.append((n, a))
            seen_n.add(n)
        if len(candidates) >= 4:
            break
    
    # Ensure best overall is included
    if best_sequence is not None:
        nb = len(best_sequence)
        if nb not in seen_n:
            candidates.append((nb, best_sequence.copy()))
    
    remaining = deadline - time.monotonic() - 250.0
    t_per = max(40.0, remaining * 0.65 / max(1, len(candidates)))
    
    for i, (n, a0) in enumerate(candidates):
        if time.monotonic() > deadline - 200:
            break
        t_alloc = min(t_per, deadline - time.monotonic() - 150)
        if t_alloc < 15:
            break
        
        print(f"  Hybrid {i}: n={n}, t={t_alloc:.0f}s")
        a_opt, obj = optimize_hybrid(a0, time_limit=t_alloc, phase_time=25.0)
        try_sequence(a_opt)
    
    print(f"Phase 2 done ({time.monotonic()-t_p2:.0f}s). Best: {best_score:.6f}")
    
    # ============================================================
    # Phase 3: Focused optimization with varied strategies on best
    # ============================================================
    print("=== Phase 3: Focused Search ===")
    t_p3 = time.monotonic()
    
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 150.0
        n_best = len(best_sequence)
        
        # Multiple focused runs with different strategies
        strategies = [
            ('gradient', lambda a, t: optimize_gradient(a, t, top_k=100, lr_init=0.005, momentum=0.94)),
            ('lp', lambda a, t: optimize_lp_iter(a, t, K=40)),

        for si, (sname, sfunc) in enumerate(strategies):

            if t_alloc < 10:
                break

            if si > 0:
                noise = rng.standard_normal(n_best) * 0.001
                a0 = np.maximum(best_sequence + noise, 0)
                s0 = a0.sum()
                if s0 > 1e-15:
                    a0 = a0 / s0
                else:
                    a0 = best_sequence.copy()
            else:
                a0 = best_sequence.copy()

    # ============================================================

    # ============================================================
    print("=== Phase 4: Length Exploration ===")
    t_p4 = time.monotonic()

        n_best = len(best_sequence)
        alt_lengths = [n_best // 2, n_best * 3 // 2, n_best * 2]

            t_alloc = min(40.0, deadline - time.monotonic() - 50)
            if t_alloc < 10:
                break
            
            # Interpolate best to new length
            a_interp = np.interp(np.linspace(0, 1, alt_n), np.linspace(0, 1, n_best), best_sequence)
            a_interp = np.maximum(a_interp, 0)
            si = a_interp.sum()
            if si > 1e-15:
                a_interp = a_interp / si
            
            a_opt, obj = optimize_hybrid(a_interp, time_limit=t_alloc, phase_time=15.0)
            try_sequence(a_opt)
    
    print(f"Phase 4 done ({time.monotonic()-t_p4:.0f}s). Best: {best_score:.6f}")
    
    # ============================================================
    # Phase 5: Final polish - short focused gradient + deflation
    # ============================================================
    print("=== Phase 5: Final Polish ===")
    t_p5 = time.monotonic()
    
    if best_sequence is not None and time.monotonic() < deadline - 10:
        t_final = min(30.0, deadline - time.monotonic() - 5)
        if t_final > 5:

            try_sequence(a_opt)
            
            if time.monotonic() < deadline - 5:
                t_d = min(15.0, deadline - time.monotonic() - 3)

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))
    
    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    
    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
