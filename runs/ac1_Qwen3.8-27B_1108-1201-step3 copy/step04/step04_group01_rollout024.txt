    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v_hat = np.zeros(n)
    t = 0

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        if time.monotonic() - t0 > time_limit:
            break

        grad = lse_gradient(f, beta)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-16:
            no_improve += 100
            break

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v_hat = beta2 * v_hat + (1 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat_corr = v_hat / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        f_new = project_simplex(f - step)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 30:
                lr *= 0.93
            if no_improve > 150:
                rng = np.random.default_rng(t * 13 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                m = np.zeros(n)
                v_hat = np.zeros(n)
                t = 0
                lr = 0.005
                beta = min(beta * 1.4, beta_end)
                no_improve = 0

        if beta < beta_end:
            beta *= 1.004

        if t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    AdamLSE t={t}, obj={best_obj:.12f}, beta={beta:.0f}, lr={lr:.6f}, {elapsed:.0f}s")

    return best_f, best_obj


def subgrad_peak_polish(f0, time_limit, top_k=1, temp=0.001):
    """Direct subgradient polishing targeting top-k peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    lr = 0.003
    momentum = 0.92
    v = np.zeros(n)
    cur_temp = temp

    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)

        # Select top-k peaks with softmax weighting
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            weights = np.ones(len(peaks)) / len(peaks)
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            pv = conv[peaks]
            scaled = pv / cur_temp
            scaled -= scaled.max()
            weights = np.exp(scaled)
            weights /= weights.sum()

        # Compute weighted subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            k = peaks[i]
            sg = sg + weights[i] * subgradient_peak_vec(f, k)

        v = momentum * v - lr * sg
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                lr *= 0.95
                momentum = max(0.85, momentum - 0.003)
            if no_improve > 200:
                rng = np.random.default_rng(it * 17 + 5)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                v = np.zeros(n)
                lr = max(lr * 0.5, 0.0003)
                no_improve = 0

        cur_temp *= 1.003

        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG t={it}, obj={best_obj:.12f}, lr={lr:.6f}, {elapsed:.0f}s")

    return best_f, best_obj


def spectral_shaping(f0, time_limit, num_iter_target=500):
    """Gerchberg-Saxton style spectral shaping for flat autocorrelation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit and it < num_iter_target:
        it += 1

        # Compute FFT of current sequence
        fft_size = 1 << (2 * n - 1).bit_length()
        A = np.fft.rfft(f, fft_size)

        # Target: flat power spectrum
        # current power spectrum
        power = A * np.conj(A)
        # We want |A|^2 to be as flat as possible
        # Iterate: set |A| to sqrt of desired, keep phase
        # Desired: uniform power
        target_power_mean = np.sum(power.real) / len(power)
        target_mag = np.sqrt(target_power_mean)

        # New A with target magnitude and current phase
        phase = A / (np.abs(A) + 1e-30)
        A_new = target_mag * phase

        # Transform back
        f_new = np.fft.irfft(A_new, fft_size)[:n]

        # Project to simplex
        f_new = project_simplex(np.maximum(f_new, 0.0))

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
        else:
            # Blend with previous
            alpha = 0.5
            f = project_simplex(np.maximum(alpha * best_f + (1 - alpha) * f_new, 0.0))
            if it % 100 == 0:
                # Occasionally accept worse to escape
                rng = np.random.default_rng(it)
                if rng.random() < 0.1:
                    f = project_simplex(np.maximum(f_new, 0.0))

    return best_f, best_obj


def momentum_lse_optimize(f0, time_limit, beta_start=100.0, beta_end=10000.0):
    """LSE with momentum (more stable than Adam for this problem)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = 1.5 / n
    mom = 0.91
    v = np.zeros(n)

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        grad = lse_gradient(f, beta)
        gn = np.linalg.norm(grad)
        if gn < 1e-16:
            no_improve += 100
            break
        grad = grad / gn

        v = mom * v - lr * grad
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.001, 3.0 / n)
        else:
            no_improve += 1
            if no_improve >= 15:
                lr *= 0.91
                mom = max(0.85, mom - 0.004)
            if no_improve > 100:
                f = best_f.copy()
                v = np.zeros(n)
                lr = max(1.5 / n, 0.002)
                beta = min(beta * 1.5, beta_end)
                mom = 0.89
                no_improve = 0

        if beta < beta_end:
            beta *= 1.003

        if it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MomLSE t={it}, obj={best_obj:.12f}, beta={beta:.0f}, lr={lr:.7f}, {elapsed:.0f}s")

    return best_f, best_obj


def try_sequence(a, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.10f} (n={len(a)})")
    return score
