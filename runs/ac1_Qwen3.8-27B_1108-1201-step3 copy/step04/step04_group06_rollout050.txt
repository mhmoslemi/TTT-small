For the Adam optimization in log-space:
- `theta` is a free vector

  where `dL/da_i = (2/beta) * sum_k softmax_k * a_{k-i} * [i in J_k]`

Wait, let me recompute. `L = logsumexp(beta * c) / beta` where `c = conv(a,a)`.
`dL/dc_k = exp(beta*c_k) / sum(exp(beta*c_l)) = w_k` (the softmax weights)
`dL/dtheta_j = sum_k w_k * dc_k/dtheta_j`
`dc_k/dtheta_j = sum_i dc_k/da_i * da_i/dtheta_j = sum_i dc_k/da_i * a_i * (delta(i,j) - a_j)`
`= (sum_i dc_k/da_i * a_i) * (-a_j) + dc_k/da_j * a_j`
`= a_j * (dc_k/da_j - sum_i dc_k/da_i * a_i)`

So `dL/dtheta_j = a_j * sum_k w_k * (dc_k/da_j - sum_i dc_k/da_i * a_i)`
`= a_j * (sum_k w_k * dc_k/da_j - sum_i a_i * sum_k w_k * dc_k/da_i)`
`= a_j * (g_j - sum_i a_i * g_i)`

where `g_i = sum_k w_k * dc_k/da_i = 2 * sum_k w_k * a_{k-i} * [i in J_k]` is the gradient of the weighted sum of convolution values.

So the gradient in theta-space is: `grad_theta_j = a_j * (g_j - dot(a, g))`

This is the "centered" gradient, which is what you get from the Fisher information metric. It's automatically orthogonal to the all-ones vector (since `sum_j a_j * (g_j - dot(a,g)) = dot(a,g) - dot(a,g) = 0`), which makes sense because the objective is invariant to adding a constant to all theta_i.

This is a clean formulation. Let me implement it.

For the gradient computation, I need `g_i = 2 * sum_k w_k * a_{k-i} * [i in J_k]`.

As I showed earlier, `[i in J_k]` is 1 iff `i <= k <= i+n-1` (i.e., `k in [i, i+n-1]`).

So `g_i = 2 * sum_{k=i}^{i+n-1} w_k * a_{k-i}`

This is a cross-correlation! `g_i / 2 = sum_{j=0}^{n-1} a_j * w_{i+j}`

- This is the cross-correlation `(u ⋆ v)[i]`

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-phase approach:
1. Adam optimization in log-softmax parametrization with annealed smooth-max
2. Exact coordinate descent with 1D golden section search
3. Local search with group perturbations
4. Multi-length exploration
"""

import numpy as np
from math import log, exp
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax(x):
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()


def logsumexp(x, beta):
    """Compute logsumexp(beta*x)/beta."""
    beta_x = beta * x
    m = beta_x.max()
    return (m + np.log(np.sum(np.exp(beta_x - m)))) / beta


def logsumexp_grad(x, beta):
    """Gradient of logsumexp(beta*x)/beta w.r.t. x."""
    beta_x = beta * x
    m = beta_x.max()
    e = np.exp(beta_x - m)
    return e / e.sum()


def compute_g(w, a):
    """Compute g_i = 2 * sum_{k=i}^{i+n-1} w_k * a_{k-i} for i=0..n-1."""
    n = len(a)
    g = np.zeros(n)
    for i in range(n):
        g[i] = 2.0 * np.dot(w[i:i+n], a)
    return g


def adam_logsoftmax(a0, time_limit, beta0=20.0, beta_end=5000.0,
                    lr0=0.01, beta_growth=1.002, verbose=False):
    """
    Adam optimization in log-softmax space with annealed smooth-max.
    """
    n = len(a0)
    # Initialize theta
    a = np.maximum(a0, 1e-15).astype(np.float64)
    a = a / a.sum()
    theta = np.log(a)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    beta = beta0
    lr = lr0
    t0 = time.monotonic()
    
    best_score = np.inf
    best_a = a.copy()
    
    c = fast_convolve(a)
    current_obj = logsumexp(c, beta_end)  # Evaluate with final beta
    
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        t += 1
        
        # Compute gradient
        c = fast_convolve(a)
        w = logsumexp_grad(c, beta)
        g = compute_g(w, a)
        
        # Gradient in theta-space: grad_j = a_j * (g_j - dot(a, g))
        ag = np.dot(a, g)
        grad_theta = a * (g - ag)
        
        # Adam update
        b1, b2, eps = 0.9, 0.999, 1e-8
        m = b1 * m + (1 - b1) * grad_theta
        v = b2 * v + (1 - b2) * grad_theta ** 2
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        
        theta -= lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Convert back to a
        a = softmax(theta)
        c = fast_convolve(a)
        
        # Evaluate with a fixed (large) beta for tracking progress
        obj = logsumexp(c, beta_end)
        if obj < best_score - 1e-10:
            best_score = obj
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 1000 == 0:
                print(f"    AdamLog iter {it}: obj={obj:.10f}, beta={beta:.1f}")
        else:
            no_improve += 1
        
        # Anneal beta
        beta = min(beta * beta_growth, beta_end)
        
        # Adaptive learning rate
        if no_improve > 2000:
            lr *= 0.5
            no_improve = 0
            if lr < 1e-6:
                break
        # Occasionally boost lr to escape local minima
        if it % 5000 == 0 and no_improve > 500:
            lr = max(lr, lr0 * 0.1)
            # Small random perturbation
            theta += np.random.default_rng(it).standard_normal(n) * 0.01
        
        # Check actual score periodically
        if it % 500 == 0:
            actual = fast_convolve(best_a).max()
            actual_score = 2 * n * actual / (best_a.sum()**2)
    
    return best_a.copy(), best_score


def coordinate_descent(a0, time_limit, n_sweeps=None):
    """
    Coordinate descent with 1D optimization for each coordinate.
    """
    n = len(a0)
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    sweep = 0
    
    while time.monotonic() - t0 < time_limit:
        sweep += 1
        order = rng.permutation(n)
        improved = False
        
        for p in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[p]
            if old_val < 1e-12:
                # Try small positive values
                candidates = [1e-4, 1e-3, 0.01, 0.05]
            else:
                # Try multiplicative perturbation of the current value
                candidates = [old_val * f for f in 
                              [0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 3.0]]
            
            best_cand = old_val
            best_cand_obj = best_obj  # Current obj is the baseline
            
            for x in candidates:
                a[p] = x
                s = a.sum()
                if s < 1e-15:
                    a[p] = old_val
                    continue
                a_norm = a / s
                c2 = fast_convolve(a_norm)
                obj2 = c2.max()
                if obj2 < best_cand_obj - 1e-10:
                    best_cand_obj = obj2
                    best_cand = x
            
            if best_cand != old_val:
                a[p] = best_cand
                s = a.sum()
                a = a / s
                c = fast_convolve(a)
                obj = c.max()
                if obj < best_obj - 1e-10:
                    best_obj = obj
                    best_a = a.copy()
                    improved = True
            else:
                a[p] = old_val
        
        if not improved and sweep > 1:
            break
    
    return best_a.copy(), best_obj


def local_search_groups(a0, time_limit, group_size=5):
    """
    Local search: perturb groups of coordinates simultaneously.
    """
    n = len(a0)
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    
    c = fast_convolve(a)
    best_obj = c.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Pick a random group of coordinates
        gsize = rng.integers(max(1, n // 20), min(group_size * 3, n // 4))
        coords = rng.choice(n, size=gsize, replace=False)
        
        # Generate random perturbation
        for attempt in range(20):
            perturbation = rng.standard_normal(gsize) * 0.1
            new_vals = np.maximum(a[coords] + perturbation * a[coords].max(), 0)
            
            save = a[coords].copy()
            a[coords] = new_vals
            s = a.sum()
            if s < 1e-15:
                a[coords] = save
                continue
            a = a / s
            c2 = fast_convolve(a)
            obj2 = c2.max()
            
            if obj2 < best_obj - 1e-10:
                best_obj = obj2
                best_a = a.copy()
                no_improve = 0
            else:
                a = best_a.copy()
        
        no_improve += 1
        if no_improve > 50:
            # Try larger perturbations
            perturb_scale = 0.5
            gsize = rng.integers(max(2, n // 10), min(n // 2, 50))
            coords = rng.choice(n, size=gsize, replace=False)
            perturbation = rng.standard_normal(gsize) * perturb_scale
            new_vals = np.maximum(a[coords] + perturbation, 0)
            save = a[coords].copy()
            a[coords] = new_vals
            s = a.sum()
            if s > 1e-10:
                a = a / s
                c2 = fast_convolve(a)
                obj2 = c2.max()
                if obj2 < best_obj - 1e-10:
                    best_obj = obj2
                    best_a = a.copy()
                    no_improve = 0
    
    return best_a.copy(), best_obj
