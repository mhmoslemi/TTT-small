def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax_safe(x):
    """Numerically stable softmax."""
    x = x - x.max()
    e = np.exp(x)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def projection_simplex(v):
    """Project vector onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_gradient(p, a):
    """
    Compute gradient of log-sum-exp smoothed max of conv(a,a) w.r.t. a.
    grad[i] = 2 * sum_j p[i+j] * a[j]
    where p is the softmax distribution over convolution lags.
    """
    n = len(a)
    # grad[i] = 2 * sum_{j=0}^{n-1} p[i+j] * a[j]
    # Use matrix formulation: W[i,j] = p[i+j], grad = 2 * W @ a
    # Efficient computation via slicing
    grad = np.zeros(n)
    for i in range(n):
        grad[i] = 2.0 * np.dot(a, p[i:i + n])
    return grad


def smooth_mirror_descent(a0, time_limit, tau_init=100.0, verbose=False):
    """
    Mirror descent (natural gradient) on the log-sum-exp smoothed objective.
    Update: a_new[i] = a[i] * exp(-eta * grad[i]), then normalize.
    This stays in the simplex automatically.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    tau = tau_init
    best_obj = np.inf
    best_a = a.copy()
    eta = 1.0

    t0 = time.monotonic()
    it = 0
    no_improve = 0
    tau_increase = 1.002

    while time.monotonic() - t0 < time_limit:
        c = fast_convolve(a)
        c_max = c.max()
        cur_obj = 2.0 * n * c_max

        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Compute softmax distribution over lags
        p = softmax_safe(tau * c)

        # Compute gradient
        grad = compute_smooth_gradient(p, a)

        # Mirror descent step: a_new[i] = a[i] * exp(-eta * grad[i] / a[i])
        # This is equivalent to: a_new[i] = a[i] * exp(-eta * grad[i]) normalized... 
        # Actually, the natural gradient on the simplex is:
        # a_new[i] = a[i] * exp(-eta * grad[i]) then normalize
        # But grad[i] can be large relative to a[i], so let's be careful.
        # 
        # Better: use the standard mirror descent:
        # min_g <grad, g> + (1/eta) KL(g || a)
        # Solution: g[i] = a[i] * exp(-eta * grad[i]) / sum_j a[j] * exp(-eta * grad[j])
        
        scaled_grad = eta * grad
        # Clip to avoid underflow
        scaled_grad = np.clip(scaled_grad, -500, 500)
        a_new = a * np.exp(-scaled_grad)
        s_new = a_new.sum()
        if s_new < 1e-30:
            a_new = a.copy()
        else:
            a = a_new / s_new

        # Adapt tau and eta
        if no_improve % 50 == 0 and no_improve > 0:
            tau *= tau_increase
            tau_increase = min(1.01, tau_increase * 1.001)
            eta *= 0.95

        # Periodic perturbation to escape local minima
        if no_improve >= 300 and it > 100:
            rng = np.random.default_rng(it * 7 + 42)
            noise = rng.standard_normal(n) * 0.003
            a = best_a + noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            no_improve = 0

        it += 1
        if verbose and it % 500 == 0:
            print(f"    MD iter {it}, obj={cur_obj:.8f}, tau={tau:.1f}, "
                  f"eta={eta:.4f}, t={time.monotonic()-t0:.1f}s")

    return best_a, best_obj


def peak_shaving(a0, time_limit, verbose=False):
    """
    Coordinate descent that directly targets the peak of the convolution.
    For the peak lag k*, reduce a[i]*a[k*-i] for the worst pair.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    best_obj = np.inf
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0

    while time.monotonic() - t0 < time_limit:
        c = fast_convolve(a)
        c_max = c.max()
        cur_obj = 2.0 * n * c_max

        if cur_obj < best_obj - 1e-15:
            best_obj = cur_obj
            best_a = a.copy()

        # Find peak lag
        k_star = int(np.argmax(c))

        # Find the worst pair (i, k_star - i)
        # conv[k_star] = sum_i a[i] * a[k_star - i]
        i_min = max(0, k_star - n + 1)
        i_max = min(n - 1, k_star)
        indices = np.arange(i_min, i_max + 1)
        products = a[indices] * a[k_star - indices]
        
        # Find the pair with largest product
        worst_idx = indices[np.argmax(products)]
        partner_idx = k_star - worst_idx

        # Redistribute: move mass from the worst pair to a "safe" location
        # Choose a location that minimizes contribution to the peak
        # Simple heuristic: move to a location with low a value that's far from peak
        amount = 0.05 * min(a[worst_idx], a[partner_idx])

        if amount > 1e-10:
            # Find a good destination (low contribution to peak)
            # Try moving to end of sequence
            if worst_idx > partner_idx:
                src1, src2 = worst_idx, partner_idx
            else:
                src1, src2 = partner_idx, worst_idx
            
            # Move from src1 to a low-value position
            dest_candidates = np.argsort(a)[:10]
            dest = None
            for d in dest_candidates:
                if d != src1 and d != src2:
                    # Check if this doesn't create a new peak
                    dist_to_peak = abs(d + src2 - k_star)  # rough check
                    if dist_to_peak > n * 0.2:
                        dest = d
                        break
            if dest is None:
                dest = dest_candidates[0]
                if dest == src1:
                    dest = dest_candidates[1] if len(dest_candidates) > 1 else 0

            a[src1] -= amount
            a[dest] += amount
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s

        # Also try a symmetric move
        it += 1
        if verbose and it % 1000 == 0:
            print(f"    PeakShave iter {it}, obj={cur_obj:.8f}, "
                  f"k*={k_star}, t={time.monotonic()-t0:.1f}s")

    return best_a, best_obj


def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def solve_lp_direction(a, M, A_ub):
    """Solve LP: maximize sum(g) s.t. a*g <= M, g >= 0. Returns normalized g."""
    n = len(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if result.success:
            g = np.maximum(result.x, 0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum, g_sum
    except Exception:
        pass
    return a.copy(), 1.0


def lp_optimize(a0, time_limit, verbose=False):
    """LP-based Frank-Wolfe optimization."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    _, conv = obj_value(a, n)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    tightness = 1.0

    while time.monotonic() - t0 < time_limit:
        A_ub = build_conv_matrix(a)
        g, g_sum = solve_lp_direction(a, M * tightness, A_ub)

        if g_sum > 1.0 + 1e-6:
            d = g - a
            # Line search
            best_t = 0.0
            best_trial = best_obj
            for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                a_try = np.maximum(a + t * d, 0.0)
                s_try = a_try.sum()
                if s_try > 1e-15:
                    a_try = a_try / s_try
                    obj_try, _ = obj_value(a_try, n)
                    if obj_try < best_trial:
                        best_trial = obj_try
                        best_t = t

            if best_t > 0 and best_trial < best_obj - 1e-14:
                a_new = np.maximum(a + best_t * d, 0.0)
                s_new = a_new.sum()
                if s_new > 1e-15:
                    a = a_new / s_new
                _, conv = obj_value(a, n)
                M = conv.max()
                best_obj = 2.0 * n * M
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        iteration += 1

        if no_improve >= 60:
            rng = np.random.default_rng(iteration * 11 + 3)
            noise = rng.standard_normal(n) * 0.003
            a = best_a + noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            _, conv = obj_value(a, n)
            M = conv.max()
            no_improve = 0

        if verbose and iteration % 50 == 0:
            print(f"    LP iter {iteration}, obj={best_obj:.8f}, "
                  f"tight={tightness:.4f}, t={time.monotonic()-t0:.1f}s")

    return best_a, best_obj


def obj_value(a, n):
    """Compute objective and convolution for normalized a (sum=1)."""
    conv = fast_convolve(a)
    obj = 2.0 * n * conv.max()
    return obj, conv


def resample_sequence(seq, new_n):
    """Resample a sequence to a different length via linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s > 1e-15:
        new_seq = new_seq / s
    return new_seq


def propose_candidate(seed=42, budget_s=1800, **kwargs):

    Hybrid optimization: mirror descent on smoothed objective + LP polishing
    + peak shaving + multi-length exploration.
    """

    best_sequence = None

    # Load parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(p)}, score={evaluate_sequence(parent.tolist()):.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768
        print("No parent available, using uniform init n=768")

    parent_n = len(parent)

    # === Phase 1: Mirror Descent on parent length ===
    print("=== Phase 1: Mirror Descent (n={}) ===".format(parent_n))
    t_start = time.monotonic()
    p1_budget = 350.0
    p1_deadline = t_start + p1_budget

    # Run mirror descent with different tau values
    md_configs = [
        {'tau_init': 50.0},
        {'tau_init': 200.0},
        {'tau_init': 500.0},
    ]

    t_per_md = p1_budget / len(md_configs)
    for ci, cfg in enumerate(md_configs):
        remaining = p1_deadline - time.monotonic()
        if remaining < 30:
            break
        t_alloc = min(t_per_md, remaining - 10)
        if t_alloc < 20:
            break

        a_opt, obj_opt = smooth_mirror_descent(parent, time_limit=t_alloc,
                                                tau_init=cfg['tau_init'], verbose=True)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  MD config {ci}: score={score:.6f}, obj={obj_opt:.6f}")

    # === Phase 2: Peak Shaving ===
    print("=== Phase 2: Peak Shaving ===")
    t_start2 = time.monotonic()
    p2_budget = 150.0
    p2_deadline = t_start2 + p2_budget

    if best_sequence is not None:
        a_opt, obj_opt = peak_shaving(best_sequence, time_limit=p2_budget, verbose=True)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a_opt.copy()
            print(f"  PeakShave: score={score:.6f}")

    # === Phase 3: LP Polishing on best ===
    print("=== Phase 3: LP Polishing ===")
    t_start3 = time.monotonic()

    remaining_for_lp = deadline - time.monotonic() - 400.0
    if remaining_for_lp < 100:
        remaining_for_lp = 100
    lp_budget = min(350.0, remaining_for_lp)
    lp_deadline = t_start3 + lp_budget

    if best_sequence is not None:
        # Multiple LP rounds with perturbation
        n_rounds = 3
        t_per_lp = lp_budget / n_rounds
        for ri in range(n_rounds):
            remaining = lp_deadline - time.monotonic()
            if remaining < 30:
                break
            t_alloc = min(t_per_lp, remaining - 10)
            if t_alloc < 20:
                break

            # Add small perturbation for subsequent rounds
            if ri > 0:
                noise = rng.standard_normal(len(best_sequence)) * 0.002
                a_pert = best_sequence + noise
                a_pert = np.maximum(a_pert, 0.0)
                a_pert = a_pert / a_pert.sum()
                a_opt, obj_opt = lp_optimize(a_pert, time_limit=t_alloc, verbose=True)
            else:
                a_opt, obj_opt = lp_optimize(best_sequence, time_limit=t_alloc, verbose=True)

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  LP round {ri}: score={score:.6f}")

    # === Phase 4: Mirror Descent again on LP result ===
    print("=== Phase 4: MD on LP result ===")
    t_start4 = time.monotonic()
    remaining_for_p4 = deadline - time.monotonic() - 200.0
    if remaining_for_p4 < 50:
        remaining_for_p4 = 50
    p4_budget = min(200.0, remaining_for_p4)

    if best_sequence is not None:
        a_opt, obj_opt = smooth_mirror_descent(best_sequence, time_limit=p4_budget,
                                               tau_init=300.0, verbose=True)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:

            best_sequence = a_opt.copy()
            print(f"  MD phase 4: score={score:.6f}")

    if remaining_for_p5 > 100:

        alt_n = 1024 if parent_n != 1024 else 768
        if remaining_for_p5 > 150:

        a_alt = resample_sequence(best_sequence if best_sequence is not None else parent, alt_n)
        
        t_alloc = min(150.0, remaining_for_p5 - 20)
        if t_alloc > 50:
            a_opt, obj_opt = smooth_mirror_descent(a_alt, time_limit=t_alloc * 0.5,
                                                   tau_init=200.0)
            # Quick LP polish
            if t_alloc > 80:
                a_opt2, obj_opt2 = lp_optimize(a_opt, time_limit=t_alloc * 0.5, verbose=True)
                if obj_opt2 < obj_opt:
                    a_opt = a_opt2

            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Alt length {alt_n}: score={score:.6f}")

    # === Phase 6: Final fine-tuning ===
    print("=== Phase 6: Final Fine-tuning ===")
    t_start6 = time.monotonic()
    remaining_final = deadline - time.monotonic() - 10.0

    if remaining_final > 30 and best_sequence is not None:
        n_best = len(best_sequence)
        # Quick LP with tight budget
        t_quick = min(60.0, remaining_final * 0.5)
        if t_quick > 15:
            a_opt, obj_opt = lp_optimize(best_sequence, time_limit=t_quick)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final LP: score={score:.6f}")

        # A few more MD steps
        remaining_md = deadline - time.monotonic() - 5.0
        if remaining_md > 20:
            a_opt, obj_opt = smooth_mirror_descent(best_sequence, time_limit=remaining_md,
                                                   tau_init=500.0)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final MD: score={score:.6f}")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
