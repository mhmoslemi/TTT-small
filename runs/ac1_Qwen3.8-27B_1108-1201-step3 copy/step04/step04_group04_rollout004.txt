def lp_min_peak_cross(a, n):
    """
    Solve: min t s.t. convolve(a, g)_k <= t for all k, sum(g)=1, g>=0
    Returns (g, t_optimal, success)
    """

    # Variables: x = [g_0, ..., g_{n-1}, t]
    n_vars = n + 1
    
    # Objective: minimize t => c = [0, 0, ..., 0, 1]
    c = np.zeros(n_vars)
    c[-1] = 1.0
    
    # Inequality: C@g - t <= 0
    # In standard form: A_ub @ x <= b_ub
    A_ub = np.zeros((2*n-1, n_vars))
    A_ub[:, :n] = C
    A_ub[:, -1] = -1.0  # -t
    b_ub = np.zeros(2*n-1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = [1.0]
    
    # Bounds: g_i >= 0, t >= 0
    bounds = [(0, None)] * n + [(0, None)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

        t = result.x[-1]

            # Hybrid: blend min-peak and max-sum directions
            alpha_blend = 0.6
            d = alpha_blend * (g - a) + (1 - alpha_blend) * (g2 - a)
        else:
            d = g - a

        # Add momentum
        d = d + momentum_vec
        momentum_vec = 0.5 * momentum_vec

        norm_d = np.linalg.norm(d)
        if norm_d < 1e-12:
            no_improve += 10
            if no_improve > 100:
                break
            continue

        # Line search
        best_alpha, new_peak = line_search_peak(a, d, M, n)

        if best_alpha > 0:
            a_new = a + best_alpha * d
            a_new = np.maximum(a_new, 0.0)
            s_new = a_new.sum()
            if s_new < 1e-12:
                a_new = a.copy()
            else:
                a_new = a_new / s_new
            conv_new = fast_convolve(a_new)
            M_new = conv_new.max()
            obj_new = 2.0 * n * M_new

            if obj_new < best_obj - 1e-13:
                best_obj = obj_new
                best_a = a_new.copy()
                no_improve = 0
            else:
                no_improve += 1
            a = a_new
            conv = conv_new
            M = M_new
            C = build_conv_matrix(a, n)
            momentum_vec = best_alpha * d / max(norm_d, 1e-10) * 0.3
        else:
            no_improve += 1

        iteration += 1

        if verbose and iteration % 50 == 0:
            print(f"    LP iter {iteration}, obj={best_obj:.8f}, t={time.monotonic()-t0:.1f}s, no_impr={no_improve}")

        # Stagnation handling
        if no_improve > 50:
            rng = np.random.default_rng(iteration * 31 + 17)
            # Structured perturbation based on peak pattern
            peak_idx = int(conv.argmax())
            pert = np.zeros(n)
            # Add anti-phase perturbation at the peak
            j_start = max(0, peak_idx - n + 1)
            j_end = min(n - 1, peak_idx)
            for j in range(j_start, j_end + 1):
                pert[j] -= a[peak_idx - j] * 0.01
            # Add small random noise
            pert += rng.standard_normal(n) * 0.002
            a = best_a + pert
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-12:
                a = best_a.copy()
            else:
                a = a / s
            conv = fast_convolve(a)
            M = conv.max()
            C = build_conv_matrix(a, n)
            no_improve = 0
            momentum_vec = np.zeros(n)

            if iteration > 500 and no_improve > 200:
                break
            if no_improve > 200:
                break

    return best_a, best_obj


def spectral_polish(a0, time_limit, verbose=False):
    """
    Spectral water-filling: iteratively modify the power spectrum to
    equalize the auto-convolution, then project back to non-negative.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, 2.0 * len(a) * fast_convolve(a).max()
    a = a / s
    n = len(a)
    N = 1 << (2 * n - 1).bit_length()

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    alpha = 0.1  # blending parameter
    iteration = 0

    while time.monotonic() - t0 < time_limit:
        # Compute frequency domain
        A = np.fft.rfft(a, N)
        P = np.abs(A) ** 2

        # Compute auto-conv from P
        p = np.fft.irfft(P, N)[:2 * n - 1]

        # Find peak and equalize
        peak_val = p.max()
        target = p.mean() * 1.5  # aim for slightly above mean
        if peak_val > target:
            # Reduce peak in time domain
            p_reduced = p.copy()
            p_reduced[p_reduced > target] = target + (p_reduced[p_reduced > target] - target) * 0.5
            p_reduced = np.maximum(p_reduced, 0)

            # Go back to frequency domain
            A_new = np.fft.rfft(p_reduced, N)
            # Phase retrieval: keep original phase, update magnitude
            A_combined = np.abs(A_new) * np.exp(1j * np.angle(A + 1e-30))
            a_candidate = np.fft.irfft(A_combined, N)[:n]

            # Blend with original
            a_trial = (1 - alpha) * a + alpha * a_candidate
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 1e-12:
                a_trial = a_trial / st
            else:
                a_trial = a.copy()

            conv_trial = fast_convolve(a_trial)
            M_trial = conv_trial.max()
            if M_trial < M - 1e-14:
                a = a_trial
                conv = conv_trial
                M = M_trial
                if 2.0 * n * M < best_obj:
                    best_obj = 2.0 * n * M
                    best_a = a.copy()
                iteration += 1
            else:
                alpha *= 0.9
                if alpha < 0.01:
                    break
        else:
            break

        if verbose and iteration % 20 == 0:
            print(f"    Spectral iter {iteration}, obj={best_obj:.8f}, alpha={alpha:.4f}")

    return best_a, best_obj


def gradient_optimize(a0, time_limit, lr_init=0.008, momentum=0.9, top_k=30, temp_init=0.002):
    """Nesterov-accelerated gradient descent with adaptive peak weighting."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    y = a.copy()
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    mom = momentum
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        y = a + mom * v
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy

        conv_y = fast_convolve(y)

        # Compute gradient using top-k peaks with softmax weighting
        peaks = np.argpartition(conv_y, -top_k)[-top_k:]
        peak_vals = conv_y[peaks]
        scaled = peak_vals / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(peak_vals)] = 1.0
        else:
            w /= w_sum

        grad = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-12:
                continue
            k = int(peaks[idx])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idxs = np.arange(j_start, j_end + 1)
            grad[idxs] += w[idx] * a[k - idxs]

        v = mom * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-13:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9997
        temp *= 1.002
        mom = min(0.96, mom + 0.0003)

        if no_improve >= 150:
            break

    return best_a, best_obj


def fine_coordinate_descent(a0, time_limit):
    """Targeted coordinate descent on peak-contributing elements."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, 2.0 * len(a) * fast_convolve(a).max()
    a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(99999)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        # Find peak and its contributing elements
        peak_idx = int(conv.argmax())
        j_start = max(0, peak_idx - n + 1)
        j_end = min(n - 1, peak_idx)
        contributing = np.arange(j_start, j_end + 1)

        # Get top contributors
        contributions = np.abs(a[contributing] * a[peak_idx - contributing])
        top_contrib = contributing[np.argsort(contributions)[-20:]]

        # Try small adjustments on top contributors
        improved = False
        for i in top_contrib:
            if time.monotonic() - t0 > time_limit - 0.1:
                break
            for delta_frac in [0.01, 0.02, 0.05, -0.01, -0.02, -0.05]:
                delta = delta_frac * a[i]
                if a[i] + delta < 0:
                    continue
                # Find where to put the mass
                # Pick a position with low contribution to peak
                low_positions = np.setdiff1d(np.arange(n), contributing[:5])
                if len(low_positions) == 0:
                    continue
                j = rng.choice(low_positions)

                a_try = a.copy()
                a_try[i] += delta
                a_try[j] -= delta
                if a_try[j] < -1e-12:
                    continue
                a_try = np.maximum(a_try, 0)
                st = a_try.sum()
                if st < 1e-12:
                    continue
                a_try = a_try / st
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj - 1e-12:
                    a = a_try
                    best_obj = obj_try
                    best_a = a.copy()
                    conv = conv_try
                    improved = True
                    no_improve = 0
                    break
            if improved:
                break

        if not improved:
            no_improve += 1
            if no_improve > 100:
                break

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations for optimization."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # Sine-based
    inits['sine1'] = (np.sin(np.pi * x / n) ** 2 + 0.01)
    inits['sine1'] /= inits['sine1'].sum()

    inits['sine2'] = (np.sin(2 * np.pi * x / n) ** 2 + 0.01)
    inits['sine2'] /= inits['sine2'].sum()

    # Bimodal
    edge = max(1, n // 4)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.25 * n)) ** 2)
    inits['gaussian'] = g / g.sum()

    # Alternating
    alt = np.ones(n)
    alt[::2] = 1.3
    alt[1::2] = 0.7
    inits['alternating'] = alt / alt.sum()

    # Triangular (inverted)
    tri = 1.0 - 2.0 * np.abs(x / n - 0.5)
    tri = np.maximum(tri, 0.01)
    inits['triangular'] = tri / tri.sum()

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps

    return inits
