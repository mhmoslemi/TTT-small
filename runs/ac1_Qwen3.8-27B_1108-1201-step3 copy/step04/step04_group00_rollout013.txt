def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_gradient(a, conv, temp, top_k=200):
    """
    Compute gradient of 2*n*LSE(conv, temp) w.r.t. a.
    LSE(x, t) = t * log(sum(exp(x_i/t)))
    d LSE / dx_i = exp(x_i/t) / sum(exp(x_j/t)) = softmax(x/t)_i
    """
    n = len(a)
    c_max = conv.max()
    logits = (conv - c_max) / temp
    logits -= logits.max()
    exp_l = np.exp(logits)
    softmax_w = exp_l / exp_l.sum()
    
    # Focus on top peaks for efficiency
    actual_top_k = min(top_k, len(conv))
    top_indices = np.argpartition(softmax_w, -actual_top_k)[-actual_top_k:]
    sw = np.zeros(len(conv))
    sw[top_indices] = softmax_w[top_indices]
    # Renormalize
    sw_sum = sw.sum()
    if sw_sum > 0:
        sw /= sw_sum
    else:
        sw = softmax_w
    
    grad = np.zeros(n)
    for k in top_indices:
        k = int(k)
        w = sw[k]
        if w < 1e-14:
            continue
        
        # d conv[k] / d a[i] for term 1: a[k-i] valid when max(0,k-n+1) <= i <= min(n-1,k)
        i1s = max(0, k - n + 1)
        i1e = min(n - 1, k)
        if i1s <= i1e:
            idx = k - np.arange(i1e, i1s - 1, -1)
            grad[i1s:i1e+1] += w * a[idx]
        
        # d conv[k] / d a[i] for term 2: a[i-k] valid when max(0,k) <= i <= min(n-1, k+n-1)
        i2s = max(0, k)
        i2e = min(n - 1, k + n - 1)
        if i2s <= i2e:
            idx = np.arange(i2s, i2e + 1) - k
            grad[i2s:i2e+1] += w * a[idx]
    
    grad *= 2.0 * n
    return grad

def bb_step(f_curr, f_prev, s_vec, y_vec):
    """Barzilai-Borwein step size (two variants, pick safer one)."""
    sy = np.dot(s_vec, y_vec)
    if abs(sy) < 1e-30:
        return 0.01
    ss = np.dot(s_vec, s_vec)
    if ss < 1e-30:
        return 0.01
    yr = np.dot(y_vec, y_vec)
    if yr < 1e-30:
        return 0.01
    
    alpha_plus = ss / sy if sy > 0 else 0.01
    alpha_minus = ss / yr
    
    # Pick the one closer to previous step
    # For simplicity, use alpha_plus
    return max(1e-6, min(0.5, alpha_plus))
