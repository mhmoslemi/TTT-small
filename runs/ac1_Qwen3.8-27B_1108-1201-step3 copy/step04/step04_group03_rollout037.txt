    # For numerical stability, subtract max
    m = conv.max()

def smooth_grad(a, tau):
    """Gradient of smooth objective."""
    n = len(a)
    conv = fast_convolve(a)
    m = conv.max()
    weights = np.exp((conv - m) / tau) / np.sum(np.exp((conv - m) / tau))

        je = min(n - 1, k)
        idx = np.arange(js, je + 1)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth log-sum-exp approximation for differentiable optimization
2. Adam optimizer with simplex projection
3. Frank-Wolfe with LP oracle for structured refinement
4. Coordinate descent polish
5. Multi-length exploration (256, 384, 512)
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_max_conv(a, tau):
    """Log-sum-exp smooth approximation of max(conv(a,a))."""
    conv = fast_convolve(a)
    m = conv.max()
    if tau < 1e-10:
        return m
    shifted = (conv - m) / tau
    # For numerical stability
    shifted = np.minimum(shifted, 50)  # clip to avoid overflow
    return m + tau * np.log(np.sum(np.exp(shifted)))


def smooth_grad(a, tau):
    """Gradient of log-sum-exp smooth approximation of max(conv(a,a))."""
    n = len(a)
    conv = fast_convolve(a)
    m = conv.max()
    shifted = (conv - m) / tau
    shifted = np.minimum(shifted, 50)
    weights = np.exp(shifted)
    ws = weights.sum()
    weights /= ws
    grad = np.zeros(n)
    for k in range(len(conv)):
        if weights[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * weights[k] * a[k - idx]
    return grad


def adam_simplex(a0, time_limit, lr=0.01, tau_init=0.001, tau_decay=0.9999,
                 beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer with simplex projection and adaptive temperature."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    m_v = np.zeros(n)  # First moment
    v_v = np.zeros(n)  # Second moment
    t = 0
    cur_tau = tau_init
    best_obj = smooth_max_conv(a, cur_tau)
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        cur_tau = max(tau_init * (tau_decay ** t), 1e-6)
        
        grad = smooth_grad(a, cur_tau)
        # Project gradient onto tangent space of simplex
        grad_tan = grad - grad.mean()
        
        m_v = beta1 * m_v + (1 - beta1) * grad_tan
        v_v = beta2 * v_v + (1 - beta2) * grad_tan ** 2
        m_hat = m_v / (1 - beta1 ** t)
        v_hat = v_v / (1 - beta2 ** t)
        
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a = project_simplex(a_new)
        
        obj = smooth_max_conv(a, cur_tau)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Restart if stuck
        if no_improve > 200:
            rng = np.random.default_rng(int(time.monotonic() * 100000) % 1000000)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.002)
            m_v = np.zeros(n)
            v_v = np.zeros(n)
            no_improve = 0
            cur_tau = tau_init * 0.5
        
    # Return the actual max conv (not smooth approximation)
    final_conv = fast_convolve(best_a)
    return best_a, final_conv.max()


def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def fw_lp_optimize(f0, time_limit):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        g = solve_lp(f, M * m_exp)
        if g is None:
            m_exp *= 1.15
            if m_exp > 5.0:
                m_exp = 1.0
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-9:
            no_improve += 1
            if no_improve > 30:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue
        
        best_alpha = 0.0
        best_val = M
        
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.08, 4.0)
        
        if no_improve > 50:
            rng = np.random.default_rng(it * 31 + 17)
            scale = 0.002 + 0.004 * (no_improve // 80)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0
    
    return best_f, best_obj


def mirror_descent_peak(a0, time_limit, lr_init=0.02, top_k=20, temp_init=0.0005):
    """Multiplicative weight / mirror descent focusing on top peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    cur_temp = temp_init
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            w = np.ones(len(peaks)) / len(peaks)
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            pv = conv[peaks]
            scaled = pv / cur_temp
            scaled -= scaled.max()
            w = np.exp(scaled)
            ws = w.sum()
            if ws < 1e-15:
                w = np.zeros(len(peaks))
                w[np.argmax(pv)] = 1.0
            else:
                w /= ws
        
        log_update = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            log_update[idx] -= cur_lr * 2.0 * w[i] * a[k - idx]
        
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-25)
        a = a / a.sum()
        
        conv = fast_convolve(a)
        obj = conv.max()
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        cur_lr *= 0.9998
        cur_temp *= 1.002
        
        if no_improve > 100:
            rng = np.random.default_rng(int(time.monotonic() * 100000) % 1000000)
            a = best_a * np.exp(rng.standard_normal(n) * 0.003)
            a = a / a.sum()
            cur_lr = max(lr_init * 0.7, 0.005)
            cur_temp = temp_init * 1.5
            no_improve = 0
    
    return best_a, best_obj


def coordinate_polish(a0, time_limit, step_factors=None):
    """Coordinate descent: try adjusting each coordinate."""
    if step_factors is None:
        step_factors = [0.3, 0.5, 0.7, 0.9, 1.0, 1.1, 1.3, 1.5, 2.0]
    
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        improved = False
        for j in range(n):
            current_val = a[j]
            if current_val < 1e-15:
                # Try small positive values
                candidates = [0.001, 0.005, 0.01]
            else:
                candidates = [current_val * f for f in step_factors]
            
            best_cand = current_val
            best_cand_obj = best_obj
            
            for cand in candidates:
                if cand < 0:
                    continue
                a[j] = cand
                # Renormalize (cheap way: just adjust one other coordinate)
                # Actually, let's just recompute conv after normalization
                ss = a.sum()
                if ss < 1e-15:
                    a[j] = current_val
                    continue
                a_norm = a / ss
                c = fast_convolve(a_norm)
                obj = c.max()
                if obj < best_cand_obj - 1e-14:
                    best_cand_obj = obj
                    best_cand = cand

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth log-sum-exp approximation with Adam optimizer on simplex
2. Frank-Wolfe with LP oracle for structured refinement
3. Mirror descent (multiplicative weight) with peak weighting
4. Coordinate descent polish
5. Multi-phase time allocation with diverse initializations

Key insight: The subgradient of max(conv(a,a)) at peak k is 2*a shifted by k,
enabling efficient gradient computation. The log-sum-exp smoothing provides
a differentiable surrogate for the non-smooth max, allowing use of Adam.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_max_conv(a, tau):
    """Log-sum-exp smooth approximation of max(conv(a,a))."""
    conv = fast_convolve(a)
    m = conv.max()
    if tau < 1e-10:
        return m
    shifted = (conv - m) / tau
    shifted = np.minimum(shifted, 50.0)
    return m + tau * np.log(np.sum(np.exp(shifted)))


def smooth_grad(a, tau):
    """Gradient of log-sum-exp of conv(a,a)."""
    n = len(a)
    conv = fast_convolve(a)
    m = conv.max()
    shifted = (conv - m) / tau
    shifted = np.minimum(shifted, 50.0)
    weights = np.exp(shifted)
    ws = weights.sum()
    weights /= ws
    grad = np.zeros(n)
    for k in range(len(conv)):
        if weights[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * weights[k] * a[k - idx]
    return grad


def adam_simplex(a0, time_limit, lr=0.005, tau_init=0.001, tau_decay=0.9999,
                 beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer with simplex projection and adaptive temperature."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    m_v = np.zeros(n)
    v_v = np.zeros(n)
    t = 0
    cur_tau = tau_init
    best_obj = smooth_max_conv(a, cur_tau)
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        t += 1
        cur_tau = max(tau_init * (tau_decay ** t), 1e-6)

        grad = smooth_grad(a, cur_tau)
        grad_tan = grad - grad.mean()

        m_v = beta1 * m_v + (1 - beta1) * grad_tan
        v_v = beta2 * v_v + (1 - beta2) * grad_tan ** 2
        m_hat = m_v / (1 - beta1 ** t)
        v_hat = v_v / (1 - beta2 ** t)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a = project_simplex(a_new)

        obj = smooth_max_conv(a, cur_tau)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 100000) % 1000000)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            m_v = np.zeros(n)
            v_v = np.zeros(n)
            t = 0
            no_improve = 0
            cur_tau = tau_init * 0.5

    final_conv = fast_convolve(best_a)
    return best_a, final_conv.max()


def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def mirror_descent_peak(a0, time_limit, lr_init=0.02, top_k=20, temp_init=0.0005):

    cur_temp = temp_init
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)

        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            w = np.ones(len(peaks)) / len(peaks)

            scaled -= scaled.max()

            cur_lr = max(lr_init * 0.7, 0.005)

        a = np.ones(len(a)) / len(a)
    else:
        a = a / s

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

            current_val = a[j]

            else:
                candidates = [current_val * f for f in step_factors]

            best_cand_obj = best_obj

                    continue

                ss = a.sum()

                c = fast_convolve(a_norm)
                obj = c.max()
                if obj < best_cand_obj - 1e-14:
                    best_cand_obj = obj
                    best_cand = cand

            if best_cand != current_val:
                a[j] = best_cand
                ss = a.sum()
                a = a / ss
                conv = fast_convolve(a)
                obj = conv.max()
                if obj < best_obj - 1e-14:

                    improved = True
                else:

                    ss = a.sum()
                    a = a / ss

        if not improved and it > 3:
            break

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    x = np.arange(n, dtype=np.float64)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits.append(g / g.sum())

    c1 = np.abs(np.cos(np.pi * x / n)) + 0.05
    inits.append(c1 / c1.sum())

    asym = np.exp(-0.5 * ((x - 0.3 * (n - 1)) / (0.25 * n)) ** 2)
    inits.append(asym / asym.sum())

    bump1 = np.exp(-0.5 * ((x - 0.25 * (n - 1)) / (0.12 * n)) ** 2)
    bump2 = np.exp(-0.5 * ((x - 0.65 * (n - 1)) / (0.12 * n)) ** 2)
    inits.append((bump1 + 0.6 * bump2) / (bump1 + 0.6 * bump2).sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search combining smooth optimization,
    Frank-Wolfe LP, mirror descent, and coordinate polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            if np.isfinite(ps) and ps < best_score:
                best_score = ps
                best_seq = parent.copy()
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(384) / 384
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
        print(f"Using uniform parent: n=384, score={best_score:.8f}")

    n_p = len(parent)

    def _eval(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        ss = a.sum()
        if ss < 1e-15:
            return float('inf')
        a_norm = a / ss
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # === Phase 1: Quick exploration (~250s) ===
    print("\n=== Phase 1: Quick Exploration ===")

    rem = deadline - time.monotonic() - 1400
    if rem > 20:
        t_alloc = min(rem, 120)
        print(f"  FW-LP on parent (n={n_p}), t={t_alloc:.0f}s")
        a_opt, _ = fw_lp_optimize(parent, time_limit=t_alloc)
        _eval(a_opt, "fw_parent")

    rem = deadline - time.monotonic() - 1200
    if rem > 20:
        t_alloc = min(rem, 80)
        print(f"  Adam on parent (n={n_p}), t={t_alloc:.0f}s")
        a_opt, _ = adam_simplex(parent, time_limit=t_alloc, lr=0.005, tau_init=0.002)
        _eval(a_opt, "adam_parent")

    # Try alternative initializations
    rem = deadline - time.monotonic() - 1100
    if rem > 20:
        inits = generate_inits(n_p, parent, rng)
        for i, init in enumerate(inits[1:5]):
            rem = deadline - time.monotonic() - 1000
            if rem < 15:
                break
            t_alloc = min(rem, 35)
            print(f"  Adam on init {i+1}, t={t_alloc:.0f}s")
            a_opt, _ = adam_simplex(init, time_limit=t_alloc, lr=0.01, tau_init=0.003)
            _eval(a_opt, f"adam_init_{i}")

    print(f"  Phase 1 best: {best_score:.8f}")

    # === Phase 2: Deep Adam optimization (~600s) ===
    print("\n=== Phase 2: Deep Adam Optimization ===")

    if best_seq is not None:
        adam_configs = [
            {'lr': 0.003, 'tau_init': 0.001, 'tau_decay': 0.9999},
            {'lr': 0.005, 'tau_init': 0.002, 'tau_decay': 0.9998},
            {'lr': 0.002, 'tau_init': 0.0005, 'tau_decay': 0.99995},
            {'lr': 0.007, 'tau_init': 0.003, 'tau_decay': 0.9997},
            {'lr': 0.004, 'tau_init': 0.0015, 'tau_decay': 0.99985},
        ]

        rem = deadline - time.monotonic() - 550
        t_per = max(20, rem / len(adam_configs))

        for ci, cfg in enumerate(adam_configs):
            rem = deadline - time.monotonic() - 450
            if rem < 20:
                break
            t_alloc = min(t_per, rem)
            print(f"  Adam config {ci} (lr={cfg['lr']}, tau={cfg['tau_init']}), t={t_alloc:.0f}s")
            a_opt, _ = adam_simplex(best_seq, time_limit=t_alloc, **cfg)
            _eval(a_opt, f"adam_deep_{ci}")

    print(f"  Phase 2 best: {best_score:.8f}")

    # === Phase 3: Deep FW-LP (~350s) ===
    print("\n=== Phase 3: Deep FW-LP ===")

    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 30:
            t_alloc = min(rem, 200)
            print(f"  Deep FW-LP, t={t_alloc:.0f}s")
            a_opt, _ = fw_lp_optimize(best_seq, time_limit=t_alloc)
            _eval(a_opt, "deep_fw")

            rem = deadline - time.monotonic() - 200
            if rem > 20:
                noise = rng.standard_normal(len(best_seq)) * 0.005
                f_pert = project_simplex(best_seq + noise)
                a_opt2, _ = fw_lp_optimize(f_pert, time_limit=min(rem, 80))
                _eval(a_opt2, "fw_perturbed")

    print(f"  Phase 3 best: {best_score:.8f}")

    # === Phase 4: Mirror Descent (~200s) ===
    print("\n=== Phase 4: Mirror Descent ===")

    if best_seq is not None:
        md_configs = [
            {'lr_init': 0.02, 'top_k': 15, 'temp_init': 0.0003},
            {'lr_init': 0.03, 'top_k': 25, 'temp_init': 0.0005},
            {'lr_init': 0.015, 'top_k': 10, 'temp_init': 0.0002},
            {'lr_init': 0.025, 'top_k': 20, 'temp_init': 0.0004},
        ]

        rem = deadline - time.monotonic() - 150
        t_per = max(15, rem / len(md_configs))

        for ci, cfg in enumerate(md_configs):
            rem = deadline - time.monotonic() - 100
            if rem < 15:
                break
            t_alloc = min(t_per, rem)
            print(f"  MD config {ci} (lr={cfg['lr_init']}, k={cfg['top_k']}), t={t_alloc:.0f}s")
            a_opt, _ = mirror_descent_peak(best_seq, time_limit=t_alloc, **cfg)
            _eval(a_opt, f"md_{ci}")

    print(f"  Phase 4 best: {best_score:.8f}")

    # === Phase 5: Polish (~100s) ===
    print("\n=== Phase 5: Final Polish ===")

    if best_seq is not None:
        # Coordinate descent
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            t_alloc = min(rem, 50)
            print(f"  Coordinate polish, t={t_alloc:.0f}s")
            a_opt, _ = coordinate_polish(best_seq, time_limit=t_alloc)
            _eval(a_opt, "coord")

        # Fine Adam
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            print(f"  Fine Adam, t={min(rem, 30):.0f}s")
            a_opt, _ = adam_simplex(best_seq, time_limit=min(rem, 30),
                                    lr=0.001, tau_init=0.0005, tau_decay=0.99995)
            _eval(a_opt, "fine_adam")

        # Tiny perturbations
        rem = deadline - time.monotonic() - 10
        if rem > 5:
            for pi in range(3):
                rem = deadline - time.monotonic() - 5
                if rem < 5:
                    break
                noise = rng.standard_normal(len(best_seq)) * 0.001
                f_pert = project_simplex(best_seq + noise)
                a_opt, _ = adam_simplex(f_pert, time_limit=min(rem, 10),
                                        lr=0.002, tau_init=0.001)
                _eval(a_opt, f"perturb_{pi}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
