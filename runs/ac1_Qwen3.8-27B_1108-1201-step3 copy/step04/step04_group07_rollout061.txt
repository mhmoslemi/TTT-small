For f = 1 (uniform), (f*f)(k) = min(k, 1, 2-k, ...), which has peak 1 at k=0.5 (in the continuous case). Actually for the discrete case with n elements, uniform gives peak n*(1/n)^2 = 1/n at the center.

For a Gaussian f(x) = exp(-x^2/(2σ^2)) normalized, the auto-convolution is a wider Gaussian with peak higher than the uniform case for the same support... hmm, this is getting complicated.

```python
"""
Multi-phase optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Final version combining improved LP, Nesterov subgradient, coordinate descent,
and multi-length exploration to beat the 1.5027 target.

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for i in range(n):
        C[i:i + n, i] = a
    return C


def obj_value(a):
    """Compute 2n * max(conv(a,a)) for the given (possibly unnormalized) a."""
    s = a.sum()
    if s < 1e-15:
        return float('inf'), np.zeros(2*len(a)-1)
    a_norm = a / s
    conv = fast_convolve(a_norm)
    return 2.0 * len(a_norm) * conv.max(), conv


def solve_lp(a, M, C):
    """Solve LP: maximize sum(g) s.t. C@g <= M, g >= 0. Returns (g_normalized, g_sum)."""
    n = len(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        result = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs',
                         options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if result.success:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g / g_sum, g_sum
    except Exception:
        pass
    return None, 0.0


def line_search(a, d, current_obj, max_evals=20):
    """Backtracking + golden section line search."""
    n = len(a)
    best_t = 0.0
    best_obj = current_obj
    
    # Coarse scan
    for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
        a_try = np.maximum(a + t * d, 0.0)
        s = a_try.sum()
        if s < 1e-15:
            continue
        a_try = a_try / s
        obj_try = obj_value(a_try)[0]
        if obj_try < best_obj:
            best_obj = obj_try
            best_t = t
    
    # Golden section refinement around best_t
    if 0.02 < best_t < 0.98:
        gr = (np.sqrt(5) - 1) / 2
        lo = max(0.005, best_t * 0.7)
        hi = min(1.5, best_t * 1.3)
        
        for _ in range(6):
            mid1 = lo + (1 - gr) * (hi - lo)
            mid2 = lo + gr * (hi - lo)
            a1 = np.maximum(a + mid1 * d, 0.0)
            a2 = np.maximum(a + mid2 * d, 0.0)
            s1, s2 = a1.sum(), a2.sum()
            o1 = obj_value(a1/s1)[0] if s1 > 1e-15 else current_obj
            o2 = obj_value(a2/s2)[0] if s2 > 1e-15 else current_obj
            if o1 < o2:
                hi = mid2
            else:
                lo = mid1
            if hi - lo < 0.005:
                break
        mid = (lo + hi) / 2
        a_mid = np.maximum(a + mid * d, 0.0)
        s_mid = a_mid.sum()
        if s_mid > 1e-15:
            o_mid = obj_value(a_mid / s_mid)[0]
            if o_mid < best_obj:
                best_obj = o_mid
                best_t = mid
    
    return best_t, best_obj

            perturb_count += 1

def nesterov_sg(a0, time_limit, lr_init=0.004, momentum=0.94, top_k=150, temp=0.0002):
    """Nesterov-accelerated subgradient with softmax peak weighting."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    v = np.zeros(n)
    y = a.copy()  # Nesterov lookahead
    _, conv = obj_value(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    step = 0
    
    while time.monotonic() - t0 < time_limit:
        # Nesterov lookahead
        y = a + momentum * v
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y
        
        # Compute subgradient at y
        conv_y = fast_convolve(y)
        
        # Softmax weights over top peaks
        idx = np.argpartition(conv_y, -top_k)[-top_k:]
        vals = conv_y[idx]
        scaled = vals / temp - vals.max()
        w = np.exp(scaled)
        w /= w.sum()
        
        sg = np.zeros(n)
        for ii in range(len(idx)):
            if w[ii] < 1e-12:
                continue
            k = int(idx[ii])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx_arr = np.arange(j_start, j_end + 1)
            sg[idx_arr] += w[ii] * 2.0 * y[k - idx_arr]
        
        v = momentum * v - lr * sg
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()
            v = np.zeros(n)
        
        step += 1
        obj_cur, _ = obj_value(a)
        
        if obj_cur < best_obj - 1e-13:
            best_obj = obj_cur
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive learning rate
        if no_improve % 100 == 0 and no_improve > 0:
            lr *= 0.995
            temp *= 1.005
        
        if no_improve >= 500:
            rng = np.random.default_rng(step * 3 + 11)
            a = best_a + rng.standard_normal(n) * 0.0015
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            v = np.zeros(n)
            no_improve = 0
    
    return best_a, best_obj


def coordinate_descent(a0, time_limit, n_perturb=50):
    """Coordinate descent: optimize one element at a time."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    best_obj = obj_value(a)[0]
    best_a = a.copy()
    
    t0 = time.monotonic()
    idx_order = np.random.default_rng(42).permutation(n)
    
    while time.monotonic() - t0 < time_limit:
        for start in range(0, n, n_perturb):
            if time.monotonic() - t0 > time_limit:
                break
            for ii in range(n_perturb):
                i = idx_order[(start + ii) % n]
                # Try to optimize element i
                # (a*a)[k] depends on a[i] through terms involving a[i]*a[k-i] and a[k-i]*a[i]
                # For the peak, we want to reduce a[i] if it contributes to the peak
                old_val = a[i]
                if old_val < 1e-10:
                    continue
                
                # Try small decrease
                for frac in [0.98, 0.95, 0.9, 0.8]:
                    a[i] = old_val * frac
                    s_new = a.sum()
                    if s_new < 1e-15:
                        a[i] = old_val
                        break
                    obj_try = obj_value(a / s_new)[0]
                    if obj_try < best_obj:
                        best_obj = obj_try
                        best_a = a.copy()
                        break
                a[i] = old_val
        
        idx_order = np.random.default_rng(int(time.monotonic() * 1000)).permutation(n)
    
    return best_a, best_obj


def resample(seq, new_n):
    """Resample a normalized sequence to a new length."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s > 1e-15:
        new_seq = new_seq / s
    return new_seq


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    if parent is not None:
        inits['parent'] = resample(parent, n)
    
    # Raised cosine
    rc = np.maximum(np.cos(np.pi * (x - 0.5 * n) / n), 0)
    inits['raised_cos'] = rc / rc.sum()
    
    # Sine
    sine = 0.6 + 0.4 * np.sin(np.pi * x / n)
    inits['sine'] = sine / sine.sum()
    
    # Triangular
    tri = 1.0 - np.abs(x - 0.5 * (n - 1)) / (0.5 * (n - 1))
    inits['tri'] = np.maximum(tri, 0.1) / np.maximum(tri, 0.1).sum()
    
    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.15 * n)) ** 2)
    inits['gauss'] = g / g.sum()
    
    # Flat top
    ft = np.ones(n)
    ft[:n//10] *= 0.5
    ft[-n//10:] *= 0.5
    inits['flattop'] = ft / ft.sum()
    
    # Dirichlet
    inits['dirichlet'] = rng.dirichlet(np.ones(n) * 0.3)
    
    # Random smooth
    freq = rng.integers(1, 5)
    smooth = 0.7 + 0.3 * np.sin(2 * np.pi * freq * x / n + rng.uniform(0, 2*np.pi))
    inits['rand_smooth'] = smooth / smooth.sum()
    
    return inits
```

Hmm, let me reconsider. The coordinate descent as I wrote it is very expensive because it calls obj_value (which does an FFT convolution) for each element perturbation. For n=768, that's 768 evaluations per pass, each requiring an FFT of size ~2048. That's too slow.

Let me replace coordinate descent with something more efficient.

Also, I realize I should think more carefully about what will actually give the 0.2% improvement needed.

Let me think about this differently. The key question is: what's the bottleneck preventing the previous algorithm from reaching 1.5027?

1. **Convergence speed**: The LP might not be converging to a truly optimal direction due to numerical precision.
2. **Local minima**: The algorithm might be getting stuck.
3. **Length choice**: Maybe a different length would work better.
4. **Initialization**: Maybe the initializations aren't good enough.

Let me focus on a different strategy: **peak reduction targeting**.

The idea: at each step, identify the single maximum peak of the auto-convolution. Then find the direction that most reduces that peak while keeping other peaks from increasing too much.

This is equivalent to solving:
```
minimize (a+d)*(a+d) at the peak location k*
subject to (a+d)*(a+d) at other locations <= M + ε
           sum(a+d) = 1
           a+d >= 0
```

For small d, this becomes:
```
minimize 2 * a * d at k*  (linearized)
subject to 2 * a * d at k <= 2*M - 2*(a*a)[k] + ε  for k != k*
           sum(d) = 0
           a + d >= 0
```

This is a linear program! And it's specifically targeting the maximum peak.

Actually, this is very similar to what the original LP does, but with a tighter focus.

Let me try a different approach: **iterative peak reduction with linearized constraints**.

At each step:
1. Find the peak k* of the auto-convolution.
2. Solve: minimize 2 * sum_i a[i] * d[i] * (contribution to peak k*)
   subject to: for each other peak k, 2 * sum_i a[i-k] * d[i] <= (a*a)[k] + slack
   sum(d) = 0
   a + d >= 0

This is getting complicated. Let me just stick with the LP approach but make it more efficient and add better heuristics.

Actually, let me try a completely different angle. What if I use **CVXPY** to solve the problem directly?

The problem is:
```
minimize t
subject to sum_i a[i]*a[k-i] <= t  for all k = 0, ..., 2n-2
           sum(a) = 1
           a >= 0
```

    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        C = build_conv_matrix(a)
        M_lp = M * tightness
        g, g_sum = solve_lp(a, M_lp, C)

        if g is None or g_sum <= 1.0 + 1e-7:
            if tightness < 1.0:
                tightness = min(1.002, tightness * 1.01)
            else:
                no_improve += 1
        else:
            d = g - a
            if np.linalg.norm(d) < 1e-12:
                no_improve += 1
            else:
                t_best, obj_best = line_search(a, d, best_obj)
                if obj_best < best_obj - 1e-11:
                    a_new = np.maximum(a + t_best * d, 0.0)
                    s_new = a_new.sum()
                    if s_new > 1e-15:
                        a = a_new / s_new
                    else:
                        a = best_a.copy()
                    _, conv = obj_value(a)
                    M = conv.max()
                    best_obj = 2.0 * n * M
                    best_a = a.copy()
                    no_improve = 0
                    tightness = max(0.9985, tightness - 0.0002)
                    if verbose and iteration % 20 == 0:
                        print(f"    LP iter {iteration}, obj={best_obj:.8f}, t={time.monotonic()-t0:.1f}s")
                else:
                    no_improve += 1

        iteration += 1

        if no_improve >= max_no_improve:
            # Peak-targeted perturbation (mix with random)
            if perturb_count % 3 == 0:
                _, conv_cur = obj_value(best_a)
                scale = 0.001 + 0.0015 * min(perturb_count, 10)
                a = peak_targeted_perturbation(best_a, conv_cur, rng, scale)
            else:
                scale = 0.001 + 0.0012 * min(perturb_count, 10)
                a = best_a + rng.standard_normal(n) * scale
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s < 1e-15:
                    a = best_a.copy()
                else:
                    a = a / s

            _, conv = obj_value(a)
            M = conv.max()
            no_improve = 0
            perturb_count += 1
            max_no_improve = min(200, 50 + 30 * perturb_count)
            tightness = 1.001
            if perturb_count > 35:
                break

    return best_a, best_obj


def nesterov_sg(a0, time_limit, lr_init=0.004, momentum=0.94, top_k=150, temp=0.0002):
    """Nesterov-accelerated subgradient with softmax peak weighting."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    _, conv = obj_value(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    step = 0

    while time.monotonic() - t0 < time_limit:
        # Nesterov lookahead
        y = a + momentum * v
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y

        conv_y = fast_convolve(y)

        # Softmax weights over top peaks
        idx = np.argpartition(conv_y, -top_k)[-top_k:]
        vals = conv_y[idx]
        scaled = vals / temp - vals.max()
        w = np.exp(scaled)
        w /= w.sum()

        # Compute subgradient
        sg = np.zeros(n)
        for ii in range(len(idx)):
            if w[ii] < 1e-12:
                continue
            k = int(idx[ii])
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            idx_arr = np.arange(j_start, j_end + 1)
            sg[idx_arr] += w[ii] * 2.0 * y[k - idx_arr]

        v = momentum * v - lr * sg
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()
            v = np.zeros(n)

        step += 1
        obj_cur = obj_value(a)[0]

        if obj_cur < best_obj - 1e-13:
            best_obj = obj_cur
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve % 80 == 0 and no_improve > 0:
            lr *= 0.996
            temp *= 1.004

        if no_improve >= 400:
            rng = np.random.default_rng(step * 3 + 11)
            a = best_a + rng.standard_normal(n) * 0.0012
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            v = np.zeros(n)
            no_improve = 0

    return best_a, best_obj
