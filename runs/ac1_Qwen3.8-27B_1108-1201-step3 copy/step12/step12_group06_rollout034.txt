    n = len(a)

        return np.ones(n) / n

    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    theta = (cssv[rho[-1]] - 1) / (rho[-1] + 1.0)
    return np.maximum(v - theta, 0)

def fast_convolve(a, b=None):
    """Fast linear convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    if b is None:
        b = a
    else:
        b = np.asarray(b, dtype=np.float64)
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def fast_auto_convolve(a):
    """Fast linear autocovolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def softmax_from_z(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

    return np.log(np.maximum(a, 1e-300))


def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_idx = np.where(cond)[0]
    if len(rho_idx) == 0:
        return np.ones(n) / n
    rho = rho_idx[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def lse_objective_grad_z(z_val, T, n):

    a = softmax_from_z(z_val)
    b = fast_auto_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    if max_Tb > 700.0:
        max_Tb = 700.0
    exp_Tb = np.exp(Tb - (b.max() if b.max() > 0 else 0))
    exp_Tb = np.exp(T * b - T * b.max())
    sum_exp = exp_Tb.sum()
    if sum_exp < 1e-300:
        w = np.ones(len(b)) / len(b)
        log_sum = 0.0
    else:
        w = exp_Tb / sum_exp
        log_sum = T * b.max() + np.log(sum_exp)

    obj = (2.0 * n / T) * log_sum

    # Gradient w.r.t. a: (4n/T) * g where g[i] = sum_k w[k]*a[k-i]
    g = compute_lse_grad_a(a, w, n)
    grad_a = (4.0 * n / T) * g

    # Gradient w.r.t. z: a * (grad_a - <grad_a, a>)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)

    return obj, grad_z


def lse_lbfgs_optimize(a0, time_limit, T_start=80.0, T_min=1.0, n_stages=30, iters_per_stage=50):
    """Adaptive LogSumExp L-BFGS with progress-based temperature scheduling."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)

    stagnant = 0
    prev_score = best_score

    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break

        frac = stage / max(n_stages - 1, 1)
        T = T_start * (T_min / T_start) ** frac
        T = max(T, T_min)

        def func_staged(zv):
            o, _ = lse_objective_grad_z(zv, T, n)
            return o

        def jac_staged(zv):
            _, gr = lse_objective_grad_z(zv, T, n)
            return gr

        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-12, 'maxfun': iters_per_stage * 500,
                         'disp': False}
            )
            z = res.x
        except Exception:
            pass

        a_check = softmax_from_z(z)
        sc = score_normalized(a_check)

        if sc < best_score:
            best_score = sc
            best_a = a_check.copy()

        # Adaptive temperature
        if prev_score - sc < 1e-8:
            stagnant += 1
        else:
            stagnant = max(0, stagnant - 1)

        if stagnant >= 3:
            T = max(T_min, T * 0.8)
            stagnant = 0
        prev_score = sc

        if (stage + 1) % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage={stage+1}/{n_stages} T={T:.2f}: score={sc:.8f} best={best_score:.8f} ({elapsed:.0f}s)")

    return best_a, best_score


def build_conv_matrix_sparse(a_fixed):
    """Build sparse convolution matrix T where (T@b)[k] = (a_fixed*b)[k]."""
    n = len(a_fixed)
    size = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for k in range(j, j + n):
            if k < size:
                rows.append(k)
                cols.append(j)
                vals.append(a_fixed[k - j])
    T = csr_matrix((vals, (rows, cols)), shape=(size, n))
    return T


def get_lp_solver():

def solve_lp_best_response(a_fixed, solver):
    """Solve the LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix_sparse(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        # Fallback: use current a
        return a_fixed.copy(), None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return a_fixed.copy(), None
    b_val = b_val / s
    m_val = float(M_var.value) if M_var.value is not None else None
    return b_val, m_val

    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_print = t0
    prev_best = best_score

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * float(it * it))) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 10:
                print(f"    NLP iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} M={M_lp if M_lp else 'N/A'}")
                last_print = elapsed
                prev_best = best_score

    return best_a, best_score


def lse_direct_optimize(a0, time_limit, T=30.0, n_steps=50000):
    """Direct LSE optimization with backtracking line search in z-space."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        obj, grad_z = lse_objective_grad_z(z, T, n)
        grad_norm = np.linalg.norm(grad_z)
        if grad_norm < 1e-14:
            break

        # Backtracking line search
        eta = 1.0 / (step + 1)
        for ls in range(20):
            z_new = z - eta * grad_z
            a_new = softmax_from_z(z_new)
            sc_new = score_normalized(a_new)
            if sc_new < obj or np.isfinite(sc_new):
                break
            eta *= 0.5
        else:
            break

        z = z - eta * grad_z
        a = softmax_from_z(z)
        sc = score_normalized(a)

        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if (step + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-Direct step={step+1} ({elapsed:.0f}s): score={sc:.8f} best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n, endpoint=False)
    new_x = np.linspace(0, 1, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate structured initial sequences."""
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif choice == 2:
        x = np.linspace(-1, 1, n)
        a = np.exp(-2 * x ** 2)
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    elif choice == 4:
        a = np.ones(n) * (1.0 + 0.2 * np.cos(2 * np.pi * 3 * np.arange(n) / n))
    elif choice == 5:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(3 * x) + 0.15 * np.cos(7 * x)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = np.abs(np.sin(np.pi * x) * np.cos(np.pi * x / 3))
    else:
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.5) ** 2) / 0.08) + 0.2 * np.exp(-((x - 0.2) ** 2) / 0.03) + 0.2 * np.exp(-((x - 0.8) ** 2) / 0.03)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def nesterov_lp_fast(a0, time_limit, solver, max_iters=200000, use_approx=False):
    """
    Nesterov LP with optional approximate best response.
    When use_approx=True, uses a few LSE gradient steps instead of full LP solve.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_print = t0
    lp_count = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        if use_approx:
            # Approximate best response: a few gradient steps on LSE of (y*b)
            b = y.copy()
            for inner in range(30):
                yb = fast_convolve(y, b)
                T_inner = 50.0
                Tb = T_inner * yb
                max_Tb = Tb.max()
                exp_Tb = np.exp(Tb - max_Tb)
                w = exp_Tb / exp_Tb.sum()
                # grad_b[i] = sum_k w[k] * y[k-i] = cross-correlation (w ⋆ y)[i]
                g_b = compute_lse_grad_a(y, w, n)
                g_b = -g_b  # negative gradient (minimizing)
                lr = 0.5 / (inner + 1)
                b = b + lr * g_b
                b = project_simplex(b)
            b_new = b
        else:
            b_new, M_lp = solve_lp_best_response(y, solver)
            if b_new is None:
                continue
            lp_count += 1

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * float(it * it))) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 15:
                print(f"    NLP-F iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} laps={lp_count}")
                last_print = elapsed

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function: orchestrates multiple optimization phases."""
    rng = np.random.default_rng(seed)
    global_deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Total budget: {budget_s:.0f}s, effective: {global_deadline - time.monotonic():.0f}s")

    # Get parent sequence
    best_seq = None
    best_score = float('inf')
    n_base = 1200

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            n_base = len(best_seq)
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.8f}")
    except Exception:
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            remain = global_deadline - time.monotonic()
            print(f"  ** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, remain={remain:.0f}s)")

    # === Phase 1: Nesterov LP (exact) - 40% time ===
    print(f"\n=== Phase 1: Nesterov LP at n={n_base} (score={best_score:.8f}) ===")
    t1 = 0.40 * (budget_s - 25)
    remain = global_deadline - time.monotonic()
    t1 = min(t1, remain)
    a_out, _ = nesterov_lp_optimize(best_seq, t1, solver)
    try_update(a_out, "nlp_exact")
    print(f"  Phase 1 done: best={best_score:.8f}")

    # === Phase 2: LSE L-BFGS - 20% time ===
    print(f"\n=== Phase 2: LSE L-BFGS (score={best_score:.8f}) ===")
    remain = global_deadline - time.monotonic()
    t2 = min(0.20 * (budget_s - 25), remain * 0.6)
    if t2 > 30:
        # High-T pass
        a_l1, _ = lse_lbfgs_optimize(best_seq, t2 * 0.5, T_start=100.0, T_min=2.0, n_stages=20, iters_per_stage=40)
        try_update(a_l1, "lse_highT")

        remain = global_deadline - time.monotonic()
        if remain > 20:
            # Medium-T pass from best
            a_l2, _ = lse_lbfgs_optimize(best_seq, t2 * 0.5, T_start=20.0, T_min=1.0, n_stages=15, iters_per_stage=35)
            try_update(a_l2, "lse_medT")

        remain = global_deadline - time.monotonic()
        if remain > 15:
            # Low-T polish
            a_l3, _ = lse_lbfgs_optimize(best_seq, remain * 0.6, T_start=5.0, T_min=0.5, n_stages=10, iters_per_stage=30)
            try_update(a_l3, "lse_lowT")
    print(f"  Phase 2 done: best={best_score:.8f}")

    # === Phase 3: Approximate Nesterov (faster iterations) - 20% time ===
    print(f"\n=== Phase 3: Approximate Nesterov (score={best_score:.8f}) ===")
    remain = global_deadline - time.monotonic()
    t3 = min(0.20 * (budget_s - 25), remain * 0.6)
    if t3 > 30:
        a_n2, _ = nesterov_lp_fast(best_seq, t3, solver, use_approx=True)
        try_update(a_n2, "nlp_approx")

        remain = global_deadline - time.monotonic()
        if remain > 20:
            # Also try exact LP from the approximate result
            a_n3, _ = nesterov_lp_optimize(a_n2, remain * 0.5, solver)
            try_update(a_n3, "nlp_exact2")
    print(f"  Phase 3 done: best={best_score:.8f}")

    # === Phase 4: Multi-length exploration - 15% time ===
    print(f"\n=== Phase 4: Multi-length (score={best_score:.8f}) ===")
    remain = global_deadline - time.monotonic()
    t4_total = min(0.15 * (budget_s - 25), remain * 0.7)
    if t4_total > 40:
        lengths = [n_base + 200, n_base + 400, n_base - 200]
        for n_try in lengths:
            remain = global_deadline - time.monotonic()
            if remain < 30:
                break
            t_per = min(t4_total / 3.0, remain * 0.45)
            if t_per < 20:
                break
            print(f"  Trying n={n_try}, budget={t_per:.0f}s")
            init_try = interpolate_sequence(best_seq, n_try)
            # Quick Nesterov
            a_try, _ = nesterov_lp_optimize(init_try, t_per * 0.5, solver, max_iters=50000)
            try_update(a_try, f"len{n_try}_nlp")
            # Quick LSE
            remain = global_deadline - time.monotonic()
            if remain > 15:
                a_try2, _ = lse_lbfgs_optimize(a_try, remain * 0.5, T_start=30.0, T_min=1.0, n_stages=10, iters_per_stage=30)
                try_update(a_try2, f"len{n_try}_lse")
    print(f"  Phase 4 done: best={best_score:.8f}")

    # === Phase 5: Final polish - remaining time ===
    print(f"\n=== Phase 5: Final polish (score={best_score:.8f}) ===")
    remain = global_deadline - time.monotonic()
    if remain > 20:
        # Short Nesterov LP
        a_f1, _ = nesterov_lp_optimize(best_seq, remain * 0.3, solver, max_iters=30000)
        try_update(a_f1, "final_nlp")

        remain = global_deadline - time.monotonic()
        if remain > 15:
            # Short LSE from very low T
            a_f2, _ = lse_lbfgs_optimize(best_seq, remain * 0.5, T_start=3.0, T_min=0.3, n_stages=8, iters_per_stage=25)
            try_update(a_f2, "final_lse")

        remain = global_deadline - time.monotonic()
        if remain > 10:
            # Direct LSE gradient descent
            a_f3, _ = lse_direct_optimize(best_seq, remain * 0.6, T=20.0, n_steps=20000)
            try_update(a_f3, "final_direct")

        remain = global_deadline - time.monotonic()
        if remain > 10:
            # One more Nesterov round
            a_f4, _ = nesterov_lp_optimize(best_seq, remain * 0.7, solver, max_iters=20000)
            try_update(a_f4, "final_nlp2")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"Time used: {time.monotonic() - (global_deadline - (budget_s - 25)):.0f}s")
    return [float(x) for x in best_seq]
