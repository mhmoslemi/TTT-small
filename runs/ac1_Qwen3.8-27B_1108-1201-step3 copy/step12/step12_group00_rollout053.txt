def build_conv_matrix(a):
    """Build the convolution matrix T such that (T@b)[k] = sum_j a[|k-j|]*b[j] for the full convolution."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, M, solver):
    """Solve the LP: max sum(b) s.t. T@b <= M, b >= 0"""
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = build_conv_matrix(a)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    return b_val, s

            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

def dual_averaging_nesterov(a0, time_limit, verbose=False):
    """Dual-averaging Nesterov for the min-max game."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    # Dual variable: history of gradients
    dual_grad = np.zeros(n)
    sum_t = 0.0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(y)
        k_star = int(np.argmax(b))
        M_curr = b[k_star]

        # Subgradient of max_k conv(y,y)_k w.r.t. y
        grad = np.zeros(n)
        for i in range(n):
            j1 = k_star - i
            if 0 <= j1 < n:
                grad[i] += y[j1]
            j2 = i - k_star
            if 0 <= j2 < n:
                grad[i] += y[j2]
        grad *= 2.0

        # Dual averaging update
        sum_t += 1.0
        dual_grad += grad / sum_t
        
        # Primal: project -dual_grad onto the L1 ball (non-negative, fixed sum)
        # Mirror descent with entropy: a[i] ∝ exp(-dual_grad[i])
        # But we need to maintain the constraint sum = norm
        max_dg = dual_grad.max()
        a_new = np.exp(-dual_grad - max_dg)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        # Nesterov momentum
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * a_new

        y = y_new
        a = a_new

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose and t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    DA-Nest t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def coord_descent_polish(a, time_limit, max_sweeps=50):
    """Coordinate descent: for each coordinate, try moving mass to reduce peak."""
    n = len(a)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for sweep in range(max_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        improved = False
        
        # Random order for coordinates
        order = rng.permutation(n)
        
        for i in order:
            a[i] = max(a[i], 0.0)
            if a[i] < 1e-10:
                continue
            
            # Compute current autocorrelation
            b = fast_autocorr(a)
            peak = b.max()
            
            # Try moving a small fraction from i to a random position
            for frac in [0.005, 0.01, 0.02, 0.05]:
                amt = a[i] * frac
                if amt < 1e-15:
                    continue
                # Try moving to a few random positions
                for _ in range(3):
                    j = rng.integers(n)
                    if j == i:
                        continue
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] < 0:
                        continue
                    s_t = a_trial.sum()
                    if s_t < 1e-15:
                        continue
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            
            if time.monotonic() - t0 > time_limit:
                break

        if not improved:
            break

    return best_a, best_score


def spectral_init(n, rng):
    """Generate a sequence using spectral shaping."""
    # Start with a random spectrum and iteratively shape it
    a = rng.random(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s
    
    for _ in range(50):
        # Compute spectrum
        fft_size = 1 << (2 * n - 1).bit_length()
        A = rfft(a, fft_size)
        power = A.real ** 2 + A.imag ** 2
        power = np.maximum(power, 1e-15)
        
        # Target: slightly modulated flat spectrum
        # a = 0.5 + 0.3*cos + 0.2*sin (in frequency domain)
        freq = np.linspace(0, 1, len(power))
        target = 0.5 + 0.25 * np.cos(2 * np.pi * 2 * freq) + 0.1 * np.cos(2 * np.pi * 5 * freq)
        target = np.maximum(target, 0.01)
        
        # Blend current and target
        power = 0.5 * power / power.max() + 0.5 * target / target.max()
        
        # Inverse: take sqrt of power, find phase, IFFT
        # Keep current phase
        phase = A / np.maximum(np.abs(A), 1e-15)
        A_new = np.sqrt(power) * phase
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new
    
    return a


def interp_to_length(a, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss


def multi_restart_lp(a0, time_limit, solver, n_restarts=3, verbose=True):
    """LP-Nesterov with periodic large restarts."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    best_a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = best_a.sum()
    if s0 < 1e-15:
        best_a = np.ones(n) * norm / n
    else:
        best_a = best_a * norm / s0
    best_score = fast_autocorr(best_a).max()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    
    chunk_time = time_limit / n_restarts
    
    for r in range(n_restarts):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        actual_chunk = min(chunk_time, remaining)
        
        if r > 0:
            # Perturb the current best
            perturb_scale = 0.3 * (r + 1) / n_restarts
            a_start = best_a * (1.0 - perturb_scale) + rng.random(n) * perturb_scale * norm
            a_start = np.maximum(a_start, 0)
            s_s = a_start.sum()
            if s_s < 1e-15:
                a_start = best_a.copy()
            else:
                a_start = a_start * norm / s_s
        
        a_out, sc = nesterov_lp_optimize(a_start, actual_chunk, solver, verbose=verbose)
        if sc < best_score:
            best_score = sc
            best_a = a_out.copy()
        
        if verbose:
            print(f"    Restart {r+1}/{n_restarts}: score={sc:.8f}, best={best_score:.8f}")
    
    return best_a, best_score

    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = build_conv_matrix(a)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        return None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    stall = 0
    stall_best = best_score

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break

        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
            stall_best = best_score

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
            stall_best = best_score

        avg_a += a
        n_avg += 1
        if n_avg % 40 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    stall = 0
                    stall_best = best_score

        if score_y >= stall_best - 1e-12:
            stall += 1
        if stall > 80 and stall % 160 == 0:
            rng = np.random.default_rng(n_iters + 42)
            perturb = rng.random(n) * 0.08 * norm / n
            y_pert = y + perturb
            y_pert = np.maximum(y_pert, 0.0)
            s_p = y_pert.sum()
            if s_p > 1e-15:
                y = y_pert * norm / s_p
            stall = 0
            stall_best = best_score

        if verbose and n_iters % 150 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def dual_avg_nesterov(a0, time_limit, verbose=False):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    dual_accum = np.zeros(n)
    sum_t = 0.0

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(y)
        k_star = int(np.argmax(b))

        grad = np.zeros(n)
        for i in range(n):
            j1 = k_star - i
            if 0 <= j1 < n:
                grad[i] += y[j1]
            j2 = i - k_star
            if 0 <= j2 < n:
                grad[i] += y[j2]
        grad *= 2.0

        sum_t += 1.0
        dual_accum += grad / sum_t

        max_dg = dual_accum.max()
        a_new = np.exp(-dual_accum - max_dg)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new * norm / s_new

        alpha = 2.0 / (t + 2.0)
        y = (1.0 - alpha) * y + alpha * a_new
        a = a_new

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if verbose and t % 3000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    DA-Nest t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def coord_descent_polish(a0, time_limit, seed=99):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a * norm / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    no_improve = 0

    total_sweeps = 0
    while time.monotonic() - t0 < time_limit:
        total_sweeps += 1
        improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < norm * 1e-8:
                continue

            # Find positions with lowest contribution to peaks
            b = fast_autocorr(a)
            top_k = np.argsort(b)[::-1][:5]

            # Identify "safe" destination positions (low in all peak neighborhoods)
            peak_mask = np.zeros(n, dtype=bool)
            for k in top_k:
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                peak_mask[js:je + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != i)[0]
            if len(candidates) == 0:
                continue

            # Pick a few candidates with low values
            k_try = min(5, len(candidates))
            cand_sorted = candidates[np.argsort(a[candidates])][:k_try]
            j = cand_sorted[rng.integers(len(cand_sorted))]

            for frac in [0.003, 0.008, 0.015, 0.03]:
                amt = a[i] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < 0:
                    continue
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    no_improve = 0
                    break
            if improved:
                break

        if not improved:
            no_improve += 1
            if no_improve > 5:
                break
            # Random exploration
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-6:
                amt = a[i] * rng.uniform(0.02, 0.08)
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial * norm / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            no_improve = 0

    return best_a, best_score


def spectral_init(n, rng):
    a = rng.random(n)
    a = np.maximum(a, 0.01)
    s = a.sum()
    a = a / s

    fft_size = 1 << (2 * n - 1).bit_length()
    for it in range(100):
        A = rfft(a, fft_size)
        power = A.real ** 2 + A.imag ** 2
        power = np.maximum(power, 1e-20)

        freq = np.linspace(0, 1, len(power))
        target = 0.6 + 0.2 * np.cos(2 * np.pi * 3 * freq) + 0.15 * np.cos(2 * np.pi * 7 * freq) + 0.05 * np.sin(2 * np.pi * 11 * freq)
        target = np.maximum(target, 0.01)

        blend = 0.3 + 0.7 * (it / 100.0)
        power_new = (1.0 - blend) * power / power.max() + blend * target / target.max()

        phase = A / np.maximum(np.abs(A), 1e-15)
        A_new = np.sqrt(power_new) * phase
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a = a_new / s_new

    return a


def interp_to_length(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0.0)
    ss = a_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return a_interp / ss


def multi_restart_lp(a0, time_limit, solver, n_restarts=3, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    best_a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s0 = best_a.sum()
    if s0 < 1e-15:
        best_a = np.ones(n) * norm / n
    else:
        best_a = best_a * norm / s0
    best_score = fast_autocorr(best_a).max()

    t0 = time.monotonic()
    rng = np.random.default_rng(555)
    chunk_time = time_limit / n_restarts

    for r in range(n_restarts):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        actual_chunk = min(chunk_time, remaining)

        if r == 0:
            a_start = best_a.copy()
        else:
            scale = 0.2 + 0.3 * r / n_restarts
            a_start = best_a * (1.0 - scale) + rng.random(n) * scale * norm
            a_start = np.maximum(a_start, 0)
            s_s = a_start.sum()
            if s_s < 1e-15:
                a_start = best_a.copy()
            else:
                a_start = a_start * norm / s_s

        a_out, sc = nesterov_lp_optimize(a_start, actual_chunk, solver, verbose=False)
        if sc < best_score:
            best_score = sc
            best_a = a_out.copy()

        if verbose:
            print(f"    Restart {r+1}/{n_restarts}: score={sc:.8f}, best={best_score:.8f}")

    return best_a, best_score


def try_update(state, a, label):
    if a is None:
        return
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 0.01:
        return
    a = a / s
    sc = evaluate_safe(a)
    if np.isfinite(sc) and sc < state['best_score']:
        state['best_score'] = sc
        state['best_seq'] = a.copy()
        elapsed = time.monotonic() - state['t_start']
        print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Uses LP-Nesterov with multi-length exploration,
    spectral initialization, coordinate descent polish, and random restarts.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1650.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")

    state = {
        'best_seq': None,
        'best_score': float('inf'),
        't_start': t_global,
        'parent': None
    }

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            state['parent'] = p / p.sum()
            p_score = evaluate_safe(state['parent'])
            state['best_seq'] = state['parent'].copy()
            state['best_score'] = p_score
            print(f"Parent: n={len(state['parent'])}, score={p_score:.6f}")
    except Exception:
        pass

    # Initialize if no parent
    if state['parent'] is None:
        n_base = 1600
        state['best_seq'] = spectral_init(n_base, rng)
        state['best_score'] = evaluate_safe(state['best_seq'])
        print(f"Spectral init: n={n_base}, score={state['best_score']:.6f}")

    n_base = len(state['best_seq'])

    # Phase 1: LP Nesterov at base length
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.25, 380.0)
    if t1 > 30:
        a_norm = state['best_seq'] * np.sqrt(2.0 * n_base) / state['best_seq'].sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True)
        try_update(state, a_out, f"lp_n{n_base}")

    # Phase 2: Multi-restart at best length
    remaining = time_budget - (time.monotonic() - t_global)
    n_cur = len(state['best_seq'])
    print(f"\n=== Phase 2: Multi-restart LP n={n_cur} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 30:
        a_norm = state['best_seq'] * np.sqrt(2.0 * n_cur) / state['best_seq'].sum()
        a_out, _ = multi_restart_lp(a_norm, t2, solver, n_restarts=3, verbose=True)
        try_update(state, a_out, f"restart_n{n_cur}")

    # Phase 3: Upscale to n+400
    n2 = n_cur + 400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: LP Nesterov n={n2} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 30:
        init_long = interp_to_length(state['best_seq'], n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t3, solver, verbose=True)
        try_update(state, a_out, f"lp_n{n2}")

    # Phase 4: Upscale to n+800
    n3 = n_cur + 800
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: LP Nesterov n={n3} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.18, 280.0)
    if t4 > 30:
        init_long3 = interp_to_length(state['best_seq'], n3)
        a_norm = init_long3 * np.sqrt(2.0 * n3) / init_long3.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t4, solver, verbose=True)
        try_update(state, a_out, f"lp_n{n3}")

    # Phase 5: Coordinate descent polish
    remaining = time_budget - (time.monotonic() - t_global)
    n_polish = len(state['best_seq'])
    print(f"\n=== Phase 5: Coord descent n={n_polish} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.12, 120.0)
    if t5 > 10:
        a_norm = state['best_seq'] * np.sqrt(2.0 * n_polish) / state['best_seq'].sum()
        a_out, _ = coord_descent_polish(a_norm, t5, seed=777)
        try_update(state, a_out, f"coord_n{n_polish}")

    # Phase 6: Dual-averaging Nesterov
    remaining = time_budget - (time.monotonic() - t_global)
    n_da = len(state['best_seq'])
    print(f"\n=== Phase 6: Dual-Avg Nesterov n={n_da} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.15, 150.0)
    if t6 > 10:
        a_norm = state['best_seq'] * np.sqrt(2.0 * n_da) / state['best_seq'].sum()
        a_out, _ = dual_avg_nesterov(a_norm, t6, verbose=False)
        try_update(state, a_out, f"da_n{n_da}")

    # Phase 7: Final LP polish on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_final = len(state['best_seq'])
    print(f"\n=== Phase 7: Final LP n={n_final} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.25, 200.0)
    if t7 > 10:
        a_norm = state['best_seq'] * np.sqrt(2.0 * n_final) / state['best_seq'].sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t7, solver, verbose=False)
        try_update(state, a_out, f"final_lp_n{n_final}")

    # Phase 8: Final coord descent
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        t8 = min(remaining * 0.5, 60.0)
        if t8 > 5:
            n_final2 = len(state['best_seq'])
            a_norm = state['best_seq'] * np.sqrt(2.0 * n_final2) / state['best_seq'].sum()
            a_out, _ = coord_descent_polish(a_norm, t8, seed=1234)
            try_update(state, a_out, "final_coord")

    # Final validation
    if state['best_seq'] is not None:
        fs = evaluate_safe(state['best_seq'])
        if np.isfinite(fs):
            state['best_score'] = min(state['best_score'], fs)
        print(f"\nFINAL: score={state['best_score']:.6f}, n={len(state['best_seq'])}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in state['best_seq']]
