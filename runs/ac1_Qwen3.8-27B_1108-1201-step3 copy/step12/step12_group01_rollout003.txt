"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Full LSE gradient descent with Adam (FFT-based autocorrelation + correlate-based gradient)
2. Multi-stage temperature annealing with adaptive restarts
3. LP Nesterov zero-sum game as secondary optimizer
4. Peak shaving local search for final polish
5. Multi-length exploration
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def compute_lse_obj_grad(a, T):
    """
    Compute full LSE objective and gradient.
    obj = (1/T) * log(sum_k exp(T * c[k])) where c = conv(a,a)
    grad_a[j] = 2 * sum_m w[j+m] * a[m] where w = softmax(T*c)
    """
    n = len(a)
    c = fast_autocorr(a)
    
    Tc = T * c
    Tc_max = Tc.max()
    exp_Tc = np.exp(Tc - Tc_max)
    w = exp_Tc / exp_Tc.sum()
    
    lse_val = (Tc_max + np.log(exp_Tc.sum())) / T
    
    # grad_a[j] = 2 * sum_{m=0}^{n-1} w[j+m] * a[m]
    grad_a = 2.0 * np.correlate(w, a, mode='valid')
    
    return lse_val, grad_a


def full_lse_adam(a0, time_limit, T_start=300.0, T_end=0.05, n_stages=50, lr_base=0.01, seed=42):
    """
    Full LSE gradient descent with Adam optimizer and multi-stage annealing.
    Works on normalized sequences (sum=1).
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    stage_time = time_limit / n_stages
    
    stall_count = 0
    patience = 800  # steps without improvement before restart
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit - 2:
            break
        
        stage_start = time.monotonic()
        lr = lr_base * (0.95 ** si)
        lr = max(lr, 5e-5)
        
        m = np.zeros(n)
        v = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.85, 0.999, 1e-8
        
        while time.monotonic() - stage_start < stage_time:
            if time.monotonic() - t0 > time_limit - 2:
                break
            
            lse_val, grad_a = compute_lse_obj_grad(a, stage_T)
            
            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad_a
            v = beta2 * v + (1 - beta2) * grad_a ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)
            
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new / s_new
            
            a = a_new
            
            # Check actual objective periodically
            if t_adam % 20 == 0:
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    stall_count = 0
                else:
                    stall_count += 1
            
            # Adaptive restart within stage
            if stall_count > patience:
                a = best_a.copy()
                m = np.zeros(n)
                v = np.zeros(n)
                t_adam = 0
                stall_count = 0
    
    # Final check
    score = 2.0 * n * fast_autocorr(a).max()
    if score < best_score:
        best_score = score
        best_a = a.copy()
    
    score2 = 2.0 * n * fast_autocorr(best_a).max()
    
    return best_a, score2


def multiplicative_weights_adam(a0, time_limit, T_start=200.0, T_end=0.05, n_stages=40, lr_base=0.01, seed=42):
    """
    Multiplicative weights (mirror descent) with LSE objective.
    a_new[i] = a[i] * exp(-lr * grad[i] / a[i]) ... actually just use log-space.
    Equivalent to Adam on log(a) but with the natural gradient.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    z = np.log(a)  # log-space parameterization

def peak_shaving(a0, time_limit, seed=123):
    """
    Peak shaving: iteratively reduce the highest peaks by moving mass.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit - 2:
        b = fast_autocorr(a)
        bmax = b.max()
        threshold = bmax * 0.997
        top_peaks = np.where(b >= threshold)[0]
        
        improved = False
        for k in top_peaks[:30]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            
            contrib = a[idx] * a[k - idx]
            max_c = contrib.max()
            if max_c < 1e-15:
                continue
            
            src = idx[np.argmax(contrib)]
            
            # Find safe destinations
            danger = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:30]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                danger[js2:je2 + 1] = True
            
            safe = np.where(~danger)[0]
            safe = safe[safe != src]
            if len(safe) == 0:
                safe = np.where(np.arange(n) != src)[0]
            
            safe_sorted = safe[np.argsort(a[safe])]
            dst_pool = safe_sorted[:max(5, len(safe_sorted) // 4)]
            if len(dst_pool) == 0:
                continue
            dst = dst_pool[rng.integers(len(dst_pool))]
            
            for frac in [0.001, 0.003, 0.007, 0.015, 0.03, 0.05]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random exploration
            a_trial = a.copy()
            for _ in range(rng.integers(1, 4)):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a_trial[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.02)
                    amt = a_trial[i] * frac
                    a_trial[i] -= amt
                    a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 1e-10:
                a_trial = a_trial / st
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
            
            if stall > 200:
                break
    
    return best_a, best_score


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 10)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 3
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 2
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.7 + 0.2 * np.cos(2 * x) + 0.1 * np.cos(5 * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-5 * (x - 0.3) ** 2) + 0.5 * np.exp(-5 * (x - 0.7) ** 2)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) * np.cos(np.pi * x / 3) ** 2
    elif choice == 7:
        x = np.linspace(-1, 1, n)
        a = 0.5 * (1 + np.cos(np.pi * x))
    elif choice == 8:
        # Flat with small perturbation
        a = np.ones(n)
        a += rng.normal(0, 0.1, n)
        a = np.maximum(a, 0)
    else:
        # Two lobes
        x = np.linspace(0, 1, n)
        a = np.exp(-8 * (x - 0.25) ** 2) + np.exp(-8 * (x - 0.75) ** 2)
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
