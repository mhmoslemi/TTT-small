def coord_descent_optimize(a0, time_limit, max_passes=200, n_evals=25):

    Coordinate descent with golden section search per coordinate.

    else:
        a = a / s

    best_score = compute_score(a)
    t0 = time.monotonic()
    phi = (1.0 + np.sqrt(5.0)) / 2.0  # golden ratio

    def eval_score_with_change(a_arr, i, t):
        """Compute score with a[i] replaced by t (no renormalization)."""
        n_arr = len(a_arr)
        a_i_old = a_arr[i]
        s_old = a_arr.sum()
        s_new = s_old - a_i_old + t
        if s_new < 1e-15:
            return float('inf')
        # Compute new autocorrelation at affected lags
        c = fast_autocorr(a_arr)
        # Update affected lags
        k_start = i
        k_end = min(i + n_arr - 1, 2 * n_arr - 2)
        max_val = 0.0

            if k_start <= k <= k_end:
                other = k - i
                if 0 <= other < n_arr:
                    if k == 2 * i:
                        c[k] = c[k] - a_i_old * a_i_old + t * t
                    else:
                        c[k] = c[k] - 2.0 * a_i_old * a_arr[other] + 2.0 * t * a_arr[other]
            if c[k] > max_val:
                max_val = c[k]
        return 2.0 * n_arr * max_val / (s_new * s_new)

    passes = 0
    stall = 0
    while time.monotonic() - t0 < time_limit - 2 and passes < max_passes:
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit - 2:
                break
            a_i = a[i]
            S = a.sum()
            # Golden section search for optimal t
            lo = max(0.0, a_i * 0.01)

            if hi <= lo:

            gr = 1.0 / phi
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            f1 = eval_score_with_change(a, i, c1)

                    c2 = c1
                    f2 = f1
                    c1 = hi - gr * (hi - lo)

                else:

                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi - lo)
                    f2 = eval_score_with_change(a, i, c2)

            t_opt = (lo + hi) / 2.0
            f_opt = eval_score_with_change(a, i, t_opt)
            # Also check the original value
            f_orig = compute_score(a)  # this is the current score

        passes += 1
        score = compute_score(a)

            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        base_lr = 3e-3 * min(1.0, 100.0 / n)
        lr = base_lr * (0.75 ** si)

        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.8, 0.999, 1e-8

        max_it = max(20, int(min(remaining, 100.0) * 5))

        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 5:
                break

            a = softmax_fn(z)
            c = fast_autocorr(a)
            k_use = min(topk, len(c))
            top_idx = np.argsort(c)[::-1][:k_use]
            top_vals = c[top_idx]

            Tc = stage_T * top_vals
            max_Tc = Tc.max()
            exp_Tc = np.exp(Tc - max_Tc)
            w = exp_Tc / exp_Tc.sum()

            # Gradient
            grad_a = np.zeros(n)
            for idx in range(k_use):
                K = top_idx[idx]
                wK = w[idx]
                js = max(0, K - (n - 1))
                je = min(n - 1, K)
                if js <= je:
                    idx_range = np.arange(js, je + 1)
                    grad_a[idx_range] += 2.0 * wK * a[K - idx_range]

            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)

            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            if (it + 1) % 10 == 0:
                a_check = softmax_fn(z)
                score = compute_score(a_check)
                if score < best_score:
                    best_score = score
                    best_a = a_check.copy()

        a_stage = softmax_fn(z)
        score_stage = compute_score(a_stage)
        if score_stage < best_score:
            best_score = score_stage
            best_a = a_stage.copy()

        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE stage {si}/{n_stages} (T={stage_T:.1f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def peak_shaving(a0, time_limit, seed=123):
    """
    Peak deflation: move mass from positions contributing to peaks to safer positions.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit - 3:
        b = fast_autocorr(a)
        bmax = b.max()
        threshold = bmax * 0.998
        top_peaks = np.where(b >= threshold)[0]
        if len(top_peaks) == 0:
            top_peaks = np.argsort(b)[::-1][:10]
        if len(top_peaks) > 30:
            top_peaks = top_peaks[:30]

        improved = False
        for k in top_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            max_c = contrib.max()
            if max_c < 1e-15:
                continue
            src = idx[np.argmax(contrib)]

            # Find safe destinations
            danger = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:15]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                danger[js2:je2 + 1] = True
            safe = np.where(~danger)[0]
            safe = safe[safe != src]
            if len(safe) == 0:
                safe = np.where(np.arange(n) != src)[0]
            safe_sorted = safe[np.argsort(a[safe])]
            dst_pool = safe_sorted[:max(3, len(safe_sorted) // 10)]
            if len(dst_pool) == 0:
                continue

            for dst in dst_pool[:3]:
                for frac in [0.002, 0.005, 0.01, 0.02, 0.04, 0.07]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    score = compute_score(a_trial)
                    if score < best_score - 1e-12:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 100:
                break
            # Random perturbation
            if stall % 10 == 0:
                n_pert = rng.integers(1, 5)
                a_trial = a.copy()
                for _ in range(n_pert):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a_trial[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.03)
                        amt = a_trial[i] * frac
                        a_trial[i] -= amt
                        a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-10:
                    a_trial = a_trial / st
                    score = compute_score(a_trial)
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def perturbed_init(a0, rng, scale=0.1):
    """Create a perturbed version for multi-start."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a0)) / len(a0)
    a = a / s
    noise = rng.normal(0, scale, len(a))
    a_new = a + noise * (a + 1e-6)
    a_new = np.maximum(a_new, 0)
    sn = a_new.sum()
    if sn < 1e-15:
        return a.copy()
    return a_new / sn


def mixed_coord_descent(a0, time_limit, n_evals=20):
    """
    Faster coordinate descent using precomputed autocorrelation and O(n) updates.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    c = fast_autocorr(a)
    c_max = c.max()
    S = s

    pass_count = 0
    stall = 0
    while time.monotonic() - t0 < time_limit - 2 and pass_count < 500:
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit - 2:
                break
            a_i = a[i]
            if a_i < 1e-12:
                # Try setting it to a small value
                t_candidates = [a_i, S * 0.001 / n, S * 0.01 / n]
            else:
                t_candidates = [a_i * f for f in [0.3, 0.5, 0.7, 1.0, 1.4, 2.0, 3.0, 5.0]]
                t_candidates.append(0.0)

            best_t = a_i
            best_f = 2.0 * n * c_max / (S * S)

            for t in t_candidates:
                if t < 0:
                    continue
                # Compute new score efficiently
                s_new = S - a_i + t
                if s_new < 1e-10:
                    continue
                # Affected lags
                k_start = i
                k_end = min(i + n - 1, 2 * n - 2)
                new_max = 0.0
                # Check unaffected lags
                if k_start > 0:
                    m1 = c[:k_start].max()
                    if m1 > new_max:
                        new_max = m1
                if k_end + 1 < 2 * n - 1:
                    m2 = c[k_end + 1:].max()
                    if m2 > new_max:
                        new_max = m2
                # Check affected lags
                for k in range(k_start, k_end + 1):
                    other = k - i
                    if 0 <= other < n:
                        if k == 2 * i:
                            new_val = c[k] - a_i * a_i + t * t
                        else:
                            new_val = c[k] - 2.0 * a_i * a[other] + 2.0 * t * a[other]
                    else:
                        new_val = c[k]
                    if new_val > new_max:
                        new_max = new_val
                f_val = 2.0 * n * new_max / (s_new * s_new)
                if f_val < best_f:
                    best_f = f_val
                    best_t = t

            if best_t != a_i and best_f < best_score - 1e-12:
                # Update a[i] and c
                s_new = S - a_i + best_t
                if s_new < 1e-10:
                    continue
                a[i] = best_t
                S = s_new
                # Update c
                k_start = i
                k_end = min(i + n - 1, 2 * n - 2)
                for k in range(k_start, k_end + 1):
                    other = k - i
                    if 0 <= other < n:
                        if k == 2 * i:
                            c[k] = c[k] - a_i * a_i + best_t * best_t
                        else:
                            c[k] = c[k] - 2.0 * a_i * a[other] + 2.0 * best_t * a[other]
                c_max = c.max()
                improved = True

        pass_count += 1
        score = 2.0 * n * c_max / (S * S)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall >= 8:
            break

        if pass_count % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"      MixedCD pass={pass_count} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score

    """

        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = compute_score(parent)
