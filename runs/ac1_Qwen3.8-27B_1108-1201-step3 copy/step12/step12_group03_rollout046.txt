def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence using the provided evaluate_sequence."""

def softmax_weights(b, tau):
    """Compute p_k = exp(tau*b_k) / sum(exp(tau*b_k)) with numerical stability."""
    shifted = tau * b - tau * b.max()
    e = np.exp(shifted)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(b)) / len(b)
    return e / s


def compute_smoothed_gradient(a, tau):

    # (p ⊛ a)_i = sum_j p_{j+i} * a_j

    # (p_rev * a)_{2n-2-i} = sum_j p_rev[2n-2-i-j] * a_j = sum_j p[i+j] * a_j = c_i
    # So c = cross-correlation = irisft(rfft(p_rev) * rfft(a)) at appropriate indices.
    
    # Simpler: just use the direct formula with FFT
    # c_i = sum_{j=0}^{n-1} p_{i+j} * a_j

def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Algorithm from Duchi et al. (2011)
    if n == 0:
        return v
    mu = np.sort(v)[::-1]
    cv = np.cumsum(mu)
    rho = np.nonzero(mu * np.arange(1, n + 1) > (cv - 1))[0][-1]
    theta = (cv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def fista_smoothed(a0, time_limit, tau_start=5.0, tau_end=500.0, n_phases=5):

        a = a / s
    y = a.copy()

    best_a = a.copy()
    b_best = fast_autocorr(a)
    best_score = 2.0 * n * b_best.max()

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    total_iters = 0
    for phase in range(n_phases):
        # Tau for this phase (geometric progression)
        tau = tau_start * (tau_end / tau_start) ** (phase / max(1, n_phases - 1))
        phase_time = time_limit / n_phases
        phase_start = time.monotonic()
        
        # Initialize FISTA for this phase
        y = a.copy()
        t = 1.0

        # Use backtracking instead

        f_val = (1.0 / tau) * (tau * b.max() + np.log(np.sum(np.exp(tau * b - tau * b.max()))))
        # Actually, f_tau = (1/tau) * log(sum exp(tau * b_k))
        # = b.max() + (1/tau) * log(sum exp(tau*(b_k - b.max())))
        
        lr = 1.0 / (tau * 4.0 * a.max() * 2.0)  # Initial step size (conservative)
        min_lr = 1e-10 * n
        
        for it in range(100000):
            if time.monotonic() - phase_start > phase_time:
                break
            if time.monotonic() - t0 > time_limit - 5:
                break

            grad = compute_smoothed_gradient(y, tau)
            
            # FISTA update
            a_new = project_simplex(y - lr * grad)

            b_new = fast_autocorr(a_new)

                continue

            t_new = 0.5 * (1 + np.sqrt(1 + 4 * t * t))

            y_s = y.sum()
            if y_s < 1e-15:

            t = t_new

        b = fast_autocorr(a)

            b_best = b.copy()
            best_score = 2.0 * n * b.max()

def active_set_polish(a0, time_limit, n_active=20):
    """
    Polishing using projected subgradient of the true max.
    At each step, compute the subgradient of max_k (a*a)_k,
    project onto simplex, line search.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    b = fast_autocorr(a)
    max_b = b.max()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * max_b
    
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    for it in range(100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_autocorr(a)
        max_b = b.max()
        
        # Find active lags (those within 1% of max)
        threshold = max_b * 0.99
        active = np.where(b >= threshold)[0]
        
        # Compute subgradient as average of gradients of active lags
        # Weighted by how close to max
        weights = np.exp(50 * (b[active] - max_b))  # Sharpen around max
        weights = weights / weights.sum()
        
        grad = np.zeros(n)
        for w, k in zip(weights, active):
            # grad_k(i) = 2 * a_{k-i} if i in S_k else 0
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            other = k - idx
            grad[idx] += 2 * w * a[other]
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm
        
        # Line search
        best_lr = 0
        best_a_ls = a.copy()
        best_obj_ls = max_b
        for lr in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
            a_trial = project_simplex(a - lr * grad)
            b_trial = fast_autocorr(a_trial)
            obj_trial = b_trial.max()
            if obj_trial < best_obj_ls:
                best_obj_ls = obj_trial
                best_lr = lr
                best_a_ls = a_trial
            if obj_trial < max_b * 0.999:
                break
        
        if best_lr == 0:
            # No improvement, try smaller steps or break
            if it > 100:
                break
            continue
        
        a = best_a_ls
        if best_obj_ls < best_score / (2.0 * n) - 1e-15:
            best_score = 2.0 * n * best_obj_ls
            best_a = a.copy()
    
    return best_a, best_score


def build_conv_matrix(a_fixed):
    """Build the (2n-1) x n convolution matrix T where T@b gives convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue
        n_iters += 1
        weight = t  # Weight for geometric averaging
        
        iterates_sum += weight * b_star
        weight_sum += weight
        
        # Nesterov update
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_new_s = y_new.sum()
        if y_new_s < 1e-15:
            y_new = y.copy()
            y_new_s = 1.0
        y_new = y_new / y_new_s
        
        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            a_s = a_new.sum()
            if a_s < 1e-15:
                a_new = y_new.copy()
                a_s = 1.0
            a_new = a_new / a_s
        
        y = y_new
        a = a_new
        
        # Score a
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        # Score y
        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        # Score geometric average
        if n_iters >= 10 and n_iters % 5 == 0:
            avg = iterates_sum / weight_sum
            avg = np.maximum(avg, 0)
            avg_s = avg.sum()
            if avg_s > 1e-15:
                avg = avg / avg_s
                score_avg = 2.0 * n * fast_autocorr(avg).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg.copy()
        
        if t % 500 == 0 and verbose:
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score, n_iters


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def fista_refined(a0, time_limit, tau_start=10.0, tau_end=2000.0, n_phases=8):
    """
    Improved FISTA with adaptive tau and multiple phases.
    Uses the smoothed max objective with backtracking.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    t0 = time.monotonic()
    best_a = a.copy()
    b = fast_autocorr(a)
    best_max = b.max()
    best_score = 2.0 * n * best_max
    
    for phase in range(n_phases):
        if time.monotonic() - t0 > time_limit - 5:
            break
        
        # Geometric tau schedule
        progress = phase / max(1, n_phases - 1)
        tau = tau_start * (tau_end / tau_start) ** progress
        
        phase_budget = (time_limit - (time.monotonic() - t0)) / (n_phases - phase)
        phase_start = time.monotonic()
        
        # FISTA initialization for this phase
        y = a.copy()
        t_f = 1.0
        lr = 1.0 / (tau * 2.0 * np.linalg.norm(a)**2 * 2.0 * n)
        lr = max(lr, 1e-12)
        min_lr = lr * 1e-8
        
        # Track for this phase
        phase_best = best_max
        phase_best_a = best_a.copy()
        stall = 0
        
        it = 0
        while time.monotonic() - phase_start < phase_budget:
            if time.monotonic() - t0 > time_limit - 2:
                break
            it += 1
            
            b_y = fast_autocorr(y)
            p = softmax_weights(b_y, tau)
            grad = compute_smoothed_gradient(y, tau)
            
            # FISTA step
            a_new = project_simplex(y - lr * grad)
            
            b_new = fast_autocorr(a_new)
            cur_max = b_new.max()
            
            # Backtracking: if objective increased too much
            if cur_max > b_y.max() + 1e-6:
                lr = max(lr * 0.5, min_lr)
                continue
            
            # Update
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_f * t_f))
            y = a_new + ((t_f - 1.0) / t_new) * (a_new - a)
            y = np.maximum(y, 0)
            y_s = y.sum()
            if y_s < 1e-15:
                y = a_new.copy()
                y_s = 1.0
            y = y / y_s
            a = a_new
            t_f = t_new
            
            # Track best
            if cur_max < phase_best - 1e-14:
                phase_best = cur_max
                phase_best_a = a.copy()
                stall = 0
            else:
                stall += 1
            
            if stall > 500 and it > 100:
                break
        
        # End of phase
        if phase_best < best_max - 1e-14:
            best_max = phase_best
            best_score = 2.0 * n * best_max
            best_a = phase_best_a.copy()
            a = phase_best_a.copy()
        
        elapsed = time.monotonic() - t0
        print(f"      FISTA phase {phase} (tau={tau:.1f}, {it} iters): best={best_score:.6f}, elapsed={elapsed:.0f}s")
    
    return best_a, best_score
