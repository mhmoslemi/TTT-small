import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_convolution(a, b):
    """Full convolution of a and b via FFT."""
    n, m = len(a), len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def fast_autocorr(a):
    """Full convolution a*a using FFT. Returns array of length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_score(a):
    """Compute 2n*max(conv(a,a))/(sum(a))^2 quickly."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)


def evaluate_safe(a):
    """Call the official evaluate_sequence safely."""
    try:
        a = np.asarray(a, dtype=np.float64)
        a = np.clip(a, 0.0, 1000.0)
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def subgradient_of_smoothed_conv(a, T):
    """
    Compute gradient of (1/T)*log(sum(exp(T*b_k))) where b = conv(a,a).
    Returns (value, gradient) where gradient has same shape as a.
    """
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    Z = exp_Tb.sum()
    obj = (Tb_max + np.log(Z)) / T
    w = exp_Tb / Z  # softmax weights over convolution positions

    # subgrad[j] = 2 * sum_{m=0}^{n-1} a[m] * w[j+m]
    # = 2 * (a * w_rev)[len(w)-n : len(w)][::-1]  where w_rev = w[::-1]
    w_rev = w[::-1]
    c = fast_convolution(a, w_rev)
    # c has length n + len(w) - 1 = n + (2n-1) - 1 = 3n-2
    # We want c[len(w)-n : len(w)][::-1]
    # len(w) = 2n-1, so indices from (2n-1)-n=n to 2n-1 (exclusive)
    subgrad = 2.0 * c[n : 2 * n - 1][::-1]

    return obj, subgrad


def project_simplex_nonneg(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    # First clip to non-negative
    v = np.maximum(v, 0)
    s = v.sum()
    if s < 1e-15:
        return np.ones(n) * target_sum / n
    # Scale to target sum
    return v * (target_sum / s)


def smooth_gradient_descent(a0, time_limit, T_init=200.0, T_min=0.1,
                            n_stages=15, mu=0.9, seed=42, verbose=False):
    """
    Nesterov-accelerated gradient descent on the smoothed objective.
    Uses simplex projection after each step.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()

    temps = np.geomspace(T_init, T_min, n_stages)
    momentum = mu
    v = np.zeros(n)  # momentum vector

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        # More iterations for later stages (lower T = sharper)
        iters = int(20 + 60 * (si / max(1, n_stages - 1)))
        iters = min(iters, max(5, int(remaining / 0.05)))

        y = a.copy()  # Nesterov lookahead point
        for it in range(iters):
            if time.monotonic() - t0 > time_limit:
                break

            # Compute gradient at lookahead point y
            _, grad_y = subgradient_of_smoothed_conv(y, stage_T)

            # Nesterov update
            v_new = momentum * v - grad_y
            y_new = a + ((1 + momentum) * v - momentum * v_new) * 0 + (v_new - v)
            # Simplified: use standard momentum
            v = v_new
            a_new = a + v

            # Project to simplex
            a_new = project_simplex_nonneg(a_new, 1.0)

            # Check score periodically
            if it % 10 == 0:
                sc = fast_score(a_new)
                if sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()

            a = a_new
            y = a  # For simplicity, use a as lookahead (standard momentum)

        # Final check for this stage
        sc = fast_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if verbose and si % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SGD stage {si+1}/{n_stages} (T={stage_T:.1f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def mirror_descent_momentum(a0, time_limit, eta0=0.08, decay_rate=0.5,
                            k_min=1, k_max=20, k_ramp=500,
                            momentum=0.7, seed=42, verbose=False):
    """
    Mirror descent (multiplicative weights) with momentum and adaptive
    top-k peak subgradient.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    v = np.zeros(n)  # "velocity" in log-space for momentum
    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall_count = 0
    last_best = best_score

    for t in range(1, 20000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        b_max = b.max()
        b_mean = b.sum() / len(b)

        # Adaptive k
        frac = min(1.0, t / float(k_ramp))
        k = max(k_min, min(k_max, int(k_min + frac * (k_max - k_min))))

        # Top-k peaks with exponential weighting
        top_idx = np.argsort(b)[::-1][:k]
        excess = b[top_idx] - b[0]  # Use non-center peak as reference
        # Actually use the mean as reference for better weighting
        excess = b[top_idx] - b_mean
        if excess.max() - excess.min() < 1e-15:
            weights = np.ones(k) / k
        else:
            e_shift = excess - excess.max()
            weights = np.exp(e_shift / max(excess.max() - excess.min(), 1e-15))
            weights /= weights.sum()

        # Compute subgradient
        grad = np.zeros(n)
        # Use FFT-based computation for efficiency
        w = np.zeros(2 * n - 1)
        for ii in range(k):
            w[int(top_idx[ii])] = weights[ii]
        w_rev = w[::-1]
        c = fast_convolution(a, w_rev)
        grad = 2.0 * c[n : 2 * n - 1][::-1]

        # Learning rate with decay
        eta = eta0 / (1.0 + t / 200.0) ** decay_rate

        # Momentum in log-space
        v = momentum * v - eta * grad
        log_update = v / max(a.sum(), 1e-10)  # Normalize by sum for scale invariance

        # Multiplicative update with momentum
        a_new = a * np.exp(np.clip(log_update, -5, 5))
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a + 0.01 * rng.standard_normal(n)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
        a_new = a_new / s_new

        a = a_new

        # Check score
        check_interval = 5 if t < 1000 else (10 if t < 10000 else 25)
        if t % check_interval == 0:
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stall_count = 0
            else:
                stall_count += 1

        # Stall handling
        if stall_count > 800:
            noise_scale = min(0.03, 0.005 * (stall_count / 200.0))
            a = a + noise_scale * rng.standard_normal(n)
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
            stall_count = 0

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD+Mom t={t} ({elapsed:.0f}s): best={best_score:.6f}, k={k}")

    return best_a, best_score


def peak_shaving_v2(a0, time_limit, seed=123, verbose=False):
    """
    Improved peak shaving: for each top peak, try moving mass from
    high-contributing source to low-contributing destination positions.
    Uses adaptive step sizes and considers multiple peaks simultaneously.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall = 0
    peak_count = min(15, n)

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_peaks = np.argsort(b)[::-1][:peak_count]

        improved = False
        # Try each peak
        for k_idx in top_peaks[:8]:
            k_idx = int(k_idx)
            js = max(0, k_idx - n + 1)
            je = min(n - 1, k_idx)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_idx - idx]

            if contrib.sum() < 1e-15:
                continue

            # Find best source (highest contribution)
            src = int(idx[np.argmax(contrib)])

            # Find good destinations: low a-value, away from peaks
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:5]:
                k2 = int(k2)
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) < 10:
                candidates = np.arange(n)
            # Pick lowest-a candidates not equal to src
            candidates = candidates[candidates != src]
            if len(candidates) == 0:
                continue
            sorted_cand = candidates[np.argsort(a[candidates])[:max(10, len(candidates) // 4)]]

            for dst in sorted_cand[:5]:
                for frac in [0.001, 0.003, 0.008, 0.02, 0.05, 0.1]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-10:
                        continue
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    sc = fast_score(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial.copy()
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation
            if rng.integers(2) == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.005, 0.05)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    sc = fast_score(a_trial)
                    if sc < best_score:
                        a = a_trial.copy()
                        best_score = sc
                        best_a = a.copy()
                        stall = 0

            if stall > 600:
                break

    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    PeakShave v2 ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def spectral_reshape(a0, time_limit, seed=77, verbose=False):
    """
    Spectral domain heuristic: try to flatten the convolution by
    adjusting the phase of the DFT to reduce time-domain peaks.
    Works by iteratively:
    1. Compute DFT of a
    2. Compute DFT of conv(a,a) = A^2
    3. Identify peaks in time domain
    4. Adjust A to reduce peak energy
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    # FFT size
    fft_size = 1 << (2 * n - 2).bit_length()
    A = rfft(a, fft_size)

    for it in range(500):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute current convolution in frequency domain
        conv_spec = A * A
        conv_time = irfft(conv_spec, fft_size)[:2 * n - 1]
        conv_max = conv_time.max()

        # Identify which frequency bins contribute most to the peak
        # The peak is at some time position; try to reduce it by
        # redistributing energy across frequency bins
        peak_pos = int(np.argmax(conv_time))

        # Compute the "pressure" on each frequency bin
        # d(conv_max)/d(A[k]) is complex; use gradient of |A[k]|^2 contribution
        # Simplified: reduce bins that contribute to the peak
        phase = np.angle(conv_spec)
        mag = np.abs(conv_spec)

        # Try to flatten: reduce high-magnitude bins slightly
        target = mag.mean()
        # Gently push toward flat spectrum
        A_new_mag = np.abs(A)
        A_new_mag = 0.98 * A_new_mag + 0.02 * (np.sum(np.abs(A)) / len(A))
        A_new_phase = np.angle(A) + 0.01 * rng.standard_normal(len(A))
        A_new = A_new_mag * np.exp(1j * A_new_phase)

        # Reconstruct and project
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            continue
        a_new = a_new / s_new

        sc = fast_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            A = rfft(a_new, fft_size)

    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    Spectral ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def lse_lbfgs(a0, time_limit, T_start=100.0, T_end=0.3, n_stages=14, seed=42):
    """
    LogSumExp-smoothed optimization with L-BFGS in log-space.
    Parameterization: a = softmax(z), optimize z.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s

    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = fast_score(a_init)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        iters = max(10, min(60, int(15 + 45 * si / max(1, n_stages))))

        def func(z_in):
            val, _ = lse_obj_grad_z(z_in, stage_T)
            return val

        def jac(z_in):
            _, g = lse_obj_grad_z(z_in, stage_T)
            return g

        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': iters, 'gtol': 1e-12,
                                    'maxfun': iters * 300, 'disp': False})
            z = res.x
            z_shift = z - z.max()
            a = np.exp(z_shift)
            a = a / a.sum()
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        except Exception:
            pass

    return best_a, best_score


def lse_obj_grad_z(z, T):
    """
    LogSumExp objective and gradient in z-space where a = softmax(z).
    """
    n = len(z)
    z_shift = z - z.max()
    e = np.exp(z_shift)
    a = e / e.sum()

    b = fast_autocorr(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    obj = (Tb_max + np.log(exp_Tb.sum())) / T
    w = exp_Tb / exp_Tb.sum()

    # grad_a = 2 * cross_corr(w, a)
    w_rev = w[::-1]
    c = fast_convolution(a, w_rev)
    grad_a = 2.0 * c[n : 2 * n - 1][::-1]

    # grad_z = a * (grad_a - dot(grad_a, a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)

    return obj, grad_z


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def diverse_init(n, rng, variant=None):
    """Generate diverse structured initializations."""
    if variant is None:
        variant = int(rng.integers(0, 12))

    if variant == 0:
        # Linear ramp (decreasing)
        x = np.linspace(0, 1, n)
        a = 1.0 - x
    elif variant == 1:
        # Exponential decay
        x = np.linspace(0, 1, n)
        a = np.exp(-2.0 * x)
    elif variant == 2:
        # Power law decay
        x = np.linspace(0.01, 1, n)
        a = x ** (-0.5)
    elif variant == 3:
        # U-shaped (edges heavy)
        x = np.linspace(0, 1, n)
        a = (x - 0.5) ** 2 + 0.1
    elif variant == 4:
        # Sine half-wave (peaked center)
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif variant == 5:
        # Asymmetric sinusoidal
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.4 * np.sin(2 * np.pi * x - np.pi / 2)
    elif variant == 6:
        # Steep ramp
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 3
    elif variant == 7:
        # Two-segment: high then low
        a = np.ones(n)
        a[:n // 2] = 2.0
    elif variant == 8:
        # Cosine-weighted ramp
        x = np.linspace(0, 1, n)
        a = (1 - x) * (0.5 + 0.5 * np.cos(np.pi * x))
    elif variant == 9:
        # Periodic with drift
        x = np.linspace(0, 1, n)
        a = (1 - x) * (1 + 0.3 * np.cos(10 * np.pi * x))
    elif variant == 10:
        # Gaussian at edge
        x = np.linspace(0, 1, n)
        a = np.exp(-20 * x ** 2)
    else:
        # Random smooth
        a = np.ones(n)
        for _ in range(3):
            center = rng.uniform(0, n)
            width = rng.uniform(n * 0.1, n * 0.4)
            amp = rng.uniform(-0.3, 0.5)
            x = np.arange(n, dtype=np.float64)
            a += amp * np.exp(-(x - center) ** 2 / (2 * width ** 2))
        a = np.maximum(a, 0)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def coordinate_ascent(a0, time_limit, seed=555, verbose=False):
    """
    Block coordinate descent: partition the sequence into blocks,
    and optimize each block while keeping others fixed.
    For a given block, the contribution to the convolution is quadratic
    in the block variables, so we can solve a small QP.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()

    # Block sizes to try
    block_sizes = [max(2, n // 100), max(4, n // 30), max(8, n // 10), max(16, n // 5)]
    block_sizes = list(set(min(bs, n // 2) for bs in block_sizes))

    stall = 0
    while time.monotonic() - t0 < time_limit:
        improved = False
        for bs in block_sizes:
            if time.monotonic() - t0 > time_limit:
                break
            # Random block
            start = int(rng.integers(0, max(1, n - bs + 1)))
            end = min(start + bs, n)

            # Optimize this block
            a_block = a[start:end].copy()
            a_rest = np.delete(a, slice(start, end))

            best_block = a_block.copy()
            best_block_score = float('inf')

            # Try a few perturbations of the block
            for trial in range(20):
                if trial == 0:
                    trial_block = a_block.copy()
                elif trial == 1:
                    # Shift mass within block
                    idx = int(rng.integers(bs))
                    jdx = int(rng.integers(bs))
                    if idx != jdx:
                        amt = a_block[idx] * rng.uniform(0.1, 0.5)
                        trial_block = a_block.copy()
                        trial_block[idx] -= amt
                        trial_block[jdx] += amt
                        trial_block = np.maximum(trial_block, 0)
                elif trial == 2:
                    # Reshape block
                    trial_block = a_block.copy()
                    trial_block = trial_block * (1 + 0.1 * rng.standard_normal(bs))
                    trial_block = np.maximum(trial_block, 0)
                elif trial == 3:
                    # Neighborhood: swap with adjacent outside
                    if start > 0:
                        trial_block = a_block.copy()
                        amt = a[start - 1] * rng.uniform(0.05, 0.3)
                        a_trial = a.copy()
                        a_trial[start - 1] -= amt
                        a_trial[start] += amt
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        sc = fast_score(a_trial)
                        if sc < best_score:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                        continue
                    elif end < n:
                        trial_block = a_block.copy()
                        amt = a[end] * rng.uniform(0.05, 0.3)
                        a_trial = a.copy()
                        a_trial[end] -= amt
                        a_trial[end - 1] += amt
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        sc = fast_score(a_trial)
                        if sc < best_score:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                        continue
                    else:
                        continue
                else:
                    trial_block = a_block + 0.01 * rng.standard_normal(bs)
                    trial_block = np.maximum(trial_block, 0)

                # Evaluate
                a_trial = a.copy()
                a_trial[start:end] = trial_block
                a_trial = a_trial / a_trial.sum()
                sc = fast_score(a_trial)
                if sc < best_block_score:
                    best_block_score = sc
                    best_block = trial_block.copy()

            # Apply best block
            if best_block_score < best_score - 1e-14:
                a[start:end] = best_block
                a = a / a.sum()
                best_score = best_block_score
                best_a = a.copy()
                improved = True

        if not improved:
            stall += 1
            if stall > 100:
                break

    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    CoordAscent ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score

    - Mirror descent with momentum

    rng = np.random.default_rng(seed)

    def try_update(a, label):

        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = fast_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
