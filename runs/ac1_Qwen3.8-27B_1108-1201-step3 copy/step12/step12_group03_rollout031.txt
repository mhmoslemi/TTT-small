import numpy as np

from scipy.fft import rfft, irfft


def fast_conv(a):
    """Full convolution a*a using FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_conv(a, b):
    """Cross-correlation: result[j] = sum_k a[k]*b[k-j] for valid indices.
    This gives the gradient structure for d(conv(b,b))[k]/db[j] = 2*b[k-j]."""
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    # We want result[j] = sum_{k valid} a[k] * b[k-j]
    # This is cross-correlation, which is conv(a, flip(b))
    B_flip = rfft(np.asarray(b[::-1], dtype=np.float64), fft_size)
    result = irfft(A * B_flip, fft_size)
    # The cross-correlation is in the latter part of the result
    # conv(a, flip(b)) has length 2n-1
    # corr[j] = result[n-1+j] for j in range(2n-1)... 
    # Actually let me just use a direct approach: conv(a, b) gives c[k] = sum_i a[i]*b[k-i]
    # We want g[j] = sum_k a[k]*b[k-j] = sum_i a[j+i]*b[i] (letting i=k-j)
    # = (a_shifted * b)[j] where a_shifted[i] = a[j+i]
    # This is actually the cross-correlation = conv(flip(a), b)[n-1+j]... 
    # Let me just compute it directly via FFT:
    # g = ifft(fft(a) * conj(fft(b))) gives circular cross-correlation
    # For linear, we need zero-padding
    A2 = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B2 = rfft(np.asarray(b, dtype=np.float64), fft_size)
    # Linear cross-correlation: g[j] = sum_i a[i+j]*b[i]
    # = conv(a, flip(b)) shifted
    # Simpler: just compute conv(a, b) and extract what we need
    result = irfft(A2 * B2, fft_size)[:2*n-1]
    return result

    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) < 1:
            return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            t0 = time.monotonic()
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a_fixed):
    """Build the (2n-1) x n matrix T where (T@b)[k] = conv(a_fixed, b)[k]."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def mirror_descent_lse(a0, time_limit, evaluate_sequence, beta_start=None, beta_end=None, n_iters=None, verbose=True):
    """
    Fast mirror descent with log-sum-exp smoothing.
    Minimizes max_k (a*a)[k] subject to sum(a)=1, a>=0.
    Uses FFT for O(n log n) gradient computation per iteration.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s

    if beta_start is None:
        beta_start = n * 0.5
    if beta_end is None:
        beta_end = n * 20.0
    if n_iters is None:
        n_iters = 100000

    t0 = time.monotonic()
    best_a = a.copy()
    best_score = float(2.0 * n * fast_conv(a).max())

    # Averaged iterate for convergence guarantee
    avg_a = np.zeros(n)
    n_avg = 0

    # Momentum
    momentum = 0.9
    v = np.zeros(n)

    # Adaptive learning rate
    lr = 1.0 / n

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute convolution
        c = fast_conv(a)
        max_c = c.max()

        # Anneal beta
        progress = min(1.0, it / (n_iters * 0.8))
        beta = beta_start + (beta_end - beta_start) * progress * progress

        # Compute log-sum-exp gradient
        # softmax: w[k] = exp(beta*c[k]) / sum exp(beta*c[k])
        c_shifted = c - c.max()
        w = np.exp(beta * c_shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.ones_like(c) / len(c)
        else:
            w = w / w_sum

        # Gradient: dG/da[j] = 2 * sum_k w[k] * a[k-j] (for valid k-j)
        # This is 2 * cross_correlation(w, a)[j]
        # cross_correlation(w, a)[j] = sum_i w[i+j] * a[i] = conv(w, flip(a))[n-1+j]... 
        # Actually: sum_k w[k] * a[k-j] where k ranges over valid indices
        # Let's compute: result[j] = sum_{k: 0<=k<2n-1, 0<=k-j<n} w[k] * a[k-j]
        # = sum_{i: 0<=i<n, 0<=i+j<2n-1} w[i+j] * a[i]
        # This is the cross-correlation of w and a, evaluated at lag j
        # = (w * flip(a))[n-1+j]... 
        # Simplest: use FFT
        fft_size = 1 << (2 * n - 1).bit_length()
        W = rfft(w, fft_size)
        A = rfft(a, fft_size)
        # We want g[j] = sum_i w[i+j] * a[i] for j = 0..n-1
        # = ifft(conj(FFT(w)) * FFT(a)) ... no
        # Cross-correlation: (w ⋆ a)[j] = sum_i w[i+j] * a[i]
        # In FFT: ifft(FFT(w) * conj(FFT(a))) gives circular cross-correlation
        # For linear, with zero-padding:
        cross = irfft(W * np.conj(A), fft_size)
        # The linear cross-correlation is in cross[n-1 : 2n-1] (or similar offset)
        # Actually for ifft(FFT(x)*conj(FFT(y))), result[k] = sum_i x[i+k] * y[i] (circular)
        # With zero-padding to fft_size >= 2n-1, the linear result is in positions 0 to 2n-2
        # cross[j] = sum_i w[i+j] * a[i] for j = 0, 1, ..., n-1 (first n values)
        grad = 2.0 * cross[:n]

        # Mirror descent step with momentum
        v = momentum * v + (1 - momentum) * grad
        # Adaptive lr: scale by 1/sqrt(t)
        lr_t = lr / np.sqrt(1.0 + it * 0.01)

        # Apply step: a_new[j] = a[j] * exp(-lr_t * v[j]) / Z
        log_a = np.log(np.maximum(a, 1e-30))
        log_a_new = log_a - lr_t * v
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new_s = a_new.sum()
        if a_new_s < 1e-30:
            a_new = a.copy()
        else:
            a_new = a_new / a_new_s
        a = a_new

        # Track best
        score = float(2.0 * n * fast_conv(a).max())
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # Averaged iterate
        avg_a += a
        n_avg += 1
        if n_avg % 100 == 0:
            avg_check = avg_a / n_avg
            avg_score = float(2.0 * n * fast_conv(avg_check).max())
            if avg_score < best_score:
                best_score = avg_score
                best_a = avg_check.copy()

        if verbose and it % 5000 == 0:
            elapsed = time.monotonic() - t0
            cur_score = float(2.0 * n * fast_conv(a).max())
            print(f"      MD-LS t={it} ({elapsed:.0f}s): best={best_score:.6f}, cur={cur_score:.6f}, beta={beta:.1f}")

    return best_a, best_score


def nesterov_lp_optimize(a0, time_limit, solver, evaluate_sequence, verbose=True):
    """
    Nesterov-accelerated LP best-response with averaged iterate tracking.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = float(2.0 * n * fast_conv(a).max())
    t0 = time.monotonic()

    # Averaged iterate
    avg_a = np.zeros(n)
    n_avg = 0

    iter_count = 0
    stall_count = 0

    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break

        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue
        iter_count += 1

        # Nesterov schedule
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_new_s = y_new.sum()
        if y_new_s < 1e-15:
            y_new = y.copy()
            y_new_s = 1.0
        y_new = y_new / y_new_s

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            a_s = a_new.sum()
            if a_s < 1e-15:
                a_new = y_new.copy()
                a_s = 1.0
            a_new = a_new / a_s

        y = y_new
        a = a_new

        # Score
        score_a = float(2.0 * n * fast_conv(a).max())
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        else:
            if score_a > best_score * 0.99999:
                stall_count += 1

        score_y = float(2.0 * n * fast_conv(y).max())
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        # Averaged iterate (important for convergence!)
        avg_a += a_new
        n_avg += 1
        if n_avg % 10 == 0:
            avg_check = avg_a / n_avg
            score_avg = float(2.0 * n * fast_conv(avg_check).max())
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
                stall_count = 0

        if t % 2000 == 0 and verbose:
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.6f}, iters={iter_count}")

        if stall_count > 5000 and t > 1000:
            if verbose:
                print(f"      Early stop at t={t}, stall={stall_count}")
            break

    return best_a, best_score, iter_count


def peak_shaving(a0, time_limit, evaluate_sequence, seed=123):
    """
    Multi-scale peak shaving with random restarts.
    Tries to reduce the peak of the convolution by moving mass.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = float(2.0 * n * fast_conv(a).max())
    t0 = time.monotonic()
    stall = 0
    max_stall = 2000

    while time.monotonic() - t0 < time_limit:
        b = fast_conv(a)
        max_b = b.max()
        peak_pos = np.argmax(b)

        # Find positions contributing most to the peak
        k = peak_pos
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        other = k - idx
        contrib = a[idx] * a[other]
        total = contrib.sum()

        if total < 1e-15:
            stall += 1
            break

        # Try moving mass from top contributors to less contributing positions
        top_contrib = np.argsort(contrib)[::-1][:min(10, len(contrib))]

        # Find "safe" target positions (far from peak)
        peak_contrib_positions = set()
        # Consider top few peaks
        n_peaks = min(20, 2 * n - 1)
        top_peaks = np.argsort(b)[::-1][:n_peaks]
        for k2 in top_peaks:
            js2 = max(0, k2 - n + 1)
            je2 = min(n - 1, k2)
            peak_contrib_positions.update(range(js2, je2 + 1))

        improved = False
        for ci in top_contrib:
            src = idx[ci]
            src_other = other[ci]

            # Score all target positions
            target_scores = []
            for j in range(n):
                if j in peak_contrib_positions:
                    continue
                # Estimate benefit: reducing a[src] and a[src_other], adding to a[j]
                benefit = len(peak_contrib_positions) - sum(1 for k2 in top_peaks[max(0, js2-n):] if 0 <= j-k2 < n or 0 <= k2-j < n)
                # Simpler: just try positions far from the peak
                dist = abs(j - (peak_pos - (n-1) // 2))
                target_scores.append((dist, j))
            target_scores.sort(reverse=True)

            for _, dst in target_scores[:20]:
                for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.04]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    if src != src_other:
                        half_amt = min(a[src_other], amt * 0.5)
                        a_trial[src_other] -= half_amt
                        a_trial[dst] += amt + half_amt
                    else:
                        a_trial[dst] += amt
                    if (a_trial < -1e-12).any():
                        continue
                    a_trial = np.maximum(a_trial, 0)
                    a_trial_s = a_trial.sum()
                    if a_trial_s < 1e-15:
                        continue
                    a_trial = a_trial / a_trial_s
                    score = float(2.0 * n * fast_conv(a_trial).max())
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation
            if stall % 5 == 1:
                n_perturb = rng.integers(1, min(10, n // 10))
                sources = rng.choice(n, size=n_perturb, replace=False)
                targets = rng.choice(n, size=n_perturb, replace=False)
                a_trial = a.copy()
                for src_idx, dst_idx in zip(sources, targets):
                    frac = rng.uniform(0.005, 0.05)
                    amt = a_trial[src_idx] * frac
                    a_trial[src_idx] -= amt
                    a_trial[dst_idx] += amt
                a_trial = np.maximum(a_trial, 0)
                a_trial_s = a_trial.sum()
                if a_trial_s > 1e-15:
                    a_trial = a_trial / a_trial_s
                    score = float(2.0 * n * fast_conv(a_trial).max())
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            if stall > max_stall:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function combining multiple optimization approaches.
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent, evaluate_sequence)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    n_base = len(best_seq) if best_seq is not None else 1600

    if parent is None:
        n_base = 1600
        # Initialize with a sine-squared shape (slightly asymmetric)
        x = np.linspace(0, np.pi, n_base)
        best_seq = np.cos(x * 0.4) ** 2 + 0.1
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq, evaluate_sequence)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_safe(a, evaluate_sequence)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # ============================================================
    # Phase 1: Fast mirror descent to get a good starting point
    # ============================================================
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: Mirror Descent LSE at n={n_base} ===")
    t1 = min(remaining * 0.10, 120.0)
    if t1 > 30:
        a_out, sc_out = mirror_descent_lse(best_seq, t1, evaluate_sequence,
                                            beta_start=n_base * 0.3,
                                            beta_end=n_base * 10.0,
                                            n_iters=50000)
        print(f"  MD-LS done: score={sc_out:.6f}")
        try_update(a_out, f"md_lse_n{n_base}")

    # ============================================================
    # Phase 2: LP Nesterov (main workhorse)
    # ============================================================
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: LP Nesterov at n={n_base} (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.45, 700.0)
    if t2 > 60:
        a_out, sc_out, iters = nesterov_lp_optimize(best_seq, t2, solver, evaluate_sequence)
        print(f"  LP Nesterov done: {iters} iters, score={sc_out:.6f}")
        try_update(a_out, f"lp_n{n_base}")

    # ============================================================
    # Phase 3: Mirror descent refinement on the LP result
    # ============================================================
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: MD refinement at n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.10, 100.0)
    if t3 > 20:
        a_out, sc_out = mirror_descent_lse(best_seq, t3, evaluate_sequence,
                                            beta_start=len(best_seq) * 5.0,
                                            beta_end=len(best_seq) * 30.0,
                                            n_iters=30000)
        print(f"  MD refine done: score={sc_out:.6f}")
        try_update(a_out, f"md_refine_n{len(best_seq)}")

    # ============================================================
    # Phase 4: Peak shaving
    # ============================================================
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: Peak Shaving at n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 15:
        a_out, sc_out = peak_shaving(best_seq, t4, evaluate_sequence, seed=456)
        print(f"  Peak shave 1: score={sc_out:.6f}")
        try_update(a_out, "peak_shave_1")

    # ============================================================
    # Phase 5: Second LP Nesterov round (explore from improved point)
    # ============================================================
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 5: LP Nesterov round 2 (remaining={remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 30:
        a_out, sc_out, iters = nesterov_lp_optimize(best_seq, t5, solver, evaluate_sequence, verbose=False)
        print(f"  LP round 2: {iters} iters, score={sc_out:.6f}")
        try_update(a_out, "lp_round2")

    # ============================================================
    # Phase 6: Try nearby lengths
    # ============================================================
    for n_try in [n_base + 100, n_base - 100, n_base + 200]:
        if n_try < 200 or n_try > 5000:
            continue
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining < 150:
            break
        print(f"\n=== Phase 6: Explore n={n_try} (remaining={remaining:.0f}s) ===")
        init_try = interpolate_sequence(best_seq, n_try)
        
        # Quick MD
        t6a = min(remaining * 0.15, 80.0)
        if t6a > 20:
            a_out, sc_out = mirror_descent_lse(init_try, t6a, evaluate_sequence,
                                                beta_start=n_try * 0.3,
                                                beta_end=n_try * 10.0,
                                                n_iters=20000, verbose=False)
            try_update(a_out, f"md_n{n_try}")

        # Quick LP
        remaining = time_budget - (tmp_global := time.monotonic()) - t_global_start
        t6b = min(remaining * 0.4, 120.0)
        if t6b > 30:
            a_out, sc_out, iters = nesterov_lp_optimize(best_seq, t6b, solver, evaluate_sequence, verbose=False)
            print(f"  LP at n={n_try}: {iters} iters, score={sc_out:.6f}")
            try_update(a_out, f"lp_n{n_try}")

    # ============================================================
    # Phase 7: Final peak shaving
    # ============================================================
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 7: Final Peak Shaving (remaining={remaining:.0f}s) ===")
    t7 = min(remaining * 0.6, 100.0)
    if t7 > 10:
        a_out, sc_out = peak_shaving(best_seq, t7, evaluate_sequence, seed=789)
        print(f"  Final peak shave: score={sc_out:.6f}")
        try_update(a_out, "peak_shave_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq, evaluate_sequence)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
