def fast_autocorr(a):
    """Full self-convolution a*a using FFT, O(n log n)."""
    a = np.asarray(a, dtype=np.float64)

    A = rfft(a, fft_size)

def build_conv_matrix(a_fixed):

    T = np.zeros((size, n))

    Solve LP to find best linearized direction to reduce max convolution.

    b = fast_autocorr(a)
    a_rev = a[::-1]

    t_var = cp.Variable()

        cp.sum(delta_var) == 0,
        delta_var >= -a,
        b + 2.0 * step_scale * (T_rev @ delta_var) <= t_var

        prob.solve(solver=solver)

        pass

                    break

                continue

        return np.zeros(n), None

    delta_val = np.asarray(delta_var.value, dtype=np.float64)
    t_val = float(t_var.value)
    return delta_val, t_val

def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def compute_lse_grad(a, T):
    """
    Compute gradient of (1/T)*log(sum exp(T*(a*a)_k)) w.r.t. a.
    Uses FFT for O(n log n).
    """
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()

    # grad[i] = 2 * sum_k w[k] * a[k-i] (valid k)
    # = 2 * (w * a_rev)[i + n - 1] where a_rev = a[::-1]
    a_rev = a[::-1].copy()
    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    if conv_len > fft_size:
        fft_size *= 2

    W = rfft(w, fft_size)
    A_rev = rfft(a_rev, fft_size)
    conv = irfft(W * A_rev, fft_size)

    grad = 2.0 * conv[n - 1:n - 1 + n]
    return grad, w


def lse_obj_grad_z(z, T):
    """Objective and gradient for z-parameterization (a = softmax(z))."""
    a = softmax_fn(z)
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    obj_val = (max_Tb + np.log(exp_Tb.sum())) / T

    grad_a, w = compute_lse_grad(a, T)

    # Transform to z-space
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj_val, grad_z


def logsumexp_optimize(a0, time_limit, T_start=100.0, T_end=0.02, n_stages=20):
    """
    LogSumExp-smoothed optimization with L-BFGS and multi-stage annealing.
    Uses FFT-based gradients for O(n log n) per iteration.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        iters_per_stage = max(10, min(60, int(20 + 40 * (si / max(1, n_stages - 1)))))

        def func(z_in):
            val, _ = lse_obj_grad_z(z_in, stage_T)
            return val

        def jac(z_in):
            _, g = lse_obj_grad_z(z_in, stage_T)
            return g

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-15, 'maxfun': iters_per_stage * 300,
                         'disp': False}
            )
            z = res.x
            a = softmax_fn(z)
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
        except Exception:
            pass

        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE stage={si}/T={stage_T:.4f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
