def fast_conv(a):
    """Full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def lse_objective(x, T, n):
    """Compute LSE approximation of 2*n*max(conv(a,a))/sum(a)^2."""
    S = x.sum()
    if S < 1e-15:
        return 1e10
    b = fast_conv(x)
    c = b / (S * S)
    Tc = T * c
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    log_sum = max_Tc + np.log(exp_Tc.sum())
    return (2.0 * n / T) * log_sum


def lse_gradient(x, T, n):
    """Compute gradient of LSE objective w.r.t. x."""
    S = x.sum()
    if S < 1e-15:
        return np.zeros(n)
    b = fast_conv(x)
    c = b / (S * S)
    Tc = T * c
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    w = exp_Tc / exp_Tc.sum()

    # G_i = sum_m a[m]*w[m+i] + sum_m a[m]*w[i-m]
    # Term 1: (a_rev * w) at positions n-1 .. 2n-2, extract [n-1 : n-1+n]
    # Term 2: (a * w) at positions 0 .. n-1
    # Linear conv length: n + (2n-1) - 1 = 3n - 2
    fft_size = 1 << (3 * n - 2).bit_length()

    A = rfft(x, fft_size)
    W = rfft(w, fft_size)
    A_rev = rfft(x[::-1], fft_size)

    conv1 = irfft(A_rev * W, fft_size)
    conv2 = irfft(A * W, fft_size)

    G = conv1[n - 1:n - 1 + n] + conv2[0:n]

    W_scalar = np.dot(w, b)
    grad = (2.0 * n / (S ** 3)) * (S * G - 2.0 * W_scalar)
    return grad


def lse_obj_grad(x, T, n):
    """Combined objective and gradient for L-BFGS-B."""
    obj = lse_objective(x, T, n)
    g = lse_gradient(x, T, n)
    return obj, g


def second_moment_obj_grad(x):
    """Compute sum(conv(a,a)^2)/sum(a)^4 and its gradient."""
    n = len(x)
    S = x.sum()
    if S < 1e-15:
        return 1e10, np.zeros(n)
    b = fast_conv(x)
    b2_sum = np.sum(b * b)
    obj = b2_sum / (S ** 4)

    # grad_i = 4*(S*G_i - b2_sum*x_i) / S^5
    # where G_i = sum_k b_k * db_k/dx_i
    # db_k/dx_i = x[k-i] + x[i-k]
    # G_i computed similarly to LSE gradient but with weights = b
    w = b / b2_sum  # normalized weights (doesn't matter for G computation)
    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(x, fft_size)
    B = rfft(b, fft_size)
    A_rev = rfft(x[::-1], fft_size)
    conv1 = irfft(A_rev * B, fft_size)
    conv2 = irfft(A * B, fft_size)
    G = conv1[n - 1:n - 1 + n] + conv2[0:n]

    grad = 4.0 * (S * G - b2_sum * x) / (S ** 5)
    return obj, grad

Key insight: the optimal sequence is NOT symmetric; the max of conv(a,a)
occurs at a non-central lag, and the sequence has a U-shaped profile.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def lse_objective(x, T, n):
    """Compute LSE approximation of 2*n*max(conv(a,a))/sum(a)^2."""
    S = x.sum()
    if S < 1e-15:
        return 1e10
    b = fast_conv(x)
    c = b / (S * S)
    Tc = T * c
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    log_sum = max_Tc + np.log(exp_Tc.sum())
    return (2.0 * n / T) * log_sum


def lse_gradient(x, T, n):
    """Compute gradient of LSE objective w.r.t. x."""
    S = x.sum()
    if S < 1e-15:
        return np.zeros(n)
    b = fast_conv(x)
    c = b / (S * S)
    Tc = T * c
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    w = exp_Tc / exp_Tc.sum()

    fft_size = 1 << (3 * n - 2).bit_length()
    A = rfft(x, fft_size)
    W = rfft(w, fft_size)
    A_rev = rfft(x[::-1], fft_size)

    conv1 = irfft(A_rev * W, fft_size)
    conv2 = irfft(A * W, fft_size)
    G = conv1[n - 1:n - 1 + n] + conv2[0:n]

    W_scalar = np.dot(w, b)
    grad = (2.0 * n / (S ** 3)) * (S * G - 2.0 * W_scalar)
    return grad


def lse_obj_grad_for_lbfgs(x, T, n):
    """Combined objective and gradient for L-BFGS-B."""
    return lse_objective(x, T, n), lse_gradient(x, T, n)


def evaluate_safe(arr):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def normalize(a):
    """Normalize a to sum to 1, clip negatives."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def lse_lbfgs_optimize(a0, time_limit, T_start=200.0, T_end=1.0, n_stages=8):
    """Multi-stage T-annealed LSE optimization using L-BFGS-B."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_safe(a)
    t0 = time.monotonic()
    bounds = [(0.0, 1000.0)] * n

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        max_iter = min(500, max(50, int(remaining * 150 / max(1, n_stages - si))))

        def fun(x):
            return lse_objective(x, T, n)

        def grad(x):
            return lse_gradient(x, T, n)

        try:
            result = minimize(fun, a, jac=grad, method='L-BFGS-B',
                              bounds=bounds, options={'maxiter': max_iter,
                                                      'ftol': 1e-14,
                                                      'gtol': 1e-10})
            a_new = np.maximum(result.x, 0)
            s = a_new.sum()
            if s > 1e-15:
                a = a_new / s
        except Exception:
            pass

        score = evaluate_safe(a)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()

        elapsed = time.monotonic() - t0
        print(f"        LSE T={T:.1f}: score={best_score:.6f} (elapsed={elapsed:.0f}s)")

    return best_a, best_score


def second_moment_optimize(a0, time_limit):
    """Optimize sum(conv^2)/sum(a)^4 using L-BFGS-B."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = evaluate_safe(a)
    t0 = time.monotonic()
    bounds = [(0.0, 1000.0)] * n

    def fun(x):
        S = x.sum()
        if S < 1e-15:
            return 1e10
        b = fast_conv(x)
        return np.sum(b * b) / (S ** 4)

    def grad(x):
        S = x.sum()
        if S < 1e-15:
            return np.zeros(n)
        b = fast_conv(x)
        b2_sum = np.sum(b * b)
        fft_size = 1 << (3 * n - 2).bit_length()
        A = rfft(x, fft_size)
        B = rfft(b, fft_size)
        A_rev = rfft(x[::-1], fft_size)
        conv1 = irfft(A_rev * B, fft_size)
        conv2 = irfft(A * B, fft_size)
        G = conv1[n - 1:n - 1 + n] + conv2[0:n]
        return 4.0 * (S * G - b2_sum * x) / (S ** 5)

    while time.monotonic() - t0 < time_limit:
        try:
            result = minimize(fun, a, jac=grad, method='L-BFGS-B',
                              bounds=bounds, options={'maxiter': 300,
                                                      'ftol': 1e-14,
                                                      'gtol': 1e-12})
            a_new = np.maximum(result.x, 0)
            s = a_new.sum()
            if s > 1e-15:
                a = a_new / s
        except Exception:
            break

        score = evaluate_safe(a)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()

        if result.status == 1 and time.monotonic() - t0 < time_limit - 2:
            continue
        break

    return best_a, best_score


def peak_shaving(a0, time_limit, seed=123):
    """Coordinate-descent peak shaving."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = 2.0 * n * fast_conv(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_conv(a)
        top_k = np.argsort(b)[::-1][:6]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            src_local = np.argmax(contrib)
            src = idx[src_local]

            peak_areas = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_areas.update(range(js2, je2 + 1))

            candidates = [j for j in range(n) if j not in peak_areas]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            candidates_sorted = sorted(candidates, key=lambda j: a[j])
            dst_pool = candidates_sorted[:max(1, len(candidates_sorted) // 4)]
            dst = dst_pool[rng.integers(len(dst_pool))]

            for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_conv(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.005, 0.02)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_conv(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            if stall > 150:
                break

    return best_a, best_score


def u_shaped_init(n, rng, variant=0):
    """Generate diverse U-shaped initializations."""
    x = np.linspace(0, 1, n)
    if variant == 0:
        a = 0.3 + 0.7 * (x ** 3 + (1 - x) ** 3)
    elif variant == 1:
        a = 0.2 + 0.8 * (x ** 2 + (1 - x) ** 2)
    elif variant == 2:
        a = 0.1 + 0.9 * np.abs(2 * x - 1)
    elif variant == 3:
        a = 0.3 + 0.7 * np.cos(np.pi * x) ** 2
        a = 1.0 / (a + 0.3)
    elif variant == 4:
        a = np.ones(n)
        a[:n // 5] = 2.0
        a[4 * n // 5:] = 2.0
    elif variant == 5:
        a = 0.5 + 0.5 * (x ** 5 + (1 - x) ** 5)
    else:
        a = rng.random(n)
    a = np.maximum(a, 0)
    return normalize(a)


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Multi-phase:
    Phase 1: LSE+L-BFGS-B at parent length
    Phase 2: Second moment optimization
    Phase 3: Length extension with LSE
    Phase 4: More length extension
    Phase 5: Peak shaving polish
    Phase 6: Random restarts at best length
    Phase 7: Final peak shaving
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 30.0, 1770.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

    if best_seq is None:
        n0 = 1600
        best_seq = u_shaped_init(n0, rng, variant=1)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    n_base = len(best_seq)

    # Phase 1: LSE + L-BFGS-B at base length
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: LSE+L-BFGS-B at n={n_base} (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.15, 200.0)
    if t1 > 10:
        a_out, sc = lse_lbfgs_optimize(best_seq, t1, T_start=300.0, T_end=1.0, n_stages=10)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print(f"  *** NEW BEST [lse_n{n_base}]: {best_score:.6f}")
        else:
            print(f"  No improvement: {sc:.6f} vs {best_score:.6f}")

    # Phase 2: Second moment optimization
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: Second moment at n={n_base} (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.10, 120.0)
    if t2 > 10:
        a_out, sc = second_moment_optimize(best_seq, t2)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print(f"  *** NEW BEST [sm_n{n_base}]: {best_score:.6f}")
        else:
            print(f"  No improvement: {sc:.6f} vs {best_score:.6f}")

    # Phase 3: Try longer length
    n_long1 = max(int(n_base * 4 / 3), n_base + 200)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: LSE at n={n_long1} (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.20, 250.0)
    if t3 > 15:
        init3 = interpolate_sequence(best_seq, n_long1)
        a_out, sc = lse_lbfgs_optimize(init3, t3, T_start=200.0, T_end=1.0, n_stages=8)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            n_base = n_long1
            print(f"  *** NEW BEST [lse_n{n_long1}]: {best_score:.6f}")
        else:
            print(f"  No improvement: {sc:.6f} vs {best_score:.6f}")

    # Phase 4: Try even longer length
    n_long2 = max(int(n_base * 4 / 3), n_base + 400)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: LSE at n={n_long2} (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 15:
        init4 = interpolate_sequence(best_seq, n_long2)
        a_out, sc = lse_lbfgs_optimize(init4, t4, T_start=200.0, T_end=1.0, n_stages=8)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            n_base = n_long2
            print(f"  *** NEW BEST [lse_n{n_long2}]: {best_score:.6f}")
        else:
            print(f"  No improvement: {sc:.6f} vs {best_score:.6f}")

    # Phase 5: Peak shaving at current best
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 5: Peak shaving at n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 150.0)
    if t5 > 5:
        a_out, sc = peak_shaving(best_seq, t5, seed=99)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print(f"  *** NEW BEST [peak_n{len(best_seq)}]: {best_score:.6f}")
        else:
            print(f"  No improvement: {sc:.6f} vs {best_score:.6f}")

    # Phase 6: Random restarts at best length
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 6: Random restarts at n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    n_restarts = max(3, min(10, int(remaining / 60)))
    t_per_restart = remaining / (n_restarts + 1)

    for ri in range(n_restarts):
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining < 30:
            break
        # Create perturbed version of best
        noise_scale = 0.05 + 0.05 * ri
        perturbed = best_seq + noise_scale * rng.standard_normal(len(best_seq))
        perturbed = normalize(perturbed)

        t_r = min(t_per_restart * 0.6, remaining * 0.4)
        if t_r < 10:
            break

        a_out, sc = lse_lbfgs_optimize(perturbed, t_r, T_start=100.0, T_end=1.0, n_stages=6)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print(f"  *** NEW BEST [restart_{ri}_n{len(best_seq)}]: {best_score:.6f}")
        else:
            print(f"  Restart {ri}: {sc:.6f} vs {best_score:.6f}")

    # Phase 7: Final peak shaving
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 7: Final peak shaving (remaining={remaining:.0f}s) ===")
    t7 = min(remaining * 0.4, 120.0)
    if t7 > 5:
        a_out, sc = peak_shaving(best_seq, t7, seed=456)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_out
            print(f"  *** NEW BEST [final_peak_n{len(best_seq)}]: {best_score:.6f}")
        else:
            print(f"  No improvement: {sc:.6f} vs {best_score:.6f}")

    # Phase 8: One more LSE polish at very low T
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 20:
        t8 = min(remaining * 0.3, 100.0)
        if t8 > 10:
            print(f"\n=== Phase 8: Final LSE polish (remaining={remaining:.0f}s) ===")
            a_out, sc = lse_lbfgs_optimize(best_seq, t8, T_start=50.0, T_end=0.5, n_stages=6)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = a_out
                print(f"  *** NEW BEST [final_lse_n{len(best_seq)}]: {best_score:.6f}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
            best_seq = np.maximum(best_seq, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
