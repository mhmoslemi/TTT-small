Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smoothed objective with softmax reparametrization
2. L-BFGS-B optimization with adaptive temperature
3. Multiple random restarts with diverse initializations
4. Fourier-based initialization
5. Final coordinate descent polishing (peak shaving)

import numpy as np
from scipy.optimize import minimize
from scipy.fft import rfft, irfft
import time

def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def softmax_fn(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def smooth_max_val(b, beta):
    """Compute log-sum-exp / beta as smooth approximation to max."""
    b_max = b.max()
    shifted = beta * (b - b_max)
    return b_max + np.log(np.sum(np.exp(shifted))) / beta

def smooth_max_weights(b, beta):
    """Compute softmax weights for the smooth max."""
    shifted = beta * (b - b.max())
    w = np.exp(shifted)
    return w / w.sum()

def build_toeplitz_grad(weights, n):
    """Build the n x n matrix T where T[m,j] = weights[m+j]."""
    row_idx = np.arange(n)[:, None]
    col_idx = np.arange(n)[None, :]
    return weights[row_idx + col_idx]

def objective_z(z, beta, n):
    """Objective and gradient with respect to z (unconstrained)."""
    a = softmax_fn(z)
    b = fast_autocorr(a)
    
    # Smooth max objective
    b_max = b.max()
    shifted = beta * (b - b_max)
    exp_shifted = np.exp(shifted)
    log_sum_exp = b_max + np.log(exp_shifted.sum())
    obj = log_sum_exp / beta
    
    # Gradient of smooth max w.r.t. a
    w = exp_shifted / exp_shifted.sum()
    
    # grad_a[m] = 2 * sum_j w[m+j] * a[j]
    row_idx = np.arange(n)[:, None]
    col_idx = np.arange(n)[None, :]
    T = w[row_idx + col_idx]
    grad_a = 2.0 * (T @ a)
    
    # Chain rule: grad_z = a * (grad_a - a @ grad_a)
    grad_z = a * (grad_a - np.dot(a, grad_a))
    
    return obj, grad_z

def optimize_smooth(a_init, n, time_limit, beta_start=20.0, beta_end=200.0, verbose=True):
    """Optimize using L-BFGS-B with adaptive beta."""
    t0 = time.monotonic()
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    
    # Initialize z from a
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()  # Center for numerical stability
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    
    if verbose:
        print(f"    Init: score={best_score:.8f}, max_b={fast_autocorr(a).max():.10f}")
    
    n_betas = 8
    time_per_beta = time_limit / n_betas
    
    for bi in range(n_betas):
        if time.monotonic() - t0 > time_limit:
            break
        
        progress = bi / max(1, n_betas - 1)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        def obj_fun(z):
            return objective_z(z, beta, n)
        
        try:
            result = minimize(obj_fun, z, method='L-BFGS-B', 
                            options={'maxiter': 500, 'maxfun': 2000, 
                                    'ftol': 1e-15, 'gtol': 1e-12,
                                    'maxls': 50})
        except Exception as e:
            if verbose:
                print(f"    L-BFGS-B failed at beta={beta:.1f}: {e}")
            break
        
        z_opt = result.x
        a_new = softmax_fn(z_opt)

        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        # Also evaluate the intermediate iterates? No, L-BFGS-B returns the best.
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    beta={beta:.1f}: score={score:.8f}, best={best_score:.8f} ({elapsed:.0f}s)")
        
        # Reset z for next beta
        z = np.log(np.maximum(a_new, 1e-30))
        z = z - z.mean()
    
    return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smoothed objective with softmax reparametrization + L-BFGS-B
2. Adaptive temperature schedule (beta annealing)
3. Subgradient / proximal gradient descent as complementary method
4. Fourier-based and random alternative initializations for diversity
5. Final coordinate descent peak shaving for polishing
6. Multi-phase time allocation with checkpointing
"""

import numpy as np
from scipy.optimize import minimize
from scipy.fft import rfft, irfft
import time


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def score_sequence(a):
    """Compute the true objective 2*n*max(conv(a,a))/(sum(a))^2."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)


def smooth_max_val(b, beta):
    """Log-sum-exp / beta as smooth approximation to max(b)."""
    b_max = b.max()
    shifted = beta * (b - b_max)
    return b_max + np.log(np.sum(np.exp(shifted))) / beta


def build_toeplitz_indices(n):
    """Precompute index matrix for Toeplitz gradient computation."""
    row_idx = np.arange(n)[:, None]
    col_idx = np.arange(n)[None, :]
    return row_idx + col_idx


def objective_z(z, beta, n, idx_mat):
    """
    Objective and gradient with respect to z (unconstrained parameters).
    a = softmax(z), objective = smooth_max(conv(a,a), beta)
    """
    a = softmax_fn(z)
    b = fast_autocorr(a)

    # Smooth max value
    b_max = b.max()
    shifted = beta * (b - b_max)
    exp_shifted = np.exp(shifted)
    log_sum_exp = b_max + np.log(exp_shifted.sum())
    obj = log_sum_exp / beta

    # Weights for smooth max
    w = exp_shifted / exp_shifted.sum()

    # grad_a[m] = 2 * sum_j w[m+j] * a[j]
    # Using precomputed index matrix
    w_sub = w[idx_mat]
    grad_a = 2.0 * (w_sub @ a)

    # Chain rule through softmax: grad_z = a * (grad_a - <a, grad_a>)
    grad_z = a * (grad_a - np.dot(a, grad_a))

    return obj, grad_z


def optimize_smooth_lbfgsb(a_init, n, time_limit, beta_start=10.0, beta_end=200.0,
                           n_betas=8, maxiter_per_beta=300, verbose=True):
    """
    Optimize using L-BFGS-B on softmax-parameterized space with adaptive beta.
    """
    t0 = time.monotonic()
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s

    best_score = score_sequence(a)

    # Precompute index matrix
    idx_mat = build_toeplitz_indices(n)

    if verbose:
        print(f"    Init score={best_score:.8f}, max_b={fast_autocorr(a).max():.2e}")

    time_per_beta = time_limit / n_betas

    for bi in range(n_betas):
        if time.monotonic() - t0 > time_limit:

        progress = bi / max(1, n_betas - 1)
        beta = beta_start * (beta_end / beta_start) ** progress

def subgradient_descent(a_init, n, time_limit, n_iters=2000, verbose=True):
    """
    Subgradient descent on the simplex with step-size schedule.
    Minimizes max_k conv(a,a)_k subject to a>=0, sum(a)=1.
    """
    t0 = time.monotonic()
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s

    best_a = a.copy()
    best_score = score_sequence(a)

    # Running average (mirror descent style averaging)
    avg_a = a.copy()
    n_avg = 0

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Step size
        t = it + 1
        lr = 1.0 / np.sqrt(t)

        # Compute autocorrelation and find peak
        b = fast_autocorr(a)
        max_b = b.max()

        # Top-5 weighted subgradient
        top_k = np.argsort(b)[::-1][:5]
        weights = np.exp((b[top_k] - max_b) * 5.0)
        weights = weights / weights.sum()

        # Compute weighted subgradient
        grad = np.zeros(n)
        for w_k, k in zip(weights, top_k):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            other = k - idx
            grad[idx] += 2.0 * w_k * a[other]
            grad[other] += 2.0 * w_k * a[idx]

        # Gradient step with projection onto simplex
        a_new = a - lr * grad
        # Project onto simplex {a >= 0, sum(a) = 1}
        a_new = np.maximum(a_new, 0)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-15:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / a_new_sum

        a = a_new

        # Update running average
        avg_a = (n_avg * avg_a + a) / (n_avg + 1)
        n_avg += 1

        # Check scores
        sc = score_sequence(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        sc_avg = score_sequence(avg_a)
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = avg_a.copy()

        if verbose and it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG it={it}: best={best_score:.8f}, cur={sc:.8f} ({elapsed:.0f}s)")

    return best_a, best_score

                        a_trial = np.maximum(a_trial, 0)
                        a_trial_s = a_trial.sum()
                        if a_trial_s < 1e-15:
                            continue
                        a_trial = a_trial / a_trial_s
                        sc = score_sequence(a_trial)
                        if sc < best_score - 1e-15:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Perturbation to escape local minimum
            if stall % 10 == 0:
                a_trial = a.copy()
                n_move = rng.integers(1, 4)
                for _ in range(n_move):
                    src = int(rng.integers(n))
                    dst = int(rng.integers(n))
                    amt = a_trial[src] * rng.uniform(0.01, 0.04)
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                a_trial_s = a_trial.sum()
                if a_trial_s > 1e-15:
                    a_trial = a_trial / a_trial_s
                    sc = score_sequence(a_trial)
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        stall = 0

            if stall > max_stall:
                break

    return best_a, best_score


def fourier_init(n, seed=0):
    """Initialize using inverse FFT of a random smooth spectrum."""
    rng = np.random.default_rng(seed)
    fft_size = 1 << (2 * n - 1).bit_length()
    # Create random phase with smooth magnitude
    freqs = np.arange(fft_size // 2 + 1)
    # Low-pass filter the magnitude for smoothness
    mag = np.exp(-0.5 * (np.linspace(0, 3, fft_size // 2 + 1))**2)
    phase = 2 * np.pi * rng.random(fft_size // 2 + 1)
    spectrum = mag * np.exp(1j * phase)

    # Hermitian symmetry
    full_spec = np.zeros(fft_size, dtype=complex)
    full_spec[:fft_size // 2 + 1] = spectrum
    for i in range(2, fft_size // 2 + 1):
        full_spec[fft_size - i] = np.conj(spectrum[i])

    a = np.fft.ifft(full_spec).real[:n]
    a = np.abs(a)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def cosine_init(n, seed=0):
    """Initialize with cosine bump."""
    x = np.linspace(0, 1, n)
    a = np.cos(np.pi * x / 2) ** 2
    s = a.sum()
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search: multi-phase optimization combining smooth L-BFGS-B,
    subgradient descent, and peak shaving.
    """
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_sequence([float(min(1000.0, x)) for x in a])
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Get parent sequence
    n_base = 1600
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p.copy()
            n_base = len(parent)
            ps = evaluate_sequence([float(x) for x in parent])
            if np.isfinite(ps):
                best_score = ps
                best_seq = parent.copy()
                print(f"Parent: n={n_base}, score={ps:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = cosine_init(n_base)
        best_score = score_sequence(parent)
        best_seq = parent.copy()
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    # === Phase 1: Primary smooth L-BFGS-B (most time) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: Smooth L-BFGS-B at n={n_base} (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.30, 450.0)
    if t1 > 40:
        a_out, sc_out = optimize_smooth_lbfgsb(parent, n_base, t1,
                                               beta_start=10.0, beta_end=150.0,
                                               n_betas=10, maxiter_per_beta=200,
                                               verbose=True)
        try_update(a_out, f"smooth1_n{n_base}")
        # Keep the internally optimized 'a' for next phase
        if a_out is not None and sc_out < best_score:
            parent = a_out

    # === Phase 2: Secondary smooth L-BFGS-B with higher beta ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: High-beta smooth L-BFGS-B (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 30:
        a_out, sc_out = optimize_smooth_lbfgsb(best_seq, len(best_seq), t2,
                                               beta_start=50.0, beta_end=500.0,
                                               n_betas=6, maxiter_per_beta=200,
                                               verbose=True)
        try_update(a_out, f"smooth2_n{len(best_seq)}")

    # === Phase 3: Subgradient descent ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: Subgradient descent (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 180.0)
    if t3 > 20:
        a_out, sc_out = subgradient_descent(best_seq, len(best_seq), t3,
                                            n_iters=5000, verbose=True)
        try_update(a_out, f"sgd_n{len(best_seq)}")

    # === Phase 4: Alternative initializations ===
    for alt_seed in [7, 99, 256]:
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining < 100:
            break
        t4 = min(remaining * 0.10, 100.0)
        if t4 < 30:
            break

        # Diverse initialization
        a_f = fourier_init(n_base, seed=alt_seed)
        a_c = cosine_init(n_base)
        a_mix = 0.4 * a_f + 0.6 * a_c
        a_mix = np.maximum(a_mix, 0)
        a_mix = a_mix / a_mix.sum()

        print(f"\n=== Phase 4: Alt init seed={alt_seed} (remaining={remaining:.0f}s) ===")
        a_out, sc_out = optimize_smooth_lbfgsb(a_mix, n_base, t4,
                                               beta_start=10.0, beta_end=100.0,
                                               n_betas=5, maxiter_per_beta=150,
                                               verbose=False)
        try_update(a_out, f"alt{alt_seed}_n{n_base}")

    # === Phase 5: Peak shaving on current best ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 5: Peak shaving (remaining={remaining:.0f}s) ===")

    t6 = min(remaining * 0.25, 200.0)
    if t6 > 20:

                                               beta_start=20.0, beta_end=400.0,
                                               n_betas=8, maxiter_per_beta=150,
                                               verbose=True)
        try_update(a_out, "smooth_final")

    # === Phase 7: Final peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 7: Final peak shaving (remaining={remaining:.0f}s) ===")
    t7 = min(remaining, 120.0)
    if t7 > 10:
        a_out, sc_out = peak_shaving(best_seq, t7, seed=999)
        try_update(a_out, "peak_shave_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_sequence([float(max(0.0, min(1000.0, x))) for x in best_seq])
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
