import numpy as np
import cvxpy as cp

from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(arr):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in arr]
        sc = evaluate_sequence(a_list)

        return float('inf')


def normalize_pos(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def score_normalized(a):
    """Compute score for a normalized non-negative sequence (sum=1)."""
    n = len(a)
    b = fast_autocorr(a)
    return 2.0 * n * b.max()


def spectral_error_reduction(a0, n_iter=2000, alpha=0.5, t_limit=None):
    """
    Gerchberg-Saxton style: iterate between flattening the power spectrum
    and projecting back to non-negative simplex.
    Target: |A(f)|^2 proportional to (n-1)/(2n-1) for flat convolution.
    """
    a = normalize_pos(a0)
    n = len(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    for i in range(n_iter):
        if t_limit is not None and time.monotonic() - t0 > t_limit:
            break
        
        A = rfft(a, fft_size)
        power = np.abs(A) ** 2
        
        # Target: flat spectrum (all frequencies equal)
        # DC is fixed at (sum a)^2 = 1, so target for non-DC is also ~1/(2n-1) * (2n-1) = 1
        # Actually for a flat convolution of value 1/(2n-1), the spectrum should be 1/(2n-1) * fft_size
        # Let's use geometric mean of non-DC as target
        non_dc_power = power[1:power.size // 2 + 1]
        target = np.exp(np.mean(np.log(non_dc_power + 1e-30)))
        
        # Update: scale non-DC toward target
        ratio = target / (power[1:] + 1e-30)
        ratio = np.clip(ratio, 0.01, 100.0)
        scale = np.ones(len(A))
        scale[1:] = np.power(ratio, alpha * 0.5)
        
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = normalize_pos(a_new)
        
        sc = score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        a = a_new
        
        if i % 200 == 0:
            print(f"    SER iter {i}: score={best_score:.6f}, current={sc:.6f}")
    
    return best_a, best_score


def build_conv_matrix_subset(a_fixed, top_indices):
    """Build convolution matrix rows for specified indices only."""
    n = len(a_fixed)
    T = np.zeros((len(top_indices), n))
    for row_idx, k in enumerate(top_indices):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        cols = np.arange(js, je + 1)
        vals = a_fixed[k - cols]
        T[row_idx, cols] = vals
    return T


def solve_reduced_lp(y, top_k_indices, solver):
    """
    Solve: min M s.t. T(y)@b <= M for top-k indices, sum(b)=1, b>=0.
    Returns (b, M, n_constraints_used).
    """
    n = len(y)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    T = build_conv_matrix_subset(y, top_k_indices)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None, len(top_k_indices)
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None, len(top_k_indices)
    b_val = b_val / s
    return b_val, float(M_var.value), len(top_k_indices)


def lp_nesterov_reduced(a0, t_limit, solver, k_start=50, k_max=300, k_growth=1.05):
    """
    Nesterov-accelerated LP with adaptive constraint reduction.
    Starts with few constraints, grows over time.
    """
    n = len(a0)
    a = normalize_pos(a0)
    y = a.copy()
    
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    k = k_start
    threshold = 1e-9  # improvement threshold to stop growing k
    
    avg_a = np.zeros(n)
    n_avg = 0
    t = 0
    
    while time.monotonic() - t0 < t_limit:
        t += 1
        remaining = t_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        # Compute current convolution and find top-k peaks
        c = fast_autocorr(y)
        top_k = np.argsort(c)[::-1][:k]
        
        # Solve reduced LP
        b_star, M_val, _ = solve_reduced_lp(y, top_k, solver)
        if b_star is None:
            continue
        
        # Nesterov update
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = normalize_pos(y_new)
        
        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = normalize_pos(a_new)
        
        y = y_new
        a = a_new
        
        # Track best
        for candidate in [a, y]:
            sc = score_normalized(candidate)
            if sc < best_score:
                best_score = sc
                best_a = candidate.copy()
        
        # Running average
        avg_a += a
        n_avg += 1
        if n_avg >= 20 and n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            sc_avg = score_normalized(avg_check)
            if sc_avg < best_score:
                best_score = sc_avg
                best_a = avg_check.copy()
        
        # Grow k periodically
        if t % 100 == 0 and k < k_max:
            k = min(k_max, int(k * k_growth) + 1)
        
        if t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest t={t} (k={k}, {elapsed:.0f}s): best={best_score:.6f}, M={M_val:.8f}")
    
    return best_a, best_score


def multiplicative_weights(a0, t_limit, eta_start=0.5, eta_decay=0.999):
    """
    Multiplicative weights (exponential weights) method for the max-conv problem.
    Uses subgradient of max_k c[k].
    """
    n = len(a0)
    a = normalize_pos(a0)
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = eta_start
    
    z = np.zeros(n)  # log-domain: a = exp(z) / sum(exp(z))
    z = np.log(np.maximum(a, 1e-30))
    
    for t in range(100000):
        if time.monotonic() - t0 > t_limit:
            break
        
        a = softmax_fn(z)
        c = fast_autocorr(a)
        
        # Find the peak and compute subgradient
        k_max = c.argmax()
        
        # Subgradient of c[k_max] w.r.t. a
        grad = np.zeros(n)
        js = max(0, k_max - n + 1)
        je = min(n - 1, k_max)
        idx = np.arange(js, je + 1)
        grad[idx] = a[k_max - idx]
        # For the case where 2*i = k_max, the contribution is doubled
        if k_max % 2 == 0:
            mid = k_max // 2
            if 0 <= mid < n:
                grad[mid] += a[mid]
        
        # Update in log domain: z -= eta * grad
        z = z - eta * grad
        eta *= eta_decay
        
        sc = score_normalized(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        if t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MW t={t} ({elapsed:.0f}s): score={best_score:.6f}, eta={eta:.4f}")
    
    return best_a, best_score

            b = fast_autocorr(a)
            Tb = stage_T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            
            # Gradient via FFT
            A_rev = rfft(a[::-1].copy(), fft_size)
            W = rfft(w, fft_size)
            conv = irfft(A_rev * W, fft_size)
            grad_a = 2.0 * conv[n - 1:n - 1 + n]
            
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)
            
            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)
            
            lr_t = lr / np.sqrt(1 + 0.01 * it)
            z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            
            if it % 30 == 0:
                a_check = softmax_fn(z)
                sc = score_normalized(a_check)
                if sc < best_score:
                    best_score = sc
                    best_a = a_check.copy()
        
        a_check = softmax_fn(z)
        sc = score_normalized(a_check)
        if sc < best_score:
            best_score = sc
            best_a = a_check.copy()
        
        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    AdamLSE stage {si}/{n_stages} (T={stage_T:.2f}, {elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score


def peak_shaving(a0, t_limit, seed=123):
    """Coordinate-descent peak shaving."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize_pos(a0)
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < t_limit:
        c = fast_autocorr(a)
        top_k = np.argsort(c)[::-1][:6]
        
        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            
            src = idx[np.argmax(contrib)]
            
            # Find destination: low-contribution position
            peak_areas = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_areas.update(range(js2, je2 + 1))
            
            candidates = [j for j in range(n) if j not in peak_areas]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            
            if not candidates:
                continue
            
            # Try moving mass from src to a random low-value candidate
            candidates_sorted = sorted(candidates, key=lambda j: a[j])
            dst_pool = candidates_sorted[:max(1, len(candidates_sorted) // 4)]
            dst = dst_pool[rng.integers(len(dst_pool))]
            
            for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                sc = score_normalized(a_trial)
                if sc < best_score - 1e-13:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    sc = score_normalized(a_trial)
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        stall = 0
        
        if stall > 200:
            break
    
    return best_a, best_score


def get_lp_solver():
    """Find a working LP solver, preferring fast ones."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize_pos(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    return normalize_pos(f_interp)


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.4 * np.cos(2 * np.pi * 3 * x) + 0.1 * np.cos(2 * np.pi * 7 * x)
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 0.5
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * (x - 0.3) ** 2) + 0.6 * np.exp(-4 * (x - 0.7) ** 2)
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = 0.7 + 0.3 * np.cos(2 * np.pi * 5 * x ** 2)
    elif choice == 4:
        # Two-peak structure
        x = np.linspace(0, 1, n)
        a = np.exp(-20 * (x - 0.2) ** 2) + np.exp(-20 * (x - 0.8) ** 2)
    else:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(2 * np.pi * 11 * x)
    return normalize_pos(a)

    solver = get_lp_solver()
    print(f"LP solver: {solver}")
    
    # Load parent
    best_seq = None
    best_score = float('inf')
    
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            best_score = evaluate_safe(best_seq)
            print(f"Parent loaded: n={len(best_seq)}, score={best_score:.6f}")
    except Exception:
        pass

        n_base = 1600
        best_seq = np.ones(n_base) / n_base

        print(f"Default init: n={n_base}, score={best_score:.6f}")
