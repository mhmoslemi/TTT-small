def fast_autocorr(a):
    """Full self-convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

        return float(sc)
    except Exception:

def normalize(a):
    """Ensure non-negative and sum to 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def lse_gradient_descent(a0, time_limit, T_start=50.0, T_end=0.1, n_stages=30):
    """
    LogSumExp-smoothed gradient descent on the simplex.
    Uses FFT for O(n log n) gradient computation.
    """
    n = len(a0)
    a = normalize(a0)
    fft_size = 1 << (2 * n - 2).bit_length()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Number of iterations for this stage
        iters_per_stage = max(20, min(200, int(100 * T_start / T) // 10))
        
        for it in range(iters_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Compute autocorrelation
            b = fast_autocorr(a)
            
            # LogSumExp weights
            Tb = T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            
            # Gradient: grad[i] = 2 * sum_k w[k] * a[k-i] (valid)
            # = 2 * (w * a_rev)[i + n - 1]
            a_rev = a[::-1]
            W = rfft(w, fft_size)
            AR = rfft(a_rev, fft_size)
            conv = irfft(W * AR, fft_size)
            grad = 2.0 * conv[n - 1:n - 1 + n]
            
            # Adaptive step size
            # We want: a - eta*grad to be non-negative
            # eta < min_i a[i] / max_i grad[i] (roughly)
            max_grad = np.abs(grad).max()
            if max_grad < 1e-15:
                break
            eta_max = 0.5 * a.sum() / max_grad  # conservative
            eta = min(0.1, eta_max)
            
            # Gradient step with projection
            a_new = a - eta * grad
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                continue
            a_new = a_new / s
            
            score = 2.0 * n * fast_autocorr(a_new).max()
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
                a = a_new
            else:
                # Backtrack
                eta *= 0.5
                a_new = a - eta * grad
                a_new = np.maximum(a_new, 0)
                s = a_new.sum()
                if s < 1e-15:
                    break
                a_new = a_new / s
                score = 2.0 * n * fast_autocorr(a_new).max()
                if score < best_score:
                    best_score = score
                    best_a = a_new.copy()
                    a = a_new
        
        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage={si}/T={T:.3f} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where (T@b) = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_best_response(a_fixed, solver):
    """
    Solve LP: min M s.t. T(a_fixed)@b <= M, sum(b)=1, b>=0.
    """
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s

def peak_targeting_descent(a0, time_limit, seed=42):
    """
    Smart coordinate descent that targets the top peaks.
    For each peak, identifies the top contributing pairs and
    redistributes mass to positions that don't contribute to any peak.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize(a0)
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    K = 5  # top K peaks
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:K]
        
        improved = False
        for k in top_k:
            # Find contributing pairs
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            
            # Top contributors
            top_contrib = np.argsort(contrib)[::-1][:5]
            for tc in top_contrib:
                src = idx[tc]
                src_val = a[src]
                if src_val < 1e-12:
                    continue
                
                # Find target: position that minimizes contribution to all top peaks
                min_score = float('inf')
                best_dst = -1
                # Sample positions (for speed with large n)
                if n <= 3000:
                    all_pos = np.setdiff1d(np.arange(n), [src])
                else:
                    all_pos = rng.choice(n, size=min(n, 500), replace=False)
                    all_pos = all_pos[all_pos != src]
                
                for dst in all_pos:
                    score = a[dst]
                    for k2 in top_k:
                        if max(0, k2 - n + 1) <= dst <= min(n - 1, k2):
                            partner = k2 - dst
                            if 0 <= partner < n:
                                score += a[dst] * a[partner] * 10
                    if score < min_score:
                        min_score = score
                        best_dst = dst
                
                if best_dst < 0:
                    continue
                
                # Try moving a small amount
                for frac in [0.001, 0.003, 0.01, 0.03, 0.05]:
                    amt = src_val * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[best_dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.001, 0.02)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
            
            if stall > 500:
                break
        else:
            stall = 0
        
        if stall > 0 and stall % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PTD stall={stall} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score

            a = a_new

            print(f"    SSG it={it} ({elapsed:.0f}s): score={best_score:.8f}")

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Spectral flattening (iterative frequency-domain amplitude adjustment)
2. LogSumExp gradient descent with adaptive annealing
3. Nesterov-accelerated LP best-response (game-theoretic)
4. Stochastic subgradient with heavy-ball momentum
5. Peak-targeting coordinate descent for final polish
6. Multi-length exploration with interpolation transfer
Key novelty: spectral flattening directly targets the power spectrum to
flatten the autocorrelation, combined with multi-method refinement pipeline.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time
import cvxpy as cp


def fast_autocorr(a):
    """Full self-convolution a*a using FFT, O(n log n)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def normalize_seq(a):
    """Ensure non-negative and sum to 1."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def score_from_autocorr(a):
    """Compute 2*n*max(conv) for normalized a (sum=1)."""
    n = len(a)
    return 2.0 * n * fast_autocorr(a).max()


def spectral_flattening(a0, time_limit, max_iters=80000):
    """
    Iteratively flatten the power spectrum to reduce autocorrelation peaks.
    Key idea: autocorrelation = IFFT(|FFT(a)|^2), so flattening |FFT(a)|^2
    directly flattens the autocorrelation. Uses amplitude modification with
    phase preservation and time-domain projection.
    """
    n = len(a0)
    a = normalize_seq(a0)
    fft_size = 1 << (2 * n - 2).bit_length()

    best_a = a.copy()
    best_score = score_from_autocorr(a)
    t0 = time.monotonic()

    eta = 0.25
    alpha = 0.4
    no_improve_count = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        A = rfft(a, fft_size)
        power = np.abs(A) ** 2
        mean_power = power.mean()

        P_new = (1.0 - eta) * power + eta * mean_power
        new_amp = np.sqrt(np.maximum(P_new, 1e-30))

        mag = np.abs(A)
        phase = A / (mag + 1e-30)
        A_new = new_amp * phase

        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            eta = max(0.01, eta * 0.5)
            no_improve_count += 1
            if no_improve_count > 100:
                break
            continue
        a_new = a_new / s

        a_blend = (1.0 - alpha) * a + alpha * a_new
        a_blend = np.maximum(a_blend, 0)
        s = a_blend.sum()
        if s < 1e-15:
            continue
        a_blend = a_blend / s

def lse_gradient_descent(a0, time_limit, T_start=60.0, T_end=0.05, n_stages=30):
    """
    LogSumExp-smoothed gradient descent on the simplex with FFT-based gradients.
    Adaptive step size with backtracking.
    """
    n = len(a0)
    a = normalize_seq(a0)
    fft_size = 1 << (2 * n - 2).bit_length()

    best_a = a.copy()
    best_score = score_from_autocorr(a)
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break

        iters_per_stage = max(30, min(300, int(80.0 * T_start / max(T, T_end) * 0.3)))

        for it in range(iters_per_stage):
            if time.monotonic() - t0 > time_limit:
                break

            b = fast_autocorr(a)

            Tb = T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()

            a_rev = a[::-1]
            W = rfft(w, fft_size)
            AR = rfft(a_rev, fft_size)
            conv = irfft(W * AR, fft_size)
            grad = 2.0 * conv[n - 1:n - 1 + n]

            max_grad = np.abs(grad).max()
            if max_grad < 1e-15:
                break

            eta = min(0.5, 0.3 * a.sum() / max_grad)

            for backtrack in range(10):
                a_new = a - eta * grad
                a_new = np.maximum(a_new, 0)
                s = a_new.sum()
                if s < 1e-15:
                    eta *= 0.5
                    continue
                a_new = a_new / s

                score = score_from_autocorr(a_new)
                if score < best_score:
                    best_score = score
                    best_a = a_new.copy()
                    a = a_new
                    break
                else:
                    eta *= 0.5

        if si % 5 == 0 or si == n_stages - 1:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage={si}/{n_stages} T={T:.3f} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where (T@b) = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_best_response(a_fixed, solver):
    """Solve LP: min M s.t. T(a_fixed)@b <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    try:
        prob.solve(solver=solver)
    except Exception:
        pass

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s


def nesterov_lp_optimize(a0, time_limit, solver, max_iters=5000):
    """Nesterov-accelerated LP best-response (one-sided)."""
    n = len(a0)
    a = normalize_seq(a0)
    y = a.copy()

    best_a = a.copy()
    best_score = score_from_autocorr(a)
    t0 = time.monotonic()

    avg_a = np.zeros(n)
    n_avg = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        b_star = solve_best_response(y, solver)
        if b_star is None:
            continue

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
