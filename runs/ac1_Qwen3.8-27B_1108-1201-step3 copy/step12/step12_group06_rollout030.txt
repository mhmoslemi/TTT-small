```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated LP best-response with proper iterate averaging
2. Fast LSE-smoothed L-BFGS with adaptive temperature
3. Direct max subgradient with projection onto simplex
4. Multi-restart and multi-length exploration
Key improvements: faster LP via HiGHS, O(n log n) convolution,
efficient LSE gradient, proper averaging, and better time management.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog

def fast_convolve(a):
    """Fast linear convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_normalized(a):
    """Score assuming sum(a)=1: returns 2*n*max(b)."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()

def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')

def softmax_from_z(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_arr = np.where(cond)[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def build_conv_matrix(a_fixed):
    """Build the convolution matrix T where (T@b)[k] = sum_j a_fixed[j]*b[k-j]."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        cols = np.arange(j, min(j + n, size))
        T[cols, j] = a_fixed[:len(cols)]
    return T

def solve_lp_best_response(a_fixed, T_matrix=None):
    """Solve the LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0 using HiGHS."""
    n = len(a_fixed)
    if T_matrix is None:
        T_matrix = build_conv_matrix(a_fixed)
    
    # Variables: [b_0, ..., b_{n-1}, M]
    # Minimize: [0, ..., 0, 1] @ [b, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: T@b - M*1 <= 0
    # i.e., T@b + [-1]*M <= 0
    A_ub = np.hstack([T_matrix, -np.ones((T_matrix.shape[0], 1))])
    b_ub = np.zeros(T_matrix.shape[0])
    
    # Equality constraint: sum(b) = 1, M is free
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: b_i >= 0, M is free
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.status != 0:
        return None, None
    
    b_val = res.x[:n].copy()
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    M_val = res.x[n]
    return b_val, float(M_val)

def nesterov_lp(a0, time_limit, max_iters=1000000):
    """Nesterov-accelerated LP best-response with proper averaging."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    # Precompute T for the current a (will be updated each iteration)
    y = a.copy()
    T_y = build_conv_matrix(y)
    
    best_a = a.copy()
    best_score = score_normalized(a)
    
    # Averaging
    avg_a = a.copy()
    
    t0 = time.monotonic()
    last_print = t0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP with T(y)
        b_new, M_lp = solve_lp_best_response(y, T_y)
        if b_new is None:
            # Fallback: use uniform
            b_new = np.ones(n) / n
        
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k
        
        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa
        
        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy
        
        a = a_new
        y = y_new
        
        # Update T matrix for next iteration
        T_y = build_conv_matrix(y)
        
        # Update average (uniform average)
        avg_a = avg_a + (a - avg_a) / (it + 1)
        
        # Check scores
        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        sc_avg = score_normalized(avg_a)
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = avg_a.copy()
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
        
        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 10:
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} M_lp={M_lp:.6f}")
                last_print = elapsed
    
    return best_a, best_score
```

```python
def lse_objective_and_grad(z, T_temp, n):
    """Compute LSE objective and gradient w.r.t. z (log-parameter)."""
    a = softmax_from_z(z)
    c = fast_convolve(a)
    
    # LSE: (2n/T) * log(sum exp(T*c_k))
    max_c = c.max()
    exp_c = np.exp(T_temp * (c - max_c))
    w = exp_c / exp_c.sum()
    log_sum_exp = max_c + np.log(exp_c.sum())
    obj = (2.0 * n / T_temp) * log_sum_exp
    
    # Gradient w.r.t. a:
    # d(obj)/da[i] = (4n/T) * sum_j w[i+j] * a[j]
    # Then d(obj)/dz[i] = a[i] * (d(obj)/da[i] - sum_j a[j] * d(obj)/da[j])
    i_idx = np.arange(n)[:, None]
    j_idx = np.arange(n)[None, :]
    G = w[i_idx + j_idx] @ a  # G[i] = sum_j w[i+j] * a[j]
    grad_a = (4.0 * n / T_temp) * G
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return obj, grad_z

def lse_lbfgs(a0, time_limit, T_start=100.0, T_min=0.1, n_stages=30, iters_per_stage=30):
    """LSE-smoothed L-BFGS with adaptive temperature."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)
    
    T = T_start
    
    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        def func(z_val):
            obj, _ = lse_objective_and_grad(z_val, T, n)
            return obj
        
        def jac(z_val):
            _, grad = lse_objective_and_grad(z_val, T, n)
            return grad
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-15, 'maxfun': iters_per_stage * 200,
                         'disp': False}
            )
            z = res.x
            a_new = softmax_from_z(z)
            sc = score_normalized(a_new)
            
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            
            # Adaptive temperature
            gap = sc - 2.0 * n / (2.0 * n - 1)  # gap to theoretical lower bound
            rel_gap = gap / (best_score + 1e-15)
            
            if rel_gap > 0.01:
                # Stuck, keep or increase T
                T = max(T, T * 1.1)
            else:
                # Progressing, decrease T
                T = max(T_min, T * 0.7)
            
            if (stage + 1) % 5 == 0:
                elapsed = time.monotonic() - t0
                print(f"    LSE stage={stage+1}/{n_stages} T={T:.2f}: score={sc:.8f} best={best_score:.8f} ({elapsed:.0f}s)")
        
        except Exception as e:
            T = max(T_min, T * 0.8)
    
    return best_a, best_score
```
