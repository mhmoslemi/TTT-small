def fast_autocorr(a):
    """Full linear convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def obj_value(a):
    """Compute 2*n*max(conv)/(sum)^2."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    c = fast_autocorr(a)
    return 2.0 * n * c.max() / (s * s)


def evaluate_safe(a):
    """Safely evaluate using evaluate_sequence."""

        a_list = [float(max(0.0, min(1000.0, x))) for x in a]

def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if n == 1:
        return np.ones(1)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.argmax(1 - u * np.arange(1, n + 1) > cssv)
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w


def qmatvec(p, a):
    """Compute Q_p @ a where Q_p[i,j] = p[i+j].
    Uses FFT-based convolution: Q_p @ a = (a_rev * p)[n-1 : 2n-2]
    where a_rev = a reversed.
    """
    n = len(a)
    a_rev = a[::-1]
    # conv(a_rev, p) has length n + len(p) - 1 = n + (2n-1) - 1 = 3n - 2
    # We need elements [n-1] through [2n-2] (inclusive), which is n elements
    size = len(a_rev) + len(p) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a_rev, fft_size)
    P = rfft(p, fft_size)
    conv = irfft(A * P, fft_size)[:size]
    return conv[n - 1: 2 * n - 1]

    y = a.copy()
    t_prev = 1.0

    for t in range(1, n_iters + 1):
        t_now = (1.0 + np.sqrt(1.0 + 4.0 * t_prev * t_prev)) / 2.0
        gamma = 1.0 - t_prev / t_now
        y = a + gamma * (a - y)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy < 1e-15:
            y = a.copy()
        else:
            y = y / sy

        grad = 2.0 * qmatvec(p, y)
        a_new = project_simplex(y - lr * grad)
        val = float(a_new @ qmatvec(p, a_new))
        if val < best_obj:
            best_obj = val
            best_a = a_new.copy()
        a = a_new
        t_prev = t_now

    return best_a, best_obj


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def get_lp_solver():
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL, cp.MOSEK, cp.GLPK]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None, None
    b_val = b_val / s
    M_val = float(M_var.value)

    p_dual = np.zeros(2 * n - 1)
    try:
        dv = np.asarray(constraints[0].dual_value, dtype=np.float64)
        if dv is not None and len(dv) == 2 * n - 1:
            p_dual = np.maximum(dv, 0)
            ps = p_dual.sum()
            if ps > 1e-15:
                p_dual = p_dual / ps
            else:
                c = fast_autocorr(b_val)
                T_sm = 5.0 / max(c.max(), 1e-15)
                p_dual = np.exp(T_sm * c - T_sm * c.max())
                p Dual_sum = p_dual.sum()
                p_dual = p_dual / p_dual_sum
    except Exception:
        c = fast_autocorr(b_val)
        cmax = c.max()
        if cmax > 1e-15:
            T_sm = 5.0 / cmax
            p_dual = np.exp(T_sm * (c - cmax))
            p_dual_sum = p_dual.sum()
            if p_dual_sum > 1e-15:
                p_dual = p_dual / p_dual_sum
            else:
                p_dual = np.zeros(2 * n - 1)
                p_dual[np.argmax(c)] = 1.0
        else:
            p_dual = np.ones(2 * n - 1) / (2 * n - 1)

    return b_val, M_val, p_dual


def nesterov_lp_loop(a0, time_limit, solver, max_iters=500000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = obj_val(a)
    t0 = time.monotonic()

    avg_long = np.zeros(n)
    n_long = 0
    stall_count = 0
    last_p = np.ones(2 * n - 1) / (2 * n - 1)

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        b_star, M_val, p_dual = solve_lp_best_response(y, solver)
        if b_star is None:
            continue
        if p_dual is not None and p_dual.sum() > 1e-15:
            last_p = p_dual

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_sum = y_new.sum()
        if y_sum < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / y_sum

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_new

        y = y_new
        a = a_new

        avg_long += a
        n_long += 1

        score_a = obj_val(a)
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_y = obj_val(y)
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if n_long >= 30 and n_long % 50 == 0:
            avg_check = avg_long / n_long
            score_avg = obj_val(avg_check)
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
                stall_count = 0

        if stall_count > 600 and t > 200:
            a = best_a.copy()
            y = best_a.copy()
            avg_long = np.zeros(n)
            n_long = 0
            stall_count = 0
            if t % 1000 == 0:
                print(f"      NLP restart at t={t}, best={best_score:.6f}")

        if t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.8f}")

    return best_a, best_score, last_p


def paired_local_search(a0, time_limit, seed=42):
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = obj_val(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit - 3:
        c = fast_autocorr(a)
        cmax = c.max()
        top_k = np.where(c >= cmax * 0.999)[0]
        if len(top_k) == 0:
            top_k = np.argsort(c)[::-1][:5]

        improved = False
        for k in top_k[:10]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]

            # Find destination: positions that don't contribute to top peaks
            danger = np.zeros(n, dtype=bool)
            for k2 in top_k[:15]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                danger[js2:je2 + 1] = True

            safe = np.where(~danger)[0]
            safe = safe[safe != src]
            if len(safe) == 0:
                safe = np.where(np.arange(n) != src)[0]
            # Prefer low-value safe positions
            safe_sorted = safe[np.argsort(a[safe])[:max(5, len(safe) // 8)]]
            if len(safe_sorted) == 0:
                safe_sorted = safe[:10]

            for dst in safe_sorted[:8]:
                for frac in [0.002, 0.005, 0.01, 0.02, 0.04, 0.07]:
                    amt = a[src] * frac
                    if amt < 1e-13:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    sc = obj_val(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape
            n_pert = rng.integers(1, 4)
            a_trial = a.copy()
            for _ in range(n_pert):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a_trial[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.03)
                    amt = a_trial[i] * frac
                    a_trial[i] -= amt
                    a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 1e-10:
                a_trial = a_trial / st
                sc = obj_val(a_trial)
                if sc < best_score:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    stall = 0
            if stall > 200:
                break

    return best_a, best_score


def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.8 + 0.15 * np.cos(3 * x) + 0.05 * np.cos(7 * x)
    elif choice == 4:
        x = np.linspace(0, 1, n)
        a = np.exp(-8 * (x - 0.35) ** 2) + 0.6 * np.exp(-8 * (x - 0.65) ** 2)
    else:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x * 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
