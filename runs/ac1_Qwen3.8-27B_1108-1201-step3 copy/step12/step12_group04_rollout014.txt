- Each constraint (a*a)[k] + 2t*(a[k-i]-a[k-j]) + t^2*coef_k <= M is a quadratic inequality in t
- The feasible set for t is the intersection of intervals (each quadratic inequality gives at most 2 intervals, but since the quadratics open upward or are linear, it's at most 1 interval per constraint)

Actually, for a quadratic at^2 + bt + c <= M:
- If a > 0: feasible set is [t1, t2] (an interval)
- If a < 0: feasible set is (-inf, t1] ∪ [t2, +inf) (two intervals)
- If a = 0: linear, gives half-line or all/none

In our case, coef_k = [k==2i] + [k==2j] - 2[k==i+j], which can be -2, -1, 0, or 1.

For coef_k = 1: quadratic opens upward → interval
For coef_k = -2: quadratic opens downward → two intervals (complement of an interval)
For coef_k = 0: linear → half-line

The intersection of all these sets gives the feasible set for t. This is a bit complex but doable.

    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)

            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    
    return (a + b) / 2

                    if src == dst or a[src] < 1e-12:
                        continue

                    t_star, f_star = pairwise_update(a, b, src, dst)

                    if t_star > 1e-13 and f_star < b_max - 1e-12:
                        a[src] -= t_star
                        a[dst] += t_star
                        a = np.maximum(a, 0)
                        ss = a.sum()
                        if ss > 1e-15:
                            a = a / ss
                        b_new = fast_autocorr(a)
                        score = 2.0 * n * b_new.max()
                        if score < best_score - 1e-13:
                            best_score = score
                            best_a = a.copy()
                            improved = True
                        b = b_new
                        b_max = b.max()
                        break

                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape
            n_trials = 10
            for _ in range(n_trials):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.02)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
                        improved = True
                        break

            if stall > 200:
                break
        else:
            stall = 0

        if stall % 50 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            print(f"      PCD stall={stall} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = int(rng.integers(0, 12))
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.8 * np.exp(-3 * (x - 0.7) ** 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.15 * np.cos(10 * np.pi * x) + 0.05 * np.cos(20 * np.pi * x)
        a = np.maximum(a, 0)
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 1.5
    elif choice == 4:
        x = np.linspace(0, 1, n)
        a = np.exp(-5 * x ** 2)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = 1.0 - 0.3 * np.sin(6 * np.pi * x) - 0.1 * np.sin(14 * np.pi * x)
        a = np.maximum(a, 0)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * x / 2) ** 2 * (1 + 0.2 * np.sin(8 * np.pi * x))
        a = np.maximum(a, 0)
    elif choice == 7:
        x = np.linspace(0, 1, n)
        a = np.exp(-2 * x) * (1 + 0.3 * np.cos(12 * np.pi * x))
        a = np.maximum(a, 0)
    elif choice == 8:
        x = np.linspace(0, 1, n)
        a = (x * (1 - x)) ** 0.5
    elif choice == 9:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * (x - 0.5) ** 2) + 0.5 * np.exp(-8 * (x - 0.2) ** 2) + 0.3 * np.exp(-8 * (x - 0.8) ** 2)
        a = np.maximum(a, 0)
    elif choice == 10:
        # Random Fourier
        coeffs = rng.standard_normal(5) * np.array([1, 0.5, 0.3, 0.15, 0.08])
        x = np.linspace(0, 1, n)
        a = 1.0
        for k, c in enumerate(coeffs, 1):
            a += c * np.cos(2 * np.pi * k * x * rng.integers(1, 4))
        a = np.maximum(a, 0.01)
    else:
        x = np.linspace(0, 1, n)
        a = np.ones(n) * (1 - 0.1 * x)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def evaluate_safe(a):
    """Safely evaluate using provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def fine_polish(a0, time_limit, seed=123):
    """
    Very fine-grained coordinate descent with small random perturbations.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = project_simplex(np.asarray(a0, dtype=np.float64))
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    step_sizes = [0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001]
    step_idx = 0

    while time.monotonic() - t0 < time_limit:
        stall = 0
        max_stall = 50
        current_step = step_sizes[min(step_idx, len(step_sizes) - 1)]

        while stall < max_stall and time.monotonic() - t0 < time_limit:
            # Pick a random coordinate and try to adjust it
            i = int(rng.integers(n))
            if a[i] < 1e-12:
                stall += 1
                continue

            # Try moving a small fraction from i to j
            j = int(rng.integers(n))
            if i == j:
                continue

            frac = current_step * rng.uniform(0.5, 1.5)
            amt = a[i] * frac
            if amt < 1e-14:
                stall += 1
                continue

            a_trial = a.copy()
            a_trial[i] -= amt
            a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-15:
                stall += 1
                continue
            a_trial = a_trial / ss

            score = 2.0 * n * fast_autocorr(a_trial).max()
            if score < best_score - 1e-14:
                a = a_trial
                best_score = score
                best_a = a.copy()
                stall = 0
            else:
                stall += 1

        if stall >= max_stall:
            # Decrease step size
            step_idx = min(step_idx + 1, len(step_sizes) - 1)
            if step_idx >= len(step_sizes) - 1:
                # Try larger random jumps
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-8:
                        amt = a[i] * rng.uniform(0.01, 0.1)
                        a_trial = a.copy()
                        a_trial[i] -= amt
                        a_trial[j] += amt
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            step_idx = max(0, step_idx - 2)
                            break

        if (step_idx > 0) and (int(time.monotonic() - t0) % 20 == 0):
            elapsed = time.monotonic() - t0
            print(f"      Fine t={elapsed:.0f}s: best={best_score:.8f}, step={current_step}")

    return best_a, best_score


def propose_candidate():
    """
    Main search: multi-phase optimization combining Adam-LSE,
    pairwise coordinate descent, and fine polishing.
    """
    rng = np.random.default_rng(2024)
    t_start = time.monotonic()
    time_budget = 1740.0

    def elapsed():
        return time.monotonic() - t_start

    def remaining_time():
        return time_budget - elapsed()

    # Initialize from parent
    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = project_simplex(p)
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={len(best_seq)}, score={best_score:.8f}")
    except Exception:
        pass

    n_base = len(best_seq) if best_seq is not None else 1600

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed():.0f}s)")

    if best_seq is None:
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    # === Phase 1: Adam-LSE optimization ===
    rem = remaining_time()
    print(f"\n=== Phase 1: Adam-LSE (remaining={rem:.0f}s) ===")
    t1 = min(rem * 0.30, 400.0)
    if t1 > 30:
        a_out, _ = adam_lse_optimize(best_seq, t1, T_start=300.0, T_end=1.0, n_stages=35, lr_init=0.08)
        try_update(a_out, "adam_lse_1")

    # === Phase 2: Pairwise coordinate descent ===
    rem = remaining_time()
    print(f"\n=== Phase 2: Pairwise CD (remaining={rem:.0f}s) ===")
    t2 = min(rem * 0.25, 350.0)
    if t2 > 30:
        a_out, _ = pairwise_coord_descent(best_seq, t2, top_peaks=12, candidates_per_peak=5, seed=77)
        try_update(a_out, "pairwise_cd_1")

    # === Phase 3: More Adam-LSE with lower temperature ===
    rem = remaining_time()
    print(f"\n=== Phase 3: Adam-LSE refine (remaining={rem:.0f}s) ===")
    t3 = min(rem * 0.20, 300.0)
    if t3 > 20:
        a_out, _ = adam_lse_optimize(best_seq, t3, T_start=100.0, T_end=0.5, n_stages=30, lr_init=0.05)
        try_update(a_out, "adam_lse_2")

    # === Phase 4: Pairwise CD again ===
    rem = remaining_time()
    print(f"\n=== Phase 4: Pairwise CD refine (remaining={rem:.0f}s) ===")
    t4 = min(rem * 0.20, 250.0)
    if t4 > 20:
        a_out, _ = pairwise_coord_descent(best_seq, t4, top_peaks=20, candidates_per_peak=6, seed=1337)
        try_update(a_out, "pairwise_cd_2")

    # === Phase 5: Multi-length exploration ===
    rem = remaining_time()
    print(f"\n=== Phase 5: Multi-length exploration (remaining={rem:.0f}s) ===")
    if rem > 100:
        n_best = len(best_seq)
        for offset in [-200, 200, 400]:
            rem = remaining_time()
            if rem < 60:
                break
            nl = n_best + offset
            if nl < 800 or nl > 3000:
                continue
            tl = min(rem * 0.25, 150.0)
            if tl < 30:
                continue
            print(f"  Trying n={nl} (time={tl:.0f}s)")
            init_nl = interpolate_sequence(best_seq, nl)
            a_out, _ = adam_lse_optimize(init_nl, tl * 0.5, T_start=200.0, T_end=5.0, n_stages=15, lr_init=0.06)
            rem2 = remaining_time()
            if rem2 > 30:
                a_out2, _ = pairwise_coord_descent(a_out, tl * 0.3, top_peaks=10, candidates_per_peak=4, seed=555)
                try_update(a_out2, f"multi_n{nl}_pcd")

    # === Phase 6: Fine polish ===
    rem = remaining_time()
    print(f"\n=== Phase 6: Fine polish (remaining={rem:.0f}s) ===")
    t6 = min(rem * 0.4, 200.0)
    if t6 > 20:
        a_out, _ = fine_polish(best_seq, t6, seed=42)
        try_update(a_out, "fine_polish")

    # === Phase 7: Final Adam-LSE with very low T ===
    rem = remaining_time()
    print(f"\n=== Phase 7: Final LSE (remaining={rem:.0f}s) ===")
    t7 = min(rem * 0.3, 150.0)
    if t7 > 15:

    print(f"\n=== Phase 8: Final PCD (remaining={rem:.0f}s) ===")

        try_update(a_out, "final_pcd")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return result
