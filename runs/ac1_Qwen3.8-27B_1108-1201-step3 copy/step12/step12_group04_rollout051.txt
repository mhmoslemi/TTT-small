This gives a very specific and efficient local search:
1. Find `k*` = argmax of convolution
2. Compute gradient `g[i] = a[k*-i]` (the reversed/shifted sequence)
3. Move a small amount of mass from a high-g position to a low-g position
4. Check if the peak decreased

This is O(n) per step (using FFT for the convolution), so we can do many steps.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated LP best-response (primary workhorse)
2. Multi-stage LogSumExp gradient descent with FFT-based gradients
3. Gradient-direction local search exploiting peak structure
4. Simulated annealing with peak-targeted moves
5. Multi-length exploration with interpolation
6. Final coordinate descent polish
Key improvements: gradient-aware local search, adaptive LP switching,
better temperature schedules, and smarter mass redistribution.
"""

import numpy as np
import cvxpy as cp
import time
import math
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_autocorr(a):
    """Full self-convolution a*a using FFT, O(n log n)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where T@b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            x = cp.Variable(50, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """
    Solve LP: min M s.t. T(a_fixed)@b <= M, sum(b)=1, b>=0.
    Returns (b, M).
    """
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    try:
        prob.solve(solver=solver)
    except Exception:
        pass

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.SCS, cp.ECOS, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    M_val = float(M_var.value)
    return b_val, M_val


def nesterov_lp_optimize(a0, time_limit, solver, max_iters=100000):
    """
    Nesterov-accelerated LP best-response iteration.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    a_prev = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    avg_a = np.zeros(n)
    n_avg = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        ss = y_new.sum()
        if ss < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / ss

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            ss = a_new.sum()
            if ss < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / ss

        y = y_new
        a_prev = a
        a = a_new

        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        avg_a += a
        n_avg += 1
        if n_avg >= 50 and n_avg % 100 == 0:
            avg_check = avg_a / n_avg
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()

        if t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.8f}, LP_M={M_val:.10f}")

    return best_a, best_score


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def compute_lse_grad(a, T):
    """
    Compute gradient of (1/T)*log(sum exp(T*(a*a)_k)) w.r.t. a.
    """
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()

    a_rev = a[::-1].copy()
    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    if conv_len > fft_size:
        fft_size *= 2

    W = rfft(w, fft_size)
    A_rev = rfft(a_rev, fft_size)
    conv = irfft(W * A_rev, fft_size)

    grad = 2.0 * conv[n - 1:n - 1 + n]
    return grad, w


def lse_obj_grad_z(z, T):
    """Objective and gradient for z-parameterization (a = softmax(z))."""
    a = softmax_fn(z)
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    obj_val = (max_Tb + np.log(exp_Tb.sum())) / T

    grad_a, w = compute_lse_grad(a, T)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj_val, grad_z


def logsumexp_optimize(a0, time_limit, T_start=100.0, T_end=0.01, n_stages=25):
    """
    LogSumExp-smoothed optimization with L-BFGS and multi-stage annealing.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        iters_per_stage = max(20, min(80, int(30 + 50 * (si / max(1, n_stages - 1)))))

        def func(z_in):
            val, _ = lse_obj_grad_z(z_in, stage_T)
            return val

        def jac(z_in):
            _, g = lse_obj_grad_z(z_in, stage_T)
            return g

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-15, 'maxfun': iters_per_stage * 500,
                         'disp': False}
            )
            z = res.x
            a = softmax_fn(z)
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
        except Exception:
            pass

        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE stage={si}/T={stage_T:.4f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def gradient_direction_search(a0, time_limit, seed=42):
    """
    Local search using gradient direction from active peak.
    The gradient of (a*a)[k*] w.r.t. a[i] is 2*a[k*-i].
    We move mass from high-gradient to low-gradient positions.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        k_star = int(np.argmax(b))
        b_max = b[k_star]

        # Gradient direction: g[i] = a[k_star - i] (reversed around k_star)
        # Positions where a[k_star - i] is large have high gradient
        # We want to decrease a[i] where a[k_star - i] is large
        grad_dir = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad_dir[i] = a[j]

        # Normalize gradient
        g_max = grad_dir.max()
        if g_max < 1e-15:
            break
        grad_norm = grad_dir / g_max

        # Find top source positions (high gradient) and bottom target positions (low gradient)
        # Also exclude positions that would create new peaks
        top_sources = np.argsort(grad_norm)[::-1][:20]
        bottom_targets = np.argsort(grad_norm)[:50]

        improved = False
        for src in top_sources:
            if a[src] < 1e-12:
                continue
            for dst in bottom_targets:
                if dst == src:
                    continue
                # Try small moves
                for frac in [0.001, 0.003, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Try random perturbations with various step sizes
            for _ in range(3):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.03)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss > 1e-15:
                        a_trial = a_trial / ss
                        score = 2.0 * n * fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
                            break

            stall += 1
            if stall > 500:
                break

        if stall % 200 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            print(f"      GDS stall={stall} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def simulated_annealing_search(a0, time_limit, seed=123):
    """
    Simulated annealing with peak-targeted moves.
    Uses a geometric cooling schedule and occasionally reheats.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    current_score = best_score
    t0 = time.monotonic()

    T_start = 0.01
    T_end = 1e-6
    n_iters = 20000
    cool_rate = (T_end / T_start) ** (1.0 / max(1, n_iters))

    stall = 0
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        T = T_start * (cool_rate ** it)

        # Occasionally reheat
        if it > 0 and it % 5000 == 0:
            T = T_start * 0.1

        b = fast_autocorr(a)
        k_star = int(np.argmax(b))

        # Choose move type
        move_type = rng.integers(0, 4)

        if move_type == 0:
            # Gradient-directed move
            grad_dir = np.zeros(n)
            for i in range(n):
                j = k_star - i
                if 0 <= j < n:
                    grad_dir[i] = a[j]
            src = rng.choice(n, p=np.maximum(grad_dir, 1e-30) / np.maximum(grad_dir, 1e-30).sum())
            dst = rng.integers(n)
        elif move_type == 1:
            # Random pair move
            src = rng.integers(n)
            dst = rng.integers(n)
        elif move_type == 2:
            # Block move: move mass from a small block
            block_size = min(3, n // 10)
            src_start = rng.integers(0, max(1, n - block_size))
            dst = rng.integers(n)
            # Use first position of block as source
            src = src_start
        else:
            # Mirror move: move from i to k_star - i
            src = rng.integers(n)
            dst = k_star - src
            if dst < 0 or dst >= n:
                dst = rng.integers(n)

        if src == dst or a[src] < 1e-12:
            continue

        frac = rng.uniform(0.001, 0.05) * (1.0 + 2.0 * T / T_start)
        amt = a[src] * frac
        a_trial = a.copy()
        a_trial[src] -= amt
        a_trial[dst] += amt
        a_trial = np.maximum(a_trial, 0)
        ss = a_trial.sum()
        if ss < 1e-15:
            continue
        a_trial = a_trial / ss

        score = 2.0 * n * fast_autocorr(a_trial).max()

        # Metropolis acceptance
        delta = score - current_score
        if delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-10)):
            a = a_trial
            current_score = score
            if score < best_score:
                best_score = score
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        if stall > 2000:
            # Reheat
            T = T_start * 0.01
            stall = 0

        if it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SA it={it} T={T:.6f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def peak_shaving_targeted(a0, time_limit, seed=42):
    """
    Targeted peak shaving: simultaneously addresses top K peaks.
    Uses a scoring function to find optimal redistribution.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    K = 5

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:K]
        b_max = b[top_k[0]]

        # Compute "danger" for each position: how much it contributes to top peaks
        danger = np.zeros(n)
        for k in top_k:
            weight = b[k] / b_max
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                partner = k - j
                if 0 <= partner < n:
                    danger[j] += weight * a[j] * a[partner]

        # Find positions with high danger to move from, and low danger to move to
        high_danger = np.where(danger > np.mean(danger) + np.std(danger))[0]
        low_danger = np.where(danger < np.mean(danger) - 0.5 * np.std(danger))[0]

        if len(high_danger) == 0 or len(low_danger) == 0:
            high_danger = np.argsort(danger)[::-1][:10]
            low_danger = np.argsort(danger)[:20]

        improved = False
        for src in high_danger[:15]:
            if a[src] < 1e-12:
                continue
            # Sort targets by danger (lowest first)
            targets = sorted(low_danger.tolist(), key=lambda j: danger[j])
            for dst in targets[:10]:
                if dst == src:
                    continue
                for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Random exploration
            for _ in range(5):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.04)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss > 1e-15:
                        a_trial = a_trial / ss
                        score = 2.0 * n * fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
                            break

            stall += 1
            if stall > 400:
                break

        if stall % 100 == 0 and stall > 0:
            elapsed = time.monotonic() - t0
            print(f"      PST stall={stall} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def dual_length_search(a0, time_limit, solver, short_n, long_n):
    """
    Search at two different lengths and interpolate back.
    """
    n0 = len(a0)
    best_a = a0.copy()
    best_score = 2.0 * n0 * fast_autocorr(a0).max()
    t0 = time.monotonic()

    # Search at short_n
    if time.monotonic() - t0 < time_limit * 0.4:
        a_short = _interpolate(a0, short_n)
        t_sub = time_limit * 0.35
        a_out, sc = nesterov_lp_optimize(a_short, t_sub, solver)
        if a_out is not None:
            score = 2.0 * short_n * fast_autocorr(a_out).max()
            if score < best_score:
                best_a = _interpolate(a_out, n0)
                best_score = 2.0 * n0 * fast_autocorr(best_a).max()

    # Search at long_n
    if time.monotonic() - t0 < time_limit * 0.8:
        a_long = _interpolate(a0, long_n)
        t_sub = time_limit * 0.3
        remaining = time_limit - (time.monotonic() - t0)
        if remaining > 10:
            a_out, sc = nesterov_lp_optimize(a_long, min(t_sub, remaining), solver)
            if a_out is not None:
                score = 2.0 * long_n * fast_autocorr(a_out).max()
                if score < best_score:
                    best_a = _interpolate(a_out, n0)
                    best_score = 2.0 * n0 * fast_autocorr(best_a).max()

    return best_a, best_score


def _interpolate(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def coordinate_descent_polish(a0, time_limit, seed=777):
    """
    Final polish: extremely fine-grained coordinate descent.
    Tries tiny moves in each coordinate.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    # Precompute convolution
    b = fast_autocorr(a)
    k_star = int(np.argmax(b))

    # Try very small adjustments
    fracs = [0.0005, 0.001, 0.002, 0.005, 0.01]

    order = rng.permutation(n)
    for _ in range(50):
        if time.monotonic() - t0 > time_limit:
            break

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            if a[i] < 1e-14:
                continue

            # Try moving from i to several targets
            # Prefer targets that are "far" in convolution sense
            j1 = (i + n // 3) % n
            j2 = (i + 2 * n // 3) % n
            j3 = k_star - i
            if j3 < 0 or j3 >= n:
                j3 = (i + 1) % n

            for j in [j1, j2, j3]:
                if j == i:
                    continue
                for frac in fracs:
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()

        # Shuffle order
        order = rng.permutation(n)

    return best_a, best_score


def propose_candidate():
    """
    Main search function. Multi-phase optimization:
    - Phase 1: Nesterov LP at parent length
    - Phase 2: LogSumExp gradient refinement
    - Phase 3: Gradient-direction local search
    - Phase 4: Simulated annealing
    - Phase 5: Peak-targeted shaving
    - Phase 6: Multi-length exploration
    - Phase 7: Final coordinate descent polish
    """
    rng = np.random.default_rng(12345)
    t_global_start = time.monotonic()
    time_budget = min(1800.0 - 60.0, 1740.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Time budget: {time_budget:.0f}s")

    # Load parent sequence
    best_seq = None
    best_score = float('inf')
    parent = None
    n_base = 1600

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            n_base = len(parent)
            p_score = evaluate_safe(parent)
            print(f"Parent: n={n_base}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        print("No valid parent, using default initialization")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is None:
        # Create a good initial sequence: slightly non-uniform
        x = np.linspace(0, 1, n_base)
        init = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.15 * np.cos(4 * np.pi * x)
        init = np.maximum(init, 0.01)
        init = init / init.sum()
        best_seq = init
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    # === Phase 1: Nesterov LP at parent length ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: Nesterov LP at n={n_base} (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.28, 420.0)
    if t1 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t1, solver)
        try_update(a_out, f"nlp_n{n_base}")

    # === Phase 2: LogSumExp gradient refinement ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: LogSumExp at n={n_base} (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.18, 280.0)
    if t2 > 15:
        a_out, _ = logsumexp_optimize(best_seq, t2, T_start=80.0, T_end=0.01, n_stages=22)
        try_update(a_out, f"lse_n{n_base}")

    # === Phase 3: Gradient-direction local search ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: Gradient-direction search at n={n_base} (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.14, 220.0)
    if t3 > 10:
        a_out, _ = gradient_direction_search(best_seq, t3, seed=42)
        try_update(a_out, f"gds_n{n_base}")

    # === Phase 4: Simulated annealing ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: Simulated annealing at n={n_base} (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.14, 220.0)
    if t4 > 10:
        a_out, _ = simulated_annealing_search(best_seq, t4, seed=123)
        try_update(a_out, f"sa_n{n_base}")

    # === Phase 5: Peak-targeted shaving ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 5: Peak-targeted shaving at n={n_base} (remaining={remaining:.0f}s) ===")
    t5 = min(remaining * 0.12, 180.0)
    if t5 > 10:
        a_out, _ = peak_shaving_targeted(best_seq, t5, seed=42)
        try_update(a_out, f"pst_n{n_base}")

    # === Phase 6: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 6: Multi-length exploration (remaining={remaining:.0f}s) ===")
    if remaining > 60:
        t6 = min(remaining * 0.2, 200.0)
        if t6 > 20:
            n_best = len(best_seq)
            short_n = max(800, n_best - 200)
            long_n = min(2400, n_best + 200)
            a_out, _ = dual_length_search(best_seq, t6, solver, short_n, long_n)
            try_update(a_out, "dual_len")

    # === Phase 7: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final polish at n={n_best} (remaining={remaining:.0f}s) ===")

    # Coordinate descent
    t7a = min(remaining * 0.3, 120.0)
    if t7a > 5:
        a_out, _ = coordinate_descent_polish(best_seq, t7a, seed=777)
        try_update(a_out, "cd_polish")

    # Another round of gradient search
    remaining = time_budget - (time.monotonic() - t_global_start)
    t7b = min(remaining * 0.25, 100.0)
    if t7b > 5:
        a_out, _ = gradient_direction_search(best_seq, t7b, seed=555)
        try_update(a_out, "gds_2")

    # Another LogSumExp with very low temperature
    remaining = time_budget - (time.monotonic() - t_global_start)
    t7c = min(remaining * 0.25, 80.0)
    if t7c > 5:
        a_out, _ = logsumexp_optimize(best_seq, t7c, T_start=20.0, T_end=0.005, n_stages=12)
        try_update(a_out, "lse_final")

    # Final coordinate descent
    remaining = time_budget - (time.monotonic() - t_global_start)
    t7d = min(remaining * 0.2, 60.0)
    if t7d > 5:
        a_out, _ = coordinate_descent_polish(best_seq, t7d, seed=999)
        try_update(a_out, "cd_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    # Ensure output is valid
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return result
