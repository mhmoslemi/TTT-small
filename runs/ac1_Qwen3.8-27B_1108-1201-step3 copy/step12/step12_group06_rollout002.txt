"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Adaptive LogSumExp L-BFGS with FFT-accelerated gradients
- Nesterov-accelerated peak-targeted subgradient with Polyak step and averaging
- LP Nash equilibrium (best-response) iteration
- Multi-length exploration and final polishing
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


# Global state for LSE callbacks (to avoid closures)
_lse_state = {'T': 1.0, 'n': 1}


def fast_convolve(a):
    """Fast linear convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score assuming sum(a)=1: returns 2*n*max(b)."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax_from_z(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Inverse of softmax (log)."""
    return np.log(np.maximum(a, 1e-300))


def lse_obj_grad_z(z, T, n):
    """
    Compute LSE-based score and gradient w.r.t. z.
    score = 2n * T * log(sum_k exp(c_k/T)) where c = conv(a,a), a = softmax(z)
    """
    a = softmax_from_z(z)
    c = fast_convolve(a)  # length 2n-1
    
    c_scaled = c / T
    c_max = c_scaled.max()
    exp_c = np.exp(c_scaled - c_max)
    w = exp_c / exp_c.sum()
    
    lse_val = T * (c_max + np.log(exp_c.sum()))
    score = 2.0 * n * lse_val
    
    # grad_a[j] = 2 * sum_k w[k] * a[k-j]
    # = 2 * (a_rev * w)[j + n - 1]
    a_rev = a[::-1]
    fft_size = 1 << (n + len(w) - 2).bit_length()  # need at least n + (2n-1) - 1 = 3n-2
    # Actually, len(a_rev) = n, len(w) = 2n-1, so conv length = n + 2n-1 - 1 = 3n-2
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    A_rev = rfft(a_rev, fft_size)
    W = rfft(w, fft_size)
    conv_aw = irfft(A_rev * W, fft_size)
    grad_a = 2.0 * conv_aw[n - 1: n - 1 + n]
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return score, grad_z


def _lse_func(z_val):
    """Callback for scipy minimize (function)."""
    obj, _ = lse_obj_grad_z(z_val, _lse_state['T'], _lse_state['n'])
    return obj


def _lse_jac(z_val):
    """Callback for scipy minimize (gradient)."""
    _, grad = lse_obj_grad_z(z_val, _lse_state['T'], _lse_state['n'])
    return grad


def lse_lbfgs(a0, time_limit, T_start=50.0, T_min=0.1, n_stages=30, iters_per_stage=40):
    """Adaptive LogSumExp L-BFGS."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)
    
    T = T_start
    stagnant = 0
    
    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        _lse_state['T'] = T
        _lse_state['n'] = n
        
        try:
            res = minimize(
                fun=_lse_func, x0=z, jac=_lse_jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-16,
                         'gtol': 1e-12, 'maxfun': iters_per_stage * 500,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(z)
            sc = score_normalized(a_check)
            
            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()
            
            # Adaptive temperature
            if prev_score is not None:
                if sc > best_score * 1.003:
                    stagnant += 1
                else:

def build_conv_matrix(a_fixed):
    """Build the convolution matrix T where (T@b)[k] = (a_fixed*b)[k]."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T

    return cp.ECOS
