If `a_j = 1 + c * (2j/(n-1) - 1)^2`, then:
- At edges: `a_0 = a_{n-1} = 1 + c`
- At center: `a_{n/2} = 1`

The convolution `(a*a)_k` will be flatter than the triangular convolution of the uniform sequence.

Let me compute: if `a_j = 1 + c * x_j^2` where `x_j = 2j/(n-1) - 1`, then
`(a*a)_k = sum_j (1 + c*x_j^2)(1 + c*x_{k-j}^2)`
`= (number of terms) + c * sum_j (x_j^2 + x_{k-j}^2) + c^2 * sum_j x_j^2 * x_{k-j}^2`

For large `n`, the sums are approximately integrals. The key is that the `c^2` term has a specific lag dependence that can help flatten the convolution.

I think a good initialization is a quadratic (parabolic) shape. Let me also consider:
- `a_j = 1 + c * cos(2*pi*j/n)` (single harmonic)
- `a_j = 1 + c * cos(4*pi*j/n)` (double harmonic)

The cosine perturbations have the property that their self-convolution is also a cosine (by the convolution theorem), so they can be analyzed exactly.

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Structured initialization (parabolic, harmonic, U-shaped)
2. Nesterov with smoothed best response (softmax-weighted multiplicative updates)
3. Nesterov with exact LP best response (cutting plane)

import numpy as np

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score_normalized(a):
    """Score assuming sum(a)=1. Returns 2*n*max(conv(a,a))."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()

def evaluate_safe(a):

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))

def build_conv_matrix(a_fixed):
    """Build the (2n-1) x n matrix T where T[k,j] = a[k-j] if valid."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a_fixed
    return T

def init_parabolic(n, c=0.5):
    """U-shaped / parabolic initialization."""
    x = np.linspace(-1, 1, n)
    a = 1.0 + c * x**2
    a = np.maximum(a, 0.01)
    return a / a.sum()

def init_harmonic(n, mode=1, c=0.3):
    """Harmonic perturbation of uniform."""
    j = np.arange(n)
    a = 1.0 + c * np.cos(2.0 * np.pi * mode * j / n)
    a = np.maximum(a, 0.01)
    return a / a.sum()

def init_bimodal(n, sep=0.3, width=0.1):
    """Two Gaussian bumps."""
    x = np.linspace(0, 1, n)
    a = np.exp(-((x - 0.5 + sep)**2) / (2*width**2)) + np.exp(-((x - 0.5 - sep)**2) / (2*width**2)) + 0.05
    return a / a.sum()

def init_exponential(n, rate=2.0):
    """Exponential decay."""
    x = np.linspace(0, 1, n)
    a = np.exp(-rate * x) + 0.1
    return a / a.sum()

def smoothed_best_response(a, T_inv=10.0):
    """
    Compute the smoothed best response to `a`.
    This is: b = argmin_{sum(b)=1, b>=0} sum_k exp(T_inv * (a*b)_k) / (2n-1)
    Approximated by: b_j proportional to exp(-T_inv * sum_k softmax_k * a_{k-j})
    where softmax_k = exp(T_inv * (a*a)_k) / sum(exp(T_inv * (a*a)))
    """
    n = len(a)
    b = fast_convolve(a)
    # Compute softmax weights over lags
    T_inv_clipped = min(T_inv, 50.0)
    shifted = T_inv_clipped * b - T_inv_clipped * b.max()
    w = np.exp(shifted)
    w = w / w.sum()
    
    # Compute the "effective gradient" for b
    # d/db_j [sum_k w_k * (a*b)_k] = sum_k w_k * a_{k-j}
    grad = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je+1):
            grad[j] += w[k] * a[k - j]
    
    # b_j proportional to exp(-T_inv * grad_j)
    # But we need to be careful: the "best response" in the smoothed game
    # is b = exp(-T_inv * grad) / sum(exp(-T_inv * grad))
    exponent = -T_inv_clipped * grad
    exponent -= exponent.max()
    b_new = np.exp(exponent)
    s = b_new.sum()
    if s < 1e-300:
        return np.ones(n) / n
    return b_new / s

def nesterov_smoothed(a0, time_limit, T_start=5.0, T_end=100.0, max_iters=100000):
    """
    Nesterov method with smoothed best response.
    The smoothed BR converges to exact BR as T_inv -> 0 (T -> inf).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    last_improve = 0
    
    for it in range(max_iters):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        if it - last_improve > 5000 and t_nest > 10:
            break
        
        # Anneal the temperature
        progress = min(1.0, it / max_iters * 3)
        T_inv = T_start * (T_end / T_start) ** progress
        
        # Nesterov extrapolation
        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            y = a + momentum * (a - a_prev)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:
                y = a.copy()
            else:
                y = y / sy
        else:
            y = a
        
        # Smoothed best response
        b_new = smoothed_best_response(y, T_inv)
        
        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        alpha = 2.0 / (t_new + 1.0)
        
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa
        
        sc_new = score_normalized(a_new)
        
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            last_improve = it
        
        # Restart logic
        if sc_new > best_score * 1.001 and t_nest > 5.0:
            t_nest = 1.0
        else:
            t_nest = t_new
        a_prev = a.copy()
        a = a_new
        
        if (it + 1) % 2000 == 0:
            print(f"    Smoothed Nesterov iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} T_inv={T_inv:.2f}")
    
    return best_a, best_score

def solve_lp_best_response_cutting_plane(a_fixed, max_rounds=20, solver='ECOS'):
    """
    Solve the best response LP using a cutting plane approach.
    Start with a small set of constraints and add violated ones.
    """
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Start with all constraints (for medium n, this is fine)
    # For large n, use cutting plane
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in ['PDLP', 'ECOS', 'SCS']:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def lse_objective_grad(z_in, T_val, n):
    """LogSumExp objective and gradient in z-space."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp
    
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-20:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z

def lbgfs_logsumexp(a0, time_limit, T_start=80.0, T_end=0.03, n_stages=20):
    """L-BFGS on LogSumExp approximation."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, stage_T in enumerate(temps):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        remaining = time_limit - elapsed
        if remaining < 3:
            break
        
        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, stage_T, n)
            return obj
        
        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, stage_T, n)
            return grad
        
        iters = 30 if si < n_stages // 2 else 15
        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-20,
                         'gtol': 1e-16, 'maxfun': iters * 200,
                         'disp': False}
            )
            z_new = res.x
            a_new = softmax_from_z(z_new)
            sc = score_normalized(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            z = z_new
        except Exception:
            pass
    
    return best_a, best_score

def coordinate_descent(a0, time_limit, max_sweeps=100):
    """
    Coordinate descent: for each coordinate, find the optimal value
    that minimizes the max convolution value.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    order = np.random.default_rng(42).permutation(n)
    
    for sweep in range(max_sweeps):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        if sweep > 10 and best_score >= score_normalized(a) * (1 - 1e-10):
            # Check for stagnation
            pass
        
        for p in order:
            # Try different values of a[p]
            old_val = a[p]
            
            # The convolution values affected by a[p] are at lags k where
            # a[p] * a[k-p] appears in (a*a)_k
            # (a*a)_k = sum_j a[j] * a[k-j]
            # a[p] appears when j=p (contributing a[p]*a[k-p]) or when k-j=p (contributing a[k-p]*a[p])
            # So (a*a)_k changes by: delta * a[k-p] + a[k-p] * delta = 2*delta*a[k-p] (for k-p valid)
            # Wait, actually in the convolution, a[p] appears once in each term:
            # (a*a)_k = ... + a[p]*a[k-p] + ... (when k-p is valid)
            # So changing a[p] to a[p]+delta changes (a*a)_k by delta * a[k-p] for valid k-p
            
            # We want to minimize max_k (a*a)_k subject to a[p] >= 0 and sum(a) = 1
            # If we change a[p] by delta, sum changes by delta, so we need to rescale
            # For simplicity, let's try a few values and pick the best
            
            # Quick approach: binary search on a[p]
            # If we increase a[p], some convolution values increase, some decrease
            # The optimal a[p] balances the "peak" lags
            
            # Try a small perturbation
            delta = old_val * 0.1
            if delta < 1e-8:
                delta = 1e-8
            
            best_local = old_val
            best_local_score = best_score  # Use current score as baseline
            
            for frac in [-0.3, -0.15, -0.05, 0.05, 0.15, 0.3]:
                new_val = max(0.0, old_val * (1.0 + frac))
                if new_val == old_val:
                    continue
                a[p] = new_val
                s_new = a.sum()
                if s_new < 1e-10:
                    a[p] = old_val
                    continue
                a_norm = a / s_new
                sc = score_normalized(a_norm)
                if sc < best_local_score:
                    best_local_score = sc
                    best_local = new_val
            
            a[p] = best_local
            s_new = a.sum()
            a = a / s_new
            sc = score_normalized(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        if (sweep + 1) % 10 == 0:
            print(f"    CoordDescent sweep={sweep+1} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score

def peak_shaving(a0, time_limit, max_iters=20000):
    """
    Targeted peak shaving: identify the lag with maximum convolution value
    and directly reduce it by adjusting the contributing coordinates.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = 0.02
    
    for it in range(max_iters):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        if it > 100 and best_score >= score_normalized(a) and it > 1000:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        # Find lags near the peak
        threshold = max_b * 0.95
        peak_lags = np.where(b >= threshold)[0]
        
        if len(peak_lags) == 0:
            peak_lags = np.array([int(np.argmax(b))])
        
        # For each peak lag k, the contributing pairs are (j, k-j) for valid j
        # To reduce b[k], we want to reduce a[j]*a[k-j] for the contributing pairs
        # Strategy: reduce the larger of the two in each pair
        
        grad = np.zeros(n)
        for k in peak_lags:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                m = k - j
                # a[j] * a[m] contributes to b[k]
                # Reduce a[j] if it's larger (more effective)
                weight = a[m]  # Gradient of b[k] w.r.t. a[j] is a[m]
                grad[j] += weight / len(peak_lags)
        
        # Multiplicative reduction
        exponent = -eta * grad
        exponent -= exponent.max()
        new_a = a * np.exp(exponent)
        sa = new_a.sum()
        if sa < 1e-15:
            eta *= 0.5
            continue
        new_a = new_a / sa
        
        sc = score_normalized(new_a)
        if sc < best_score:
            best_score = sc
            best_a = new_a.copy()
            a = new_a
            eta = min(eta * 1.01, 0.1)
        else:
            eta *= 0.9
            if eta < 1e-10:
                break
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Resample a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        arr = np.maximum(arr, 0).astype(np.float64)
        s = arr.sum()
        if s < 0.01:
            return
        arr = arr / s
        sc = evaluate_safe(arr)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = arr.copy()
            best_n = len(arr)
            elapsed = time.monotonic()
            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(arr)}) t={elapsed:.0f}s")
    
    # === Phase 1: Quick structured init search (30s) ===
    print(f"\n=== Phase 1: Structured init search ===")
    t1_available = min(30.0, (deadline - time.monotonic()) * 0.05)
    if t1_available > 5:
        # Try different initializations and pick the best
        inits = [
            init_parabolic(best_n, 0.3),
            init_parabolic(best_n, 0.5),
            init_parabolic(best_n, 0.8),
            init_parabolic(best_n, 1.2),
            init_harmonic(best_n, 1, 0.3),
            init_harmonic(best_n, 2, 0.2),
            init_bimodal(best_n, 0.25, 0.12),
            init_bimodal(best_n, 0.35, 0.08),
        ]
        for i, a_init in enumerate(inits):
            a_init = np.maximum(a_init, 0)
            s = a_init.sum()
            if s > 1e-10:
                a_init = a_init / s
                sc = score_normalized(a_init)
                if sc < best_score:
                    best_score = sc
                    best_seq = a_init.copy()
                    print(f"  Better init #{i}: score={sc:.8f}")
    
    # === Phase 2: Smoothed Nesterov (fast, no LP) (300s) ===
    print(f"\n=== Phase 2: Smoothed Nesterov (n={best_n}, score={best_score:.8f}) ===")
    remaining = deadline - time.monotonic()
    t2 = min(300.0, remaining * 0.2)
    if t2 > 20:
        a_out, _ = nesterov_smoothed(best_seq, t2, T_start=3.0, T_end=80.0, max_iters=200000)
        try_update(a_out, "smoothed_nesterov_p2")
    print(f"  Phase 2 done: best_score={best_score:.8f}")
    
    # === Phase 3: LP Nesterov (main workhorse) (500s) ===
    print(f"\n=== Phase 3: Nesterov LP (n={best_n}, score={best_score:.8f}) ===")
    remaining = deadline - time.monotonic()
    t3 = min(500.0, remaining * 0.35)
    if t3 > 30:
        a_out, _ = nesterov_lp(best_seq, t3, solver='ECOS', max_iters=50000)
        try_update(a_out, "nesterov_lp_main")
    print(f"  Phase 3 done: best_score={best_score:.8f}")
