        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS)
        except Exception:
            try:
                prob.solve(solver=cp.SCS)
            except Exception:
                return None, None
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """
    Nesterov-accelerated LP best-response for bilinear minimax.
    Tracks both running iterate and averaged iterate.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    iters = 0
    stall = 0

    # Exponential moving average
    ema = a.copy()
    ema_alpha = 0.03

    for t in range(1, 500000):
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue
        iters += 1

        # Nesterov schedule
        alpha_t = 2.0 / (t + 2.0)

        # Update y (lookahead point)
        y_new = (1.0 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        yn = y_new.sum()
        if yn < 1e-15:
            y_new = y.copy()
            yn = 1.0
        y_new = y_new / yn

        # Update a with momentum
        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            an = a_new.sum()
            if an < 1e-15:
                a_new = y_new.copy()
                an = 1.0
            a_new = a_new / an

        y = y_new
        a = a_new

        # Score and track
        ac = fast_autocorr(a)
        score_a = 2.0 * n * ac.max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Also check y
        yc = fast_autocorr(y)
        score_y = 2.0 * n * yc.max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0

        # EMA
        ema = (1.0 - ema_alpha) * ema + ema_alpha * a
        ema = np.maximum(ema, 0)
        em = ema.sum()
        if em > 1e-15:
            ema = ema / em
            score_ema = 2.0 * n * fast_autocorr(ema).max()
            if score_ema < best_score:
                best_score = score_ema
                best_a = ema.copy()
                stall = 0

        # Adapt EMA
        if stall > 500:
            ema_alpha = min(0.1, ema_alpha * 1.03)

        if verbose and t % 500 == 0:
            print(f"    NLP t={t} ({elapsed:.0f}s): best={best_score:.8f}, M={M_val:.8f}")

        # Early stop
        if stall > 5000 and t > 1000:
            if verbose:
                print(f"    Early stop NLP at t={t}")
            break

    return best_a, best_score, iters


def qp_polish(a0, time_limit, max_constraints=60):
    """
    Adaptive QP polishing: iteratively add active constraints and solve.
    min M s.t. (a*a)_k <= M for k in active_set, sum(a)=1, a>=0
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    active_lags = set()

    for round_num in range(50):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_autocorr(a)
        max_c = c.max()

        # Select active lags: top K by value
        K = min(max_constraints, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:K]
        new_active = set(top_idx.tolist())

        # Also include previously active
        active_lags.update(new_active)
        if len(active_lags) > max_constraints:
            # Keep only the top ones by current c value
            active_lags = set(np.argsort(c)[::-1][:max_constraints].tolist())

        # Build QP
        a_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        constraints = [cp.sum(a_var) == 1, M_var >= 0]

        for k in sorted(active_lags):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            other = k - idx
            q_expr = cp.sum(cp.multiply(a_var[idx], a_var[other]))
            constraints.append(q_expr <= M_var)

        prob = cp.Problem(cp.Minimize(M_var), constraints)
        solved = False
        for solver_try in [cp.ECOS, cp.SCS, cp.CLARABEL]:
            try:
                prob.solve(solver=solver_try, max_iter=50000)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
                    break
            except Exception:
                continue

        if not solved:
            break

        a_new = np.asarray(a_var.value, dtype=np.float64)
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new = a_new / sn

        score_new = 2.0 * n * fast_autocorr(a_new).max()
        if score_new < best_score - 1e-12:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new
            if verbose_qp():
                print(f"    QP r{round_num}: {best_score:.8f} ({len(active_lags)} constraints)")
        else:
            # Try mixing
            improved_mix = False
            for frac in [0.2, 0.4, 0.6, 0.8]:
                a_mix = (1.0 - frac) * a + frac * a_new
                a_mix = np.maximum(a_mix, 0)
                sm = a_mix.sum()
                if sm < 1e-15:
                    continue
                a_mix = a_mix / sm
                sc_mix = 2.0 * n * fast_autocorr(a_mix).max()
                if sc_mix < best_score - 1e-12:
                    best_score = sc_mix
                    best_a = a_mix.copy()
                    a = a_mix
                    improved_mix = True
                    break
            if not improved_mix and round_num >= 3:
                break

    return best_a, best_score


def verbose_qp():
    return True


def frank_wolfe_polish(a0, time_limit, n_iters=None):
    """
    Frank-Wolfe (conditional gradient) on the simplex.
    Uses smoothed subgradient from top peaks.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    if n_iters is None:
        n_iters = 200000

    temp = 0.01  # Smoothing temperature

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_autocorr(a)
        max_c = c.max()

        # Smoothed subgradient: weighted average of top peaks
        n_top = min(10, 2 * n - 1)
        top_idx = np.argsort(c)[::-1][:n_top]
        weights = np.exp((c[top_idx] - max_c) / temp)
        weights = weights / weights.sum()

        # Compute weighted gradient
        grad = np.zeros(n)
        for w, k in zip(weights, top_idx):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            other = k - idx
            # For c_k = sum a_j * a_{k-j}, grad w.r.t a_i:
            # If i is in idx: contribution is 2*w*a[other] (or w*a[i] if i==other)
            # If i is in other: contribution is 2*w*a[idx]
            for j_pos, (j, o) in enumerate(zip(idx, other)):
                if j == o:
                    grad[j] += 2.0 * w * a[j]
                else:
                    grad[j] += 2.0 * w * a[o]
                    grad[o] += 2.0 * w * a[j]

        # LMO for simplex: e_{argmin(grad)}
        min_idx = np.argmin(grad)
        LW = np.zeros(n)
        LW[min_idx] = 1.0

        # Step size with line search (simplified)
        gamma = 2.0 / (it + 2.0)

        # Try a few step sizes
        best_trial_score = best_score
        best_trial_a = a.copy()
        for g in [gamma * 0.5, gamma, gamma * 1.5, 2.0 / (it + 2.0) * 2.0]:
            if g > 1.0:
                continue
            a_trial = (1.0 - g) * a + g * LW
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st < 1e-15:
                continue
            a_trial = a_trial / st
            sc = 2.0 * n * fast_autocorr(a_trial).max()
            if sc < best_trial_score:
                best_trial_score = sc
                best_trial_a = a_trial.copy()

        if best_trial_score < best_score - 1e-14:
            best_score = best_trial_score
            best_a = best_trial_a.copy()
            a = best_trial_a.copy()
        else:
            # Decay temperature slowly (annealing)
            if it % 1000 == 0:
                temp = max(0.001, temp * 0.95)
            # Accept anyway for exploration (with small probability)
            if it % 10 == 0:
                a = (1.0 - gamma * 0.1) * a + gamma * 0.1 * LW
                a = np.maximum(a, 0)
                sa = a.sum()
                if sa > 1e-15:
                    a = a / sa

    return best_a, best_score


def peak_shave(a0, time_limit, seed=42):
    """
    Targeted peak shaving: identify peak-contributing pairs and redistribute.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(a)
        max_c = c.max()
        n_peaks = min(8, 2 * n - 1)
        top_peaks = np.argsort(c)[::-1][:n_peaks]

        improved = False
        for k in top_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            other = k - idx
            contrib = a[idx] * a[other]
            top_ci = np.argsort(contrib)[::-1][:3]

            for ci in top_ci:
                src = idx[ci]
                src_o = other[ci]

                # Try moving mass from src (and src_o) to a distant position
                # Choose target: position far from peak-contributing positions
                peak_positions = set()
                for k2 in top_peaks[:4]:
                    js2 = max(0, k2 - n + 1)
                    je2 = min(n - 1, k2)
                    if js2 <= je2:
                        peak_positions.update(range(js2, je2 + 1))

                # Score candidate targets
                candidates = []
                for j in range(n):
                    if j in peak_positions:
                        continue
                    dist = min(abs(j - p) for p in peak_positions) if peak_positions else n
                    candidates.append((dist, -a[j], j))
                candidates.sort(key=lambda x: (x[0], x[1]), reverse=True)

                for _, _, dst in candidates[:5]:
                    for frac in [0.005, 0.01, 0.02, 0.04]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        if src != src_o and a_trial[src_o] >= amt * 0.5:
                            a_trial[src_o] -= amt * 0.5
                        a_trial[dst] += amt
                        if (a_trial < -1e-12).any():
                            continue
                        a_trial = np.maximum(a_trial, 0)
                        st = a_trial.sum()
                        if st < 1e-15:
                            continue
                        a_trial = a_trial / st
                        sc = 2.0 * n * fast_autocorr(a_trial).max()
                        if sc < best_score - 1e-14:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape
            if stall % 5 == 0 and stall < 200:
                npert = rng.integers(1, 4)
                sources = rng.choice(n, size=npert, replace=False)
                targets = rng.choice(n, size=npert, replace=False)
                a_trial = a.copy()
                for src, dst in zip(sources, targets):
                    frac = rng.uniform(0.005, 0.03)
                    amt = a_trial[src] * frac
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-15:
                    a_trial = a_trial / st
                    sc = 2.0 * n * fast_autocorr(a_trial).max()
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        stall = 0
            if stall > 300:
                break

    return best_a, best_score


def interpolate_to_length(a, new_n):
    """Linearly interpolate a normalized sequence to a new length."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n, endpoint=False)
    new_x = np.linspace(0, 1, new_n, endpoint=False)
    a_new = np.interp(new_x, old_x, a_norm)
    a_new = np.maximum(a_new, 0)
    sn = a_new.sum()
    if sn > 1e-15:
        a_new = a_new / sn
    return a_new


def coordinate_descent_polish(a0, time_limit, n_iters=50000):
    """
    Simple coordinate descent: at each step, pick the coordinate that most
    reduces the max autocorrelation, and adjust its value.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        c = fast_autocorr(a)
        max_c = c.max()
        k_star = np.argmax(c)

        # Try moving mass from high-contributing pair at k_star
        # to a low-contributing position
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        if js > je:
            continue
        idx = np.arange(js, je + 1)
        other = k_star - idx
        contrib = a[idx] * a[other]

        # Find the pair with max contribution
        ci = np.argmax(contrib)
        src = idx[ci]
        src_o = other[ci]

        # Find best target
        # Score: how much does adding mass at position j increase max autocorr?
        # Approximate: look at c at lags involving j
        # For speed, just try a few random + heuristic targets
        targets = []
        # Heuristic: positions far from src
        for j in range(n):
            if j == src or (src_o != src and j == src_o):
                continue
            targets.append(j)
        # Shuffle and try first few
        rng.shuffle(targets)

        improved = False
        for dst in targets[:20]:
            # Binary search for best amount to move
            lo, hi = 0.0, a[src] * 0.5
            best_amt = 0.0
            best_sc = best_score
            for _ in range(8):
                mid = (lo + hi) / 2.0
                a_trial = a.copy()
                a_trial[src] -= mid
                if src != src_o and a_trial[src_o] >= mid * 0.5:
                    a_trial[src_o] -= mid * 0.5
                a_trial[dst] += mid
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st < 1e-15:
                    break
                a_trial = a_trial / st
                sc = 2.0 * n * fast_autocorr(a_trial).max()
                if sc < best_sc:
                    best_sc = sc
                    best_amt = mid
                    lo = mid
                else:
                    hi = mid

            if best_amt > 0 and best_sc < best_score - 1e-14:
                a_trial = a.copy()
                a_trial[src] -= best_amt
                if src != src_o and a_trial[src_o] >= best_amt * 0.5:
                    a_trial[src_o] -= best_amt * 0.5
                a_trial[dst] += best_amt
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-15:
                    a_trial = a_trial / st
                    a = a_trial
                    best_score = best_sc
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            break

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Combines LP Nesterov, QP polishing,
    Frank-Wolfe, and peak shaving across multiple lengths.
    """
    t_start = time.monotonic()
    time_budget = min(float(budget_s) - 100.0, 1700.0)

    solver = find_lp_solver()
    print(f"LP solver: {solver}, time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            best_seq = parent.copy()
            best_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600

        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)

        if s < 0.01:
            return

        if np.isfinite(sc) and sc < best_score:
            best_score = sc

    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 2: QP Polish at n={len(best_seq)} ({remaining:.0f}s remaining) ===")

    if t2 > 20:
        a_out, sc_out = qp_polish(best_seq, t2, max
