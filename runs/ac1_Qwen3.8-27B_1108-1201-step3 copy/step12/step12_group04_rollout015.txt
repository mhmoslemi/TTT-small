            W = rfft(w, fft_size)
            Y_rev = rfft(y_rev, fft_size)
            conv_wy = irfft(W * Y_rev, fft_size)
            grad = (2.0 / T) * conv_wy[n - 1:2 * n - 1]

            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break

            # Nesterov momentum
            if t == 1:
                a_new = y - eta * grad
            else:
                momentum = (t - 1.0) / (t + 2.0)
                a_new = y + momentum * (y - a_curr) - eta * grad

            # Project onto simplex
            a_new = np.maximum(a_new, 0)
            ss = a_new.sum()
            if ss < 1e-15:
                a_new = y.copy()
            else:
                a_new = a_new / ss

            # Extrapolate for next y
            if t == 1:
                y = a_new
            else:
                y = (1.0 - 2.0 / (t + 2.0)) * y + (2.0 / (t + 2.0)) * a_new
                y = np.maximum(y, 0)
                ss = y.sum()
                if ss < 1e-15:
                    y = a_new.copy()
                else:
                    y = y / ss

            a_prev = a_curr
            a_curr = a_new

            # Periodic score check
            if t % 100 == 0:
                score = 2.0 * n * fast_autocorr(a_curr).max()
                if score < best_score:
                    best_score = score
                    best_a = a_curr.copy()
                    stall_count = 0
                else:
                    stall_count += 1

        a = a_curr
        score = 2.0 * n * fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
            prev_best = best_score

        # Check stagnation: if no improvement across many stages, increase T faster
        if stall_count > len(T_stages) // 3 and si < len(T_stages) - 5:
            # Skip to a higher T
            skip = min(5, len(T_stages) - si - 1)
            if si + skip < len(T_stages):
                pass  # Just continue to next stage

        if si % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLS stage={si}/{n_stages} T={T:.2f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def directed_peak_shave(a0, time_limit, seed=42):
    """
    Directed peak shaving: identifies peak partner positions and moves mass
    away from them toward positions that contribute least to all top peaks.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > 500:
            break

        b = fast_autocorr(a)
        b_max = b.max()
        k_peak = int(np.argmax(b))

        # Positions contributing to this peak
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        if js > je:
            break
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_peak - idx]

        # Top contributors to the peak
        top_contrib = np.argsort(contrib)[::-1][:5]

        # Find top 5 peaks for target scoring
        top_peaks = np.argsort(b)[::-1][:5]

        improved = False
        for tc in top_contrib:
            src = int(idx[tc])
            src_val = a[src]
            if src_val < 1e-12:
                continue

            # Find best target: minimize contribution to all top peaks
            # Sample candidates
            n_cand = min(n, 300)
            candidates = rng.integers(0, n, size=n_cand)
            candidates = np.unique(candidates)

            best_target = -1
            best_cost = float('inf')

            for j in candidates:
                j = int(j)
                if j == src:
                    continue
                cost = 0.0
                for k2 in top_peaks:
                    js2 = max(0, k2 - n + 1)
                    je2 = min(n - 1, k2)
                    if js2 <= j <= je2:
                        partner = k2 - j
                        if 0 <= partner < n:
                            cost += a[partner]
                cost += 0.005 * a[j]
                if cost < best_cost:
                    best_cost = cost
                    best_target = j

            if best_target < 0:
                continue

            dst = best_target
            for frac in [0.01, 0.005, 0.02, 0.003, 0.03, 0.001, 0.04]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    no_improve = 0
                    break

            if improved:
                break

        if not improved:
            no_improve += 1
            # Random exploration
            n_trials = 5
            for _ in range(n_trials):
                i1 = int(rng.integers(n))
                i2 = int(rng.integers(n))
                if i1 == i2 or a[i1] < 1e-10:
                    continue
                frac = float(rng.uniform(0.002, 0.03))
                amt = a[i1] * frac
                a_trial = a.copy()
                a_trial[i1] -= amt
                a_trial[i2] += amt
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    no_improve = 0
                    break

        if no_improve % 100 == 0 and no_improve > 0:
            elapsed = time.monotonic() - t0
            print(f"      DPS no_improve={no_improve} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def micro_polish(a0, time_limit, seed=123):
    """Very fine coordinate descent for final polish."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        k_peak = int(np.argmax(b))

        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        if js > je:
            break
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_peak - idx]
        src = int(idx[np.argmax(contrib)])

        if a[src] < 1e-14:
            break

        # Try very small transfers
        dst = int(rng.integers(n))
        if dst == src:
            continue

        improved = False
        for frac in [0.001, 0.002, 0.0005, 0.003, 0.0002]:
            amt = a[src] * frac
            if amt < 1e-16:
                continue
            a_trial = a.copy()
            a_trial[src] -= amt
            a_trial[dst] += amt
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-15:
                continue
            a_trial = a_trial / ss
            score = 2.0 * n * fast_autocorr(a_trial).max()
            if score < best_score - 1e-14:
                a = a_trial
                best_score = score
                best_a = a.copy()
                improved = True
                break

        if not improved:
            # Try cyclic coordinate: scan all positions
            found = False
            for src2 in range(0, n, max(1, n // 200)):
                if a[src2] < 1e-12:
                    continue
                for dst2 in range(0, n, max(1, n // 200)):
                    if dst2 == src2:
                        continue
                    amt = a[src2] * 0.001
                    a_trial = a.copy()
                    a_trial[src2] -= amt
                    a_trial[dst2] += amt
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        found = True
                        break
                if found:
                    break
            if not found:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = int(rng.integers(0, 10))
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 2
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(3 * x) + 0.1 * np.cos(7 * x) + 0.05 * np.cos(13 * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * (x - 0.3) ** 2) + 0.7 * np.exp(-4 * (x - 0.7) ** 2)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = (1 - x) * (0.5 + 0.5 * np.sin(10 * np.pi * x))
    elif choice == 7:
        x = np.linspace(0, 1, n)
        a = np.maximum(0, 1 - 2 * np.abs(x - 0.5)) + 0.2 * np.abs(np.sin(5 * np.pi * x))
        a = np.maximum(a, 0)
    elif choice == 8:
        x = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * (1 - np.cos(2 * np.pi * x * (n // 30 + 1)))
    else:
        x = np.linspace(0, 1, n)
        a = np.exp(-2 * (x - 0.2) ** 2) + 0.5 * np.exp(-2 * (x - 0.5) ** 2) + 0.3 * np.exp(-2 * (x - 0.8) ** 2)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate():
    """
    Main search function. Multi-phase optimization:
    - Phase 1: Nesterov LP at parent length
    - Phase 2: Nesterov-accelerated LogSumExp (fast gradient)
    - Phase 3: Directed peak shaving
    - Phase 4: Multi-length exploration
    - Phase 5: Final micro-polish
    """
    rng = np.random.default_rng(12345)
    t_global_start = time.monotonic()
    time_budget = min(1800.0 - 60.0, 1740.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None
    n_base = 1600

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            n_base = len(parent)
            p_score = evaluate_safe(parent)
            print(f"Parent: n={n_base}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        print("No valid parent, using default initialization")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is None:
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    # === Phase 1: Nesterov LP at parent length ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: Nesterov LP at n={n_base} (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.30, 420.0)
    if t1 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t1, solver)
        try_update(a_out, f"nlp_n{n_base}")

    # === Phase 2: Nesterov-accelerated LogSumExp ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: Nesterov LSE at n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.30, 450.0)
    if t2 > 20:
        a_out = nesterov_lse(best_seq, t2, T_start=2.0, T_end=400.0, n_stages=40)
        try_update(a_out, f"nlse_n{len(best_seq)}")

    # === Phase 3: Directed peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: Directed peak shaving at n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.25, 350.0)
    if t3 > 15:
        a_out = directed_peak_shave(best_seq, t3, seed=42)
        try_update(a_out, f"dps_n{len(best_seq)}")

    # === Phase 4: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: Multi-length exploration (remaining={remaining:.0f}s) ===")

    n_current = len(best_seq)
    candidate_lengths = []
    for offset in [-200, 200, -400, 400, -600, 600]:
        nl = n_current + offset
        if nl >= 600 and nl <= 3000:
            candidate_lengths.append(nl)

    candidate_lengths = sorted(set(candidate_lengths), key=lambda x: abs(x - n_current))

    time_per_length = min(remaining * 0.60 / max(1, len(candidate_lengths)), 200.0)

    for nl in candidate_lengths:
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining < 30:
            break
        tl = min(time_per_length, remaining * 0.35)
        if tl < 15:
            continue

        print(f"  Trying n={nl} (time={tl:.0f}s)")
        init_long = interpolate_sequence(best_seq, nl)

        # Quick Nesterov LSE (faster than LP for this purpose)
        t_lse = tl * 0.5
        a_out = nesterov_lse(init_long, t_lse, T_start=5.0, T_end=200.0, n_stages=20)
        try_update(a_out, f"nlse_n{nl}")

        # Quick directed peak shave
        remaining_sub = tl - (time.monotonic() - t_global_start - (time.monotonic() - t_global_start))
        rem2 = time_budget - (time.monotonic() - t_global_start)
        t_dps = max(5, min(tl * 0.35, rem2 * 0.08))
        if t_dps > 5:
            a_out2 = directed_peak_shave(a_out, t_dps, seed=nl)
            try_update(a_out2, f"dps_n{nl}")

    # === Phase 5: Final refinement on best sequence ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Final refinement at n={n_best} (remaining={remaining:.0f}s) ===")

    # More Nesterov LSE with tighter T range
    t5a = min(remaining * 0.30, 200.0)
    if t5a > 10:
        a_out = nesterov_lse(best_seq, t5a, T_start=50.0, T_end=1000.0, n_stages=30)
        try_update(a_out, f"nlse_refine_n{n_best}")

    # Directed peak shaving again
    remaining = time_budget - (time.monotonic() - t_global_start)
    t5b = min(remaining * 0.30, 200.0)
    if t5b > 10:
        a_out = directed_peak_shave(best_seq, t5b, seed=777)
        try_update(a_out, f"dps_refine_n{n_best}")

    # Micro polish
    remaining = time_budget - (time.monotonic() - t_global_start)
    t5c = min(remaining * 0.25, 150.0)
    if t5c > 5:
        a_out = micro_polish(best_seq, t5c, seed=555)
        try_update(a_out, f"micro_n{n_best}")

    # One more LSE pass
    remaining = time_budget - (time.monotonic() - t_global_start)
    t5d = min(remaining * 0.20, 100.0)
    if t5d > 5:
        a_out = nesterov_lse(best_seq, t5d, T_start=200.0, T_end=2000.0, n_stages=15)
        try_update(a_out, f"nlse_final_n{n_best}")

    # === Final verification ===
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return result
