def fast_autocorr(a):
    """Full self-convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def subgradient_step(a, eta):
    """
    Compute subgradient of max(conv(a,a)) at a.
    The subgradient is 2*a[shifted] where the shift depends on argmax.
    Returns the projected subgradient (zero-sum).
    """
    n = len(a)
    c = fast_autocorr(a)
    # Use softmax over top peaks for a smoother subgradient
    top_k = np.argsort(c)[::-1][:min(10, len(c))]
    weights = np.zeros(len(c))
    top_vals = c[top_k]
    max_val = top_vals[0]
    # Temperature for softmax
    T = max_val * 0.1 if max_val > 0 else 1.0
    exps = np.exp((top_vals - max_val) / T)
    weights[top_k] = exps / exps.sum()
    
    # g[j] = 2 * sum_k weights[k] * a[k-j] (if valid)
    # This is 2 * (weights * a_rev)[j] at the right offset
    a_rev = a[::-1]
    # conv_len for weights (2n-1) and a_rev (n)
    conv_len = (2*n-1) + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(weights, fft_size)
    AR = rfft(a_rev, fft_size)
    conv = irfft(W * AR, fft_size)
    # The result at index j+n-1 gives sum_k weights[k] * a_rev[k-j] = sum_k weights[k] * a[n-1-k+j]
    # Hmm, let me think about this more carefully.
    # weights has length 2n-1, a_rev has length n
    # conv[m] = sum_i weights[i] * a_rev[m-i]
    # We want g[j] = 2 * sum_k weights[k] * a[k-j] for 0<=k-j<n
    # = 2 * sum_k weights[k] * a_rev[n-1-(k-j)] for 0<=k-j<n
    # Let p = n-1-(k-j), so k = j + n-1-p
    # g[j] = 2 * sum_p weights[j+n-1-p] * a_rev[p]
    # = 2 * (weights * a_rev)[j+n-1]
    # where (weights * a_rev) is the linear convolution
    
    g = 2.0 * conv[n-1:n-1+n]
    # Project to zero-sum
    g = g - g.mean()
    return g, max_val


def nesterov_subgradient(a0, time_limit, max_iters=100000):
    """Nesterov-accelerated subgradient method."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    a_prev = a.copy()
    y = a.copy()
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        alpha_t = 2.0 / (t + 2.0)
        
        # Compute subgradient at y
        g_y, max_val = subgradient_step(y, alpha_t)
        
        # Step size
        eta = 1.0 / (math.sqrt(t) * max(max_val, 1e-10))
        
        # Update
        y_new = y - eta * g_y
        y_new = np.maximum(y_new, 0)
        ss = y_new.sum()
        if ss < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / ss
        
        if t == 1:
            a_new = y_new.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            ss = a_new.sum()
            if ss < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / ss
        
        y = y_new
        a_prev = a
        a = a_new
        
        score = 2.0 * n * fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SUB t={t} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where (T@b) = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a_fixed
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.3])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver, n_peaks=None):
    """
    Solve LP: min M s.t. conv(a_fixed, b) <= M, sum(b)=1, b>=0.
    Optionally, only enforce constraints at top peaks for speed.
    """
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    if n_peaks is not None and n_peaks < 2*n - 1:
        # Only enforce at top peaks (faster)
        b_approx = fast_autocorr(a_fixed)  # Use as proxy for where peaks will be
        top_idx = np.argsort(b_approx)[::-1][:n_peaks]
        T_full = build_conv_matrix(a_fixed)
        T = T_full[top_idx, :]
        constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    else:
        T = build_conv_matrix(a_fixed)
        constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    M_val = float(M_var.value)
    return b_val, M_val

Now let me think about the LogSumExp approach more carefully. The key insight is that we can replace max with a smooth approximation:

max_k c[k] ≈ (1/T) * log(sum_k exp(T * c[k]))

As T → ∞, this approaches the max. For finite T, it's a smooth upper bound, and we can use gradient-based methods.

The gradient of the LogSumExp objective with respect to a is:
d/d(a[j]) (1/T) log(sum_k exp(T*c[k])) = sum_k (exp(T*c[k]) / sum exp(T*c[k'])) * (d*c[k]/d*a[j])

= sum_k w[k] * (d*c[k]/d*a[j])

where w[k] = exp(T*c[k]) / sum exp(T*c[k]) is a softmax distribution over the lags.

And d*c[k]/d*a[j] = 2*a[k-j] (if 0<=k-j<n) else 0 (as derived earlier).

So grad[j] = 2 * sum_k w[k] * a[k-j] (valid) = 2 * (w * a_rev)[j+n-1]

This can be computed with FFT in O(n log n)!

For the L-BFGS optimization, we parameterize with a = softmax(z) to handle the simplex constraint. The gradient in z-space is:
grad_z[j] = a[j] * (grad_a[j] - sum_i a[i]*grad_a[i])

Key improvements: adaptive LP constraint subsets, iterate averaging,

import numpy as np
import cvxpy as cp
import time
import math
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_self_conv(a):
    """Full linear self-convolution a*a using FFT, O(n log n)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def compute_lse_grad(a, T):
    """
    Compute (objective_value, gradient) of (1/T)*log(sum exp(T*(a*a))) w.r.t. a.
    Uses FFT for O(n log n).
    """
    n = len(a)
    b = fast_self_conv(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    obj_val = (max_Tb + np.log(exp_Tb.sum())) / T
    w = exp_Tb / exp_Tb.sum()
    # grad[j] = 2 * sum_k w[k] * a[k-j] (valid) = 2 * (w * a_rev)[j+n-1]
    a_rev = a[::-1].copy()
    conv_len = (2 * n - 1) + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(w, fft_size)
    AR = rfft(a_rev, fft_size)
    conv = irfft(W * AR, fft_size)
    grad = 2.0 * conv[n - 1:n - 1 + n]
    return obj_val, grad, w


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def lse_z_objective(z, T):
    """Objective for z-parameterization (a = softmax(z))."""
    a = softmax_fn(z)
    val, _, _ = compute_lse_grad(a, T)
    return val


def lse_z_gradient(z, T):
    """Gradient for z-parameterization."""
    a = softmax_fn(z)
    _, grad_a, _ = compute_lse_grad(a, T)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return grad_z


def lse_nesterov_optimize(a0, time_limit, T_start=80.0, T_end=1.0, n_stages=15):
    """
    Multi-stage LogSumExp Nesterov gradient optimization.
    T decreases geometrically from T_start to T_end.
    Uses mirror descent step (multiplicative updates) for simplicity and stability.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_self_conv(a).max()
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        # Adaptive iterations per stage
        iters = max(50, min(200, int(30 * (n_stages - si) / n_stages + 30)))

        for it in range(iters):
            if time.monotonic() - t0 > time_limit:
                break

            obj, grad_a, w = compute_lse_grad(a, stage_T)

            # Mirror descent step (multiplicative)
            eta = 1.0 / (stage_T * max(fast_self_conv(a).max(), 1e-10) + 1e-10)
            eta = min(eta, 0.5)

            # a_new[i] proportional to a[i] * exp(-eta * grad_a[i])
            log_a = np.log(np.maximum(a, 1e-30))
            log_a_new = log_a - eta * grad_a
            log_a_new -= log_a_new.max()
            a_new = np.exp(log_a_new)
            a_new = np.maximum(a_new, 0)
            ss = a_new.sum()
            if ss < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new / ss

            score = 2.0 * n * fast_self_conv(a_new).max()
            if score < best_score:
                best_score = score
                best_a = a_new.copy()

            # Damping for stability
            if it < 5:
                a = 0.7 * a + 0.3 * a_new
            else:
                a = a_new

        if si % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage={si}/T={stage_T:.3f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where (T@b) = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.3])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver, n_top=None):
    """
    Solve LP: min M s.t. conv(a_fixed, b) <= M (at all or top n_top lags), sum(b)=1, b>=0.
    """
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T_full = build_conv_matrix(a_fixed)

    if n_top is not None and n_top < 2 * n - 1:
        # Use autocorrelation of a_fixed as proxy for peak locations
        proxy = fast_self_conv(a_fixed)
        top_idx = np.argsort(proxy)[::-1][:n_top]
        T = T_full[top_idx, :]
    else:
        T = T_full

    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        pass

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    if b_val is None:
        return None, None
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    M_val = float(M_var.value)
    return b_val, M_val


def nesterov_lp_optimize(a0, time_limit, solver, max_iters=50000):
    """
    Nesterov-accelerated LP best-response.
    Tracks running average of iterates for improved endpoint.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    a_prev = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_self_conv(a).max()
    t0 = time.monotonic()

    avg_a = np.zeros(n)
    n_avg = 0
    avg_window = 80

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        # Use partial constraints periodically for speed
        if t % 3 == 0 and n > 1000:
            n_top = max(80, n // 2)
        else:
            n_top = None

        b_star, M_val = solve_lp_best_response(y, solver, n_top=n_top)
        if b_star is None:
            continue

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        ss = y_new.sum()
        if ss < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / ss

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            ss = a_new.sum()
            if ss < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / ss

        y = y_new
        a_prev = a
        a = a_new

        # Evaluate and track best
        score_a = 2.0 * n * fast_self_conv(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = 2.0 * n * fast_self_conv(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        # Exponential moving average
        avg_a = (avg_a * min(n_avg, avg_window) + a) / (min(n_avg, avg_window) + 1)
        n_avg += 1
        if n_avg >= 20 and n_avg % 20 == 0:
            score_avg = 2.0 * n * fast_self_conv(avg_a).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_a.copy()

        if t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP t={t} ({elapsed:.0f}s): best={best_score:.8f}, M={M_val:.10f}")

    return best_a, best_score


def lbfgs_lse_optimize(a0, time_limit, T_start=20.0, T_end=0.05, n_stages=12):
    """
    L-BFGS optimization of LogSumExp surrogate with z-parameterization.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_self_conv(a_init).max()

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        iters_per_stage = max(15, min(80, int(20 + 60 * (si / max(1, n_stages - 1)))))

        def func(z_in):
            return lse_z_objective(z_in, stage_T)

        def jac(z_in):
            return lse_z_gradient(z_in, stage_T)

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': iters_per_stage * 200,
                         'disp': False}
            )
            z = res.x
            a_eval = softmax_fn(z)
            score = 2.0 * n * fast_self_conv(a_eval).max()
            if score < best_score:
                best_score = score
                best_a = a_eval.copy()
        except Exception:
            pass

        if si % 4 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LBF stage={si}/T={stage_T:.4f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def peak_shaving(a0, time_limit, seed=42):
    """
    Targeted peak shaving: identify top peaks, find top contributors,
    and redistribute mass to non-contributing positions.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_self_conv(a).max()
    t0 = time.monotonic()
    stall = 0
    K = 6

    while time.monotonic() - t0 < time_limit:
        b = fast_self_conv(a)
        top_k_idx = np.argsort(b)[::-1][:K]
        b_max = b[top_k_idx[0]]

        improved = False
        for k in top_k_idx:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            top_contrib = np.argsort(contrib)[::-1][:4]

            for tc in top_contrib:
                src = int(idx[tc])
                src_val = a[src]
                if src_val < 1e-14:
                    continue

                # Find positions with minimal contribution to all top peaks
                min_contrib = np.full(n, np.inf)
                for k2 in top_k_idx:
                    js2 = max(0, k2 - n + 1)
                    je2 = min(n - 1, k2)
                    for j2 in range(js2, je2 + 1):
                        partner = k2 - j2
                        if 0 <= partner < n:
                            min_contrib[j2] = min(min_contrib[j2],
                                                   (a[j2] * a[partner] + a[j2] * a[j2]) / (2.0 * b_max + 1e-30))

                # Exclude source and positions already in peak areas
                in_peak = np.zeros(n, dtype=bool)
                for k2 in top_k_idx:
                    js2 = max(0, k2 - n + 1)
                    je2 = min(n - 1, k2)
                    in_peak[js2:je2 + 1] = True
                in_peak[src] = False

                candidates = np.where(~in_peak)[0]
                if len(candidates) == 0:
                    candidates = np.where(np.arange(n) != src)[0]

                # Sort by contribution
                scored = np.array([min_contrib[j] for j in candidates])
                order = np.argsort(scored)
                dst_pool = candidates[order[:min(8, len(candidates))]]

                for dst in dst_pool:
                    dst = int(dst)
                    if dst == src:
                        continue
                    for frac in [0.001, 0.003, 0.007, 0.015, 0.03, 0.05]:
                        amt = src_val * frac
                        if amt < 1e-16:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial[src] < -1e-15:
                            continue
                        a_trial = np.maximum(a_trial, 0)
                        ss = a_trial.sum()
                        if ss < 1e-15:
                            continue
                        a_trial = a_trial / ss
                        score = 2.0 * n * fast_self_conv(a_trial).max()
                        if score < best_score - 1e-15:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation
            for _ in range(8):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-12:
                    continue
                frac = rng.uniform(0.001, 0.03)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                score = 2.0 * n * fast_self_conv(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
                    break

            if stall > 500:
                break

        if stall > 0 and stall % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PS stall={stall} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def simulated_annealing(a0, time_limit, seed=123):
    """
    Simulated annealing local search with geometric cooling.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_self_conv(a).max()
    current_score = best_score
    t0 = time.monotonic()

    T_start = max(0.01, (best_score - 1.0) * 0.1)
    T_end = T_start * 0.001
    n_steps = 5000
    T_schedule = np.geomspace(T_start, T_end, n_steps)

    for it in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        T = T_schedule[it]

        # Random move: transfer mass from i to j
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j or a[i] < 1e-14:
            continue

        frac = rng.uniform(0.001, 0.05)
        amt = a[i] * frac
        a_trial = a.copy()
        a_trial[i] -= amt
        a_trial[j] += amt
        if a_trial[i] < -1e-15:
            continue
        a_trial = np.maximum(a_trial, 0)
        ss = a_trial.sum()
        if ss < 1e-15:
            continue
        a_trial = a_trial / ss

        new_score = 2.0 * n * fast_self_conv(a_trial).max()
        delta = new_score - current_score

        if delta < 0 or rng.random() < math.exp(-delta / T):
            a = a_trial
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()

        if it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SA it={it} T={T:.6f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length using cubic spline."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 5)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 1.5
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(3 * x) + 0.15 * np.cos(7 * x) + 0.05 * np.cos(13 * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * (x - 0.3) ** 2) + 0.7 * np.exp(-4 * (x - 0.7) ** 2)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = (1 - x) * (0.6 + 0.4 * np.sin(8 * np.pi * x))
        a = np.maximum(a, 0)
    else:
        x = np.linspace(0, 1, n)
        a = 0.5 * np.ones(n)
        a[:n // 2] = 1.0 - np.linspace(0, 0.3, n // 2)
        a[n // 2:] = 0.7 + 0.3 * np.linspace(0, 1, n - n // 2)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate():
    """
    Main search function. Multi-phase optimization combining:
    - LogSumExp smoothed gradient (fast warmup)
    - LP best-response Nesterov (main workhorse)
    - Multi-length exploration
    - L-BFGS refinement
    - Peak shaving and simulated annealing (final polish)
    """
    rng = np.random.default_rng(2024)
    t_global_start = time.monotonic()
    time_budget = 1740.0  # Leave 60s buffer

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Time budget: {time_budget:.0f}s")

    # Load parent sequence
    best_seq = None
    best_score = float('inf')
    n_base = 1600

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            n_base = len(best_seq)
            p_score = evaluate_safe(best_seq)
            print(f"Parent: n={n_base}, score={p_score:.8f}")
            best_score = p_score
    except Exception:
        print("No valid parent, using structured initialization")
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Init: n={n_base}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: LogSumExp gradient warmup ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: LSE gradient warmup n={n_base} (rem={remaining:.0f}s) ===")
    t1 = min(remaining * 0.12, 200.0)
    if t1 > 30:
        a_out, _ = lse_nesterov_optimize(best_seq, t1, T_start=60.0, T_end=2.0, n_stages=12)
        try_update(a_out, f"lse_warm_n{n_base}")

    # === Phase 2: LP Nesterov at base length ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    n_cur = len(best_seq)
    print(f"\n=== Phase 2: LP Nesterov n={n_cur} (rem={remaining:.0f}s) ===")
    t2 = min(remaining * 0.35, 580.0)
    if t2 > 60:
        a_out, _ = nesterov_lp_optimize(best_seq, t2, solver)
        try_update(a_out, f"nlp_n{n_cur}")

    # Second LP pass with more time if available
    remaining = time_budget - (time.monotonic() - t_global_start)
    t2b = min(remaining * 0.15, 200.0)
    if t2b > 30:
        a_out2, _ = nesterov_lp_optimize(best_seq, t2b, solver)
        try_update(a_out2, f"nlp2_n{n_cur}")

    # === Phase 3: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    n_best = len(best_seq)
    print(f"\n=== Phase 3: Multi-length exploration (rem={remaining:.0f}s) ===")

    candidate_lengths = []
    for offset in [-300, -100, 100, 300, 600, 1000]:
        nl = n_best + offset
        if nl >= 800 and nl <= 4000:
            candidate_lengths.append(nl)
    candidate_lengths = sorted(set(candidate_lengths), key=lambda x: abs(x - n_best))

    n_lengths_to_try = min(len(candidate_lengths), 4)
    time_per_length = min(remaining * 0.55 / max(1, n_lengths_to_try), 350.0)

    for nl in candidate_lengths[:n_lengths_to_try]:
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining < 40:
            break
        tl = min(time_per_length, remaining * 0.45)
        if tl < 25:
            continue

        print(f"  Trying n={nl} (time={tl:.0f}s)")
        init_long = interpolate_sequence(best_seq, nl)

        # LP Nesterov
        t_lp = tl * 0.6
        a_out, _ = nesterov_lp_optimize(init_long, t_lp, solver)

        rem_sub = time_budget - (time.monotonic() - t_global_start)
        t_lse = max(5, min(tl * 0.25, rem_sub * 0.15))
        if t_lse > 5:
            a_out2, _ = lse_nesterov_optimize(a_out, t_lse, T_start=30.0, T_end=1.0, n_stages=8)
            try_update(a_out2, f"lse_n{nl}")
        try_update(a_out, f"nlp_n{nl}")

    # === Phase 4: L-BFGS refinement at best length ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: L-BFGS refinement n={n_best} (rem={remaining:.0f}s) ===")
    t4 = min(remaining * 0.18, 200.0)
    if t4 > 20:
        a_out, _ = lbfgs_lse_optimize(best_seq, t4, T_start=15.0, T_end=0.05, n_stages=10)
        try_update(a_out, f"lbf_n{n_best}")

    # === Phase 5: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: Peak shaving n={n_best} (rem={remaining:.0f}s) ===")
    t5 = min(remaining * 0.18, 180.0)
    if t5 > 15:
        a_out, _ = peak_shaving(best_seq, t5, seed=777)
        try_update(a_out, f"ps_n{n_best}")

    # === Phase 6: Simulated annealing ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Simulated annealing n={n_best} (rem={remaining:.0f}s) ===")
    t6 = min(remaining * 0.15, 120.0)
    if t6 > 10:
        a_out, _ = simulated_annealing(best_seq, t6, seed=456)
        try_update(a_out, f"sa_n{n_best}")

    # === Phase 7: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final polish n={n_best} (rem={remaining:.0f}s) ===")

    # Quick LP pass
    t7a = min(remaining * 0.3, 80.0)
    if t7a > 10:
        a_out, _ = nesterov_lp_optimize(best_seq, t7a, solver)
        try_update(a_out, "final_nlp")

    # Peak shaving again
    remaining = time_budget - (time.monotonic() - t_global_start)
    t7b = min(remaining * 0.3, 60.0)
    if t7b > 10:
        a_out, _ = peak_shaving(best_seq, t7b, seed=999)
        try_update(a_out, "final_ps")

    # Final SA
    remaining = time_budget - (time.monotonic() - t_global_start)
    t7c = min(remaining * 0.3, 40.0)
    if t7c > 5:
        a_out, _ = simulated_annealing(best_seq, t7c, seed=555)
        try_update(a_out, "final_sa")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = min(best_score, fs)
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return result
