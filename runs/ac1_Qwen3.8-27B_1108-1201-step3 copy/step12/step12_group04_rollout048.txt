"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast Nesterov-accelerated LP with top-K peak constraints
2. Exact 1D coordinate descent with analytical line search
3. LogSumExp-smoothed L-BFGS optimization
4. Adaptive simulated annealing for escaping local minima
Key innovations: top-K LP for speed, exact coordinate descent exploiting
the structure of the max-of-linear-functions objective, and adaptive
temperature scheduling.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_autocorr(a):
    """Full self-convolution a*a using FFT, O(n log n)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def normalize(a):
    """Normalize a non-negative array to sum to 1."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.3])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_topk_lp(a_fixed, top_k_indices, solver):
    """
    Solve LP: min M s.t. (a_fixed * b)_k <= M for k in top_k_indices,
    sum(b) = 1, b >= 0.
    Returns (b, M) or (None, None).
    """
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    constraints = [cp.sum(b_var) == 1]
    
    for k in top_k_indices:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        # (a_fixed * b)_k = sum_{j=js}^{je} a_fixed[k-j] * b_j
        row = np.zeros(n)
        for j in range(js, je + 1):
            row[j] = a_fixed[k - j]
        if row.sum() > 0:
            constraints.append(row @ b_var <= M_var)
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    M_val = float(M_var.value)
    return b_val, M_val


def nesterov_topk_lp(a0, time_limit, solver, top_k=30, max_iters=50000):
    """
    Nesterov-accelerated iteration with top-K LP best response.
    """
    n = len(a0)
    a = normalize(np.asarray(a0, dtype=np.float64))
    
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    avg_a = np.zeros(n)
    n_avg = 0
    
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_y = fast_autocorr(y)
        top_k_idx = np.argsort(b_y)[::-1][:top_k]
        
        b_star, M_val = solve_topk_lp(y, top_k_idx, solver)
        if b_star is None:
            continue
        
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        ss = y_new.sum()
        if ss < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / ss
        
        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            ss = a_new.sum()
            if ss < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / ss
        
        y = y_new
        a = a_new
        
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        avg_a += a
        n_avg += 1
        if n_avg >= 50 and n_avg % 100 == 0:
            avg_check = avg_a / n_avg
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()
        
        if t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP t={t} ({elapsed:.0f}s): best={best_score:.8f}, M={M_val:.10f}")
    
    return best_a, best_score


def coordinate_descent(a0, time_limit, n_peaks=30, max_sweeps=5000):
    """
    Exact 1D coordinate descent.
    For each coordinate i, find the optimal delta by analyzing the
    max of (b_k + 2*delta*a_{k-i}) / (S+delta)^2 over top peaks.
    """
    n = len(a0)
    a = normalize(np.asarray(a0, dtype=np.float64))
    S = a.sum()
    b = fast_autocorr(a)
    current_max_b = b.max()
    
    best_a = a.copy()
    best_score = 2.0 * n * current_max_b
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    no_improve_sweeps = 0
    
    for sweep in range(max_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        
        sweep_improved = False
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            a_i = a[i]
            if a_i < 1e-15 and S < 1e-10:
                continue
            
            # Get top peaks
            top_k = np.argsort(b)[::-1][:n_peaks]
            
            # Find best delta for this coordinate
            best_delta = 0.0
            best_new_score = 2.0 * n * current_max_b  # current score (with S=1)
            
            # Candidates: delta=0, delta for each top peak, and some random
            candidates = [0.0]
            
            for k in top_k:
                partner = k - i
                if partner < 0 or partner >= n:
                    continue
                a_partner = a[partner]
                if a_partner < 1e-15:
                    continue
                # Optimal delta for this peak: delta* = S - b[k]/a[partner]
                # (derived from setting d/delta [(b_k + 2*delta*a_p)/(S+delta)^2] = 0)
                delta_star = S - b[k] / a_partner
                # Clamp: delta >= -a_i (so a_i + delta >= 0)
                if delta_star < -a_i:
                    delta_star = -a_i
                candidates.append(delta_star)
            
            # Also try some small random deltas
            for _ in range(3):
                candidates.append(rng.uniform(-a_i * 0.5, a_i * 0.5))
            
            # Evaluate candidates
            for delta in candidates:
                if delta < -a_i - 1e-15:
                    delta = -a_i
                if abs(delta) < 1e-18:
                    continue
                
                new_S = S + delta
                if new_S < 1e-10:
                    continue
                
                # Compute new max autocorrelation (approximately)
                # Only need to check top peaks + the changed ones
                new_max = 0.0
                for k in top_k:
                    partner = k - i
                    if 0 <= partner < n:
                        nb = b[k] + 2.0 * delta * a[partner]
                    else:
                        nb = b[k]
                    if nb > new_max:
                        new_max = nb
                
                new_score = 2.0 * n * new_max / (new_S * new_S)
                
                if new_score < best_new_score * (1 - 1e-12):
                    best_new_score = new_score
                    best_delta = delta
            
            if best_delta != 0.0 and best_new_score < 2.0 * n * current_max_b / (S * S) * (1 - 1e-10):
                # Apply the change
                a[i] = max(a[i] + best_delta, 0.0)
                S = a.sum()
                if S < 1e-15:
                    a = np.ones(n) / n
                    S = 1.0
                    b = fast_autocorr(a)
                    current_max_b = b.max()
                    continue
                
                # Recompute autocorrelation
                b = fast_autocorr(a)
                current_max_b = b.max()
                
                actual_score = 2.0 * n * current_max_b / (S * S)
                if actual_score < best_score * (1 - 1e-12):
                    best_score = actual_score
                    best_a = a.copy()
                    sweep_improved = True
        
        if not sweep_improved:
            no_improve_sweeps += 1
            if no_improve_sweeps >= 3:
                break
        else:
            no_improve_sweeps = 0
        
        if sweep % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD sweep={sweep} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def compute_lse_grad(a, T):
    """
    Compute gradient of (1/T)*log(sum exp(T*(a*a)_k)) w.r.t. a.
    """
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    
    # grad[i] = 2 * sum_k w[k] * a[k-i] (valid k)
    a_rev = a[::-1].copy()
    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    if conv_len > fft_size:
        fft_size *= 2
    
    W = rfft(w, fft_size)
    A_rev = rfft(a_rev, fft_size)
    conv = irfft(W * A_rev, fft_size)
    
    grad = 2.0 * conv[n - 1:n - 1 + n]
    return grad, w


def lse_obj_grad_z(z, T):
    """Objective and gradient for z-parameterization (a = softmax(z))."""
    a = softmax_fn(z)
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    obj_val = (max_Tb + np.log(exp_Tb.sum())) / T
    grad_a, _ = compute_lse_grad(a, T)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj_val, grad_z


def logsumexp_optimize(a0, time_limit, T_start=80.0, T_end=0.01, n_stages=15, max_iter_per_stage=80):
    """LogSumExp-smoothed optimization with L-BFGS and multi-stage annealing."""
    n = len(a0)
    a_init = normalize(np.asarray(a0, dtype=np.float64))
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        def func(z_in):
            val, _ = lse_obj_grad_z(z_in, stage_T)
            return val
        
        def jac(z_in):
            _, g = lse_obj_grad_z(z_in, stage_T)
            return g
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': max_iter_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': max_iter_per_stage * 200,
                         'disp': False}
            )
            z = res.x
            a = softmax_fn(z)
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
        except Exception:
            pass
        
        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage={si} T={stage_T:.4f} ({elapsed:.0f}s): best={best_score:.8f}")
    
    return best_a, best_score


def simulated_annealing(a0, time_limit, n_peaks=10, initial_temp=0.001, cooling=0.9997):
    """
    Simulated annealing with autocorrelation-aware moves.
    Moves: transfer mass between two positions, informed by peak structure.
    """
    n = len(a0)
    a = normalize(np.asarray(a0, dtype=np.float64))
    t0 = time.monotonic()
    rng = np.random.default_rng(777)
    
    b = fast_autocorr(a)
    current_max_b = b.max()
    current_score = 2.0 * n * current_max_b
    
    best_a = a.copy()
    best_score = current_score
    temp = initial_temp
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Choose move type
        if rng.random() < 0.7:
            # Transfer mass: pick source (high value) and target (low contribution to peaks)
            # Source: weighted random from top 30% values
            sorted_idx = np.argsort(a)[::-1]
            top_30 = sorted_idx[:max(1, n // 3)]
            weights = a[top_30]
            weights = weights / weights.sum()
            src_idx = top_30[rng.choice(len(top_30), p=weights)]
            
            # Target: pick a position that contributes least to top peaks
            top_k = np.argsort(b)[::-1][:n_peaks]
            # Score each position by its contribution to top peaks
            contrib = np.zeros(n)
            for k in top_k:
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                for j in range(js, je + 1):
                    contrib[j] += a[j] * a[k - j]
            
            # Prefer positions with low contribution and low value
            target_score = contrib + 0.1 * a
            # Exclude source
            target_score[src_idx] = np.inf
            bottom_50 = np.argsort(target_score)[:n // 2]
            dst_idx = bottom_50[rng.integers(len(bottom_50))]
            
            frac = rng.uniform(0.01, 0.1)
            amt = a[src_idx] * frac
            if amt < 1e-15:
                continue
            
            a_new = a.copy()
            a_new[src_idx] -= amt
            a_new[dst_idx] += amt
            a_new = np.maximum(a_new, 0)
            a_new = normalize(a_new)
            
        else:
            # Gaussian perturbation
            a_new = a + rng.normal(0, 0.005 * a.max(), n)
            a_new = np.maximum(a_new, 0)
            a_new = normalize(a_new)
        
        # Evaluate
        b_new = fast_autocorr(a_new)
        new_max_b = b_new.max()
        new_score = 2.0 * n * new_max_b
        
        delta = new_score - current_score
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-10)):
            a = a_new
            b = b_new
            current_max_b = new_max_b
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
        
        temp *= cooling
        
        if it % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SA it={it} ({elapsed:.0f}s): T={temp:.6f}, best={best_score:.8f}")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
