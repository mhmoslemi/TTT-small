import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2*n-1, n))
    for j in range(n):
        T[j:j+n, j] = a
    return T

def solve_lp(a, solver, warm_start=None):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    prob.solve(solver=solver)
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b = np.asarray(b_var.value, dtype=np.float64)
    b = np.maximum(b, 0)
    s = b.sum()
    if s < 1e-15:
        return None, None
    b /= s
    return b, float(M_var.value)

def nesterov_lp(a0, time_limit, solver, verbose=False):
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    a /= a.sum()
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    
    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_star, M = solve_lp(y, solver)
        if b_star is None:
            continue
        
        alpha = 2.0 / (t + 2.0)
        y_new = (1 - alpha) * y + alpha * b_star
        y_new = np.maximum(y_new, 0)
        y_new /= y_new.sum()
        
        if t == 1:
            a_new = y_new.copy()
        else:
            momentum = (t-1.0) / (t+2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            a_new /= a_new.sum()
        
        y = y_new
        a = a_new
        avg_a += a
        n_avg += 1
        
        # Check scores
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
        
        if t % 10 == 0:
            score_avg = 2.0 * n * fast_autocorr(avg_a / n_avg).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = (avg_a / n_avg).copy()
        
        if verbose and t % 500 == 0:
            print(f"  t={t}, M={M:.6f}, best={best_score:.6f}")
    
    return best_a, best_score

This creates a matrix where column `j` has `a_fixed` starting at row `j`. So `T[k, j] = a_fixed[k-j]` if `0 <= k-j < n`, else 0.

Then `T @ b = sum_j T[k,j] * b[j] = sum_j a_fixed[k-j] * b[j]` (for valid `j`). This is indeed `convolve(a_fixed, b)[k]`.

Good, this is correct.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Extended LP Nesterov with multi-scale averaging and warm-start tracking
2. Subgradient descent with adaptive step size and active-set focusing
3. Coordinate descent polishing with analytical 1D optimization
4. Multi-length exploration with interpolation-based initialization
5. Randomized restarts for escaping local minima
"""

import numpy as np
import time
import cvxpy as cp
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def score_of(a):
    """Compute the objective value for a sequence."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def get_lp_solver():
    """Find the fastest working LP solver."""
    best_solver = cp.ECOS
    best_time = float('inf')
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(50, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            t0 = time.monotonic()
            prob.solve(solver=solver_name)
            elapsed = time.monotonic() - t0
            if prob.status in ('optimal', 'optimal_inaccurate') and elapsed < best_time:
                best_time = elapsed
                best_solver = solver_name
        except Exception:
            continue
    return best_solver


def build_conv_matrix(a_fixed):
    """Build the (2n-1) x n convolution matrix T where T@b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver, max_iter=None):
    """Solve LP: min M s.t. conv(a_fixed, b) <= M, sum(b)=1, b>=0. Returns (b, M)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    kwargs = {}
    if max_iter is not None:
        kwargs['max_iter'] = max_iter
    try:
        prob.solve(solver=solver, **kwargs)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, **kwargs)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """
    Nesterov-accelerated LP best-response with multi-scale tracking.
    Tracks: best iterate, short-window avg, long-window avg, and the current iterate.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()

    # Multiple averaging windows
    avg_short = np.zeros(n)
    n_short = 0
    avg_long = np.zeros(n)
    n_long = 0
    avg_med = np.zeros(n)
    n_med = 0

    # EMA
    ema_a = a.copy()
    ema_alpha = 0.03

    iter_count = 0
    stump = 0

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue
        iter_count += 1

        # Nesterov step size
        alpha_t = 2.0 / (t + 2.0)

        # Update y (dual-tracking point)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_new_s = y_new.sum()
        if y_new_s < 1e-15:
            y_new = y.copy()
            y_new_s = 1.0
        y_new = y_new / y_new_s

        # Update a (primal, with Nesterov momentum)
        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            a_s = a_new.sum()
            if a_s < 1e-15:
                a_new = y_new.copy()
                a_s = 1.0
            a_new = a_new / a_s

        y = y_new
        a = a_new
        iter_count += 0  # just for tracking

        # Update averages
        avg_short += a_new
        n_short += 1
        avg_long += a_new
        n_long += 1
        if t > 500:
            avg_med += a_new
            n_med += 1

        # EMA
        ema_a = (1 - ema_alpha) * ema_a + ema_alpha * a_new

        # Score check (a_new)
        sc = score_of(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            stump = 0
        else:
            stump += 1

        # Periodic score checks on averages
        if t % 5 == 0:
            sc_short = score_of(avg_short / n_short)
            if sc_short < best_score:
                best_score = sc_short
                best_a = (avg_short / n_short).copy()
                stump = 0

        if t % 10 == 0:
            sc_med = score_of(avg_med / max(1, n_med))
            if sc_med < best_score:
                best_score = sc_med
                best_a = (avg_med / max(1, n_med)).copy()
                stump = 0

        if t % 20 == 0:
            sc_long = score_of(avg_long / n_long)
            if sc_long < best_score:
                best_score = sc_long
                best_a = (avg_long / n_long).copy()
                stump = 0

            sc_ema = score_of(ema_a)
            if sc_ema < best_score:
                best_score = sc_ema
                best_a = ema_a.copy()
                stump = 0

        if verbose and t % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): M={M_val:.8f}, best={best_score:.7f}, stump={stump}")

        # Adaptive: if stalled for a long time, boost EMA
        if stump > 500:
            ema_alpha = min(0.1, ema_alpha * 1.02)

        # Early termination if severely stalled
        if stump > 5000 and t > 1000:
            if verbose:
                print(f"      Early stop at t={t}, stump={stump}")
            break

    return best_a, best_score, iter_count


def subgradient_polish(a0, time_limit, verbose=True):
    """
    Subgradient descent on the simplex with adaptive step size.
    Uses top-K peaks for subgradient computation.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()

    lr = 0.05
    n_top = 5
    patience = 0
    max_patience = 5000

    for it in range(200000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        max_b = b.max()
        cur_score = 2.0 * n * max_b

        # Find top peaks
        top_peaks = np.argsort(b)[::-1][:n_top]
        # Weight by proximity to max
        weights = np.exp((b[top_peaks] - max_b) * 20.0)
        weights = weights / weights.sum()

        # Compute subgradient
        grad = np.zeros(n)
        for w, k in zip(weights, top_peaks):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            other = k - idx
            grad[idx] += 2 * w * a[other]
            grad[other] += 2 * w * a[idx]

        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Step with backtracking
        improved = False
        for attempt in range(10):
            step = lr / (1.0 + 0.1 * attempt)
            a_trial = a - step * grad
            a_trial = np.maximum(a_trial, 0)
            a_trial_s = a_trial.sum()
            if a_trial_s < 1e-15:
                continue
            a_trial = a_trial / a_trial_s
            sc = score_of(a_trial)
            if sc < best_score:
                best_score = sc
                best_a = a_trial.copy()
                a = a_trial
                improved = True
                patience = 0
                break
            elif sc < cur_score * 1.001:
                a = a_trial
                # Accept even if not improving global best (for progress)
                if sc < cur_score:
                    patience = 0
                else:
                    patience += 1
                break
        else:
            patience += 1
            # Reduce learning rate
            lr *= 0.95

        # Decay learning rate gradually
        lr *= 0.9999

        if patience > max_patience:
            break

        if verbose and it % 2000 == 0:
            print(f"      SG it={it}: best={best_score:.7f}, cur={cur_score:.7f}, lr={lr:.6f}")

    return best_a, best_score


def coordinate_polish(a0, time_limit, seed=42, verbose=True):
    """
    Coordinate descent: for each coordinate, find the optimal value
    that minimizes the max convolution, given other coordinates fixed.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()

    def try_coordinate(a, i, best_score):
        """Try changing coordinate i to reduce the max."""
        # Save original value
        ai = a[i]
        if ai < 1e-15:
            return a, best_score, False

        # Compute current convolution
        b = fast_autocorr(a)
        max_b = b.max()

        # The change in conv[k] when a[i] changes by delta:
        # conv[k] changes by delta * (a[k-i] + a[i-k]) + delta^2 (if k=2i)
        # We want to find delta that minimizes max_k (conv[k] + delta * c[k] + delta^2 * d[k])
        # For simplicity, treat as: minimize max_k (conv[k] + delta * c[k])

        c = np.zeros(2 * n - 1)
        for k in range(2 * n - 1):
            # contribution of a[i] to conv[k]
            if 0 <= k - i < n:
                c[k] += a[k - i]
            if 0 <= i - k < n:
                c[k] += a[i - k]
        # Note: c[i] includes 2*a[i] from the two terms (unless k-i = i-k, i.e., k=2i)
        # Actually for k = 2i: c[2i] = a[i] + a[i] = 2*a[i] (if 0<=i<n which is true)
        # But the actual change is delta*c[k] + delta^2 (for k=2i where both terms give a[i])
        # Let's be more careful:
        # conv_new[k] = conv[k] + delta*(a[k-i] if 0<=k-i<n else 0) + delta*(a[i-k] if 0<=i-k<n else 0)
        # For k=2i: both terms give a[i], so change = delta*2*a[i]... but wait
        # Actually conv[k] = sum_j a[j]*a[k-j]. The terms involving a[i] are:
        # j=i: a[i]*a[k-i] (if 0<=k-i<n)
        # j=k-i: a[k-i]*a[i] (if 0<=k-i<n and k-i != i)
        # If k-i = i (k=2i): only one term a[i]*a[i], change = delta*a[i] + a[i]*delta = 2*delta*a[i]
        #   plus delta^2
        # If k-i != i: two terms, change = delta*a[k-i] + a[k-i]*delta = 2*delta*a[k-i]
        # Wait, I'm confusing myself. Let me just directly compute.

        # When we change a[i] by delta:
        # conv_new[k] = conv[k] + delta * (a[k-i] if valid) + delta * (a[i-k] if valid)
        #              + delta^2 * (1 if k-i == i and 0<=k-i<n, else 0)
        # But actually, if we change a[i] to a[i]+delta:
        # Old contribution to conv[k]: a[i]*a[k-i] + a[k-i]*a[i] (if i != k-i)
        #                              or a[i]^2 (if i == k-i)
        # New contribution: (a[i]+delta)*a[k-i] + a[k-i]*(a[i]+delta) (if i != k-i)
        #                    or (a[i]+delta)^2 (if i == k-i)
        # Change (if i != k-i): 2*delta*a[k-i]
        # Change (if i == k-i, i.e., k=2i): 2*delta*a[i] + delta^2

        # So the linear coefficient c[k] is:
        # c[k] = a[k-i] (if 0<=k-i<n and k-i != i) + a[i-k] (if 0<=i-k<n and i-k != i)
        # For k=2i: c[2i] = 2*a[i] (and there's also a delta^2 term)

        # Let me just compute c properly:
        c[:] = 0
        # Term 1: a[i] * a[k-i], changes to (a[i]+delta) * a[k-i], change = delta * a[k-i]
        k1 = i  # k such that k-i = i... no
        # conv[k] includes a[j]*a[k-j] for all valid j
        # The index j=i gives: a[i]*a[k-i], valid if 0<=k-i<n
        # The index j=k-i gives: a[k-i]*a[i], valid if 0<=k-i<n and k-i != i
        # So change in conv[k]:
        #   delta * a[k-i] (from j=i term, if 0<=k-i<n)
        #   + delta * a[i] (from j=k-i term, if 0<=k-i<n and k-i != i)
        #   + delta^2 (if k-i == i, i.e., k=2i, from the j=i term becoming (a[i]+delta)*a[i])

        # Simpler: just evaluate directly for a few candidate deltas
        best_delta = 0.0
        best_max_b = max_b

        # Try reducing a[i]
        for frac in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
            delta = -ai * frac
            a_trial = a.copy()
            a_trial[i] = a_trial[i] + delta
            if a_trial[i] < 0:
                continue
            a_trial_s = a_trial.sum()
            if a_trial_s < 0.01:
                continue
            b_trial = fast_autocorr(a_trial)
            mb = b_trial.max()
            if mb < best_max_b:
                best_max_b = mb
                best_delta = delta

        # Try increasing a[i] (by taking from other coordinates)
        # This is handled by the "move" operation below

        if best_delta != 0:
            a[i] += best_delta
            new_score = 2.0 * n * fast_autocorr(a).max() / (a.sum() ** 2)
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                return a, best_score, True

        return a, best_score, False

    # Main loop: iterate over coordinates in random order
    order = np.arange(n)
    rounds = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        np.random.default_rng(rounds * 1000 + seed).shuffle(order)
        improved = False
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            a, best_score, imp = try_coordinate(a, i, best_score)
            if imp:
                improved = True
                stall = 0

        if not improved:
            # Try random pairwise moves
            for _ in range(50):
                if time.monotonic() - t0 > time_limit:
                    break
                i = rng.integers(0, n)
                j = rng.integers(0, n)
                if i == j:
                    continue
                ai = a[i]
                if ai < 1e-10:
                    continue
                for frac in [0.01, 0.05, 0.1, 0.2]:
                    amt = ai * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial_s = a_trial.sum()
                    if a_trial_s < 0.01:
                        continue
                    sc = score_of(a_trial)
                    if sc < best_score:
                        best_score = sc
                        best_a = a_trial.copy()
                        a = a_trial
                        improved = True
                        break

        rounds += 1
        if not improved:
            stall += 1
            if stall > 20:
                break
        else:
            stall = 0

        if verbose and rounds % 5 == 0:
            print(f"      CD round={rounds}: best={best_score:.7f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    # === Phase 4: Multi-length exploration (20%) ===
    lengths_to_try = [n_base - 100, n_base + 100, n_base + 200, n_base - 200]
    lengths_to_try = [l for l in lengths_to_try if l >= 400]
    # Shuffle for diversity
    rng.shuffle(lengths_to_try)

    for n_try in lengths_to_try:
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining < 100:
            break
        print(f"\n=== Phase 4: Length n={n_try} (budget={remaining:.0f}s) ===")
        t4 = min(remaining * 0.15, 180.0)
        if t4 < 40:
            continue

        # Initialize by interpolation
        init_try = interpolate_sequence(best_seq, n_try)
        a_out, sc_out, iters = nesterov_lp_optimize(init_try, t4, solver, verbose=False)
        print(f"  LP at n={n_try}: {iters} iters, score={sc_out:.7f}")
        try_update(a_out, f"lp_n{n_try}")

        # Quick subgradient polish
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining > 50 and a_out is not None:
            t4b = min(remaining * 0.06, 60.0)
            if t4b > 15:
                a_out2, sc_out2 = subgradient_polish(a_out, t4b, verbose=False)
                try_update(a_out2, f"sg_n{n_try}")

    # === Phase 5: Second LP round on best (10%) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 5: LP Round 2 at n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.10, 150.0)
    if t5 > 30:
        a_out, sc_out, iters = nesterov_lp_optimize(best_seq, t5, solver, verbose=False)
        print(f"  LP round 2: {iters} iters, score={sc_out:.7f}")
        try_update(a_out, f"lp2_n{len(best_seq)}")

    # === Phase 6: Final coordinate + subgradient polish (5%) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 6: Final Polish (budget={remaining:.0f}s) ===")
    if remaining > 30:
        t6a = min(remaining * 0.5, 80.0)
        if t6a > 15:
            a_out, sc_out = subgradient_polish(best_seq, t6a, verbose=False)
            try_update(a_out, "final_sg")

        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining > 15:
            t6b = min(remaining * 0.7, 60.0)
            if t6b > 10:
                a_out, sc_out = coordinate_polish(best_seq, t6b, seed=789, verbose=False)
                try_update(a_out, "final_cd")

    # === Phase 7: Randomized restarts (remaining time) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 7: Randomized perturbations (budget={remaining:.0f}s) ===")
    if remaining > 60:
        for restart in range(5):
            remaining = time_budget - (time.monotonic() - t_global_start)
            if remaining < 30:
                break
            # Perturb the best sequence
            perturbation = rng.standard_normal(len(best_seq)) * 0.01
            a_perturbed = best_seq + perturbation
            a_perturbed = np.maximum(a_perturbed, 0)
            a_perturbed_s = a_perturbed.sum()
            if a_perturbed_s < 0.01:
                continue
            a_perturbed = a_perturbed / a_perturbed_s

            t7 = min(remaining * 0.15, 60.0)
            if t7 < 15:
                continue
            a_out, sc_out, _ = nesterov_lp_optimize(a_perturbed, t7, solver, verbose=False)
            try_update(a_out, f"restart_{restart}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.7f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
