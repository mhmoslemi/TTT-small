def nesterov_fw(a0, time_limit, fw_iters=80, lp_every=500, solver=None):
    n = len(a0)
    a = normalize_nonneg(a0)
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    t = 0
    while time.monotonic() - t0 < time_limit:
        t += 1
        
        # Compute best response
        if lp_every > 0 and t % lp_every == 0 and solver is not None:
            b_star, _ = solve_lp_exact(y, solver)
            if b_star is None:
                b_star = fw_best_response(y, fw_iters)
        else:
            b_star = fw_best_response(y, fw_iters)
        
        # Nesterov update
        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * b_star
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s > 1e-15:
            y_new = y_new / s
        
        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s > 1e-15:
                a_new = a_new / s
            else:
                a_new = y_new.copy()
        
        y = y_new
        a = a_new
        
        # Check improvement every 10 steps
        if t % 10 == 0 or t % 200 == 0:
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
            score_y = 2.0 * n * fast_autocorr(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y.copy()
        
        if t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"        NFW t={t} ({elapsed:.0f}s): best={best_score:.8f}")
    
    # Final check
    score_a = 2.0 * n * fast_autocorr(a).max()
    if score_a < best_score:
        best_score = score_a
        best_a = a.copy()
    score_y = 2.0 * n * fast_autocorr(y).max()
    if score_y < best_score:
        best_score = score_y
        best_a = y.copy()

def solve_lp_exact(y, solver):
    """Solve the exact LP: min M s.t. (y*b)[k] <= M, sum(b)=1, b>=0."""
    n = len(y)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build convolution matrix T where (T*b)[k] = (y*b)[k]
    # T[k, j] = y[k-j] if 0 <= k-j < n, else 0
    # This is a (2n-1) x n matrix
    T = cp.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je >= js:
            # T[k, js:je+1] = y[k-je : k-js+1][::-1]
            T[k, js:je+1] = y[k - je:k - js + 1][::-1]
    
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        return None, None
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def peak_shaving(a0, time_limit, seed=123):
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize_nonneg(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:10]
        
        improved = False
        for k in top_k:
            # Find the pair (i,j) with i+j=k (or i+j=2n-2-k) that contributes most
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            src = int(idx[np.argmax(contrib)])
            
            # Find a good destination: low a value, far from peak area
            peak_area = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                for j2 in range(js2, je2 + 1):
                    peak_area.add(j2)
            
            candidates = [j for j in range(n) if j not in peak_area and j != src]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            
            # Try moving a small fraction to a random candidate
            for _ in range(5):
                dst = int(rng.integers(len(candidates)))
                dst_idx = candidates[dst]
                for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst_idx] += amt
                    a_trial = normalize_nonneg(a_trial)
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j and a[i] > 1e-12:
                frac = float(rng.uniform(0.003, 0.02))
                amt = a[i] * frac
                a_trial = a.copy()

                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()

            if stall > 500:
                break
