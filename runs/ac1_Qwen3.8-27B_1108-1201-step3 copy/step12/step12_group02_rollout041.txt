"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Ramp-based initialization (key insight: monotone sequences minimize center autocorrelation)
2. FFT-based LogSumExp optimization with multi-stage annealing (L-BFGS in softmax space)
3. Mirror descent with adaptive peak-weighted subgradient
4. Peak flattening via coordinate descent
5. Multiple structural initializations (ramps, power ramps, modified shapes)
Key insight: pure linear ramp gives score ~4/3 for large n, far below previous 1.504.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_autocorr(a):
    """Full linear autocorrelation (convolution of a with itself) using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_score(a):
    """Compute 2n*max(b)/(sum(a))^2 quickly."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)


def lse_value_and_w(a, T):
    """Compute log-sum-exp value and softmax weights w for T*b."""
    b = fast_autocorr(a)
    Tb = T * b
    Tb_shift = Tb - Tb.max()
    exp_Tb = np.exp(Tb_shift)
    val = (Tb.max() + np.log(exp_Tb.sum())) / T
    w = exp_Tb / exp_Tb.sum()
    return val, w


def lse_gradient(a, T):
    """
    Gradient of (1/T)*log(sum(exp(T*b[k]))) w.r.t. a.
    grad[i] = 2 * sum_j a[j] * w[i+j]
    Computed via FFT-based convolution.
    """
    n = len(a)
    _, w = lse_value_and_w(a, T)

    a_pad = np.zeros(2 * n - 1)
    a_pad[:n] = a
    w_rev = w[::-1]

    conv_len = (2 * n - 1) * 2 - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    Ap = rfft(a_pad, fft_size)
    Wr = rfft(w_rev, fft_size)
    conv_result = irfft(Ap * Wr, fft_size)[:conv_len]

    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * conv_result[idx]
    return grad


def lse_objective_z(z, T):
    """Objective in z-space where a = softmax(z)."""
    z_shift = z - z.max()
    e = np.exp(z_shift)
    a = e / e.sum()
    val, _ = lse_value_and_w(a, T)
    return val


def lse_grad_z(z, T):
    """Gradient in z-space where a = softmax(z)."""
    z_shift = z - z.max()
    e = np.exp(z_shift)
    a = e / e.sum()
    grad_a = lse_gradient(a, T)
    # grad_z = a * (grad_a - dot(grad_a, a))
    S = np.dot(grad_a, a)
    return a * (grad_a - S)


def lse_lbfgs_optimize(a0, time_limit, T_start=200.0, T_end=0.3, n_stages=15, seed=42):
    """Multi-stage log-sum-exp optimization with L-BFGS in softmax space."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    s = a_init.sum()
    a_init = a_init / s
    z = np.log(a_init)
    t0 = time.monotonic()

    best_a = a_init.copy()
    best_score = fast_score(a_init)

    temps = np.geomspace(T_start, T_end, n_stages)
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        iters = max(20, min(100, int(20 + 80 * (si / max(1, n_stages - 1)))))

        def func(z_in):
            return lse_objective_z(z_in, stage_T)

        def jac(z_in):
            return lse_grad_z(z_in, stage_T)

        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': iters, 'ftol': 1e-18,
                                    'gtol': 1e-12, 'maxfun': iters * 300,
                                    'disp': False})
            z = res.x
            z_shift = z - z.max()
            a = np.exp(z_shift)
            a = a / a.sum()
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        except Exception:
            pass

        if si % 4 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage {si+1}/{n_stages} (T={stage_T:.1f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def mirror_descent_peaks(a0, time_limit, eta0=0.03, decay_rate=0.6,
                         k_min=1, k_max=20, k_ramp=500, seed=42):
    """
    Mirror descent (multiplicative weights) with adaptive top-k Gaussian-weighted subgradient.
    O(n log n) per iteration via FFT.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall_count = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        b_max = b.max()
        b_mean = b.sum() / len(b)

        frac = min(1.0, t / float(k_ramp))
        k = max(k_min, min(k_max, int(k_min + frac * (k_max - k_min))))

        top_idx = np.argsort(b)[::-1][:k]
        excess = b[top_idx] - b_mean
        if excess.max() < 1e-15:
            weights = np.ones(k) / k
        else:
            excess_shift = excess - excess.max()
            weights = np.exp(excess_shift) / np.exp(excess_shift).sum()

        grad = np.zeros(n)
        for ii in range(k):
            m = int(top_idx[ii])
            w_m = weights[ii] * 2.0
            j_start = max(0, m - n + 1)
            j_end = min(n - 1, m)
            js = np.arange(j_start, j_end + 1)
            grad[js] += w_m * a[m - js]

        eta = eta0 / (1.0 + t / 400.0) ** decay_rate
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a + 0.01 * rng.standard_normal(n)
            a_new = np.maximum(a_new, 0)
            a_new = a_new / max(a_new.sum(), 1e-15)
        else:
            a_new = a_new / s_new

        a = a_new

        if t < 200:
            check_interval = 2
        elif t < 2000:
            check_interval = 5
        else:
            check_interval = 15

        if t % check_interval == 0:
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stall_count = 0
            else:
                stall_count += 1

        if stall_count > 300:
            noise_scale = min(0.03, 0.005 * (stall_count / 50.0))
            a = a + noise_scale * rng.standard_normal(n)
            a = np.maximum(a, 0)
            a = a / a.sum()
            stall_count = 0

        if t % 5000 == 0 and t > 0:
            elapsed = time.monotonic() - t0
            print(f"    MD t={t} ({elapsed:.0f}s): best={best_score:.6f}, bmax={b_max:.6e}")

    return best_a, best_score


def peak_flatten(a0, time_limit, seed=123):
    """
    Coordinate descent: iteratively reduce the highest autocorrelation peak
    by redistributing mass from contributing positions.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        b_max_idx = int(np.argmax(b))
        top_k = np.argsort(b)[::-1][:8]

        improved = False
        for k_idx in top_k:
            k_idx = int(k_idx)
            js = max(0, k_idx - n + 1)
            je = min(n - 1, k_idx)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_idx - idx]

            if contrib.sum() < 1e-15:
                continue

            src = int(idx[np.argmax(contrib)])

            # Find low-contribution target positions
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k[:4]:
                k2 = int(k2)
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) < 5:
                candidates = np.where(np.arange(n) != src)[0]
            sorted_cands = candidates[np.argsort(a[candidates])[:max(3, len(candidates) // 5)]]
            if len(sorted_cands) == 0:
                continue
            dst = int(sorted_cands[rng.integers(len(sorted_cands))])

            for frac in [0.001, 0.003, 0.008, 0.015, 0.03, 0.06]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                sc = fast_score(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial.copy()
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break

            if improved:
                break

        if not improved:
            stall += 1
            if rng.integers(4) == 0 and stall < 200:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.015)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= 0:
                        a_trial = a_trial / a_trial.sum()
                        sc = fast_score(a_trial)
                        if sc < best_score:
                            a = a_trial.copy()
                            best_score = sc
                            best_a = a.copy()
                            stall = 0
            if stall > 300:
                break

    return best_a, best_score


def ramp_init(n, power=1.0):
    """Generate a power-ramp initialization: a[i] = ((i+1)/n)^power."""
    x = (np.arange(n) + 1) / n
    a = x ** power
    a = a / a.sum()
    return a


def modified_ramp_init(n, mode=0):
    """Generate modified ramp initializations."""
    x = np.arange(n, dtype=np.float64) / n
    if mode == 0:
        # Pure linear ramp
        a = x + 0.01
    elif mode == 1:
        # Quadratic ramp
        a = x ** 2 + 0.01
    elif mode == 2:
        # Cube root ramp (more mass at start)
        a = x ** (1.0 / 3.0) + 0.01
    elif mode == 3:
        # Ramp with slight concavity
        a = x + 0.3 * x * (1 - x)
    elif mode == 4:
        # Ramp with slight convexity
        a = x - 0.2 * x * (1 - x) + 0.01
    elif mode == 5:
        # Log-ramp
        a = np.log(x + 0.1) + 1.0
    elif mode == 6:
        # Exponential ramp
        a = np.exp(2.0 * x)
    elif mode == 7:
        # Ramp with quadratic correction
        a = x + 0.1 * (x - 0.5) ** 2
    elif mode == 8:
        # Sqrt ramp
        a = np.sqrt(x) + 0.01
    else:
        a = x + 0.01

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

        n_long = min(int(n_parent * 1.3), 2400)
        # Create long ramp
        a_long = ramp_init(n_long, power=1.0)
        sc_long = fast_score(a_long)
        print(f"  Long ramp n={n_long}: score={sc_long:.6f}")

        t_md_long = min(remaining * 0.2, 200.0)
        if t_md_long > 15:
            a_out, _ = mirror_descent_peaks(a_long, t_md_long, eta0=0.02, decay_rate=0.5,
                                            k_min=1, k_max=12, k_ramp=300, seed=55)
            try_update(a_out, f"md_n{n_long}")

        remaining = time_budget - (time.monotonic() - t_global)
        t_lse_long = min(remaining * 0.15, 150.0)
        if t_lse_long > 15 and (best_seq is None or best_score > 1.35):
            a_out, _ = lse_lbfgs_optimize(a_long, t_lse_long, T_start=100.0, T_end=0.2,
                                          n_stages=10, seed=66)
            try_update(a_out, f"lse_n{n_long}")

    # === Phase 7: Final polish ===
    print("\n=== Phase 7: Final Polish ===")
    remaining = time_budget - (time.monotonic() - t_global)
    if best_seq is not None and remaining > 10:
        # Short MD
        t_f1 = min(remaining * 0.3, 100.0)
        if t_f1 > 5:
            a_out, _ = mirror_descent_peaks(best_seq, t_f1, eta0=0.01, decay_rate=0.7,
                                            k_min=3, k_max=20, k_ramp=100, seed=77)
            try_update(a_out, "final_md")

        # Peak flatten
        remaining = time_budget - (time.monotonic() - t_global)
        t_f2 = min(remaining * 0.4, 80.0)
        if t_f2 > 5:
            a_out, _ = peak_flatten(best_seq, t_f2, seed=222)
            try_update(a_out, "final_pf")

        # Very low temp LSE
        remaining = time_budget - (time.monotonic() - t_global)
        t_f3 = min(remaining * 0.5, 60.0)
        if t_f3 > 5:
            a_out, _ = lse_lbfgs_optimize(best_seq, t_f3, T_start=20.0, T_end=0.05,
                                          n_stages=8, seed=88)
            try_update(a_out, "final_lse")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            if fs < best_score:
                best_score = fs
            print(f"\nFINAL (official): score={fs:.6f}")
        print(f"FINAL (fast): score={best_score:.6f}, n={len(best_seq)}")
    else:
        print("ERROR: No valid sequence found! Using fallback.")
        # Fallback: use ramp
        best_seq = ramp_init(1600, power=1.0)
        best_score = fast_score(best_seq)
        print(f"FALLBACK: score={best_score:.6f}")

    return [float(x) for x in best_seq]
