"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated LogSumExp smooth approximation with L-BFGS
2. Continuation method (increasing temperature)
3. Direct subgradient steps on the exact max
4. Adaptive restarts with perturbation
5. Multi-length exploration
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Fast linear convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        s = sum(a_list)
        if s < 0.01:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def score_normalized(a):
    """Score assuming sum(a)=1: returns 2*n*max(b)."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def softmax_stable(z):
    """Stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def lse_obj_grad(z, T, n):
    """
    Compute LSE objective and gradient w.r.t. z.
    a = softmax(z), R = conv(a,a)
    LSE = (1/T) * log(sum exp(T * R))
    Returns (2*n*LSE, d(2*n*LSE)/dz)
    """
    a = softmax_stable(z)
    R = fast_convolve(a)  # length 2n-1
    
    # LSE with temperature T
    T_R = T * R
    T_R_max = T_R.max()
    exp_T_R = np.exp(T_R - T_R_max)
    sum_exp = exp_T_R.sum()
    LSE = (T_R_max + np.log(sum_exp)) / T
    
    # Softmax weights
    w = exp_T_R / sum_exp  # length 2n-1
    
    # Gradient w.r.t. a: dLSE/da[i] = 2 * sum_j a[j] * w[i+j]
    # This is 2 * cross_correlation(a, w)
    # Use FFT for speed
    fft_size = 1 << (3 * n - 2 - 1).bit_length()
    A = rfft(a, fft_size)
    w_rev = w[::-1]  # length 2n-1
    W = rfft(w_rev, fft_size)
    conv_full = irfft(A * W, fft_size)[:3 * n - 2]
    # grad_a[i] = conv_full[2n-2-i] for i=0..n-1
    grad_a = 2.0 * conv_full[n - 1:2 * n - 1][::-1]
    
    # Convert to gradient w.r.t. z: dz[i] = a[i] * (ga[i] - sum(a*ga))
    S = np.dot(a, grad_a)
    grad_z = a * (grad_a - S)
    
    obj = 2.0 * n * LSE
    grad_z *= 2.0 * n
    
    return obj, grad_z


def run_lbfgs(a0, T, time_limit, max_iter=100):
    """Run L-BFGS on the LSE objective at fixed temperature T."""
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300))
    
    t0 = time.monotonic()
    
    def func(z_val):
        obj, _ = lse_obj_grad(z_val, T, n)
        return obj
    
    def jac(z_val):
        _, grad = lse_obj_grad(z_val, T, n)
        return grad
    
    try:
        res = minimize(
            fun=func, x0=z, jac=jac, method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': 1e-15,
                     'gtol': 1e-10, 'maxfun': max_iter * 200,
                     'disp': False}
        )
        z_out = res.x
    except Exception:
        z_out = z
    
    a_out = softmax_stable(z_out)
    s = a_out.sum()
    if s > 1e-15:
        a_out = a_out / s
    
    return a_out


def subgradient_exact(a0, n_steps=200, time_limit=None):
    """Subgradient method on the exact max function."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_score = score_normalized(a)
    
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time_limit is not None and time.monotonic() - t0 > time_limit:
            break
        
        R = fast_convolve(a)
        max_R = R.max()
        k_star = int(np.argmax(R))
        
        # Subgradient: dR[k*]/da[i] = 2*a[k*-i] if 0<=k*-i<n else 0
        g = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        for i in range(lo, hi + 1):
            g[i] = 2.0 * a[k_star - i]
        
        # Project onto tangent space
        g_mean = g.mean()
        g_tan = g - g_mean
        
        # Polyak-like step size
        # Lower bound on optimal value (conservative)
        lower_bound = 1.0  # Very conservative
        gap = 2.0 * n * max_R - lower_bound
        grad_norm_sq = np.dot(g_tan, g_tan)
        if grad_norm_sq < 1e-30:
            break
        eta = gap / (2.0 * n * n * grad_norm_sq)
        eta = min(eta, 1.0 / n)
        
        # Take step and project onto simplex
        a_new = a - eta * g_tan
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s_new
        
        sc_new = score_normalized(a_new)
        
        # Backtracking if no improvement
        if sc_new > 2.0 * n * max_R * 1.01:
            # Try smaller step
            for _ in range(10):
                eta *= 0.5
                a_new = a - eta * g_tan
                a_new = np.maximum(a_new, 0)
                s_new = a_new.sum()
                if s_new > 1e-15:
                    a_new = a_new / s_new
                sc_new = score_normalized(a_new)
                if sc_new <= 2.0 * n * max_R:
                    break
        
        a = a_new
        if sc_new < best_score:
            best_score = sc_new
            best_a = a.copy()
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = np.interp(new_x, old_x, f_norm)
        f_interp = np.maximum(f_interp, 0)
        ss = f_interp.sum()
        if ss > 1e-15:
            f_interp = f_interp / ss
    return f_interp

    """
