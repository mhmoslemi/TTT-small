        a_new = np.maximum(a_new, 0)
        a_new_s = a_new.sum()
        if a_new_s < 1e-15:
            break
        a_new = a_new / a_new_s

        sc = score_of(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()

        a = a_new
        lr *= lr_decay

        if it % 500 == 0:
            print(f"    SGD it={it}: best={best_score:.7f}, cur={sc:.7f}, T={temperature:.3f}")

        # Anneal temperature
        if it % 1000 == 0:
            temperature *= 0.9

    return best_a, best_score


def coordinate_descent(a0, time_limit, seed=42):
    """Coordinate descent: move mass from peak-contributing positions to low-impact positions."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = score_of(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(a)
        max_c = c.max()
        top_peaks = np.argsort(c)[::-1][:min(5, 2 * n - 1)]

        improved = False
        for k in top_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            top_contrib = np.argsort(contrib)[::-1][:3]

            for ci in top_contrib:
                src = idx[ci]
                # Try moving mass from src to random positions
                for trial in range(20):
                    dst = rng.integers(0, n)
                    if dst == src:
                        continue
                    for frac in [0.001, 0.003, 0.01, 0.03]:
                        amt = a[src] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial.min() < -1e-12:
                            continue
                        a_trial = np.maximum(a_trial, 0)
                        ts = a_trial.sum()
                        if ts < 1e-15:
                            continue
                        a_trial = a_trial / ts
                        sc = score_of(a_trial)
                        if sc < best_score - 1e-15:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
            if improved:
                break
            if contrib[top_contrib[0]] < 1e-15:
                continue

        if not improved:
            stall += 1
            if stall % 5 == 0:
                # Random perturbation
                n_p = rng.integers(2, 6)
                srcs = rng.choice(n, size=n_p, replace=False)
                dsts = rng.choice(n, size=n_p, replace=False)
                a_trial = a.copy()
                for sr, dt in zip(srcs, dsts):
                    amt = a_trial[sr] * rng.uniform(0.005, 0.03)
                    a_trial[sr] -= amt
                    a_trial[dt] += amt
                a_trial = np.maximum(a_trial, 0)
                ts = a_trial.sum()
                if ts > 1e-15:
                    a_trial = a_trial / ts
                    sc = score_of(a_trial)
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        stall = 0
            if stall > 200:
                break

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Resample a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def spectral_init(n, peakiness=1.0):
    """
    Generate initial sequence by shaping the spectrum.
    Target: flat |X|^2 to minimize autocorrelation peak.
    """
    # Start with a shape that has a broad spectrum
    # Use a raised cosine or sinc-like shape
    x = np.linspace(-1, 1, n)
    a = np.exp(-peakiness * x ** 2)  # Gaussian
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    return a


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Uses LP Nesterov, smoothed gradient descent,
    and coordinate descent to minimize the evaluation score.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = find_good_solver()
    print(f"Solver: {solver}, Budget: {time_budget:.0f}s")

    # Get starting sequence
    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p.copy()
            best_score = score_of(p)
            print(f"Parent: n={len(p)}, score={best_score:.7f}")
    except Exception:
        pass

    if best_seq is None:
        n_base = 1600
        best_seq = np.ones(n_base) / n_base
        best_score = score_of(best_seq)
        print(f"Default uniform init: n={n_base}, score={best_score:.7f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        sc = score_of(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            el = time.monotonic() - t_start
            print(f"  *** NEW BEST [{label}]: {sc:.7f} (n={len(a)}, t={el:.0f}s)")

    n_base = len(best_seq)

    # Phase 1: LP Nesterov (main workhorse)
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 1: LP Nesterov n={n_base}, time={remaining:.0f}s ===")
    t1 = min(remaining * 0.40, 600.0)
    if t1 > 30:
        a_out, sc_out, iters = nesterov_lp(best_seq, t1, solver, verbose=True)
        print(f"  NLP done: {iters} iters, score={sc_out:.7f}")
        try_update(a_out, f"nlp_n{n_base}")

    # Phase 2: Smoothed gradient descent
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 2: Smoothed GD n={len(best_seq)}, time={remaining:.0f}s ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 20:
        a_out, sc_out = smooth_gradient_descent(best_seq, t2, lr=0.02, temperature=0.5)
        print(f"  SGD done: score={sc_out:.7f}")
        try_update(a_out, f"sgd_n{len(best_seq)}")

    # Phase 3: Coordinate descent
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 3: Coord Descent n={len(best_seq)}, time={remaining:.0f}s ===")
    t3 = min(remaining * 0.20, 250.0)
    if t3 > 15:
        a_out, sc_out = coordinate_descent(best_seq, t3, seed=123)
        print(f"  CD done: score={sc_out:.7f}")
        try_update(a_out, f"cd_n{len(best_seq)}")

    # Phase 4: LP Nesterov round 2 (exploit improvements)
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 4: LP Nesterov round 2, time={remaining:.0f}s ===")
    t4 = min(remaining * 0.25, 300.0)
    if t4 > 30:
        a_out, sc_out, iters = nesterov_lp(best_seq, t4, solver, verbose=False)
        print(f"  NLP2 done: {iters} iters, score={sc_out:.7f}")
        try_update(a_out, "nlp2")

    # Phase 5: Try different lengths
    for n_try in [n_base + 100, n_base - 100, n_base + 200]:
        if n_try < 200:
            continue
        remaining = time_budget - (time.monotonic() - t_start)
        if remaining < 100:
            break
        print(f"\n=== Phase 5: LP at n={n_try}, time={remaining:.0f}s ===")
        t5 = min(remaining * 0.15, 150.0)
        if t5 < 20:
            continue
        init_try = interpolate_seq(best_seq, n_try)
        a_out, sc_out, iters = nesterov_lp(init_try, t5, solver, verbose=False)
        print(f"  NLP n={n_try}: {iters} iters, score={sc_out:.7f}")
        try_update(a_out, f"nlp_n{n_try}")

    # Phase 6: Final polishing
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 6: Final polish, time={remaining:.0f}s ===")
    if remaining > 30:
        t6a = min(remaining * 0.3, 60.0)
        if t6a > 10:
            a_out, sc_out = coordinate_descent(best_seq, t6a, seed=999)
            try_update(a_out, "cd_final")

        remaining = time_budget - (time.monotonic() - t_start)
        if remaining > 60:
            t6b = min(remaining * 0.5, 120.0)
            if t6b > 20:
                a_out, sc_out, iters = nesterov_lp(best_seq, t6b, solver, verbose=False)
                try_update(a_out, "nlp_final")

    # Final check
    if best_seq is not None:
        final_score = score_of(best_seq)
        print(f"\nFINAL: score={final_score:.7f}, n={len(best_seq)}")
        return [float(x) for x in best_seq]
    else:
        return [1.0]
