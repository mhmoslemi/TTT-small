import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time

def fast_autocorr(a):
    """Full self-convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_lse_gradient(a, T):
    """Compute gradient of LSE(T * (a*a)) w.r.t. a using FFT."""
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    
    # grad[j] = 2 * sum_k w[k] * a[k-j] for valid k,j
    # = 2 * (w * a_rev)[n-1 + j]
    a_rev = a[::-1]
    conv_len = len(w) + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(w, fft_size)
    A_rev = rfft(a_rev, fft_size)
    conv = irfft(W * A_rev, fft_size)
    grad = 2.0 * conv[n - 1:n - 1 + n]
    return grad, w

def lse_objective_and_gradient(a, T):
    """Compute LSE objective and gradient."""
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    obj = (max_Tb + np.log(exp_Tb.sum())) / T
    
    grad, w = compute_lse_gradient(a, T)
    return obj, grad

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-stage LogSumExp optimization with L-BFGS-B (primary workhorse)
2. Nesterov-accelerated LP best-response (refinement)
3. Gradient-based peak chasing (targeted peak reduction)
4. Final coordinate descent polish
Key improvements: slower LogSumExp cooling, direct peak-targeting,
and more focused optimization at the parent length.

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_autocorr(a):
    """Full self-convolution a*a using FFT, O(n log n)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_lse_grad(a, T):
    """Compute gradient of LSE(T*(a*a)) w.r.t. a using FFT."""
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    a_rev = a[::-1]
    conv_len = len(w) + n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(w, fft_size)
    A_rev = rfft(a_rev, fft_size)
    conv = irfft(W * A_rev, fft_size)
    grad = 2.0 * conv[n - 1:n - 1 + n]
    return grad, w


def lse_fun_jac(z, T):
    """Objective and gradient for z-parameterization (a = softmax(z))."""
    a = softmax_fn(z)
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    obj_val = (max_Tb + np.log(exp_Tb.sum())) / T
    grad_a, w = compute_lse_grad(a, T)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj_val, grad_z


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def logsumexp_lbfgs(a0, time_limit, T_start=200.0, T_end=0.005, n_stages=40):
    """

def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where (T@b) = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL, cp.MOSEK, cp.GLPK]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver):
    """
    Solve LP: min M s.t. conv(a_fixed, b) <= M, sum(b)=1, b>=0.
    Returns (b, M).
    """
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    try:
        prob.solve(solver=solver)
    except Exception:
        pass

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL, cp.MOSEK, cp.GLPK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    M_val = float(M_var.value)
    return b_val, M_val

def subgradient_max_optimize(a0, time_limit, n_iters=50000):

        # Top K peaks

        a_rev = a[::-1]

        fft_size = 1 << (conv_len - 1).bit_length()

        # For each peak position k, the gradient of (a*a)[k] w.r.t. a is:
        # grad[j] = 2*a[k-j] (if 0<=k-j<n and k!=2j) or a[k-j] (if k==2j)

        # More precisely, sum over k in top_k: indicator vectors

        # Build the averaged "weight" vector w where w[k] = 1/K for top_k
        w = np.zeros(len(b))
        for k in top_k:
            w[k] = 1.0 / K

        # grad = 2 * conv(w, a_rev)[n-1:n-1+n]
        W = rfft(w, fft_size)
        A_rev = rfft(a_rev, fft_size)
        conv = irfft(W * A_rev, fft_size)

        grad_proj = grad - np.mean(grad)

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            n_base = len(parent)
            p_score = evaluate_safe(parent)
            print(f"Parent: n={n_base}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        print("No valid parent, using default initialization")
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: LogSumExp with L-BFGS ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: LogSumExp L-BFGS at n={n_base} (remaining={remaining:.0f}s) ===")
    t1 = remaining * 0.35
    if t1 > 30:
        a_out, _ = logsumexp_lbfgs(best_seq, t1, T_start=200.0, T_end=0.005, n_stages=45)
        try_update(a_out, "lse1")

    # === Phase 2: Nesterov LP ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: Nesterov LP at n={n_base} (remaining={remaining:.0f}s) ===")
    t2 = remaining * 0.25
    if t2 > 30:
        a_out, _ = nesterov_lp_optimize(best_seq, t2, solver)
        try_update(a_out, "nlp")

    # === Phase 3: Subgradient max optimization ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: Subgradient optimization (remaining={remaining:.0f}s) ===")
    t3 = remaining * 0.15
    if t3 > 10:
        a_out, _ = subgradient_max_optimize(best_seq, t3, n_iters=100000)
        try_update(a_out, "sg")

    # === Phase 4: Peak chasing ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: Peak chasing (remaining={remaining:.0f}s) ===")
    t4 = remaining * 0.15
    if t4 > 10:
        a_out, _ = peak_chasing(best_seq, t4, seed=999)
        try_update(a_out, "pc1")

    # Second round of LogSumExp (finer tuning)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4b: LogSumExp refinement (remaining={remaining:.0f}s) ===")
    t4b = remaining * 0.15
    if t4b > 10:
        a_out, _ = logsumexp_lbfgs(best_seq, t4b, T_start=50.0, T_end=0.002, n_stages=30)
        try_update(a_out, "lse2")

    # === Phase 5: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 5: Final polish (remaining={remaining:.0f}s) ===")
    t5 = remaining * 0.4
    if t5 > 5:
        a_out, _ = peak_chasing(best_seq, t5, seed=123)
        try_update(a_out, "pc2")

    remaining = time_budget - (time.monotonic() - t_global_start)
    t5b = remaining * 0.4
    if t5b > 5:
        a_out, _ = coordinate_descent_final(best_seq, t5b)
        try_update(a_out, "cd")

    # Final LogSumExp with very low T
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 20:
        t5c = remaining * 0.5
        a_out, _ = logsumexp_lbfgs(best_seq, t5c, T_start=10.0, T_end=0.001, n_stages=15)
        try_update(a_out, "lse_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    # Ensure output is valid
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return result
