def solve_lp_fast(a_fixed, active_lags=None, solver='highs'):
    n = len(a_fixed)
    if active_lags is None:
        active_lags = np.arange(2 * n - 1)
    
    K = len(active_lags)
    
    # c = [0, ..., 0, 1] (minimize M)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # A_ub: K rows of [T_a_row, -1]
    A_ub = np.zeros((K, n + 1))
    for i, k in enumerate(active_lags):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        other = k - idx
        A_ub[i, other] = a_fixed[idx]
        A_ub[i, n] = -1.0
    b_ub = np.zeros(K)
    
    # A_eq: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: b >= 0, M free (or >= 0)
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')

    if s < 1e-15:

    b_val = b_val / s

For the active set approach, we only need K rows of T_a, so we can compute them on the fly without building the full matrix.

OK let me just write the code. I'll focus on:
1. Fast LP using scipy.optimize.linprog with HiGHS
2. Active set management
3. Nesterov acceleration with averaging
4. Multiple phases with time management
5. Polishing at the end

Let me also add a "dual tracking" approach: maintain the dual variables (the optimal M values and the LP dual) to get a certificate of optimality and know when to stop.

Actually, for the Nesterov LP method, we can track both the primal and dual values:
- Primal value: max_k (a^t * a^t)_k (using the running average of a)
- Dual value: the optimal M from the LP (or an average of M values)

The gap between primal and dual values gives a certificate of how close we are to optimal.

Let me write the final code:

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast active-set LP Nesterov with HiGHS solver
2. Dual-gap tracking for convergence detection
3. Averaging-based solution extraction
4. QP polishing on active constraint set
5. Multi-length exploration
6. Coordinate descent final refinement
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import linprog, minimize


def fast_autocorr(a):
    """Full autocorrelation a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_corr(a, b):
    """Full cross-correlation a*b using FFT."""
    n = max(len(a), len(b))
    size = len(a) + len(b) - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def solve_lp_active(a_fixed, active_lags):
    """
    Solve LP: min M s.t. (a_fixed * b)_k <= M for k in active_lags,
                sum(b) = 1, b >= 0.
    Uses scipy linprog with HiGHS.
    Returns (b, M) or (None, None).
    """
    n = len(a_fixed)
    K = len(active_lags)
    
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize M
    
    A_ub = np.zeros((K, n + 1))
    for i, k in enumerate(active_lags):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        idx = np.arange(js, je + 1)
        other = k - idx
        A_ub[i, other] = a_fixed[idx]
        A_ub[i, n] = -1.0
    b_ub = np.zeros(K)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True})
    except Exception:
        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
        except Exception:
            return None, None
    
    if res.status != 0:
        return None, None
    
    b_val = res.x[:n].copy()
    M_val = float(res.x[n])
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, M_val


def nesterov_active_set(a0, time_limit, K_init=30, K_max=100, verbose=True):
    """
    Nesterov-accelerated LP method with active set management.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    
    t0 = time.monotonic()
    
    # Running average for solution extraction (Nesterov's averaging trick)
    avg_a = np.zeros(n)
    n_avg = 0
    
    # Dual tracking
    M_values = []
    
    # Active set initialization
    b_curr = fast_autocorr(a)
    K = K_init
    active_set = list(np.argsort(b_curr)[::-1][:K])
    
    iter_count = 0
    last_print = t0
    stall = 0
    best_score_at_stall = best_score
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP with active set
        b_star, M_val = solve_lp_active(y, active_set)
        if b_star is None:
            # Try with more constraints
            b_curr_y = cross_corr(y, y)
            active_set = list(np.argsort(b_curr_y)[::-1][:min(K_max, 2*n-1)])
            b_star, M_val = solve_lp_active(y, active_set)
            if b_star is None:
                continue
        
        iter_count += 1
        
        # Check for violated constraints (using cross-correlation)
        cross = cross_corr(y, b_star)
        cross_max = cross.max()
        if cross_max > M_val + 1e-8:
            # Find violated lags
            violated = np.where(cross > M_val + 1e-8)[0]
            # Add top violations to active set
            violated_sorted = violated[np.argsort(cross[violated])[::-1]]
            add_count = min(len(violated_sorted), K - len(active_set) + 5)
            if add_count > 0:
                new_lags = list(violated_sorted[:add_count])
                active_set = list(set(active_set + new_lags))
                # Trim if too large
                if len(active_set) > K:
                    # Keep the most binding ones
                    cross_active = [cross[k] for k in active_set]
                    active_set = [active_set[i] for i in np.argsort(cross_active)[::-1][:K]]
                # Re-solve
                b_star, M_val = solve_lp_active(y, active_set)
                if b_star is None:
                    active_set = list(np.argsort(cross)[::-1][:K])
                    b_star, M_val = solve_lp_active(y, active_set)
                    if b_star is None:
                        continue
        
        # Nesterov update
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_sum = y_new.sum()
        if y_sum < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / y_sum
        
        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            a_sum = a_new.sum()
            if a_sum < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / a_sum
        
        y = y_new
        a = a_new
        
        # Update running average (harmonic average schedule)
        # weight_t = 1/t (standard Nesterov averaging)
        n_avg += 1
        avg_a = avg_a * (1.0 - 1.0/n_avg) + a_new * (1.0/n_avg)
        
        # Score and update best
        autocorr_a = fast_autocorr(a)
        score_a = 2.0 * n * autocorr_a.max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        
        score_avg = 2.0 * n * fast_autocorr(avg_a).max()
        if score_avg < best_score:
            best_score = score_avg
            best_a = avg_a.copy()
            stall = 0
        
        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        
        # Track dual
        M_values.append(M_val)
        
        # Adapt K: increase when stalled (need more constraints for precision)
        if stall > 500 and K < K_max:
            K = min(K * 2, K_max)
            b_curr_a = fast_autocorr(a)
            active_set = list(np.argsort(b_curr_a)[::-1][:K])
            stall = 0
        
        # Periodic print
        if time.monotonic() - last_print > 10 and verbose:
            elapsed = time.monotonic() - t0
            avg_M = np.mean(M_values[-100:]) if len(M_values) >= 100 else np.mean(M_values)
            print(f"  LAS t={t} ({elapsed:.0f}s): best={best_score:.6f}, M_avg={avg_M:.8f}, K={K}, iters={iter_count}")
            last_print = time.monotonic()
        
        # Stall detection
        if score_a > best_score * 1.00001:
            stall += 1
        
        # Adaptive restart: if severely stalled, try a different approach
        if stall > 5000:
            # Reset Nesterov momentum (keep the best point)
            y = best_a.copy()
            a = best_a.copy()
            alpha_base = t  # effectively restart the counter
            stall = 0
            if verbose:
                print(f"  LAS: Restart at t={t}")
            # Shift t counter conceptually (we can't easily, so just reset momentum)
            # Actually, let's just continue with the current t
    
    return best_a, best_score, iter_count


def full_lp_nesterov(a0, time_limit, verbose=True):
    """Full LP Nesterov (no active set) for final precision."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()

    if s < 1e-15:
        a = np.ones(n) / n

    y = a.copy()

    best_score = 2.0 * n * fast_autocorr(a).max()
    
    t0 = time.monotonic()

    n_avg = 0
    
    full_lags = list(range(2 * n - 1))

            break

        y_new = (1.0 - alpha_t) * y + alpha_t * b_star

            a_new = a_new / a_new.sum()
        
        y = y_new

        n_avg += 1

def qp_polish(a0, time_limit, n_active=30):
    """
    QP polishing: minimize M subject to (a*a)_k <= M for active k,
    sum(a) = 1, a >= 0.
    Uses scipy minimize with SLSQP.
    """
    import cvxpy as cp
    
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    b = fast_autocorr(a)
    max_b = b.max()
    
    best_a = a.copy()
    best_score = 2.0 * n * max_b
    t0 = time.monotonic()
    
    for round_num in range(10):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Active set: lags where (a*a)_k is close to max
        threshold = max_b * 0.99
        active = list(np.where(b >= threshold)[0][:n_active])
        
        # QP: min M s.t. (a*a)_k <= M for k in active, sum(a)=1, a>=0
        a_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        
        constraints = [cp.sum(a_var) == 1]
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            other = k - idx
            q = cp.sum(a_var[idx] * a_var[other])
            constraints.append(q <= M_var)
        
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iter=5000)
        except Exception:
            try:
                prob.solve(solver=cp.CLARABEL)
            except Exception:
                break
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        
        a_new = np.asarray(a_var.value, dtype=np.float64)
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            b = fast_autocorr(a)
            max_b = b.max()
            print(f"  QP round {round_num}: {best_score:.6f}")
        else:
            # Try mixing
            improved = False
            for frac in [0.1, 0.3, 0.5]:
                a_mix = (1 - frac) * a + frac * a_new
                a_mix = np.maximum(a_mix, 0)
                a_mix = a_mix / a_mix.sum()
                score_mix = 2.0 * n * fast_autocorr(a_mix).max()
                if score_mix < best_score:
                    best_score = score_mix
                    best_a = a_mix.copy()
                    a = a_mix
                    b = fast_autocorr(a)
                    max_b = b.max()
                    improved = True
                    break
            if not improved and round_num >= 2:
                break
    
    return best_a, best_score


def coordinate_descent_polish(a0, time_limit):
    """
    Coordinate descent: for each coordinate, find the optimal value
    given all others fixed.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    rng = np.random.default_rng(42)
    
    for outer in range(200):
        if time.monotonic() - t0 > time_limit:
            break
        
        improved = False
        # Random order for diversity
        order = rng.permutation(n)
        
        for i in order:
            # Fix all coordinates except a[i]
            # The total sum must be 1, so a[i] = 1 - sum(others)
            # But we also need a[i] >= 0
            others_sum = 1.0 - a[i]
            if others_sum < 0:
                continue
            
            # The objective as a function of a[i] (call it x):
            # (a*a)_k = x * (sum of a[j] where i+j=k) + (contributions not involving i)
            # For each k, the contribution involving a[i]=x is x * (a[i+k] + a[k-i] - x if k=2i... )
            # This is getting complicated. Let's just do a simple 1D optimization.
            
            # Try a few values around current
            x0 = a[i]
            candidates = [max(0, x0 * 0.5), max(0, x0 * 0.75), x0, x0 * 1.25, x0 * 1.5]
            candidates = [c for c in candidates if c <= others_sum + x0]  # must maintain sum=1
            
            best_x = x0
            best_sc = best_score
            
            for x in candidates:
                if x < 0 or x > 1:
                    continue
                a_trial = a.copy()
                a_trial[i] = x
                # Renormalize
                a_trial = a_trial / a_trial.sum()
                sc = 2.0 * n * fast_autocorr(a_trial).max()
                if sc < best_sc:
                    best_sc = sc
                    best_x = x
            
            if best_x != x0:
                a[i] = best_x
                a = a / a.sum()
                improved = True
        
        sc = 2.0 * n * fast_autocorr(a).max()
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        if not improved:
            break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
