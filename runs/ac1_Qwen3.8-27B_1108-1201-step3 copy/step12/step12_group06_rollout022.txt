def fast_convolve(a):
    """Fast linear convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score assuming sum(a)=1: returns 2*n*max(b)."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def softmax_from_z(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s

def nesterov_projected_gradient(a0, time_limit, max_iters=500000, k_active=20):

def gs_time_domain(a0, time_limit, max_iters=200000, k_active=10):
    """
    Time-domain error reduction: iteratively reduce the peak of the convolution
    by moving mass from high-contribution positions to low-contribution ones.
    This is a fast gradient-free heuristic.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / a.sum()

    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_print = t0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        peak_idx = int(np.argmax(b))

        # Compute contribution of each a[i] to the peak
        lo = max(0, peak_idx - n + 1)
        hi = min(n - 1, peak_idx)
        contrib = np.zeros(n)
        idx = np.arange(lo, hi + 1)
        contrib[idx] = a[peak_idx - idx]

        # Move mass from high-contribution to low-contribution positions
        # Use a soft update: redistribute based on contribution
        a_mean = 1.0 / n
        # Reduce where contrib is high relative to a_mean, increase where low
        delta = contrib - a_mean
        # Scale: we want to remove a small fraction of the excess
        scale = 0.1 / (np.max(np.abs(delta)) + 1e-10)
        a_new = a - scale * delta * a

        # Also consider secondary peaks
        sorted_idx = np.argsort(b)[-k_active:]
        for k2 in sorted_idx[1:]:
            lo2 = max(0, k2 - n + 1)
            hi2 = min(n - 1, k2)
            idx2 = np.arange(lo2, hi2 + 1)
            contrib2 = np.zeros(n)
            contrib2[idx2] = b[k2] * a[k2 - idx2]
            a_new = a_new - 0.3 * scale * (contrib2 - np.mean(contrib2)) * a_new

        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        sc = score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            # If no improvement, use smaller step
            a_new2 = a - 0.5 * scale * delta * a
            a_new2 = np.maximum(a_new2, 0)
            s2 = a_new2.sum()
            if s2 > 1e-10:
                a_new2 = a_new2 / s2
                sc2 = score_normalized(a_new2)
                if sc2 < best_score:
                    best_score = sc2
                    best_a = a_new2.copy()
                    a = a_new2.copy()
                else:
                    # Reduce step further
                    a_new3 = a - 0.1 * scale * delta * a
                    a_new3 = np.maximum(a_new3, 0)
                    s3 = a_new3.sum()
                    if s3 > 1e-10:
                        a_new3 = a_new3 / s3
                        a = a_new3.copy()

        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 10:
                print(f"    GSTD iter={it+1} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score

def lse_objective_grad(z_in, T_val, n):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    max_b = b.max()
    if max_b > 700:
        max_b = 700
    exp_b = np.exp((b - max_b) / T_val)
    w = exp_b / exp_b.sum()
    log_sum_exp = max_b + T_val * np.log(exp_b.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z

            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, T, n)
            return grad

        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-16, 'maxfun': iters_per_stage * 500,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(z)
            sc = score_normalized(a_check)

            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()

            # Adaptive temperature update
            improvement = prev_score - sc
            if improvement < 1e-8:
                stagnant += 1
            else:
                stagnant = max(0, stagnant - 2)

            if stagnant >= 3:
                T = max(T, T * 1.5)
                stagnant = 0
            else:
                T = max(T_min, T * 0.7)

            prev_score = sc

            if (stage + 1) % 5 == 0:
                print(f"    LSE stage={stage+1}/{n_stages} T={T:.3f}: score={sc:.8f} best={best_score:.8f}")

        except Exception:
            T = max(T_min, T * 0.8)

    return best_a, best_score


def weighted_subgradient_nesterov(a0, time_limit, max_iters=500000, k_active=30):
    """
    Weighted subgradient with Nesterov acceleration and iterate averaging.
    Uses exponential weighting of active constraints for smoother gradients.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / a.sum()

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    lower_bound = 2.0 * n / (2.0 * n - 1.0)

    t0 = time.monotonic()
    last_print = t0

    # Exponential moving average for iterate averaging
    avg_a = a.copy()
    avg_weight = 0.0
    beta_avg = 0.999

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(y)
        max_b = b.max()
        cur_score = 2.0 * n * max_b

        # Weighted active set: use all indices with weight proportional to exp(b/T)
        T_ws = max_b * 0.05 + 1e-10
        w = np.exp((b - max_b) / T_ws)
        w_sum = w.sum()
        if w_sum < 1e-30:
            break

        # Compute weighted gradient
        g = np.zeros(n)
        for k in range(len(b)):
            if w[k] / w_sum < 1e-10:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += 2.0 * (w[k] / w_sum) * b[k] * y[k - idx]

        g_norm_sq = np.dot(g, g)
        if g_norm_sq < 1e-30:
            break

        # Nesterov step
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        # Polyak step
        gap = cur_score - lower_bound
        if gap < 1e-12:
            break
        eta = gap / (g_norm_sq * (t_k + 1.0))
        eta = min(eta, 2.0 / (t_k))

        a_old = a.copy()
        a_new = project_simplex(y - alpha * eta * g)

        y_new = a_new + momentum * (a_new - a_old)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        # Iterate averaging (exponential)
        avg_weight = avg_weight * beta_avg + 1.0
        avg_a = (avg_a * (avg_weight - 1.0) / max(avg_weight, 1e-10) + a) if avg_weight > 1 else a.copy()
        avg_a = avg_a / avg_a.sum() if avg_a.sum() > 1e-10 else a.copy()

        sc = score_normalized(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        sc_avg = score_normalized(avg_a)
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = avg_a.copy()

        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 10:
                print(f"    WSGN iter={it+1} ({elapsed:.0f}s): best={best_score:.8f}")
                last_print = elapsed

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng, method=0):
    """Generate structured initial sequences."""
    if method == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif method == 1:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif method == 2:
        x = np.linspace(0, 1, n)
        a = np.exp(-5 * (x - 0.5) ** 2)
    elif method == 3:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x)
    elif method == 4:
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.5) ** 2) / 0.08)
    elif method == 5:
        x = np.linspace(0, 1, n)
        sigma = 0.15
        a = np.exp(-((x - 0.5) ** 2) / (2 * sigma ** 2)) + 0.2 * np.exp(-((x - 0.3) ** 2) / (2 * 0.05 ** 2)) + 0.2 * np.exp(-((x - 0.7) ** 2) / (2 * 0.05 ** 2))
    elif method == 6:
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x)) ** 0.5
    else:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(2 * np.pi * 2 * x) + 0.1 * np.cos(2 * np.pi * 5 * x)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def spectral_shaping(a0, time_limit, max_iters=100000):
    """
    Spectral shaping: alternate between time-domain (non-negativity) and
    frequency-domain (peak reduction) operations.
    In frequency domain, we reduce the peak of |F|^2 by redistributing energy.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / a.sum()

    best_a = a.copy()
    best_score = score_normalized(a)
    N = 1 << (2 * n - 1).bit_length()

    t0 = time.monotonic()
    last_print = t0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute FFT
        F = rfft(a, N)
        power = F.real ** 2 + F.imag ** 2

        # In frequency domain: reduce peak power
        # Target: make power spectrum more uniform
        mean_power = power.mean()
        peak_power = power.max()
        if peak_power <= mean_power * 1.01:
            break

        # Soft peak reduction: compress the dynamic range
        # |F_new| = |F| * (mean/target)^beta for peaks
        beta = 0.3  # softness parameter
        mag = np.sqrt(power)
        mean_mag = np.sqrt(mean_power)
        ratio = mag / (mag + 1e-15)
        # Compress: new_mag = mag - beta * max(0, mag - mean_mag)
        excess = np.maximum(mag - mean_mag, 0)
        mag_new = mag - beta * excess
        mag_new = np.maximum(mag_new, 1e-10)

        # Apply new magnitude, keep phase
        phase = np.angle(F)
        F_new = mag_new * (np.cos(phase) + 1j * np.sin(phase))

        # Inverse FFT
        a_new = irfft(F_new, N)[:n]

        # Time domain: enforce non-negativity
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a.copy()
        else:
            a_new = a_new / s_new

        sc = score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new.copy()
        else:
            # Blend with previous
            a_new = 0.5 * a + 0.5 * a_new
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                sc2 = score_normalized(a_new)
                if sc2 < best_score:
                    best_score = sc2
                    best_a = a_new.copy()
                    a = a_new.copy()

        if (it + 1) % 10000 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 10:
                print(f"    Spectral iter={it+1} ({elapsed:.0f}s): best={best_score:.8f}")
                last_print = elapsed

    return best_a, best_score


def coordinate_descent_max(b, a, n, max_coords=50):
    """
    Coordinate descent: try swapping mass between pairs of coordinates
    to reduce the max convolution value.
    """
    a = a.copy()
    max_b = b.max()
    peak_idx = int(np.argmax(b))

    # For the peak at index k, the contribution of a[i] is a[k-i]
    # Reducing a[i] where a[k-i] is large reduces the peak
    lo = max(0, peak_idx - n + 1)
    hi = min(n - 1, peak_idx)
    idx = np.arange(lo, hi + 1)
    contrib = a[peak_idx - idx]

    # Find top contributors
    top_contrib = np.argsort(contrib)[-max_coords:]
    top_positions = idx[top_contrib]

    # Find positions with low contribution (good places to add mass)
    all_contrib = np.zeros(n)
    all_contrib[idx] = contrib
    low_positions = np.argsort(all_contrib)[:max_coords]

    # Try small transfers
    epsilon = 0.01 / len(a)
    a_new = a.copy()
    for src in top_positions[:5]:
        a_new[src] -= epsilon
    for dst in low_positions[:5]:
        a_new[dst] += epsilon

    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s

    return a_new


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Combines multiple optimization strategies:
    1. Fast weighted subgradient with Nesterov acceleration
    2. Nesterov LP best-response
    3. Adaptive LogSumExp L-BFGS
    4. Spectral shaping (frequency-domain peak reduction)
    5. Time-domain error reduction
    6. Multi-length exploration
    7. Final combined polishing
    """
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Total budget: {budget_s:.0f}s")

    # Get parent sequence or default
    best_seq = None
    best_score = float('inf')
    n_base = 1200

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            n_base = len(best_seq)
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.8f}")
    except Exception:
        best_seq = structured_init(n_base, rng, 0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            remain = total_deadline - time.monotonic()
            print(f"  ** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, remaining={remain:.0f}s)")

    # === Phase 1: Weighted Subgradient with Nesterov - 25% time ===
    print(f"\n=== Phase 1: Weighted Subgradient (score={best_score:.8f}) ===")
    t1 = 0.25 * (budget_s - 30)
    a_out, _ = weighted_subgradient_nesterov(best_seq, t1, max_iters=800000, k_active=30)
    try_update(a_out, "wsgn_main")
    print(f"  Phase 1 done: best_score={best_score:.8f}")

    # === Phase 2: Nesterov LP - 25% time ===
    print(f"\n=== Phase 2: Nesterov LP (score={best_score:.8f}) ===")
    remaining = total_deadline - time.monotonic()
    t2 = min(0.25 * (budget_s - 30), remaining * 0.5)
    if t2 > 20:
        a_out2, _ = nesterov_lp(best_seq, t2, solver=solver, max_iters=300000)
        try_update(a_out2, "nesterov_lp")
    print(f"  Phase 2 done: best_score={best_score:.8f}")

    # === Phase 3: Adaptive LSE L-BFGS - 25% time ===
    print(f"\n=== Phase 3: Adaptive LSE (score={best_score:.8f}) ===")
    remaining = total_deadline - time.monotonic()
    t3 = min(0.25 * (budget_s - 30), remaining * 0.5)
    if t3 > 30:
        # First pass: high temperature
        a_lse1, _ = lsbfgs_lse_adaptive(best_seq, t3 * 0.55, T_start=100.0, T_min=0.1, n_stages=25, iters_per_stage=35)
        try_update(a_lse1, "lse_highT")

        remaining = total_deadline - time.monotonic()
        if remaining > 20:
            # Second pass: low temperature from best
            a_lse2, _ = lsbfgs_lse_adaptive(best_seq, remaining * 0.55, T_start=20.0, T_min=0.05, n_stages=15, iters_per_stage=30)
            try_update(a_lse2, "lse_lowT")

        remaining = total_deadline - time.monotonic()
        if remaining > 20:
            # Third pass: from perturbed
            pert = best_seq + 0.03 * rng.normal(0, 1, len(best_seq))
            pert = np.maximum(pert, 0)
            pert = pert / pert.sum()
            a_lse3, _ = lsbfgs_lse_adaptive(pert, remaining * 0.55, T_start=50.0, T_min=0.1, n_stages=15, iters_per_stage=30)
            try_update(a_lse3, "lse_pert")
    print(f"  Phase 3 done: best_score={best_score:.8f}")

    # === Phase 4: Spectral Shaping + Time-domain ER - 10% time ===
    print(f"\n=== Phase 4: Spectral/Time-domain (score={best_score:.8f}) ===")
    remaining = total_deadline - time.monotonic()
    t4 = min(0.10 * (budget_s - 30), remaining * 0.5)
    if t4 > 15:
        a_spec, _ = spectral_shaping(best_seq, t4 * 0.5, max_iters=100000)
        try_update(a_spec, "spectral")

        remaining = total_deadline - time.monotonic()
        if remaining > 15:
            a_td, _ = gs_time_domain(best_seq, remaining * 0.5, max_iters=150000, k_active=8)
            try_update(a_td, "time_domain")
    print(f"  Phase 4 done: best_score={best_score:.8f}")

    # === Phase 5: Projected Gradient (fast) - 8% time ===
    print(f"\n=== Phase 5: Projected Gradient (score={best_score:.8f}) ===")
    remaining = total_deadline - time.monotonic()
    t5 = min(0.08 * (budget_s - 30), remaining * 0.5)
    if t5 > 15:
        a_pg, _ = nesterov_projected_gradient(best_seq, t5, max_iters=500000, k_active=25)
        try_update(a_pg, "proj_grad")
    print(f"  Phase 5 done: best_score={best_score:.8f}")

    # === Phase 6: Multi-length exploration - 5% time ===
    print(f"\n=== Phase 6: Multi-length (score={best_score:.8f}) ===")
    remaining = total_deadline - time.monotonic()
    t6_total = min(0.05 * (budget_s - 30), remaining * 0.6)
    if t6_total > 20:
        lengths_to_try = [n_base + 100, n_base + 300, n_base + 500]
        for n_try in lengths_to_try:
            remaining = total_deadline - time.monotonic()
            if remaining < 20:
                break
            t6 = min(t6_total / 3.0, remaining * 0.4)
            if t6 < 10:
                break
            print(f"  Trying n={n_try}, budget={t6:.0f}s")
            init_try = interpolate_sequence(best_seq, n_try)
            a_try, _ = nesterov_lp(init_try, t6 * 0.6, solver=solver, max_iters=50000)
            try_update(a_try, f"len_{n_try}_lp")
            remaining = total_deadline - time.monotonic()
            if remaining > 10:
                a_try2, _ = lsbfgs_lse_adaptive(a_try, remaining * 0.5, T_start=30.0, T_min=0.2, n_stages=8, iters_per_stage=20)
                try_update(a_try2, f"len_{n_try}_lse")
    print(f"  Phase 6 done: best_score={best_score:.8f}")

    # === Phase 7: Final combined polish - 2% time ===
    print(f"\n=== Phase 7: Final polish (score={best_score:.8f}) ===")
    remaining = total_deadline - time.monotonic()
    if remaining > 10:
        # Quick LSE from very low T
        a_f1, _ = lsbfgs_lse_adaptive(best_seq, remaining * 0.4, T_start=10.0, T_min=0.05, n_stages=8, iters_per_stage=20)
        try_update(a_f1, "final_lse")

        remaining = total_deadline - time.monotonic()
        if remaining > 8:
            # Quick subgradient
            a_f2, _ = weighted_subgradient_nesterov(best_seq, remaining * 0.5, max_iters=200000, k_active=20)
            try_update(a_f2, "final_wsgn")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    elapsed_total = time.monotonic() - (total_deadline - (budget_s - 30))
    print(f"Time used: {elapsed_total:.0f}s / {budget_s:.0f}s")
    return [float(x) for x in best_seq]
