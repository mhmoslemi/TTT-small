def fast_autocorr(a):
    """Compute full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):

def project_simplex(z):
    """Project z onto the probability simplex {a: a>=0, sum(a)=1}."""
    n = len(z)
    if z.sum() <= 0:
        return np.ones(n) / n
    # Water-filling
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(z - theta, 0)
    return w / w.sum()

            je2 = min(n - 1, k)
            idx2 = np.arange(js2, je2 + 1)
            danger[idx2] += a[k - idx2]  # approximate danger

        # Try both src and src_partner as source
        for source in [src, src_partner] if src != src_partner else [src]:
            amt_options = [a[source] * f for f in [0.001, 0.003, 0.005, 0.01, 0.02]]
            target_positions = np.argsort(danger)[:20]  # least dangerous

                if amt < 1e-15 or amt > a[source]:
                    continue
                for dst in target_positions:
                    if dst == source and dst == src_partner:
                        continue
                    a_trial = a.copy()
                    a_trial[source] -= amt
                    if src != src_partner:
                        # Only reduce one of the pair
                        pass
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
                        break
                if stall == 0:
                    break
            if stall == 0:
                break
        else:
            stall += 1
            if stall > 200:
                # Random perturbation
                n_pert = rng.integers(2, 8)
                sources = rng.choice(n, size=n_pert, replace=False)
                targets = rng.choice(n, size=n_pert, replace=False)

                break

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated zero-sum game (LP best responses)
2. Active-set local refinement via linearized subproblems
3. Smooth peak-shaving with line search
4. Multi-length exploration
5. Running-average convergence tracking

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def project_simplex(z):
    """Euclidean projection onto probability simplex."""
    n = len(z)
    if n == 0:
        return z
    if z.sum() <= 1e-30:
        return np.ones(n) / n
    u = np.sort(z)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(z - theta, 0)
    s = w.sum()
    if s < 1e-30:
        return np.ones(n) / n
    return w / s


def find_solver():
    """Find a working LP/QP solver."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_best_response(y, solver, fallback_solvers):
    """Solve LP: min M s.t. conv(y,b)_k <= M, sum(b)=1, b>=0."""
    n = len(y)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    # Build constraints using Toeplitz structure
    # conv(y, b)[k] = sum_j y[k-j] * b[j] for valid indices
    # This is (T @ b)[k] where T[k,j] = y[k-j] if 0<=k-j<n
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = y

    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    tried = [solver] + [s for s in fallback_solvers if s != solver]
    for s in tried:
        try:
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                break
        except Exception:
            continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s


def nesterov_game(a0, time_limit, solver, fallback_solvers, verbose=True):
    """
    Nesterov-accelerated best-response iteration for the zero-sum game.
    Both players minimize max_k conv(a,b)[k]. At equilibrium, a=b gives
    the minmax of max_k conv(a,a)[k].
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    y = a.copy()  # Nesterov momentum point

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    # Running averages at different windows
    avg_long = np.zeros(n)
    n_long = 0
    avg_short = np.zeros(n)
    n_short = 0

    stall_count = 0
    stall_threshold = best_score * 1.000001

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        # Solve best response LP
        b_star = solve_best_response(y, solver, fallback_solvers)
        if b_star is None:
            continue

        # Nesterov schedule
        alpha_t = 2.0 / (t + 2.0)

        # Update momentum point
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        yn = y_new.sum()
        if yn < 1e-15:
            y_new = y.copy()
            yn = 1.0
        y_new = y_new / yn

        # Nesterov extrapolation
        if t == 1:
            a_new = b_star.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            a_new = y_new + beta * (y_new - y)
            a_new = np.maximum(a_new, 0)
            an = a_new.sum()
            if an < 1e-15:
                a_new = y_new.copy()
                an = 1.0
            a_new = a_new / an

        y = y_new
        a = a_new

        # Update running averages
        avg_long += a_new
        n_long += 1
        avg_short = (avg_short * 0.95 + a_new * 0.05) if n_short > 0 else a_new.copy()
        n_short += 1

        # Score check (every 5 iterations to save time)
        if t % 5 == 0:
            sc_a = 2.0 * n * fast_autocorr(a).max()
            if sc_a < best_score:
                best_score = sc_a
                best_a = a.copy()
                stall_count = 0
            else:
                stall_count += 1

        # Check running averages periodically
        if n_long % 50 == 0:
            avg = avg_long / n_long
            sc_avg = 2.0 * n * fast_autocorr(avg).max()
            if sc_avg < best_score:
                best_score = sc_avg
                best_a = avg.copy()
                stall_count = 0

        if n_short % 100 == 0 and n_short > 50:
            sc_short = 2.0 * n * fast_autocorr(avg_short).max()
            if sc_short < best_score:
                best_score = sc_short
                best_a = avg_short.copy()
                stall_count = 0

        if t % 2000 == 0 and verbose:
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}")

        # Early stopping if severely stalled
        if stall_count > 5000 and t > 2000:
            if verbose:
                print(f"      NLP early stop at t={t}")
            break

    # Final check of running average
    if n_long > 100:
        avg = avg_long / n_long
        sc_avg = 2.0 * n * fast_autocorr(avg).max()
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = avg.copy()

    return best_a, best_score


def local_subgradient_refine(a0, time_limit, verbose=True):
    """
    Projected subgradient method with adaptive step size and multi-peak weighting.
    Minimizes max_k (a*a)[k] on the simplex.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    lr = 0.02
    lr_min = 1e-6
    no_improve = 0

    for it in range(500000):
        if time.monotonic() - t0 > time_limit:
            break
        if lr < lr_min:
            break

        b = fast_autocorr(a)
        max_b = b.max()

        # Multi-peak weighted subgradient
        n_peaks = min(8, 2 * n - 1)
        peak_lags = np.argsort(b)[::-1][:n_peaks]
        # Exponential weighting based on proximity to max
        weights = np.exp((b[peak_lags] - max_b) * 8.0)
        weights = weights / weights.sum()

        grad = np.zeros(n)
        for w, k in zip(weights, peak_lags):
            k = int(k)
            # grad[i] += 2*w*a[k-i] for valid i (0 <= k-i < n)
            i_start = max(0, k - n + 1)
            i_end = min(n - 1, k)
            if i_start <= i_end:
                idx = np.arange(i_start, i_end + 1)
                grad[idx] += 2.0 * w * a[k - idx]

        # Projected step with backtracking
        a_trial = a - lr * grad
        a_trial = project_simplex(a_trial)
        sc_trial = 2.0 * n * fast_autocorr(a_trial).max()

        # Backtracking line search
        if sc_trial >= best_score:
            for shrink in [0.5, 0.25, 0.125, 0.0625]:
                a_small = a - lr * shrink * grad
                a_small = project_simplex(a_small)
                sc_small = 2.0 * n * fast_autocorr(a_small).max()
                if sc_small < best_score:
                    a_trial = a_small
                    sc_trial = sc_small
                    break
            else:
                # No improvement, shrink learning rate
                lr *= 0.9
                no_improve += 1
                if no_improve > 5000:
                    break
                continue
        else:
            no_improve = 0
            lr = min(lr * 1.01, 0.05)  # Slowly grow if improving

        a = a_trial
        if sc_trial < best_score:
            best_score = sc_trial
            best_a = a.copy()

        if it % 5000 == 0 and verbose:
            print(f"      SubGrad it={it}: best={best_score:.6f}, lr={lr:.6f}")

    return best_a, best_score


def peak_shave_refine(a0, time_limit, seed=42, verbose=True):
    """
    Direct peak shaving: identify peak-contributing pairs and redistribute mass.
    Uses multiple strategies including pair reduction and random exploration.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    max_stall = 3000

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        k_star = int(np.argmax(b))

        # Strategy 1: Reduce top contributing pairs
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_star - idx]

        improved = False
        # Try top 10 contributors
        top_ci = np.argsort(contrib)[::-1][:10]
        for ci in top_ci:
            src = int(idx[ci])
            partner = int(k_star - src)

            # Compute "danger" of each position (how much it contributes to peaks)
            top_peaks = np.argsort(b)[::-1][:5]
            danger = np.zeros(n)
            for k in top_peaks:
                k = int(k)
                s2 = max(0, k - n + 1)
                e2 = min(n - 1, k)
                if s2 <= e2:
                    ix = np.arange(s2, e2 + 1)
                    danger[ix] += a[k - ix]

            # Try moving from source to least-dangerous positions
            safety = np.argsort(danger)[:30]
            for dst in safety:
                if dst == src and dst == partner:
                    continue
                for frac in [0.001, 0.003, 0.005, 0.01, 0.015, 0.02]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < -1e-12:
                        continue
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    sc = 2.0 * n * fast_autocorr(a_trial).max()
                    if sc < best_score - 1e-15:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        # Strategy 2: Reduce both members of a pair
        if not improved:
            for ci in top_ci[:5]:
                src = int(idx[ci])
                partner = int(k_star - src)
                if src == partner:
                    continue
                safety = np.argsort(danger)[:20]
                for dst in safety:
                    if dst in (src, partner):
                        continue
                    for frac in [0.002, 0.005, 0.01, 0.015]:
                        amt = min(a[src], a[partner]) * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[partner] -= amt
                        a_trial[dst] += 2 * amt
                        if (a_trial < -1e-12).any():
                            continue
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        sc = 2.0 * n * fast_autocorr(a_trial).max()
                        if sc < best_score - 1e-15:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            # Random perturbation every 50 stalls
            if stall % 50 == 0:
                n_pert = int(rng.integers(3, 10))
                sources = rng.choice(n, size=n_pert, replace=False)
                targets = rng.choice(n, size=n_pert, replace=False)
                a_trial = a.copy()
                for s2, d2 in zip(sources, targets):
                    f = float(rng.uniform(0.002, 0.01))
                    amt2 = a_trial[s2] * f
                    a_trial[s2] -= amt2
                    a_trial[d2] += amt2
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                sc = 2.0 * n * fast_autocorr(a_trial).max()
                if sc < best_score:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    stall = 0
            if stall > max_stall:
                break

        if it := (int((time.monotonic() - t0) * 10)) % 5000 == 0 and verbose:
            print(f"      PeakShave: best={best_score:.6f}, stall={stall}")

    return best_a, best_score


def gaussian_smooth(a, sigma):
    """Apply Gaussian smoothing to a sequence."""
    n = len(a)
    kernel_size = int(6 * sigma) | 1  # odd
    x = np.arange(kernel_size) - kernel_size // 2
    kernel = np.exp(-0.5 * (x / sigma) ** 2)
    kernel = kernel / kernel.sum()
    smoothed = np.convolve(a, kernel, mode='same')
    smoothed = np.maximum(smoothed, 0)
    s = smoothed.sum()
    if s > 1e-15:
        smoothed = smoothed / s
    return smoothed


def try_smooth_variants(a, n_trials=5):
    """Try several smoothing levels and return the best."""
    n = len(a)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    for sigma in [1.0, 2.0, 3.0, 5.0, 8.0, 12.0]:
        a_s = gaussian_smooth(a, sigma)
        sc = 2.0 * n * fast_autocorr(a_s).max()
        if sc < best_score:
            best_score = sc
            best_a = a_s.copy()
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate to new length preserving shape."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n, endpoint=False)
    new_x = np.linspace(0, 1, new_n, endpoint=False)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Uses Nesterov game theory + local refinement
    to minimize 2*n*max(conv(a,a))/(sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)

    solver = find_solver()
    fallbacks = [s for s in [cp.ECOS, cp.SCS, cp.PDLP] if s != solver]
    print(f"Solver: {solver}, Fallbacks: {fallbacks}")
    print(f"Time budget: {time_budget:.0f}s")

    # Get parent sequence
    best_seq = None
    best_score = float('inf')
    n_base = 1600

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.all(np.isfinite(p)) and p.sum() > 1e-10:
            n_base = len(p)
            p = p / p.sum()
            ps = evaluate_safe(p)
            print(f"Parent: n={n_base}, score={ps:.6f}")
            best_seq = p.copy()
            best_score = ps
    except Exception:
        pass

    if best_seq is None:
        # Cosine initialization
        x = np.linspace(0, np.pi, n_base)
        best_seq = np.cos(x / 2) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    def check_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: Nesterov Game (primary optimization) ===
    remaining = time_budget - (time.monotonic() - t_start)
    t1 = min(remaining * 0.45, 700.0)
    print(f"\n=== Phase 1: Nesterov LP (t={t1:.0f}s, n={n_base}) ===")
    if t1 > 60:
        a_out, sc_out = nesterov_game(best_seq, t1, solver, fallbacks, verbose=True)
        check_update(a_out, f"nesterov_n{n_base}")
        # Also try smoothed version
        a_smooth, sc_smooth = try_smooth_variants(a_out)
        check_update(a_smooth, f"smooth_n{n_base}")

    # === Phase 2: Local Subgradient Refinement ===
    remaining = time_budget - (time.monotonic() - t_start)
    t2 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 2: Subgradient (t={t2:.0f}s) ===")
    if t2 > 30:
        a_out, sc_out = local_subgradient_refine(best_seq, t2, verbose=True)
        check_update(a_out, "subgrad")

    # === Phase 3: Peak Shaving ===
    remaining = time_budget - (time.monotonic() - t_start)
    t3 = min(remaining * 0.20, 300.0)
    print(f"\n=== Phase 3: Peak Shaving (t={t3:.0f}s) ===")
    if t3 > 20:
        a_out, sc_out = peak_shave_refine(best_seq, t3, seed=123, verbose=True)
        check_update(a_out, "peakshave_1")

    # === Phase 4: Nesterov restart from current best ===
    remaining = time_budget - (time.monotonic() - t_start)
    t4 = min(remaining * 0.15, 200.0)
    print(f"\n=== Phase 4: Nesterov Round 2 (t={t4:.0f}s) ===")
    if t4 > 30:
        a_out, sc_out = nesterov_game(best_seq, t4, solver, fallbacks, verbose=False)
        check_update(a_out, "nesterov_r2")

    # === Phase 5: Try different length ===
    remaining = time_budget - (time.monotonic() - t_start)
    if remaining > 100:
        n_alt = n_base + 200
        t5 = min(remaining * 0.3, 150.0)
        print(f"\n=== Phase 5: Nesterov at n={n_alt} (t={t5:.0f}s) ===")
        init_alt = interpolate_sequence(best_seq, n_alt)
        a_out, sc_out = nesterov_game(init_alt, t5, solver, fallbacks, verbose=False)
        check_update(a_out, f"nesterov_n{n_alt}")

        # Peak shave on the new length
        remaining = time_budget - (time.monotonic() - t_start)
        if remaining > 30:
            t5b = min(remaining * 0.3, 80.0)
            a_out2, sc_out2 = peak_shave_refine(a_out, t5b, seed=456, verbose=False)
            check_update(a_out2, f"peakshave_n{n_alt}")

    # === Phase 6: Final peak shaving on global best ===
    remaining = time_budget - (time.monotonic() - t_start)
    t6 = min(remaining * 0.5, 120.0)
    print(f"\n=== Phase 6: Final Peak Shave (t={t6:.0f}s) ===")
    if t6 > 10:
        a_out, sc_out = peak_shave_refine(best_seq, t6, seed=789, verbose=True)
        check_update(a_out, "final_peakshave")

    # === Phase 7: Final subgradient polish ===
    remaining = time_budget - (time.monotonic() - t_start)
    t7 = min(remaining * 0.5, 60.0)
    if t7 > 10:
        a_out, sc_out = local_subgradient_refine(best_seq, t7, verbose=False)
        check_update(a_out, "final_subgrad")

    # Final smoothing check
    a_smooth, sc_smooth = try_smooth_variants(best_seq)
    check_update(a_smooth, "final_smooth")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}")
    else:
        print("ERROR: No valid sequence!")
        return [1.0]

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
