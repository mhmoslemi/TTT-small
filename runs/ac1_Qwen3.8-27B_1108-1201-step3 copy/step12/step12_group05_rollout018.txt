    n = len(a_fixed)

    bounds = [(0, None)] * n + [(None, None)]

                     bounds=bounds, method='highs')

        M_val = result.x[n]
        b_val = np.maximum(b_val, 0)
        s = b_val.sum()
        if s > 1e-15:
            b_val = b_val / s
        return b_val, M_val
    return None, None

def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.where(cond)[0][-1] + 1
    theta = cssv[i-1] / i
    return np.maximum(v - theta, 0)

    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_adam = 0

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        iters = max(10, int(remaining * 60 / max(1, n_stages - si)))

        for it in range(iters):
            if time.monotonic() - t0 > time_limit:
                break

            a = softmax_fn(z)
            b = fast_autocorr(a)
            Tb = T_start * b  # Use stage temperature
            Tb = stage_T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()

            # Gradient computation
            size = 2 * n - 1
            fft_size = 1 << (size - 1).bit_length()
            A_rev = rfft(a[::-1].copy(), fft_size)
            W_fft = rfft(w, fft_size)
            conv = irfft(A_rev * W_fft, fft_size)
            grad_a = 2.0 * conv[n - 1:n - 1 + n]

            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)

            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)

            lr_t = lr / np.sqrt(1 + 0.005 * it)
            z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)

            if it % 20 == 0:
                a_check = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a_check).max()
                if score < best_score:
                    best_score = score
                    best_a = a_check.copy()

        a = softmax_fn(z)
        score = 2.0 * n * fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Adam-LSE stage {si}/{n_stages} (T={stage_T:.2f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def build_T_matrix(a_fixed):
    """Build convolution matrix T where (T @ b) = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_scipy(a_fixed, prev_b=None):
    """Solve LP: min M s.t. T(a_fixed)@b <= M, sum(b)=1, b>=0 using scipy HiGHS."""
    n = len(a_fixed)
    T = build_T_matrix(a_fixed)

    # Decision variables: [b_0, ..., b_{n-1}, M]
    c = np.zeros(n + 1)
    c[-1] = 1.0

    # Inequality: T @ b - M <= 0
    A_ub = np.hstack([T, -np.ones((2 * n - 1, 1))])
    b_ub = np.zeros(2 * n - 1)

    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    # Bounds
    bounds = [(0, None)] * n + [(None, None)]

    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
        if result.status == 0:
            b_val = result.x[:n]
            M_val = result.x[n]
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s < 1e-15:
                return None, None
            b_val = b_val / s
            return b_val, float(M_val)
    except Exception:
        pass

    return None, None


def nesterov_lp_optimize(a0, time_limit, max_iters=1000000, topk_grad=5):
    """
    Nesterov-accelerated LP best-response iteration.
    Uses scipy HiGHS for fast LP solves.
    Tracks multiple averaging windows.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    # Multiple averaging windows
    avg_short = np.zeros(n)  # Last 50 iterates
    avg_med = np.zeros(n)    # Last 200 iterates
    avg_long = np.zeros(n)   # All iterates
    n_short = 0
    n_med = 0
    n_long = 0
    short_window = 50
    med_window = 200

    # Circular buffers for short/med averaging
    short_buf = np.zeros((short_window, n))
    med_buf = np.zeros((med_window, n))
    short_idx = 0
    med_idx = 0

    prev_b = None
    lp_times = []

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        t_lp_start = time.monotonic()
        b_star, M_val = solve_lp_scipy(y, prev_b)
        t_lp = time.monotonic() - t_lp_start
        lp_times.append(t_lp)

        if b_star is None:
            continue

        prev_b = b_star

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s

        y = y_new
        a = a_new

        # Update averages
        avg_short += a
        n_short += 1
        avg_med += a
        n_med += 1
        avg_long += a
        n_long += 1

        # Circular buffer updates
        short_buf[short_idx] = a
        short_idx = (short_idx + 1) % short_window
        med_buf[med_idx] = a
        med_idx = (med_idx + 1) % med_window

        # Check scores
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        # Check averaged solutions
        if n_long >= 10:
            avg_check = avg_long / n_long
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()

        if n_short >= short_window and t % 10 == 0:
            avg_s = short_buf.sum(axis=0) / short_window
            score_s = 2.0 * n * fast_autocorr(avg_s).max()
            if score_s < best_score:
                best_score = score_s
                best_a = avg_s.copy()

        if n_med >= med_window and t % 20 == 0:
            avg_m = med_buf.sum(axis=0) / med_window
            score_m = 2.0 * n * fast_autocorr(avg_m).max()
            if score_m < best_score:
                best_score = score_m
                best_a = avg_m.copy()

        if t % 100 == 0:
            elapsed = time.monotonic() - t0
            avg_lp_t = np.mean(lp_times[-50:]) if len(lp_times) >= 50 else np.mean(lp_times)
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}, LP_M={M_val:.8f}, avg_lp_t={avg_lp_t:.3f}s")

    return best_a, best_score


def subgradient_peak_shaving(a0, time_limit, topk=3):
    """
    Subgradient descent on the max autocorrelation peak.
    Uses simplex projection for constraint handling.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    lr = 0.01
    min_lr = 1e-6
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break

        grad, max_val, top_indices = compute_grad_max(a, topk=topk)

        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm

        # Try step
        a_trial = a - lr * grad
        # Project onto simplex {x >= 0, sum(x) = 1}
        a_trial = project_simplex(a_trial)

        score = 2.0 * n * fast_autocorr(a_trial).max()
        if score < best_score - 1e-12:
            a = a_trial
            best_score = score
            best_a = a.copy()
            no_improve = 0
            lr = min(lr * 1.1, 0.05)
        else:
            no_improve += 1
            lr *= 0.8
            if lr < min_lr:
                # Try random perturbation
                rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-10:
                    amt = a[i] * 0.01
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss > 1e-15:
                        a_trial = a_trial / ss
                        score = 2.0 * n * fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            no_improve = 0
                            lr = 0.005

            if no_improve > 200:
                break

    return best_a, best_score


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=np.float64)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.ones(n) / n
    i = np.where(cond)[0][-1] + 1
    theta = cssv[i - 1] / i
    return np.maximum(v - theta, 0.0)


def spectral_shaping(a, n_iter=500, alpha=0.3):
    """Spectral shaping: flatten power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2

        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))

        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.05, 20.0)
        scale[1:] = np.power(ratio, alpha / 2.0)

        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a


def coordinate_peak_shaving(a0, time_limit, seed=42, topk_peaks=16):
    """
    Coordinate descent peak shaving: move mass from peak contributors to quiet areas.
    More aggressive than the previous version.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:topk_peaks]
        max_b = b.max()

        improved = False
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            if contrib.sum() < 1e-15:
                continue

            src_local = np.argmax(contrib)
            src = idx[src_local]

            # Find quiet locations (low contribution to all top peaks)
            peak_areas = set()
            for k2 in top_k[:8]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_areas.update(range(js2, je2 + 1))

            candidates = [j for j in range(n) if j not in peak_areas]
            if not candidates:
                candidates = [j for j in range(n) if j != src]

            # Sort by a[j] ascending (move to already-small values)
            candidates_sorted = sorted(candidates, key=lambda j: a[j])
            dst_pool = candidates_sorted[:max(1, len(candidates_sorted) // 4)]
            dst = dst_pool[rng.integers(len(dst_pool))]

            for frac in [0.001, 0.003, 0.007, 0.015, 0.03, 0.05]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break

            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation
            for _ in range(5):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-12:
                    frac = rng.uniform(0.002, 0.02)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= 0:
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
                            break

            if stall > 500:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

        print(f"Total time: {elapsed:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
