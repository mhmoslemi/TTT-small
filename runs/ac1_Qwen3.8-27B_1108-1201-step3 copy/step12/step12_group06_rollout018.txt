```python
import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import cvxpy as cp

def fast_convolve(a):
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def score(a):
    """Score = 2*n*max(conv(a,a))/(sum(a))^2"""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    b = fast_convolve(a_norm)
    return 2.0 * n * b.max()

def solve_lp_active(y, active_idx, M, solver):
    """Solve LP: max 1^T b s.t. (y*b)[k] <= M for k in active_idx, b >= 0"""
    n = len(y)
    K = len(active_idx)
    # Build constraint matrix: (y*b)[k] = sum_j y[(k-j) mod ...] 
    # Actually (y*b)[k] = sum_j y_j * b_{k-j} for valid j, k-j
    # As a matrix: row k of T has T[j, k] = y_{k-j} if 0 <= k-j < n
    # Wait, (y*b)[k] = sum_j y_j * b_{k-j}
    # So the k-th row of the constraint matrix has entry (k,j) = y_{k-j} if 0 <= k-j < n, else 0
    
    b_var = cp.Variable(n, nonneg=True)
    # Build constraint rows
    T_rows = []
    for k in active_idx:
        row = np.zeros(n)
        for j in range(min(n, k+1)):
            if k-j < n:
                row[j] = y[k-j]
        T_rows.append(row)
    T = np.array(T_rows)
    
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    prob.solve(solver=solver)
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s
```

```python
2. Smoothed gradient L-BFGS with adaptive temperature

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize
import cvxpy as cp


def fast_convolve(a):
    """Fast linear convolution via real FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_norm(a):
    """Score assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def build_conv_rows(y, active_idx):
    """Build constraint rows for LP: (y*b)[k] = sum_j y[j]*b[k-j]."""
    n = len(y)
    K = len(active_idx)
    T = np.zeros((K, n))
    for i, k in enumerate(active_idx):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            # T[i, j] = y[k-j] for j in [j_start, j_end]
            idx = np.arange(j_start, j_end + 1)
            T[i, idx] = y[k - idx][::-1]
    return T


def get_lp_solver():
    """Find the best available LP solver."""
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CBC]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Maximize(cp.sum(x)), [cp.sum(x) <= 10])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_active(y, active_idx, M, solver):
    """Solve LP: max 1^T b s.t. (y*b)[k] <= M for k in active_idx, b >= 0.
    Returns normalized b (sum=1) or None."""
    n = len(y)
    K = len(active_idx)
    if K == 0:
        return None

    T = build_conv_rows(y, active_idx)
    b_var = cp.Variable(n, nonneg=True)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)

    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        pass

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        # Try alternative solver
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s


def check_violations(y, b, M, n, top_k=200):
    """Check which convolution constraints are violated by b.
    Returns indices of violated constraints (beyond the top_k already included)."""
    conv_yb = fast_convolve(y) * 0 + fast_convolve(y)  # Just compute (y*b)
    # Actually compute (y*b) properly
    yy = y
    bb = b
    conv = fast_convolve(np.concatenate([yy, np.zeros(0)]))  # This is y*y, not y*b
    # Need cross-correlation: (y*b)[k] = sum_j y[j]*b[k-j]
    # Use FFT
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    Y = rfft(yy, fft_size)
    B = rfft(bb, fft_size)
    conv_yb = irfft(Y * B, fft_size)[:size]

    violations = np.where(conv_yb > M * 1.001 + 1e-10)[0]
    return violations


def nesterov_lp_adaptive(a0, time_limit, solver, K_init=200, K_max=600,
                          max_iters=500000, smooth_eps=0.0):
    """Nesterov accelerated LP with adaptive active set and cutting-plane.
    
    Args:
        a0: initial sequence (will be normalized)
        time_limit: seconds to run
        solver: cvxpy solver
        K_init: initial active set size
        K_max: maximum active set size
        max_iters: maximum iterations
        smooth_eps: add epsilon*uniform to LP solution for diversity
    """
    n = len(a0)
    a = normalize(a0)
    y = a.copy()
    best_a = a.copy()
    best_score = score_norm(a)
    t0 = time.monotonic()
    last_print = t0

    # Average tracking
    avg_a = a.copy()
    avg_count = 0

    # Active set (start with top K_init)
    b_conv = fast_convolve(a)
    active_set = np.argsort(b_conv)[-K_init:][::-1]
    active_set = np.sort(active_set)
    K = K_init

    stalled = 0
    best_score_stall = best_score

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute current max
        b_y = fast_convolve(y)
        M = b_y.max()

        # Solve LP with current active set
        b_sol = solve_lp_active(y, active_set, M, solver)
        if b_sol is None:
            # Fallback: use uniform
            b_sol = np.ones(n) / n
        else:
            # Add small uniform for diversity
            if smooth_eps > 0:
                b_sol = (1 - smooth_eps) * b_sol + smooth_eps / n
                b_sol = b_sol / b_sol.sum()

            # Check for violations (cutting-plane)
            if K < K_max and it % 5 == 0:
                violations = check_violations(y, b_sol, M, n)
                if len(violations) > 0:
                    # Add violated constraints to active set
                    new_active = np.union1d(active_set, violations)
                    new_K = min(len(new_active), K_max)
                    if new_K > len(active_set):
                        # Keep top K_max by violation magnitude
                        conv_yb = np.zeros(2 * n - 1)
                        fft_size = 1 << (2 * n - 2).bit_length()
                        Y = rfft(y, fft_size)
                        B = rfft(b_sol, fft_size)
                        conv_yb = irfft(Y * B, fft_size)[:2 * n - 1]
                        all_candidates = np.unique(np.concatenate([active_set, violations]))
                        if len(all_candidates) > K_max:
                            # Keep the most violated
                            viol_vals = conv_yb[all_candidates]
                            top_idx = np.argsort(viol_vals)[-K_max:]
                            active_set = np.sort(all_candidates[top_idx])
                        else:
                            active_set = np.sort(all_candidates)
                        K = len(active_set)

        # Nesterov update
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_sol
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_sol.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        # Update best
        sc_a = score_norm(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
            best_score_stall = best_score
            stalled = 0
        else:
            stalled += 1

        sc_y = score_norm(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()
            best_score_stall = best_score
            stalled = 0

        # Running average (exponential)
        if avg_count < 500:
            avg_a = (avg_count * avg_a + a) / (avg_count + 1)
            avg_count += 1
        else:
            avg_a = 0.999 * avg_a + 0.001 * a

        sc_avg = score_norm(avg_a)
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = avg_a.copy()

        # Adaptive: if stalled, try larger active set or reset
        if stalled > 200 and K < K_max:
            K = min(K + 50, K_max)
            b_y = fast_convolve(y)
            active_set = np.argsort(b_y)[-K:][::-1]
            active_set = np.sort(active_set)
            stalled = 0
        elif stalled > 500:
            # Reset to best + small perturbation
            a = best_a + 0.01 * np.random.default_rng(it).normal(0, 1, n)
            a = normalize(a)
            y = a.copy()
            b_y = fast_convolve(y)
            K = K_init
            active_set = np.argsort(b_y)[-K:][::-1]
            active_set = np.sort(active_set)
            stalled = 0

        if (it + 1) % 500 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 10:
                print(f"    NLP-Adaptive iter={it+1} ({elapsed:.0f}s): "
                      f"score={best_score:.8f} K={K} stalled={stalled}")
                last_print = elapsed

    return best_a, best_score


def softmax_from_z(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Log of a (inverse softmax)."""
    return np.log(np.maximum(a, 1e-300))


def lse_obj_grad(z, T, n):
    """LogSumExp objective and gradient w.r.t. z for softmax parameterization."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
        Tb = np.clip(Tb, None, 700)
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp

    # Gradient: 2 * sum_k w_k * P_k @ a
    # (P_k @ a)[m] = a[k-m] if valid
    # So grad_a[m] = 2 * sum_k w_k * a[k-m]
    # This is 2 * cross_correlation(w, a)[m]
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lse_lbfgs_refine(a0, time_limit, T_start=80.0, T_min=0.2, n_stages=20):
    """Adaptive LSE L-BFGS refinement."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_norm(a0)
    T = T_start
    stagnant = 0

    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        iters = max(10, int(remaining * 8 / (n_stages - stage)))

        def func(z_val):
            obj, _ = lse_obj_grad(z_val, T, n)
            return obj

        def jac(z_val):
            _, grad = lse_obj_grad(z_val, T, n)
            return grad

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': iters * 300, 'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(z)
            sc = score_norm(a_check)

            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()
                stagnant = 0
            else:
                stagnant += 1

            if stagnant >= 3:
                T = max(T, T * 1.3)
                stagnant = 0
            else:
                T = max(T_min, T * 0.7)

        except Exception:
            T = max(T_min, T * 0.8)

        if (stage + 1) % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage={stage+1}/{n_stages} T={T:.2f}: "
                  f"score={sc:.8f} best={best_score:.8f} ({elapsed:.0f}s)")

    return best_a, best_score


def coordinate_descent(a0, time_limit, n_sweeps=100):
    """Coordinate descent polishing. Cycle through coordinates, try local moves."""
    n = len(a0)
    a = normalize(a0.copy())
    best_a = a.copy()
    best_score = score_norm(a)
    t0 = time.monotonic()

    # Precompute which k positions each coordinate affects
    # Coordinate i affects b[k] for k in [i, i+n-1]
    
    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_limit:
            break

        # Shuffle coordinate order
        order = np.random.default_rng(sweep * 42).permutation(n)

        improved = False
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[i]
            # Try a few multipliers
            for mult in [0.5, 0.75, 1.0, 1.25, 1.5, 2.0]:
                trial = a.copy()
                trial[i] = old_val * mult
                if trial[i] < 0:
                    continue
                trial = normalize(trial)
                sc = score_norm(trial)
                if sc < best_score - 1e-12:
                    a = trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break

        if not improved and sweep > 10:
            # Try random perturbations
            for _ in range(20):
                idx = np.random.default_rng().integers(0, n)
                for mult in [0.3, 0.6, 1.5, 3.0]:
                    trial = a.copy()
                    trial[idx] *= mult
                    trial = normalize(trial)
                    sc = score_norm(trial)
                    if sc < best_score - 1e-12:
                        a = trial
                        best_score = sc
                        best_a = a.copy()
                        break

        if (sweep + 1) % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc sweep={sweep+1} ({elapsed:.0f}s): score={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):
    """Generate structured initial sequences."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif choice == 2:
        a = np.exp(-2 * (x - 0.5) ** 2 / 0.1)
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    elif choice == 4:
        a = 0.5 + 0.25 * np.cos(3 * 2 * np.pi * x) + 0.15 * np.cos(7 * 2 * np.pi * x)
    else:
        a = np.sin(np.pi * x) * np.cos(np.pi * x / 3) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    return normalize(f_interp)

    a_out1, _ = nesterov_lp_adaptive(best_seq, t1 * 0.7, solver, K_init=150, K_max=500)
    try_update(a_out1, "nlp_K150")

    remaining = global_deadline - time.monotonic()
    if remaining > 30:
        a_out2, _ = nesterov_lp_adaptive(best_seq, remaining * 0.6, solver, K_init=300, K_max=800)
        try_update(a_out2, "nlp_K300")

    print(f"  Phase 1 done: best={best_score:.8f}")

    # === Phase 2: LSE-LBFGS refinement - 20% time ===
    remaining = global_deadline - time.monotonic()
    t2 = 0.20 * (budget_s - 20)
    if remaining > 30 and t2 > 20:
        print(f"\n=== Phase 2: LSE-LBFGS, budget={min(t2, remaining*0.8):.0f}s ===")
        a_lse1, _ = lse_lbfgs_refine(best_seq, min(t2, remaining * 0.5), T_start=100.0, T_min=0.2, n_stages=20)
        try_update(a_lse1, "lse_high_T")

        remaining = global_deadline - time.monotonic()
        if remaining > 20:
            a_lse2, _ = lse_lbfgs_refine(best_seq, min(t2, remaining * 0.5), T_start=20.0, T_min=0.1, n_stages=15)
            try_update(a_lse2, "lse_low_T")

    print(f"  Phase 2 done: best={best_score:.8f}")

    # === Phase 3: Coordinate descent - 15% time ===
    remaining = global_deadline - time.monotonic()
    t3 = min(0.15 * (budget_s - 20), remaining * 0.7)
    if t3 > 15:
        print(f"\n=== Phase 3: Coordinate Descent, budget={t3:.0f}s ===")
        a_cd, _ = coordinate_descent(best_seq, t3)
        try_update(a_cd, "coord_desc")

    print(f"  Phase 3 done: best={best_score:.8f}")

    # === Phase 4: Multi-length exploration - 10% time ===
    remaining = global_deadline - time.monotonic()
    t4_total = min(0.10 * (budget_s - 20), remaining * 0.8)
    if t4_total > 30:
        print(f"\n=== Phase 4: Multi-length, budget={t4_total:.0f}s ===")
        lengths_to_try = [n_base + 300, n_base - 300, n_base + 600]
        for n_try in lengths_to_try:
            if n_try < 200:
                continue
            remaining = global_deadline - time.monotonic()
            if remaining < 30:
                break
            t_try = min(t4_total / 3.0, remaining * 0.6)
            if t_try < 15:
                break
            print(f"  Trying n={n_try}, budget={t_try:.0f}s")
            init_try = interpolate_sequence(best_seq, n_try)
            a_try, _ = nesterov_lp_adaptive(init_try, t_try * 0.6, solver, K_init=150, K_max=400)
            try_update(a_try, f"len_{n_try}")

            remaining = global_deadline - time.monotonic()
            if remaining > 15:
                a_try2, _ = coordinate_descent(a_try, min(remaining * 0.4, 30))
                try_update(a_try2, f"len_{n_try}_cd")

    print(f"  Phase 4 done: best={best_score:.8f}")

    # === Phase 5: Final polish - 5% time ===
    remaining = global_deadline - time.monotonic()
    if remaining > 15:
        print(f"\n=== Phase 5: Final polish, budget={remaining:.0f}s ===")
        # Quick Nesterov from best
        a_f1, _ = nesterov_lp_adaptive(best_seq, min(remaining * 0.4, 60), solver, K_init=200, K_max=500)
        try_update(a_f1, "final_nlp")

        remaining = global_deadline - time.monotonic()
        if remaining > 10:
            a_f2, _ = coordinate_descent(best_seq, min(remaining * 0.5, 30), n_sweeps=50)
            try_update(a_f2, "final_cd")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL: score={best_score:.8f} (n={len(best_seq)})")
    print(f"Time used: {time.monotonic() - (global_deadline - (budget_s - 20)):.0f}s")
    return [float(x) for x in best_seq]
```
