def qp_peak_shave(a0, time_limit, n_iters_max=500, seed=42):
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    
    for it in range(n_iters_max):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_conv(a)
        b_max = b.max()
        
        # Adaptive temperature
        T = 1.0 + 10.0 * (it / n_iters_max)
        
        # Softmax weights
        w = np.exp(b / (T * b_max + 1e-15))
        w /= w.sum()
        
        # Solve QP
        a_qp = solve_qp_hankel(w, n, max_iters=50, tol=1e-10)
        
        if a_qp is not None:
            # Blend with current solution
            step = 0.3 + 0.5 * (it / n_iters_max)  # increase step over time
            a_new = (1 - step) * a + step * a_qp
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new / s_new
                sc = fast_score(a_new)
                if sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                a = a_new
        else:
            # Fallback: gradient step
            grad = hankel_matvec(w, a) * 2
            eta = 0.1 / (1 + it * 0.01)
            a_new = a - eta * grad
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new > 1e-15:
                a_new = a_new / s_new
                sc = fast_score(a_new)
                if sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                a = a_new
        
        if it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    QP iter {it} ({elapsed:.0f}s): best={best_score:.6f}, T={T:.1f}")
    
    return best_a, best_score

def mirror_descent(a0, time_limit, eta0=0.05, decay=0.6, k_min=1, k_max=15, k_ramp=300, seed=42):
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall = 0
    
    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_conv(a)
        b_max = b.max()
        b_mean = b.mean()
        
        frac = min(1.0, t / float(k_ramp))
        k = max(k_min, min(k_max, int(k_min + frac * (k_max - k_min))))
        
        top_idx = np.argsort(b)[::-1][:k]
        excess = b[top_idx] - b_mean
        if excess.max() < 1e-15:
            weights = np.ones(k) / k
        else:
            excess_shift = excess - excess.max()
            weights = np.exp(excess_shift) / np.exp(excess_shift).sum()
        
        # Subgradient
        grad = np.zeros(n)
        for ii in range(k):
            m = int(top_idx[ii])
            w_m = weights[ii] * 2.0
            j_start = max(0, m - n + 1)
            j_end = min(n - 1, m)
            js = np.arange(j_start, j_end + 1)
            grad[js] += w_m * a[m - js]
        
        eta = eta0 / (1.0 + t / 300.0) ** decay
        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a + 0.02 * rng.standard_normal(n)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            a_new = a_new / max(s_new, 1e-15)
        else:
            a_new = a_new / s_new
        a = a_new
        
        # Check score
        interval = 3 if t < 500 else (10 if t < 5000 else 30)
        if t % interval == 0:
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        
        if stall > 300:
            noise = min(0.05, 0.005 * (stall / 50.0))
            a = a + noise * rng.standard_normal(n)
            a = np.maximum(a, 0)
            a = a / a.sum()
            stall = 0
    
    return best_a, best_score
