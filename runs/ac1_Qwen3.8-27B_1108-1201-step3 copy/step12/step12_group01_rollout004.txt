"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast multiplicative weights (mirror descent) with adaptive step size - primary optimizer
2. Top-k LogSumExp gradient descent with Adam for smooth refinement
3. Multi-scale peak shaving for final polish
4. Multi-start diversity with multiplicative weights restarts
5. Adaptive multi-length exploration
Key improvement: replaces expensive LP-based Nesterov with O(n log n) multiplicative weights
allowing 100x more iterations in the same time budget.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def normalize_positive(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s


def multiplicative_weights_optimize(a0, time_limit, eta_start=None, eta_end=None,
                                     topk=5, use_avg=True):
    """
    Fast multiplicative weights (mirror descent on simplex) for min max_k conv(a,a)[k].
    Each iteration is O(n log n) due to FFT.
    """
    n = len(a0)
    a = normalize_positive(np.asarray(a0, dtype=np.float64))
    if eta_start is None:
        eta_start = n * 0.8
    if eta_end is None:
        eta_end = n * 0.01

    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()

    # Running average buffer
    avg_buf = np.zeros((500, n), dtype=np.float64)
    avg_count = 0

    # Estimate max iterations: each iter ~ 0.1ms for n=1600
    est_iters = int(time_limit * 8000)
    est_iters = min(est_iters, 50_000_000)

    no_improve = 0
    check_interval = max(100, n // 10)

    for t in range(1, est_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        # Adaptive eta: cosine schedule
        progress = min(1.0, t / est_iters)
        eta = eta_end + 0.5 * (eta_start - eta_end) * (1 + np.cos(np.pi * progress))

        # Compute autocorrelation
        c = fast_autocorr(a)

        # Find top-k peaks
        k_use = min(topk, len(c))
        if k_use == 1:
            k_star = int(np.argmax(c))
            g = np.zeros(n)
            js = max(0, k_star - n + 1)
            je = min(n - 1, k_star)
            idx = np.arange(js, je + 1)
            g[idx] = 2.0 * a[k_star - idx]
        else:
            top_idx = np.argsort(c)[::-1][:k_use]
            top_vals = c[top_idx]
            # Softmax weights over top peaks
            tw = top_vals - top_vals.max()
            tw = np.exp(tw * 2.0)
            tw = tw / tw.sum()
            g = np.zeros(n)
            for ii in range(k_use):
                k = top_idx[ii]
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                g[idx] += 2.0 * tw[ii] * a[k - idx]

        # Mirror descent update: a[j] <- a[j] * exp(-eta * g[j])
        # Clip eta*g to prevent overflow
        exponents = -eta * g
        exponents = np.clip(exponents, -50, 50)
        a_new = a * np.exp(exponents)
        s = a_new.sum()
        if s < 1e-300:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / s
        a = a_new

        # Check score periodically
        if t % check_interval == 0:
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

            # Check running average
            if use_avg and avg_count >= 50:
                avg_a = avg_buf[:avg_count].mean(axis=0)
                avg_a = normalize_positive(avg_a)
                avg_score = 2.0 * n * fast_autocorr(avg_a).max()
                if avg_score < best_score:
                    best_score = avg_score
                    best_a = avg_a.copy()
                    no_improve = 0

            # Store in average buffer
            avg_count = (avg_count + 1) % 500
            avg_buf[avg_count] = a

        # Adaptive restart if stuck for too long
        if no_improve > 2000 and t > 5000:
            a = best_a.copy()
            no_improve = 0
            # Reduce eta for finer search
            eta_start = eta_start * 0.5
            eta_end = eta_end * 0.3
            est_iters = t + int(est_iters * 0.5)

        if t % 100000 == 0:
            elapsed = time.monotonic() - t0
            print(f"        MW t={t} ({elapsed:.0f}s): score={best_score:.6f}, eta={eta:.1f}")

    # Final check
    score = 2.0 * n * fast_autocorr(a).max()
    if score < best_score:
        best_score = score
        best_a = a.copy()

    if use_avg and avg_count > 50:
        avg_a = avg_buf[:avg_count].mean(axis=0)
        avg_a = normalize_positive(avg_a)
        avg_score = 2.0 * n * fast_autocorr(avg_a).max()
        if avg_score < best_score:
            best_score = avg_score
            best_a = avg_a.copy()

    return best_a, best_score


def multiplicative_weights_anneal(a0, time_limit, n_phases=5, topk=5):
    """
    Multi-phase multiplicative weights with decreasing eta for convergence.
    Each phase uses a smaller eta range for finer refinement.
    """
    n = len(a0)
    a = normalize_positive(np.asarray(a0, dtype=np.float64))
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()

    for phase in range(n_phases):
        if time.monotonic() - t0 > time_limit - 5:
            break
        phase_time = (time_limit - (time.monotonic() - t0)) / (n_phases - phase)
        # Decreasing eta range per phase
        scale = 0.5 ** phase
        eta_s = n * 0.8 * scale
        eta_e = n * 0.05 * scale

        a_out, sc = multiplicative_weights_optimize(
            a, phase_time, eta_start=eta_s, eta_end=eta_e, topk=topk
        )
        if sc < best_score:
            best_score = sc
            best_a = a_out.copy()
        a = a_out
        elapsed = time.monotonic() - t0
        print(f"        MW-anneal phase {phase+1}/{n_phases} ({elapsed:.0f}s): score={best_score:.6f}")

    return best_a, best_score


def compute_topk_grad(a, T, topk):
    """Compute top-k LSE objective and gradient w.r.t. a."""
    n = len(a)
    c = fast_autocorr(a)
    k_use = min(topk, len(c))
    top_idx = np.argsort(c)[::-1][:k_use]
    top_vals = c[top_idx]

    Tc = T * top_vals
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    lse_val = (max_Tc + np.log(exp_Tc.sum())) / T
    w = exp_Tc / exp_Tc.sum()

    grad_a = np.zeros(n)
    for idx in range(k_use):
        K = top_idx[idx]
        wK = w[idx]
        js = max(0, K - (n - 1))
        je = min(n - 1, K)
        if js <= je:
            idx_range = np.arange(js, je + 1)
            grad_a[idx_range] += 2.0 * wK * a[K - idx_range]

    return lse_val, grad_a


def topk_lse_grad_z(z, T, topk):
    """Objective and gradient in z-space for top-k LSE."""
    a = softmax_fn(z)
    obj, grad_a = compute_topk_grad(a, T, topk)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def adam_topk_lse(a0, time_limit, T_start=50.0, T_end=0.5, n_stages=15, topk=8):
    """Adam-optimized top-k LSE with multi-stage annealing."""
    n = len(a0)
    a_init = normalize_positive(np.asarray(a0, dtype=np.float64))
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit - 5:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        base_lr = 3e-3 * min(1.0, 100.0 / n)
        lr = base_lr * (0.75 ** si)

        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.8, 0.999, 1e-8

        max_it = max(50, int(remaining * 5))

        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 3:
                break

            obj, grad_z = topk_lse_grad_z(z, stage_T, topk)

            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            if (it + 1) % 10 == 0:
                a = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    if (it + 1) % 50 == 0:
                        z = np.log(np.maximum(best_a, 1e-30))

        a_stage = softmax_fn(z)
        score_stage = 2.0 * n * fast_autocorr(a_stage).max()
        if score_stage < best_score:
            best_score = score_stage
            best_a = a_stage.copy()

        if si % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"        LSE stage {si}/{n_stages} (T={stage_T:.1f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def peak_shaving(a0, time_limit, seed=123):
    """
    Multi-scale peak shaving with adaptive step sizes.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize_positive(np.asarray(a0, dtype=np.float64))
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit - 3:
        b = fast_autocorr(a)
        bmax = b.max()
        threshold = bmax * 0.995
        top_peaks = np.where(b >= threshold)[0]
        if len(top_peaks) == 0:
            top_peaks = np.argsort(b)[::-1][:10]

        improved = False
        for k in top_peaks[:15]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            max_contrib = contrib.max()
            if max_contrib < 1e-15:
                continue

            src = idx[np.argmax(contrib)]

            # Compute danger zone
            danger = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:15]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                danger[js2:je2 + 1] = True

            safe = np.where(~danger)[0]
            safe = safe[safe != src]
            if len(safe) == 0:
                safe = np.where(np.arange(n) != src)[0]

            safe_sorted = safe[np.argsort(a[safe])]
            dst_pool = safe_sorted[:max(3, len(safe_sorted) // 5)]
            if len(dst_pool) == 0:
                continue
            dst = dst_pool[rng.integers(len(dst_pool))]

            for frac in [0.002, 0.005, 0.01, 0.02, 0.04, 0.07, 0.1]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = normalize_positive(a_trial)
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            n_pert = rng.integers(1, 4)
            a_trial = a.copy()
            for _ in range(n_pert):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a_trial[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.02)
                    amt = a_trial[i] * frac
                    a_trial[i] -= amt
                    a_trial[j] += amt
            a_trial = normalize_positive(a_trial)
            score = 2.0 * n * fast_autocorr(a_trial).max()
            if score < best_score:
                a = a_trial
                best_score = score
                best_a = a.copy()
                stall = 0

            if stall > 200:
                break

    return best_a, best_score


def gradient_descent_nesterov(a0, time_limit, topk=5):
    """
    Nesterov-accelerated gradient descent in the probability simplex.
    Uses the subgradient of max_k conv(a,a)[k] directly.
    """
    n = len(a0)
    a = normalize_positive(np.asarray(a0, dtype=np.float64))
    a_prev = a.copy()
    b = a.copy()  # Nesterov extrapolation point
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()

    t_gamma = 1.0
    base_lr = 0.1 / n

    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        t_gamma += 1.0
        gamma = 2.0 / (t_gamma + 1.0)

        # Nesterov extrapolation
        b = a + (t_gamma - 1.0) / (t_gamma + 1.0) * (a - a_prev)
        b = normalize_positive(b)

        # Compute subgradient at b
        c = fast_autocorr(b)
        k_use = min(topk, len(c))
        top_idx = np.argsort(c)[::-1][:k_use]
        top_vals = c[top_idx]
        tw = top_vals - top_vals.max()
        tw = np.exp(tw)
        tw = tw / tw.sum()

        g = np.zeros(n)
        for ii in range(k_use):
            k = top_idx[ii]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            g[idx] += 2.0 * tw[ii] * b[k - idx]

        # Projected gradient step
        a_prev = a.copy()
        lr = base_lr / np.sqrt(t_gamma)
        a_new = a - lr * g
        a = normalize_positive(a_new)

        if t % 500 == 0:
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

    score = 2.0 * n * fast_autocorr(a).max()
    if score < best_score:
        best_score = score
        best_a = a.copy()

    return best_a, best_score


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 10)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 3
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 2
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.7 + 0.2 * np.cos(2 * x) + 0.1 * np.cos(5 * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-5 * (x - 0.3) ** 2) + 0.5 * np.exp(-5 * (x - 0.7) ** 2)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) * np.cos(np.pi * x / 3) ** 2
    elif choice == 7:
        # Exponential decay (skewed)
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * x)
    elif choice == 8:
        # Bimodal
        x = np.linspace(0, 1, n)
        a = np.exp(-20 * (x - 0.25) ** 2) + np.exp(-20 * (x - 0.75) ** 2)
    else:
        # Power law
        x = np.linspace(0.01, 1, n)
        a = x ** 0.5
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def perturbed_init(a0, rng, scale=0.1):
    """Create a perturbed version of a sequence for multi-start."""
    a = normalize_positive(np.asarray(a0, dtype=np.float64))
    noise = rng.normal(0, scale, len(a))
    a_new = a + noise * (a + 1e-6)
    a_new = np.maximum(a_new, 0)
    return normalize_positive(a_new)

            try_update(a_out, f"nest_gd_n{n_base}")

    # === Phase 5: Multi-start with MW ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 5: Multi-start ({remaining:.0f}s left) ===")
    t5_total = min(remaining * 0.15, 180.0)
    if t5_total > 30:
        n_starts = 3
        t5_each = t5_total / n_starts
        for si in range(n_starts):
            if time.monotonic() - t_global_start > time_budget - 60:
                break
            pert = perturbed_init(best_seq, rng, scale=0.03 * (si + 1))
            a_out, _ = multiplicative_weights_optimize(
                pert, t5_each,
                eta_start=n_base * 0.5 * (0.7 ** si),
                eta_end=n_base * 0.02,
                topk=4
            )
            try_update(a_out, f"multi_start_{si}")

    # === Phase 6: Try longer length ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 60:
        n_long = n_base + 200
        t6 = min(remaining * 0.18, 200.0)
        if t6 > 30:
            print(f"=== Phase 6: MW n={n_long} ({remaining:.0f}s left) ===")
            init_long = interpolate_sequence(best_seq, n_long)
            a_out, _ = multiplicative_weights_anneal(init_long, t6, n_phases=3, topk=5)
            try_update(a_out, f"mw_n{n_long}")

    # === Phase 7: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 7: Peak shaving ({remaining:.0f}s left) ===")
    t7 = min(remaining * 0.12, 100.0)
    if t7 > 10:
        a_out, _ = peak_shaving(best_seq, t7, seed=456)
        try_update(a_out, "peak_shave_1")

    # === Phase 8: Final MW polish ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 20:
        t8 = min(remaining * 0.20, 100.0)
        if t8 > 10:
            print(f"=== Phase 8: Final MW polish ({remaining:.0f}s left) ===")
            a_out, _ = multiplicative_weights_optimize(
                best_seq, t8,
                eta_start=n_base * 0.05, eta_end=n_base * 0.002,
                topk=3
            )
            try_update(a_out, "final_mw")

    # === Phase 9: Final peak shave ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 10:
        t9 = min(remaining * 0.5, 50.0)
        if t9 > 5:
            a_out, _ = peak_shaving(best_seq, t9, seed=789)
            try_update(a_out, "peak_shave_2")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
