def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def build_conv_matrix(a_fixed):
    """Build the (2n-1) x n convolution matrix where (T@b) = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver, max_iters=100000):
    """Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0. Returns (b, M, dual_vars)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    cons_M = T @ b_var <= M_var
    constraints = [cons_M, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver, max_iter=max_iters)
    except Exception:
        pass
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, max_iter=max_iters)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None, None
    b_val = b_val / s

    except Exception:

        x = np.abs(np.arange(n) - center) / center
        a = (1 - x**2) ** p
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 0:
            candidates.append(a / s)
    
    # Random
    for _ in range(5):
        a = rng.exponential(1.0, size=n)
        candidates.append(a / a.sum())
    
    # Find the best
    best_a = None
    best_score = float('inf')
    for a in candidates:
        a = np.maximum(a, 0)
        s = a.sum()
        if s < 1e-15:
            continue
        a = a / s
        score = 2.0 * n * fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_a, best_score


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True, warmup_frac=0.1):
    """
    Nesterov-accelerated LP best-response with improved scheduling.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # EMA tracking
    ema_a = a.copy()
    ema_alpha = 0.03
    
    # For averaging: maintain a list of recent iterates
    recent_iterates = []
    
    iter_count = 0
    stall_count = 0
    
    # Determine warmup iterations
    # During warmup, use larger steps to explore
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_star, M_val, duals = solve_lp_best_response(y, solver)
        if b_star is None:
            continue
        iter_count += 1
        
        # Adaptive Nesterov schedule
        if t <= 50:
            # Warmup: larger steps
            alpha_t = 1.0 / (1 + t * 0.1)
        else:
            # Standard Nesterov
            alpha_t = 2.0 / (t + 2.0)
        
        # Nesterov update for y
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_new_s = y_new.sum()
        if y_new_s < 1e-15:
            y_new = y.copy()
            y_new_s = 1.0
        y_new = y_new / y_new_s
        
        # Nesterov momentum for a
        if t == 1:
            a_new = b_star.copy()
        else:
            beta_t = (t - 1.0) / (t + 2.0)
            a_new = y_new + beta_t * (y_new - y)
            a_new = np.maximum(a_new, 0)
            a_s = a_new.sum()
            if a_s < 1e-15:
                a_new = y_new.copy()
                a_s = 1.0
            a_new = a_new / a_s
        
        y = y_new
        a = a_new
        
        # Track best
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0
        
        # EMA
        ema_a = (1 - ema_alpha) * ema_a + ema_alpha * a_new
        
        # Periodically check EMA and running average
        if t % 10 == 0:
            score_ema = 2.0 * n * fast_autocorr(ema_a).max()
            if score_ema < best_score:
                best_score = score_ema
                best_a = ema_a.copy()
                stall_count = 0
        
        # Keep recent iterates for averaging
        if t > 50:
            recent_iterates.append(a_new.copy())
            if len(recent_iterates) > 100:
                recent_iterates.pop(0)
        
        if t % 50 == 0 and len(recent_iterates) >= 50:
            avg_a = np.mean(recent_iterates, axis=0)
            avg_a = np.maximum(avg_a, 0)
            ss = avg_a.sum()
            if ss > 1e-15:
                avg_a = avg_a / ss
                score_avg = 2.0 * n * fast_autocorr(avg_a).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_a.copy()
                    stall_count = 0
        
        # Adapt EMA rate
        if t > 100 and stall_count > 500:
            ema_alpha = min(0.15, ema_alpha * 1.02)
        
        if t % 1000 == 0 and verbose:
            elapsed = time.monotonic() - t0
            print(f"    NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.6f}, iters/s={iter_count/elapsed:.1f}")
        
        # Early stopping
        if stall_count > 5000 and t > 500:
            if verbose:
                print(f"    Early stop at t={t}")
            break
    
    return best_a, best_score, iter_count


def subgradient_refine(a0, time_limit, seed=42):
    """
    Subgradient method with randomized lag selection and averaging.
    Different from Nesterov LP: uses direct subgradients of the max function.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    avg_a = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    n_total = 20000  # target number of iterations
    
    for it in range(1, n_total + 1):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute convolution
        b = fast_autocorr(a)
        max_b = b.max()
        current_score = 2.0 * n * max_b
        
        # Learning rate: O(1/sqrt(t))
        eta = 1.0 / np.sqrt(it + 1)
        
        # Randomized subgradient: pick lag with probability proportional to exp(beta * b[k])
        beta = 20.0  # temperature parameter
        weights = b - max_b
        weights = np.exp(weights * beta)
        weights /= weights.sum()
        k = rng.choice(len(b), p=weights)
        
        # Subgradient of conv(a,a) at lag k
        grad = np.zeros(n)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            other = k - j
            if j == other:
                grad[j] += a[j]  # 2*a[j] but we count it once since j==other
            else:
                grad[j] += a[other]
                grad[other] += a[j]
        
        # Wait, let me recompute. c[k] = sum_j a[j]*a[k-j]
        # dc[k]/da[i] = a[k-i] + (a[i] if k-i==i else 0) + ... 
        
        # Actually: c[k] = sum_{j: valid} a[j] * a[k-j]
        # dc[k]/da[i]: 
        #   - term where j=i: a[k-i] (if k-i is valid)
        #   - term where k-j=i, i.e., j=k-i: a[k-i] (if k-i is valid and k-i != i)
        #   - if k-i == i (i.e., k=2i): both terms give a[i], so total is 2*a[i]... 
        #     wait no, it's a[i]*a[i] and the derivative is 2*a[i]
        #   Actually if k=2i, then j=i gives a[i]*a[i], derivative = 2*a[i]
        #   If k != 2i and both i and k-i are valid: derivative = a[k-i] + a[k-i] = 2*a[k-i]
        #   If only i is valid (k-i out of range): derivative = 0 (the term j=i doesn't exist)
        #   If only k-i is valid (i out of range): derivative = 0
        
        # Let me redo this properly:
        # c[k] = sum_{j=max(0,k-n+1)}^{min(n-1,k)} a[j] * a[k-j]
        # dc[k]/da[i] = a[k-i] * 1[i in range] + a[k-i] * 1[k-i in range and k-i != i]
        #              + 2*a[i] * 1[k=2i]  ... no this is getting confusing
        
        # Let me just use: dc[k]/da[i] = 2*a[k-i] if 0<=k-i<n and k-i != i
        #                               a[i] if k == 2*i (and 0<=i<n)
        #                               0 otherwise
        # Hmm, that's not right either. Let me just compute it directly.
        
        # c[k] involves terms a[j]*a[k-j] for valid j.
        # The derivative with respect to a[i] picks out terms where i appears:
        #   - a[i] appears as a[j] when j=i: contributes a[k-i] if k-i is in [0,n-1]
        #   - a[i] appears as a[k-j] when k-j=i, i.e., j=k-i: contributes a[k-i] if k-i is in [0,n-1]
        # If i = k-i (i.e., k=2i), these are the same term a[i]*a[i], and the derivative is 2*a[i].
        # If i != k-i, these are two different terms, each contributing a[k-i].
        # So: dc[k]/da[i] = 
        #   2*a[i] if k=2i
        #   2*a[k-i] if k-i valid and k-i != i
        #   a[k-i] if only one of i or k-i is valid (can't happen since validity is symmetric)
        #   0 otherwise
        
        # Wait, I think the issue is:
        # c[k] = sum_j a[j] * a[k-j]
        # If I differentiate w.r.t. a[i]:
        #   d/da[i] of a[j]*a[k-j] = delta_{j,i} * a[k-j] + a[j] * delta_{k-j,i}
        #                            = delta_{j,i} * a[k-j] + a[j] * delta_{j,k-i}
        # Summing over j:
        #   = a[k-i] (from j=i, if valid) + a[k-i] (from j=k-i, if valid)
        #   But if i = k-i (k=2i), these are the same j, so we get:
        #   = a[i] + a[i] = 2*a[i]
        
        # So: dc[k]/da[i] = a[k-i] * [k-i valid] + a[k-i] * [k-i valid] 
        #                    BUT if k=2i, it's 2*a[i], not a[k-i]+a[k-i]
        #                    Actually it IS a[k-i] + a[k-i] = 2*a[k-i] = 2*a[i] when k=2i.
        #                    So the formula is simply: 2*a[k-i] if k-i is in [0,n-1], else 0.
        
        # Let me verify: if k=0, then c[0] = a[0]*a[0]. dc[0]/da[0] = 2*a[0]. 
        # Formula: 2*a[0-0] = 2*a[0]. ✓
        # If k=1, c[1] = a[0]*a[1] + a[1]*a[0] = 2*a[0]*a[1]. 
        # dc[1]/da[0] = 2*a[1]. Formula: 2*a[1-0] = 2*a[1]. ✓
        # dc[1]/da[1] = 2*a[0]. Formula: 2*a[1-1] = 2*a[0]. ✓
        # dc[1]/da[2] = 0. Formula: 2*a[1-2] = 2*a[-1] -> invalid, so 0. ✓
        
        # Great, so the subgradient is simply 2*a[(k-i) mod something]... no, just:
        # grad[i] = 2*a[k-i] if 0 <= k-i <= n-1, else 0
        
        grad = np.zeros(n)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            grad[j] = 2.0 * a[k - j]
        
        # Gradient step
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
            s_new = 1.0
        a_new = a_new / s_new
        
        # Update average (Polyak averaging)
        avg_a = (it / (it + 1)) * avg_a + (1 / (it + 1)) * a_new
        
        a = a_new
        
        # Track best
        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()
        
        score_avg = 2.0 * n * fast_autocorr(avg_a).max()
        if score_avg < best_score:
            best_score = score_avg
            best_a = avg_a.copy()
        
        if it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG it={it} ({elapsed:.0f}s): best={best_score:.6f}, cur={current_score:.6f}")
    
    return best_a, best_score


def qp_refine(a0, time_limit, n_active_max=50):
    """QP refinement on active constraint set."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    b = fast_autocorr(a)
    max_b = b.max()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * max_b
    
    for round_num in range(30):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Active set: lags within 1% of max
        threshold = max_b * 0.99
        active_lags = np.where(b >= threshold)[0]
        if len(active_lags) > n_active_max:
            active_lags = np.argsort(-b[active_lags])[:n_active_max] + np.where(b >= threshold)[0][:n_active_max]
            active_lags = np.unique(active_lags)
            active_lags = active_lags[:n_active_max]
        
        a_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        
        constraints = [cp.sum(a_var) == 1, M_var >= 0]
        for k in active_lags:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            other_idx = k - idx
            q_expr = cp.sum(a_var[idx] * a_var[other_idx])
            constraints.append(q_expr <= M_var)
        
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=cp.CLARABEL)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS)
            except Exception:
                try:
                    prob.solve(solver=cp.SCS)
                except Exception:
                    break
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            break
        
        a_new = np.asarray(a_var.value, dtype=np.float64)
        a_new = np.maximum(a_new, 0)
        a_new_s = a_new.sum()
        if a_new_s < 1e-15:
            break
        a_new = a_new / a_new_s
        
        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            b = fast_autocorr(a)
            max_b = b.max()
            print(f"    QP round {round_num}: {best_score:.6f}")
        else:
            # Try mixing
            improved = False
            for frac in [0.2, 0.4, 0.6, 0.8]:
                a_mix = (1 - frac) * a + frac * a_new
                a_mix = np.maximum(a_mix, 0)
                a_mix_s = a_mix.sum()
                if a_mix_s < 1e-15:
                    continue
                a_mix = a_mix / a_mix_s
                score_mix = 2.0 * n * fast_autocorr(a_mix).max()
                if score_mix < best_score:
                    best_score = score_mix
                    best_a = a_mix.copy()
                    a = a_mix
                    b = fast_autocorr(a)
                    max_b = b.max()
                    improved = True
                    break
            if not improved and round_num >= 5:
                break
    
    return best_a, best_score
