"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-phase optimization:
1. Active-set LSE with L-BFGS (FFT-accelerated gradients)
2. Nesterov-accelerated LP best-response (warm-started)
3. Peak mirror descent with Polyak step and averaging
Innovations: FFT-based cross-correlation for O(n log n) gradients,

import numpy as np

from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_corr(w, a):
    """Compute result[i] = sum_k w[k]*a[k-i] via FFT (linear cross-correlation)."""
    n = len(a)
    m = len(w)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(np.conj(W) * A, fft_size)
    return corr[:n]


def score_normalized(a):
    """Score for normalized (sum=1) sequence: 2*n*max(conv(a,a))."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def lse_obj_grad(z, beta, n, top_k=None):
    """Compute LSE objective and gradient w.r.t. z.
    
    Uses active set: only top_k peaks contribute.
    """
    a = softmax_from_z(z)
    b = fast_convolve(a)
    
    if top_k is not None and top_k < len(b):
        # Only use top_k peaks
        top_idx = np.argpartition(b, -top_k)[-top_k:]
        b_active = b[top_idx]
        max_b = b_active.max()
        exp_b = np.exp(beta * (b_active - max_b))
        w_active = exp_b / exp_b.sum()
        log_sum_exp = max_b + np.log(exp_b.sum())
        
        # Build full w vector (sparse)
        w = np.zeros(len(b))
        w[top_idx] = w_active
    else:
        max_b = b.max()
        exp_b = np.exp(beta * (b - max_b))
        w = exp_b / exp_b.sum()
        log_sum_exp = max_b + np.log(exp_b.sum())
    
    obj = (2.0 * n / beta) * log_sum_exp
    
    # Gradient: (4n/beta) * cross_corr(w, a)
    grad_a = (4.0 * n / beta) * fast_cross_corr(w, a)
    
    # Project to simplex tangent space (for softmax parametrization)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return obj, grad_z


def lse_lbfgs(a0, time_limit, beta_start=20.0, beta_end=500.0, 
              n_stages=20, iters_per_stage=50, top_k_start=10, top_k_end=None):
    """Multi-stage LSE with L-BFGS and progressive active set."""
    n = len(a0)
    if top_k_end is None:
        top_k_end = 2 * n - 1
    
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)
    
    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Progress from 0 to 1
        frac = stage / max(1, n_stages - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        top_k = int(top_k_start + frac * (top_k_end - top_k_start))
        top_k = min(top_k, 2 * n - 1)
        
        def func(z_val):
            obj, _ = lse_obj_grad(z_val, beta, n, top_k)
            return obj
        
        def jac(z_val):
            _, grad = lse_obj_grad(z_val, beta, n, top_k)
            return grad
        
        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                          options={'maxiter': iters_per_stage, 'ftol': 1e-16,
                                   'gtol': 1e-10, 'maxfun': iters_per_stage * 200,
                                   'disp': False})
            z = res.x
            a_check = softmax_from_z(z)
            sc = score_normalized(a_check)
            
            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()
        except Exception:
            pass
    
    return best_a, best_score

def solve_lp_fast(a_fixed, n, prev_b=None):
    """Solve LP: min M s.t. (a_fixed*b)[k] <= M for all k, sum(b)=1, b>=0.
    
    Uses cvxpy with warm-start if available.
    """
    import cvxpy as cp
    
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build constraint: T @ b_var <= M_var
    T = build_conv_matrix_banded(a_fixed, n)
    
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    
    obj = cp.Minimize(M_var)
    prob = cp.Problem(obj, constraints)
    
    # Try solvers in order
    for solver in [cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                break
        except Exception:
            continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp(a0, time_limit, max_iters=50000):
    """Nesterov-accelerated LP best-response."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_new, M_lp = solve_lp_fast(y, n)
        if b_new is None:
            continue
        
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k
        
        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa
        
        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy
        
        a = a_new
        y = y_new
        
        sc = score_normalized(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    return best_a, best_score


def peak_mirror_descent(a0, time_limit, n_steps=100000):
    """Peak-targeted mirror descent (multiplicative weights) with Polyak step."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    best_a = a.copy()
    best_score = score_normalized(a)
    lower_bound = 2.0 * n / (2.0 * n - 1.0)
    
    t0 = time.monotonic()
    avg_a = np.zeros(n)
    avg_count = 0
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        cur_score = 2.0 * n * max_b
        
        # Find active peaks
        threshold = max_b * 0.99
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        if len(active) > 30:
            top_idx = np.argsort(b[active])[-30:]
            active = active[top_idx]
        
        # Subgradient: average gradient at active peaks
        # grad_a[i] = 2 * sum_k w_k * a[k-i] where w_k = 1/|active| for k in active
        w = np.zeros(len(b))
        w[active] = 1.0 / len(active)
        grad_a = 2.0 * fast_cross_corr(w, a)
        
        # Mirror descent: a_new ∝ a * exp(-eta * grad_a)
        # Use Polyak step
        gap = cur_score - lower_bound
        if gap < 1e-12:
            break
        grad_norm_sq = np.sum(grad_a ** 2)
        if grad_norm_sq < 1e-30:
            break
        eta = gap / (2.0 * grad_norm_sq)
        eta = min(eta, 1.0)
        
        log_a = np.log(np.maximum(a, 1e-300))
        log_a_new = log_a - eta * grad_a
        a_new = softmax_from_z(log_a_new)
        sc_new = score_normalized(a_new)
        
        # Line search if needed
        if sc_new > cur_score * 1.05:
            for _ in range(8):
                eta *= 0.5
                log_a_new = log_a - eta * grad_a
                a_new = softmax_from_z(log_a_new)
                sc_new = score_normalized(a_new)
                if sc_new <= cur_score:
                    break
        
        a = a_new
        
        # Running average (exponential)
        avg_a = (avg_count * avg_a + a) / (avg_count + 1)
        avg_count += 1
        
        if sc_new < best_score:
            best_score = sc_new
            best_a = a.copy()
        
        sc_avg = score_normalized(avg_a)
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = avg_a.copy()
    
    return best_a, best_score


def structured_init(n, rng, mode=0):

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:

One important insight: the peaks of b(k) = (a*a)(k) for non-negative a are likely concentrated around k=0 and k=n-1 (the "edges" of the convolution). The middle of the convolution (around k=n-1) might naturally have higher values because more terms contribute.

b(k) = sum_j a(j) * a(k-j)

For k=0: b(0) = a(0)^2
For k=n-1: b(n-1) = sum_{j=0}^{n-1} a(j) * a(n-1-j) = sum_j a(j)*a(n-1-j)
For k=2(n-1): b(2(n-1)) = a(n-1)^2

The middle value b(n-1) has n terms in the sum, while b(0) has only 1 term. So if a is roughly uniform, b(n-1) ≈ n * (1/n)^2 = 1/n, while b(0) ≈ 1/n^2. So the middle is much larger.

This means the max of b is likely at the center (k≈n-1), and the challenge is to make this as small as possible while keeping sum(a) = 1.

The minimum possible value of b(n-1) = sum_j a(j)*a(n-1-j) subject to sum(a)=1, a>=0 is... if a is uniform, it's 1/n. Can we do better?

"""

2. Nesterov-accelerated LP best-response with warm-starting

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Fast linear convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_corr(w, a):
    """Compute result[i] = sum_k w[k]*a[k-i] via FFT (linear cross-correlation)."""
    n = len(a)
    m = len(w)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    corr = irfft(W * np.conj(A), fft_size)
    return corr[:n]


def score_normalized(a):
    """Score for normalized (sum=1) sequence: 2*n*max(conv(a,a))."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax_from_z(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Inverse of softmax."""
    return np.log(np.maximum(a, 1e-300))


def lse_obj_grad(z, beta, n, top_k):
    """Compute LSE objective and gradient w.r.t. z using active set of top_k peaks."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    m = len(b)

    if top_k is not None and top_k < m:
        top_idx = np.argpartition(b, -top_k)[-top_k:]
        b_active = b[top_idx]
        max_b = b_active.max()
        exp_b = np.exp(beta * (b_active - max_b))
        w_active = exp_b / exp_b.sum()
        log_sum_exp = max_b + np.log(exp_b.sum())
        w = np.zeros(m)
        w[top_idx] = w_active
    else:
        max_b = b.max()
        exp_b = np.exp(beta * (b - max_b))
        w = exp_b / exp_b.sum()
        log_sum_exp = max_b + np.log(exp_b.sum())

    obj = (2.0 * n / beta) * log_sum_exp
    grad_a = (4.0 * n / beta) * fast_cross_corr(w, a)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lse_lbfgs(a0, time_limit, beta_start, beta_end, n_stages, iters_per_stage, top_k_start, top_k_end):
    """Multi-stage LSE with L-BFGS and progressive active set."""
    n = len(a0)
    if top_k_end is None or top_k_end > 2 * n - 1:
        top_k_end = 2 * n - 1

    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)

    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break

        frac = stage / max(1, n_stages - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        top_k = int(top_k_start + frac * (top_k_end - top_k_start))
        top_k = min(top_k, 2 * n - 1)

        def func(z_val):
            obj, _ = lse_obj_grad(z_val, beta, n, top_k)
            return obj

        def jac(z_val):
            _, grad = lse_obj_grad(z_val, beta, n, top_k)
            return grad

        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                          options={'maxiter': iters_per_stage, 'ftol': 1e-16,
                                   'gtol': 1e-10, 'maxfun': iters_per_stage * 200,
                                   'disp': False})
            z = res.x
            a_check = softmax_from_z(z)
            sc = score_normalized(a_check)
            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()
        except Exception:
            pass

    return best_a, best_score


def build_conv_matrix(a_fixed, n):
    """Build the convolution constraint matrix T where (T@b)[k] = (a_fixed*b)[k]."""
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_fast(a_fixed, n):
    """Solve LP: min M s.t. (a_fixed*b)[k] <= M for all k, sum(b)=1, b>=0."""
    import cvxpy as cp

    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed, n)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    obj = cp.Minimize(M_var)
    prob = cp.Problem(obj, constraints)

    for solver in [cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                break
        except Exception:
            continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp(a0, time_limit, max_iters=50000):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_lp = solve_lp_fast(y, n)
        if b_new is None:
            continue

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        sc = score_normalized(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def peak_mirror_descent(a0, time_limit, n_steps=100000):
    """Peak-targeted mirror descent (multiplicative weights) with Polyak step and averaging."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = score_normalized(a)
    lower_bound = 2.0 * n / (2.0 * n - 1.0)

    t0 = time.monotonic()
    avg_a = np.zeros(n)
    avg_count = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        cur_score = 2.0 * n * max_b

        threshold = max_b * 0.99
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        if len(active) > 30:
            top_idx = np.argsort(b[active])[-30:]
            active = active[top_idx]

        w = np.zeros(len(b))
        w[active] = 1.0 / len(active)
        grad_a = 2.0 * fast_cross_corr(w, a)

        gap = cur_score - lower_bound
        if gap < 1e-12:
            break
        grad_norm_sq = np.sum(grad_a ** 2)
        if grad_norm_sq < 1e-30:
            break
        eta = gap / (2.0 * grad_norm_sq)
        eta = min(eta, 1.0)

        log_a = np.log(np.maximum(a, 1e-300))
        log_a_new = log_a - eta * grad_a
        a_new = softmax_from_z(log_a_new)
        sc_new = score_normalized(a_new)

        if sc_new > cur_score * 1.05:
            for _ in range(8):
                eta *= 0.5
                log_a_new = log_a - eta * grad_a
                a_new = softmax_from_z(log_a_new)
                sc_new = score_normalized(a_new)
                if sc_new <= cur_score:
                    break

        a = a_new
        avg_a = (avg_count * avg_a + a) / (avg_count + 1)
        avg_count += 1

        if sc_new < best_score:
            best_score = sc_new
            best_a = a.copy()

        sc_avg = score_normalized(avg_a)
        if sc_avg < best_score:
            best_score = sc_avg
            best_a = avg_a.copy()

    return best_a, best_score


def structured_init(n, rng, mode):
    """Generate diverse structured initializations."""
    x = np.linspace(0, 1, n)
    if mode == 0:
        a = np.sin(np.pi * x) ** 2
    elif mode == 1:
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif mode == 2:
        a = np.exp(-4.0 * (x - 0.5) ** 2)
    elif mode == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    elif mode == 4:
        a = np.ones(n) * (1.0 + 0.3 * np.cos(2 * np.pi * 5 * x))
    elif mode == 5:
        a = 0.5 + 0.2 * np.cos(3 * 2 * np.pi * x) + 0.1 * np.cos(7 * 2 * np.pi * x)
    elif mode == 6:
        a = np.abs(np.sin(np.pi * x * 3)) ** 0.5
    elif mode == 7:
        a = np.exp(-((x - 0.3) ** 2) / 0.02) + np.exp(-((x - 0.7) ** 2) / 0.02) + 0.3
    elif mode == 8:
        freqs = rng.integers(1, 10, size=5)
        phases = rng.uniform(0, 2 * np.pi, size=5)
        amps = rng.uniform(0.1, 0.5, size=5)
        a = np.ones(n) * 0.5
        for i in range(5):
            a += amps[i] * np.cos(2 * np.pi * freqs[i] * x + phases[i])
    else:
        a = np.ones(n)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def local_peak_refine(a0, time_limit, n_peaks=5, max_iters=5000):
    """Local refinement: minimize the top-K peaks simultaneously via projected gradient."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        sorted_idx = np.argsort(b)[::-1]
        top_idx = sorted_idx[:n_peaks]

        # Compute gradient that reduces all top peaks
        w = np.zeros(len(b))
        w[top_idx] = 1.0 / n_peaks
        grad_a = 2.0 * fast_cross_corr(w, a)

        # Project gradient onto tangent space of simplex
        S = np.dot(grad_a, a)
        grad_tangent = grad_a - S

        # Adaptive step size
        max_b = b[top_idx].max()
        cur_score = 2.0 * n * max_b
        lower_bound = 2.0 * n / (2.0 * n - 1.0)
        gap = cur_score - lower_bound
        grad_norm_sq = np.dot(grad_tangent, grad_tangent)
        if grad_norm_sq < 1e-20:
            break
        eta = min(gap / (2.0 * grad_norm_sq), 0.8)

        # Multiplicative update (mirror descent)
        log_a = np.log(np.maximum(a, 1e-300))
        log_a_new = log_a - eta * grad_tangent
        a_new = softmax_from_z(log_a_new)
        sc_new = score_normalized(a_new)

        if sc_new > cur_score * 1.02:
            eta *= 0.5
            log_a_new = log_a - eta * grad_tangent
            a_new = softmax_from_z(log_a_new)
            sc_new = score_normalized(a_new)

        a = a_new
        if sc_new < best_score:
            best_score = sc_new
            best_a = a.copy()

    return best_a, best_score


def get_initial_sequence(rng, n_base):
    """Get the initial sequence from height_sequence_1 or generate one."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            return p / p.sum(), len(p), "parent"
    except Exception:
        pass
    a = structured_init(n_base, rng, 0)
    return a, n_base, "default"


def propose_candidate(seed=42, budget_s=1800.0):
    """Main search function that minimizes the evaluation sequence score."""
    rng = np.random.default_rng(seed)
    total_budget = max(1.0, float(budget_s))
    global_deadline = time.monotonic() + (total_budget - 30.0)

    best_seq, n_base, src = get_initial_sequence(rng, 1200)
    best_score = evaluate_safe(best_seq)
    print(f"Initial: n={n_base}, score={best_score:.8f} (source: {src})")

    # Helper to update best (inline to avoid closure)
    def _try_update(a, label):
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        return a, sc

    # === Phase 1: LSE-LBFGS (main workhorse) - 40% time ===
    print(f"\n=== Phase 1: LSE-LBFGS ===")
    remaining = global_deadline - time.monotonic()
    t1 = min(0.40 * (total_budget - 30), remaining * 0.7)

    if t1 > 30:
        a_out, _ = lse_lbfgs(best_seq, t1 * 0.4, beta_start=20.0, beta_end=300.0,
                             n_stages=15, iters_per_stage=40, top_k_start=5, top_k_end=2 * n_base - 1)
        a_new, sc_new = _try_update(a_out, "lse1")
        if sc_new < best_score:
            best_score = sc_new
            best_seq = a_new
            print(f"  ** LSE-1: {best_score:.8f}")

    remaining = global_deadline - time.monotonic()
    if remaining > 20:
        a_out, _ = lse_lbfgs(best_seq, remaining * 0.5, beta_start=50.0, beta_end=800.0,
                             n_stages=15, iters_per_stage=30, top_k_start=3, top_k_end=2 * n_base - 1)
        a_new, sc_new = _try_update(a_out, "lse2")
        if sc_new < best_score:
            best_score = sc_new
            best_seq = a_new
            print(f"  ** LSE-2: {best_score:.8f}")

    remaining = global_deadline - time.monotonic()
    if remaining > 20:
        pert = best_seq + 0.02 * rng.normal(0, 1, len(best_seq))
        pert = np.maximum(pert, 0)
        pert = pert / max(1e-15, pert.sum())
        a_out, _ = lse_lbfgs(pert, remaining * 0.5, beta_start=30.0, beta_end=500.0,
                             n_stages=12, iters_per_stage=30, top_k_start=5, top_k_end=2 * n_base - 1)
        a_new, sc_new = _try_update(a_out, "lse_pert")
        if sc_new < best_score:
            best_score = sc_new
            best_seq = a_new
            print(f"  ** LSE-pert: {best_score:.8f}")

    print(f"  Phase 1 done: best_score={best_score:.8f}")

    # === Phase 2: Nesterov LP - 25% time ===
    print(f"\n=== Phase 2: Nesterov LP ===")
    remaining = global_deadline - time.monotonic()
    t2 = min(0.25 * (total_budget - 30), remaining * 0.6)
    if t2 > 30:
        a_lp, _ = nesterov_lp(best_seq, t2)
        a_new, sc_new = _try_update(a_lp, "lp")
        if sc_new < best_score:
            best_score = sc_new
            best_seq = a_new
            print(f"  ** LP: {best_score:.8f}")

    remaining = global_deadline - time.monotonic()
    if remaining > 30:
        a_lp2, _ = nesterov_lp(best_seq, remaining * 0.5)
        a_new, sc_new = _try_update(a_lp2, "lp2")
        if sc_new < best_score:
            best_score = sc_new
            best_seq = a_new
            print(f"  ** LP-2: {best_score:.8f}")

    print(f"  Phase 2 done: best_score={best_score:.8f}")

    # === Phase 3: Peak Mirror Descent - 12% time ===
    print(f"\n=== Phase 3: Peak MD ===")
    remaining = global_deadline - time.monotonic()
    t3 = min(0.12 * (total_budget - 30), remaining * 0.6)
    if t3 > 20:
        a_md, _ = peak_mirror_descent(best_seq, t3)
        a_new, sc_new = _try_update(a_md, "md")
        if sc_new < best_score:
            best_score = sc_new
            best_seq = a_new
            print(f"  ** MD: {best_score:.8f}")

    remaining = global_deadline - time.monotonic()
    if remaining > 15:
        a_md2, _ = peak_mirror_descent(best_seq, remaining * 0.6)
        a_new, sc_new = _try_update(a_md2, "md2")
        if sc_new < best_score:
            best_score = sc_new
            best_seq = a_new
            print(f"  ** MD-2: {best_score:.8f}")

    print(f"  Phase 3 done: best_score={best_score:.8f}")

    # === Phase 4: Local Peak Refine - 8% time ===
    print(f"\n=== Phase 4: Local Peak Refine ===")
    remaining = global_deadline - time.monotonic()
    t4 = min(0.08 * (total_budget - 30), remaining * 0.6)
    if t4 > 20:
        a_ref, _ = local_peak_refine(best_seq, t4, n_peaks=3, max_iters=20000)
        a_new, sc_new = _try_update(a_ref, "refine")
        if sc_new < best_score:
            best_score = sc_new
            best_seq = a_new
            print(f"  ** Refine: {best_score:.8f}")

        remaining = global_deadline - time.monotonic()
        if remaining > 15:
            a_ref2, _ = local_peak_refine(best_seq, remaining * 0.6, n_peaks=5, max_iters=15000)
            a_new, sc_new = _try_update(a_ref2, "refine5")
            if sc_new < best_score:
                best_score = sc_new
                best_seq = a_new
                print(f"  ** Refine5: {best_score:.8f}")

    print(f"  Phase 4 done: best_score={best_score:.8f}")

    # === Phase 5: Multi-start with diverse inits - 8% time ===
    print(f"\n=== Phase 5: Multi-start ===")
    remaining = global_deadline - time.monotonic()
    t5_total = min(0.08 * (total_budget - 30), remaining * 0.5)
    if t5_total > 60:
        n_starts = 5
        t5_per = t5_total / n_starts
        for mode in range(n_starts):
            remaining = global_deadline - time.monotonic()
            if remaining < 25:
                break
            init = structured_init(n_base, rng, mode + 1)
            a_ms, _ = lse_lbfgs(init, min(t5_per, remaining * 0.7),
                                beta_start=50.0, beta_end=400.0,
                                n_stages=10, iters_per_stage=30,
                                top_k_start=5, top_k_end=2 * n_base - 1)
            a_new, sc_new = _try_update(a_ms, f"ms{mode}")
            if sc_new < best_score:
                best_score = sc_new
                best_seq = a_new
                print(f"  ** Multi-start-{mode}: {best_score:.8f}")

    print(f"  Phase 5 done: best_score={best_score:.8f}")

    # === Phase 6: Final polish with combined methods - 7% time ===
    print(f"\n=== Phase 6: Final Polish ===")
    remaining = global_deadline - time.monotonic()
    if remaining > 25:
        # High-temperature LSE for final squeeze
        a_f, _ = lse_lbfgs(best_seq, remaining * 0.35, beta_start=300.0, beta_end=1500.0,
                           n_stages=10, iters_per_stage=30, top_k_start=2, top_k_end=2 * n_base - 1)
        a_new, sc_new = _try_update(a_f, "final_lse")
        if sc_new < best_score:
            best_score = sc_new
            best_seq = a_new
            print(f"  ** Final LSE: {best_score:.8f}")

    remaining = global_deadline - time.monotonic()
    if remaining > 15:
        a_f2, _ = local_peak_refine(best_seq, remaining * 0.4, n_peaks=3, max_iters=15000)
        a_new, sc_new = _try_update(a_f2, "final_refine")
        if sc_new < best_score:
            best_score = sc_new
            best_seq = a_new
            print(f"  ** Final Refine: {best_score:.8f}")

    remaining = global_deadline - time.monotonic()
    if remaining > 15:
        a_f3, _ = peak_mirror_descent(best_seq, remaining * 0.5)
        a_new, sc_new = _try_update(a_f3, "final_md")
        if sc_new < best_score:
            best_score = sc_new
            best_seq = a_new
            print(f"  ** Final MD: {best_score:.8f}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
