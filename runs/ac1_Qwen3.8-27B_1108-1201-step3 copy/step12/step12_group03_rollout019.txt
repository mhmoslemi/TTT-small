def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_weights(b, beta):
    """Compute softmax(beta * b) with numerical stability."""
    beta_b = beta * (b - b.max())
    # Clip to avoid overflow/underflow
    beta_b = np.clip(beta_b, -700, 700)
    e = np.exp(beta_b)
    s = e.sum()
    if s < 1e-300:
        # Fallback: uniform on argmax
        w = np.zeros(len(b))
        w[b >= b.max() - 1e-10] = 1.0
        w /= w.sum()
        return w
    return e / s


def soft_max_value(b, beta):
    """Compute soft_max(b, beta) = (1/beta) * log(sum(exp(beta*b)))."""
    bm = b.max()
    beta_b = beta * (b - bm)
    beta_b = np.clip(beta_b, -700, 700)
    e = np.exp(beta_b)
    s = e.sum()
    if s < 1e-300:
        return bm
    return bm + np.log(s) / beta


def simplex_project(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    # Algorithm from Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.argmax(1 - u * np.arange(1, n+1) / (cssv) > 0)
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)


def compute_gradient_soft(a, beta):
    """Compute gradient of soft_max(conv(a,a), beta) with respect to a."""
    n = len(a)
    b = fast_autocorr(a)
    w = softmax_weights(b, beta)
    # grad_i = 2 * sum_j a_j * w_{i+j}
    # This is 2 * correlate(w, a, 'valid')
    g = 2.0 * np.correlate(w, a, mode='valid')
    # Verify: g should have length n
    assert len(g) == n, f"Gradient length {len(g)} != {n}"
    return g, b

def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build constraint: convolve(a_fixed, b_var) <= M
    # We can't use cvxpy's convolve, so build the matrix
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

            a_new = np.maximum(a_new, 0)
            a_s = a_new.sum()
            if a_s < 1e-15:
                a_new = y_new.copy()
                a_s = 1.0

def qp_active_set_refinement(a0, time_limit, n_active_max=50):
    """QP refinement on active constraint set."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    b = fast_autocorr(a)
    max_b = b.max()
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * max_b
    
    n_active = min(15, n_active_max)
    
    for round_num in range(30):
        if time.monotonic() - t0 > time_limit:
            break
        
        threshold = max_b * (1 - 0.005)
        active_lags = np.where(b >= threshold)[0]
        if len(active_lags) > n_active:
            active_lags = active_lags[np.argsort(b[active_lags])[::-1][:n_active]]
        
        a_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        
        constraints = [cp.sum(a_var) == 1, M_var >= 0]
        for k in active_lags:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            other_idx = k - idx
            q_expr = cp.sum(a_var[idx] * a_var[other_idx])
            constraints.append(q_expr <= M_var)
        
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        
        solved = False
        for solver in [cp.ECOS, cp.SCS, cp.CLARABEL]:
            try:
                prob.solve(solver=solver, max_iter=20000)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    solved = True
                    break
            except Exception:
                continue
        
        if not solved:
            break
        
        a_new = np.asarray(a_var.value, dtype=np.float64)
        a_new = np.maximum(a_new, 0)
        a_new_s = a_new.sum()
        if a_new_s < 1e-15:
            break
        a_new = a_new / a_new_s
        
        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            b = fast_autocorr(a)
            max_b = b.max()
            n_active = min(n_active + 5, n_active_max)
        else:
            for frac in [0.2, 0.4, 0.6]:
                a_mix = (1 - frac) * a + frac * a_new
                a_mix = np.maximum(a_mix, 0)
                a_mix_s = a_mix.sum()
                if a_mix_s < 1e-15:
                    continue
                a_mix = a_mix / a_mix_s
                score_mix = 2.0 * n * fast_autocorr(a_mix).max()
                if score_mix < best_score:
                    best_score = score_mix
                    best_a = a_mix.copy()
                    a = a_mix
                    b = fast_autocorr(a)
                    max_b = b.max()
                    break
            else:
                if round_num >= 5:
                    break
                n_active = min(n_active + 5, n_active_max)
    
    return best_a, best_score
