def fast_autocorr(a):

    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)

def softmax_fn(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:

def compute_lse_grad_a(a, T):
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    
    fft_size = 1 << (2 * n - 1).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)
    
    grad = 2.0 * conv[n - 1:n - 1 + n]
    return grad, w

def lse_obj_grad_z(z, T):

    w = exp_Tb / exp_Tb.sum()

    W = rfft(w, fft_size)

    grad_a = 2.0 * conv[n - 1:n - 1 + n]
    
    S = np.dot(grad_a, a)

def spectral_shaping(a, n_iter=500, alpha=0.3):
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    
    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.05, 20.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:

        a_new = a_new / s

        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
    
    return best_a

def get_lp_solver():

        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s

def gradient_peak_shaving(a0, time_limit, seed=123):
    """Gradient-guided peak shaving: use LSE gradient to identify best moves."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:

    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:10]
        
        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            
            src_local = np.argmax(contrib)
            src = idx[src_local]
            
            # Compute gradient to guide destination selection
            Tb = 50.0 * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()
            
            fft_size = 1 << (2 * n - 1).bit_length()
            A_rev = rfft(a[::-1].copy(), fft_size)
            W = rfft(w, fft_size)
            conv = irfft(A_rev * W, fft_size)
            grad = 2.0 * conv[n - 1:n - 1 + n]
            
            # Find destinations with low gradient (less likely to create new peaks)
            peak_areas = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_areas.update(range(js2, je2 + 1))
            
            candidates = [j for j in range(n) if j not in peak_areas]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            
            # Sort by gradient (ascending) - low gradient means safe to increase
            candidates_sorted = sorted(candidates, key=lambda j: grad[j])
            
            for dst in candidates_sorted[:5]:
                for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation to escape local minima
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            if stall > 500:
                break
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

import numpy as np

from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(arr):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def compute_lse_grad_a(a, T):
    """Compute gradient of (1/T)*log(sum exp(T*(a*a)_k)) w.r.t. a using FFT."""
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()

    fft_size = 1 << (2 * n - 1).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)

    grad = 2.0 * conv[n - 1:n - 1 + n]
    return grad, w


def lse_obj_grad_z(z, T):
    """Objective and gradient for z-parameterization (a = softmax(z))."""
    a = softmax_fn(z)
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    obj_val = (max_Tb + np.log(exp_Tb.sum())) / T

    w = exp_Tb / exp_Tb.sum()

    fft_size = 1 << (2 * n - 1).bit_length()
    A_rev = rfft(a[::-1].copy(), fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A_rev * W, fft_size)
    grad_a = 2.0 * conv[n - 1:n - 1 + n]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj_val, grad_z


def spectral_shaping(a, n_iter=500, alpha=0.3):
    """
    Spectral shaping: iteratively flatten the power spectrum to reduce
    autocorrelation peaks via frequency-domain scaling.
    """
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2

        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))

        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.05, 20.0)
        scale[1:] = np.power(ratio, alpha / 2.0)

        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a


def adam_lse_optimize(a0, time_limit, T_start=100.0, T_end=0.05, n_stages=20, lr=0.01):
    """
    LogSumExp-smoothed optimization with Adam optimizer and multi-stage annealing.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_adam = 0

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        iters = max(20, min(200, int(remaining * 100 / max(1, (n_stages - si)))))

        for it in range(iters):
            if time.monotonic() - t0 > time_limit:
                break

            obj, grad = lse_obj_grad_z(z, stage_T)

            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)

            lr_t = lr / np.sqrt(1 + 0.005 * it)
            z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)

            if it % 10 == 0:
                a = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()

        a = softmax_fn(z)
        score = 2.0 * n * fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Adam-LSE stage {si}/{n_stages} (T={stage_T:.2f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a_fixed):
    """Build the (2n-1) x n convolution matrix T where (T@b) = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0. Returns (b, M)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, max_iters=100000):
    """
    Nesterov-accelerated LP best-response iteration for the zero-sum game.
    Uses exponential moving average for tracking the best iterate.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    # Exponential moving average for tracking
    ema = a.copy()
    ema_beta = 0.97

    # Also track uniform average over last K iterations
    window = []
    window_size = 50

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s

        y = y_new
        a = a_new

        # Update EMA
        ema = ema_beta * ema + (1 - ema_beta) * a
        ema = np.maximum(ema, 0)
        s_ema = ema.sum()
        if s_ema > 1e-15:
            ema = ema / s_ema

        # Update sliding window average
        window.append(a.copy())
        if len(window) > window_size:
            window.pop(0)

        # Check scores
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_ema = 2.0 * n * fast_autocorr(ema).max()
        if score_ema < best_score:
            best_score = score_ema
            best_a = ema.copy()

        if len(window) >= 10:
            win_avg = np.mean(window, axis=0)
            win_avg = np.maximum(win_avg, 0)
            s_win = win_avg.sum()
            if s_win > 1e-15:
                win_avg = win_avg / s_win
                score_win = 2.0 * n * fast_autocorr(win_avg).max()
                if score_win < best_score:
                    best_score = score_win
                    best_a = win_avg.copy()

        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        if t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}, LP_M={M_val:.8f}")

    return best_a, best_score


def gradient_peak_shaving(a0, time_limit, seed=123):
    """
    Gradient-guided coordinate descent for peak shaving.
    Uses LSE gradient to identify safe destinations for mass transfer.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    fft_size = 1 << (2 * n - 1).bit_length()

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:10]

        # Compute gradient using moderate temperature
        Tb = 30.0 * b
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()
        A_rev = rfft(a[::-1].copy(), fft_size)
        W = rfft(w, fft_size)
        conv = irfft(A_rev * W, fft_size)
        grad = 2.0 * conv[n - 1:n - 1 + n]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            src_local = np.argmax(contrib)
            src = idx[src_local]

            peak_areas = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_areas.update(range(js2, je2 + 1))

            candidates = [j for j in range(n) if j not in peak_areas]
            if not candidates:
                candidates = [j for j in range(n) if j != src]

            # Sort by gradient (ascending) - low gradient means safe to increase
            candidates_sorted = sorted(candidates, key=(lambda j: grad[j]))

            for dst in candidates_sorted[:8]:
                for frac in [0.002, 0.005, 0.01, 0.02, 0.03, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            if stall > 400:
                # Try a larger random perturbation to escape
                for _ in range(20):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.01, 0.05)
                        amt = a[i] * frac
                        a_trial = a.copy()
                        a_trial[i] -= amt
                        a_trial[j] += amt
                        if a_trial[i] >= 0:
                            a_trial = a_trial / a_trial.sum()
                            score = 2.0 * n * fast_autocorr(a_trial).max()
                            if score < best_score:
                                a = a_trial
                                best_score = score
                                best_a = a.copy()
                                stall = 0
                                break
                if stall > 500:
                    break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    Phase 4: Gradient-guided peak shaving
    Phase 5: Longer length - spectral + Adam + LP

    Phase 7: Final peak shaving
    Phase 8: Extra LP Nesterov if time remains
