For fixed p, the inner problem is:

Actually, for fractional quadratic: min a^T A a / (1^T a)² subject to a >= 0.

This is a quadratic program! And the optimal value is related to the minimum eigenvalue.

For our problem, T_p is a sum of "shifted" outer products, so it's PSD. The minimum of a^T T_p a on the simplex is the Rayleigh quotient of the minimum eigenvalue (if the corresponding eigenvector is non-negative) or a boundary solution.

        # Mirror descent step
        a_new = mirror_descent_step(a, grad, lr)

        # Evaluate
        c_new = fast_conv(a_new)
        score = 2.0 * n * c_new.max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        a = a_new
        it += 1

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SMD it={it}, tau={tau:.1f}, best={best_score:.8f}, ({elapsed:.0f}s)")

    return best_a


def coordinate_descent_pass(a, c, n, active_k=None):
    """
    One full pass of coordinate descent. For each coordinate i, find the optimal
    value given all others fixed. Uses exact 1D optimization.
    Returns updated a and c.
    """
    S = a.sum()
    if S < 1e-15:
        return a, c

    # Precompute active k set (top peaks)
    if active_k is None:
        top_m = min(8, 2 * n - 1)
        active_k = np.argsort(c)[::-1][:top_m]

    new_a = a.copy()

    for i in range(n):
        ai = new_a[i]
        if ai < 1e-15:
            # Try setting to a small positive value
            pass

        # Compute d_k = 2 * a_{k-i} for k != 2i, d_{2i} = 2*a_i
        # c_k' = c_k + delta * d_k (approx, ignoring delta^2 at k=2i)
        # where delta = a_i' - a_i

        # d_k = 2 * a_{k-i} where a_j = 0 for j outside [0, n)
        # For k in [0, 2n-2]: a_{k-i} is valid when 0 <= k-i < n, i.e., i <= k <= i+n-1
        d = np.zeros(2 * n - 1)
        k_start = max(0, i)
        k_end = min(2 * n - 2, i + n - 1)
        d[k_start:k_end + 1] = 2.0 * a[i - i:k_end - i + 1]  # a[0 : k_end-i+1]
        # Special case k = 2i: d_{2i} = 2*a_i (already correct since a_{2i-i} = a_i)

        # For each active k, compute optimal delta
        best_delta = 0.0
        best_f = 2.0 * n * c.max() / (S ** 2)

        for k in active_k:
            dk = d[k]
            if dk < 1e-15:
                continue
            # Optimal: delta* = S - 2*c_k/dk
            delta_opt = S - 2.0 * c[k] / dk
            # Clamp: delta >= -ai (so a_i' >= 0), and S + delta > 0
            delta_opt = max(delta_opt, -ai + 1e-15)
            if delta_opt < -S + 1e-15:
                continue

            ai_trial = ai + delta_opt
            S_trial = S + delta_opt
            if S_trial < 0.01:
                continue

            # Compute new max convolution (approximate: only update the active k's)
            # For efficiency, just check the active k's
            max_c_trial = 0.0
            for k2 in active_k:
                d2 = d[k2]
                if k2 == 2 * i:
                    c2_trial = c[k2] + delta_opt * 2.0 * ai + delta_opt ** 2
                else:
                    c2_trial = c[k2] + delta_opt * d2
                if c2_trial > max_c_trial:
                    max_c_trial = c2_trial

            # Also need to check if any non-active k becomes the max
            # Approximation: the new max is at most max(active max, old second max + some bound)
            # For now, just use the active k's (good enough in practice)

            f_trial = 2.0 * n * max_c_trial / (S_trial ** 2)
            if f_trial < best_f - 1e-12:
                best_f = f_trial
                best_delta = delta_opt

        # Also try delta = -ai (set to zero)
        if ai > 1e-10:
            delta_zero = -ai
            S_zero = S - ai
            if S_zero > 0.01:
                max_c_zero = 0.0
                for k2 in active_k:
                    d2 = d[k2]
                    if k2 == 2 * i:
                        c2_zero = c[k2] + delta_zero * 2.0 * ai + delta_zero ** 2
                    else:
                        c2_zero = c[k2] + delta_zero * d2
                    c2_zero = max(0.0, c2_zero)
                    if c2_zero > max_c_zero:
                        max_c_zero = c2_zero
                f_zero = 2.0 * n * max_c_zero / (S_zero ** 2)
                if f_zero < best_f - 1e-12:
                    best_f = f_zero
                    best_delta = delta_zero

        # Apply the best delta
        new_a[i] = max(0.0, ai + best_delta)

    # Update c at the end
    s_new = new_a.sum()
    if s_new < 1e-15:
        return a, c
    # Renormalize to maintain sum
    # Actually, for coordinate descent we should NOT renormalize (the objective handles it)
    # But for numerical stability, let's keep the sum close to 1
    if abs(s_new - 1.0) > 0.01:
        new_a = new_a / s_new

    c_new = fast_conv(new_a)
    return new_a, c_new


def coordinate_descent(a0, time_limit, max_passes=10000, verbose=True):
    """Multiple passes of exact coordinate descent."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    # Don't normalize - let the objective handle it
    # But keep sum roughly 1 for numerical stability
    a = a / s

    c = fast_conv(a)
    best_a = a.copy()
    best_score = 2.0 * n * c.max() / (a.sum() ** 2)
    t0 = time.monotonic()
    stall = 0

    for p in range(max_passes):
        if time.monotonic() - t0 > time_limit:
            break

        # Update active set
        top_m = min(10, 2 * n - 1)
        active_k = np.argsort(c)[::-1][:top_m]

        a, c = coordinate_descent_pass(a, c, n, active_k)

        score = 2.0 * n * c.max() / (a.sum() ** 2)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if verbose and (p + 1) % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass={p+1}, best={best_score:.8f}, stall={stall} ({elapsed:.0f}s)")

        if stall > 100:
            break

    return best_a


def enhance_coordinate_descent(a0, time_limit, verbose=True):
    """
    Enhanced coordinate descent that periodically updates the active set
    and uses a finer search over the 1D subproblem.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s

    c = fast_conv(a)
    best_a = a.copy()
    best_score = 2.0 * n * c.max() / (a.sum() ** 2)
    t0 = time.monotonic()
    stall = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        # Update active set
        top_m = min(12, 2 * n - 1)
        active_k = np.argsort(c)[::-1][:top_m]
        S = a.sum()

        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            ai = a[i]

            # d_k values
            k_start = max(0, i)
            k_end = min(2 * n - 2, i + n - 1)
            d_seg = 2.0 * a[:k_end - i + 1]

            # For each active k, compute optimal a_i'
            best_ai = ai
            best_f = 2.0 * n * max(c[active_k]) / (S * S)

            for k in active_k:
                if k < k_start or k > k_end:
                    dk = 0.0
                else:
                    dk = d_seg[k - i]

                if dk < 1e-12:
                    continue

                # delta* = S - 2*c_k/dk
                delta_star = S - 2.0 * c[k] / dk
                ai_candidate = ai + delta_star

                if ai_candidate < 0:
                    ai_candidate = 0.0

                S_cand = S - ai + ai_candidate
                if S_cand < 0.01:
                    continue

                # Compute new c at active positions
                delta = ai_candidate - ai
                max_c_new = 0.0
                for k2 in active_k:
                    if k2 < k_start or k2 > k_end:
                        c2 = c[k2]
                    elif k2 == 2 * i:
                        c2 = c[k2] + delta * 2.0 * ai + delta * delta
                    else:
                        c2 = c[k2] + delta * d_seg[k2 - i]
                    c2 = max(0.0, c2)
                    if c2 > max_c_new:
                        max_c_new = c2

                f_cand = 2.0 * n * max_c_new / (S_cand * S_cand)
                if f_cand < best_f - 1e-14:
                    best_f = f_cand
                    best_ai = ai_candidate

            # Also try ai = 0
            if ai > 1e-8:
                S_cand = S - ai
                if S_cand > 0.01:
                    delta = -ai
                    max_c_new = 0.0
                    for k2 in active_k:
                        if k2 < k_start or k2 > k_end:
                            c2 = c[k2]
                        elif k2 == 2 * i:
                            c2 = c[k2] + delta * 2.0 * ai + delta * delta
                        else:
                            c2 = c[k2] + delta * d_seg[k2 - i]
                        c2 = max(0.0, c2)
                        if c2 > max_c_new:
                            max_c_new = c2
                    f_cand = 2.0 * n * max_c_new / (S_cand * S_cand)
                    if f_cand < best_f - 1e-14:
                        best_f = f_cand
                        best_ai = 0.0

            # Apply
            if abs(best_ai - ai) > 1e-14:
                a[i] = best_ai
                improved = True

        if improved:
            # Recompute c
            c = fast_conv(a)
            score = 2.0 * n * c.max() / (a.sum() ** 2)
            if score < best_score - 1e-14:
                best_score = score
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        else:
            stall += 1

        it += 1
        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ECD pass={it}, best={best_score:.8f}, stall={stall} ({elapsed:.0f}s)")

        if stall > 50:
            break

    return best_a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length, preserving sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def oscillatory_init(n, freq, rng):
    """Generate an oscillatory non-negative initialization."""
    x = np.arange(n)
    phase = rng.uniform(0, 2 * np.pi)
    a = 1.0 + np.cos(2 * np.pi * freq * x / n + phase)
    a = np.maximum(a, 0.0)
    # Mix multiple frequencies
    for f in [2, 3, 5, 7]:
        a = a * (1.0 + 0.3 * np.cos(2 * np.pi * f * x / n + rng.uniform(0, 2 * np.pi)))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Combines mirror descent, coordinate descent,
    and multi-length exploration to minimize the objective.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)
    print(f"Time budget: {time_budget:.0f}s, seed={seed}")

    best_seq = None
    best_score = float('inf')

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = evaluate_sequence([float(x) for x in parent])
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        # Scale to ensure sum is reasonable (avoid clipping issues)
        a_eval = a / s  # normalize to sum=1
        sc = evaluate_sequence([float(x) for x in a_eval])
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a_eval.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is None:
        n_base = 1600
        a_init = oscillatory_init(n_base, 3, rng)
        best_seq = a_init
        best_score = evaluate_sequence([float(x) for x in best_seq])
        print(f"Default init: n={n_base}, score={best_score:.6f}")
        n_base = len(best_seq)
    else:
        n_base = len(best_seq)

    # === Phase 1: Mirror descent on base length ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Mirror Descent n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.25, 350.0)
    if t1 > 30:
        a_out = smooth_mirror_descent(best_seq, t1, tau_start=3.0, tau_end=1500.0, verbose=True)
        try_update(a_out, f"smd_n{n_base}")

    # === Phase 2: Coordinate descent on base length ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Enhanced CD n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.25, 350.0)
    if t2 > 30:
        a_out = enhance_coordinate_descent(best_seq, t2, verbose=True)
        try_update(a_out, f"ecd_n{len(best_seq)}")

    # === Phase 3: Mirror descent again (continued annealing) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: SMD continued n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 20:
        a_out = smooth_mirror_descent(best_seq, t3, tau_start=50.0, tau_end=5000.0, verbose=True)
        try_update(a_out, f"smd2_n{len(best_seq)}")

    # === Phase 4: Try longer length ===
    n_long = n_base + 400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Long length n={n_long} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.20, 250.0)
    if t4 > 30:
        a_long = interpolate_sequence(best_seq, n_long)
        a_out = smooth_mirror_descent(a_long, t4 * 0.6, tau_start=5.0, tau_end=1000.0, verbose=True)
        try_update(a_out, f"smd_n{n_long}")
        # CD on the long sequence
        remaining2 = time_budget - (time.monotonic() - t_global)
        t4b = min(remaining2 * 0.15, 150.0)
        if t4b > 20:
            a_out2 = enhance_coordinate_descent(a_out, t4b, verbose=False)
            try_update(a_out2, f"ecd_n{n_long}")

    # === Phase 5: Try even longer length ===
    n_longer = n_base + 800
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Longer length n={n_longer} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.15, 200.0)
    if t5 > 30:
        a_longer = interpolate_sequence(best_seq, n_longer)
        a_out = smooth_mirror_descent(a_longer, t5 * 0.5, tau_start=5.0, tau_end=800.0, verbose=True)
        try_update(a_out, f"smd_n{n_longer}")
        remaining2 = time_budget - (time.monotonic() - t_global)
        t5b = min(remaining2 * 0.15, 120.0)
        if t5b > 20:
            a_out2 = enhance_coordinate_descent(a_out, t5b, verbose=False)
            try_update(a_out2, f"ecd_n{n_longer}")

    # === Phase 6: Final CD polish on best ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final CD n={n_best} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.20, 200.0)
    if t6 > 10:
        a_out = enhance_coordinate_descent(best_seq, t6, verbose=True)
        try_update(a_out, "final_cd")

    # === Phase 7: Final SMD with very high tau ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final SMD n={n_best} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.20, 150.0)
    if t7 > 10:
        a_out = smooth_mirror_descent(best_seq, t7, tau_start=200.0, tau_end=10000.0, verbose=True)
        try_update(a_out, "final_smd")

    # === Phase 8: Rapid coordinate descent with local restarts ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: CD with restarts n={n_best} (budget={remaining:.0f}s) ===")
    t8 = min(remaining * 0.30, 200.0)
    if t8 > 10:
        a_cur = best_seq.copy()
        # Do short CD bursts with perturbations
        n_bursts = max(3, int(t8 / 15))
        burst_time = t8 / n_bursts
        for bi in range(n_bursts):
            if time.monotonic() - t_global > time_budget - 30:
                break
            a_out = enhance_coordinate_descent(a_cur, burst_time, verbose=False)
            try_update(a_out, f"cd_burst_{bi}")
            a_cur = best_seq.copy()
            # Perturb
            a_cur = a_cur + rng.normal(0, 0.001, a_cur.shape)
            a_cur = np.maximum(a_cur, 0.0)
            a_cur = a_cur / a_cur.sum()

    # === Final verification ===
    if best_seq is not None:
        final_score = evaluate_sequence([float(x) for x in best_seq])
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, "
              f"time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0] * 10

    # Return the sequence (normalized to sum=1 for safety)
    s_final = best_seq.sum()
    if s_final < 0.01:
        best_seq = best_seq * (1.0 / s_final)
    return [float(x) for x in best_seq]
