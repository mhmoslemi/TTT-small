"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adaptive p-norm gradient descent (Adam in log-domain) - main workhorse
2. LP Nesterov (zero-sum game) with warm starts - complementary optimizer

4. Multi-length exploration
5. Peak shaving for final polish
Key innovation: p-norm considers ALL peaks (not just top-k), giving more balanced gradients.

import numpy as np
import time
from scipy.fft import fft, ifft, rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_corr(a, b):
    """Cross-correlation: result[j] = sum_k a[k]*b[k-j] (valid range)."""
    n = len(b)
    la, lb = len(a), len(b)
    L = 1 << (la + lb - 1 - 1).bit_length()
    A = fft(a, L)
    B = fft(b, L)
    return ifft(A * np.conj(B)).real[:n]


def pnorm_obj_and_grad(a, p):
    """
    Compute p-norm objective and gradient w.r.t. a (not z).
    obj = sum_k c[k]^p where c = conv(a,a)
    grad[j] = 2p * (d * a)[j] where d = c^(p-1)
    Returns (obj, grad) where grad has same length as a.
    """
    n = len(a)
    c = fast_autocorr(a)
    # Use stable power
    c_clipped = np.maximum(c, 0.0)
    if p == 1:
        d = np.ones_like(c)
    else:
        d = np.power(c_clipped, p - 1)
    obj = np.sum(np.power(c_clipped, p))
    # grad[j] = 2p * sum_k d[k]*a[k-j] = 2p * cross_corr(d, a)[j]
    cc = cross_corr(d, a)
    grad = 2.0 * p * cc
    return obj, grad


def pnorm_grad_z(z, p):
    """
    Compute p-norm objective and gradient w.r.t. z (log-domain).
    a = softmax(z)
    grad_z[j] = 2p * a[j] * ((d*a)[j] - obj)
    """
    # Stable softmax
    z_shift = z - z.max()
    e = np.exp(z_shift)
    s = e.sum()
    if s < 1e-300:
        a = np.ones(len(z)) / len(z)
    else:
        a = e / s
    obj, grad_a = pnorm_obj_and_grad(a, p)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def get_lp_solver():
    """Find a working LP solver."""
    import cvxpy as cp
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL, cp.MOSEK, cp.GLPK]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver, prev_b=None):
    """Solve LP: min M s.t. conv(a_fixed, b) <= M, sum(b)=1, b>=0."""
    import cvxpy as cp
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    # Build constraint matrix
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL, cp.MOSEK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    b_val = b_val - b_val.min()
    s = b_val.sum()
    if s < 1e-15:
        b_val = np.ones(n) / n
    else:
        b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, max_iters=200000):
    """Nesterov-accelerated LP best-response with adaptive restart."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall_count = 0
    prev_b = None

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit - 2:
            break

        b_star, M_val = solve_lp_best_response(y, solver, prev_b)
        if b_star is None:
            continue
        prev_b = b_star

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_sum = y_new.sum()
        if y_sum < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / y_sum

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            if s_new < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s_new

        y = y_new
        a = a_new

        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1

        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        if stall_count > 500 and t > 100:
            a = best_a.copy()
            y = best_a.copy()
            stall_count = 0
            prev_b = None

        if t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.8f}")

    return best_a, best_score


def pnorm_adam_optimize(a0, time_limit, p_start=4.0, p_end=100.0, n_stages=30, lr_base=0.05, seed=42):
    """
    Adam-optimized p-norm with multi-stage annealing of p.
    This is the main optimizer - uses all peaks (not just top-k).
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()
    rng = np.random.default_rng(seed)

    # Multi-stage schedule
    for si in range(n_stages):
        if time.monotonic() - t0 > time_limit - 5:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        # Interpolate p
        frac = si / max(1, n_stages - 1)
        p_cur = p_start * (p_end / p_start) ** frac

        # Learning rate: start higher, decay
        lr = lr_base * (0.85 ** si)
        # Scale by n for longer sequences
        lr = lr * min(1.0, 200.0 / n)

        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.8, 0.999, 1e-8

        max_it = max(20, int(remaining * 2))
        no_improve = 0
        stage_best = best_score

        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 3:
                break

            obj, grad_z = pnorm_grad_z(z, p_cur)

            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            if (it + 1) % 10 == 0:
                a = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    no_improve = 0
                else:
                    no_improve += 1

                # Periodic reset to prevent drift
                if (it + 1) % 50 == 0:
                    z = np.log(np.maximum(best_a, 1e-30))
                    m_z = np.zeros(n)
                    v_z = np.zeros(n)
                    t_adam = 0

                # Adaptive restart with noise if stuck
                if no_improve > 60:
                    z = np.log(np.maximum(best_a, 1e-30))
                    z += rng.normal(0, 0.01, n)
                    m_z = np.zeros(n)
                    v_z = np.zeros(n)
                    t_adam = 0
                    no_improve = 0

        a_stage = softmax_fn(z)
        score_stage = 2.0 * n * fast_autocorr(a_stage).max()
        if score_stage < best_score:
            best_score = score_stage
            best_a = a_stage.copy()

        if si % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"      PNorm stage {si}/{n_stages} (p={p_cur:.1f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score

def peak_shaving(a0, time_limit, seed=456):
    """
    Multi-scale peak shaving with adaptive step sizes.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit - 3:
        b = fast_autocorr(a)
        bmax = b.max()
        threshold = bmax * 0.998
        top_peaks = np.where(b >= threshold)[0]
        if len(top_peaks) == 0:
            top_peaks = np.argsort(b)[::-1][:10]

        improved = False
        for k in top_peaks[:30]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            max_contrib = contrib.max()
            if max_contrib < 1e-15:
                continue

            src = idx[np.argmax(contrib)]

            # Find safe destinations
            danger = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:15]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                danger[js2:je2 + 1] = True

            safe = np.where(~danger)[0]
            safe = safe[safe != src]
            if len(safe) == 0:
                safe = np.where(np.arange(n) != src)[0]

            safe_sorted = safe[np.argsort(a[safe])]
            dst_pool = safe_sorted[:max(5, len(safe_sorted) // 4)]
            if len(dst_pool) == 0:
                continue
            dst = dst_pool[rng.integers(len(dst_pool))]

            for frac in [0.002, 0.005, 0.01, 0.02, 0.04, 0.07, 0.1]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            n_pert = rng.integers(1, 3)
            a_trial = a.copy()
            for _ in range(n_pert):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a_trial[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.03)
                    amt = a_trial[i] * frac
                    a_trial[i] -= amt
                    a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 1e-10:
                a_trial = a_trial / st
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
            if stall > 200:
                break

    return best_a, best_score


def subgradient_simplex(a0, time_limit, seed=789):
    """
    Subgradient method with projection onto simplex.
    Uses the active set of peaks for the subgradient.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    def project_simplex(v):
        """Project v onto the simplex {x >= 0, sum(x) = 1}."""
        n = len(v)
        if n == 0:
            return v
        mu = 0.0
        sv = np.sort(v)[::-1]
        cumulative = np.cumsum(sv)
        rho = np.nonzero(sv * np.arange(1, n + 1) > (cumulative - 1))[0][-1]
        theta = (cumulative[rho] - 1) / (rho + 1.0)
        return np.maximum(v - theta, 0)

    # Adaptive step size
    step = 0.1
    t_count = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit - 3:
        t_count += 1
        t = max(1, int(t_count * 0.1))
        eta = step / np.sqrt(t)

        c = fast_autocorr(a)
        cmax = c.max()
        # Active set: peaks within 1% of max
        active = np.where(c >= cmax * 0.99)[0]

        # Compute subgradient: average of gradients of active peaks
        sg = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * a[k - idx]
        sg /= max(1, len(active))

        # Update
        a_trial = project_simplex(a - eta * sg)
        score = 2.0 * n * fast_autocorr(a_trial).max()

        if score < best_score:
            best_score = score
            best_a = a_trial.copy()
            a = a_trial
            no_improve = 0
        else:
            # Occasionally accept worse to escape
            if rng.random() < 0.05 * np.exp(-(score - best_score) * 10):
                a = a_trial
            no_improve += 1

        if no_improve > 100:
            a = best_a.copy()
            step = max(0.01, step * 0.5)
            t_count = 0
            no_improve = 0

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 10)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 3
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 2
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.7 + 0.2 * np.cos(2 * x) + 0.1 * np.cos(5 * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-5 * (x - 0.3) ** 2) + 0.5 * np.exp(-5 * (x - 0.7) ** 2)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) * np.cos(np.pi * x / 3) ** 2
    elif choice == 7:
        x = np.linspace(-1, 1, n)
        a = np.where(np.abs(x) <= 1, 0.5 * (1 + np.cos(np.pi * x)), 0)
    elif choice == 8:
        # Nearly uniform with small sinusoidal perturbation
        a = np.ones(n) + 0.1 * np.sin(2 * np.pi * np.arange(n) / n * 3)
    else:
        # Random smooth
        a = np.ones(n)
        for _ in range(5):
            center = rng.uniform(0, n)
            width = rng.uniform(n/10, n/3)
            amplitude = rng.uniform(-0.3, 0.3)
            a += amplitude * np.exp(-(np.arange(n) - center)**2 / (2*width**2))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def perturbed_init(a0, rng, scale=0.05):
    """Create a perturbed version of a sequence."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a0)) / len(a0)
    a = a / s
    noise = rng.normal(0, scale, len(a))
    a_new = a + noise * (a + 1e-6)
    a_new = np.maximum(a_new, 0)
    sn = a_new.sum()
    if sn < 1e-15:
        return a.copy()
    return a_new / sn


def hybrid_optimize(a0, time_limit, solver, seed=42):
    """
    Hybrid: alternate between p-norm Adam and LP Nesterov.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    # Alternate: p-norm for a chunk, then LP for a chunk
    chunk = 30  # seconds per sub-chunk
    p_cur = 4.0
    p_mult = 1.3

    while time.monotonic() - t0 < time_limit - 5:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 15:
            break

        # P-norm Adam chunk
        t_chunk = min(chunk, remaining * 0.6)
        if t_chunk > 5:
            a_out, _ = pnorm_adam_optimize(a, t_chunk, p_start=p_cur, p_end=p_cur * p_mult**20,
                                           n_stages=10, lr_base=0.04, seed=seed)
            sc = 2.0 * n * fast_autocorr(a_out).max()
            if sc < best_score:
                best_score = sc
                best_a = a_out.copy()
            a = a_out
            p_cur *= 1.2

        # LP Nesterov chunk
        remaining = time_limit - (time.monotonic() - t0)
        t_lp = min(chunk * 0.5, remaining * 0.35)
        if t_lp > 5:
            a_out, _ = nesterov_lp_optimize(a, t_lp, solver)
            sc = 2.0 * n * fast_autocorr(a_out).max()
            if sc < best_score:
                best_score = sc
                best_a = a_out.copy()
            a = a_out

    return best_a, best_score


def propose_candidate(seed=42, time_budget=1800.0):
    """
    Multi-phase optimization combining p-norm gradient descent, LP Nesterov,
    and peak shaving. The key innovation is using the p-norm (all peaks) for
    gradient computation, which provides more balanced optimization signals.
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_budget = min(float(time_budget) - 100.0, 1700.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Time budget: {time_budget:.0f}s")

    # Get parent sequence
    best_seq = None
    best_score = float('inf')
    parent = None

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(arr, label):
        nonlocal best_seq, best_score
        if arr is None:
            return
        arr = np.maximum(np.asarray(arr, dtype=np.float64), 0)
        s = arr.sum()
        if s < 0.01:
            return
        arr = arr / s
        sc = evaluate_safe(arr)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = arr.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(arr)}, t={elapsed:.0f}s)")

    n_base = len(best_seq) if best_seq is not None else 1600

    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    # === Phase 1: Quick LP Nesterov baseline ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    t1 = min(remaining * 0.08, 100.0)
    if t1 > 20:
        print(f"\n=== Phase 1: Quick LP Nesterov n={n_base} ({t1:.0f}s) ===")
        a_out, _ = nesterov_lp_optimize(best_seq, t1, solver)
        try_update(a_out, f"lp_quick_n{n_base}")

    # === Phase 2: p-norm Adam (main workhorse, all peaks) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 30:
        print(f"=== Phase 2: p-norm Adam n={n_base} ({t2:.0f}s) ===")
        a_out, _ = pnorm_adam_optimize(best_seq, t2, p_start=4.0, p_end=80.0,
                                       n_stages=30, lr_base=0.05, seed=42)
        try_update(a_out, f"pnorm1_n{n_base}")

    # === Phase 3: LP Nesterov refinement ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 20:
        print(f"=== Phase 3: LP Nesterov refinement n={n_base} ({t3:.0f}s) ===")
        a_out, _ = nesterov_lp_optimize(best_seq, t3, solver)
        try_update(a_out, f"lp_refine_n{n_base}")

    # === Phase 4: p-norm Adam variant 2 (different hyperparams) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 20:
        print(f"=== Phase 4: p-norm Adam v2 n={n_base} ({t4:.0f}s) ===")
        a_out, _ = pnorm_adam_optimize_v2(best_seq, t4, p_start=6.0, p_end=200.0,
                                          n_stages=35, lr_base=0.03, seed=123)
        try_update(a_out, f"pnorm2_n{n_base}")

    # === Phase 5: Hybrid optimization ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    t5 = min(remaining * 0.12, 150.0)
    if t5 > 20:
        print(f"=== Phase 5: Hybrid n={n_base} ({t5:.0f}s) ===")
        a_out, _ = hybrid_optimize(best_seq, t5, solver, seed=777)
        try_update(a_out, f"hybrid_n{n_base}")

    # === Phase 6: Multi-start at longer lengths ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    t6_total = min(remaining * 0.12, 150.0)
    if t6_total > 30:
        print(f"=== Phase 6: Multi-length ({t6_total:.0f}s) ===")
        for n_try in [n_base + 100, n_base + 200, n_base + 400]:
            if time.monotonic() - t_global_start > time_budget - 60:
                break
            t_each = t6_total / 3.0
            if t_each < 15:
                break
            init_long = interpolate_sequence(best_seq, n_try)
            a_out, _ = pnorm_adam_optimize(init_long, t_each, p_start=4.0, p_end=50.0,
                                           n_stages=15, lr_base=0.04, seed=n_try)
            try_update(a_out, f"pnorm_long_n{n_try}")

    # === Phase 7: Subgradient method ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    t7 = min(remaining * 0.06, 60.0)
    if t7 > 10:
        print(f"=== Phase 7: Subgradient n={n_base} ({t7:.0f}s) ===")
        a_out, _ = subgradient_simplex(best_seq, t7, seed=999)
        try_update(a_out, f"subgrad_n{n_base}")

    # === Phase 8: Final p-norm Adam polish ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    t8 = min(remaining * 0.10, 100.0)
    if t8 > 10:
        print(f"=== Phase 8: Final p-norm polish ({t8:.0f}s) ===")
        a_out, _ = pnorm_adam_optimize(best_seq, t8, p_start=20.0, p_end=300.0,
                                       n_stages=20, lr_base=0.02, seed=555)
        try_update(a_out, "final_pnorm")

    # === Phase 9: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    t9 = min(remaining * 0.15, 100.0)
    if t9 > 10:
        print(f"=== Phase 9: Peak shaving ({t9:.0f}s) ===")
        a_out, _ = peak_shaving(best_seq, t9, seed=456)
        try_update(a_out, "peak_shave")

    # === Phase 10: Final tiny LP ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 15:
        t10 = min(remaining * 0.5, 60.0)
        if t10 > 10:
            print(f"=== Phase 10: Final LP ({t10:.0f}s) ===")
            a_out, _ = nesterov_lp_optimize(best_seq, t10, solver)
            try_update(a_out, "final_lp")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
