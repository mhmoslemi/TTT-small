- But we need to check that other lags don't increase

- Identify the top contributing pairs to the peak lag
- For each such pair `(p, q)` with `p + q = k*`, try moving mass from `p` (or `q`) to a position that's far from all peak lags

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize as scipy_minimize

def fast_convolution(a, b=None):
    """Full convolution of a with b (or a with itself if b is None)."""
    if b is None:
        b = a
    n1, n2 = len(a), len(b)
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]

def objective_value(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2 for normalized a (sum=1)."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    conv = fast_convolution(a_norm)
    return 2.0 * n * conv.max()

def mirror_descent_best_response(a_fixed, b_init=None, n_inner=30, eta=None):
    """
    Solve: min max_k conv(a_fixed, b)_k s.t. b >= 0, sum(b) = 1
    using mirror descent (exponential gradient) on the simplex.
    Returns approximate minimizer b.
    """
    n = len(a_fixed)
    if b_init is None:
        b = np.ones(n) / n
    else:
        b = np.maximum(np.asarray(b_init, dtype=np.float64), 1e-30)
        b = b / b.sum()
    
    # Compute subgradient norm bound for step size
    # The subgradient is a shifted version of a_fixed
    # Its L1 norm = sum(a_fixed) = 1 (assuming a_fixed is normalized)
    if eta is None:
        eta = 1.0 / (n ** 0.25)  # Heuristic step size
    
    for it in range(n_inner):
        # Compute conv(a_fixed, b)
        c = fast_convolution(a_fixed, b)
        max_val = c.max()
        
        # Compute subgradient: gradient of conv(a_fixed, b)_{k*} w.r.t. b
        k_star = np.argmax(c)
        # grad[j] = a_fixed[k_star - j] for valid j
        grad = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        if js <= je:

            grad[idx] = a_fixed[k_star - idx]

        log_b -= log_b.max()

        b_new = b_new / b_new.sum()
        b = b_new
    
    return b

def coordinate_descent_refine(a0, time_limit, seed=42):
    """
    Coordinate descent: move mass from peak-contributing positions to non-peak positions.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolution(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolution(a)
        max_conv = conv.max()
        
        # Find top peaks
        n_peaks = min(20, 2 * n - 1)
        peak_lags = np.argsort(conv)[::-1][:n_peaks]
        
        improved = False
        for k in peak_lags:
            if conv[k] < max_conv * 0.95:
                break
            
            # Find contributing pairs
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            total = contrib.sum()
            if total < 1e-15:
                continue
            
            # Top contributors
            top_c = np.argsort(contrib)[::-1][:min(5, len(idx))]
            for ci in top_c:
                src = idx[ci]
                other = k - src
                
                # Find good target: position that minimizes contribution to all top peaks
                # Score each position by its "exposure" to peaks
                exposure = np.zeros(n)
                for k2 in peak_lags[:10]:
                    js2 = max(0, k2 - n + 1)
                    je2 = min(n - 1, k2)
                    idx2 = np.arange(js2, je2 + 1)
                    exposure[idx2] += a[k2 - idx2]
                    exposure[k2 - idx2] += a[idx2]
                
                # Target: low exposure, low current value
                target_score = -exposure - 0.1 * a
                # Exclude source
                target_score[src] = -np.inf
                if src != other:
                    target_score[other] -= exposure[other]  # penalty
                
                dst = np.argmax(target_score)
                if dst == src:
                    continue
                
                # Try moving mass
                for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolution(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            if stall % 5 == 0:
                n_pert = rng.integers(1, 6)
                for _ in range(n_pert):
                    src = rng.integers(0, n)
                    dst = rng.integers(0, n)
                    if src == dst:
                        continue
                    frac = rng.uniform(0.001, 0.01)
                    a_trial = a.copy()
                    amt = a_trial[src] * frac
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolution(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
            
            if stall > 2000:
                break
    
    return best_a, best_score

def subgradient_refine(a0, time_limit, n_iters=20000):
    """
    Subgradient method on the actual (non-convex) objective:
    f(a) = max_k conv(a,a)_k
    Using projected subgradient descent on the simplex.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolution(a).max()
    t0 = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolution(a)
        max_val = conv.max()
        
        # Subgradient of max_k conv(a,a)_k
        # d/da_i [conv(a,a)_k] = a[k-i] + a[i-k] (for valid indices)
        top_peaks = np.argsort(conv)[::-1][:5]
        weights = np.exp((conv[top_peaks] - max_val) * 5.0)
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for w, k in zip(weights, top_peaks):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            other = k - idx
            grad[idx] += w * a[other]
            grad[other] += w * a[idx]
        
        # Gradient step with projection onto simplex
        lr = 0.1 / (it + 10) ** 0.5
        a_new = a - lr * grad
        # Project onto simplex {x >= 0, sum(x) = 1}
        # Simple approach: clip and normalize
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        score = 2.0 * n * fast_convolution(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a, best_score

def interpolate_to_length(a, new_n):
    """Interpolate a sequence to a new length, preserving normalization."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_interp = np.interp(new_x, old_x, a_norm)
    a_interp = np.maximum(a_interp, 0)
    ss = a_interp.sum()
    if ss > 1e-15:
        a_interp = a_interp / ss
    return a_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated Nesterov iteration with mirror descent inner oracle (no LP needed)

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_conv(a, b=None):
    """Full convolution using FFT. If b is None, computes a*a."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    B = rfft(np.asarray(b, dtype=np.float64), fft_size)
    return irfft(A * B, fft_size)[:size]


def objective(a):
    """Compute the score for a sequence (scale-invariant)."""
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    an = a / s
    n = len(an)
    c = fast_conv(an)
    return 2.0 * n * c.max()


def mirror_descent_response(a_fixed, b_init=None, n_inner=40, base_eta=None):
    """
    Approximate best response: min max_k conv(a_fixed, b)_k s.t. b>=0, sum(b)=1.
    Uses mirror descent (exponential gradient) on the simplex.
    """
    n = len(a_fixed)
    af = np.asarray(a_fixed, dtype=np.float64)
    s = af.sum()
    if s < 1e-15:
        return np.ones(n) / n
    af = af / s

    if b_init is None:
        b = np.ones(n) / n
    else:
        b = np.maximum(np.asarray(b_init, dtype=np.float64), 1e-30)
        b = b / b.sum()

    if base_eta is None:
        base_eta = 1.0 / (n ** 0.3)

    eta = base_eta

    for it in range(n_inner):
        # Adaptive eta: decrease slowly
        eta = base_eta / (1.0 + 0.01 * it)

        # Compute conv(af, b)
        c = fast_conv(af, b)
        max_val = c.max()

        # Subgradient: use top-3 peaks with exponential weights
        top3 = np.argsort(c)[::-1][:3]
        w = np.exp((c[top3] - max_val) * 8.0)
        w = w / w.sum()

        grad = np.zeros(n)
        for wi, k in zip(w, top3):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            grad[idx] += wi * af[k - idx]

        # Mirror descent step
        log_b = np.log(b) - eta * grad
        log_b -= log_b.max()
        b_new = np.exp(log_b)
        b_new = b_new / b_new.sum()
        b = b_new

    return b


def nesterov_fft(a0, time_limit, verbose=True, damping=1.0):
    """
    Nesterov-accelerated iteration with FFT-based mirror descent oracle.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()
    b_prev = None

    best_a = a.copy()
    best_score = 2.0 * n * fast_conv(a).max()

    t0 = time.monotonic()
    n_iters = 0
    stall = 0
    best_at_stall = best_score

    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break

        # Inner oracle: more iterations for better accuracy as we progress
        n_inner = min(60, 15 + t // 50)
        eta = max(0.01, 1.0 / (n ** 0.3) * (1.0 / (1.0 + 0.005 * t)))
        b_star = mirror_descent_response(y, b_init=b_prev, n_inner=n_inner, base_eta=eta)
        b_prev = b_star
        n_iters += 1

        # Nesterov update
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_new_s = y_new.sum()
        if y_new_s < 1e-15:
            y_new = y.copy()
            y_new_s = 1.0
        y_new = y_new / y_new_s

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            a_new_s = a_new.sum()
            if a_new_s < 1e-15:
                a_new = y_new.copy()
                a_new_s = 1.0
            a_new = a_new / a_new_s

        # Damping
        if damping < 1.0:
            a_new = (1.0 - damping) * y + damping * a_new
            a_new = np.maximum(a_new, 0)
            a_new_s = a_new.sum()
            if a_new_s < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / a_new_s

        y = y_new
        a = a_new

        # Track best
        score_a = 2.0 * n * fast_conv(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        score_y = 2.0 * n * fast_conv(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0

        # Check b_star directly
        score_b = 2.0 * n * fast_conv(b_star).max()
        if score_b < best_score:
            best_score = score_b
            best_a = b_star.copy()
            stall = 0

        if verbose and n_iters % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NF iter={n_iters} ({elapsed:.0f}s): best={best_score:.6f}")

        if stall > 5000 and n_iters > 5000:
            if verbose:
                print(f"      NF stalled at iter={n_iters}, best={best_score:.6f}")
            break

    return best_a, best_score, n_iters


def coord_descent(a0, time_limit, seed=42):
    """
    Coordinate descent: move mass from high-peak contributors to low-exposure positions.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, float('inf'), 0
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_conv(a).max()
    t0 = time.monotonic()
    n_iters = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_conv(a)
        max_conv = conv.max()
        n_iters += 1

        # Top peaks
        n_peaks = min(15, 2 * n - 1)
        peak_lags = np.argsort(conv)[::-1][:n_peaks]

        improved = False

        # Compute exposure map: how much each position contributes to top peaks
        exposure = np.zeros(n)
        for k in peak_lags:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            weight = conv[k] / max_conv
            exposure[idx] += weight * a[k - idx]
            exposure[k - idx] += weight * a[idx]

        for k in peak_lags[:8]:
            if conv[k] < max_conv * 0.9:
                break
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            if contrib.sum() < 1e-15:
                continue

            top_c = np.argsort(contrib)[::-1][:min(3, len(idx))]
            for ci in top_c:
                src = idx[ci]
                other = k - src

                # Target: minimize exposure
                target_score = -exposure - 0.05 * a
                target_score[src] = -np.inf
                dst = np.argmax(target_score)

                for frac in [0.002, 0.005, 0.01, 0.02, 0.03, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_conv(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall % 20 == 1:
                # Random multi-move perturbation
                n_pert = rng.integers(2, 8)
                a_trial = a.copy()
                for _ in range(n_pert):
                    src = rng.integers(0, n)
                    dst = rng.integers(0, n)
                    if src == dst:
                        continue
                    frac = rng.uniform(0.002, 0.02)
                    amt = a_trial[src] * frac
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss < 1e-15:
                    continue
                a_trial = a_trial / ss
                score = 2.0 * n * fast_conv(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0

            if stall > 3000:
                if n_iters > 50:
                    break

        if n_iters % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      CD iter={n_iters} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score, n_iters


def smooth_subgradient(a0, time_limit, n_iters=50000, beta=30.0):
    """
    Subgradient method with log-sum-exp smoothing of the max.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, float('inf'), 0
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_conv(a).max()
    t0 = time.monotonic()
    n_done = 0

    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        n_done += 1

        conv = fast_conv(a)
        max_val = conv.max()

        # Smoothed subgradient: weighted average of top peaks
        top_k = np.argsort(conv)[::-1][:8]
        w = np.exp((conv[top_k] - max_val) * beta)
        w = w / w.sum()

        grad = np.zeros(n)
        for wi, k in zip(w, top_k):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            other = k - idx
            grad[idx] += wi * a[other]
            grad[other] += wi * a[idx]

        # Gradient step with adaptive learning rate
        lr = 0.3 / (it + 20) ** 0.6
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0)
        ss = a_new.sum()
        if ss < 1e-15:
            break
        a_new = a_new / ss

        score = 2.0 * n * fast_conv(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()

        a = a_new

        if it % 10000 == 0:
            print(f"      SG iter={it}: best={best_score:.6f}")

    return best_a, best_score, n_done


def interpolate_length(a, new_n):
    """Interpolate sequence to new length, preserving non-negativity and normalization."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    an = a / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    a_interp = np.interp(new_x, old_x, an)
    a_interp = np.maximum(a_interp, 0)
    ss = a_interp.sum()
    if ss > 1e-15:
        a_interp = a_interp / ss
    return a_interp

            t4b = min(remaining * 0.10, 60.0)
            if t4b > 10:
                a_out2, sc_out2, _ = coord_descent(a_out, t4b, seed=456)
                try_update(a_out2, f"cd_n{n_try}")

    # Phase 5: Second Nesterov round on global best
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = min(remaining * 0.25, 200.0)
    if t5 > 30:
        print(f"\n=== Phase 5: Nesterov round2 n={len(best_seq)} (t={t5:.0f}s) ===")
        a_out, sc_out, iters5 = nesterov_fft(best_seq, t5, verbose=False)
        try_update(a_out, "nf_r2")

    # Phase 6: Final CD
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 20:
        t6 = min(remaining * 0.5, 120.0)
        if t6 > 10:
            print(f"\n=== Phase 6: Final CD (t={t6:.0f}s) ===")
            a_out, sc_out, _ = coord_descent(best_seq, t6, seed=789)
            try_update(a_out, "cd_final")

    # Final verification
    if best_seq is not None:
        a_final = [float(max(0.0, min(1000.0, x))) for x in best_seq]
        final_score = evaluate_sequence(a_final)
        print(f"\nFINAL: score={final_score:.6f} (n={len(best_seq)})")
        return a_final
    else:
        print("ERROR: No valid sequence!")
        return [1.0]
