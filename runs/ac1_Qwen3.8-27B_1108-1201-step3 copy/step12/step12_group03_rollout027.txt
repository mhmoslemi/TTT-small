def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def softmax(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def compute_gradient(a, top_k=1):
    """
    Compute subgradient of max_k b_k where b = conv(a,a).
    Returns gradient vector and the active lags.
    Uses top_k lags weighted by proximity to max.
    """
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    
    # Get top lags
    sorted_idx = np.argsort(b)[::-1]
    top_lags = sorted_idx[:top_k]
    
    # Weight by softness (how close to max)
    weights = np.exp((b[top_lags] - max_b) * 20)  # Sharp: only max matters
    weights = weights / weights.sum()
    
    grad = np.zeros(n)
    for w, k in zip(weights, top_lags):
        # db_k/da_i = 2*a_{k-i} (if 0 <= k-i < n)
        # So grad_i = 2*w*a_{k-i}
        # a_{k-i} is valid when 0 <= k-i < n, i.e., max(0, k-n+1) <= i <= min(n-1, k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            grad[lo:hi+1] += 2 * w * a[k - np.arange(lo, hi + 1)]
    
    return grad, max_b, top_lags


def adam_simplex_optimize(a0, time_limit, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8, 
                          top_k=3, verbose=True):
    """
    Optimize max_k b_k using Adam with softmax parametrization.
    a = softmax(z), so we optimize z directly.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a = a / a.sum()
    
    # Initialize z from a: z = log(a)
    z = np.log(a)
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Adaptive learning rate schedule
    base_lr = lr
    decay_start = int(time_limit * 0.7)  # Start decaying after 70% of time
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        
        # Compute gradient
        grad_z, max_b, top_lags = compute_gradient(a, top_k=top_k)
        
        # Convert gradient to z-space
        # d(f)/dz_i = a_i * (grad_i - sum_j grad_j * a_j)
        mean_grad = np.dot(grad_z, a)
        grad_z_space = a * (grad_z - mean_grad)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_z_space
        v = beta2 * v + (1 - beta2) * grad_z_space ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        # Adaptive LR: decay
        if elapsed < decay_start:
            current_lr = base_lr
        else:
            progress = (elapsed - decay_start) / max(1, time_limit - decay_start)
            current_lr = base_lr * (1 - progress) ** 2
        
        z -= current_lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Convert back to a
        a_new = softmax(z)
        
        # Evaluate
        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and t % 1000 == 0:
            print(f"    Adam t={t} ({elapsed:.0f}s): best={best_score:.6f}, cur={score:.6f}")
        
        # Check for numerical issues
        if not np.isfinite(score):
            break
    
    return best_a, best_score


def peak_equalization(a0, time_limit, n_active=5, verbose=True):
    """
    Try to equalize the top peaks by adjusting the sequence.
    Uses a simple approach: for each peak, compute its "gradient direction"
    and take a small step to reduce it while not increasing others too much.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    a = a / a.sum()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_autocorr(a)
        max_b = b.max()
        top_lags = np.argsort(b)[::-1][:n_active]
        
        # Try to reduce each top peak
        improved = False
        for k in top_lags:
            # Gradient of b_k: 2*a shifted
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            shift = np.zeros(n)
            shift[lo:hi+1] = 2 * a[k - np.arange(lo, hi + 1)]
            
            # Project gradient to simplex tangent space
            mean_g = np.dot(shift, a)
            grad_proj = a * (shift - mean_g)  # In z-space
            
            # Take a small step in negative gradient direction (in z-space)
            # z = log(a), so a_new = a * exp(-lr * grad_proj / a) = a * exp(-lr * (shift - mean_g))
            # Wait, that's not quite right. In z-space, the step is -lr * grad_z_space
            # a_new = softmax(z - lr * grad_z_space) = softmax(z) * exp(-lr * grad_z_space) / Z
            # = a * exp(-lr * (shift - mean_g)) / Z
            
            for lr in [0.001, 0.005, 0.01, 0.02, 0.05]:
                # In z-space, grad = a * (shift - mean_g), so step in z is -lr * a * (shift - mean_g)
                # a_new[i] = a[i] * exp(-lr * a[i] * (shift[i] - mean_g)) ... no
                # Actually: z_new = z - lr * grad_z_space
                # a_new = exp(z_new) / sum(exp(z_new)) = exp(z) * exp(-lr * grad_z_space) / Z
                # = a * exp(-lr * a * (shift - mean_g)) / Z
                exponent = -lr * a * (shift - mean_g)
                a_trial = a * np.exp(exponent)
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score:
                    best_score = score
                    best_a = a_trial.copy()
                    a = a_trial
                    improved = True
                    break
        
        if not improved:
            # Try random perturbation
            rng = np.random.default_rng(it)
            for _ in range(20):
                idx = rng.choice(n, size=2, replace=False)
                frac = rng.uniform(0.001, 0.01)
                amt = a[idx[0]] * frac
                a_trial = a.copy()
                a_trial[idx[0]] -= amt
                a_trial[idx[1]] += amt
                if (a_trial >= 0).all():
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        best_score = score
                        best_a = a_trial.copy()
                        a = a_trial
                        improved = True
                        break
        
        if not improved:

            if it % 500 == 0 and verbose:
                print(f"    PEAK it={it}: best={best_score:.6f}, n_active={n_active}")
            if n_active < 2:
                break
    
    return best_a, best_score

        #
        # Let me just use: the Hessian approach.
        # f(a) = sum_k p_k * (a*a)_k
        # grad_f = 2 * (p a) where (p a)_i = sum_k p_k a_{k-i} (valid)
        # This is a linear operator applied to a, so grad is linear in a.
        # 
        # To minimize f(a) over the simplex, we can use mirror descent:
        # a_new = a * exp(-eta * grad_f) / Z
        
        # Compute grad_f = 2 * linear_op(p) @ a
        # linear_op(p) @ a has components: [linear_op(p) @ a]_i = sum_k p_k * a_{k-i}
        # This is a cross-correlation.
        
        # Using FFT: if we define the cross-correlation properly...
        # Actually, let's just compute it directly for now (might be slow for large n).
        # For efficiency, use the fact that it's a Toeplitz matrix-vector product.
        
        # [linear_op(p) @ a]_i = sum_{k=0}^{2n-2} p_k * a_{k-i} (where 0 <= k-i < n)
        # = sum_{j=0}^{n-1} p_{i+j} * a_j  (substituting j = k-i, so k = i+j)
        # Valid when 0 <= i+j <= 2n-2, i.e., j <= 2n-2-i (always true for j < n when i >= 0)
        # So [linear_op(p) @ a]_i = sum_{j=0}^{n-1} p_{i+j} * a_j
        
        # This is a Toeplitz matrix-vector product! The matrix T has T[i,j] = p_{i+j}.
        # We can compute this efficiently using FFT:
        # T @ a = ifft(fft(p_full) * fft(a_padded)) ... not exactly.
        
        # Actually, for a Toeplitz matrix with T[i,j] = p_{i+j}, we can write:
        # (T@a)_i = sum_j p_{i+j} a_j
        # Let's define the vector p_ext of length 2n-1 (already have it as p).
        # Then (T@a)_i = sum_{j=0}^{n-1} p_ext[i+j] * a_j
        # This is the i-th element of the "anti-convolution" or we can use:
        # Reverse a: a_rev[j] = a[n-1-j]
        # Then (T@a)_i = sum_j p_ext[i+j] * a[n-1-j] = (p_ext * a_rev)[n-1+i] ... 
        # Hmm, let me just use the direct formula with numpy.
        
        # For n up to ~5000, the direct O(n^2) computation might be too slow.
        # Let me use a different approach: since f(a) = sum_k p_k b_k and b = conv(a,a),
        # f(a) = sum_k p_k b_k = dot(p, b) = dot(p, conv(a,a))
        # The gradient of dot(p, conv(a,a)) w.r.t. a is:
        # = 2 * conv(p, a) ... no, that's not right either.
        #
        # Let's use the property: d/da [conv(a,a)] = 2 * conv(a, .) (as an operator)
        # So grad = 2 * conv(p_rev, a) where p_rev = p[::1] (but we need to be careful about indexing)
        #
        # Actually, the cleanest way: use the fact that for any vectors u, v, w:
        # d/da [u^T conv(a, v)] = 2 * conv(u_rev, v) + conv(u, v_rev) ... no.
        #
        # Let me just use the subgradient approach with the top lag.
        # For the purpose of the Nesterov iteration, I'll use:
        # - Player 2's strategy: put weight on top lags
        # - Player 1's response: minimize the weighted sum using a gradient step
        
        # Update p (Player 2's strategy) based on current a:
        b = fast_autocorr(a)
        # Softmax over b to get p (higher b_k means player 2 plays k more)
        temp = 1.0 / (max_b if (max_b := b.max()) > 0 else 1.0)
        p_new = np.exp(b * temp - b.max())
        p_s = p_new.sum()
        if p_s < 1e-300:
            p_new = np.ones(n_lags) / n_lags
        else:
            p_new = p_new / p_s
        
        # Compute gradient of dot(p_new, b) w.r.t. a
        # = 2 * (p_new cross-correlate a) ... let me just use the top-lag approximation
        # For efficiency, use the top few lags:
        top_idx = np.argsort(b)[::-1][:5]
        top_weights = p_new[top_idx]
        top_weights = top_weights / max(top_weights.sum(), 1e-300)
        
        grad = np.zeros(n)
        for w, k in zip(top_weights, top_idx):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                grad[lo:hi+1] += 2 * w * a[k - np.arange(lo, hi + 1)]
        
        # Mirror descent step for player 1:
        eta = 2.0 / (t + 2.0)
        # a_new = a * exp(-eta * grad) / Z
        log_update = -eta * grad
        log_update -= log_update.max()
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 0)
        a_s = a_new.sum()
        if a_s < 1e-30:
            a_new = a.copy()
        else:
            a_new = a_new / a_s
        
        p = p_new
        a = a_new
        
        # Evaluate
        score = 2.0 * n * fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest t={t} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score


def general_result_double_or_nothing(a0, time_limit, solver, verbose=True):
    """
    Generalized Nesterov with the LP best response.
    This is the approach from the previous program but with improvements.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    for t in range(1, 200000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # LP best response: minimize M s.t. T(a)@b <= M, sum(b)=1, b>=0
        # where T is the (2n-1) x n convolution matrix
        # This is equivalent to: find b that minimizes max_k (a*b)_k
        # which is the same game from the other side.
        
        # For large n, solving the LP each iteration is expensive.
        # Instead, use the analytic best response:
        # Given a, the best response b minimizes max_k (a*b)_k.
        # This is a convex optimization problem (min of max of linear functions).
        # We can solve it with a simple bisection + LP, or use the dual.
        
        # For now, use a simpler approach: the mirror descent step
        # with the subgradient of max_k b_k.
        
        b = fast_autocorr(a)
        max_b = b.max()
        
        # Subgradient approach: use top lags
        top_k_count = min(5, 2 * n - 1)
        top_lags = np.argsort(b)[::-1][:top_k_count]
        weights = np.exp((b[top_lags] - max_b) * 10)
        weights = weights / weights.sum()
        
        grad = np.zeros(n)
        for w, k in zip(weights, top_lags):
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            if lo <= hi:
                grad[lo:hi+1] += 2 * w * a[k - np.arange(lo, hi + 1)]
        
        # Nesterov step
        alpha_t = 2.0 / (t + 2.0)
        
        # Mirror descent on y
        eta = alpha_t * 0.1  # Scale factor
        log_update = -eta * grad
        log_update -= log_update.max()
        y_new = y * np.exp(log_update)
        y_new = np.maximum(y_new, 0)
        y_s = y_new.sum()
        if y_s < 1e-30:
            y_new = y.copy()
        else:
            y_new = y_new / y_s
        
        if t == 1:
            a_new = y_new.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            a_s = a_new.sum()
            if a_s < 1e-30:
                a_new = y_new.copy()
            else:
                a_new = a_new / a_s
        
        y = y_new
        a = a_new
        
        score = 2.0 * n * fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if verbose and t % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    GenRes t={t} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver, verbose=False):
    """Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build constraints efficiently
    # (T @ b)[k] = sum_j a_fixed[j] * b[k-j] (valid indices)
    # For k = 0..2n-2:
    #   (T@b)[k] = sum_{j=max(0,k-n+1)}^{min(n-1,k)} a_fixed[j] * b[k-j]
    
    # This creates a lot of constraints. For large n, this is expensive.
    # Use a subsample of lags for scalability.
    n_lags = 2 * n - 1
    # Sample lags: always include 0 and the extremes, plus evenly spaced
    if n_lags <= 200:
        lag_indices = np.arange(n_lags)
    else:
        # Sample ~200 lags
        n_sample = 200
        lag_indices = np.unique(np.concatenate([
            [0, n - 1, 2 * n - 2],  # Key lags
            np.linspace(0, n_lags - 1, n_sample).astype(int)
        ]))
    
    constraints = [cp.sum(b_var) == 1]
    for k in lag_indices:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        idx = np.arange(js, je + 1)
        b_idx = k - idx
        expr = a_fixed[idx] @ b_var[b_idx]
        constraints.append(expr <= M_var)
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_full(a0, time_limit, solver, verbose=True):
    """
    Full LP Nesterov with actual LP solving for best response.
    More expensive per iteration but potentially better convergence.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    n_iters = 0
    
    for t in range(1, 50000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Solve LP best response
        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue
        n_iters += 1
        
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_new_s = y_new.sum()
        if y_new_s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / y_new_s
        
        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            a_s = a_new.sum()
            if a_s < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / a_s
        
        y = y_new
        a = a_new
        
        score = 2.0 * n * fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        if verbose and n_iters % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Nest t={t} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.8f}")
    
    return best_a, best_score, n_iters


def coordinate_descent(a0, time_limit, verbose=True):
    """
    Coordinate descent: adjust pairs of coordinates to reduce the max autocorrelation.
    """
    n = len(a0)
    rng = np.random.default_rng(42)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    stall = 0
    max_stall = 200
    
    while time.monotonic() - t0 < time_limit and stall < max_stall:
        b = fast_autocorr(a)
        max_b = b.max()
        top_lag = np.argmax(b)
        
        improved = False
        
        # Try random pairs
        n_tries = min(500, max(50, n // 10))
        for _ in range(n_tries):
            i, j = rng.choice(n, size=2, replace=False)
            # Try moving mass from i to j
            for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
                amt = a[i] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if improved:
            stall = 0
        else:
            stall += 1
        
        if verbose and stall % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD ({elapsed:.0f}s): best={best_score:.6f}, stall={stall}")
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search combining multiple optimization approaches:
    1. Adam with analytic gradients (fast, good for local refinement)
    2. LP Nesterov (global exploration)
    3. Peak equalization (targeted peak reduction)
    4. Coordinate descent (fine-tuning)
    5. Multi-length exploration
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    
    # Get parent sequence
    best_seq = None
    best_score = float('inf')
    
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                best_seq = parent.copy()
                best_score = p_score
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            else:
                best_seq = parent.copy()
                best_score = 2.0 * len(parent) * fast_autocorr(parent).max()
                print(f"Parent (local eval): n={len(parent)}, score={best_score:.6f}")
        else:
            raise ValueError("Parent invalid")
    except Exception:
        # Default initialization
        n_base = 1600
        x = np.linspace(0, np.pi, n_base)
        best_seq = np.cos(x / 2) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = 2.0 * n_base * fast_autocorr(best_seq).max()
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        # Use local evaluation for speed, verify with evaluate_sequence at the end
        sc = 2.0 * len(a) * fast_autocorr(a).max()
        if sc < best_score - 1e-10:
            # Verify with actual evaluator (but this is expensive, so only do it sometimes)
            if len(a) <= 3000:
                actual_sc = evaluate_safe(a)
                if np.isfinite(actual_sc) and actual_sc < best_score - 1e-10:
                    best_score = actual_sc
                    best_seq = a.copy()
                    elapsed = time.monotonic() - t_global_start
                    print(f"  *** NEW BEST [{label}]: {best_score:.6f} (n={len(a)}, t={elapsed:.0f}s)")
                    return
            else:
                # For large sequences, trust the local eval
                best_score = sc
                best_seq = a.copy()
                elapsed = time.monotonic() - t_global_start
                print(f"  *** NEW BEST [{label}]: {best_score:.6f} (n={len(a)}, t={elapsed:.0f}s}")
    
    n_base = len(best_seq)
    remaining = time_budget - (time.monotonic() - t_global_start)
    
    # Phase 1: Adam optimization at base length (fast, good local optimizer)
    print(f"\n=== Phase 1: Adam at n={n_base} (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.25, 350.0)
    if t1 > 30:
        a_out, sc_out = adam_simplex_optimize(best_seq, t1, lr=0.02, top_k=3, verbose=True)
        print(f"  Adam done: score={sc_out:.6f}")
        try_update(a_out, f"adam_n{n_base}")
    
    # Phase 2: LP Nesterov (global exploration)
    remaining = time_budget - (time.monotonic() - t_global_start)
    solver = get_lp_solver()
    print(f"\n=== Phase 2: LP Nesterov at n={n_base} (remaining={remaining:.0f}s, solver={solver}) ===")
    t2 = min(remaining * 0.30, 450.0)
    if t2 > 60:
        a_out, sc_out = nesterov_lp_full(best_seq, t2, solver, verbose=True)
        print(f"  LP Nesterov done: score={sc_out:.6f}")
        try_update(a_out, f"lp_n{n_base}")
    
    # Phase 3: Second Adam pass (refine after LP)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: Adam refinement at n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 20:
        a_out, sc_out = adam_simplex_optimize(best_seq, t3, lr=0.005, top_k=5, verbose=True)
        print(f"  Adam refine: score={sc_out:.6f}")
        try_update(a_out, f"adam2_n{len(best_seq)}")
    
    # Phase 4: Peak equalization
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: Peak equalization at n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 150.0)
    if t4 > 15:
        a_out, sc_out = peak_equalization(best_seq, t4, n_active=5, verbose=True)
        print(f"  Peak eq: score={sc_out:.6f}")
        try_update(a_out, f"peak_eq_n{len(best_seq)}")
    
    # Phase 5: Try nearby lengths
    for n_try in [n_base + 100, n_base - 100, n_base + 200, n_base + 300]:
        if n_try < 500:
            continue
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining < 100:
            break
        print(f"\n=== Phase 5: Adam at n={n_try} (remaining={remaining:.0f}s) ===")
        t5 = min(remaining * 0.20, 150.0)
        if t5 < 30:
            continue
        init_try = interpolate_sequence(best_seq, n_try)
        a_out, sc_out = adam_simplex_optimize(init_try, t5, lr=0.01, top_k=3, verbose=False)
        print(f"  Adam n={n_try}: score={sc_out:.6f}")
        try_update(a_out, f"adam_n{n_try}")
        
        # Quick MC refinement
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining > 50:
            t5b = min(remaining * 0.10, 60.0)
            if t5b > 10:
                a_out2, sc_out2 = coordinate_descent(a_out, t5b, verbose=False)
                try_update(a_out2, f"cd_n{n_try}")
    
    # Phase 6: Final Adam + Coordinate descent on global best
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 6: Final refinement at n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    if remaining > 30:
        t6 = min(remaining * 0.4, 120.0)
        a_out, sc_out = adam_simplex_optimize(best_seq, t6, lr=0.001, top_k=5, verbose=True)
        try_update(a_out, "final_adam")
    
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 20:
        t6b = min(remaining * 0.4, 80.0)
        a_out, sc_out = coordinate_descent(best_seq, t6b, verbose=True)
        try_update(a_out, "final_cd")
    
    # Final verification with actual evaluator
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        else:
            # Fallback: use parent if we have one
            try:
                p = np.asarray(height_sequence_1, dtype=np.float64)
                p = np.clip(p, 0.0, 1000.0)
                if p.sum() > 0.01:
                    fs = evaluate_safe(p)
                    if np.isfinite(fs):
                        best_seq = p
                        best_score = fs
            except Exception:
                pass
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adam optimizer with softmax parametrization and analytic subgradients
2. LP Nesterov (minimax game theory) for global exploration
3. Peak equalization via targeted gradient steps
4. Multi-length and multi-start exploration
5. Coordinate descent fine-tuning

import numpy as np
import time
from scipy.fft import rfft, irfft
import cvxpy as cp


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def objective_local(a):
    """Compute objective 2*n*max(conv(a,a))/(sum(a))^2 locally."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_np = np.asarray(a, dtype=np.float64)
        a_list = [float(max(0.0, min(1000.0, x))) for x in a_np]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def compute_subgradient(a, top_k=3, sharpness=15.0):
    """
    Compute subgradient of max_k b_k where b = conv(a,a).
    Uses soft weighting over top lags for smoother gradients.
    Returns (gradient vector, max_b, top_lags).
    """
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    if max_b < 1e-30:
        return np.zeros(n), 0.0, np.array([0])
    
    sorted_idx = np.argsort(b)[::-1]
    top_lags = sorted_idx[:min(top_k, 2 * n - 1)]
    
    # Softmax weighting over top lags
    weights = np.exp((b[top_lags] - max_b) * sharpness)
    wsum = weights.sum()
    if wsum < 1e-300:
        weights = np.ones(len(top_lags)) / len(top_lags)
    else:
        weights = weights / wsum
    
    grad = np.zeros(n)
    for w, k in zip(weights, top_lags):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            indices = np.arange(lo, hi + 1)
            grad[indices] += 2.0 * w * a[k - indices]
    
    return grad, max_b, top_lags


def adam_optimize(a0, time_limit, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8,
                  top_k=3, sharpness=15.0, verbose=False, verbose_interval=500):
    """
    Adam optimizer with softmax parametrization for minimizing max_k b_k.
    a = softmax(z), optimize z directly.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a = a / a.sum()
    
    # Initialize z from a
    z = np.log(a)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    best_a = a.copy()
    best_score = objective_local(a)
    t0 = time.monotonic()
    
    # Adaptive learning rate
    base_lr = lr
    warmup = min(100, time_limit // 10)
    decay_start_ratio = 0.7
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        elapsed = time.monotonic() - t0
        
        # Compute subgradient in a-space
        grad_a, max_b, _ = compute_subgradient(a, top_k=top_k, sharpness=sharpness)
        if not np.isfinite(max_b):
            break
        
        # Convert to z-space gradient
        # d(f)/dz_i = a_i * (grad_a_i - sum_j grad_a_j * a_j)
        mean_grad = np.dot(grad_a, a)
        grad_z = a * (grad_a - mean_grad)
        
        # Check gradient magnitude
        grad_norm = np.linalg.norm(grad_z)
        if grad_norm < 1e-15:
            break
        
        # Learning rate schedule
        if t < warmup:
            current_lr = base_lr * (t / warmup)
        else:
            decay_start = decay_start_ratio * time_limit
            if elapsed < decay_start:
                current_lr = base_lr
            else:
                progress = (elapsed - decay_start) / max(1.0, time_limit - decay_start)
                current_lr = base_lr * max(0.0, (1.0 - progress) ** 2)
        
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * grad_z * grad_z
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        
        z -= current_lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Convert back
        a_new = softmax_fn(z)
        
        # Periodically renormalize z to prevent numerical drift
        if t % 1000 == 0:
            z = np.log(np.maximum(a_new, 1e-30))
        
        score = objective_local(a_new)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        
        if verbose and t % verbose_interval == 0:
            print(f"    Adam t={t} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score


def nesterov_mirror(a0, time_limit, top_k=5, sharpness=10.0, scale=0.1, verbose=False, verbose_interval=500):
    """
    Nesterov-accelerated mirror descent on the simplex.
    Uses multiplicative weights (exponential) update rule.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()
    
    best_a = a.copy()
    best_score = objective_local(a)
    t0 = time.monotonic()
    stall = 0
    stall_patience = 5000
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute subgradient
        grad_a, max_b, _ = compute_subgradient(a, top_k=top_k, sharpness=sharpness)
        if not np.isfinite(max_b):
            break
        
        alpha_t = 2.0 / (t + 2.0)
        eta = alpha_t * scale
        
        # Mirror descent step on y
        log_update = -eta * grad_a
        log_update -= log_update.max()
        y_new = y * np.exp(log_update)
        y_new = np.maximum(y_new, 0)
        y_s = y_new.sum()
        if y_s < 1e-30:
            y_new = y.copy()
        else:
            y_new = y_new / y_s
        
        # Nesterov extrapolation
        if t == 1:
            a_new = y_new.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            a_s = a_new.sum()
            if a_s < 1e-30:
                a_new = y_new.copy()
            else:
                a_new = a_new / a_s
        
        y = y_new
        a = a_new
        
        score = objective_local(a)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if verbose and t % verbose_interval == 0:
            elapsed = time.monotonic() - t0
            print(f"    Nesterov t={t} ({elapsed:.0f}s): best={best_score:.6f}")
        
        if stall > stall_patience:
            # Reset with perturbation
            rng = np.random.default_rng(t)
            a_pert = a + rng.normal(0, 0.01, n)
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            a = a_pert
            y = a_pert.copy()
            stall = 0
    
    return best_a, best_score

def solve_lp_best_response(a_fixed, solver, n_max_constraints=300):
    """
    Solve LP: min M s.t. (a_fixed * b)_k <= M for sampled k, sum(b)=1, b>=0.
    Uses subsampled constraints for scalability.
    """
    n = len(a_fixed)
    n_lags = 2 * n - 1
    
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Sample important lags
    if n_lags <= n_max_constraints:
        lag_indices = np.arange(n_lags)
    else:
        # Always include key lags + evenly spaced
        key_lags = np.array([0, n // 2, n - 1, 2 * n - 2])
        even = np.linspace(0, n_lags - 1, n_max_constraints - len(key_lags)).astype(int)
        lag_indices = np.unique(np.concatenate([key_lags, even]))
    
    constraints = [cp.sum(b_var) == 1]
    for k in lag_indices:
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        idx = np.arange(js, je + 1)
        b_idx = k - idx
        expr = a_fixed[idx] @ b_var[b_idx]
        constraints.append(expr <= M_var)
    
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def generate_initialization(n, method='cosine', seed=42):
    """Generate various initializations."""
    if method == 'cosine':
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif method == 'sine':
        x = np.linspace(0, np.pi, n)
        a = np.sin(x / 2) ** 2
    elif method == 'bump':
        x = np.linspace(0, 1, n)
        sigma = 0.3
        a = np.exp(-((x - 0.5) ** 2) / (2 * sigma ** 2))
    elif method == 'plateau':
        x = np.linspace(0, 1, n)
        a = np.ones(n)
        taper = int(0.2 * n)
        if taper > 0:
            ramp = np.linspace(0, 1, taper)
            a[:taper] = ramp
            a[-taper:] = ramp[::-1]
    elif method == 'random_smooth':
        rng = np.random.default_rng(seed)
        a = rng.random(n)
        # Smooth with a moving average
        k = max(3, n // 50)
        kernel = np.ones(k) / k
        a = np.convolve(a, kernel, mode='same')
    else:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search: multi-phase optimization combining Adam, Nesterov, LP,
    peak reduction, and coordinate descent at multiple lengths.
    """
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    
    # Get parent sequence
    best_seq = None
    best_score = float('inf')
    
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 0.01:
            parent = p / p.sum()
            p_score = objective_local(parent)
            best_seq = parent.copy()
            best_score = p_score
            print(f"Parent: n={len(parent)}, local_score={p_score:.6f}")
            # Verify with actual evaluator
            actual = evaluate_safe(parent)
            if np.isfinite(actual):
                best_score = actual
                print(f"Parent verified score: {actual:.6f}")
        else:
            raise ValueError("Parent invalid")
    except Exception:
        n_base = 1600
        best_seq = generate_initialization(n_base, 'cosine')
        best_score = objective_local(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    
    n_base = len(best_seq)
    
    def update_best(a, label, use_evaluator=True):
        """Try to update the best sequence found so far."""
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        sc = objective_local(a)
        if sc < best_score - 1e-10:
            # Verify with evaluator for significant improvements
            if use_evaluator and np.isfinite(sc):
                actual_sc = evaluate_safe(a)
                if np.isfinite(actual_sc) and actual_sc < best_score - 1e-10:
                    best_score = actual_sc
                    best_seq = a.copy()
                    elapsed = time.monotonic() - t_global_start
                    print(f"  *** NEW BEST [{label}]: {best_score:.6f} (n={len(a)}, t={elapsed:.0f}s)")
                    return
            elif sc < best_score - 1e-10:
                best_score = sc
                best_seq = a.copy()
                elapsed = time.monotonic() - t_global_start
                print(f"  *** NEW BEST [{label}]: {best_score:.6f} (n={len(a)}, t={elapsed:.0f}s}")
    
    # Get solver
    solver = get_lp_solver()
    print(f"LP solver: {solver}")
    
    remaining = time_budget - (time.monotonic() - t_global_start)
    
    # === Phase 1: Adam optimization (fast, effective local optimizer) ===
    print(f"\n=== Phase 1: Adam at n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.20, 300.0)
    if t1 > 30:
        a_out, sc_out = adam_optimize(best_seq, t1, lr=0.02, top_k=3, sharpness=15.0,
                                      verbose=True, verbose_interval=1000)
        print(f"  Adam1 done: score={sc_out:.6f}")
        update_best(a_out, f"adam1_n{n_base}")
    
    # === Phase 2: Nesterov mirror descent ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: Nesterov mirror at n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 30:
        a_out, sc_out = nesterov_mirror(best_seq, t2, top_k=5, sharpness=10.0,
                                        scale=0.05, verbose=True, verbose_interval=2000)
        print(f"  Nesterov done: score={sc_out:.6f}")
        update_best(a_out, f"nesterov_n{len(best_seq)}")
    
    # === Phase 3: LP Nesterov (global) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: LP Nesterov at n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.25, 400.0)
    if t3 > 60:
        a_out, sc_out, n_solves = nesterov_lp(best_seq, t3, solver,
                                              verbose=True, verbose_interval=5)
        print(f"  LP done: {n_solves} solves, score={sc_out:.6f}")
        update_best(a_out, f"lp_n{len(best_seq)}")
    
    # === Phase 4: Second Adam pass (refine) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: Adam refine at n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 20:
        a_out, sc_out = adam_optimize(best_seq, t4, lr=0.005, top_k=5, sharpness=20.0,
                                      verbose=True, verbose_interval=1000)
        print(f"  Adam2 done: score={sc_out:.6f}")
        update_best(a_out, f"adam2_n{len(best_seq)}")
    
    # === Phase 5: Peak reduction ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 5: Peak reduction at n={len(best_seq)} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.10, 100.0)
    if t5 > 15:
        a_out, sc_out = peak_reduction(best_seq, t5, verbose=True)
        print(f"  PeakRed done: score={sc_out:.6f}")
        update_best(a_out, f"peakred_n{len(best_seq)}")
    
    # === Phase 6: Multi-length exploration ===
    lengths_to_try = [n_base + 100, n_base - 100, n_base + 200, n_base + 300, n_base + 500]
    for n_try in lengths_to_try:
        if n_try < 500:
            continue
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining < 80:
            break
        print(f"\n=== Phase 6: Explore n={n_try} (budget={remaining:.0f}s) ===")
        t6 = min(remaining * 0.15, 120.0)
        if t6 < 20:
            continue
        init_try = interpolate_sequence(best_seq, n_try)
        a_out, sc_out = adam_optimize(init_try, t6, lr=0.01, top_k=3, sharpness=15.0,
                                      verbose=False)
        print(f"  Adam n={n_try}: score={sc_out:.6f}")
        update_best(a_out, f"adam_n{n_try}")
        
        # Quick coordinate descent
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining > 40:
            t6b = min(remaining * 0.08, 50.0)
            if t6b > 10:
                a_out2, sc_out2 = coordinate_descent(a_out, t6b, n_tries_per_iter=100)
                update_best(a_out2, f"cd_n{n_try}")
    
    # === Phase 7: Multi-start exploration ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 7: Multi-start (budget={remaining:.0f}s) ===")
    if remaining > 60:
        for method in ['sine', 'bump', 'random_smooth']:
            remaining = time_budget - (time.monotonic() - t_global_start)
            if remaining < 40:
                break
            print(f"  Trying init: {method}")
            init = generate_initialization(n_base, method, seed=seed + 7)
            t7 = min(remaining * 0.25, 100.0)
            if t7 < 20:
                continue
            a_out, sc_out = adam_optimize(init, t7, lr=0.02, top_k=3, sharpness=10.0,
                                          verbose=False)
            print(f"    {method}: score={sc_out:.6f}")
            update_best(a_out, f"multi_{method}")
    
    # === Phase 8: Final refinement ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 8: Final refinement at n={len(best_seq)} (budget={remaining:.0f}s) ===")
    if remaining > 20:
        t8 = min(remaining * 0.5, 80.0)
        a_out, sc_out = adam_optimize(best_seq, t8, lr=0.001, top_k=5, sharpness=25.0,
                                      verbose=True, verbose_interval=500)
        update_best(a_out, "final_adam")
    
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 15:
        t8b = min(remaining * 0.5, 50.0)
        a_out, sc_out = coordinate_descent(best_seq, t8b, n_tries_per_iter=300)
        update_best(a_out, "final_cd")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        else:
            try:
                p = np.asarray(height_sequence_1, dtype=np.float64)
                p = np.clip(p, 0.0, 1000.0)
                if p.sum() > 0.01:
                    best_seq = p
                    best_score = evaluate_safe(p)
            except Exception:
                pass
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]
    
    return [float(x) for x in best_seq]
