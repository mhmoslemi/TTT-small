def fast_autocorr(a):
    """Compute full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a.astype(np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_score(a):
    """Compute 2n*max(b)/(sum(a))^2."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)

    """Safely call evaluate_sequence."""
    try:
        a = np.asarray(a, dtype=np.float64)
        a = np.clip(a, 0.0, 1000.0)
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

    """Compute g[i] = sum_m w[i+m] * a[m] for i=0..n-1 using FFT."""
    n = len(a)
    w_ext = np.zeros(3 * n - 2)
    w_ext[:len(w)] = w
    a_pad = np.zeros(3 * n - 2)
    a_pad[:n] = a
    fft_size = 1 << (2 * (3 * n - 2) - 1).bit_length()
    W = rfft(w_ext, fft_size)
    A = rfft(a_pad, fft_size)
    cc = irfft(np.conj(W) * A, fft_size)
    return cc[:n]

    """Compute gradient of LSE objective (1/T)*log(sum exp(T*b)) w.r.t. a."""
    b = fast_autocorr(a)
    Tb = T * b

    exp_Tb = np.exp(Tb_shift)
    w = exp_Tb / exp_Tb.sum()
    # grad[i] = (2/T) * sum_m w[i+m] * a[m]
    cc = cross_corr_w_a(w, a)
    return (2.0 / T) * cc

    """Compute LSE objective value."""
    b = fast_autocorr(a)
    Tb = T * b
    return (Tb.max() + np.log(np.exp(Tb - Tb.max()).sum())) / T

- `FFT(W)` is the DFT of W
- `conj(FFT(W)) * FFT(A)` is the DFT of the cross-correlation (W ⋆ A)
- `IFFT(conj(FFT(W)) * FFT(A))` gives (W ⋆ A)[i] = sum_k W[k] * A[(k-i) mod N]

With zero-padding (N >= 3n-2), the circular and linear correlations agree for i = 0, ..., n-1.

And as I derived: (W ⋆ A)[i] = sum_m W[i+m] * a[m] = sum_m w[i+m] * a[m] = g[i] ✓

"""
Minimizes 2n*max(conv(a,a))/(sum(a))^2 using:
1. LSE-smoothed L-BFGS optimization (multi-stage temperature annealing)
2. Adaptive mirror descent with softmax-weighted subgradient and T-annealing
3. Stochastic local search (simulated annealing with structured moves)
4. Multi-length exploration with interpolation transfer
5. Final coordinate descent peak shaving
All core operations use FFT for O(n log n) per iteration.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a.astype(np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_score(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)


def evaluate_safe(a):

        a = np.asarray(a, dtype=np.float64)
        a = np.clip(a, 0.0, 1000.0)
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:

def cross_corr_w_a(w, a):
    """Compute g[i] = sum_{m=0}^{n-1} w[i+m] * a[m] using FFT."""
    n = len(a)
    N = 3 * n - 2
    fft_size = 1 << (N - 1).bit_length()
    W = np.zeros(fft_size)
    W[:len(w)] = w
    A = np.zeros(fft_size)
    A[:n] = a
    C = irfft(rfft(W, fft_size).conj() * rfft(A, fft_size), fft_size)
    return C[:n]


def softmax_weights(b, T):
    """Compute softmax weights over b with temperature T."""
    Tb = T * b
    Tb_shift = Tb - Tb.max()
    exp_Tb = np.exp(Tb_shift)
    return exp_Tb / exp_Tb.sum()


def subgradient_smoothed(a, T=5.0):
    """Compute smoothed subgradient of max(b) using softmax weights."""
    b = fast_autocorr(a)
    w = softmax_weights(b, T)
    return 2.0 * cross_corr_w_a(w, a), b


def lse_obj_and_grad_a(a, T):
    """LSE objective and its gradient w.r.t. a (not z)."""
    b = fast_autocorr(a)
    Tb = T * b
    Tb_shift = Tb - Tb.max()
    exp_Tb = np.exp(Tb_shift)
    obj = (Tb.max() + np.log(exp_Tb.sum())) / T
    w = exp_Tb / exp_Tb.sum()
    grad_a = (2.0 / T) * cross_corr_w_a(w, a)
    return obj, grad_a


def lse_obj_grad_z(z, T):
    """LSE objective and gradient in z-space (a = softmax(z))."""
    n = len(z)
    z_shift = z - z.max()
    e = np.exp(z_shift)
    a = e / e.sum()
    obj, grad_a = lse_obj_and_grad_a(a, T)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def optimize_lse(a0, time_limit, T_start=50.0, T_end=1.0, n_stages=10, seed=42):
    """Multi-stage LSE optimization with L-BFGS in z-space."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    a_init = a_init / a_init.sum()
    z = np.log(a_init)
    t0 = time.monotonic()
    best_a = a_init.copy()
    best_score = fast_score(a_init)
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        iters = max(10, min(60, int(20 + 30 * si / max(1, n_stages))))
        
        def func(z_in):
            val, _ = lse_obj_grad_z(z_in, T)
            return val
        
        def jac(z_in):
            _, g = lse_obj_grad_z(z_in, T)
            return g
        
        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                          options={'maxiter': iters, 'ftol': 1e-18,
                                   'gtol': 1e-12, 'maxfun': iters * 500})
            z = res.x
            z_shift = z - z.max()
            a = np.exp(z_shift)
            a = a / a.sum()
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        except Exception:
            pass
    
    return best_a, best_score


def mirror_descent_adaptive(a0, time_limit, T_start=20.0, T_end=2.0,
                            eta0=0.08, decay=0.6, seed=42, eps=1e-12):
    """Mirror descent with annealed softmax subgradient."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / max(s, 1e-15)
    
    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall = 0
    n_iter = 0
    
    while time.monotonic() - t0 < time_limit:
        n_iter += 1
        # Anneal T from T_start to T_end
        frac = min(1.0, n_iter / 5000.0)
        T = T_start * (T_end / T_start) ** frac
        
        b = fast_autocorr(a)
        w = softmax_weights(b, T)
        g = 2.0 * cross_corr_w_a(w, a)
        
        # Adaptive learning rate
        g_scale = g.max() if g.max() > 1e-15 else 1.0
        eta = eta0 / (1.0 + n_iter / 300.0) ** decay / g_scale * 0.1
        
        # Multiplicative update with floor
        a_new = (a + eps) * np.exp(-eta * g)
        a_new = np.maximum(a_new, eps)
        s_new = a_new.sum()
        a = a_new / s_new
        
        # Check score
        interval = 3 if n_iter < 1000 else 10 if n_iter < 10000 else 20
        if n_iter % interval == 0:
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        
        # Stall handling with increasing noise
        if stall > 200:
            noise = min(0.05, 0.005 * (stall / 50.0))
            a = a + noise * rng.standard_normal(n)
            a = np.maximum(a, 0)
            ss = a.sum()
            if ss > 1e-10:
                a = a / ss
            stall = 0
        
        if n_iter % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD iter={n_iter} ({elapsed:.0f}s): best={best_score:.6f}, T={T:.2f}")
    
    return best_a, best_score


def stochastic_local_search(a0, time_limit, seed=123, T_init=0.01, T_final=0.0001):
    """Simulated annealing with structured moves on the simplex."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / max(s, 1e-15)
    
    best_a = a.copy()
    best_score = fast_score(a)
    current_score = best_score
    t0 = time.monotonic()
    n_iter = 0
    
    while time.monotonic() - t0 < time_limit:
        n_iter += 1
        # Annealing schedule
        frac = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_init * (T_final / T_init) ** frac
        
        # Choose move type
        move_type = rng.integers(3)
        a_trial = a.copy()
        
        if move_type == 0:
            # Mass transfer between two random positions
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac_transfer = rng.uniform(0.01, 0.1)
                amt = a[i] * frac_transfer
                a_trial[i] -= amt
                a_trial[j] += amt
        elif move_type == 1:
            # Gaussian bump add/subtract
            center = rng.uniform(0, n)
            width = rng.uniform(n * 0.02, n * 0.2)
            amp = rng.uniform(-0.02, 0.02)
            x = np.arange(n, dtype=np.float64)
            bump = amp * np.exp(-(x - center) ** 2 / (2 * width ** 2))
            a_trial += bump
        else:
            # Block shift: shift a random contiguous block
            block_len = rng.integers(max(3, n // 50), max(5, n // 10))
            start = rng.integers(0, n - block_len)
            shift = rng.integers(-block_len, block_len + 1)
            if shift != 0:
                new_start = start + shift
                new_end = start + block_len + shift
                if new_start >= 0 and new_end <= n:
                    a_trial = a.copy()
                    block = a_trial[start:start+block_len].copy()
                    a_trial[start:start+block_len] = 0.0
                    a_trial[new_start:new_end] += block
                    # Redistribute any overflow
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss > 1e-10:
                        a_trial = a_trial / ss
        
        a_trial = np.maximum(a_trial, 0)
        ss = a_trial.sum()
        if ss < 1e-10:
            continue
        a_trial = a_trial / ss
        
        trial_score = fast_score(a_trial)
        delta = trial_score - current_score
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_trial
            current_score = trial_score
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()
        
        if n_iter % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SLS iter={n_iter} ({elapsed:.0f}s): best={best_score:.6f}, T={T:.6f}")
    
    return best_a, best_score


def coordinate_peak_shave(a0, time_limit, seed=99):
    """Coordinate descent: reduce top peaks by redistributing mass."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / max(s, 1e-15)
    
    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:min(8, 2*n-1)]
        improved = False
        
        for k_idx in top_k:
            k_idx = int(k_idx)
            js = max(0, k_idx - n + 1)
            je = min(n - 1, k_idx)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_idx - idx]
            
            if contrib.sum() < 1e-15:
                continue
            
            src = int(idx[int(np.argmax(contrib))])
            
            # Find low-value target positions
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k[:4]:
                k2 = int(k2)
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2+1] = True
            
            candidates = np.where(~peak_mask)[0]
            if len(candidates) < 5:
                candidates = np.where(np.arange(n) != src)[0]
            
            # Try top few lowest-value candidates
            sorted_c = candidates[np.argsort(a[candidates])[:max(3, len(candidates)//10)]]
            
            for dst in sorted_c[:5]:
                dst = int(dst)
                for frac in [0.001, 0.005, 0.01, 0.03]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    sc = fast_score(a_trial)
                    if sc < best_score:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            if rng.integers(2) == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    amt = a[i] * rng.uniform(0.002, 0.01)
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= 0:
                        a_trial = a_trial / a_trial.sum()
                        sc = fast_score(a_trial)
                        if sc < best_score:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            stall = 0
            if stall > 300:
                break
    
    return best_a, best_score
