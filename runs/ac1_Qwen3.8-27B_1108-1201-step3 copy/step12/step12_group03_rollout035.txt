import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.sparse import csr_matrix
import cvxpy as cp

def fast_conv(a):
    """Full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def build_T(y):
    """Build sparse convolution matrix T where (T@b)[k] = conv(y,b)[k]."""
    n = len(y)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
            vals.append(y[k - i])
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

def score(a):
    """Compute the objective value."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fast_conv(a)
    return 2.0 * n * c.max() / (s * s)

def solve_fw_subproblem(y, solver=cp.ECOS):
    """Solve: min M s.t. T(y)@b <= M, sum(b)=1, b>=0."""
    n = len(y)
    b = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    T = build_T(y)
    constraints = [T @ b <= M, cp.sum(b) == 1]
    prob = cp.Problem(cp.Minimize(M), constraints)
    prob.solve(solver=solver)
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val /= s
    return b_val, float(M.value)

def nesterov_fw(a0, time_limit, solver, verbose=True):
    """Nesterov-accelerated Frank-Wolfe."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0)
    a /= a.sum()
    y = a.copy()
    
    best_a = a.copy()
    best_score_val = score(a)
    t0 = time.monotonic()
    
    for t in range(1, 500000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_star, M_val = solve_fw_subproblem(y, solver)
        if b_star is None:
            continue
        
        # Nesterov step size
        if t == 1:
            a_new = b_star.copy()
        else:
            alpha = 2.0 / (t + 2.0)
            y_update = (1 - alpha) * y + alpha * b_star
            y_update = np.maximum(y_update, 0)
            y_update /= y_update.sum()
            
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_update + momentum * (y_update - y)
            a_new = np.maximum(a_new, 0)
            a_new /= a_new.sum()
        
        y = y_update if t > 1 else a_new
        
        sc = score(a_new)
        if sc < best_score_val:
            best_score_val = sc
            best_a = a_new.copy()
        
        sc_y = score(y)
        if sc_y < best_score_val:
            best_score_val = sc_y
            best_a = y.copy()
        
        if verbose and t % 2000 == 0:
            print(f"    FW t={t}: best={best_score_val:.6f}, M={M_val:.8f}")
    
    return best_a, best_score_val

import numpy as np
import time
from scipy.fft import rfft, irfft, fft, ifft
import cvxpy as cp

def fast_conv(a, b=None):
    """Linear convolution. If b is None, compute a*a."""
    if b is None:
        b = a
    na, nb = len(a), len(b)
    size = na + nb - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]

def score(a):
    """Compute objective value."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fast_conv(a)
    return 2.0 * n * c.max() / (s * s)

def softmax(x, temp=1.0):
    """Numerically stable softmax."""
    x = x / temp
    x = x - x.max()
    e = np.exp(x)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s

def solve_fw_lp(y, solver=None):
    """Exact F-W subproblem solver."""
    if solver is None:
        solver = cp.CLARABEL
    n = len(y)
    b = cp.Variable(n, nonneg=True)
    M = cp.Variable()
    # Build constraints efficiently
    # (conv(y,b))[k] <= M for k = 0,...,2n-2
    # (conv(y,b))[k] = sum_{i: 0<=k-i<=n-1} y[k-i]*b[i]
    constraints = [cp.sum(b) == 1]
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1)
        y_vals = y[k - idx]
        constraints.append(y_vals @ b[idx] <= M)
    prob = cp.Problem(cp.Minimize(M), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            try:
                prob.solve(solver=alt, verbose=False)
                break
            except:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s
