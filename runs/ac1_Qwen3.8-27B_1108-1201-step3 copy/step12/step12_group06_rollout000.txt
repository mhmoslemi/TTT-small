def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):

def softmax_from_z(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def project_simplex(v):
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


# Cache for constraint matrices
_conv_matrix_cache = {}

def get_conv_matrix(a_fixed):
    key = hash(tuple(np.round(a_fixed, 10)))
    if key in _conv_matrix_cache:
        return _conv_matrix_cache[key]
    T = build_conv_matrix(a_fixed)
    if len(_conv_matrix_cache) > 20:
        _conv_matrix_cache.clear()
    _conv_matrix_cache[key] = T
    return T


def get_lp_solver():
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp(a_fixed, solver):
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        for alt in [cp.ECOS, cp.PDLP, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val /= s
    return b_val, float(np.asarray(M_var.value).flat[0])

def lse_obj_grad(z, T, n):
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = np.clip(Tb.max(), 0, 700)
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    lse = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * lse

    # Gradient
    g = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        g[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(g, a)
    grad_z = a * (g - S)
    return obj, grad_z

            improvement = (prev_score - sc) / (prev_score + 1e-15)

def peak_subgrad_nesterov(a0, time_limit, max_iters=500000):

    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()

    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    # Nesterov variables
    y = a.copy()

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(y)
        max_b = b.max()
        cur_score = 2.0 * n * max_b

        # Find top peaks
        threshold = max_b * 0.95
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(b))])
        if len(active) > 30:
            top_idx = np.argsort(b[active])[-30:]
            active = active[top_idx]

        # Subgradient
        g = np.zeros(n)
        for k in active:
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += 2.0 * y[k - idx]
        g /= len(active)

        # Project to tangent space
        S = np.dot(g, a)
        grad_z = a * (g - S)
        grad_norm = np.linalg.norm(grad_z)
        if grad_norm < 1e-15:
            break

        # Adaptive step size
        eta = 1.0 / (1.0 + it * 0.01)
        eta = min(eta, 0.5 / (grad_norm + 1e-10))
        eta = min(eta, 1.0)

        z_new = z - eta * grad_z
        a_new = softmax_from_z(z_new)
        sc_new = score_normalized(a_new)

        # Line search if needed
        if sc_new > cur_score * 1.02:
            for _ in range(10):
                eta *= 0.5
                z_new = z - eta * grad_z
                a_new = softmax_from_z(z_new)
                sc_new = score_normalized(a_new)
                if sc_new <= cur_score:
                    break

        # Nesterov update
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * float(it + 1) ** 2)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a = a_new
        y = a + momentum * (a - a_old)
        y = np.maximum(y, 0)
        sy = y.sum()
        if sy > 1e-15:
            y /= sy

        if sc_new < best_score:
            best_score = sc_new
            best_a = a.copy()

            print(f"    PSG iter={it+1} ({elapsed:.0f}s): cur={cur_score:.8f} best={best_score:.8f}")

    return best_a, best_score


def interpolate(a, new_n):
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    s = a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    a_norm = a / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    result = np.interp(new_x, old_x, a_norm)
    result = np.maximum(result, 0)
    ss = result.sum()
    if ss > 1e-15:
        result /= ss
    return result
