"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smoothed logsumexp optimization with Adam (FFT-accelerated gradients)
2. Pairwise exchange coordinate descent with bisection line search
3. Multi-length exploration with interpolation
4. Frequency-domain perturbations for escape from local minima
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if not a_list:
            return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def project_simplex(v):
    """Project vector v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def smoothed_autocorr_grad(a, beta):
    """
    Compute gradient of (1/beta)*log(sum_k exp(beta*(a*a)_k)).
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    b = fast_autocorr(a)
    # Softmax weights
    b_shifted = b - b.max()
    exp_b = np.exp(beta * b_shifted)
    w_sum = exp_b.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(b)
        argmax = np.argmax(b)
        w[argmax] = 1.0
    else:
        w = exp_b / w_sum
    
    # Cross-correlation: grad[i] = 2 * sum_{m=0}^{n-1} w[i+m] * a[m]
    # Use FFT: ifft(FFT(w) * conj(FFT(a)))[:n]
    N = 1 << (3 * n - 2 - 1).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    R = irfft(W * np.conj(A), N)
    grad = 2.0 * R[:n]
    return grad


def adam_optimize(a0, time_limit, beta_schedule=None, lr=0.01, verbose=True):
    """
    Adam optimization of smoothed max autocorrelation over the simplex.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    if beta_schedule is None:
        beta_schedule = [5, 20, 50, 100, 200, 500, 1000]
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    
    for beta in beta_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        
        # Adjust learning rate for this beta phase
        phase_lr = lr * min(1.0, 50.0 / beta)  # Smaller lr for larger beta
        
        phase_steps = 0
        phase_start_time = time.monotonic()
        
        while time.monotonic() - t0 < time_limit:
            step += 1
            phase_steps += 1
            
            # Compute gradient
            grad = smoothed_autocorr_grad(a, beta)
            
            # Evaluate current
            b_curr = fast_autocorr(a)
            score_curr = 2.0 * n * b_curr.max()
            if score_curr < best_score:
                best_score = score_curr
                best_a = a.copy()
            
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad ** 2
            m_hat = m / (1 - beta1 ** step)
            v_hat = v / (1 - beta2 ** step)
            a_new = a - phase_lr * m_hat / (np.sqrt(v_hat) + eps)
            
            # Project to simplex
            a_new = project_simplex(a_new)
            
            # Check for improvement
            a = a_new
            
            # Periodic evaluation
            if phase_steps % 500 == 0:
                b_check = fast_autocorr(a)
                score_check = 2.0 * n * b_check.max()
                if score_check < best_score:
                    best_score = score_check
                    best_a = a.copy()
                if verbose and phase_steps % 5000 == 0:
                    elapsed = time.monotonic() - t0
                    print(f"    Adam beta={beta} step={phase_steps} ({elapsed:.0f}s): "
                          f"best={best_score:.6f}, cur={score_check:.6f}")
            
            # Break if no progress for a while (check less frequently)
            if phase_steps >= 1000 and phase_steps % 2000 == 0:
                # Check if we're stalling by comparing with best
                if score_check > best_score * 1.0001:
                    if phase_steps > 20000:
                        break
    
    return best_a, best_score


def pairwise_exchange(a0, time_limit, n_rounds=200, verbose=True):
    """
    Pairwise exchange: for each pair (i,j), find the best amount to transfer.
    This is a coordinate descent on the pairwise transfers.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(42)
    
    for rnd in range(n_rounds):
        if time.monotonic() - t0 > time_limit:
            break
        
        improved = False
        # Try random pairs (more efficient than all pairs for large n)
        n_pairs = min(n * 2, 500)
        for _ in range(n_pairs):
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            
            # For this pair, find best transfer amount
            # Try a few fractions
            best_frac = 0
            best_pair_score = best_score
            
            for frac_sign in [1, -1]:
                max_frac = a[i] if frac_sign == 1 else a[j]
                for frac in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.4]:
                    if frac > max_frac:
                        continue
                    if frac_sign == 1:
                        # Move from i to j
                        a_trial = a.copy()
                        a_trial[i] -= frac
                        a_trial[j] += frac
                    else:
                        # Move from j to i
                        a_trial = a.copy()
                        a_trial[j] -= frac
                        a_trial[i] += frac
                    
                    a_trial = np.maximum(a_trial, 0)
                    st = a_trial.sum()
                    if st < 1e-15:
                        continue
                    a_trial = a_trial / st
                    
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_pair_score:
                        best_pair_score = score
                        best_frac = frac * frac_sign
            
            if best_frac != 0 and best_pair_score < best_score:
                if best_frac > 0:
                    a[i] = max(0, a[i] - best_frac)
                    a[j] += best_frac
                else:
                    a[j] = max(0, a[j] - (-best_frac))
                    a[i] += (-best_frac)
                a = np.maximum(a, 0)
                st = a.sum()
                if st > 1e-15:
                    a = a / st
                best_score = best_pair_score
                best_a = a.copy()
                improved = True
        
        if verbose and rnd % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PE round={rnd} ({elapsed:.0f}s): best={best_score:.6f}")
        
        if not improved and rnd > 10:
            # Stalled, try random perturbation
            perturb_mag = 0.01
            a_pert = a + rng.normal(0, perturb_mag, n)
            a_pert = project_simplex(a_pert)
            score_pert = 2.0 * n * fast_autocorr(a_pert).max()
            if score_pert < best_score:
                a = a_pert
                best_a = a.copy()
                best_score = score_pert
    
    return best_a, best_score

    # Actually let me redo this more carefully
    # s[k] = a[k-j] - a[k-i] where a[m] = 0 if m not in [0,n-1]
    
    # For a[k-j]: nonzero when 0 <= k-j < n, i.e., j <= k < j+n
    # a[k-j] = a[0], a[1], ... for k = j, j+1, ...

    # a[k-i] = a[0], a[1], ... for k = i, i+1, ...
    
    # So s[k] = a[k-j] for k in [j, j+n-1] minus a[k-i] for k in [i, i+n-1]
    
    # This is: s = shift(a, j) - shift(a, i) where shift(a, p) is a shifted by p
    # (with zero padding)
    
    # Using numpy:
    s = np.zeros(2 * n - 1)
    if j + n <= 2 * n - 1:
        s[j:j+n] += a
    else:
        s[j:] += a[:2*n-1-j]
    if i + n <= 2 * n - 1:
        s[i:i+n] -= a
    else:
        s[i:] -= a[:2*n-1-i]
    
    # Now new_b[t][k] ≈ b[k] + 2t * s[k]
    # We want to minimize max_k (b[k] + 2t * s[k]) over t in [0, t_max]

    lo = 0.0
    hi = t_max

        # Compute new max at t = mid
        new_b = b + 2.0 * mid * s
        new_max = new_b.max()

        k_star = np.argmax(new_b)
        subgrad = s[k_star]
        
        if subgrad > 0:
            # Increasing t increases the max, so decrease t
            hi = mid
        elif subgrad < 0:
            # Increasing t decreases the max, so increase t
            lo = mid
        else:

    best_t = (lo + hi) / 2.0
    new_b = b + 2.0 * best_t * s
    predicted_max = new_b.max()
    
    return best_t, predicted_max

"""

2. Efficient pairwise exchange with bisection (O(n) per pair, 30 bisection steps)

import numpy as np
import time

def fast_autocorr(a):
    """Full linear convolution a*a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if not a_list:
            return float('inf')
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def project_simplex(v):
    """Project vector v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def smoothed_autocorr_grad(a, beta):
    """
    Compute gradient of (1/beta)*log(sum_k exp(beta*(a*a)_k)).
    Uses FFT cross-correlation for O(n log n) computation.
    """
    n = len(a)
    b = fast_autocorr(a)
    # Softmax weights over autocorrelation lags
    b_shifted = b - b.max()
    exp_b = np.exp(beta * b_shifted)
    w_sum = exp_b.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(b)
        argmax = np.argmax(b)
        w[int(argmax)] = 1.0
    else:
        w = exp_b / w_sum

    # grad[i] = 2 * sum_{m=0}^{n-1} w[m+i] * a[m]
    # This is 2 * cross_correlation(w, a) at lag i
    N = 1 << (3 * n - 2 - 1).bit_length()
    W = rfft(w, N)
    A = rfft(a, N)
    R = irfft(W * np.conj(A), N)
    grad = 2.0 * R[:n]
    return grad


def compute_slope_vector(a, i, j):
    """
    Compute s[k] = a[k-j] - a[k-i] (zero-padded outside [0,n-1]).
    This is the slope of autocorrelation w.r.t. transfer t from i to j.
    """
    n = len(a)
    s = np.zeros(2 * n - 1)
    # a[k-j] nonzero for k in [j, j+n-1] (clipped to [0, 2n-2])
    lo1 = j
    hi1 = min(j + n, 2 * n - 1)
    cnt1 = hi1 - lo1
    if cnt1 > 0:
        s[lo1:hi1] += a[:cnt1]
    # a[k-i] nonzero for k in [i, i+n-1] (clipped to [0, 2n-2])
    lo2 = i
    hi2 = min(i + n, 2 * n - 1)
    cnt2 = hi2 - lo2
    if cnt2 > 0:
        s[lo2:hi2] -= a[:cnt2]
    return s


def find_best_transfer_t(b, a, i, j, t_max):
    """
    Find best t in [0, t_max] to transfer from i to j,
    using bisection on the subgradient of the linear approximation.
    Returns (best_t, predicted_new_max_b).
    """
    if t_max <= 0:

    n = len(a)
    s = compute_slope_vector(a, i, j)

    # Bisection: minimize max_k (b[k] + 2*t*s[k])
    lo = 0.0
    hi = t_max

    for _ in range(30):
        mid = (lo + hi) / 2.0
        # Compute new max
        new_b = b + 2.0 * mid * s
        new_max_val = new_b.max()
        k_star = int(np.argmax(new_b))
        subgrad = s[k_star]

        if subgrad > 1e-15:
            hi = mid
        elif subgrad < -1e-15:

    return best_t, predicted_max


def adam_smoothed_optimize(a0, time_limit, beta_schedule=None, lr_base=0.05, verbose=True, label=""):
    """
    Adam optimization of smoothed max autocorrelation over the simplex.
    Uses FFT-accelerated gradients.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    if beta_schedule is None:
        beta_schedule = [5, 15, 40, 100, 250, 500, 1000, 2000]

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-8

    for beta in beta_schedule:
        phase_t0 = time.monotonic()
        phase_steps = 0
        # Learning rate scales inversely with beta (gradients scale with beta)
        phase_lr = lr_base * min(1.0, 30.0 / beta)
        # Max steps per phase
        max_phase_steps = max(2000, int(60.0 / (n * 1e-4)))  # roughly time-limited

        while phase_steps < max_phase_steps:
            if time.monotonic() - t0 > time_limit:
                break

            step += 1
            phase_steps += 1

            # Compute gradient
            grad = smoothed_autocorr_grad(a, beta)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** step)
            v_hat = v / (1.0 - beta2 ** step)
            a_new = a - phase_lr * m_hat / (np.sqrt(v_hat) + eps)

            # Project to simplex
            a_new = project_simplex(a_new)
            a = a_new

            # Periodic evaluation
            if phase_steps % 200 == 0:
                b_curr = fast_autocorr(a)
                score_curr = 2.0 * n * b_curr.max()
                if score_curr < best_score:
                    best_score = score_curr
                    best_a = a.copy()

            # Check for early termination of this phase
            if phase_steps >= 1000 and phase_steps % 1000 == 0:
                elapsed_phase = time.monotonic() - phase_t0
                # If we've been running for over 30s in this phase and not improving much
                if elapsed_phase > 30.0:
                    break

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    Adam {label} beta={beta}: {phase_steps} steps, best={best_score:.7f} (t={elapsed:.0f}s)")

    return best_a, best_score

                    best_target_score = score_j
                    best_target = dst

            if best_target >= 0:
                # Try transferring a small amount
                amt = a[src] * 0.05
                if src != src_other:
                    amt = min(amt, a[src_other] * 0.05)
                if amt < 1e-15:
                    continue

                a_trial = a.copy()
                a_trial[src] -= amt
                if src != src_other:
                    a_trial[src_other] -= amt * 0.3
                a_trial[best_target] += amt
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st

                b_trial = fast_autocorr(a_trial)
                score_trial = 2.0 * n * b_trial.max()
                if score_trial < best_score - 1e-15:
                    a = a_trial
                    b = b_trial
                    best_score = score_trial
                    best_a = a.copy()
                    improved = True
                    break

        if not improved:
            # Slight random perturbation
            n_pert = int(rng.integers(1, 5))
            sources = rng.choice(n, size=n_pert, replace=False)
            targets = rng.choice(n, size=n_pert, replace=False)
            a_trial = a.copy()
            for src_p, dst_p in zip(sources, targets):
                frac = rng.uniform(0.001, 0.01)
                amt = a_trial[int(src_p)] * frac
                a_trial[int(src_p)] -= amt
                a_trial[int(dst_p)] += amt
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 1e-15:
                a_trial = a_trial / st
            b_trial = fast_autocorr(a_trial)
            score_trial = 2.0 * n * b_trial.max()
            if score_trial < best_score:
                a = a_trial
                b = b_trial
                best_a = a.copy()
                best_score = score_trial

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    # Handle endpoints: pad with 0 at both ends for better interpolation
    f_padded = np.zeros(old_n + 2)
    f_padded[1:-1] = f_norm
    old_x = np.linspace(0, 1, old_n + 2)
    new_x = np.linspace(0, 1, new_n + 2)
    f_interp = np.interp(new_x, old_x, f_padded)
    f_result = f_interp[1:-1]
    f_result = np.maximum(f_result, 0)
    ss = f_result.sum()
    if ss > 1e-15:
        f_result = f_result / ss
    else:
        f_result = np.ones(new_n) / new_n
    return f_result


def freq_perturbation(a0, rng, n_modes=5, scale=0.01):
    """
    Apply a small perturbation in the frequency domain to escape local minima.
    """
    n = len(a0)
    a = a0.copy()
    # FFT of a (padded)
    fft_size = 1 << (n - 1).bit_length()
    A = rfft(a, fft_size)
    # Add small random phases to a few frequency bins
    for _ in range(n_modes):
        k = int(rng.integers(1, fft_size // 2))
        A[k] += scale * A[k] * (rng.normal() + 1j * rng.normal())
        A[fft_size - k] = np.conj(A[k])
    # IFFT back
    a_new = irfft(A, fft_size)[:n]
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a0.copy()
    a_new = a_new / s
    return a_new


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function combining multiple optimization strategies:
    1. Smoothed Adam (logsumexp objective, FFT gradients)
    2. Pairwise exchange with bisection
    3. Peak shaving coordinate descent
    4. Multi-length exploration
    5. Frequency domain perturbations
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1680.0)

    # Get parent sequence
    best_seq = None
    best_score = float('inf')
    parent = None

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            if np.isfinite(p_score):
                print(f"Parent: n={len(parent)}, score={p_score:.6f}")
                best_seq = parent.copy()
                best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Set up initial sequence if no parent
    if parent is None:
        n_base = 1600
        # Try a few different initializations
        inits = []
        # Cosine bump
        x = np.linspace(0, np.pi, n_base)
        init1 = np.cos(x / 2) ** 4
        inits.append(init1 / init1.sum())
        # Exponential decay
        init2 = np.exp(-np.linspace(0, 3, n_base))
        inits.append(init2 / init2.sum())
        # Bell curve
        t_arr = np.linspace(-1, 1, n_base)
        init3 = np.exp(-4 * t_arr ** 2)
        inits.append(init3 / init3.sum())

        for idx_init, init_try in enumerate(inits):
            sc = evaluate_safe(init_try)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = init_try.copy()
                print(f"Default init {idx_init}: n={n_base}, score={sc:.6f}")

    n_base = len(best_seq)
    print(f"Starting: n={n_base}, best_score={best_score:.6f}")
    print(f"Time budget: {time_budget:.0f}s")

    # === Phase 1: Smoothed Adam optimization at base length ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: Smoothed Adam at n={n_base} (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.30, 500.0)
    if t1 > 60:
        a_out, sc_out = adam_smoothed_optimize(
            best_seq, t1,
            beta_schedule=[10, 30, 80, 200, 500, 1000],
            lr_base=0.03, verbose=True, label="base"
        )
        try_update(a_out, f"adam_n{n_base}")

    # === Phase 2: Pairwise exchange optimization ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: Pairwise Exchange at n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.25, 400.0)
    if t2 > 30:
        a_out, sc_out = pairwise_exchange_optimize(
            best_seq, t2, n_pairs_per_round=200, max_rounds=3000,
            verbose=True, label="pe"
        )
        try_update(a_out, f"pe_n{len(best_seq)}")

    # === Phase 3: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: Peak Shaving at n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 20:
        a_out, sc_out = peak_shaving_direct(
            best_seq, t3, verbose=True, label="peak"
        )
        try_update(a_out, f"peak_n{len(best_seq)}")

    # === Phase 4: Another Adam round (exploit improvements from PE/peak shaving) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: Adam Round 2 at n={len(best_seq)} (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.20, 300.0)
    if t4 > 60:
        a_out, sc_out = adam_smoothed_optimize(
            best_seq, t4,
            beta_schedule=[200, 500, 1000, 2000],
            lr_base=0.02, verbose=True, label="r2"
        )
        try_update(a_out, f"adam_r2_n{len(best_seq)}")

    # === Phase 5: Multi-length exploration ===
    for n_try in [n_base - 200, n_base - 100, n_base + 100, n_base + 200, n_base + 400]:
        if n_try < 400:
            continue
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining < 100:
            break
        print(f"\n=== Phase 5: Multi-length at n={n_try} (remaining={remaining:.0f}s) ===")
        t5 = min(remaining * 0.20, 180.0)
        if t5 < 40:
            continue
        init_try = interpolate_sequence(best_seq, n_try)
        # Quick Adam
        a_out, sc_out = adam_smoothed_optimize(
            init_try, t5 * 0.6,
            beta_schedule=[50, 200, 500],
            lr_base=0.03, verbose=False, label=f"n{n_try}"
        )
        # Quick PE
        t5_pe = min(remaining * 0.08, 60.0)
        if t5_pe > 15:
            a_out2, sc_out2 = pairwise_exchange_optimize(
                a_out, t5_pe, n_pairs_per_round=100, max_rounds=1000,
                verbose=False, label=f"pe_n{n_try}"
            )
            if sc_out2 < sc_out:
                a_out = a_out2
                sc_out = sc_out2
        try_update(a_out, f"multi_n{n_try}")

    # === Phase 6: Frequency perturbation + re-optimize ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 6: Freq Perturbation + Re-optimize (remaining={remaining:.0f}s) ===")
    if remaining > 120:
        t6 = min(remaining * 0.25, 200.0)
        # Try a few frequency perturbations
        for fp_idx in range(3):
            if time.monotonic() - t_global_start > time_budget - 60:
                break
            a_pert = freq_perturbation(best_seq, rng, n_modes=3 + fp_idx * 2, scale=0.005)
            # Quick optimize the perturbed version
            t6_sub = min(t6 / 3.0, 60.0)
            if t6_sub > 20:
                a_out, sc_out = adam_smoothed_optimize(
                    a_pert, t6_sub,
                    beta_schedule=[100, 300, 500],
                    lr_base=0.02, verbose=False, label=f"fp{fp_idx}"
                )
                try_update(a_out, f"freq_pert{fp_idx}")

    # === Phase 7: Final combined polish ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 7: Final Polish (remaining={remaining:.0f}s) ===")
    if remaining > 60:
        # Final PE on the best
        t7a = min(remaining * 0.4, 80.0)
        if t7a > 15:
            a_out, sc_out = pairwise_exchange_optimize(
                best_seq, t7a, n_pairs_per_round=300, max_rounds=2000,
                verbose=False, label="final_pe"
            )
            try_update(a_out, "final_pe")

        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining > 30:
            t7b = min(remaining * 0.5, 50.0)
            if t7b > 15:
                a_out, sc_out = peak_shaving_direct(
                    best_seq, t7b, verbose=False, label="final_peak"
                )
                try_update(a_out, "final_peak")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = min(best_score, fs)
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        # Fallback: uniform sequence
        return [1.0 / 1000] * 1000

    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
