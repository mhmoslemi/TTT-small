def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def cross_corr(g, a):
    """Compute cross-correlation sum_k g[k]*a[k-i] for all i, using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    G = fft(g, fft_size)
    A = fft(a, fft_size)
    # Cross-correlation: sum_k g[k] * a[k-i] = ifft(fft(g) * conj(fft(a))) shifted
    cc = np.real(ifft(G * np.conj(A), fft_size))
    # The result at index i gives sum_k g[k] * a[k-i] (with wraparound)
    # We need the linear cross-correlation, so take the appropriate part
    # ifft(fft(g) * conj(fft(a))) gives circular cross-correlation
    # For linear: result[i] = sum_k g[k] * a[k-i] for i=0,...,n-1
    # This is the first n elements shifted appropriately
    # Actually: circular cross-correlation r[i] = sum_k g[(k+i) mod N] * a[k]
    # We want: c[i] = sum_k g[k] * a[k-i] = sum_k g[k+i] * a[k] (reindexing)
    # So c[i] = r_circ[i] where r_circ is the circular cross-correlation
    # But we need to handle the linear case (no wraparound)
    # For our purposes with zero-padded signals, the first n elements of
    # ifft(fft(g) * conj(fft(a))) give the linear cross-correlation
    return cc[:n]


def powmean_gradient(a, p):
    """Compute gradient of sum_k (a*a)[k]^p w.r.t. a (unconstrained)."""
    b = fast_convolve(a)
    n = len(a)
    if p == 1:
        g = np.ones_like(b)
    else:
        g = np.maximum(b, 0) ** (p - 1)
    # Gradient: 2*p * cross_corr(g, a)
    grad = 2.0 * p * cross_corr(g, a)
    return grad


def variance_gradient(a, b):
    """Compute gradient of variance of b w.r.t. a."""
    b_mean = b.sum() / len(b)
    b_centered = b - b_mean
    grad = 4.0 * cross_corr(b_centered, a)
    return grad


def power_mean_anneal(a0, time_limit, p_start=4.0, p_end=500.0, n_stages=30,
                      var_weight=0.01, max_iter_per_stage=200):
    """
    Power-mean gradient descent in log-space with annealing.
    Minimizes sum_k (a*a)[k]^p + lambda * Var(a*a) for decreasing... no, INCREASING p.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-30)
    s = a.sum()
    a = a / s
    z = z_from_a(a).copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    ps = np.geomspace(p_start, p_end, n_stages)
    eta = 0.5
    global_var_weight = var_weight

    for si, p_val in enumerate(ps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        adj_iters = int(max_iter_per_stage * min(1.0, remaining / 10.0))
        if adj_iters < 5:
            adj_iters = 5

        for it in range(adj_iters):
            if time.monotonic() - t0 > time_limit:
                break

            # Compute gradient
            b = fast_convolve(a)
            g = np.maximum(b, 0) ** (p_val - 1)
            grad_a = 2.0 * p_val * cross_corr(g, a)

            # Add variance gradient
            if global_var_weight > 0:
                grad_a += global_var_weight * variance_gradient(a, b)

            # Convert to log-space gradient
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)

            # Adaptive step size
            z_new = z - eta * grad_z
            a_new = softmax_from_z(z_new)

            sc = score_normalized(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                z = z_new
                a = a_new
                eta = min(eta * 1.01, 2.0)
            else:
                eta *= 0.7
                if eta < 1e-10:
                    break

        if (si + 1) % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PowMean stage={si+1}/{n_stages} p={p_val:.1f}: score={best_score:.8f} ({elapsed:.0f}s)")

            if time.monotonic() - t0 > time_limit:
                break

            b = fast_convolve(a)

            # Power mean gradient
            g = np.maximum(b, 0) ** (p_val - 1)
            grad_pm = 2.0 * p_val * cross_corr(g, a)

            # Variance gradient
            grad_var = variance_gradient(a, b)

            # Combined gradient
            grad_a = grad_pm + lam * grad_var

            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)

            z_new = z - eta * grad_z
            a_new = softmax_from_z(z_new)

            sc = score_normalized(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                z = z_new
                a = a_new
                eta = min(eta * 1.005, 1.0)
            else:
                eta *= 0.8
                if eta < 1e-10:
                    break

    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    """LogSumExp smoothed objective and gradient."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = np.clip(Tb.max(), 0, 700)
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    # Gradient: 2*n/T * sum_k w[k] * a[k-i]
    grad_a = 2.0 * cross_corr(w, a)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbgfs_logsumexp(a0, time_limit, T_start=50.0, T_end=0.02, n_stages=25):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)

    temps = np.geomspace(T_start, T_end, n_stages)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, stage_T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, stage_T, n)
            return grad

        iters = 50 if si < n_stages // 2 else 25
        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-25,
                         'gtol': 1e-20, 'maxfun': iters * 200,
                         'disp': False}
            )
            z_new = res.x
            a_new = softmax_from_z(z_new)
            sc = score_normalized(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
            z = z_new
        except Exception:
            pass

    return best_a, best_score


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


_lp_solver = None


def get_lp_solver():
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp(a0, time_limit, solver=None, max_iters=500000):
    """Nesterov accelerated LP-based best response."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    last_print = t0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            y = a + momentum * (a - a_prev)
            y = np.maximum(y, 0)
            sy = y.sum()
            y = y / sy if sy > 1e-15 else a.copy()
        else:
            y = a

        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        alpha = 2.0 / (t_new + 1.0)

        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        a_new = a_new / sa if sa > 1e-15 else b_new.copy()

        sc_new = score_normalized(a_new)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()

        sc_cur = score_normalized(a)
        if sc_new > sc_cur * 1.001 and t_nest > 5.0:
            t_nest = 1.0
            a_prev = a_new.copy()
        else:
            a_prev = a.copy()
            t_nest = t_new

        a = a_new

        if (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 10:
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.8f}")
                last_print = elapsed

    return best_a, best_score


def peak_peeling(a0, time_limit, max_steps=100000):
    """Direct peak-peeling: reduce the max lag by adjusting contributing elements."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-10)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = 0.1

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step % 5000 == 0 and step > 0:
            if score_normalized(a) > best_score * 1.01:
                a = best_a.copy()
                eta *= 0.5

        b = fast_convolve(a)
        max_b = b.max()
        max_idx = np.argmax(b)

        # Find which pairs contribute to the max lag
        # (a*a)[max_idx] = sum_m a[m] * a[max_idx - m]
        contrib = np.zeros(n)
        for m in range(n):
            other = max_idx - m
            if 0 <= other < n:
                contrib[m] = a[other]
                contrib[other] += a[m]
                if m == other:
                    contrib[m] -= a[m]  # avoid double count

        # Reduce elements that contribute most to the peak
        # Use multiplicative update: a_i *= exp(-eta * contrib_i / max_contrib)
        max_contrib = contrib.max()
        if max_contrib < 1e-15:
            break
        exponent = -eta * contrib / max_contrib
        exponent -= exponent.max()
        new_a = a * np.exp(exponent)
        sa = new_a.sum()
        if sa < 1e-10:
            eta *= 0.5
            continue
        new_a = new_a / sa

        sc = score_normalized(new_a)
        if sc < best_score:
            best_score = sc
            best_a = new_a.copy()
            a = new_a
            eta = min(eta * 1.003, 0.5)
        else:
            eta *= 0.92
            if eta < 1e-8:
                break

    return best_a, best_score


def init_oscillatory(n, freq, amplitude, rng):
    """Oscillatory initialization."""
    i = np.arange(n)
    a = 1.0 + amplitude * np.cos(2.0 * np.pi * freq * i / n + rng.uniform(0, 2 * np.pi))
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def init_edge(n, beta, rng):
    """Edge-concentrated initialization."""
    i = np.arange(n)
    x = i / (n - 1)
    a = np.exp(beta * (x - 0.5) ** 2)
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def init_two_peak(n, beta, rng):
    """Two-peaked initialization."""
    i = np.arange(n)
    x = i / (n - 1)
    p1 = rng.uniform(0.1, 0.4)
    p2 = 1.0 - p1
    a = np.exp(-beta * (x - p1) ** 2) + 0.8 * np.exp(-beta * (x - p2) ** 2)
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def init_random_smooth(n, rng, smoothness=0.3):
    """Random smooth initialization."""
    a = rng.random(n)
    # Smooth with a kernel
    kernel_size = max(3, n // 50)
    kernel = np.ones(kernel_size) / kernel_size
    a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def init_asymmetric(n, rng):
    """Asymmetric structured initialization."""
    choice = rng.integers(0, 6)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = 1.0 + 0.7 * (2.0 * x - 1.0)
    elif choice == 1:
        a = np.exp(-3.0 * x) + 0.3
    elif choice == 2:
        a = np.exp(-5.0 * (x - 0.3) ** 2) + 0.2
    elif choice == 3:
        a = 0.5 + 0.4 * np.cos(2 * np.pi * (x + 0.15))
    elif choice == 4:
        a = (x + 0.1) ** 0.7
    else:
        a = 0.8 + 0.3 * np.sin(4 * np.pi * x + 1.0)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def subgradient_fast(a0, time_limit, max_steps=200000):
    """Fast subgradient method targeting the top peaks."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-10)
    s = a.sum()
    a = a / s
    z = z_from_a(a).copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = 0.2
    t_nest = 1.0
    z_prev = z.copy()

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step % 10000 == 0 and step > 0:
            if score_normalized(a) > best_score * 1.02:
                a = best_a.copy()
                z = z_from_a(a)
                t_nest = 1.0

        b = fast_convolve(a)
        max_b = b.max()
        # Use top 5% of lags
        threshold = max_b * 0.9
        topk = np.where(b >= threshold)[0]
        if len(topk) == 0:
            topk = np.array([int(np.argmax(b))])

        # Gradient: average of contributions from top lags
        grad = np.zeros(n)
        for k in topk:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += a[k - idx]
        grad /= len(topk)

        S = np.dot(grad, a)
        grad_z = a * (grad - S)

        # Nesterov extrapolation
        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            z_extrap = z + momentum * (z - z_prev)
        else:
            z_extrap = z

        z_new = z_extrap - eta * grad_z
        a_new = softmax_from_z(z_new)

        sc = score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            z_prev = z
            z = z_new
            a = a_new
            t_nest = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        else:
            if sc > best_score * 1.005:
                t_nest = 1.0
                eta *= 0.95
            else:
                z_prev = z
                z = z_new
                a = a_new
                t_nest = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))

    return best_a, best_score

        p = np.asarray(height_sequence_1, dtype=np.float64)

        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()

    best_n = n_base

            return

        s = arr.sum()
        if s < 0.01:
            return
        arr = arr / s
        sc = evaluate_safe(arr)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = arr.copy()

            print(f"  NEW BEST [{label}]: {sc:.8f} (n={len(arr)})")

    # === Phase 0: Multi-start initialization (120s) ===

    }

    # Quick evaluate all initializations
    candidates = []
    for name, init_a in initializations.items():
        init_a = np.maximum(init_a, 0)
        s = init_a.sum()
        init_a = init_a / s if s > 0 else np.ones(n_base) / n_base
        sc = score_normalized(init_a)
