"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated LP (Nash equilibrium) with warm-started solver
2. Multi-stage adaptive LogSumExp L-BFGS
3. Peak-directed coordinate descent with line search
4. Fourier-domain spectral flattening (Gerchberg-Saxton variant)
5. Multi-length exploration and final combined polish
Key improvements: better time allocation, local refinement, Fourier methods,
and multi-start diversification.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft, fft, ifft
from scipy.optimize import minimize


def fast_convolve(a):
    """Fast linear convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    """Score assuming sum(a)=1: returns 2*n*max(b)."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    """Safely call evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax_from_z(z):
    """Stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Inverse of softmax."""
    return np.log(np.maximum(a, 1e-300))


def build_conv_matrix(a_fixed):
    """Build the convolution matrix T where (T@b)[k] = (a_fixed*b)[k]."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


_lp_solver = None


def get_lp_solver():
    """Find the best available LP solver."""
    global _lp_solver
    if _lp_solver is not None:
        return _lp_solver
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CBC]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _lp_solver = solver_name
                return solver_name
        except Exception:
            continue
    _lp_solver = cp.ECOS
    return _lp_solver


def solve_lp_best_response(a_fixed, solver=None):
    """Solve the LP: min M s.t. (a_fixed*b)[k] <= M for all k, sum(b)=1, b>=0."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS, cp.CBC]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp(a0, time_limit, solver=None, max_iters=500000):
    """Nesterov-accelerated LP best-response iteration."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    last_print = t0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * it * it)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        sc_a = score_normalized(a)
        if sc_a < best_score:
            best_score = sc_a
            best_a = a.copy()
        sc_y = score_normalized(y)
        if sc_y < best_score:
            best_score = sc_y
            best_a = y.copy()

        if (it + 1) % 1000 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 20:
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): best={best_score:.8f} M_lp={M_lp:.6f}")
                last_print = elapsed

    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    """Compute LogSumExp objective and gradient w.r.t. z."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-18:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lse_lbfgs(a0, time_limit, T_start=80.0, T_min=0.2, n_stages=30, iters_per_stage=40):
    """Multi-stage adaptive LogSumExp L-BFGS."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)

    T = T_start
    stagnant = 0

    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, T, n)
            return obj

        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, T, n)
            return grad

        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-18,
                         'gtol': 1e-14, 'maxfun': iters_per_stage * 300,
                         'disp': False}
            )
            z = res.x
            a_check = softmax_from_z(z)
            sc = score_normalized(a_check)

            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()

            if stage > 0:
                obj_ratio = (sc - best_score) / (best_score + 1e-15)
                if obj_ratio > 0.003:
                    stagnant += 1
                else:
                    stagnant = max(0, stagnant - 1)

            if stagnant >= 3:
                T = max(T, T * 1.3)
                stagnant = 0
            else:
                T = max(T_min, T * 0.7)

            if (stage + 1) % 5 == 0:
                print(f"    LSE stage={stage+1}/{n_stages} T={T:.2f}: sc={sc:.8f} best={best_score:.8f}")

        except Exception:
            T = max(T_min, T * 0.8)

    return best_a, best_score


def peak_coordinate_descent(a0, time_limit, n_steps=500000):
    """Peak-directed coordinate descent with line search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    no_improve = 0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if no_improve > 50000:
            break

        b = fast_convolve(a)
        k_star = int(np.argmax(b))
        cur_score = 2.0 * n * b[k_star]

        # Gradient of conv[k_star] w.r.t. a
        g = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        idx = np.arange(lo, hi + 1)
        g[idx] = 2.0 * a[k_star - idx]

        # Find the coordinate with highest g (most to reduce) and lowest g (most to increase)
        # Move mass from high-g to low-g coordinates
        active = np.where(a > 1e-10)[0]
        if len(active) < 2:
            break

        g_active = g[active]
        a_active = a[active]

        # Source: coordinates with high gradient
        src_thresh = np.percentile(g_active, 90)
        sources = active[g_active >= src_thresh]
        # Destination: coordinates with low gradient
        dst_thresh = np.percentile(g_active, 10)
        dests = active[g_active <= dst_thresh]

        if len(sources) == 0 or len(dests) == 0:
            no_improve += 1
            # Try random perturbation
            rng = np.random.default_rng(step)
            pert = rng.normal(0, 0.01, n)
            a_new = a + pert
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            sc_new = score_normalized(a_new)
            if sc_new < best_score:
                best_score = sc_new
                best_a = a_new.copy()
                a = a_new
                no_improve = 0
            continue

        # Try moving a small fraction of mass
        for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
            total_src_mass = a[sources].sum()
            move = total_src_mass * frac
            if move < 1e-15:
                continue
            a_new = a.copy()
            # Scale down sources proportionally
            a_new[sources] -= a_new[sources] * frac
            # Distribute to dests proportionally to their current mass
            dst_mass = a_new[dests].sum()
            if dst_mass < 1e-15:
                a_new[dests] += move / len(dests)
            else:
                a_new[dests] += a_new[dests] * (move / dst_mass)
            a_new = np.maximum(a_new, 0)
            sn = a_new.sum()
            if sn < 1e-15:
                continue
            a_new = a_new / sn
            sc_new = score_normalized(a_new)

            if sc_new < cur_score * 0.9999:
                a = a_new
                no_improve = 0
                break
            else:
                continue
        else:
            no_improve += 1
            # If stuck for a while, try a random perturbation
            if no_improve % 100 == 0:
                rng = np.random.default_rng(step + 7)
                pert = rng.normal(0, 0.005, n)
                a_new = a + pert
                a_new = np.maximum(a_new, 0)
                a_new = a_new / a_new.sum()
                sc_new = score_normalized(a_new)
                if sc_new < best_score:
                    best_score = sc_new
                    best_a = a_new.copy()
                    a = a_new
                    no_improve = 0

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        if (step + 1) % 20000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CoordDesc step={step+1} ({elapsed:.0f}s): cur={cur_score:.8f} best={best_score:.8f} no_impr={no_improve}")

    return best_a, best_score


def fourier_flattening(a0, time_limit, n_iters=20000):
    """Gerchberg-Saxton style iteration in Fourier domain."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    # Target: make conv as flat as possible
    # In Fourier domain: |FFT(a)|^2 should be concentrated at DC
    # Equivalent: make FFT(a) have a "water-filled" magnitude profile

    beta = 0.5  # Interpolation parameter
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break

        A = fft(a)
        # Desired magnitude: proportional to 1/sqrt(k+1) to flatten the convolution
        # (This is a heuristic; the exact optimal profile is unknown)
        freqs = np.arange(n)
        desired_mag = 1.0 / np.sqrt(freqs + 1.0)
        desired_mag = desired_mag / desired_mag.sum() * np.abs(A).sum()

        # Interpolate current and desired magnitudes
        new_mag = (1 - beta) * np.abs(A) + beta * desired_mag
        # Preserve phase
        phase = np.angle(A)
        A_new = new_mag * np.exp(1j * phase)

        a_new = np.real(ifft(A_new))
        a_new = np.maximum(a_new, 0)
        sn = a_new.sum()
        if sn < 1e-15:
            break
        a_new = a_new / sn

        sc_new = score_normalized(a_new)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()

        # Accept if improvement, otherwise revert
        if sc_new < score_normalized(a):
            a = a_new
        else:
            # Try smaller step
            A revert = A  # keep old
            beta *= 0.9
            if beta < 0.01:
                beta = 0.01

        if (it + 1) % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Fourier it={it+1} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def structured_init(n, rng):
    """Generate structured initial sequences."""
    choice = rng.integers(0, 10)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 2
    elif choice == 2:
        x = np.linspace(-1, 1, n)
        a = np.exp(-2 * x ** 2)
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    elif choice == 4:
        a = np.ones(n) * (1.0 + 0.2 * np.cos(2 * np.pi * 3 * np.arange(n) / n))
    elif choice == 5:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.25 * np.cos(3 * x) + 0.15 * np.cos(7 * x)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) * np.cos(np.pi * x / 3) ** 2
    elif choice == 7:
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.5) ** 2) / 0.05) + 0.3 * np.exp(-((x - 0.2) ** 2) / 0.02) + 0.3 * np.exp(-((x - 0.8) ** 2) / 0.02)
    elif choice == 8:
        # Almost uniform with small structured perturbation
        a = np.ones(n)
        x = np.linspace(0, 1, n)
        a *= (1.0 + 0.05 * np.cos(2 * np.pi * 5 * x))
    else:
        # Binary-like sequence (motivated by difference sets)
        a = np.ones(n) * 0.1
        step = max(1, n // 7)
        for i in range(0, n, step):
            a[i:i + step // 3] = 1.0
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def multi_start_nesterov(a0, time_limit, solver, n_starts=3):
    """Run Nesterov LP from multiple slightly perturbed starting points."""
    n = len(a0)
    rng = np.random.default_rng(12345)
    best_overall_a = a0.copy()
    best_overall_score = score_normalized(a0)

    per_start_time = time_limit / n_starts
    for s_idx in range(n_starts):
        remaining = time_limit - (time.monotonic() - (time.monotonic() - per_start_time * s_idx))
        if remaining < 10:
            break

        if s_idx == 0:
            init = a0.copy()
        else:
            # Perturb the best so far
            pert = rng.normal(0, 0.02, n)
            init = best_overall_a + pert
            init = np.maximum(init, 0)
            init = init / init.sum()

        t_rem = time_limit - (time.monotonic() - (time.monotonic() - per_start_time * s_idx))
        t_rem = min(per_start_time * 1.2, t_rem)
        if t_rem < 5:
            break

        a_out, sc = nesterov_lp(init, t_rem, solver=solver, max_iters=200000)
        if a_out is not None and sc < best_overall_score:
            best_overall_score = sc
            best_overall_a = a_out.copy()
            print(f"    MultiStart[{s_idx}]: score={sc:.8f}")

    return best_overall_a, best_overall_score

            perturbed = perturbed / perturbed.sum()
            a_lse3, _ = lse_lbfgs(perturbed, remaining * 0.5, T_start=40.0, T_min=0.15, n_stages=15, iters_per_stage=30)
            try_update(a_lse3, "lse_perturbed")
    print(f"  Phase 3 done: best_score={best_score:.8f}")

    # === Phase 4: Peak coordinate descent - 15% time ===
    print(f"\n=== Phase 4: Peak Coordinate Descent (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t4 = min(0.15 * (budget_s - 30), remaining * 0.6)
    if t4 > 20:
        a_cd, _ = peak_coordinate_descent(best_seq, t4, n_steps=300000)
        try_update(a_cd, "coord_descent")

        # Second pass from the result
        remaining = global_deadline - time.monotonic()
        if remaining > 15:
            a_cd2, _ = peak_coordinate_descent(best_seq, remaining * 0.6, n_steps=200000)
            try_update(a_cd2, "coord_descent2")
    print(f"  Phase 4 done: best_score={best_score:.8f}")

    # === Phase 5: Fourier flattening - 5% time ===
    print(f"\n=== Phase 5: Fourier Flattening (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t5 = min(0.05 * (budget_s - 30), remaining * 0.5)
    if t5 > 15:
        a_ff, _ = fourier_flattening(best_seq, t5, n_iters=10000)
        try_update(a_ff, "fourier")
    print(f"  Phase 5 done: best_score={best_score:.8f}")

    # === Phase 6: Multi-length exploration - 5% time ===
    print(f"\n=== Phase 6: Multi-length (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t6_total = min(0.05 * (budget_s - 30), remaining * 0.6)
    if t6_total > 25:
        lengths_to_try = [n_base + 200, n_base - 200, n_base + 400]
        for n_try in lengths_to_try:
            remaining = global_deadline - time.monotonic()
            if remaining < 20:
                break
            t6 = min(t6_total / 3.0, remaining * 0.45)
            if t6 < 10:
                break
            print(f"  Trying n={n_try}, budget={t6:.0f}s")
            init_try = interpolate_sequence(best_seq, n_try)
            a_try, _ = nesterov_lp(init_try, t6 * 0.6, solver=solver, max_iters=50000)
            try_update(a_try, f"len_{n_try}")

            remaining = global_deadline - time.monotonic()
            if remaining > 10:
                a_try2, _ = lse_lbfgs(a_try, remaining * 0.5, T_start=30.0, T_min=0.2, n_stages=10, iters_per_stage=20)
                try_update(a_try2, f"len_{n_try}_lse")
    print(f"  Phase 6 done: best_score={best_score:.8f}")

    # === Phase 7: Final combined polish - 5% time ===
    print(f"\n=== Phase 7: Final polish (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    if remaining > 15:
        # Short Nesterov
        a_f1, _ = nesterov_lp(best_seq, remaining * 0.3, solver=solver, max_iters=30000)
        try_update(a_f1, "final_nesterov")

        remaining = global_deadline - time.monotonic()
        if remaining > 10:
            # Short LSE
            a_f2, _ = lse_lbfgs(best_seq, remaining * 0.4, T_start=8.0, T_min=0.1, n_stages=8, iters_per_stage=20)
            try_update(a_f2, "final_lse")

        remaining = global_deadline - time.monotonic()
        if remaining > 8:
            # Short coordinate descent
            a_f3, _ = peak_coordinate_descent(best_seq, remaining * 0.5, n_steps=50000)
            try_update(a_f3, "final_coord")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]


def propose_candidate(seed=42, budget_s=1800.0):
