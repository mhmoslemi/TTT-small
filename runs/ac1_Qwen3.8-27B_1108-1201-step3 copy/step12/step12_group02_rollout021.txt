    n = len(a)
    size = 2 * n - 1

    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def fast_score(a):
    """Compute 2n*max(b)/(sum(a))^2."""
    a = np.maximum(a, 0)
    n = len(a)
    s = a.sum()
    if s < 0.01:

    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)

def softmax_weights(values, temperature=1.0):
    """Compute softmax weights for a set of values."""
    v = np.asarray(values)
    v_shift = v - v.max()
    e = np.exp(v_shift / temperature)
    return e / e.sum()

def compute_subgradient_all(b, a, threshold_factor=0.5):
    """
    Compute subgradient of max(b) using all peaks above threshold.
    Returns gradient vector of length n.
    """
    n = len(a)

    b_threshold = threshold_factor * b_max

    mask = b >= b_threshold
    if not mask.any():

    # Weight by excess above mean (or by softmax)
    selected_b = b[mask]
    weights = selected_b / selected_b.sum()  # or softmax

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Accelerated mirror descent (multiplicative weights) with adaptive-temperature
   all-peaks softmax subgradient - O(n log n) per iteration via FFT
2. Peak equalization phase: targeted water-filling to flatten top autocorrelation peaks
3. Frequency-domain spectral shaping as an alternative optimization pathway
4. Multi-length exploration with diverse structured initializations
5. Adaptive restart and combinatorial polish
Key differences from prior approach: adaptive temperature annealing, all-peaks
(soft) subgradient instead of top-k, peak equalization via direct mass transfer,
and frequency-domain optimization for final refinement.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_autocorr(a):
    """Full linear autocorrelation a*a via FFT. Returns array of length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_score(a):
    """Compute 2n*max(b)/(sum(a))^2 quickly."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)


def evaluate_safe(a):
    """Call the official evaluate_sequence safely."""
    try:
        a = np.asarray(a, dtype=np.float64)
        a = np.clip(a, 0.0, 1000.0)
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def cross_corr_weights(w, a):
    """
    Compute (w cross a)[j] = sum_m w[m] * a[m-j] for valid indices.
    w has length 2n-1, a has length n, result has length n.
    """
    n = len(a)
    a_rev = a[::-1].copy()
    conv_len = n + len(w) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a_rev, fft_size)
    conv = irfft(W * A, fft_size)
    return conv[n - 1:n - 1 + n]


def mirror_descent_adaptive(a0, time_limit, T_start=50.0, T_end=1.0,
                            eta0=0.08, decay=0.55, seed=42, verbose=True):
    """
    Mirror descent with adaptive-temperature softmax subgradient.
    Uses all peaks (weighted by softmax) for a smoother gradient signal.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall = 0
    total_iters = 0

    while time.monotonic() - t0 < time_limit:
        # Adapt temperature: anneal from T_start to T_end
        frac = min(1.0, (time.monotonic() - t0) / (time_limit * 0.8))
        T = T_start * (T_end / T_start) ** frac

        b = fast_autocorr(a)
        b_max = b.max()

        # Softmax weights over all autocorrelation bins
        Tb = T * b
        Tb_shift = Tb - Tb.max()
        exp_Tb = np.exp(Tb_shift)
        w = exp_Tb / exp_Tb.sum()

        # Subgradient: grad = 2 * cross_corr(w, a)
        grad = 2.0 * cross_corr_weights(w, a)

        # Learning rate with power decay
        total_iters += 1
        eta = eta0 / (1.0 + total_iters / 500.0) ** decay

        # Mirror descent update (multiplicative weights on simplex)
        log_update = -eta * grad
        log_update -= log_update.max()  # numerical stability
        a_new = a * np.exp(log_update)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a + 0.01 * rng.standard_normal(n)
            a_new = np.maximum(a_new, 0)
            s_new = a_new.sum()
            a_new = a_new / max(s_new, 1e-15)
        else:
            a_new = a_new / s_new
        a = a_new

        # Check score periodically
        if total_iters < 200:
            check_int = 2
        elif total_iters < 2000:
            check_int = 5
        elif total_iters < 10000:
            check_int = 10
        else:
            check_int = 25

        if total_iters % check_int == 0:
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stall = 0
            else:
                stall += 1

        # Stall handling: inject noise
        if stall > 300:
            noise = min(0.04, 0.005 + 0.002 * (stall / 50.0))
            a = a + noise * rng.standard_normal(n)
            a = np.maximum(a, 0)
            a = a / a.sum()
            stall = 0

        if verbose and total_iters % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD t={total_iters} ({elapsed:.0f}s): best={best_score:.6f}, T={T:.1f}")

    return best_a, best_score


def peak_equalization(a0, time_limit, seed=77, verbose=True):
    """
    Targeted peak equalization: iteratively identify the top peaks and
    move mass from peak-contributing positions to low-activity positions.
    Uses a "water-filling" strategy to flatten the autocorrelation.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        b_max = b.max()

        # Get top peaks
        top_k = min(8, 2 * n - 1)
        top_idx = np.argsort(b)[::-1][:top_k]

        # For each top peak, find the dominant contributing pair
        improved = False
        for k in range(top_k):
            m = int(top_idx[k])
            if b[m] < b_max * 0.8:  # only work on significant peaks
                break

            js = max(0, m - n + 1)
            je = min(n - 1, m)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[m - idx]

            if contrib.sum() < 1e-15:
                continue

            # Find the position with highest contribution
            src_local = int(np.argmax(contrib))
            src = int(idx[src_local])

            # Find target: position with lowest "activity" (low a value
            # and not contributing to any top peak)
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_idx[:5]:
                k2 = int(k2)
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            candidates = np.where(~peak_mask)[0]
            if len(candidates) < 3:
                candidates = np.where(np.arange(n) != src)[0]

            # Pick the candidate with lowest a value
            sorted_cand = candidates[np.argsort(a[candidates])[:max(8, len(candidates) // 8)]]
            if len(sorted_cand) == 0:
                continue
            dst = int(sorted_cand[rng.integers(len(sorted_cand))])

            # Try multiple transfer fractions
            for frac in [0.001, 0.003, 0.007, 0.015, 0.03, 0.06]:
                amt = a[src] * frac
                if amt < 1e-15 or a[src] - amt < 0:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                a_trial = a_trial / a_trial.sum()
                sc = fast_score(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial.copy()
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break

            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape
            if rng.integers(4) == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.015)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= 0:
                        a_trial = a_trial / a_trial.sum()
                        sc = fast_score(a_trial)
                        if sc < best_score:
                            a = a_trial.copy()
                            best_score = sc
                            best_a = a.copy()
                            stall = 0
            if stall > 300:
                break

    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    PeakEq ({elapsed:.0f}s): best={best_score:.6f}")
    return best_a, best_score


def freq_domain_optimize(a0, time_limit, seed=13, verbose=True):
    """
    Optimize in the frequency domain. The autocorrelation b = IFFT(|A|^2)
    where A = FFT(a_padded). We directly shape |A|^2 to reduce peaks.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    fft_size = 1 << (2 * n - 1).bit_length() - 1
    if not (fft_size & (fft_size - 1) == 0):
        fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    lr = 0.02

    for it in range(200000):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute A and b
        A = rfft(a, fft_size)
        power = A.real ** 2 + A.imag ** 2
        b = irfft(power, fft_size)[:2 * n - 1]
        b_max = b.max()

        # Gradient of b_max w.r.t. |A|^2 is delta at the peak position
        # In frequency domain, changing |A[k]|^2 affects all b[m]
        # Use the relation: db[m]/d|A[k]|^2 = (1/N)*exp(2*pi*i*k*m/N)
        # So to reduce b at the peak, we should reduce |A[k]|^2 weighted by
        # the real part of exp(2*pi*i*k*m_peak/N)

        peak_idx = int(np.argmax(b))
        # Softmax over top peaks for smoothness
        T = 20.0
        Tb = T * b
        Tb_shift = Tb - Tb.max()
        w = np.exp(Tb_shift)
        w = w / w.sum()

        # Gradient in frequency domain: for each frequency bin k,
        # the contribution to reducing weighted peaks is:
        # grad_power[k] = sum_m w[m] * (1/N) * cos(2*pi*k*m/N)
        # We want to DECREASE power where grad is positive
        k_bins = np.arange(fft_size // 2 + 1)
        m_range = np.arange(2 * n - 1)
        phase = 2.0 * np.pi * np.outer(k_bins, m_range) / fft_size
        # This is too large (O(n^2)), use FFT instead
        # The gradient of sum_m w[m]*b[m] w.r.t. power[k] is:
        # (1/N) * Re(FFT(w)[k]) ... approximately
        W = rfft(w, fft_size)
        grad_power = W.real * 2.0  # simplified

        # Update: reduce power where gradient is high
        power_new = power - lr * grad_power
        power_new = np.maximum(power_new, 1e-20)

        # Reconstruct a from new power (phase-preserving)
        phase_A = np.angle(A)
        A_new = np.sqrt(power_new) * np.exp(1j * phase_A)
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a_new = a_new / s_new
            sc = fast_score(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
            else:
                # Reduce learning rate
                lr *= 0.95
                a = a  # keep old
        else:
            lr *= 0.9

        if it % 100 == 0:
            lr = 0.02 * (0.99 ** (it // 100))

    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    FreqOpt ({elapsed:.0f}s): best={best_score:.6f}")
    return best_a, best_score


def diverse_init(n, rng, variant=None):
    """Generate diverse structured initializations."""
    if variant is None:
        variant = int(rng.integers(0, 12))

    if variant == 0:
        x = np.linspace(0, 1, n)
        a = np.maximum(1 - 2 * abs(x - 0.5), 0)
    elif variant == 1:
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif variant == 2:
        x = np.linspace(-1, 1, n)
        a = np.exp(-3 * x ** 2)
    elif variant == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-25 * (x - 0.2) ** 2) + np.exp(-25 * (x - 0.8) ** 2)
    elif variant == 4:
        x = np.linspace(0, 1, n)
        a = x ** 2 + (1 - x) ** 2 + 0.05
    elif variant == 5:
        x = np.linspace(0, 1, n)
        a = 1 + 0.4 * np.cos(4 * np.pi * x) + 0.1 * np.cos(8 * np.pi * x)
    elif variant == 6:
        x = np.linspace(-1, 1, n)
        a = np.sinc(x) ** 2
    elif variant == 7:
        # Raised cosine
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x)) ** 2
    elif variant == 8:
        # Multi-harmonic
        x = np.linspace(0, 2 * np.pi, n)
        a = 1 + 0.25 * np.cos(2 * x) + 0.15 * np.cos(5 * x) + 0.05 * np.cos(9 * x)
        a = np.maximum(a, 0)
    elif variant == 9:
        # Triangular (optimal for some low-autocorrelation problems)
        x = np.linspace(0, 1, n)
        a = np.maximum(1 - 2 * abs(x - 0.5), 0) ** 2
    elif variant == 10:
        # Sidelnbounded-like: interpolate known good pattern
        x = np.linspace(0, 1, n)
        a = 0.6 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.1 * np.cos(2 * np.pi * 7 * x)
        a = np.maximum(a, 0)
    else:
        # Random smooth
        a = np.ones(n)
        for _ in range(8):
            center = rng.uniform(0, n)
            width = rng.uniform(n * 0.05, n * 0.3)
            amp = rng.uniform(-0.4, 0.8)
            x = np.arange(n, dtype=np.float64)
            a += amp * np.exp(-(x - center) ** 2 / (2 * width ** 2))
        a = np.maximum(a, 0)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def lse_refine(a0, time_limit, T_start=50.0, T_end=0.3, n_stages=10, seed=55):
    """
    LogSumExp-smoothed refinement with L-BFGS in z-space (a = softmax(z)).
    Multi-stage temperature annealing.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s

    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = fast_score(a_init)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        iters = max(10, min(50, int(20 + 30 * si / max(1, n_stages))))

        def func(z_in):
            n_in = len(z_in)
            z_shift = z_in - z_in.max()
            e = np.exp(z_shift)
            a_in = e / e.sum()
            b = fast_autocorr(a_in)
            Tb = stage_T * b
            Tb_shift = Tb - Tb.max()
            obj = (Tb.max() + np.log(np.exp(Tb_shift).sum())) / stage_T
            return obj

        def jac(z_in):
            n_in = len(z_in)
            z_shift = z_in - z_in.max()
            e = np.exp(z_shift)
            a_in = e / e.sum()
            b = fast_autocorr(a_in)
            Tb = stage_T * b
            Tb_shift = Tb - Tb.max()
            exp_Tb = np.exp(Tb_shift)
            w = exp_Tb / exp_Tb.sum()
            cc = cross_corr_weights(w, a_in)
            grad_a = 2.0 * cc
            S = np.dot(grad_a, a_in)
            grad_z = a_in * (grad_a - S)
            return grad_z

        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': iters, 'ftol': 1e-14,
                                    'gtol': 1e-8, 'maxfun': iters * 300,
                                    'disp': False})
            z = res.x
            z_shift = z - z.max()
            a = np.exp(z_shift)
            a = a / a.sum()
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        except Exception:
            pass

        if si % 3 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage {si+1}/{n_stages} (T={stage_T:.1f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Multi-phase optimization combining:
    - Adaptive-temperature mirror descent (primary driver)
    - Peak equalization (targeted polish)
    - Frequency-domain optimization (alternative pathway)
    - LSE refinement (smooth local optimization)
    - Multi-length exploration
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 20.0, 1770.0)
    rng = np.random.default_rng(seed)

    best_seq = None
    best_score = float('inf')

    # Try to use parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = fast_score(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = fast_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Determine starting length
    if best_seq is not None:
        n_start = len(best_seq)
    else:
        n_start = 1600
        best_seq = diverse_init(n_start, rng, variant=0)
        best_score = fast_score(best_seq)
        print(f"Default init: n={n_start}, score={best_score:.6f}")

    # === PHASE 1: Primary mirror descent at parent length ===
    print(f"\n=== Phase 1: Adaptive MD at n={n_start} ===")
    remaining = time_budget - (time.monotonic() - t_global)
    t_md1 = min(remaining * 0.30, 500.0)
    if t_md1 > 30:
        a_out, _ = mirror_descent_adaptive(best_seq, t_md1, T_start=60.0, T_end=2.0,
                                            eta0=0.06, decay=0.5, seed=1, verbose=True)
        try_update(a_out, f"md1_n{n_start}")

    # === PHASE 2: Second MD run with different params ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: MD variant at n={n_start} ===")
    t_md2 = min(remaining * 0.12, 200.0)
    if t_md2 > 20 and best_seq is not None:
        a_out, _ = mirror_descent_adaptive(best_seq, t_md2, T_start=30.0, T_end=0.5,
                                            eta0=0.04, decay=0.7, seed=2, verbose=False)
        try_update(a_out, f"md2_n{n_start}")

    # === PHASE 3: Peak equalization ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Peak Equalization at n={n_start} ===")
    t_pe1 = min(remaining * 0.12, 200.0)
    if t_pe1 > 20 and best_seq is not None:
        a_out, _ = peak_equalization(best_seq, t_pe1, seed=33, verbose=True)
        try_update(a_out, f"pe1_n{n_start}")

    # === PHASE 4: MD again on the improved sequence ===
    remaining = time_budget - (time.monotonic() - t_global)
    t_md3 = min(remaining * 0.12, 200.0)
    if t_md3 > 20 and best_seq is not None:
        a_out, _ = mirror_descent_adaptive(best_seq, t_md3, T_start=20.0, T_end=1.0,
                                            eta0=0.03, decay=0.6, seed=4, verbose=False)
        try_update(a_out, f"md3_n{n_start}")

    # === PHASE 5: LSE refinement ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: LSE Refinement at n={n_start} ===")
    t_lse1 = min(remaining * 0.12, 200.0)
    if t_lse1 > 20 and best_seq is not None:
        a_out, _ = lse_refine(best_seq, t_lse1, T_start=40.0, T_end=0.5,
                              n_stages=10, seed=55)
        try_update(a_out, f"lse_n{n_start}")

    # === PHASE 6: Peak equalization again ===
    remaining = time_budget - (time.monotonic() - t_global)
    t_pe2 = min(remaining * 0.08, 120.0)
    if t_pe2 > 15 and best_seq is not None:
        a_out, _ = peak_equalization(best_seq, t_pe2, seed=66, verbose=False)
        try_update(a_out, f"pe2_n{len(best_seq)}")

    # === PHASE 7: Try a longer length ===
    n_long = int(round(n_start * 1.3))
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 7: Longer length n={n_long} ===")
    t_md4 = min(remaining * 0.15, 250.0)
    if t_md4 > 30 and best_seq is not None:
        init_long = interpolate_to_length(best_seq, n_long)
        a_out, _ = mirror_descent_adaptive(init_long, t_md4, T_start=50.0, T_end=2.0,
                                            eta0=0.04, decay=0.5, seed=7, verbose=True)
        try_update(a_out, f"md_long_n{n_long}")

    # Peak equalization at longer length
    remaining = time_budget - (time.monotonic() - t_global)
    t_pe3 = min(remaining * 0.08, 100.0)
    if t_pe3 > 15 and best_seq is not None and len(best_seq) == n_long:
        a_out, _ = peak_equalization(best_seq, t_pe3, seed=77, verbose=False)
        try_update(a_out, f"pe_long_n{n_long}")

    # === PHASE 8: Try a shorter length (sometimes shorter is better) ===
    n_short = int(round(n_start * 0.75))
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 8: Shorter length n={n_short} ===")
    t_md5 = min(remaining * 0.08, 120.0)
    if t_md5 > 20 and best_seq is not None:
        init_short = interpolate_to_length(best_seq, n_short)
        a_out, _ = mirror_descent_adaptive(init_short, t_md5, T_start=40.0, T_end=1.0,
                                            eta0=0.05, decay=0.5, seed=8, verbose=False)
        try_update(a_out, f"md_short_n{n_short}")

    # === PHASE 9: Diverse initialization at best length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best_len = len(best_seq) if best_seq is not None else n_start
    print(f"\n=== Phase 9: Diverse inits at n={n_best_len} ===")
    t_div = min(remaining * 0.08, 100.0)
    if t_div > 20:
        for v in [4, 7, 10]:
            t_each = t_div / 3.0
            init = diverse_init(n_best_len, rng, variant=v)
            a_out, _ = mirror_descent_adaptive(init, t_each, T_start=50.0, T_end=2.0,
                                                eta0=0.06, decay=0.5, seed=v * 17, verbose=False)
            try_update(a_out, f"div_v{v}")

    # === PHASE 10: Final polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 10: Final Polish ===")
    if best_seq is not None:
        # Quick MD with very low T (focus tightly on the max peak)
        t_f1 = min(remaining * 0.25, 80.0)
        if t_f1 > 10:
            a_out, _ = mirror_descent_adaptive(best_seq, t_f1, T_start=100.0, T_end=5.0,
                                                eta0=0.02, decay=0.7, seed=99, verbose=False)
            try_update(a_out, "final_md")

        # Quick peak equalization
        remaining = time_budget - (time.monotonic() - t_global)
        t_f2 = min(remaining * 0.35, 80.0)
        if t_f2 > 10:
            a_out, _ = peak_equalization(best_seq, t_f2, seed=111, verbose=False)
            try_update(a_out, "final_pe")

        # Final very short MD
        remaining = time_budget - (time.monotonic() - t_global)
        t_f3 = min(remaining * 0.40, 60.0)
        if t_f3 > 5:
            a_out, _ = mirror_descent_adaptive(best_seq, t_f3, T_start=200.0, T_end=10.0,
                                                eta0=0.01, decay=0.8, seed=222, verbose=False)
            try_update(a_out, "final_md2")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        elapsed = time.monotonic() - t_global
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={elapsed:.0f}s")
    else:
        print("ERROR: No valid sequence found!")
        return [1.0] * 100

    return [float(x) for x in best_seq]
