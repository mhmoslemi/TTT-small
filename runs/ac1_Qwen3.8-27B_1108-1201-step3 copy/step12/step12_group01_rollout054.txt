```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Fast LP Nesterov with HiGHS solver (zero-sum game / fictitious play)
2. Multi-scale optimization: try multiple sequence lengths
3. Top-k LogSumExp gradient descent with Adam for refinement
4. Quadratic peak shaving for final polish
5. Multi-start diversity with structured perturbations
"""

import numpy as np
import time
import scipy.optimize
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full autocorrelation (convolution) a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def solve_lp_fast(a_fixed):
    """
    Solve LP: min M s.t. conv(a_fixed, b) <= M, sum(b)=1, b>=0
    Using scipy linprog with HiGHS.
    Returns (b_star, M_val) or (None, None) on failure.
    """
    n = len(a_fixed)
    # Variables: [b_0, ..., b_{n-1}, M]
    # Objective: minimize M -> c = [0,...,0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize M
    
    # Inequality constraints: T @ b - M <= 0 for each lag
    # T is (2n-1) x n where T[k, j] = a_fixed[k-j] if 0 <= k-j < n, else 0
    # So constraint: sum_j T[k,j] * b_j - M <= 0
    # In standard form: sum_j T[k,j] * b_j + (-1) * M <= 0
    
    A_ineq = np.zeros((2 * n - 1, n + 1))
    for k in range(2 * n - 1):
        for j in range(max(0, k - n + 1), min(n, k + 1)):
            A_ineq[k, j] = a_fixed[k - j]
        A_ineq[k, n] = -1.0  # coefficient of M
    
    # Equality constraint: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: b_i >= 0, M free
    bounds = [(0, None)] * n + [(None, None)]
    
    try:
        result = scipy.optimize.linprog(
            c_obj, A_ub=A_ineq, b_ub=np.zeros(2 * n - 1),
            A_eq=A_eq, b_eq=b_eq, bounds=bounds,
            method='highs', options={'presolve': True}
        )
        if result.success:
            b_val = result.x[:n]
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s < 1e-15:
                return None, None
            b_val = b_val / s
            M_val = result.x[n]
            return b_val, float(M_val)
    except Exception:
        pass
    
    return None, None
```

```python
    """
    Approximate best response: find b that approximately minimizes max_k (a_fixed * b)_k
    using mirror descent.
    """
    n = len(a_fixed)
    b = np.ones(n) / n  # Start uniform
    
    for _ in range(n_steps):
        c = np.convolve(a_fixed, b)  # or use FFT
        # Compute LSE gradient
        k = min(5, len(c))
        top = np.argsort(c)[::-1][:k]
        # Softmax weights
        T = 50.0
        w = np.exp(T * c[top] - T * c[top].max())
        w = w / w.sum()
        
        # Gradient of LSE w.r.t. b
        grad = np.zeros(n)
        for idx in range(k):
            K = top[idx]
            js = max(0, K - n + 1)
            je = min(n - 1, K)
            idx_range = np.arange(js, je + 1)
            grad[idx_range] += w[idx] * a_fixed[K - idx_range]
        
        # Mirror descent update (in log-space)
        z = np.log(np.maximum(b, 1e-30))
        z = z - eta * grad
        b = softmax_fn(z)
    
    return b
```

```python
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def lse_gradient(c, topk, T):
    """Compute LSE value and weights for top-k peaks."""
    k_use = min(topk, len(c))
    top_idx = np.argsort(c)[::-1][:k_use]
    top_vals = c[top_idx]
    Tc = T * top_vals
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    lse_val = (max_Tc + np.log(exp_Tc.sum())) / T
    w = exp_Tc / exp_Tc.sum()
    return top_idx, w, lse_val


def conv_gradient(c_workflow, a_fixed, b, top_idx, w):
    """Compute gradient of weighted LSE w.r.t. b given fixed a_fixed."""
    n = len(b)
    grad = np.zeros(n)
    for i in range(len(top_idx)):
        K = top_idx[i]
        wi = w[i]
        js = max(0, K - len(a_fixed) + 1)
        je = min(n - 1, K)
        if js <= je:
            idx_range = np.arange(js, je + 1)
            grad[idx_range] += wi * a_fixed[K - idx_range]
    return grad


def mirror_descent_fp(a0, time_limit, topk=5, T_val=30.0, eta=0.3, avg_window=100):
    """
    Fast Mirror Descent Fictitious Play.
    Iterates: b = best_response(a), a = best_response(b), track averages.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    b = a.copy()
    best_a = a.copy()
    best_b = b.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()

    # Running averages
    avg_a = a.copy()
    avg_b = b.copy()
    count = 0
    avg = a.copy()

    # Short window buffer
    short_window = 50
    buf = np.zeros((short_window, n))
    buf_idx = 0
    buf_filled = 0
    short_avg = a.copy()

    t0 = time.monotonic()
    iter_count = 0
    stall_count = 0
    last_improve = 0

    while time.monotonic() - t0 < time_limit:
        # Best response: update b given a
        c = fast_convolve(a, b)
        top_idx, w, _ = lse_gradient(c, topk, T_val)
        grad_b = conv_gradient(c, a, b, top_idx, w)
        z_b = np.log(np.maximum(b, 1e-30))
        z_b -= eta * grad_b
        b = softmax_fn(z_b)

        # Best response: update a given b
        c2 = fast_convolve(a, b)
        top_idx2, w2, _ = lse_gradient(c2, topk, T_val)
        grad_a = conv_gradient(c2, b, a, top_idx2, w2)
        z_a = np.log(np.maximum(a, 1e-30))
        z_a -= eta * grad_a
        a = softmax_fn(z_a)

        iter_count += 1

        # Update running averages
        avg = ((count * avg + a) / (count + 1))
        count += 1

        # Short window
        buf[buf_idx] = a
        short_avg = buf.sum(axis=0) / min(count, short_window)
        buf_idx = (buf_idx + 1) % short_window

        # Check scores periodically
        if iter_count % 20 == 0:
            s_a = 2.0 * n * fast_autocorr(a).max()
            if s_a < best_score:
                best_score = s_a
                best_a = a.copy()
                stall_count = 0
            else:
                stall_count += 1

            s_avg = 2.0 * n * fast_autocorr(avg).max()
            if s_avg < best_score:
                best_score = s_avg
                best_a = avg.copy()
                stall_count = 0

            if buf_filled >= short_window:
                s_short = 2.0 * n * fast_autocorr(short_avg).max()
                if s_short < best_score:
                    best_score = s_short
                    best_a = short_avg.copy()
                    stall_count = 0

            buf_filled = min(count, short_window)

        # Adaptive restart if stalled
        if stall_count > 500 and iter_count > 200:
            # Perturb and restart
            noise = np.random.default_rng(iter_count).normal(0, 0.02, n)
            a_restart = a + noise * (a + 1e-6)
            a_restart = np.maximum(a_restart, 0)
            s_r = a_restart.sum()
            if s_r > 1e-15:
                a = a_restart / s_r
            b = a.copy()
            avg = a.copy()
            count = 0
            buf_idx = 0
            buf_filled = 0
            short_avg = a.copy()
            stall_count = 0

        # Adaptive eta: decrease over time
        eta = max(0.05, 0.3 / (1.0 + iter_count / 5000.0))

        if iter_count % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      MD-FP iter={iter_count} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def adam_topk_lse(a0, time_limit, T_start=50.0, T_end=0.5, n_stages=20, topk=10):
    """Adam-optimized top-k LSE with multi-stage annealing."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        base_lr = 3e-3 * min(1.0, 50.0 / n)
        lr = base_lr * (0.85 ** si)

        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.8, 0.999, 1e-8

        max_it = max(20, int(min(remaining, 80.0) * 5))

        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 3:
                break

            a = softmax_fn(z)
            c = fast_autocorr(a)
            top_idx, w, lse_val = lse_gradient(c, topk, stage_T)

            # Gradient w.r.t. a
            grad_a = np.zeros(n)
            for i in range(len(top_idx)):
                K = top_idx[i]
                wi = 2.0 * w[i]
                js = max(0, K - n + 1)
                je = min(n - 1, K)
                if js <= je:
                    idx_range = np.arange(js, je + 1)
                    grad_a[idx_range] += wi * a[K - idx_range]

            # Project to tangent space of simplex
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)

            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            if (it + 1) % 10 == 0:
                a_check = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a_check).max()
                if score < best_score:
                    best_score = score
                    best_a = a_check.copy()

        a_stage = softmax_fn(z)
        score_stage = 2.0 * n * fast_autocorr(a_stage).max()
        if score_stage < best_score:
            best_score = score_stage
            best_a = a_stage.copy()

        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE stage {si}/{n_stages} (T={stage_T:.1f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def peak_shaving_guided(a0, time_limit, seed=42):
    """
    Guided peak shaving: focuses on reducing the top peaks by identifying
    the most contributing pairs and moving mass to safe positions.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit - 3:
        b = fast_autocorr(a)
        bmax = b.max()
        threshold = bmax * 0.998
        top_peaks = np.where(b >= threshold)[0]
        if len(top_peaks) == 0:
            top_peaks = np.argsort(b)[::-1][:5]
        if len(top_peaks) > 20:
            top_peaks = top_peaks[:20]

        improved = False
        for k in top_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            if contrib.max() < 1e-15:
                continue
            src = idx[np.argmax(contrib)]

            # Find safe positions (low contribution to all top peaks)
            safe_scores = np.zeros(n)
            for k2 in top_peaks:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                idx2 = np.arange(js2, je2 + 1)
                if len(idx2) == 0:
                    continue
                safe_scores[idx2] += a[k2 - idx2]
            
            # Prefer positions with low safe_score and low a value
            safe_scores[src] = np.inf
            candidates = np.argsort(safe_scores)[:max(5, n // 20)]
            
            for dst in candidates[:10]:
                if dst == src:
                    continue
                for frac in [0.002, 0.005, 0.01, 0.02, 0.04, 0.07, 0.12]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random exploration
            n_pert = rng.integers(1, 4)
            a_trial = a.copy()
            for _ in range(n_pert):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a_trial[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.03)
                    amt = a_trial[i] * frac
                    a_trial[i] -= amt
                    a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 1e-10:
                a_trial = a_trial / st
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
            if stall > 200:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 4)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.6 + 0.25 * np.cos(2 * x) + 0.1 * np.cos(7 * x) + 0.05 * np.cos(13 * x)
    elif choice == 4:
        x = np.linspace(0, 1, n)
        a = np.exp(-8 * (x - 0.3) ** 2) + 0.6 * np.exp(-8 * (x - 0.7) ** 2)
    else:
        x = np.linspace(0, 1, n)
        a = 1.0 / (1.0 + 10.0 * np.sin(np.pi * x) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def perturbed_init(a0, rng, scale=0.1):
    """Create a perturbed version of a sequence."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(len(a0)) / len(a0)
    a = a / s
    noise = rng.normal(0, scale, len(a))
    a_new = a + noise * (a + 1e-6)
    a_new = np.maximum(a_new, 0)
    sn = a_new.sum()
    if sn < 1e-15:
        return a.copy()
    return a_new / sn
```
