**Coordinate Descent Polish:**

Given a near-optimal sequence a (normalized to sum = sqrt(2n)), the goal is to reduce max_k conv(a,a)[k].

Key idea: if the peak is at lag k*, we want to reduce the contributions to conv(a,a)[k*].
The contributions are: a[j] * a[k*-j] for valid j.

Strategy:
1. Find the top K peak lags
2. For each peak, identify the "heavy" contributors (pairs with large product)
3. Move mass from heavy positions to light positions
4. Check if the peak decreased

A more efficient approach:
- Compute the gradient of max_k conv(a,a)[k] (subgradient at the peak)
- Do a line search in the negative gradient direction, projected onto the simplex

The subgradient at a when peak is at k* is:
g[i] = 2 * a[k*-i] if 0 <= k*-i <= n-1, else 0

The direction to move is: -g + (mean(-g)) * ones (to stay on the sum constraint)
Or more precisely: move from a towards the point that minimizes g^T x on the simplex.
The minimizer is: x = e_{argmin(g)} (a vertex of the simplex).

So the FW direction is: e_{argmin(g)} - a (normalized).

def fast_autocorr(a):

def dual_greedy(a, M):
    n = len(a)
    a_safe = np.maximum(a, 1e-15)
    y = np.zeros(2 * n - 1)
    window_sum = 0.0
    for j in range(n):
        needed = 1.0 / a_safe[j] - window_sum
        if needed > 0:
            y[j + n - 1] += needed
            window_sum += needed
        window_sum -= y[j]
    return y, M * y.sum()


def primal_from_dual(a, y_dual, M):
    n = len(a)
    a_safe = np.maximum(a, 1e-15)
    y_csum = np.zeros(2 * n)
    y_csum[1:] = np.cumsum(y_dual)
    ws = y_csum[n:n * 2 - 1] - y_csum[:n - 1]
    thresholds = 1.0 / a_safe
    tol = 1e-9 * np.max(thresholds) + 1e-15
    tight = ws >= (thresholds - tol)
    if not tight.any():
        b = np.ones(n)
        ab = fast_cross(a, b)
        max_ab = ab.max()
        if max_ab > M and max_ab > 1e-15:
            b *= M / max_ab
        return b
    b = np.zeros(n)
    b[tight] = 1.0 / a_safe[tight]
    ab = fast_cross(a, b)
    max_ab = ab.max()
    if max_ab > M and max_ab > 1e-15:
        b *= M / max_ab
    return b

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss

        a = np.ones(n)
        for freq in [3, 7, 13]:
            a += 0.3 * np.cos(2 * np.pi * freq * np.arange(n) / n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 120.0, 1680.0)

    print(f"Total time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    if parent is None:
        n_base = 2000
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")
    else:
        n_base = len(best_seq)

    # Phase 1: Polish parent with Nesterov fast LP
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Nesterov LP n={n_base} ===")
    t1 = min(remaining * 0.12, 180.0)
    if t1 > 20:
        a_work = best_seq * np.sqrt(2.0 * n_base) / best_seq.sum()
        a_out, _ = nesterov_fast_lp(a_work, t1, verbose=True)
        try_update(a_out, f"nlp_n{n_base}")

    # Phase 2: Grow to n2
    n2 = n_base + 400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Nesterov LP n={n2} ===")
    t2 = min(remaining * 0.20, 300.0)
    if t2 > 30:
        init2 = interpolate_sequence(best_seq, n2)
        a_work = init2 * np.sqrt(2.0 * n2) / init2.sum()
        a_out, _ = nesterov_fast_lp(a_work, t2, verbose=True)
        try_update(a_out, f"nlp_n{n2}")

    # Phase 3: Grow to n3
    n3 = n_base + 800
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Nesterov LP n={n3} ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 30:
        init3 = interpolate_sequence(best_seq, n3)
        a_work = init3 * np.sqrt(2.0 * n3) / init3.sum()
        a_out, _ = nesterov_fast_lp(a_work, t3, verbose=True)
        try_update(a_out, f"nlp_n{n3}")

    # Phase 4: Grow to n4
    n4 = n_base + 1400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Nesterov LP n={n4} ===")
    t4 = min(remaining * 0.20, 280.0)
    if t4 > 30:
        init4 = interpolate_sequence(best_seq, n4)
        a_work = init4 * np.sqrt(2.0 * n4) / init4.sum()
        a_out, _ = nesterov_fast_lp(a_work, t4, verbose=True)
        try_update(a_out, f"nlp_n{n4}")

    # Phase 5: Even larger n5
    n5 = n_base + 2400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Nesterov LP n={n5} ===")
    t5 = min(remaining * 0.15, 220.0)
    if t5 > 30:
        init5 = interpolate_sequence(best_seq, n5)
        a_work = init5 * np.sqrt(2.0 * n5) / init5.sum()
        a_out, _ = nesterov_fast_lp(a_work, t5, verbose=True)
        try_update(a_out, f"nlp_n{n5}")

    # Phase 6: Peak polish on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Peak polish n={n_best} ===")
    t6 = min(remaining * 0.15, 200.0)
    if t6 > 10:
        a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = peak_polish(a_work, t6, seed=777)
        try_update(a_out, "polish")

    # Phase 7: Fine coordinate descent
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Coord descent n={n_best} ===")
    t7 = min(remaining * 0.15, 180.0)
    if t7 > 10:
        a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = fine_coord_descent(a_work, t7, seed=123)
        try_update(a_out, "coord")

    # Phase 8: Final peak polish
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        n_best = len(best_seq)
        print(f"\n=== Phase 8: Final polish n={n_best} ===")
        t8 = min(remaining * 0.5, 120.0)
        if t8 > 5:
            a_work = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
            a_out, _ = peak_polish(a_work, t8, seed=999)
            try_update(a_out, "final_polish")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
