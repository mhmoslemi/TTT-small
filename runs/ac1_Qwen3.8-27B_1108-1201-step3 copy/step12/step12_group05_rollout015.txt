from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def compute_subgrad_fft(a, w):
    """
    Compute subgradient: grad_j = 2 * sum_k w_k * a[k-j]
    where w is a probability distribution over convolution positions.
    Uses FFT for O(n log n) computation.
    
    a has length n, w has length 2n-1 (convolution length).
    grad_j = 2 * sum_k w_k * a[k-j] for k-j in [0, n-1]
    = 2 * (w * a_reversed) where * is cross-correlation
    
    Actually: grad_j = 2 * sum_k w_k * a[k-j]
    Let's define c_k = w_k, d_m = a_m
    Then grad_j = 2 * sum_k c_k * d_{k-j}
    This is the cross-correlation of c and d at lag j.
    
    Cross-correlation via FFT:
    corr(j) = sum_k c_k * d_{k-j} = ifft(fft(c) * conj(fft(d)))
    
    But we need to be careful with the indexing and lengths.
    """
    n = len(a)
    conv_len = 2 * n - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    
    # We want grad_j = 2 * sum_k w_k * a[k-j] for j = 0,...,n-1
    # where a[k-j] = 0 if k-j not in [0, n-1]
    
    # This is the cross-correlation of w and a at lag j
    # = sum_k w[k] * a[k-j]
    
    # Using FFT: corr = ifft(conj(fft(a)) * fft(w))
    # But lengths need to match. Let's pad a to length conv_len.
    
    a_padded = np.zeros(conv_len)
    a_padded[:n] = a
    
    F_a = rfft(a_padded, fft_size)
    F_w = rfft(w, fft_size)
    
    # Cross-correlation: sum_k w[k] * a[k-j] 
    # = ifft(fft(w) * conj(fft(a)))
    # Wait, let me be more careful.
    # 
    # cross_corr[j] = sum_k w[k] * a[k-j]
    # = sum_k w[k] * a[k-j]
    # 
    # In terms of convolution: if we define a_rev[m] = a[-m mod n]... 
    # Actually, let's just use the direct formula.
    #
    # sum_k w[k] * a[k-j] where a[i] = 0 for i < 0 or i >= n
    # = sum_{k=j}^{j+n-1} w[k] * a[k-j]  (only valid a indices)
    #
    # This is (w * a_rev)[j + n - 1] where a_rev is a reversed and padded
    # 
    # Alternatively: define b[m] = a[n-1-m] (reversed a, length n)
    # Then sum_k w[k] * a[k-j] = sum_k w[k] * b[n-1-(k-j)]
    # = (w * b)[j + n - 1]  (full convolution)
    
    a_rev = a[::-1]  # length n
    # Full convolution of w (length 2n-1) and a_rev (length n) has length 3n-2
    fft_size2 = 1 << ((conv_len + n - 1) - 1).bit_length()
    F_w2 = rfft(w, fft_size2)
    F_ar = rfft(a_rev, fft_size2)
    conv = irfft(F_w2 * F_ar, fft_size2)
    
    # grad_j = 2 * conv[j + n - 1]
    grad = 2.0 * conv[n - 1: n - 1 + n]
    return grad


def mirror_descent_optimize(a0, time_limit, T_schedule=None, lr_schedule=None):
    """
    Mirror descent (multiplicative weights) for minimizing max_k conv(a,a)_k
    subject to a >= 0, sum(a) = 1.
    
    Uses softmax-approximated subgradients with adaptive temperature.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Nesterov momentum on the "virtual" iterate
    y = a.copy()
    
    n_iter = 0
    for it in range(1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Compute subgradient at y
        b_y = fast_autocorr(y)
        
        # Adaptive temperature: start high (spread out), decrease (concentrate)
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        if T_schedule is None:
            T = 50.0 * (1.0 - progress) + 5.0 * progress
        else:
            T = T_schedule(progress)
        
        # Softmax weights over peaks
        Tb = T * b_y
        max_Tb = Tb.max()
        exp_Tb = np.exp(Tb - max_Tb)
        w = exp_Tb / exp_Tb.sum()
        
        # Compute subgradient
        grad = compute_subgrad_fft(y, w)
        
        # Adaptive step size (based on gradient norm)
        if lr_schedule is None:
            # Use inverse square root schedule
            lr = 1.0 / np.sqrt(it + 1.0)
        else:
            lr = lr_schedule(it)
        
        # Scale gradient so that max |grad| ~ 1 (important for step size)
        grad_max = np.abs(grad).max()
        if grad_max > 1e-15:
            grad = grad / grad_max
        
        # Mirror descent update (multiplicative weights)
        # a_new_j ∝ y_j * exp(-lr * grad_j)
        log_a = np.log(np.maximum(y, 1e-30)) - lr * grad
        log_a -= log_a.max()  # for numerical stability
        a_new = np.exp(log_a)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-15:
            a_new = y.copy()
        else:
            a_new = a_new / a_new_sum
        
        # Nesterov update for y
        if it == 0:
            y = a_new.copy()
        else:
            momentum = (it - 1.0) / (it + 2.0)
            y_new = a_new + momentum * (a_new - y)
            y_new = np.maximum(y_new, 0)
            s2 = y_new.sum()
            if s2 < 1e-15:
                y = a_new.copy()
            else:
                y = y_new / s2
        
        a = a_new  # keep track of the "real" iterate
        
        # Check score periodically
        if it % 10 == 0:
            score = 2.0 * n * fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()
            
            # Also check y (virtual iterate can sometimes be better)
            score_y = 2.0 * n * fast_autocorr(y).max()
            if score_y < best_score:
                best_score = score_y
                best_a = y / y.sum()
        
        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"        MD it={it} ({elapsed:.1f}s): best={best_score:.6f}, T={T:.2f}")
    
    return best_a, best_score

    for si, (T_s, T_e, base_lr, frac) in enumerate(stages):

                break
            
            progress = min(1.0, elapsed_stage / max(stage_time, 1e-6))
            T = T_s + (T_e - T_s) * progress
            
            # Compute subgradient
            b_y = fast_autocorr(y)
            Tb = T * b_y
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)

            # Mirror descent update
            log_a = np.log(np.maximum(y, 1e-30)) - lr * grad

            else:
                a_new = a_new / a_new_sum
            
            # Nesterov
            if it == 0:
                y = a_new.copy()
            else:
                momentum = (it - 1.0) / (it + 2.0)
                y_new = a_new + momentum * (a_new - y)
                y_new = np.maximum(y_new, 0)
                s2 = y_new.sum()
                if s2 < 1e-15:
                    y = a_new.copy()
                else:
                    y = y_new / s2
            
            a = a_new
            
            if it % 5 == 0:
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                score_y = 2.0 * n * fast_autocorr(y).max()
                if score_y < best_score:
                    best_score = score_y
                    best_a = y / y.sum()
            
            if it % 1000 == 0:
                elapsed = time.monotonic() - t0

def spectral_shaping(a, n_iter=500, alpha=0.3):
    """Spectral shaping: flatten power spectrum to reduce autocorrelation peaks."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2

        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))

        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.05, 20.0)
        scale[1:] = np.power(ratio, alpha / 2.0)

        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a


def peak_shaving(a0, time_limit, seed=123):
    """Coordinate-descent peak shaving."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:8]

        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            src_local = np.argmax(contrib)
            src = idx[src_local]

            peak_areas = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_areas.update(range(js2, je2 + 1))

            candidates = [j for j in range(n) if j not in peak_areas]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            candidates_sorted = sorted(candidates, key=lambda j: a[j])
            dst_pool = candidates_sorted[:max(1, len(candidates_sorted) // 3)]
            dst = dst_pool[rng.integers(len(dst_pool))]

            for frac in [0.003, 0.007, 0.015, 0.03, 0.05]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            if stall > 300:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""

def nesterov_lp_optimize(a0, time_limit, solver, max_iters=100000):

    n = len(a0)

    s = a.sum()
    if s < 1e-15:

    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    avg_a = np.zeros(n)
    n_avg = 0

    def build_conv_matrix(a_fixed):
        nn = len(a_fixed)
        T = np.zeros((2 * nn - 1, nn))
        for j in range(nn):
            T[j:j + nn, j] = a_fixed
        return T

    def solve_lp_best_response(a_fixed, solver_name):
        import cvxpy as cp
        nn = len(a_fixed)
        b_var = cp.Variable(nn, nonneg=True)
        M_var = cp.Variable()
        T_mat = build_conv_matrix(a_fixed)
        constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=solver_name)
        except Exception:
            pass
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
                if alt == solver_name:
                    continue
                try:
                    prob.solve(solver=alt)
                    if prob.status in ('optimal', 'optimal_inaccurate'):
                        break
                except Exception:
                    continue
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            return None, None
        b_val = np.asarray(b_var.value, dtype=np.float64)
        b_val = np.maximum(b_val, 0)
        s_val = b_val.sum()
        if s_val < 1e-15:
            return None, None
        b_val = b_val / s_val
        return b_val, float(M_var.value)

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s2 = y_new.sum()
        if s2 < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s2

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s3 = a_new.sum()
            if s3 < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s3

        y = y_new
        a = a_new

        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y / y.sum()

        avg_a += a
        n_avg += 1
        if n_avg >= 10 and n_avg % 20 == 0:
            avg_check = avg_a / n_avg
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()

        if t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"        NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}, LP_M={M_val:.8f}")

    return best_a, best_score
