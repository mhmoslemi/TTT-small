"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using FFT-based accelerated gradient descent
with adaptive LogSumExp smoothing, combined with multi-length exploration and LP polishing.
Key improvements over previous approach: O(n log n) per gradient iteration (vs O(n^2) LP),
allowing 50-200x more optimization steps. Uses Nesterov acceleration in log-space,
adaptive temperature annealing, random restarts, and a short LP polishing phase.

def fft_convolve_self(a):
    """Compute linear self-convolution a * a via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fft_convolve_cross(a, b):
    """Compute linear convolution of a (len na) and b (len nb) via FFT."""
    na = len(a)
    nb = len(b)
    size = na + nb - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def stable_softmax(x, T=None):
    """Stable softmax. If T is given, compute softmax(T*x)."""
    if T is not None:
        x = T * x
    x = x - x.max()
    # Clamp for numerical stability
    x = np.clip(x, -700, 700)
    e = np.exp(x)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(x)) / len(x)
    return e / s


def compute_grad_a(a, w):
    """Compute gradient of LogSumExp objective w.r.t. a.
    grad_a[i] = 2 * sum_j a[j] * w[i+j]
    where w has length 2n-1 and a has length n.
    """
    n = len(a)
    # Use cross-correlation via FFT
    w_rev = w[::-1]
    conv = fft_convolve_cross(a, w_rev)
    # grad[i] = 2 * conv[n-1 : 2n-1][::-1]
    grad = 2.0 * conv[n - 1:2 * n - 1][::-1]
    return grad


def score_normalized(a):
    """Score assuming sum(a)=1: returns 2*n*max(conv(a,a))."""
    n = len(a)
    b = fft_convolve_self(a)
    return 2.0 * n * b.max()

def project_simplex(v):
    """Project onto probability simplex (sum=1, non-negative)."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def nesterov_fft_optimize(a0, time_limit, T_start=50.0, T_end=1.0,
                          n_anneal=50, eta0=0.5, rng=None):
    """Nesterov-accelerated gradient descent in log-space with adaptive LogSumExp.
    
    Works in the log-space z = log(a) to naturally handle the simplex constraint.
    Uses Nesterov momentum and backtracking line search.
    """
    if rng is None:
        rng = np.random.default_rng(0)
    
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 1e-15)
    a = a / a.sum()
    
    # Work in log-space
    z = np.log(a)
    z_prev = z.copy()
    
    best_a = a.copy()
    best_score = score_normalized(a)
    t_start = time.monotonic()
    
    # Nesterov parameters
    t_k = 1.0
    
    for it in range(1, 10000000):
        if time.monotonic() - t_start > time_limit:
            break
        
        # Adaptive temperature: anneal from T_start to T_end over n_anneal "phases"
        # Use a smooth schedule based on iteration
        progress = min(1.0, it / (n_anneal * 100))
        T = T_start * (T_end / T_start) ** progress
        
        # Nesterov extrapolation in log-space
        t_k_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        alpha = (1.0 - t_k) / t_k_new
        momentum = (t_k - 1.0) / t_k_new
        
        z_curr = z + alpha * (z - z_prev)
        a_curr = np.exp(z_curr - z_curr.max())
        a_curr = a_curr / a_curr.sum()
        a_curr = np.maximum(a_curr, 1e-15)
        a_curr = a_curr / a_curr.sum()
        
        # Compute gradient at extrapolated point
        b = fft_convolve_self(a_curr)
        w = stable_softmax(b, T)
        g_a = compute_grad_a(a_curr, w)
        
        # Project to tangent space of simplex (for log-parameterization)
        g_z = a_curr * (g_a - np.dot(g_a, a_curr))
        
        # Gradient step with backtracking line search
        grad_norm = np.linalg.norm(g_z)
        if grad_norm < 1e-18:
            # Converged
            break
        
        # Initial step size
        eta = eta0 / np.sqrt(it + 1)
        
        # Backtracking line search (max 10 iterations)
        obj_curr = T + np.log(w.max() * len(w))  # approximate current obj
        accepted = False
        for ls in range(12):
            z_trial = z - eta * g_z
            a_trial = np.exp(z_trial - z_trial.max())
            a_trial = np.maximum(a_trial, 1e-15)
            a_trial = a_trial / a_trial.sum()
            
            # Check if score improved
            sc_trial = score_normalized(a_trial)
            sc_curr = score_normalized(a)
            
            if sc_trial < sc_curr * 1.001 + 1e-12:
                accepted = True
                break
            eta *= 0.5
        
        if not accepted:
            # Very small step
            eta = 1e-8
            z_trial = z - eta * g_z
            a_trial = np.exp(z_trial - z_trial.max())
            a_trial = np.maximum(a_trial, 1e-15)
            a_trial = a_trial / a_trial.sum()
        
        z_prev = z.copy()
        z = z_trial
        a = a_trial
        
        sc = score_normalized(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        # Occasional random perturbation to escape local minima
        if it % 200 == 0 and it > 100:
            perturb_scale = 0.02 / np.sqrt(it / 100.0)
            z_pert = z + perturb_scale * rng.normal(0, 1, n)
            a_pert = np.exp(z_pert - z_pert.max())
            a_pert = np.maximum(a_pert, 1e-15)
            a_pert = a_pert / a_pert.sum()
            sc_pert = score_normalized(a_pert)
            if sc_pert < best_score:
                best_score = sc_pert
                best_a = a_pert.copy()
                z = z_pert.copy()
                a = a_pert.copy()
        
        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    FFT-Nesterov it={it+1} ({elapsed:.0f}s): score={best_score:.8f} T={T:.2f}")
    
    return best_a, best_score

def build_conv_matrix(a_fixed):
    """Build the convolution matrix T where (T@b)[k] = (a_fixed*b)[k]."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver=None):
    """Solve the LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0."""
    if solver is None:
        solver = cp.ECOS
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS, cp.GLPK, cp.CBC]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    b_val = b_val / s
    return b_val

def structured_init(n, rng):

    choice = rng.integers(0, 10)

    elif choice == 1:
        a = np.sin(np.pi * x) ** 4

    elif choice == 3:
        # Bessel-like
        a = np.exp(-2 * (x - 0.5) ** 2) * (1 + 0.2 * np.cos(6 * np.pi * x))
    elif choice == 4:
        # Double hump
        a = np.exp(-8 * (x - 0.3) ** 2) + np.exp(-8 * (x - 0.7) ** 2)
    elif choice == 5:
        # Chirp-like amplitude
        a = np.sin(np.pi * x) * (0.5 + 0.5 * np.cos(2 * np.pi * 2 * x))
    elif choice == 6:
        a = np.sin(np.pi * x) ** 2 * (1 + 0.5 * np.sin(8 * np.pi * x))
    elif choice == 7:
        # Flat top with tapered edges
        a = np.clip((np.sin(np.pi * x) - 0.3) / 0.7, 0, 1) ** 2
    elif choice == 8:

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    # Get parent sequence
    best_seq = None
    best_score = float('inf')
    n_base = 1200
    
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            n_base = len(best_seq)
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.8f}")
    except Exception:
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0).astype(np.float64)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            remain = global_deadline - time.monotonic()
            print(f"  ** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, remaining={remain:.0f}s)")
    
    # === Phase 1: Multi-temperature FFT optimization at base n (40% time) ===
    print(f"\n=== Phase 1: Multi-T FFT at n={n_base} (score={best_score:.8f}) ===")
    t1 = 0.40 * (budget_s - 30)
    a_out, _ = multi_temperature_optimize(best_seq, t1, rng)
    try_update(a_out, "multiT_main")
    print(f"  Phase 1 done: best_score={best_score:.8f}")
    
    # === Phase 2: Nesterov FFT with line search (20% time) ===
    print(f"\n=== Phase 2: FFT Nesterov w/ line search (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t2 = min(0.20 * (budget_s - 30), remaining * 0.5)
    if t2 > 30:
        a_out2, _ = nesterov_fft_optimize(best_seq, t2, T_start=20.0, T_end=0.3,
                                          n_anneal=40, eta0=0.8, rng=rng)
        try_update(a_out2, "nesterov_fs")
    print(f"  Phase 2 done: best_score={best_score:.8f}")
    
    # === Phase 3: Multi-length exploration (20% time) ===
    print(f"\n=== Phase 3: Multi-length exploration (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t3_total = min(0.20 * (budget_s - 30), remaining * 0.6)
    if t3_total > 30:
        # Try several lengths
        n_candidates = [n_base, n_base + 200, n_base + 500, n_base + 1000,
                        n_base - 200, n_base + 2000]
        # Remove duplicates and sort by proximity to n_base
        n_candidates = sorted(set(n_candidates), key=lambda x: abs(x - n_base))
        
        time_per_n = t3_total / len(n_candidates)
        for n_try in n_candidates:
            remaining = global_deadline - time.monotonic()
            if remaining < 15:
                break
            t_n = min(time_per_n, remaining * 0.45)
            if t_n < 10 or n_try == n_base:
                continue
            if n_try < 100 or n_try > 10000:
                continue
            
            print(f"  Trying n={n_try}, budget={t_n:.0f}s")
            init_try = interpolate_sequence(best_seq, n_try)
            
            # Quick multi-T optimization
            a_try, _ = multi_temperature_optimize(init_try, t_n * 0.7, rng)
            try_update(a_try, f"len_{n_try}")
            
            remaining = global_deadline - time.monotonic()
            if remaining > 10:
                a_try2, _ = nesterov_fft_fast(a_try, remaining * 0.4, T=2.0, eta0=0.5, rng=rng)
                try_update(a_try2, f"len_{n_try}_fast")
    print(f"  Phase 3 done: best_score={best_score:.8f}")
    
    # === Phase 4: Random restarts (10% time) ===
    print(f"\n=== Phase 4: Random restarts (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t4 = min(0.10 * (budget_s - 30), remaining * 0.5)
    if t4 > 20:
        n_restarts = 5
        t_per_restart = t4 / n_restarts
        for r in range(n_restarts):
            remaining = global_deadline - time.monotonic()
            if remaining < 10:
                break
            # Random structured init at best n
            n_cur = len(best_seq)
            init_r = structured_init(n_cur, rng)
            # Blend with current best
            blend = 0.5 * init_r + 0.5 * best_seq
            blend = np.maximum(blend, 0)
            blend = blend / blend.sum()
            
            a_r, _ = nesterov_fft_fast(blend, t_per_restart * 0.8, T=10.0, eta0=1.0, rng=rng)
            try_update(a_r, f"restart_{r}")
            a_r2, _ = nesterov_fft_fast(a_r, t_per_restart * 0.3, T=2.0, eta0=0.5, rng=rng)
            try_update(a_r2, f"restart_{r}_T2")
    print(f"  Phase 4 done: best_score={best_score:.8f}")
    
    # === Phase 5: LP polishing (7% time) ===
    print(f"\n=== Phase 5: LP polishing (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    t5 = min(0.07 * (budget_s - 30), remaining * 0.5)
    if t5 > 20:
        a_lp, _ = nesterov_lp_short(best_seq, t5, solver=cp.ECOS)
        try_update(a_lp, "lp_polish")
    print(f"  Phase 5 done: best_score={best_score:.8f}")
    
    # === Phase 6: Final FFT polish (remaining time) ===
    print(f"\n=== Phase 6: Final FFT polish (score={best_score:.8f}) ===")
    remaining = global_deadline - time.monotonic()
    if remaining > 10:
        # Low temperature fine-tuning
        a_fin, _ = nesterov_fft_optimize(best_seq, remaining * 0.6,
                                         T_start=1.0, T_end=0.1,
                                         n_anneal=20, eta0=0.2, rng=rng)
        try_update(a_fin, "final_fft")
        
        remaining = global_deadline - time.monotonic()
        if remaining > 5:
            a_fin2, _ = nesterov_fft_fast(best_seq, remaining * 0.7, T=0.5, eta0=0.3, rng=rng)
            try_update(a_fin2, "final_fast")
    print(f"  Phase 6 done: best_score={best_score:.8f}")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
    
    elapsed = time.monotonic() - t_global_start
    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)}, time={elapsed:.0f}s)")
    return [float(x) for x in best_seq]
