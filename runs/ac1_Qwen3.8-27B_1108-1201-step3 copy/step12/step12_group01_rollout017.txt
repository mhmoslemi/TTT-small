"""

2. Top-k LogSumExp gradient descent (Adam) for final polish

import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def compute_topk_grad(a, T, topk):
    """Compute top-k LSE objective and gradient w.r.t. a."""
    n = len(a)
    c = fast_autocorr(a)
    k_use = min(topk, len(c))
    top_idx = np.argsort(c)[::-1][:k_use]
    top_vals = c[top_idx]

    Tc = T * top_vals
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    lse_val = (max_Tc + np.log(exp_Tc.sum())) / T
    w = exp_Tc / exp_Tc.sum()

    grad_a = np.zeros(n)
    for idx in range(k_use):
        K = top_idx[idx]
        wK = w[idx]
        js = max(0, K - (n - 1))
        je = min(n - 1, K)
        if js <= je:
            idx_range = np.arange(js, je + 1)
            grad_a[idx_range] += 2.0 * wK * a[K - idx_range]

    return lse_val, grad_a


def topk_lse_grad_z(z, T, topk):
    """Objective and gradient in z-space for top-k LSE."""
    a = softmax_fn(z)
    n = len(a)
    obj, grad_a = compute_topk_grad(a, T, topk)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL, cp.MOSEK, cp.GLPK]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where T@b = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. conv(a_fixed, b) <= M, sum(b)=1, b>=0. Returns (b, M, dual_y)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    dual_y = None
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL, cp.MOSEK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None, None
    b_val = b_val / s

    # Try to extract dual variables from the inequality constraints
    try:
        # The inequality constraints T @ b_var <= M_var have dual values
        # In cvxpy, we can access them via the constraint objects
        # But this is solver-dependent, so we'll compute them heuristically
        dvals = None
        for c in constraints:
            if hasattr(c, 'dual_value') and c.dual_value is not None:
                dv = np.asarray(c.dual_value, dtype=np.float64)
                if dv.shape[0] == 2 * n - 1:
                    dvals = np.maximum(dv, 0)
                    break
        if dvals is not None and dvals.sum() > 1e-15:
            dual_y = dvals / dvals.sum()
    except Exception:
        pass

    return b_val, float(M_var.value), dual_y

def adam_topk_lse(a0, time_limit, T_start=50.0, T_end=0.1, n_stages=30, topk=20):

        a_init = np.ones(n) / n
    else:
        a_init = a_init / s

    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        base_lr = 3e-3 * min(1.0, 100.0 / n)
        lr = base_lr * (0.85 ** si)

        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.8, 0.999, 1e-8

        max_it = max(20, int(min(remaining, 60.0) * 5))

        for it in range(max_it):

                break

            obj, grad_z = topk_lse_grad_z(z, stage_T, topk)

            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            if (it + 1) % 5 == 0:
                a = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    if (it + 1) % 30 == 0:
                        z = np.log(np.maximum(best_a, 1e-30))

        a_stage = softmax_fn(z)
        score_stage = 2.0 * n * fast_autocorr(a_stage).max()
        if score_stage < best_score:
            best_score = score_stage
            best_a = a_stage.copy()

        if si % 8 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE stage {si}/{n_stages} (T={stage_T:.2f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def peak_shaving_focused(a0, time_limit, seed=123):
    """
    Focused peak shaving: iteratively reduce the single highest peak
    by finding the best source-destination pair that reduces it most.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit - 2:
        b = fast_autocorr(a)
        bmax = b.max()
        k_max = b.argmax()

        # Find the top contributing pair to the peak
        js = max(0, k_max - n + 1)
        je = min(n - 1, k_max)
        idx = np.arange(js, je + 1)
        if len(idx) == 0:
            break
        contrib = a[idx] * a[k_max - idx]
        max_contrib_idx = idx[np.argmax(contrib)]

        # Try multiple reduction strategies
        improved = False

        # Strategy 1: Move mass from max contributor to a "safe" location
        for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
            amt = a[max_contrib_idx] * frac
            if amt < 1e-12:
                continue
            # Find safe destination: minimize impact on all top peaks
            top_peaks = np.argsort(b)[::-1][:30]
            danger_score = np.zeros(n)
            for k2 in top_peaks:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                danger_score[js2:je2+1] += b[k2] * a[js2:je2+1]

            # Prefer low danger_score and low a[j]
            candidates = np.where((np.arange(n) != max_contrib_idx) & (a > 0.001))[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != max_contrib_idx)[0]
            scores = danger_score[candidates] + 0.1 * a[candidates]
            best_dst = candidates[np.argmin(scores)]

            a_trial = a.copy()
            a_trial[max_contrib_idx] -= amt
            a_trial[best_dst] += amt
            a_trial = a_trial / a_trial.sum()
            score = 2.0 * n * fast_autocorr(a_trial).max()
            if score < best_score - 1e-14:
                a = a_trial
                best_score = score
                best_a = a.copy()
                stall = 0
                improved = True
                break

        # Strategy 2: Split mass from top-2 contributors
        if not improved and len(idx) >= 2:
            top2 = idx[np.argsort(contrib)[::-1]][:2]
            for src in top2:
                for frac in [0.003, 0.008, 0.015]:
                    amt = a[src] * frac
                    if amt < 1e-12:
                        continue
                    dest_candidates = np.where(
                        (np.arange(n) != src) &
                        (np.abs(np.arange(n) - src) > n // 4)
                    )[0]
                    if len(dest_candidates) == 0:
                        dest_candidates = np.where(np.arange(n) != src)[0]
                    # Pick the one with least contribution to the peak
                    dist_to_peak = np.abs(dest_candidates - (k_max - src))
                    best_dst = dest_candidates[np.argmin(dist_to_peak)]

                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[best_dst] += amt
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
                        improved = True
                        break
                if improved:
                    break

        if not improved:
            stall += 1
            # Random perturbation to escape
            n_pert = rng.integers(2, 5)
            a_trial = a.copy()
            for _ in range(n_pert):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a_trial[i] > 1e-8:
                    frac = rng.uniform(0.001, 0.01)
                    amt = a_trial[i] * frac
                    a_trial[i] -= amt
                    a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 1e-10:
                a_trial = a_trial / st
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0

            if stall > 200:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 4)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.6 + 0.25 * np.cos(2 * x) + 0.1 * np.cos(5 * x)
    elif choice == 4:
        x = np.linspace(0, 1, n)
        a = np.exp(-8 * (x - 0.3) ** 2) + 0.6 * np.exp(-8 * (x - 0.7) ** 2)
    else:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) * (0.5 + 0.5 * np.cos(3 * np.pi * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, time_budget=1800.0):
    """
    Multi-phase optimization:
    1. LP Nesterov at base length (primary workhorse, ~60% of time)
    2. LSE gradient descent polish
    3. Multi-length exploration
    4. Final peak shaving
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_budget = min(float(time_budget) - 60.0, 1740.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Time budget: {time_budget:.0f}s")

    # Get parent sequence
    best_seq = None
    best_score = float('inf')
    parent = None

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    n_base = len(best_seq) if best_seq is not None else 1600

    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    # === Phase 1: LP Nesterov at base length (primary) ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} ({remaining:.0f}s left) ===")
    t1 = min(remaining * 0.55, 950.0)
    if t1 > 60:
        a_out, _ = nesterov_lp_optimize(best_seq, t1, solver)
        try_update(a_out, f"lp_n{n_base}")

    # === Phase 2: LSE gradient descent polish ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 2: LSE polish n={n_base} ({remaining:.0f}s left) ===")
    t2 = min(remaining * 0.15, 250.0)
    if t2 > 20:
        a_out, _ = adam_topk_lse(best_seq, t2, T_start=40.0, T_end=0.05, n_stages=25, topk=25)
        try_update(a_out, f"lse_n{n_base}")

    # === Phase 3: Try longer length ===
    n_long = n_base + 200
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 3: LP Nesterov n={n_long} ({remaining:.0f}s left) ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 40:
        init_long = interpolate_sequence(best_seq, n_long)
        a_out, _ = nesterov_lp_optimize(init_long, t3, solver)
        try_update(a_out, f"lp_n{n_long}")

    # === Phase 4: LSE at longer length ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 30:
        t4 = min(remaining * 0.12, 150.0)
        if t4 > 15:
            print(f"=== Phase 4: LSE n={n_long} ({remaining:.0f}s left) ===")
            a_out, _ = adam_topk_lse(best_seq, t4, T_start=20.0, T_end=0.1, n_stages=15, topk=20)
            try_update(a_out, f"lse_nlong_{n_long}")

    # === Phase 5: Focused peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"=== Phase 5: Peak shaving ({remaining:.0f}s left) ===")
    t5 = min(remaining * 0.15, 150.0)
    if t5 > 10:
        a_out, _ = peak_shaving_focused(best_seq, t5, seed=789)
        try_update(a_out, "peak_shave")

    # === Phase 6: Final LSE with very low temperature ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 15:
        t6 = min(remaining * 0.20, 120.0)
        if t6 > 10:
            print(f"=== Phase 6: Final LSE low-T ({remaining:.0f}s left) ===")
            a_out, _ = adam_topk_lse(best_seq, t6, T_start=5.0, T_end=0.02, n_stages=15, topk=15)
            try_update(a_out, "final_lse_lowT")

    # === Phase 7: Very short peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 8:
        t7 = min(remaining * 0.3, 60.0)
        if t7 > 5:
            print(f"=== Phase 7: Final peak shave ({remaining:.0f}s left) ===")
            a_out, _ = peak_shaving_focused(best_seq, t7, seed=101)
            try_update(a_out, "final_shave")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
