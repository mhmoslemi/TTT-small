def fast_autocorr(a):
    """Full linear convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(arr):

def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def project_simplex(v):
    """Project vector v onto the probability simplex {x: sum x = 1, x >= 0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.argmax(1 - u / np.arange(1, n + 1) > 0)
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def l2_normalize(a):
    """Normalize a to sum to 1, non-negative."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s


def lp_norm_grad_descent(a0, time_limit, p_start=8, p_end=128, n_stages=8, lr=0.01, momentum=0.9):

def lse_optimize(a0, time_limit, T_start=50.0, T_end=0.05, n_stages=12, lr=0.005):
    """LogSumExp-smoothed optimization with Adam optimizer."""
    n = len(a0)
    a_init = l2_normalize(a0)
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_adam = 0

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        iters = max(20, min(120, int(remaining * 60 / (n_stages - si))))
        fft_size = 1 << (3 * n - 1).bit_length()

        for it in range(iters):
            if time.monotonic() - t0 > time_limit:
                break

            a = softmax_fn(z)
            b = fast_autocorr(a)
            Tb = stage_T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()

            # Gradient: 2 * (a * w) centered
            Wr = rfft(w, fft_size)
            Arev = rfft(a[::-1].copy(), fft_size)
            conv = irfft(Arev * Wr, fft_size)
            grad_a = 2.0 * conv[n - 1:n - 1 + n]

            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)

            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)

            lr_t = lr / np.sqrt(1 + 0.01 * it)
            z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)

            if it % 15 == 0:
                a = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()

        a = softmax_fn(z)
        score = 2.0 * n * fast_autocorr(a).max()
        if score < best_score:
            best_score = score

        if si % 4 == 0:

            print(f"      LSE stage {si}/{n_stages} (T={stage_T:.2f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def get_lp_solver():
    """Find a working LP solver, prefer fast ones."""
    for solver_name in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_fast(a_fixed, solver, previous_b=None):
    """
    Solve LP: min M s.t. conv(a_fixed, b)_k <= M, sum(b)=1, b>=0.
    Uses the convolution structure implicitly via cvxpy.
    """
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    # Build constraints using the convolution structure
    # conv(a, b)_k = sum_j a_j * b_{k-j} for valid j
    # We express this as: for each k, sum_j a_{k-j} * b_j <= M
    # This is a_matrix @ b <= M where a_matrix[k,j] = a_{k-j} if valid

    # For efficiency, build the matrix
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed

    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    try:
        prob.solve(solver=solver)
    except Exception:
        pass

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, max_iters=50000):
    """
    Nesterov-accelerated LP best-response iteration.
    At each step: solve LP to get b*, then update a using Nesterov extrapolation.
    """
    n = len(a0)
    a = l2_normalize(a0)
    y = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    avg_a = np.zeros(n)
    n_avg = 0

    # Pre-build T matrix once (it changes when a changes, so we rebuild each iter)
    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        b_star, M_val = solve_lp_fast(y, solver)
        if b_star is None:
            continue

        alpha_t = 2.0 / (t + 2.0)

        # Update y (the "center" of Nesterov)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = l2_normalize(y_new)

        # Nesterov extrapolation for a
        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = l2_normalize(a_new)

        y = y_new
        a = a_new

        # Check best at a, y, and running average
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        avg_a += a
        n_avg += 1
        if n_avg >= 5 and n_avg % 10 == 0:
            avg_check = avg_a / n_avg
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()

        if t % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.8f}")

    return best_a, best_score


def spectral_shaping(a, n_iter=500, alpha=0.3):
    """Spectral shaping: flatten power spectrum to reduce convolution peaks."""
    n = len(a)
    a = l2_normalize(a)
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.05, 20.0)
        scale = np.ones(len(A))
        scale[1:] = np.power(ratio, alpha / 2.0)
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = l2_normalize(a_new)
        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a


def peak_shaving(a0, time_limit, seed=123):
    """Coordinate-descent peak shaving."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = l2_normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:6]
        improved = False

        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            src_local = np.argmax(contrib)
            src = idx[src_local]

            peak_areas = set()
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_areas.update(range(js2, je2 + 1))

            candidates = [j for j in range(n) if j not in peak_areas]
            if not candidates:
                candidates = [j for j in range(n) if j != src]
            candidates_sorted = sorted(candidates, key=lambda j: a[j])
            dst_pool = candidates_sorted[:max(1, len(candidates_sorted) // 3)]
            if not dst_pool:
                continue
            dst = dst_pool[rng.integers(len(dst_pool))]

            for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < 0:
                    continue
                a_trial = l2_normalize(a_trial)
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    break
            if improved:
                break

        if not improved:
            stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.003, 0.02)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = l2_normalize(a_trial)
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            if stall > 200:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f = l2_normalize(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def dual_lp_solve(a_fixed, solver, n_dual_iters=50):
    """
    Approximate LP best response using dual gradient method.
    Dual: max nu s.t. T^T lambda >= nu*1, sum lambda = 1, lambda >= 0
    Equivalent: max min_j (T^T lambda)_j s.t. lambda in simplex
    Use smoothed version with exponentiated gradient.
    """
    n = len(a_fixed)
    m = 2 * n - 1  # number of convolution positions
    fft_size_dual = 1 << (m + n - 1).bit_length()

    # Initialize lambda uniformly
    lam = np.ones(m) / m
    best_b = None
    best_M = float('inf')

    for it in range(n_dual_iters):
        # Compute T^T lambda: (T^T lam)_j = sum_k T[k,j] * lam[k] = sum_k a[k-j] * lam[k]
        # = sum_{k=j}^{j+n-1} a[k-j] * lam[k]
        # This is cross-correlation of a and lam
        Af = rfft(a_fixed, fft_size_dual)
        Lf = rfft(lam, fft_size_dual)
        # (T^T lam)_j = sum_m a[m] * lam[m+j] (where m+j < m_total)
        # = ifft(conj(Af) * Lf) at positions n-1 .. 2n-2
        cc = irfft(np.conj(Af) * Lf, fft_size_dual)
        TL = cc[n - 1:2 * n - 1]

        # Find the minimum and update lambda (exponentiated gradient for the dual)
        min_val = TL.min()
        max_TL = TL.max()
        # Softmax weights concentrating on minimum
        w = np.exp(-50.0 * (TL - min_val))
        w = w / w.sum()

        # Update lambda toward w (mirror descent on simplex)
        eta = 1.0 / (it + 1)
        lam = (1 - eta) * lam + eta * w
        lam = np.maximum(lam, 0)
        lam = lam / lam.sum()

        # Extract b from the support of TL (where TL is close to max)
        threshold = min_val + 0.5 * (max_TL - min_val)
        support = np.where(TL >= threshold)[0]
        if len(support) == 0:
            support = np.where(TL >= max(TL) - 1e-6)[0]
        if len(support) == 0:
            support = np.array([np.argmax(TL)])

        b_trial = np.zeros(n)
        b_trial[support] = 1.0 / len(support)

        # Evaluate
        conv_val = fast_autocorr(b_trial)  # Wrong - we need conv(a_fixed, b_trial)
        # Actually compute conv(a_fixed, b_trial)
        Ab = rfft(a_fixed, 1 << (2 * n - 1).bit_length())
        Bb = rfft(b_trial, 1 << (2 * n - 1).bit_length())
        conv_ab = irfft(Ab * Bb, 1 << (2 * n - 1).bit_length())[:2 * n - 1]
        M_trial = conv_ab.max()

        if M_trial < best_M:
            best_M = M_trial
            best_b = b_trial.copy()

    return best_b, best_M
