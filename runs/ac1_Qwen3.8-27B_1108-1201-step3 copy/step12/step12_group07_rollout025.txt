    size = 2 * n - 1

    A = rfft(a, fft_size)

    a_sym = np.maximum(a_sym, 0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym = a_sym / s
    return a_sym


def softmax_from_z(z):
    """Numerically stable softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Log of sequence (for z-space parameterization)."""
    return np.log(np.maximum(a, 1e-300))


def get_lp_solver():
    """Find the best available LP solver."""

def solve_lp_best_response(a_fixed, solver=None):
    """
    Solve the LP: min M s.t. conv(a_fixed, b)_k <= M for all k, b >= 0, sum(b) = 1.
    Returns (b_optimal, M_value).
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    size = 2 * n - 1
    
    # Build convolution matrix T where T[k, j] * b[j] contributes to conv[k]
    # conv[k] = sum_j a_fixed[min(k, n-1) - max(0, k-n+1+j) + ... ]
    # Actually: (a*b)_k = sum_j a[k-j] * b[j] where valid indices
    # T[k, j] = a[k-j] if 0 <= k-j <= n-1, else 0
    
    # For efficiency, build T as a banded matrix conceptually
    # But for cvxpy, just build it directly
    T = np.zeros((size, n))
    for j in range(n):
        start = j
        end = start + n
        T[start:end, j] = a_fixed
    
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        # Try alternative solvers
        for alt in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_symmetric(a0, time_limit, solver=None, max_iters=500000):
    """
    Nesterov LP with symmetry enforcement.
    """
    n = len(a0)
    a = enforce_symmetry(np.asarray(a0, dtype=np.float64))
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    a_prev = a.copy()
    restart_count = 0
    last_improve_iter = 0
    iteration = 0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if it % 2000 == 0:
            remaining = time_limit - (time.monotonic() - t0)
            if remaining < 2:
                break
        
        # Nesterov extrapolation
        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            y = a + momentum * (a - a_prev)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:
                y = a.copy()
            else:
                y = y / sy
            y = enforce_symmetry(y)
        else:
            y = a
        
        # Solve LP for best response
        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        
        # Enforce symmetry on the LP solution
        b_new_sym = enforce_symmetry(b_new)
        
        # Nesterov update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        alpha = 2.0 / (t_new + 1.0)
        
        a_new = (1.0 - alpha) * a + alpha * b_new_sym
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new_sym.copy()
        else:
            a_new = a_new / sa
        a_new = enforce_symmetry(a_new)
        
        sc_new = score_normalized(a_new)
        
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            last_improve_iter = it
        
        # Restart logic
        if sc_new > best_score * 1.001 and t_nest > 10.0:
            t_nest = 1.0
            restart_count += 1
        
        a_prev = a
        a = a_new
        t_nest = t_new
        iteration = it + 1
        
        if (it + 1) % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP-Sym iter={it+1} ({elapsed:.0f}s): score={best_score:.8f} "
                  f"t={t_nest:.1f} restarts={restart_count}")
    
    return best_a, best_score


def subgradient_nesterov_symmetric(a0, time_limit, max_steps=200000):
    """Fast subgradient Nesterov with symmetry."""
    n = len(a0)
    a = enforce_symmetry(np.asarray(a0, dtype=np.float64))
    best_a = a.copy()
    best_score = score_normalized(a)
    z = z_from_a(a).copy()
    a_prev = a.copy()
    z_prev = z.copy()
    t0 = time.monotonic()
    t_nest = 1.0
    eta = 0.3
    last_improve = 0
    
    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step - last_improve > 3000:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        # Find peaks above threshold
        threshold = max_b * 0.95
        topk = np.where(b >= threshold)[0]
        if len(topk) == 0:
            topk = np.array([int(np.argmax(b))])
        
        # Gradient: for each peak lag k, the gradient w.r.t. a_i is 2*a[k-i] (for the i that makes k-i valid)
        grad_a = np.zeros(n)
        for k in topk:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * a[k - idx]
        grad_a /= len(topk)
        
        # Gradient in z-space
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)
        
        # Nesterov in z-space
        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            z_extrap = z + momentum * (z - z_prev)
        else:
            z_extrap = z
        
        z_new = z_extrap - eta * grad_z
        a_new = softmax_from_z(z_new)
        a_new = enforce_symmetry(a_new)
        
        sc = score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            z_prev = z
            a_prev = a
            a = a_new
            z = z_new
            t_nest = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
            last_improve = step
        else:
            if sc > best_score * 1.005 and t_nest > 3.0:
                t_nest = 1.0
                eta *= 0.9
            else:
                z_prev = z
                a_prev = a
                z = z_new
                a = a_new
                t_nest = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
            eta = max(eta * 0.9995, 0.005)
    
    return best_a, best_score


def peak_targeted_refine_symmetric(a0, time_limit, max_steps=50000):
    """Multiplicative weights update targeting top peaks, with symmetry."""
    n = len(a0)
    a = enforce_symmetry(np.asarray(a0, dtype=np.float64))
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = 0.03
    last_improve_step = 0
    
    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step - last_improve_step > 3000:
            break
        
        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.90
        
        topk = np.where(b >= threshold)[0]
        if len(topk) == 0:
            topk = np.array([int(np.argmax(b))])
        
        # Compute gradient toward peaks
        grad = np.zeros(n)
        for k in topk:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += a[k - idx]
        grad /= len(topk)
        
        # Multiplicative update
        exponent = -eta * grad
        exponent -= exponent.max()
        new_a = a * np.exp(exponent)
        new_a = enforce_symmetry(new_a)
        
        sc = score_normalized(new_a)
        if sc < best_score:
            best_score = sc
            best_a = new_a.copy()
            a = new_a
            last_improve_step = step
            eta = min(eta * 1.001, 0.1)
        else:
            eta *= 0.9
            if eta < 1e-10:
                break
    
    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    """LogSumExp objective and gradient."""
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp
    
    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-20:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbgfs_logsumexp(a0, time_limit, T_start=100.0, T_end=0.02, n_stages=25):
    """L-BFGS on LogSumExp smoothed objective."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break
        
        def func_staged(z_val):
            obj, _ = lse_objective_grad(z_val, stage_T, n)
            return obj
        
        def jac_staged(z_val):
            _, grad = lse_objective_grad(z_val, stage_T, n)
            return grad
        
        iters = 50 if si < n_stages // 3 else 30
        try:
            res = minimize(
                fun=func_staged, x0=z, jac=jac_staged, method='L-BFGS-B',
                options={'maxiter': iters, 'ftol': 1e-25,
                         'gtol': 1e-20, 'maxfun': iters * 500,
                         'disp': False}
            )
            z_new = res.x
            a_new = softmax_from_z(z_new)
            a_new = enforce_symmetry(a_new)
            sc = score_normalized(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                z = z_new
            else:
                z = z_new
        except Exception:
            pass
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init_symmetric(n, rng):
    """Generate a symmetric structured initialization."""
    half = n // 2 + 1
    x = np.linspace(0, 1, half)
    choice = rng.integers(0, 6)
    if choice == 0:
        a_half = 1.0 + 0.5 * np.cos(np.pi * x)
    elif choice == 1:
        a_half = np.exp(-2.0 * (x - 0.5) ** 2) + 0.3
    elif choice == 2:
        a_half = 0.5 + 0.4 * np.cos(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x)
        a_half = np.maximum(a_half, 0.05)
    elif choice == 3:
        a_half = 1.0 + 0.3 * (1 - 2*x) * np.cos(np.pi * x)
        a_half = np.maximum(a_half, 0.05)
    elif choice == 4:
        a_half = np.exp(-3.0 * x) + 0.5
    else:
        a_half = 1.0 + 0.2 * np.sin(3 * np.pi * x)
    
    # Mirror to make full symmetric sequence
    if n % 2 == 0:
        a = np.concatenate([a_half, a_half[-2::-1]])
    else:
        a = np.concatenate([a_half, a_half[-1::-1]])
    
    a = np.maximum(a, 0.01)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()

def enforce_symmetry(a):
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym = a_sym / s
    return a_sym


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))

def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    size = 2 * n - 1

    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed

    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    try:
        kwargs = {}
        if solver == cp.CLARABEL:
            kwargs = {'eps_abs': 1e-10, 'eps_rel': 1e-10, 'max_iter': 100000}
        elif solver == cp.PDLP:
            kwargs = {}
        prob.solve(solver=solver, **kwargs)
    except Exception:
        try:
            prob.solve(solver=solver)
        except Exception:
            pass

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.CLARABEL, cp.PDLP, cp.ECOS, cp.SCS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def subgradient_nesterov(a0, time_limit, symmetric=False, max_steps=200000):

    best_a = a.copy()

    t_nest = 1.0
    eta = 0.2
    last_improve = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:

        if step - last_improve > 3000:
            break

        b = fast_convolve(a)
        max_b = b.max()

        topk = np.where(b >= threshold)[0]
        if len(topk) == 0:
            topk = np.array([int(np.argmax(b))])

        grad_a = np.zeros(n)
        for k in topk:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * a[k - idx]
        grad_a /= len(topk)

            z_extrap = z

        a_new = softmax_from_z(z_new)
        if symmetric:
            a_new = enforce_symmetry(a_new)

        sc = score_normalized(a_new)

            best_score = sc
            best_a = a_new.copy()
            z_prev = z
            a = a_new
            z = z_new

def lse_objective(z_in, T_val, n):
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-20:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lse_fun_wrapper(z_val, T_val, n):

def lse_jac_wrapper(z_val, T_val, n):

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init_symmetric(n, rng):
    half = n // 2 + 1
    x = np.linspace(0, 1, half)
    choice = int(rng.integers(0, 5))
    if choice == 0:
        a_half = 1.0 + 0.4 * np.cos(np.pi * x)
    elif choice == 1:
        a_half = np.exp(-2.0 * (x - 0.5) ** 2) + 0.3
    elif choice == 2:
        a_half = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x)
        a_half = np.maximum(a_half, 0.05)
    elif choice == 3:
        a_half = 1.0 + 0.3 * (1 - 2 * x) * np.cos(np.pi * x)
        a_half = np.maximum(a_half, 0.05)
    else:
        a_half = 1.0 + 0.2 * np.sin(3 * np.pi * x)

    if n % 2 == 0:
        a = np.concatenate([a_half, a_half[-2::-1]])
    else:
        a = np.concatenate([a_half, a_half[-1::-1]])

    a = np.maximum(a, 0.01)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

        a_out2, _ = nesterov_lp(best_seq, t6, solver=solver, symmetric=False)
        _check_and_store(a_out2, "lp_r2")
    print(f"  Phase 6 done: best={best_score:.8f}")

    # === Phase 7: Length exploration ===
    print(f"\n=== Phase 7: Length exploration ===")
    remaining = _global_deadline - time.monotonic()
    if remaining > 50:
        lengths_to_try = [best_n + 50, best_n - 50, best_n + 100, best_n + 200, best_n - 100]
        for n_try in lengths_to_try:
            remaining = _global_deadline - time.monotonic()
            if remaining < 15:
                break
            if n_try < 100 or n_try > 5000:
                continue
            t7 = min(25.0, remaining * 0.2)
            if t7 < 8:
                break
            print(f"  Trying n={n_try} (budget={t7:.0f}s)")
            init_try = interpolate_sequence(best_seq, n_try)
            a_try, _ = nesterov_lp(init_try, t7, solver=solver, symmetric=False, max_iters=50000)
            _check_and_store(a_try, f"len_{n_try}")
    print(f"  Phase 7 done: best={best_score:.8f}")

    # === Phase 8: Final short polish ===
    remaining = _global_deadline - time.monotonic()
    if remaining > 8 and best_seq is not None:
        t8 = min(remaining * 0.7, 20.0)
        if t8 > 4:
            a_final, _ = peak_targeted_refine(best_seq, t8, symmetric=False, max_steps=10000)
            _check_and_store(a_final, "final_pt")

    # Final verification with actual evaluator
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'='*50}")
    return [float(x) for x in best_seq]
