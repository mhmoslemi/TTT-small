def coordinate_descent_polish(a0, time_limit, seed=42):
    """
    Coordinate descent: try adjusting pairs of elements to reduce the max autocorrelation.
    More systematic than random peak shaving.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:10]
        
        improved = False
        # Try moving mass from elements that contribute to peaks
        # to elements that are in "quiet" regions
        peak_elements = set()
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                peak_elements.add(j)
                peak_elements.add(k - j)
        
        # Find elements NOT in peak regions
        quiet_elements = [j for j in range(n) if j not in peak_elements]
        
        if not quiet_elements:
            quiet_elements = [j for j in range(n) if a[j] < np.median(a)]
        
        # Try transfers
        sources = sorted(peak_elements, key=lambda j: a[j], reverse=True)[:20]
        targets = sorted(quiet_elements, key=lambda j: a[j])[:20]
        
        tried = 0
        for src in sources:
            for dst in targets:
                if src == dst:
                    continue
                tried += 1
                if tried > 400:
                    break
                
                for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-15 or a[src] - amt < 0:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score - 1e-12:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
            if improved:
                break
            if tried > 400:
                break
        
        if not improved:
            # Random perturbation
            stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.001, 0.02)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            
            if stall > 500:
                break
    
    return best_a, best_score

def get_lp_solver():
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):

        except Exception:
            continue
    return cp.ECOS

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def diverse_initializations(n, rng, count=5):

            a = np.sin(np.pi * x) ** 2

        elif c == 3:
            a = 0.5 + 0.4 * np.cos(2 * np.pi * rng.integers(2, 10) * x)

            a = np.maximum(a, 0)
        a = np.maximum(a, 0)
        s = a.sum()
        if s > 1e-15:
            inits.append(a / s)
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Constraint-generation LP with Nesterov acceleration (main workhorse)
2. Spectral shaping for initialization and intermediate polish
3. Adam-based LogSumExp gradient optimization
4. Coordinate descent for final polish
5. Multi-length exploration with interpolation transfer
Key innovation: constraint generation reduces LP from O(n) constraints to O(K),
enough 10-50x more LP iterations in the same time budget.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(arr):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in arr]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name, verbose=False)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_reduced(a_fixed, active_indices, solver):
    """
    Solve LP: min M s.t. (conv(a_fixed, b))[k] <= M for k in active_indices,
    sum(b) = 1, b >= 0.
    Returns (b, M) or (None, None) on failure.
    """
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()

    # Build constraint matrix for only active indices
    T = np.zeros((len(active_indices), n))
    for i, k in enumerate(active_indices):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            T[i, k - j] = a_fixed[j]

    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        pass

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_constrained(a0, time_limit, solver, K_init=50, K_max=200, max_iters=100000):
    """
    Nesterov-accelerated LP with adaptive constraint generation.
    Starts with K_init constraints, grows K if violations detected.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    K = K_init
    fail_count = 0
    lp_solves = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        # Compute autocorrelation of y to find active constraints
        b_y = fast_autocorr(y)
        top_k = np.argsort(b_y)[::-1][:K]

        # Solve LP
        b_star, M_val = solve_lp_reduced(y, top_k, solver)
        lp_solves += 1

        if b_star is None:
            fail_count += 1
            if fail_count > 5:
                # Try with more constraints
                K = min(K * 2, K_max)
                fail_count = 0
            continue

        # Check for violations of inactive constraints
        conv_check = fast_autocorr(b_star)
        max_inactive = 0.0
        for k in range(len(conv_check)):
            if k not in top_k and conv_check[k] > M_val * (1 + 0.01):
                max_inactive = max(max_inactive, conv_check[k])

        if max_inactive > M_val * 1.1 and K < K_max:
            K = min(K + 20, K_max)

        # Nesterov step
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s

        y = y_new
        a = a_new

        # Check score
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        # Running average (often better than individual iterates)
        if t % 50 == 0:
            avg_score = 2.0 * n * fast_autocorr(y).max()
            if avg_score < best_score:
                best_score = avg_score
                best_a = y.copy()

        if t % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s, K={K}, LPs={lp_solves}): best={best_score:.6f}")

    return best_a, best_score


def spectral_shaping(a, n_iter=1000, alpha=0.3):
    """Spectral shaping: iteratively flatten the power spectrum."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2

        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))

        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.01, 50.0)
        scale = np.ones(len(A))
        scale[1:] = np.power(ratio, alpha / 2.0)

        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a


def adam_lse_optimize(a0, time_limit, T_start=50.0, T_end=0.01, n_stages=25, lr=0.005):
    """LogSumExp-smoothed optimization with Adam optimizer."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    beta1, beta2, eps = 0.9, 0.999, 1e-8

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        m = np.zeros(n)
        v = np.zeros(n)
        t_adam = 0

        iters = max(30, min(300, int(remaining * 80 / max(1, n_stages - si))))

        for it in range(iters):
            if time.monotonic() - t0 > time_limit:
                break

            a = softmax_fn(z)
            b = fast_autocorr(a)
            Tb = stage_T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()

            conv_len = n + (2 * n - 1) - 1
            fft_size = 1 << (conv_len - 1).bit_length()
            A_rev = rfft(a[::-1].copy(), fft_size)
            W = rfft(w, fft_size)
            conv = irfft(A_rev * W, fft_size)
            grad_a = 2.0 * conv[n - 1:n - 1 + n]

            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)

            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)

            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            if it % 20 == 0:
                a = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()

        a = softmax_fn(z)
        score = 2.0 * n * fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if si % 8 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Adam-LSE stage {si}/{n_stages} (T={stage_T:.3f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def coordinate_descent(a0, time_limit, seed=42):
    """Coordinate descent: systematically move mass to reduce autocorrelation peaks."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:10]

        # Identify peak-contributing elements
        peak_elements = set()
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                peak_elements.add(j)
                peak_elements.add(k - j)

        quiet_elements = [j for j in range(n) if j not in peak_elements]
        if not quiet_elements:
            quiet_elements = [j for j in range(n) if a[j] < np.median(a) * 0.5]
        if not quiet_elements:
            quiet_elements = list(range(n))

        sources = sorted(list(peak_elements), key=lambda j: a[j], reverse=True)[:15]
        targets = sorted(quiet_elements, key=lambda j: a[j])[:15]

        improved = False
        for src in sources:
            for dst in targets:
                if src == dst:
                    continue
                for frac in [0.003, 0.008, 0.02, 0.05]:
                    amt = a[src] * frac
                    if amt < 1e-15 or a[src] - amt < 0:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.002, 0.03)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            if stall > 400:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
