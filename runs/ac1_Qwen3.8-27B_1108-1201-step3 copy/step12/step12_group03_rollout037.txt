def get_lp_solver():
    for solver_name in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver):
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

    buf = [a.copy()]

            a_s = a_new.sum()
            if a_s < 1e-15:

        a = a_new

        avg_a += a_new

            buf_idx = (buf_idx + 1) % buf_size
        
        # Check score of a
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0

def peak_shading(a0, time_limit, seed=42):
    """
    Peak shaving: identify top peaks and reduce them by moving mass.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    max_stall = 300
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        current_score = 2.0 * n * max_b
        
        # Top peaks
        n_peaks = min(8, 2 * n - 1)
        top_peaks = np.argsort(b)[::-1][:n_peaks]
        
        improved = False
        for k in top_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            total = contrib.sum()
            if total < 1e-15:
                continue
            
            # Target: reduce the top contributing pair
            top_ci = np.argsort(contrib)[::-1][:3]
            for ci in top_ci:
                i1 = idx[ci]
                i2 = k - i1
                
                # Find a good target position (far from peaks)
                # Score: distance from peak positions minus current value
                peak_lags = top_peaks[:5]
                good_targets = []
                for j in range(n):
                    # How much would adding to j increase the peaks?
                    penalty = 0
                    for k2 in peak_lags:
                        # If j contributes to lag k2, it would increase that peak
                        if 0 <= k2 - j < n:
                            penalty += a[k2 - j]
                        if 0 <= j - k2 < n:  # This is conv[j][k2-j] but we need conv[k2]
                            # Actually conv[k2] = sum_i a[i] * a[k2-i]
                            # Adding to a[j] increases conv[k2] by a[k2-j] if k2-j is valid
                            penalty += a[k2 - j] if 0 <= k2 - j < n else 0
                    dist_score = min(abs(j - i1), abs(j - i2)) if n > 1 else 0
                    good_targets.append((dist_score * 0.01 - penalty - a[j] * 0.001, j))
                
                good_targets.sort(reverse=True)
                
                for _, dst in good_targets[:5]:
                    if dst == i1 and dst == i2:
                        continue
                    for frac in [0.001, 0.003, 0.005, 0.01]:
                        amt = min(a[i1], a[i2]) * frac if i1 != i2 else a[i1] * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[i1] -= amt
                        if i1 != i2:
                            a_trial[i2] -= amt * 0.5
                        a_trial[dst] += amt
                        if (a_trial < -1e-12).any():
                            continue
                        a_trial = np.maximum(a_trial, 0)
                        a_trial_s = a_trial.sum()
                        if a_trial_s < 1e-15:
                            continue
                        a_trial = a_trial / a_trial_s
                        score = 2.0 * n * fast_autocorr(a_trial).max()
                        if score < best_score - 1e-12:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            if stall > 20 and stall % 5 == 0:
                a_pert = a.copy()
                for _ in range(rng.integers(1, 4)):
                    src = rng.integers(n)
                    dst = rng.integers(n)
                    if src == dst:
                        continue

                    a_pert[src] -= amt
                    a_pert[dst] += amt

                        a = a_pert
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            
            if stall > max_stall:
                break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

            buf_idx = 0
            if verbose:
                print(f"    Restart t={t} ({time.monotonic()-t0:.0f}s), best={best_score:.8f}")

        if t % 1000 == 0 and verbose:
            elapsed = time.monotonic() - t0
            print(f"    NLP t={t} ({elapsed:.0f}s): best={best_score:.8f}, M={M_val:.8f}, iters={iter_count}")

        if stall > 8000:
            break

    return best_a, best_score, iter_count


def peak_shading(a0, time_limit, seed=42):
    """Targeted peak shaving via mass redistribution."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    max_stall = 600

    # Precompute candidate targets (sampled positions)
    n_targets = min(500, n)
    target_positions = np.linspace(0, n - 1, n_targets, dtype=int)

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()

        n_peaks = min(6, 2 * n - 1)
        top_peaks = np.argsort(b)[::-1][:n_peaks]

        improved = False
        for k in top_peaks:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            top_ci = np.argsort(contrib)[::-1][:4]
            for ci in top_ci:
                i1 = idx[ci]
                i2 = k - i1

                # Score target positions by penalty
                best_target = None
                best_target_score = -1e10
                for j in target_positions:
                    penalty = 0.0
                    for k2 in top_peaks[:4]:
                        if 0 <= k2 - j < n:
                            penalty += a[k2 - j]
                    dist = min(abs(j - i1), abs(j - i2)) / (n + 1)
                    score = dist - penalty * 0.5 - a[j] * 0.01
                    if score > best_target_score:
                        best_target_score = score
                        best_target = j

                if best_target is None:
                    continue

                for frac in [0.001, 0.003, 0.005, 0.01, 0.02]:
                    if i1 == i2:
                        amt = a[i1] * frac
                    else:
                        amt = min(a[i1], a[i2]) * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[i1] -= amt
                    if i1 != i2:
                        a_trial[i2] -= amt * 0.3
                    a_trial[best_target] += amt
                    if (a_trial < -1e-12).any():
                        continue
                    a_trial = np.maximum(a_trial, 0)
                    ts = a_trial.sum()
                    if ts < 1e-15:
                        continue
                    a_trial = a_trial / ts
                    sc = 2.0 * n * fast_autocorr(a_trial).max()
                    if sc < best_score - 1e-11:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 50 and stall % 10 == 0:
                a_pert = a.copy()
                for _ in range(rng.integers(2, 5)):
                    src = rng.integers(n)
                    dst = rng.integers(n)
                    if src == dst:
                        continue
                    amt = a_pert[src] * rng.uniform(0.002, 0.015)
                    a_pert[src] -= amt
                    a_pert[dst] += amt
                a_pert = np.maximum(a_pert, 0)
                ps = a_pert.sum()
                if ps > 1e-15:
                    a_pert = a_pert / ps
                    sc = 2.0 * n * fast_autocorr(a_pert).max()
                    if sc < best_score:
                        a = a_pert
                        best_score = sc
                        best_a = a.copy()
                        stall = 0
            if stall > max_stall:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Linearly interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def polish_active_set(a0, time_limit, verbose=False):
    """Refine using active-set LP with tight constraints."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        b_autocorr = fast_autocorr(y)
        max_b = b_autocorr.max()
        # Use very tight active set
        threshold = max_b * (1.0 - 0.002 * min(1.0, t / 100.0))
        active = np.where(b_autocorr >= threshold)[0]
        if len(active) > 50:
            active = np.argsort(b_autocorr)[::-1][:50]
        if len(active) == 0:
            active = np.array([np.argmax(b_autocorr)])

        b_star, M_val = solve_lp_active_set(y, active)
        if b_star is None:
            b_star, M_val = solve_lp_best_response(y)
        if b_star is None:
            break

        alpha = 2.0 / (t + 2.0)
        y_new = (1 - alpha) * y + alpha * b_star
        y_new = np.maximum(y_new, 0)
        ys = y_new.sum()
        if ys < 1e-15:
            y_new = y.copy()
            ys = 1.0
        y_new = y_new / ys

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            a_s = a_new.sum()
            if a_s < 1e-15:
                a_new = y_new.copy()
                a_s = 1.0
            a_new = a_new / a_s

        y = y_new
        a = a_new
        avg_a += a_new
        n_avg += 1

        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        if n_avg >= 50 and n_avg % 100 == 0:
            avg_check = avg_a / n_avg
            score_avg = 2.0 * n * fast_autocorr(avg_check).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()

        if verbose and t % 500 == 0:
            print(f"    Polish t={t}: best={best_score:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search: multi-phase LP Nesterov with scipy HiGHS, active-set polishing,
    peak shading, and multi-length exploration.
    """
    t_start = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={ps:.6f}")
            best_seq = parent.copy()
            best_score = ps
    except Exception:
        pass

    if parent is None:
        n_base = 1600
        x = np.linspace(0, np.pi, n_base)
        best_seq = np.cos(x / 2) ** 2
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)

    def do_update(cand, label):
        nonlocal best_seq, best_score
        if cand is None:
            return
        cand = np.maximum(np.asarray(cand, dtype=np.float64), 0)
        s = cand.sum()
        if s < 0.01:
            return
        sc = evaluate_safe(cand)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            elapsed = time.monotonic() - t_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(cand)}, t={elapsed:.0f}s)")

    # Phase 1: Main LP Nesterov
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 1: LP Nesterov n={n_base} ({remaining:.0f}s) ===")
    t1 = min(remaining * 0.45, 700.0)
    if t1 > 60:
        a_out, sc_out, iters = nesterov_lp_optimize(best_seq, t1, verbose=True)
        print(f"  LP Nesterov done: {iters} iters, local={sc_out:.8f}")
        do_update(a_out, f"lp_n{n_base}")

    # Phase 2: Active-set polishing
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 2: Active-set Polish ({remaining:.0f}s) ===")
    t2 = min(remaining * 0.20, 250.0)
    if t2 > 30:
        a_out, sc_out = polish_active_set(best_seq, t2, verbose=False)
        print(f"  Polish done: local={sc_out:.8f}")
        do_update(a_out, "polish1")

    # Phase 3: Peak shading
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 3: Peak Shading ({remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 180.0)
    if t3 > 15:
        a_out, sc_out = peak_shading(best_seq, t3, seed=42)
        print(f"  Peak shade done: local={sc_out:.8f}")
        do_update(a_out, "peak1")

    # Phase 4: Another LP round
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 4: LP Nesterov round 2 ({remaining:.0f}s) ===")
    t4 = min(remaining * 0.25, 250.0)
    if t4 > 40:
        a_out, sc_out, iters = nesterov_lp_optimize(best_seq, t4, verbose=False)
        print(f"  LP round2 done: {iters} iters, local={sc_out:.8f}")
        do_update(a_out, "lp_round2")

    # Phase 5: Try nearby lengths
    for n_try in [n_base + 100, n_base - 100, n_base + 200]:
        if n_try < 400:
            continue
        remaining = time_budget - (time.monotonic() - t_start)
        if remaining < 80:
            break
        print(f"\n=== Phase 5: LP at n={n_try} ({remaining:.0f}s) ===")
        t5 = min(remaining * 0.20, 120.0)
        if t5 < 30:
            continue
        init_try = interpolate_sequence(best_seq, n_try)
        a_out, sc_out, iters = nesterov_lp_optimize(init_try, t5, verbose=False)
        print(f"  n={n_try}: {iters} iters, local={sc_out:.8f}")
        do_update(a_out, f"lp_n{n_try}")

    # Phase 6: Final polish + peak shade
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 6: Final Polish ({remaining:.0f}s) ===")
    t6 = min(remaining * 0.30, 100.0)
    if t6 > 15:
        a_out, sc_out = polish_active_set(best_seq, t6, verbose=False)
        do_update(a_out, "polish_final")

    remaining = time_budget - (time.monotonic() - t_start)
    if remaining > 20:
        t7 = min(remaining * 0.4, 60.0)
        if t7 > 10:
            a_out, sc_out = peak_shading(best_seq, t7, seed=789)
            do_update(a_out, "peak_final")

    # Final evaluation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
