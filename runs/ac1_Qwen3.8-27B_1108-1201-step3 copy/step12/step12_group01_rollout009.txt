    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    for T in T_stages:
        if time.monotonic() - t0 > time_limit - 10:
            break
        
        z0 = np.log(np.maximum(best_a, 1e-30))
        
        def fun_and_grad(z):
            a = np.exp(z - z.max())
            a = a / a.sum()
            c = fast_autocorr(a)
            top_idx = np.argsort(c)[::-1][:K]
            top_vals = c[top_idx]
            Tc = T * top_vals
            max_Tc = Tc.max()
            exp_Tc = np.exp(Tc - max_Tc)

                    grad_a[j_idx] += 2.0 * wk * a[k - j_idx]

            grad_z = a * (grad_a - Sa)

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the objective value."""
    n = len(a)
    c = fast_autocorr(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * n * c.max() / (s * s)


def obj_grad_lse(z, n, K, T):
    """LSE-smoothed max of top-K autocorrelation peaks. Returns (obj, grad)."""
    a = np.exp(z - z.max())
    a = a / a.sum()
    c = fast_autocorr(a)
    top_idx = np.argsort(c)[::-1][:K]
    top_vals = c[top_idx]
    Tc = T * top_vals
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    lse = (max_Tc + np.log(exp_Tc.sum())) / T
    w = exp_Tc / exp_Tc.sum()
    grad_a = np.zeros(n)
    for idx in range(K):
        k = top_idx[idx]
        wk = w[idx]
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            j_idx = np.arange(js, je + 1)
            grad_a[j_idx] += 2.0 * wk * a[k - j_idx]
    Sa = np.dot(grad_a, a)
    grad_z = a * (grad_a - Sa)
    return lse, grad_z

def perturb_sequence(a, rng, mode=0, scale=0.05):
    """Create a perturbed version of a sequence."""
    n = len(a)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    a = a / s

    if mode == 0:
        # Gaussian noise proportional to value
        noise = rng.normal(0, scale, n)
        a_new = a + noise * (a + 1e-8)
    elif mode == 1:
        # Random mass transfer
        a_new = a.copy()
        n_transfers = rng.integers(3, 10)
        for _ in range(n_transfers):
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a_new[i] > 1e-10:
                amt = a_new[i] * rng.uniform(0.01, 0.1)
                a_new[i] -= amt
                a_new[j] += amt
    elif mode == 2:
        # Small random blur
        kernel_size = rng.integers(3, 7) | 1
        kernel = np.ones(kernel_size) / kernel_size
        a_new = np.convolve(a, kernel, mode='same')
    elif mode == 3:
        # Frequency domain perturbation
        fft_size = 1 << (2 * n - 1).bit_length()
        A = rfft(a, fft_size)
        freq_noise = rng.normal(0, scale * 0.1, len(A))
        A[1:] += freq_noise[1:]
        a_new = irfft(A, fft_size)[:n]
    else:
        # Mixed perturbation
        noise = rng.normal(0, scale * 0.5, n)
        a_new = a + noise * (a + 1e-8)
        i = rng.integers(n)
        j = rng.integers(n)
        if i != j and a_new[i] > 1e-10:
            amt = a_new[i] * rng.uniform(0.01, 0.05)
            a_new[i] -= amt
            a_new[j] += amt

    a_new = np.maximum(a_new, 0)
    s_new = a_new.sum()
    if s_new < 1e-15:
        return a.copy()
    return a_new / s_new


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 10)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 3
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 2
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.7 + 0.2 * np.cos(2 * x) + 0.1 * np.cos(5 * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-5 * (x - 0.3) ** 2) + 0.5 * np.exp(-5 * (x - 0.7) ** 2)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) * np.cos(np.pi * x / 3) ** 2
    elif choice == 7:
        x = np.linspace(-1, 1, n)
        a = 0.5 * (1 + np.cos(np.pi * x))
    elif choice == 8:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(4 * np.pi * x) + 0.1 * np.sin(7 * np.pi * x)
    else:
        x = np.linspace(0, 1, n)
        a = np.exp(-2 * (x - 0.5) ** 2) * (1 + 0.5 * np.cos(6 * np.pi * x))
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def eval_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def propose_candidate(seed=42, time_budget=1800.0):
    """
    Multi-phase optimization combining L-BFGS quasi-Newton methods,
    random restarts, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    budget = min(float(time_budget) - 60.0, 1740.0)

    # Load parent sequence
    best_seq = None
    best_score = float('inf')

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            p = p / p.sum()
            best_seq = p.copy()
            best_score = eval_safe(p)
            print(f"Parent: n={len(p)}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        n_init = 1600
        best_seq = structured_init(n_init, rng)
        best_score = compute_score(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return False
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return False
        a = a / s
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # === Phase 1: Direct L-BFGS optimization on parent ===
    remaining = budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 1: L-BFGS on n={len(best_seq)} ({remaining:.0f}s) ===")
    t1 = min(remaining * 0.25, 300.0)
    a_opt, _ = optimize_single(best_seq, t1, "phase1")
    try_update(a_opt, "phase1")

    # === Phase 2: Random restarts with L-BFGS refinement ===
    remaining = budget - (time.monotonic() - t_start)
    print(f"=== Phase 2: Random restarts ({remaining:.0f}s) ===")
    n_restarts = 40
    t_per_restart = remaining * 0.35 / n_restarts
    t_per_restart = max(t_per_restart, 3.0)

    for ri in range(n_restarts):
        if time.monotonic() - t_start > budget - 60:
            break
        mode = ri % 5
        scale = 0.02 + 0.03 * (ri % 4)
        pert = perturb_sequence(best_seq, rng, mode=mode, scale=scale)
        a_opt, _ = optimize_single(pert, t_per_restart, f"restart{ri}")
        try_update(a_opt, f"restart{ri}")

    # === Phase 3: Multi-length exploration ===
    remaining = budget - (time.monotonic() - t_start)
    print(f"=== Phase 3: Multi-length ({remaining:.0f}s) ===")
    n_base = len(best_seq)
    lengths_to_try = [n_base - 100, n_base + 100, n_base + 200, n_base + 400]
    lengths_to_try = [l for l in lengths_to_try if l > 200]

    for n_try in lengths_to_try:
        if time.monotonic() - t_start > budget - 60:
            break
        t_len = min(remaining * 0.12, 120.0)
        if t_len < 10:
            continue
        print(f"  Trying n={n_try} ({t_len:.0f}s)")
        init_try = interpolate_sequence(best_seq, n_try)
        a_opt, _ = optimize_single(init_try, t_len, f"len{n_try}")
        try_update(a_opt, f"len{n_try}")
        remaining = budget - (time.monotonic() - t_start)

    # === Phase 4: More aggressive restarts from scratch ===
    remaining = budget - (time.monotonic() - t_start)
    print(f"=== Phase 4: Fresh starts ({remaining:.0f}s) ===")
    n_fresh = len(best_seq)
    n_fresh_starts = 8
    t_fresh = remaining * 0.15 / max(1, n_fresh_starts)
    t_fresh = max(t_fresh, 5.0)

    for fi in range(n_fresh_starts):
        if time.monotonic() - t_start > budget - 60:
            break
        init_fresh = structured_init(n_fresh, rng)
        a_opt, _ = optimize_single(init_fresh, t_fresh, f"fresh{fi}")
        try_update(a_opt, f"fresh{fi}")

    # === Phase 5: Ultra-fine polish ===
    remaining = budget - (time.monotonic() - t_start)
    print(f"=== Phase 5: Ultra-fine polish ({remaining:.0f}s) ===")
    t5 = min(remaining * 0.3, 200.0)
    if t5 > 10:
        # Very low temperature LSE
        n = len(best_seq)
        a_curr = best_seq.copy()
        for K in [3, 5, 10, 20]:
            for T in [0.5, 0.3, 0.1, 0.05]:
                if time.monotonic() - t_start > budget - 30:
                    break
                z0 = np.log(np.maximum(a_curr, 1e-30))
                z_opt = run_lbfgs(z0, obj_grad_lse, (n, K, T), maxiter=500, gtol=1e-13)
                a_opt = np.exp(z_opt - z_opt.max())
                a_opt = a_opt / a_opt.sum()
                score = compute_score(a_opt)
                if score < best_score:
                    best_score = score
                    best_seq = a_opt.copy()
                    a_curr = a_opt.copy()
                    print(f"    Polish K={K},T={T}: {score:.8f}")

    # === Phase 6: Final random micro-perturbations ===
    remaining = budget - (time.monotonic() - t_start)
    if remaining > 20:
        print(f"=== Phase 6: Micro-perturbations ({remaining:.0f}s) ===")
        t6 = min(remaining * 0.6, 150.0)
        if t6 > 10:
            n = len(best_seq)
            t_per = t6 / 60
            t_per = max(t_per, 1.0)
            for mi in range(60):
                if time.monotonic() - t_start > budget - 20:
                    break
                pert = perturb_sequence(best_seq, rng, mode=rng.integers(5), scale=0.01)
                z0 = np.log(np.maximum(pert, 1e-30))
                for K, T in [(5, 1), (3, 0.5), (10, 2)]:
                    z_opt = run_lbfgs(z0, obj_grad_lse, (n, K, T), maxiter=100, gtol=1e-12)
                    a_opt = np.exp(z_opt - z_opt.max())
                    a_opt = a_opt / a_opt.sum()
                    try_update(a_opt, f"micro{mi}")

    # Final verification
    if best_seq is not None:
        final_score = eval_safe(best_seq)
        if np.isfinite(final_score):
            best_score = final_score
        print(f"\nFINAL: score={best_score:.8f}, n={len(best_seq)}")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
