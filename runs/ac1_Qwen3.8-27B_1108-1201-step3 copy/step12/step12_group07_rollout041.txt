Let me now think about a truly novel approach that might give a breakthrough:

**"Convolution deconvolution" approach:**

The idea: if we want `conv(a, a) ≈ c * 1` (flat), then in the "multiplicative" sense (in the Fourier domain), we want `A(f)^2 ≈ c * FFT(1)` where A(f) is the DFT of a. But this doesn't work directly because of the non-negativity constraint and the linear (not circular) nature of the convolution.

However, we can use it as a heuristic:
1. Start with a candidate `a`
2. Compute `b = conv(a, a)`
3. The "ideal" a should have `conv(a, a)` flatter than the current b
4. Use `a_new = some_function(b)` as a new candidate
5. What function? If `b ≈ c * 1`, we want `a` such that `conv(a,a) ≈ c*1`. This is like taking the "convolution square root."

The "convolution square root" can be computed in the frequency domain: if `B(f) = A(f)^2`, then `A(f) = sqrt(B(f))`. But we need to be careful about phases.

For real non-negative sequences, the DFT has conjugate symmetry, and the phase of `A(f)` is related to the "center of mass" of `a`. This gets complicated.

A simpler heuristic: 
1. Compute `b = conv(a, a)`
2. Replace `b` with a "flattened" version: `b_flat = b + alpha * (mean(b) - b)` (reduce contrast)
3. Try to find `a` such that `conv(a, a) ≈ b_flat` (this is a "deconvolution" or "non-negative square root" problem)
4. Use this as the new `a`

The non-negative deconvolution is hard, but we can approximate it with a few iterations of:
- `a_new[j] = max(0, a[j] - eta * (conv(a,a)[j+n-1] - target[j]))` (gradient descent in the time domain)

This is essentially what the previous methods do, just framed differently.

- Optimization over a "control" sequence that generates `a` through a deterministic mapping

For example:

- `a[j] = u[j] / sum(u)` where `u = max(0, v)` for some `v`

**My main improvement idea: Diversified multi-start with shared information**

1. Create several diverse starting points (from the parent sequence + perturbations)
2. Run a short optimization on each
3. Keep the best one and continue optimizing
4. Periodically generate new perturbations from the current best

This is like a "chain" of optimizations where each one starts from the previous one but with a perturbation.

**Second improvement idea: "Convolution-aware" perturbation**

When perturbing the sequence, use information about the convolution peaks to guide the perturbation. Specifically:
- At the peak lag k, the convolution value is `sum_j a[j]*a[k-j]`
- To reduce this peak, reduce products `a[j]*a[k-j]` where both factors are large
- This suggests reducing `a[j]` for j in the "heavy" part of the sequence at the peak lag

**Third improvement idea: "Complementary" sequences**

The convolution `conv(a, a)` at lag k involves pairs `(j, k-j)`. If we think of the sequence as being split into pairs, we want the products within each pair to be small. This is similar to the "balancing" problem.

def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j+n, j] = a_fixed

def solve_lp_best_response(a_fixed, solver=None):
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.ECOS, cp.SCS]:
            if alt is None or alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None
    return b_val / s

def nesterov_lp(a0, time_limit, solver=None, max_iters=500000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    restarts = 0
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if t_nest > 1.0:
            mom = (t_nest - 1.0) / t_nest
            y = a + mom * (a - a_prev)
            y = np.maximum(y, 0)
            sy = y.sum()
            y = y / sy if sy > 1e-15 else a.copy()
        else:
            y = a
        
        b_new = solve_lp_best_response(y, solver)
        if b_new is None:
            continue
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        alpha = 2.0 / (t_new + 1.0)
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        a_new = a_new / sa if sa > 1e-15 else b_new.copy()
        
        sc = score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        sc_cur = score_normalized(a)
        if sc > sc_cur * 1.001 and t_nest > 5.0:
            t_nest = 1.0
            a_prev = a_new.copy()
            restarts += 1
        else:
            a_prev = a.copy()
            t_nest = t_new
        a = a_new
    
    return best_a, best_score

def conv_aware_perturb(a, rng, strength=0.1):
    """Perturb a in a way informed by convolution peak structure."""
    n = len(a)
    b = fast_convolve(a)
    peak_k = int(np.argmax(b))
    # Get the "partner" indices for the peak lag
    # At peak_k, the products a[j]*a[peak_k - j] contribute
    js = max(0, peak_k - n + 1)
    je = min(n - 1, peak_k)
    partners = np.arange(js, je + 1)
    partner_vals = a[peak_k - partners]
    # Products that are "bad" (large) are where both a[j] and a[peak_k-j] are large
    product = a[partners] * partner_vals
    # Reduce the larger of each pair
    mask_large = a[partners] > partner_vals
    # Create perturbation: reduce elements that are "responsible" for the peak
    pert = np.zeros(n)
    for j, p in zip(partners, product):
        if p > product.mean():
            if mask_large[j - js]:
                pert[j] = -strength * a[j]
            else:
                pert[peak_k - j] = -strength * a[peak_k - j]
    # Add random component
    pert += rng.normal(0, strength * 0.1, n)
    a_new = a + pert
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def stochastic_local_search(a0, time_limit, rng, n_restarts=50):
    """Stochastic perturbation + local search."""
    n = len(a0)
    best_a = a0.copy()
    best_score = score_normalized(a0)
    t0 = time.monotonic()
    current_a = a0.copy()
    current_score = best_score
    
    for restart in range(n_restarts):
        if time.monotonic() - t0 > time_limit:
            break
        per_iter = (time_limit - (time.monotonic() - t0)) / max(1, n_restarts - restart)
        
        # Perturb
        strength = 0.05 + 0.1 * rng.random()
        a_try = conv_aware_perturb(current_a, rng, strength)
        sc_try = score_normalized(a_try)
        
        # Accept if better or with some probability (simulated annealing)
        if sc_try < current_score or rng.random() < np.exp(-(sc_try - current_score) / 0.01):
            current_a = a_try
            current_score = sc_try
            if sc_try < best_score:
                best_score = sc_try
                best_a = a_try.copy()
    
    return best_a, best_score

def peak_targeted_refine_v2(a0, time_limit, max_steps=100000):
    """Improved peak-targeted multiplicative refinement with top-K awareness."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s if s > 1e-15 else np.ones(n) / n

    t0 = time.monotonic()
    eta = 0.08

        # Use top 20% of peaks
        threshold = max_b * 0.85

            topk = np.array([int(np.argmax(b))])

        weights = (b[topk] - threshold)

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Top-K LogSumExp L-BFGS for fast shape optimization
2. LP-based Nesterov with adaptive momentum (main workhorse)
3. Convolution-aware stochastic perturbation for escaping local minima
4. Peak-targeted multiplicative refinement with weighted top-K
5. Multi-diversification: perturbed restarts from current best
6. Final polishing with fine-grained LSE
Key novel ideas: conv-aware perturbation, multi-diversification pool,

import numpy as np

def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def conv_grad_from_weights(weights, a, n):
    grad = np.zeros(n)
    for k in range(len(weights)):
        w = weights[k]
        if abs(w) < 1e-20:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        idx = np.arange(js, je + 1)
        grad[idx] += w * a[k - idx]
    return grad


def topk_lse_obj_grad(z, T_val, K_top, n):
    a = softmax_from_z(z)
    b = fast_convolve(a)
    K = min(K_top, len(b))
    if K == len(b):
        top_idx = np.arange(len(b))
        top_vals = b.copy()
    else:
        top_idx = np.argpartition(-b, K - 1)[:K]
        top_vals = b[top_idx].copy()
    max_v = top_vals.max()
    shifted = (top_vals - max_v) / T_val
    exp_v = np.exp(shifted)
    exp_sum = exp_v.sum()
    lse = max_v + T_val * np.log(exp_sum)
    obj = 2.0 * n * lse
    w = exp_v / exp_sum
    bw = np.zeros(len(b))
    bw[top_idx] = w
    grad_a = conv_grad_from_weights(bw, a, n)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbgfs_topk(a0, time_limit, T_start=50.0, T_end=0.05, n_stages=15, K_top=25):
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)
    temps = np.geomspace(T_start, T_end, n_stages)
    for si in range(n_stages):
        T = temps[si]
        if time.monotonic() - t0 > time_limit:
            break
        if time_limit - (time.monotonic() - t0) < 5:
            break

        def _fun(zv):
            obj, _ = topk_lse_obj_grad(zv, T, K_top, n)
            return obj

        def _jac(zv):
            _, g = topk_lse_obj_grad(zv, T, K_top, n)
            return g

        max_it = 150 if si < n_stages // 3 else 80
        try:
            res = minimize(_fun, z, jac=_jac, method='L-BFGS-B',
                           options={'maxiter': max_it, 'ftol': 1e-18,
                                    'gtol': 1e-15, 'maxfun': max_it * 60})
            z = res.x
            a_new = softmax_from_z(z)
            sc = score_normalized(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        except Exception:
            pass
    return best_a, best_score


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T

    b_val = np.asarray(b_var.value, dtype=np.float64)

def nesterov_lp(a0, time_limit, solver=None, max_iters=500000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    restarts = 0
    stall_counter = 0

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        if t_nest > 1.0:
            mom = (t_nest - 1.0) / t_nest
            y = a + mom * (a - a_prev)
            y = np.maximum(y, 0)
            sy = y.sum()
            y = y / sy if sy > 1e-15 else a.copy()
        else:
            y = a

        b_new = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        alpha = 2.0 / (t_new + 1.0)
        a_new = (1.0 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        a_new = a_new / sa if sa > 1e-15 else b_new.copy()

        sc = score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            stall_counter = 0
        else:
            stall_counter += 1

        sc_cur = score_normalized(a)
        if sc > sc_cur * 1.0005 and t_nest > 3.0:
            t_nest = 1.0
            a_prev = a_new.copy()
            restarts += 1
        else:
            a_prev = a.copy()
            t_nest = t_new
        a = a_new

    return best_a, best_score


def conv_aware_perturb(a, rng, strength=0.08):
    n = len(a)
    b = fast_convolve(a)
    # Top few peaks
    K = min(10, len(b))
    top_peaks = np.argpartition(-b, K - 1)[:K]
    max_b = b.max()
    pert = np.zeros(n)
    for k in top_peaks:
        if b[k] < max_b * 0.8:
            break
        weight = (b[k] / max_b) ** 2
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        partners = np.arange(js, je + 1)
        partner_idx = k - partners
        products = a[partners] * a[partner_idx]
        mean_prod = products.mean() if len(products) > 0 else 0
        for j, p in zip(partners, products):
            if p > mean_prod * 1.2:
                pert[j] -= weight * strength * a[j] * 0.5
                pert[partner_idx[(j - js)]] -= weight * strength * a[partner_idx[(j - js)]] * 0.5
    # Add Gaussian noise
    pert += rng.normal(0, strength * 0.05, n)
    a_new = a + pert
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s

def peak_targeted_refine(a0, time_limit, max_steps=100000):
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s if s > 1e-15 else np.ones(n) / n
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    eta = 0.06
    last_improve = 0

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        if step - last_improve > 8000:
            break

        b = fast_convolve(a)
        max_b = b.max()
        threshold = max_b * 0.80
        topk = np.where(b >= threshold)[0]
        if len(topk) == 0:
            topk = np.array([int(np.argmax(b))])

        # Weight peaks by their excess above threshold
        excess = b[topk] - threshold
        w_sum = excess.sum()
        if w_sum > 1e-20:
            weights = excess / w_sum
        else:
            weights = np.ones(len(topk)) / len(topk)

        grad = np.zeros(n)
        for k, w in zip(topk, weights):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += w * a[k - idx]

            if eta < 1e-12:
                break

def smoothness_constrained_lse(a0, time_limit, T_start=20.0, T_end=0.05,
                                n_stages=10, K_top=20, smooth_lambda=0.01):
    """LSE with a smoothness penalty to avoid oscillatory solutions."""
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)
    temps = np.geomspace(T_start, T_end, n_stages)

    for si in range(n_stages):
        T = temps[si]
        if time.monotonic() - t0 > time_limit:
            break
        if time_limit - (time.monotonic() - t0) < 5:
            break

        def _fun_s(zv):
            a_t = softmax_from_z(zv)
            b_t = fast_convolve(a_t)
            K = min(K_top, len(b_t))
            if K == len(b_t):
                tv = b_t
            else:
                ti = np.argpartition(-b_t, K - 1)[:K]
                tv = b_t[ti]
            mv = tv.max()
            ev = np.exp((tv - mv) / T)
            lse = mv + T * np.log(ev.sum())
            # Smoothness penalty
            diff = np.diff(a_t)
            smooth = smooth_lambda * np.sum(diff * diff)
            return 2.0 * n * lse + smooth

        def _jac_s(zv):
            obj, gz = topk_lse_obj_grad(zv, T, K_top, n)
            a_t = softmax_from_z(zv)
            diff = np.diff(a_t)
            smooth_grad_a = np.zeros(n)
            smooth_grad_a[1:-1] += 2 * smooth_lambda * (diff[:-1] - diff[1:])
            smooth_grad_a[0] += 2 * smooth_lambda * diff[0]
            smooth_grad_a[-1] += -2 * smooth_lambda * diff[-1]
            S = np.dot(smooth_grad_a, a_t)
            gz_smooth = a_t * (smooth_grad_a - S)
            return gz + gz_smooth

        max_it = 100
        try:
            res = minimize(_fun_s, z, jac=_jac_s, method='L-BFGS-B',
                           options={'maxiter': max_it, 'ftol': 1e-18,
                                    'gtol': 1e-14, 'maxfun': max_it * 50})
            z = res.x
            a_new = softmax_from_z(z)
            sc = score_normalized(a_new)
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
        except Exception:
            pass

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

            best_seq = p / p.sum()
            n_base = len(best_seq)
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.8f}")
    except Exception:
        pass

    if best_seq is None:
        x = np.linspace(0, 1, n_base)
        a1 = np.exp(-4.0 * x) + 0.2
        a2 = 0.5 + 0.35 * np.cos(2 * np.pi * (x * 0.8 + 0.15))
        a3 = (1.0 - x) ** 0.7 + 0.1
        best_seq = 0.4 * a1 + 0.35 * a2 + 0.25 * a3
        best_seq = np.maximum(best_seq, 0.01)
        best_seq = best_seq / best_seq.sum()
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    n = len(best_seq)

    def _try_update(arr, label):
        nonlocal best_seq, best_score
        if arr is None:
            return
        arr = np.maximum(arr, 0).astype(np.float64)
        s = arr.sum()
        if s < 0.01:
            return
        arr = arr / s
        sc = evaluate_safe(arr)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = arr.copy()
            print(f"  NEW BEST [{label}]: {sc:.8f}")

    # === Phase 0: Quick LSE shape improvement (60s) ===
    print(f"\n=== Phase 0: Quick LSE shape (score={best_score:.8f}) ===")
    t0 = _global_deadline - time.monotonic()
    t_b = min(60.0, t0 * 0.05)
    if t_b > 10:
        a_out, _ = lbgfs_topk(best_seq, t_b, T_start=30.0, T_end=0.5,
                              n_stages=8, K_top=30)
        _try_update(a_out, "lse_q0")
    print(f"  Phase 0 done: best={best_score:.8f}")

    # === Phase 1: LP Nesterov main (600s) ===
    print(f"\n=== Phase 1: LP Nesterov main (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t_b = min(600.0, remaining * 0.42)
    if t_b > 30:
        a_out, _ = nesterov_lp(best_seq, t_b, solver=solver)
        _try_update(a_out, "lp_main")
    print(f"  Phase 1 done: best={best_score:.8f}")

    # === Phase 2: Stochastic diversification (200s) ===
    print(f"\n=== Phase 2: Stochastic diversify (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t_b = min(200.0, remaining * 0.18)
    if t_b > 15:
        a_out, _ = stochastic_diversify(best_seq, t_b, rng, n_tries=50)
        _try_update(a_out, "stoch_div")
    print(f"  Phase 2 done: best={best_score:.8f}")

    # === Phase 3: Peak-targeted refinement (200s) ===
    print(f"\n=== Phase 3: Peak-targeted (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t_b = min(200.0, remaining * 0.18)
    if t_b > 10:
        a_out, _ = peak_targeted_refine(best_seq, t_b, max_steps=80000)
        _try_update(a_out, "peak_p3")
    print(f"  Phase 3 done: best={best_score:.8f}")

    # === Phase 4: LP Nesterov round 2 (300s) ===
    print(f"\n=== Phase 4: LP Nesterov r2 (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t_b = min(300.0, remaining * 0.3)
    if t_b > 20:
        a_out, _ = nesterov_lp(best_seq, t_b, solver=solver)
        _try_update(a_out, "lp_r2")
    print(f"  Phase 4 done: best={best_score:.8f}")

    # === Phase 5: Smooth-constrained LSE (100s) ===
    print(f"\n=== Phase 5: Smooth LSE (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t_b = min(100.0, remaining * 0.1)
    if t_b > 10:
        a_out, _ = smoothness_constrained_lse(best_seq, t_b, T_start=15.0,
                                               T_end=0.03, n_stages=8,
                                               K_top=20, smooth_lambda=0.005)
        _try_update(a_out, "smooth_lse")
    print(f"  Phase 5 done: best={best_score:.8f}")

    # === Phase 6: Fine LSE polish (100s) ===
    print(f"\n=== Phase 6: Fine LSE polish (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    t_b = min(100.0, remaining * 0.1)
    if t_b > 10:
        a_out, _ = lbgfs_topk(best_seq, t_b, T_start=8.0, T_end=0.02,
                              n_stages=10, K_top=12)
        _try_update(a_out, "fine_lse")
    print(f"  Phase 6 done: best={best_score:.8f}")

    # === Phase 7: Final peak refinement (50s) ===
    remaining = _global_deadline - time.monotonic()
    if remaining > 15:
        t_b = min(50.0, remaining * 0.4)
        if t_b > 8:
            a_out, _ = peak_targeted_refine(best_seq, t_b, max_steps=30000)
            _try_update(a_out, "peak_final")

    # === Phase 8: Length exploration (remaining) ===
    remaining = _global_deadline - time.monotonic()
    if remaining > 40:
        print(f"\n=== Phase 8: Length exploration (score={best_score:.8f}) ===")
        candidates_n = [n + 150, n - 100, n + 300, n - 200]
        for n_try in candidates_n:
            if n_try < 100:
                continue
            remaining = _global_deadline - time.monotonic()
            if remaining < 25:
                break
            t_b = min(50.0, remaining * 0.45)
            if t_b < 12:
                break
            print(f"  Trying n={n_try}, budget={t_b:.0f}s")
            a_interp = interpolate_sequence(best_seq, n_try)
            a_out, _ = nesterov_lp(a_interp, t_b * 0.5, solver=solver)
            _try_update(a_out, f"lp_n{n_try}")
            a_out2, _ = lbgfs_topk(a_interp, t_b * 0.5, T_start=10.0,
                                    T_end=0.05, n_stages=8, K_top=20)
            _try_update(a_out2, f"lse_n{n_try}")
        print(f"  Phase 8 done: best={best_score:.8f}")

    # Final: one more quick peak refinement on the absolute best
    remaining = _global_deadline - time.monotonic()
    if remaining > 10:
        t_b = min(30.0, remaining * 0.8)
        if t_b > 5:
            a_out, _ = peak_targeted_refine(best_seq, t_b, max_steps=15000)
            _try_update(a_out, "peak_very_final")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
