Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

2. Simulated annealing with two-point mass transfer moves

Key innovations: Adam with adaptive LR, SA phase for escaping local minima,

import numpy as np
import time
import math
from scipy.fft import rfft, irfft

def fast_autocorr(a):
    """Full self-convolution a*a using FFT, O(n log n)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u > (cssv - 1.0) / rho
    rho_hat = np.max(rho[cond]) if np.any(cond) else n
    theta = (cssv[rho_hat - 1] - 1.0) / rho_hat
    return np.maximum(v - theta, 0.0)

def lse_gradient(a, T):
    """
    Compute the gradient of (1/T)*log(sum_k exp(T*c[k])) w.r.t. a.
    Uses FFT for O(n log n).
    Returns (lse_value, gradient).
    """
    n = len(a)
    c = fast_autocorr(a)
    Tc = T * c
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    sum_exp = exp_exp = exp_Tc.sum()
    if sum_exp < 1e-300:
        sum_exp = 1e-300
    w = exp_Tc / sum_exp  # softmax weights
    lse_val = (max_Tc + math.log(sum_exp)) / T
    
    # grad[j] = 2 * sum_{m=0}^{n-1} a[m] * w[m+j]
    # = 2 * (flip(a) * w)[n-1+j]
    a_flip = a[::-1].copy()
    conv_len = n + (2 * n - 1) - 1  # = 3n - 2
    fft_size = 1 << (conv_len - 1).bit_length()
    AF = rfft(a_flip, fft_size)
    WF = rfft(w, fft_size)
    conv = irfft(AF * WF, fft_size)
    grad = 2.0 * conv[n - 1:2 * n - 1]  # length n
    
    return lse_val, grad

def adam_lse_optimize(a0, time_limit, T_stages=None, n_iters_per_stage=200, lr_init=0.01):
    """
    Adam optimizer on the LSE objective with multi-stage temperature annealing.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    if T_stages is None:
        T_stages = [500, 300, 200, 150, 100, 75, 50, 35, 25, 18, 13, 10, 7, 5, 4, 3, 2.5, 2, 1.5, 1.2]
    
    # Adam state
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    for si, T in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        # Adaptive learning rate: start higher, decay within stage
        lr = lr_init / (1 + 0.1 * si)
        
        iters = 0
        for it in range(n_iters_per_stage):
            if time.monotonic() - t0 > time_limit:
                break
            if time.monotonic() - t0 > time_limit - remaining * 0.8:  # leave some time
                break
            
            t += 1
            lse_val, grad = lse_gradient(a, T)
            
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            
            # Step in z-space (a = softmax(z))... actually, let's just do gradient step + projection
            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            # Project onto simplex
            a_new = project_simplex(a_new)
            a = a_new
            iters += 1
            
            # Track best using actual max
            if it % 5 == 0:

                if score < best_score:
                    best_score = score
                    best_a = a.copy()
        
        elapsed = time.monotonic() - t0
        print(f"      LSE-Adam stage={si}/T={T}: best={best_score:.8f} ({elapsed:.0f}s, {iters} iters)")
    
    return best_a, best_score

                    if ss < 1e-15:

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-based LogSumExp gradient descent with Adam optimizer and multi-stage annealing
2. Targeted peak shaving via greedy contributor identification
3. Simulated annealing with two-point mass transfer for escaping local minima
4. Coordinate descent with adaptive step sizes for fine-tuning
5. Multi-length exploration with interpolation transfer

Key innovations over previous approach: Adam-based LSE (faster, more robust than L-BFGS),
greedy peak peeling with top-contributor identification, simulated annealing phase,
and more aggressive multi-stage temperature schedules.

import numpy as np
import time
import math
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full self-convolution a*a using FFT, O(n log n)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u > (cssv - 1.0) / rho
    if not np.any(cond):
        rho_hat = n
    else:
        rho_hat = int(np.max(rho[cond]))
    theta = (cssv[rho_hat - 1] - 1.0) / rho_hat
    return np.maximum(v - theta, 0.0)


def lse_gradient(a, T):
    """
    Compute gradient of (1/T)*log(sum_k exp(T*c[k])) w.r.t. a using FFT.
    Returns (lse_value, gradient_array).
    """
    n = len(a)
    c = fast_autocorr(a)
    Tc = T * c
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    sum_exp = exp_Tc.sum()
    if sum_exp < 1e-300:
        sum_exp = 1e-300
    w = exp_Tc / sum_exp
    lse_val = (max_Tc + math.log(sum_exp)) / T

    # grad[j] = 2 * sum_{m=0}^{n-1} a[m] * w[m+j]
    a_flip = a[::-1].copy()
    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    AF = rfft(a_flip, fft_size)
    WF = rfft(w, fft_size)
    conv = irfft(AF * WF, fft_size)
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return lse_val, grad


def adam_lse_optimize(a0, time_limit, T_stages, n_iters_per_stage, lr_init):
    """
    Adam optimizer on LSE objective with multi-stage temperature annealing.
    Projects onto simplex after each step.
    """

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    s = a.sum()
    if s > 0:
        a = a / s

    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    total_iters = 0

    for si, T in enumerate(T_stages):

        remaining = time_limit - (time.monotonic() - t0)

def simulated_annealing(a0, time_limit, T_start=0.003, T_end=0.00005, seed=42):

    """

    t0 = time.monotonic()

        while time.monotonic() - batch_start < 0.5:  # 0.5s batches

            frac = total_moves / max(1.0, total_moves + 10000)

            # Choose move type
            if rng.random() < 0.7:
                # Two-point transfer
                i = int(rng.integers(n))
                j = int(rng.integers(n))

                continue
            for frac in fracs:
                for sign in (1, -1):
                    delta = sign * a[j] * frac
                    if a[j] + delta < -1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[j] += delta
                    a_trial = np.maximum(a_trial, 0)
                    ss = a_trial.sum()
                    if ss < 1e-15:
                        continue
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score - 1e-15:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                continue

        sweep += 1
        if not improved:
            # Try pairwise moves
            for _ in range(20):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or a[i] < 1e-10:
                    continue
                frac = rng.uniform(0.001, 0.04)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                ss = a_trial.sum()
                if ss > 1e-15:
                    a_trial = a_trial / ss
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
            if not improved:
                break

        if sweep % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"      CD sweep={sweep} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def evaluate_safe(a):
    """Safely evaluate a sequence using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = int(rng.integers(0, 8))
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        x = np.linspace(0, 1, n)
        a = np.exp(-3.0 * (x - 0.5) ** 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.1 * np.cos(2 * np.pi * 7 * x)
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 1.5
    elif choice == 4:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * (x - 0.3) ** 2) + 0.7 * np.exp(-4 * (x - 0.7) ** 2)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 4
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = (1 - x) * (0.5 + 0.5 * np.sin(8 * np.pi * x))
    else:
        x = np.linspace(0, 1, n)
        a = 0.3 + 0.4 * np.exp(-5 * (x - 0.4) ** 2) + 0.3 * np.exp(-5 * (x - 0.8) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate():
    """
    Main search: multi-phase optimization combining LSE-Adam, greedy peak shaving,
    simulated annealing, and coordinate descent with multi-length exploration.
    """
    rng = np.random.default_rng(12345)
    t_global = time.monotonic()
    time_budget = 1740.0

    # Get parent sequence
    best_seq = None
    best_score = float('inf')
    n_base = 1600

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            n_base = len(parent)
            p_score = evaluate_safe(parent)
            print(f"Parent: n={n_base}, score={p_score:.8f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        print("No valid parent, using default init")
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s_val = a.sum()
        if s_val < 0.01:
            return
        a = a / s_val
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")

    # === Phase 1: LSE-Adam at parent length ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.40
    print(f"\n=== Phase 1: LSE-Adam at n={n_base} (time={t1:.0f}s) ===")
    T_stages_main = [400, 300, 220, 160, 120, 90, 70, 55, 42, 32, 24, 18, 14, 10, 8, 6, 5, 4, 3, 2.5, 2, 1.5, 1.2, 1.0, 0.8, 0.6]
    a_out, _ = adam_lse_optimize(best_seq, t1, T_stages=T_stages_main, n_iters_per_stage=500, lr_init=0.004)
    try_update(a_out, f"lse_n{n_base}")

    # === Phase 2: Greedy peak shaving ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = remaining * 0.20
    print(f"\n=== Phase 2: Peak shaving at n={n_base} (time={t2:.0f}s) ===")
    if t2 > 10:
        a_out, _ = peak_shave_greedy(best_seq, t2, seed=42)
        try_update(a_out, f"ps_n{n_base}")

    # === Phase 3: Simulated annealing ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = remaining * 0.15
    print(f"\n=== Phase 3: Simulated annealing at n={n_base} (time={t3:.0f}s) ===")
    if t3 > 10:
        a_out, _ = simulated_annealing(best_seq, t3, T_start=0.002, T_end=0.00001, seed=77)
        try_update(a_out, f"sa_n{n_base}")

    # === Phase 4: Coordinate descent ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = remaining * 0.15
    print(f"\n=== Phase 4: Coordinate descent at n={n_base} (time={t4:.0f}s) ===")
    if t4 > 10:
        a_out, _ = coordinate_descent(best_seq, t4, seed=123)
        try_update(a_out, f"cd_n{n_base}")

    # === Phase 5: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Multi-length exploration (remaining={remaining:.0f}s) ===")
    candidate_lengths = sorted(set([n_base + 200, n_base + 400, n_base + 800]), key=lambda x: abs(x - n_base))
    candidate_lengths = [nl for nl in candidate_lengths if 600 <= nl <= 5000]
    time_per_length = min(remaining * 0.7 / max(1, len(candidate_lengths)), 180.0)

    for nl in candidate_lengths:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 30:
            break
        tl = min(time_per_length, remaining * 0.45)
        if tl < 20:
            continue
        print(f"  Trying n={nl} (time={tl:.0f}s)")
        init = interpolate_sequence(best_seq, nl)
        # LSE-Adam
        t_sub = tl * 0.6
        a_out, _ = adam_lse_optimize(init, t_sub,
                                     T_stages=[200, 120, 80, 50, 30, 20, 12, 8, 5, 3, 2, 1.5, 1],
                                     n_iters_per_stage=200, lr_init=0.003)
        try_update(a_out, f"lse_n{nl}")
        # Quick CD
        remaining_sub = time_budget - (time.monotonic() - t_global)
        if remaining_sub > 20:
            a_out2, _ = coordinate_descent(a_out, min(tl * 0.3, remaining_sub * 0.3), seed=nl)
            try_update(a_out2, f"cd_n{nl}")

    # === Phase 6: Final polish at best length ===
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 6: Final polish at n={n_best} (remaining={remaining:.0f}s) ===")

    if remaining > 30:
        # Low-T LSE
        t6a = remaining * 0.3
        a_out, _ = adam_lse_optimize(best_seq, t6a,
                                     T_stages=[5, 3, 2, 1.5, 1, 0.8, 0.6, 0.5, 0.4],
                                     n_iters_per_stage=300, lr_init=0.001)
        try_update(a_out, "final_lse")

        # Final peak shaving
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 20:
            t6b = remaining * 0.3
            a_out, _ = peak_shave_greedy(best_seq, t6b, seed=999)
            try_update(a_out, "final_ps")

        # Final CD
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining > 15:
            t6c = remaining * 0.8
            a_out, _ = coordinate_descent(best_seq, t6c, seed=555)
            try_update(a_out, "final_cd")

    # Second pass: if we have time, try LSE from a different init
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 60:
        print(f"\n=== Phase 7: Alternative init exploration (remaining={remaining:.0f}s) ===")
        alt_init = structured_init(n_base, rng)
        # Blend with current best
        blend = 0.3 * alt_init + 0.7 * best_seq
        blend = np.maximum(blend, 0)
        blend = blend / blend.sum()
        t7 = remaining * 0.5
        a_out, _ = adam_lse_optimize(blend, t7,
                                     T_stages=[100, 60, 40, 25, 15, 10, 7, 5, 3, 2, 1.5, 1],
                                     n_iters_per_stage=300, lr_init=0.002)
        try_update(a_out, "alt_lse")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return result
