```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated LogSumExp optimization (O(n log n) per gradient step)
2. Nesterov LP acceleration for coarse optimization
3. Direct subgradient with backtracking for final polishing
4. Multi-length exploration with interpolation
Key improvement: FFT-based gradient computation enables 100x more
LSE iterations than the previous O(n^2) approach.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_general(x, y):
    n = len(x) + len(y) - 1
    fft_size = 1 << (n - 1).bit_length()
    X = rfft(x, fft_size)
    Y = rfft(y, fft_size)
    return irfft(X * Y, fft_size)[:n]


def score_normalized(a):
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()


def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if len(a_list) == 0:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def lse_obj_grad(a, T):
    """
    Compute LSE-smoothed objective and gradient w.r.t. a.
    obj = (1/T) * log(sum_k exp(T * b_k)) where b = conv(a,a)
    grad = 2 * conv(W, a)[n-1 : 2n-1] where W = softmax(T*b)
    """
    n = len(a)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    W = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = log_sum_exp / T
    
    # Gradient: 2 * conv(W, a)[n-1 : 2n-1]
    conv_Wa = fast_convolve_general(W, a)
    grad = 2.0 * conv_Wa[n-1:2*n-1]
    
    return obj, grad


def lse_obj_grad_z(z, T):
    """LSE objective and gradient in log-domain (z = log(a))."""
    n = len(z)
    # Softmax to get a from z
    z_shift = z - z.max()
    exp_z = np.exp(z_shift)
    a = exp_z / exp_z.sum()
    
    obj_a, grad_a = lse_obj_grad(a, T)
    
    # Chain rule: d obj / d z_i = sum_j (d obj / d a_j) * (d a_j / d z_i)
    # d a_j / d z_i = a_j * (delta_{ij} - a_i)
    # So grad_z_i = a_i * (grad_a_i - sum_j grad_a_j * a_j)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    
    return obj_a, grad_z


def lse_lbfgs(a0, time_limit, T_start=50.0, T_end=1.0, n_stages=30,
              iters_per_stage=50, tol=1e-16):
    """
    L-BFGS optimization on the LSE-smoothed objective with
    adaptive temperature scheduling.
    """
    n = len(a0)
    z = np.log(np.maximum(a0, 1e-300)).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)
    
    # Temperature schedule: geometric from T_start to T_end
    if n_stages <= 1:
        T_schedule = [T_start]
    else:
        ratio = (T_end / T_start) ** (1.0 / (n_stages - 1))
        T_schedule = [T_start * (ratio ** i) for i in range(n_stages)]
    
    prev_score = best_score
    stagnant = 0
    
    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        T = T_schedule[stage]
        
        def func(z_val):
            obj, _ = lse_obj_grad_z(z_val, T)
            return obj
        
        def jac(z_val):
            _, grad = lse_obj_grad_z(z_val, T)
            return grad
        
        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'gtol': 1e-12,
                         'maxfun': iters_per_stage * 200, 'disp': False}
            )
            z = res.x
            a_check = np.exp(z - z.max())
            a_check = a_check / a_check.sum()
            a_check = np.maximum(a_check, 0)
            sc = score_normalized(a_check)
            
            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()
            
            # Adapt temperature
            if sc > prev_score * 1.001:
                stagnant += 1
            else:
                stagnant = max(0, stagnant - 1)
            prev_score = sc
            
            if stagnant >= 3:
                # Reset to a slightly higher temperature
                T_schedule[stage:] = [max(T_schedule[stage], T * 1.5)] * (n_stages - stage)
                stagnant = 0
            
        except Exception as e:
            print(f"      LSE stage {stage} error: {e}")
        
        if (stage + 1) % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE stage {stage+1}/{n_stages} T={T:.1f}: "
                  f"cur={sc:.8f} best={best_score:.8f} ({elapsed:.0f}s)")
    
    return best_a, best_score


def subgradient_descent(a0, time_limit, n_steps=1000000, top_frac=0.1):
    """
    Direct subgradient method on the non-smooth objective.
    Uses FFT for O(n log n) gradient computation.
    """
    n = len(a0)
    a = np.maximum(a0.copy(), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:

def nesterov_lp(a0, time_limit, max_iters=50000):
    """
    Nesterov-accelerated best-response iteration.
    The best response is computed via a simple water-filling / 
    projected gradient step (approximation to the LP solution).
    """
    n = len(a0)
    a = np.maximum(a0.copy(), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s
    
    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Best response: given y, find b that minimizes max_k conv(y,b)_k
        # subject to sum(b)=1, b>=0.
        # Approximate via: b ∝ max(0, 1 - lambda * conv(y, y_at_best))
        # Better: use the gradient of max_k (y*b)_k at the current b
        
        # Actually, let's use a simpler approach: 
        # The best response to y minimizes max_k sum_i y_i * b_{i+k-(n-1)}
        # This is equivalent to: for each k, (T_k b) ≤ M where (T_k b)_i = y_i * b_{i+k-(n-1)}
        # The LP is: min M s.t. sum_i y_i * b_{i+k-(n-1)} ≤ M for all k, sum(b)=1, b≥0
        
        # For efficiency, use a subgradient step on the dual:
        # Start with b = uniform, then optimize
        
        # Actually, let's just do a few steps of projected gradient on b
        # to approximate the best response
        b = y.copy()  # warm start from previous iterate
        
        for _ in range(3):  # 3 inner steps
            conv_yb = fast_convolve_general(y, b)
            max_val = conv_yb.max()
            # Subgradient of max_k conv_yb_k w.r.t. b
            peak = np.argmax(conv_yb)
            # d conv_yb_peak / db_i = y_{peak - i + (n-1)} (with bounds)
            d = peak - (n - 1)
            grad_b = np.zeros(n)
            for i in range(n):
                idx = i + d
                if 0 <= idx < n:
                    grad_b[i] = y[idx]
                idx2 = i - d
                if 0 <= idx2 < n:
                    grad_b[i] += y[idx2]
            
            # Projected gradient step
            step = 0.01
            b_new = b - step * grad_b
            b_new = np.maximum(b_new, 0)
            sb = b_new.sum()
            if sb < 1e-15:
                b_new = b.copy()
            else:
                b_new = b_new / sb
            b = b_new
        
        # Nesterov update
        t_k = (1.0 + np.sqrt(1.0 + 4.0 * (it + 1) ** 2)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k
        
        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b.copy()
        else:
            a_new = a_new / sa
        
        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy
        
        a = a_new
        y = y_new
        
        sc = score_normalized(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
```

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FFT-accelerated LogSumExp (LSE) optimization with L-BFGS - O(n log n) per gradient
2. Adaptive temperature scheduling with stagnation detection and restarts

4. Approximate best-response (Nesterov-style) for diversity

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Fast linear convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_general(x, y):
    """Linear convolution of two arrays via FFT."""
    n = len(x) + len(y) - 1
    fft_size = 1 << (n - 1).bit_length()
    X = rfft(x, fft_size)
    Y = rfft(y, fft_size)
    return irfft(X * Y, fft_size)[:n]


def score_normalized(a):
    """Score assuming sum(a)=1: returns 2*n*max(b)."""
    n = len(a)
    b = fast_convolve(a)
    return 2.0 * n * b.max()

    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    W = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = log_sum_exp / T

    # Gradient: 2 * conv(W, a)[n-1 : 2n-1]
    conv_Wa = fast_convolve_general(W, a)
    grad = 2.0 * conv_Wa[n - 1:2 * n - 1]

    return obj, grad


def lse_obj_grad_z(z, T):
    """LSE objective and gradient in log-domain (z = log(a) + const)."""
    n = len(z)
    z_shift = z - z.max()
    exp_z = np.exp(z_shift)
    a = exp_z / exp_z.sum()

    obj, grad_a = lse_obj_grad_a(a, T)

    # Chain rule for softmax: grad_z_i = a_i * (grad_a_i - dot(grad_a, a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)

    return obj, grad_z


def z_from_a(a):
    """Convert a to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def a_from_z(z):
    """Convert from log-domain to a (softmax)."""
    z_shift = z - z.max()
    exp_z = np.exp(z_shift)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def lse_lbfgs(a0, time_limit, T_start=80.0, T_end=0.5, n_stages=40,
              iters_per_stage=60, restart_every=8):
    """
    L-BFGS on LSE-smoothed objective with adaptive temperature.
    Uses FFT-based O(n log n) gradient computation.
    """
    n = len(a0)
    z = z_from_a(a0).copy()
    t0 = time.monotonic()
    best_a = a0.copy()
    best_score = score_normalized(a0)
    prev_score = best_score
    stagnant = 0
    total_iters = 0

    if n_stages <= 1:
        T_schedule = [T_start]
    else:
        ratio = (T_end / T_start) ** (1.0 / (n_stages - 1))
        T_schedule = [T_start * (ratio ** i) for i in range(n_stages)]

    for stage in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1.5:
            break

        T = T_schedule[stage]

        def func_lse(z_val):
            obj, _ = lse_obj_grad_z(z_val, T)
            return obj

        def jac_lse(z_val):
            _, grad = lse_obj_grad_z(z_val, T)
            return grad

        try:
            res = minimize(
                fun=func_lse, x0=z, jac=jac_lse, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'gtol': 1e-10,
                         'maxfun': iters_per_stage * 300, 'disp': False}
            )
            z = res.x
            total_iters += iters_per_stage
            a_check = a_from_z(z)
            sc = score_normalized(a_check)

            if sc < best_score:
                best_score = sc
                best_a = a_check.copy()

            # Stagnation detection
            if sc > prev_score * 1.0005:
                stagnant += 1
            else:
                stagnant = max(0, stagnant - 1)
            prev_score = sc

            # If very stagnant, perturb and restart
            if stagnant >= restart_every:
                # Perturb in log-space
                rng_p = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
                z = z + 0.05 * rng_p.normal(0, 1, n)
                stagnant = 0
                # Boost temperature
                for i in range(stage, n_stages):
                    T_schedule[i] = max(T_schedule[i], T * 1.3)

        except Exception:
            pass

        if (stage + 1) % 10 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE stage {stage+1}/{n_stages} T={T:.1f}: "
                  f"cur={sc:.8f} best={best_score:.8f} ({elapsed:.0f}s)")

    return best_a, best_score


def subgradient_descent(a0, time_limit, n_steps=2000000, top_frac=0.05):
    """
    Direct subgradient on non-smooth objective with FFT-based gradient.
    Uses Polyak step size with backtracking.
    """
    n = len(a0)
    a = np.maximum(a0.copy(), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    z = z_from_a(a).copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    lower_bound = 2.0 * n / (2.0 * n - 1.0)
    t0 = time.monotonic()
    step_scale = 1.0

    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        cur_score = 2.0 * n * max_b

        # Active set: top fraction
        threshold = max_b * (1.0 - top_frac)
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])
        if len(active) > 80:
            top_idx = np.argsort(b[active])[-80:]
            active = active[top_idx]

        # FFT-based subgradient
        W_active = np.zeros(2 * n - 1)
        W_active[active] = 1.0 / len(active)
        conv_Wa = fast_convolve_general(W_active, a)
        grad_a = 2.0 * conv_Wa[n - 1:2 * n - 1]

        # Log-domain gradient
        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)

        grad_norm_sq = np.dot(grad_z, grad_z)
        if grad_norm_sq < 1e-30:
            break

        # Polyak step
        gap = cur_score - lower_bound
        if gap < 1e-10:
            break
        eta = step_scale * gap / (grad_norm_sq + 1e-15)
        eta = min(eta, 0.05)

        z_new = z - eta * grad_z
        a_new = a_from_z(z_new)
        sc_new = score_normalized(a_new)

        # Backtracking
        if sc_new > cur_score * 1.002:
            for _ in range(4):
                eta *= 0.5
                z_new = z - eta * grad_z
                a_new = a_from_z(z_new)
                sc_new = score_normalized(a_new)
                if sc_new <= cur_score:
                    break
            step_scale *= 0.6
        else:
            step_scale = min(step_scale * 1.05, 2.0)

        z = z_new
        a = a_new

        if sc_new < best_score:
            best_score = sc_new
            best_a = a.copy()

        if (step + 1) % 100000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      SG step={step+1} ({elapsed:.0f}s): "
                  f"cur={cur_score:.8f} best={best_score:.8f}")

    return best_a, best_score


def best_response_approx(y, n_inner=5):
    """
    Approximate best response to y: find b minimizing max_k conv(y,b)_k
    subject to sum(b)=1, b>=0. Uses subgradient iterations.
    """
    n = len(y)
    b = y.copy()

    for _ in range(n_inner):
        conv_yb = fast_convolve_general(y, b)
        # Subgradient at the peak
        peak = int(np.argmax(conv_yb))
        d = peak - (n - 1)
        grad_b = np.zeros(n)
        lo = max(0, -d)
        hi = min(n, n - d)
        for i in range(lo, hi):
            idx = i + d
            if 0 <= idx < n:
                grad_b[i] += y[idx]
        d2 = -d
        lo2 = max(0, -d2)
        hi2 = min(n, n - d2)
        for i in range(lo2, hi2):
            idx = i + d2
            if 0 <= idx < n:
                grad_b[i] += y[idx]

        # Projected gradient step
        step = 0.05
        b_new = b - step * grad_b
        b_new = np.maximum(b_new, 0)
        sb = b_new.sum()
        if sb < 1e-15:
            b_new = b.copy()
        else:
            b_new = b_new / sb
        b = b_new

    return b


def nesterov_best_response(a0, time_limit, max_iters=30000):
    """Nesterov-accelerated best-response iteration."""
    n = len(a0)
    a = np.maximum(a0.copy(), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b = best_response_approx(y, n_inner=4)

        t_k = (1.0 + np.sqrt(1.0 + 4.0 * (it + 1) ** 2)) / 2.0
        alpha = 2.0 / (t_k + 1.0)
        momentum = (t_k - 1.0) / t_k

        a_old = a.copy()
        a_new = (1.0 - alpha) * a + alpha * b
        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b.copy()
        else:
            a_new = a_new / sa

        y_new = a_new + momentum * (a_new - a_old)
        y_new = np.maximum(y_new, 0)
        sy = y_new.sum()
        if sy < 1e-15:
            y_new = a_new.copy()
        else:
            y_new = y_new / sy

        a = a_new
        y = y_new

        sc = score_normalized(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_a, best_score


def structured_init(n, method=0):
    """Generate structured initial sequences."""
    if method == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif method == 1:
        x = np.linspace(0, 1, n)
        a = np.cos(np.pi * (x - 0.5)) ** 4
    elif method == 2:
        x = np.linspace(-1, 1, n)
        a = np.exp(-3 * x ** 2)
    elif method == 3:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    elif method == 4:
        x = np.linspace(0, 1, n)
        a = 0.3 + 0.7 * (x ** 2 + (1 - x) ** 2)
    elif method == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-20 * (x - 0.3) ** 2) + np.exp(-20 * (x - 0.7) ** 2) + 0.1
    elif method == 6:
        x = np.linspace(0.01, 1, n)
        a = x ** 0.5 + (1 - x) ** 0.5
    elif method == 7:
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.5) ** 2) / 0.08)
    elif method == 8:
        # Try to match a "flat convolution" shape
        x = np.linspace(0, 1, n)
        a = 1.0 / (1.0 + 50.0 * np.sin(np.pi * x) ** 2)
    else:
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - 0.5) ** 2) / 0.15) + 0.5 * np.exp(-((x - 0.2) ** 2) / 0.03) + 0.5 * np.exp(-((x - 0.8) ** 2) / 0.03)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
