"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Mirror descent with Nesterov momentum and adaptive multi-peak subgradient
2. IAM (Iterative Amplitude Modification) / Gerchberg-Saxton spectral flattening

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_autocorr(a):
    """Full autocorrelation a*a using FFT. Returns array of length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_score(a):
    """Compute 2n*max(b)/(sum(a))^2 quickly."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)


def compute_subgradient(a, b, k_idx, weights):
    """Compute weighted subgradient at top peaks."""
    n = len(a)
    grad = np.zeros(n)
    for ii in range(len(k_idx)):
        m = int(k_idx[ii])
        w_m = weights[ii] * 2.0
        j_start = max(0, m - n + 1)
        j_end = min(n - 1, m)
        js = np.arange(j_start, j_end + 1)
        grad[js] += w_m * a[m - js]
    return grad


def mirror_descent_momentum(a0, time_limit, eta0=0.05, beta=0.7, decay_rate=0.6,
                            k_min=1, k_max=20, k_ramp=500, seed=42):
    """Mirror descent with Nesterov momentum."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    v = np.zeros(n)  # momentum
    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall_count = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        b_max = b.max()
        b_mean = b.sum() / len(b)

        frac = min(1.0, t / float(k_ramp))
        k = max(k_min, min(k_max, int(k_min + frac * (k_max - k_min))))

        top_idx = np.argsort(b)[::-1][:k]
        excess = b[top_idx] - b_mean
        if excess.max() < 1e-15:
            weights = np.ones(k) / k
        else:
            excess_shift = excess - excess.max()
            weights = np.exp(excess_shift) / np.exp(excess_shift).sum()

        grad = compute_subgradient(a, b, top_idx, weights)

        eta = eta0 / (1.0 + t / 500.0) ** decay_rate

        # Nesterov-style momentum
        v = beta * v - eta * grad
        a_new = a * np.exp(-v)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a + 0.01 * rng.standard_normal(n)
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
        else:
            a_new = a_new / s_new

        a = a_new

        if t < 300:
            check_interval = 2
        elif t < 3000:
            check_interval = 5
        else:
            check_interval = 20

        if t % check_interval == 0:
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stall_count = 0
            else:
                stall_count += 1

        if stall_count > 300:
            noise_scale = min(0.03, 0.005 * (stall_count / 100.0))
            a = a + noise_scale * rng.standard_normal(n)
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
            stall_count = 0

    return best_a, best_score


def iam_optimize(a0, time_limit, alpha=0.7, seed=42):
    """
    IAM (Iterative Amplitude Modification) / Gerchberg-Saxton.
    Alternates between:
    - Frequency domain: compress dynamic range of |A|
    - Time domain: clip to non-negative
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        # Forward FFT
        A = rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)

        # Compress dynamic range: mag_new = mag^alpha (alpha < 1 compresses)
        mag_max = mag.max()
        if mag_max < 1e-15:
            break
        mag_norm = mag / mag_max
        mag_new = mag_norm ** alpha * mag_max

        # Inverse FFT with modified magnitude
        A_new = mag_new * np.exp(1j * phase)
        a_new = irfft(A_new, fft_size)[:n]

        # Clip to non-negative
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a = a_new

        # Check score periodically
        if t < 100:
            check_interval = 5
        else:
            check_interval = 20

        if t % check_interval == 0:
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a, best_score


def lse_obj_grad_z(z, T, n):
    """Objective and gradient in z-space where a = softmax(z)."""
    z_shift = z - z.max()
    e = np.exp(z_shift)
    a = e / e.sum()

    b = fast_autocorr(a)
    Tb = T * b
    Tb_shift = Tb - Tb.max()
    exp_Tb = np.exp(Tb_shift)
    obj = (Tb.max() + np.log(exp_Tb.sum())) / T

    w = exp_Tb / exp_Tb.sum()

    # grad_a[j] = 2 * sum_m w[m] * a[m-j] (when valid)
    grad_a = np.zeros(n)
    for m in range(len(w)):
        if w[m] < 1e-10:
            continue
        j_start = max(0, m - n + 1)
        j_end = min(n - 1, m)
        js = np.arange(j_start, j_end + 1)
        grad_a[js] += 2.0 * w[m] * a[m - js]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)

    return obj, grad_z


def lse_optimize(a0, time_limit, T_start=80.0, T_end=0.3, n_stages=15, seed=42):
    """LogSumExp-smoothed optimization with L-BFGS and multi-stage annealing."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s

    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = fast_score(a_init)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        iters = max(20, min(100, int(25 + 50 * si / max(1, n_stages))))

        def func(z_in):
            val, _ = lse_obj_grad_z(z_in, stage_T, n)
            return val

        def jac(z_in):
            _, g = lse_obj_grad_z(z_in, stage_T, n)
            return g

        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': iters, 'ftol': 1e-18,
                                    'gtol': 1e-12, 'maxfun': iters * 300,
                                    'disp': False})
            z = res.x
            z_shift = z - z.max()
            a = np.exp(z_shift)
            a = a / a.sum()
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        except Exception:
            pass

    return best_a, best_score


def peak_shadvanced(a0, time_limit, seed=123):
    """
    Advanced peak shaving: uses quadratic local model to determine
    optimal mass transfers, and tries multi-position moves.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:min(5, n)]
        b_max = b[top_k[0]]

        improved = False
        for k_idx in top_k[:3]:
            k_idx = int(k_idx)
            js = max(0, k_idx - n + 1)
            je = min(n - 1, k_idx)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_idx - idx]

            if contrib.sum() < 1e-15:
                continue

            # Find the top contributors
            top_contrib_idx = np.argsort(contrib)[::-1][:5]
            
            for ci in top_contrib_idx:
                src = int(idx[ci])
                
                # Find good targets: positions that would reduce the peak
                # A position dst helps if it's far from the peak region
                peak_mask = np.zeros(n, dtype=bool)
                for k2 in top_k[:3]:
                    k2 = int(k2)
                    js2 = max(0, k2 - n + 1)
                    je2 = min(n - 1, k2)
                    peak_mask[js2:je2 + 1] = True

                candidates = np.where(~peak_mask)[0]
                if len(candidates) < 10:
                    candidates = np.where(np.arange(n) != src)[0]
                
                # Prefer low-a candidates (less likely to create new peaks)
                sorted_candidates = candidates[np.argsort(a[candidates])[:max(10, len(candidates)//8)]]
                if len(sorted_candidates) == 0:
                    continue
                
                dst = int(sorted_candidates[rng.integers(len(sorted_candidates))])

                # Try multiple transfer fractions
                for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.04]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    sc = fast_score(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial.copy()
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break

                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation
            if stall % 10 == 0:
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.002, 0.015)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= 0:
                        a_trial = a_trial / a_trial.sum()
                        sc = fast_score(a_trial)
                        if sc < best_score:
                            a = a_trial.copy()
                            best_score = sc
                            best_a = a.copy()
                            stall = 0

            if stall > 500:
                break

    return best_a, best_score


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def spline_init(n, rng, n_control=50):
    """Initialize using a smooth spline with random control points."""
    x = np.linspace(0, 1, n)
    ctrl_x = np.linspace(0, 1, n_control)
    # Random smooth control values
    ctrl_vals = 1.0 + 0.3 * np.sin(2 * np.pi * np.arange(n_control) / n_control * rng.integers(1, 5))
    ctrl_vals += 0.1 * rng.standard_normal(n_control)
    ctrl_vals = np.maximum(ctrl_vals, 0.01)
    a = np.interp(x, ctrl_x, ctrl_vals)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Mirror descent with Nesterov momentum and adaptive multi-peak subgradient
2. IAM (Iterative Amplitude Modification) / Gerchberg-Saxton spectral flattening
3. FFT-based LogSumExp gradient optimization (L-BFGS)
4. Advanced peak shaving with multi-position moves
5. Multi-length exploration with cross-length transfer
Key innovations: momentum-accelerated MD, frequency-domain spectral shaping via IAM,
quadratic-aware peak shaving, and adaptive resolution scaling.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_autocorr(a):
    """Full autocorrelation a*a using FFT. Returns array of length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_score(a):
    """Compute 2n*max(b)/(sum(a))^2 quickly."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)


def compute_subgradient(a, k_idx, weights):
    """Compute weighted subgradient at top peaks."""
    n = len(a)
    grad = np.zeros(n)
    for ii in range(len(k_idx)):
        m = int(k_idx[ii])
        w_m = weights[ii] * 2.0
        j_start = max(0, m - n + 1)
        j_end = min(n - 1, m)
        js = np.arange(j_start, j_end + 1)
        grad[js] += w_m * a[m - js]
    return grad


def mirror_descent_momentum(a0, time_limit, eta0=0.04, beta=0.7, decay_rate=0.55,
                            k_min=1, k_max=15, k_ramp=400, seed=42):
    """Mirror descent with Nesterov momentum."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    v = np.zeros(n)
    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall_count = 0

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        b_mean = b.sum() / len(b)

        frac = min(1.0, t / float(k_ramp))
        k = max(k_min, min(k_max, int(k_min + frac * (k_max - k_min))))

        top_idx = np.argsort(b)[::-1][:k]
        excess = b[top_idx] - b_mean
        if excess.max() < 1e-15:
            weights = np.ones(k) / k
        else:
            excess_shift = excess - excess.max()
            weights = np.exp(excess_shift) / np.exp(excess_shift).sum()

        grad = compute_subgradient(a, top_idx, weights)
        eta = eta0 / (1.0 + t / 500.0) ** decay_rate

        v = beta * v - eta * grad
        a_new = a * np.exp(-v)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a + 0.01 * rng.standard_normal(n)
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
        else:
            a_new = a_new / s_new

        a = a_new

        if t < 300:
            check_interval = 2
        elif t < 3000:
            check_interval = 5
        else:
            check_interval = 15

        if t % check_interval == 0:
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stall_count = 0
            else:
                stall_count += 1

        if stall_count > 300:
            noise_scale = min(0.03, 0.005 * (stall_count / 100.0))
            a = a + noise_scale * rng.standard_normal(n)
            a = np.maximum(a, 0)
            a = a / a.sum()
            v = np.zeros(n)
            stall_count = 0

    return best_a, best_score


def iam_optimize(a0, time_limit, alpha=0.7, seed=42):
    """
    IAM (Iterative Amplitude Modification) / Gerchberg-Saxton spectral flattening.
    Alternates between frequency-domain magnitude compression and time-domain clipping.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 2).bit_length()
    stall = 0

    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break

        A = rfft(a, fft_size)
        mag = np.abs(A)
        phase = np.angle(A)

        mag_max = mag.max()
        if mag_max < 1e-15:
            break
        mag_norm = mag / mag_max
        mag_new = mag_norm ** alpha * mag_max

        A_new = mag_new * np.exp(1j * phase)
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        a = a_new

        if t < 200:
            check_interval = 5
        elif t < 2000:
            check_interval = 15
        else:
            check_interval = 30

        if t % check_interval == 0:
            sc = fast_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stall = 0
            else:
                stall += 1

        if stall > 200:
            a = a + 0.01 * rng.standard_normal(n)
            a = np.maximum(a, 0)
            a = a / a.sum()
            stall = 0

    return best_a, best_score


def lse_obj_grad_z(z, T, n):
    """Objective and gradient in z-space where a = softmax(z)."""
    z_shift = z - z.max()
    e = np.exp(z_shift)
    a = e / e.sum()

    b = fast_autocorr(a)
    Tb = T * b
    Tb_shift = Tb - Tb.max()
    exp_Tb = np.exp(Tb_shift)
    obj = (Tb.max() + np.log(exp_Tb.sum())) / T

    w = exp_Tb / exp_Tb.sum()

    grad_a = np.zeros(n)
    for m in range(len(w)):
        if w[m] < 1e-12:
            continue
        j_start = max(0, m - n + 1)
        j_end = min(n - 1, m)
        js = np.arange(j_start, j_end + 1)
        grad_a[js] += 2.0 * w[m] * a[m - js]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)

    return obj, grad_z


def lse_optimize(a0, time_limit, T_start=50.0, T_end=0.3, n_stages=15, seed=42):
    """LogSumExp-smoothed optimization with L-BFGS and multi-stage annealing."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s

    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = fast_score(a_init)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 10:
            break

        iters = max(20, min(100, int(25 + 50 * si / max(1, n_stages))))

        def func(z_in):
            val, _ = lse_obj_grad_z(z_in, stage_T, n)
            return val

        def jac(z_in):
            _, g = lse_obj_grad_z(z_in, stage_T, n)
            return g

        try:
            res = minimize(fun=func, x0=z, jac=jac, method='L-BFGS-B',
                           options={'maxiter': iters, 'ftol': 1e-18,
                                    'gtol': 1e-12, 'maxfun': iters * 300,
                                    'disp': False})
            z = res.x
            z_shift = z - z.max()
