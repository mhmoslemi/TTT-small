        perm = rng.permutation(n)

        for idx in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            i = perm[idx]
            # Try a few random j's
            for attempt in range(3):
                j = int(rng.integers(n))
                if j == i:
                    continue
                if a[i] < norm * 1e-8:
                    continue

                # Compute gain vector g_k for transferring delta from i to j
                # First order: change in conv_k = delta * (a_{k-j} + a_{j-k} - a_{k-i} - a_{i-k})
                ai = a[i]
                g = np.zeros(size)
                # a_{k-i} for k in [i, i+n-1] -> a[0:n] shifted by i
                lo1 = max(0, i)
                hi1 = min(size, i + n)
                if hi1 > lo1:
                    g[lo1:hi1] += a[max(0, lo1 - i):max(0, hi1 - i)]
                lo2 = max(0, n - 1 - i)
                hi2 = min(size, n + i)
                if hi2 > lo2:
                    idx2 = np.arange(lo2, hi2)
                    src2 = n - 1 - idx2
                    valid = (src2 >= 0) & (src2 < n)
                    g[lo2:hi2][valid] += a[src2[valid]]

                lo3 = max(0, j)
                hi3 = min(size, j + n)
                if hi3 > lo3:
                    g[lo3:hi3] -= a[max(0, lo3 - j):max(0, hi3 - j)]
                lo4 = max(0, n - 1 - j)
                hi4 = min(size, n + j)
                if hi4 > lo4:
                    idx4 = np.arange(lo4, hi4)
                    src4 = n - 1 - idx4
                    valid4 = (src4 >= 0) & (src4 < n)
                    g[lo4:hi4][valid4] -= a[src4[valid4]]

                # Find optimal delta in [0, ai] minimizing max_k(conv_k + delta*g_k)
                # Plus second order: at k=2i, term is (ai-delta)^2 vs ai^2, change = -2*ai*delta + delta^2
                # At k=2j, term is (aj+delta)^2 vs aj^2, change = 2*aj*delta + delta^2
                # For simplicity use first order + check boundaries

                # Evaluate at several delta values
                best_d = 0.0
                best_val = conv.max()
                n_samples = 20
                deltas = np.linspace(0.0, ai, n_samples)
                for d in deltas:
                    if d == 0:
                        continue
                    new_max = (conv + d * g).max()
                    # Second order correction for k=2i and k=2j
                    if 2 * i < size:
                        new_max_2i = conv[2 * i] - 2 * ai * d + d * d
                        new_max = max(new_max, new_max_2i)
                    if 2 * j < size:
                        new_max_2j = conv[2 * j] + 2 * a[j] * d + d * d
                        new_max = max(new_max, new_max_2j)
                    if new_max < best_val - 1e-12:
                        best_val = new_max
                        best_d = d

                if best_d > 0:
                    a[i] -= best_d
                    a[j] += best_d
                    s = a.sum()
                    if s < 1e-15:
                        a[i] += best_d
                        a[j] -= best_d
                        continue
                    a = a * norm / s
                    conv = fast_autocorr(a)
                    cur_max = conv.max()
                    if cur_max < best_score - 1e-12:
                        best_score = cur_max
                        best_a = a.copy()
                        pass_improved = True
                        no_improve = 0
                    else:
                        # Revert
                        a[j] -= best_d
                        a[i] += best_d
                        a = a * norm / a.sum()
                        conv = fast_autocorr(a)

        if not pass_improved:
            no_improve += 1
            if no_improve >= 3:
                break

        if verbose and n_passes % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PCD pass={n_passes} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def lse_gradient_optimize(a0, time_limit, verbose=True):
    """LSE (log-sum-exp) smooth approximation gradient descent."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()

    # Anneal temperature
    n_stages = 8
    temps = np.geomspace(20.0, 0.5, n_stages)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    for si, T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 1:
            break
        n_steps = max(10, min(100, int(remaining * 3)))

        for it in range(n_steps):
            if time.monotonic() - t0 > time_limit:
                break
            conv = fast_autocorr(a)
            # LSE weights
            max_conv = conv.max()
            exp_conv = np.exp((conv - max_conv) / T)
            w = exp_conv / exp_conv.sum()

            # Gradient: grad_i = 2 * sum_k w_k * (a_{k-i} + a_{i-k})
            # Use FFT: grad = 2 * (w * a + w * a_flipped) projected to length n
            A = rfft(a, fft_size)
            W = rfft(w, fft_size)
            conv_wa = irfft(A * W, fft_size)[:size]
            # a_{k-i}: this is the cross-correlation of w and a
            # grad_i = 2 * sum_k w_k * a_{k-i} + 2 * sum_k w_k * a_{i-k}
            # sum_k w_k * a_{k-i} = conv(w, a)[i] (full conv, position i)
            wa_conv = fast_autocorr_helper(w, a, fft_size)
            # Actually let's just compute directly
            # sum_k w_k * a_{k-i}: for fixed i, sum over k where 0<=k-i<n
            # This is the (i)-th element of full conv(w, a)
            wa = irfft(rfft(w, fft_size) * A, fft_size)[:size]
            # wa[k] = sum_j w_j * a[k-j], so wa[i] = sum_j w_j * a[i-j]
            # We want sum_k w_k * a[k-i] = wa[n-1-i] (by reversal)
            # Actually: sum_k w_k * a[k-i]. Let m=k-i, then k=m+i. sum_m w_{m+i} * a_m.
            # This is the correlation of w (shifted) with a.
            # In terms of convolution: (w * a_flipped)[n-1-i] where a_flipped = a[::-1]
            # Let's just compute: grad_i = 2 * (sum_k w_k * a_{k-i} + sum_k w_k * a_{i-k})
            # sum_k w_k * a_{i-k} = conv(w,a)[i] (with appropriate indexing)
            # For the full convolution conv(w,a) of length 2n-1:
            # conv(w,a)[k] = sum_j w_j * a[k-j]
            # sum_k w_k * a_{i-k} = sum_k w_k * a[i-k] -- let j=k, = sum_j w_j * a[i-j] = conv(w,a)[i]
            # sum_k w_k * a_{k-i} -- let l=k-i, k=l+i: sum_l w_{l+i} * a_l
            # This is conv(w_rev, a_rev)[...] -- getting complicated.
            # Simpler: use the fact that for the full conv of w and a (length 2n-1),
            # element at position k is sum_j w_j * a_{k-j} (valid j).
            # So sum_k w_k * a_{k-i} = element at position (something) of conv of reversed stuff.
            # Let me just do it directly with the FFT approach:
            # grad = 2 * (cross_corr(w,a) + cross_corr(w, a[::-1]))[0:n]
            # where cross_corr is computed via FFT.

            # Compute sum_k w_k * a_{k-i} for all i:
            # This is the cross-correlation of w with a, evaluated at lag i.
            # cross_corr_wa[i] = sum_k w_k * a_{k+i} (with valid indices)
            # Using FFT: ifft(fft(w) * conj(fft(a))) gives the cross-correlation.
            W_fft = rfft(w, fft_size)
            A_fft = rfft(a, fft_size)
            # cross_corr = ifft(W_fft * conj(A_fft)) -- but we need to be careful with one-sided FFT
            # For real signals: cross_corr[i] = sum_k w[k] * a[k+i]
            # = ifft(FFT(w) * conj(FFT(a)))
            cross1 = irfft(W_fft * np.conj(A_fft), fft_size)
            # cross1[i] = sum_k w[k]*a[k+i] for i>=0 (circular correlation)
            # We need the linear version, but for our purposes the circular version
            # is close enough if fft_size > 2n.
            g1 = cross1[:n]  # sum_k w_k * a_{k+i} for i=0..n-1

            # sum_k w_k * a_{i-k} = cross_corr at negative lag = cross_corr of a with w at lag i
            cross2 = irfft(A_fft * np.conj(W_fft), fft_size)
            g2 = cross2[fft_size - n + 1:fft_size + 1] if fft_size >= n else np.zeros(n)
            # Actually: cross2[i] = sum_k a[k]*w[k+i], so cross2[fft_size-i] for negative lag
            # sum_k w_k * a_{i-k} = sum_k a_{i-k} * w_k. Let j=i-k: sum_j a_j * w_{i-j} = cross2[i]
            # Wait: cross2 = ifft(FFT(a) * conj(FFT(w))), so cross2[m] = sum_k a[k]*w[k+m]
            # We want sum_k w_k * a_{i-k}. Let j = i-k: sum_j w_{i-j} * a_j = sum_j a_j * w_{i-j}
            # = cross2 evaluated at... cross2[m] = sum_k a[k] w[k+m]. Set k=j: sum_j a[j] w[j+m].
            # We want sum_j a[j] w[i-j] = sum_j a[j] w[-(j-i)] = cross2 at m = -(j-i)...
            # This is getting confusing. Let me just use a simpler approach.

            # Direct computation via the convolution matrix (fine for n <= 4000)
            grad = np.zeros(n)
            for i in range(n):
                # sum_k w_k * a_{k-i}: valid k from max(0,i) to min(n-1, size-1+i)
                k_lo = max(0, i)
                k_hi = min(n, i + n)  # k-i must be in [0,n-1], so k in [i, i+n-1]
                if k_hi > k_lo:
                    grad[i] += np.dot(w[k_lo:k_hi], a[k_lo - i:k_hi - i])
                # sum_k w_k * a_{i-k}: valid k from max(0, i-n+1) to min(n-1, i)
                # i-k in [0,n-1] means k in [i-n+1, i]
                k_lo2 = max(0, i - n + 1)
                k_hi2 = min(n, i + 1)
                if k_hi2 > k_lo2:
                    grad[i] += np.dot(w[k_lo2:k_hi2], a[i - np.arange(k_lo2, k_hi2)])
            grad *= 2.0

            # Projected gradient step on simplex {a >= 0, sum(a) = norm}
            step = norm / (T * n) / (1.0 + 0.05 * it)
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = a.copy()
            else:
                a_new = a_new * norm / s
            a = a_new

            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage T={T:.1f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def fast_autocorr_helper(w, a, fft_size=None):
    if fft_size is None:
        n = len(a)
        size = len(w) + n - 1
        fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    return irfft(W * A, fft_size)


def structured_init(n, rng):
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-6.0 * (x - 0.3) ** 2) + 0.7 * np.exp(-6.0 * (x - 0.7) ** 2)
    elif choice == 4:
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 3
    else:
        x = np.linspace(0.0, 1.0, n)
        a = 0.3 + 0.7 * np.sin(np.pi * x) ** 0.7
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n)
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n)
    return f_interp / ss


def subgrad_fw(a0, time_limit, verbose=False):
    """Frank-Wolfe with subgradient (fast, no LP needed)."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    size = 2 * n - 1

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0

    for t in range(1, 100000000):
        if time.monotonic() - t0 > time_limit:
            break

        conv_y = fast_autocorr(y)
        k_star = int(np.argmax(conv_y))

        # Subgradient of max_k conv(a,a)_k w.r.t. a_i
        # d/d a_i conv_k = a_{k-i} + a_{i-k} (for valid indices)
        grad = np.zeros(n)
        # For k = k_star:
        for i in range(n):
            v = 0.0
            j1 = k_star - i
            if 0 <= j1 < n:
                v += y[j1]
            j2 = i - k_star
            if 0 <= j2 < n:
                v += y[j2]
            grad[i] = v
        grad *= 2.0

        # FW step: go to e_{argmin grad} * norm
        i_min = int(np.argmin(grad))
        v = np.zeros(n)
        v[i_min] = norm

        alpha = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha) * y + alpha * v
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = v.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        score = fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 50 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                sc = fast_autocorr(avg_check).max()
                if sc < best_score:
                    best_score = sc
                    best_a = avg_check.copy()

        if verbose and t % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FW t={t} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1700.0)

    solver = get_solver()
    print(f"Time budget: {time_budget:.0f}s")

    best_seq = None
    best_score = float('inf')
    parent = None

    # Try to load parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def do_try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # Initialize
    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")
    else:
        n_base = len(best_seq)

    # Phase 1: LP Nesterov at base length (main workhorse)
    remaining = time_budget - (time.monotonic() - t_global)
    n1 = n_base
    print(f"\n=== Phase 1: LP Nesterov n={n1} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.35, 500.0)
    if t1 > 30:
        a_norm = best_seq.copy()
        s_a = a_norm.sum()
        if s_a > 1e-15:
            a_norm = a_norm * np.sqrt(2.0 * n1) / s_a
        a_out, _ = nesterov_lp_optimize(a_norm, t1, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n1}")

    # Phase 2: Pair coordinate descent (key improvement)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 2: Pair CD n={n_best} (budget={remaining:.0f}s) ===")
    t2 = min(remaining * 0.20, 250.0)
    if t2 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = pair_coordinate_descent(a_norm, t2, rng, verbose=True)
        do_try_update(a_out, f"pcd_n{n_best}")

    # Phase 3: Try longer length
    n2 = n_base + 400
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: LP Nesterov n={n2} (budget={remaining:.0f}s) ===")
    t3 = min(remaining * 0.20, 300.0)
    if t3 > 30:
        init_long = interpolate_sequence(best_seq, n2)
        a_norm = init_long * np.sqrt(2.0 * n2) / init_long.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t3, solver, verbose=True)
        do_try_update(a_out, f"lp_n{n2}")

    # Phase 4: Pair CD on longer
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Pair CD n={n_best} (budget={remaining:.0f}s) ===")
    t4 = min(remaining * 0.15, 200.0)
    if t4 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = pair_coordinate_descent(a_norm, t4, rng, verbose=True)
        do_try_update(a_out, f"pcd2_n{n_best}")

    # Phase 5: LSE gradient on best
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 5: LSE gradient n={n_best} (budget={remaining:.0f}s) ===")
    t5 = min(remaining * 0.12, 150.0)
    if t5 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = lse_gradient_optimize(a_norm, t5, verbose=True)
        do_try_update(a_out, f"lse_n{n_best}")

    # Phase 6: Shorter length exploration (sometimes shorter is better)
    n_short = max(800, n_base - 400)
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: LP Nesterov n={n_short} (budget={remaining:.0f}s) ===")
    t6 = min(remaining * 0.12, 120.0)
    if t6 > 20:
        init_short = interpolate_sequence(best_seq, n_short)
        a_norm = init_short * np.sqrt(2.0 * n_short) / init_short.sum()
        a_out, _ = nesterov_lp_optimize(a_norm, t6, solver, verbose=False)
        do_try_update(a_out, f"lp_n{n_short}")
        # Also try from scratch
        a_init2 = structured_init(n_short, rng)
        a_norm2 = a_init2 * np.sqrt(2.0 * n_short) / a_init2.sum()
        t6b = min(time_budget - (time.monotonic() - t_global) - 120.0, 60.0)
        if t6b > 10:
            a_out2, _ = nesterov_lp_optimize(a_norm2, t6b, solver, verbose=False)
            do_try_update(a_out2, f"lp_n{n_short}_fresh")

    # Phase 7: Final pair CD
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 7: Final Pair CD n={n_best} (budget={remaining:.0f}s) ===")
    t7 = min(remaining * 0.25, 100.0)
    if t7 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = pair_coordinate_descent(a_norm, t7, rng, verbose=True)
        do_try_update(a_out, f"pcd_final_n{n_best}")

    # Phase 8: FW subgradient (fast, can run many iterations)
    remaining = time_budget - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 8: FW subgrad n={n_best} (budget={remaining:.0f}s) ===")
    t8 = min(remaining * 0.15, 80.0)
    if t8 > 10:
        a_norm = best_seq * np.sqrt(2.0 * n_best) / best_seq.sum()
        a_out, _ = subgrad_fw(a_norm, t8, verbose=False)
        do_try_update(a_out, f"fw_n{n_best}")

    # Final check
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
