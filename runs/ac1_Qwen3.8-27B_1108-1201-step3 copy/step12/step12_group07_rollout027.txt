def fast_convolve(a):

def build_conv_matrix(a_fixed):

def solve_lp_best_response(a_fixed, solver=None):
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
            if alt == solver:

                    break

                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None

def mw_best_response(a_fixed, T_inner=80, eta=None):
    """Fast multiplicative weights approximate best response."""
    n = len(a_fixed)
    b = np.ones(n) / n
    if eta is None:
        eta = 2.0 / np.sqrt(n)
    b_avg = np.zeros(n)
    for t in range(T_inner):
        fft_size = 1 << (2 * n - 1).bit_length()
        A = rfft(a_fixed, fft_size)
        B = rfft(b, fft_size)
        c = irfft(A * B, fft_size)[:2 * n - 1]
        k_star = int(np.argmax(c))
        grad = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        grad[idx] = a_fixed[k_star - idx]
        b = b * np.exp(-eta * grad)
        s = b.sum()
        if s < 1e-300:
            b = np.ones(n) / n
        else:
            b = b / s
        b_avg += b
    return b_avg / T_inner

def nesterov_lp_adaptive(a0, time_limit, solver=None, max_iters=1000000, line_search=True):
    """Nesterov LP with adaptive restarts and optional line search."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    a = a / s

    a_prev = a.copy()
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()
    t_nest = 1.0
    last_print = t0
    restart_count = 0
    prev_score = best_score

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        # Extrapolation
        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            y = a + momentum * (a - a_prev)
            y = np.maximum(y, 0)
            sy = y.sum()
            if sy < 1e-15:
                y = a.copy()
            else:
                y = y / sy
        else:
            y = a

        b_new, M_lp = solve_lp_best_response(y, solver)
        if b_new is None:
            continue

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        alpha = 2.0 / (t_new + 1.0)

        if line_search:
            # Try several step sizes and pick the best
            best_alpha_used = alpha
            best_sc_ls = score_normalized((1.0 - alpha) * a + alpha * b_new)
            for frac in [0.5, 0.75, 1.25, 1.5]:
                alpha_try = alpha * frac
                if alpha_try <= 0 or alpha_try > 1.5:
                    continue
                candidate = (1.0 - alpha_try) * a + alpha_try * b_new
                candidate = np.maximum(candidate, 0)
                sc_candidate = candidate.sum()
                if sc_candidate < 1e-15:
                    continue
                candidate = candidate / sc_candidate
                sc_try = score_normalized(candidate)
                if sc_try < best_sc_ls:
                    best_sc_ls = sc_try
                    best_alpha_used = alpha_try
            a_new = (1.0 - best_alpha_used) * a + best_alpha_used * b_new
        else:
            a_new = (1.0 - alpha) * a + alpha * b_new

        a_new = np.maximum(a_new, 0)
        sa = a_new.sum()
        if sa < 1e-15:
            a_new = b_new.copy()
        else:
            a_new = a_new / sa

        sc_new = score_normalized(a_new)

        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()

        # Adaptive restart based on gap
        sc_cur = score_normalized(a)
        improvement = sc_cur - sc_new
        if sc_new > sc_cur * 1.001 and t_nest > 5.0:
            t_nest = 1.0
            a_prev = a_new.copy()
            restart_count += 1
        elif sc_new > prev_score * 1.002 and t_nest > 10.0:
            # Stagnation detection
            t_nest = 1.0
            a_prev = a_new.copy()
            restart_count += 1
        else:
            a_prev = a.copy()
            t_nest = t_new

        a = a_new
        prev_score = sc_new

        if (it + 1) % 200 == 0:
            elapsed = time.monotonic() - t0
            if elapsed - last_print > 10:
                print(f"    NesterovLP iter={it+1} ({elapsed:.0f}s): score={best_score:.10f} t={t_nest:.1f} restarts={restart_count}")
                last_print = elapsed

    return best_a, best_score


def lse_objective_grad(z_in, T_val, n):
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-20:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z

        if len(topk) == 0:
            topk = np.array([int(np.argmax(b))])
        if len(topk) > 50:
            # Keep only top 50
            vals = b[topk]
            top50 = np.argsort(vals)[-50:]
            topk = topk[top50]

        # Gradient in a-space
        grad_a = np.zeros(n)
        for k in topk:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_a[idx] += 2.0 * a[k - idx]
        grad_a /= len(topk)

        S = np.dot(grad_a, a)
        grad_z = a * (grad_a - S)

        # Nesterov in z-space
        if t_nest > 1.0:
            momentum = (t_nest - 1.0) / t_nest
            z_extrap = z + momentum * (z - z_from_a(a_prev))
        else:
            z_extrap = z

        z_new = z_extrap - eta * grad_z
        a_new = softmax_from_z(z_new)

        sc = score_normalized(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a_prev = a
            a = a_new
            z = z_new
            t_nest = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
            last_improve = step
        else:
            if sc > best_score * 1.0005:
                t_nest = 1.0
                eta *= 0.95
            else:
                t_nest = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
                a_prev = a
                a = a_new
                z = z_new
            eta = max(eta * 0.9995, 0.005)

    return best_a, best_score


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def diverse_multi_start(n, rng, budget_s, solver):
    """Try multiple diverse initializations and return the best."""
    best_a = None
    best_score = float('inf')
    n_starts = 6
    per_start_time = budget_s / n_starts

    for i in range(n_starts):
        if time.monotonic() > _global_deadline:
            break
        remaining = _global_deadline - time.monotonic()
        if remaining < per_start_time * 0.5:
            break
        a_init = structured_init(n, rng, variant=i % 12)
        # Quick MW pre-pass
        a_mw = a_init.copy()
        for _ in range(5):
            b_res = mw_best_response(a_mw, T_inner=30)
            a_mw = 0.7 * a_mw + 0.3 * b_res
            a_mw = np.maximum(a_mw, 0)
            a_mw = a_mw / a_mw.sum()
        sc = score_normalized(a_mw)
        if sc < best_score:
            best_score = sc
            best_a = a_mw.copy()
            print(f"    Start {i+1}/{'%d' % n_starts}: score={sc:.10f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=1800.0):
    global _global_deadline
    rng = np.random.default_rng(seed)
    _global_deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Total budget: {budget_s:.0f}s, deadline in {budget_s - 30:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    n_base = 1200
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_seq = p / p.sum()
            n_base = len(best_seq)
            best_score = evaluate_safe(best_seq)
            print(f"Parent: n={n_base}, score={best_score:.10f}")
    except Exception:
        pass

    if best_seq is None:
        best_seq = structured_init(n_base, rng, 0)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.10f}")

    best_n = n_base

    def try_update(arr, label):
        nonlocal best_seq, best_score, best_n
        if arr is None:
            return
        arr = np.maximum(arr, 0).astype(np.float64)
        s = arr.sum()
        if s < 0.01:
            return
        arr = arr / s
        sc = evaluate_safe(arr)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = arr.copy()
            best_n = len(arr)
            print(f"  NEW BEST [{label}]: {sc:.10f} (n={len(arr)})")

    # === Phase 1: Quick MW warm-up on parent (30s) ===
    print(f"\n=== Phase 1: MW warm-up at n={n_base} ===")
    t1 = min(30.0, (_global_deadline - time.monotonic()) * 0.03)
    if t1 > 5:
        a_warm = best_seq.copy()
        t_start = time.monotonic()
        while time.monotonic() - t_start < t1:
            b_res = mw_best_response(a_warm, T_inner=60)
            a_new = 0.8 * a_warm + 0.2 * b_res
            a_new = np.maximum(a_new, 0)
            a_new = a_new / a_new.sum()
            sc = score_normalized(a_new)
            if sc < score_normalized(a_warm):
                a_warm = a_new
            else:
                break
        try_update(a_warm, "mw_warmup")
    print(f"  Phase 1 done: best_score={best_score:.10f}")

    # === Phase 2: Nesterov LP main phase (700s) ===
    print(f"\n=== Phase 2: Nesterov LP main (score={best_score:.10f}) ===")
    remaining = _global_deadline - time.monotonic()
    t2 = min(700.0, remaining * 0.5)
    if t2 > 30:
        a_out, _ = nesterov_lp_adaptive(best_seq, t2, solver=solver, line_search=True)
        try_update(a_out, "nesterov_lp_main")
    print(f"  Phase 2 done: best_score={best_score:.10f}")

    # === Phase 3: LSE L-BFGS multi-stage (350s) ===
    print(f"\n=== Phase 3: LSE L-BFGS polish (score={best_score:.10f}) ===")
    remaining = _global_deadline - time.monotonic()
    t3 = min(350.0, remaining * 0.25)
    if t3 > 20:
        a_lse = lbgfs_logsumexp(best_seq, t3, T_start=200.0, T_end=0.02, n_stages=30)
        try_update(a_lse, "lse_polish")
    print(f"  Phase 3 done: best_score={best_score:.10f}")

    # === Phase 4: Nesterov LP round 2 (250s) ===
    print(f"\n=== Phase 4: Nesterov LP round 2 (score={best_score:.10f}) ===")
    remaining = _global_deadline - time.monotonic()
    t4 = min(250.0, remaining * 0.2)
    if t4 > 20:
        a_out2, _ = nesterov_lp_adaptive(best_seq, t4, solver=solver, line_search=False)
        try_update(a_out2, "nesterov_lp_r2")
    print(f"  Phase 4 done: best_score={best_score:.10f}")

    # === Phase 5: Creeping subgradient (150s) ===
    print(f"\n=== Phase 5: Creeping subgradient (score={best_score:.10f}) ===")
    remaining = _global_deadline - time.monotonic()
    t5 = min(150.0, remaining * 0.15)
    if t5 > 10:
        a_cree, _ = creeping_subgradient(best_seq, t5, max_steps=500000)
        try_update(a_cree, "creeping")
    print(f"  Phase 5 done: best_score={best_score:.10f}")

    # === Phase 6: Final LSE with very low T (100s) ===
    print(f"\n=== Phase 6: Final LSE ultra-low T (score={best_score:.10f}) ===")
    remaining = _global_deadline - time.monotonic()
    if remaining > 30:
        t6 = min(100.0, remaining * 0.2)
        if t6 > 10:
            a_lse2 = lbgfs_logsumexp(best_seq, t6, T_start=50.0, T_end=0.005, n_stages=20)
            try_update(a_lse2, "lse_final")
    print(f"  Phase 6 done: best_score={best_score:.10f}")

    # === Phase 7: Multi-length exploration ===
    print(f"\n=== Phase 7: Length exploration (score={best_score:.10f}) ===")
    remaining = _global_deadline - time.monotonic()
    if remaining > 60:
        lengths_to_try = []
        # Try nearby lengths and some "nice" lengths
        nearby = [best_n - 50, best_n + 50, best_n + 100, best_n - 100, best_n + 200]
        nice = [1024, 1536, 2048, 1000, 1500]
        for nl in nearby + nice:
            if nl >= 500 and nl not in lengths_to_try:
                lengths_to_try.append(nl)

        for n_try in lengths_to_try[:5]:
            remaining = _global_deadline - time.monotonic()
            if remaining < 20:
                break
            t7 = min(45.0, remaining * 0.35)
            if t7 < 10:
                break
            print(f"  Trying n={n_try}, budget={t7:.0f}s")
            init_try = interpolate_sequence(best_seq, n_try)
            # Quick LP optimization
            a_try, _ = nesterov_lp_adaptive(init_try, t7, solver=solver, max_iters=100000, line_search=True)
            try_update(a_try, f"len_{n_try}")

    # === Phase 8: Final creeping on best ===
    remaining = _global_deadline - time.monotonic()
    if remaining > 20:
        print(f"\n=== Phase 8: Final creeping (score={best_score:.10f}) ===")
        t8 = min(remaining * 0.8, 60.0)
        if t8 > 10:
            a_final, _ = creeping_subgradient(best_seq, t8, max_steps=300000)
            try_update(a_final, "final_creeper")
    print(f"  Phase 8 done: best_score={best_score:.10f}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
