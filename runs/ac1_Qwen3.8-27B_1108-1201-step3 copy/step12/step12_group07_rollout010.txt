def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def score_normalized(a):
    n = len(a)
    b = fast_convolve(a)

def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def build_conv_matrix(a_fixed):
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T

def lse_objective_grad(z_in, T_val, n):
    a = softmax_from_z(z_in)
    b = fast_convolve(a)
    Tb = T_val * b
    max_Tb = Tb.max()
    if max_Tb > 700:
        max_Tb = 700.0
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T_val) * log_sum_exp

    grad_a = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-20:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad_a[idx] += 2.0 * w[k] * a[k - idx]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

    print(f"\n=== Phase 6: Final refinement (score={best_score:.8f}) ===")
    remaining = _global_deadline - time.monotonic()
    if remaining > 20:
        t6a = min(50.0, remaining * 0.4)
        if t6a > 5:
            a_sp, _ = peak_shave_single(best_seq, t6a, max_steps=50000)
            try_update(a_sp, "single_peak")
        remaining = _global_deadline - time.monotonic()
        if remaining > 10:
            t6b = min(50.0, remaining * 0.5)
            if t6b > 5:
                a_lse2, _ = lse_phase(best_seq, t6b, T_start=15.0, T_end=0.005, n_stages=20)
                try_update(a_lse2, "lse_final")
    print(f"  Phase 6 done: best_score={best_score:.8f}")

    # === Phase 7: Additional LP + peak cycle if time remains ===
    remaining = _global_deadline - time.monotonic()
    print(f"\n=== Phase 7: Extra cycles (score={best_score:.8f}), remaining={remaining:.0f}s ===")
    if remaining > 60:
        # Quick LP
        t7a = min(100.0, remaining * 0.4)
        if t7a > 15:
            a_out3, _ = nesterov_lp_improved(best_seq, t7a, solver=solver)
            try_update(a_out3, "lp_r3")
        remaining = _global_deadline - time.monotonic()
        # Quick peak shave
        if remaining > 30:
            t7b = min(60.0, remaining * 0.4)
            if t7b > 5:
                a_pt2, _ = peak_shave(best_seq, t7b, max_steps=50000)
                try_update(a_pt2, "peak_r3")
        remaining = _global_deadline - time.monotonic()
        # Final LSE
        if remaining > 20:
            t7c = min(40.0, remaining * 0.5)
            if t7c > 5:
                a_lse3, _ = lse_phase(best_seq, t7c, T_start=10.0, T_end=0.003, n_stages=15)
                try_update(a_lse3, "lse_r3")

    # === Phase 8: Length exploration (if time remains) ===
    remaining = _global_deadline - time.monotonic()
    print(f"\n=== Phase 8: Length exploration, remaining={remaining:.0f}s ===")
    if remaining > 80:
        for n_try in [best_n + 50, best_n - 50, best_n + 100]:
            remaining = _global_deadline - time.monotonic()
            if remaining < 30:
                break
            if n_try < 100:
                continue
            t8 = min(40.0, remaining * 0.3)
            if t8 < 10:
                break
            print(f"  Trying n={n_try}, budget={t8:.0f}s")
            init_try = interpolate_sequence(best_seq, n_try)
            a_try, _ = nesterov_lp_improved(init_try, t8, solver=solver)
            try_update(a_try, f"len_{n_try}")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
