def fast_autocorr(a):
    """Full self-convolution a*a using FFT, O(n log n)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_corr(a, b):
    """Full cross-convolution a*b using FFT."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n1, n2 = len(a), len(b)
    size = n1 + n2 - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    B = rfft(b, fft_size)
    return irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0} in O(n log n)."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def evaluate_safe(a):
    """Safely evaluate a sequence using the provided evaluate_sequence."""
    try:

        sc = evaluate_sequence(a_list)
        return float(sc)

def normalized_score(a):
    """Compute 2*n*max_b/(sum_a)^2 directly (for speed, avoids calling evaluate_sequence)."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    n = len(a)
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where (T@b) = convolve(a_fixed, b)."""
    n = len(a_fixed)
    size = 2 * n - 1
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.3])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def solve_lp_best_response(a_fixed, T_mat, solver):
    """
    Solve LP: min M s.t. T@b <= M, sum(b)=1, b>=0.
    T is precomputed. Returns (b, M).
    """
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    constraints = [T_mat @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)

    try:
        prob.solve(solver=solver)
    except Exception:
        pass

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue

    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None

    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    M_val = float(M_var.value)
    if not np.isfinite(M_val):
        M_val = fast_cross_corr(a_fixed, b_val).max()
    return b_val, M_val

    else:

        if time.monotonic() - t0 > time_limit:

        # For efficiency, rebuild every 50 iterations or use current a

        b_star, M_val = solve_lp_best_response(y, T_mat, solver)
        if b_star is None:
            continue

        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        ss = y_new.sum()
        if ss < 1e-15:
            y_new = b_star.copy()
        else:
            y_new = y_new / ss

def mirror_descent_optimize(a0, time_limit, T_start=50.0, T_end=1.0, eta_init=0.05, max_iters=500000):
    """
    Mirror descent (Exponentiated Gradient) on the simplex.
    Uses LogSumExp-smoothed gradient with adaptive temperature.
    Much faster per iteration than LP: O(n log n) each.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = normalized_score(a)
    t0 = time.monotonic()

    eta = eta_init
    n_stages = 20
    temps = np.geomspace(T_start, T_end, n_stages)
    iters_per_stage = max(100, max_iters // n_stages)

    iter_count = 0
    stall_count = 0

    for stage in range(n_stages):
        T = temps[stage]
        stage_iters = min(iters_per_stage, max_iters - iter_count)

        for it in range(stage_iters):
            if time.monotonic() - t0 > time_limit:
                break
            if iter_count >= max_iters:
                break

            # Compute convolution
            b = fast_autocorr(a)

            # Compute softmax weights
            Tb = T * b
            max_Tb = Tb.max()
            exp_Tb = np.exp(Tb - max_Tb)
            w = exp_Tb / exp_Tb.sum()

            # Compute gradient: grad[i] = 2 * sum_k w[k] * a[k-i]
            # This is 2 * (w * a_rev)[i + n - 1] where a_rev = a[::-1]
            a_rev = a[::-1].copy()
            conv_len = len(w) + n - 1
            fft_size = 1 << (conv_len - 1).bit_length()
            if conv_len > fft_size:
                fft_size *= 2
            W = rfft(w, fft_size)
            A_rev = rfft(a_rev, fft_size)
            conv = irfft(W * A_rev, fft_size)
            grad = 2.0 * conv[n - 1:n - 1 + n]

            # Mirror descent update (exponentiated gradient)
            # a_new[i] = a[i] * exp(-eta * grad[i]) / Z
            # For numerical stability, scale grad
            grad_max = np.abs(grad).max()
            if grad_max < 1e-10:
                break
            scaled = -eta * grad / grad_max
            # Clip for numerical stability
            scaled = np.clip(scaled, -50, 50)
            updates = np.exp(scaled)
            a_new = a * updates
            ss = a_new.sum()
            if ss < 1e-300:
                a_new = a.copy()
            else:
                a_new = a_new / ss

            a = a_new
            iter_count += 1

            # Evaluate periodically
            if it % 20 == 0:
                score = normalized_score(a)
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                    stall_count = 0
                else:
                    stall_count += 1

                # Adaptive step size
                if stall_count > 50:
                    eta *= 0.8
                    stall_count = 0
                elif best_score < 2.0:
                    eta = min(eta * 1.0

1, 0.5)

        if stage % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      MD stage={stage}/T={T:.3f} ({elapsed:.0f}s): best={best_score:.8f}, eta={eta:.6f}")

    return best_a, best_score


def nesterov_subgrad_optimize(a0, time_limit, max_iters=300000):
    """
    Nesterov-accelerated subgradient descent on max_k (a*a)[k].
    Uses the exact subgradient at the maximizing lag (fast: O(n) per step after FFT).
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    a_prev = a.copy()
    y = a.copy()

    best_a = a.copy()
    best_score = normalized_score(a)
    t0 = time.monotonic()

    avg_a = np.zeros(n)
    n_avg = 0

    gamma = math.sqrt(2.0)  # Initial step size parameter

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov momentum step
        if t == 1:
            y = a.copy()
        else:
            beta = (t - 1.0) / (t + 2.0)
            y = a + beta * (a - a_prev)
            y = np.maximum(y, 0)
            ys = y.sum()
            if ys < 1e-15:
                y = a_prev.copy()
            else:
                y = y / ys

        # Compute subgradient at y
        b = fast_autocorr(y)
        k_star = int(np.argmax(b))

        # grad[j] = 2 * y[k_star - j] if 0 <= k_star - j < n, else 0
        grad = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        for j in range(lo, hi + 1):
            grad[j] = 2.0 * y[k_star - j]

        # Step size
        eta_t = gamma / math.sqrt(t)

        # Gradient step + projection
        a_new = y - eta_t * grad
        a_new = project_simplex(a_new)

        a_prev = a
        a = a_new

        # Evaluate
        if t % 10 == 0:
            score = normalized_score(a)
            if score < best_score:
                best_score = score
                best_a = a.copy()

        avg_a += a
        n_avg += 1
        if n_avg % 500 == 0:
            avg_check = avg_a / n_avg
            score_avg = normalized_score(avg_check)
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_check.copy()

        if t % 5000 == 0:
            elapsed = time.monotonic() - t0
            b_max = fast_autocorr(a).max()
            print(f"      NSG t={t} ({elapsed:.0f}s): best={best_score:.8f}, bmax={b_max:.8f}")

    return best_a, best_score


def compute_lse_grad(a, T):
    """
    Compute gradient of (1/T)*log(sum exp(T*(a*a)_k)) w.r.t. a.
    """
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    w = exp_Tb / exp_Tb.sum()

    a_rev = a[::-1].copy()
    conv_len = n + (2 * n - 1) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    W = rfft(w, fft_size)
    A_rev = rfft(a_rev, fft_size)
    conv = irfft(W * A_rev, fft_size)
    grad = 2.0 * conv[n - 1:n - 1 + n]
    return grad, w


def lse_obj_grad_z(z, T):
    """Objective and gradient for z-parameterization (a = softmax(z))."""
    z = z - z.max()
    e = np.exp(z)
    a = e / e.sum()
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    obj_val = (max_Tb + np.log(exp_Tb.sum())) / T

    grad_a, w = compute_lse_grad(a, T)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj_val, grad_z


def logsumexp_optimize(a0, time_limit, T_start=100.0, T_end=0.02, n_stages=20):
    """
    LogSumExp-smoothed optimization with L-BFGS and multi-stage annealing.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = normalized_score(a_init)

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        iters_per_stage = max(10, min(80, int(20 + 60 * (si / max(1, n_stages - 1)))))

        def func(z_in):
            val, _ = lse_obj_grad_z(z_in, stage_T)
            return val

        def jac(z_in):
            _, g = lse_obj_grad_z(z_in, stage_T)
            return g

        try:
            res = minimize(
                fun=func, x0=z, jac=jac, method='L-BFGS-B',
                options={'maxiter': iters_per_stage, 'ftol': 1e-20,
                         'gtol': 1e-15, 'maxfun': iters_per_stage * 200,

            )
            z = res.x
            a_new = np.exp(z - z.max())
            a_new = a_new / a_new.sum()
            a_new = np.maximum(a_new, 0)
            score = normalized_score(a_new)
            if score < best_score:
                best_score = score
                best_a = a_new.copy()
                z = np.log(np.maximum(a_new, 1e-30))
        except Exception:
            pass

        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE stage={si}/T={stage_T:.4f} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def peak_shaving_advanced(a0, time_limit, seed=42):

    """

            # Top contributors

            for tc in top_contrib[:2]:
                src = int(idx[tc])
                src_val = a[src]
                if src_val < 1e-14:
                    continue

                def compute_target_penalty(j):
                    total = 0.0
                    for k2 in top_k:
                        js2 = max(0, k2 - n + 1)
                        je2 = min(n - 1, k2)
                        if js2 <= j <= je2:
                            partner = k2 - j
                            if 0 <= partner < n:
                                total += a[partner]
                    return total + 0.001 * a[j]

                # Sample candidate targets (for speed)
                n_samples = min(50, n // 2)
                candidates = rng.integers(0, n, size=n_samples)
                candidates = np.unique(candidates)
                candidates = [j for j in candidates if j != src]

                if not candidates:
                    continue

                penalties = [compute_target_penalty(j) for j in candidates]
                candidates_sorted = sorted(zip(penalties, candidates))[:5]

                for _, dst in candidates_sorted[:3]:
                    for frac in [0.003, 0.007, 0.015, 0.03, 0.06]:
                        amt = src_val * frac
                        if amt < 1e-15:
                            continue
                        a_trial = a.copy()
                        a_trial[src] -= amt
                        a_trial[dst] += amt
                        if a_trial[src] < -1e-15:
                            continue
                        a_trial = np.maximum(a_trial, 0)
                        ss = a_trial.sum()
                        if ss < 1e-15:
                            continue
                        a_trial = a_trial / ss
                        score = normalized_score(a_trial)
                        if score < best_score - 1e-14:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            stall = 0
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random exploration
            for _ in range(10):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.04)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= -1e-15:
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        score = normalized_score(a_trial)
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
                            break
            if stall > 500:
                break

    return best_a, best_score

    best_a = a.copy()
    best_score = normalized_score(a)
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        k_star = int(np.argmax(b))
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)

        # Try adjusting coordinates in the peak region
        for _ in range(100):
            if time.monotonic() - t0 > time_limit:
                break
            i = int(rng.integers(js, je + 1))
            if a[i] < 1e-12:
                continue
            delta = a[i] * rng.uniform(-0.05, 0.05)
            if abs(delta) < 1e-15:
                continue
            # Redistribute to a random other coordinate
            j = int(rng.integers(n))
            if j == i:
                continue
            a_trial = a.copy()
            a_trial[i] += delta
            a_trial[j] -= delta
            if a_trial[i] < -1e-15 or a_trial[j] < -1e-15:
                continue
            a_trial = np.maximum(a_trial, 0)
            ss = a_trial.sum()
            if ss < 1e-15:
                continue
            a_trial = a_trial / ss
            score = normalized_score(a_trial)
            if score < best_score:
                a = a_trial
                best_score = score
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        if stall > 2000:
            break

    return best_a, best_score


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = int(rng.integers(0, 10))
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 2
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(3 * x) + 0.1 * np.cos(7 * x) + 0.05 * np.cos(13 * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * (x - 0.3) ** 2) + 0.7 * np.exp(-4 * (x - 0.7) ** 2)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = (1 - x) * (0.5 + 0.5 * np.sin(10 * np.pi * x))
    elif choice == 7:
        x = np.linspace(0, 1, n)
        a = np.maximum(0, 1 - 2 * np.abs(x - 0.5)) + 0.2 * np.sin(5 * np.pi * x)
        a = np.maximum(a, 0)
    elif choice == 8:
        # Parabolic
        x = np.linspace(0, 1, n)
        a = x * (1 - x)
    else:
        # Multi-peak
        x = np.linspace(0, 1, n)
        a = np.exp(-20*(x-0.2)**2) + 0.8*np.exp(-20*(x-0.5)**2) + 0.6*np.exp(-20*(x-0.8)**2)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
