For the gradient computation:
- conv(a, a) via FFT: O(n log n)
- LSE gradient: need to compute sum_k w_k * (T @ a)_k where w = softmax(conv/T)
- This can be done as: grad = 2 * (T^T @ w) where T is the convolution matrix
- T^T @ w can be computed as: for each i, (T^T @ w)[i] = sum_k w[k] * T[k][i] = sum_k w[k] * a[k-i] (when valid)
- This is just a convolution of w with a (reversed)! So it's another FFT.

So the full gradient computation is:
1. FFT(a) → compute conv(a,a) [one FFT + pointwise square + IFFT]
2. Compute w = softmax(conv/T)
3. Compute w * a (element-wise) [O(n)]
4. FFT of (reversed a) and FFT of w, multiply, IFFT → gives T^T @ w [two FFTs + IFFT]

Total: ~4 FFTs per gradient step, each O(n log n). Very fast!

For simplex projection: sort + binary search, O(n log n).

So each FISTA iteration is O(n log n). For n=2000, this is very fast - we can do thousands of iterations per second.

```python
from scipy.fft import rfft, irfft

    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def lse_gradient(a, T):

    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    # Softmax weights
    exp_b = np.exp((b - max_b) / T)
    w = exp_b / exp_b.sum()

    # T^T @ w at index i = sum_k w[k] * T[k,i] = sum_k w[k] * a[k-i] (when 0<=k-i<n)
    # This is the cross-correlation of w and a, at lag i... 
    # Actually: sum_k w[k] * a[k-i] where 0 <= k-i < n means k-i in [0, n-1]
    # Let j = k-i, so k = j+i. Sum over j: w[j+i] * a[j] for j in [0, n-1] and j+i in [0, 2n-2]
    # This is (w * a)[i] where * denotes cross-correlation
    # In convolution terms: conv(w, a_reversed)[i + n - 1] where a_reversed = a[::-1]
    
    # More carefully: we want g[i] = sum_k w[k] * T[k,i]
    # T[k,i] = a[k-i] if 0 <= k-i < n, else 0
    # So g[i] = sum_{k: 0<=k-i<n} w[k] * a[k-i]
    # Let j = k-i: g[i] = sum_{j: 0<=j<n, 0<=j+i<2n-1} w[j+i] * a[j]

    # Wait, we need 0 <= j+i <= 2n-2, i.e., -i <= j <= 2n-2-i
    # Since j >= 0 and j <= n-1, we need max(0, -i) <= j <= min(n-1, 2n-2-i)

    # For i in [0, n-1]: 2n-2-i >= 2n-2-(n-1) = n-1, so j ranges from 0 to n-1
    # So g[i] = sum_{j=0}^{n-1} w[j+i] * a[j] for i in [0, n-1]
    
    # This is the cross-correlation of (w, a) at lag i.
    # In terms of np.convolve: if we compute np.convolve(w, a[::-1]), 
    # then result[i + n - 1] = sum_j w[j] * a[::-1][i+n-1-j] = sum_j w[j] * a[n-1-i-n+1+j] = ...
    
    # Let me just use FFT:
    # g[i] = sum_{j=0}^{n-1} w[j+i] * a[j]
    # Let's define u = w (padded to length 2n-1) and v = a (padded)
    # g[i] = sum_j u[j+i] * v[j] = (u ⊛ v)[i] where ⊛ is cross-correlation
    # Cross-correlation can be computed as: IFFT(FFT(u) * conj(FFT(v)))
    # But we need to be careful with the indexing.
    
    # Actually the simplest way: g = 2 * np.correlate(w, a, mode='valid')
    # np.correlate(w, a, 'valid')[i] = sum_j w[j+i] * a[j] (for the 'valid' part)
    # This is exactly what we want! And 'valid' gives length max(len(w), len(a)) - min(len(w), len(a)) + 1
    # Since both have length... wait, w has length 2n-1 and a has length n.
    # 'valid' gives length (2n-1) - n + 1 = n.

    fft_size = 1 << (3 * n - 2).bit_length()
    W = rfft(w, fft_size)
    A_rev = rfft(a[::-1].copy(), fft_size)

    # We want sum_j w[j+i] * a[j] = sum_j w[j'] * a[j'-i] where j' = j+i
    # = sum_j w[j'] * a_rev[n-1-(j'-i)] = ... this is getting confusing.
    
    # Let me just use the direct formula with FFT:
    # g[i] = sum_{j=0}^{n-1} w[j+i] * a[j]
    # 
    # Define: extend w to length 3n-2 (padded with zeros), extend a to same length.
    # Then g[i] = (w * a_flip)[i + n - 1] - (w * a_flip)[i] ... no this is getting too complicated.

    # Actually for n=2000, the direct computation is O(n^2) = 4M operations per gradient step.
    # That's too slow for thousands of iterations.

    # We want g[i] = sum_{j=0}^{n-1} w[j+i] * a[j] for i = 0, ..., n-1
    # 
    # Let's compute the full cross-correlation c[m] = sum_j w[j] * a[j+m] for all m
    # (with appropriate boundary handling)
    # Then g[i] = c[-i] = sum_j w[j] * a[j-i] ... no.
    # 
    # Actually: g[i] = sum_j w[j+i] * a[j]. Let k = j+i, so j = k-i.
    # g[i] = sum_k w[k] * a[k-i] (where a[k-i] = 0 if k-i not in [0, n-1])

    # 
    # Cross-correlation R_wa[i] = sum_k w[k] * a[k-i] (with a extended by zeros)
    # This equals conv(w, a_flipped) where a_flipped = a[::-1]
    # Specifically: conv(w, a_flip)[i + n - 1] = sum_k w[k] * a_flip[i+n-1-k]
    #            = sum_k w[k] * a[n-1-(i+n-1-k)] = sum_k w[k] * a[k-i]
    # 
    # So g[i] = conv(w, a_flip)[i + n - 1] for i = 0, ..., n-1
    # And we can compute conv(w, a_flip) via FFT!
    
    a_flip = a[::-1].copy()
    fft_size2 = 1 << ((2 * n - 1) - 1).bit_length()  # size of conv(w, a_flip) is (2n-1) + n - 1 = 3n-2
    # Actually conv of length-(2n-1) and length-n gives length 3n-2

    conv_w_af = irfft(W_fft * AF_fft, fft_size2)

    g = 2.0 * conv_w_af[n - 1: 2 * n - 1]
    
    return g
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FISTA with FFT-accelerated LSE-surrogate gradients (primary, very fast)
2. Multi-temperature geometric annealing schedule
3. Aggressive coordinate peak-shaving polish
4. Multi-length exploration with interpolation transfer (n=1600..3200)
5. Multiple structural initializations to avoid local minima
6. Adaptive restarts on stagnation
Key advantage: no LP solver needed; each iteration is ~3 FFTs + simplex projection.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Compute full autocorrelation (convolution with self) via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def lse_gradient(a, T):
    """Compute gradient of T*log(sum exp(conv(a,a)/T)) w.r.t. a."""
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    exp_b = np.exp((b - max_b) / T)
    w = exp_b / exp_b.sum()

    # grad[i] = 2 * sum_k w[k] * a[k-i] (valid range)
    # = 2 * conv(w, a_flip)[n-1 + i] for i=0..n-1
    a_flip = a[::-1].copy()
    conv_len = (2 * n - 1) + n - 1  # = 3n - 2
    fft_size = 1 << (conv_len - 1).bit_length()
    W_fft = rfft(w, fft_size)
    AF_fft = rfft(a_flip, fft_size)
    conv_result = irfft(W_fft * AF_fft, fft_size)
    grad = 2.0 * conv_result[n - 1: 2 * n - 1]
    return grad


def simplex_project(v):
    """Project vector v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(x) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def fista_lse(a0, time_limit, T_start=5.0, T_end=0.05, verbose=False):
    """
    FISTA with adaptive LSE temperature annealing.
    Minimizes max_k conv(a,a)[k] subject to a>=0, sum(a)=1.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    y = a.copy()
    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    t0 = time.monotonic()
    t_fista = 1.0
    step_size = 1.0 / (2.0 * n)  # initial step
    n_iters = 0
    stall_count = 0
    no_improve = 0

    # Precompute fft_size for gradient
    conv_len = 3 * n - 2
    fft_size = 1 << (conv_len - 1).bit_length()
    a_flip_buf = np.zeros(n)

    # Temperature schedule: geometric from T_start to T_end over expected iterations
    # We'll adapt T based on progress
    current_T = T_start

    while time.monotonic() - t0 < time_limit:
        # Adaptive temperature: decrease every 5000 iterations or when stalled
        if n_iters > 0 and n_iters % 3000 == 0:
            ratio = (time.monotonic() - t0) / time_limit
            current_T = T_start * (T_end / T_start) ** min(ratio * 1.2, 1.0)
            current_T = max(current_T, T_end)

        # Compute gradient at y
        b_y = fast_autocorr(y)
        max_by = b_y.max()
        exp_by = np.exp((b_y - max_by) / current_T)
        w = exp_by / exp_by.sum()

        a_flip_buf[:] = y[::-1]
        W_fft = rfft(w, fft_size)
        AF_fft = rfft(a_flip_buf, fft_size)
        conv_result = irfft(W_fft * AF_fft, fft_size)
        grad = 2.0 * conv_result[n - 1: 2 * n - 1]

        # Backtracking line search for step size
        # Lipschitz estimate: L ~ 2 * max(w) * n / T (rough)
        L_est = 2.0 * w.max() * n / current_T
        step = 1.0 / L_est

        # Gradient step + projection
        z = y - step * grad
        x_new = simplex_project(z)

        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_fista * t_fista)) / 2.0
        momentum = (t_fista - 1.0) / t_new
        y_new = x_new + momentum * (x_new - y)
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = x_new.copy()
        else:
            y_new = y_new / s_y

        y = y_new
        t_fista = t_new
        n_iters += 1

        # Evaluate and track best
        score = fast_autocorr(x_new).max()
        if score < best_score:
            best_score = score
            best_a = x_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Check y as well (sometimes better)
        score_y = fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            no_improve = 0

        # Restart on prolonged stagnation
        if no_improve > 20000:
            # Perturb and restart
            rng = np.random.default_rng(n_iters)
            pert = rng.normal(0, 0.1 / n, n)
            a_pert = x_new + pert
            a_pert = simplex_project(a_pert)
            y = a_pert.copy()
            t_fista = 1.0
            no_improve = 0
            current_T = max(T_start * 0.5, current_T)

        if verbose and n_iters % 50000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FISTA iter={n_iters} (T={current_T:.4f}, {elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def peak_shave(a, time_limit, seed=42, verbose=False):
    """
    Aggressive coordinate-wise peak shaving.
    Targets top convolution peaks by redistributing mass.
    """
    n = len(a)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    n_iters = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        # Get top peaks
        n_peaks = min(5, 2 * n - 1)
        top_idx = np.argsort(b)[::-1][:n_peaks]
        M = b[top_idx[0]]

        improved = False
        for k in top_idx:
            # Find contributors to peak k: pairs (i, k-i) with 0<=i<n, 0<=k-i<n
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            total_contrib = contrib.sum()
            if total_contrib < 1e-15:
                continue

            # Find the largest contributor
            src = idx[np.argmax(contrib)]
            src_val = a[src]
            if src_val < 1e-10:
                continue

            # Find destination: position that minimizes addition to all top peaks
            # For each candidate position p, compute how much it would add to top peaks
            # conv[p] += a[p]*a[k-p] for each peak k
            # We want p where this is small
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_idx:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True

            # Prefer positions NOT in any peak's support, with low current value
            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]

            # Among candidates, pick ones with lowest value
            if len(candidates) > n // 8:
                candidates = candidates[np.argsort(a[candidates])][:n // 8]
            if len(candidates) == 0:
                candidates = np.array([src])

            dst = candidates[rng.integers(len(candidates))]

            # Try moving some mass
            for frac in [0.001, 0.002, 0.005, 0.01, 0.02, 0.05]:
                amt = src_val * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    stall = 0
                    n_iters += 1
                    break

        if not improved:
            # Random exploration move
            if a.sum() > 1e-10:
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.005, 0.03)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    a_trial = np.maximum(a_trial, 0.0)
                    s_t = a_trial.sum()
                    if s_t > 1e-15:
                        a_trial = a_trial / s_t
                        score = fast_autocorr(a_trial).max()
                        if score < best_score:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            stall = 0
                            n_iters += 1
            stall += 1
            if stall > 500:
                break

        if verbose and n_iters % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PeakShave iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss


def make_init(n, method, rng):
    """Create an initial sequence of length n using the given method."""
    if method == 0:  # Uniform
        return np.ones(n) / n
    elif method == 1:  # Cosine bell
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif method == 2:  # Two-bump
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-20.0 * (x - 0.25) ** 2) + np.exp(-20.0 * (x - 0.75) ** 2)
    elif method == 3:  # Random softmax
        a = np.abs(rng.normal(0, 1, n))
    elif method == 4:  # Triangle
        x = np.linspace(0.0, 1.0, n)
        a = 1.0 - np.abs(2.0 * x - 1.0)
    else:  # Gaussian center
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-8.0 * (x - 0.5) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Returns the best sequence found within budget.
    """
    rng = np.random.default_rng(seed)
    t_global = time.monotonic()
    max_time = min(float(budget_s) - 120.0, 1650.0)
    print(f"Total time budget: {max_time:.0f}s")

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        s = a.sum()
        if s < 0.01:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")

    # If no parent, create one
    if parent is None:
        best_seq = make_init(1600, 0, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={len(best_seq)}, score={best_score:.6f}")

    # Phase 1: FISTA at parent length
    n_base = len(best_seq)
    remaining = max_time - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: FISTA n={n_base} (budget={remaining:.0f}s) ===")
    t1 = min(remaining * 0.30, 400.0)
    if t1 > 30:
        a_out, _ = fista_lse(best_seq, t1, T_start=3.0, T_end=0.02, verbose=True)
        try_update(a_out, f"fista_n{n_base}")
        # Peak shave on result
        remaining = max_time - (time.monotonic() - t_global)
        t1b = min(remaining * 0.10, 100.0)
        if t1b > 5:
            a_out2, _ = peak_shave(best_seq, t1b, seed=123, verbose=False)
            try_update(a_out2, f"shave_n{n_base}")

    # Phase 2: Try multiple initializations at n_base
    remaining = max_time - (time.monotonic() - t_global)
    n_inits = 3
    t_per_init = min(remaining * 0.15 / n_inits, 120.0)
    if t_per_init > 10:
        for init_method in [1, 3, 5]:
            remaining = max_time - (time.monotonic() - t_global)
            if remaining < 20:
                break
            init_a = make_init(n_base, init_method, rng)
            t_init = min(t_per_init, remaining * 0.5)
            a_out, _ = fista_lse(init_a, t_init, T_start=2.0, T_end=0.05)
            try_update(a_out, f"init{init_method}_n{n_base}")

    # Phase 3: Longer lengths
    for length_mult in [1.25, 1.5, 1.75, 2.0]:
        n_try = int(n_base * length_mult)
        remaining = max_time - (time.monotonic() - t_global)
        if remaining < 80:
            break
        print(f"\n=== Phase 3: FISTA n={n_try} (budget={remaining:.0f}s) ===")

        init_long = interpolate_sequence(best_seq, n_try)
        t_fista = min(remaining * 0.40, 300.0)
        if t_fista > 30:
            a_out, _ = fista_lse(init_long, t_fista, T_start=3.0, T_end=0.03, verbose=True)
            try_update(a_out, f"fista_n{n_try}")

        # Also try a random init at this length
        remaining = max_time - (time.monotonic() - t_global)
        t_rand = min(remaining * 0.15, 100.0)
        if t_rand > 10:
            init_rand = make_init(n_try, rng.integers(0, 6), rng)
            a_out_r, _ = fista_lse(init_rand, t_rand, T_start=2.0, T_end=0.1)
            try_update(a_out_r, f"rand_n{n_try}")

        # Peak shave
        remaining = max_time - (time.monotonic() - t_global)
        t_shave = min(remaining * 0.15, 80.0)
        if t_shave > 5:
            a_out_s, _ = peak_shave(best_seq, t_shave, seed=n_try, verbose=False)
            try_update(a_out_s, f"shave_n{n_try}")

    # Phase 4: Final intensive polish on best
    remaining = max_time - (time.monotonic() - t_global)
    n_best = len(best_seq)
    print(f"\n=== Phase 4: Final polish n={n_best} (budget={remaining:.0f}s) ===")

    if remaining > 20:
        # FISTA with very low temperature
        t4 = min(remaining * 0.4, 200.0)
        if t4 > 10:
            a_out, _ = fista_lse(best_seq, t4, T_start=0.5, T_end=0.005, verbose=True)
            try_update(a_out, "fista_polish")

        # Aggressive peak shaving
        remaining = max_time - (time.monotonic() - t_global)
        t4b = min(remaining * 0.4, 200.0)
        if t4b > 5:
            a_out2, _ = peak_shave(best_seq, t4b, seed=777, verbose=True)
            try_update(a_out2, "shave_polish")

        # Another FISTA round on the polished result
        remaining = max_time - (time.monotonic() - t_global)
        t4c = min(remaining * 0.3, 100.0)
        if t4c > 5:
            a_out3, _ = fista_lse(best_seq, t4c, T_start=0.3, T_end=0.002)
            try_update(a_out3, "fista_final")

        # Final peak shave
        remaining = max_time - (time.monotonic() - t_global)
        t4d = min(remaining * 0.5, 100.0)
        if t4d > 3:
            a_out4, _ = peak_shave(best_seq, t4d, seed=999)
            try_update(a_out4, "shave_final")

    # Final validation
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}, time={time.monotonic()-t_global:.0f}s")
    else:
        print("\nERROR: No valid sequence!")
        return [1.0]

    return [float(x) for x in best_seq]
```
