def fast_autocorr(a):

def softmax_fn(z):

    e = np.exp(z)

    # If i = k-i (i.e., k=2i): these are the same term, and the derivative of a[i]^2 w.r.t. a[i] is 2*a[i]

def lse_minimize_lbfgs(a0, time_limit, T_stages=None):
    """L-BFGS-B optimization with LogSumExp smoothing."""
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    
    if T_stages is None:
        T_stages = [200.0, 100.0, 50.0, 25.0, 10.0, 5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05]
    
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()
    
    for si, T in enumerate(T_stages):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        def func(z_val):
            obj, _ = lse_objective_and_grad(z_val, T)
            return obj
        
        def func_grad(z_val):
            _, grad = lse_objective_and_grad(z_val, T)
            return grad
        
        max_iters = max(20, min(200, int(remaining * 5 / (len(T_stages) - si))))
        
        try:
            result = minimize(func, z, jac=func_grad, method='L-BFGS-B',
                            options={'maxiter': max_iters, 'maxls': 20, 'gtol': 1e-6})
        except Exception:
            break
        
        z = result.x
        a = softmax_fn(z)
        score = 2.0 * n * fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        elapsed = time.monotonic() - t0
        print(f"    L-BFGS-B stage {si}/{len(T_stages)} (T={T:.2f}, {elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score

def peak_targeting_cd(a0, time_limit, seed=42):
    """Coordinate descent that specifically targets the maximum peak."""
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_k = np.argmax(b)
        max_val = b[max_k]
        
        # Try to reduce the peak at max_k
        # (a*a)[max_k] = sum_i a[i]*a[max_k-i]
        # To reduce this, we want to reduce products a[i]*a[max_k-i]
        # Strategy: find the pair (i, max_k-i) with the largest product and perturb it
        
        js = max(0, max_k - n + 1)
        je = min(n - 1, max_k)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[max_k - idx]
        
        # Sort by contribution (descending)
        order = np.argsort(contrib)[::-1]
        
        improved = False
        for oi in order[:10]:  # Try top 10 contributors
            src = idx[oi]
            other = max_k - src
            
            # Find a good destination: index j where a[j] is small and
            # moving from src to j reduces the peak
            # The change in (a*a)[max_k] from moving amt from src to j:
            # Old: a[src]*a[other] (if other != src)
            # New: (a[src]-amt)*a[other] + amt*a[other] = a[src]*a[other] (no change if other is involved)
            # Wait, we need to be more careful.
            
            # Moving amt from src to dst:
            # a[src] -> a[src] - amt
            # a[dst] -> a[dst] + amt
            # Change in (a*a)[k] = -amt*a[k-src] + amt*a[k-dst] - amt*a[src-(k-src)] ... 
            # This is getting complicated. Let me just try small moves and check.
            
            # Try moving a small amount from src to a random low-value position
            for frac in [0.01, 0.02, 0.05, 0.1]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                
                # Choose destination: prefer positions that don't create new peaks
                candidates = np.argsort(a)[:n//4]  # bottom quarter
                for dst in rng.choice(candidates, size=min(3, len(candidates)), replace=False):
                    if dst == src:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score - 1e-14:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation to escape
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                frac = rng.uniform(0.01, 0.05)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0
            
            if stall > 500:
                break
    
    return best_a, best_score

def spectral_shaping_v2(a, n_iter=300, alpha=0.2):
    """Improved spectral shaping with adaptive alpha."""
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    fft_size = 1 << (2 * n - 1).bit_length()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    
    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        
        # Target: geometric mean of non-DC spectral components
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        
        # Adaptive alpha: reduce when close to optimum
        alpha_i = alpha * (1.0 - i / n_iter * 0.5)
        
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.1, 10.0)
        scale[1:] = np.power(ratio, alpha_i / 2.0)
        
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new
    
    return best_a

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 12)
    x = np.linspace(0, 1, n)
    if choice == 0:
        a = np.sin(np.pi * x) ** 2
    elif choice == 1:
        a = np.exp(-8 * (x - 0.5) ** 2)
    elif choice == 2:
        a = (1 - 2 * np.abs(x - 0.5)) ** 0.7  # triangular with soft edges
    elif choice == 3:
        a = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.1 * np.cos(4 * np.pi * x)
    elif choice == 4:
        a = np.exp(-4 * (x - 0.3) ** 2) + 0.5 * np.exp(-4 * (x - 0.7) ** 2)
    elif choice == 5:
        a = np.sin(np.pi * x) * np.cos(np.pi * x / 2)
    elif choice == 6:
        a = x ** 0.5 * (1 - x) ** 0.5  # semi-circle
    elif choice == 7:
        a = np.exp(-6 * x) * (1 - np.exp(-6 * (1 - x)))
    elif choice == 8:
        a = np.maximum(0, 1 - 4 * (x - 0.5) ** 2 + 0.1 * np.cos(8 * np.pi * x))
    elif choice == 9:
        a = np.exp(-10 * (x - 0.5) ** 2) + 0.1
    elif choice == 10:
        a = np.maximum(0, np.sqrt(1 - (2*x - 1)**2))  # upper half circle
    else:
        a = (1 - x) ** 1.3 + 0.05
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
