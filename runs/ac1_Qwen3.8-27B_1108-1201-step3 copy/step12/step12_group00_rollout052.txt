"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP best-response with 
spectral shaping, mirror descent variants, multi-length exploration, 
and coordinate descent polish.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft

_global_solver = None

def get_solver():
    global _global_solver
    if _global_solver is not None:
        return _global_solver
    for s in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(4, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.4])
            prob.solve(solver=s)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                _global_solver = s
                return s
        except Exception:
            continue
    _global_solver = cp.ECOS
    return _global_solver

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_local(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0, 1000.0)
    n = len(a)
    if n == 0:
        return float('inf')
    b = fast_autocorr(a)
    max_b = b.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    return float(2.0 * n * max_b / (sum_a ** 2))

def build_conv_matrix(a):
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

def lp_best_response(a, M, solver):
    n = len(a)
    b_var = cp.Variable(n, nonneg=True)
    T = build_conv_matrix(a)
    constraints = [T @ b_var <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), constraints)
    prob.solve(solver=solver, verbose=False)
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None
    b_val = np.maximum(np.asarray(b_var.value, dtype=np.float64), 0.0)
    if b_val.sum() < 1e-15:
        return None
    return b_val

def gs_spectral_step(a, alpha=0.7, n_iters=1):
    """Gerchberg-Saxton spectral shaping step."""
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return a.copy()
    a_work = a.copy()
    for _ in range(n_iters):
        N = 1 << (2 * n - 1).bit_length()
        F = rfft(a_work, N)
        P = F.real * F.real + F.imag * F.imag
        P_min = P.min()
        P_range = P.max() - P_min
        if P_range < 1e-15:
            break
        # Flatten: P_new = P^alpha (compresses dynamic range)
        P_norm = (P - P_min) / (P.max() + 1e-30)
        P_new = P_min + P_range * (P_norm ** alpha)
        # Reconstruct with same phase
        mag_old = np.sqrt(P)
        mag_new = np.sqrt(P_new)
        phase = F / (mag_old + 1e-30)
        F_new = phase * mag_new
        a_work = irfft(F_new, N)[:n]
        a_work = np.maximum(a_work, 0.0)
        s = a_work.sum()
        if s < 1e-15:
            a_work = a.copy()
            break
        a_work = a_work * S / s
    return a_work

def mirror_descent_step(a, grad, eta, S):
    """One step of mirror descent with entropy kernel."""
    n = len(a)
    # a_new_i ∝ a_i * exp(-eta * grad_i)
    log_a = np.where(a > 1e-30, np.log(a + 1e-30), -10.0)
    log_new = log_a - eta * grad
    log_new -= log_new.max()  # numerical stability
    new = np.exp(log_new)
    s = new.sum()
    if s < 1e-15:
        return a * S / a.sum()
    return new * S / s

def nesterov_lp_optimize(a0, time_limit, solver, use_spectral=True, 
                          spectral_interval=50, spectral_iters=3, verbose=True):
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()

    y = a.copy()
    avg_a = np.zeros(n)
    n_avg = 0
    t0 = time.monotonic()
    n_iters = 0
    stall = 0
    prev_best = best_score

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        M_curr = fast_autocorr(y).max()
        if M_curr < 1e-12:
            break
        
        b = lp_best_response(y, M_curr, solver)
        if b is None:
            continue

        b_sum = b.sum()
        if b_sum < 1e-15:
            continue
        g = b * norm / b_sum
        n_iters += 1

        alpha_step = 2.0 / (t + 2.0)
        y_new = (1.0 - alpha_step) * y + alpha_step * g
        y_new = np.maximum(y_new, 0.0)
        s_y = y_new.sum()
        if s_y < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new * norm / s_y

        if t == 1:
            a_new = g.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0.0)
            s_a = a_new.sum()
            if s_a < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new * norm / s_a

        y = y_new
        a = a_new

        # Spectral shaping every spectral_interval iterations
        if use_spectral and n_iters % spectral_interval == 0:
            a = gs_spectral_step(a, alpha=0.7, n_iters=spectral_iters)
            s_a = a.sum()
            if s_a > 1e-15:
                a = a * norm / s_a
            y = a.copy()  # Reset Nesterov momentum after spectral step

        score_a = fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        avg_a += a
        n_avg += 1
        if n_avg % 20 == 0:
            avg_check = avg_a / n_avg
            s_avg = avg_check.sum()
            if s_avg > 1e-15:
                avg_check = avg_check * norm / s_avg
                score_avg = fast_autocorr(avg_check).max()
                if score_avg < best_score:
                    best_score = score_avg
                    best_a = avg_check.copy()
                    stall = 0

        if verbose and n_iters % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP iter={n_iters} ({elapsed:.0f}s): best={best_score:.8f}")

        if stall > 2000:
            # Perturb to escape
            rng = np.random.default_rng(n_iters)
            idx = rng.integers(n)
            a[idx] *= 0.9
            s_a = a.sum()
            a = a * norm / s_a
            y = a.copy()
            stall = 0

    return best_a, best_score

def coordinate_descent_polish(a0, time_limit, n_top_peaks=5, verbose=True):
    """Polish by moving mass from peak-contributing coords to non-contributing ones."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    iteration = 0
    while time.monotonic() - t0 < time_limit:
        iteration += 1
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:n_top_peaks]
        
        improved = False
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            
            if len(idx) == 0:
                continue
            
            src = idx[np.argmax(contrib)]
            
            # Build mask of positions contributing to any top peak
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                if je2 >= js2:
                    peak_mask[js2:je2 + 1] = True
            
            candidates = np.where(~peak_mask)[0]
            if len(candidates) == 0:
                candidates = np.where(np.arange(n) != src)[0]
            
            if len(candidates) == 0:
                continue
            
            # Try to move some mass from src to a random candidate
            dst = candidates[rng.integers(len(candidates))]
            
            for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.05]:
                amt = a[src] * frac
                if amt < 1e-20:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-15:
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial * norm / s_t
                score = fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            # Random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > norm * 1e-12:
                frac = rng.uniform(0.005, 0.03)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial * norm / s_t
                    score = fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()

        if verbose and iteration % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD iter={iteration} ({elapsed:.0f}s): best={best_score:.8f}")

        if iteration > 50000 and not improved:
            break

    return best_a, best_score

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0.0)
    ss = f_interp.sum()
    if ss < 1e-15:
        return np.ones(new_n) / new_n
    return f_interp / ss

def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0.0, np.pi, n)
        a = np.cos(x / 2.0) ** 2
    elif choice == 1:
        x = np.linspace(0.0, 1.0, n)
        a = np.sin(np.pi * x) ** 1.5
    elif choice == 2:
        # Multi-frequency cosine
        x = np.linspace(0.0, 2.0 * np.pi, n)
        a = 0.5 + 0.3 * np.cos(2.0 * x) + 0.15 * np.cos(5.0 * x) + 0.05 * np.cos(9.0 * x)
    elif choice == 3:
        # Two Gaussian bumps
        x = np.linspace(0.0, 1.0, n)
        a = np.exp(-4.0 * (x - 0.25) ** 2) + 0.6 * np.exp(-4.0 * (x - 0.75) ** 2)
    elif choice == 4:
        # Triangle
        x = np.linspace(0.0, 1.0, n)
        a = (1.0 - np.abs(2.0 * x - 1.0)) ** 2
    elif choice == 5:
        # M-sequence inspired: pseudo-random with structure
        a = np.abs(np.sin(np.pi * np.arange(n) * 1.618)) **+ 0.1
    elif choice == 6:
        # Periodic with linear decay
        x = np.linspace(0.0, 1.0, n)
        a = (0.5 + 0.5 * np.cos(20 * np.pi * x)) * (1.0 - x) ** 0.5
    else:
        # Pure noise
        a = rng.random(n) ** 2
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

def spectral_polish(a0, time_limit, verbose=True):
    """Iterative spectral shaping to flatten autocorrelation."""
    n = len(a0)
    norm = np.sqrt(2.0 * n)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0)
    s0 = a.sum()
    if s0 < 1e-15:
        a = np.ones(n) * norm / n
    else:
        a = a * norm / s0

    best_a = a.copy()
    best_score = fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Anneal alpha from aggressive to mild
    for stage in range(10):
        if time.monotonic() - t0 > time_limit:
            break
        alpha = 0.5 + 0.05 * stage  # 0.5, 0.55, ..., 0.95
        for it in range(20):
            if time.monotonic() - t0 > time_limit:
                break
            a = gs_spectral_step(a, alpha=alpha, n_iters=1)
            s = a.sum()
            if s > 1e-15:
                a = a * norm / s
            score = fast_autocorr(a).max()
            if score < best_score:
                best_score = score
                best_a = a.copy()

    return best_a, best_score
