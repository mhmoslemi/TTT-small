def fast_autocorr(a):
    """Full self-convolution a*a using FFT. Returns array of length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_score(a):
    """Compute 2n*max(b)/(sum(a))^2 quickly."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0)
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)

def lse_obj_grad_z(z, T, n_cache=None):
    """
    Objective and gradient in z-space where a = softmax(z).
    Objective: (1/T)*log(sum(exp(T*b_k)))
    """
    n = len(z)
    z_max = z.max()
    e = np.exp(z - z_max)
    a = e / e.sum()

    b = fast_autocorr(a)
    Tb = T * b
    Tb_max = Tb.max()
    exp_Tb = np.exp(Tb - Tb_max)
    obj = (Tb_max + np.log(exp_Tb.sum())) / T

    w = exp_Tb / exp_Tb.sum()

    # grad_a[j] = 2 * sum_k w[k] * a[k-j] (for valid k-j)
    # This is 2 * cross_corr(w, a)
    cc = cross_corr_w_a(w, a)
    grad_a = 2.0 * cc

    # grad_z = a * (grad_a - dot(grad_a, a))
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)

    return obj, grad_z


def mirror_descent_soft(a0, time_limit, eta0=0.05, decay_rate=0.6,

        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()

    for t in range(1, 10000000):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_autocorr(a)
        b_max = b.max()
        b_mean = b.sum() / len(b)

        # Soft weights: softmax over all peaks with temperature
        # Use excess over mean to focus on peaks
        excess = b - b_mean
        # Clip negative excess to 0 (only penalize peaks)
        excess = np.maximum(excess, 0)
        
        if t < k_ramp:
            # Ramp: start with top-k, gradually include more
            frac = t / float(k_ramp)
            k = max(k_min, int(k_min + frac * (k_max - k_min)))
            top_idx = np.argsort(b)[::-1][:k]
            # Weight top-k by their excess

            if w_k.sum() < 1e-15:
                w_k = np.ones(k) / k
            else:
                w_k = w_k / w_k.sum()
            grad = np.zeros(n)
            for ii in range(k):
                m = int(top_idx[ii])
                w_m = w_k[ii] * 2.0
                j_start = max(0, m - n + 1)

                js = np.arange(j_start, j_end + 1)
                grad[js] += w_m * a[m - js]
        else:
            # Full soft gradient using cross-correlation
            # weights proportional to exp((b - b_max) / temp)
            b_shift = (b - b_max) / temp
            w_full = np.exp(b_shift)

        eta = eta0 / (1.0 + t / 500.0) ** decay_rate

        a_new = a * np.exp(-eta * grad)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        a_new = a_new / s_new
        a = a_new

        # Check score periodically

            check_interval = 10
        else:
            check_interval = 25

        if t % check_interval == 0:
            sc = fast_score(a)
            if sc < best_score:

                stall_count = 0
            else:
                stall_count += 1

        # Stall handling
        if stall_count > 300:
            if stall_count % 100 == 0:
                # Inject structured noise
                noise_scale = min(0.03, 0.005 * (stall_count / 100.0))
                a = a + noise_scale * rng.standard_normal(n)

                a = a / a.sum()
                stall_count = 0

        if verbose and t % 5000 == 0:

def spectral_shaping(a0, time_limit, seed=42, verbose=True):

    representation to reduce time-domain peaks.

    shape |FFT(a)|^2 to be more "spread out" in frequency.

    1. Compute current b and its peaks
    2. Compute a "target" |A|^2 that would give a flatter b
    3. Interpolate between current and target
    4. Recover a via IFFT and project onto {a >= 0, sum(a)=1}

    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()

    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_score = fast_score(a)

            break

                if stall > 200:
                    break

        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Spectral it={it} ({elapsed:.0f}s): best={best_score:.6f}, bmax={b_max:.6e}")

    return best_a, best_score


def block_peak_shaving(a0, time_limit, seed=123, verbose=True):
    """
    Block-based peak shaving: redistribute mass in small blocks (5-20 elements)
    rather than single pairs. More effective at reducing multiple peaks simultaneously.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = fast_score(a)
    t0 = time.monotonic()
    stall = 0

    block_sizes = [3, 5, 8, 12, 16, 25]

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        top_k = np.argsort(b)[::-1][:min(8, n)]

        improved = False
        for k_idx in top_k:
            k_idx = int(k_idx)
            
            # Find the positions contributing most to this peak
            js = max(0, k_idx - n + 1)
            je = min(n - 1, k_idx)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k_idx - idx]
            
            if contrib.sum() < 1e-15:
                continue
            
            # Source: block around the highest contributing position
            src_center = int(idx[np.argmax(contrib)])
            bs = rng.choice(block_sizes)
            half_bs = bs // 2
            src_start = max(0, src_center - half_bs)
            src_end = min(n, src_start + bs)
            src_start = max(0, src_end - bs)
            
            # Target: block in a low-contribution region
            # Find positions not in any top peak's contribution range
            peak_mask = np.zeros(n, dtype=bool)
            for k2 in top_k[:4]:
                k2 = int(k2)
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                peak_mask[js2:je2 + 1] = True
            
            # Target: lowest-sum region not in peak mask
            candidate_regions = []
            step = max(1, n // 50)
            for start in range(0, n - bs + 1, step):
                if not peak_mask[start:start+bs].any():
                    candidate_regions.append((start, a[start:start+bs].sum()))
            
            if not candidate_regions:
                # Fallback: any region not containing src
                candidate_regions = [(i, a[i:i+bs].sum()) for i in range(0, n-bs+1, step) 
                                     if i + bs <= src_start or i >= src_end]
            
            if not candidate_regions:
                continue
            
            # Pick the lowest-sum region
            candidate_regions.sort(key=lambda x: x[1])
            dst_start = candidate_regions[0][0]
            
            # Try moving different fractions of the block mass
            src_mass = a[src_start:src_end].sum()
            if src_mass < 1e-15:
                continue
                
            for frac in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
                # Create trial: move frac of src block to dst block
                a_trial = a.copy()
                src_block = a_trial[src_start:src_end].copy()
                # Reduce src block proportionally
                a_trial[src_start:src_end] = src_block * (1 - frac)
                # Add to dst block proportionally
                dst_block = a_trial[dst_start:dst_start+bs]
                if dst_block.sum() > 1e-15:
                    # Distribute proportionally
                    a_trial[dst_start:dst_start+bs] = dst_block + src_block * frac * (dst_block / dst_block.sum())
                else:
                    # Distribute uniformly
                    a_trial[dst_start:dst_start+bs] = dst_block + src_block.sum() * frac / bs
                
                a_trial = np.maximum(a_trial, 0)
                s_t = a_trial.sum()
                if s_t > 1e-15:
                    a_trial = a_trial / s_t
                    sc = fast_score(a_trial)
                    if sc < best_score - 1e-12:
                        a = a_trial.copy()
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
            
            if improved:
                break

        if not improved:
            stall += 1
            if rng.integers(5) == 0:
                # Random perturbation
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i != j and a[i] > 1e-10:
                    frac = rng.uniform(0.01, 0.05)
                    amt = a[i] * frac
                    a_trial = a.copy()
                    a_trial[i] -= amt
                    a_trial[j] += amt
                    if a_trial[i] >= 0:
                        a_trial = a_trial / a_trial.sum()
                        sc = fast_score(a_trial)
                        if sc < best_score:
                            a = a_trial.copy()
                            best_score = sc
                            best_a = a.copy()
                            stall = 0

            if stall > 500:
                break

    if verbose:
        elapsed = time.monotonic() - t0
        print(f"    BlockShave ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def diverse_init(n, rng, variant=None):
    """Generate diverse structured initializations."""
    if variant is None:
        variant = int(rng.integers(0, 12))

    if variant == 0:
        x = np.linspace(0, 1, n)
        a = np.maximum(1 - 2 * abs(x - 0.5), 0)
    elif variant == 1:
        x = np.linspace(0, 1, n)
        a = 0.5 * (1 + np.cos(2 * np.pi * x))
    elif variant == 2:
        x = np.linspace(-1, 1, n)
        a = np.exp(-3 * x ** 2)
    elif variant == 3:
        x = np.linspace(0, 1, n)
        a = np.exp(-20 * (x - 0.25) ** 2) + np.exp(-20 * (x - 0.75) ** 2)
    elif variant == 4:
        x = np.linspace(0, 1, n
        a = x ** 2 + (1 - x) ** 2 + 0.1
    elif variant == 5:
        x = np.linspace(0, 1, n)
        a = 1 + 0.5 * np.cos(4 * np.pi * x)
    elif variant == 6:
        x = np.linspace(-1, 1, n)
        a = np.sinc(x) ** 2
    elif variant == 7:
        x = np.linspace(0, 1, n)
        a = x + 0.1
    elif variant == 8:
        x = np.linspace(0, 2 * np.pi, n)
        a = 1 + 0.3 * np.cos(3 * x) + 0.1 * np.cos(7 * x)
    elif variant == 9:
        x = np.linspace(0, 1, n)
        a = abs(x - 0.5) + 0.2
    elif variant == 10:
        # Spectral initialization: IFFT of a shaped spectrum
        fft_size = 1 << (2*n-1).bit_length()
        # Create a spectrum with cosine roll-off
        freqs = np.arange(fft_size) / fft_size
        # Low-pass with smooth transition
        cutoff = 0.3
        mag = np.where(freqs < cutoff, 1.0, np.exp(-5*(freqs-cutoff)/0.1))
        # Symmetrize
        mag = np.maximum(mag, mag[fft_size - np.arange(fft_size)])
        A = mag * np.exp(1j * rng.standard_normal(fft_size))
        # Fix Hermitian symmetry
        for k in range(1, fft_size // 2 + 1):
            A[fft_size - k] = np.conj(A[k])
        A[0] = np.abs(A[0])
        a = ifft(A).real
        a = np.maximum(a, 0)
    elif variant == 11:
        # Multi-Gaussian
        a = np.zeros(n)
        for _ in range(rng.integers(2, 6)):
            center = rng.uniform(0, n)
            width = rng.uniform(n * 0.05, n * 0.3)
            amp = rng.uniform(0.5, 2.0)
            x = np.arange(n, dtype=np.float64)
            a += amp * np.exp(-(x - center) ** 2 / (2 * width ** 2))
    else:
        a = np.ones(n)

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp
