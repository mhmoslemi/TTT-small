def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(arr):
    """Safely evaluate a sequence."""

def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

    a = np.maximum(np.asarray(a, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:

    return a / s

def spectral_shaping(a, n_iter=500, alpha=0.3):
    """Spectral shaping: iteratively flatten the power spectrum."""
    n = len(a)
    a = normalize(a)
    fft_size = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()

    for i in range(n_iter):
        A = rfft(a, fft_size)
        B = np.abs(A) ** 2
        non_dc = B[1:]
        target = np.exp(np.mean(np.log(non_dc + 1e-30)))
        scale = np.ones(len(A))
        ratio = target / (B[1:] + 1e-30)
        ratio = np.clip(ratio, 0.05, 20.0)
        scale[1:] = np.power(ratio, alpha / 2.0)
        A_new = A * scale
        a_new = irfft(A_new, fft_size)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a

def adam_lse_optimize(a0, time_limit, T_start=100.0, T_end=0.05, n_stages=20, lr=0.01):
    """LogSumExp-smoothed optimization with Adam optimizer."""
    n = len(a0)
    a_init = normalize(a0)
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_adam = 0

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        iters = max(20, min(200, int(remaining * 100 / (n_stages - si))))

        for it in range(iters):
            if time.monotonic() - t0 > time_limit:
                break
            obj, grad = lse_obj_grad_z(z, stage_T)
            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)
            lr_t = lr / np.sqrt(1 + 0.01 * it)
            z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)

            if it % 10 == 0:
                a = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a).max()
                if score < best_score:
                    best_score = score
                    best_a = a.copy()

        a = softmax_fn(z)
        score = 2.0 * n * fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()

        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      Adam-LSE stage {si}/{n_stages} (T={stage_T:.2f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def get_lp_solver():
    """Find a working LP solver, preferring faster ones."""
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(5, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a_fixed):
    """Build the (2n-1) x n matrix T where (T@b) = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. conv(a_fixed, b) <= M, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, max_iters=100000, use_smooth_br=False, smooth_br_steps=30):
    """
    Nesterov-accelerated LP best-response with:
    - Adaptive step size based on observed range
    - Sliding window averaging (last 500 iterates)
    - Optional smooth best response (faster than LP)
    """
    n = len(a0)
    a = normalize(a0)
    y = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    # Sliding window for averaging
    window_size = 500
    window = [a.copy()]
    window_sum = a.copy()

    # Adaptive parameters
    M_values = []
    fft_size = 1 << (2 * n - 1).bit_length()
    T = a.copy()  # alias for current y
    T_mat = build_conv_matrix(a)

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        # Get best response to y
        if use_smooth_br:
            b_star = smooth_best_response(y, T=50.0, n_steps=smooth_br_steps, fft_size=fft_size)
            M_val = None
        else:
            b_star, M_val = solve_lp_best_response(y, solver)
            if b_star is None:
                # Fallback to smooth best response
                b_star = smooth_best_response(y, T=50.0, n_steps=20, fft_size=fft_size)
                M_val = None

        # Nesterov update
        alpha_t = 2.0 / (t + 2.0)
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        s = y_new.sum()
        if s < 1e-15:
            y_new = y.copy()
        else:
            y_new = y_new / s

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                a_new = y_new.copy()
            else:
                a_new = a_new / s

        y = y_new
        a = a_new

        # Track M values for adaptive step
        if M_val is not None:
            M_values.append(M_val)
            if len(M_values) > 200:
                M_values.pop(0)

        # Update sliding window
        window_sum = window_sum - window[0] + a_new
        if len(window) >= window_size:
            window.pop(0)
        window.append(a_new.copy())

        # Evaluate
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()

        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()

        # Check window average periodically
        if t % 50 == 0 and len(window) >= 100:
            avg_a = window_sum / len(window)
            score_avg = 2.0 * n * fast_autocorr(avg_a).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_a.copy()

        if t % 500 == 0:
            elapsed = time.monotonic() - t0
            msg = f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}"
            if M_val is not None:
                msg += f", LP_M={M_val:.8f}"
            print(msg)

    return best_a, best_score


def directed_local_search(a0, time_limit, seed=42):
    """
    Directed local search: uses convolution structure to make targeted moves.
    Identifies peak contributors and redistributes mass optimally.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    fft_size = 1 << (2 * n - 1).bit_length()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        peak_idx = np.argmax(b)
        peak_val = b[peak_idx]

        # Find top contributors to the peak
        js = max(0, peak_idx - n + 1)
        je = min(n - 1, peak_idx)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[peak_idx - idx]
        top_contrib_idx = idx[np.argsort(contrib)[::-1][:5]]

        # Try moving mass from top contributors to random other positions
        improved = False
        for src in top_contrib_idx:
            if a[src] < 1e-12:
                continue
            # Choose destination: prefer positions that don't contribute to top peaks
            top_peaks = np.argsort(b)[::-1][:10]
            bad_positions = set()
            for kp in top_peaks:
                js2 = max(0, kp - n + 1)
                je2 = min(n - 1, kp)
                for j2 in range(js2, je2 + 1):
                    if a[src] * a[j2] > 0.01 * peak_val:
                        bad_positions.add(j2)

            # Pick a good destination
            good_positions = [j for j in range(n) if j not in bad_positions and j != src]
            if not good_positions:
                good_positions = [j for j in range(n) if j != src]

            # Try several destinations with different fractions
            for _ in range(5):
                dst = good_positions[rng.integers(len(good_positions))]
                for frac in [0.001, 0.003, 0.007, 0.015, 0.03]:
                    amt = a[src] * frac
                    if amt < 1e-15:
                        continue
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score - 1e-12:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Random exploration
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-12:
                frac = rng.uniform(0.002, 0.02)
                amt = a[i] * frac
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                if a_trial[i] >= 0:
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            if stall > 200:
                break

    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 8)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 2)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 1.5
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.6 + 0.3 * np.cos(3 * x) + 0.1 * np.cos(7 * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-3 * (x - 0.3) ** 2) + 0.5 * np.exp(-3 * (x - 0.7) ** 2)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.cos(4 * np.pi * x ** 2)
        a = np.maximum(a, 0)
    else:
        x = np.linspace(0, 1, n)
        tri = 1 - 2 * np.abs(x - 0.5)
        a = tri * (1 + 0.2 * np.cos(6 * np.pi * x))
        a = np.maximum(a, 0)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s

        a_spec = spectral_shaping(init_long, n_iter=800, alpha=0.2)
        sc = evaluate_safe(a_spec)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = normalize(a_spec)
            print(f"  *** NEW BEST [spectral_n{n_long}]: {best_score:.6f} (t={time.monotonic()-t_global_start:.0f}s)")
        t4b = t4 * 0.6
        a_out, _ = adam_lse_optimize(a_spec, t4b, T_start=100.0, T_end=0.05, n_stages=15, lr=0.003)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = normalize(a_out)
            print(f"  *** NEW BEST [adam_lse_n{n_long}]: {best_score:.6f} (t={time.monotonic()-t_global_start:.0f}s)")

    # Phase 5: Nesterov LP at longer length (~300s)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 5: Nesterov LP at n={n_long} (remaining={remaining:.0f}s) ===")
    t5 = min(remaining * 0.25, 350.0)
    if t5 > 30:
        init5 = interpolate_sequence(best_seq, n_long)
        a_out, _ = nesterov_lp_optimize(init5, t5, solver, max_iters=100000, use_smooth_br=True, smooth_br_steps=25)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = normalize(a_out)
            print(f"  *** NEW BEST [lp_smooth_n{n_long}]: {best_score:.6f} (t={time.monotonic()-t_global_start:.0f}s)")

    # Phase 6: Back to base length Nesterov LP with full solver (~200s)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 6: Nesterov LP full at best n (remaining={remaining:.0f}s) ===")
    t6 = min(remaining * 0.20, 200.0)
    if t6 > 30:
        n_best = len(best_seq)
        a_out, _ = nesterov_lp_optimize(best_seq, t6, solver, max_iters=100000, use_smooth_br=False)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = normalize(a_out)
            print(f"  *** NEW BEST [lp_full_n{n_best}]: {best_score:.6f} (t={time.monotonic()-t_global_start:.0f}s)")

    # Phase 7: Directed local search (~100s)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 7: Directed local search (remaining={remaining:.0f}s) ===")
    t7 = min(remaining * 0.20, 120.0)
    if t7 > 5:
        a_out, _ = directed_local_search(best_seq, t7, seed=123)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = normalize(a_out)
            print(f"  *** NEW BEST [local_search]: {best_score:.6f} (t={time.monotonic()-t_global_start:.0f}s)")

    # Phase 8: Final Adam LSE polish (~60s)
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 8: Final Adam LSE polish (remaining={remaining:.0f}s) ===")
    t8 = min(remaining * 0.20, 60.0)
    if t8 > 10:
        a_out, _ = adam_lse_optimize(best_seq, t8, T_start=5.0, T_end=0.01, n_stages=10, lr=0.002)
        sc = evaluate_safe(a_out)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = normalize(a_out)
            print(f"  *** NEW BEST [final_adam]: {best_score:.6f} (t={time.monotonic()-t_global_start:.0f}s)")

    # Phase 9: One more local search pass (~50s)
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 10:
        t9 = min(remaining * 0.30, 50.0)
        if t9 > 5:
            a_out, _ = directed_local_search(best_seq, t9, seed=456)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = normalize(a_out)
                print(f"  *** NEW BEST [local_search_2]: {best_score:.6f} (t={time.monotonic()-t_global_start:.0f}s)")

    # Phase 10: Final spectral touch (~20s)
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 5:
        t10 = min(remaining * 0.4, 20.0)
        if t10 > 3:
            a_out = spectral_shaping(best_seq, n_iter=200, alpha=0.05)
            sc = evaluate_safe(a_out)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_seq = normalize(a_out)
                print(f"  *** NEW BEST [final_spectral]: {best_score:.6f} (t={time.monotonic()-t_global_start:.0f}s)")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
            best_seq = np.maximum(best_seq, 0)
            s = best_seq.sum()
            if s > 0:
                best_seq = best_seq / s
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
