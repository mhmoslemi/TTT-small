Wait, that's not right either. If i ≠ k/2, the two terms j=i and j=k-i are DIFFERENT terms in the sum, and both contribute. So:

= ... + 2*a[i]*a[k-i] + ...
df_k/da[i] = 2*a[k-i]

If i = k/2: f_k = ... + a[i]*a[i] + ... (one term)
= ... + a[i]^2 + ...
df_k/da[i] = 2*a[i] = 2*a[k-i] (since k-i = i)

So in ALL cases: df_k/da[i] = 2*a[k-i] (with a[k-i]=0 if out of range).

Great, so the gradient of f_k with respect to a is simply 2 * reverse_shift(a, k), i.e., 2 * a_flipped where a_flipped[i] = a[k-i] (with zero padding).

This means the subgradient of max_k f_k(a) is 2 * a[k*-i] where k* is the argmax lag.

This is very simple! We can use this gradient directly in a gradient-based method.

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def get_lp_solver():
    """Find a working LP solver, preferring faster ones."""
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            t0 = time.monotonic()

def solve_lp_best_response(a_fixed, solver):
    """Solve LP: min M s.t. conv(a_fixed, b)[k] <= M for all k, sum(b)=1, b>=0."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build convolution matrix T where (T @ b)[k] = conv(a_fixed, b)[k]
    # T has shape (2n-1, n), T[k, j] = a_fixed[k-j] if 0<=k-j<n else 0
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        lo = j
        hi = j + n
        T[lo:hi, j] = a_fixed
    
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    try:
        prob.solve(solver=solver)
    except Exception:
        pass
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def compute_gradient_smooth(a, b, t_smooth=20.0):
    """
    Compute smooth approximation gradient of max_k conv(a,a)[k].
    Uses log-sum-exp smoothing with parameter t_smooth.
    Gradient of f_k(a) = conv(a,a)[k] is 2*a[k-i] (reversed/shifted).
    """
    n = len(a)
    b = fast_autocorr(a)
    max_b = b.max()
    
    # Softmax weights over top lags
    shifted = b - max_b
    weights = np.exp(t_smooth * shifted)
    wsum = weights.sum()
    if wsum < 1e-300:
        weights = np.zeros_like(b)
        weights[np.argmax(b)] = 1.0
        wsum = 1.0
    weights /= wsum
    
    # Gradient: weighted sum of 2*a[k-i] for each k
    grad = np.zeros(n)
    # For efficiency, only process lags with significant weight
    threshold = weights.max() * 0.001
    active = np.where(weights > threshold)[0]
    
    for k in active:
        w = weights[k]
        if w < 1e-10:
            continue
        # grad contribution: 2*w*a[k-i] for i in valid range
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        other = k - idx
        grad[idx] += 2.0 * w * a[other]
    
    return grad, max_b, len(active)


def fista_smooth(a0, time_limit, t_smooth=50.0):
    """
    FISTA-like accelerated gradient method with smooth approximation.
    Minimizes max_k conv(a,a)[k] on the simplex using log-sum-exp smoothing.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()
    t_param = 1.0
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Simplex projection
    def project_simplex(v):
        d = len(v)
        if d == 1:
            return np.array([max(v[0], 0.0)])
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u)
        rho = np.nonzero(u * np.arange(1, d + 1) > (cssv - 1))[0][-1]
        theta = (cssv[rho] - 1) / (rho + 1.0)
        proj = np.maximum(v - theta, 0)
        return proj
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        # Compute smooth gradient at y
        grad, max_b, n_active = compute_gradient_smooth(y, None, t_smooth)
        
        # Adaptive step size: use Lipschitz estimate
        # The Hessian of f_k has eigenvalues bounded by ~4*n (rough bound)
        # For smooth approx, use a practical step size
        lr = 1.0 / (2.0 * np.sqrt(n) + 1)
        lr = min(lr, 0.5 / max(np.max(np.abs(grad)), 1e-10))
        lr = max(lr, 1e-6)
        
        # Gradient step
        z = y - lr * grad
        z_new = project_simplex(z)
        z_new_s = z_new.sum()
        if z_new_s < 1e-15:
            z_new = np.ones(n) / n
        else:
            z_new = z_new / z_new_s
        
        # FISTA update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param))
        y_new = z_new + ((t_param - 1.0) / t_new) * (z_new - y)
        y_new = np.maximum(y_new, 0)
        y_new_s = y_new.sum()
        if y_new_s < 1e-15:
            y_new = z_new.copy()
        else:
            y_new = y_new / y_new_s
        
        y = y_new
        t_param = t_new
        a = z_new
        it += 1
        
        # Evaluate
        score = 2.0 * n * fast_autocorr(a).max()
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
        
        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    FISTA it={it} ({elapsed:.0f}s): best={best_score:.6f}, cur={score:.6f}")
        
        # Adapt smoothing: decrease as we converge
        if it % 500 == 0:
            t_smooth = max(10.0, t_smooth * 0.8)
    
    return best_a, best_score


def active_set_qp(a0, time_limit, n_active_max=40):
    """
    QP refinement on active constraint set.
    Identifies lags where conv(a,a) is near max and solves a QP.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    
    for round_num in range(50):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_autocorr(a)
        max_b = b.max()
        
        # Identify active lags (within 1% of max)
        threshold = max_b * 0.99
        active_lags = np.where(b >= threshold)[0]
        
        # Keep top n_active_max
        if len(active_lags) > n_active_max:
            active_lags = active_lags[np.argsort(b[active_lags])[::-1][:n_active_max]]
        
        if len(active_lags) == 0:
            active_lags = np.array([np.argmax(b)])
        
        # Build QP: min M s.t. a^T B_k a <= M for k in active_lags, sum(a)=1, a>=0
        a_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        
        constraints = [cp.sum(a_var) == 1, M_var >= 0]
        for k in active_lags:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            other_idx = k - idx
            q_expr = cp.sum(a_var[idx] * a_var[other_idx])
            constraints.append(q_expr <= M_var)
        
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iter=20000, tol_abs=1e-8, tol_rel=1e-8)
        except Exception:
            try:
                prob.solve(solver=cp.SCS, max_iter=20000)
            except Exception:
                break
        
        if prob.status not in ('optimal', 'optimal_inaccurate'):
            try:
                prob.solve(solver=cp.CLARABEL)
            except Exception:
                break
            if prob.status not in ('optimal', 'optimal_inaccurate'):
                break
        
        a_new = np.asarray(a_var.value, dtype=np.float64)
        a_new = np.maximum(a_new, 0)
        a_new_s = a_new.sum()
        if a_new_s < 1e-15:
            break
        a_new = a_new / a_new_s
        
        score = 2.0 * n * fast_autocorr(a_new).max()
        improved = False
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            improved = True
            print(f"    QP round {round_num}: {best_score:.6f} (active={len(active_lags)})")
        else:
            # Try mixing
            for frac in [0.3, 0.5, 0.7, 0.9]:
                a_mix = (1 - frac) * a + frac * a_new
                a_mix = np.maximum(a_mix, 0)
                a_mix_s = a_mix.sum()
                if a_mix_s < 1e-15:
                    continue
                a_mix = a_mix / a_mix_s
                score_mix = 2.0 * n * fast_autocorr(a_mix).max()
                if score_mix < best_score:
                    best_score = score_mix
                    best_a = a_mix.copy()
                    a = a_mix
                    improved = True
                    print(f"    QP mix {frac} round {round_num}: {best_score:.6f}")
                    break
        
        if not improved and round_num >= 5:
            # Widen active set slightly
            threshold = max(fast_autocorr(a).max() * 0.97, 0)
            break
    
    return best_a, best_score


def coordinate_descent(a0, time_limit, seed=42):
    """
    Coordinate descent: move mass from positions contributing to the max peak
    to positions that reduce the peak.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_autocorr(a)
        max_b = b.max()
        k_max = np.argmax(b)
        
        # Find the lag with maximum and identify contributing pairs
        js = max(0, k_max - n + 1)
        je = min(n - 1, k_max)
        idx = np.arange(js, je + 1)
        other = k_max - idx
        contrib = a[idx] * a[other]
        total = contrib.sum()
        
        if total < 1e-15:
            break
        
        # Find top contributors to the peak
        n_top = min(8, len(idx))
        top_ci = np.argsort(contrib)[::-1][:n_top]
        
        improved_this_round = False
        for ci in top_ci:
            src = idx[ci]
            src_other = other[ci]
            
            if src == src_other:
                # Move from a[src] to some other position
                sources = [(src, a[src])]
            else:
                sources = [(src, a[src]), (src_other, a[src_other])]
            
            for s_pos, s_val in sources:
                if s_val < 1e-10:
                    continue
                
                # Find best target: minimize contribution to top peaks
                top_lags = np.argsort(b)[::-1][:5]
                # Score each position by its "peak involvement"
                peak_inv = np.zeros(n)
                for k2 in top_lags:
                    js2 = max(0, k2 - n + 1)
                    je2 = min(n - 1, k2)
                    idx2 = np.arange(js2, je2 + 1)
                    other2 = k2 - idx2
                    peak_inv[idx2] += a[other2]
                    peak_inv[other2] += a[idx2]
                
                # Target: position with lowest peak involvement and current value
                exclusion = set()
                if src != src_other:
                    exclusion.add(src_other)
                exclusion.add(s_pos)
                
                target_scores = []
                for j in range(n):
                    if j in exclusion:
                        continue
                    # Lower peak involvement is better; also prefer positions with lower a[j]
                    score_j = -peak_inv[j] - 0.1 * a[j] * n
                    target_scores.append((score_j, j))
                
                target_scores.sort(reverse=True)
                
                for _, dst in target_scores[:5]:
                    for frac in [0.001, 0.003, 0.005, 0.01, 0.02, 0.04]:
                        amt = s_val * frac
                        if amt < 1e-12:
                            continue
                        a_trial = a.copy()
                        a_trial[s_pos] -= amt
                        a_trial[dst] += amt
                        if a_trial[s_pos] < -1e-12:
                            continue
                        a_trial = np.maximum(a_trial, 0)
                        a_trial_s = a_trial.sum()
                        if a_trial_s < 1e-15:
                            continue
                        a_trial = a_trial / a_trial_s
                        score = 2.0 * n * fast_autocorr(a_trial).max()
                        if score < best_score - 1e-12:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved_this_round = True
                            stall = 0
                            break
                    if improved_this_round:
                        break
                if improved_this_round:
                    break
            if improved_this_round:
                break
        
        if not improved_this_round:
            stall += 1
            # Random perturbation every 20 stalls
            if stall % 20 == 0:
                n_pert = rng.integers(1, 4)
                srcs = rng.choice(n, size=n_pert, replace=False)
                dsts = rng.choice(n, size=n_pert, replace=False)
                a_trial = a.copy()
                for s_p, d_p in zip(srcs, dsts):
                    frac = rng.uniform(0.005, 0.03)
                    amt = a_trial[s_p] * frac
                    a_trial[s_p] -= amt
                    a_trial[d_p] += amt
                a_trial = np.maximum(a_trial, 0)
                a_trial_s = a_trial.sum()
                if a_trial_s < 1e-15:
                    continue
                a_trial = a_trial / a_trial_s
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
            
            if stall > 2000:
                break
    
    return best_a, best_score


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """
    LP Nesterov (multiplicative weights) with:
    - Adaptive momentum
    - Running average of all iterates (best for convergence)
    - Periodic smoothing of the iterate
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Running average (this gives the best convergence rate for MW)
    run_avg = a.copy()
    n_avg = 1
    
    # Also keep an EMA with adaptive rate
    ema = a.copy()
    ema_rate = 0.03
    
    iter_count = 0
    last_improve_iter = 0
    
    for t in range(1, 1000000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue
        iter_count += 1
        
        # Nesterov step size
        if t < 50:
            alpha_t = 2.0 / (t + 2.0)
        else:
            alpha_t = 2.0 / (t + 2.0)
        
        # Update y
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_new_s = y_new.sum()
        if y_new_s < 1e-15:
            y_new = y.copy()
            y_new_s = 1.0
        y_new = y_new / y_new_s
        y = y_new
        
        # Compute a (accelerated)
        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y_new + (y_new - y_new))  # placeholder
            # Standard Nesterov: a = y + (t-1)/(t+2) * (y - y_prev)
            # But we need y_prev. Let's use a simpler scheme:
            a_new = y_new.copy()
        a_new = np.maximum(a_new, 0)
        a_s = a_new.sum()
        if a_s < 1e-15:
            a_new = y.copy()
            a_s = 1.0
        a_new = a_new / a_s
        a = a_new
        
        # Update running average
        run_avg = (run_avg * n_avg + a_new) / (n_avg + 1)
        n_avg += 1
        
        # Update EMA
        ema = (1 - ema_rate) * ema + ema_rate * a_new
        
        # Score multiple candidates
        for candidate, label in [(a_new, 'a'), (y, 'y'), (run_avg, 'avg'), (ema, 'ema')]:
            score = 2.0 * n * fast_autocorr(candidate).max()
            if score < best_score:
                best_score = score
                best_a = candidate.copy()
                last_improve_iter = t
        
        # Adapt EMA rate
        if t - last_improve_iter > 500:
            ema_rate = min(0.1, ema_rate * 1.1)
        elif t - last_improve_iter < 50:
            ema_rate = max(0.01, ema_rate * 0.95)
        
        if verbose and t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.8f}")
        
        # Early stopping
        if t - last_improve_iter > 10000 and t > 2000:
            if verbose:
                print(f"    Early stop at t={t} (no improvement for 10000 iters)")
            break
    
    return best_a, best_score, iter_count


def make_exponential(n, alpha=0.5, center=False):
    """Create an exponential decay sequence."""
    if center:
        x = np.abs(np.arange(n) - n // 2)
        a = np.exp(-alpha * x)
    else:
        x = np.arange(n)
        a = np.exp(-alpha * x)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_triangle(n, sharpness=1.0):
    """Create a triangle/cone-shaped sequence."""
    x = np.arange(n, dtype=np.float64)
    peak = int(n * 0.3)  # Peak at 30% from left (lopsided)
    a = np.exp(-sharpness * np.abs(x - peak) / n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_sine(n, freq=1.0, phase=0.0):
    """Create a sine-based sequence (always non-negative)."""
    x = np.linspace(0, np.pi, n)
    a = np.cos(x / 2 + phase * (x / np.pi - 0.5)) ** 2
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search function. Multi-phase optimization combining LP Nesterov,
    smooth FISTA, active-set QP, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    t_global_start = time.monotonic()
    time_budget = min(float(budget_s) - 60.0, 1740.0)
    
    solver = get_lp_solver()
    print(f"Using LP solver: {solver}")
    print(f"Time budget: {time_budget:.0f}s")
    
    # --- Initialization ---
    best_seq = None
    best_score = float('inf')
    parent = None
    n_base = 1600
    
    # Try to get parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
            n_base = len(parent)
    except Exception:
        pass
    
    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            elapsed = time.monotonic() - t_global_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
    
    # Generate multiple initial candidates at n_base
    inits = {
        'exp_r10': make_exponential(n_base, alpha=10.0 / n_base),
        'exp_r50': make_exponential(n_base, alpha=50.0 / n_base),
        'exp_r100': make_exponential(n_base, alpha=100.0 / n_base),
        'tri_05': make_triangle(n_base, sharpness=0.5),
        'tri_10': make_triangle(n_base, sharpness=1.0),
        'tri_20': make_triangle(n_base, sharpness=2.0),
        'sine': make_sine(n_base),
        'sine_p05': make_sine(n_base, phase=0.5),
        'sine_p1': make_sine(n_base, phase=1.0),
    }
    
    # Rank inits by score
    init_scores = {}
    for name, init in inits.items():
        sc = evaluate_safe(init)
        init_scores[name] = (sc, init)
        print(f"  Init '{name}': score={sc:.6f}")
    
    # If we have parent, include it
    if parent is not None:
        init_scores['parent'] = (best_score, parent)
    
    # Sort by score
    sorted_inits = sorted(init_scores.items(), key=lambda x: x[1][0])
    print(f"\n  Best init: {sorted_inits[0][0]} with score {sorted_inits[0][1][0]:.6f}")
    
    # Use the best init as starting point
    if sorted_inits[0][1][0] < best_score:
        best_seq = sorted_inits[0][1][1].copy()
        best_score = sorted_inits[0][1][0]
        n_base = len(best_seq)
    
    # --- Phase 1: LP Nesterov at base length ---
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 1: LP Nesterov at n={n_base} (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.40, 600.0)
    if t1 > 60:
        a_out, sc_out, iters = nesterov_lp_optimize(best_seq, t1, solver, verbose=True)
        print(f"  LP Nesterov done: {iters} iters, score={sc_out:.6f}")
        try_update(a_out, f"lp1_n{n_base}")
    
    # --- Phase 2: FISTA smooth at base length ---
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 2: FISTA at n={n_base} (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.15, 200.0)
    if t2 > 30:
        a_out, sc_out = fista_smooth(best_seq, t2, t_smooth=30.0)
        print(f"  FISTA done: score={sc_out:.6f}")
        try_update(a_out, f"fista_n{n_base}")
    
    # --- Phase 3: Active-set QP ---
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 3: QP at n={n_base} (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.15, 150.0)
    if t3 > 20:
        a_out, sc_out = active_set_qp(best_seq, t3, n_active_max=30)
        print(f"  QP done: score={sc_out:.6f}")
        try_update(a_out, f"qp_n{n_base}")
    
    # --- Phase 4: Coordinate descent ---
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 4: Coord Descent at n={n_base} (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.10, 120.0)
    if t4 > 15:
        a_out, sc_out = coordinate_descent(best_seq, t4, seed=123)
        print(f"  CD done: score={sc_out:.6f}")
        try_update(a_out, f"cd_n{n_base}")
    
    # --- Phase 5: Try alternative inits at base length ---
    remaining = time_budget - (time.monotonic() - t_global_start)
    for rank, (name, (sc, init)) in enumerate(sorted_inits[1:4]):
        if remaining < 60:
            break
        print(f"\n=== Phase 5: LP from init '{name}' (score={sc:.4f}) (remaining={remaining:.0f}s) ===")
        t5 = min(remaining * 0.15, 100.0)
        if t5 < 30:
            break
        a_out, sc_out, iters = nesterov_lp_optimize(init, t5, solver, verbose=False)
        print(f"  LP from '{name}': {iters} iters, score={sc_out:.6f}")
        try_update(a_out, f"lp_init_{name}")
        remaining = time_budget - (time.monotonic() - t_global_start)
    
    # --- Phase 6: Try nearby lengths ---
    for n_try in [n_base - 100, n_base + 100, n_base - 200, n_base + 200]:
        if n_try < 200:
            continue
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining < 100:
            break
        print(f"\n=== Phase 6: n={n_try} (remaining={remaining:.0f}s) ===")
        init_n = interpolate_sequence(best_seq, n_try)
        t6 = min(remaining * 0.15, 120.0)
        if t6 < 30:
            continue
        a_out, sc_out, iters = nesterov_lp_optimize(init_n, t6, solver, verbose=False)
        print(f"  LP at n={n_try}: {iters} iters, score={sc_out:.6f}")
        try_update(a_out, f"lp_n{n_try}")
        
        # Quick QP
        remaining = time_budget - (time.monotonic() - t_global_start)
        if remaining > 40:
            t_qp = min(remaining * 0.1, 50.0)
            if t_qp > 10:
                a_qp, sc_qp = active_set_qp(a_out, t_qp, n_active_max=20)
                try_update(a_qp, f"qp_n{n_try}")
    
    # --- Phase 7: Second round of refinement on best ---
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 7: Refinement round 2 (remaining={remaining:.0f}s) ===")
    if remaining > 60:
        t7a = min(remaining * 0.20, 100.0)
        if t7a > 20:
            a_out, sc_out, iters = nesterov_lp_optimize(best_seq, t7a, solver, verbose=False)
            print(f"  LP round2: {iters} iters, score={sc_out:.6f}")
            try_update(a_out, "lp_round2")
    
    remaining = time_budget - (time.monotonic() - t_global_start)
    if remaining > 30:
        t7b = min(remaining * 0.20, 60.0)
        if t7b > 10:
            a_out, sc_out = active_set_qp(best_seq, t7b, n_active_max=25)
            print(f"  QP round2: score={sc_out:.6f}")
            try_update(a_out, "qp_round2")
    
    # --- Phase 8: Final coordinate descent polish ---
    remaining = time_budget - (time.monotonic() - t_global_start)
    print(f"\n=== Phase 8: Final CD polish (remaining={remaining:.0f}s) ===")
    if remaining > 15:
        t8 = min(remaining * 0.5, 80.0)
        if t8 > 5:
            a_out, sc_out = coordinate_descent(best_seq, t8, seed=999)
            print(f"  Final CD: score={sc_out:.6f}")
            try_update(a_out, "cd_final")
    
    # --- Final verification ---
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found! Using fallback.")
        # Fallback: uniform sequence
        fallback = np.ones(100) / 100
        print(f"  Fallback score: {evaluate_safe(fallback):.6f}")
        return [float(x) for x in fallback]
    
    return [float(x) for x in best_seq]
