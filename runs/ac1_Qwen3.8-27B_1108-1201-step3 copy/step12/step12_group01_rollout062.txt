"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Nesterov-accelerated mirror descent for the zero-sum game (no LP needed, O(n log n) per iteration)
2. Top-k LogSumExp gradient descent with Adam for refinement
3. Multi-start with different initializations and lengths
4. Peak shaving for final polish
Key innovation: Uses FFT-based cross-correlation to compute S(sigma)@v in O(n log n),
enabling thousands of game iterations vs hundreds of LP iterations.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time

def fast_autocorr(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def evaluate_safe(a):
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')

def softmax_fn(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s

def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def S_sigma_v(sigma, v):
    """Compute S(sigma) @ v = cross-correlation of sigma (len 2n-1) and v (len n)."""
    n = len(v)
    n_sigma = len(sigma)
    v_flipped = v[::-1]
    total_len = n_sigma + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    S = rfft(sigma, fft_size)
    V = rfft(v_flipped, fft_size)
    conv = irfft(S * V, fft_size)
    return conv[n - 1:2 * n - 1]

def game_optimize(a0, time_limit, seed=42):
    """
    Nesterov-accelerated mirror descent for the zero-sum game.
    minimizer: a (distribution over [n])
    maximizer: sigma (distribution over lags [0, 2n-2])
    Payoff: max_k (a*a)[k]
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    # sigma: distribution over lags
    sigma = np.ones(2 * n - 1) / (2 * n - 1)
    
    y = a.copy()  # Nesterov "y" point
    v_mom = np.zeros(n)  # Momentum for Nesterov
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Running average
    avg_a = a.copy()
    avg_count = 1
    WINDOW = 100
    buf = [a.copy() for _ in range(WINDOW)]
    buf_idx = 0
    
    stall = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit - 5:
        it += 1
        progress = min(1.0, it / 5000.0)
        
        # Learning rates
        lr_a = 0.5 / (1.0 + 0.1 * it)  # Decays
        lr_sigma = 0.3 / (1.0 + 0.05 * it)
        
        # Step 1: Compute gradient S(sigma) @ y
        g = S_sigma_v(sigma, y)
        
        # Step 2: Mirror descent step (use exponential / multiplicative update)
        # p_new ∝ exp(-lr * g) * y (natural exponential family)
        # But this might be unstable. Use additive with projection instead.
        y_new = y - lr_a * g
        y_new = project_simplex(y_new)
        
        # Step 3: Nesterov acceleration
        momentum = (it - 1.0) / (it + 2.0) if it > 1 else 0.0
        a_new = y_new + momentum * (y_new - y)
        a_new = project_simplex(a_new)
        
        y = y_new
        a = a_new
        
        # Step 4: Compute autocorrelation and update sigma
        c = fast_autocorr(a)
        # Soft update: use softmax of autocorrelation
        beta = 5.0 + 10.0 * progress  # Increasing sharpness
        sigma_new_target = softmax_fn(beta * c / max(c.max(), 1e-15))
        sigma = (1.0 - lr_sigma) * sigma + lr_sigma * sigma_new_target
        ss = sigma.sum()
        if ss > 1e-15:
            sigma = sigma / ss
        
        # Step 5: Track best
        score_a = 2.0 * n * c.max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Check y as well
        c_y = fast_autocorr(y)
        score_y = 2.0 * n * c_y.max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall = 0
        
        # Running average
        if buf[buf_idx] is not None:
            avg_a -= buf[buf_idx] / WINDOW
        buf[buf_idx] = a.copy()
        avg_a += a / WINDOW
        buf_idx = (buf_idx + 1) % WINDOW
        
        if it % 500 == 0:
            score_avg = 2.0 * n * fast_autocorr(avg_a).max()
            if score_avg < best_score:
                best_score = score_avg
                best_a = avg_a.copy()
                stall = 0
            elapsed = time.monotonic() - t0
            print(f"    Game it={it} ({elapsed:.0f}s): best={best_score:.6f}, score_y={score_y:.6f}")
        
        # Adaptive restart
        if stall > 500 and it > 100:
            # Reset to best with some perturbation
            a = best_a.copy()
            y = best_a.copy()
            sigma = softmax_fn(3.0 * fast_autocorr(a) / max(fast_autocorr(a).max(), 1e-15))
            buf = [a.copy() for _ in range(WINDOW)]
            avg_a = a.copy()
            stall = 0
            if it % 1000 == 0:
                print(f"    Game restart at it={it}, best={best_score:.6f}")
    
    return best_a, best_score

def topk_lse_optimize(a0, time_limit, T_start=100.0, T_end=0.2, n_stages=30, topk=15, seed=0):
    """
    Top-k LSE gradient descent with Adam.
    Uses the efficient gradient: grad = 2 * S_sigma_v(w_topk, a)
    where w_topk is the softmax of top-k autocorrelation values.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    
    # Work in log-space for stability
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit - 5:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        base_lr = 0.02 * min(1.0, 100.0 / n)
        lr = base_lr * (0.85 ** si)
        
        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.8, 0.999, 1e-8
        
        max_it = max(20, int(min(remaining, 60.0) * 5))
        
        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 3:
                break
            
            a = softmax_fn(z)
            c = fast_autocorr(a)
            
            # Top-k LSE
            k_use = min(topk, len(c))
            top_idx = np.argsort(c)[::-1][:k_use]
            top_vals = c[top_idx]
            
            Tc = stage_T * top_vals
            max_Tc = Tc.max()
            exp_Tc = np.exp(Tc - max_Tc)
            w_top = exp_Tc / exp_Tc.sum()
            
            # Build full weight vector for S_sigma_v
            w_full = np.zeros(2 * n - 1)
            w_full[top_idx] = w_top
            
            # Gradient in a-space: grad_a = 2 * S_sigma_v(w_full, a)
            grad_a = 2.0 * S_sigma_v(w_full, a)
            
            # Convert to z-space gradient
            S_dot = np.dot(grad_a, a)
            grad_z = a * (grad_a - S_dot)
            
            # Adam update
            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            # Periodic evaluation
            if (it + 1) % 5 == 0:
                a_check = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a_check).max()
                if score < best_score:
                    best_score = score
                    best_a = a_check.copy()
        
        # End of stage
        a_stage = softmax_fn(z)
        score_stage = 2.0 * n * fast_autocorr(a_stage).max()
        if score_stage < best_score:
            best_score = score_stage
            best_a = a_stage.copy()
        
        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage {si}/{n_stages} (T={stage_T:.1f}, {elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score

def peak_shaving_v2(a0, time_limit, seed=42):
    """
    Improved peak shaving: focus on reducing the top autocorrelation peaks
    by moving mass from high-contributing positions to low-impact positions.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit - 3:
        c = fast_autocorr(a)
        cmax = c.max()
        threshold = cmax * 0.99
        top_peaks = np.where(c >= threshold)[0]
        if len(top_peaks) == 0:
            top_peaks = np.argsort(c)[::-1][:10]
        
        improved = False
        for k in top_peaks[:15]:
            # Find which (i, k-i) pairs contribute most to c[k]
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            
            # Try to reduce the maximum contributing pair
            src = idx[np.argmax(contrib)]
            dst_candidate = k - src  # the other element in the pair
            
            # If we reduce a[src], the peak at k decreases
            # But we need to check that we don't create new peaks elsewhere
            # Strategy: move small amount from src to a "safe" position
            
            # Find safe positions (those that don't contribute to top peaks)
            danger = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:10]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                danger[js2:je2 + 1] = True
            
            safe_pos = np.where(~danger)[0]
            safe_pos = safe_pos[safe_pos != src]
            if len(safe_pos) == 0:
                safe_pos = np.array([j for j in range(n) if j != src])
            
            # Sort safe positions by current value (prefer low)
            safe_sorted = safe_pos[np.argsort(a[safe_pos])]
            pool_size = max(2, len(safe_sorted) // 4)
            dst_pool = safe_sorted[:pool_size]
            
            for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                amt = a[src] * frac
                if amt < 1e-14:
                    break
                for dst in rng.choice(dst_pool, size=min(3, len(dst_pool)), replace=False):
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:

def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]

def softmax_fn(z):
    """Numerically stable softmax."""
    z = np.asarray(z, dtype=np.float64)
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def project_simplex(v):
    """Project vector v onto the probability simplex {x: x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_arr = np.where(cond)[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

            momentum = (it - 1.0) / (it + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = project_simplex(a_new)
        else:
            a_new = y_new.copy()

        y = y_new
        a = a_new

        # Step 4: Compute autocorrelation and update sigma
        c = fast_autocorr(a)
        cmax = c.max()
        if cmax < 1e-15:
            beta = 5.0
        else:
            beta = 5.0 + 20.0 * progress
        sigma_target = softmax_fn(beta * c / cmax)
        sigma = (1.0 - lr_sigma) * sigma + lr_sigma * sigma_target
        ss = sigma.sum()
        if ss > 1e-15:
            sigma = sigma / ss

        # Step 5: Track best solutions
        score_a = 2.0 * n * cmax
        if score_a < best_score - 1e-12:
            best_score = score_a
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Also check y
        c_y = fast_autocorr(y)
        score_y = 2.0 * n * c_y.max()
        if score_y < best_score - 1e-12:
            best_score = score_y
            best_a = y.copy()
            stall = 0

        # Update running average
        avg_buf[buf_idx] = a
        buf_idx = (buf_idx + 1) % WINDOW
        if it > WINDOW:
            avg_filled = WINDOW

        # Periodically check running average
        if it % 300 == 0 and avg_filled == WINDOW:
            avg_a = avg_buf.mean(axis=0)
            score_avg = 2.0 * n * fast_autocorr(avg_a).max()
            if score_avg < best_score - 1e-12:
                best_score = score_avg
                best_a = avg_a.copy()
                stall = 0

        if it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Game it={it} ({elapsed:.0f}s): best={best_score:.6f}")

        # Adaptive restart with perturbation
        if stall > 600 and it > 200:
            a = best_a.copy()
            y = best_a.copy()
            c_best = fast_autocorr(a)
            sigma = softmax_fn(4.0 * c_best / max(c_best.max(), 1e-15))
            # Small random perturbation
            noise = rng.normal(0, 0.02, n) * a
            a_pert = a + noise
            a = project_simplex(a_pert)
            y = a.copy()
            for i in range(WINDOW):
                avg_buf[i] = a
            stall = 0
            if it % 2000 == 0:
                print(f"    Game restart at it={it}, best={best_score:.6f}")

    return best_a, best_score


def topk_lse_optimize(a0, time_limit, T_start=100.0, T_end=0.2, n_stages=30, topk=15, seed=0):
    """
    Top-k LSE gradient descent with Adam in log-space.
    Uses efficient gradient via S_sigma_v.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s

    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()

    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()

    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit - 5:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        base_lr = 0.015 * min(1.0, 120.0 / n)
        lr = base_lr * (0.88 ** si)

        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.8, 0.999, 1e-8

        max_it = max(20, int(min(remaining, 50.0) * 6))

        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 3:
                break

            a = softmax_fn(z)
            c = fast_autocorr(a)

            # Top-k selection
            k_use = min(topk, len(c))
            top_idx = np.argsort(c)[::-1][:k_use]
            top_vals = c[top_idx]

            # LSE weights
            Tc = stage_T * top_vals
            max_Tc = Tc.max()
            exp_Tc = np.exp(Tc - max_Tc)
            w_top = exp_Tc / exp_Tc.sum()

            # Build full weight vector
            w_full = np.zeros(2 * n - 1)
            w_full[top_idx] = w_top

            # Gradient in a-space
            grad_a = 2.0 * S_sigma_v(w_full, a)

            # Convert to z-space
            S_dot = np.dot(grad_a, a)
            grad_z = a * (grad_a - S_dot)

            # Adam
            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            # Periodic evaluation
            if (it + 1) % 5 == 0:
                a_check = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a_check).max()
                if score < best_score:
                    best_score = score
                    best_a = a_check.copy()
                    # Occasionally snap back to best to prevent drift
                    if (it + 1) % 25 == 0:
                        z = np.log(np.maximum(best_a, 1e-30))

        # End of stage
        a_stage = softmax_fn(z)
        score_stage = 2.0 * n * fast_autocorr(a_stage).max()
        if score_stage < best_score:
            best_score = score_stage
            best_a = a_stage.copy()

        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage {si}/{n_stages} (T={stage_T:.2f}, {elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def peak_shaving_opt(a0, time_limit, seed=42):
    """
    Peak shaving: iteratively reduce top autocorrelation peaks by moving mass
    from high-contributing positions to low-impact positions.
    """
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit - 3:
        c = fast_autocorr(a)
        cmax = c.max()
        threshold = cmax * 0.992
        top_peaks = np.where(c >= threshold)[0]
        if len(top_peaks) == 0:
            top_peaks = np.argsort(c)[::-1][:10]

        improved = False
        for k in top_peaks[:20]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            max_c_idx = np.argmax(contrib)
            src = idx[max_c_idx]

            # Build danger zone
            danger = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:15]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                danger[js2:je2 + 1] = True

            safe_pos = np.where(~danger)[0]
            safe_pos = safe_pos[safe_pos != src]
            if len(safe_pos) == 0:
                safe_pos = np.array([j for j in range(n) if j != src])
            if len(safe_pos) == 0:
                continue

            safe_sorted = safe_pos[np.argsort(a[safe_pos])]
            pool = safe_sorted[:max(3, len(safe_sorted) // 5)]

            for frac in [0.002, 0.005, 0.01, 0.02, 0.04, 0.06]:
                amt = a[src] * frac
                if amt < 1e-14:
                    break
                n_try = min(3, len(pool))
                for _ in range(n_try):
                    dst = pool[rng.integers(len(pool))]
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    a_trial /= a_trial.sum()
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 30:
                # Random exploration
                n_pert = rng.integers(1, 5)
                a_trial = a.copy()
                for _ in range(n_pert):
                    i = rng.integers(n)
                    j = rng.integers(n)
                    if i != j and a_trial[i] > 1e-10:
                        amt = a_trial[i] * rng.uniform(0.002, 0.04)
                        a_trial[i] -= amt
                        a_trial[j] += amt
                a_trial = np.maximum(a_trial, 0)
                st = a_trial.sum()
                if st > 1e-10:
                    a_trial /= st
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        stall = 0

            if stall > 300:
                break

    return best_a, best_score


def structured_init(n, rng):
    """Generate a structured initialization."""
    choice = rng.integers(0, 6)
    if choice == 0:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x)
    elif choice == 1:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.8 + 0.2 * np.cos(3 * x)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.exp(-4 * (x - 0.3) ** 2) + 0.6 * np.exp(-4 * (x - 0.7) ** 2)
    elif choice == 3:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 4:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 2 * (1 + 0.3 * np.cos(10 * np.pi * x))
    else:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.3 * np.sin(4 * np.pi * x) + 0.2 * np.cos(7 * np.pi * x)
    a = np.maximum(a, 0)
    sv = a.sum()
    if sv < 1e-15:
        return np.ones(n) / n
    return a / sv


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def propose_candidate(seed=42, time_budget=1800.0):
    """
    Multi-phase optimization:
    1. FFT-based game (Nesterov mirror descent) - main workhorse, O(n log n) per iter
    2. Top-k LSE with Adam - gradient refinement
    3. Second game round from improved start
    4. Peak shaving - local polish
    5. Longer length exploration
    6. Final LSE + peak shave
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    time_budget = min(float(time_budget) - 120.0, 1680.0)

    # Get parent sequence
    best_seq = None
    best_score = float('inf')
    parent = None

    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = evaluate_safe(parent)
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass

    if parent is None:
        n_base = 1600
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Default init: n={n_base}, score={best_score:.6f}")

    n_base = len(best_seq)

    # Helper: attempt to update best (inlined to avoid closure)
    def do_update(a, label, best_seq_ref, best_score_val):
        if a is None:
            return best_seq_ref, best_score_val
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 0.01:
            return best_seq_ref, best_score_val
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score_val:
            best_score_val = sc
            best_seq_ref = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  *** NEW BEST [{label}]: {sc:.6f} (n={len(a)}, t={elapsed:.0f}s)")
        return best_seq_ref, best_score_val

    # === Phase 1: Game at base length (main workhorse) ===
    remaining = time_budget - (time.monotonic() - t_start)
    print(f"\n=== Phase 1: Game n={n_base} ({remaining:.0f}s left) ===")
    t1 = min(remaining * 0.32, 550.0)
    if t1 > 30:
        a_out, _ = game_optimize(best_seq, t1, seed=seed)
        best_seq, best_score = do_update(a_out, f"game1_n{n_base}", best_seq, best_score)

    # === Phase 2: LSE polish ===
    remaining = time_budget - (time.monotonic() - t_start)
    t2 = min(remaining * 0.18, 220.0)
    if t2 > 20:
        print(f"=== Phase 2: LSE n={n_base} ({remaining:.0f}s left) ===")
        a_out, _ = topk_lse_optimize(best_seq, t2, T_start=80.0, T_end=0.2,
                                      n_stages=25, topk=12)
        best_seq, best_score = do_update(a_out, f"lse_n{n_base}", best_seq, best_score)

    # === Phase 3: Second game round ===
    remaining = time_budget - (time.monotonic() - t_start)
    t3 = min(remaining * 0.15, 200.0)
    if t3 > 20:
        print(f"=== Phase 3: Game round 2 n={n_base} ({remaining:.0f}s left) ===")
        a_out, _ = game_optimize(best_seq, t3, seed=seed + 100)
        best_seq, best_score = do_update(a_out, f"game2_n{n_base}", best_seq, best_score)

    # === Phase 4: Peak shaving ===
    remaining = time_budget - (time.monotonic() - t_start)
    t4 = min(remaining * 0.12, 120.0)
    if t4 > 10:
        print(f"=== Phase 4: Peak shaving ({remaining:.0f}s left) ===")
        a_out, _ = peak_shaving_opt(best_seq, t4, seed=789)
        best_seq, best_score = do_update(a_out, "peak_shave_1", best_seq, best_score)

    # === Phase 5: Longer length game ===
    remaining = time_budget - (time.monotonic() - t_start)
    if remaining > 80:
        n_long = n_base + 200
        t5 = min(remaining * 0.20, 220.0)
        if t5 > 30:
            print(f"=== Phase 5: Game n={n_long} ({remaining:.0f}s left) ===")
            init_long = interpolate_sequence(best_seq, n_long)
            a_out, _ = game_optimize(init_long, t5, seed=seed + 200)
            best_seq, best_score = do_update(a_out, f"game_n{n_long}", best_seq, best_score)

    # === Phase 6: LSE at longer length ===
    remaining = time_budget - (time.monotonic() - t_start)
    if remaining > 40:
        n_long2 = n_base + 200
        t6 = min(remaining * 0.12, 100.0)
        if t6 > 20:
            print(f"=== Phase 6: LSE n={n_long2} ({remaining:.0f}s left) ===")
            init_long2 = interpolate_sequence(best_seq, n_long2)
            a_out, _ = topk_lse_optimize(init_long2, t6, T_start=50.0, T_end=0.5,
                                          n_stages=15, topk=10)
            best_seq, best_score = do_update(a_out, f"lse_n{n_long2}", best_seq, best_score)

    # === Phase 7: Final peak shave ===
    remaining = time_budget - (time.monotonic() - t_start)
    if remaining > 10:
        t7 = min(remaining * 0.35, 80.0)
        if t7 > 5:
            print(f"=== Phase 7: Final peak shave ({remaining:.0f}s left) ===")
            a_out, _ = peak_shaving_opt(best_seq, t7, seed=101)
            best_seq, best_score = do_update(a_out, "final_shave", best_seq, best_score)

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    return [float(x) for x in best_seq]
