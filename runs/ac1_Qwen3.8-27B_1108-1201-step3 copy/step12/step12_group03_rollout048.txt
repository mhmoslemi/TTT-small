def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to new length."""

    old_x = np.linspace(0, 1, old_n)

            x_new = y_new + momentum * (y_new - y)
            x_new = np.maximum(x_new, 0.0)
            s_x = x_new.sum()
            if s_x < 1e-15:
                x_new = y_new.copy()
            else:
                x_new = x_new / s_x

        y = y_new
        a = x_new

        # Update running average (key for convergence guarantee)
        sum_b += b_star
        count += 1

        # Periodic scoring
        if t % score_interval == 0:
            avg_b = sum_b / count
            sc_avg = score_normalized(avg_b)
            if sc_avg < best_score:
                best_score = sc_avg
                best_a = avg_b.copy()

            sc_a = score_normalized(a)
            if sc_a < best_score:
                best_score = sc_a
                best_a = a.copy()

            if verbose and t % max(500, score_interval * 5) == 0:
                elapsed = time.monotonic() - t0
                print(f"    NLP t={t} ({elapsed:.0f}s): best={best_score:.6f}, M={M_val:.8f}")

    # Final scoring of all candidates
    avg_b = sum_b / count
    sc_avg = score_normalized(avg_b)
    if sc_avg < best_score:
        best_score = sc_avg
        best_a = avg_b.copy()

    sc_a = score_normalized(a)
    if sc_a < best_score:
        best_score = sc_a
        best_a = a.copy()

    return best_a, best_score, n_iters


def coord_descent_refine(a0, time_limit, verbose=True):
    """
    Coordinate descent refinement: iteratively move mass to reduce peak autocorrelation.
    """
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = score_normalized(a)
    rng = np.random.default_rng(99999)

    t0 = time.monotonic()
    stall = 0
    max_stall = 2000

    while time.monotonic() - t0 < time_limit and stall < max_stall:
        b = fast_autocorr(a)
        max_b = b.max()
        max_idx = np.argmax(b)

        # Get top peaks
        n_peaks = min(15, 2 * n - 1)
        top_peaks = np.argsort(b)[::-1][:n_peaks]

        improved = False
        for k in top_peaks:
            if b[k] < max_b * 0.995:
                break

            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]
            total = contrib.sum()
            if total < 1e-15:
                continue

            # Top contributors to this peak
            n_top = min(8, len(idx))
            top_ci = np.argsort(contrib)[::-1][:n_top]

            for ci in top_ci:
                src = int(idx[ci])
                src_other = int(k - src)

                # Score each target position: prefer positions far from peaks
                target_scores = np.empty(n)
                for j in range(n):
                    # Distance-based scoring: prefer positions that minimize
                    # contribution to all top peaks
                    penalty = 0.0
                    for k2 in top_peaks[:7]:
                        # How much would a[j] contribute to peak k2?
                        # contrib = a[j] * a[k2-j]
                        other_idx = k2 - j
                        if 0 <= other_idx < n:
                            penalty += a[other_idx]
                        # Also a[other] * a[j]
                        other_idx2 = k2 - j
                        if 0 <= other_idx2 < n and other_idx2 != j:
                            penalty += a[other_idx2] * 0.5
                    target_scores[j] = -penalty  # We want low penalty = high score
                    target_scores[j] -= 0.1 * a[j] * n

                top_targets = np.argsort(target_scores)[::-1][:15]

                for dst in top_targets:
                    dst = int(dst)
                    if dst == src and dst == src_other:
                        continue
                    if dst == src and src == src_other:
                        continue

                    for frac in [0.002, 0.005, 0.01, 0.02, 0.04]:
                        if src != src_other:
                            amt = min(a[src], a[src_other]) * frac
                        else:
                            amt = a[src] * frac
                        if amt < 1e-15:
                            continue

                        a_trial = a.copy()
                        a_trial[src] -= amt
                        if src != src_other:
                            a_trial[src_other] -= amt * 0.5
                        a_trial[dst] += amt * (1.5 if src != src_other else 1.0)

                        if (a_trial < -1e-12).any():
                            continue
                        a_trial = np.maximum(a_trial, 0.0)
                        s_t = a_trial.sum()
                        if s_t < 1e-15:
                            continue
                        a_trial = a_trial / s_t

                        sc = score_normalized(a_trial)
                        if sc < best_score - 1e-13:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            stall = 0
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            stall += 1
            # Periodic random perturbation
            if stall % 25 == 1:
                n_pert = int(rng.integers(2, 6))
                sources = rng.choice(n, size=n_pert, replace=False)
                targets = rng.choice(n, size=n_pert, replace=False)
                a_trial = a.copy()
                for s_i, d_i in zip(sources, targets):
                    frac = rng.uniform(0.003, 0.015)
                    amt = a_trial[s_i] * frac
                    a_trial[s_i] -= amt
                    a_trial[d_i] += amt
                if (a_trial < -1e-12).any():
                    continue
                a_trial = np.maximum(a_trial, 0.0)
                s_t = a_trial.sum()
                if s_t < 1e-15:
                    continue
                a_trial = a_trial / s_t
                sc = score_normalized(a_trial)
                if sc < best_score:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    stall = 0

        if verbose and stall % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD stall={stall} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def mirror_descent_step(a, lr, b):
    """Single mirror descent step on the simplex using the entropy mirror map."""
    n = len(a)
    max_b = b.max()
    # Use top peaks with soft weighting
    top_k = np.argsort(b)[::-1][:8]
    weights = np.exp((b[top_k] - max_b) * 15.0)
    weights = weights / weights.sum()

    grad = np.zeros(n)
    for w, k in zip(weights, top_k):
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        other = k - idx
        grad[idx] += 2.0 * w * a[other]
        grad[other] += 2.0 * w * a[idx]

    # Mirror descent update
    log_a = np.log(np.maximum(a, 1e-30))
    log_a_new = log_a - lr * grad
    log_a_new -= log_a_new.max()
    a_new = np.exp(log_a_new)
    s = a_new.sum()
    if s < 1e-30:
        return a.copy()
    return a_new / s


def mirror_descent_refine(a0, time_limit, verbose=True):
    """Mirror descent refinement with adaptive learning rate."""
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = score_normalized(a)
    t0 = time.monotonic()

    n_iters = 3000
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        progress = it / max(1, n_iters)
        lr = 0.08 * (0.005 / 0.08) ** progress

        b = fast_autocorr(a)
        a = mirror_descent_step(a, lr, b)

        if it % 100 == 0:
            sc = score_normalized(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
            if verbose and it % 500 == 0:
                print(f"    MD it={it}: best={best_score:.6f}")

    sc = score_normalized(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

    return best_a, best_score


def get_best_solver():
    """Find the best available LP solver by testing on a small problem."""
    for name in [cp.CLARABEL, cp.ECOS, cp.PDLP, cp.SCS]:
        try:
            x = cp.Variable(20, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            t0 = time.monotonic()
            prob.solve(solver=name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                elapsed = time.monotonic() - t0
                print(f"  Solver {name}: OK in {elapsed:.3f}s")
                return name
        except Exception:
            continue
    return cp.ECOS


def interpolate_to_length(f, new_n):
    """Interpolate a normalized sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    f_norm = normalize(f)
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    return normalize(f_interp)


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Main search: Nesterov accelerated LP as primary optimizer,
    with coordinate descent and mirror descent for refinement.
    """
    t_global = time.monotonic()
    time_budget = min(float(budget_s) - 90.0, 1710.0)

    solver = get_best_solver()
    print(f"Using solver: {solver}, time budget: {time_budget:.0f}s")

    n = 1600
    best_a = None
    best_score = float('inf')

    # Initialize from parent if available
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            best_a = normalize(p)
            best_score = score_normalized(best_a)
            n = len(best_a)
            print(f"Parent: n={n}, score={best_score:.6f}")
    except Exception:
        pass

    if best_a is None:
        # Cosine bump initialization
        x = np.linspace(0.0, np.pi, n)
        best_a = np.cos(x / 2.0) ** 2
        best_a = normalize(best_a)
        best_score = score_normalized(best_a)
        print(f"Default cosine init: n={n}, score={best_score:.6f}")

    # === Phase 1: Main Nesterov LP (55% of budget) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t1 = remaining * 0.55
    print(f"\n=== Phase 1: Nesterov LP at n={n} (budget={t1:.0f}s) ===")
    if t1 > 30:
        a_out, sc_out, iters = nesterov_lp(best_a, t1, solver, verbose=True)
        print(f"  NLP done: {iters} iters, score={sc_out:.6f}")
        if sc_out < best_score:
            best_score = sc_out
            best_a = a_out.copy()
            print(f"  *** Updated best to {best_score:.6f}")

    # === Phase 2: Coordinate Descent (20% of budget) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t2 = remaining * 0.22
    print(f"\n=== Phase 2: Coord Descent (budget={t2:.0f}s) ===")
    if t2 > 10:
        a_out, sc_out = coord_descent_refine(best_a, t2)
        print(f"  CD done: score={sc_out:.6f}")
        if sc_out < best_score:
            best_score = sc_out
            best_a = a_out.copy()
            print(f"  *** Updated best to {best_score:.6f}")

    # === Phase 3: Second Nesterov round on improved point (12% of budget) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t3 = remaining * 0.15
    print(f"\n=== Phase 3: Nesterov Round 2 (budget={t3:.0f}s) ===")
    if t3 > 20:
        a_out, sc_out, iters = nesterov_lp(best_a, t3, solver, verbose=False)
        print(f"  NLP2 done: {iters} iters, score={sc_out:.6f}")
        if sc_out < best_score:
            best_score = sc_out
            best_a = a_out.copy()
            print(f"  *** Updated best to {best_score:.6f}")

    # === Phase 4: Mirror Descent (6% of budget) ===
    remaining = time_budget - (time.monotonic() - t_global)
    t4 = remaining * 0.08
    print(f"\n=== Phase 4: Mirror Descent (budget={t4:.0f}s) ===")
    if t4 > 8:
        a_out, sc_out = mirror_descent_refine(best_a, t4)
        print(f"  MD done: score={sc_out:.6f}")
        if sc_out < best_score:
            best_score = sc_out
            best_a = a_out.copy()
            print(f"  *** Updated best to {best_score:.6f}")

    # === Phase 5: Final coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    t5 = remaining * 0.4
    print(f"\n=== Phase 5: Final CD Polish (budget={t5:.0f}s) ===")
    if t5 > 5:
        a_out, sc_out = coord_descent_refine(best_a, t5, verbose=False)
        print(f"  Final CD: score={sc_out:.6f}")
        if sc_out < best_score:
            best_score = sc_out
            best_a = a_out.copy()
            print(f"  *** Updated best to {best_score:.6f}")

    # === Final verification and output ===
    best_a = np.maximum(best_a, 0.0)
    best_a = np.minimum(best_a, 1000.0)
    final_sc = score_normalized(normalize(best_a))
    print(f"\nFINAL: score={final_sc:.6f}, n={len(best_a)}")
    print(f"Total time: {time.monotonic() - t_global:.1f}s")

    return [float(x) for x in best_a]
