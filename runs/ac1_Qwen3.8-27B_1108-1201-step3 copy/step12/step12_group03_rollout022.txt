2. KKT-guided peak equalization (direct analytic gradient method)
3. Softmax-annealed mirror descent for smooth landscape traversal
4. Adaptive coordinate descent with line search for final polish
5. Multi-length exploration
"""

import numpy as np
import cvxpy as cp
import time
from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full autocorrelation (convolution with self) using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate using the provided evaluate_sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        sc = evaluate_sequence(a_list)
        return float(sc)
    except Exception:
        return float('inf')


def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name, max_iter=1000)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a_fixed):
    """Build (2n-1) x n matrix T where T@b gives convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver, verbose=False):
    """
    Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0.
    Returns (b, M) or (None, None) on failure.
    """
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    solvers_to_try = [solver]
    for alt in [cp.ECOS, cp.SCS, cp.PDLP, cp.CLARABEL]:
        if alt not in solvers_to_try:
            solvers_to_try.append(alt)
    
    for s in solvers_to_try:
        try:
            prob.solve(solver=s, max_iter=50000)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                break
        except Exception:
            continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def nesterov_lp_optimize(a0, time_limit, solver, verbose=True):
    """
    Nesterov-accelerated LP best-response with adaptive parameters.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    y = a.copy()

    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()

    # Adaptive parameters
    momentum_base = 0.0
    stall_count = 0
    ema_a = a.copy()
    ema_alpha = 0.03

    for t in range(1, 100000):
        if time.monotonic() - t0 > time_limit:
            break

        b_star, M_val = solve_lp_best_response(y, solver)
        if b_star is None:
            continue

        # Nesterov schedule
        if t <= 10:
            alpha_t = 2.0 / (t + 1.0)
        else:
            alpha_t = 2.0 / (t + 2.0)

        # Nesterov update
        y_new = (1 - alpha_t) * y + alpha_t * b_star
        y_new = np.maximum(y_new, 0)
        y_s = y_new.sum()
        if y_s < 1e-15:
            y_new = y.copy()
            y_s = 1.0
        y_new = y_new / y_s

        if t == 1:
            a_new = b_star.copy()
        else:
            momentum = (t - 1.0) / (t + 2.0)
            a_new = y_new + momentum * (y_new - y)
            a_new = np.maximum(a_new, 0)
            a_s = a_new.sum()
            if a_s < 1e-15:
                a_new = y_new.copy()
                a_s = 1.0
            a_new = a_new / a_s

        y = y_new
        a = a_new

        # Score
        score_a = 2.0 * n * fast_autocorr(a).max()
        if score_a < best_score:
            best_score = score_a
            best_a = a.copy()
            stall_count = 0
        else:
            if score_a > best_score * 1.000001:
                stall_count += 1

        score_y = 2.0 * n * fast_autocorr(y).max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            stall_count = 0

        # EMA
        ema_a = (1 - ema_alpha) * ema_a + ema_alpha * a_new
        if t > 30 and t % 10 == 0:
            score_ema = 2.0 * n * fast_autocorr(ema_a).max()
            if score_ema < best_score:
                best_score = score_ema
                best_a = ema_a.copy()
                stall_count = 0

        # Adapt EMA rate
        if stall_count > 100:
            ema_alpha = min(0.15, ema_alpha * 1.1)
        elif stall_count == 0:
            ema_alpha = max(0.01, ema_alpha * 0.95)

        if t % 300 == 0 and verbose:
            elapsed = time.monotonic() - t0
            print(f"      NLP t={t} ({elapsed:.0f}s): best={best_score:.8f}")

        if stall_count > 2000 and t > 200:
            break

    return best_a, best_score

    a = a / s

    temperature = max(best_score / (2.0 * n) * 0.1, 1e-8)

            if wk < 1e-15:
                continue

            # a_{i+k} is nonzero when 0 <= i+k < n, i.e., -k <= i < n-k

            if k < n:
                # i + k < n => i < n - k
                for i in range(n - k):
                    f[i] += wk * a[i + k]
            # a_{i-k} for i=0..n-1
            # a_{i-k} is nonzero when 0 <= i-k < n, i.e., k <= i < n+k
            # For i in [0, n-1]: a_{i-k} = a[i-k] if i >= k else 0
            if k < n:
                for i in range(k, n):
                    f[i] += wk * a[i - k]
            elif k >= n:
                # k >= n: a_{i+k} = 0 always (since i+k >= n)
                # a_{i-k}: i-k < 0 always (since i < n <= k)
                pass

        # KKT condition: f_i should be constant for all active i
        # Active i: where a_i > 0
        active = a > 1e-10
        if active.sum() == 0:
            break
        
        f_mean = f[active].mean()

        lr = min(lr, 0.5)  # Cap to prevent instability

        log_a_new = log_a - lr * grad
        log_a_new -= log_a_new.max()
        a_new = np.exp(log_a_new)
        a_new_s = a_new.sum()
        if a_new_s < 1e-30:
            break
        a_new = a_new / a_new_s
        a_new = np.maximum(a_new, 0)
        a_new_s = a_new.sum()
        if a_new_s < 1e-30:
            break
        a_new = a_new / a_new_s

        score = 2.0 * n * fast_autocorr(a_new).max()
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            # Decrease temperature as we converge
            temperature *= 0.95
        else:
            a = a_new  # Still update for the algorithm to progress
            # Try smaller step
            temperature = max(temperature * 1.05, 1e-10)

        if it % 200 == 0:
            print(f"      KKT it={it}: best={best_score:.8f}, temp={temperature:.2e}")

    return best_a, best_score
