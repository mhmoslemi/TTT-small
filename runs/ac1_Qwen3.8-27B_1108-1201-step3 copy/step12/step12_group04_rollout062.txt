def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    # Term 1: sum_k w[k] * a[k-j] = (w * a_rev_full)[j + n - 1] where a_rev_full is a reversed and padded

    #   a_rev = a[::-1] (length n)
    #   (w * a_rev)[m] = sum_k w[k] * a_rev[m-k] = sum_k w[k] * a[n-1-(m-k)] = sum_k w[k] * a[n-1-m+k]

    #   This is getting messy.
    
    # Simplest correct approach for moderate n:
    # Compute grad directly as a sum of shifted w weighted by a.
    # For n=1600 and 2n-1=3199, this is O(n * (2n-1)) = O(n^2) which is ~5M operations.
    # That's fine for a gradient computation.
    
    # Actually, even better: use the FFT convolution trick.
    # 
    # grad[j] = sum_k w[k] * a[k-j] + sum_k w[k] * a[j-k]

    # For the first term: let's define the convolution (w * a_padded) where a_padded has a in the first n positions.

    #

    #
    # Wait, let me verify:
    # (a * w)[j] = sum_i a[i] * w[j-i]
    # 
    # First term: sum_k w[k] * a[k-j]. Let m = k-j, so k = m+j:
    #   = sum_m w[m+j] * a[m]
    #
    # Second term: sum_k w[k] * a[j-k]. Let m = j-k, so k = j-m:
    #   = sum_m w[j-m] * a[m]
    #
    # (a * w)[j] = sum_m a[m] * w[j-m]
    # 
    # So the second term = (a*w)[j].

    # First term: sum_m a[m] * w[m+j]

    #

    # grad[j] = h[j] + s[j] = (a_rev * w)[n-1+j] + (a*w)[j]
    # 
    # Hmm, these are different indices into different convolutions. Let me recheck.
    #
    # h[j] = (a_rev * w)[n-1+j]

    #

    #
    # Let me verify with a small example.
    # a = [a0, a1, a2], n=3
    # c = a*a = [a0^2, 2a0a1, 2a0a2+a1^2, 2a1a2, a2^2]

    #
    # grad[0] = w0*a0 + w1*a0 + w2*a0 + w3*a0 + w4*a0 ... no wait.

    # grad[j] = sum_k w[k] * (a[k-j] + a[j-k])
    # grad[0] = sum_k w[k] * (a[k] + a[-k]) = sum_k w[k] * a[k] (since a[-k]=0 for k>0, a[0]=a[0] for k=0)

    #

    # [6]: a2*w4

    # T2[1] = (a*w)[1] = a0*w1 + a1*w0

    #
    # grad[0] = T1[0] + T2[0] = a2*w2 + a1*w1 + a0*w0 + a0*w0 = 2*a0*w0 + a1*w1 + a2*w2 ✓

    #
    # Let me verify directly:
    # grad[1] = sum_k w[k] * (a[k-1] + a[1-k])
    # k=0: w0*(a[-1] + a[1]) = w0*a1
    # k=1: w1*(a[0] + a[0]) = 2*w1*a0
    # k=2: w2*(a[1] + a[-1]) = w2*a1
    # k=3: w3*(a[2] + a[-2]) = w3*a2
    # k=4: w4*(a[3] + a[-3]) = 0
    # Total: a1*w0 + 2*a0*w1 + a1*w2 + a2*w3 ✓

    # Now compute using FFT:
    a_rev = a[::-1]
    conv_len = 3 * n - 2  # length of a_rev * w
    fft_size = 1 << (conv_len - 1).bit_length()
    
    A = rfft(a, fft_size)
    AR = rfft(a_rev, fft_size)
    W = rfft(w, fft_size)
    
    conv_aw = irfft(A * W, fft_size)[:n]  # (a*w)[0:n]
    conv_arw = irfft(AR * W, fft_size)[n-1:2*n-1]  # (a_rev*w)[n-1:2n-1]
    
    grad = conv_arw + conv_aw
    
    return val, grad, w, c


def lse_optimize(a0, time_limit, T_start=200.0, T_end=1.0, n_stages=30):
    """LogSumExp-smoothed optimization with projected gradient + Nesterov momentum."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Nesterov momentum parameters
        y = a.copy()
        a_mom = a.copy()
        
        # Adaptive step size: start with a reasonable guess
        # Lipschitz constant of the gradient is roughly 2*T*max(a)*2n (very rough)
        # Use backtracking instead
        
        alpha = 1.0 / (2.0 * stage_T * n)  # initial step size guess
        
        max_iters = 200
        
        for it in range(max_iters):
            if time.monotonic() - t0 > time_limit:
                break
            
            # Nesterov step
            if it == 0:
                a_curr = y
            else:
                momentum = (np.sqrt(1 + 4 * it * it) - 1) / 2
                a_curr = y + momentum * (y - a_mom)

```python
def fast_autocorr(a):
    """Full self-convolution a*a using FFT, O(n log n)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def lse_grad(a, T):
    """
    Compute value and gradient of (1/T)*log(sum_k exp(T*(a*a)_k)).
    Returns (value, gradient). Uses FFT for O(n log n).
    """
    n = len(a)
    c = fast_autocorr(a)
    Tc = T * c
    max_Tc = Tc.max()
    exp_Tc = np.exp(Tc - max_Tc)
    W = exp_Tc / exp_Tc.sum()
    val = (max_Tc + np.log(exp_Tc.sum())) / T

    a_rev = a[::-1]
    conv_len = 3 * n - 2
    fft_size = 1 << (conv_len - 1).bit_length()
    A = rfft(a, fft_size)
    AR = rfft(a_rev, fft_size)
    WF = rfft(W, fft_size)
    conv_aw = irfft(A * WF, fft_size)[:n]
    conv_arw = irfft(AR * WF, fft_size)[n - 1:2 * n - 1]
    grad = conv_arw + conv_aw
    return val, grad, W


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def dense_subproblem(a, max_iters=200, tol=1e-13):
    """
    Exact solver for min_{p in simplex} max_k (a*p)_k.
    Since n is moderate (~1600), builds the (2n-1) x n matrix explicitly
    and uses Frank-Wolfe / away-steps for the convex max-of-linear problem.
    """
    n = len(a)
    size = 2 * n - 1
    # Build T: T[k, j] = a[j] if 0 <= k-j < n else 0
    T = np.zeros((size, n))
    for j in range(n):
        T[j:j + n, j] = a
    # Normalize T rows so it's well-conditioned
    # Each row k has min(n, k+1, 2n-1-k) non-zero entries
    row_counts = np.minimum(np.minimum(np.arange(size) + 1, n), size - np.arange(size))
    row_counts = np.maximum(row_counts, 1)
    Tn = T / row_counts[:, None]

    p = np.ones(n) / n
    best_obj = np.inf
    L = np.linalg.norm(Tn, ord=np.inf) * np.linalg.norm(Tn, ord=1)
    step = 2.0 / (t + 2.0) if False else 0.0  # will use adaptive

    for it in range(max_iters):
        obj_vals = Tn @ p
        M = obj_vals.max()
        if M < best_obj - tol:
            best_obj = M
        # Find vertex that minimizes Tn @ e_j = min column of Tn
        col_mins = Tn.min(axis=0)
        j_star = int(np.argmin(col_mins))
        # Away-step Frank-Wolfe
        # Check if moving away from an existing vertex helps
        active = np.nonzero(p > 1e-10)[0]
        best_impr = 0
        best_move = None
        # Try moving toward j_star
        gamma = 2.0 / (it + 3)
        p_trial = (1 - gamma) * p
        p_trial[j_star] += gamma
        if p_trial[j_star] > 1:
            gamma = 1 - p[j_star]
            p_trial = (1 - gamma) * p
            p_trial[j_star] += gamma
        obj_trial = (Tn @ p_trial).max()
        if obj_trial < M - 1e-15:
            best_impr = M - obj_trial
            best_move = p_trial.copy()
        # Try away steps
        for idx in active[:min(len(active), 50)]:
            if p[idx] < 1e-10:
                continue
            # Move away from idx toward j_star
            p_away = p.copy()
            p_away[idx] = 0
            p_away[j_star] += p[idx]
            p_away = np.maximum(p_away, 0)
            s = p_away.sum()
            if s < 1e-15:
                continue
            p_away /= s
            # Blend with current
            for frac in [0.5, 0.7, 0.9]:
                p_mix = (1 - frac) * p + frac * p_away
                p_mix = np.maximum(p_mix, 0)
                sm = p_mix.sum()
                if sm < 1e-15:
                    continue
                p_mix /= sm
                obj_mix = (Tn @ p_mix).max()
                if obj_mix < M - 1e-15 and obj_mix < (best_move is not None and (Tn @ best_move).max() or M):
                    best_move = p_mix.copy()
        if best_move is not None:
            p = best_move
        else:
            # No improvement found
            if best_impr < 1e-12 and it > 10:
                break
            # Shrink step
            gamma = 2.0 / (it + 3)
            p = (1 - gamma) * p
            p[j_star] += gamma
            p = np.maximum(p, 0)
            p /= p.sum()
    p = np.maximum(p, 0)
    s = p.sum()
    if s > 0:
        p /= s
    return p


def levelset_cap(a, lambda_val, iters=30):
    """
    Level-set method: iteratively cap a at lambda so that max(a*a) <= lambda.
    Each cap operation reduces the peak while preserving non-negativity.
    Returns improved sequence.
    """
    n = len(a)
    a = np.maximum(np.asarray(a, dtype=np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a
    a /= s
    best_a = a.copy()
    best_max = fast_autocorr(a).max()

    for it in range(iters):
        c = fast_autocorr(a)
        M = c.max()
        if M <= lambda_val * best_max * 1.001:
            break
        # Find positions where a is large and contributes to peaks
        # Cap: set a[i] = min(a[i], cap_level) where cap_level is chosen
        # to reduce the max
        # Simple approach: find top contributors to the max peak
        k_max = int(np.argmax(c))
        js = max(0, k_max - n + 1)
        je = min(n - 1, k_max)
        idx = np.arange(js, je + 1)
        contrib = a[idx] * a[k_max - idx]
        total = contrib.sum()
        if total < 1e-15:
            break
        # Fractional scaling: scale down the contributors proportionally
        # to reduce the peak by a factor
        target_M = M * 0.85
        if target_M > best_max:
            target_M = best_max * 0.99
        scale = target_M / M
        if scale >= 1:
            break
        # Scale only the top contributors
        top_frac = 0.5
        n_top = max(1, int(n_top := len(idx) * top_frac))
        top_idx = idx[np.argsort(contrib)[::-1][:n_top]]
        a_new = a.copy()
        for i in top_idx:
            a_new[i] *= scale
        s2 = a_new.sum()
        if s2 < 1e-15:
            break
        a_new /= s2
        # Check if this helps
        c_new = fast_autocorr(a_new)
        M_new = c_new.max()
        if M_new < best_max:
            best_max = M_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try more aggressive capping
            for frac in [0.3, 0.7, 0.9, 1.0]:
                a_try = a.copy()
                ti2 = idx[np.argsort(contrib)[::-1][:max(1, int(len(idx) * frac))]]
                for i in ti2:
                    a_try[i] *= 0.7
                a_try = np.maximum(a_try, 0)
                s2 = a_try.sum()
                if s2 < 1e-15:
                    continue
                a_try /= s2
                M_try = fast_autocorr(a_try).max()
                if M_try < best_max:
                    best_max = M_try
                    best_a = a_try.copy()
                    a = a_try
                    break
    return best_a


def lse_multiscale(a0, time_limit):
    """
    Multiscale FFT-based LogSumExp optimization.
    Start at low resolution, optimize, then upsample and refine.
    """
    n0 = len(a0)
    a_base = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_base.sum()
    if s < 1e-15:
        a_base = np.ones(n0) / n0
    else:
        a_base = a_base / s

    t0 = time.monotonic()
    best_a = a_base.copy()
    best_score = 2.0 * n0 * fast_autocorr(a_base).max()

    # Start at half resolution, then full
    for n in [max(64, n0 // 4), max(128, n0 // 2), n0]:
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        if n == n0:
            a = best_a.copy() if len(best_a) == n0 else np.interp(
                np.linspace(0, 1, n), np.linspace(0, 1, len(best_a)), best_a)
        else:
            src = best_a if len(best_a) >= n else a_base
            a = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(src)), src)
            a = np.maximum(a, 0)
            s = a.sum()
            a /= s if s > 1e-15 else 1

        # LSE optimization at this resolution
        T_start, T_end = 50.0, 5.0
        n_stages = 15
        temps = np.geomspace(T_start, T_end, n_stages)
        y = a.copy()
        a_prev = a.copy()

        for si, T in enumerate(temps):
            if time.monotonic() - t0 > time_limit:
                break
            for it in range(50):
                val, grad, _ = lse_grad(y, T)
                alpha = 1.0 / (T * n)
                y_new = y - alpha * grad
                y_new = project_simplex(y_new)
                # Check improvement
                c = fast_autocorr(y_new)
                score = 2.0 * n * c.max()
                if score < 2.0 * n * fast_autocorr(y).max():
                    a_prev = y
                    y = y_new
                else:
                    alpha *= 0.5
                    y_new = y - alpha * grad
                    y_new = project_simplex(y_new)
                    a_prev = y
                    y = y_new

                if time.monotonic() - t0 > time_limit:
                    break

            a = project_simplex(y)
            score = 2.0 * n * fast_autocorr(a).max()
            if n == n0 and score < best_score:
                best_score = score
                best_a = a.copy()

        # Store result for next scale
        if n < n0:
            a_interp = np.interp(np.linspace(0, 1, n0), np.linspace(0, 1, n), a)
            a_interp = np.maximum(a_interp, 0)
            s = a_interp.sum()
            if s > 1e-15:
                a_interp /= s
            sc = 2.0 * n0 * fast_autocorr(a_interp).max()
            if sc < best_score:
                best_score = sc
                best_a = a_interp.copy()

        elapsed = time.monotonic() - t0
        print(f"    LSE-MS n={n} ({elapsed:.0f}s): best={best_score:.8f}")

    return best_a, best_score


def subgradient_nesterov(a0, time_limit, max_iters=50000):
    """
    Nesterov-accelerated projected subgradient descent on f(a) = max_k (a*a)_k.
    Step size: alpha_t = c / sqrt(t) with adaptive c.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()

    y = a.copy()
    a_prev = a.copy()
    c_param = 0.01  # step size parameter
    n_impacts = 0

    for t in range(1, max_iters + 1):
        if time.monotonic() - t0 > time_limit:
            break

        if t == 1:
            a_curr = y
        else:
            momentum = (np.sqrt(1 + 4 * (t - 1) ** 2) - 1) / 2
            a_curr = y + momentum * (y - a_prev)
            a_curr = project_simplex(a_curr)

        c = fast_autocorr(a_curr)
        c_max = c.max()
        # Active set: all k where c[k] is close to max
        threshold = c_max * 0.999
        active = np.nonzero(c >= threshold)[0]

        # Averaged subgradient
        grad = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                val = a_curr[k - j]
                if k - j != j:
                    val += a_curr[k - 2 * j + j]  # a_curr[j-k] if valid
                # Actually: dc_k/da_j = a[k-j] + a[j-k]
                # a[k-j] is valid when 0<=k-j<n
                # a[j-k] is valid when 0<=j-k<n
                # Let me recompute properly
            pass

        # Recompute gradient properly using FFT
        n_active = len(active)
        if n_active > 0:
            w_active = np.zeros(2 * n - 1)
            w_active[active] = 1.0 / n_active
            # Use the same FFT trick as in lse_grad but with uniform weights on active set
            a_rev = a_curr[::-1]
            conv_len = 3 * n - 2
            fft_size = 1 << (conv_len - 1).bit_length()
            A = rfft(a_curr, fft_size)
            AR = rfft(a_rev, fft_size)
            WF = rfft(w_active, fft_size)
            conv_aw = irfft(A * WF, fft_size)[:n]
            conv_arw = irfft(AR * WF, fft_size)[n - 1:2 * n - 1]
            grad = conv_arw + conv_aw

        alpha = c_param / np.sqrt(t)
        y_new = a_curr - alpha * grad
        y_new = project_simplex(y_new)

        a_prev = y
        y = y_new

        # Track best
        c_y = fast_autocorr(y)
        score_y = 2.0 * n * c_y.max()
        if score_y < best_score:
            best_score = score_y
            best_a = y.copy()
            n_impacts += 1

        # Periodic best update from averaged iterate
        if t % 1000 == 0:
            c_check = fast_autocorr(y)
            score_check = 2.0 * n * c_check.max()
            if score_check < best_score:
                best_score = score_check
                best_a = y.copy()

        if t % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PG t={t} ({elapsed:.0f}s): best={best_score:.8f}, c_max={c_y.max():.10f}")

    return best_a, best_score


def coordinate_descent(a0, time_limit, seed=42):
    """
    Advanced coordinate descent: for each coordinate, find the optimal
    transfer to a partner coordinate that most reduces the max convolution.
    Uses FFT for fast evaluation.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    t0 = time.monotonic()
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        c = fast_autocorr(a)
        c_max = c.max()

        # Find top peaks
        top_k = np.argsort(c)[::-1][:10]

        improved = False
        # For each top peak, try to reduce it by moving mass
        for k in top_k:
            if time.monotonic() - t0 > time_limit:
                break
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = a[idx] * a[k - idx]

            # Find the largest contributor
            top_contrib = np.argsort(contrib)[::-1][:5]
            for tc in top_contrib:
                src = idx[tc]
                src_val = a[src]
                if src_val < 1e-12:
                    continue

                # Find good destinations: positions that minimize contribution to all top peaks
                # Score each position by total contribution to top peaks
                scores = np.zeros(n)
                for k2 in top_k[:5]:
                    js2 = max(0, k2 - n + 1)
                    je2 = min(n - 1, k2)
                    for j2 in range(js2, je2 + 1):
                        partner = k2 - j2
                        if 0 <= partner < n:
                            scores[j2] += a[j2] * a[partner]

                # Best destinations are those with lowest scores (not in peak areas)
                scores[src] = np.inf
                best_dsts = np.argsort(scores)[:20]

                for dst in best_dsts[:5]:
                    # Binary search for optimal transfer amount
                    lo, hi = 0.0, src_val * 0.5
                    best_amt = 0
                    best_sc = best_score
                    for _ in range(8):
                        mid = (lo + hi) / 2
                        a_trial = a.copy()
                        a_trial[src] -= mid
                        a_trial[dst] += mid
                        a_trial = np.maximum(a_trial, 0)
                        a_trial /= a_trial.sum()
                        sc = 2.0 * n * fast_autocorr(a_trial).max()
                        if sc < best_sc:
                            best_sc = sc
                            best_amt = mid
                            lo = mid
                        else:
                            hi = mid

                    if best_amt > 1e-12 and best_sc < best_score - 1e-14:
                        a[src] -= best_amt
                        a[dst] += best_amt
                        a = np.maximum(a, 0)
                        a /= a.sum()
                        best_score = best_sc
                        best_a = a.copy()
                        improved = True
                        break

            if improved:
                break

        if not improved:
            stall += 1
            # Random perturbation to escape
            if stall % 50 == 0:
                n_random = 10
                for _ in range(n_random):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i != j and a[i] > 1e-10:
                        frac = rng.uniform(0.001, 0.03)
                        amt = a[i] * frac
                        a_trial = a.copy()
                        a_trial[i] -= amt
                        a_trial[j] += amt
                        a_trial = np.maximum(a_trial, 0)
                        a_trial /= a_trial.sum()
                        sc = 2.0 * n * fast_autocorr(a_trial).max()
                        if sc < best_score:
                            a = a_trial
                            best_score = sc
                            best_a = a.copy()
                            stall = 0
                            break
            if stall > 500:
                break

    return best_a, best_score

def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

            print(f"Parent: n={n_base}, score={best_score:.8f}")
    except Exception:
        print("No valid parent, using structured initialization")

    if best_seq is None:
        best_seq = structured_init(n_base, rng)
        best_score = evaluate_safe(best_seq)
        print(f"Init: n={n_base}, score={best_score:.8f}")

    def try_update(a, label):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(np.asarray(a, dtype=np.float64), 0)
        s = a.sum()
        if s < 1e-10:
            return
        a = a / s
        sc = evaluate_safe(a)
        if np.isfinite(sc) and sc < best_score:
            prev = best_score
            best_score = sc
            best_seq = a.copy()
            el = time.monotonic() - t_global
            print(f"  *** NEW BEST [{label}]: {sc:.8f} (was {prev:.8f}, n={len(a)}, t={el:.0f}s)")

    # === Phase 1: Dense subproblem (exact inner solver) ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 1: Dense subproblem (remaining={remaining:.0f}s) ===")
    t1 = min(remaining * 0.20, 200.0)
    if t1 > 10:
        # Iterate: solve inner problem, update outer
        a_curr = best_seq.copy()
        for outer_it in range(20):
            if time.monotonic() - t_global > time_budget - remaining * 0.8:
                break
            p_star = dense_subproblem(a_curr, max_iters=100)
            # Blend
            for t_mix in [0.3, 0.5, 0.7, 1.0]:
                a_trial = (1 - t_mix) * a_curr + t_mix * p_star
                a_trial = np.maximum(a_trial, 0)
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial /= s
                sc = 2.0 * n_base * fast_autocorr(a_trial).max()
                if sc < 2.0 * n_base * fast_autocorr(a_curr).max():
                    a_curr = a_trial
                    break
            try_update(a_curr, f"dense_it{outer_it}")
            if time.monotonic() - t_global > time_budget * 0.3:
                break

    # === Phase 2: Level-set cap ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 2: Level-set cap (remaining={remaining:.0f}s) ===")
    t2 = min(remaining * 0.10, 80.0)
    if t2 > 10:
        a_out = levelset_cap(best_seq, 1.0, iters=50)
        try_update(a_out, "levelset")

    # === Phase 3: Multiscale LSE ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 3: Multiscale LSE (remaining={remaining:.0f}s) ===")
    t3 = min(remaining * 0.20, 200.0)
    if t3 > 15:
        a_out, _ = lse_multiscale(best_seq, t3)
        try_update(a_out, "lse_ms")

    # === Phase 4: Nesterov subgradient ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 4: Nesterov subgradient (remaining={remaining:.0f}s) ===")
    t4 = min(remaining * 0.20, 250.0)
    if t4 > 20:
        a_out, _ = subgradient_nesterov(best_seq, t4, max_iters=20000)
        try_update(a_out, "nesterov_pg")

    # === Phase 5: Multi-length exploration ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 5: Multi-length exploration (remaining={remaining:.0f}s) ===")
    n_best = len(best_seq)
    candidate_lengths = sorted(set(
        [n_best - 100, n_best + 100, n_best + 200, n_best + 400, n_best + 800]
    ), key=lambda x: abs(x - n_best))
    candidate_lengths = [nl for nl in candidate_lengths if 800 <= nl <= 4000]

    time_per = min(remaining * 0.4 / max(1, len(candidate_lengths)), 100.0)
    for nl in candidate_lengths[:4]:
        remaining = time_budget - (time.monotonic() - t_global)
        if remaining < 30:
            break
        tl = min(time_per, remaining * 0.3)
        if tl < 15:
            continue
        print(f"  Trying n={nl} (time={tl:.0f}s)")
        init = interpolate_to_length(best_seq, nl)
        a_out, _ = lse_multiscale(init, tl * 0.6)
        try_update(a_out, f"lse_n{nl}")
        rem2 = time_budget - (time.monotonic() - t_global)
        if rem2 > 20:
            a_out2, _ = subgradient_nesterov(a_out, min(tl * 0.4, rem2 * 0.5), max_iters=5000)
            try_update(a_out2, f"pg_n{nl}")

    # === Phase 6: Final coordinate descent polish ===
    remaining = time_budget - (time.monotonic() - t_global)
    print(f"\n=== Phase 6: Coordinate descent polish (remaining={remaining:.0f}s) ===")
    if remaining > 15:
        t6 = min(remaining * 0.5, 120.0)
        a_out, _ = coordinate_descent(best_seq, t6, seed=123)
        try_update(a_out, "coord_desc")

    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 15:
        t6b = min(remaining * 0.4, 80.0)
        a_out, _ = coordinate_descent(best_seq, t6b, seed=456)
        try_update(a_out, "coord_desc2")

    # Final level-set attempt
    remaining = time_budget - (time.monotonic() - t_global)
    if remaining > 10:
        a_out = levelset_cap(best_seq, 1.0, iters=30)
        try_update(a_out, "final_levelset")

    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]

    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    return result
```
