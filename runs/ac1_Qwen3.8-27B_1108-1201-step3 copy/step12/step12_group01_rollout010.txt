    n = len(a0)
    a = normalize(a0)
    
    avg = a.copy()
    total_weight = 1.0
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    
    t0 = time.monotonic()
    eval_count = 0
    stall = 0
    
    for t in range(1, 10**8):
        if time.monotonic() - t0 > time_limit - 3:
            break
        
        # Build convolution matrix and solve LP
        T_mat = np.zeros((2*n-1, n))
        for j in range(n):
            T_mat[j:j+n, j] = avg
        b_star, M_val = solve_lp_best_response(avg, solver, T_mat)
        if b_star is None:
            continue
        
        # Update step-size weighted average
        alpha = 2.0 / (t + 2.0)
        avg = avg + alpha * (b_star - avg)
        avg = normalize(avg)
        
        # Evaluate (not every iteration to save time)
        if t % 20 == 0:
            score = 2.0 * n * fast_autocorr(avg).max()
            if score < best_score:
                best_score = score
                best_a = avg.copy()
                stall = 0
            else:
                stall += 1
            eval_count += 1
        
        # Restart if stalled
        if stall > 100 and t > 200:
            avg = best_a.copy()
            stall = 0
            if t % 1000 == 0:
                print(f"    ZSG restart t={t}, best={best_score:.6f}")
        
        if t % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    ZSG t={t} ({elapsed:.0f}s, {t/elapsed:.0f} iter/s): best={best_score:.6f}")
    
    return best_a, best_score


def lse_grad_descent(a0, time_limit, T_start=50.0, T_end=0.5, n_stages=15, topk=20):
    """
    Top-k LogSumExp gradient descent with Adam.
    """
    n = len(a0)
    a = normalize(a0)
    z = np.log(np.maximum(a, 1e-30))
    
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    
    t0 = time.monotonic()
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit - 5:
            break
        
        lr = 0.01 * min(1.0, 50.0 / n)
        m = np.zeros(n)
        v = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.8, 0.999, 1e-8
        
        max_it = max(20, int((time_limit - (time.monotonic() - t0)) * 5 / (n_stages - si)))
        
        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 3:
                break
            
            a = softmax_fn(z)
            c = fast_autocorr(a)
            k_use = min(topk, len(c))
            top_idx = np.argsort(c)[::-1][:k_use]
            top_vals = c[top_idx]
            
            Tc = stage_T * top_vals
            max_Tc = Tc.max()
            exp_Tc = np.exp(Tc - max_Tc)
            w = exp_Tc / exp_Tc.sum()
            
            # Gradient computation
            grad_a = np.zeros(n)
            for idx in range(k_use):
                K = top_idx[idx]
                wK = w[idx]
                js = max(0, K - (n - 1))
                je = min(n - 1, K)
                if js <= je:
                    idx_range = np.arange(js, je + 1)
                    grad_a[idx_range] += 2.0 * wK * a[K - idx_range]
            
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)
            
            t_adam += 1
            m = beta1 * m + (1 - beta1) * grad_z
            v = beta2 * v + (1 - beta2) * grad_z ** 2
            m_hat = m / (1 - beta1 ** t_adam)
            v_hat = v / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            if (it + 1) % 5 == 0:
                a_check = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a_check).max()
                if score < best_score:
                    best_score = score
                    best_a = a_check.copy()
    
    return best_a, best_score


def softmax_fn(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def peak_shave(a0, time_limit, seed=42):
    """Multi-scale peak shaving."""
    rng = np.random.default_rng(seed)
    n = len(a0)
    a = normalize(a0)
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit - 3:
        b = fast_autocorr(a)
        bmax = b.max()
        threshold = bmax * 0.998
        top_peaks = np.where(b >= threshold)[0]
        if len(top_peaks) == 0:
            top_peaks = np.argsort(b)[::-1][:5]
        
        improved = False
        for k in top_peaks[:10]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            src = idx[np.argmax(contrib)]
            
            # Find safe destination
            danger = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:10]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                danger[js2:je2+1] = True
            safe = np.where(~danger)[0]
            safe = safe[safe != src]
            if len(safe) == 0:
                safe = np.where(np.arange(n) != src)[0]
            safe = safe[np.argsort(a[safe])]
            dst_pool = safe[:max(5, len(safe)//10)]
            
            for dst in dst_pool[:5]:
                for frac in [0.002, 0.005, 0.01, 0.02, 0.05]:
                    amt = a[src] * frac
                    a_trial = a.copy()
                    a_trial[src] -= amt
                    a_trial[dst] += amt
                    if a_trial[src] < 0:
                        continue
                    a_trial = normalize(a_trial)
                    score = 2.0 * n * fast_autocorr(a_trial).max()
                    if score < best_score:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            if stall > 100:
                break
            # Small random perturbation
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j and a[i] > 1e-10:
                amt = a[i] * rng.uniform(0.001, 0.02)
                a_trial = a.copy()
                a_trial[i] -= amt
                a_trial[j] += amt
                a_trial = normalize(a_trial)
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
    
    return best_a, best_score
