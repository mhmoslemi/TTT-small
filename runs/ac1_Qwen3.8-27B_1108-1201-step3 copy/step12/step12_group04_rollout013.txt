from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_autocorr(a):
    """Full self-convolution a*a using FFT, O(n log n)."""
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):
    """Safely evaluate a sequence."""
    try:
        a_list = [float(max(0.0, min(1000.0, x))) for x in a]
        if sum(a_list) < 0.01:
            return float('inf')
        return float(evaluate_sequence(a_list))
    except Exception:
        return float('inf')


def softmax_fn(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return e / s


def lse_obj_grad_z(z, T):
    """Compute LSE objective and gradient w.r.t. z (a = softmax(z))."""
    a = softmax_fn(z)
    n = len(a)
    b = fast_autocorr(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    obj_val = max_Tb + np.log(exp_Tb.sum())
    w = exp_Tb / exp_Tb.sum()
    
    # Gradient w.r.t. a: grad_a[i] = 2 * sum_k w[k] * a[k-i]
    a_rev = a[::-1]
    conv_len = n + len(w) - 1
    fft_size = 1 << (conv_len - 1).bit_length()
    A = rfft(a_rev, fft_size)
    W = rfft(w, fft_size)
    conv = irfft(A * W, fft_size)
    grad_a = 2.0 * conv[n-1 : n-1+n]
    
    # Transform to z-space
    sat = np.dot(grad_a, a)
    grad_z = a * (grad_a - sat)
    return obj_val, grad_z


def score_seq(a):
    """Compute the score 2*n*max_b/(sum_a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    b = fast_autocorr(a)
    return 2.0 * n * b.max() / (s * s)


def lse_optimize(a_init, time_limit, T_start=300.0, T_end=1.0, n_stages=25, reheat_every=5):
    """
    LSE-based optimization with L-BFGS and adaptive annealing/reheating.
    """
    n = len(a_init)
    a = np.maximum(np.asarray(a_init, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    z = np.log(a + 1e-30)
    best_a = a.copy()
    best_score = score_seq(a)
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    
    for si in range(n_stages):
        if time.monotonic() - t0 > time_limit:
            break
        
        T = temps[si]
        
        # L-BFGS at this temperature
        max_iter = max(50, int(300 * (1.0 - si / n_stages)))
        
        def func(z_in):
            val, _ = lse_obj_grad_z(z_in, T)
            return val
        
        def jac(z_in):
            _, g = lse_obj_grad_z(z_in, T)
            return g
        
        try:
            res = minimize(func, z, jac=jac, method='L-BFGS-B',
                          options={'maxiter': max_iter, 'gtol': 1e-12, 'ftol': 1e-16,
                                   'maxfun': max_iter * 200, 'disp': False})
            z = res.x
        except Exception:
            pass
        
        a = softmax_fn(z)
        score = score_seq(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Print progress
        if si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE stage={si}/{n_stages}, T={T:.2f} ({elapsed:.0f}s): score={best_score:.8f}")
    
    return best_a, best_score
