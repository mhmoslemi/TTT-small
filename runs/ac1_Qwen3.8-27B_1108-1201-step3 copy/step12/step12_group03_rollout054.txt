import numpy as np

from scipy.fft import rfft, irfft


def fast_autocorr(a):
    """Full convolution a*a using FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(np.asarray(a, dtype=np.float64), fft_size)
    return irfft(A * A, fft_size)[:size]


def evaluate_safe(a):

def get_lp_solver():
    """Find a working LP solver."""
    for solver_name in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
        try:
            x = cp.Variable(10, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1, x <= 0.5])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return solver_name
        except Exception:
            continue
    return cp.ECOS


def build_conv_matrix(a_fixed):
    """Build the (2n-1) x n convolution matrix T where (T@b) = convolve(a_fixed, b)."""
    n = len(a_fixed)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    return T


def solve_lp_best_response(a_fixed, solver, extra_kwargs=None):
    """Solve LP: min M s.t. T(a)@b <= M, sum(b)=1, b>=0. Returns (b, M)."""
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [T @ b_var <= M_var, cp.sum(b_var) == 1]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    
    kwargs = {}
    if extra_kwargs:
        kwargs = extra_kwargs
    
    try:
        prob.solve(solver=solver, **kwargs)
    except Exception:
        pass
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        for alt in [cp.PDLP, cp.CLARABEL, cp.SCS, cp.ECOS]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                if prob.status in ('optimal', 'optimal_inaccurate'):
                    break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)

def qp_active_set_refinement(a0, time_limit, n_active_start=15):

    QP refinement: identify active constraints and solve QP iteratively.
    """

    a = a / s

        if time.monotonic() - t0 > time_limit:
            break
        
        threshold = max_b * (1 - 0.005)
        active_lags = np.where(b >= threshold)[0]
        if len(active_lags) > n_active:
            active_lags = np.argsort(b[active_lags])[::-1][:n_active]
        active_lags = np.sort(active_lags)
        
        a_var = cp.Variable(n, nonneg=True)
        M_var = cp.Variable()
        
        constraints = [cp.sum(a_var) == 1, M_var >= 0]
        for k in active_lags:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            other_idx = k - idx
            q_expr = cp.sum(cp.multiply(a_var[idx], a_var[other_idx]))
            constraints.append(q_expr <= M_var)
        
        prob = cp.Problem(cp.Minimize(M_var), constraints)
        solved = False
        for solver in [cp.CLARABEL, cp.ECOS, cp.SCS, cp.PDLP]:

                    break

                continue

            break

        a_new = np.maximum(a_new, 0)

        if a_new_s < 1e-15:
            break
        a_new = a_new / a_new_s

def gradient_peak_shaving(a0, time_limit, seed=123):
    """
    Gradient-based peak shaving using subgradient of max_k (a*a)(k).
    Uses projected mirror descent on the simplex.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    for it in range(200000):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_autocorr(a)
        max_b = b.max()
        
        # Compute subgradient: weighted average of top peaks' gradients
        n_top = min(10, 2 * n - 1)
        top_peaks = np.argsort(b)[::-1][:n_top]
        weights = np.exp((b[top_peaks] - max_b) * 50.0)
        w_sum = weights.sum()
        if w_sum < 1e-30:
            weights = np.ones(n_top) / n_top
        else:
            weights = weights / w_sum
        
        grad = np.zeros(n)
        for w, k in zip(weights, top_peaks):
            # grad[j] += 2*w*a(k-j) if 0 <= k-j <= n-1
            k_min = max(0, k - (n - 1))
            k_max = min(n - 1, k)
            js = np.arange(k_min, k_max + 1)
            others = k - js
            grad[js] += 2 * w * a[others]
        
        # Mirror descent step with adaptive learning rate
        base_lr = 0.01 / (1 + it * 0.001)
        lr = base_lr
        
        # Try a few learning rates
        best_lr_score = best_score
        for lr_try in [lr * 0.1, lr * 0.5, lr, lr * 2, lr * 5]:
            log_a = np.log(np.maximum(a, 1e-30))
            log_a_new = log_a - lr_try * grad
            log_a_new -= log_a_new.max()
            a_new = np.exp(log_a_new)
            a_new_s = a_new.sum()
            if a_new_s < 1e-30:
                continue
            a_new = a_new / a_new_s
            score = 2.0 * n * fast_autocorr(a_new).max()
            if score < best_lr_score:
                best_lr_score = score
                best_candidate = a_new.copy()
        
        if best_lr_score < best_score - 1e-12:
            a = best_candidate
            best_score = best_lr_score
            best_a = a.copy()
            stall = 0
        else:
            # Try adding small noise to escape
            if stall % 50 == 0 and stall > 100:
                noise = rng.normal(0, 0.01, n)
                a_noisy = a + noise * a
                a_noisy = np.maximum(a_noisy, 0)
                a_noisy_s = a_noisy.sum()
                if a_noisy_s > 1e-15:
                    a_noisy = a_noisy / a_noisy_s
                    score_noisy = 2.0 * n * fast_autocorr(a_noisy).max()
                    if score_noisy < best_score:
                        a = a_noisy
                        best_score = score_noisy
                        best_a = a.copy()
                        stall = 0
            stall += 1
        
        if stall > 2000:
            break
        
        if it % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"      GPS it={it} ({elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score


def coordinate_descent_refine(a0, time_limit, seed=42):
    """
    Coordinate descent with line search on individual pairs of coordinates.
    """
    n = len(a0)
    rng = np.random.default_rng(seed)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    
    # Precompute which lags are "hot"
    b = fast_autocorr(a)
    max_b = b.max()
    hot_threshold = max_b * 0.95
    hot_lags = set(np.where(b >= hot_threshold)[0].tolist())
    
    iterations = 0
    while time.monotonic() - t0 < time_limit:
        # Pick a random pair of coordinates to swap mass
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        if a[i] < 1e-10:
            continue
        
        # Try moving a small amount from i to j
        for frac in [0.001, 0.003, 0.01, 0.03, 0.1]:
            amt = a[i] * frac
            a_trial = a.copy()
            a_trial[i] -= amt
            a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            a_trial_s = a_trial.sum()
            if a_trial_s < 1e-15:
                continue
            a_trial = a_trial / a_trial_s
            score = 2.0 * n * fast_autocorr(a_trial).max()
            if score < best_score - 1e-13:
                a = a_trial
                best_score = score
                best_a = a.copy()
                b = fast_autocorr(a)
                max_b = b.max()
                hot_lags = set(np.where(b >= max_b * 0.95)[0].tolist())
                break
        
        iterations += 1
    
    return best_a, best_score, iterations


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_mirror = f_interp / ss
        # Also try the reversed version
        f_mirror = f_interp[::-1] / ss
    return f_interp

            if ss > 1e-15:
                a_avg = a_avg / ss
                try_update(a_avg, "avg_fwd_rev")
    
    # Final verification
    if best_seq is not None:
        fs = evaluate_safe(best_seq)
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    else:
        print("\nERROR: No valid sequence found!")
        return [1.0]
    
    return [float(max(0.0, min(1000.0, x))) for x in best_seq]
