    A = rfft(a, fft_size)

    lse_val = (Tc_max + np.log(exp_Tc.sum())) / T
    w = exp_Tc / exp_Tc.sum()

    # We need N >= n + (2n-1) - 1 = 3n-2
    # But with the flip trick: g = (a_flip * w)[n-1:2n-1]
    # a_flip has length n, w has length 2n-1
    # Linear conv has length n + 2n-1 - 1 = 3n-2
    # We need N >= 3n-2
    
    # Simpler: just use N >= 2n-1 and compute the cross-correlation directly
    # g[i] = sum_{m=0}^{n-1} a[m] * w[i+m]

    # Let's pad a to N and w to N, compute ifft(conj(fft(a,N)) * fft(w,N))
    # This gives circular cross-correlation, which equals linear for valid indices if N is large enough.
    
    # For the linear cross-correlation of a (len n) and w (len 2n-1):
    # We need N >= n + (2n-1) = 3n-1... actually the output length is n + (2n-1) - 1 = 3n-2
    # But we only need the first n values (for i=0 to n-1)
    # With N >= 2*(n-1) + (2n-1) = 4n-3 we'd be safe, but N >= 3n-2 should work
    # Actually, the standard requirement for linear cross-correlation via FFT:
    # N >= max(len(a), len(w)) + min(len(a), len(w)) - 1 = (2n-1) + n - 1 = 3n-2
    
    N = 1 << (3 * n - 2).bit_length()
    
    a_pad = np.zeros(N)

def nesterov_fw_optimize(a0, time_limit, n_fw_iters=80, verbose=True):

        a = a / s
    y = a.copy()

    t0 = time.monotonic()
    
    # Precompute FFT size
    fft_size = 1 << (2 * n - 1).bit_length()
    # For cross-correlation in FW
    corr_fft_size = 1 << (3 * n - 2).bit_length()
    
    stall_count = 0
    fw_step_size_base = 2.0
    
    for t in range(1, 5000000):
        if time.monotonic() - t0 > time_limit:
            break

        for fw_it in range(n_fw_iters):
            # Compute conv(y, b)

            c = irfft(Y_ff * B_ff, fft_size)[:2*n-1]
            
            # Softmax weights (for smooth FW)

def adam_lse_optimize(a0, time_limit, T_start=50.0, T_end=0.5, n_stages=30, verbose=True):
    """
    Adam optimization with full log-sum-exp (top-k=all) and multi-stage annealing.
    """
    n = len(a0)
    a_init = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a_init.sum()
    if s < 1e-15:
        a_init = np.ones(n) / n
    else:
        a_init = a_init / s
    z = np.log(np.maximum(a_init, 1e-30))
    t0 = time.monotonic()
    
    temps = np.geomspace(T_start, T_end, n_stages)
    best_a = a_init.copy()
    best_score = 2.0 * n * fast_autocorr(a_init).max()
    
    for si, stage_T in enumerate(temps):
        if time.monotonic() - t0 > time_limit:
            break
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break
        
        base_lr = 3e-3 * min(1.0, 100.0 / n)
        lr = base_lr * (0.75 ** si)
        
        m_z = np.zeros(n)
        v_z = np.zeros(n)
        t_adam = 0
        beta1, beta2, eps = 0.8, 0.999, 1e-8
        
        max_it = max(20, int(min(remaining, 60.0) * 5))
        
        for it in range(max_it):
            if time.monotonic() - t0 > time_limit - 3:
                break
            
            a = softmax_fn(z)
            obj, grad_a = lse_grad_fast(a, stage_T)
            S = np.dot(grad_a, a)
            grad_z = a * (grad_a - S)
            
            t_adam += 1
            m_z = beta1 * m_z + (1 - beta1) * grad_z
            v_z = beta2 * v_z + (1 - beta2) * grad_z ** 2
            m_hat = m_z / (1 - beta1 ** t_adam)
            v_hat = v_z / (1 - beta2 ** t_adam)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            if (it + 1) % 10 == 0:
                a_check = softmax_fn(z)
                score = 2.0 * n * fast_autocorr(a_check).max()
                if score < best_score:
                    best_score = score
                    best_a = a_check.copy()
        
        a_stage = softmax_fn(z)
        score_stage = 2.0 * n * fast_autocorr(a_stage).max()
        if score_stage < best_score:
            best_score = score_stage
            best_a = a_stage.copy()
        
        if verbose and si % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE stage {si}/{n_stages} (T={stage_T:.2f}, {elapsed:.0f}s): best={best_score:.6f}")
    
    return best_a, best_score

    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_autocorr(a).max()
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit - 3:
        b = fast_autocorr(a)
        bmax = b.max()
        threshold = bmax * 0.998
        top_peaks = np.where(b >= threshold)[0]
        if len(top_peaks) == 0:
            top_peaks = np.argsort(b)[::-1][:10]
        
        improved = False
        for k in top_peaks[:15]:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            if len(idx) == 0:
                continue
            contrib = a[idx] * a[k - idx]
            max_contrib = contrib.max()
            if max_contrib < 1e-15:
                continue
            src = idx[np.argmax(contrib)]
            
            # Danger zone
            danger = np.zeros(n, dtype=bool)
            for k2 in top_peaks[:15]:
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                danger[js2:je2 + 1] = True
            
            safe = np.where(~danger)[0]
            safe = safe[safe != src]
            if len(safe) == 0:
                safe = np.where(np.arange(n) != src)[0]
            safe_sorted = safe[np.argsort(a[safe])]
            dst_pool = safe_sorted[:max(3, len(safe_sorted) // 4)]
            if len(dst_pool) == 0:
                continue
            dst = dst_pool[rng.integers(len(dst_pool))]
            
            for frac in [0.0005, 0.002, 0.005, 0.01, 0.02, 0.04, 0.07]:
                amt = a[src] * frac
                if amt < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[src] -= amt
                a_trial[dst] += amt
                if a_trial[src] < -1e-15:
                    continue
                a_trial[src] = max(0, a_trial[src])
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            stall += 1
            # Random perturbation
            n_pert = rng.integers(1, 4)
            a_trial = a.copy()
            for _ in range(n_pert):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and a_trial[i] > 1e-10:
                    frac = rng.uniform(0.001, 0.03)
                    amt = a_trial[i] * frac
                    a_trial[i] -= amt
                    a_trial[j] += amt
            a_trial = np.maximum(a_trial, 0)
            st = a_trial.sum()
            if st > 1e-10:
                a_trial = a_trial / st
                score = 2.0 * n * fast_autocorr(a_trial).max()
                if score < best_score:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    stall = 0
            
            if stall > 500:
                break
    
    return best_a, best_score


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def structured_init(n, rng):
    """Generate diverse structured initializations."""
    choice = rng.integers(0, 10)
    if choice == 0:
        x = np.linspace(0, np.pi, n)
        a = np.cos(x / 2) ** 2
    elif choice == 1:
        x = np.linspace(-1, 1, n)
        a = np.exp(-x ** 2 * 3)
    elif choice == 2:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) ** 3
    elif choice == 3:
        x = np.linspace(0, 1, n)
        a = (1 - x) ** 2
    elif choice == 4:
        x = np.linspace(0, 2 * np.pi, n)
        a = 0.7 + 0.2 * np.cos(2 * x) + 0.1 * np.cos(5 * x)
    elif choice == 5:
        x = np.linspace(0, 1, n)
        a = np.exp(-5 * (x - 0.3) ** 2) + 0.5 * np.exp(-5 * (x - 0.7) ** 2)
    elif choice == 6:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x) * np.cos(np.pi * x / 3) ** 2
    elif choice == 7:
        # Golden ratio spacing
        phi = (1 + np.sqrt(5)) / 2
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.sin(2 * np.pi * phi * x * 10)
    elif choice == 8:
        # Multiple frequency
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.2 * np.sin(2 * np.pi * 3.7 * x) + 0.1 * np.sin(2 * np.pi * 7.3 * x)
    else:
        # Random with smoothing
        a = np.random.default_rng(choice * 1000).uniform(0.5, 1.5, n)
        # Simple smoothing
        kernel = np.ones(5) / 5
        a = np.convolve(a, kernel, mode='same')
    
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s
