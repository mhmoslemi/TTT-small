import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def subgradient_optimize(f0, time_limit, verbose=False):
    """Fast subgradient method with active set."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    
    lr = 0.01
    no_improve = 0
    consecutive_worse = 0
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        # Find active set: lags within 5% of max
        threshold = M * 0.95
        active = np.where(conv >= threshold)[0]
        
        # Compute averaged subgradient
        sg = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * f[k - idx] / len(active)
        
        # Step
        trial = f - lr * sg
        trial = project_simplex(trial)
        
        new_conv = fast_convolve(trial)
        new_M = new_conv.max()
        
        if new_M < best_obj - 1e-15:
            best_obj = new_M
            best_f = trial.copy()
            f = trial
            no_improve = 0
            consecutive_worse = 0
            lr = min(lr * 1.1, 0.05)
        elif new_M < M + 1e-10:
            f = trial
            no_improve += 1
            consecutive_worse = 0
            if new_M < M - 1e-15:
                lr = min(lr * 1.05, 0.05)
            else:
                lr *= 0.9
        else:
            consecutive_worse += 1
            lr *= 0.8
            no_improve += 1
        
        if no_improve > 100:
            # Restart with perturbation
            rng = np.random.default_rng(it * 7 + 3)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.01, 0)
            f = f / f.sum()
            lr = 0.01
            no_improve = 0
            consecutive_worse = 0
        
        if consecutive_worse > 50:
            lr *= 0.5
            consecutive_worse = 0
        
        if verbose and it % 100 == 0:
            print(f"  SG iter {it}: obj={best_obj:.10f}, lr={lr:.6f}")
    
    return best_f, best_obj

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, conv, frac=0.97):
    """Compute weighted subgradient using active set of near-max lags."""
    n = len(f)
    M = conv.max()
    threshold = M * frac
    active = np.where(conv >= threshold)[0]
    
    #Weights: softmax over active lags
    active_vals = conv[active]
    weights = np.exp((active_vals - active_vals.max()) * 100.0)
    weights /= weights.sum()
    
    sg = np.zeros(n)
    for i, k in enumerate(active):
        if weights[i] < 1e-10:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * weights[i] * f[k - idx]
    
    return sg, M, active

def make_gaussian(n, sigma_frac=0.25):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bimodal(n):
    """Bimodal: high at two positions."""
    a = np.zeros(n)
    pos1 = n // 4
    pos2 = 3 * n // 4
    sigma = n * 0.05
    x = np.arange(n)
    a += np.exp(-0.5 * ((x - pos1) / sigma) ** 2)
    a += np.exp(-0.5 * ((x - pos2) / sigma) ** 2)
    a = a / a.sum()
    return a

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using fast first-order methods:
- Projected subgradient with Nesterov momentum and adaptive step size
- Peak-targeted coordinate descent for local refinement
- Greedy LP-direction approximation (fast alternative to full LP solves)
- Multi-start with structured perturbations and multi-length exploration
- Final fine-grained polish
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, conv, frac):
    """Compute weighted subgradient using active set of near-max lags."""
    n = len(f)
    M = conv.max()
    threshold = M * frac
    active = np.where(conv >= threshold)[0]
    active_vals = conv[active]
    w = np.exp((active_vals - active_vals.max()) * 200.0)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.zeros(len(active))
        w[0] = 1.0
    else:
        w = w / wsum
    sg = np.zeros(n)
    for i in range(len(active)):
        if w[i] < 1e-12:
            continue
        k = int(active[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k - idx]
    return sg, M


def greedy_direction(f, M, n):
    """
    Fast greedy approximation of LP-Frank-Wolfe direction.
    Tries pairs of coordinates to build a direction that reduces the peak.
    """
    # For each coordinate j, the "cost" of increasing it is how much it
    # contributes to each lag k: cost[j] = max_k f[k-j] (the partner values)
    # We want to increase coordinates where the partner values are small.
    
    # Compute for each j the vector of partner values f[k-j] over valid k
    # The "bottleneck" is max over k of f[k-j]
    # We want to increase j where this bottleneck is small
    
    # For efficiency, compute the max partner for each j
    max_partner = np.zeros(n)
    for j in range(n):
        # k ranges from j to j+n-1, clamped to [0, 2n-2]
        k_min = 0
        k_max = min(2 * n - 2, j + n - 1)
        # f[k-j] for k in [k_min, k_max] where 0 <= k-j < n
        lo = max(0, -j)
        hi = min(n - 1, k_max - j)
        if lo <= hi:
            max_partner[j] = f[lo:hi + 1].max()
        else:
            max_partner[j] = 0.0
    
    # Direction: increase coordinates with smallest max_partner
    # Use a top-K approach
    k_top = min(8, n)
    top_idx = np.argsort(max_partner)[:k_top]
    
    g = np.zeros(n)
    for j in top_idx:
        delta = M / (max_partner[j] + 1e-15)
        g[j] = delta
    
    gsum = g.sum()
    if gsum < 1e-15:
        g = np.ones(n) / n
    else:
        g = g / gsum
    
    return g


def fw_greedy_optimize(f0, time_limit, verbose, lr_init):
    """Frank-Wolfe with greedy direction (fast, no LP solves)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 1.0:
            break
        
        conv = fast_convolve(f)
        M = conv.max()
        
        g = greedy_direction(f, M, n)
        
        # Line search: try several step sizes
        best_alpha = 0.0
        best_val = M
        for alpha in [0.01, 0.03, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha < 1e-10:
            no_improve += 1
            if no_improve > 30:
                # Perturb
                rng = np.random.default_rng(it * 31 + 11)
                f = np.maximum(best_f + rng.standard_normal(n) * 0.005, 0)
                f = f / f.sum()
                no_improve = 0
            continue
        
        f = f + best_alpha * (g - f)
        f = np.maximum(f, 0.0)
        f = f / f.sum()
        conv = fast_convolve(f)
        M = conv.max()
        
        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 50 == 0:
                print(f"    FW-G iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
    
    return best_f, best_obj


def subgradient_momentum(f0, time_limit, verbose, lr_init, momentum):
    """Projected subgradient with Nesterov momentum."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    
    lr = lr_init
    v = np.zeros(n)
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        
        sg, M_val = compute_subgradient(f, conv, 0.97)
        
        v = momentum * v + lr * sg
        trial = f - v
        trial = project_simplex(trial)
        
        new_conv = fast_convolve(trial)
        new_M = new_conv.max()
        
        if new_M < best_obj - 1e-15:
            best_obj = new_M
            best_f = trial.copy()
            no_improve = 0
            lr = min(lr * 1.01, lr_init * 3.0)
        else:
            no_improve += 1
            if new_M < M - 1e-15:
                f = trial
                lr = min(lr * 1.005, lr_init * 3.0)
            else:
                lr *= 0.97
        
        if no_improve > 300:
            rng = np.random.default_rng(it * 13 + 7)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.004, 0)
            f = f / f.sum()
            v = np.zeros(n)
            lr = lr_init
            no_improve = 0
        
        if verbose and it % 500 == 0:
            print(f"    SG-M iter {it}: obj={best_obj:.10f}, lr={lr:.6f}")
    
    return best_f, best_obj


def peak_coord_descent(f0, time_limit, verbose):
    """Peak-targeted coordinate descent with smart moves."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if no_improve > 500:
            break
        
        conv = fast_convolve(f)
        M = conv.max()
        
        # Get top peaks
        k_top = min(5, 2 * n - 1)
        top_peaks = np.argpartition(conv, -k_top)[-k_top:]
        top_peaks = top_peaks[np.argsort(conv[top_peaks])[::-1]]
        
        improved = False
        for pk in top_peaks:
            if time.monotonic() - t0 > time_limit:
                break
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            idx = np.arange(js, je + 1)
            products = f[idx] * f[pk - idx]
            
            if products.sum() < 1e-15:
                continue
            
            # Try reducing top contributors
            order = np.argsort(products)[::-1]
            for oi in range(min(3, len(order))):
                max_i = order[oi]
                src = int(idx[max_i])
                partner = pk - src
                
                for frac in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4]:
                    delta = f[src] * frac
                    if delta < 1e-15:
                        continue
                    
                    trial = f.copy()
                    trial[src] -= delta
                    
                    # Move mass to non-contributing coordinates
                    non_contrib = np.ones(n, dtype=bool)
                    non_contrib[idx] = False
                    nc_idx = np.where(non_contrib)[0]
                    if len(nc_idx) == 0:
                        continue
                    
                    nc_vals = trial[nc_idx]
                    nc_sum = nc_vals.sum()
                    if nc_sum < 1e-15:
                        trial[nc_idx] += delta / len(nc_idx)
                    else:
                        trial[nc_idx] += delta * nc_vals / nc_sum
                    
                    trial = np.maximum(trial, 0)
                    ts = trial.sum()
                    if ts < 1e-15:
                        continue
                    trial = trial / ts
                    
                    new_M = fast_convolve(trial).max()
                    if new_M < best_obj - 1e-14:
                        best_obj = new_M
                        best_f = trial.copy()
                        f = trial
                        no_improve = 0
                        improved = True
                        break
                    elif new_M < M - 1e-12:
                        f = trial
                        no_improve = 0
                        improved = True
                        break
                
                if improved:
                    break
            
            if improved:
                break
        
        if not improved:
            no_improve += 1
            # Try swap-based moves
            if no_improve % 10 == 0:
                rng = np.random.default_rng(it * 7 + 3)
                for _ in range(5):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i == j:
                        continue
                    for mix in [0.1, 0.2, 0.3, 0.5]:
                        trial = f.copy()
                        tmp = f[i] * mix
                        if tmp < 1e-15:
                            continue
                        trial[i] -= tmp
                        trial[j] += tmp
                        if trial[i] < 0:
                            continue
                        trial = trial / trial.sum()
                        new_M = fast_convolve(trial).max()
                        if new_M < best_obj - 1e-14:
                            best_obj = new_M
                            best_f = trial.copy()
                            f = trial
                            no_improve = 0
                            improved = True
                            break
                    if improved:
                        break
        
        if verbose and it % 100 == 0:
            print(f"    PCD iter {it}: obj={best_obj:.10f}, no_improve={no_improve}")
    
    return best_f, best_obj


def make_gaussian(n, sigma_frac):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    sm = a.sum()
    if sm > 1e-15:
        a /= sm
    return a


def make_bimodal(n):
    a = np.zeros(n)
    pos1 = n // 4
    pos2 = 3 * n // 4
    sigma = max(1.0, n * 0.04)
    x = np.arange(n)
    a += np.exp(-0.5 * ((x - pos1) / sigma) ** 2)
    a += np.exp(-0.5 * ((x - pos2) / sigma) ** 2)
    sm = a.sum()
    if sm > 1e-15:
        a /= sm
    return a


def make_multi_peak(n, n_peaks, seed):
    rng = np.random.default_rng(seed)
    a = np.ones(n) / n
    sigma = max(1.0, n * 0.03)
    x = np.arange(n)
    for _ in range(n_peaks):
        pos = int(rng.integers(0, n))
        a += np.exp(-0.5 * ((x - pos) / sigma) ** 2)
    a = np.maximum(a, 0)
    sm = a.sum()
    if sm > 1e-15:
        a /= sm
    return a


def make_u_shaped(n, sharpness):
    """U-shaped: high at edges, low in middle."""
    x = np.arange(n) - n / 2.0
    a = np.cosh(x / (n * sharpness))
    a = a / a.sum()
    return a


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def objective_quick(a):
    """Quick objective for normalized sequence."""
    a = np.asarray(a, dtype=np.float64)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    conv = fast_convolve(a_norm)
    return 2.0 * len(a) * conv.max()
