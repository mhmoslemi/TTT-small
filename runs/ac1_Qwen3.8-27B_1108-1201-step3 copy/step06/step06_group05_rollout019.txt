def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def make_half_power_shape(n):
    """Create sequence based on (1-x)^(-1/2) coefficients: binom(2k,k)/4^k."""
    k = np.arange(n)
    # Use log for numerical stability
    log_c = np.array([np.log(np.math.comb(2*i, i)) - i * np.log(4) for i in range(n)])
    # Actually for large n, use Stirling or direct computation
    # c_k = binom(2k,k) / 4^k
    c = np.ones(n)
    for i in range(1, n):
        c[i] = c[i-1] * (2*i - 1) / (2*i)  # recurrence: c_k = c_{k-1} * (2k-1)/(2k)
    s = c.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return c / s

def make_symmetric_half_power(n):

def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. conv(f,g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    # Build constraint matrix A: A[k, j] = f[k-j] if 0 <= k-j < n
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j+n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def spectral_equalize(f, n_iter=50, alpha=0.5):
    """
    Frequency-domain shaping: tilt the spectrum to flatten the convolution.
    """
    a = np.maximum(f, 0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        return f
    a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    for it in range(n_iter):
        A = np.fft.fft(a)
        power = np.abs(A) ** 2
        # IFFT of power is the convolution (circular, but close for our purposes)
        conv_circ = np.fft.ifft(power).real
        # Only consider the valid part
        conv_valid = np.concatenate([conv_circ[n-1:], conv_circ[:n-1]])
        # Actually, let's just use the full circular conv and take the relevant part
        # The circular conv of a with itself: conv_circ[k] = sum_i a[i]*a[(k-i)%n]
        # The linear conv has 2n-1 values; the circular conv "wraps" them.
        # For our purpose, let's just use the circular conv peak.
        
        peak = conv_circ.max()
        mean = conv_circ.mean()
        
        if peak < 1e-15:
            break
        
        # Tilt: reduce power where conv is high
        tilt = np.exp(-alpha * (conv_circ - mean) / mean)
        A_new = A * np.sqrt(tilt)
        a_new = np.fft.ifft(A_new).real
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        obj = fast_convolve(a_new).max()
        if obj < best_obj:
            best_obj = obj
            best_a = a_new.copy()
        else:
            a = best_a.copy()  # revert
            alpha *= 0.8
    
    return best_a

def gradient_descent_smoothed(f0, time_limit, verbose=False):
    """Smoothed gradient descent with LSE."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    c = fast_convolve(f)
    best_obj = c.max()
    
    t0 = time.monotonic()
    
    beta_schedule = [4, 8, 16, 32, 64, 128]
    momentum = np.zeros(n)
    mu = 0.85
    
    for beta in beta_schedule:
        if time.monotonic() - t0 > time_limit * 0.9:
            break
        
        it = 0
        no_improve = 0
        lr = 1.0 / (beta * 2.0 * np.sqrt(n))
        
        stage_start = time.monotonic()
        stage_time = min(time_limit * 0.1, (time_limit - (time.monotonic() - t0)) * 0.2)
        
        while time.monotonic() - stage_start < stage_time and time.monotonic() - t0 < time_limit * 0.9:
            it += 1
            c = fast_convolve(f)
            
            # Compute smoothed gradient
            scaled = beta * c
            scaled -= scaled.max()
            exp_scaled = np.exp(scaled)
            w = exp_scaled / exp_scaled.sum()
            
            # g_j = 2 * sum_i f_i * w_{j+i} (cross-correlation)
            w_rev = w[::-1].copy()
            conv_len = n + len(w_rev) - 1
            fft_size = 1 << (conv_len - 1).bit_length()
            F = np.fft.rfft(f, fft_size)
            W = np.fft.rfft(w_rev, fft_size)
            conv = np.fft.irfft(F * W, fft_size)[:conv_len]
            indices = np.arange(2 * n - 2, 2 * n - 2 - n, -1)
            g = 2.0 * conv[indices]
            
            g_norm = np.linalg.norm(g)
            if g_norm < 1e-15:
                break
            
            lr = min(lr, 1.0 / (g_norm + 1e-10))
            momentum = mu * momentum - lr * g
            f_trial = project_simplex(f + momentum)
            
            c_trial = fast_convolve(f_trial)
            obj_trial = c_trial.max()
            
            if obj_trial <= c.max() + 1e-10:
                f = f_trial
                if obj_trial < best_obj - 1e-12:
                    best_obj = obj_trial
                    best_f = f.copy()
                    no_improve = 0
                    lr = min(lr * 1.05, 1.0 / (g_norm * 0.5 + 1e-10))
                else:
                    no_improve += 1
            else:
                no_improve += 1
                lr *= 0.7
            
            if no_improve > 60:
                rng = np.random.default_rng(it * 13 + int(beta) * 7)
                noise_scale = 0.002
                f = project_simplex(best_f + rng.standard_normal(n) * noise_scale)
                momentum = np.zeros(n)
                lr = 1.0 / (beta * 2.0 * np.sqrt(n))
                no_improve = 0
    
    return best_f, best_obj

def coordinate_polish(f0, time_limit, verbose=False):
    """Fine coordinate descent."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        return f0, fast_convolve(f0).max()
    f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        c = fast_convolve(f)
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-12:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in [0.7, 0.8, 0.9, 1.1, 1.2, 1.4]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                
                f_trial = f.copy()
                f_trial[j] = new_val
                f_trial = np.maximum(f_trial, 0)
                ss = f_trial.sum()
                if ss < 1e-15:
                    continue
                f_trial = f_trial / ss
                
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                f[j] = best_j_val
                f = np.maximum(f, 0)
                f = f / f.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_f, best_obj
