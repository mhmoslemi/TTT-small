Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smoothed gradient descent
with adaptive temperature annealing, multi-length exploration, and
LP-Frank-Wolfe refinement.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def cross_correlate(w, p):
    """Compute cross-correlation (w*p)[j] = sum_{j'} w[j+j']*p[j'] for j=0..n-1."""
    n = len(p)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    P = np.fft.rfft(p, fft_size)
    # Cross-correlation: sum_j' w[j+j'] * p[j']
    # This is the correlation of w with p
    # corr[j] = sum_j' w[j+j'] * p[j']
    # In FFT terms: if we reverse w, then corr = ifft(conj(FFT(w_rev)) * FFT(p))
    # Or: corr = ifft(FFT(w) * conj(FFT(p))) ... let me think.
    # Actually: sum_j' w[j+j'] * p[j'] = sum_j' w[j'] * p[j] ... no.
    # Let me define: c[j] = sum_{j'=0}^{n-1} w[j+j'] * p[j'] (with w[k]=0 for k>=len(w))
    # This is a cross-correlation. In terms of convolution:
    # c[j] = sum_{j'} w[j+j'] * p[j'] = (p * w_flipped)[j + ...]
    # Actually, let's just use the direct formula with FFT.
    # c[j] = sum_{j'} w[j+j'] * p[j']
    # Let u = j'. Then c[j] = sum_u w[j+u] * p[u]
    # This is the cross-correlation of w and p at lag j.
    # In FFT: cross-correlation can be computed as ifft(conj(FFT(w)) * FFT(p))
    # But we need to be careful about the definition.
    # 
    # Standard cross-correlation: (w ⋆ p)[j] = sum_k w[k+j] * conjugate(p[k])
    # In our case (real-valued): (w ⋆ p)[j] = sum_k w[k+j] * p[k]
    # This is what we want: c[j] = sum_k w[k+j] * p[k]
    # FFT: (w ⋆ p) = ifft(conj(FFT(w)) * FFT(p)) (with zero-padding)
    
    W = np.fft.rfft(w, fft_size)
    P = np.fft.rfft(p, fft_size)
    result = np.fft.irfft(np.conj(W) * P, fft_size)
    # The result has length fft_size. We want c[0], c[1], ..., c[n-1]
    # From the FFT definition, result[m] = sum_k w[k+m] * p[k] for m >= 0
    # (with appropriate wrap-around, but since we zero-padded, it's correct for m < n)
    return result[:n]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def objective_normalized(a):
    """Compute 2*n*max(conv)/sum^2 for normalized a (sum=1)."""
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max()

    Uses log-sum-exp smoothing with adaptive temperature.

    n = len(p0)
    p = np.maximum(p0.astype(np.float64), 0.0)
    s = p.sum()
    if s < 1e-15:
        p = np.ones(n) / n
    else:
        p = p / s

    t = t_init
    best_obj = fast_convolve(p).max()
    best_p = p.copy()
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(p)
        
        # Compute smoothed gradient
        # w = softmax(t * conv)
        tc = t * conv
        tc_max = tc.max()
        w = np.exp(tc - tc_max)
        w_sum = w.sum()
        if w_sum < 1e-300:

            w[conv.argmax()] = 1.0
        else:
            w = w / w_sum

        # grad[j] = (2/t) * sum_{j'} w[j+j'] * p[j']
        # This is (2/t) * cross_correlate(w, p)
        grad = (2.0 / t) * cross_correlate(w, p)
        
        # Step size

            no_improve += 1

            # Small perturbation

        f = np.ones(len(f0)) / len(f0)

    it = 0

        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 1.0:
            break

        # Using the primal formulation

        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):

        b_ub = np.full(2 * n - 1, M)

            if no_improve > 5:

            conv = fast_convolve(f)
            M = conv.max()
            continue

        g = res.x
        g_sum = g.sum()
        if g_sum < 1e-12:
            no_improve += 1
            if no_improve > 5:
                break
            continue
        g = g / g_sum

        diff = np.max(np.abs(g - f))
        if diff < 1e-10:
            no_improve += 1
            if no_improve > 30:
                break
            continue
        no_improve = 0

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:

            no_improve += 1
            if no_improve > 30:
                break
            continue

        f_new = np.maximum(f + best_alpha * (g - f), 0.0)
        f_new = f_new / f_new.sum()
        conv = fast_convolve(f_new)
        M = conv.max()
        f = f_new

        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()

    return best_f, best_obj


def peak_descent(f0, time_limit, verbose=False):
    """Peak-targeted coordinate descent with pair operations."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        M = conv.max()
        peak_k = int(np.argmax(conv))

        # Find contributing indices
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        if js > je:
            no_improve += 1
            if no_improve > 100:
                break
            # Try a different peak
            conv[peak_k] = -np.inf
            peak_k = int(np.argmax(conv))
            conv[peak_k] = M  # restore
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            if js > je:
                break
            continue

        contrib_idx = np.arange(js, je + 1)
        products = f[contrib_idx] * f[peak_k - contrib_idx]
        
        # Try reducing the top contributing pair
        top_prod_pos = np.argmax(products)
        src = contrib_idx[top_prod_pos]
        partner = peak_k - src

        # Strategy: move mass from src to a random non-contributing position
        non_contrib = np.ones(n, dtype=bool)
        non_contrib[contrib_idx] = False
        if non_contrib.sum() < 1:
            non_contrib[:] = True
            non_contrib[src] = False
        
        improved = False
        for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
            diff = f[src] * frac
            if diff < 1e-15:
                continue
            trial = f.copy()
            trial[src] -= diff
            # Distribute to non-contributing positions
            nc_idx = np.where(non_contrib)[0]
            # Pick a few random target positions
            targets = rng.choice(nc_idx, size=min(5, len(nc_idx)), replace=False)
            for target in targets:
                trial[target] += diff / len(targets)
            trial = np.maximum(trial, 0.0)
            ts = trial.sum()
            if ts < 1e-15:
                continue
            trial = trial / ts
            obj = fast_convolve(trial).max()
            if obj < best_obj - 1e-13:
                f = trial
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
                improved = True
                break

        if not improved:
            # Try pair swap
            if no_improve < 50:
                # Swap a high value with a low value at non-contributing position
                nc_idx = np.where(non_contrib)[0]
                if len(nc_idx) > 0:
                    # Find the highest and lowest values
                    hi = int(np.argmax(f))
                    lo_candidates = nc_idx[f[nc_idx] == f[nc_idx].min()]
                    if len(lo_candidates) > 0 and hi not in contrib_idx:
                        lo = lo_candidates[0]
                        if f[hi] > f[lo] + 1e-6:
                            # Try partial swap
                            for swap_frac in [0.1, 0.3, 0.5, 0.8]:
                                amt = (f[hi] - f[lo]) * swap_frac
                                trial = f.copy()
                                trial[hi] -= amt
                                trial[lo] += amt
                                trial = np.maximum(trial, 0.0)
                                trial = trial / trial.sum()
                                obj = fast_convolve(trial).max()
                                if obj < best_obj - 1e-13:
                                    f = trial
                                    best_obj = obj
                                    best_f = f.copy()
                                    no_improve = 0
                                    improved = True
                                    break
            no_improve += 1
            if no_improve > 200:
                # Perturb
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                no_improve = 0

    return best_f, best_obj


def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_two_bump(n, frac=0.3, gap=0.1):
    """Two bump initialization."""
    a = np.zeros(n)
    m = int(n * frac)
    center1 = m // 2
    center2 = n - m // 2
    for j in range(n):
        d1 = abs(j - center1)
        d2 = abs(j - center2)
        a[j] = max(0, 1 - d1 / (m * 0.6)) + max(0, 1 - d2 / (m * 0.6))
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_cosine_mod(n, freq, amp):
    x = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * x / n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
