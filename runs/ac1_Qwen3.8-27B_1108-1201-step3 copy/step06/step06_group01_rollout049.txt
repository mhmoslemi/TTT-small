import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def cross_corr(w, a):
    """Cross-correlation: result[i] = sum_k w[k] * a[k-i] (valid indices only)."""
    n = len(a)
    m = len(w)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    return irfft(np.conj(W) * A, fft_size)[:n]


def softmax_from_z(z):
    """Convert log-domain z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector a to log-domain z."""
    return np.log(np.maximum(a, 1e-300))


def soft_max_obj_grad(z, T, n):
    """
    Soft-max smoothed objective and gradient in log-space.
    f(a) = (2n/T) * logsumexp(T * conv(a,a))
    """
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb / T + np.log(exp_Tb.sum()) / T
    obj = 2.0 * n * log_sum_exp
    W = exp_Tb / exp_Tb.sum()
    # Gradient w.r.t. a: grad_a[i] = 2 * sum_k (2n * W[k]) * a[k-i] = 4n * cross_corr(W, a)[i]
    # But for L-BFGS, the constant factor doesn't matter, so use 2n for numerical stability
    grad_a = 2.0 * n * cross_corr(W, a)
    # Gradient w.r.t. z: grad_z[i] = a[i] * (grad_a[i] - S) where S = dot(grad_a, a)
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - S)
    return obj, grad_z


def lbgfs_refine(a0, time_limit, T=100, maxiter=5000, verbose=False):
    """L-BFGS refinement in log-space with soft-max smoothing."""
    n = len(a0)
    z0 = z_from_a(a0)
    t0 = time.monotonic()
    stopped = [False]

    def func(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        obj, _ = soft_max_obj_grad(z, T, n)
        return obj

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return np.zeros_like(z)
        _, g = soft_max_obj_grad(z, T, n)
        return g

    best_z = z0.copy()
    best_obj = func(z0)
    
    # Run L-BFGS
    try:
        res = minimize(
            fun=func,
            x0=z0,
            jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10,
                     'maxfun': maxiter * 50, 'disp': False}
        )
        if res.x is not None:
            a_candidate = softmax_from_z(res.x)
            score_candidate = 2.0 * n * fast_convolve(a_candidate).max()
            a0_score = 2.0 * n * fast_convolve(a0).max()
            if score_candidate < a0_score:
                best_z = res.x.copy()
                best_obj = score_candidate
    except Exception:
        pass

    a_out = softmax_from_z(best_z)
    if verbose:
        elapsed = time.monotonic() - t0
        final_score = 2.0 * n * fast_convolve(a_out).max()
        print(f"    L-BFGS T={T} ({elapsed:.0f}s): score={final_score:.6f}")
    return a_out


def build_conv_matrix_rows(a, k_list):
    """Build rows of the convolution matrix for specific lags."""
    n = len(a)
    rows = []
    for k in k_list:
        row = np.zeros(n)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            row[j] = a[k - j]
        rows.append(row)
    return np.array(rows)


def solve_lp_best_response(a_fixed, solver=None):
    """
    Given fixed a (non-negative, sum=1), find b (non-negative, sum=1)
    minimizing max_k conv(a, b)[k]. Returns (b, M).
    """
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    
    # Build convolution matrix efficiently
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a_fixed
    
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.GLPK]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt)
                break
            except Exception:
                continue
    
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def get_lp_solver():
    """Find a working LP solver."""
    global _lp_solver

    for solver_name in [cp.MOSEK, cp.GUROBI, cp.GLPK, cp.ECOS, cp.SCS]:

def nesterov_subgradient(a0, time_limit, learning_rate=0.005, nesterov_momentum=0.9):
    """Nesterov accelerated subgradient method."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    y = a.copy()  # Nesterov lookahead
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    no_improve = 0
    lr = learning_rate

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Compute subgradient at y
        b = fast_convolve(y)
        max_b = b.max()
        # Active set: indices where b is close to max
        threshold = max_b - 0.01 * max_b
        active = np.where(b >= threshold)[0]
        
        # Subgradient: average of gradients at active peaks
        grad = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * y[k - idx]
        grad /= len(active)
        grad *= 2.0 * n  # Scale for the objective
        
        # Nesterov step
        t_k = 2.0 / (it + 1 + np.sqrt(1 + 4 * it * it))
        x_new = y - lr * grad
        # Project to simplex
        x_new = simplex_project(x_new)
        y = (1 - t_k) * x_new + t_k * y  # Wait, this isn't right
        # Actually for Nesterov: y_{k+1} = x_{k+1} + (t_k - 1)/t_{k+1} * (x_{k+1} - x_k)
        # Simplified: y_new = x_new + momentum * (x_new - a)
        
        # Simplified Nesterov-like update
        a_new = x_new
        y = a_new + nesterov_momentum * (a_new - a)
        y = simplex_project(y)
        a = a_new
        
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        lr *= 0.999
        if no_improve > 200:
            # Reset with perturbation
            rng = np.random.default_rng(it)
            a = best_a + rng.standard_normal(n) * 0.002

            y = a.copy()

            no_improve = 0

            elapsed = time.monotonic() - t0
            print(f"    Nesterov {elapsed:.0f}s: score={best_score:.6f} lr={lr:.6f}")

    return best_a, best_score


def simplex_project(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if cond.any():
        rho_val = np.nonzero(cond)[0][-1]
        theta = (cssv[rho_val] - 1.0) / (rho_val + 1)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def peak_equalization(a0, time_limit, max_rounds=200):
    """
    Iteratively equalize the top peaks of the autocorrelation.
    At each step, find the highest peak and try to reduce it by
    adjusting the contributing elements.
    """
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0).copy()
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for round_idx in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        max_b = b.max()
        
        # Find all peaks within 1% of max
        threshold = max_b * 0.99
        peaks = np.where(b >= threshold)[0]
        
        if len(peaks) == 0:
            break

        # For each peak, try to reduce it
        improved = False
        for k in sorted(peaks, key=lambda x: -b[x]):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[k - idx]
            
            if contribs.max() < 1e-15:
                continue

            # Try reducing the largest contributor
            top_i = idx[np.argmax(contribs)]
            for factor in [0.95, 0.9, 0.85, 0.8, 0.7]:
                diff = a[top_i] * (1 - factor)
                if diff < 1e-15:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[top_i] = False
                others = a[mask]
                if others.sum() < 1e-15:
                    continue
                a_trial = a.copy()
                a_trial[top_i] *= factor
                a_trial[mask] = others + diff * others / others.sum()
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            # Try pairwise reduction
            for k in sorted(peaks, key=lambda x: -b[x]):
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                contribs = a[idx] * a[k - idx]
                top_two = np.argsort(contribs)[::-1][:2]
                if len(top_two) < 2:
                    continue
                i1, i2 = idx[top_two[0]], idx[top_two[1]]
                if i1 == i2:
                    continue
                for f1 in [0.9, 0.8]:
                    for f2 in [0.9, 0.8]:
                        a_trial = a.copy()
                        a_trial[i1] *= f1
                        a_trial[i2] *= f2
                        a_trial = np.maximum(a_trial, 0)
                        a_trial = a_trial / a_trial.sum()
                        score = 2.0 * n * fast_convolve(a_trial).max()
                        if score < best_score - 1e-13:
                            a = a_trial
                            best_score = score
                            best_a = a.copy()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            if not improved:
                break

    return best_a, best_score


def damped_lp_refine(a0, time_limit, solver=None, max_iters=3000):
    """Damped alternating LP refinement."""
    n = len(a0)
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - t0 > time_limit:
            break

        b_new, M_b = solve_lp_best_response(a, solver)
        if b_new is None:
            continue

        alpha = min(1.0, 3.0 / (it + 1))
        a_new = (1 - alpha) * a + alpha * b_new
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()
        a = a_new

    return best_a, best_score
