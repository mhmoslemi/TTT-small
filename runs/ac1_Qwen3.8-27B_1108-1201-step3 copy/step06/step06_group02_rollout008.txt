def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)

    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)

    cond = u * np.arange(1, n + 1) > (cssv - 1)

        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp_primal(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)

        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def objective_normalized(a):
    """Compute 2*n*max(conv) for normalized a (sum=1)."""
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max()


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j, j + n] = f[::-1]
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={"presolve": True, "dual_feasibility_tolerance": 1e-9})
    if res.success:
        return res.x
    return None

def swap_coordinate_descent(f0, time_limit, verbose=False):
    """
    Swap-based coordinate descent: move mass from peak-contributing
    coordinates to non-contributing ones.
    """

    n = len(f)

    it = 0

        it += 1
        if no_improve > 300:
            break
        if time.monotonic() - t0 < time_limit and (time.monotonic() - t0) > time_limit - 2:
            break

        conv = fast_convolve(f)
        M = conv.max()
        peak_k = int(np.argmax(conv))

        # Contributing indices for the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        contrib_idx = np.arange(js, je + 1)

        if len(contrib_idx) == 0:
            break

        products = f[contrib_idx] * f[peak_k - contrib_idx]
        total = products.sum()
        if total < 1e-15:
            break

        # Find top contributing sources
        sorted_products = np.sort(products)[::-1]
        top_k = min(5, len(contrib_idx))
        top_srcs = contrib_idx[np.argsort(products)[::-1][:top_k]]

        # Target candidates: coordinates far from the source
        found_improvement = False
        for src in top_srcs:
            if found_improvement:
                break
            # Compute distances to find good targets
            dists = np.abs(np.arange(n) - src)
            # Prefer targets with low current value and high distance
            target_score = dists * (1.0 / (f + 1e-10))
            target_score[src] = 0
            num_targets = min(30, n)
            target_candidates = np.argsort(target_score)[::-1][:num_targets]

            for tgt in target_candidates:
                if found_improvement:
                    break
                if tgt == src:
                    continue
                for frac in [0.03, 0.05, 0.1, 0.15, 0.2]:
                    diff = f[src] * frac
                    if diff < 1e-15:
                        continue
                    trial = f.copy()
                    trial[src] -= diff
                    trial[tgt] += diff
                    if trial[src] < -1e-12:
                        continue
                    trial = np.maximum(trial, 0.0)
                    trial = trial / trial.sum()
                    obj = fast_convolve(trial).max()
                    if obj < best_obj - 1e-14:
                        f = trial
                        best_obj = obj

                        break

            no_improve += 1

            if no_improve % 50 == 0:
                rng = np.random.default_rng(it * 31 + 7)
                f = project_simplex(f + rng.standard_normal(n) * 0.001)

    return best_f, best_obj


def multiplicative_descent(f0, time_limit, verbose=False):
    """Multiplicative weight descent with adaptive temperature."""
    f = np.maximum(f0.astype(np.float64), 1e-12)
    f = f / f.sum()
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    lr = 0.003
    no_improve = 0
    temperature = 0.001
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Compute weighted subgradient from top peaks
        k_actual = min(20, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = pv / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        # Multiplicative update
        f = f * np.exp(-lr * sg)
        f = np.maximum(f, 1e-15)
        f = f / f.sum()

        co = fast_convolve(f).max()
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 50 == 0:
                print(f"    MD iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        lr *= 0.9997
        temperature *= 1.005

        if no_improve > 300:
            rng = np.random.default_rng(it * 41 + 3)
            f = np.maximum(best_f + rng.standard_normal(n) * 0.002, 1e-15)
            f = f / f.sum()
            lr = 0.002
            no_improve = 0

        if lr < 1e-9:
            break

    return best_f, best_obj


def make_gaussian(n, sigma_frac=0.25):
    """Gaussian initial sequence."""
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.5):
    """Bell-shaped initial sequence."""
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 6
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_cosine_mod(n, freq, amp):
    """Cosine-modulated uniform sequence."""
    x = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * x / n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_multi_cosine(n, n_freqs=4, max_amp=0.25, seed=0):
    """Multi-frequency cosine-modulated sequence."""
    rng = np.random.default_rng(seed)
    a = np.ones(n)
    for m in range(1, n_freqs + 1):
        amp = rng.uniform(0.03, max_amp)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2.0 * np.pi * m * np.arange(n) / n + phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_flat_top(n, taper=0.15):
    """Flat-top with cosine taper at edges."""
    a = np.ones(n)
    t = int(n * taper)
    if t > 0:
        edge = 0.5 * (1 + np.cos(np.pi * np.arange(t) / t))
        a[:t] = edge
        a[-t:] = edge[::-1]
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

            print(f"  Final LP-FW: alloc={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "final_lpfw")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(768) / 768
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
