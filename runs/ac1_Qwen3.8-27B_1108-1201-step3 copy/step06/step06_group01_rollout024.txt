    cond = u * ind > (cssv - 1)
    if cond.any():
        rho = ind[cond][-1]
        theta = (cssv[rho - 1] - 1.0) / rho
    else:
        return np.ones(n) / n
    return np.maximum(v - theta, 0.0)

def get_peaks(b, top_k=10):
    """Get indices of top-k peaks in convolution."""
    k = min(top_k, len(b))
    return np.argpartition(b, -k)[-k:]

def subgradient_step(a, lr, mom, v, top_k=20):
    """One step of subgradient descent with momentum."""
    n = len(a)
    b = fast_convolve(a)
    peaks = get_peaks(b, top_k)
    
    # Temperature-weighted peaks
    pv = b[peaks]
    temp = 0.01 * max(pv.max(), 1e-10)
    w = np.exp((pv - pv.max()) / temp)
    w /= w.sum()
    
    # Compute subgradient
    sg = np.zeros(n)
    for i, k in enumerate(peaks):
        if w[i] < 1e-12:
            continue
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += w[i] * a[k - idx]
    
    v = mom * v - lr * sg
    a_new = project_simplex(a + v)
    return a_new, v

def peak_directed_moves(a, n_moves=100):

            k_j = peak_k - j
            prod = a[j] * a[k_j]
            if prod < 1e-10:
                continue

            for factor in [0.9, 0.8, 0.7]:
                diff = a[j] * (1 - factor)
                if diff < 1e-12:
                    continue

                # (i.e., the position where adding mass increases the max least)
                # Approximate: add to position that's far from peak_k
                a_trial = a.copy()
                a_trial[j] *= factor
                # Add the mass to a "random" far position or the position
                # that minimizes the resulting peak
                # For efficiency, try adding to a few candidate positions
                candidates = [0, n-1, n//2]
                # Remove j from candidates
                candidates = [c for c in candidates if c != j]
                if not candidates:
                    candidates = [n - 1 - j]
                
                for tgt in candidates:
                    a_trial2 = a_trial.copy()
                    a_trial2[tgt] += diff
                    a_trial2 = np.maximum(a_trial2, 0)
                    s = a_trial2.sum()
                    if s < 1e-10:
                        continue

                    if sc < best_score - best_improvement:
                        best_improvement = best_score - sc
                        best_new_a = a_trial2.copy()

            a = best_new_a
            best_a = a.copy()

                            a = a_trial
                            best_a = a.copy()
                            best_score = sc
                            improved = True
                            break
                    if improved:
                        break
                if improved:

            if not improved:

def simulated_annealing(a0, time_limit, rng):
    """Simulated annealing on the simplex."""
    n = len(a0)
    a = a0.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    T_start = 0.01
    T_end = 0.0001
    n_steps = 0
    
    while time.monotonic() - t0 < time_limit:
        n_steps += 1
        T = T_start * (T_end / T_start) ** (n_steps / max(10000, int(time_limit * 100)))
        
        # Generate a perturbation
        perturbation_scale = T * 0.1
        a_perturbed = a + rng.standard_normal(n) * perturbation_scale
        a_perturbed = project_simplex(a_perturbed)
        
        score_new = compute_score(a_perturbed)
        score_curr = compute_score(a)
        
        delta = score_new - score_curr
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            a = a_perturbed
            if score_new < best_score:
                best_score = score_new
                best_a = a.copy()
    
    return best_a, best_score

def random_pairwise(a, n_moves=1000, rng=None):
    """Random pairwise mass transfer."""
    if rng is None:

    best_a = a.copy()
    best_score = compute_score(a)
    
    for _ in range(n_moves):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = rng.uniform(0.001, 0.01) * best_a.max()
        
        for direction in [1, -1]:
            a_trial = a.copy()
            a_trial[i] += direction * delta
            a_trial[j] -= direction * delta
            if a_trial[i] < 0 or a_trial[j] < 0:

            a_trial /= a_trial.sum()
            sc = compute_score(a_trial)
            if sc < best_score - 1e-14:
                a = a_trial
                best_a = a.copy()
                best_score = sc

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

and multi-peak awareness.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = fast_convolve(a)
    return 2.0 * len(a) * b.max() / (s * s)


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    ind = np.arange(1, n + 1)
    cond = u * ind > (cssv - 1)
    if cond.any():
        rho = ind[cond][-1]
        theta = (cssv[rho - 1] - 1.0) / rho
    else:
        return np.ones(n) / n
    return np.maximum(v - theta, 0.0)


def get_top_peaks(b, top_k=10):
    k = min(top_k, len(b))
    return np.argpartition(b, -k)[-k:]


def demand_vector(a, top_k=20):
    """Compute a 'demand' vector: how much each position 'wants' mass.
    Positions with low demand are good targets for redistribution."""
    n = len(a)
    b = fast_convolve(a)
    peaks = get_top_peaks(b, top_k)
    demand = np.zeros(n)
    for k in peaks:
        k = int(k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        demand[idx] += a[k - idx]
    return demand


def targeted_local_search(a0, time_limit, rng, verbose=True):
    """
    Aggressive peak-directed local search with smart redistribution.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a0, compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    iterations = 0
    no_improve = 0
    stuck_count = 0

    while time.monotonic() - t0 < time_limit:
        iterations += 1
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]

        # Get top contributing pairs for the peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]

        # Also consider second and third peaks
        b_sorted = b.copy()
        b_sorted[peak_k] = -np.inf
        peak_k2 = int(np.argmax(b_sorted))
        b_sorted[peak_k2] = -np.inf
        peak_k3 = int(np.argmax(b_sorted))

        improved = False

        # Strategy 1: Reduce top contributors of main peak, redistribute to low-demand positions
        top_ci = np.argsort(contribs)[::-1][:8]
        demand = demand_vector(a, top_k=10)
        min_demand = demand.min()
        max_demand = demand.max()

        for ci in top_ci:
            j = int(idx[ci])
            k_j = peak_k - j
            if a[j] < 1e-12:
                continue

            for frac in [0.05, 0.1, 0.15, 0.2, 0.3]:
                delta = a[j] * frac
                if delta < 1e-14:
                    continue

                # Find best target: low demand, not j or k_j
                target_score = demand.copy()
                target_score[j] = np.inf
                if k_j < n:
                    target_score[k_j] = np.inf
                # Prefer positions with truly low demand
                best_tgt = int(np.argmin(target_score))

                a_trial = a.copy()
                a_trial[j] -= delta
                a_trial[best_tgt] += delta
                if a_trial[j] < 0:
                    continue
                a_trial = np.maximum(a_trial, 0)
                a_trial /= a_trial.sum()
                sc = compute_score(a_trial)
                if sc < best_score - 1e-14:
                    a = a_trial
                    best_a = a.copy()
                    best_score = sc
                    improved = True
                    break

            if improved:
                break

        # Strategy 2: Pair reduction (reduce both elements of a contributing pair)
        if not improved:
            for ci in top_ci[:5]:
                j = int(idx[ci])
                k_j = peak_k - j
                if j == k_j or a[k_j] < 1e-12:
                    continue
                for f in [0.95, 0.9, 0.85, 0.8]:
                    a_trial = a.copy()
                    old_j = a_trial[j]
                    old_k = a_trial[k_j]
                    a_trial[j] = old_j * f
                    a_trial[k_j] = old_k * f
                    freed = (old_j - a_trial[j]) + (old_k - a_trial[k_j])
                    # Redistribute to lowest demand positions
                    target_score = demand.copy()
                    target_score[j] = np.inf
                    target_score[k_j] = np.inf
                    best_tgt = int(np.argmin(target_score))
                    a_trial[best_tgt] += freed
                    a_trial = np.maximum(a_trial, 0)
                    a_trial /= a_trial.sum()
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_a = a.copy()
                        best_score = sc
                        improved = True
                        break
                if improved:
                    break

        # Strategy 3: Reduce contributor of second peak
        if not improved:
            js2 = max(0, peak_k2 - n + 1)
            je2 = min(n - 1, peak_k2)
            idx2 = np.arange(js2, je2 + 1)
            contribs2 = a[idx2] * a[peak_k2 - idx2]
            top_ci2 = np.argsort(contribs2)[::-1][:5]
            for ci in top_ci2:
                j = int(idx2[ci])
                k_j = peak_k2 - j
                if a[j] < 1e-12:
                    continue
                for frac in [0.1, 0.2, 0.3]:
                    delta = a[j] * frac
                    target_score = demand.copy()
                    target_score[j] = np.inf
                    if k_j < n:
                        target_score[k_j] = np.inf
                    best_tgt = int(np.argmin(target_score))
                    a_trial = a.copy()
                    a_trial[j] -= delta
                    a_trial[best_tgt] += delta
                    if a_trial[j] < 0:
                        continue
                    a_trial = np.maximum(a_trial, 0)
                    a_trial /= a_trial.sum()
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_a = a.copy()
                        best_score = sc
                        improved = True
                        break
                if improved:
                    break

        # Strategy 4: Small Gaussian perturbation
        if not improved:
            no_improve += 1
            scale = 0.001 * (1 + no_improve / 100.0)
            a_trial = a + rng.standard_normal(n) * scale
            a_trial = project_simplex(a_trial)
            sc = compute_score(a_trial)
            if sc < best_score - 1e-14:
                a = a_trial
                best_a = a.copy()
                best_score = sc
                improved = True
                no_improve = 0
            elif no_improve > 200:
                # Larger restart
                scale = 0.01
                a_trial = best_a + rng.standard_normal(n) * scale
                a_trial = project_simplex(a_trial)
                sc = compute_score(a_trial)
                if sc < best_score + 0.005:
                    a = a_trial
                    no_improve = 0
                else:
                    a = best_a.copy()
                    no_improve = 0
                    stuck_count += 1
                    if stuck_count > 5:
                        break
            else:
                stuck_count = 0
        else:
            no_improve = 0
            stuck_count = 0

        if verbose and iterations % 5000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    TLS iter={iterations} ({elapsed:.0f}s): best={best_score:.6f}")

    return best_a, best_score


def subgradient_momentum(a0, time_limit, rng, top_k=20, verbose=True):
    """Subgradient method with adaptive learning rate and momentum."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a0, compute_score(a0)
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    v = np.zeros(n)
    mom = 0.93
    lr = 0.003
    t0 = time.monotonic()
    iters = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        iters += 1
        b = fast_convolve(a)
        peaks = get_top_peaks(b, top_k)
        pv = b[peaks]

        # Temperature-weighted
        temp = 0.005 * max(pv.max(), 1e-10)
        w = np.exp((pv - pv.max()) / temp)
        w /= w.sum()

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += w[i] * a[k - idx]

        v = mom * v - lr * sg
        a_new = project_simplex(a + v)
        sc = compute_score(a_new)

        if sc < best_score - 1e-14:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive LR
        if no_improve % 50 == 0:
            lr *= 0.95
        if no_improve > 300:
            # Restart with perturbation
            a = best_a + rng.standard_normal(n) * 0.005
            a = project_simplex(a)
            v = np.zeros(n)
            lr = max(lr * 2, 0.001)
            no_improve = 0

        if verbose and iters % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    SG iter={iters} ({elapsed:.0f}s): best={best_score:.6f} lr={lr:.6f}")

    return best_a, best_score


def multi_peak_reduction(a0, n_rounds=5, rng=None):
    """Try to reduce multiple peaks simultaneously."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0).copy()
    s = a.sum()
    if s < 1e-15:
        return a0
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)

    for rnd in range(n_rounds):
        b = fast_convolve(a)
        peaks = get_top_peaks(b, 5)

        for peak_k in peaks:
            peak_k = int(peak_k)
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            idx = np.arange(js, je + 1)
            contribs = a[idx] * a[peak_k - idx]
            top_ci = np.argsort(contribs)[::-1][:3]

            demand = demand_vector(a, top_k=10)

            for ci in top_ci:
                j = int(idx[ci])
                k_j = peak_k - j
                if a[j] < 1e-12:
                    continue
                for frac in [0.1, 0.15, 0.2]:
                    delta = a[j] * frac
                    target_score = demand.copy()
                    target_score[j] = np.inf
                    if k_j < n:
                        target_score[k_j] = np.inf
                    best_tgt = int(np.argmin(target_score))
                    a_trial = a.copy()
                    a_trial[j] -= delta
                    a_trial[best_tgt] += delta
                    if a_trial[j] < 0:
                        continue
                    a_trial = np.maximum(a_trial, 0)
                    a_trial /= a_trial.sum()
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-14:
                        a = a_trial
                        best_a = a.copy()
                        best_score = sc
                        break
                else:
                    continue
                break

    return best_a


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def structured_initialization(n, rng):
    """Create a structured initial sequence with ripple pattern."""
    a = np.ones(n) / n
    # Add sinusoidal perturbation
    freq = rng.integers(1, 5)
    t = np.linspace(0, freq * 2 * np.pi, n)
    amplitude = rng.uniform(0.05, 0.15)
    a = a * (1 + amplitude * np.cos(t))
    a = np.maximum(a, 0)
    a /= a.sum()
    return a
