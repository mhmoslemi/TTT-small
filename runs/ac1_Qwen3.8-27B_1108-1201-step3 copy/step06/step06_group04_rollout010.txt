                diff = 1.0 - f.sum()
                non_peak = np.ones(n, dtype=bool)
                non_peak[js:je + 1] = False
                if non_peak.sum() > 0:
                    w = f[non_peak]
                    wsum = w.sum()
                    if wsum > 1e-15:
                        f[non_peak] += diff * w / wsum
                f = np.maximum(f, 0)
                f = f / f.sum()
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            continue

        g = np.maximum(g, 0)
        g_sum = g.sum()
        if g_sum < 1e-15:
            m_exp *= 1.1
            continue
        g = g / g_sum

        diff = np.max(np.abs(g - f))
        if diff < 1e-12:
            no_improve += 1
            if no_improve > 20:
                m_exp = min(m_exp * 1.06, 6.0)
            if no_improve > 80:
                rng = np.random.default_rng(it * 31 + 7)
                scale = 0.003 * min(m_exp, 3.0)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)
                M = conv.max()
                m_exp = 1.0
                no_improve = 0
            continue
        else:
            if diff > 1e-5:
                no_improve = 0
            elif diff > 1e-7:
                no_improve += 1
            else:
                no_improve += 3

        for step_i in range(3):
            if time.monotonic() - t0 > time_limit:
                break
            alpha, alpha_val = line_search(f, g, n, max_iter=20)
            if alpha < 1e-8:
                break
            f_new = f + alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            sn = f_new.sum()
            if sn < 1e-15:
                break
            f_new = f_new / sn
            f = f_new
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            current_obj = 2.0 * n * M
            if current_obj < best_obj - 1e-15:
                best_obj = current_obj
                best_f = f.copy()

        if no_improve > 100:
            break

    return best_f, best_obj


def nesterov_subgrad(f0, time_limit, verbose=False):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_obj = 2.0 * n * fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    lr = 0.01
    momentum = 0.9
    v = np.zeros(n)
    no_improve = 0
    f_prev = f.copy()
    temperature = 0.005

    while time.monotonic() - t0 < time_limit:
        y = f + momentum * (f - f_prev)
        y = np.maximum(y, 0)
        ys = y.sum()
        if ys < 1e-15:
            break
        y = y / ys

        conv = fast_convolve(y)
        M = conv.max()

        k_actual = min(30, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = pv / temperature
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            kk = int(peaks[i])
            js = max(0, kk - n + 1)
            je = min(n - 1, kk)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * y[kk - idx]

        v = momentum * v + lr * sg
        f_new = y - v
        f_new = project_simplex(f_new)

        f_prev = f
        f = f_new
        co = 2.0 * n * fast_convolve(f).max()
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9995
        temperature *= 1.005

        if no_improve > 200:
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            f_prev = f.copy()
            v = np.zeros(n)
            lr = max(lr, 0.005)
            no_improve = 0

        if lr < 1e-7:
            break

    return best_f, best_obj


def pair_swap_heuristic(f0, time_limit, verbose=False):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit and no_improve < 60:
        conv = fast_convolve(f)
        M = conv.max()
        k_peak = int(np.argmax(conv))

        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        idx = np.arange(js, je + 1)
        products = f[idx] * f[k_peak - idx]
        total = products.sum()
        if total < 1e-15:
            break

        order = np.argsort(products)[::-1]
        improved = False

        for oi in range(min(6, len(order))):
            j_max = int(idx[order[oi]])
            j_partner = int(k_peak - j_max)
            old_val = f[j_max]
            if old_val < 1e-12:
                continue

            for mult in [0.3, 0.5, 0.7, 0.8, 1.2, 1.4, 1.6]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.5:
                    continue
                d = old_val - new_val
                if abs(d) < 1e-14:
                    continue

                trial = f.copy()
                trial[j_max] = new_val

                non_peak = np.ones(n, dtype=bool)
                non_peak[js:je + 1] = False
                cnt = non_peak.sum()
                if cnt > 0:
                    w = f[non_peak]
                    wsum = w.sum()
                    if wsum > 1e-15:
                        trial[non_peak] += d * w / wsum

                trial = np.maximum(trial, 0)
                tsum = trial.sum()
                if tsum < 1e-15:
                    continue
                trial = trial / tsum
                obj = 2.0 * n * fast_convolve(trial).max()
                if obj < best_obj - 1e-13:
                    best_obj = obj
                    best_f = trial.copy()
                    f = trial
                    improved = True
                    break

            if improved:
                break

        if not improved:
            no_improve += 1
            if no_improve > 15:
                k2 = int(np.argmax(fast_convolve(f)))
                js2 = max(0, k2 - n + 1)
                je2 = min(n - 1, k2)
                trial = f.copy()
                trial[js2:je2 + 1] *= 0.9
                d = f.sum() - trial.sum()
                non_contrib = np.ones(n, dtype=bool)
                non_contrib[js2:je2 + 1] = False
                if non_contrib.sum() > 0:
                    nc_w = f[non_contrib]
                    nw = nc_w.sum()
                    if nw > 1e-15:
                        trial[non_contrib] += d * nc_w / nw
                trial = np.maximum(trial, 0)
                trial = trial / trial.sum()
                obj = 2.0 * n * fast_convolve(trial).max()
                if obj < best_obj:
                    f = trial
                    best_obj = obj
                    best_f = f.copy()
                    no_improve = 0
                else:
                    break
        else:
            no_improve = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian(n, sigma_frac=0.25):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_geometric(n, ratio=0.95):
    a = np.array([ratio ** j for j in range(n)], dtype=np.float64)
    a = a / a.sum()
    return a


def make_power(n, exponent=2.0):
    x = np.arange(n, dtype=np.float64) / n
    a = (1.0 - x) ** exponent
    a = a / a.sum()
    return a


def make_bell(n, width=0.7):
    x = np.arange(n) - n / 2.0
    half_w = n * width / 2.0
    t = np.clip(np.abs(x) / half_w, 0, 1.0)
    a = (1.0 - t) ** 3
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def try_sequence(a, best_seq, best_score, label=""):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    rng = np.random.default_rng(seed)

    best_seq = None

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)

        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_gaussian(768, 0.25)
        best_seq, best_score = try_sequence(parent, best_seq, best_score, "default")

    n_parent = len(parent)

        init_candidates.append(make_geometric(n_try, 0.98))
        init_candidates.append(make_power(n_try, 1.5))
        init_candidates.append(make_power(n_try, 2.5))

        init_candidates.append(make_bell(n_try, 0.7))
        init_candidates.append(make_bell(n_try, 0.9))

    best_init = None
    best_init_obj = float('inf')
    for ic in init_candidates:
        s = ic.sum()
        if s < 1e-10:
            continue
        a_norm = ic / s
        obj = obj_normalized(a_norm)
        if obj < best_init_obj:
            best_init_obj = obj
            best_init = a_norm.copy()
        best_seq, best_score = try_sequence(a_norm, best_seq, best_score,
                                            f"init_n{len(a_norm)}")

    print(f"  Best init: obj={best_init_obj:.6f} (n={len(best_init)})")

    # Phase 2: LP-FW on best init
    print(f"\n=== Phase 2: LP-FW (n={len(best_init)}) ===")
    rem = deadline - time.monotonic() - 600
    if rem > 50:
        t_alloc = min(rem, 250)
        print(f"  LP-FW: t={t_alloc:.0f}s")

        best_seq, best_score = try_sequence(a_opt, best_seq, best_score, "lpfw_main")

    # Phase 3: Nesterov subgradient
    if best_seq is not None:
        print(f"\n=== Phase 3: Nesterov Subgrad (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 400

            t_alloc = min(rem, 120)

    # Phase 4: Pair swap

        print(f"\n=== Phase 4: Pair Swap (n={len(best_seq)}) ===")
        rem = deadline - time.monotonic() - 300

    if best_seq is not None:
        print(f"\n=== Phase 5: LP-FW Refine (n={len(best_seq)}) ===")

    # Phase 6: Multi-length exploration
    base = best_seq if best_seq is not None else best_init
    print(f"\n=== Phase 6: Multi-length Exploration ===")
    for target_n in [640, 700, 768, 832, 900, 1024]:
        if len(base) == target_n:

    if best_seq is not None:

    if best_seq is None:

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
