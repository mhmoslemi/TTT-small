Let me also think about the "culling" idea more carefully:

**Culling**: Select a subset S of indices to set to zero. The remaining indices are optimized.

The benefit: reducing the support can reduce peaks that are caused by interactions between the culled indices.
The cost: we lose the ability to distribute mass over those indices.

When is culling beneficial?
- If the culled indices are "dynamically important" - they contribute to multiple peaks.
- If the remaining indices can compensate (i.e., the peaks they create are lower than the peaks we're eliminating).

A simple culling strategy:
1. Find the worst peak k*.
2. Identify the indices j that contribute to peak k*: these are j and k*-j.
3. For each such j, compute its "total peak contribution": sum over all peaks k' of f[j]*f[k'-j].
4. Cull the index j with the highest "total peak contribution" (it's the most problematic).
5. Redistribute its mass to nearby indices that have low peak contribution.
6. Repeat.

This is a greedy algorithm that should work well in practice.

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using hybrid optimization:
LP-Frank-Wolfe, multiplicative peeling, targeted culling, and coordinate descent.
"""

import numpy as np
from scipy.optimize import linprog
import time

# --- Utility functions ---

def fast_convolve(a):
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def obj_norm(f):
    """Objective for sum-normalized sequence: 2*n*max(conv)."""
    n = len(f)
    return 2.0 * n * fast_convolve(f).max()

def normalize(f):
    s = f.sum()
    if s < 1e-15:
        return np.ones(len(f)) / len(f)
    return np.maximum(f, 0) / s

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)

def build_lp_matrix(f, n):
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2 * n - 2)
        length = k_end - k_start + 1
        A[k_start:k_end + 1, j] = f[:length]
    return A

def solve_lp(f, M, n):
    A = build_lp_matrix(f, n)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def multiplicative_peel(f, n, K=20, eta=0.1):
    """Multiplicative update targeting top-K peaks."""
    conv = fast_convolve(f)
    if len(conv) <= K:
        peaks = np.argsort(conv)[::-1]
    else:
        peaks = np.argpartition(conv, -K)[-K:]
    
    M = conv.max()
    # Weights: exponential decay from peak value
    pv = conv[peaks]
    w = np.exp(-(M - pv) / (0.01 * M + 1e-15))
    w = w / w.sum()
    
    # Compute cost for each index
    cost = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-10:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        cost[idx] += w[i] * f[k - idx]
    
    # Multiplicative update
    f_new = f * np.exp(-eta * cost)
    f_new = np.maximum(f_new, 1e-15)
    return normalize(f_new)

def line_search_convex(f, g, n, max_evals=30):
    """Line search: find t in [0,1] minimizing obj of normalize(f + t*(g-f))."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c_pt = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    
    def eval_t(t):
        ft = f + t * (g - f)
        ft = np.maximum(ft, 0)
        s = ft.sum()
        if s < 1e-15:
            return obj_norm(np.ones(n) / n)
        return obj_norm(ft / s)
    
    c_val = eval_t(c_pt)
    d_val = eval_t(d_pt)
    best_t = c_pt if c_val < d_val else d_pt
    best_v = min(c_val, d_val)
    
    for _ in range(max_evals):
        if hi - lo < 1e-8:
            break
        if c_val < d_val:
            hi = d_pt
            d_pt = c_pt
            d_val = c_val
            c_pt = hi - gr * (hi - lo)
            c_val = eval_t(c_pt)
        else:
            lo = c_pt
            c_pt = d_pt
            c_val = d_val
            d_pt = lo + gr * (hi - lo)
            d_val = eval_t(d_pt)
        
        mid = (lo + hi) / 2.0
        mid_v = eval_t(mid)
        if mid_v < best_v:
            best_v = mid_v
            best_t = mid
    
    return best_t, best_v

    f = normalize(f0.astype(np.float64))

def cull_optimize(f0, time_limit, max_culs=5, verbose=False):
    """Try culling (zeroing) indices that cause peaks."""
    f = normalize(f0.astype(np.float64))
    n = len(f)
    best_obj = obj_norm(f)
    best_f = f.copy()
    
    t0 = time.monotonic()
    
    for cul in range(max_culs):
        if time.monotonic() - t0 > time_limit:
            break
        
        conv = fast_convolve(f)
        k_star = int(np.argmax(conv))
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        prods = f[idx] * f[k_star - idx]
        
        # Find the worst contributor
        worst = int(idx[np.argmax(prods)])
        if f[worst] < 1e-8:
            break
        
        # Cull this index
        trial = f.copy()
        freed = trial[worst]
        trial[worst] = 0.0
        
        # Redistribute to non-peak indices
        non_peak = np.ones(n, dtype=bool)
        non_peak[js:je+1] = False
        non_peak[worst] = False
        if non_peak.sum() > 0:
            w = f[non_peak]
            ws = w.sum()
            if ws > 1e-15:
                trial[non_peak] += freed * w / ws
        
        trial = np.maximum(trial, 0)
        trial = trial / trial.sum()
        o = obj_norm(trial)
        
        if o < best_obj - 1e-10:
            best_obj = o
            best_f = trial.copy()
            f = trial
            if verbose:
                print(f"    Cull {cul+1}: zeroed idx {worst}, obj={best_obj:.8f}")
        else:
            break
    
    return best_f, best_obj

def interpolate_seq(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_new = np.interp(new_x, old_x, f)
    return normalize(f_new)

def make_gaussian(n, sigma=0.25):
    x = np.arange(n) - n / 2.0
    a = np.exp(-0.5 * (x / (n * sigma)) ** 2)
    return normalize(a)

def make_smooth_noise(n, seed=0):
    rng = np.random.default_rng(seed)
    a = np.abs(rng.standard_normal(n)) + 0.5
    # Smooth it
    kernel = np.ones(5) / 5
    a = np.convolve(a, kernel, mode='same')
    return normalize(a)
