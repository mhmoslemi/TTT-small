def project_simplex(v, target_sum=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)] * 1 * target_sum)
    v = np.asarray(v, dtype=np.float64)
    s = v.sum()
    if s < 1e-15:

    v = v / s  # normalize to sum 1

    N = 1 << (n + m - 1 - 1).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(weights, N)
    result = np.fft.irfft(W * np.conj(A), N)
    return result[:n]


def single_peak_subgradient(a, k_star):
    """Compute subgradient for a single peak at k_star."""
    n = len(a)
    sg = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(n - 1, k_star)
    for j in range(js, je + 1):
        sg[j] = a[k_star - j]
    return sg


def make_gaussian(n, sigma_frac=0.25):
    """Gaussian initial sequence normalized to sum 1."""
    x = np.arange(n) - (n - 1) / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a /= s
    return a


def make_flat(n):
    """Uniform sequence."""
    return np.ones(n) / n


def make_randn(n, rng):
    """Random non-negative sequence."""
    a = np.abs(rng.standard_normal(n))
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return a / s


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def objective_val(a):
    """Compute 2*n*max(conv) for a sum-normalized sequence."""
    n = len(a)
    conv = fast_convolve(a)
    return 2.0 * n * conv.max()


def mirror_descent_optimize(f0, time_limit, verbose=False, momentum=0.9, 
                            init_alpha=0.5, temp_init=0.01, top_k=10):
    """
    Mirror descent with softmax-weighted subgradients, momentum, and adaptive step.
    """
    f = np.maximum(np.asarray(f0, dtype=np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    obj = 2.0 * n * conv.max()
    best_obj = obj
    best_f = f.copy()

    t0 = time.monotonic()
    alpha = init_alpha
    temp = temp_init
    v = np.zeros(n)  # momentum
    no_improve = 0
    it = 0
    best_temp = temp_init
    alpha_decay = 0.99995
    temp_growth = 1.002

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        # Compute convolution
        conv = fast_convolve(f)
        obj = 2.0 * n * conv.max()

        # Compute weighted subgradient
        if top_k is not None and top_k < len(conv):
            # Use top-k peaks with softmax weights
            peak_idx = np.argpartition(conv, -top_k)[-top_k:]
            peak_vals = conv[peak_idx]
            w_peaks = safe_softmax(peak_vals, temp)
            weights = np.zeros(len(conv))
            weights[peak_idx] = w_peaks
        else:
            weights = safe_softmax(conv, temp)

        sg = weighted_subgradient_fft(f, weights)

        # Momentum
        v = momentum * v + (1.0 - momentum) * sg

        # Mirror descent step with adaptive alpha
        improved = False
        for attempt in range(5):
            f_new = f * np.exp(-alpha * v)
            f_new = np.maximum(f_new, 1e-30)
            f_new = f_new / f_new.sum()
            
            new_conv = fast_convolve(f_new)
            new_obj = 2.0 * n * new_conv.max()
            
            if new_obj < obj:
                f = f_new
                obj = new_obj
                improved = True
                if obj < best_obj:
                    best_obj = obj
                    best_f = f.copy()
                    no_improve = 0
                break
            else:
                alpha *= 0.5
        
        if not improved:
            no_improve += 1
            # Restart from best if stuck
            if no_improve > 50:
                f = best_f.copy()
                v = np.zeros(n)
                alpha = max(init_alpha * 0.5, 0.01)
                temp = temp_init
                no_improve = 0
            elif no_improve > 20:
                # Add small perturbation
                rng = np.random.default_rng(it * 7 + 3)
                f = f + rng.standard_normal(n) * 0.001
                f = np.maximum(f, 0)
                f = f / f.sum()
                v = np.zeros(n)

        # Adapt temperature and alpha
        alpha *= alpha_decay
        temp *= temp_growth

        if verbose and it % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD iter {it}: obj={best_obj:.10f}, alpha={alpha:.6f}, "
                  f"temp={temp:.4f}, no_improve={no_improve}, t={elapsed:.0f}s")

        if alpha < 1e-8:
            # Reset alpha
            alpha = init_alpha * 0.3
            v = np.zeros(n)

    return best_f, best_obj


def lp_refine(f0, time_limit, verbose=False, n_constraints=None):
    """
    LP-based refinement: at each step, solve an LP to find a direction
    that maintains the convolution bound while maximizing the sum.
    Uses only the top constraints for speed.
    """
    f = np.maximum(np.asarray(f0, dtype=np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    # Number of constraints to use (top peaks)
    if n_constraints is None:
        n_constraints = min(2 * n - 1, max(50, n // 2))

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        M = conv.max()

        # Select top constraints
        top_idx = np.argsort(conv)[-n_constraints:]
        
        # Build LP: max sum(g) s.t. (f*g)[k] <= M for k in top_idx, g >= 0
        # (f*g)[k] = sum_j f[j] * g[k-j] for valid j
        A_ub = np.zeros((len(top_idx), n))
        for ii, k in enumerate(top_idx):
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                A_ub[ii, j] = f[k - j]

        c = -np.ones(n)
        b_ub = np.full(len(top_idx), M)
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            no_improve += 1
            if no_improve > 20:
                break
            continue

        g = res.x
        g_sum = g.sum()
        if g_sum < 1e-15:
            no_improve += 1
            continue
        g = g / g_sum

        # Check if g is different from f
        diff = np.max(np.abs(g - f))
        if diff < 1e-10:
            no_improve += 1
            if no_improve > 30:
                break
            continue

        # Line search: find best alpha in [0,1] for f + alpha*(g-f)
        best_alpha = 0.0
        best_alpha_obj = 2.0 * n * M
        for alpha_try in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            f_try = f + alpha_try * (g - f)
            f_try = np.maximum(f_try, 0)
            st = f_try.sum()
            if st < 1e-15:
                continue
            f_try = f_try / st
            obj_try = 2.0 * n * fast_convolve(f_try).max()
            if obj_try < best_alpha_obj:
                best_alpha_obj = obj_try
                best_alpha = alpha_try

        if best_alpha < 1e-8:
            no_improve += 1
            if no_improve > 30:
                break
            continue

        f = f + best_alpha * (g - f)
        f = np.maximum(f, 0)
        f = f / f.sum()

        obj = 2.0 * n * fast_convolve(f).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 40:
                break

        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LP iter {it}: obj={best_obj:.10f}, t={elapsed:.0f}s")

    return best_f, best_obj


def coordinate_polish(f0, time_limit, verbose=False):
    """
    Coordinate descent polish: for each peak in the convolution,
    try to reduce it by adjusting the contributing elements.
    """
    f = np.maximum(np.asarray(f0, dtype=np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = 2.0 * n * conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit and no_improve < 100:
        it += 1
        conv = fast_convolve(f)
        obj = 2.0 * n * conv.max()
        improved = False

        # Focus on top peaks
        n_peaks = min(8, len(conv))
        peak_idx = np.argsort(conv)[-n_peaks:]

        for k in peak_idx:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue

            idx = np.arange(js, je + 1)
            products = f[idx] * f[k - idx]
            total = products.sum()
            if total < 1e-15:
                continue

            # Try reducing the top 3 contributors
            order = np.argsort(products)[::-1]
            for oi in range(min(3, len(order))):
                j_max = int(idx[order[oi]])
                old_val = f[j_max]
                if old_val < 1e-12:
                    continue

                for mult in [0.7, 0.8, 0.9, 1.2]:
                    new_val = old_val * mult
                    trial = f.copy()
                    trial[j_max] = new_val
                    d = old_val - new_val
                    if abs(d) > 1e-15:
                        # Redistribute
                        mask = np.ones(n, dtype=bool)
                        mask[js:je + 1] = False
                        if mask.sum() > 0:
                            w = f[mask]
                            wsum = w.sum()
                            if wsum > 1e-15:
                                trial[mask] += d * w / wsum
                    trial = np.maximum(trial, 0)
                    tsum = trial.sum()
                    if tsum < 1e-15:
                        continue
                    trial = trial / tsum
                    obj_try = 2.0 * n * fast_convolve(trial).max()
                    if obj_try < best_obj - 1e-15:
                        best_obj = obj_try
                        best_f = trial.copy()
                        f = trial
                        improved = True

        if not improved:
            no_improve += 1
        else:
            no_improve = 0
            if verbose and it % 10 == 0:
                print(f"    CD iter {it}: obj={best_obj:.10f}")

    return best_f, best_f_obj(best_obj) if False else (best_f, best_obj)


def try_sequence(a, best_seq, best_score, label=""):
    """Evaluate a candidate sequence, return updated best."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score
