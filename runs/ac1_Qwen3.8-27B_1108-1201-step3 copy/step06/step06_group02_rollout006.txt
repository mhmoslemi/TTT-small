Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a smooth-max subgradient method

import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho_arr = np.nonzero(cond)[0]
    if len(rho_arr) == 0:
        return np.ones(n) / n
    rho = rho_arr[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_gradient(p, conv, T, top_k=50):
    """Compute the gradient of the softmax-smoothed max of conv."""
    n = len(p)
    scaled = conv / max(T, 1e-10)
    scaled -= scaled.max()
    weights = np.exp(scaled)
    weights /= weights.sum()
    
    # Select top peaks
    if top_k < len(weights):
        top_idx = np.argsort(weights)[-top_k:]
        w_top = weights[top_idx]
        w_sum = w_top.sum()
        if w_sum > 1e-15:
            w_top /= w_sum
        else:
            w_top = np.ones(len(top_idx)) / len(top_idx)
    else:
        top_idx = np.arange(len(weights))
        w_top = weights
    
    grad = np.zeros(n)
    for i in range(len(top_idx)):
        k = int(top_idx[i])
        w = w_top[i]
        if w < 1e-12:
            continue
        # Term 1: p[k-j] for valid j
        lo1 = max(0, k - n + 1)
        hi1 = min(n - 1, k)
        idx1 = np.arange(lo1, hi1 + 1)
        grad[idx1] += w * p[k - idx1]
        # Term 2: p[j-k] for valid j
        lo2 = max(0, k)
        hi2 = min(n - 1, k + n - 1)
        idx2 = np.arange(lo2, hi2 + 1)
        grad[idx2] += w * p[idx2 - k]
    
    return grad

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    s = f_interp.sum()
    if s > 1e-15:
        f_interp /= s
    return f_interp
