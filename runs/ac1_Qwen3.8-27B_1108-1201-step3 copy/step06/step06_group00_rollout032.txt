"""

exact active-set handling, followed by L-BFGS and coordinate descent polishing.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build the convolution matrix C where C[k,j] = a[k-j] if valid."""
    n = len(a)
    n_conv = 2 * n - 1
    C = np.zeros((n_conv, n))
    for j in range(n):
        C[j:j+n, j] = a
    return C


def lp_optimization_step(a, use_all_constraints=True, n_top=100):
    """
    One LP-based optimization step.
    Returns (direction d, estimated new M).
    """
    n = len(a)
    conv = fast_convolve(a)
    n_conv = 2 * n - 1
    
    C = build_conv_matrix(a)
    
    if use_all_constraints:
        A_ub = np.hstack([C, -np.ones((n_conv, 1))])
        b_ub = -conv.copy()
    else:
        # Only use top-k peaks
        top_idx = np.argpartition(conv, -n_top)[-n_top:]
        A_ub = np.hstack([C[top_idx], -np.ones((n_top, 1))])
        b_ub = -conv[top_idx].copy()
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    m_upper = conv.max() + 1e-10
    bounds = [(-a[i], None) for i in range(n)] + [(0, m_upper)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n].copy(), float(res.x[-1])
    except Exception as e:
        print(f"  LP error: {e}")
    
    return None, None


def lp_optimize(a0, time_limit, use_all=True, n_top=100, tol=1e-10,

    LP-based optimization loop.

    rng = np.random.default_rng(12345)
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    S = a.sum()
    a = a / S
    n = len(a)
    
    cur_conv = fast_convolve(a)
    cur_max = cur_conv.max()
    best_a = a.copy()
    best_max = cur_max
    stagnant = 0
    
    while time.monotonic() - t0 < time_limit:
        d, M_est = lp_optimization_step(a, use_all=use_all, n_top=n_top)
        
        if d is None:
            break
        
        # Check if improvement is significant
        expected_improve = cur_max - M_est
        if expected_improve < tol:
            stagnant += 1
        else:
            stagnant = 0
        
        # Try step with line search
        improved = False
        for alpha in [1.0, 0.5, 0.25, 0.1, 0.05]:
            a_trial = a + alpha * d
            if a_trial.min() < -1e-12:
                continue
            a_trial = np.maximum(a_trial, 0)
            # renormalize (shouldn't be needed if sum(d)=0, but just in case)
            s = a_trial.sum()
            if s < 1e-10:
                continue
            a_trial = a_trial / s
            
            trial_conv = fast_convolve(a_trial)
            trial_max = trial_conv.max()
            
            if trial_max < cur_max - 1e-14:
                a = a_trial
                cur_conv = trial_conv
                cur_max = trial_max
                improved = True
                break
        
        if improved:
            if cur_max < best_max - 1e-14:
                best_max = cur_max
                best_a = a.copy()
        else:
            stagnant += 1
        
        if stagnant > max_stagnant:
            # Perturbation
            a = best_a + rng.standard_normal(n) * perturb_size
            a = np.maximum(a, 0)
            a = a / a.sum()
            cur_conv = fast_convolve(a)
            cur_max = cur_conv.max()
            perturb_size *= 0.5
            stagnant = 0
    
    return best_a, best_max


def softmax_from_z(z):
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    return np.log(np.maximum(a, 1e-300))


def soft_max_obj_grad(z, T, n):
    """Soft-max smoothed objective and gradient in log-space."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    log_sum_exp = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * log_sum_exp
    w = exp_Tb / exp_Tb.sum()
    
    # Gradient of max_soft(conv) w.r.t. a
    # grad_a[i] = sum_k w[k] * a[k-i] (for valid k-i)
    n_a = len(a)

    for k in range(len(b)):
        if w[k] < 1e-14:
            continue
        lo = max(0, k - n_a + 1)
        hi = min(n_a - 1, k)
        grad_a[lo:hi+1] += w[k] * a[k:lo-1:-1] if hi >= lo else 0
    
    # Hmm, this loop is slow. Let me use the cross-correlation approach.
    # Actually, grad_a = cross_correlation(w, a) where:
    # grad_a[i] = sum_k w[k] * a[k-i] for 0 <= k-i < n

    # i.e., grad_a[i] = sum_{j} w[i+j] * a[j] where j = k-i, so k = i+j

    # This is a "cross-correlation" which can be computed via FFT
    size = 2 * n_a - 1
    fft_size = 1 << (size - 1).bit_length()
    W = rfft(w, fft_size)
    A = rfft(a, fft_size)
    # Cross-correlation: corr[i] = sum_j w[i+j] * a[j]
    # In FFT: this is the inverse FFT of conj(W) * A, but shifted
    # Actually, let me think more carefully.
    # w has length 2n-1, a has length n.
    # We want: grad_a[i] = sum_{j=0}^{n-1} w[i+j] * a[j] for i=0,...,n-1
    # This is a "partial cross-correlation" or "cross-correlation at lags 0 to n-1"
    
    # Using FFT: let W_full = fft(w, fft_size), A_full = fft(a, fft_size)
    # The linear cross-correlation is ifft(conj(W_full) * A_full)
    # corr[m] = sum_j w[j] * a[j+m] (for m >= 0) ... hmm, need to be careful with conventions
    
    # Let me just use the direct formula for now (it's O(n^2) but n is at most a few thousand)
    # Actually for n=1152, O(n^2) = 1.3M operations, which is fine.
    
    # Redo: grad_a[i] = sum_{k=i}^{i+n-1} w[k] * a[k-i] (for valid k)
    # Let j = k-i: grad_a[i] = sum_{j=0}^{n-1} w[i+j] * a[j] (for 0 <= i+j < 2n-1)
    
    # This is the same as: grad_a = ifft(fft(w) * conj(fft(reversed_a)))  ... no
    # Let me just do it directly with numpy:
    grad_a = np.zeros(n_a)
    for i in range(n_a):
        # grad_a[i] = sum_j w[i+j] * a[j] for j=0..n-1 where i+j < 2n-1
        max_j = min(n_a, 2*n_a - 1 - i)
        if max_j > 0:
            grad_a[i] = np.dot(w[i:i+max_j], a[:max_j])
    
    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    return obj, grad_z


def lbgfs_refine(a0, T, time_limit, maxiter=5000):
    """L-BFGS in log-space with soft-max objective."""
    t0 = time.monotonic()
    n = len(a0)
    z0 = z_from_a(a0)
    
    def func(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        obj, _ = soft_max_obj_grad(z, T, n)
        return obj
    
    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        _, g = soft_max_obj_grad(z, T, n)
        return g
    
    try:
        res = minimize(
            fun=func, x0=z0, jac=jac, method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': maxiter * 20, 'disp': False}
        )
        final_obj = func(res.x)
        if np.isfinite(final_obj):
            a_out = softmax_from_z(res.x)
            return a_out
    except Exception:
        pass
    
    return a0


def coordinate_polish(a0, time_limit):
    """Coordinate descent polish."""
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    
    cur_conv = fast_convolve(a)
    best_obj = cur_conv.max()
    
    multipliers = np.array([0.5, 0.7, 0.85, 1.15, 1.3, 1.5])
    
    improved = True
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a
            old_val = a[j]
            if old_val < 1e-14:
                continue
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
    return best_a


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using LP-based optimization with
exact active-set handling, L-BFGS soft-max refinement, and coordinate descent.

Key improvements over previous subgradient approach:
- LP-based direction finding solves the exact linearized Chebyshev subproblem
- No learning rate tuning or momentum needed
- Handles all active constraints simultaneously via the LP

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize, linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C where C[k,j] = a[k-j] if 0<=k-j<n, else 0."""
    n = len(a)
    n_conv = 2 * n - 1
    C = np.zeros((n_conv, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def lp_optimization_step(a, n_top=None):
    """
    Solve LP to find optimal descent direction.
    Returns (direction d, estimated new max M) or (None, None).
    """
    n = len(a)
    conv = fast_convolve(a)
    n_conv = 2 * n - 1

    C = build_conv_matrix(a)

    # Include all constraints (2n-1) for correctness
    A_ub = np.hstack([C, -np.ones((n_conv, 1))])
    b_ub = -conv.copy()

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    m_upper = conv.max() + 1e-10
    bounds = [(-a[i], None) for i in range(n)] + [(0, m_upper)]

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[:n].astype(np.float64).copy(), float(res.x[-1])
    except Exception:
        pass

    return None, None


def lp_optimize(a0, time_limit, tol=1e-10, max_stagnant=300, seed=12345):

    """

    t0 = time.monotonic()

    S = a.sum()
    a = a / S
    n = len(a)

    cur_conv = fast_convolve(a)
    cur_max = cur_conv.max()
    best_a = a.copy()
    best_max = cur_max
    stagnant = 0
    perturb_size = 0.01
    step = 0

    while time.monotonic() - t0 < time_limit:
        d, M_est = lp_optimization_step(a)
        step += 1

        if d is None:

        if expected_improve < tol:

        else:

            if s < 1e-10:

def project_simplex(v):
    """Euclidean projection onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_from_z(z):
    """Convert z to probability vector via softmax."""
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    s = exp_z.sum()
    if s < 1e-300:
        return np.ones(len(z)) / len(z)
    return exp_z / s


def z_from_a(a):
    """Convert probability vector to log-domain."""
    return np.log(np.maximum(a, 1e-300))


def soft_max_obj_grad(z, T, n):
    """Soft-max smoothed objective and gradient in log-space."""
    a = softmax_from_z(z)
    b = fast_convolve(a)
    nb = len(b)
    Tb = T * b
    max_Tb = Tb.max()
    exp_Tb = np.exp(Tb - max_Tb)
    lse = max_Tb + np.log(exp_Tb.sum())
    obj = (2.0 * n / T) * lse
    w = exp_Tb / exp_Tb.sum()

    # grad_a[i] = sum_j w[i+j] * a[j] for i=0..n-1
    grad_a = np.zeros(n)
    for j in range(n):
        grad_a += a[j] * w[j:j + n]

    S = np.dot(grad_a, a)
    grad_z = a * (grad_a - a * S)
    return obj, grad_z

        return obj

    def jac(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros_like(z)
        _, g = soft_max_obj_grad(z, T, n)
        return g

    try:
        res = minimize(
            fun=func, x0=z0, jac=jac, method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                     'maxfun': maxiter * 20, 'disp': False}
        )
        a_out = softmax_from_z(res.x)
        cur_conv = fast_convolve(a_out)
        if cur_conv.max() < fast_convolve(a0).max():
            return a_out
    except Exception:
        pass
    return a0.copy()


def coordinate_polish(a0, time_limit):
    """Coordinate descent polish on individual elements."""
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)

    cur_conv = fast_convolve(a)
    best_obj = cur_conv.max()
    best_a = a.copy()

    multipliers = np.array([0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0])

    improved = True
    while improved and time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a
            old_val = a[j]
            if old_val < 1e-14:
                continue
            best_j_obj = best_obj
            best_j_val = old_val

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.9999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
    return best_a


def subgradient_optimize(a0, time_limit, lr_init=0.003, momentum=0.9,
                         temp_init=0.0005, kick_interval=1000,
                         kick_size=0.003, seed=42):
    """Fallback subgradient method with Nesterov momentum."""
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    v = np.zeros(n)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    cur_lr = lr_init
    cur_temp = temp_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        b = fast_convolve(a)
        b_max = b.max()

        scaled = b / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-30:
            w = np.zeros(len(b))
            w[b.argmax()] = 1.0
        else:
            w /= w_sum

        n_top = max(20, n // 10)
        top_idx = np.argpartition(b, -n_top)[-n_top:]
        w_sel = w[top_idx]
        w_sel /= w_sel.sum()
        w_full = np.zeros(len(b))
        w_full[top_idx] = w_sel
        w = w_full

        g = np.zeros(n)
        for k in range(len(b)):
            if w[k] < 1e-12:
                continue
            lo = max(0, k - n + 1)
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] += w[k] * a[k - idx]

        v = momentum * v - cur_lr * g
        a_new = project_simplex(a + v)

        b_new = fast_convolve(a_new)
        obj_new = b_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        if no_improve > kick_interval:
            a = project_simplex(best_a + rng.standard_normal(n) * kick_size)
            v = np.zeros(n)
            cur_lr = max(cur_lr * 0.7, lr_init * 0.1)
            no_improve = 0
        elif no_improve > kick_interval // 2:
            cur_lr *= 0.95

        cur_temp *= 1.02
        if cur_temp > 0.5:
            cur_temp = 0.5

    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
