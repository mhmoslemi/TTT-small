```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Fast mirror descent with log-sum-exp smoothed gradients (O(n log n) per iteration)
- Nesterov-style momentum in log-domain
- LP-Frank-Wolfe for final polishing
- Multi-length exploration with adaptive time allocation
- Annealed temperature for the smoothing parameter
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_smoothed_gradient(a, tau, top_k=None):
    """
    Compute gradient of (1/tau)*log(sum_k exp(tau * L_k)) where L_k = conv(a,a)[k].
    Returns gradient vector of length n.
    """
    n = len(a)
    conv = fast_convolve(a)
    m = len(conv)  # 2n-1
    
    if top_k is not None and top_k < m:
        idx = np.argpartition(conv, -top_k)[-top_k:]
        vals = conv[idx]
    else:
        vals = conv
        idx = np.arange(m)
    
    max_val = vals.max()
    exp_vals = np.exp(tau * (vals - max_val))
    wsum = exp_vals.sum()
    if wsum < 1e-300:
        return np.zeros(n), conv.max()
    weights = exp_vals / wsum
    
    # Build full weight vector
    w_full = np.zeros(m)
    w_full[idx] = weights
    
    # Compute gradient: grad[i] = 2 * sum_j a[j] * w_full[i+j]
    # = 2 * convolve(a[::-1], w_full)[n-1 : 2n-1]
    a_rev = a[::-1]
    conv_len = n + m - 1  # 3n-2
    fft_size = 1 << (conv_len - 1).bit_length()
    A_rev = np.fft.rfft(a_rev, fft_size)
    W = np.fft.rfft(w_full, fft_size)
    conv_result = np.fft.irfft(A_rev * W, fft_size)[:conv_len]
    grad = 2.0 * conv_result[n-1 : 2*n-1]
    
    return grad, conv.max()


def mirror_descent(a_init, time_limit, tau_start=3.0, tau_end=50.0,
                   eta_init=0.3, top_k=50, momentum_init=0.0, momentum_end=0.7,
                   verbose=False):
    """Mirror descent with annealed temperature and momentum."""
    a = a_init.astype(np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    a_prev = a.copy()
    v = np.zeros(n)  # momentum
    
    t0 = time.monotonic()
    best_a = a.copy()
    best_val = fast_convolve(a).max()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        
        # Anneal parameters
        tau = tau_start * (tau_end / tau_start) ** progress
        eta = eta_init * (1.0 - 0.5 * progress)
        momentum = momentum_init + (momentum_end - momentum_init) * progress
        
        # Compute gradient
        grad, current_max = compute_smoothed_gradient(a, tau, top_k)
        
        # Normalize gradient
        gnorm = np.mean(np.abs(grad))
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        
        # Momentum update (in log-domain)
        v = momentum * v + (1.0 - momentum) * grad
        
        # Multiplicative update
        log_update = -eta * v
        # Clamp to prevent underflow/overflow
        log_update = np.clip(log_update, -5.0, 5.0)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()
        
        # Evaluate
        current_val = fast_convolve(a).max()
        if current_val < best_val - 1e-12:
            best_val = current_val
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"      MD iter {it}: obj={best_val:.10f} (tau={tau:.1f})")
        else:
            no_improve += 1
        
        # Restart if stuck
        if no_improve > 500:
            rng = np.random.default_rng(it * 7919 + 3)
            a = a * (1.0 + 0.02 * rng.standard_normal(n))
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            v = np.zeros(n)
            no_improve = 0
            if verbose:
                print(f"      MD restart at iter {it}")
        
        a_prev = a.copy()
    
    return best_a, best_val


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_cosine_mod(n, freq, amp):
    x = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * x / n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_multi_cosine(n, n_freqs=3, max_amp=0.3, seed=0):
    rng = np.random.default_rng(seed)
    a = np.ones(n)
    for m in range(1, n_freqs + 1):
        amp = rng.uniform(0.05, max_amp)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2.0 * np.pi * m * np.arange(n) / n + phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
- Fast mirror descent with log-sum-exp smoothed gradients (O(n log n) per iteration)
- Nesterov-style momentum in log-domain with annealed temperature
- LP-Frank-Wolfe for final polishing
- Multi-length exploration with adaptive time allocation
- Coordinate descent refinement on peak-contributing coordinates
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_smoothed_gradient(a, tau, top_k=None):
    """
    Compute gradient of (1/tau)*log(sum_k exp(tau * L_k)) where L_k = conv(a,a)[k].
    Returns (gradient vector of length n, current max convolution value).
    """
    n = len(a)
    conv = fast_convolve(a)
    m = len(conv)  # 2n-1

    if top_k is not None and top_k < m:
        idx = np.argpartition(conv, -top_k)[-top_k:]
        vals = conv[idx]
    else:
        vals = conv
        idx = np.arange(m)

    max_val = vals.max()
    exp_vals = np.exp(tau * (vals - max_val))
    wsum = exp_vals.sum()
    if wsum < 1e-300:
        return np.zeros(n), conv.max()
    weights = exp_vals / wsum

    # Build full weight vector
    w_full = np.zeros(m)
    w_full[idx] = weights

    # Compute gradient: grad[i] = 2 * sum_j a[j] * w_full[i+j]
    # = 2 * convolve(a[::-1], w_full)[n-1 : 2n-1]
    a_rev = a[::-1]
    conv_len = n + m - 1  # 3n-2
    fft_size = 1 << (conv_len - 1).bit_length()
    A_rev = np.fft.rfft(a_rev, fft_size)
    W = np.fft.rfft(w_full, fft_size)
    conv_result = np.fft.irfft(A_rev * W, fft_size)[:conv_len]
    grad = 2.0 * conv_result[n - 1: 2 * n - 1]

    return grad, conv.max()


def mirror_descent(a_init, time_limit, tau_start=3.0, tau_end=80.0,
                   eta_init=0.3, top_k=30, momentum_init=0.0, momentum_end=0.6,
                   verbose=False):
    """Mirror descent with annealed temperature, momentum, and restarts."""
    a = a_init.astype(np.float64).copy()
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    v = np.zeros(n)  # momentum (in log-domain)

    t0 = time.monotonic()
    best_a = a.copy()
    best_val = fast_convolve(a).max()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)

        # Anneal parameters
        tau = tau_start * (tau_end / tau_start) ** progress
        eta = eta_init * (1.0 - 0.4 * progress)
        momentum = momentum_init + (momentum_end - momentum_init) * progress

        # Compute gradient
        grad, current_max = compute_smoothed_gradient(a, tau, top_k)

        # Normalize gradient
        gnorm = np.mean(np.abs(grad))
        if gnorm < 1e-15:
            break
        grad = grad / gnorm

        # Momentum update
        v = momentum * v + (1.0 - momentum) * grad

        # Multiplicative update with clamping
        log_update = -eta * v
        log_update = np.clip(log_update, -4.0, 4.0)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        # Evaluate
        current_val = fast_convolve(a).max()
        if current_val < best_val - 1e-12:
            best_val = current_val
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"      MD iter {it}: obj={best_val:.10f} (tau={tau:.1f}, eta={eta:.3f})")
        else:
            no_improve += 1

        # Restart if stuck
        if no_improve > 400:
            rng = np.random.default_rng(it * 7919 + 3)
            a = a * (1.0 + 0.015 * rng.standard_normal(n))
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            v = np.zeros(n)
            no_improve = 0
            if verbose:
                print(f"      MD restart at iter {it}, val={fast_convolve(a).max():.10f}")

    return best_a, best_val


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-Frank-Wolfe with multi-level M relaxation and line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_levels = [1.0, 1.02, 1.05, 1.1, 1.15, 1.3, 1.6, 2.0, 3.0]
    m_idx = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        rem = time_limit - (time.monotonic() - t0)
        if rem < 2.0:
            break

        g = None
        for level_try in range(m_idx, len(m_levels)):
            M_try = conv.max() * m_levels[level_try]
            g = solve_lp_direction(f, M_try, n)
            if g is not None and g.sum() > 1e-12:
                m_idx = level_try
                break
            m_idx = level_try + 1

        if g is None or g.sum() < 1e-12:
            no_improve += 1
            if no_improve > 40:
                break
            rng = np.random.default_rng(it * 13 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            conv = fast_convolve(f)
            m_idx = 0
            continue

        g = g / g.sum()
        diff = np.max(np.abs(g - f))
        if diff < 1e-10:
            no_improve += 1
            m_idx = min(m_idx + 1, len(m_levels) - 1)
            if no_improve > 30:
                break
            continue
        no_improve = 0

        # Line search
        best_alpha = 0.0
        best_val = conv.max()
        alphas = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
        for alpha in alphas:
            ft = np.maximum(f + alpha * (g - f), 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha < 1e-10:
            no_improve += 1
            m_idx = min(m_idx + 1, len(m_levels) - 1)
            continue

        f_new = np.maximum(f + best_alpha * (g - f), 0.0)
        f_new = f_new / f_new.sum()
        conv = fast_convolve(f_new)
        M = conv.max()
        f = f_new
        m_idx = 0

        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()
            if verbose and it % 10 == 0:
                print(f"      LP-FW iter {it}: obj={best_obj:.10f}")

    return best_f, best_obj


def coordinate_descend(f0, time_limit, verbose=False):
    """Targeted coordinate descent: reduce peak by redistributing mass."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if no_improve > 80:
            break

        conv = fast_convolve(f)
        M = conv.max()
        peak_k = int(np.argmax(conv))

        # Find contributing indices
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        if js > je:
            break
        contrib_idx = np.arange(js, je + 1)
        products = f[contrib_idx] * f[peak_k - contrib_idx]
        total = products.sum()
        if total < 1e-15:
            break

        # Try reducing top contributors and moving mass to non-contributing coords
        sorted_idx = contrib_idx[np.argsort(products)[::-1]]
        improved = False

        for src in sorted_idx[:5]:
            for frac in [0.05, 0.1, 0.15, 0.2, 0.3]:
                diff = f[src] * frac
                if diff < 1e-15:
                    continue
                trial = f.copy()
                trial[src] -= diff
                # Move to non-contributing coordinates
                non_contrib = np.ones(n, dtype=bool)
                non_contrib[contrib_idx] = False
                nc_vals = trial[non_contrib]
                nc_sum = nc_vals.sum()
                if nc_sum < 1e-15:
                    nc_targets = np.ones(n) / n
                else:
                    nc_targets = np.zeros(n)
                    nc_targets[non_contrib] = nc_vals / nc_sum
                trial += diff * nc_targets
                trial = np.maximum(trial, 0.0)
                trial = trial / trial.sum()
                obj = fast_convolve(trial).max()
                if obj < best_obj - 1e-14:
                    f = trial
                    best_obj = obj
                    best_f = f.copy()
                    no_improve = 0
                    improved = True
                    break
            if improved:
                break

        if not improved:
            no_improve += 1
            # Try a different peak
            top_peaks = np.argsort(conv)[-3:]
            found = False
            for pk in top_peaks:
                pk = int(pk)
                if pk == peak_k:
                    continue
                js2 = max(0, pk - n + 1)
                je2 = min(n - 1, pk)
                if js2 > je2:
                    continue
                cidx2 = np.arange(js2, je2 + 1)
                prods2 = f[cidx2] * f[pk - cidx2]
                src2 = cidx2[np.argmax(prods2)]
                for frac2 in [0.1, 0.2, 0.3]:
                    diff2 = f[src2] * frac2
                    trial = f.copy()
                    trial[src2] -= diff2
                    nc2 = np.ones(n, dtype=bool)
                    nc2[cidx2] = False
                    nc2_vals = trial[nc2]
                    nc2_sum = nc2_vals.sum()
                    if nc2_sum < 1e-15:
                        continue
                    trial[nc2] += diff2 * nc2_vals / nc2_sum
                    trial = np.maximum(trial, 0.0)
                    trial = trial / trial.sum()
                    obj2 = fast_convolve(trial).max()
                    if obj2 < best_obj - 1e-14:
                        f = trial
                        best_obj = obj2
                        best_f = f.copy()
                        no_improve = 0
                        found = True
                        break
                if found:
                    break
            if not found:
                no_improve += 1

    return best_f, best_obj


def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n / 2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x / sigma) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_bell(n, width_frac=0.6):
    x = np.arange(n) - n / 2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x) / half_width, 0, 1.0)
    a = 1.0 - t ** 4
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_cosine_mod(n, freq, amp):
    x = np.arange(n)
    a = 1.0 + amp * np.cos(2.0 * np.pi * freq * x / n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def make_multi_cosine(n, n_freqs=3, max_amp=0.3, seed=0):
    rng = np.random.default_rng(seed)
    a = np.ones(n)
    for m in range(1, n_freqs + 1):
        amp = rng.uniform(0.05, max_amp)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.cos(2.0 * np.pi * m * np.arange(n) / n + phase)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def objective_normalized(a):
    """Compute 2*n*max(conv)/sum^2 for normalized a (sum=1)."""
    conv = fast_convolve(a)
    return 2.0 * len(a) * conv.max()


def try_seq_and_update(a, label, best_seq, best_score):
    """Try a sequence, return (best_seq, best_score) possibly updated."""
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return best_seq, best_score
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return best_seq, best_score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimization:
    - Phase 1: Diverse initializations
    - Phase 2: Fast mirror descent with smoothed gradients
    - Phase 3: LP-Frank-Wolfe polishing
    - Phase 4: Multi-length exploration + coordinate descent
    - Phase 5: Final high-precision polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 12.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            if p_score < best_score:
                best_score = p_score
                best_seq = parent.copy()
    except Exception:
        pass

    if parent is None:
        parent = make_gaussian(768)
        best_seq, best_score = try_seq_and_update(parent, "default_parent", best_seq, best_score)

    # === Phase 1: Diverse initializations ===
    print("\n=== Phase 1: Diverse Initializations ===")
    t0 = time.monotonic()

    n_values = [512, 640, 768, 896, 1024, 1280]
    init_candidates = []

    for n_try in n_values:
        init_candidates.append(make_gaussian(n_try, 0.25))
        init_candidates.append(make_gaussian(n_try, 0.35))
        init_candidates.append(make_bell(n_try, 0.5))
        init_candidates.append(make_cosine_mod(n_try, 1, 0.15))
        init_candidates.append(make_multi_cosine(n_try, 4, 0.2, seed=n_try))
        init_candidates.append(np.ones(n_try) / n_try)

    if parent is not None:
        init_candidates.append(parent)

    scored = []
    for a_init in init_candidates:
        sc = objective_normalized(a_init)
        scored.append((sc, a_init))
    scored.sort(key=lambda x: x[0])

    print(f"  Best init obj: {scored[0][0]:.6f} (n={len(scored[0][1])})")
    print(f"  Phase 1 time: {time.monotonic()-t0:.1f}s")

    # === Phase 2: Mirror Descent (main optimization) ===
    print("\n=== Phase 2: Mirror Descent ===")
    t0 = time.monotonic()

    # Try top 3 initializations with different parameters
    for trial in range(min(3, len(scored))):
        rem = deadline - time.monotonic() - 500
        if rem < 30:
            break
        sc_init, a_init = scored[trial]
        t_alloc = min(rem * 0.45, 250)
        if t_alloc < 20:
            break
        print(f"  MD trial {trial}: n={len(a_init)}, init_obj={sc_init:.6f}, alloc={t_alloc:.0f}s")
        a_opt, obj = mirror_descent(a_init, time_limit=t_alloc,
                                     tau_start=3.0, tau_end=100.0,
                                     eta_init=0.35, top_k=25,
                                     momentum_init=0.0, momentum_end=0.65,
                                     verbose=True)
        best_seq, best_score = try_seq_and_update(a_opt, f"md_t{trial}", best_seq, best_score)

    # Run MD on the parent with different parameters
    if parent is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            t_alloc = min(rem, 200)
            print(f"  MD parent: n={len(parent)}, alloc={t_alloc:.0f}s")
            a_opt, obj = mirror_descent(parent, time_limit=t_alloc,
                                         tau_start=8.0, tau_end=120.0,
                                         eta_init=0.25, top_k=15,
                                         momentum_init=0.2, momentum_end=0.55,
                                         verbose=True)
            best_seq, best_score = try_seq_and_update(a_opt, "md_parent", best_seq, best_score)

    # Additional MD run with different settings if time allows
    if best_seq is not None:
        rem = deadline - time.monotonic() - 350
        if rem > 30:
            t_alloc = min(rem, 150)
            print(f"  MD grind: n={len(best_seq)}, alloc={t_alloc:.0f}s")
            a_opt, obj = mirror_descent(best_seq, time_limit=t_alloc,
                                         tau_start=20.0, tau_end=200.0,
                                         eta_init=0.15, top_k=10,
                                         momentum_init=0.4, momentum_end=0.6,
                                         verbose=True)
            best_seq, best_score = try_seq_and_update(a_opt, "md_grind", best_seq, best_score)

    print(f"  Phase 2 time: {time.monotonic()-t0:.1f}s")

    # === Phase 3: LP-Frank-Wolfe polishing ===
    print("\n=== Phase 3: LP-Frank-Wolfe ===")
    t0 = time.monotonic()

    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            t_alloc = min(rem, 500)
            print(f"  LP-FW: n={len(best_seq)}, alloc={t_alloc:.0f}s")
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=True)
            best_seq, best_score = try_seq_and_update(a_opt, "lpfw", best_seq, best_score)

    print(f"  Phase 3 time: {time.monotonic()-t0:.1f}s")

    # === Phase 4: Multi-length exploration + coordinate descent ===
    print("\n=== Phase 4: Multi-Length + Coord Descent ===")
    t0 = time.monotonic()

    # Coordinate descent on current best
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_alloc = min(rem, 120)
            print(f"  CoordDescend: n={len(best_seq)}, alloc={t_alloc:.0f}s")
            a_opt, obj = coordinate_descend(best_seq, time_limit=t_alloc, verbose=True)
            best_seq, best_score = try_seq_and_update(a_opt, "coord", best_seq, best_score)

    # Multi-length exploration
    if best_seq is not None:
        base_n = len(best_seq)
        for target_n in [512, 640, 896, 1024, 1280]:
            if target_n == base_n:
                continue
            rem = deadline - time.monotonic() - 50
            if rem < 15:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            best_seq, best_score = try_seq_and_update(f_interp, f"interp_n{target_n}", best_seq, best_score)

            if best_score < 1.51:
                t_alloc = min(rem * 0.25, 50)
                if t_alloc > 10:
                    a_opt, obj = mirror_descent(f_interp, time_limit=t_alloc,
                                                 tau_start=10.0, tau_end=80.0,
                                                 eta_init=0.25, top_k=20,
                                                 verbose=False)
                    best_seq, best_score = try_seq_and_update(a_opt, f"md_n{target_n}", best_seq, best_score)

    print(f"  Phase 4 time: {time.monotonic()-t0:.1f}s")

    # === Phase 5: Final polish ===
    print("\n=== Phase 5: Final Polish ===")
    t0 = time.monotonic()

    if best_seq is not None:
        # High-tau MD for precise peak reduction
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            t_alloc = min(rem * 0.4, 40)
            a_opt, obj = mirror_descent(best_seq, time_limit=t_alloc,
                                         tau_start=50.0, tau_end=300.0,
                                         eta_init=0.08, top_k=5,
                                         momentum_init=0.3, momentum_end=0.5,
                                         verbose=True)
            best_seq, best_score = try_seq_and_update(a_opt, "md_final", best_seq, best_score)

        # LP-FW final squeeze
        rem = deadline - time.monotonic() - 10
        if rem > 8:
            t_alloc = min(rem, 30)
            a_opt, obj = lp_fw_optimize(best_seq, time_limit=t_alloc, verbose=False)
            best_seq, best_score = try_seq_and_update(a_opt, "lpfw_final", best_seq, best_score)

        # One more coordinate descent
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            a_opt, obj = coordinate_descend(best_seq, time_limit=min(rem, 15), verbose=False)
            best_seq, best_score = try_seq_and_update(a_opt, "coord_final", best_seq, best_score)

    print(f"  Phase 5 time: {time.monotonic()-t0:.1f}s")

    # === Phase 6: Perturbation restarts with remaining time ===
    print("\n=== Phase 6: Perturbation Restarts ===")

    if best_seq is not None and time.monotonic() < deadline - 8:
        n_best = len(best_seq)
        r = 0
        while time.monotonic() < deadline - 8:
            t_alloc = min(12.0, (deadline - time.monotonic() - 8) * 0.6)
            if t_alloc < 4:
                break
            rng_r = np.random.default_rng(seed + 5000 + r)
            strategy = r % 5
            if strategy == 0:
                a_pert = best_seq * (1.0 + 0.008 * rng_r.standard_normal(n_best))
            elif strategy == 1:
                shift = int(rng_r.integers(1, max(2, n_best // 20)))
                a_pert = np.roll(best_seq, shift)
            elif strategy == 2:
                mix = rng_r.uniform(0.03, 0.12)
                a_pert = (1 - mix) * best_seq + mix * (np.ones(n_best) / n_best)
            elif strategy == 3:
                a_pert = 0.75 * best_seq + 0.25 * best_seq[::-1]
            else:
                # Squeeze center
                center = n_best // 2
                width = max(5, n_best // 10)
                mask = np.ones(n_best)
                mask[center - width:center + width] = 0.85
                a_pert = best_seq * mask

            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()

            a_opt, obj = mirror_descent(a_pert, time_limit=t_alloc,
                                         tau_start=15.0, tau_end=100.0,
                                         eta_init=0.2, top_k=10,
                                         momentum_init=0.3, momentum_end=0.5,
                                         verbose=False)
            best_seq, best_score = try_seq_and_update(a_opt, f"pert{r}", best_seq, best_score)
            r += 1

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"Total time: {float(budget_s) - 12.0 + time.monotonic() - deadline:.1f}s")
    return [float(x) for x in best_seq]
```
