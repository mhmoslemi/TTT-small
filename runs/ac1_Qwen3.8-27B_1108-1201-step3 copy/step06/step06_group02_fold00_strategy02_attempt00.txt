def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n) - n/2.0
    sigma = n * sigma_frac
    a = np.exp(-0.5 * (x/sigma)**2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a

def make_bell(n, width_frac=0.6):
    x = np.arange(n) - n/2.0
    half_width = n * width_frac / 2.0
    t = np.clip(np.abs(x)/half_width, 0, 1.0)
    a = np.maximum(1.0 - t**4, 0)  # flatten bell with F4 taper
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a

def make_cosine_mod(n, freq, amp):
    x = np.arange(n)
    a = 1.0 + amp * np.cos(2.0*np.pi*freq*x/n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a

def make_multi_cosine(n, n_freqs=3, max_amp=0.3):
    a = np.ones(n)
    for i in range(1, n_freqs+1):

    a = np.exp(a) - 1  # Make tall spikes
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a
