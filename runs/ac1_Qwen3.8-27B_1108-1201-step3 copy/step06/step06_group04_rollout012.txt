Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using mirror descent with adaptive
temperature, Nesterov momentum, multi-length exploration, and peak-targeted
coordinate descent polish. Key ideas: softmax-smoothed subgradient computed
via FFT, multiplicative weight updates for simplex constraints, simulated
annealing for escaping local minima, and a final coordinate descent phase.

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.zeros(0)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    idx = np.nonzero(cond)[0][-1]
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_sgrad(a, temperature):
    """Compute soft-max subgradient of conv(a,a) w.r.t. a, and return conv."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = conv / temperature
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p /= p_sum
    
    # subgrad[j] = sum_i p[j+i] * a[i] for i=0..n-1 (p out of bounds = 0)
    # = cr[n-1+j] where cr = convolve(p, a_reversed)
    a_rev = a[::-1]
    cr_size = len(p) + n - 1
    fft_size = 1 << (cr_size - 1).bit_length()
    P = np.fft.rfft(p, fft_size)
    R = np.fft.rfft(a_rev, fft_size)
    cr = np.fft.irfft(P * R, fft_size)[:cr_size]
    subgrad = cr[n-1 : 2*n-1]
    return subgrad, conv

        else:

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
