```python
def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
```

```python
    """FISTA with log-sum-exp smoothing for the auto-correlation peak minimization."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Initial parameters
    mu = 0.01  # smoothing parameter
    mu_decay = 0.95
    step_size = 0.5  # initial step (will be adjusted)
    
    k = 0  # FISTA counter
    y = a.copy()  # FISTA momentum variable
    
    while time.monotonic() - t0 < time_limit:
        if mu < 1e-7:
            break
            
        # Compute convolution at y
        conv_y = fast_convolve(y)
        
        # Log-sum-exp weights
        max_cv = conv_y.max()
        shifted = (conv_y - max_cv) / mu
        weights = np.exp(shifted)
        weights /= weights.sum()
        
        # Gradient of smoothed objective
        grad = np.zeros(n)
        # Only use top-k weights for efficiency
        top_k = min(100, len(conv_y))
        top_idx = np.argpartition(conv_y, -top_k)[-top_k:]
        top_weights = weights[top_idx]
        top_weights /= top_weights.sum()  # renormalize
        
        for idx in top_idx:
            k_val = int(idx)
            if top_weights[list(top_idx).index(idx)] < 1e-10:
                continue
            js = max(0, k_val - n + 1)
            je = min(n - 1, k_val)
            js_arr = np.arange(js, je + 1)
            wk = top_weights[list(top_idx).index(idx)]
            grad[js_arr] += 2.0 * wk * y[k_val - js_arr]
        
        # FISTA update
        alpha = 1.0 / (2.0 + 1.0/mu)  # step size (Lipschitz bound)
        z = np.maximum(y - alpha * grad, 0.0)
        z = z / z.sum() if z.sum() > 0 else a
        
        # Check for improvement
```

```python
def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
```

    """Solve KKT system using Newton's method."""
    n = len(a0)
    A = list(active_set)
    m = len(A)

    a = np.maximum(a0.astype(np.float64), 0.0)

        for j in range(n):
