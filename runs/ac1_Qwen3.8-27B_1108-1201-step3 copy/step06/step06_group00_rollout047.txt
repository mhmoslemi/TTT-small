def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def frank_wolfe_lp(a, active_set, max_iter=100):
    """
    Solve the Frank-Wolfe LP:
    minimize t
    subject to: grad_i . d <= t for i in active_set
                a + d >= 0
                sum(d) = 0
    
    Returns the direction d (or None if infeasible).
    """

    m = len(active_set)

    # We can compute all gradients at once using cross-correlation

    w[active_set] = 1.0
    all_grads = 2.0 * cross_corr(w, a)  # This gives sum_i w[i] * grad_i ... no

    # grad_i[j] = 2 * a[i-j] if 0 <= i-j < n else 0

    # For the LP:
    # Variables: d_0, ..., d_{n-1}, t (n+1 variables)
    # Minimize: t (the last variable)
    # Constraints:
    #   For each i in active_set: sum_j grad_i[j] * d_j - t <= 0  (m constraints)
    #   For each j: d_j >= -a[j]  (n constraints)
    #   sum_j d_j = 0  (1 equality)
    
    # Total: n+1 variables, m+n inequality constraints, 1 equality constraint

    # Instead of the full LP, we can use a simpler heuristic:
    # d = -projected_subgradient / ||projected_subgradient||
    # with a backtracking line search.

    # Compute the "average" subgradient over the active set
    w_avg = np.zeros(2 * n - 1)
    w_avg[active_set] = 1.0 / m
    g = 2.0 * cross_corr(w_avg, a)
    g = g - g.mean()  # project to simplex tangent

    # Backtracking line search
    a_curr = a.copy()
    b_curr = fast_convolve(a_curr)
    obj_curr = b_curr.max()
    
    alpha = 1.0
    for _ in range(30):
        a_try = project_simplex(a_curr + alpha * d)
        b_try = fast_convolve(a_try)
        obj_try = b_try.max()
        if obj_try < obj_curr - 1e-15:
            return a_try, obj_try

def coordinate_descent_polish(a0, time_limit, n_sweep=20):
    """Coordinate descent: optimize one element at a time."""
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            
            old_val = a[j]
            if old_val < 1e-15:
                continue
            
            # Try a grid of new values
            # a[j] = new_val, scale others to maintain sum=1
            best_new_val = old_val
            best_new_obj = best_obj
            
            # Log-scale around old value
            ratios = np.logspace(-1.5, 1.5, n_sweep)
            for ratio in ratios:
                new_val = old_val * ratio
                if new_val < 0 or new_val >= 1.0:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0 or scale > 100:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                obj = fast_convolve(a_trial).max()
                if obj < best_new_obj - 1e-15:
                    best_new_obj = obj
                    best_new_val = new_val
            
            if best_new_val != old_val:
                denom = 1.0 - old_val
                scale = (1.0 - best_new_val) / denom
                a[j] = best_new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = a / a.sum()
                if best_new_obj < best_obj - 1e-15:
                    best_obj = best_new_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_a, best_obj

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
