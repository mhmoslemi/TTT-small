```python
"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses: (1) Adam subgradient with temperature annealing, (2) LP-based direction
finding via cutting planes, (3) FISTA, (4) coordinate descent polish,
(5) multi-length exploration.

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_convolve_cross(a, g):
    """Compute cross-correlation (a * g) where both have length n."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    G = rfft(g, fft_size)
    return irfft(A * G, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def eval_obj(a):
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (S * S))


def compute_subgradient(a, b, temp, n_top=None):
    """Compute temperature-weighted subgradient of max(conv)."""
    n = len(a)
    scaled = b / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(b))
        w[b.argmax()] = 1.0
    else:
        w /= w_sum
    
    if n_top is not None and n_top < len(b):
        top_idx = np.argpartition(b, -n_top)[-n_top:]
        w_sel = w[top_idx]
        ws = w_sel.sum()
        if ws > 1e-30:
            w_sel /= ws
        else:
            w_sel = np.zeros(len(top_idx))
            w_sel[top_idx == b.argmax()] = 1.0
        w_full = np.zeros(len(b))
        w_full[top_idx] = w_sel
        w = w_full
    
    # g[i] = sum_k w[k] * a[k-i] for valid k-i
    g = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        g[idx] += w[k] * a[k - idx]
    return g


def adam_subgradient(a0, time_limit, lr_init=0.01, beta1=0.9, beta2=0.999,
                     temp_init=0.001, temp_growth=1.02, eps=1e-8,
                     kick_interval=1000, kick_size=0.003, seed=42):
    """Adam-optimized subgradient on the simplex."""
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    cur_temp = temp_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        # Compute subgradient
        b = fast_convolve(a)
        g = compute_subgradient(a, b, cur_temp, n_top=max(20, n // 15))
        
        t += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        # Adam step
        a_new = a - lr_init * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)
        
        # Evaluate
        b_new = fast_convolve(a_new)
        obj_new = b_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        # Periodic adjustments
        if no_improve > kick_interval:
            a = project_simplex(best_a + rng.standard_normal(n) * kick_size)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            lr_init *= 0.8
            no_improve = 0
        
        cur_temp *= temp_growth
        if cur_temp > 0.5:
            cur_temp = 0.5
    
    return best_a, best_obj


def fista_optimize(a0, time_limit, lr_init=0.01, temp_init=0.001,
                   temp_growth=1.02, kick_interval=1000, kick_size=0.003, seed=42):
    """FISTA (accelerated proximal gradient) on the simplex."""
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    y = a.copy()
    t_curr = 1.0
    cur_lr = lr_init
    cur_temp = temp_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(y)
        g = compute_subgradient(y, b, cur_temp, n_top=max(20, n // 15))
        
        # FISTA step
        a_new = project_simplex(y - cur_lr * g)
        t_next = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_curr ** 2))
        y = a_new + ((t_curr - 1.0) / t_next) * (a_new - a)
        y = project_simplex(y)  # Keep y in simplex
        t_curr = t_next
        
        b_new = fast_convolve(a_new)
        obj_new = b_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        if no_improve > kick_interval:
            a = project_simplex(best_a + rng.standard_normal(n) * kick_size)
            y = a.copy()
            t_curr = 1.0
            cur_lr *= 0.8
            no_improve = 0
        
        cur_temp *= temp_growth
        if cur_temp > 0.5:
            cur_temp = 0.5
    
    return best_a, best_obj


def lp_direction(a, M, n_constraints=100, solver='highs'):
    """
    Find the LP direction: max sum(g) s.t. g >= 0, (a*g)_k <= M for active k.
    Returns the optimal g (unnormalized) or None if LP fails.
    """
    n = len(a)
    b = fast_convolve(a)
    
    # Find active constraints: positions where b is close to M
    threshold = M * 0.95
    active = np.where(b >= threshold)[0]
    
    if len(active) == 0:
        active = np.array([b.argmax()])
    
    # Use at most n_constraints
    if len(active) > n_constraints:
        # Pick the most active ones
        idx = np.argsort(b[active])[::-1][:n_constraints]
        active = active[idx]
    
    # Build constraint matrix
    # Row j: sum_i a[k_j - i] * g[i] <= M for valid i
    m = len(active)
    A_ub = np.zeros((m, n))
    for j, k in enumerate(active):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for i in range(lo, hi + 1):
            A_ub[j, i] = a[i]
    
    # Objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=np.full(m, M), bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 10})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            if g.sum() > 1e-10:
                return g
    except Exception as e:
        pass
    
    return None


def lp_refine(a, time_limit, n_lp_iter=20, step_size=0.1, n_constraints=80, seed=42):
    """
    LP-based refinement: iteratively find LP direction and update.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a)
    a = a.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    for it in range(n_lp_iter):
        if time.monotonic() - t0 > time_limit:
            break
        
        M = fast_convolve(a).max()
        g = lp_direction(a, M, n_constraints=n_constraints)
        
        if g is None:
            break
        
        # Rescale g to sum=1
        g = g / g.sum()
        
        # Blend
        a_new = (1 - step_size) * a + step_size * g
        a_new = project_simplex(a_new)
        
        b_new = fast_convolve(a_new)
        obj_new = b_new.max()
        
        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            a_new2 = (1 - step_size * 0.5) * a + step_size * 0.5 * g
            a_new2 = project_simplex(a_new2)
            obj_new2 = fast_convolve(a_new2).max()
            if obj_new2 < best_obj - 1e-15:
                best_obj = obj_new2
                best_a = a_new2.copy()
                a = a_new2
            else:

    return best_a, best_obj


def coordinate_polish(a0, time_limit, n_trials_per_elem=6):
    """Coordinate descent: try scaling each element by various multipliers."""
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    multipliers = np.array([0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0])
    
    improved_global = True
    while improved_global and time.monotonic() - t0 < time_limit:
        improved_global = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-14:
                continue
            best_j_obj = best_obj
            best_j_val = old_val
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                ss = a_trial.sum()
                if ss < 1e-10:
                    continue
                a_trial = a_trial / ss
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                ss = a.sum()
                if ss > 1e-10:
                    a = a / ss
                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved_global = True
    return best_a, best_obj


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian_init(n, sigma_frac=0.3):
    x = np.linspace(0, 1, n)
    center = 0.5
    sigma = sigma_frac
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_cosine_init(n):
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_tukey_init(n, r=0.5):
    x = np.linspace(0, 1, n)
    a = np.ones(n)
    taper = int(r * n // 2)
    if taper > 0:
        t = np.linspace(0, 1, taper)
        a[:taper] = 0.5 * (1 + np.cos(np.pi * (2 * t / r - 1)))
        a[-taper:] = a[:taper][::-1]
    a = np.maximum(a, 1e-10)
    return a / a.sum()
```

```python
"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Key innovation: LP cutting-plane direction finding provides exact best-response
directions for this piecewise-linear objective, combined with Adam subgradient
for fast convergence and coordinate descent for final polishing.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def fast_cross_convolve(a, g):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    G = rfft(g, fft_size)
    return irfft(A * G, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_vals = np.arange(1, n + 1)
    cond = u * rho_vals > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def eval_obj(a):
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (S * S))


def compute_subgradient(a, b, temp, n_top=None):
    n = len(a)
    scaled = b / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(len(b))
        w[b.argmax()] = 1.0
    else:
        w /= w_sum

    if n_top is not None and n_top < len(b):
        top_idx = np.argpartition(b, -n_top)[-n_top:]
        w_sel = w[top_idx]
        ws = w_sel.sum()
        if ws > 1e-30:
            w_sel /= ws
        else:
            w_sel = np.zeros(len(top_idx))
            w_sel[np.argmax(b[top_idx])] = 1.0
        w_full = np.zeros(len(b))
        w_full[top_idx] = w_sel
        w = w_full

    g = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-14:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        idx = np.arange(lo, hi + 1)
        g[idx] += w[k] * a[k - idx]
    return g


def adam_subgradient(a0, time_limit, lr_init=0.005, beta1=0.9, beta2=0.999,
                     temp_init=0.0005, temp_growth=1.02, eps=1e-8,
                     kick_interval=800, kick_size=0.002, seed=42):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    cur_temp = temp_init
    no_improve = 0
    cur_lr = lr_init

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        g = compute_subgradient(a, b, cur_temp, n_top=max(20, n // 15))

        t += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        a_new = a - cur_lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)

        b_new = fast_convolve(a_new)
        obj_new = b_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        if no_improve > kick_interval:

            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            cur_lr *= 0.75
            no_improve = 0
        elif no_improve > kick_interval // 2:
            cur_lr *= 0.95

        cur_temp *= temp_growth
        if cur_temp > 0.5:
            cur_temp = 0.5

    return best_a, best_obj


def build_lp_matrix(a, positions):
    n = len(a)
    m = len(positions)
    A = np.zeros((m, n))
    for j, k in enumerate(positions):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        cols = np.arange(lo, hi + 1)
        A[j, cols] = a[k - cols]
    return A


def lp_solve(a, M, positions, time_limit_lp=5):
    n = len(a)
    m = len(positions)
    if m == 0:
        return None
    A_ub = build_lp_matrix(a, positions)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=np.full(m, M), bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': time_limit_lp})
        if res.status == 0:
            g = np.maximum(res.x, 0)
            if g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None

              n_constraints_max=300, seed=42, max_rounds=50):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    for it in range(max_rounds):
        if time.monotonic() - t0 > time_limit:
            break

        b = fast_convolve(a)
        M = b.max()

        # Find active constraint positions
        threshold = M * 0.92
        active = np.where(b >= threshold)[0]
        if len(active) > n_constraints_start:
            idx = np.argsort(b[active])[::-1][:n_constraints_start]
            active = active[idx]
        if len(active) == 0:
            active = np.array([b.argmax()])

        # Solve LP
        time_left = time_limit - (time.monotonic() - t0)
        lp_time = min(5.0, time_left * 0.1)
        if lp_time < 0.5:
            break

        g = lp_solve(a, M, active, time_limit_lp=lp_time)
        if g is None:
            break

        # Check feasibility over ALL positions
        cross = fast_cross_convolve(a, g)
        violated = np.where(cross > M * (1 + 1e-8))[0]

        if len(violated) == 0:
            # LP solution is feasible - great direction
            g_norm = g / g.sum()
            a_new = (1 - step_size) * a + step_size * g_norm
            a_new = project_simplex(a_new)
            b_new = fast_convolve(a_new)
            obj_new = b_new.max()

            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
            else:
                # Try smaller step
                for frac in [0.5, 0.25, 0.1]:
                    a_try = (1 - step_size * frac) * a + step_size * frac * g_norm
                    a_try = project_simplex(a_try)
                    obj_try = fast_convolve(a_try).max()
                    if obj_try < best_obj - 1e-15:
                        best_obj = obj_try
                        best_a = a_copy = a_try.copy()
                        a = a_try
                        break
                else:
                    break
        else:
            # Add violated constraints and re-solve
            all_positions = list(active) + list(violated)
            all_positions = list(set(all_positions))
            if len(all_positions) > n_constraints_max:
                # Keep most violated
                b_pos = cross[all_positions]
                idx = np.argsort(b_pos)[::-1][:n_constraints_max]
                all_positions = [all_positions[i] for i in idx]

            g2 = lp_solve(a, M, np.array(all_positions), time_limit_lp=lp_time)
            if g2 is None:
                break
            g_norm = g2 / g2.sum()
            a_new = (1 - step_size) * a + step_size * g_norm
            a_new = project_simplex(a_new)
            b_new = fast_convolve(a_new)
            obj_new = b_new.max()

            if obj_new < best_obj - 1e-15:
                best_obj = obj_new
                best_a = a_new.copy()
                a = a_new
            else:
                break

    return best_a, best_obj


def fista_optimize(a0, time_limit, lr_init=0.005, temp_init=0.0005,
                   temp_growth=1.02, kick_interval=800, kick_size=0.002, seed=42):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    y = a.copy()
    t_curr = 1.0
    cur_lr = lr_init
    cur_temp = temp_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(y)
        g = compute_subgradient(y, b, cur_temp, n_top=max(20, n // 15))

        a_new = project_simplex(y - cur_lr * g)
        t_next = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_curr ** 2))
        y = a_new + ((t_curr - 1.0) / t_next) * (a_new - a)
        y = np.maximum(y, 0)
        ys = y.sum()
        if ys > 1e-15:
            y = y / ys
        t_curr = t_next

        b_new = fast_convolve(a_new)
        obj_new = b_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        if no_improve > kick_interval:
            a = project_simplex(best_a + rng.standard_normal(n) * kick_size)
            y = a.copy()
            t_curr = 1.0
            cur_lr *= 0.75
            no_improve = 0

        cur_temp *= temp_growth
        if cur_temp > 0.5:
            cur_temp = 0.5

    return best_a, best_obj


def coordinate_polish(a0, time_limit):
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = a / a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    multipliers = np.array([0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0])

    improved_global = True
    while improved_global and time.monotonic() - t0 < time_limit:
        improved_global = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            old_val = a[j]
            if old_val < 1e-14:
                continue
            best_j_obj = best_obj
            best_j_val = old_val

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                ss = a_trial.sum()
                if ss < 1e-10:
                    continue
                a_trial = a_trial / ss
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                ss = a.sum()
                if ss > 1e-10:
                    a = a / ss
                if best_j_obj < best_obj - 1e-15:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved_global = True
    return best_a, best_obj


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_gaussian_init(n, sigma_frac=0.3):
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma_frac) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_tukey_init(n, r=0.5):
    x = np.linspace(0, 1, n)
    a = np.ones(n)
    taper = int(r * n // 2)
    if taper > 0:
        t = np.linspace(0, 1, taper)
        a[:taper] = 0.5 * (1 + np.cos(np.pi * (2 * t / r - 1)))
        a[-taper:] = a[:taper][::-1]
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_cosine_init(n):
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization: Adam subgradient -> LP cutting-plane -> FISTA
    -> LP again -> multi-length -> coordinate descent -> micro polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = make_tukey_init(1152, 0.5)
        print("No parent, using Tukey init n=1152")

    n_parent = len(parent)

    # === Phase 1: Adam subgradient warmup ===
    print(f"\n=== Phase 1: Adam warmup (n={n_parent}) ===")
    rem = deadline - time.monotonic() - 600
    if rem > 60:
        t_alloc = min(rem, 200.0)
        print(f"  t={t_alloc:.0f}s")
        a_sg, _ = adam_subgradient(
            parent, time_limit=t_alloc,
            lr_init=0.004, beta1=0.9, beta2=0.999,
            temp_init=0.0003, temp_growth=1.02,
            kick_interval=600, kick_size=0.002, seed=seed
        )
        try_seq(a_sg, "adam1")

    # === Phase 2: LP cutting-plane (primary improvement) ===
    print(f"\n=== Phase 2: LP cutting-plane (score={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 500
        if rem > 30:
            t_alloc = min(rem * 0.5, 200.0)
            print(f"  LP round 1, t={t_alloc:.0f}s, n={len(best_seq)}")
            a_lp, _ = lp_refine(
                best_seq, time_limit=t_alloc,
                step_size=0.1, n_constraints_start=50,
                n_constraints_max=250, seed=seed, max_rounds=40
            )
            try_seq(a_lp, "lp1")

    # === Phase 3: Adam on LP result ===
    print(f"\n=== Phase 3: Adam post-LP (score={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            t_alloc = min(rem * 0.4, 100.0)
            print(f"  t={t_alloc:.0f}s")
            a_sg2, _ = adam_subgradient(
                best_seq, time_limit=t_alloc,
                lr_init=0.002, beta1=0.95, beta2=0.999,
                temp_init=0.0001, temp_growth=1.01,
                kick_interval=500, kick_size=0.001, seed=seed + 100
            )
            try_seq(a_sg2, "adam2")

    # === Phase 4: Second LP round ===
    print(f"\n=== Phase 4: LP round 2 (score={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 20:
            t_alloc = min(rem * 0.5, 150.0)
            print(f"  t={t_alloc:.0f}s")
            a_lp2, _ = lp_refine(
                best_seq, time_limit=t_alloc,
                step_size=0.05, n_constraints_start=80,
                n_constraints_max=300, seed=seed + 200, max_rounds=40
            )
            try_seq(a_lp2, "lp2")

    # === Phase 5: FISTA ===
    print(f"\n=== Phase 5: FISTA (score={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 200
        if rem > 20:
            t_alloc = min(rem * 0.4, 80.0)
            print(f"  t={t_alloc:.0f}s")
            a_f, _ = fista_optimize(
                best_seq, time_limit=t_alloc,
                lr_init=0.003, temp_init=0.0002,
                temp_growth=1.01, kick_interval=500, kick_size=0.001,
                seed=seed + 300
            )
            try_seq(a_f, "fista")

    # === Phase 6: Third LP round (more constraints) ===
    print(f"\n=== Phase 6: LP round 3 (score={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 150
        if rem > 15:
            t_alloc = min(rem * 0.5, 100.0)
            print(f"  t={t_alloc:.0f}s")
            a_lp3, _ = lp_refine(
                best_seq, time_limit=t_alloc,
                step_size=0.03, n_constraints_start=100,
                n_constraints_max=400, seed=seed + 400, max_rounds=30
            )
            try_seq(a_lp3, "lp3")

    # === Phase 7: Multi-length exploration ===
    print(f"\n=== Phase 7: Multi-length (score={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        target_lengths = []
        for delta in [-32, -16, -8, 8, 16, 32, 64, 128]:
            tl = n_best + delta
            if 256 <= tl <= 4096:
                target_lengths.append(tl)

        for target_n in target_lengths[:5]:
            rem = deadline - time.monotonic() - 80
            if rem < 15:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.3, 40.0)
            if t_alloc < 5:
                continue
            print(f"  n={target_n}, t={t_alloc:.0f}s")
            a_opt, _ = adam_subgradient(
                f_interp, time_limit=t_alloc,
                lr_init=0.003, temp_init=0.0005,
                temp_growth=1.02, kick_interval=400, kick_size=0.002,
                seed=seed + target_n
            )
            try_seq(a_opt, f"len_{target_n}")
            # Quick LP polish on best length
            if best_seq is not None and len(best_seq) == target_n:
                rem = deadline - time.monotonic() - 60
                if rem > 10:
                    a_lp_l, _ = lp_refine(
                        best_seq, time_limit=min(rem * 0.3, 30.0),
                        step_size=0.05, n_constraints_start=50,
                        n_constraints_max=200, seed=seed + target_n + 50
                    )
                    try_seq(a_lp_l, f"lp_len_{target_n}")

    # === Phase 8: Deep Adam ===
    print(f"\n=== Phase 8: Deep Adam (score={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 80
        if rem > 20:
            t_alloc = min(rem * 0.5, 60.0)
            print(f"  t={t_alloc:.0f}s")
            a_deep, _ = adam_subgradient(
                best_seq, time_limit=t_alloc,
                lr_init=0.0008, beta1=0.97, beta2=0.999,
                temp_init=0.00005, temp_growth=1.003,
                kick_interval=1500, kick_size=0.0008, seed=seed + 777
            )
            try_seq(a_deep, "deep_adam")

    # === Phase 9: Final LP polish ===
    print(f"\n=== Phase 9: Final LP (score={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 50
        if rem > 10:
            t_alloc = min(rem * 0.5, 40.0)
            a_lpf, _ = lp_refine(
                best_seq, time_limit=t_alloc,
                step_size=0.02, n_constraints_start=150,
                n_constraints_max=500, seed=seed + 888, max_rounds=20
            )
            try_seq(a_lpf, "lp_final")

    # === Phase 10: Coordinate descent ===
    print(f"\n=== Phase 10: CD polish (score={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 25
        if rem > 15:
            t_cd = min(rem, 40.0)
            print(f"  t={t_cd:.0f}s")
            a_cd, _ = coordinate_polish(best_seq, time_limit=t_cd)
            try_seq(a_cd, "cd")

    # === Phase 11: Micro optimization ===
    print(f"\n=== Phase 11: Micro (score={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 5
        if rem > 8:
            a_micro, _ = adam_subgradient(
                best_seq, time_limit=min(rem * 0.5, 20.0),
                lr_init=0.0003, beta1=0.98, beta2=0.999,
                temp_init=0.00001, temp_growth=1.001,
                kick_interval=2000, kick_size=0.0003, seed=seed + 999
            )
            try_seq(a_micro, "micro")

            rem = deadline - time.monotonic() - 3
            if rem > 5:
                for attempt in range(3):
                    rem = deadline - time.monotonic() - 2
                    if rem < 2:
                        break
                    noise = rng.standard_normal(len(best_seq)) * 0.0008
                    a_pert = project_simplex(best_seq + noise)
                    a_p, _ = adam_subgradient(
                        a_pert, time_limit=min(rem * 0.4, 8.0),
                        lr_init=0.0005, temp_init=0.0001,
                        temp_growth=1.005, kick_interval=200, kick_size=0.0005,
                        seed=seed + attempt * 31337
                    )
                    try_seq(a_p, f"pert_{attempt}")

    # Fallback
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")

    return [float(x) for x in best_seq]
```
