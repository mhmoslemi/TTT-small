```python
import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def lp_step(f, M, maxiter=100):
    """
    Solve the LP: maximize sum(g) subject to g >= 0, (f*g)_k <= M for all k.
    Returns the optimal g (unscaled).
    """
    n = len(f)
    
    # Build constraint matrix A_ub and b_ub
    # For each k = 0, ..., 2n-2: sum_i f_i * g_{k-i} <= M
    # where i ranges over max(0, k-n+1) to min(n-1, k)
    
    # A_ub will have (2n-1) rows and n columns
    # Row k: for i in valid range, A_ub[k, i] = f_{k-i}
    
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for i in range(js, je + 1):
            A_ub[k, i] = f[k - i]
    
    b_ub = np.full(2 * n - 1, M)
    
    # Objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    
    if result.success:
        g = result.x
        g = np.maximum(g, 0)
        return g
    else:
        return None
```

```python
def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. g >= 0, row_k @ g <= M for all k.
    row_k[i] = f[k-i] if 0 <= k-i < n, else 0."""
    n = len(f)
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for i in range(js, je + 1):
            A_ub[k, i] = f[k - i]
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, 
                  method='highs', options={'presolve': True})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None

def coordinate_descent(a, time_limit, verbose=False, label=""):
    """Coordinate descent with line search."""
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)
    
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue
            denom = 1.0 - old_val
            if denom < 1e-10:
                continue
            
            best_j_obj = best_obj
            best_j_val = old_val
            for mult in [0.8, 0.9, 1.1, 1.2]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.4:
                    continue
                scale = (1.0 - new_val) / denom
                trial = a.copy()
                trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                trial[mask] *= scale
                trial = np.maximum(trial, 0.0)
                s = trial.sum()
                if s < 1e-15:
                    continue
                trial = trial / s
                val = fast_convolve(trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_a, best_obj


def make_init(n, kind="gauss"):
    """Create initial sequence."""
    if kind == "gauss":
        x = np.arange(n) - n / 2.0
        sigma = n * 0.3
        a = np.exp(-0.5 * (x / sigma) ** 2)
    elif kind == "bell":
        x = np.arange(n) - n / 2.0
        t = np.clip(np.abs(x) / (n * 0.3), 0, 1)
        a = 1.0 - t ** 4
    elif kind == "ramp":
        a = np.linspace(0.1, 1.0, n)
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a
```

Looking at this problem, I need to minimize `2*n*max(conv(a,a))/(sum(a))^2` for a non-negative sequence. The previous approach used mirror descent and achieved 1.505. Let me think about a fundamentally different approach.

**Key insight**: The "state of the art" mentions solving a linear program at each step. Given current sequence `f` with peak `M`, the LP `maximize sum(g) s.t. g>=0, (f*g)_k <= M` finds the most "spread out" sequence compatible with the current peak level. """
```python
LP-based iterative optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Primary method: at each step, solve LP max sum(g) s.t. g>=0, (f*g)_k <= M
to find the optimal redistribution direction. Combined with mirror descent
and coordinate descent for multi-phase optimization.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def solve_lp(f, M):
    n = len(f)
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for i in range(js, je + 1):
            rows.append(k)
            cols.append(i)
            data.append(f[k - i])
    A_ub = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True})
    if res.success:
        return np.maximum(res.x, 0.0)
    return None


def lp_optimize(a, time_limit, verbose=False, label=""):
    a = np.maximum(a.astype(np.float64), 0.0)
    ss = a.sum()
    if ss < 1e-10:
        return a, float('inf')
    a = a / ss
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()

        g = solve_lp(a, M * 1.001)
        if g is None or g.sum() < 1e-10:
            g = solve_lp(a, M)
            if g is None or g.sum() < 1e-10:
                break
        g = g / g.sum()

        best_t = 0.0
        best_trial = M
        for t in [0.3, 0.5, 0.7, 0.9, 1.0]:
            a_t = (1 - t) * a + t * g
            a_t = np.maximum(a_t, 0)
            s = a_t.sum()
            if s < 1e-10:
                continue
            a_t = a_t / s
            obj_t = fast_convolve(a_t).max()
            if obj_t < best_trial:
                best_trial = obj_t
                best_t = t

        if best_t > 0:
            a = (1 - best_t) * a + best_t * g
            a = np.maximum(a, 0)
            a = a / a.sum()
            stall = 0
        else:
            stall += 1
            if stall > 20:
                rng = np.random.default_rng(it * 104729 + 3)
                a = best_a + rng.standard_normal(n) * 0.001
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                stall = 0
                continue

        a_sym = 0.5 * (a + a[::-1])
        a_sym = np.maximum(a_sym, 0)
        s = a_sym.sum()
        if s > 1e-10:
            a_sym = a_sym / s
            sym_obj = fast_convolve(a_sym).max()
            if sym_obj < best_obj - 1e-14:
                best_obj = sym_obj
                best_a = a_sym.copy()
                a = a_sym.copy()

        cur = fast_convolve(a).max()
        if cur < best_obj - 1e-13:
            best_obj = cur
            best_a = a.copy()
            if verbose and it % 10 == 0:
                print(f"    {label} LP it={it}: obj={best_obj:.10f}")
        else:
            best_a = a.copy()

    return best_a, best_obj


def mirror_descent(a, time_limit, top_k=30, temp_init=0.0003, lr_init=0.008,
                   verbose=False, label=""):
    a = np.maximum(a.astype(np.float64), 1e-15)
    a = a / a.sum()
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()

        K = min(top_k, len(conv))
        if K < len(conv):
            peak_indices = np.argpartition(conv, -K)[-K:]
        else:
            peak_indices = np.arange(len(conv))

        pv = conv[peak_indices]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peak_indices))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        g = np.zeros(n)
        for idx2 in range(len(peak_indices)):
            k = int(peak_indices[idx2])
            w_k = w[idx2]
            if w_k < 1e-15:
                continue
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if je < js:
                continue
            g[js:je + 1] += 2.0 * w_k * a[k - np.arange(js, je + 1)]

        best_lr = lr * 0.1
        best_trial_obj = M * 1.01
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            if trial_lr < 1e-10:
                continue
            log_update = np.clip(-trial_lr * g, -50, 50)
            a_trial = a * np.exp(log_update)
            a_trial = np.maximum(a_trial, 1e-15)
            ss = a_trial.sum()
            if ss < 1e-10:
                continue
            a_trial = a_trial / ss
            trial_obj = fast_convolve(a_trial).max()
            if trial_obj <= best_trial_obj:
                best_lr = trial_lr
                best_trial_obj = trial_obj
                if trial_obj < M * 0.999:
                    break

        log_update = np.clip(-best_lr * g, -50, 50)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        new_obj = fast_convolve(a).max()
        if new_obj < best_obj - 1e-13:
            best_obj = new_obj
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    {label} MD it={it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        temp = min(temp * 1.002, 0.05)
        if no_improve > 100:
            lr *= 0.5
            no_improve = 0
        lr = max(lr * 0.9999, 1e-8)

        if no_improve > 500:
            rng = np.random.default_rng(it * 7919 + 13)
            a = np.maximum(best_a + rng.standard_normal(n) * 0.001, 1e-15)
            a = a / a.sum()
            lr = lr_init * 0.3
            no_improve = 0

        if it % 500 == 0:
            a_s = 0.5 * (a + a[::-1])
            a_s = np.maximum(a_s, 1e-15)
            a_s = a_s / a_s.sum()
            if fast_convolve(a_s).max() < best_obj:
                best_obj = fast_convolve(a_s).max()
                best_a = a_s.copy()
                a = a_s.copy()

        if lr < 1e-9:
            break

    return best_a, best_obj


def coordinate_descent(a, time_limit, verbose=False, label=""):
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    n = len(a)

    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-12:
                continue
            denom = 1.0 - old_val
            if denom < 1e-10:
                continue

            best_j_obj = best_obj
            best_j_val = old_val
            for mult in [0.8, 0.9, 0.95, 1.05, 1.1, 1.25]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.4:
                    continue
                scale = (1.0 - new_val) / denom
                trial = a.copy()
                trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                trial[mask] *= scale
                trial = np.maximum(trial, 0.0)
                s = trial.sum()
                if s < 1e-15:
                    continue
                trial = trial / s
                val = fast_convolve(trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                scale = (1.0 - best_j_val) / denom
                a[j] = best_j_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a[mask] *= scale
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved:
            break

    return best_a, best_obj


def symmetrize(a):
    a_s = 0.5 * (a + a[::-1])
    a_s = np.maximum(a_s, 0)
    s = a_s.sum()
    if s > 1e-15:
        a_s = a_s / s
    return a_s


def make_init(n, kind="gauss"):
    if kind == "gauss":
        x = np.arange(n) - n / 2.0
        sigma = n * 0.28
        a = np.exp(-0.5 * (x / sigma) ** 2)
    elif kind == "bell":
        x = np.arange(n) - n / 2.0
        t = np.clip(np.abs(x) / (n * 0.3), 0, 1)
        a = 1.0 - t ** 4
    elif kind == "superbell":
        x = np.arange(n) - n / 2.0
        t = np.clip(np.abs(x) / (n * 0.35), 0, 1)
        a = np.cos(t * np.pi / 2) ** 2
    else:
        a = np.ones(n)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    return a
```
