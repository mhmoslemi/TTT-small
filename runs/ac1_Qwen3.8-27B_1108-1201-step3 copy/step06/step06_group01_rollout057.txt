"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using convex optimization techniques.
Key insight: the problem is convex (max of quadratic forms over the simplex),
so global convergence is possible. Uses adaptive soft-max L-BFGS with
homotopy, mirror descent, and multi-length exploration.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    """Compute the objective score (normalized)."""
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    a_norm = a / s
    b = fast_convolve(a_norm)
    return 2.0 * len(a_norm) * b.max()


def smooth_obj_and_grad(a, T):
    """
    Compute soft-max smoothed objective and gradient.
    a: probability vector (sum=1, non-negative)
    T: temperature
    Returns: (smoothed_max_conv, gradient_wrt_a)
    """
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    shifted = (b - max_b) / T
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    lse = max_b + T * np.log(sum_exp)
    w = exp_shifted / sum_exp
    
    # Gradient: grad_a[j] = 2 * sum_m a[m] * w[m+j]
    # This is 2 * cross_correlation(a, w) where cross_corr[i] = sum_m a[m]*w[m+i]
    # Using FFT: cross_corr = ifft(FFT(a) * conj(FFT(w)))... 
    # Actually: sum_m a[m]*w[m+i] for i=0,...,n-1
    # = conv(a, w)[i + n - 1]... no
    # Let me use: flip_a = a[::-1], then conv(flip_a, w)[n-1+i] = sum_p flip_a[p]*w[n-1+i-p]
    # = sum_p a[n-1-p]*w[n-1+i-p]
    # Let q = n-1-p: sum_q a[q]*w[i+q] = cross_corr[i] ✓
    
    fft_size = 1 << (3 * n).bit_length()
    A = rfft(a, fft_size)
    W = rfft(w, fft_size)
    AF = rfft(a[::-1], fft_size)
    cross = irfft(AF * W, fft_size)
    grad_a = 2.0 * cross[n-1:2*n-1]
    
    return lse, grad_a


def simplex_project(v):
    """Project v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lbgfs_homotopy(a0, time_limit, T_start=0.01, T_end=1e-6, n_sweeps=15):
    """
    L-BFGS with geometric homotopy on temperature.
    Starts with high T (smooth) and decreases to low T (sharp).
    """
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    # Geometric schedule of temperatures
    T_schedule = np.geomspace(T_start, T_end, n_sweeps)
    
    for T in T_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        
        # Max iterations based on remaining time
        rem = time_limit - (time.monotonic() - t0)
        maxiter = max(500, int(rem * 100))
        
        def func(z):
            # z is in log-domain
            z_shifted = z - z.max()
            exp_z = np.exp(z_shifted)
            a_local = exp_z / exp_z.sum()
            obj, _ = smooth_obj_and_grad(a_local, T)
            return obj
        
        def jac(z):
            z_shifted = z - z.max()
            exp_z = np.exp(z_shifted)
            a_local = exp_z / exp_z.sum()
            obj, grad_a = smooth_obj_and_grad(a_local, T)
            # Chain rule: d obj / d z[j] = grad_a[j] * a[j] - a[j] * sum_i grad_a[i] * a[i]
            S = np.dot(grad_a, a_local)
            grad_z = a_local * (grad_a - a_local * S)
            return grad_z
        
        # Convert to log-domain
        a_safe = np.maximum(a, 1e-300)
        z0 = np.log(a_safe)
        
        try:
            res = minimize(
                fun=func,
                x0=z0,
                jac=jac,
                method='L-BFGS-B',
                options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12,
                         'maxfun': maxiter * 20, 'disp': False}
            )
            z_opt = res.x
            z_shifted = z_opt - z_opt.max()
            a_new = np.exp(z_shifted)
            a_new = a_new / a_new.sum()
        except Exception as e:
            print(f"  L-BFGS T={T:.2e} failed: {e}")
            a_new = a
        
        # Evaluate
        score_new = compute_score(a_new)
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
        
        a = a_new
        elapsed = time.monotonic() - t0
        print(f"  Homotopy T={T:.2e} ({elapsed:.0f}s): score={best_score:.6f}")
    
    return best_a, best_score


def mirror_descent(a0, time_limit, eta_start=0.1, eta_decay=0.999):
    """Mirror descent (multiplicative weight) on the simplex."""
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    a = a / s
    
    best_a = a.copy()
    best_score = compute_score(a)
    eta = eta_start
    t0 = time.monotonic()
    T = 0.001  # Fixed smooth temperature
    
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        obj, grad_a = smooth_obj_and_grad(a, T)
        
        # Mirror descent update
        a_new = a * np.exp(-eta * grad_a)
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        score = compute_score(a_new)
        it += 1
        
        if score < best_score - 1e-13:
            best_score = score
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
        else:
            no_improve += 1
        
        eta *= eta_decay
        if no_improve > 500:
            # Reset with perturbation
            a = best_a + np.random.default_rng(it).standard_normal(n) * 0.001
            a = np.maximum(a, 0)
            a = a / a.sum()
            eta = eta_start * 0.1
            T = T * 0.5
            no_improve = 0
        
        if it % 1000 == 0:
            elapsed = time.monotonic() - t0
            print(f"  MirrorDesc it={it} ({elapsed:.0f}s): score={best_score:.6f} eta={eta:.4f} T={T:.2e}")
    
    return best_a, best_score

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def peak_reduction(a0, time_limit, n_steps=5000):
    """
    Direct peak reduction: identify the peak lag and try to reduce it
    by adjusting the contributing coefficients.
    """
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    s = a.sum()
    a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    for step in range(n_steps):
        if time.monotonic() - t0 > time_limit:
            break
        
        b = fast_convolve(a)
        peak_k = int(np.argmax(b))
        peak_val = b[peak_k]
        
        # Find contributing pairs
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        # Find the top contributor
        top_i = idx[np.argmax(contribs)]
        
        # Try reducing top_i and distributing to others
        for factor in [0.9, 0.85, 0.8, 0.7]:
            new_val = a[top_i] * factor
            diff = a[top_i] - new_val
            if diff < 1e-15:
                continue
            
            a_trial = a.copy()
            a_trial[top_i] = new_val
            # Distribute diff to other coordinates proportionally
            mask = np.ones(n, dtype=bool)
            mask[top_i] = False
            others_sum = a[mask].sum()
            if others_sum < 1e-15:
                continue
            a_trial[mask] += diff * a[mask] / others_sum
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            
            score = compute_score(a_trial)
            if score < best_score - 1e-13:
                a = a_trial
                best_a = a.copy()
                best_score = score
                break
    
    return best_a, best_score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive L-BFGS with
geometric temperature homotopy, multi-start diversification, and
multi-length exploration. Exploits the structure of the auto-correlation
minimization problem with careful gradient computation via FFT.
"""

import numpy as np
import time
from scipy.fft import rfft, irfft
from scipy.optimize import minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def compute_score(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    a_norm = a / s
    b = fast_convolve(a_norm)
    return 2.0 * len(a_norm) * b.max()


def smooth_obj_and_grad(a, T):
    n = len(a)
    b = fast_convolve(a)
    max_b = b.max()
    shifted = (b - max_b) / T
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    lse = max_b + T * np.log(sum_exp)
    w = exp_shifted / sum_exp
    
    # grad of max_conv: grad_a[j] = 2 * sum_m a[m] * w[m+j]
    fft_size = 1 << (3 * n).bit_length()
    AF = rfft(a[::-1], fft_size)
    W = rfft(w, fft_size)
    cross = irfft(AF * W, fft_size)
    grad_conv = 2.0 * cross[n-1:2*n-1]
    
    # Full objective: 2*n*max_conv, so gradient is 2*n*grad_conv
    obj = 2.0 * n * lse
    grad = 2.0 * n * grad_conv
    
    return obj, grad


def softmax(z):
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def lbgfs_optimize(a0, T, maxiter=20000, time_check=None):
    """Run L-BFGS in log-domain for a given temperature."""
    n = len(a0)
    a_safe = np.maximum(a0, 1e-300)
    z0 = np.log(a_safe)
    
    def func(z):
        if time_check is not None and time_check():
            return 1e15
        a = softmax(z)
        obj, _ = smooth_obj_and_grad(a, T)
        return obj
    
    def jac(z):
        if time_check is not None and time_check():
            return np.zeros(n)
        a = softmax(z)
        _, grad_a = smooth_obj_and_grad(a, T)
        S = np.dot(grad_a, a)
        return a * (grad_a - S)
    
    try:
        res = minimize(
            fun=func, x0=z0, jac=jac,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                     'maxfun': maxiter * 30}
        )
        return softmax(res.x)
    except Exception:
        return a0


def lbgfs_homotopy(a0, time_limit, T_start=0.01, T_end=1e-7, n_sweeps=12):
    n = len(a0)
    a = np.maximum(a0, 0).astype(np.float64)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    t0 = time.monotonic()
    
    T_schedule = np.geomspace(T_start, T_end, n_sweeps)
    
    for i, T in enumerate(T_schedule):
        if time.monotonic() - t0 > time_limit:
            break
        rem = time_limit - (time.monotonic() - t0)
        maxiter = max(1000, int(rem * 50))
        
        deadline = [time.monotonic() + rem * 0.95]
        def time_check():
            return time.monotonic() > deadline[0]
        
        a_new = lbgfs_optimize(a, T, maxiter=maxiter, time_check=time_check)
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        score = compute_score(a_new)
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
        
        a = a_new
        print(f"    T={T:.1e}: score={best_score:.6f} (elapsed={time.monotonic()-t0:.0f}s)")
    
    return best_a, best_score
