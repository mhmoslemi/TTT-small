One more thought: **the "away step" Frank-Wolfe** might be very effective here. The idea is:
- Standard FW: move from `a` toward a vertex `e_j`
- Away step FW: if the current point has a specific element `a_i` that is "in the way" (i.e., removing it would decrease the objective), move away from `a_i` first, then toward the new vertex.

This can lead to much faster convergence.

For our problem:
- The "away" direction is: remove mass from coordinate `i` (the one whose removal most decreases the peak)
- The "toward" direction is: add mass to coordinate `j` (the one whose addition least increases the peak)

The combined direction is: `(e_j - e_i) / ||e_j - e_i||` projected onto the simplex.

Let me implement this:

```python
def away_step_fw_step(a, gradient, step_size):
    n = len(a)
    # Find the coordinate to move toward (min gradient)
    j = np.argmin(gradient)
    # Find the coordinate to move away from (max a[i] * gradient[i] among positive a[i])
    # Away step: remove from i where a[i]*gradient[i] is maximized (or where removing a[i] most helps)
    # Simple heuristic: i = argmax(a)
    support = np.where(a > 1e-10)[0]
    if len(support) == 0:
        return a, 0
    
    # For away step, we want to remove from the coordinate where the marginal cost of keeping it is highest
    # This is approximately a[i] * gradient[i] (the contribution of coordinate i to the gradient)
    # Or simply: the coordinate with the largest a[i] (since having mass there forces the peak)
    i = support[np.argmax(a[support])]
    
    if i == j:
        return a, 0
    
    # Move: a_new = a - step*a[i]/sum(a[support]) * e_i + step * e_j
    # But we need to stay in the simplex
    # Simple version: a_new = (1 - 2*step)*a + step*e_j + step*(a[i]/sum(a)) * (1 - e_i)
    # Actually, the standard away-step FW is:
    # 1. Move away: b = (a - step*a)/(1-step) (remove all mass proportionally, or just from i)
    # 2. Move toward: a_new = (1-step)*b + step*e_j
    
    # Simpler: just do a line search on the direction e_j - a[i]*e_i
    # a_new(t) = (1-t)*a + t*(e_j - a[i]*e_i) ... but this doesn't stay in simplex
    
    # Standard away-step FW (Lanckriet & Kuncheva 2007):
    # Step 1: a1 = (a - γ*a[s_i]*e_{s_i}) / (1 - γ*a[s_i])  (away step)
    # Step 2: a2 = (1-γ)*a1 + γ*e_j  (toward step)
    
    gamma = min(step_size, 0.5)
    a_i_val = a[i]
    if a_i_val > 1e-15:
        a1 = a.copy()
        a1[i] -= gamma * a_i_val
        a1 /= (1 - gamma * a_i_val)
    else:
        a1 = a
    
    a2 = (1 - gamma) * a1 + gamma * np.eye(n)[j]
    return a2, gamma
```

    if n == 1:

    rho = np.nonzero(cond)[0][-1]

def compute_smooth_obj_grad(f, tau):
    """Compute log-sum-exp objective and its gradient."""
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / tau
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    obj = tau * (np.log(w_sum) + scaled.max())  # = tau * log(sum exp(conv/tau))
    w = w / w_sum

    w_rev = w[::-1]

    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def adam_smooth(f0, time_limit, tau=None, lr=0.005, verbose=False):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    conv0 = fast_convolve(f)
    if tau is None:
        tau = 0.08 * conv0.max()

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    t = 0
    best_obj = conv0.max()
    best_f = f.copy()
    no_imp = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        t += 1
        conv = fast_convolve(f)
        scaled = conv / tau
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        w = w / w_sum

        w_rev = w[::-1]
        tl = n + len(w) - 1
        fs = 1 << (tl - 1).bit_length()
        A = np.fft.rfft(f, fs)
        W = np.fft.rfft(w_rev, fs)
        cr = np.fft.irfft(A * W, fs)
        grad = (2.0 / tau) * cr[n - 1:2 * n - 1][::-1]

        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        f = project_simplex(f - lr * m_hat / (np.sqrt(v_hat) + eps))

        M = fast_convolve(f).max()
        if M < best_obj - 1e-14:
            best_obj = M
            best_f = f.copy()
            no_imp = 0
        else:
            no_imp += 1

        tau = max(tau * 0.9993, 1e-8)

        if no_imp > 150:
            rng = np.random.default_rng((t * 7919 + 13) & 0xFFFFFFFF)
            noise = 0.002 + 0.005 * min(no_imp / 500.0, 2.0)
            f = project_simplex(best_f + rng.standard_normal(n) * noise)
            m = np.zeros(n)
            v = np.zeros(n)
            no_imp = 0
            tau = 0.04 * fast_convolve(f).max()
            lr = 0.003
            if verbose:
                print(f"        Adam restart t={t}: obj={2 * n * best_obj:.8f}")

        if verbose and t % 500 == 0:
            print(f"        Adam t={t}: obj={2 * n * best_obj:.8f}, tau={tau:.2e}")

    return best_f, 2 * n * best_obj


def coord_descent(f0, time_limit, verbose=False):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_imp = 0
    it = 0

    # Precompute shifted versions for reuse
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Get top peaks
        n_peaks = min(3, 2 * n - 1)
        peak_idx = np.argsort(conv)[-n_peaks:]

        improved = False
        for pk in peak_idx:
            pk = int(pk)
            js = max(0, pk - n + 1)
            je = min(n - 1, pk)
            if je < js:
                continue
            idx = np.arange(js, je + 1)
            products = f[idx] * f[pk - idx]
            j_star = int(idx[np.argmax(products)])

            # Find best target using linear approximation
            shifted_from = np.zeros(2 * n - 1)
            valid_from = np.arange(2 * n - 1) >= j_star
            inner_from = (np.arange(2 * n - 1) - j_star) < n
            mask_from = valid_from & inner_from
            shifted_from[mask_from] = f[np.arange(2 * n - 1)[mask_from] - j_star]

            rng_local = np.random.default_rng((it * 1009 + int(pk)) & 0xFFFFFFFF)
            if n > 300:
                candidates = rng_local.choice(n, size=min(80, n), replace=False)
            else:
                candidates = np.arange(n)

            best_j = -1
            best_new_max = M
            for jp in candidates:
                jp = int(jp)
                if jp == j_star:
                    continue
                if f[jp] > 0.95 * f[j_star]:
                    continue
                shifted_to = np.zeros(2 * n - 1)
                valid_to = np.arange(2 * n - 1) >= jp
                inner_to = (np.arange(2 * n - 1) - jp) < n
                mask_to = valid_to & inner_to
                shifted_to[mask_to] = f[np.arange(2 * n - 1)[mask_to] - jp]

                for frac in [0.1, 0.3, 0.5, 0.8]:
                    delta = frac * f[j_star]
                    if delta < 1e-13:
                        continue
                    new_conv = conv + 2.0 * delta * (shifted_from - shifted_to)
                    new_max = new_conv.max()
                    if new_max < best_new_max:
                        best_new_max = new_max
                        best_j = jp
                        best_frac = frac

            if best_j >= 0 and best_new_max < M - 1e-14:
                delta = best_frac * f[j_star]
                f[j_star] -= delta
                f[best_j] += delta
                f = np.maximum(f, 0)
                f /= f.sum()
                conv = fast_convolve(f)
                M = conv.max()
                if M < best_obj - 1e-14:
                    best_obj = M
                    best_f = f.copy()
                improved = True

        if not improved:
            no_imp += 1
            if no_imp > 10:
                rng = np.random.default_rng((it * 31 + 7) & 0xFFFFFFFF)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                no_imp = 0
        else:
            no_imp = 0

        if verbose and it % 100 == 0:
            print(f"        CD it={it}: obj={2 * n * best_obj:.8f}")

    return best_f, 2 * n * best_obj


def away_fw(f0, time_limit, verbose=False):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    n = len(f)
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    t0 = time.monotonic()
    it = 0
    gamma = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()
        peak = int(np.argmax(conv))

        # Gradient of (a*a)[peak]: grad[j] = 2*a[peak-j] if 0<=peak-j<n
        grad = np.zeros(n)
        js = max(0, peak - n + 1)
        je = min(n - 1, peak)
        grad[js:je + 1] = 2.0 * f[peak - np.arange(js, je + 1)]

        # Toward: argmin grad
        j_to = int(np.argmin(grad))
        # Away: argmax a[j] among support
        support = np.where(f > 1e-8)[0]
        if len(support) == 0:
            break
        j_away = int(support[np.argmax(f[support])])

        if j_to == j_away:
            gamma = max(gamma * 0.95, 0.01)
            continue

        # Away step + toward step
        a_away = f[j_away]
        g_use = min(gamma, 0.5 * a_away)
        if g_use < 1e-12:
            continue
        f1 = f.copy()
        f1[j_away] -= g_use
        f1 /= (1.0 - g_use)
        f2 = (1.0 - g_use) * f1 + g_use * np.zeros(n)
        f2[j_to] = g_use

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
            ft = f + alpha * (f2 - f)
            ft = np.maximum(ft, 0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft /= st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-6:
            f_new = f + best_alpha * (f2 - f)
            f_new = np.maximum(f_new, 0)
            st = f_new.sum()
            if st > 1e-15:
                f_new /= st
            val = fast_convolve(f_new).max()
            if val < best_obj - 1e-14:
                best_obj = val
                best_f = f_new.copy()
            f = f_new
            gamma = 1.0
        else:
            gamma = max(gamma * 0.9, 0.01)

        if verbose and it % 100 == 0:
            print(f"        AW it={it}: obj={2 * n * best_obj:.8f}, gamma={gamma:.3f}")

    return best_f, 2 * n * best_obj


def global_search_init(n, rng):
    """Generate a diverse random initialization."""
    strategy = rng.integers(0, 6)
    if strategy == 0:
        return make_asym_gaussian(n, 0.3 + 0.3 * rng.random(), 0.15 + 0.2 * rng.random())
    elif strategy == 1:
        return make_gaussian(n, 0.3 + 0.4 * rng.random(), 0.1 + 0.3 * rng.random())
    elif strategy == 2:
        return make_super_gaussian(n, 2 + 4 * rng.integers(0, 4))
    elif strategy == 3:
        return make_cosine_bell(n)
    elif strategy == 4:
        a = rng.dirichlet(np.ones(n) * 0.5)
        return a
    else:
        a = rng.dirichlet(np.ones(n) * 2.0)
        return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            return float('inf')
        a_norm = a / s
        try:
            score = float(evaluate_sequence(a_norm.tolist()))
        except Exception:
            return float('inf')
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
        return score

    # === Phase 1: Diverse initialization ===
    print("=== Phase 1: Initialization ===")
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = try_seq(parent, "parent")
            print(f"  Parent score: {ps:.6f}")
    except NameError:
        pass
    except Exception:
        pass

    all_inits = []
    if parent is not None:
        all_inits.append((parent, "parent"))

    for n_try in [512, 768, 1024]:
        for i in range(3):
            all_inits.append((global_search_init(n_try, rng), f"rand_n{n_try}_{i}"))
        all_inits.append((make_asym_gaussian(n_try, 0.35, 0.25), f"asym_n{n_try}"))
        all_inits.append((make_cosine_bell(n_try), f"cos_n{n_try}"))

    for a_init, label in all_inits:
        if time.monotonic() > deadline - 1500:
            break
        try_seq(a_init, label)

    if best_seq is None:
        best_seq = make_asym_gaussian(768, 0.35, 0.25)
        best_score = try_seq(best_seq, "fallback")

    print(f"  Best init: n={len(best_seq)}, score={best_score:.6f}")

    # === Phase 2: Main Adam optimization ===
    print("\n=== Phase 2: Adam Smooth (main) ===")
    rem = deadline - time.monotonic()
    t2 = min(rem, 600)
    if t2 > 50:
        print(f"  Time: {t2:.0f}s, n={len(best_seq)}")
        a_opt, obj = adam_smooth(best_seq, time_limit=t2 * 0.55, lr=0.005, verbose=True)
        try_seq(a_opt, "adam1")

        if time.monotonic() < deadline - 500:
            rem2 = min(deadline - time.monotonic() - 400, 200)
            if rem2 > 30:
                tau_val = 0.03 * fast_convolve(best_seq).max()
                a_opt2, obj2 = adam_smooth(best_seq, time_limit=rem2, tau=tau_val, lr=0.003, verbose=False)
                try_seq(a_opt2, "adam2")

    # === Phase 3: Coordinate descent ===
    print("\n=== Phase 3: Coordinate Descent ===")
    rem = deadline - time.monotonic() - 350
    if rem > 30:
        t3 = min(rem, 250)
        print(f"  Time: {t3:.0f}s, n={len(best_seq)}")
        a_opt, obj = coord_descent(best_seq, time_limit=t3, verbose=True)
        try_seq(a_opt, "cd")

    # === Phase 4: Away-step FW ===
    print("\n=== Phase 4: Away-step FW ===")
    rem = deadline - time.monotonic() - 250
    if rem > 30:
        t4 = min(rem, 150)
        print(f"  Time: {t4:.0f}s, n={len(best_seq)}")
        a_opt, obj = away_fw(best_seq, time_limit=t4, verbose=True)
        try_seq(a_opt, "away_fw")

    # === Phase 5: Multi-length exploration ===
    print("\n=== Phase 5: Multi-Length Exploration ===")
    if best_seq is not None:
        for target_n in [1024, 1280, 640, 512]:
            rem = deadline - time.monotonic() - 150
            if rem < 40:
                break
            f_interp = interpolate_sequence(best_seq, target_n)
            sc = try_seq(f_interp, f"interp_{target_n}")
            t5 = min(rem * 0.35, 100)
            if t5 > 20:
                a_opt, obj = adam_smooth(f_interp, time_limit=t5, lr=0.004, verbose=False)
                try_seq(a_opt, f"adam_{target_n}")
                rem2 = deadline - time.monotonic() - 100
                if rem2 > 20:
                    a_opt2, _ = coord_descent(a_opt, time_limit=min(rem2, 50), verbose=False)
                    try_seq(a_opt2, f"cd_{target_n}")

    # === Phase 6: Fresh exploration with random restarts ===
    print("\n=== Phase 6: Random Restart Exploration ===")
    rem = deadline - time.monotonic() - 100
    if rem > 60:
        n_explore = len(best_seq)
        for trial in range(5):
            rem_check = deadline - time.monotonic() - 80
            if rem_check < 40:
                break
            a_rand = project_simplex(best_seq + rng.standard_normal(n_explore) * 0.03)
            t6 = min(rem_check * 0.3, 80)
            if t6 > 15:
                a_opt, _ = adam_smooth(a_rand, time_limit=t6, lr=0.005, verbose=False)
                try_seq(a_opt, f"restart_{trial}")

    # === Phase 7: Final polish ===
    print("\n=== Phase 7: Final Polish ===")
    rem = deadline - time.monotonic() - 10
    if rem > 15 and best_seq is not None:
        t7 = min(rem, 60)
        tau_final = 0.02 * fast_convolve(best_seq).max()
        a_opt, _ = adam_smooth(best_seq, time_limit=t7, tau=tau_final, lr=0.002, verbose=False)
        try_seq(a_opt, "polish_adam")

        rem2 = deadline - time.monotonic() - 5
        if rem2 > 10 and best_seq is not None:
            a_opt2, _ = coord_descent(best_seq, time_limit=min(rem2, 30), verbose=False)
            try_seq(a_opt2, "polish_cd")

    # Fallback
    if best_seq is None:
        best_seq = make_asym_gaussian(768, 0.35, 0.25)
        best_score = try_seq(best_seq, "final_fallback")

    # Final verification
    final_score = try_seq(best_seq, "final_check")
    print(f"\nFINAL: score={best_score:.6f}, n={len(best_seq)}")
    return [float(x) for x in best_seq]
