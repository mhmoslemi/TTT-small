def spectral_flattening(a, time_limit, alpha=0.1):
    n = len(a)
    a = a / a.sum()
    N = 1 << (2*n-2).bit_length()  # Next power of 2 >= 2n-1
    
    A = np.fft.rfft(a, N)
    A_mag = np.abs(A)
    A_phase = A / np.where(A_mag > 0, A_mag, 1)
    
    best_obj = float('inf')
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    current_alpha = alpha
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Compute autocorrelation
        power = A_mag ** 2
        c = np.fft.irfft(power, N)[:2*n-1]
        c = c / N  # Normalization
        peak = c.max()
        obj = 2 * n * peak
        
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()
        
        # Update magnitude to flatten spectrum
        # Where c is large, reduce power; where c is small, increase power
        # Use exponential weighting
        c_clipped = np.maximum(c, 1e-15)
        update = np.exp(-current_alpha * c_clipped / peak)
        A_mag = A_mag * np.sqrt(update)
        
        # Reconstruct a
        A_new = A_phase * A_mag
        a_new = np.fft.irfft(A_new, N)[:n]
        
        # Handle negative entries
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using mirror descent with asymmetric

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, peak_indices, weights):
    """Compute weighted subgradient of max_k (a*a)_k."""
    n = len(a)
    g = np.zeros(n)
    for idx in range(len(peak_indices)):
        k = int(peak_indices[idx])
        w_k = weights[idx]
        if w_k < 1e-15:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je < js:
            continue
        g[js:je + 1] += 2.0 * w_k * a[k - np.arange(js, je + 1)]
    return g


def spectral_flatten_step(a, alpha=0.5):
    """
    Apply a spectral flattening step: modify the power spectrum to be flatter,
    then project back to non-negativity.
    """
    n = len(a)
    fft_size = 1 << (2 * n - 2).bit_length()
    
    A = np.fft.rfft(a, fft_size)
    A_mag = np.abs(A)
    A_phase = np.angle(A)
    
    # Compute current autocorrelation
    power = A_mag ** 2
    c = np.fft.irfft(power, fft_size)[:2 * n - 1]
    c = c / fft_size
    peak = c.max()
    
    if peak < 1e-15:
        return a.copy()
    
    # Flattening: reduce power where autocorrelation is high
    c_clipped = np.maximum(c, 1e-15)
    # Desired spectrum is proportional to 1/c (more power where autocorr is low)
    # Use a soft version
    update = np.exp(-alpha * c_clipped / peak)
    A_mag_new = A_mag * np.sqrt(update)
    
    # Reconstruct
    A_new = A_mag_new * np.exp(1j * A_phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    
    # Project to non-negativity
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    a_new = a_new / s
    
    return a_new

def make_bartlett_asymmetric(n, asym=0.05):
    """Asymmetric Bartlett (triangular) window."""
    center = n / 2.0 + n * asym

    a = np.maximum(a, 0)

    return a
