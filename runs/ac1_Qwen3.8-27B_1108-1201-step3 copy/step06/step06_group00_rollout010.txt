import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize_scalar
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def eval_local(a):
    """Compute objective 2n*max(conv)/(sum)^2 locally."""
    n = len(a)
    S = a.sum()
    if S < 0.01:
        return float('inf')
    b = fast_convolve(a)
    return float(2.0 * n * b.max() / (S * S))


def subgradient_at(a, b, temp, n_top=None):
    """
    Compute subgradient of max_k b_k with temperature weighting.
    Returns gradient vector g of length n.
    """
    n = len(a)
    scaled = b / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    
    if n_top is not None and n_top < len(b):
        top_idx = np.argpartition(b, -n_top)[-n_top:]
        w_sel = w[top_idx]
        w_sel /= w_sel.sum()
        w_full = np.zeros(len(b))
        w_full[top_idx] = w_sel
        w = w_full
    else:
        w /= w.sum()
    
    g = np.zeros(n)
    for k in range(len(b)):
        if w[k] < 1e-14:
            continue
        if k == 0:
            g += w[k] * 2.0 * a
        else:
            # db_k/da_i = a_{i+k} (if valid) + a_{i-k} (if valid)
            for i in range(n - k):
                g[i] += w[k] * a[i + k]
                g[i + k] += w[k] * a[i]
    return g


def subgradient_fast(a, b, temp, n_top=None):
    """
    Fast subgradient computation using FFT for cross-correlation.
    """
    n = len(a)
    scaled = b / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    
    if n_top is not None and n_top < len(b):
        top_idx = np.argpartition(b, -n_top)[-n_top:]
        w_sel = w[top_idx]
        w_sel /= w_sel.sum()
        w_full = np.zeros(len(b))
        w_full[top_idx] = w_sel
        w = w_full
    else:
        w /= w.sum()
    
    # For k=0: contribution is 2*a*w[0]
    g = np.zeros(n)
    if w[0] > 1e-14:
        g += 2.0 * w[0] * a
    
    # For k>0: g_i += w[k] * (a_{i+k} + a_{i-k}) with bounds
    # This is: g += w_cross_correlate
    # g_i = sum_k w[k] * a[i+k] + sum_k w[k] * a[i-k]
    #      = (w * a)_i + (w * a_flipped)_i (with appropriate shifts)
    
    # Use FFT for the cross-correlation
    # sum_k w[k] * a[i+k] for valid indices = cross-correlation of w and a
    # Actually, let's just do it directly with numpy for clarity
    # g[i] = sum_{k: i+k<n} w[k]*a[i+k] + sum_{k: i-k>=0} w[k]*a[i-k]
    #      = (w reverse-convolve a)[i] + (w convolve a)[i] (partial)
    
    # Simpler: for each k, add w[k] * shift(a, k) + w[k] * shift(a, -k)
    # Using FFT:
    w_pos = np.zeros(2 * n)
    w_pos[:n] = w[1:n]  # positive lags
    w_neg = np.zeros(2 * n)
    w_neg[:n] = w[1:n][::-1]  # negative lags (reversed)
    
    # cross_corr1[i] = sum_k w[k] * a[i+k] (k from 1 to n-1)
    # This is (w * flip(a))[i] shifted... 
    # Actually: sum_k w[k] a[i+k] = (w * a)_i where * is cross-correlation
    # = ifft(conj(fft(w)) * fft(a))
    
    # Let's just do it with a loop over nonzero w values (typically few after top-k)
    nonzero_k = np.where(w > 1e-12)[0]
    for k in nonzero_k:
        if k == 0:
            continue
        wk = w[k]
        # g[i] += wk * a[i+k] for i = 0..n-1-k
        g[:n-k] += wk * a[k:]
        # g[i] += wk * a[i-k] for i = k..n-1
        g[k:] += wk * a[:n-k]
    
    return g


def mirror_descent_optimize(a0, time_limit, lr_init=0.5, temp_init=0.001,
                            temp_growth=1.01, n_top=None, seed=42,
                            kick_interval=500, kick_size=0.002,
                            lr_decay=0.7):
    """
    Mirror descent (multiplicative weights) for minimizing max autocorrelation.
    Works on simplex (sum=1, a>=0).
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    a /= a.sum()
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    cur_lr = lr_init
    cur_temp = temp_init
    no_improve = 0
    it_count = 0
    
    while time.monotonic() - t0 < time_limit:
        # Batch of 20 iterations
        batch_end = it_count + 20
        while it_count < batch_end and time.monotonic() - t0 < time_limit:
            it_count += 1
            b = fast_convolve(a)
            b_max = b.max()
            
            if b_max < best_obj - 1e-14:
                best_obj = b_max
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            # Compute subgradient with temperature weighting
            g = subgradient_fast(a, b, cur_temp, n_top)
            
            # Multiplicative weights update: a_i *= exp(-lr * g_i)
            # Normalize to maintain sum=1
            exponent = -cur_lr * g
            exponent -= exponent.max()  # numerical stability
            multiplier = np.exp(exponent)
            a = a * multiplier
            a_sum = a.sum()
            if a_sum < 1e-15:
                a = best_a.copy()
                break
            a /= a_sum
            # Ensure non-negativity
            a = np.maximum(a, 0.0)
            a /= a.sum()
        
        # Periodic adjustments
        if no_improve > kick_interval:
            # Restart from best with noise
            a = best_a + rng.standard_normal(n) * kick_size
            a = np.maximum(a, 0.0)
            a /= a.sum()
            cur_lr = max(cur_lr * lr_decay, lr_init * 0.05)
            no_improve = 0
        elif no_improve > kick_interval // 3:
            cur_lr *= 0.92
        
        # Grow temperature
        cur_temp *= temp_growth
        if cur_temp > 0.3:
            cur_temp = 0.3
    
    return best_a, best_obj


def coordinate_descent_exact(a0, time_limit, n_sweeps_max=100):
    """
    Exact coordinate descent: for each coordinate, use golden section search
    to find the optimal value (minimizing the full objective).
    """
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return a0, float('inf')
    a = a / S
    n = len(a)
    
    best_a = a.copy()
    best_obj = eval_local(a)
    
    for sweep in range(n_sweeps_max):
        if time.monotonic() - t0 > time_limit:
            break
        
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            
            a_j_old = a[j]
            if a_j_old < 1e-15:
                # Try small values
                candidates = [a.sum() * 0.001, a.sum() * 0.01]
            else:
                # Golden section search around current value
                # Search range: [0, 2*a_j_old] or broader if at boundary
                if a_j_old < 0.01:
                    lo, hi = 0.0, a.sum() * 0.5
                elif a_j_old > 0.5:
                    lo, hi = a.sum() * 0.1, min(2.0 * a_j_old, 0.99 * a.sum())
                else:
                    lo, hi = max(0.0, a_j_old * 0.1), min(2.0 * a_j_old, 0.99 * a.sum())
            
            def obj_1d(x):
                a_trial = a.copy()
                a_trial[j] = x
                s = a_trial.sum()
                if s < 1e-10:
                    return 1e10
                b = fast_convolve(a_trial)
                return 2.0 * n * b.max() / (s * s)
            
            try:
                res = minimize_scalar(obj_1d, bounds=(lo, hi), method='bounded',
                                      options={'xatol': 1e-10, 'maxiter': 50})
                if res.success and res.fun < best_obj - 1e-12:
                    a[j] = res.x
                    a = np.maximum(a, 0.0)
                    s = a.sum()
                    if s < 1e-10:
                        a = best_a.copy()
                        break
                    a /= s
                    best_obj = res.fun
                    best_a = a.copy()
                    improved = True
            except Exception:
                pass
        
        if not improved and sweep > 2:
            break
    
    return best_a, best_obj


def coordinate_descent_fast(a0, time_limit):
    """
    Faster coordinate descent using the analytical structure:
    b_k is linear in a_j for k>0, quadratic for k=0.
    Use this to compute the objective analytically in 1D.
    """
    t0 = time.monotonic()
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    S = a.sum()
    if S < 1e-15:
        return a0, float('inf')
    a = a / S
    n = len(a)
    
    best_a = a.copy()
    best_obj = eval_local(a)
    b = fast_convolve(a)
    
    sweep = 0
    while time.monotonic() - t0 < time_limit and sweep < 200:
        sweep += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            
            a_j = a[j]
            
            # Precompute T1_k for this j (coefficient of x in b_k when a_j=x)
            # b_k_new(x) = b_k + (x - a_j) * T1_k for k > 0
            # b_0_new(x) = b_0 - a_j^2 + x^2
            T1 = np.zeros(len(b))
            for k in range(1, n):
                val = 0.0
                if j + k < n:
                    val += a[j + k]
                if j - k >= 0:
                    val += a[j - k]
                T1[k] = val
            
            b0_base = b[0] - a_j * a_j
            
            def obj_analytic(x):
                S_new = S - a_j + x
                if S_new < 1e-10:
                    return 1e10
                # b_0
                b0 = b0_base + x * x
                # b_k for k > 0
                max_bk = b0
                for k in range(1, n):
                    if T1[k] == 0.0:
                        bk = b[k]
                    else:
                        bk = b[k] + (x - a_j) * T1[k]
                    if bk > max_bk:
                        max_bk = bk
                return 2.0 * n * max_bk / (S_new * S_new)
            
            # Determine search bounds
            if a_j < 1e-8:
                lo, hi = 0.0, max(0.1 * S, 1e-6)
            else:
                lo = max(0.0, a_j * 0.3)
                hi = min(0.9 * S, a_j * 3.0 + 0.01 * S)
                if hi - lo < 1e-10:
                    lo = max(0.0, a_j - 0.01 * S)
                    hi = min(0.99 * S, a_j + 0.01 * S)
            
            try:
                res = minimize_scalar(obj_analytic, bounds=(lo, hi), method='bounded',
                                      options={'xatol': 1e-10, 'maxiter': 40})
                if res.fun < best_obj - 1e-12:
                    a[j] = res.x
                    S = a.sum()
                    a = np.maximum(a, 0.0)
                    a /= a.sum()
                    S = a.sum()
                    b = fast_convolve(a)
                    best_obj = eval_local(a)
                    best_a = a.copy()
                    improved = True
            except Exception:
                pass
        
        if not improved and sweep > 3:
            break
    
    return best_a, best_obj


def make_fejer_init(n):
    """Fejér window (sinc-squared) initialization."""
    x = np.arange(n)
    center = (n - 1) / 2.0
    a = np.sinc((x - center) / (n / 4.0)) ** 2
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_prolate_init(n, B=0.3):
    """
    Approximate prolate spheroidal wave function using iterative DFT.
    B is the bandwidth parameter (fraction of Nyquist).
    """
    a = np.ones(n) / n
    iters = 20
    for _ in range(iters):
        # DFT
        A = rfft(a, 2 * n)
        # Mask to bandwidth
        idx = np.arange(len(A))
        mask = idx <= B * n
        A[~mask] = 0
        a = np.maximum(irfft(A, 2 * n)[:n], 0)
        s = a.sum()
        if s < 1e-15:
            a = np.ones(n) / n
        else:
            a /= s
    return a


def make_comb_init(n, period=7, duty=0.3):
    """Comb-like initialization with duty cycle."""
    a = np.zeros(n)
    width = max(1, int(duty * period))
    for start in range(0, n, period):
        end = min(start + width, n)
        a[start:end] = 1.0
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_gaussian_init(n, sigma_frac=0.3):
    """Gaussian-shaped initialization."""
    x = np.linspace(0, 1, n)
    a = np.exp(-0.5 * ((x - 0.5) / sigma_frac) ** 2)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_cosine_init(n):
    """Raised cosine window."""
    x = np.linspace(0, 1, n)
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_smoothstep_init(n):
    """Smoothstep (6th order polynomial) window."""
    x = np.linspace(0, 1, n)
    # 6th order smoothstep: 31x^6 - 90x^5 + 84x^4 - 20x^3
    t = np.clip(x, 0, 1)
    a = 31 * t**6 - 90 * t**5 + 84 * t**4 - 20 * t**3
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_symmetric_init(n, shape):
    """Create symmetric (palindromic) initialization."""
    half = n // 2
    if shape == 'gauss':
        x = np.linspace(0, 0.5, half)
        h = np.exp(-0.5 * (x / 0.25) ** 2)
    elif shape == 'linear':
        h = np.linspace(0.3, 1.0, half)
    elif shape == 'quadratic':
        x = np.linspace(0, 0.5, half)
        h = 1.0 - 4.0 * x * x
    else:
        h = np.ones(half) * 0.5
    h = np.maximum(h, 1e-10)
    if n % 2 == 0:
        a = np.concatenate([h[::-1], h])
    else:
        a = np.concatenate([h[::-1], h[1:]])
    return a / a.sum()


def peak_chasing_optimize(a0, time_limit, n_peaks=5, seed=42):
    """
    Peak chasing: explicitly target the highest autocorrelation peaks
    and use Newton-like steps to reduce them.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    n = len(a0)
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 1e-15)
    a /= a.sum()
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    lr = 0.01
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        b = fast_convolve(a)
        b_max = b.max()
        
        if b_max < best_obj - 1e-14:
            best_obj = b_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Find top peaks
        top_k = np.argsort(b)[-n_peaks:]
        
        # Compute gradient as average over top peaks
        g = np.zeros(n)
        for k in top_k:
            if k == 0:
                g += 2.0 * a
            else:
                for i in range(n - k):
                    g[i] += a[i + k]
                    g[i + k] += a[i]
        g /= n_peaks
        
        # Project gradient onto tangent space of simplex
        # (subtract mean for positive components)
        positive = a > 1e-10
        if positive.sum() > 0:
            g_proj = g - g[positive].mean()
        else:
            g_proj = g
        
        # Step
        a_new = a - lr * g_proj
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a = best_a + rng.standard_normal(n) * 0.001
            a = np.maximum(a, 0.0)
            a /= a.sum()
            lr *= 0.5
            continue
        a_new /= s
        a = a_new
        
        # Adaptive learning rate
        if no_improve > 200:
            lr *= 0.7
            if lr < 1e-6:
                # Restart
                a = best_a + rng.standard_normal(n) * 0.003
                a = np.maximum(a, 0.0)
                a /= a.sum()
                lr = 0.01
                no_improve = 0
        elif no_improve > 50:
            lr *= 0.95
    
    return best_a, best_obj


def pairwise_coordinate_descent(a0, time_limit, n_pairs=20):
    """
    Pairwise coordinate descent: optimize pairs of coordinates together.
    For each pair (i,j), solve a 2D subproblem.
    """
    t0 = time.monotonic()
    rng = np.random.default_rng(123)
    a = a0.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    n = len(a)
    best_a = a.copy()
    best_obj = eval_local(a)
    
    sweep = 0
    while time.monotonic() - t0 < time_limit and sweep < 50:
        sweep += 1
        improved = False
        
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit:
                return best_a, best_obj
            
            # Random pair
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            
            a_i, a_j = a[i], a[j]
            
            def obj_2d(params):
                x, y = params
                a_trial = a.copy()
                a_trial[i] = x
                a_trial[j] = y
                s = a_trial.sum()
                if s < 1e-10:
                    return 1e10
                a_trial = np.maximum(a_trial, 0)
                b = fast_convolve(a_trial)
                return 2.0 * n * b.max() / (s * s)
            
            # Bounds
            lo = [max(0, a_i * 0.2), max(0, a_j * 0.2)]
            hi = [min(0.9, a_i * 3 + 0.05), min(0.9, a_j * 3 + 0.05)]
            
            try:
                res = minimize_scalar(
                    lambda t: obj_2d([a_i + t * (hi[0] - lo[0]) - a_i,
                                       a_j + t * (hi[1] - lo[1]) - a_j]),
                    bounds=(0, 1), method='bounded', options={'xatol': 1e-6}
                )
                # Actually, let's just do a simple grid + local search
                # 3x3 grid around current
                best_local = (a_i, a_j)
                best_local_val = obj_2d([a_i, a_j])
                for di in [-0.5, -0.2, 0, 0.2, 0.5]:
                    for dj in [-0.5, -0.2, 0, 0.2, 0.5]:
                        xi = max(0, a_i * (1 + di))
                        xj = max(0, a_j * (1 + dj))
                        val = obj_2d([xi, xj])
                        if val < best_local_val:
                            best_local_val = val
                            best_local = (xi, xj)
                
                if best_local_val < best_obj - 1e-12:
                    a[i] = best_local[0]
                    a[j] = best_local[1]
                    a = np.maximum(a, 0.0)
                    a /= a.sum()
                    best_obj = eval_local(a)
                    best_a = a.copy()
                    improved = True
            except Exception:
                pass
        
        if not improved and sweep > 3:
            break
    
    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length preserving the sum."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def propose_candidate(seed=42, budget_s=1800.0):
    """
    Multi-phase optimization:
    1. Diverse initialization + mirror descent
    2. Peak chasing
    3. Exact coordinate descent
    4. Multi-length exploration
    5. Final pairwise polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    
    best_seq = None
    best_score = float('inf')
    
    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
            best_seq = parent.copy()
            best_score = p_score
    except Exception:
        pass
    
    if parent is None:
        parent = np.ones(1024) / 1024
        print("No parent, using uniform n=1024")
        best_seq = parent.copy()
        best_score = eval_local(parent)
    
    n_parent = len(parent)
    
    # === Phase 1: Mirror descent from parent ===
    print(f"\n=== Phase 1: Mirror descent (n={n_parent}) ===")
    rem = deadline - time.monotonic()
    t_alloc = min(rem * 0.25, 300.0)
    if t_alloc > 30:
        print(f"  t={t_alloc:.0f}s")
        a_md, _ = mirror_descent_optimize(
            parent, time_limit=t_alloc,
            lr_init=0.3, temp_init=0.0005, temp_growth=1.008,
            n_top=max(10, n_parent // 15),
            kick_interval=300, kick_size=0.002,
            seed=seed
        )
        try_seq(a_md, "mirror_parent")
    
    # === Phase 2: Peak chasing ===
    print(f"\n=== Phase 2: Peak chasing (score={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic()
        t_alloc = min(rem * 0.2, 200.0)
        if t_alloc > 20:
            print(f"  t={t_alloc:.0f}s")
            a_pc, _ = peak_chasing_optimize(
                best_seq, time_limit=t_alloc,
                n_peaks=5, seed=seed + 1
            )
            try_seq(a_pc, "peak_chasing")
    
    # === Phase 3: Exact coordinate descent ===
    print(f"\n=== Phase 3: Coordinate descent (score={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t_alloc = min(rem * 0.25, 400.0)
        if t_alloc > 30:
            print(f"  Exact CD: t={t_alloc * 0.5:.0f}s")
            a_cd, _ = coordinate_descent_fast(best_seq, time_limit=t_alloc * 0.5)
            try_seq(a_cd, "coord_descent_fast")
            
            rem = deadline - time.monotonic()
            if rem > 20:
                t_cd2 = min(rem * 0.4, 120.0)
                print(f"  Exact CD2: t={t_cd2:.0f}s")
                a_cd2, _ = coordinate_descent_exact(best_seq, time_limit=t_cd2)
                try_seq(a_cd2, "coord_descent_exact")
    
    # === Phase 4: Diverse initializations + mirror descent ===
    print(f"\n=== Phase 4: Diverse initializations (score={best_score:.6f}) ===")
    if best_seq is not None:
        n_base = len(best_seq)
    else:
        n_base = n_parent
    
    initializations = {}
    initializations["fejer"] = make_fejer_init(n_base)
    initializations["prolate_0.2"] = make_prolate_init(n_base, B=0.2)
    initializations["prolate_0.3"] = make_prolate_init(n_base, B=0.3)
    initializations["prolate_0.4"] = make_prolate_init(n_base, B=0.4)
    initializations["smoothstep"] = make_smoothstep_init(n_base)
    initializations["comb_7"] = make_comb_init(n_base, period=7, duty=0.3)
    initializations["comb_5"] = make_comb_init(n_base, period=5, duty=0.4)
    initializations["gauss_0.25"] = make_gaussian_init(n_base, 0.25)
    initializations["gauss_0.35"] = make_gaussian_init(n_base, 0.35)
    initializations["sym_gauss"] = make_symmetric_init(n_base, 'gauss')
    initializations["sym_quad"] = make_symmetric_init(n_base, 'quadratic')
    
    # Perturbed parent
    initializations["parent_perturb"] = np.maximum(
        best_seq + rng.standard_normal(n_base) * 0.02, 0)
    initializations["parent_perturb"] /= initializations["parent_perturb"].sum()
    
    for name, init_a in initializations.items():
        rem = deadline - time.monotonic()
        if rem < 15:
            break
        t_alloc = min(rem * 0.08, 30.0)
        if t_alloc < 5:
            continue
        print(f"  {name}: t={t_alloc:.0f}s", end=" ")
        a_opt, _ = mirror_descent_optimize(
            init_a, time_limit=t_alloc,
            lr_init=0.5, temp_init=0.001, temp_growth=1.02,
            n_top=max(5, n_base // 30),
            kick_interval=100, kick_size=0.005,
            seed=seed + hash(name) % 100000
        )
        score = try_seq(a_opt, f"div_{name}")
        print(f"-> {score:.6f}")
    
    # === Phase 5: Multi-length exploration ===
    print(f"\n=== Phase 5: Multi-length (score={best_score:.6f}) ===")
    if best_seq is not None:
        target_lengths = set()
        for factor in [0.5, 0.6, 0.75, 0.8, 1.1, 1.2, 1.33, 1.5, 2.0]:
            tl = int(n_parent * factor)
            tl = max(256, min(3000, tl))
            if tl != n_parent and tl not in target_lengths:
                target_lengths.add(tl)
        
        target_lengths = sorted(target_lengths, key=lambda x: abs(x - n_parent))
        
        for target_n in target_lengths[:5]:
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            
            f_interp = interpolate_sequence(best_seq, target_n)
            t_alloc = min(rem * 0.12, 60.0)
            if t_alloc < 10:
                continue
            print(f"  n={target_n}: t={t_alloc:.0f}s", end=" ")
            a_opt, _ = mirror_descent_optimize(
                f_interp, time_limit=t_alloc,
                lr_init=0.3, temp_init=0.0005, temp_growth=1.01,
                n_top=max(5, target_n // 20),
                kick_interval=200, kick_size=0.003,
                seed=seed + target_n
            )
            score = try_seq(a_opt, f"len_{target_n}")
            print(f"-> {score:.6f}")
    
    # === Phase 6: Coordinate descent on best ===
    print(f"\n=== Phase 6: Final coordinate descent (score={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t_cd = min(rem * 0.3, 300.0)
        if t_cd > 20:
            print(f"  CD: t={t_cd:.0f}s")
            a_cd, _ = coordinate_descent_fast(best_seq, time_limit=t_cd)
            try_seq(a_cd, "final_cd")
    
    # === Phase 7: Pairwise polish ===
    print(f"\n=== Phase 7: Pairwise polish (score={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t_pw = min(rem * 0.2, 120.0)
        if t_pw > 15:
            print(f"  Pairwise: t={t_pw:.0f}s")
            a_pw, _ = pairwise_coordinate_descent(best_seq, time_limit=t_pw)
            try_seq(a_pw, "pairwise")
            
            # Another round of CD
            rem = deadline - time.monotonic()
            if rem > 15:
                t_cd2 = min(rem * 0.5, 60.0)
                a_cd2, _ = coordinate_descent_fast(best_seq, time_limit=t_cd2)
                try_seq(a_cd2, "cd_after_pw")
    
    # === Phase 8: Final perturbation + reoptimize ===
    print(f"\n=== Phase 8: Final attempts (score={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for attempt in range(5):
            rem = deadline - time.monotonic()
            if rem < 10:
                break
            noise_scale = 0.005 / (attempt + 1)
            a_pert = np.maximum(best_seq + rng.standard_normal(n_best) * noise_scale, 0)
            a_pert /= a_pert.sum()
            t_alloc = min(rem * 0.3, 30.0)
            if t_alloc < 5:
                break
            # Quick mirror descent + coordinate descent
            a_md, _ = mirror_descent_optimize(
                a_pert, time_limit=t_alloc * 0.5,
                lr_init=0.1, temp_init=0.0002, temp_growth=1.005,
                n_top=max(5, n_best // 20),
                kick_interval=50, kick_size=0.001,
                seed=seed + attempt * 1000
            )
            try_seq(a_md, f"perturb_{attempt}")
            
            rem = deadline - time.monotonic()
            if rem > 5:
                a_cd, _ = coordinate_descent_fast(a_md, time_limit=min(rem * 0.4, 15.0))
                try_seq(a_cd, f"perturb_{attempt}_cd")
    
    # Final verification
    if best_seq is None:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = parent.copy()
        best_score = float(evaluate_sequence(parent.tolist()))
    
    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    print(f"{'='*50}")
    
    return [float(x) for x in best_seq]
