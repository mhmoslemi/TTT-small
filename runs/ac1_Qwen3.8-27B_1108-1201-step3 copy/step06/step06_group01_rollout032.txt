def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1

    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]

    """Build the (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T

        try:

            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)

                _lp_solver_cache = solver_name
                return solver_name

    # cross_corr[k] = sum_d a[d] * w_full[k+d] = ifft(conj(a_fft) * w_fft)[k]? 
    # Let me verify: ifft(conj(A) * W)[k] = (1/(2N)) sum_m conj(A[m]) W[m] exp(2πikm/(2N))
    # A[m] = sum_d a[d] exp(-2πidm/(2N))
    # conj(A[m]) = sum_d a[d] exp(2πidm/(2N)) (since a is real)
    # W[m] = sum_e w[e] exp(-2πiem/(2N))
    # Product = sum_de a[d] w[e] exp(2πidm/(2N)) exp(-2πiem/(2N))
    # = sum_de a[d] w[e] exp(2πim(d-e)/(2N))
    # ifft gives: (1/(2N)) sum_m [sum_de a[d]w[e] exp(2πim(d-e)/(2N))] exp(2πikm/(2N))
    # = sum_de a[d]w[e] (1/(2N)) sum_m exp(2πim(d-e+k)/(2N))
    # = sum_de a[d]w[e] * delta_{d-e+k, 0 mod 2N}
    # = sum_d a[d] * w[d+k] (when d+k < 2N, which is our case)
    # So ifft(conj(A)*W)[k] = sum_d a[d]*w[d+k] = cross_corr[k] ✓
    
    cross = irfft(np.conj(a_fft) * w_fft, 2 * N)[:n]
    grad = 2.0 * cross
    
    # Now we need the gradient of the power mean, not just the weighted sum.
    # power_mean = b_max * (sum (b/b_max)^p)^(1/p)
    # d power_mean / d a = b_max * (1/p) * (sum (b/b_max)^p)^(1/p - 1) * p * sum_d (b_d/b_max)^(p-1) * (1/b_max) * (db_d/da)
    # = (sum (b/b_max)^p)^(1/p - 1) * sum_d (b_d/b_max)^(p-1) * (db_d/da)
    # = (S)^(1/p-1) * sum_d (b_d/b_max)^(p-1) * (db_d/da) where S = sum (b/b_max)^p
    # Note that w_d = (b_d/b_max)^p / S, so (b_d/b_max)^(p-1) = w_d * b_max / b_d
    # Hmm, let me use a cleaner formulation.
    # 
    # F = (sum_i b_i^p)^(1/p)
    # dF/da_j = (1/p) (sum b_i^p)^(1/p - 1) * p * sum_i b_i^(p-1) * db_i/da_j
    # = (sum b_i^p)^(1/p-1) * sum_i b_i^(p-1) * db_i/da_j
    # 
    # Let c_i = b_i^(p-1) / (sum b_i^p)^(1-1/p)
    # Then dF/da_j = sum_i c_i * db_i/da_j
    # And the "effective" weights for the correlation are c_i.
    # 
    # So: grad = 2 * cross_corr(c_padded, a) where c_i = b_i^(p-1) / (sum b_i^p)^(1-1/p)
    
    b_safe = np.maximum(b, 0)
    b_p = b_safe ** p
    S = b_p.sum()
    if S < 1e-30:
        return np.zeros(n)
    c = b_safe ** (p - 1) / S ** (1 - 1.0 / p)
    # Now grad[j] = 2 * sum_i c[i] * a[i-j] for valid i-j
    # = 2 * sum_d a[d] * c[j+d] (substituting d=i-j)
    # Same cross-correlation structure!
    
    c_full = c  # length 2n-1
    c_fft = rfft(c_full, 2 * N)
    cross2 = irfft(np.conj(a_fft) * c_fft, 2 * N)[:n]
    grad_final = 2.0 * cross2
    return grad_final

def power_mean_obj(a, p):
    """Objective: 2*n*power_mean(conv(a,a), p) with sum(a)=1."""
    b = fast_convolve(a)
    return 2.0 * len(a) * power_mean_smooth(b, p)

def power_mean_obj_grad(a, p):
    """Objective and gradient (on the simplex, using log-parametrization handled externally)."""
    b = fast_convolve(a)
    obj = 2.0 * len(a) * power_mean_smooth(b, p)
    # Gradient of the sum part (before the 2*n scaling and normalization)
    # We need the gradient with respect to a directly, then handle the simplex constraint
    b_safe = np.maximum(b, 0)
    b_p = b_safe ** p
    S = b_p.sum()
    if S < 1e-30:
        return obj, np.zeros(len(a))
    c = b_safe ** (p - 1) / S ** (1 - 1.0 / p)
    n = len(a)
    N = 1 << (2 * n).bit_length()
    a_fft = rfft(a, N)
    c_fft = rfft(c, N)
    cross = irfft(np.conj(a_fft) * c_fft, N)[:n]
    grad = 4.0 * n * cross  # 2 (from conv) * 2n (scaling) ... wait
    # d/d a_j [2n * (sum b^p)^(1/p)] = 2n * (sum b^p)^(1/p-1) * sum_i b_i^(p-1) * db_i/da_j
    # = 2n * (sum b^p)^(1/p-1) * sum_i b_i^(p-1) * 2*a[i-j] (valid)
    # The c I defined = b_i^(p-1) / S^(1-1/p) where S = sum b^p
    # So (sum b^p)^(1/p-1) * sum_i b_i^(p-1) * (db_i/da_j) = (1/S^(1-1/p)) * sum_i c_i * S^(1-1/p) * (db_i/da_j) / S^(1-1/p)
    # Hmm let me redo.
    # F = (sum_i b_i^p)^(1/p)
    # dF/da_j = (sum b^p)^(1/p - 1) * sum_i b_i^(p-1) * (db_i/da_j)
    # Let S = sum b^p. Then (sum b^p)^(1/p-1) = S^(1/p-1) = S^(-1+1/p)
    # dF/da_j = S^(-1+1/p) * sum_i b_i^(p-1) * (db_i/da_j)
    # = sum_i [b_i^(p-1) / S^(1-1/p)] * (db_i/da_j)
    # = sum_i c_i * (db_i/da_j) where c_i = b_i^(p-1) / S^(1-1/p) ✓
    # 
    # db_i/da_j = 2 * sum_d a[d]*c[i-j+d]... no wait.
    # db_i/da_j = 2*a[i-j] if 0<=i-j<n
    # sum_i c_i * 2*a[i-j] = 2 * sum_i c_i * a[i-j] = 2 * sum_d a[d] * c[j+d]
    # = 2 * cross[j] where cross[j] = sum_d a[d]*c[j+d]
    # So dF/da_j = 2 * cross[j]
    # And d(2n*F)/da_j = 4n * cross[j]
    
    return obj, 4.0 * n * cross

def simplex_project(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if cond.any():
        rho = np.nonzero(cond)[0][-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        return np.maximum(v - theta, 0.0)
    else:
        return np.ones(n) / n

def lbfgs_power_mean(a0, time_limit, p=20, maxiter=5000):
    """L-BFGS optimization using power-mean smoothing, with simplex projection."""
    n = len(a0)
    a = a0.copy()
    best_a = a.copy()
    best_obj = power_mean_obj(a, p)
    t0 = time.monotonic()
    
    # Use log-parametrization: a = softmax(z), z = log(a)
    a_safe = np.maximum(a, 1e-300)
    z = np.log(a_safe)
    
    def func_z(z):
        if time.monotonic() - t0 > time_limit:
            return 1e15
        z_shifted = z - z.max()
        a_z = np.exp(z_shifted)
        a_z = a_z / a_z.sum()
        return power_mean_obj(a_z, p)
    
    def jac_z(z):
        if time.monotonic() - t0 > time_limit:
            return np.zeros(n)
        z_shifted = z - z.max()
        a_z = np.exp(z_shifted)
        a_z = a_z / a_z.sum()
        _, grad_a = power_mean_obj_grad(a_z, p)
        # grad_z = a * (grad_a - sum(a * grad_a))
        S = np.dot(grad_a, a_z)
        grad_z = a_z * (grad_a - S)
        return grad_z
    
    try:
        res = minimize(
            fun=func_z, x0=z, jac=jac_z,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12, 'maxfun': maxiter * 50, 'disp': False}
        )
        z_out = res.x
        z_shifted = z_out - z_out.max()
        a_out = np.exp(z_shifted)
        a_out = a_out / a_out.sum()
        a_out = np.maximum(a_out, 0)
        return a_out
    except Exception as e:
        print(f"  L-BFGS error: {e}")
        return best_a

def fictitious_play(a_init, time_limit, solver=None, max_iters=500):
    """Fictitious play with running averages."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_init)
    a = a_init.copy()
    b = a.copy()
    a_avg = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    iters = 0
    
    while time.monotonic() - t0 < time_limit and iters < max_iters:
        iters += 1
        b_new, _ = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new
        a_new, _ = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new
        
        a_avg = (iters * a_avg + a) / (iters + 1)
        
        candidates = [a, a_avg, 0.5*(a + a_avg)]
        for cand in candidates:
            cand = np.maximum(cand, 0)
            s = cand.sum()
            if s < 1e-10:
                continue
            cand = cand / s
            score = 2.0 * n * fast_convolve(cand).max()
            if score < best_score:
                best_score = score
                best_a = cand.copy()
    
    return best_a, best_score, iters

def subgradient_optimize(a0, time_limit, p_init=10):
    """Subgradient method with power-mean objective, Nesterov momentum."""
    n = len(a0)
    a = a0.copy()
    y = a.copy()  # Nesterov momentum
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    beta = 0.7  # momentum
    lr = 0.01
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Nesterov step
        y_new = y + beta * (y - a)  # wait, Nesterov: y = x + (t-1)/(t+2) * (x - x_prev)
        # Simpler: use heavy ball
        y = a + beta * (a - y) if it > 1 else a  # hmm, this is wrong
        # Let me use the standard Nesterov:
        # t_k = (1+sqrt(1+4*t_{k-1}^2))/2
        # y_k = x_k + (t_{k-1}-1)/t_k * (x_k - x_{k-1})
        # x_{k+1} = y_k - lr * grad(y_k)
        # This is getting complicated. Let me just use momentum SGD.
        
        _, grad = power_mean_obj_grad(a, p_init)
        
        # Projected gradient step with momentum
        a_prev = a.copy()
        a = a - lr * grad
        a = simplex_project(a)
        
        score = 2.0 * n * fast_convolve(a).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
        
        lr *= 0.999
        if it % 1000 == 0:
            print(f"    SubGrad iter={it}: score={best_score:.6f} lr={lr:.6f}")
    
    return best_a, best_score

def peak_shave(a0, time_limit, max_steps=5000):
    """Aggressive peak shaving with redistribution."""
    n = len(a0)
    a = a0.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    
    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        b = fast_convolve(a)
        # Find the dominant peak (excluding endpoints which are always small)
        b_mid = b.copy()
        b_mid[0] = 0
        b_mid[-1] = 0
        peak_k = int(np.argmax(b_mid))
        peak_val = b[peak_k]
        
        if peak_val <= best_score / (2.0 * n) * 1.001:
            break
        
        # Find top contributors to this peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        
        # Try reducing the top contributor
        top_idx_local = np.argsort(contribs)[::-1][:5]
        improved = False
        
        for ci in top_idx_local:
            i = idx[ci]
            j = peak_k - i
            if a[i] < 1e-14:
                continue
            for factor in [0.92, 0.85, 0.75, 0.6]:
                a_trial = a.copy()
                diff = a_trial[i] * (1 - factor)
                a_trial[i] *= factor
                # Redistribute to other elements (proportionally)
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others_sum = a_trial[mask].sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] += diff * a_trial[mask] / others_sum
                a_trial = a_trial / a_trial.sum()
                
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score - 1e-13:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Try pairwise reduction
            for ci1 in top_idx_local[:3]:
                for ci2 in top_idx_local[:3]:
                    if ci1 == ci2:
                        continue
                    i1 = idx[ci1]
                    i2 = idx[ci2]
                    j1 = peak_k - i1
                    j2 = peak_k - i2
                    if i1 == j2 or i2 == j1:
                        continue
                    if a[i1] < 1e-14 or a[i2] < 1e-14:
                        continue
                    a_trial = a.copy()
                    diff1 = a_trial[i1] * 0.1
                    diff2 = a_trial[i2] * 0.1
                    a_trial[i1] -= diff1
                    a_trial[i2] -= diff2
                    mask = np.ones(n, dtype=bool)
                    mask[i1] = False
                    mask[i2] = False
                    others_sum = a_trial[mask].sum()
                    if others_sum < 1e-15:
                        continue
                    a_trial[mask] += (diff1 + diff2) * a_trial[mask] / others_sum
                    a_trial = np.maximum(a_trial, 0)
                    a_trial = a_trial / a_trial.sum()
                    score = 2.0 * n * fast_convolve(a_trial).max()
                    if score < best_score - 1e-13:
                        a = a_trial
                        best_score = score
                        best_a = a.copy()
                        improved = True
                        break
                if improved:
                    break
        
        if not improved:
            # Random perturbation to escape
            if step % 100 == 0:
                rng = np.random.default_rng(step)
                a_trial = a + rng.standard_normal(n) * 0.002
                a_trial = np.maximum(a_trial, 0)
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()
                if score < best_score * 1.05:  # Allow slight worsening to escape
                    a = a_trial
            else:
                break
    
    return best_a, best_score

def interpolate_seq(f, new_n):
    """Interpolate sequence to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp

def check_update(a, best_seq, best_score, label=""):
    """Evaluate and update best."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return best_seq, best_score
    a_norm = a / s
    score = float(evaluate_sequence(a_norm.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_norm)})")
    return best_seq, best_score

import numpy as np
import cvxpy as cp
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = rfft(a, fft_size)
    return irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build (2n-1) x n matrix T such that T @ b = conv(a, b)."""
    n = len(a)
    T = np.zeros((2 * n - 1, n))
    for j in range(n):
        T[j:j + n, j] = a
    return T


def get_lp_solver():
    """Find a working LP solver."""
    global _lp_solver_cache
    if '_lp_solver_cache' in dir():
        return _lp_solver_cache
    for solver_name in [cp.ECOS, cp.SCS, cp.CLARABEL, cp.PDLP]:
        try:
            x = cp.Variable(3, nonneg=True)
            prob = cp.Problem(cp.Minimize(cp.sum(x)), [cp.sum(x) == 1])
            prob.solve(solver=solver_name)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                globals()['_lp_solver_cache'] = solver_name
                return solver_name
        except Exception:
            continue
    globals()['_lp_solver_cache'] = cp.ECOS
    return cp.ECOS


def solve_lp_best_response(a_fixed, solver=None):
    """Given fixed a (sum=1), find b (sum=1, b>=0) minimizing max_k conv(a,b)[k]."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_fixed)
    b_var = cp.Variable(n, nonneg=True)
    M_var = cp.Variable()
    T = build_conv_matrix(a_fixed)
    constraints = [
        T @ b_var <= M_var,
        cp.sum(b_var) == 1
    ]
    prob = cp.Problem(cp.Minimize(M_var), constraints)
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        for alt in [cp.ECOS, cp.SCS, cp.PDLP]:
            if alt == solver:
                continue
            try:
                prob.solve(solver=alt, verbose=False)
                break
            except Exception:
                continue
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        return None, None
    b_val = np.asarray(b_var.value, dtype=np.float64)
    b_val = np.maximum(b_val, 0)
    s = b_val.sum()
    if s < 1e-15:
        return None, None
    b_val = b_val / s
    return b_val, float(M_var.value)


def power_mean_val(b, p):
    """Compute p-th power mean of b (smooth approximation to max)."""
    b_max = b.max()
    if b_max <= 1e-30:
        return 0.0
    scaled = (b / b_max) ** p
    return b_max * (scaled.sum()) ** (1.0 / p)


def power_mean_obj_grad(a, p):
    """Compute objective 2n * PM(conv(a,a), p) and its gradient."""
    n = len(a)
    b = fast_convolve(a)
    b_max = b.max()
    if b_max <= 1e-30:
        return 2.0 * n * 0.0, np.zeros(n)
    pm = b_max * ((b / b_max) ** p).sum() ** (1.0 / p)
    obj = 2.0 * n * pm

    # Gradient: df/da_j = 4n * sum_d a[d] * c[j+d]
    # where c_i = b_i^(p-1) / (sum b^p)^(1-1/p)
    b_safe = np.maximum(b, 0.0)
    S = (b_safe ** p).sum()
    if S < 1e-30:
        return obj, np.zeros(n)
    c = b_safe ** (p - 1) / S ** (1.0 - 1.0 / p)

    # cross_corr[j] = sum_d a[d] * c[j+d] = (c cross-correlate a)[j]
    N = 1 << (2 * n).bit_length()
    a_fft = rfft(a, N)
    c_fft = rfft(c, N)
    cross = irfft(np.conj(a_fft) * c_fft, N)[:n]
    grad = 4.0 * n * cross
    return obj, grad


def lbfgs_power_mean(a0, time_limit, p=16, maxiter=5000):
    """L-BFGS with power-mean smoothing using log-parametrization."""
    n = len(a0)
    a_safe = np.maximum(a0, 1e-300)
    z0 = np.log(a_safe)
    t0 = time.monotonic()
    stopped = [False]

    def func_z(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return 1e15
        z_shift = z - z.max()
        a_z = np.exp(z_shift)
        a_z = a_z / a_z.sum()
        obj, _ = power_mean_obj_grad(a_z, p)
        return obj

    def jac_z(z):
        if time.monotonic() - t0 > time_limit:
            stopped[0] = True
            return np.zeros(n)
        z_shift = z - z.max()
        a_z = np.exp(z_shift)
        a_z = a_z / a_z.sum()
        _, grad_a = power_mean_obj_grad(a_z, p)
        S = np.dot(grad_a, a_z)
        return a_z * (grad_a - S)

    try:
        res = minimize(
            fun=func_z, x0=z0, jac=jac_z,
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12,
                     'maxfun': maxiter * 60, 'disp': False}
        )
        z_out = res.x
        z_shift = z_out - z_out.max()
        a_out = np.exp(z_shift)
        a_out = a_out / a_out.sum()
        return np.maximum(a_out, 0.0)
    except Exception:
        return a0.copy()


def fictitious_play(a_init, time_limit, solver=None, max_iters=1000):
    """Fictitious play with running averages."""
    if solver is None:
        solver = get_lp_solver()
    n = len(a_init)
    a = a_init.copy()
    b = a.copy()
    a_avg = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    iters = 0

    while time.monotonic() - t0 < time_limit and iters < max_iters:
        iters += 1
        b_new, _ = solve_lp_best_response(a, solver)
        if b_new is not None:
            b = b_new
        a_new, _ = solve_lp_best_response(b, solver)
        if a_new is not None:
            a = a_new
        a_avg = (iters * a_avg + a) / (iters + 1)

        for cand in [a, a_avg, 0.5 * (a + a_avg)]:
            cand = np.maximum(cand, 0)
            s = cand.sum()
            if s < 1e-10:
                continue
            cand = cand / s
            score = 2.0 * n * fast_convolve(cand).max()
            if score < best_score:
                best_score = score
                best_a = cand.copy()

    return best_a, best_score, iters


def nesterov_subgrad(a0, time_limit, p=16, lr_init=0.02):
    """Nesterov-accelerated subgradient with power-mean objective."""
    n = len(a0)
    a = a0.copy()
    x_prev = a.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    lr = lr_init
    t0 = time.monotonic()
    it = 0
    t_nest = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        t_nest_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        beta = (t_nest - 1.0) / t_nest_new
        # Nesterov extrapolation
        x_nest = a + beta * (a - x_prev)
        x_nest = np.maximum(x_nest, 0)
        s = x_nest.sum()
        if s < 1e-15:
            x_nest = a.copy()
        else:
            x_nest = x_nest / s

        _, grad = power_mean_obj_grad(x_nest, p)

        # Gradient step with projection
        a_new = x_nest - lr * grad
        # Project onto simplex
        u = np.sort(a_new)[::-1]
        cssv = np.cumsum(u)
        cond = u * np.arange(1, n + 1) > (cssv - 1)
        if cond.any():
            rho = np.nonzero(cond)[0][-1]
            theta = (cssv[rho] - 1.0) / (rho + 1.0)
            a_new = np.maximum(a_new - theta, 0.0)
        else:
            a_new = np.ones(n) / n
        a_new = a_new / a_new.sum()

        score = 2.0 * n * fast_convolve(a_new).max()
        if score < best_score - 1e-14:
            best_score = score
            best_a = a_new.copy()

        x_prev = a
        a = a_new
        t_nest = t_nest_new
        lr = lr_init / (1.0 + it * 0.01)

        if it % 2000 == 0:
            print(f"    Nesterov iter={it}: score={best_score:.6f} lr={lr:.6f}")

    return best_a, best_score


def peak_shave_anneal(a0, time_limit, max_steps=10000, T_anneal=0.5):
    """Peak shaving with simulated annealing acceptance criterion."""
    n = len(a0)
    a = a0.copy()
    best_a = a.copy()
    best_score = 2.0 * n * fast_convolve(a).max()
    t0 = time.monotonic()
    T = T_anneal
    rng = np.random.default_rng(12345)

    for step in range(max_steps):
        if time.monotonic() - t0 > time_limit:
            break
        T *= 0.9995

        b = fast_convolve(a)
        b_mid = b.copy()
        b_mid[0] = 0
        b_mid[-1] = 0
        peak_k = int(np.argmax(b_mid))

        if b[peak_k] <= best_score / (2.0 * n) * 1.0005:
            if step > 100:
                break

        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        contribs = a[idx] * a[peak_k - idx]
        n_top = min(8, len(idx))
        top_local = np.argsort(contribs)[::-1][:n_top]

        improved = False
        for ci in top_local:
            i = int(idx[ci])
            if a[i] < 1e-14:
                continue
            for factor in [0.93, 0.87, 0.8, 0.7, 0.55]:
                a_trial = a.copy()
                diff = a_trial[i] * (1.0 - factor)
                a_trial[i] *= factor
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                others = a_trial[mask]
                others_sum = others.sum()
                if others_sum < 1e-15:
                    continue
                a_trial[mask] = others + diff * others / others_sum
                a_trial = a_trial / a_trial.sum()
                score = 2.0 * n * fast_convolve(a_trial).max()

                if score < best_score - 1e-14:
                    a = a_trial
                    best_score = score
                    best_a = a.copy()
                    improved = True
                    break
                elif T > 0.01 and rng.random() < np.exp(-(score - best_score) / T):
                    a = a_trial
                    improved = True
                    break
            if improved:
                break

        if not improved and step % 50 == 0:
            # Random perturbation
            a_trial = a + rng.standard_normal(n) * 0.003
            a_trial = np.maximum(a_trial, 0)
            a_trial = a_trial / a_trial.sum()
            score = 2.0 * n * fast_convolve(a_trial).max()
            if score < best_score * 1.02 or (T > 0.1 and rng.random() < np.exp(-(score - best_score) / T)):
                a = a_trial

    return best_a, best_score


def interpolate_seq(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    s = f.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    f_norm = f / s
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f_norm)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp = f_interp / ss
    return f_interp


def check_update(a, best_seq, best_score, label=""):
    """Evaluate candidate and update best if improved."""
    a = np.maximum(a, 0.0)

        best_seq = a_norm.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a_norm)})")
    return best_seq, best_score

        n_init = 768
        t = np.linspace(0, 2 * np.pi, n_init)
        init = 1.0 + 0.3 * np.cos(2 * t) + 0.1 * np.cos(5 * t) + 0.05 * np.sin(3 * t)
        init = np.maximum(init, 0)
        init = init / init.sum()
        p_score = float(evaluate_sequence(init.tolist()))
        best_seq = init.copy()
        best_score = p_score
        print(f"Initialized: n={n_init}, score={p_score:.8f}")

    solver = get_lp_solver()
    print(f"LP solver: {solver}")
    print(f"Start time, deadline in {budget_s - 25:.0f}s")

    # === Phase 1: LP Fictitious Play ===
    print(f"\n=== Phase 1: LP Fictitious Play (best={best_score:.8f}) ===")
    n_best = len(best_seq)
    n_values = [n_best, int(n_best * 0.875), int(n_best * 1.125)]
    n_values = sorted(set([x for x in n_values if 256 <= x <= 2048]),
                      key=lambda x: abs(x - n_best))

    fp_total = min(250.0, (deadline - time.monotonic()) * 0.15)
    t_per_n = fp_total / len(n_values)

    for n_try in n_values[:3]:
        rem = deadline - time.monotonic() - 700
        if rem < 30:
            break
        t_alloc = min(t_per_n, rem * 0.9)
        if t_alloc < 20:
            continue
        print(f"  FP n={n_try}, budget={t_alloc:.0f}s")
        init_fp = interpolate_seq(best_seq, n_try) if len(best_seq) != n_try else best_seq.copy()
        a_fp, score_fp, iters = fictitious_play(init_fp, t_alloc, solver=solver)
        best_seq, best_score = check_update(a_fp, best_seq, best_score, f"FP_n{n_try}_it{iters}")

    # === Phase 2: Power-Mean Continuation L-BFGS ===
    print(f"\n=== Phase 2: L-BFGS Continuation (best={best_score:.8f}) ===")
    n_focus = len(best_seq)
    p_schedule = [4, 8, 12, 16, 24, 36, 50, 72]

    for p_val in p_schedule:
        rem = deadline - time.monotonic() - 250
        if rem < 15:
            break
        t_alloc = min(100.0, rem * 0.25)
        print(f"  L-BFGS p={p_val}, n={n_focus}, budget={t_alloc:.0f}s")
        a_lb = lbfgs_power_mean(best_seq, t_alloc, p=p_val, maxiter=4000)
        best_seq, best_score = check_update(a_lb, best_seq, best_score, f"LB_p{p_val}")
        n_focus = len(best_seq)

    # === Phase 3: Nesterov Subgradient ===
    print(f"\n=== Phase 3: Nesterov SubGrad (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 250
    if rem > 30:
        t_alloc = min(150.0, rem * 0.5)
        print(f"  Nesterov n={len(best_seq)}, budget={t_alloc:.0f}s, p=16")
        a_nest, _ = nesterov_subgrad(best_seq, t_alloc, p=16, lr_init=0.015)
        best_seq, best_score = check_update(a_nest, best_seq, best_score, "nesterov")

    # === Phase 4: Annealed Peak Shaving ===
    print(f"\n=== Phase 4: Annealed Peak Shave (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 200
    if rem > 20:
        t_alloc = min(150.0, rem * 0.55)
        print(f"  PeakShave n={len(best_seq)}, budget={t_alloc:.0f}s")
        a_ps = peak_shave_anneal(best_seq, t_alloc, T_anneal=0.8)
        best_seq, best_score = check_update(a_ps, best_seq, best_score, "peak_shave")

    # === Phase 5: L-BFGS Round 2 ===
    print(f"\n=== Phase 5: L-BFGS Round 2 (best={best_score:.8f}) ===")
    for p_val in [20, 40, 60]:
        rem = deadline - time.monotonic() - 100
        if rem < 15:
            break
        t_alloc = min(80.0, rem * 0.3)
        print(f"  L-BFGS2 p={p_val}, budget={t_alloc:.0f}s")
        a_lb2 = lbfgs_power_mean(best_seq, t_alloc, p=p_val, maxiter=3000)
        best_seq, best_score = check_update(a_lb2, best_seq, best_score, f"LB2_p{p_val}")

    # === Phase 6: Alternative lengths ===
    print(f"\n=== Phase 6: Alt Lengths (best={best_score:.8f}) ===")
    n_cur = len(best_seq)
    alt_factors = [0.85, 1.1, 1.25]
    for factor in alt_factors:
        alt_n = int(n_cur * factor)
        if alt_n < 256 or alt_n > 3000 or abs(alt_n - n_cur) < 50:
            continue
        rem = deadline - time.monotonic() - 50
        if rem < 20:
            break
        t_alloc = min(60.0, rem * 0.4)
        init_alt = interpolate_seq(best_seq, alt_n)
        a_alt = lbfgs_power_mean(init_alt, t_alloc, p=20, maxiter=3000)
        best_seq, best_score = check_update(a_alt, best_seq, best_score, f"alt_n{alt_n}")

    # === Phase 7: Final polish ===
    print(f"\n=== Phase 7: Final Polish (best={best_score:.8f}) ===")
    rem = deadline - time.monotonic() - 10
    if rem > 15:
        t_f = min(40.0, rem * 0.5)
        print(f"  Final L-BFGS p=50, budget={t_f:.0f}s")
        a_f = lbfgs_power_mean(best_seq, t_f, p=50, maxiter=4000)
        best_seq, best_score = check_update(a_f, best_seq, best_score, "final_LB")

    rem = deadline - time.monotonic() - 5
    if rem > 10:
        t_ps = min(20.0, rem * 0.6)
        a_ps2 = peak_shave_anneal(best_seq, t_ps, T_anneal=0.3)
        best_seq, best_score = check_update(a_ps2, best_seq, best_score, "final_ps")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(512) / 512
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
